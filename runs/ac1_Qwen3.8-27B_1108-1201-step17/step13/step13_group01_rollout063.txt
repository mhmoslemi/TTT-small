def fast_conv(a):
    """Compute full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute the objective score: 2*n*max(conv)/(sum)^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize_sum(a, target_sum):
    """Normalize a to have the given sum, preserving non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * target_sum / s


def softmax_parametrize(v, target_sum):
    """Convert unconstrained v to a non-negative vector with given sum."""
    v_shifted = v - v.max()
    exp_v = np.exp(np.minimum(v_shifted, 50.0))
    s = exp_v.sum()
    if s < 1e-30:
        return np.full_like(v, target_sum / len(v))
    return target_sum * exp_v / s


def power_objective_and_grad(v, n, p, target_sum):
    """
    Objective: sum_k (b[k]/M)^p where b = conv(a,a), M = max(b), a = softmax(v)*target_sum
    Gradient computed via FFT.

    a = softmax_parametrize(v, target_sum)
    b = fast_conv(a)
    M = b.max()
    if M < 1e-30:
        return 1e10, np.zeros(n)

    # We minimize sum(b[k]/M)^p, gradient w.r.t. a is:
    # grad_a[i] = sum_k p*(b[k]/M)^(p-1)/M * (a[k-i] + a[i-k])

    fft_size = 1 << (3 * n - 2).bit_length()

    w_rev = w[::-1]
    W_rev = rfft(w_rev, fft_size)

    T2 = conv_warev[n - 1:2 * n - 1][::-1]

    grad_v = a * (grad_a - dot_val)

    obj_val = np.sum(np.power(b / M, p))
    return float(obj_val), grad_v


def hinge_objective_and_grad(v, n, threshold, target_sum, p=2):
    """
    Objective: sum_k max(0, b[k]-threshold)^p / threshold^p
    More targeted than power objective - only penalizes peaks above threshold.
    """
    a = softmax_parametrize(v, target_sum)
    b = fast_conv(a)
    excess = np.maximum(b - threshold, 0.0)
    obj_val = np.sum(np.power(excess / (threshold + 1e-30), p))

    if np.all(excess < 1e-15):
        return float(obj_val), np.zeros(n)

    # Gradient: w[k] = p * max(0, b[k]-T)^(p-1) / (T^p * sum_count)
    w = p * np.power(np.maximum(excess, 1e-30), p - 1) / (np.power(threshold + 1e-30, p))
    # Only nonzero where excess > 0
    w = w * (excess > 0).astype(float)

    if w.sum() < 1e-30:
        return float(obj_val), np.zeros(n)

    # FFT-based gradient
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    conv_wa = irfft(W * A, fft_size)
    T1 = conv_wa[:n]

    w_rev = w[::-1]
    W_rev = rfft(w_rev, fft_size)
    conv_warev = irfft(W_rev * A, fft_size)
    T2 = conv_warev[n - 1:2 * n - 1][::-1]

    grad_a = 2.0 * (T1 + T2)
    dot_val = np.dot(grad_a, a) / target_sum
    grad_v = a * (grad_a - dot_val)

    return float(obj_val), grad_v


def lse_objective_and_grad(v, n, temperature, target_sum):
    """
    LSE approximation of max(b). Smooth and well-behaved.
    Objective: (T/T) * max_b + log(sum(exp((b-max_b)/T)))/T
    Actually just: lse = max_b + log(sum(exp((b-max_b)/T)))/T
    We minimize lse directly (since score = 2n*max_b/sum^2 and sum is fixed).
    """
    a = softmax_parametrize(v, target_sum)
    b = fast_conv(a)
    max_b = b.max()
    if max_b < 1e-30:
        return 1e10, np.zeros(n)

    exp_vals = np.exp((b - max_b) / temperature)
    lse_val = max_b + np.log(exp_vals.sum()) / temperature
    w = exp_vals / exp_vals.sum() / temperature

    # FFT-based gradient of lse w.r.t. a
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    conv_wa = irfft(W * A, fft_size)
    T1 = conv_wa[:n]

    w_rev = w[::-1]
    W_rev = rfft(w_rev, fft_size)
    conv_warev = irfft(W_rev * A, fft_size)
    T2 = conv_warev[n - 1:2 * n - 1][::-1]

    grad_a = 2.0 * (T1 + T2)
    dot_val = np.dot(grad_a, a) / target_sum
    grad_v = a * (grad_a - dot_val)

    return float(lse_val), grad_v


def run_lbfgs_power(a_init, n, p, time_limit, target_sum):
    """Run L-BFGS-B with power objective."""
    a = normalize_sum(a_init, target_sum)
    # Initialize v from a: v = log(a / (target_sum/n)) approximately
    a_safe = np.maximum(a, 1e-15)
    v0 = np.log(a_safe * n / target_sum)

    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()

    def obj_func(v):
        return power_objective_and_grad(v, n, p, target_sum)[0]

    def obj_grad_func(v):
        return power_objective_and_grad(v, n, p, target_sum)

    # Run L-BFGS-B
    max_iter = max(50, int(1000 * (time_limit / 60.0)))
    try:
        res = scipy_minimize(obj_grad_func, v0, method='L-BFGS-B',
                             jac=True, options={'maxiter': max_iter,
                                                'maxfun': max_iter * 3,
                                                'gtol': 1e-6})
        if res.x is not None:
            a_new = softmax_parametrize(res.x, target_sum)
            sc = eval_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new
    except Exception:
        pass

    return best_a, best_score


def run_lbfgs_lse(a_init, n, T_start, T_end, n_stages, time_limit, target_sum):
    """Run L-BFGS-B with LSE objective, annealing temperature."""
    a = normalize_sum(a_init, target_sum)
    a_safe = np.maximum(a, 1e-15)
    v0 = np.log(a_safe * n / target_sum)

    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        stage_time = min(remaining / (n_stages - si), 120.0)

        def obj_grad(v):
            return lse_objective_and_grad(v, n, T, target_sum)

        max_iter = max(30, int(800 * (stage_time / 60.0)))
        try:
            res = scipy_minimize(obj_grad, v0, method='L-BFGS-B',
                                 jac=True, options={'maxiter': max_iter,
                                                    'maxfun': max_iter * 3,
                                                    'gtol': 1e-5})
            if res.x is not None:
                a_new = softmax_parametrize(res.x, target_sum)
                sc = eval_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    v0 = res.x.copy()
        except Exception:
            pass

    return best_a, best_score


def run_lbfgs_hinge(a_init, n, time_limit, target_sum):
    """Run L-BFGS-B with adaptive hinge-loss objective."""
    a = normalize_sum(a_init, target_sum)
    a_safe = np.maximum(a, 1e-15)
    v0 = np.log(a_safe * n / target_sum)

    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()

    # Adaptive threshold: start at current max, gradually decrease
    b = fast_conv(a)
    current_max = b.max()
    target_ratio = 0.98  # aim to reduce peak by 2%

    for round_num in range(20):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / max(1, 20 - round_num), 60.0)

        threshold = current_max * target_ratio
        p = 2 + round_num // 5  # increase power over time

        def obj_grad(v):
            return hinge_objective_and_grad(v, n, threshold, target_sum, p=p)

        max_iter = max(20, int(500 * (stage_time / 60.0)))
        try:
            res = scipy_minimize(obj_grad, v0, method='L-BFGS-B',
                                 jac=True, options={'maxiter': max_iter,
                                                    'maxfun': max_iter * 2,
                                                    'gtol': 1e-5})
            if res.x is not None:
                a_new = softmax_parametrize(res.x, target_sum)
                sc = eval_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    v0 = res.x.copy()
                    b = fast_conv(a_new)
                    current_max = b.max()
        except Exception:
            pass

    return best_a, best_score


def lp_best_response(a, M, solver):
    """Solve the LP best-response: max sum(b) s.t. T*b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    # Build constraint matrix T (sparse would be better but dense works for n~2000)
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, target_sum, verbose=False):
    """Nesterov-accelerated LP best-response iterations."""
    n = len(a0)
    a = normalize_sum(a0, target_sum)
    best_a = a.copy()
    best_score = eval_score(a)
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    last_improve = t0
    stall_count = 0
    iter_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_conv(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        g = b * target_sum / b_sum
        iter_count += 1

        # Adaptive step size
        ratio = M_curr / (best_score / (2.0 * n) * (target_sum ** 2) / (target_sum ** 2))
        # Score = 2n*max_b/sum^2, with sum=target_sum: score = 2n*max_b/target_sum^2
        # So max_b = score * target_sum^2 / (2n)
        target_max = best_score * target_sum ** 2 / (2.0 * n)
        ratio = M_curr / max(target_max, 1e-15)
        alpha = 2.0 / (t + 2.0)
        alpha = min(alpha * min(ratio + 0.5, 3.0), 1.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_sum(y_new, target_sum)

        # Nesterov momentum
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_sum(a_new, target_sum)

        y = y_new
        a = a_new

        # Check scores
        for candidate in [y, a]:
            sc = eval_score(candidate)
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
                last_improve = time.monotonic()

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = normalize_sum(avg_a / n_avg, target_sum)
            sc_avg = eval_score(avg_check)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_check.copy()
                last_improve = time.monotonic()

        # Stall detection
        if time.monotonic() - last_improve > 45.0:
            stall_count += 1
            if stall_count > 15:
                break
        else:
            stall_count = 0

        if verbose and iter_count % 100 == 0:
            print(f"    LP iter={iter_count} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_directed_coord_descent(a0, time_limit, target_sum, seed=42):
    """Coordinate descent that focuses on reducing the top peaks."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = normalize_sum(a0, target_sum)
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        top_k = 6
        peak_indices = np.argsort(b)[::-1][:top_k]

        improved = False
        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break

            # Find which a[i] contribute most to this peak
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[pk - idx]
            top_contrib = idx[np.argsort(contrib)[::-1][:4]]

            # Build mask of positions involved in any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for pk2 in peak_indices:
                js2 = max(0, pk2 - n + 1)
                je2 = min(n - 1, pk2)
                peak_mask[js2:je2 + 1] = True

            # Try moving mass from high-contrib positions to low-risk positions
            for src in top_contrib:
                if a[src] < 1e-10:
                    continue
                # Find destination: positions NOT in any peak
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                # Prefer positions with small current value (less likely to create new peak)
                candidates_sorted = candidates[np.argsort(a[candidates])[:max(1, len(candidates) // 4)]]

                found_improve = False
                for dst in candidates_sorted[:5]:
                    if found_improve:
                        break
                    for frac in [0.02, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        a_trial = normalize_sum(a_trial, target_sum)
                        sc = eval_score(a_trial)
                        if sc < best_score - 1e-10:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            found_improve = True
                            break
                if found_improve:
                    break

        if not improved:
            stall += 1
            # Random perturbation when stuck
            if stall % 5 == 0:
                for _ in range(30):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.01, 0.08)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = normalize_sum(a_trial, target_sum)
                    sc = eval_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
            if stall > 300:
                break

    return best_a, best_score


def structured_perturb(a, rng, strength=0.1):
    """Apply structured perturbation to escape local minima."""
    n = len(a)
    choice = rng.integers(0, 5)
    a_new = a.copy()
    if choice == 0:
        # Additive Gaussian noise
        noise = rng.normal(0, strength / np.sqrt(n), n)
        a_new = a + noise
    elif choice == 1:
        # Multiplicative noise
        noise = rng.uniform(1 - strength, 1 + strength, n)
        a_new = a * noise
    elif choice == 2:
        # Reverse and blend
        rev = a[::-1]
        t = rng.uniform(0.2, 0.8)
        a_new = (1 - t) * a + t * rev
    elif choice == 3:
        # Circular shift and blend
        shift = rng.integers(1, n)
        shifted = np.roll(a, shift)
        t = rng.uniform(0.1, 0.5)
        a_new = (1 - t) * a + t * shifted
    else:
        # Local smoothing followed by noise
        kernel = np.ones(3) / 3.0
        smoothed = np.convolve(a, kernel, mode='same')
        t = rng.uniform(0.3, 0.7)
        a_new = (1 - t) * a + t * smoothed + rng.normal(0, strength * 0.05, n)

    a_new = np.maximum(a_new, 0.0)
    return normalize_sum(a_new, a.sum())


def interpolate_to_length(a, new_n):
    """Interpolate sequence to a different length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n)
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = eval_score(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
            best_seq = parent.copy()
            best_score = parent_score
    except Exception:
        parent = None

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = eval_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = eval_score(best_seq)
        print(f"Default uniform init: n={n_base}, score={best_score:.6f}")

    # === Phase 0: Quick multi-power L-BFGS at base length (300s) ===
    n = len(best_seq)
    target_sum = np.sqrt(2.0 * n)
    remaining = time_budget - (time.monotonic() - t_global)
    t0_phase = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 0: Power L-BFGS n={n} (budget={t0_phase:.0f}s) ===")

    a_work = normalize_sum(best_seq, target_sum)
    for p_val in [4, 6, 8]:
        chunk_time = t0_phase / 3.0
        if time.monotonic() - t_global > time_budget:
            break
        a_out, sc_out = run_lbfgs_power(a_work, n, p_val, chunk_time, target_sum)
        update_best(a_out, f"power_p{p_val}")
        a_work = normalize_sum(best_seq, target_sum)

    # === Phase 1: LSE annealing (300s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 1: LSE anneal n={n} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_work = normalize_sum(best_seq, target_sum)
        a_out, sc_out = run_lbfgs_lse(a_work, n, T_start=10.0, T_end=0.5,
                                       n_stages=6, time_limit=t1, target_sum=target_sum)
        update_best(a_out, "lse_anneal")

    # === Phase 2: Hinge-loss targeted optimization (250s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 2: Hinge n={n} (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_work = normalize_sum(best_seq, target_sum)
        a_out, sc_out = run_lbfgs_hinge(a_work, n, t2, target_sum)
        update_best(a_out, "hinge")

    # === Phase 3: LP best-response with Nesterov (400s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 3: LP+Nesterov n={n} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_work = normalize_sum(best_seq, target_sum)
        a_out, sc_out = nesterov_lp_optimize(a_work, t3, solver, target_sum, verbose=True)
        update_best(a_out, f"lp_nesterov")

    # === Phase 4: Structured restarts + gradient optimization (250s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: Restart+optimize n={n} (budget={t4:.0f}s) ===")
    if t4 > 30:
        n_restarts = max(2, int(t4 / 40.0))
        for ri in range(n_restarts):
            if time.monotonic() - t_global > time_budget:
                break
            a_pert = structured_perturb(best_seq, rng, strength=0.15)
            a_pert = normalize_sum(a_pert, target_sum)

            # Alternate between power and hinge
            if ri % 2 == 0:
                a_out, _ = run_lbfgs_power(a_pert, n, 6, min(60.0, t4 / n_restarts), target_sum)
            else:
                a_out, _ = run_lbfgs_hinge(a_pert, n, min(60.0, t4 / n_restarts), target_sum)
            update_best(a_out, f"restart_{ri}")

    # === Phase 5: Try nearby lengths (150s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 5: Length exploration (budget={t5:.0f}s) ===")
    if t5 > 30:
        for n_try in [n - 200, n + 200, n + 400]:
            if time.monotonic() - t_global > time_budget:
                break
            if n_try < 100:
                continue
            chunk = min(t5 / 3.0, 60.0)
            a_interp = interpolate_to_length(best_seq, n_try)
            ts = np.sqrt(2.0 * n_try)
            a_interp = normalize_sum(a_interp, ts)
            a_out, _ = run_lbfgs_power(a_interp, n_try, 6, chunk, ts)
            update_best(a_out, f"len_{n_try}")

    # === Phase 6: Final polish - coordinate descent (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    target_sum = np.sqrt(2.0 * n_best)
    t6 = max(remaining * 0.4, 30.0)
    print(f"\n=== Phase 6: Coord descent n={n_best} (budget={t6:.0f}s) ===")
    a_work = normalize_sum(best_seq, target_sum)
    a_out, _ = peak_directed_coord_descent(a_work, t6, target_sum, seed=12345)
    update_best(a_out, "coord_polish")

    # === Phase 7: Final LSE + coord from best (if time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    target_sum = np.sqrt(2.0 * n_best)
    t7 = max(remaining * 0.3, 10.0)
    print(f"\n=== Phase 7: Final LSE n={n_best} (budget={t7:.0f}s) ===")
    if t7 > 10:
        a_work = normalize_sum(best_seq, target_sum)
        a_out, _ = run_lbfgs_lse(a_work, n_best, T_start=3.0, T_end=0.3,
                                  n_stages=4, time_limit=t7 * 0.6, target_sum=target_sum)
        update_best(a_out, "final_lse")

    # Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    target_sum = np.sqrt(2.0 * n_best)
    t8 = max(remaining * 0.5, 5.0)
    if t8 > 5:
        a_work = normalize_sum(best_seq, target_sum)
        a_out, _ = peak_directed_coord_descent(a_work, t8, target_sum, seed=99999)
        update_best(a_out, "final_coord")

    # Final verification
    if best_seq is not None:
        final_score = eval_score(best_seq)
        print(f"\nFINAL: score={final_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
        return [float(x) for x in best_seq]
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]
