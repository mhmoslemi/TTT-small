def _spectral_flatten(a, time_limit):
    """Iteratively flatten the periodogram to reduce autocorrelation peaks."""
    n = len(a)
    a = a.copy()
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s > 0:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    fft_size = next_fast_len(2*n-1)
    
    while time.monotonic() - t0 < time_limit:
        # Compute autocorrelation
        R = _fast_autocorr(a)
        R_max = R.max()
        
        # Compute spectrum
        A = rfft(a, fft_size)
        P = A.real**2 + A.imag**2  # |A|^2
        
        # Target: flatten the spectrum
        # Apply soft thresholding to the periodogram
        P_target = np.minimum(P, np.mean(P) * 1.5)  # Soft cap
        P_new = P * 0.9 + P_target * 0.1  # Interpolation
        
        # Reconstruct: keep phase, modify magnitude
        mag_new = np.sqrt(P_new)
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        
        a_new = irfft(A_new, fft_size)[:n]

        s_new = a_new.sum()
        if s_new < 1e-15:

        a_new = a_new * norm / s_new
        
        score = _fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, next_fast_len

_global_solver = None

def _get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    return _global_solver


def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = next_fast_len(size)
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def _evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def _lse_grad(y, T):
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    
    # Compute gradient via FFT
    fft_size = next_fast_len(3 * n - 2)
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad, ac.max()


def _lse_value(y, T):
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    return T * np.log(np.sum(np.exp(shifted))) + ac_max


def _adaptive_lse_fista(a0, time_limit, verbose=True):
    """Adaptive LSE-FISTA with backtracking and auto-restart."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_ac_max = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Temperature schedule
    T_current = 5.0
    T_min = 0.02
    T_decay = 0.95
    stall_count = 0
    stall_threshold = 3000
    L = n / 20.0  # initial Lipschitz estimate
    t_fista = 1
    prev_a = a.copy()
    
    eval_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute gradient and value
        grad, ac_max_y = _lse_grad(y, T_current)
        
        # Backtracking line search
        f_y = _lse_value(y, T_current)
        L_bt = max(L, 1.0)
        for bt in range(30):
            step = 1.0 / L_bt
            z = y - step * grad
            a_new = _proj_simplex(z)
            
            # Quadratic upper bound check
            f_a_new = _lse_value(a_new, T_current)
            quad = f_y + np.dot(grad, a_new - y) + 0.5 * L_bt * np.sum((a_new - y)**2)
            
            if f_a_new <= quad + 1e-10:
                break
            L_bt *= 2.0
        L = L_bt
        
        # FISTA update
        if t_fista == 1:
            y_new = a_new.copy()
        else:
            mom = (t_fista - 1.0) / (t_fista + 2.0)
            y_new = a_new + mom * (a_new - prev_a)
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y > 1e-15:
                y_new = y_new / s_y
            else:
                y_new = a_new.copy()
        
        prev_a = a_new.copy()
        a = a_new
        y = y_new
        t_fista += 1
        eval_count += 1
        
        # Check improvement
        if eval_count % 200 == 0:
            ac_max_a = _fast_autocorr(a).max()
            if ac_max_a < best_ac_max - 1e-10:
                best_ac_max = ac_max_a
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
            
            # Adaptive temperature
            if stall_count > stall_threshold:
                T_current = max(T_current * T_decay, T_min)
                if T_current == T_min and stall_count > stall_threshold * 3:
                    # Restart with perturbation
                    rng = np.random.default_rng(int(time.monotonic() * 1000))
                    a = best_a + rng.normal(0, 0.01, n)
                    a = np.maximum(a, 0)
                    s_a = a.sum()
                    if s_a > 0:
                        a = a / s_a
                    y = a.copy()
                    t_fista = 1
                    L = n / 20.0
                    stall_count = 0
                    if verbose:
                        print(f"    LSE restart at T={T_current:.3f}, best={best_ac_max:.8f}")
        
        if verbose and eval_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA iter={eval_count} T={T_current:.3f} ({elapsed:.0f}s): best={best_ac_max:.8f}")

    return best_a, best_ac_max


def _spectral_shape(a0, time_limit, verbose=False):
    """Spectral periodogram equalization to reduce autocorrelation peaks."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    fft_size = next_fast_len(2 * n - 1)
    
    # Vary the flattening strength
    for alpha in [0.3, 0.5, 0.7, 0.9, 1.2, 1.5, 2.0]:
        if time.monotonic() - t0 > time_limit:
            break
        
        a_try = a.copy()
        for it in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            
            R = _fast_autocorr(a_try)
            A = rfft(a_try, fft_size)
            P = np.abs(A)**2
            
            # Soft threshold: reduce peaks in periodogram
            P_mean = np.mean(P)
            P_cap = P_mean * (1.0 + alpha)
            P_new = np.minimum(P, P_cap)
            
            # Blend
            blend = 0.1
            P_blend = P * (1 - blend) + P_new * blend
            
            # Reconstruct
            mag_new = np.sqrt(P_blend)
            phase = np.angle(A)
            A_new = mag_new * np.exp(1j * phase)
            a_new = irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new * norm / s_new
            
            score = _fast_autocorr(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            
            a_try = a_new
        
        a = best_a  # Reset to best
    
    return best_a, best_score


def _lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix: (b * a)[k] <= M for all k
    # (b * a)[k] = sum_i b[i] * a[k-i] (with bounds)
    # This is T_a @ b where T_a is (2n-1) x n
    # T_a[k, i] = a[k-i] if 0 <= k-i < n, else 0
    
    # For efficiency, build sparse
    from scipy.sparse import diags, csr_matrix
    rows = []
    cols = []
    vals = []
    for i in range(n):
        for k in range(i, i + n):  # k-i ranges from 0 to n-1
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    
    T = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_val = b_val[~np.isnan(b_val)]
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def _lp_nesterov(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = _fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        # Add small slack to allow feasibility
        M_slack = M_curr * 1.001
        
        b = _lp_best_response(y, M_slack, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = _fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _peak_shave(a0, time_limit, seed=42):
    """Targeted peak reduction via gradient-based local search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    fft_size = next_fast_len(2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        R = _fast_autocorr(a)
        top_k = np.argsort(R)[::-1][:10]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find good destination: low contribution to all peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2+1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            # Pick from low-value candidates
            candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates)//4)]
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = _fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            if a.any():
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-10:
                    frac = rng.uniform(0.002, 0.05)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = _fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0

            if stall > 300:
                break

    return best_a, best_score


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _gradient_descent_peaks(a0, time_limit, verbose=False):
    """Direct gradient descent on LSE-smoothed peak autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    T = 0.1  # Fixed low temperature
    lr = 0.01
    decay = 0.999
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        grad, _ = _lse_grad(a, T)
        
        # Gradient descent with projection
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            lr *= 0.5
            continue
        a_new = a_new * norm / s_new
        
        score = _fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.95
        
        if verbose and (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GD iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}, lr={lr:.6f}")
        
        if lr < 1e-10:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: multi-method optimization with adaptive scheduling."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = _evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        # U-shaped initialization
        best_seq = 1.0 / (1.0 + 20.0 * (x - 0.5)**2)
        best_seq = best_seq / best_seq.sum()
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq)
    
    # === Phase 1: Adaptive LSE-FISTA ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        print(f"\n=== Phase 1: Adaptive LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        a_out, _ = _adaptive_lse_fista(best_seq, t1, verbose=True)
        _try_update(a_out, "lse1")

    # === Phase 2: Spectral shaping ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        print(f"\n=== Phase 2: Spectral shaping n={n_cur} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _spectral_shape(a_norm, t2, verbose=True)
        _try_update(a_out, "spectral")

    # === Phase 3: LP Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LP-Nesterov n={n_cur} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _lp_nesterov(a_norm, t3, solver, verbose=True)
        _try_update(a_out, "lp1")

    # === Phase 4: LSE-FISTA polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        print(f"\n=== Phase 4: LSE-FISTA polish n={n_cur} (budget={t4:.0f}s) ===")
        a_out, _ = _adaptive_lse_fista(best_seq, t4, verbose=True)
