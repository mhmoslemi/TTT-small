def make_left_flattening(n):
    """Generate the left-flattening sequence: a_k = binom(2k,k)/4^k."""
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = (1.0 - np.sum(a[1:k] * a[k-1:0:-1])) / 2.0
    # Normalize
    s = a.sum()
    if s < 1e-15:
        return np.ones(n)
    return a / s


def make_gaussian(n, sigma_frac=0.3):
    """Generate a Gaussian-like centered sequence."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    s = a.sum()
    return a / s

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M, b >= 0."""
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_best_response_reduced(a, M, solver, max_rounds=3):
    """LP with constraint reduction for speed."""
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Full solve first
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)

        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    
    # Reduce to active constraints
    for _ in range(max_rounds):
        circ = T @ b_val
        tol = 1e-6 * M
        active = circ >= M - tol
        n_active = int(active.sum())
        if n_active >= (2 * n - 1) - 10:
            break
        if n_active < 30:
            break
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-5):
            b_val = b_new
        else:
            # Add violated constraints
            violated = circ_new > M * (1 + 1e-8)
            active = active | violated
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M])

            except Exception:

            if prob.status not in ('optimal', 'optimal_inaccurate'):
                break
            b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
            if (T @ b_new).max() <= M * (1 + 1e-5):
                b_val = b_new
    
    if b_val.sum() < 1e-15:
        return None
    return b_val
