def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgradient(a, max_positions):
    """Compute subgradient of max autocorrelation at given a."""
    n = len(a)
    g = np.zeros(n)
    for k in max_positions:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    return g


def subgradient_warmup(a0, time_limit, verbose=True):
    """Fast subgradient method with Polyak step for warm-up."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / a.sum()
    
    LB = 2.0 * n / (2.0 * n - 1.0)
    best_a = a.copy()
    best_f = fast_autocorr(a).max()
    t0 = time.monotonic()
    iters = 0
    
    # Running average for smoothing
    avg_a = a.copy()
    n_avg = 1
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        f_val = b.max()
        
        # Find positions within 1e-8 of max
        tol = 1e-8 * f_val
        max_pos = np.where(b >= f_val - tol)[0]
        # Limit to top 5 positions for speed
        if len(max_pos) > 5:
            max_pos = max_pos[np.argsort(b[max_pos])[::-1][:5]]
        
        g = compute_subgradient(a, max_pos)
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            break
        
        # Polyak step
        if f_val > LB * (1 + 1e-10):
            eta = (f_val - LB) / g_norm_sq
        else:
            eta = 1e-6
        
        # Cap the step
        eta = min(eta, 1.0 / (2.0 * norm * np.linalg.norm(a, 2)))
        
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        # Update average (exponential)
        decay = 0.999
        avg_a = decay * avg_a + (1 - decay) * a_new
        
        a = a_new
        iters += 1
        
        # Check best every 100 iterations
        if iters % 100 == 0:
            f_new = fast_autocorr(a).max()
            if f_new < best_f:
                best_f = f_new
                best_a = a.copy()
            
            f_avg = fast_autocorr(avg_a * norm / avg_a.sum()).max()
            if f_avg < best_f:
                best_f = f_avg
                best_a = avg_a * norm / avg_a.sum()
        
        if verbose and iters % 10000 == 0:
            elapsed = time.monotonic() - t0
            f_new = fast_autocorr(a).max()
            print(f"    SubGrad iter={iters} ({elapsed:.0f}s): f={f_new:.8f}, best={best_f:.8f}")
    
    f_final = fast_autocorr(a).max()
    if f_final < best_f:
        best_f = f_final
        best_a = a.copy()
    
    return best_a, best_f


def build_conv_matrix_sparse(a):
    """Build the convolution matrix T where (T @ b) = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def lp_best_response_reduced(a, M, solver, max_constraints=500):
    """LP with aggressive constraint reduction."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix_sparse(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Start with all constraints, then reduce
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    
    # Iterative constraint reduction
    for round in range(5):
        circ = T @ b_val
        # Find active constraints (within 1e-6 of M)
        tol = max(1e-8 * M, 1e-12)
        active = circ >= M - tol
        
        # Also include some "near-active" constraints for robustness
        if active.sum() < 50:
            # Add top constraints by violation
            violations = M - circ
            top_idx = np.argsort(violations)[::-1][:max(50, active.sum())]
            active[top_idx] = True
        
        n_active = active.sum()
        if n_active >= max_constraints:
            # Keep top max_constraints
            circ_active = circ[active]
            idx = np.where(active)[0]
            order = np.argsort(circ_active)[::-1]
            keep = idx[order[:max_constraints]]
            active = np.zeros(size, dtype=bool)
            active[keep] = True
            n_active = max_constraints
        
        if n_active < 10:
            break
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-5):
            b_val = b_new
        else:
            break
    
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / a.sum()
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    t0 = time.monotonic()
    iters = 0
    stall = 0
    best_iter = 0
    
    # Momentum buffer
    prev_y = a.copy()
    
    while iters < 100000 and time.monotonic() - t0 < time_limit:
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response_reduced(y, M_curr, solver)
        if b is None:
            stall += 10
            if stall > 100:
                # Perturbation
                rng = np.random.default_rng(iters * 7 + 3)
                y = np.maximum(y + rng.normal(0, 0.01 * y.mean(), n), 0)
                y = y * norm / y.sum()
                prev_y = y.copy()
                stall = 0
            iters += 1
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            iters += 1
            continue
        
        g = b * norm / b_sum
        
        # Nesterov step
        iters += 1
        alpha = 2.0 / (iters + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if iters == 1:
            a_new = g.copy()
        else:
            momentum = (iters - 1.0) / (iters + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        prev_y = y.copy()
        y = y_new
        a = a_new
        
        # Check scores
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
            best_iter = iters
        else:
            stall += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
            best_iter = iters
        
        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")
        
        # Restart if stalled
        if stall > 200:
            rng = np.random.default_rng(iters * 13 + 7)
            perturb_scale = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)

            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            if verbose:

    return best_a, best_score


def multiplicative_weights(a0, time_limit, verbose=True):
    """Mirror descent with multiplicative weights update."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / a.sum()
    
    best_a = a.copy()
    best_f = fast_autocorr(a).max()
    t0 = time.monotonic()
    iters = 0
    
    # Initialize step size
    eta = 1.0 / (4.0 * np.sum(a))
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        f_val = b.max()
        
        tol = 1e-8 * f_val
        max_pos = np.where(b >= f_val - tol)[0]
        if len(max_pos) > 5:
            max_pos = max_pos[np.argsort(b[max_pos])[::-1][:5]]
        
        g = compute_subgradient(a, max_pos)
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            break
        
        # Multiplicative weights update: a_i <- a_i * exp(-eta * g_i)
        # Use log-sum-exp for numerical stability
        log_updates = -eta * g
        max_log = log_updates.max()
        updates = np.exp(log_updates - max_log)
        
        a_new = a * updates
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        iters += 1
        
        # Adaptive step size
        eta *= 0.999  # slowly decrease
        
        if iters % 50 == 0:
            f_new = fast_autocorr(a).max()
            if f_new < best_f:
                best_f = f_new
                best_a = a.copy()
        
        if verbose and iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            f_new = fast_autocorr(a).max()
            print(f"    MW iter={iters} ({elapsed:.0f}s): f={f_new:.8f}, best={best_f:.8f}, eta={eta:.8f}")
    
    f_final = fast_autocorr(a).max()
    if f_final < best_f:
        best_f = f_final
        best_a = a.copy()
    
    return best_a, best_f


def peak_shave(a0, time_limit, seed=42, verbose=True):
    """Peak shaving with simulated annealing."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / a.sum()
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = max(0.001, best_score * 0.01 * (1 - elapsed_frac))
        
        b = fast_autocorr(a)
        # Top peak
        k = np.argmax(b)
        peak_val = b[k]
        
        # Find the source that most contributes to this peak
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k - idx]
        
        if contrib.max() < 1e-18:
            n_stall += 1
            if n_stall > 100:
                break
            continue
        
        src = idx[np.argmax(contrib)]
        
        # Try to move mass from src to a random other position
        # Prefer positions that don't contribute to top peaks
        top_k = np.argsort(b)[::-1][:10]
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_k:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2+1] = True
        
        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.where(np.arange(n) != src)[0]
        
        # Try multiple random destinations
        improved = False
        for attempt in range(20):
            if attempt < 10 and len(non_peak) > 0:
                dst = non_peak[rng.integers(len(non_peak))]
            else:
                dst = rng.integers(n)
            
            if dst == src:
                continue
            
            # Move a fraction of src's mass
            frac = rng.choice([0.05, 0.1, 0.15, 0.2, 0.3])
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            
            if a_trial[src] < -1e-12:
                continue
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t
            
            score = fast_autocorr(a_trial).max()
            
            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                n_improve += 1
                improved = True
                break
            elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                a = a_trial
                improved = True
                break
        
        if not improved:
            n_stall += 1
            if n_stall > 500:
                # Larger perturbation
                idx_pert = rng.choice(n, size=max(5, n // 100), replace=False)
                a_trial = a.copy()
                for idx_p in idx_pert:
                    amt = a[idx_p] * rng.uniform(0.2, 0.5)
                    j = rng.integers(n)
                    a_trial[idx_p] -= amt
                    a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score * 1.02:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                n_stall = 0
        
        if verbose and n_improve % 5000 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss
