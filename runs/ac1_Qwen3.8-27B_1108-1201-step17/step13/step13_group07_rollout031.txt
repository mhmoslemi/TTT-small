def pair_swap_optimize(a, i, j):
    """Find optimal redistribution between a[i] and a[j]."""
    n = len(a)
    S = a[i] + a[j]
    if S < 1e-15:
        return a[i], a[j]
    
    b = a.copy()
    b[i] = 0
    b[j] = 0
    
    C_b = np.convolve(b, b)  # Length 2n-1
    
    # For x = a[i], a[j] = S - x:
    # C[k] = C_b[k] + 2x*b[k-i] + 2(S-x)*b[k-j] + x^2*δ(k,2i) + (S-x)^2*δ(k,2j) + 2x(S-x)*δ(k,i+j)
    
    # Linear coefficient of x for each k:
    # 2*b[k-i] - 2*b[k-j]
    # Plus quadratic terms
    
    # Let's compute f(x) = max_k C[k] for x in [0, S]
    # and find x that minimizes it.
    
    # For efficiency, precompute the linear part:
    lin = np.zeros(2*n-1)
    const = C_b.copy()
    
    # 2x*b[k-i]: adds 2*b[k-i] to coefficient of x
    for k in range(2*n-1):
        idx = k - i
        if 0 <= idx < n:
            lin[k] += 2.0 * b[idx]
    # 2(S-x)*b[k-j] = 2S*b[k-j] - 2x*b[k-j]
    for k in range(2*n-1):
        idx = k - j
        if 0 <= idx < n:
            lin[k] += -2.0 * b[idx]
            const[k] += 2.0 * S * b[idx]
    
    # Quadratic terms:
    # x^2 at k=2i: adds x^2
    # (S-x)^2 at k=2j: adds S^2 - 2Sx + x^2 = (S^2) - 2Sx + x^2
    # 2x(S-x) at k=i+j: adds 2Sx - 2x^2
    
    quad_i = 2*i if 2*i < 2*n-1 else -1
    quad_j = 2*j if 2*j < 2*n-1 else -1
    cross_k = i + j if i + j < 2*n-1 else -1
    
    # For each k, C[k] as function of x:
    # Base: const[k] + lin[k]*x
    # Plus: x^2 if k == quad_i
    # Plus: x^2 - 2Sx + S^2 if k == quad_j
    # Plus: 2Sx - 2x^2 if k == cross_k
    
    # So the quadratic coefficient q[k] for x^2:
    q = np.zeros(2*n-1)
    if quad_i >= 0:
        q[quad_i] = 1.0
    if quad_j >= 0:
        q[quad_j] = 1.0
        const[quad_j] += S*S
        lin[quad_j] += -2.0*S
    if cross_k >= 0 and cross_k != quad_i and cross_k != quad_j:
        q[cross_k] += -2.0
        lin[cross_k] += 2.0*S
    elif cross_k >= 0:
        # cross_k coincides with quad_i or quad_j
        # If cross_k == quad_i (i+j = 2i, i.e., j = i, but i != j so this can't happen)
        # If cross_k == quad_j (i+j = 2j, i.e., i = j, also can't happen)
        pass
    
    # Now f(x) = max_k (const[k] + lin[k]*x + q[k]*x^2)
    # Each term is a quadratic (possibly degenerate). The max is convex.
    # Find x in [0, S] that minimizes f(x).
    
    # Use golden section search or ternary search.
    # f is convex, so ternary search works.
    
    def eval_f(x):
        vals = const + lin * x + q * x * x
        return vals.max()
    
    # Ternary search
    a_lo, a_hi = 0.0, S
    ratio = (np.sqrt(5) - 1) / 2  # 0.618...
    
    c = a_hi - ratio * (a_hi - a_lo)
    d = a_lo + ratio * (a_hi - a_lo)
    fc = eval_f(c)
    fd = eval_f(d)
    
    for _ in range(100):  # 100 iterations gives ~100 * log(1/0.618) bits of precision
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - ratio * (a_hi - a_lo)
            fc = eval_f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + ratio * (a_hi - a_lo)
            fd = eval_f(d)
        if a_hi - a_lo < 1e-15:
            break
    
    x_opt = (a_lo + a_hi) / 2
    return x_opt, S - x_opt
