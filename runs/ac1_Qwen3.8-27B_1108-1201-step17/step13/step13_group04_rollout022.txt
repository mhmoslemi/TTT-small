"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Corrected LSE gradient (fixing previous bug) with Nesterov acceleration
2. Dual LSE multiplicative best-response (fast alternative to LP)
3. Direct max subgradient with active-set tracking
4. Enhanced peak-targeting local search
5. Multi-length exploration

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_corr_with_w(a, w, fft_size=None):
    """
    Compute cross-correlation (a ⊛ w)[m] = sum_j a[j] * w[j+m] for m=0..n-1.
    This equals convolve(w, a[::-1])[n-1 : 2n-1]
    """
    n = len(a)
    if fft_size is None:
        total_len = n + len(w) - 1
        fft_size = 1 << (total_len - 1).bit_length()
    A = rfft(a[::-1], fft_size)
    W = rfft(w, fft_size)
    result = irfft(A * W, fft_size)
    return result[n-1 : n-1+n]

def lse_gradient(a, T):
    """
    Correct gradient of LSE surrogate: T*log(sum_k exp(c[k]/T)) where c = a*a.
    grad[m] = 2 * sum_j a[j] * w[m+j] = 2 * (a ⊛ w)[m]
    We return without the factor 2 (can be absorbed into step size).
    """
    n = len(a)
    c = fast_autocorr(a)
    max_c = c.max()
    w = np.exp((c - max_c) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum
    
    # grad[m] = sum_j a[j] * w[m+j] = cross-correlation
    grad = cross_corr_with_w(a, w)
    return grad

def lse_value(a, T):
    """Compute LSE surrogate value."""
    c = fast_autocorr(a)
    max_c = c.max()
    log_sum = max_c / T + np.log(np.sum(np.exp((c - max_c) / T)))
    return T * log_sum

def nesterov_lse(a0, time_limit, T_start=15.0, T_end=0.1, n_stages=16, verbose=True):
    """Nesterov-accelerated LSE gradient descent with geometric cooling."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()

    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_iters_stage = max(30, int(remaining * 2.5 / max(1, n_stages - si)))
        
        # Reset momentum at each temperature stage
        y_prev = y.copy()
        
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break

            # Nesterov extrapolation
            if it == 0:
                y_extrap = y.copy()
            else:
                momentum = (np.sqrt(1 + 4*it*it) - 1) / (2*it)
                y_extrap = y + momentum * (y - y_prev)
                y_extrap = np.maximum(y_extrap, 0)
                s_ye = y_extrap.sum()
                if s_ye > 1e-15:
                    y_extrap = y_extrap * norm / s_ye

            grad = lse_gradient(y_extrap, T)
            
            # Step size: use 1/(it+1) style with scaling
            lr = norm / (T * (1 + it * 0.1))
            y_new = y_extrap - lr * grad
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            y_prev = y.copy()
            y = y_new
            it_count += 1

            # Track best
            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()

            if it_count % 200 == 0:
                elapsed = time.monotonic() - t0
                if verbose:
                    print(f"    LSE stage={si} T={T:.3f} it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def dual_lse_best_response(a, T, max_iters=100, verbose=False):
    """
    Compute approximate LP best response using multiplicative weights on the dual.
    Finds b >= 0, sum(b)=1 that approximately minimizes max_k (a*b)_k.
    """
    n = len(a)
    b = a.copy()
    s = b.sum()
    if s < 1e-15:
        b = np.ones(n) / n
    else:
        b = b / s
    
    log_b = np.log(b + 1e-30)
    
    for it in range(max_iters):
        # Compute d = convolve(a, b)
        d = np.convolve(a, b, mode='full')
        max_d = d.max()
        
        # Weights
        w = np.exp((d - max_d) / T)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(d))
            w[n-1] = 1.0
        else:
            w /= w_sum
        
        # Gradient of LSE(d) with respect to log(b)
        # d[k] = sum_i a[i]*b[k-i]
        # dd[k]/db[j] = a[k-j] (if valid)
        # d(log_b[j]) -> db[j] -> contribution to d[k] is a[k-j]*db[j]
        # So d(d[k])/d(log_b[j]) = a[k-j]*b[j]
        # d(LSE)/d(log_b[j]) = sum_k w[k] * a[k-j] * b[j]
        
        # sum_k w[k] * a[k-j] for k where 0<=k-j<n, i.e., k in [j, j+n-1]
        # = sum_i w[j+i] * a[i] (substituting i = k-j)
        # = cross_correlation(a, w)[j]
        
        grad = cross_corr_with_w(a, w)
        grad = grad * b  # chain rule: d/d(log_b) = b * d/db
        
        # Update
        eta = 1.0 / (T * (1 + it * 0.05))
        log_b -= eta * grad
        log_b -= log_b.max()
        b = np.exp(log_b)
        s = b.sum()
        if s < 1e-15:
            break
        b = b / s
        
        if verbose and it % 20 == 0:
            print(f"      dual_lse it={it}: max_d={max_d:.6f}")
    
    return b


def lp_best_response_fast(a, M, solver=None, verbose=False):
    """
    Solve LP: min t s.t. (a*b)_k <= t for all k, sum(b)=1, b>=0.
    This gives the best response direction.
    """
    import cvxpy as cp
    from scipy.sparse import lil_matrix
    
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    
    # Build constraint matrix (sparse)
    T = lil_matrix((2 * n - 1, n))
    for j in range(n):
        for i in range(n):
            k = i + j
            T[k, j] = a[i]
    T = T.tocsr()
    
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    if solver is None:
        solver = cp.CLARABEL
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def biconvex_optimize(a0, time_limit, verbose=True, use_lp=False):
    """
    Biconvex optimization: alternate between optimizing a (given b) and b (given a).
    Uses LSE gradient for a-optimize step and dual LSE or LP for b-optimize step.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Initialize b
    b = a / a.sum()
    
    outer_it = 0
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5.0:
            break
        
        # Step 1: Optimize a given b (minimize max_k (a*b)_k s.t. sum(a)=norm, a>=0)
        # Use LSE gradient: grad[m] = sum_j b[j] * (something)
        # Actually, (a*b)_k = sum_i a[i]*b[k-i]
        # d(a*b)_k / da[m] = b[k-m] (if valid)
        # So LSE gradient for a: grad_a[m] = sum_k w[k] * b[k-m] = cross_corr(b, w)[m]
        
        n_inner = max(20, int(remaining * 0.4 / max(1, outer_it + 1)))
        T_inner = max(0.5, 5.0 / (outer_it + 1))
        
        for inner_it in range(n_inner):
            if time.monotonic() - t0 > time_limit:
                break
            
            d = np.convolve(a, b, mode='full')
            max_d = d.max()
            w = np.exp((d - max_d) / T_inner)
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.zeros(2*n-1)
                w[n-1] = 1.0
            else:
                w /= w_sum
            
            grad_a = cross_corr_with_w(b, w)
            
            lr = norm / (T_inner * (1 + inner_it * 0.1))
            a_new = a - lr * grad_a
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_a
            a = a_new
        
        # Check score
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Step 2: Optimize b given a (minimize max_k (a*b)_k s.t. sum(b)=1, b>=0)
        if use_lp and remaining > 30:
            b_new = lp_best_response_fast(a / norm, 0, verbose=False)
            if b_new is not None:
                b = b_new
        else:
            # Dual LSE approach for b
            n_inner_b = max(10, int(remaining * 0.2))
            T_b = max(0.3, 3.0 / (outer_it + 1))
            log_b = np.log(b + 1e-30)
            
            for inner_it in range(n_inner_b):
                if time.monotonic() - t0 > time_limit:
                    break
                
                d = np.convolve(a, b, mode='full')
                max_d = d.max()
                w = np.exp((d - max_d) / T_b)
                w_sum = w.sum()
                if w_sum < 1e-300:
                    w = np.zeros(2*n-1)
                    w[n-1] = 1.0
                else:
                    w /= w_sum
                
                grad = cross_corr_with_w(a, w) * b
                eta = 1.0 / (T_b * (1 + inner_it * 0.05))
                log_b -= eta * grad
                log_b -= log_b.max()
                b = np.exp(log_b)
                s = b.sum()
                if s < 1e-15:
                    break
                b = b / s
        
        outer_it += 1
        
        if verbose and outer_it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BiConvex outer={outer_it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def direct_max_subgradient(a0, time_limit, verbose=True):
    """
    Direct subgradient method on max_k (a*a)_k with Nesterov acceleration.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    y_prev = a.copy()
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        # Nesterov extrapolation
        if it == 0:
            y_extrap = y.copy()
        else:
            momentum = (np.sqrt(1 + 4*it*it) - 1) / (2*it)
            y_extrap = y + momentum * (y - y_prev)
            y_extrap = np.maximum(y_extrap, 0)
            s_ye = y_extrap.sum()
            if s_ye > 1e-15:
                y_extrap = y_extrap * norm / s_ye
        
        # Compute subgradient of max(c) at y_extrap
        c = fast_autocorr(y_extrap)
        max_c = c.max()
        
        # Active set: indices where c[k] > max_c - epsilon
        epsilon = max_c * 0.01 + 1e-10
        active = np.where(c >= max_c - epsilon)[0]
        
        # Subgradient = average of gradients at active indices
        # grad[k][m] = 2*a[k-m] * 1{0<=k-m<n}
        # Average over active: (1/|A|) * sum_{k in A} 2*a[k-m] * 1{0<=k-m<n}
        
        # This is: (2/|A|) * sum_{k in A} a[k-m] for valid k-m
        # = (2/|A|) * (indicator_A ⊛ a)[m] where indicator_A has 1s at active positions
        
        ind = np.zeros(2 * n - 1)
        ind[active] = 1.0
        ind /= len(active)
        
        grad = cross_corr_with_w(a, ind)  # Note: we use a here, not y_extrap
        # Actually should use y_extrap for consistency
        grad = cross_corr_with_w(y_extrap, ind)
        
        # Step
        lr = norm / (it + 10)
        y_new = y_extrap - lr * grad
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        y_prev = y.copy()
        y = y_new
        it += 1
        
        score = fast_autocorr(y).max()
        if score < best_score:
            best_score = score
            best_a = y.copy()
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DirectMax it={it} ({elapsed:.0f}s): best={best_score:.8f}, |A|={len(active)}")
    
    return best_a, best_score


def peak_targeting_search(a0, time_limit, seed=42, verbose=True):
    """
    Enhanced peak-targeting local search with multiple move types.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        top_ks = np.argsort(c)[::-1][:8]
        max_c = c[top_ks[0]]
        
        improved = False
        for k_star in top_ks:
            # Find contributing pairs
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_star - idx]
            
            # Top contributors
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:5]]
            
            for src in top_contrib_idx:
                if a[src] < norm * 1e-9:
                    continue
                
                # Build peak mask: positions that contribute to any top peak
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in top_ks:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_mask[js2:je2 + 1] = True
                
                # Candidates: positions NOT contributing to peaks, with low values
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                
                # Sort by value (prefer low values)
                candidates = candidates[np.argsort(a[candidates])[:20]]
                
                for dst in candidates[:8]:
                    if abs(dst - src) < 2:
                        continue
                    for frac in [0.001, 0.003, 0.01, 0.03, 0.07, 0.15]:
                        amt = a[src] * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            it_count += 1
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            if stall % 50 == 0:
                for _ in range(10):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > norm * 1e-8:
                        frac = rng.uniform(0.01, 0.1)
                        amt = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0
                                break
            
            if stall > 500:
                if verbose:
                    print(f"    PeakSearch: stalled at {best_score:.8f} ({time.monotonic()-t0:.0f}s)")
                break
        
        if verbose and it_count % 200 == 0:
            print(f"    PeakSearch it={it_count} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
