import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize a to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(y, T, solver, warm_b=None):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    n = len(y)
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:

        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def exact_subgradient_nesterov(a0, time_limit, lr_init=0.01, momentum_init=0.9):
    """
    Exact subgradient descent with Nesterov acceleration.
    Uses the true subgradient (uniform over argmax set) of the max autocorrelation.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()  # Nesterov position
    v = np.zeros(n)  # velocity
    beta = momentum_init

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute autocorrelation at Nesterov position
        b = fast_autocorr(y)
        bmax = b.max()

        # Find argmax set (within tolerance for numerical stability)
        tol = bmax * 1e-6 + 1e-12
        argmax_set = np.where(b >= bmax - tol)[0]

        # Compute exact subgradient: average gradient over argmax set
        grad = np.zeros(n)
        for k in argmax_set:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * y[k - idx]
        grad /= len(argmax_set)

        # Scale by 2n (the objective is 2n * max_b)
        grad *= 2.0 * n

        # Nesterov step
        lr = lr_init / (1.0 + 0.01 * t ** 0.5)
        v = beta * v - lr * grad
        y = y + v
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y / s_y

        # Evaluate
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            a = y.copy()  # reset Nesterov from best

        # Decay momentum
        beta = 0.999 * beta + 0.001 * 0.9
        beta = min(beta, 0.99)

        if t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def coord_descent_anneal(a0, time_limit, seed=42):
    """
    Coordinate descent with simulated annealing.
    For each top peak, find the best (src, dst) transfer that reduces the peak.
    Uses annealing to escape local optima.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    T_current = 0.05  # initial temperature
    T_min = 0.001
    T_decay = 0.9995

    while time.monotonic() - t0 < time_limit - 5:
        b = fast_autocorr(a)
        bmax = b.max()

        # Find top peaks
        sorted_k = np.argsort(b)[::-1]
        top_peaks = sorted_k[:8]

        improved = False

        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            # For each source in the peak's contribution, try to move mass
            src_candidates = idx[np.argsort(contrib)[::-1][:5]]

            for src in src_candidates:
                if a[src] < 1e-10:
                    continue

                # Find safe destinations (not contributing to top peaks)
                danger = np.zeros(n, dtype=bool)
                for k2 in top_peaks:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    danger[js2:je2 + 1] = True

                # Get safe positions sorted by current value (prefer low)
                safe_mask = ~danger
                safe_mask[src] = False
                if safe_mask.sum() > 0:
                    safe_positions = np.where(safe_mask)[0]
                    safe_positions = safe_positions[np.argsort(a[safe_positions])]
                    dst_pool = safe_positions[:10]
                else:
                    dst_pool = np.where(np.arange(n) != src)[0][:10]

                for dst in dst_pool[:6]:
                    # Try different fractions
                    for frac in [0.003, 0.01, 0.03, 0.05, 0.08, 0.12]:
                        amt = a[src] * frac
                        if amt < 1e-14:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-12:
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-12:
                            continue
                        a_trial = a_trial / st

                        new_bmax = fast_autocorr(a_trial).max()
                        new_score = 2.0 * n * new_bmax

                        if new_score < best_score:
                            # Accept (improvement)
                            a = a_trial
                            best_score = new_score
                            best_a = a.copy()
                            improved = True
                        elif rng.random() < np.exp(-(new_score - best_score) / T_current):
                            # Accept with probability (simulated annealing)
                            a = a_trial
                            improved = True

                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        T_current = max(T_min, T_current * T_decay)

        if not improved and T_current <= T_min * 1.01:
            break

    return best_a, best_score


def interpolate_n(a, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = normalize(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp


def peak_shave_final(a0, time_limit, seed=123):
    """
    Final aggressive peak shaving: iteratively reduce the single largest peak
    by finding the optimal redistribution.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        k_max = np.argmax(b)
        bmax = b[k_max]

        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_max - idx]

        if contrib.max() < 1e-15:
            break

        # Find the most contributing source
        src = idx[np.argmax(contrib)]
        src_contrib = contrib.max()

        # For destination, prefer positions that don't contribute to other peaks
        # Compute "danger" for each position
        danger = np.zeros(n)
        top_k = np.argsort(b)[::-1][:12]
        for k2 in top_k:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            danger[js2:je2 + 1] += b[k2]

        # Safe destinations: low danger, not src
        danger[src] = np.inf
        safe_sorted = np.argsort(danger)
        dst_pool = safe_sorted[:max(5, n // 20)]

        improved_this_iter = False
        for dst in dst_pool[:12]:
            # Binary search for maximum fraction we can transfer
            lo, hi = 0.0, a[src] * 0.5
            best_frac = 0.0
            for _ in range(8):
                mid = (lo + hi) / 2
                if mid < 1e-15:
                    lo = mid
                    continue
                a_try = a.copy()
                a_try[src] -= mid
                a_try[dst] += mid
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st < 1e-12:
                    hi = mid
                    continue
                a_try = a_try / st
                new_bmax = fast_autocorr(a_try).max()
                if new_bmax < bmax * (1 - 1e-9):
                    lo = mid
                    best_frac = mid
                else:
                    hi = mid

            if best_frac > 1e-14:
                a_try = a.copy()
                a_try[src] -= best_frac
                a_try[dst] += best_frac
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st > 1e-12:
                    a_try = a_try / st
                    new_bmax = fast_autocorr(a_try).max()
                    new_score = 2.0 * n * new_bmax
                    if new_score < best_score:
                        a = a_try
                        best_score = new_score
                        best_a = a.copy()
                        improved_this_iter = True
                        break

        if not improved_this_iter:
            # Try random perturbation
            stall += 1
            if stall > 5:
                n_pert = rng.integers(1, 4)
                a_try = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_try[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.02)
                        amt = a_try[i] * frac
                        a_try[i] -= amt
                        a_try[j] += amt
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st > 1e-10:
                    a_try = a_try / st
                    score = 2.0 * n * fast_autocorr(a_try).max()
                    if score < best_score:
                        a = a_try
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 50:
                break

    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Enhanced LP Nesterov zero-sum game with adaptive restart
2. Exact subgradient descent with projected Nesterov momentum
3. Aggressive coordinate descent with simulated annealing
4. Multi-length exploration with interpolation
5. Final peak-shaving polish with binary search

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_seq(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Project a vector onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted) - 1.0
    ind = np.arange(1, n + 1)
    cond = v_sorted - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    k = np.max(np.where(cond)[0]) + 1
    tau = cssv[k - 1] / k
    result = np.maximum(v - tau, 0)
    s = result.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return result / s


def build_toeplitz(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(y, T, solver, warm_b=None):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    n = len(y)
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_nesterov_optimize(a0, time_limit, solver, max_iters=500000):
    """Enhanced Nesterov LP game with adaptive restart and running average tracking."""
    n = len(a0)
    a = normalize_seq(a0)
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    warm_b = None
    no_improve = 0
    patience = 400

    avg_buf = np.zeros(n)
    avg_count = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        T = build_toeplitz(y)
        b_star, M_val = solve_lp_best_response(y, T, solver, warm_b)
        if b_star is None:
            warm_b = None
            no_improve += 1
            if no_improve > patience:
                y = best_a.copy()
                a = best_a.copy()
                warm_b = None
                no_improve = 0
                patience = max(150, patience // 2)
            continue
        warm_b = b_star

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        avg_buf += a
        avg_count += 1

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            no_improve = 0

        if avg_count >= 30 and avg_count % 30 == 0:
            avg_a = avg_buf / avg_count
            score_avg = 2.0 * n * fast_autocorr(avg_a).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_a.copy()
                no_improve = 0

        if no_improve > patience:
            a = best_a.copy()
            y = best_a.copy()
            avg_buf = np.zeros(n)
            avg_count = 0
            no_improve = 0
            warm_b = None
            patience = max(150, patience // 2)

        if t % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LPGame t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def subgradient_projected(a0, time_limit):
    """
    Projected exact subgradient with Nesterov momentum.
    Uses uniform subgradient over argmax set, projects onto simplex.
    """
    n = len(a0)
    a = normalize_seq(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    v = np.zeros(n)
    beta = 0.9
    lr = 0.05

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(y)
        bmax = b.max()

        tol = max(bmax * 1e-5, 1e-12)
        argmax_set = np.where(b >= bmax - tol)[0]

        # Exact subgradient: average over argmax set
        grad = np.zeros(n)
        for k in argmax_set:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * y[k - idx]
        grad /= len(argmax_set)
        grad *= 2.0 * n

        # Nesterov step with projection
        lr_t = lr / (1.0 + 0.05 * np.sqrt(t))
        v = beta * v - lr_t * grad
        y_new = y + v
        y = project_simplex(y_new)

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Adaptive learning rate: decrease if stalling
        if t % 500 == 0 and score_y > best_score * 1.01:
            lr *= 0.9
            beta = min(beta * 1.001, 0.99)

        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def coord_descent_anneal(a0, time_limit, seed=42):
    """
    Coordinate descent with simulated annealing.
    Tries to reduce top peaks by redistributing mass from contributing
    positions to safe positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_seq(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    T_curr = 0.03
    T_min = 0.0005
    T_decay = 0.9997

    iter_count = 0
    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        top_peaks = np.argsort(b)[::-1][:10]

        improved = False

        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.sum() < 1e-15:
                continue

            # Top contributing sources
            src_order = idx[np.argsort(contrib)[::-1][:6]]

            # Compute danger map
            danger = np.zeros(n)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] += b[k2]

            for src in src_order:
                if a[src] < 1e-10:
                    continue
                danger[src] = np.inf
                safe_mask = danger < np.inf
                safe_positions = np.where(safe_mask)[0]
                if len(safe_positions) == 0:
                    safe_positions = np.where(np.arange(n) != src)[0]
                safe_sorted = safe_positions[np.argsort(danger[safe_positions])]
                dst_pool = safe_sorted[:8]

                for dst in dst_pool:
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.06, 0.10]:
                        amt = a[src] * frac
                        if amt < 1e-14:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-12:
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-12:
                            continue
                        a_trial = a_trial / st

                        new_score = 2.0 * n * fast_autocorr(a_trial).max()

                        if new_score < best_score:
                            a = a_trial
                            best_score = new_score
                            best_a = a.copy()
                            improved = True
                        elif T_curr > T_min and rng.random() < np.exp(-(new_score - best_score) / T_curr):
                            a = a_trial
                            improved = True

                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        iter_count += 1
        T_curr = max(T_min, T_curr * T_decay)

        if not improved and T_curr <= T_min * 1.005 and iter_count > 100:
            break

    return best_a, best_score


def peak_shave_binary(a0, time_limit, seed=123):
    """
    Final peak shaving using binary search for optimal transfer amount.
    Focuses on the single largest peak and finds the maximum safe transfer.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_seq(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        k_max = int(np.argmax(b))

        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_max - idx]

        if contrib.max() < 1e-15:
            break

        # Try top-3 contributing sources
        src_order = idx[np.argsort(contrib)[::-1][:3]]

        # Danger map for destinations
        danger = np.zeros(n)
        top_k = np.argsort(b)[::-1][:15]
        for k2 in top_k:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            danger[js2:je2 + 1] += b[k2]

        improved_this = False
        for src in src_order:
            danger[src] = np.inf
            safe_sorted = np.argsort(danger)
            dst_pool = safe_sorted[:10]

            for dst in dst_pool:
                # Binary search for max safe fraction
                lo = 0.0
                hi = a[src] * 0.4
                best_frac = 0.0

                for _ in range(10):
                    mid = (lo + hi) / 2.0
                    if mid < 1e-14:
                        lo = mid
                        continue
                    a_try = a.copy()
                    a_try[src] -= mid
                    a_try[dst] += mid
                    a_try = np.maximum(a_try, 0)
                    st = a_try.sum()
                    if st < 1e-12:
                        hi = mid
                        continue
                    a_try = a_try / st
                    new_bmax = fast_autocorr(a_try).max()
                    if new_bmax < bmax * (1.0 - 1e-9):
                        lo = mid
                        best_frac = mid
                    else:
                        hi = mid

                if best_frac > 1e-13:
                    a_try = a.copy()
                    a_try[src] -= best_frac
                    a_try[dst] += best_frac
                    a_try = np.maximum(a_try, 0)
                    st = a_try.sum()
                    if st > 1e-12:
                        a_try = a_try / st
                        new_score = 2.0 * n * fast_autocorr(a_try).max()
                        if new_score < best_score:
                            a = a_try
                            best_score = new_score
                            best_a = a.copy()
                            improved_this = True
                            break
            if improved_this:
                break

        if not improved_this:
            stall += 1
            if stall > 3:
                # Random kick
                n_pert = int(rng.integers(1, 4))
                a_try = a.copy()
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a_try[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.03)
                        amt = a_try[i] * frac
                        a_try[i] -= amt
                        a_try[j] += amt
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st > 1e-10:
                    a_try = a_try / st
                    score = 2.0 * n * fast_autocorr(a_try).max()
                    if score < best_score:
                        a = a_try
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 40:
                break

    return best_a, best_score


def interpolate_seq(a, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = normalize_seq(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    else:
        a_interp = np.ones(new_n) / new_n
    return a_interp


def try_update(state, a, label):
    """Update best sequence if candidate is better."""
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < state['best_score']:
        state['best_score'] = sc
        state['best_seq'] = a.copy()
        elapsed = time.monotonic() - state['t0']
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization:
    1. LP Nesterov game (35% time)
    2. Projected subgradient (15% time)
    3. Coordinate descent + annealing (20% time)
    4. Multi-length exploration (10% time)
    5. LP game round 2 on best (10% time)
    6. Final peak shaving (10% time)
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(time_budget) - 80.0, 1700.0)

    solver = find_solver()
    print(f"LP solver: {solver}, total budget: {time_budget:.0f}s")

    state = {'best_seq': None, 'best_score': float('inf'), 't0': t_global}

    # Try to use parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            state['best_seq'] = parent.copy()
            state['best_score'] = p_score
    except Exception:
        pass

    if state['best_seq'] is None:
        n0 = 1600
        state['best_seq'] = np.ones(n0) / n0
        state['best_score'] = 2.0
        print(f"Initialized uniform: n={n0}, score={state['best_score']:.6f}")

    n_base = len(state['best_seq'])

    # === Phase 1: LP Nesterov game ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 600.0)
    print(f"\n=== Phase 1: LP Nesterov n={n_base}, time={t1:.0f}s ===")
    if t1 > 60:
        a_out, s_out = lp_nesterov_optimize(state['best_seq'], t1, solver)
        try_update(state, a_out, f"lp_n{n_base}")

    # === Phase 2: Projected subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 260.0)
    print(f"=== Phase 2: Subgradient n={n_base}, time={t2:.0f}s ===")
    if t2 > 30:
        a_out, s_out = subgradient_projected(state['best_seq'], t2)
        try_update(state, a_out, f"subgrad_n{n_base}")

    # === Phase 3: Coordinate descent + annealing ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 340.0)
    print(f"=== Phase 3: Coord+anneal n={n_base}, time={t3:.0f}s ===")
    if t3 > 20:
        a_out, s_out = coord_descent_anneal(state['best_seq'], t3, seed=777)
        try_update(state, a_out, f"coord_n{n_base}")

    # === Phase 4: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 170.0)
    print(f"=== Phase 4: Multi-length, time={t4:.0f}s ===")
    if t4 > 50:
        n_long = n_base + 300
        t4a = t4 * 0.6
        init_long = interpolate_seq(state['best_seq'], n_long)
        a_out, _ = lp_nesterov_optimize(init_long, t4a, solver)
        try_update(state, a_out, f"lp_long_{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            n_long2 = n_base + 500
            t4b = min(remaining * 0.5, 80.0)
            if t4b > 20:
                init_long2 = interpolate_seq(state['best_seq'], n_long2)
                a_out, _ = lp_nesterov_optimize(init_long2, t4b, solver)
                try_update(state, a_out, f"lp_long2_{n_long2}")

    # === Phase 5: LP game round 2 on current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 200.0)
    print(f"=== Phase 5: LP round2, time={t5:.0f}s ===")
    if t5 > 30:
        a_out, _ = lp_nesterov_optimize(state['best_seq'], t5, solver)
        try_update(state, a_out, "lp_round2")

    # === Phase 6: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.25, 100.0)
    print(f"=== Phase 6: Peak shave, time={t6:.0f}s ===")
    if t6 > 15:
        a_out, _ = peak_shave_binary(state['best_seq'], t6, seed=999)
        try_update(state, a_out, "peak_final")

    # === Phase 7: Final coord polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t7 = min(remaining * 0.5, 50.0)
        if t7 > 10:
            print(f"=== Phase 7: Final coord polish ===")
            a_out, _ = coord_descent_anneal(state['best_seq'], t7, seed=2024)
            try_update(state, a_out, "coord_final")

    # Final verification
    if state['best_seq'] is not None:
        fs = evaluate_safe(state['best_seq'])
        if np.isfinite(fs):
            state['best_score'] = min(state['best_score'], fs)
        print(f"\nFINAL: score={state['best_score']:.6f}, n={len(state['best_seq'])}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in state['best_seq']]
