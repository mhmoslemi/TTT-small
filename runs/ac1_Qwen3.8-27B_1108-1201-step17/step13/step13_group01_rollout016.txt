def lse_obj_and_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    exp_vals = np.exp(T_temp * (b - max_b))
    w = exp_vals / exp_vals.sum()
    lse = max_b + np.log(exp_vals.sum()) / T_temp
    obj_val = 2.0 * n * lse / (s * s)

    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]

    conv_grad = 4.0 * cross
    grad = conv_grad / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad.ravel()

            _, g = lse_obj_and_grad(x, n, T)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.15 * np.cos(4.0 * x) + 0.08 * np.cos(8.0 * x) + 0.03 * np.cos(14.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-5.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 4.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 0.6 + 0.3 * np.sin(3.0 * np.pi * x) + 0.1 * np.sin(7.0 * np.pi * x)
        a = np.abs(a)
    else:
        a = 1.0 / (1.0 + np.exp(-8.0 * (np.linspace(0.0, 1.0, n) - 0.5)))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Multi-length scan
    print("\n=== Phase 0: Multi-length scan ===")
    scan_lengths = [1600, 2000, 2400]
    scan_results = {}

    for idx_scan, n_scan in enumerate(scan_lengths):
        elapsed_so_far = time.monotonic() - t_global
        if elapsed_so_far > time_budget * 0.12:
            break
        remaining_scan = time_budget * 0.12 - elapsed_so_far
        scan_time = max(10.0, remaining_scan / max(1, len(scan_lengths) - idx_scan))

        if best_seq is not None:
            init_seq = interpolate_sequence(best_seq, n_scan)
        else:
            init_seq = structured_init(n_scan, rng)

        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, scan_time, solver, verbose=False,
                                              use_linesearch=False)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: score={sc_out:.8f} (t={time.monotonic()-t_global:.0f}s)")

    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        scan_best_a, scan_best_sc = scan_results[best_scan_n]
        do_try_update(scan_best_a, f"scan_n{best_scan_n}")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Deep LP-Nesterov at base length with line search
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Deep LP n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.42
    a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True,
                                     use_linesearch=True, sparse_top_k=None)
    do_try_update(a_out, f"lp_deep_n{n_base}")

    # Phase 2: Try longer length if we have time
    n_long = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP n={n_long} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.18, 250.0)
    if t2 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True,
                                         use_linesearch=True, sparse_top_k=None)
        do_try_update(a_out, f"lp_long_n{n_long}")

    # Phase 3: LSE + L-BFGS at current best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LSE scipy n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_lbfgs_optimize(a_norm, t3, T_start=50.0, T_end=1.5, n_stages=7)
        do_try_update(a_out, f"lse_scipy_n{n_best}")

    # Phase 4: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t4, seed=12345)
        do_try_update(a_out, "coord_polish")

    # Phase 5: Multi-restart exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-restart n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 20:
        n_restarts = 5
        per_restart = t5 / n_restarts
        for ri in range(n_restarts):
            if time.monotonic() - t_global > time_budget:
                break
            if per_restart < 5:
                break
            # Create perturbed version
            perturbation = rng.random(n_best) * 0.1
            a_perturbed = best_seq + perturbation
            a_perturbed = np.maximum(a_perturbed, 0.0)
            s_p = a_perturbed.sum()
            if s_p < 1e-10:
                continue
            a_perturbed = a_perturbed * np.sqrt(2.0 * n_best) / s_p
            a_out, _ = nesterov_lp_optimize(a_perturbed, per_restart, solver,
                                             verbose=False, use_linesearch=False)
            do_try_update(a_out, f"restart_{ri}")

    # Phase 6: Mirror descent for diverse angle
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Mirror descent n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = mirror_descent_optimize(a_norm, t6, seed=999, verbose=False)
        do_try_update(a_out, "mirror")

    # Phase 7: Final coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord n={n_best} (budget={remaining:.0f}s) ===")
    t7 = remaining * 0.5
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t7, seed=77777)
        do_try_update(a_out, "coord_final")

    # Phase 8: Final LSE polish at very low temperature
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 8: LSE final n={n_best} (budget={remaining:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_lbfgs_optimize(a_norm, remaining * 0.4, T_start=5.0, T_end=0.3, n_stages=4)
        do_try_update(a_out, "lse_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
