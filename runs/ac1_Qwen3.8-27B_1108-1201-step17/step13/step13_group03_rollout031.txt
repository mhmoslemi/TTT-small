def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def _lse_grad(y, T):
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def _build_conv_matrix(a, n):
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j: j + n, j] = a
    return T


def _lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    A_fft = rfft(a, fft_size)

    # Build constraint matrix using the Toeplitz structure
    # conv(a, b)[k] = sum_j a[j] * b[k-j]
    # For each k in 0..2n-2: sum over valid j of a[j]*b[k-j] <= M
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(2 * n - 1, j + n)
        T[start:end, j] = a[j:j + (end - start)]

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = _fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = _fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        # Store intermediate averages at specific points
        if n_avg in [100, 500, 2000, 10000]:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = _fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            avg_lp = np.mean(lp_times[-20:]) if len(lp_times) >= 20 else np.mean(lp_times)
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s, lp={avg_lp:.2f}s/iter): best={best_score:.8f}")

    return best_a, best_score


def _gradient_local_search(a0, time_limit, verbose=False, top_k=12):
    """Targeted local search using analytical gradient of peak autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, _fast_autocorr(a).max()
    a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:top_k]
        improved = False

        for k in top_peaks:
            # Compute which indices contribute to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            total = contrib.sum()
            if total < 1e-15:
                continue

            # Find the index with highest contribution
            src = idx[np.argmax(contrib)]

            # Build mask of indices involved in top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                if len(idx := np.arange(js2, je2 + 1)) > 0:
                    peak_mask[idx] = True
                peak_mask[k - idx if False else np.arange(js2, je2 + 1)] = True

            # Find destination: low-value indices not in peak regions
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.setdiff1d(np.arange(n), [src])

            # Sort candidates by ascending value (move to low-value spots)
            candidates_sorted = candidates[np.argsort(a[candidates])][:max(5, len(candidates) // 8)]

            for dst in candidates_sorted[:5]:
                if dst == src:
                    continue
                for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18 or a[src] < amt:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = _fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            rng = np.random.default_rng(time.monotonic() * 1000 % 100000)
            n_random = min(20, n // 10)
            for _ in range(n_random):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = _fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                stall = 0
                                break

            if stall > 300:
                break

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GradLocalSearch t={elapsed:.0f}s: best={best_score:.8f}, stall={stall}")

    return best_a, best_score


def _spectral_reweight(a0, time_limit, verbose=False):
    """Use spectral information to guide shape optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, _fast_autocorr(a).max()
    a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()

    # Compute power spectrum
    A = rfft(a, 2 * n)
    P = np.abs(A) ** 2

    # Find frequency bins with highest power (these contribute most to autocorrelation peaks)
    # Apply spectral subtraction: reduce amplitude at peak frequencies
    for iteration in range(50):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, 2 * n)
        P = np.abs(A) ** 2
        P_max = P.max()

        # Apply soft thresholding (spectral shrinkage)
        threshold = P_max * 0.7
        A_modified = A.copy()
        mask = P > threshold
        A_modified[mask] *= (1.0 - 0.1 * (P[mask] - threshold) / P_max)

        # Recover time domain
        a_new = irfft(A_modified, 2 * n)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm / s_new

        score = _fast_autocorr(a_new).max()
        if score < best_score - 1e-14:
            a = a_new
            best_score = score
            best_a = a.copy()
        else:
            # Try less aggressive threshold
            for thresh_frac in [0.8, 0.9, 0.95]:
                A_mod2 = A.copy()
                mask2 = P > P_max * thresh_frac
                A_mod2[mask2] *= (1.0 - 0.05 * (P[mask2] - P_max * thresh_frac) / P_max)
                a_try = irfft(A_mod2, 2 * n)[:n]
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                sc_try = _fast_autocorr(a_try).max()
                if sc_try < best_score - 1e-14:
                    a = a_try
                    best_score = sc_try
                    best_a = a.copy()
                    break
            break

        if verbose and iteration % 5 == 0:
            print(f"    SpectralReweight iter={iteration} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _interpolate_sequence(f, new_n, noise_level=0.0):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    if noise_level > 0:
        rng = np.random.default_rng(42 + new_n)
        f_interp += noise_level * rng.random(new_n)
        f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def _dual_guided_update(a, M, solver, time_limit_per_lp=5.0):
    """Use LP dual variables to guide reweighting of the sequence."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0) * norm / max(a.sum(), 1e-15)

    b_var = cp.Variable(n, nonneg=True)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j: j + n, j] = a[j:j + (2*n-1-j)]

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None

    # Try to get dual values for guidance
    # The dual variables tell us which constraints are binding
    b_sum = b_val.sum()
    g = b_val * norm / b_sum
    return g


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: LP-Nesterov with multi-length and multi-method strategy."""

    best_seq = None

    parent = None

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            best_seq = parent.copy()
            best_score = _evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={best_score:.6f}")
    except Exception:
        pass

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LP-Nesterov at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 300.0)
    if t1 > 30:

    # === Phase 3: Gradient local search ===

        return [1.0]

    return [float(x) for x in best_seq]
