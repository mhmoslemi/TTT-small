import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.signal import fftconvolve


def fast_autocorr(a):
    """Compute full autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the global evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def lse_gradient_step(a, T, fft_size_cache=None):
    """
    Compute one gradient step for the LSE-smoothed objective.
    grad_i = (2/T) * [sum_j a[j]*w[j+i] + sum_k w[k]*a[i-k]]
    where w_k = exp(conv_k/T) / sum exp(conv_k/T)
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Compute autocorrelation
    A = rfft(a, fft_size)
    conv = irfft(A * A, fft_size)[:size]
    
    # Compute weights
    max_conv = conv.max()
    exp_conv = np.exp((conv - max_conv) / T)
    w = exp_conv / exp_conv.sum()
    
    # Compute gradient components using FFT convolution
    # s2[i] = sum_k=0^i w[k] * a[i-k] = conv(a, w[:n])[i]
    w_n = w[:n]
    W_n = rfft(w_n, fft_size)
    s2 = irfft(A * W_n, fft_size)[:n]
    
    # s1[i] = sum_j=0^{n-1} a[j] * w[j+i]
    # = conv(a, w_rev)[2n-2-i] where w_rev = w[::-1]
    w_rev = w[::-1]
    W_rev = rfft(w_rev, fft_size)
    full_conv_s1 = irfft(A * W_rev, fft_size)
    # full_conv_s1 has length 3n-2 (we need fft_size >= 3n-2 for this)
    # Actually, fft_size >= 2n-1 (for conv of a and w), but conv(a, w_rev) has length n + (2n-1) - 1 = 3n-2
    # So we need fft_size >= 3n-2
    # Let me use a larger FFT size
    fft_size2 = 1 << (3 * n - 2 - 1).bit_length()
    A2 = rfft(a, fft_size2)
    W_rev2 = rfft(w_rev, fft_size2)
    full_conv_s1 = irfft(A2 * W_rev2, fft_size2)[:3*n-2]
    
    # s1[i] = full_conv_s1[2n-2-i] for i=0,...,n-1
    s1 = full_conv_s1[2*n-2 : n-2 : -1]
    # In Python: full_conv_s1[2n-2], full_conv_s1[2n-3], ..., full_conv_s1[n-1]
    # That's n elements: indices 2n-2, 2n-3, ..., n-1
    # Python slice: [2n-2 : n-2 : -1] gives indices 2n-2, 2n-3, ..., n-1 ✓
    
    grad = (2.0 / T) * (s1 + s2)
    return grad, conv, w


def lse_gradient_descent(a0, time_limit, verbose=True):
    """
    LSE gradient descent with adaptive temperature and step size.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Temperature schedule: start high, decrease geometrically
    conv0 = fast_autocorr(a)
    T_start = conv0.max()  # Initial temperature = peak value
    T_end = T_start * 0.01  # Final temperature
    n_stages = 12
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        # Number of steps proportional to remaining time and T
        n_steps = int(remaining * 2)  # Try ~2 steps per second initially
        n_steps = max(5, min(n_steps, 200))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            grad, conv, w = lse_gradient_step(a, T)
            
            # Adaptive step size: use the gradient norm
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            
            # Step size: want to move by roughly norm * 0.1 in the direction opposite to grad
            # projected onto the simplex
            step = 0.1 / grad_norm * norm
            
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            
            # Check improvement
            score = fast_autocorr(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            
            # Accept if not too bad (allow some uphill for exploration)
            if score < conv.max() * 1.05:
                a = a_new
            else:
                # Revert
                a = best_a.copy() if score > best_score else a
        
        if verbose and (si % 3 == 0 or si == len(temps) - 1):
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si}/{n_stages} T={T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def pair_cd_fast(a0, time_limit, rng, verbose=True):
    """
    Fast pair coordinate descent targeting the peak positions.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    n_passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        
        # Find peak positions
        peak_threshold = best_score * 0.8
        peak_positions = np.where(conv >= peak_threshold)[0]
        
        # For each peak position k, the pairs (i, j) with i+j=k (or |i-j|=k dependence)
        # are the most important. Target pairs that affect peaks.
        
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            
            # Try transferring from i to j for a few j values
            # Choose j to target peak reduction
            candidates = []
            for k in peak_positions[:5]:  # Top 5 peaks
                j = k - i  # If we move mass to j, it affects peak k
                if 0 <= j < n and j != i:
                    candidates.append(j)
                j2 = i + k  # Another influence
                if 0 <= j2 < n and j2 != i:
                    candidates.append(j2)
            
            if not candidates:
                candidates = [int(rng.integers(n))]
            
            for j in candidates[:2]:
                if j == i or a[i] < norm * 1e-6:
                    continue
                
                # Binary search for optimal delta
                lo, hi = 0.0, a[i]
                best_d = 0.0
                best_val = best_score
                
                # First, try a few sample points
                for d in [a[i] * f for f in [0.1, 0.25, 0.5, 0.75, 0.9]]:
                    a_trial = a.copy()
                    a_trial[i] -= d
                    a_trial[j] += d
                    a_trial = a_trial * norm / a_trial.sum()
                    val = fast_autocorr(a_trial).max()
                    if val < best_val:
                        best_val = val
                        best_d = d
                
                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    a = np.maximum(a, 0.0)
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)
                    cur = conv.max()
                    if cur < best_score - 1e-10:
                        best_score = cur
                        best_a = a.copy()
                        pass_improved = True
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 8)
    if choice == 0:
        # Cosine bump
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        # Sine power
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        # Multi-frequency
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        # Two Gaussian bumps
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.75) ** 2)
    elif choice == 4:
        # Triangle^3
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        # Asymmetric (skewed to reduce peak)
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 0.5 * np.exp(-2.0 * x)
    elif choice == 6:
        # "U-shape" - large at ends, small in middle
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * (2.0 * x - 1.0) ** 2
    else:
        # Random smooth
        freqs = rng.integers(1, 10, size=5)
        amps = rng.uniform(0.1, 1.0, size=5)
        phases = rng.uniform(0, 2*np.pi, size=5)
        x = np.linspace(0, 2*np.pi, n)
        a = np.ones(n) * 0.5
        for f, am, ph in zip(freqs, amps, phases):
            a += am * np.cos(f * x + ph)
        a = np.maximum(a, 0.1)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

Let me also think about the step size more carefully. The LSE approximation with temperature T approximates the max function. The gradient of LSE is:
grad = (1/T) * sum_k w_k * d(conv_k)/d(a)

where w_k = exp(conv_k/T) / sum exp(conv_j/T)

When T is small, w_k is concentrated on the peak(s), and the gradient is essentially the subgradient of the max function. When T is large, it's a smoothed version.

For the step size, I want to move in the direction that reduces the objective. The objective is f(a) = max_k conv_k (after normalization). The LSE approximation is F_T(a) = T * log(sum exp(conv_k/T)).

The gradient descent step is: a_new = a - lr * grad(F_T, a)

But I also need to maintain the constraint a >= 0 and sum(a) = norm.

A common approach is:
1. Take the gradient step
2. Project onto {a >= 0, sum(a) = norm}

The projection is: a_proj = max(0, a - step * grad), then normalize to sum = norm.

Now let me also think about a key insight: the optimal sequence for this problem likely has a specific structure. Let me think about what the "best response" looks like.

Given a current sequence a, the best response b maximizes sum(b) subject to conv(a,b)_k <= M for all k.

The active constraints are the positions k where conv(a,a)_k is close to M (the peak positions). At these positions, the constraint is tight.

If the peak is at position k*, then the constraint is:

This is a linear constraint on b. The best response tries to put as much mass as possible into b while respecting these constraints.

Intuitively, if the peak is at the center (k* = n-1), then the constraint is:

This means b should be large where a[n-1-j] is small. So b is "anti-correlated" with the reversed a.

This is the key structural insight: the best response is concentrated where the current sequence is small (at the peak lag positions).

Over iterations, this creates a "ping-pong" effect where the sequence mass moves to positions that are complementary to the current peak structure.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LSE gradient descent with FFT-based O(n log n) gradient (key innovation)
2. Multiplicative (mirror) descent for natural constraint handling
3. Pair coordinate descent with peak-targeted selection
4. Multi-length exploration with interpolation transfer
The gradient simplifies to a single cross-correlation (not two), enabling
~3 FFTs per step, allowing 100x more iterations than O(n^2) approaches.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_autocorr(a):
    """Compute full autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using global evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def compute_lse_gradient(a, T):
    """
    Compute gradient of LSE-smoothed objective.
    grad[i] = (2/T) * sum_{j=0}^{n-1} a[j] * w[j+i]
    where w = softmax(conv(a,a)/T)
    Returns (gradient, conv, w)
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Autocorrelation
    A = rfft(a, fft_size)
    conv = irfft(A * A, fft_size)[:size]
    
    # Softmax weights (shifted for numerical stability)
    max_conv = conv.max()
    exp_conv = np.exp((conv - max_conv) / T)
    w = exp_conv / exp_conv.sum()
    
    # Cross-correlation: cross[i] = sum_j a[j] * w[j+i]
    # Use FFT: irfft(FFT(a) * conj(FFT(w)))
    # Need N >= n + (2n-1) - 1 = 3n-2
    N = 1 << (3 * n - 2 - 1).bit_length()
    A_N = rfft(a, N)
    W_N = rfft(w, N)
    cross = irfft(A_N * np.conj(W_N), N)[:n]
    
    grad = (2.0 / T) * cross
    return grad, conv, w


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE gradient descent with temperature annealing and adaptive steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0  # Normalize to sum=1 (score is scale-invariant)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    conv0 = fast_autocorr(a)
    T_start = conv0.max()
    T_end = max(T_start * 0.003, 1e-8)
    n_stages = 15
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break
        
        # More steps at lower T (sharper landscape needs finer steps)
        frac = si / max(n_stages - 1, 1)
        n_steps = max(20, int(remaining * (2 + 3 * (1 - frac))))
        n_steps = min(n_steps, 300)
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            grad, conv, w = compute_lse_gradient(a, T)
            gmax = np.abs(grad).max()
            if gmax < 1e-15:
                break
            
            # Adaptive step size
            lr = 0.3 / (gmax + 1e-15) * a.max()
            lr = min(lr, a.max() * 2.0)
            
            # Gradient step with projection
            a_trial = a - lr * grad
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            a_trial = a_trial / s_trial
            
            score = fast_autocorr(a_trial).max()
            
            if score < best_score:
                best_score = score
                best_a = a_trial.copy()
            
            # Accept/reject
            cur_max = conv.max()
            if score <= cur_max * 1.01:
                a = a_trial
            elif score < best_score:
                a = a_trial
            else:
                # Halve step and retry once
                a_trial2 = a - 0.5 * lr * grad
                a_trial2 = np.maximum(a_trial2, 0.0)
                s2 = a_trial2.sum()
                if s2 > 1e-15:
                    a_trial2 = a_trial2 / s2
                    score2 = fast_autocorr(a_trial2).max()
                    if score2 < cur_max:
                        a = a_trial2
                        if score2 < best_score:
                            best_score = score2
                            best_a = a_trial2.copy()
        
        if verbose and (si % 3 == 0 or si == n_stages - 1):
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si}/{n_stages-1} T={T:.6f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def multiplicative_descent(a0, time_limit, verbose=True):
    """
    Mirror descent (multiplicative weight updates) on the LSE-smoothed objective.
    a_new[i] = a[i] * exp(-lr * grad[i]) / Z
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-12)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    conv0 = fast_autocorr(a)
    T_start = conv0.max()
    T_end = max(T_start * 0.005, 1e-8)
    n_stages = 12
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break
        n_steps = max(10, int(remaining * 3))
        n_steps = min(n_steps, 200)
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            grad, conv, w = compute_lse_gradient(a, T)
            gmax = np.abs(grad).max()
            if gmax < 1e-15:
                break
            
            # Learn rate for multiplicative update
            lr = 0.5 / (gmax + 1e-15)
            
            # Multiplicative update
            log_a = np.log(a)
            log_a_new = log_a - lr * grad
            # Shift for numerical stability
            log_a_new = log_a_new - log_a_new.max()
            a_trial = np.exp(log_a_new)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            a_trial = a_trial / s_trial
            
            score = fast_autocorr(a_trial).max()
            
            if score < best_score:
                best_score = score
                best_a = a_trial.copy()
            
            cur_max = conv.max()
            if score <= cur_max * 1.02:
                a = a_trial
            else:
                # Try smaller step
                log_a_new2 = log_a - 0.3 * lr * grad
                log_a_new2 = log_a_new2 - log_a_new2.max()
                a_trial2 = np.exp(log_a_new2)
                s2 = a_trial2.sum()
                if s2 > 1e-15:
                    a_trial2 = a_trial2 / s2
                    score2 = fast_autocorr(a_trial2).max()
                    if score2 < best_score:
                        best_score = score2
                        best_a = a_trial2.copy()
                    if score2 < cur_max:
                        a = a_trial2
        
        if verbose and (si % 3 == 0 or si == n_stages - 1):
            elapsed = time.monotonic() - t0
            print(f"    MIR stage {si}/{n_stages-1} T={T:.6f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
