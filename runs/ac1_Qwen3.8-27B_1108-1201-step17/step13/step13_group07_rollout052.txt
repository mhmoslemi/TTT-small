def peak_shaving(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_k = b.argmax()
        
        # Find top contributor to peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        src = idx[np.argmax(contrib)]
        
        # Find best dst: position that minimizes increase in max autocorrelation
        # Quick heuristic: move to a position where a is small and it doesn't create new peaks
        # Try a few random positions
        best_trial = None
        best_trial_score = best_score
        
        for _ in range(50):
            dst = rng.integers(n)
            if dst == src:
                continue
            for frac in [0.001, 0.005, 0.01, 0.02]:
                amt = a[src] * frac
                if amt < 1e-15:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_trial_score:
                    best_trial_score = score
                    best_trial = a_trial.copy()
        
        if best_trial is not None:
            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
        else:
            # Try random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:

    return irfft(A * A, fft_size)[:size]


def fast_conv(x, y):
    """Full convolution of x and y via FFT."""
    n = len(x) + len(y) - 1
    fft_size = 1 << (n - 1).bit_length()
    X = rfft(np.asarray(x, dtype=np.float64), fft_size)
    Y = rfft(np.asarray(y, dtype=np.float64), fft_size)
    return irfft(X * Y, fft_size)[:n]


def internal_score(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)

def build_T(y):

def solve_lp_dual(y):
    """Solve dual LP: max mu s.t. sum(lambda)=1, lambda>=0, (T^T@lambda)_j <= mu for all j.
    This gives a lower bound on the optimal value."""
    n = len(y)

    # Variables: [lambda (2n-1); mu]

    A_ub[:, :2*n-1] = T.T

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        ss = result.sum()
        if ss > 0:
            result = result / ss * target_sum
        else:
            result = np.full(n, target_sum / n)
    return result


def spectral_shaping(a, n_iter=400, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = internal_score(a)
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    return best_a

    stall_count = 0
    stall_threshold = 6000
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Solve dual LP: given lambda, find best response b
        # Actually, let's solve the primal LP with the current a as the fixed sequence
        b_star, M_val = solve_lp_fast(a)
        if b_star is None:
            continue
        
        # Update a via Nesterov
        alpha_t = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha_t) * a + alpha_t * b_star
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Nesterov extrapolation
        if t == 1:
            a_extrap = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_extrap = a_new + momentum * (a_new - a)
            a_extrap = project_simplex(a_extrap)
        
        a = a_new
        # Use extrapolated point for next LP
        a_for_lp = a_extrap
        
        # Update geometric average with a (not extrapolated, for stability)
        current_weight *= geo_rate
        geo_avg += current_weight * a
        geo_weight_sum += current_weight
        
        score_a = internal_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        score_extrap = internal_score(a_extrap)
        if score_extrap < best_score:
            best_score = score_extrap
            best_a = a_extrap.copy()
            stall_count = 0
        
        if t % 300 == 0:
            geo_check = geo_avg / geo_weight_sum
            score_geo = internal_score(geo_check)
            if score_geo < best_score:
                best_score = score_geo
                best_a = geo_check.copy()
                stall_count = 0
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NL-dual t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M_val:.8f}")
        
        if stall_count > stall_threshold:
            a = best_a.copy()
            t = max(1, t // 4)
            stall_count = 0
            geo_avg = np.zeros(n)
            geo_weight_sum = 0.0
            current_weight = 1.0
    
    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """Coordinate-descent peak shaving with adaptive moves."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_k = int(b.argmax())
        
        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        src_ranking = idx[np.argsort(contrib)[::-1]]
        
        improved = False
        for src in src_ranking[:3]:
            if a[src] < 1e-12:
                continue
            # Find dst candidates: positions not in peak region, with small a
            peak_areas = set()
            for k2 in np.argsort(b)[::-1][:5]:
                js2 = max(0, int(k2) - n + 1)
                je2 = min(n - 1, int(k2))
                peak_areas.update(range(js2, je2 + 1))
            
            candidates = [j for j in range(n) if j not in peak_areas and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            # Sort by a[j] ascending (prefer moving to low positions)
            candidates_arr = np.array(candidates)
            sorted_candidates = candidates_arr[np.argsort(a[candidates_arr])[:min(20, len(candidates_arr))]]
            
            for dst in sorted_candidates[:5]:
                for frac in [0.001, 0.003, 0.008, 0.015, 0.03]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            # Random perturbation to escape local minima
            if no_improve % 10 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        no_improve = 0
            if no_improve > 3000:
                break
    
    return best_a, best_score


def subgradient_optimize(a0, time_limit, verbose=True):
    """Subgradient optimization with smoothening."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    v = np.zeros(n)
    beta = 0.9
    base_lr = 0.05
    t = 0
    avg = np.zeros(n)
    n_avg = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(a)
        # Softmax weights on peaks
        T_smooth = max(0.5, 30.0 * np.exp(-t / 5000.0))
        Tb = T_smooth * (b - b.min())
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        
        # Gradient via convolution
        fft_size = 1 << (3 * n - 2).bit_length()
        A_rev = rfft(a[::-1].copy(), fft_size)
        W = rfft(w, fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]
        
        lr_t = base_lr / (1.0 + 0.005 * t)
        v = beta * v + (1 - beta) * grad
        a_new = a - lr_t * v
        a_new = project_simplex(a_new)
        
        avg += a_new
        n_avg += 1
        
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        if n_avg % 500 == 0:
            avg_check = avg / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall = 0
        
        a = a_new
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.0f}s): best={best_score:.8f}")
        
        if stall > 10000:
            break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Spectral shaping initialization
    Phase 2: Nesterov LP (primary, most time)
    Phase 3: Second Nesterov LP round with different params
    Phase 4: Multi-length exploration
    Phase 5: Subgradient optimization
    Phase 6: Peak shaving (final refinement)
    """
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = float('inf')
    
    # Load starting point
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = internal_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = internal_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)
    
    # Phase 1: Spectral shaping (quick)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 5:
        for alpha_val in [0.4, 0.25, 0.12]:
            a_out = spectral_shaping(best_seq, n_iter=300, alpha=alpha_val)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Spectral alpha={alpha_val}: {best_score:.8f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 1: best={best_score:.8f} (t={elapsed:.0f}s)")
    
    # Phase 2: Nesterov LP - primary optimizer (give most time)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Nesterov LP n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.40, 800.0)
    if t2 > 60:
        a_out, sc_out = nesterov_lp_optimize(best_seq, t2)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [NL_round1]: {best_score:.8f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 2: best={best_score:.8f} (t={elapsed:.0f}s)")
    
    # Phase 3: Second Nesterov LP round (continue from current best)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Nesterov LP round 2 n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30:
        a_out, sc_out = nesterov_lp_optimize(best_seq, t3)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [NL_round2]: {best_score:.8f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 3: best={best_score:.8f} (t={elapsed:.0f}s)")
    
    # Phase 4: Try longer length
    n_long = int(round(len(best_seq) * 1.25))
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Longer n={n_long} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 300.0)

        init_long = interpolate_sequence(best_seq, n_long)
        # Quick spectral

            best_score = sc
            best_seq = a_spec.copy()

    remaining = time_budget - (time.monotonic() - t_global_start)

        a_out, sc_out = subgradient_optimize(best_seq, t5)
        sc = internal_score(a_out)
        if sc < best_score:

            best_seq = a_out.copy()

    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 5: best={best_score:.8f} (t={elapsed:.0f}s)")
    
    # Phase 6: Peak shaving (final refinement)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Peak shaving n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.25, 200.0)
    if t6 > 10:
        a_out, sc_out = peak_shaving(best_seq, t6, seed=123)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [peak_shave_1]: {best_score:.8f}")
    
    # Phase 7: One more LP round if time remains
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final LP n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.40, 250.0)
    if t7 > 30:
        a_out, sc_out = nesterov_lp_optimize(best_seq, t7, verbose=False)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [LP_final]: {best_score:.8f}")
    
    # Phase 8: Final peak shave
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t8 = min(remaining * 0.50, 100.0)
        if t8 > 5:
            a_out, sc_out = peak_shaving(best_seq, t8, seed=789)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [peak_shave_2]: {best_score:.8f}")
    
    # Final verification
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
