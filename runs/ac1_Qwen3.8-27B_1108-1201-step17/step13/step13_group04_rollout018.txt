import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csr_matrix

# Pre-compute helper functions

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score_normed(a):
    """Compute score assuming sum(a) = 1."""
    n = len(a)
    b = fast_autocorr(a)
    return 2.0 * n * b.max()

def build_toeplitz_sparse(a):
    """Build the sparse constraint matrix for the LP."""
    n = len(a)
    T = lil_matrix((2*n-1, n))
    for j in range(2*n-1):
        k_min = max(0, j - (n-1))
        k_max = min(n-1, j)
        for k in range(k_min, k_max+1):
            T[j, k] = a[j-k]
    return T.tocsr()

def solve_lp(a, M):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    T = build_toeplitz_sparse(a)
    b_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    prob.solve(solver=cp.CLARABEL, verbose=False)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    return np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    n = len(a)

    m_start1 = max(0, k_star - n + 1)

    """Weighted subgradient using top-K peaks with exponential weighting."""
    n = len(a)
    # Get top K peaks
    if K < 2 * n - 1:
        top_idx = np.argpartition(b, -K)[-K:]
    else:
        top_idx = np.arange(len(b))
    
    # Exponential weights
    max_b = b[top_idx].max()
    w = np.exp((b[top_idx] - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Uniform over top K
        w = np.ones(len(top_idx)) / len(top_idx)
    else:
        w = w / w_sum
    
    # Compute weighted gradient
    g = np.zeros(n)
    for k_star, wk in zip(top_idx, w):
        if wk < 1e-15:
            continue
        g += wk * subgrad_hardmax(a, int(k_star))
    return g

def proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    if not cond.any():
        return np.ones(n) * (target / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a,'full'))/(sum(a))^2 using:
1. Fast subgradient methods (O(n) gradient) with Nesterov acceleration
2. Exponentiated gradient (mirror descent) for stability
3. LP best-response with barycentric averaging for precision
4. Multi-length exploration, symmetrization, and local polish
Key: O(n) subgradient enables 10x more iterations than FFT-gradient methods.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix

def fast_autocorr(a):
    """Full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_autocorr(a, b):
    """Compute score given a and its autocorrelation b."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    return float(2.0 * n * b.max() / (s * s))

def subgrad_at_peak(a, k_star):
    """O(n) subgradient of the autocorrelation peak at k_star."""
    n = len(a)
    g = np.zeros(n)
    # Term 1: a[k_star - m] for valid m
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] += a[k_star - idx]
    # Term 2: a[m - k_star] for valid m
    lo2 = max(0, k_star)
    hi2 = min(n - 1, k_star + n - 1)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2 + 1)
        g[idx] += a[idx - k_star]
    return g

def weighted_subgrad(a, b, T, K):
    """Weighted subgradient using top-K peaks."""
    n = len(a)
    b_len = len(b)
    K_eff = min(K, b_len)
    top_idx = np.argpartition(b, -K_eff)[-K_eff:]
    max_bk = b[top_idx].max()
    w = np.exp((b[top_idx] - max_bk) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(K_eff) / K_eff
    else:
        w /= w_sum
    g = np.zeros(n)
    for ks, wi in zip(top_idx, w):
        if wi > 1e-15:
            g += wi * subgrad_at_peak(a, int(ks))
    return g

def proj_simplex(v, target=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    if not cond.any():
        return np.full(n, target / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def normalize_to_sum1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def nesterov_subgrad_opt(a0, time_limit, T_init=5.0, T_end=0.1, K=5, verbose=True):
    """
    Nesterov-accelerated subgradient method on the simplex.
    Minimizes max(a*a) subject to sum(a)=1, a>=0.
    """
    n = len(a0)
    a = normalize_to_sum1(a0)
    best_a = a.copy()
    best_b = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    
    # Barycentric tracking
    bary_a = np.zeros(n)
    bary_wsum = 0.0
    
    # Temperature schedule
    total_est = int(time_limit * 200)  # estimate total iterations
    temps = np.geomspace(T_init, max(T_end, 0.01), max(10, total_est // 5000))
    
    it = 0
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break
        n_iters = max(200, int(remaining * 150 / max(1, len(temps) - si)))
        
        for _ in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1
            
            b = fast_autocorr(x)
            if T < 0.05:
                k_star = b.argmax()
                g = subgrad_at_peak(x, k_star)
            else:
                g = weighted_subgrad(x, b, T, K)
            
            # Nesterov
            alpha = 2.0 / (it + 2.0)
            if it == 1:
                y = x - alpha * g
            else:
                momentum = (it - 1.0) / (it + 2.0)
                y_extrap = x + momentum * (x - x_prev)
                y = y_extrap - alpha * g
            
            x_prev = x
            x = proj_simplex(y, 1.0)
            
            # Barycentric
            w_t = 2.0 / (it + 2.0)
            bary_a += w_t * x
            bary_wsum += w_t
            
            # Track best
            score = fast_autocorr(x).max()
            if score < best_b:
                best_b = score
                best_a = x.copy()
            
            # Periodic barycentric check
            if it % 200 == 0:
                bc = bary_a / bary_wsum
                bc = np.maximum(bc, 0)
                s_bc = bc.sum()
                if s_bc > 1e-15:
                    bc = bc / s_bc
                    sc_bc = fast_autocorr(bc).max()
                    if sc_bc < best_b:
                        best_b = sc_bc
                        best_a = bc.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    NSU stage {si+1}/{len(temps)} T={T:.3f} it={it} ({elapsed:.0f}s): best={best_b:.8f}")
    
    return best_a, best_b

def mirror_descent_subgrad(a0, time_limit, T=1.0, K=5, verbose=True):
    """
    Exponentiated gradient (mirror descent) with subgradient.
    """
    n = len(a0)
    a = normalize_to_sum1(a0)
    best_a = a.copy()
    best_b = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    it = 0
    for it in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        if T < 0.05:
            k_star = b.argmax()
            g = subgrad_at_peak(a, k_star)
        else:
            g = weighted_subgrad(a, b, T, K)
        
        eta = 5.0 / (it + 1.0) ** 0.7
        log_a = np.log(a + 1e-30) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        s = a_new.sum()
        if s < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        score = fast_autocorr(a).max()
        if score < best_b:
            best_b = score
            best_a = a.copy()
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MDS it={it} ({elapsed:.0f}s): best={best_b:.8f}")
    
    return best_a, best_b

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    # Build constraint: conv(a, b) <= M
    # conv(a,b)[j] = sum_i a[i] * b[j-i] for valid (i, j-i)
    T_mat = lil_matrix((2 * n - 1, n))
    for j in range(2 * n - 1):
        k_lo = max(0, j - (n - 1))
        k_hi = min(n - 1, j)
        for k in range(k_lo, k_hi + 1):
            T_mat[j, k] = a[j - k]
    T_mat = T_mat.tocsr()
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T_mat @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_nesterov_opt(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and barycentric averaging."""
    n = len(a0)
    a = normalize_to_sum1(a0)
    best_a = a.copy()
    best_b = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    y_prev = a.copy()
    bary_a = np.zeros(n)
    bary_wsum = 0.0
    
    it = 0
    for it in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-15:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        
        g = b / b.sum()  # normalize to sum = 1
        
        alpha = 2.0 / (it + 2.0)
        if it == 1:
            y_new = (1.0 - alpha) * y + alpha * g
        else:
            momentum = (it - 1.0) / (it + 2.0)
            y_extrap = y + momentum * (y - y_prev)
            y_new = (1.0 - alpha) * y_extrap + alpha * g
        
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new / y_new.sum()
        
        y_prev = y
        y = y_new
        
        # Barycentric
        w_t = 2.0 / (it + 2.0)
        bary_a += w_t * y
        bary_wsum += w_t
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_b:
            best_b = score_y
            best_a = y.copy()
        
        if it % 10 == 0:
            bc = bary_a / bary_wsum
            bc = np.maximum(bc, 0)
            s_bc = bc.sum()
            if s_bc > 1e-15:
                bc = bc / s_bc
                sc = fast_autocorr(bc).max()
                if sc < best_b:
                    best_b = sc
                    best_a = bc.copy()
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPN it={it} ({elapsed:.0f}s): best={best_b:.8f}")
    
    return best_a, best_b

def symmetrize(a):
    """Average with reverse; may reduce peak by convexity."""
    return (a + a[::-1]) / 2.0

def local_polish_peak(a0, time_limit, seed=42, verbose=True):
    """Peak-targeting local search: move mass from peak contributors to low-impact positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_to_sum1(a0)
    best_a = a.copy()
    best_b = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Top peaks
        top_ks = np.argsort(b)[::-1][:3]
        
        improved = False
        for k_star in top_ks:
            # Find top contributors to this peak
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_star - idx]
            top_idx = idx[np.argsort(contrib)[::-1][:3]]
            
            for src in top_idx:
                if a[src] < 1e-10:
                    continue
                # Find low-impact positions (not contributing to top peaks)
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in top_ks:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_mask[js2:je2 + 1] = True
                
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                # Pick low-value candidates
                candidates = candidates[np.argsort(a[candidates])[:5]]
                
                for dst in candidates:
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0)
                        s_t = a_t.sum()
                        if s_t < 1e-15:
                            continue
                        a_t = a_t / s_t
                        sc = fast_autocorr(a_t).max()
                        if sc < best_b - 1e-12:
                            a = a_t
                            best_b = sc
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random kick
            if rng.integers(10) == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a_t = a.copy()
                    a_t[i] -= amt
                    a_t[j] += amt
                    a_t = np.maximum(a_t, 0)
                    s_t = a_t.sum()
                    if s_t > 1e-15:
                        a_t = a_t / s_t
                        sc = fast_autocorr(a_t).max()
                        if sc < best_b:
                            a = a_t
                            best_b = sc
                            best_a = a.copy()
                            stall = 0
            if stall > 500:
                break
        
        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Polish stall={stall} ({elapsed:.0f}s): best={best_b:.8f}")
    
    return best_a, best_b

def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
