import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Full autocorrelation a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_weights(b, T):
    """Compute softmax weights for smoothing."""
    b_shifted = b - b.max()
    w = np.exp(b_shifted / T)
    s = w.sum()
    if s < 1e-300:
        return np.ones(len(b)) / len(b)
    return w / s

def smooth_gradient(a, T):
    """Compute smoothed gradient of max(autocorr)."""
    n = len(a)
    b = fast_autocorr(a)
    w = softmax_weights(b, T)
    # grad_i = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is np.correlate(w, a, 'valid')
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return b.max(), grad

def adam_step(params, grads, beta1, beta2, eps, t):
    """Adam update step."""
    m = params['m']
    v = params['v']
    m = beta1 * m + (1 - beta1) * grads
    v = beta2 * v + (1 - beta2) * grads ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    step = m_hat / (np.sqrt(v_hat) + eps)
    params['m'] = m
    params['v'] = v
    return step
```

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def softmax_vjpd(z, grad_a):
    """VJP of softmax: convert gradient w.r.t. a to gradient w.r.t. z."""
    a = softmax_fn(z)
    S = np.dot(grad_a, a)
    return a * (grad_a - S)
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast FFT-based smooth gradient descent (log-sum-exp with Adam)
2. LP Nesterov zero-sum game with constraint generation
3. Peak shaving refinement
4. Multi-length exploration
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_smooth_grad_fft(a, T):
    """
    Compute smoothed gradient of max(conv(a,a)) w.r.t. a using FFT.
    Returns (max_value, gradient).
    """
    n = len(a)
    c = fast_autocorr(a)  # length 2n-1
    c_shifted = c - c.max()
    w = np.exp(c_shifted / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w = w / w_sum
    
    # grad_i = 2 * sum_{k: valid} w[k] * a[k-i] where k-i in [0, n-1]
    # Equivalently: grad_i = 2 * sum_{j: 0<=j<=n-1} w[i+j] * a[j]
    # This is a cross-correlation: grad = 2 * correlate(w, a, 'valid')
    # Using FFT: conv(w, a_flip) then extract
    a_flip = a[::-1]
    # conv length = (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (3*n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_flip, fft_size)
    conv = irfft(W * A, fft_size)[:3*n-2]
    # grad[i] = 2 * conv[i + n - 1] for i = 0, ..., n-1
    grad = 2.0 * conv[n-1:2*n-1]
    return c.max(), grad

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def softmax_vjpd(z, grad_a):
    """VJP of softmax: grad_a -> grad_z."""
    a = softmax_fn(z)
    S = np.dot(grad_a, a)
    return a * (grad_a - S)

def smooth_adam_optimize(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=30, lr=0.005):

    Adam-optimized smooth gradient descent with temperature annealing.
    Uses FFT-based gradient computation for speed.
    """

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    z = np.log(np.maximum(a, 1e-30))
    t0 = time.monotonic()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Adjust learning rate
        lr_s = lr * min(1.0, 100.0 / n) * (0.95 ** si)
        
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        stall = 0
        
        # Estimate iterations: each iteration ~ 3-5 FFTs of size ~4096-8192
        # Should take ~0.05-0.2ms for n=1600
        # So ~5000-20000 iters per second
        max_it = int(remaining * 5000)
        max_it = max(max_it, 1000)
        
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break
            
            a = softmax_fn(z)
            b_max, grad_a = compute_smooth_grad_fft(a, T)
            grad_z = softmax_vjpd(z, grad_a)
            
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast FFT-based smooth gradient descent (full log-sum-exp with Adam)
2. LP Nesterov zero-sum game with adaptive constraint generation
3. Multi-scale peak shaving refinement
4. Multi-length exploration with interpolation
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_smooth_grad_fft(a, T):
    """
    Compute smoothed gradient of max(conv(a,a)) w.r.t. a using FFT.
    Returns (max_value, gradient).
    """
    n = len(a)
    c = fast_autocorr(a)
    c_shifted = c - c.max()
    w = np.exp(c_shifted / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w = w / w_sum

    # grad_i = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # = 2 * cross-correlation, computed via FFT
    a_flip = a[::-1]
    conv_len = len(w) + n - 1
    fft_size = 1 << (conv_len).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_flip, fft_size)
    conv = irfft(W * A, fft_size)[:conv_len]
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return c.max(), grad


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def softmax_vjpd(z, grad_a):
    """VJP of softmax: convert grad wrt output to grad wrt log-input."""
    a = softmax_fn(z)
    S = np.dot(grad_a, a)
    return a * (grad_a - S)

        b_star = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
        bs = b_star.sum()
        if bs < 1e-15:
            warm_b = None
            continue
        b_star = b_star / bs
        warm_b = b_star

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        ys = y_new.sum()
        if ys < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ys

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        if stall > 300 and t > 50:
            a = best_a.copy()
            y = best_a.copy()
            stall = 0
            warm_b = None

        if t - last_print > 500:
            elapsed = time.monotonic() - t0
            print(f"      LP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={float(M_var.value):.8f}")
            last_print = t

    return best_a, best_score


def peak_shaving_refine(a0, time_limit, seed=42):
    """
    Multi-scale peak shaving with global verification.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        c = fast_autocorr(a)
        cmax = c.max()
        threshold = cmax * 0.98
        top_peaks = np.where(c >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(c)[::-1][:10]

        danger = np.zeros(n, dtype=bool)
        for kp in top_peaks[:20]:
            js = max(0, kp - (n - 1))
            je = min(n - 1, kp)
            danger[js:je + 1] = True

        improved = False
        for k in top_peaks[:15]:
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue

            top_srcs = idx[np.argsort(contrib)[::-1][:3]]

            safe = np.where(~danger)[0]
            safe = safe[safe != top_srcs[0]]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != top_srcs[0])[0]
            safe_sorted = safe[np.argsort(a[safe])]

            for src in top_srcs:
                for dst in safe_sorted[:8]:
                    if dst == src:
                        continue
                    for frac in [0.003, 0.008, 0.015, 0.03, 0.06]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-12:
                            continue
                        a_trial = a_trial / st
                        new_max = fast_autocorr(a_trial).max()
                        if new_max < cmax * (1 - 1e-9):
                            a = a_trial
                            new_score = 2.0 * n * new_max
                            if new_score < best_score:
                                best_score = new_score
                                best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 8 == 0:
                n_pert = rng.integers(1, 5)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.04)
                        amt = a_trial[i]
                        amt = a_trial
