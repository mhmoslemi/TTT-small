def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    # Build constraint matrix T where (T @ b)[k] = conv(a, b)[k]
    # conv(a, b)[k] = sum_i a[i] * b[k-i]
    # For fixed k, this involves b[0..n-1] with coefficients from a
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    T = np.zeros((size, n))
    for j in range(n):
        # b[j] contributes to conv positions k where a[k-j] is valid
        # i.e., k-j in [0, n-1], so k in [j, j+n-1]
        start = j
        end = min(j + n, size)
        # T[k, j] = a[k-j] for k in [j, j+n-1]
        T[start:end, j] = a[0:end-j]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Adaptive step size
        alpha = 2.0 / (t + 2.0)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        start = j
        end = min(j + n, size)
        T[start:end, j] = a[0:end - j]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def ftrl_lp_optimize(a0, time_limit, solver, verbose=True):
    """FTRL-accelerated LP best-response iteration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    nesterov_m = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # FTRL step size
        gamma_t = 1.0 / np.sqrt(t)
        # Blend Nesterov and FTRL
        alpha = 2.0 / (t + 2.0)
        alpha = max(alpha, 0.1 * gamma_t)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Nesterov extrapolation
        if t == 1:
            a_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            a_new = y_new + beta * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        if time.monotonic() - last_improve > 40.0:
            stall_count += 1
            if stall_count > 10:
                if verbose:
                    print(f"    FTRL stalling at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0

        if verbose and n_iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FTRL iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_obj_grad_single(x, n, T):
    """LSE objective and gradient for a single temperature."""
    a = np.maximum(x.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    exp_vals = np.exp(T * (b - max_b))
    log_sum = np.log(exp_vals.sum())
    lse = max_b + log_sum / T
    w = exp_vals / exp_vals.sum()

    size = 2 * n - 1
    fft_size = 1 << (2 * size).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size)
    w_pad[:size] = w
    A_f = rfft(a_pad, fft_size)
    W_f = rfft(w_pad, fft_size)
    cross = irfft(A_f * np.conj(W_f), fft_size)[:n]
    conv_grad = 2.0 * cross / T
    obj_val = 2.0 * n * lse / (s * s)
    grad = conv_grad / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad.ravel()


def lse_multi_temperature_obj(x, n, temps, weights):
    """Combined multi-temperature LSE objective."""
    a = np.maximum(x.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    total_lse = 0.0
    for T, w in zip(temps, weights):
        exp_vals = np.exp(T * (b - max_b))
        total_lse += w * (max_b + np.log(exp_vals.sum()) / T)
    return 2.0 * n * total_lse / (s * s)

        lse_T = max_b + log_sum / T
        w_k = exp_vals / exp_vals.sum()
        total_lse += w * lse_T
        w_pad = np.zeros(fft_size)
        w_pad[:size] = w_k
        W_f = rfft(w_pad, fft_size)
        cross = irfft(A_f * np.conj(W_f), fft_size)[:n]
        total_conv_grad += w * 2.0 * cross / T

    obj_val = 2.0 * n * total_lse / (s * s)
    grad = total_conv_grad / (s * s) - 2.0 * obj_val * a / s
    return grad.ravel()


def lse_scipy_optimize(a0, time_limit, T_list, weights=None):
    """LSE multi-temperature L-BFGS optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    if weights is None:
        weights = np.ones(len(T_list)) / len(T_list)

    def obj_func(x):
        return lse_multi_temperature_obj(x, n, T_list, weights)

    def grad_func(x):
        return lse_multi_temperature_grad(x, n, T_list, weights)

    try:
        res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                             jac=grad_func,
                             options={'maxiter': 500, 'maxfun': 2000,
                                      'gtol': 1e-8, 'time_limit': time_limit})
        if res.x is not None:
            a_candidate = np.maximum(res.x, 0.0)
            s_c = a_candidate.sum()
            if s_c > 1e-10:
                a_candidate = a_candidate * norm / s_c
                score_c = fast_autocorr(a_candidate).max()
                if score_c < best_score:
                    best_score = score_c
                    best_a = a_candidate.copy()
    except Exception as e:
        print(f"    LSE optimize error: {e}")

    return best_a, best_score


def lse_staged_optimize(a0, time_limit):
    """Multi-stage LSE optimization with decreasing temperatures."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    stages = [
        ([50.0, 20.0, 10.0], [0.2, 0.3, 0.5]),
        ([20.0, 10.0, 5.0], [0.2, 0.3, 0.5]),
        ([10.0, 5.0, 2.0], [0.2, 0.3, 0.5]),
        ([5.0, 2.0, 1.0], [0.3, 0.4, 0.3]),
        ([2.0, 1.0, 0.5], [0.3, 0.4, 0.3]),
    ]

    n_stages = len(stages)
    for si, (T_list, weights) in enumerate(stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        stage_time = min(remaining / (n_stages - si), 150.0)
        a_out, sc_out = lse_scipy_optimize(a, stage_time, T_list, weights)
        if sc_out < best_score:
            best_score = sc_out
            best_a = a_out.copy()
            a = a_out

    return best_a, best_score


def particle_move_polish(a0, time_limit, seed=42):
    """Peak-targeted particle moves for final polishing."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        # Find all peaks above 90% of max
        threshold = 0.90 * max_b
        peak_indices = np.where(b >= threshold)[0]
        if len(peak_indices) == 0:
            peak_indices = [int(np.argmax(b))]

        improved = False
        for k in peak_indices:
            # Find contributing pairs to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue
            # Top contributing source positions
            top_frac = 0.15
            top_n = max(1, int(len(idx) * top_frac))
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:top_n]]

            for src in top_contrib_idx:
                if a[src] < 1e-10:
                    continue
                # Find low-value destinations
                dest_mask = np.ones(n, dtype=bool)
                for k2 in peak_indices:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    dest_mask[js2:je2 + 1] = False
                candidates = np.where(dest_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                # Sort by value (move to lower values)
                candidates_sorted = candidates[np.argsort(a[candidates])]
                n_try = min(5, len(candidates_sorted))

                for dst in candidates_sorted[:n_try]:
                    for frac in [0.02, 0.05, 0.10, 0.20]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        new_max = fast_autocorr(a_trial).max()
                        if new_max < best_score - 1e-12:
                            a = a_trial
                            best_score = new_max
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation attempts
            n_rand = min(30, max(5, int(time_limit / 5)))
            for _ in range(n_rand):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.15)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm / s_t
                new_max = fast_autocorr(a_trial).max()
                if new_max < best_score:
                    a = a_trial
                    best_score = new_max
                    best_a = a.copy()
                    stall = 0
                    break
            if stall > 300:
                break

    return best_a, best_score


def subgrad_nesterov(a0, time_limit, verbose=False):
    """Subgradient Nesterov method (fast per iteration)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(y)
        k_star = int(np.argmax(b))
        M_curr = b[k_star]

        # Compute gradient of max at k_star
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * y[k_star - idx]

        # Best response to gradient: put mass where gradient is lowest
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) * (1.0 + 0.3 * np.sin(4 * np.pi * x))
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 * np.sin(np.pi * x) ** 2 + 0.3 * np.sin(2 * np.pi * x) ** 2 + 0.2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns best sequence found."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Quick multi-length scan with structured inits
    print("\n=== Phase 0: Multi-length scan ===")
    scan_lengths = [1200, 1600, 2000, 2400]
    scan_results = {}

    for i_scan, n_scan in enumerate(scan_lengths):
        elapsed_scan = time.monotonic() - t_global
        if elapsed_scan > time_budget * 0.12:
            break
        remaining_for_scan = time_budget * 0.12 - elapsed_scan
        scan_time = min(remaining_for_scan / max(1, len(scan_lengths) - i_scan), 60.0)
        if scan_time < 8:
            break

        if best_seq is not None:
            init_seq = interpolate_sequence(best_seq, n_scan)
        else:
            init_seq = structured_init(n_scan, rng)

        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        a_out, sc_out = ftrl_lp_optimize(a_norm, scan_time, solver, verbose=False)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: score={sc_out:.8f} ({time.monotonic() - t_global:.0f}s)")

    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        scan_best_a, scan_best_sc = scan_results[best_scan_n]
        do_try_update(scan_best_a, f"scan_n{best_scan_n}")

    # Determine base length
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Deep FTRL-LP at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: FTRL-LP n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.35
    a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
    a_out, _ = ftrl_lp_optimize(a_norm, t1, solver, verbose=True)
    do_try_update(a_out, f"ftrl_n{n_base}")

    # Phase 2: LSE staged optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LSE staged n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 180.0)
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_staged_optimize(a_norm, t2)
        do_try_update(a_out, f"lse_staged_n{n_best}")

    # Phase 3: Try longer length
    n_long = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: FTRL-LP n={n_long} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = ftrl_lp_optimize(a_norm, t3, solver, verbose=True)
        do_try_update(a_out, f"ftrl_long_n{n_long}")

    # Phase 4: Subgradient Nesterov (different trajectory)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: SubGrad n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgrad_nesterov(a_norm, t4, verbose=False)
        do_try_update(a_out, f"subgrad_n{n_best}")

    # Phase 5: LSE with high temperatures (peak targeting)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LSE high-T n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t5, T_list=[100.0, 50.0, 20.0],
                                      weights=[0.5, 0.3, 0.2])
        do_try_update(a_out, f"lse_highT_n{n_best}")

    # Phase 6: Particle move polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Particle moves n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.20, 200.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = particle_move_polish(a_norm, t6, seed=12345)
        do_try_update(a_out, "particle_polish")

    # Phase 7: Final LSE with very low temperatures
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: LSE low-T n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 120.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t7, T_list=[5.0, 2.0, 1.0],
                                      weights=[0.3, 0.4, 0.3])
        do_try_update(a_out, "lse_lowT_final")

    # Phase 8: Final particle polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final particle n={n_best} (budget={remaining:.0f}s) ===")
    t8 = max(remaining * 0.4, 10.0)
    if t8 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = particle_move_polish(a_norm, t8, seed=77777)
        do_try_update(a_out, "particle_final")

    # Phase 9: Additional LSE refinement if time remains
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 9: Extra LSE n={n_best} (budget={remaining:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, min(remaining * 0.5, 100.0),
                                      T_list=[30.0, 10.0, 3.0], weights=[0.4, 0.4, 0.2])
        do_try_update(a_out, "lse_extra")

        # Final particle pass
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 15:
            n_best = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = particle_move_polish(a_norm, min(remaining2 * 0.6, 80.0), seed=9999)
            do_try_update(a_out, "particle_extra")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
