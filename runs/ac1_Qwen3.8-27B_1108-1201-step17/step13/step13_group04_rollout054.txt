def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_gradient(a, T):
    """
    Compute gradient of LSE surrogate: T*log(sum_k exp(b_k/T)) where b = autocorr(a).
    dL/da[m] = r1[m] + r2[m] using two FFT convolutions.
    """
    n = len(a)
    size = 2 * n - 1
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(size)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)

    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]

    return r1 + r2

                elapsed = time.monotonic() - t0
                if verbose:
                    print(f"    LSE-v2 stage={si}/{n_stages} T={T:.3f} it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_perturbation_descent(a0, time_limit, seed=42, verbose=True):
    """
    Coordinate descent with exact incremental autocorrelation tracking.
    For each coordinate, tries a set of candidate values and keeps the best.
    Uses the fact that changing a[m] by delta changes b[k] by delta*(a[k-m]+a[m-k]).
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    b = fast_autocorr(a)
    best_score = b.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        order = rng.permutation(n)
        sweep_improved = 0

        for m in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Compute c[k] = db[k]/da[m] = a[k-m] + a[m-k]
            c = np.zeros(2 * n - 1)
            lo1 = max(0, m)
            hi1 = min(2 * n - 2, m + n - 1)
            c[lo1:hi1 + 1] += a[lo1 - m:hi1 - m + 1]
            lo2 = max(0, m - n + 1)
            hi2 = min(2 * n - 2, m)
            c[lo2:hi2 + 1] += a[m - lo2:m - hi2]

            # Try several candidate deltas
            current_val = a[m]
            mean_val = a.mean()
            candidates = []
            if current_val > 1e-15:
                for frac in [-0.7, -0.5, -0.3, -0.15, 0.15, 0.3, 0.5]:
                    candidates.append(current_val * (1 + frac))
            candidates.append(0.0)
            candidates.append(mean_val * 0.5)
            candidates.append(mean_val * 1.5)
            candidates.append(mean_val)

            best_delta_score = b.max()
            best_new_val = current_val

            for new_val in candidates:
                new_val = max(0.0, new_val)
                delta = new_val - current_val
                if abs(delta) < 1e-16:
                    continue
                b_trial_max = np.max(b + delta * c)
                # Also account for sum change in the score
                # score = 2*n*max(b)/(sum(a))^2
                # new_sum = s0 + delta (where s0 = current sum)
                # But since score is scale-invariant, just compare max(b)
                if b_trial_max < best_delta_score - 1e-12:
                    best_delta_score = b_trial_max
                    best_new_val = new_val

            if best_new_val != current_val:
                delta = best_new_val - a[m]
                a[m] = best_new_val
                b += delta * c
                sweep_improved += 1

        # Periodic exact recomputation and renormalization
        if sweep % 3 == 0:
            b = fast_autocorr(a)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * norm / n
                b = fast_autocorr(a)
            else:
                a = a * norm / s
                b = fast_autocorr(a)

        current_score = b.max()
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Random perturbation if stalled
        if no_improve > 30:
            n_perturb = max(5, n // 50)
            for _ in range(n_perturb):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.05, 0.2)
                    c_i = np.zeros(2 * n - 1)
                    lo1 = max(0, i)
                    hi1 = min(2 * n - 2, i + n - 1)
                    c_i[lo1:hi1 + 1] += a[lo1 - i:hi1 - i + 1]
                    lo2 = max(0, i - n + 1)
                    hi2 = min(2 * n - 2, i)
                    c_i[lo2:hi2 + 1] += a[i - lo2:i - hi2]

                    c_j = np.zeros(2 * n - 1)
                    lo1 = max(0, j)
                    hi1 = min(2 * n - 2, j + n - 1)
                    c_j[lo1:hi1 + 1] += a[lo1 - j:hi1 - j + 1]
                    lo2 = max(0, j - n + 1)
                    hi2 = min(2 * n - 2, j)
                    c_j[lo2:hi2 + 1] += a[j - lo2:j - hi2]

                    b_trial = b - amt * c_i + amt * c_j
                    if b_trial.max() < best_score:
                        a[i] -= amt
                        a[j] += amt
                        b = b_trial
                        no_improve = 0

            no_improve = 0  # Reset after perturbation attempt

        sweep += 1
        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordPert sweep={sweep} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. (a*b)_j <= M for all j, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    from scipy.sparse import lil_matrix
    T = lil_matrix((2 * n - 1, n))
    for j in range(2 * n - 1):
        k_start = max(0, j - n + 1)
        k_end = min(n - 1, j)
        for k in range(k_start, k_end + 1):
            T[j, k] = a[j - k]
    T = T.tocsr()

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_nesterov_isotropic(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration + isotropic averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    a_nest = a.copy()
    iso_avg = np.zeros(n)
    n_iso = 0
    n_iters = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 5.0:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_nest = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
            a_nest = a_new

        y = y_new

        # Isotropic averaging
        iso_avg += a_nest
        n_iso += 1

        # Check candidates
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_nest = fast_autocorr(a_nest).max()
        if score_nest < best_score:
            best_score = score_nest
            best_a = a_nest.copy()

        if n_iso % 30 == 0:
            iso_sol = iso_avg / n_iso
            s_iso = iso_sol.sum()
            if s_iso > 1e-15:
                iso_sol = iso_sol * norm / s_iso
                score_iso = fast_autocorr(iso_sol).max()
                if score_iso < best_score:
                    best_score = score_iso
                    best_a = iso_sol.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-iso iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def spectral_peak_reduction(a0, time_limit, verbose=True):
    """
    Uses frequency domain analysis to identify which bins contribute to the peak
    and redistributes mass accordingly.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    iteration = 0
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_idx = np.argmax(b)
        peak_val = b[peak_idx]

        # Find contributing pairs
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_idx - idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-15:
            break

        # Find top contributors
        top_n = min(10, len(idx))
        top_local = np.argsort(contrib)[::-1][:top_n]
        top_idx = idx[top_local]

        # Find "safe" destinations (low contribution to peak and to other top peaks)
        # Compute contribution to ALL top peaks
        top_peaks = np.argsort(b)[::-1][:min(5, len(b))]
        danger = np.zeros(n)
        for kp in top_peaks:
            js2 = max(0, kp - n + 1)
            je2 = min(n - 1, kp)
            for i2 in range(js2, je2 + 1):
                danger[i2] += a[kp - i2]

        # Safe indices: low danger, low current value
        safety = danger + 0.1 * a
        safe_sorted = np.argsort(safety)

        improved = False
        for src in top_idx[:5]:
            if a[src] < norm * 1e-8:
                continue
            for dst in safe_sorted[:10]:
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-16:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try random moves
            moved = False
            for _ in range(20):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-8:
                    frac = rng.uniform(0.02, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            moved = True
                            break
            if not moved:
                # Larger perturbation
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j:
                    amt = a.mean() * 0.5
                    a_trial = a.copy()
                    a_trial[i] = max(0, a_trial[i] - amt)
                    a_trial[j] = a_trial[j] + amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score * 1.1:  # Allow slight increase
                            a = a_trial
                            if score < best_score:
                                best_score = score
                                best_a = a.copy()

        iteration += 1
        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SpectralRed iter={iteration} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.1 * np.cos(7.0 * x) + 0.05 * np.cos(13.0 * x) + 0.03 * np.cos(23.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.5
    elif choice == 5:
        # Multi-gaussian
        a = np.zeros(n)
        for _ in range(rng.integers(3, 7)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.03, n * 0.15)
            amp = rng.uniform(0.3, 1.0)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif choice == 6:
        # Smoothed random
        a = rng.uniform(0.3, 1.0, n)
        # Smooth with moving average
        kernel_size = max(3, n // 50)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
    else:
        # Cosine with random phases
        a = np.zeros(n)
        for k in range(rng.integers(3, 10)):
            freq = rng.integers(1, n // 2)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.1, 0.5) / (k + 1)
            a += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        a += 1.0  # Ensure positivity

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function combining multiple optimization strategies.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver={solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    n_base = 1600
    if parent is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # ============ Phase 1: LSE Nesterov v2 at base length ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LSE-Nesterov-v2 n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.28, 420.0)
    if t1 > 30:
        a_work = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = lse_nesterov_v2(a_work, t1, T_start=18.0, T_end=0.15, n_stages=16, verbose=True)
        do_try_update(a_out, f"lse_{n_base}")

    # ============ Phase 2: Coordinate Perturbation Descent ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_now = len(best_seq)
    print(f"\n=== Phase 2: CoordPertDescent n={n_now} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 380.0)
    if t2 > 20:
        a_work = best_seq * np.sqrt(2.0 * n_now) / best_seq.sum()
        a_out, _ = coordinate_perturbation_descent(a_work, t2, seed=1234, verbose=True)
        do_try_update(a_out, f"coord_{n_now}")

    # ============ Phase 3: LP Nesterov with isotropic averaging ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_now = len(best_seq)
    print(f"\n=== Phase 3: LP-Isotropic n={n_now} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.22, 330.0)
    if t3 > 30:
        a_work = best_seq * np.sqrt(2.0 * n_now) / best_seq.sum()
        a_out, _ = lp_nesterov_isotropic(a_work, t3, solver, verbose=True)
        do_try_update(a_out, f"lp_{n_now}")

    # ============ Phase 4: Spectral peak reduction ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_now = len(best_seq)
    print(f"\n=== Phase 4: SpectralPeakRed n={n_now} (budget={remaining:.0f}s) ===")

    if t4 > 15:
        a_work = best_seq * np.sqrt(2.0 * n_now) / best_seq.sum()
        a_out, _ = spectral_peak_reduction(a_work, t4, verbose=True)
        do_try_update(a_out, f"spectral_{n_now}")

    # ============ Phase 5: Try longer length ============
    n_long = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LSE n={n_long} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:

        a_out, _ = lse_nesterov_v2(a_work, t5, T_start=15.0, T_end=0.2, n_stages=12, verbose=True)
        do_try_update(a_out, f"lse_{n_long}")

    # ============ Phase 6: Final coordinate descent polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_now = len(best_seq)
    print(f"\n=== Phase 6: CoordPert final n={n_now} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.30, 400.0)
    if t6 > 10:

    remaining = time_budget - (time.monotonic() - t_global)
    n_now = len(best_seq)
    if remaining > 10:
        t8 = min(remaining * 0.6, 200.0)
        print(f"\n=== Phase 8: Final coord n={n_now} (budget={t8:.0f}s) ===")
        a_work = best_seq * np.sqrt(2.0 * n_now) / best_seq.sum()
        a_out, _ = coordinate_perturbation_descent(a_work, t8, seed=4321, verbose=True)
        do_try_update(a_out, "coord_very_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0] * 100

    return [float(x) for x in best_seq]
