However, we can smooth the objective using log-sum-exp:
LSE_T(f) = (1/T) * log(sum_k exp(T * (f*f)_k))

This converges to ||f*f||_inf as T -> infinity. The gradient is:
grad LSE_T(f)_j = 2 * sum_k w_k * f_{k-j} where w_k = exp(T*(f*f)_k) / sum_l exp(T*(f*f)_l)

This is a weighted average of the subgradients, which is a true gradient (the objective is smooth).

3. LSE-Adam optimization - secondary polish

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full autocorrelation a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    b = fast_autocorr(a)
    return 2.0 * n * b.max()


def evaluate_safe(arr):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-12:
        return v
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_candidates = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        return np.full(n, target_sum / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        result[result.argmax()] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 0:
            result = result / s * target_sum
    return result


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.PDLP


class LPBestResponse:
    """LP best response with constraint generation and warm-starting."""
    
    def __init__(self, n, solver):
        self.n = n
        self.solver = solver
        self.b_warm = None
        self.active_set = None
        self.b_var = None
    
    def solve(self, a, M):
        """Solve max sum(b) s.t. T@b <= M, b>=0 using constraint generation."""
        n = self.n
        T = build_conv_matrix(a)
        
        if self.active_set is None:
            # Initialize active set with top constraints
            b_init = np.ones(n) / n
            T_b = T @ b_init
            self.active_set = np.argsort(T_b)[::-1][:min(400, 2*n-1)]
        
        b_var = cp.Variable(n, nonneg=True)
        if self.b_warm is not None:
            b_var.value = self.b_warm
        
        constraints = [T[self.active_set] @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        prob.solve(solver=self.solver, max_iter=100)
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            # Fallback: use all constraints
            b_var = cp.Variable(n, nonneg=True)
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
            prob.solve(solver=self.solver, max_iter=200)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                return None
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        
        # Constraint generation: check and add violated constraints
        for _ in range(15):
            T_b = T @ b_val

def fast_autocorr(a):

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if len(v) == 0:
        return v
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-12:
        return v
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_candidates = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        return np.full(n, target_sum / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        idx_max = result.argmax()
        result[idx_max] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 0:
            result = result / s * target_sum
    return result


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)

        if s < 1e-15:

        score = internal_score(a_new)

            best_a = a_new.copy()
        a = a_new

    return best_a

        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    t = 0
    base_lr = 0.5

    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(y)

        # Softmax weights on peaks (annealed)
        T_smooth = max(1.0, 200.0 * (1.0 - t / max(1, int(time_limit * 20))))
        Tb = T_smooth * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        # Gradient: grad_j = 2 * sum_k w_k * y_{k-j}
        conv_len = n + (2 * n - 1) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        if conv_len > fft_size:
            fft_size *= 2
        A_rev = rfft(y[::-1].copy(), fft_size)
        W = rfft(w, fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]

        # Nesterov acceleration
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * project_simplex(y - base_lr / (1 + 0.01 * t) * grad)
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        score_a = internal_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = internal_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.01, n_stages=20, lr=0.003):
    """LogSumExp-smoothed optimization with Adam."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = internal_score(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters = max(20, min(100, int(remaining * 80 / (n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_fn(z)
            b = fast_autocorr(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            # Gradient through a
            conv_len = n + (2 * n - 1) - 1
            fft_size = 1 << (conv_len - 1).bit_length()
            if conv_len > fft_size:
                fft_size *= 2
            A_rev = rfft(a[::-1].copy(), fft_size)
            W = rfft(w, fft_size)
            conv = irfft(A_rev * W, fft_size)
            grad_a = 2.0 * conv[n - 1:n - 1 + n]

            # Gradient through z (softmax)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            lr_t = lr / np.sqrt(1 + 0.005 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 15 == 0:
                a_check = softmax_fn(z)
                score = internal_score(a_check)
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        a = softmax_fn(z)
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam-LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving with random restarts."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            src = idx[np.argmax(contrib)]
            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            if not candidates:
                continue
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.001, 0.003, 0.008, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.002, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 300:
                # Reset to best and try different seed region
                a = best_a.copy()
                stall = 0
                for _ in range(50):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > 1e-12:
                        amt = a[i] * rng.uniform(0.001, 0.01)
                        a[i] -= amt
                        a[j] += amt
                a = a / a.sum()
                stall = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Spectral shaping
    Phase 2: Nesterov LP with constraint generation (primary)
    Phase 3: Subgradient optimization
    Phase 4: Adam LSE
    Phase 5: Multi-n exploration
    Phase 6: Peak shaving + final LP
    """
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load starting point from parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = internal_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = internal_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Spectral shaping (quick)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping n={n_base} (rem={remaining:.0f}s) ===")
    t1 = min(remaining * 0.015, 15.0)
    if t1 > 3:
        for alpha_val in [0.4, 0.2, 0.1]:
            a_out = spectral_shaping(best_seq, n_iter=400, alpha=alpha_val)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Spectral alpha={alpha_val}: {best_score:.8f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 1: best={best_score:.8f} (t={elapsed:.0f}s)")

    # Phase 2: Nesterov LP (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Nesterov LP n={n_base} (rem={remaining:.0f}s) ===")
    t2 = min(remaining * 0.55, 1000.0)
    if t2 > 30:
        a_out, sc_out = nesterov_lp_optimize(best_seq, t2, solver)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP_n{n_base}]: {best_score:.8f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 2: best={best_score:.8f} (t={elapsed:.0f}s)")

    # Phase 3: Subgradient optimization
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Subgradient n={n_base} (rem={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 10:
        a_out, _ = subgradient_optimize(best_seq, t3)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [SubGrad_n{n_base}]: {best_score:.8f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 3: best={best_score:.8f} (t={elapsed:.0f}s)")

    # Phase 4: Adam LSE
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Adam LSE n={n_base} (rem={remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 10:
        a_out, _ = adam_lse_optimize(best_seq, t4, T_start=30.0, T_end=0.01, n_stages=15, lr=0.002)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [Adam_LSE]: {best_score:.8f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 4: best={best_score:.8f} (t={elapsed:.0f}s)")

    # Phase 5: Try longer n
    n_long = int(round(n_base * 1.2))
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Try n={n_long} (rem={remaining:.0f}s) ===")
    t5 = min(remaining * 0.18, 250.0)
    if t5 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        # Quick spectral
        a_spec = spectral_shaping(init_long, n_iter=200, alpha=0.25)
        sc = internal_score(a_spec)
        if sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_spec, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [spectral_n{n_long}]: {best_score:.8f}")

        # LP at longer length
        t5b = t5 * 0.7
        init_lp = interpolate_sequence(best_seq, n_long)
        a_out, _ = nesterov_lp_optimize(init_lp, t5b, solver, verbose=False)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP_n{n_long}]: {best_score:.8f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 5: best={best_score:.8f} (t={elapsed:.0f}s)")

    # Phase 6: Peak shaving + final LP
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Peak shaving + final LP n={len(best_seq)} (rem={remaining:.0f}s) ===")

    t6a = min(remaining * 0.25, 80.0)
    if t6a > 5:

        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [peak_shave]: {best_score:.8f}")

    t6b = min(remaining * 0.5, 150.0)
    if t6b > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t6b, solver, verbose=False)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP_final]: {best_score:.8f}")

        if t7 > 5:
