    """Flatten autocorrelation via spectral shaping."""

    c = np.sqrt(2.0 * n)
    a = a * c / a.sum()

    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Flatten: new_mag = mag^alpha (alpha < 1 flattens)
        alpha = 0.8  # Could schedule this
        new_mag = mag ** alpha
        new_A = new_mag * np.exp(1j * phase)
        a_new = irfft(new_A, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new * c / s
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_m
