def lp_adam(a0, tl, lrb=0.05):
    pvals = [2, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 4096]
    n = len(a0)
    a = np.maximum(np.asarray(a0, float), 0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    ba = a.copy()
    bs = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    ns = len(pvals)
    for pi, p in enumerate(pvals):
        if time.monotonic() - t0 > tl:
            break
        rem = tl - (time.monotonic() - t0)
        if rem < 5:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        ta = 0
        lr = lrb * (0.75 ** pi) * min(1.0, 100.0 / n)
        mi = max(20, int(rem * 0.9 / max(1, ns - pi)))
        for it in range(mi):
            if time.monotonic() - t0 > tl - 2:
                break
            gr, _ = lp_grad(a, float(p))
            ta += 1
            m = 0.8 * m + 0.2 * gr
            v = 0.999 * v + 0.001 * gr ** 2
            mh = m / (1 - 0.8 ** ta)
            vh = v / (1 - 0.999 ** ta)
            a = proj(a - lr * mh / (np.sqrt(vh) + 1e-8))
            if (it + 1) % 4 == 0:
                sc = 2.0 * n * fast_autocorr(a).max()
                if sc < bs:
                    bs, ba = sc, a.copy()
        sc = 2.0 * n * fast_autocorr(a).max()
        if sc < bs:
            bs, ba = sc, a.copy()
        if pi % 4 == 0:
            print(f"      Lp p={p} ({time.monotonic()-t0:.0f}s): {bs:.6f}")
    return ba, bs


def mirror(a0, tl, eta0=0.08, ntop=3):
    n = len(a0)
    a = np.maximum(np.asarray(a0, float), 0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    ba = a.copy()
    bs = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = eta0
    stall = 0
    rng = np.random.default_rng(777)
    while time.monotonic() - t0 < tl - 2:
        c = fast_autocorr(a)
        tp = np.argsort(c)[::-1][:ntop]
        imp = np.zeros(n)
        for k in tp:
            js, je = max(0, k - n + 1), min(n - 1, k)
            ix = np.arange(js, je + 1)
            imp[ix] += a[k - ix]
        im = imp.max()
        if im < 1e-15:
            break
        la = np.log(np.maximum(a, 1e-30)) - eta * (imp / im)
        la -= la.max()
        an = np.exp(la)
        sn = an.sum()
        if sn < 1e-15:
            break
        an /= sn
        sc = 2.0 * n * fast_autocorr(an).max()
        if sc < bs:
            a, ba, bs, stall = an, an.copy(), sc, 0
        else:
            stall += 1
            if stall % 50 == 0:
                eta *= 0.6
            if eta < 1e-7:
                eta = eta0
                a = np.maximum(best_a_local + rng.normal(0, 0.002, n), 0)
                a = a / a.sum() if a.sum() > 1e-10 else ba.copy()
                stall = 0
        if stall > 500:
            break
    return ba, bs
