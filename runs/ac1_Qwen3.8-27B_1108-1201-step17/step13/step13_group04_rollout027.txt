def fast_autocorr(a):
    """Compute full autocorrelation (linear convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_grad(a, T):
    """Gradient of LSE surrogate for max autocorrelation."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum
    # Gradient: for each m, grad[m] = sum_k w[k] * (a[k-m] + a[m-k])
    # = (reverse(a) * w)[m + n - 1] + (a * w)[m]
    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)
    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)
    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]
    return r1 + r2

def get_lp_structure(n):
    """Get or create pre-computed LP matrix structure for given n."""
    key = n
    if key in _lp_structure:
        return _lp_structure[key]
    # Build row and column indices
    rows_list = []
    cols_list = []
    for j in range(2 * n - 1):
        k_min = max(0, j - (n - 1))
        k_max = min(n - 1, j)
        for k in range(k_min, k_max + 1):
            rows_list.append(j)
            cols_list.append(k)
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    idx = rows - cols  # index into a: a[j-k]
    _lp_structure[key] = (rows, cols, idx)
    return _lp_structure[key]

def solve_lp_best_response(a, M, n):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0. Returns b or None."""
    rows, cols, idx = get_lp_structure(n)
    data = a[idx]
    T = csr_matrix((data, rows, cols), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            b_val = np.maximum(res.x, 0.0)
            if b_val.sum() > 1e-15:
                return b_val
    except Exception:
        pass
    return None

def lse_nesterov_loop(a0, time_limit, T_start=15.0, T_end=0.3, n_stages=12, verbose=True):

    t0 = time.monotonic()
    
    y = a.copy()
    y_prev = a.copy()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        n_iters_stage = max(30, int(remaining * 2.5 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break
            
            if it == 0:
                y_search = y
            else:
                momentum = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * (it + 2.0))
                y_search = y + momentum * (y - y_prev)
                y_search = np.maximum(y_search, 0.0)
                s_s = y_search.sum()
                if s_s < 1e-15:
                    y_search = y
                else:
                    y_search = y_search * norm / s_s
            
            grad = lse_grad(y_search, T)
            step = 1.0 / (it + 3.0)
            y_new = y_search - step * grad
            y_prev = y.copy()
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            y = y_new
            it_count += 1
            
            if it_count % 50 == 0:
                score = fast_autocorr(y).max()
                if score < best_score:
                    best_score = score
                    best_a = y.copy()
            
            if verbose and it_count % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"  LSE stage={si}/{n_stages} T={T:.3f} it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")
    
    # Final check
    score = fast_autocorr(y).max()
    if score < best_score:
        best_score = score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Primary: LP best-response with adaptive step (fast via scipy HiGHS solver)
- Secondary: LSE Nesterov gradient descent with cooling schedule
- Final: Peak-targeting local search
Key improvements: pre-allocated sparse LP structure for speed, more LP iterations,
adaptive step sizes, multi-length exploration, and better time allocation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

_lp_structure_cache = {}


def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT (length 2n-1)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def get_lp_structure(n):
    """Get or precompute the sparse matrix structure for the LP constraint matrix."""
    if n in _lp_structure_cache:
        return _lp_structure_cache[n]
    rows_list = []
    cols_list = []
    for j in range(2 * n - 1):
        k_min = max(0, j - (n - 1))
        k_max = min(n - 1, j)
        for k in range(k_min, k_max + 1):
            rows_list.append(j)
            cols_list.append(k)
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    idx = rows - cols
    _lp_structure_cache[n] = (rows, cols, idx)
    return _lp_structure_cache[n]


def solve_lp_fast(a, M, n):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0. Returns b or None."""
    rows, cols, idx = get_lp_structure(n)
    data = a[idx]
    T = csr_matrix((data, rows, cols), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
        if res.success and res.x is not None:
            b_val = np.maximum(res.x, 0.0)
            if b_val.sum() > 1e-15:
                return b_val
    except Exception:
        pass
    return None
