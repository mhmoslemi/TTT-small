def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} using the alpha method."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, active_set
    
    # Check if any dropped constraints are violated
    if len(active_set) < size:
        T_full = build_conv_matrix_rows(a, np.arange(size))
        full_conv = T_full @ b_val
        violated = np.where(full_conv > M * (1 + 1e-6))[0]
        if len(violated) > 0:
            # Expand active set and re-solve
            new_active = np.union1d(active_set, violated)
            if len(new_active) > 0.85 * size:
                # Just use all constraints
                T_all = build_conv_matrix_rows(a, np.arange(size))
                b_var2 = cp.Variable(n, nonneg=True)
                constraints2 = [T_all @ b_var2 <= M]
                prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), constraints2)
                try:
                    prob2.solve(solver=solver, verbose=False)
                except Exception:
                    return b_val, active_set
                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                active_set = np.arange(size)
            else:
                T_new = build_conv_matrix_rows(a, new_active)
                b_var2 = cp.Variable(n, nonneg=True)
                constraints2 = [T_new @ b_var2 <= M]
                prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), constraints2)
                try:
                    prob2.solve(solver=solver, verbose=False)
                except Exception:
                    return b_val, active_set
                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                active_set = new_active
    
    s = b_val.sum()
    if s < 1e-15:
        return None, active_set
    return b_val, active_set

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response for max-conv minimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0  # normalize to sum=1
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    active_set = None
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-14:
            break
        
        b, active_set = lp_best_response(y, M_curr, solver, active_set)
        if b is None:
            active_set = None
            # Fallback: use uniform
            b = np.ones(n)
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum  # normalize to sum=1
        n_iters += 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = np.maximum((1.0 - alpha) * y + alpha * g, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = np.maximum(y_new + momentum * (y_new - y), 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a
        
        y = y_new
        a = a_new
        
        # Check improvement
        score_y = fast_autocorr(y).max()
        if score_y < best_M:
            best_M = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_M:
            best_M = score_a
            best_a = a.copy()
        
        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_M:
                    best_M = score_avg
                    best_a = avg_check.copy()
        
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best_M={best_M:.10f}")
    
    return best_a, best_M * 2 * n  # Return actual objective value (with sum=1, obj = 2n*max_conv)

def lse_gradient_optimize(a0, time_limit, verbose=True):
    """LSE gradient descent with FFT-based gradient and projection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    
    size = 2 * n - 1
    fft_size = next_pow2(3 * n - 2)
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Anneal temperature
    initial_T = best_M * 0.5
    final_T = best_M * 0.01
    n_stages = 10
    temps = np.geomspace(max(initial_T, 1e-8), max(final_T, 1e-10), n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        # Decreasing step size
        step0 = min(0.1, 1.0 / (T * np.sqrt(n)))
        
        n_steps = max(20, min(200, int(remaining * 2)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            grad = lse_gradient_fft(a, T, fft_size)
            
            # Gradient step with backtracking
            step = step0 / (1.0 + 0.1 * it)
            cur_obj = fast_autocorr(a).max()
            
            a_new = a - step * grad
            # Project onto simplex
            a_new = project_simplex(a_new)
            
            new_obj = fast_autocorr(a_new).max()
            
            if new_obj < cur_obj:
                a = a_new
                if new_obj < best_M:
                    best_M = new_obj
                    best_a = a.copy()
            else:
                # Halve step (simple backtracking)
                step /= 2
                a_new = project_simplex(a - step * grad)
                new_obj = fast_autocorr(a_new).max()
                if new_obj < cur_obj:
                    a = a_new
                    if new_obj < best_M:
                        best_M = new_obj
                        best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si+1}/{len(temps)} T={T:.6f} ({elapsed:.0f}s): best_M={best_M:.10f}")
    
    return best_a, best_M * 2 * n

def pair_coordinate_descent(a0, time_limit, rng, verbose=True):
    """Systematic pair coordinate descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    
    conv = fast_autocorr(a)
    best_M = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    n_passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        
        # For each i, find the best j to transfer to
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < 1e-10:
                continue
            
            # Compute the gain vector for removing from i
            # delta_conv[k] = -a[k-i] - a[i-k] (for unit delta)
            # Then adding to j: +a[k-j] + a[j-k]
            # Net: a[k-j] + a[j-k] - a[k-i] - a[i-k]
            
            # Precompute the "removal" pattern for i
            rem = np.zeros(size)
            # -a[k-i] for valid k-i
            lo1 = max(0, i)
            hi1 = min(size, i + n)
            if hi1 > lo1:
                rem[lo1:hi1] -= a[max(0, lo1 - i):max(0, hi1 - i)]
            lo2 = max(0, n - 1 - i)
            hi2 = min(size, n + i)
            if hi2 > lo2:
                idx2 = np.arange(lo2, hi2)
                src2 = n - 1 - idx2
                valid = (src2 >= 0) & (src2 < n)
                rem[lo2:hi2][valid] -= a[src2[valid]]
            
            # Find best j by trying a few candidates
            # Strategy: try positions that are "far" from i, and positions where conv is high
            candidates = []
            # Add some randoms
            for _ in range(5):
                j = int(rng.integers(n))
                if j != i:
                    candidates.append(j)
            # Add positions near the peak
            peak_pos = np.argmax(conv)
            for offset in [-2, -1, 1, 2]:
                j = peak_pos + offset
                if 0 <= j < n and j != i:
                    candidates.append(j)
            # Add symmetric partner
            j_sym = n - 1 - i
            if j_sym != i:
                candidates.append(j_sym)
            
            best_d = 0.0
            best_val = conv.max()
            best_j = -1
            
            for j in candidates:
                if j == i or j < 0 or j >= n:
                    continue
                
                # Compute gain for transferring from i to j
                g = np.zeros(size)
                lo3 = max(0, j)
                hi3 = min(size, j + n)
                if hi3 > lo3:
                    g[lo3:hi3] += a[max(0, lo3 - j):max(0, hi3 - j)]
                lo4 = max(0, n - 1 - j)
                hi4 = min(size, n + j)
                if hi4 > lo4:
                    idx4 = np.arange(lo4, hi4)
                    src4 = n - 1 - idx4
                    valid4 = (src4 >= 0) & (src4 < n)
                    g[lo4:hi4][valid4] += a[src4[valid4]]
                
                g += rem  # Add the removal effect
                
                # Find best delta in [0, a[i]]
                ai = a[i]
                n_samples = 15
                deltas = np.linspace(0.0, ai * 0.9, n_samples)
                for d in deltas:
                    if d < 1e-12:
                        continue
                    new_max = (conv + d * g).max()
                    # Second order corrections
                    if 2 * i < size:
                        new_max = max(new_max, conv[2*i] - 2*ai*d + d*d)
                    if 2 * j < size:
                        new_max = max(new_max, conv[2*j] + 2*a[j]*d + d*d)
                    if i + j < size:
                        new_max = max(new_max, conv[i+j] - 2*d*d)
                    if new_max < best_val - 1e-12:
                        best_val = new_max
                        best_d = d
                        best_j = j
            
            if best_d > 0 and best_j >= 0:
                a[i] -= best_d
                a[best_j] += best_d
                s = a.sum()
                if s < 1e-15:
                    a[i] += best_d
                    a[best_j] -= best_d
                    continue
                a = a / s
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_M - 1e-12:
                    best_M = cur_max
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0
                else:
                    a[best_j] -= best_d
                    a[i] += best_d
                    a = a / a.sum()
                    conv = fast_autocorr(a)
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best_M={best_M:.10f}")
    
    return best_a, best_M * 2 * n

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3*np.cos(2.0*x) + 0.15*np.cos(5.0*x) + 0.05*np.cos(9.0*x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0*(x-0.3)**2) + 0.7*np.exp(-6.0*(x-0.7)**2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0*x - 1.0))**3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7*np.sin(np.pi*x)**0.7
    elif choice == 6:
        # Bimodal
        x = np.linspace(0.0, 1.0, n)
