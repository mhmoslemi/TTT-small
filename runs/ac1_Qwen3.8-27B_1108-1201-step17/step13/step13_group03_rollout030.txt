def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out

def _lse_grad(y, T):

    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T

import numpy as np
import time
from scipy.fft import rfft, irfft

def _fast_autocorr(a):
    """Compute full linear convolution of a with a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def _proj_simplex(v):
    """Project onto probability simplex {x: sum=1, x>=0}."""
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out

def _lse_smoothed_max(a, T):
    """Compute LSE-smoothed max of conv(a,a)."""
    ac = _fast_autocorr(a)
    return T * np.log(np.sum(np.exp((ac - ac.max()) / T))) + ac.max()

def _lse_grad(a, T):
    """Gradient of LSE-smoothed max of conv(a,a)."""
    n = len(a)
    ac = _fast_autocorr(a)
    shifted = (ac - ac.max()) / T
    shifted = np.clip(shifted, -100.0, 100.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    
    # Gradient computation via FFT
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[:n]
    return grad

    """FISTA with LSE-smoothed objective and temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    if s < 1e-15:

        a = a / s

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for stage_i, T in enumerate(T_schedule):
        # Lipschitz constant estimate
        L = _estimate_L(a, T)
        t_fista = 1
        z = a.copy()

        max_iters = iters_per_stage
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score
            
            grad = _lse_grad(y, T)
            step = 1.0 / L

            x_new = y - step * grad
            x_new = _proj_simplex(x_new)
            
            # Nesterov update
            if t_fista == 1:
                y_new = x_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = x_new + mom * (x_new - z)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()

                else:
                    y_new = x_new.copy()

            a = x_new
            y = y_new

                if sc < best_score:

                    best_a = a.copy()

            print(f"  FISTA stage={stage_i} T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

def _estimate_L(a, T):
    """Estimate Lipschitz constant for the LSE gradient."""
    n = len(a)
    # The gradient is approximately 2 * conv(w, a), and ||w||_1 = 1
    # So ||grad|| <= 2 * ||a||_inf * sum(w) = 2 * ||a||_inf
    # But we need the Lipschitz constant of the gradient, which is bounded by 2 * ||w||_2
    # A safe overestimate: L = 4n (conservative)
    # Better: L = 2 * max(a) * (2n) / T ... too complex
    # Just use a reasonable heuristic
    return 4.0 * n / max(T, 0.01)

For the LSE-smoothed objective f_T(a) = T * log(sum exp(ac_k/T)), the gradient is:
grad = sum_k w_k * d(ac_k)/d(a)

where w = softmax(ac/T).

The Hessian of f_T is:

A simpler bound: ||grad(a) - grad(b)|| <= L * ||a - b||
Since grad is a sum of w_k * (linear operator applied to a), and the linear operator is the convolution with a (which has norm bounded by ||a||_1 = 1), we have roughly L ~ 2/T for the LSE part.

The Lipschitz constant of the gradient is bounded by:
L <= (1/T) * max_k ||H_k|| + (1/T) * sum_k (w_k)^2 * ||G_k||^2

where H_k is the Hessian of ac_k and G_k = d(ac_k)/d(a).
