            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _lp_solver = cp.SCS
    print("  LP solver (fallback): SCS")
    return _lp_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def eval_seq(a):
    try:
        a_list = [float(max(0.0, x)) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def make_gaussian(n, width_frac=0.3):
    x = np.linspace(0, 1, n)
    center = 0.5
    width = width_frac
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_raised_cosine(n, power=1.0):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * (x - 0.5)) ** (2 * power)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_super_gaussian(n, p=3.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5) * 4) ** p)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_bimodal(n, sep=0.3, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5 + sep) / width) ** 2) + np.exp(-0.5 * ((x - 0.5 - sep) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def compute_subgrad_full(a, ac, beta=2.0):
    """Full subgradient with exponential weighting over all lags."""
    n = len(a)
    ac_pos = np.maximum(ac, 0)
    w = ac_pos ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.ones(n) / n
    w = w / w_sum
    grad = np.zeros(n)
    for k in range(len(ac)):
        if w[k] < 1e-15:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j = k - idx
        grad[idx] += 2.0 * w[k] * a[j]
    return grad

def compute_subgrad_topk(a, ac, top_k=10, beta=3.0):
    """Top-k subgradient with exponential weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 0)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-15:
            continue
        i_start = max(0, int(k) - n + 1)
        i_end = min(n, int(k) + 1)
        idx = np.arange(i_start, i_end)
        j = int(k) - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def mirror_descent(a0, time_limit, top_k=8, beta=2.5, step_decay=0.6, verbose=True):
    """Mirror descent with polynomial step decay and full subgradient mixture."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a * c / a.sum()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    avg_a = a.copy()
    n_avg = 1
    gamma = 1.5

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        step = gamma / ((t + 1) ** step_decay)
        ac = fast_autocorr(a)
        M = ac.max()
        g = compute_subgrad_topk(a, ac, top_k=top_k, beta=beta)
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * c / s_new
        a = a_new
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * c / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m

def peak_shaving(a0, time_limit, n_peaks=5, n_candidates=20, verbose=True):
    """Direct peak shaving: move mass from peak contributors to safe positions."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), fast_autocorr(a0).max()
    a = a * c / s
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M = ac.max()
        top_peaks = np.argsort(ac)[::-1][:n_peaks]
        # Build set of "dangerous" indices (those that appear in top peak contributions)
        danger = np.zeros(n, dtype=np.float64)
        for k in top_peaks:
            i_start = max(0, int(k) - n + 1)
            i_end = min(n, int(k) + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[int(k) - idx]
            danger[idx] += contrib
        # Safe positions: low danger, relatively low value
        safe_score = danger + 0.1 * a
        safe_candidates = np.argsort(safe_score)[:n_candidates]
        improved = False
        for k in top_peaks:
            i_start = max(0, int(k) - n + 1)
            i_end = min(n, int(k) + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[int(k) - idx]
            # Source: highest contributor
            src_local = np.argmax(contrib)
            src = idx[src_local]
            if a[src] < 1e-12:
                continue
            # Try moving to safe candidates
            for dst in safe_candidates:
                if dst == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.03]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * c / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > c * 1e-6:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a * c / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        if verbose and (stall % 100 == 0 or improved) and (time.monotonic() - t0) % 5 < 0.1:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    PeakShave stall={stall} ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    # Constraint: (a*b)_k <= M for all k
    # (a*b)_k = sum_j a_j * b_{k-j} (valid indices)
    # Use Toeplitz-like structure
    # For efficiency, only add constraints for lags that are currently near M
    ac_ab = fast_autocorr(np.concatenate([a, a]))  # Not quite right, let's do it properly
    # Actually: (a*b)_k = sum_{j: 0<=j<n, 0<=k-j<n} a_j * b_{k-j}
    # We can write this as: for each k, sum_j a_{k-j} * b_j (where 0<=k-j<n)
    # This is (T_k) @ b where T_k[j] = a_{k-j} if 0<=k-j<n else 0
    # For efficiency, only constrain the top lags
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    # Only add top constraint lags for speed
    # But for correctness, add all (it's 2n-1 constraints, n variables - manageable for n<=3000)
    if n <= 2000:
        constraints = [T @ b_var <= M]
    else:
        # For larger n, subsample constraints
        step_c = max(1, n // 500)
        constraints = [T[::step_c] @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), fast_autocorr(a0).max()
    a = a * c / s
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * c / b_sum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c / s_y
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * c / s_x
        y = y_new
        score_y = fast_autocorr(y).max()
        if score_y < best_m:
            best_m = score_y
            best_a = y.copy()
        score_x = fast_autocorr(x_new).max()
        if score_x < best_m:
            best_m = score_x
            best_a = x_new.copy()
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 0.999, old_n)
    new_x = np.linspace(0.0, 0.999, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def random_perturb_search(a0, time_limit, n_perturb=100, verbose=True):
    """Random perturbation local search."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), fast_autocorr(a0).max()
    a = a * c / s
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    for it in range(n_perturb):
        if time.monotonic() - t0 > time_limit:
            break
        # Pick 2-5 random positions to perturb
        n_move = rng.integers(2, 6)
        srcs = rng.choice(n, size=n_move, replace=False)
        dsts = rng.choice(n, size=n_move, replace=False)
        fracs = rng.uniform(0.001, 0.02, size=n_move)
        a_trial = a.copy()
        valid = True
        for s_i, d_i, f in zip(srcs, dsts, fracs):
            amt = a_trial[s_i] * f
            if amt < 0 or a_trial[s_i] - amt < -1e-15:
                valid = False
                break
            a_trial[s_i] -= amt
            a_trial[d_i] += amt
        if not valid:
            continue
        a_trial = np.maximum(a_trial, 0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial * c / s_t
        M_trial = fast_autocorr(a_trial).max()
        if M_trial < best_m - 1e-15:
            a = a_trial
            best_m = M_trial
            best_a = a.copy()
            if verbose and it % 10 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m / (c * c)
                print(f"    RandPerturb it={it} ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """Main search: multi-init, multi-method, multi-length optimization."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = get_lp_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = eval_seq(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    n_parent = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_seq(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
            n_parent = len(parent)
    except Exception:
        print("Warning: could not load parent, using default")

    if best_seq is None:
        best_seq = make_gaussian(n_parent, 0.3)
        best_score = eval_seq(best_seq)
        print(f"Default init: n={n_parent}, score={best_score:.6f}")

    # ============ Phase 1: Multiple initializations at parent length ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Multi-init MD at n={n_parent} (budget {remaining:.0f}s) ===")
    inits = [
        ("parent", best_seq.copy()),
        ("gauss_0.2", make_gaussian(n_parent, 0.2)),
        ("gauss_0.25", make_gaussian(n_parent, 0.25)),
        ("gauss_0.3", make_gaussian(n_parent, 0.3)),
        ("gauss_0.4", make_gaussian(n_parent, 0.4)),

        ("bimodal", make_bimodal(n_parent, 0.25, 0.12)),
    ]

    # Evaluate all inits first
    init_scores = []
    for name, init in inits:
        sc = eval_seq(init)
        init_scores.append((sc, name, init))
        print(f"    Init {name}: score={sc:.6f}")
    init_scores.sort()

    # Run MD on top 4 initializations
    t_per_init = min(remaining * 0.12 / 4, 120.0)
    for sc, name, init in init_scores[:4]:
        if time.monotonic() - t_global > time_budget - 300:
            break
        t_avail = min(t_per_init, time_budget - (time.monotonic() - t_global))
        if t_avail < 5:
            break
        c_init = np.sqrt(2.0 * n_parent)
        a_init = init * c_init / init.sum()
        a_out, _ = mirror_descent(a_init, t_avail, top_k=8, beta=2.5, step_decay=0.55, verbose=False)
        try_update(a_out, f"md_{name}")

    # ============ Phase 2: Peak shaving at best ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Peak shaving n={n_best} (budget {remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 10:
        c_best = np.sqrt(2.0 * n_best)
        a_in = best_seq * c_best / best_seq.sum()
        a_out, _ = peak_shaving(a_in, t2, n_peaks=5, n_candidates=30, verbose=True)
        try_update(a_out, f"peakshave_n{n_best}")

    # ============ Phase 3: LP Nesterov at best ============

        a_in = best_seq * c_best / best_seq.sum()

        try_update(a_out, f"md_n{n2}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Peak shaving n={n_cur} ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 10:
        c_cur = np.sqrt(2.0 * n_cur)
        a_in = best_seq * c_cur / best_seq.sum()
        a_out, _ = peak_shaving(a_in, t6, n_peaks=6, n_candidates=25, verbose=True)
        try_update(a_out, f"peakshave3_n{n_cur}")

    # ============ Phase 7: Scale up further ============
    n3 = n_best + 1600
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: MD at n={n3} (budget {remaining:.0f}s) ===")

        a_out, _ = mirror_descent(a_in, t7, top_k=5, beta=1.8, step_decay=0.45, verbose=True)
        try_update(a_out, f"md_n{n3}")

    # ============ Phase 8: LP at best ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: LP Nesterov round 2 n={n_best} ===")
    t8 = min(remaining * 0.12, 150.0)
    if t8 > 30:
        c_best = np.sqrt(2.0 * n_best)
        a_in = best_seq * c_best / best_seq.sum()
        a_out, _ = nesterov_lp(a_in, t8, solver, verbose=True)
        try_update(a_out, f"lp2_n{n_best}")

    # ============ Phase 9: Final peak shaving ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final peak shaving n={n_best} ===")
    t9 = min(remaining * 0.15, 150.0)
    if t9 > 10:
        c_best = np.sqrt(2.0 * n_best)
        a_in = best_seq * c_best / best_seq.sum()
        a_out, _ = peak_shaving(a_in, t9, n_peaks=10, n_candidates=40, verbose=True)
        try_update(a_out, f"peakshave_final_n{n_best}")

    # ============ Phase 10: Random perturbation search ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 10: Random perturbation n={n_best} ===")
    t10 = min(remaining * 0.15, 100.0)
    if t10 > 5:
        c_best = np.sqrt(2.0 * n_best)
        a_in = best_seq * c_best / best_seq.sum()
        a_out, _ = random_perturb_search(a_in, t10, n_perturb=200, verbose=True)
        try_update(a_out, f"randperturb_n{n_best}")

    # ============ Phase 11: Another MD with different params ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 11: Final MD n={n_best} ===")
    t11 = min(remaining * 0.20, 150.0)
    if t11 > 10:
        c_best = np.sqrt(2.0 * n_best)
        a_in = best_seq * c_best / best_seq.sum()
        a_out, _ = mirror_descent(a_in, t11, top_k=12, beta=1.5, step_decay=0.4, verbose=True)
        try_update(a_out, f"md_final_n{n_best}")

    # ============ Phase 12: Final peak shave + random ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 12: Final polish n={n_best} ===")
    t12a = min(remaining * 0.4, 80.0)
    t12b = min(remaining * 0.4, 60.0)
    if t12a > 5:
        c_best = np.sqrt(2.0 * n_best)
        a_in = best_seq * c_best / best_seq.sum()
        a_out, _ = peak_shaving(a_in, t12a, n_peaks=10, n_candidates=50, verbose=True)
        try_update(a_out, f"peakshave_p12a_n{n_best}")
    if t12b > 5 and time.monotonic() - t_global < time_budget - 30:
        c_best = np.sqrt(2.0 * len(best_seq))
        a_in = best_seq * c_best / best_seq.sum()
        a_out, _ = random_perturb_search(a_in, t12b, n_perturb=300, verbose=True)
        try_update(a_out, f"randperturb_p12b_n{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = eval_seq(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
