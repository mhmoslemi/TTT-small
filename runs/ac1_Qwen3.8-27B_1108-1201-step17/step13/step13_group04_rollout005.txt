    grad = np.zeros(n)

    for k in top_idx:
        w = (b[k] / b_max) ** alpha
        if w < 1e-15:
            continue
        m_start = max(0, k - n + 1)
        m_end = min(k, n - 1)
        if m_end >= m_start:
            indices = np.arange(m_start, m_end + 1)
            grad[indices] += w * a[k - indices]

    return grad, b_max


def peak_gradient_descent(a0, time_limit, K=15, alpha=2.0, verbose=True):
    """Top-K peak gradient descent with adaptive step size and momentum."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    # Work with sum=1, objective is 2*n*max_b
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.9
    lr = 1.0 / max(n, 100)
    it_count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        grad, b_max = topk_gradient(a, K=K, alpha=alpha)

        # Adaptive step size with backtracking
        for attempt in range(20):
            step = lr / (1.0 + attempt * 0.5)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / sn
            obj_new = 2.0 * n * fast_autocorr(a_new).max()
            if obj_new < best_obj:
                momentum = beta * momentum + (1 - beta) * grad
                a = a_new
                best_obj = obj_new
                best_a = a.copy()
                stall = 0
                break
        else:
            # Backtracking failed, use momentum step with smaller lr
            a_new = a - lr * 0.1 * momentum
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn > 1e-15:
                a_new = a_new / sn
                obj_new = 2.0 * n * fast_autocorr(a_new).max()
                if obj_new < best_obj:
                    a = a_new
                    best_obj = obj_new
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

        it_count += 1

        # Anneal: slowly increase K and decrease alpha to focus on the single peak
        progress = min(1.0, it_count / max(1, int(time_limit * 2)))
        K_eff = max(3, int(K * (1.0 - 0.7 * progress)))
        alpha_eff = alpha + 5.0 * progress

        if verbose and it_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PGD it={it_count} ({elapsed:.0f}s): obj={best_obj:.8f}, K={K_eff}")

        if stall > 500:
            break

    return best_a, best_obj


def simulated_annealing_peak(a0, time_limit, verbose=True, seed=42):
    """Simulated annealing with peak convolution as energy."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    current_obj = best_obj
    t0 = time.monotonic()

    T_start = best_obj * 0.05
    T_end = best_obj * 0.001
    total_iters = max(500, int(time_limit * 50))

    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = it / max(1, total_iters)
        T = T_start * (T_end / T_start) ** progress

        # Propose move: transfer mass from one position to another
        # Weight sampling towards positions contributing to peaks
        b = fast_autocorr(a)
        peak_k = np.argmax(b)

        # Source: biased towards peak contributors
        m_start = max(0, peak_k - n + 1)
        m_end = min(peak_k, n - 1)
        contrib = a[m_start:m_end + 1] * a[peak_k - np.arange(m_start, m_end + 1)]
        if contrib.sum() > 1e-30:
            src_probs = contrib / contrib.sum()
            src_local = rng.choice(len(src_probs), p=src_probs)
            src = m_start + src_local
        else:
            src = rng.integers(n)

        # Destination: avoid peak contributors
        dst_mask = np.ones(n, dtype=bool)
        dst_mask[m_start:m_end + 1] = False
        if dst_mask.sum() > n * 0.3:
            candidates = np.where(dst_mask)[0]
        else:
            candidates = np.where(np.arange(n) != src)[0]
        dst = candidates[rng.integers(len(candidates))]

        if src == dst or a[src] < 1e-15:
            continue

        # Fraction to move
        frac = rng.uniform(0.01, 0.15)
        amt = a[src] * frac

        a_trial = a.copy()
        a_trial[src] -= amt
        a_trial[dst] += amt
        a_trial = np.maximum(a_trial, 0.0)
        st = a_trial.sum()
        if st < 1e-15:
            continue
        a_trial = a_trial / st

        obj_trial = 2.0 * n * fast_autocorr(a_trial).max()
        delta = obj_trial - current_obj

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-30)):
            a = a_trial
            current_obj = obj_trial
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_a = a.copy()

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA it={it} ({elapsed:.0f}s): T={T:.4f}, best={best_obj:.8f}")

    return best_a, best_obj


def freq_perturbation(a0, time_limit, verbose=True, seed=123):
    """Explore solution space via frequency-domain phase perturbation."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Keep current FFT magnitude
    fft_size = 1 << (2 * n - 1).bit_length()
    A_mag = np.abs(rfft(a, fft_size))

    it_count = 0
    while time.monotonic() - t0 < time_limit:
        # Generate random phase perturbation
        phase = rfft(a, fft_size)
        phase = phase / np.maximum(np.abs(phase), 1e-30)

        # Perturb phase smoothly (low-frequency perturbation)
        pert_interp = np.interp(
            np.arange(fft_size // 2 + 1),
            [0, (fft_size // 2 + 1) // 4, (fft_size // 2 + 1) // 2, fft_size // 2],
            [0, rng.uniform(-0.5, 0.5), rng.uniform(-0.3, 0.3), 0]
        )
        new_phase = phase * np.exp(1j * pert_interp)
        a_trial = irfft(new_phase * A_mag, fft_size)[:n]
        a_trial = np.maximum(a_trial, 0.0)
        st = a_trial.sum()
        if st < 1e-15 or st < 0.9 * a.sum():
            continue
        a_trial = a_trial / st

        obj_trial = 2.0 * n * fast_autocorr(a_trial).max()
        if obj_trial < best_obj:
            best_obj = obj_trial
            best_a = a_trial.copy()
            a = a_trial.copy()
            A_mag = np.abs(rfft(a, fft_size))

        it_count += 1
        if verbose and it_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FreqPert it={it_count} ({elapsed:.0f}s): best={best_obj:.8f}")

    return best_a, best_obj


def batch_local_search(a0, time_limit, seed=777, verbose=True):
    """
    Batch local search: simultaneously move mass from multiple peak-contributing
    positions to multiple non-contributing positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_ks = np.argsort(b)[::-1][:8]

        # Identify all positions contributing to top peaks
        peak_contrib = np.zeros(n)
        for k in top_ks:
            m_start = max(0, k - n + 1)
            m_end = min(k, n - 1)
            indices = np.arange(m_start, m_end + 1)
            peak_contrib[indices] = np.maximum(peak_contrib[indices], a[indices] * a[k - indices])

        # Sources: positions with high peak contribution
        src_threshold = peak_contrib.max() * 0.5
        sources = np.where(peak_contrib > src_threshold)[0]
        sources = sources[a[sources] > 1e-10]

        # Destinations: positions with low peak contribution
        dst_threshold = peak_contrib.max() * 0.1
        dests_all = np.where(peak_contrib < dst_threshold)[0]
        if len(dests_all) == 0:
            dests_all = np.where(peak_contrib == peak_contrib.min())[0]

        if len(sources) == 0 or len(dests_all) == 0:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.01, 0.1)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial /= st
                    obj = 2.0 * n * fast_autocorr(a_trial).max()
                    if obj < best_obj:
                        a = a_trial
                        best_obj = obj
                        best_a = a.copy()
                        stall = 0
                    else:
                        stall += 1
            else:
                stall += 1
            it_count += 1
            if stall > 300:
                break
            continue

        # Try batch moves: pick 2-4 sources and 2-4 destinations
        n_src = min(len(sources), rng.integers(2, 5))
        n_dst = min(len(dests_all), rng.integers(2, 5))

        src_pick = sources[rng.choice(len(sources), n_src, replace=False)]
        dst_pick = dests_all[rng.choice(len(dests_all), n_dst, replace=False)]

        # Ensure no overlap
        overlap = np.intersect1d(src_pick, dst_pick)
        if len(overlap) > 0:
            dst_pick = np.setdiff1d(dst_pick, overlap)
            if len(dst_pick) == 0:
                stall += 1
                it_count += 1
                continue

        total_src_mass = a[src_pick].sum()
        if total_src_mass < 1e-15:
            stall += 1
            it_count += 1
            continue

        # Try different fractions
        improved = False
        for frac in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            move_amt = total_src_mass * frac
            portion_per_src = move_amt / len(src_pick)
            portion_per_dst = move_amt / len(dst_pick)

            a_trial = a.copy()
            valid = True
            for s in src_pick:
                a_trial[s] -= portion_per_src
                if a_trial[s] < -1e-12:
                    valid = False
                    break
            if not valid:
                continue
            for d in dst_pick:
                a_trial[d] += portion_per_dst

            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st

            obj = 2.0 * n * fast_autocorr(a_trial).max()
            if obj < best_obj - 1e-12:
                a = a_trial
                best_obj = obj
                best_a = a.copy()
                improved = True
                stall = 0
                it_count += 1
                break

        if not improved:
            stall += 1
            it_count += 1
            if stall > 400:
                break

        if verbose and it_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BatchLS it={it_count} ({elapsed:.0f}s): best={best_obj:.8f} stall={stall}")

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. (a*b)_j <= M for all j, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    T = lil_matrix((2 * n - 1, n))
    for j in range(2 * n - 1):
        for k in range(n):
            i = j - k
            if 0 <= i < n:
                T[j, k] = a[i]
    T = T.tocsr()

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    n_iters = 0

    for t in range(1, 5000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        if t == 1:
            a_new = b.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y_prev = y
        y = y_new

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_obj:
            best_obj = score_y
            best_a = y.copy()

        score_a = 2.0 * n * fast_autocorr(a_new).max()
        if score_a < best_obj:
            best_obj = score_a
            best_a = a_new.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesh iter={n_iters} ({elapsed:.0f}s): best={best_obj:.8f}")

    return best_a, best_obj


def evaluate_safe(a):
    """Safely evaluate a sequence via the provided evaluate_sequence function."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def structured_init(n, seed):
    """Generate a diverse structured initialization."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.1 * np.cos(7.0 * x) + 0.05 * np.cos(13.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        # Random smooth
        a = np.ones(n) * 0.5
        for _ in range(rng.integers(3, 8)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.3, 0.5)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif choice == 6:
        # Frequency domain: random smooth magnitude
        fft_size = 1 << (n - 1).bit_length()
        mag = np.ones(fft_size // 2 + 1) * 0.5
        for _ in range(rng.integers(2, 5)):
            f0 = rng.uniform(0, fft_size // 2)
            bw = rng.uniform(5, fft_size // 4)
            mag += rng.uniform(0.1, 0.4) * np.exp(-0.5 * (np.arange(fft_size // 2 + 1) - f0) ** 2 / bw ** 2)
        phase = rng.uniform(-np.pi, np.pi, size=fft_size // 2 + 1)
        phase[0] = 0
        a = irfft(mag * np.exp(1j * phase), fft_size)[:n]
    else:
        # Power-law shape
        x = np.linspace(0.01, 1.0, n)
        p = rng.uniform(0.3, 2.0)
        a = x ** p + (1 - x) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses multiple optimization strategies:
    - Top-K peak gradient descent (new)
    - Simulated annealing (new)
    - Frequency perturbation (new)
    - Batch local search (improved)
    - LP Nesterov (kept from previous)
    Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver={solver}")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # ============ Phase 1: Top-K Peak Gradient Descent ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: TopK-Gradient n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.22, 350.0)
    if t1 > 30:
        a_work = best_seq.copy()
        a_out, _ = peak_gradient_descent(a_work, t1, K=20, alpha=2.0, verbose=True)
        do_try_update(a_out, f"topk_grad_n{n_base}")

    # ============ Phase 2: Simulated Annealing ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: SimulatedAnnealing n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 20:
        a_out, _ = simulated_annealing_peak(best_seq, t2, verbose=True, seed=42)
        do_try_update(a_out, f"sa_n{len(best_seq)}")

    # ============ Phase 3: Frequency Perturbation ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: FreqPerturbation n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 160.0)
    if t3 > 15:
        a_out, _ = freq_perturbation(best_seq, t3, verbose=True, seed=123)
        do_try_update(a_out, f"freq_n{len(best_seq)}")

    # ============ Phase 4: Batch Local Search ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: BatchLocalSearch n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_out, _ = batch_local_search(best_seq, t4, seed=777, verbose=True)
        do_try_update(a_out, f"batch_n{len(best_seq)}")

    # ============ Phase 5: LP Nesterov ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP-Nesterov n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 320.0)
    if t5 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver, verbose=True)
        do_try_update(a_out, f"lp_n{len(best_seq)}")

    # ============ Phase 6: Second round Top-K Gradient (finer) ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: TopK-Gradient (fine) n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 20:
        a_work = best_seq.copy()
        a_out, _ = peak_gradient_descent(a_work, t6, K=8, alpha=4.0, verbose=True)
        do_try_update(a_out, f"topk_fine_n{len(best_seq)}")

    # ============ Phase 7: Batch Local Search (second pass) ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: BatchLocalSearch (pass2) n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.12, 180.0)
    if t7 > 15:
        a_out, _ = batch_local_search(best_seq, t7, seed=999, verbose=True)
        do_try_update(a_out, f"batch2_n{len(best_seq)}")

    # ============ Phase 8: Multi-length exploration ============
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = n_base + 300
    print(f"\n=== Phase 8: Multi-length n={n2} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.15, 200.0)
    if t8 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_out, _ = peak_gradient_descent(init_long, t8 * 0.5, K=15, alpha=2.0, verbose=True)
        do_try_update(a_out, f"topk_long_n{n2}")
        if time.monotonic() - t_global < time_budget - 30:
            a_out2, _ = batch_local_search(a_out, (time_budget - (time.monotonic() - t_global)) * 0.5, seed=555, verbose=True)
            do_try_update(a_out2, f"batch_long_n{n2}")

    # ============ Phase 9: Final polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    t9 = min(remaining * 0.5, 150.0)
    if t9 > 10:
        a_out, _ = batch_local_search(best_seq, t9, seed=4242, verbose=True)
        do_try_update(a_out, "final_polish")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
