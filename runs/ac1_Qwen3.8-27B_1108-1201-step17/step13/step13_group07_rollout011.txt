def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_autocorr(a):
    n = len(a)

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_corr(a, b):
    """Compute a*b cross-correlation using FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

    return irfft(A * B, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    idx = np.arange(1, n + 1)
    cond = v_sorted * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        mask = result > 0
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result = np.maximum(result, 0)
            result = result / result.sum() * target_sum
    return result
