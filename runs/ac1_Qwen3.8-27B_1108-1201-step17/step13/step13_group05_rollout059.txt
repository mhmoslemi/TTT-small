    Mirror descent with exact subgradient on the simplex {a >= 0, sum(a) = c}.
    Uses Hedge-style averaging for better convergence.
    Returns (best_a, best_M) where best_M = max autocorrelation.
    """
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a * c / s

    best_M = fast_autocorr(a).max()

    avg_a = np.zeros(n)
    count = 0

    eta0 = 1.0

        count += 1

            if s_avg > 1e-15:

def fast_autocorr(a):
    """Compute linear autocorrelation a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_subgrad(a, ac, top_k=0):
    """
    Compute subgradient of max(ac) w.r.t. a.
    If top_k > 0, use only top-k lags (weighted by value).
    If top_k == 0, use exact subgradient (uniform over all argmax).
    Returns gradient vector of shape (n,).
    """
    n = len(a)
    M = ac.max()
    if M <= 0:
        return np.zeros(n)
    
    if top_k > 0:
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[top_idx].copy()
        top_vals = np.maximum(top_vals, 1e-30)
        w = top_vals
        w = w / w.sum()
        g = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, top_idx):
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                mir = k - idx
                g[idx] += 2.0 * wi * a[mir]
        return g
    else:
        tol = max(1e-12, M * 1e-10)
        K = np.where(ac >= M - tol)[0]
        if len(K) == 0:
            K = np.array([int(np.argmax(ac))])
        g = np.zeros(n, dtype=np.float64)
        for k in K:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                mir = k - idx
                g[idx] += 2.0 * a[mir]
        g /= float(len(K))
        return g

def mirror_descent(a0, time_limit, top_k=0, eta_scale=1.0, verbose=True):
    """
    Mirror descent with subgradient on simplex {a >= 0, sum(a) = sqrt(2n)}.
    Uses Hedge-style running average.
    Returns (best_a, best_M).
    """
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a * c / s
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    
    avg_a = np.zeros(n, dtype=np.float64)
    count = 0
    t0 = time.monotonic()
    
    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        g = compute_subgrad(a, ac, top_k=top_k)
        g_norm = np.linalg.norm(g, ord=np.inf)
        if g_norm < 1e-30:
            break
        
        eta = eta_scale / (g_norm * (np.sqrt(t) + 1.0))
        # Cap step size to prevent overshooting
        eta = min(eta, 2.0 / g_norm)
        
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - eta * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * c / s_new
        a = a_new
        
        M_new = fast_autocorr(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
        
        avg_a += a
        count += 1
        if count % 100 == 0:
            avg_check = avg_a / count
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * c / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_M:
                    best_M = M_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M / (c * c)
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_M

def coordinate_polish(a0, time_limit, verbose=True):
    """
    Targeted coordinate descent: move mass from high contributors to low-peak positions.
    Returns (best_a, best_M).
    """
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a * c / s
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        M_tol = max(1e-12, M * 1e-8)
        peak_lags = np.where(ac >= M - M_tol)[0]
        
        if len(peak_lags) == 0:
            peak_lags = np.array([int(np.argmax(ac))])
        
        improved = False
        
        for k in peak_lags[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            mir = k - idx
            contrib = a[idx] * a[mir]
            
            if contrib.max() < 1e-20:
                continue
            
            src_local = np.argmax(contrib)
            src = idx[src_local]
            
            # Find target: position with minimal contribution to all peak lags
            target_score = np.zeros(n, dtype=np.float64)
            for k2 in peak_lags[:10]:
                k2 = int(k2)
                js = max(0, k2 - n + 1)
                je = min(n, k2 + 1)
                if js < je:
                    idx2 = np.arange(js, je)
                    mir2 = k2 - idx2
                    target_score[idx2] += a[mir2]
            
            # Prefer positions with low target_score and not equal to src
            target_score[src] = np.inf
            cands = np.where(target_score < np.inf)[0]
            if len(cands) == 0:
                cands = np.arange(n)
                cands = cands[cands != src]
            
            # Pick from lowest-scoring candidates
            sorted_cands = cands[np.argsort(target_score[cands])]
            n_try = max(3, len(sorted_cands) // 10)
            n_try = min(n_try, len(sorted_cands))
            
            for dst in sorted_cands[:n_try]:
                if time.monotonic() - t0 > time_limit:
                    break
                for frac in [0.01, 0.03, 0.08, 0.15]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * c / s_t
                    ac_t = fast_autocorr(a_trial)
                    M_t = ac_t.max()
                    if M_t < best_M - 1e-14:
                        a = a_trial
                        best_M = M_t
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            # Random perturb
            if stall > 50:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > c * 1e-6:
                    frac = rng.uniform(0.02, 0.1)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a * c / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_M:
                            best_M = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = a * c / a.sum()
        
        if verbose and (stall % 50 == 0 or improved):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M / (c * c)
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_M

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving normalized shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def generate_init(n, init_type, rng):
    """Generate an initial sequence of length n."""
    x = np.linspace(0, 1, n)
    if init_type == 'uniform':
        a = np.ones(n)
    elif init_type == 'sine':
        a = np.sin(np.pi * x)
    elif init_type == 'gaussian':
        sigma = 0.25
        a = np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2))
    elif init_type == 'cosine_bell':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif init_type == 'raised_cos':
        a = np.sin(np.pi * x) ** 2
    elif init_type == 'flat_top':
        w = 0.3
        a = np.ones(n)
        a[:int(w * n)] = np.linspace(0.3, 1.0, int(w * n))
        a[-int(w * n):] = np.linspace(1.0, 0.3, int(w * n))
    elif init_type == 'bimodal':
        a = (np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02))
    elif init_type == 'random':
        a = rng.random(n)
    elif init_type == 'random_smooth':
        a = rng.random(n)
        # Smooth with a small kernel
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a

            c_b = np.sqrt(2.0 * len(best_seq))
            a_norm2 = best_seq * c_b / best_seq.sum()
            a_out2, _ = coordinate_polish(a_norm2, t4b, verbose=True)
            try_update(a_out2, f"coord_long_n{n_long}")
    
    # ===== Phase 5: Try even longer =====
    remaining = time_budget - (time.monotonic() - t_global)
    n_long2 = int(len(best_seq) * 2.0)
    n_long2 = min(n_long2, 5000)
    if n_long2 > n_long and remaining > 150:
        print(f"\n=== Phase 5: Even longer (n={n_long2}) ===")
        t5 = min(remaining * 0.3, 300.0)
        if t5 > 30:
            a_long2 = interpolate_seq(best_seq, n_long2)
            c_l2 = np.sqrt(2.0 * n_long2)
            a_norm = a_long2 * c_l2 / a_long2.sum()
            a_out, _ = mirror_descent(a_norm, t5 * 0.6, top_k=3, eta_scale=1.0, verbose=True)
            try_update(a_out, f"md_long2_n{n_long2}")
            
            remaining5 = time_budget - (time.monotonic() - t_global)
            if remaining5 > 30:
                c_b = np.sqrt(2.0 * len(best_seq))
                a_norm2 = best_seq * c_b / best_seq.sum()
                a_out2, _ = coordinate_polish(a_norm2, min(remaining5 * 0.3, 100.0), verbose=True)
                try_update(a_out2, f"coord_long2_n{n_long2}")
    
    # ===== Phase 6: Final intensive polish on best =====
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_final}) ===")
    if remaining > 30:
        c_final = np.sqrt(2.0 * n_final)
        a_norm = best_seq * c_final / best_seq.sum()
        
        t6a = min(remaining * 0.3, 100.0)
        if t6a > 10:
            a_out, _ = mirror_descent(a_norm, t6a, top_k=3, eta_scale=0.5, verbose=False)
            try_update(a_out, "md_final")
        
        remaining6 = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining6 * 0.5, 150.0)
        if t6b > 10:
            c_b = np.sqrt(2.0 * len(best_seq))
            a_norm2 = best_seq * c_b / best_seq.sum()
            a_out2, _ = coordinate_polish(a_norm2, t6b, verbose=True)
            try_update(a_out2, "coord_final")
    
    # ===== Phase 7: Last resort - short MD with different hyperparams =====
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_final = len(best_seq)
        c_final = np.sqrt(2.0 * n_final)
        a_norm = best_seq * c_final / best_seq.sum()
        a_out, _ = mirror_descent(a_norm, remaining * 0.7, top_k=8, eta_scale=2.0, verbose=False)
        try_update(a_out, "md_last")
    
    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
