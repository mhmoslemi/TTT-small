import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def compute_lse_grad(a, w, fft_size):
    """Compute gradient of LSE(conv(a,a), T) w.r.t. a using FFT.
    grad_i = (1/T) * [sum_j a[j]*w[j+i] + (a*w)[i]]
    First term = cross-correlation, second = convolution.
    """
    n = len(a)
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    C1 = irfft(A * np.conj(W), fft_size)[:n]
    C2 = irfft(A * W, fft_size)[:n]
    return C1 + C2


def simplex_project(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    if target_sum <= 0:
        return np.zeros(n)
    # Binary search for theta: sum(max(v_i - theta, 0)) = target_sum
    lo = v.min() - target_sum
    hi = v.max()
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.maximum(v - mid, 0.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    return np.maximum(v - theta, 0.0)


def lse_gradient_descent(a0, time_limit, n_temps=12, verbose=True):
    """FFT-based LSE gradient descent with adaptive temperature annealing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Temperature schedule: geometric from high to low
    T_high = max(5.0, best_score * 2.0)
    T_low = max(0.1, best_score * 0.01)
    temps = np.geomspace(T_high, T_low, n_temps)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Number of steps: more for high T (smoother), fewer for low T
        steps_per_temp = max(30, min(int(remaining * 4), 500))

        for it in range(steps_per_temp):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            # Compute LSE weights
            max_c = conv.max()
            shifted = (conv - max_c) / T
            # Clip to avoid overflow
            shifted = np.clip(shifted, -500, 0)
            exp_c = np.exp(shifted)
            w_sum = exp_c.sum()
            if w_sum < 1e-300:
                w = np.zeros(size)
                w[int(np.argmax(conv))] = 1.0
            else:
                w = exp_c / w_sum

            # Compute gradient via FFT
            grad = compute_lse_grad(a, w, fft_size)
            grad /= T

            # Adaptive step size: based on gradient norm
            gnorm = np.sqrt(np.sum(grad**2) + 1e-30)
            step = (T * norm) / (gnorm * np.sqrt(n) * (1.0 + 0.1 * it))
            step = min(step, norm * 0.5)

            # Take step
            a_new = a - step * grad
            a_new = simplex_project(a_new, norm)

            # Check improvement
            score = fast_autocorr(a_new).max()
            if score <= conv.max() * 1.001:
                a = a_new
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
            else:
                # Halve step
                step *= 0.5
                a_new = a - step * grad
                a_new = simplex_project(a_new, norm)
                score = fast_autocorr(a_new).max()
                if score < conv.max():
                    a = a_new
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FFT T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_gradient_descent_v2(a0, time_limit, verbose=True):
    """Improved LSE gradient descent with momentum and better step size."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Momentum
    velocity = np.zeros(n)
    beta = 0.9

    # Multi-stage annealing
    n_stages = 15
    T_high = max(3.0, best_score * 1.5)
    T_low = max(0.05, best_score * 0.005)
    temps = np.geomspace(T_high, T_low, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        n_steps = max(20, int(remaining * 2))
        n_steps = min(n_steps, 300)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            max_c = conv.max()

            # LSE weights with clipping
            shifted = (conv - max_c) / T
            shifted = np.clip(shifted, -500, 0)
            exp_c = np.exp(shifted)
            w_sum = exp_c.sum()
            if w_sum < 1e-300:
                w = np.zeros(size)
                w[int(np.argmax(conv))] = 1.0
            else:
                w = exp_c / w_sum

            # FFT gradient
            grad = compute_lse_grad(a, w, fft_size) / T

            # Momentum update
            velocity = beta * velocity + grad
            vnorm = np.sqrt(np.sum(velocity**2) + 1e-30)

            # Step size: scale with T and inversely with velocity norm
            lr = (T * norm * 0.5) / (vnorm * np.sqrt(n) + 1e-10)
            lr = min(lr, norm * 0.3)
            # Decay learning rate within stage
            lr *= (1.0 - 0.3 * it / max(n_steps, 1))

            a_new = a - lr * velocity
            a_new = simplex_project(a_new, norm)

            score = fast_autocorr(a_new).max()
            if score < max_c + 1e-10:
                a = a_new
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
            else:
                # Try smaller step
                a_new = a - 0.5 * lr * velocity
                a_new = simplex_project(a_new, norm)
                score = fast_autocorr(a_new).max()
                if score < max_c:
                    a = a_new
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                else:
                    velocity *= 0.5

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE2 T={T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_precise(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with bisection line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            # Try 2 random targets
            for attempt in range(2):
                j = int(rng.integers(n))
                if j == i:
                    continue

                ai = a[i]
                if ai < 1e-15:
                    continue

                # Gain vector for transferring delta from i to j
                g = np.zeros(size)
                # Contribution from removing mass at i: -a shifted by i and -a shifted by n-1-i
                lo1 = max(0, i)
                hi1 = min(size, i + n)
                if hi1 > lo1:
                    g[lo1:hi1] -= a[max(0, lo1 - i):min(n, hi1 - i)]
                lo2 = max(0, n - 1 - i)
                hi2 = min(size, n - 1 - i + n)
                if hi2 > lo2:
                    g[lo2:hi2] -= a[max(0, lo2 - (n - 1 - i)):min(n, hi2 - (n - 1 - i))]

                # Contribution from adding mass at j: +a shifted by j and +a shifted by n-1-j
                lo3 = max(0, j)
                hi3 = min(size, j + n)
                if hi3 > lo3:
                    g[lo3:hi3] += a[max(0, lo3 - j):min(n, hi3 - j)]
                lo4 = max(0, n - 1 - j)
                hi4 = min(size, n - 1 - j + n)
                if hi4 > lo4:
                    g[lo4:hi4] += a[max(0, lo4 - (n - 1 - j)):min(n, hi4 - (n - 1 - j))]

                # Bisection for optimal delta in [0, ai]
                # f(delta) = max_k(conv[k] + delta*g[k] + quad_correction)
                # Quad correction: at k=2i (or n-1+i for the other direction), there's a -delta^2 term
                # Simplify: ignore quad for now (it's small for reasonable delta)

                lo = 0.0
                hi = ai
                best_d = 0.0
                best_val = conv.max()

                # First check if any improvement is possible
                if (conv + ai * g).max() >= best_val - 1e-12:
                    # Quick check with a few samples
                    for frac in [0.25, 0.5, 0.75]:
                        d = ai * frac
                        val = (conv + d * g).max()
                        if val < best_val - 1e-12:
                            break
                    else:
                        continue

                # Coarse search
                n_coarse = 10
                for ci in range(n_coarse + 1):
                    d = ai * ci / n_coarse
                    val = (conv + d * g).max()
                    if val < best_val:
                        best_val = val
                        best_d = d

                if best_d <= 0:
                    continue

                # Bisection refinement around best_d
                d_lo = max(0, ai * (int(best_d / ai * n_coarse) - 1) / n_coarse)
                d_hi = min(ai, ai * (int(best_d / ai * n_coarse) + 1) / n_coarse)
                for _ in range(15):
                    d_mid = (d_lo + d_hi) / 2
                    val_lo = (conv + d_lo * g).max()
                    val_mid = (conv + d_mid * g).max()
                    val_hi = (conv + d_hi * g).max()
                    if val_mid <= val_lo and val_mid <= val_hi:
                        break
                    elif val_lo < val_hi:
                        d_hi = d_mid
                    else:
                        d_lo = d_mid
                best_d = (d_lo + d_hi) / 2
                best_val = (conv + best_d * g).max()

                if best_val < conv.max() - 1e-12:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-12:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def multi_direction_cd(a0, time_limit, rng, verbose=True):
    """CD that tries multiple source-target pairs per iteration."""
    n = len(a0)

    s0 = a.sum()

    conv = fast_autocorr(a)
    best_score = conv.max()

    while time.monotonic() - t0 < time_limit:
        improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-9:
                continue

            # Pick 3 random targets
            targets = rng.integers(0, n, size=3)
            for j in targets:
                j = int(j)
                if j == i:
                    continue
                ai = a[i]
                if ai < 1e-15:
                    break

                # Simple: try delta = ai * fraction for several fractions
                for frac in [0.1, 0.3, 0.5, 0.7]:
                    d = ai * frac
                    # Approximate new max
                    # Fast check: only check a subset of conv positions
                    # (the ones likely to be affected)
                    new_a = a.copy()
                    new_a[i] = ai - d
                    new_a[j] = a[j] + d
                    # Only recompute conv if promising (check a few key positions)
                    # For speed, just do the full recompute
                    new_conv = fast_autocorr(new_a)
                    new_max = new_conv.max()
                    if new_max < conv.max() - 1e-12:
                        a = new_a
                        s = a.sum()
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        if conv.max() < best_score:
                            best_score = conv.max()
                            best_a = a.copy()
                        improved = True
                        conv = fast_autocorr(a)
                        break

        if not improved:
            break

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    MCD ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):

        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def perturb_and_reoptimize(a0, n_perturb, reopt_time, rng, lse_func, verbose=False):
    """Add random perturbations and re-optimize each."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    best_a = a0.copy()
    best_score = fast_autocorr(a0).max()
    t0 = time.monotonic()

    for p in range(n_perturb):
        if time.monotonic() - t0 > reopt_time:
            break
        remaining = reopt_time - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Random perturbation
        noise_scale = 0.1 + 0.3 * rng.random()
        noise = rng.standard_normal(n) * noise_scale
        a_pert = a0 + noise
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-15:
            continue
        a_pert = a_pert * norm / s

        # Re-optimize
        t_per = max(2.0, remaining / (n_perturb - p))
        a_out, _ = lse_func(a_pert, t_per, verbose=False)
        score = fast_autocorr(a_out).max()
        if score < best_score:
            best_score = score
            best_a = a_out.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Perturb {p}: {best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1750.0)

    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    n_best_current = len(best_seq)

    # === Phase 1: LSE gradient descent v2 on parent (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LSE-FFT v2 n={n_best_current} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 400.0)
    if t1 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best_current) / best_seq.sum()
        a_out, _ = lse_gradient_descent_v2(a_norm, t1, verbose=True)
        do_try_update(a_out, f"lse2_n{n_best_current}")

    # === Phase 2: LSE gradient descent (original) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best_current = len(best_seq)
    print(f"\n=== Phase 2: LSE-FFT v1 n={n_best_current} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 250.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best_current) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t2, verbose=True)
        do_try_update(a_out, f"lse1_n{n_best_current}")

    # === Phase 3: Pair CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best_current = len(best_seq)
    print(f"\n=== Phase 3: Pair CD n={n_best_current} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best_current) / best_seq.sum()
        a_out, _ = pair_cd_precise(a_norm, t3, rng, verbose=True)
        do_try_update(a_out, f"pcd_n{n_best_current}")

    # === Phase 4: Try longer length ===
    n_long = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LSE-FFT n={n_long} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 20:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = lse_gradient_descent_v2(a_norm, t4, verbose=True)
        do_try_update(a_out, f"lse2_n{n_long}")
        # CD on the longer
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            n_cur = len(best_seq)
            t4b = min(remaining * 0.5, 100.0)
            if t4b > 10:
                a_norm2 = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
                a_out2, _ = pair_cd_precise(a_norm2, t4b, rng, verbose=False)
                do_try_update(a_out2, f"pcd_n{n_cur}_long")

    # === Phase 5: Perturbation + re-optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best_current = len(best_seq)
    print(f"\n=== Phase 5: Perturbation n={n_best_current} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best_current) / best_seq.sum()
        a_out, _ = perturb_and_reoptimize(a_norm, n_perturb=5, reopt_time=t5,
                                           rng=rng, lse_func=lse_gradient_descent_v2, verbose=True)
        do_try_update(a_out, f"perturb_n{n_best_current}")

    # === Phase 6: Try even longer ===
    n_mega = n_base + 800
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: LSE-FFT n={n_mega} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 20:
        init_mega = interpolate_sequence(best_seq, n_mega)
        a_norm = init_mega * np.sqrt(2.0 * n_mega) / init_mega.sum()
        a_out, _ = lse_gradient_descent_v2(a_norm, t6, verbose=True)
        do_try_update(a_out, f"lse2_n{n_mega}")
        # Quick CD
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            n_cur = len(best_seq)
            t6b = min(remaining * 0.6, 80.0)
            if t6b > 10:
                a_norm2 = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
                a_out2, _ = pair_cd_precise(a_norm2, t6b, rng, verbose=False)
                do_try_update(a_out2, f"pcd_n{n_cur}_mega")

    # === Phase 7: Final polish with pair CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best_current = len(best_seq)
    print(f"\n=== Phase 7: Final PCD n={n_best_current} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.25, 150.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best_current) / best_seq.sum()
        a_out, _ = pair_cd_precise(a_norm, t7, rng, verbose=True)
        do_try_update(a_out, f"pcd_final_n{n_best_current}")

    # === Phase 8: Another LSE pass on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best_current = len(best_seq)
    print(f"\n=== Phase 8: Final LSE n={n_best_current} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.30, 100.0)
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best_current) / best_seq.sum()
        a_out, _ = lse_gradient_descent_v2(a_norm, t8, verbose=True)
        do_try_update(a_out, f"lse_final_n{n_best_current}")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
