5. The most efficient way is to reduce the larger of a[i] and a[k*-i] (since the product is more sensitive to the larger factor)
6. Move the mass to a location that doesn't contribute to any peak

    # Transfer a small amount

    # Renormalize

import numpy as np

from scipy.fft import rfft, irfft

_global_solver = None


def _get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return _global_solver
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    print(f"  LP solver (fallback): CLARABEL")
    return _global_solver


def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def _lse_gradient(y, T, K=3):
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -80.0, 80.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()

    fft_size2 = 1 << (3 * n - 2).bit_length()
    W_full = rfft(w, fft_size2)
    Y_full = rfft(y, fft_size2)
    cross = irfft(W_full * np.conj(Y_full), fft_size2)
    grad = 2.0 * cross[:n]
    return grad


def _lse_fista(a0, time_limit, n_peak_weights=3, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()

    T_schedule = [50.0, 15.0, 5.0, 2.0, 0.8, 0.3, 0.1, 0.04]
    iters_per_T = [3000, 4000, 5000, 6000, 8000, 10000, 12000, 15000]

    for Ti, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        L = max(n / (T * 3.0), 1.0)
        t_f = 1.0
        y = a.copy()
        z_prev = a.copy()
        max_it = iters_per_T[Ti] if Ti < len(iters_per_T) else 15000

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit:
                break
            grad = _lse_gradient(y, T, n_peak_weights)
            step = 1.0 / L
            z = y - step * grad
            a_new = _proj_simplex(z)

            if t_f == 1.0:
                y_new = a_new.copy()
            else:
                mom = (t_f - 1.0) / (t_f + 2.0)
                y_new = a_new + mom * (a_new - z_prev)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()

            z_prev = a_new.copy()
            a = a_new
            y = y_new
            t_f += 1

            if it % 1000 == 0:
                sc = _fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = _fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _peak_pair_cd(a0, time_limit, seed=42, verbose=True):
    """Coordinate descent on peak-contributing pairs."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = _fast_autocorr(a)
        top_peaks = np.argsort(c)[::-1][:15]
        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-18:
                continue
            src_rank = np.argsort(contrib)[::-1][:5]
            src_pool = idx[src_rank]

            c_min_idx = np.argmin(c)
            c_bottom = np.argsort(c)[:max(5, n // 10)]

            for si in range(len(src_pool)):
                src = src_pool[si]
                for dst in c_bottom[rng.integers(len(c_bottom))]:
                    if dst == src:
                        continue
                    if a[src] < 1e-15:
                        continue
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                        amt = a[src] * frac
                        if a[src] - amt < -1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        sc = _fast_autocorr(a_trial).max()
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.003, 0.05)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= -1e-15:
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial / s_t
                            sc = _fast_autocorr(a_trial).max()
                            if sc < best_score:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                stall = 0
            if stall > 1000:
                break

        if verbose and int(time.monotonic() - t0) % 30 < 1:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

    return best_a, best_score


def _build_conv_matrix(a, n):
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def _lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a, n)
    constraints = [T @ b_var <= M, cp.sum(b_var) <= 10.0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def _lp_nesterov(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = _fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = _lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = _fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _try_parametric_init(n, time_budget, verbose=True):
    """Try various parametric bump functions and return the best."""
    best_a = None
    best_score = float('inf')
    x = np.linspace(0.0, 1.0, n)

    params = [
        ('sin_pi_1', np.sin(np.pi * x) ** 1.0),
        ('sin_pi_1.5', np.sin(np.pi * x) ** 1.5),
        ('sin_pi_2', np.sin(np.pi * x) ** 2.0),
        ('sin_pi_3', np.sin(np.pi * x) ** 3.0),
        ('sin_pi_4', np.sin(np.pi * x) ** 4.0),
        ('sin_pi_6', np.sin(np.pi * x) ** 6.0),
        ('x_1_1mx', x * (1.0 - x)),
        ('x_2_2mx', (x * (1.0 - x)) ** 2.0),
        ('x_3_3mx', (x * (1.0 - x)) ** 3.0),
        ('gauss', np.exp(-((x - 0.5) ** 2) / (2 * 0.1 ** 2))),
        ('raised_cos', 0.5 * (1 + np.cos(2 * np.pi * x))),
        ('trapezoid', np.clip(x / 0.1, 0, 1) * np.clip((1 - x) / 0.1, 0, 1)),
    ]

    for name, a in params:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            continue
        a = a / s
        sc = _fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            if verbose:
                print(f"      Init {name}: max_ac={sc:.6f}")

    if best_a is None:
        best_a = np.ones(n) / n
        best_score = _fast_autocorr(best_a).max()
    return best_a, best_score

            _try_update(a_out, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
