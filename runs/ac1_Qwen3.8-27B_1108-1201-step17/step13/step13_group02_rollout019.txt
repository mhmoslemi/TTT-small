    b_prev = None

        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y)
        M_max = M_curr.max()
        if M_max < 1e-12:
            break

        if t < 30:
            b = lp_best_response(y, M_max, solver)
        else:
            b = lp_best_response_fast(y, M_max, solver, b_prev)

        if b is None:
            b_prev = None
            continue
        b_prev = b.copy()

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

        if stall > 300:
            rng = np.random.default_rng(n_iters * 13 + 7)
            perturb_scale = 0.015 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            sy = y.sum()
            if sy > 1e-15:
                y = y * norm / sy
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a * norm / sa
            b_prev = None
            stall = 0
            if verbose:
                print(f"    [LP Restart at iter {n_iters}]")

    return best_a, best_score


def compute_lse_gradient(a, w):
    """Gradient of sum_k w_k * (a*a)_k via FFT.
    d/d a[i] of conv(a,a)[k] = a[k-i] (for valid k-i)
    grad[i] = sum_k w[k] * a[k-i] = (w_reversed * a)[i + n - 1] ... 
    Actually: grad[i] = sum_k w[k] * (d conv[k]/d a[i]) = sum_k w[k] * a[k-i]
    Let j = k-i, so k = i+j: grad[i] = sum_j w[i+j] * a[j]
    This is a cross-correlation: grad = correlate(w, a) restricted to valid range.
    Equivalently: grad[i] = (w_flip * a)[i + len(w) - 1 - (n-1)] ... 
    Let me just use: grad[i] = sum_{j=0}^{n-1} w[i+j] * a[j] for valid i+j < 2n-1
    This equals the cross-correlation of w (length 2n-1) and a (length n).
    Using FFT: convolve(w, a_reversed) gives us what we need.
    """
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << ((2 * n - 1) + n - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return conv[n - 1:2 * n - 1]


def lse_adam_optimize(a0, time_limit, verbose=True):
    """LSE-smoothed objective with Adam optimizer and cosine annealing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_base = 2.0 / (n ** 0.5)

    T_start = max(0.5, best_score * 0.1)
    T_end = max(0.0005, best_score * 0.003)

    # Cosine schedule with warm restarts
    n_restarts = 3
    restart_period = max(50, int(time_limit / (0.005 / lr_base)))

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Cosine annealing with restarts
        phase = (steps % restart_period) / restart_period
        T = T_end + 0.5 * (T_start - T_end) * (1 + np.cos(np.pi * elapsed_frac))
        lr = lr_base * 0.5 * (1 + np.cos(np.pi * elapsed_frac))

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        # LSE weights
        scaled = (b - max_b) / max(T, 1e-10)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient
        grad = compute_lse_gradient(a, w)

        # Adam update
        steps += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** steps)
        v_hat = v / (1 - beta2 ** steps)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam step={steps} ({elapsed:.0f}s): T={T:.5f}, lr={lr:.6f}, best={best_score:.8f}")

    return best_a, best_score


def multi_peak_shave(a0, time_limit, seed=42, verbose=True):
    """Advanced peak shaving considering all top peaks simultaneously."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    T_start = max(0.3, best_score * 0.04)
    T_end = max(0.01, best_score * 0.001)

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** elapsed_frac

        b = fast_autocorr(a)
        # Consider top 12 peaks
        top_k_count = min(12, 2 * n - 1)
        top_k = np.argsort(b)[::-1][:top_k_count]

        improved = False
        for _ in range(3):  # Try 3 moves per outer iteration
            for k in top_k:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                contrib = a[idx] * a[k - idx]

                if contrib.max() < 1e-18:
                    continue

                src = idx[np.argmax(contrib)]

                # Build combined peak mask
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in top_k[:6]:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_mask[js2:je2 + 1] = True
                peak_mask[src] = False

                non_peak = np.where(~peak_mask)[0]
                if len(non_peak) < 2:
                    continue

                # Select destinations: prefer low value AND low contribution to peaks
                dst_candidates = non_peak[:min(50, len(non_peak))]
                # Score each destination by how much it would add to peaks
                dst_scores = a[dst_candidates].copy()
                for di, dst in enumerate(dst_candidates):
                    for k2 in top_k[:4]:
                        js2 = max(0, k2 - n + 1)
                        je2 = min(n - 1, k2)
                        if js2 <= dst <= je2:
                            partner = k2 - dst
                            if 0 <= partner <= n - 1:
                                dst_scores[di] += 0.5 * a[partner]
                sorted_dst = dst_candidates[np.argsort(dst_scores)][:8]
                n_dest = rng.integers(2, min(5, len(sorted_dst)) + 1)
                selected_dst = sorted_dst[rng.choice(len(sorted_dst), n_dest, replace=False)]

                frac = rng.choice([0.03, 0.05, 0.08, 0.12, 0.18])
                amt_total = a[src] * frac
                if amt_total < 1e-18:
                    continue

                dst_vals = a[selected_dst] + 1e-10
                weights = 1.0 / dst_vals
                weights = weights / weights.sum()
                amounts = amt_total * weights

                a_trial = a.copy()
                a_trial[src] -= amt_total
                for di, dst in enumerate(selected_dst):
                    a_trial[dst] += amounts[di]

                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t

                score = fast_autocorr(a_trial).max()

                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    n_improve += 1
                    break
                elif score < b[k] * 1.01 and rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                    a = a_trial
                    improved = True
                    break
            if improved:
                break

        if not improved:
            n_stall += 1
            # Random pair move
            n_trials = min(5, n // 100 + 1)
            for _ in range(n_trials):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < norm * 1e-10:
                    continue
                frac = rng.uniform(0.05, 0.2)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    n_improve += 1
                    break
                elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                    a = a_trial
                    break

            if n_stall > 150:
                # Larger perturbation
                n_pert = max(3, n // 80)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for idx in idx_pert:
                    if a[idx] < 1e-15:
                        continue
                    amt = a[idx] * rng.uniform(0.1, 0.4)
                    j = rng.integers(n)
                    a_trial[idx] -= amt
                    a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score * 1.02:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                n_stall = 0

        if verbose and n_improve % 1500 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_shaping(a0, time_limit, seed=0, verbose=True):
    """Try to reduce peaks by modifying the sequence in frequency domain.
    The idea: convolve(a,a) = IFFT(FFT(a)^2). If we can make FFT(a)^2 more
    concentrated (in frequency), the time-domain convolution becomes smoother.
    We try small multiplicative perturbations in frequency domain.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Zero-pad to convolution length
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

    steps = 0
    while time.monotonic() - t0 < time_limit:
        steps += 1
        # Random small frequency-domain perturbation
        n_bands = rng.integers(3, 15)
        band_edges = np.sort(rng.uniform(0, fft_size // 2, n_bands + 1)).astype(int)
        delta = np.zeros(fft_size // 2 + 1, dtype=np.float64)
        for bi in range(n_bands):
            lo = max(0, band_edges[bi] - 1)
            hi = min(fft_size // 2, band_edges[bi + 1])
            if hi > lo:
                delta[lo:hi] = rng.normal(0, 0.01)

        A_trial = A * (1.0 + delta)
        a_trial = irfft(A_trial, fft_size)[:n]
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial * norm / s_t

        score = fast_autocorr(a_trial).max()
        if score < best_score:
            a = a_trial
            best_score = score
            best_a = a.copy()
            A = A_trial  # Keep the successful frequency modification

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral step={steps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
