        M_use = M_curr * (1.0 + 1e-7)

        # Use full LP early, reduced LP later
        if t < 20:
            b = lp_best_response(y, M_use, solver)
        else:
            b = lp_best_response_reduced(y, M_use, solver)

        if b is None:
            # Fallback: use current y as best response (no progress)
            n_iters += 1
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(n_iters * 17 + 3)
                perturb_scale = 0.02 * a.mean()
                y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
                sy = y.sum()
                y = y * norm / sy
                a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
                sa = a.sum()
                a = a * norm / sa
                stall = 0
                patience = max(200, patience // 2)
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Check improvement
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
            best_stall = 0
        else:
            stall += 1
            best_stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
            best_stall = 0

        if verbose and n_iters % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

        # Adaptive restart
        if stall > patience:
            rng = np.random.default_rng(n_iters * 23 + 11)
            # Try different perturbation scales
            scale = rng.choice([0.01, 0.02, 0.05]) * a.mean()
            y = np.maximum(y + rng.normal(0, scale, n), 0)
            sy = y.sum()
            y = y * norm / sy
            a = np.maximum(a + rng.normal(0, scale, n), 0)
            sa = a.sum()
            a = a * norm / sa
            stall = 0
            patience = max(200, int(patience * 0.8))
            if verbose:
                print(f"    [Restart at iter {n_iters}, scale={scale:.4f}]")

        # Early termination if stuck for a very long time
        if best_stall > 3000:
            break

    return best_a, best_score


def compute_lse_gradient(a, w):
    """Compute gradient of sum_k w_k * (a*a)_k using FFT."""
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def lse_gradient_descent(a0, time_limit, verbose=True):
    """Smooth gradient descent using LSE approximation of max."""
    n = len(a0)
    norm = sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = binomial_init(n) * norm
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    # Multi-stage annealing
    T_start = max(0.5, best_score * 0.15)
    T_mid = max(0.1, best_score * 0.03)
    T_end = max(0.01, best_score * 0.003)

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        if elapsed_frac < 0.3:
            T = T_start * (T_mid / T_start) ** (elapsed_frac / 0.3)
        elif elapsed_frac < 0.7:
            T = T_mid * (T_end / T_mid) ** ((elapsed_frac - 0.3) / 0.4)
        else:
            progress = (elapsed_frac - 0.7) / 0.3
            T = T_end * (1.0 / T_end) ** (-progress * 0)  # Stay at T_end
            T = T_end

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        # Softmax weights with numerical stability
        shifted = (b - max_b) / T
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Compute gradient
        grad = compute_lse_gradient(a, w)

        # Adaptive step size with backtracking
        base_step = 1.0 / (n * 0.3)
        a_new = a - base_step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            # Try smaller step
            base_step *= 0.1
            a_new = a - base_step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Periodic random perturbation to escape local minima
        if steps % 500 == 0:
            rng = np.random.default_rng(steps * 7 + 1)
            n_pert = max(1, n // 100)
            idx = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in idx:
                a_trial[i] *= rng.uniform(0.7, 1.3)
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial = a_trial * norm / st
                sc = fast_autocorr(a_trial).max()
                if sc < best_score * 1.001:
                    a = a_trial
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()

        if verbose and steps % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")

    return best_a, best_score


def mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent (multiplicative weights) with LSE smoothing."""
    n = len(a0)
    norm = sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    s0 = a.sum()
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    T_start = max(0.3, best_score * 0.1)
    T_end = max(0.01, best_score * 0.002)
    eta_start = 1.0 / (n ** 0.5)
    eta_end = 10.0 / (n ** 0.5)

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** frac
        eta = eta_start * (eta_end / eta_start) ** frac

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        shifted = (b - max_b) / T
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        grad = compute_lse_gradient(a, w)

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDescent step={steps} ({elapsed:.0f}s): eta={eta:.4f}, best={best_score:.8f}")

    return best_a, best_score


def peak_shave_local_search(a0, time_limit, seed=42, verbose=True):
    """Targeted peak shaving: identify peaks and redistribute mass."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = binomial_init(n) * norm
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    T_cur = max(0.1, best_score * 0.02)
    T_min = best_score * 0.001

    while time.monotonic() - t0 < time_limit:
        # Anneal temperature
        T_cur = max(T_min, T_cur * 0.999)

        b = fast_autocorr(a)
        max_b = b.max()
        peak_lags = np.argsort(b)[::-1][:10]

        improved = False
        for k in peak_lags:
            if b[k] < max_b * 0.95:
                break

            # Find contributing pairs
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-18:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-15:
                continue

            # Build mask of positions involved in top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in peak_lags:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) == 0:
                non_peak = np.where(np.arange(n) != src)[0]

            # Pick destinations: prefer low-value positions not in peak mask
            n_dest = min(8, len(non_peak))
            sorted_dst = non_peak[np.argsort(a[non_peak])][:n_dest * 4]
            selected_dst = sorted_dst[rng.choice(len(sorted_dst), min(n_dest, len(sorted_dst)), replace=False)]

            # Try different fractions
            for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.25]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue

                # Distribute to destinations
                dst_weights = 1.0 / (a[selected_dst] + 1e-10)
                dst_weights = dst_weights / dst_weights.sum()
                amounts = amt * dst_weights

                a_trial = a.copy()
                a_trial[src] -= amt
                for di, dst in enumerate(selected_dst):
                    a_trial[dst] += amounts[di]

                if a_trial[src] < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t

                score = fast_autocorr(a_trial).max()

                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    n_improve += 1
                    break
                elif rng.random() < np.exp(-(score - best_score) / max(T_cur, 1e-10)):
                    a = a_trial
                    improved = True
                    break

        if not improved:
            n_stall += 1

            # Random pair swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.02, 0.2)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        n_improve += 1
                    elif rng.random() < np.exp(-(score - best_score) / max(T_cur, 1e-10)):
                        a = a_trial

            # Larger perturbation if stuck
            if n_stall > 100:
                n_pert = max(2, n // 100)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for idx in idx_pert:
                    amt = a[idx] * rng.uniform(0.1, 0.4)
                    j = rng.integers(n)
                    a_trial[idx] -= amt
                    a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score * 1.005:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                            n_improve += 1
                n_stall = 0

        if verbose and n_improve % 2000 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: finds best sequence minimizing the evaluation score."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    rng = np.random.default_rng(seed)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize with binomial coefficients if no parent
    if best_seq is None:
        n_base = 1600
        best_seq = binomial_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Binomial init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Also try binomial init at base length and compare
    binom = binomial_init(n_base)
    binom_score = evaluate_safe(binom)
    if binom_score < best_score:
        best_score = binom_score
        best_seq = binom.copy()
        print(f"  Binomial init better: {best_score:.6f}")

    # === Phase 1: LP Nesterov at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_cur}")

    # === Phase 2: LSE Gradient Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LSE-GD n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t2, verbose=True)
        do_try_update(a_out, f"lse_n{n_cur}")

    # === Phase 3: Peak Shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: PeakShave n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave_local_search(a_norm, t3, seed=123, verbose=True)
        do_try_update(a_out, f"ps_n{n_cur}")

    # === Phase 4: Mirror Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: MirrorDescent n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = mirror_descent(a_norm, t4, verbose=True)
        do_try_update(a_out, f"md_n{n_cur}")

    # === Phase 5: Try longer length (n+200) ===
    n2 = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.18, 250.0)
    if t5 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        # Also try binomial at new length
        binom_long = binomial_init(n2)
        sc_init = evaluate_safe(init_long)
        sc_binom = evaluate_safe(binom_long)
        if sc_binom < sc_init:
            init_long = binom_long.copy()
        a_norm = init_long * sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n2}")

    # Phase 5b: LSE polish at new length
    remaining = time_budget - (time.monotonic() - t_global)
    t5b = min(remaining * 0.06, 80.0)
    if t5b > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t5b, verbose=False)
        do_try_update(a_out, f"lse_polish_n{n_cur}")

    # === Phase 6: Final LP on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.20, 250.0)
    if t6 > 20:
        a_norm = best_seq * sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final PeakShave n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 10:
        a_norm = best_seq * sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_shave_local_search(a_norm, t7, seed=999, verbose=True)
        do_try_update(a_out, f"ps_final_n{n_best}")

    # === Phase 8: Final LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LSE n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.15, 120.0)
    if t8 > 10:
        a_norm = best_seq * sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t8, verbose=True)
        do_try_update(a_out, f"lse_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence! Returning default.")
        return [1.0] * 1600

    return [float(x) for x in best_seq]
