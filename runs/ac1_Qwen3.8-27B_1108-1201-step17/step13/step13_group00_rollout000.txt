    # This is the same as the first term with roles of w and a swapped, evaluated at i.
    # First term (swapped): sum_k a_k * w_{k-i}
    # That's cross1_swapped[i] = ifft(FFT(a) * conj(FFT(w)))[i] = cross2[i]
    # But we want sum_k a_k * w_{i-k} = sum_k a_k * w_{-(k-i)}
    # = cross2[fft_size - i] (for the circular version)
    
    # Ugh, let me just verify: for small n, compute both ways.
    # Actually, for the purpose of this problem, note that:
    # sum_k w_k * a_{k-i} (first term, i from 0 to n-1)
    # sum_k w_k * a_{i-k} (second term, i from 0 to n-1)
    # The second term with i -> i is the same as the first term with the roles reversed.
    # Specifically: sum_k w_k * a_{i-k} = sum_j a_j * w_{i-j} (j = i-k)
    # And the first term is: sum_k w_k * a_{k-i}
    # These are different unless w and a are symmetric.
    
    # Let me just compute the second term as:
    # sum_k w_k * a_{i-k} for k in [max(0, i-n+1), min(i, n-1)]
    # This is the (i)-th element of the convolution (a * w_reversed) where w_reversed = w[::-1]
    # Because (a * w_rev)[i] = sum_j a_j * w_rev[i-j] = sum_j a_j * w[n-1-(i-j)] = sum_j a_j * w[n-1-i+j]
    # Hmm, that's getting complicated.
    
    # Simplest: just use the first term result and note that:
    # sum_k w_k * a_{i-k} = sum_k a_k * w_{i-k} (just renaming the dummy index)
    # = (a conv w)[i] where we use the standard convolution definition
    # = irfft(FFT(a) * FFT(w))[i] (linear convolution, valid for i < fft_size)
    
    conv_aw = irfft(A * W, fft_size)[:size]  # This is the linear convolution of a and w
    # conv_aw[i] = sum_j a_j * w[i-j] for valid j
    # We want sum_k w_k * a_{i-k} = sum_k a_{i-k} * w_k
    # Let j = i-k: sum_j a_j * w_{i-j} = conv_aw[i]
    
    g2 = conv_aw[:n]
    
    grad = g1 + g2
    return grad, conv

def adam_lse_optimize(a0, time_limit, verbose=True):
    """LSE gradient descent with Adam optimizer."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    conv0 = fast_autocorr(a)
    best_score = conv0.max()
    t0 = time.monotonic()
    
    # Adam parameters
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr_base = 0.5
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    # Temperature schedule: geometric from high to low
    n_stages = 12
    temps = np.geomspace(50.0, 0.1, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        n_steps = max(50, int(remaining * 2))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t += 1
            
            grad, conv = lse_grad_fft(a, T)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            lr = lr_base / (1.0 + 0.1 * it)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new * norm / s
            else:
                a_new = a.copy()
            a = a_new
            
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Adam-LSE stage T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def frank_wolfe_optimize(a0, time_limit, verbose=True):
    """Frank-Wolfe (conditional gradient) with LSE and Nesterov momentum."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()  # Nesterov point
    avg_a = np.zeros(n)
    n_avg = 0
    
    n_stages = 10
    temps = np.geomspace(30.0, 0.2, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        n_steps = max(20, int(remaining * 3))
        
        for it in range(1, n_steps + 1):
            if time.monotonic() - t0 > time_limit:
                break
            
            grad, conv = lse_grad_fft(y, T)
            
            # FW step: find v = argmin_{x in feasible} <grad, x>
            # Feasible set: x >= 0, sum(x) = norm
            # argmin is: v_i = norm if grad_i = min(grad), else 0
            # But for LSE (smooth), we can do better:
            # v_i = max(0, lambda - grad_i) normalized to sum = norm
            # This is the "soft" FW step (projected onto the simplex-like set)
            
            # Actually, the feasible set is {x >= 0, sum(x) = norm}
            # The linear minimizer is: v = norm * e_{argmin(grad)}
            # But we can use a "lazy" version: v_i = max(0, c - grad_i)
            # Choose c so that sum(v) = norm
            
            g_min = grad.min()
            g_max = grad.max()
            if g_max - g_min < 1e-15:
                v = np.ones(n) * norm / n
            else:
                # Binary search for c in [g_min, g_max] such that sum(max(0, c - grad)) = norm
                lo, hi = g_min, g_max
                for _ in range(50):
                    mid = (lo + hi) / 2
                    v_trial = np.maximum(0.0, mid - grad)
                    s = v_trial.sum()
                    if s > norm:
                        hi = mid
                    else:
                        lo = mid
                c = (lo + hi) / 2
                v = np.maximum(0.0, c - grad)
                s = v.sum()
                if s > 1e-15:
                    v = v * norm / s
                else:
                    v = np.ones(n) * norm / n
            
            # Nesterov update
            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * v
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y > 1e-15:
                y_new = y_new * norm / s_y
            
            if it == 1:
                a_new = v.copy()
            else:
                momentum = (it - 1.0) / (it + 2.0)
                a_new = y_new + momentum * (y_new - y)
                a_new = np.maximum(a_new, 0.0)
                s_a = a_new.sum()
                if s_a > 1e-15:
                    a_new = a_new * norm / s_a
                else:
                    a_new = y_new.copy()
            
            y = y_new
            a = a_new
            
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            avg_a += a
            n_avg += 1
            if n_avg % 20 == 0:
                avg_check = avg_a / n_avg
                s_avg = avg_check.sum()
                if s_avg > 1e-15:
                    avg_check = avg_check * norm / s_avg
                    sc = fast_autocorr(avg_check).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = avg_check.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    FW-LSE T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def peak_activated_cd(a, time_limit, rng, verbose=True):
    """
    Coordinate descent that focuses on pairs affecting the current peaks.
    For the top K peaks, identify which a_i values most influence them,
    and try to redistribute mass.
    """
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    n_passes = 0
    no_improve = 0
    K = 5  # Number of peaks to consider
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        
        # Find top K peaks
        peak_indices = np.argsort(conv)[-K:][::-1]
        
        # For each peak, compute the "influence" of each a_i
        # Peak at k: conv_k = sum_i a_i * a_{k-i}
        # d(conv_k)/da_i = a_{k-i} + a_{i-k} (for valid indices)
        # We want to reduce the peaks, so we want to decrease a_i where influence is high
        # and increase where influence is low.
        
        # Combine influences from all top peaks
        influence = np.zeros(n)
        for k in peak_indices:
            weight = conv[k] - conv.min()  # How "bad" is this peak?
            for i in range(n):
                j1 = k - i
                if 0 <= j1 < n:
                    influence[i] += weight * a[j1]
                j2 = i - k
                if 0 <= j2 < n:
                    influence[i] += weight * a[j2]
        
        # Try to transfer from high-influence to low-influence indices
        # Sort by influence
        order = np.argsort(influence)
        
        # Try transfers from high to low
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick a high-influence source and low-influence target
            i = int(order[rng.integers(n // 2, n)])  # High influence
            j = int(order[rng.integers(0, n // 2)])  # Low influence
            
            if i == j or a[i] < 1e-10:
                continue
            
            # Try to transfer a small amount from i to j
            # Use the influence to estimate the benefit
            max_transfer = a[i] * 0.1  # Don't transfer too much
            n_trials = 10
            best_d = 0.0
            best_val = best_score
            
            for trial in range(n_trials):
                d = max_transfer * (trial + 1) / (n_trials + 1)
                # Approximate new conv at the peaks
                # conv_k changes by d * (a_{k-j} - a_{k-i} + a_{j-k} - a_{i-k})
                # For simplicity, just evaluate at current peaks
                new_peak_val = 0.0
                for k in peak_indices:
                    delta = 0.0
                    j1 = k - j
                    if 0 <= j1 < n:
                        delta += a[j1]
                    j2 = j - k
                    if 0 <= j2 < n:
                        delta += a[j2]
                    j3 = k - i
                    if 0 <= j3 < n:
                        delta -= a[j3]
                    j4 = i - k
                    if 0 <= j4 < n:
                        delta -= a[j4]
                    new_peak_val = max(new_peak_val, conv[k] + d * delta)
                
                if new_peak_val < best_val - 1e-12:
                    best_val = new_peak_val
                    best_d = d
            
            if best_d > 0:
                a[i] -= best_d
                a[j] += best_d
                s = a.sum()
                if s > 1e-15:
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-12:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)
                else:
                    a[j] -= best_d
                    a[i] += best_d
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0
        
        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, 1.0, n)
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))  # Sigmoid
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)  # Bump
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2  # Triangle
    elif choice == 6:
        # Random smooth
        a = rng.random(n)
        # Smooth with a window
        kernel = np.ones(21) / 21
        a = np.convolve(a, kernel, mode='same')
    else:
        # Fejer-like kernel
        x = np.linspace(0.0, 1.0, n)
        a = np.sinc(5 * (x - 0.5)) ** 2
        a = np.abs(a)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def perturb(a, rng, strength=0.1):
    """Add random perturbation and renormalize."""
    n = len(a)
    norm = a.sum()
    noise = rng.standard_normal(n) * strength
    a_new = a + noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * norm / s
