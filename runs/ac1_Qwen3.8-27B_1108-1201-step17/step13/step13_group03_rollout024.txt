def _autocorr(a):
    """Compute full autocorrelation conv(a,a) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def _j_score(a):
    """Compute J(a) = 2*n*max_b/(sum_a)^2"""
    n = len(a)
    b = _autocorr(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-15:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

    T = np.zeros((2 * n - 1, n))

    return T


def _lp_best_response(a, M, solver):

        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

def _active_set_minimax(a, K, time_limit, solver, verbose=True):
    """
    Active set approach: identify top K peaks, solve QP to minimize their max.
    """
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _autocorr(a).max()
    t0 = time.monotonic()

    # Build the full autocorrelation gradient for active peaks
    for outer in range(100):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        b = _autocorr(a)
        # Get top K peak indices
        top_idx = np.argsort(b)[::-1][:K]
        M_target = b[top_idx[-1]]  # The K-th largest peak

        # Build constraint matrix for only top K peaks
        a_norm = a / a.sum()  # normalize for the LP
        T_sub = np.zeros((len(top_idx), n))
        for i, k in enumerate(top_idx):
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    T_sub[i, j] = a_norm[idx]

        # Solve: max sum(b_var) s.t. T_sub @ b_var <= M_target * (a.sum())^2, b_var >= 0
        # Actually, let's just use the full LP but with a tighter M
        M_curr = b.max()
        resp = _lp_best_response(a_norm * a.sum(), M_curr, solver)
        if resp is None:
            break
        resp = resp * a.sum() / resp.sum() if resp.sum() > 1e-15 else a.copy()

        # Nesterov update
        t = outer + 1
        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * resp
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new * norm / s_new

        score = _autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def _local_search(a, time_limit, seed=42, verbose=True):
    """
    Local search with perturbations targeting peak reduction.
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = _autocorr(a)
        top_k = np.argsort(b)[::-1][:min(20, 2 * n - 1)]

        # Strategy 1: Move mass from peak-contributing elements to non-contributing
        improved = False
        for k in top_k[:10]:
            # Find elements that contribute most to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-18:
                continue
            src = idx[np.argmax(contrib)]

            # Find a good destination: low value, far from peaks
            # Compute "peak exposure" for each element
            exposure = np.zeros(n)
            for k2 in top_k[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                exposure[js2: je2 + 1] += 1

            # Prefer destinations with low value and low exposure
            scores_dest = a * (1.0 + 0.1 * exposure)
            candidates = np.where(exposure == 0)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            candidates = candidates[np.argsort(scores_dest[candidates])][:max(1, len(candidates) // 5)]

            if len(candidates) == 0:
                continue

            dst = candidates[rng.integers(len(candidates))]

            # Try different transfer amounts
            for frac in [0.001, 0.003, 0.007, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = _autocorr(a_trial).max()
                if score < best_score * (1.0 - 1e-10):
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        # Strategy 2: Random perturbation (for exploration)
        if not improved:
            no_improve += 1
            if no_improve > 200:
                # Try a bigger random perturbation
                n_perturb = max(1, n // 100)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                noise = rng.normal(0, 0.01 * a.mean(), size=n)
                a_trial = a + noise
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = _autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        no_improve = 0
                        if verbose:
                            print(f"    LocalSearch (restart) ({time.monotonic()-t0:.0f}s): {best_score:.8f}")
                if no_improve > 2000:
                    break
            else:
                # Small random move
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-12:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = _autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                no_improve = 0

    return best_a, best_score


def _interpolate_seq(f, new_n):
    """Upsample a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * np.sqrt(2.0 * new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n + 1)
    old_x[0] = -f_norm[0] / (2.0 * old_n)  # linear extrapolation
    old_x[-1] = 1.0 + f_norm[-1] / (2.0 * old_n)
    # Actually, just use simple interpolation
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * np.sqrt(2.0 * new_n) / new_n
    return f_interp * np.sqrt(2.0 * new_n) / ss

(The full convolution has R(k) at position k and R(n-1-k) at position n-1-k... actually it's more complex for the full convolution.)

For the full convolution conv(a,a):
- Position 0: a_0^2
- Position 1: 2*a_0*a_1
- ...
- Position n-1: sum_{i=0}^{n-1} a_i * a_{n-1-i}
- ...
- Position 2n-2: a_{n-1}^2

The "center" of the convolution (position n-1) is always the largest for a "concentrated" sequence. To minimize the max, we want to spread out the sequence.

**Idea: Use a "target autocorrelation" and solve for a that achieves it**

If we want conv(a,a) = c * 1 (constant), then in the Fourier domain:
|A(f)|^2 = c * (1 + 2*sum_k cos(2*pi*f*k/(2n-1)))... this doesn't quite work because the convolution length is 2n-1, not a power of 2.

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. LSE-smoothed FISTA with adaptive temperature (fast bulk optimization)
2. LP best-response with Nesterov acceleration (precise minimax)
3. Targeted peak-shaving local search (micro-improvements)
4. Multi-length strategy with interpolation
Key improvements: adaptive temperature scheduling, sparse LP matrices,
averaged iterates, and aggressive peak-targeting local search.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix

_global_solver = None


def _get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return _global_solver
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    print(f"  LP solver (fallback): CLARABEL")
    return _global_solver


def _autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def _j_from_autocorr(a, b):
    n = len(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-15:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def _lse_gradient(y, T):
    n = len(y)
    ac = _autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def _lse_fista(a0, T_schedule, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    t_fista = 1
    staging = a.copy()

    for stage_i, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        L = max(n / (T * 1.5), 2.0)
        max_iters = 50000
        stage_start = time.monotonic()

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score

            grad = _lse_gradient(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = _proj_simplex(z)

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()

            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1

            if it % 2000 == 0:
                sc = _autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = _autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    FISTA stage={stage_i} T={T:.3f} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def _build_sparse_conv_matrix(a, n):
    T = lil_matrix((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        T[j: j + n, j] = a
    return T.tocsr()


def _lp_solve(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_sparse_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=solver, verbose=False)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def _lp_nesterov(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_print = 0
    stale = 0

    for t in range(1, 1000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        M_curr = _autocorr(y).max()
        if M_curr < 1e-15:
            break

        b = _lp_solve(y, M_curr, solver)
        if b is None:
            stale += 1
            if stale > 20:
                break
            continue

        stale = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = _autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = _autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 5 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = _autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters - last_print >= 30:
            last_print = n_iters
            j_val = 2.0 * n * best_score / (norm * norm)
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): max_b={best_score:.6f}, J={j_val:.6f}")

    if n_avg > 0:
        avg_final = avg_a / n_avg
        s_af = avg_final.sum()
        if s_af > 1e-15:
            avg_final = avg_final * norm / s_af
            score_avg = _autocorr(avg_final).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_final.copy()

    return best_a, best_score


def _peak_shave(a, time_limit, seed=42, verbose=True):
    n = len(a)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = _autocorr(a)
        top_k = np.argsort(b)[::-1][:min(15, 2 * n - 1)]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-18:
                continue
            src = idx[np.argmax(contrib)]

            exposure = np.zeros(n)
            for k2 in top_k[:8]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                exposure[js2: je2 + 1] += 1.0

            candidates = np.where(exposure < 1.0)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            if len(candidates) == 0:
                continue

            scores_dest = a + 0.01 * exposure
            candidates = candidates[np.argsort(scores_dest[candidates])][:max(1, len(candidates) // 4)]
            dst = candidates[rng.integers(len(candidates))]

            for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = _autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break

            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve % 100 == 0 and no_improve > 100:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-12:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = _autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                no_improve = 0
                    if verbose and no_improve % 500 == 0:
                        print(f"    PeakShave ({time.monotonic()-t0:.0f}s): score={best_score:.8f}, stall={no_improve}")
            if no_improve > 3000:
                break

    return best_a, best_score


def _interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * np.sqrt(2.0 * new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * np.sqrt(2.0 * new_n) / new_n
    return f_interp * np.sqrt(2.0 * new_n) / ss
