_solver = None

def _get_solver():

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out

def _lse_grad(y, T):
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def _lse_fista(a0, T_schedule, iters_per_stage, time_limit, verbose=False):

            # Score: prefer low current value and low "gradient damage"
            # Gradient of C(k*) at position i: a[k*-i] + a[i-k*] (if in range)
            grad_vals = np.zeros(len(candidates))
            for ci, ci_idx in enumerate(candidates):
                g = 0.0
                j1 = k - ci_idx
                if 0 <= j1 < n:
                    g += a[j1]
                j2 = ci_idx - k
                if 0 <= j2 < n:
                    g += a[j2]
                grad_vals[ci] = g
            
            # Select bottom 10% by score = grad + 0.5 * a_val
            scores = grad_vals + 0.5 * a[candidates]
            sorted_scores = np.argsort(scores)
            top_candidates = candidates[sorted_scores[:max(5, len(candidates)//20)]]
            dst = top_candidates[rng.integers(len(top_candidates))]
            
            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = _fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 100 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-10:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    s_t = a_trial.sum()
                    if s_t > 0:
                        a_trial = a_trial * norm / s_t
                        score = _fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 5000:
                break
    return best_a, best_score

def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _eval_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def _simplex_param_optimize(a0, time_limit, n_basis=30, verbose=False):
    """Optimize a sequence represented as convex combination of basis functions."""
    n = len(a0)
    a0 = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a0.sum()
    if s < 1e-15:
        a0 = np.ones(n) / n
        s = 1.0
    a0 = a0 / s
    
    # Create basis functions: shifted and scaled versions of a0
    # Plus some parametric ones
    rng = np.random.default_rng(0)
    basis = [a0.copy()]
    # Gaussian bumps
    x = np.arange(n) / n
    for b_i in range(min(n_basis - 1, 20)):
        center = 0.15 + 0.7 * b_i / max(1, min(n_basis-1, 20) - 1)
        width = 0.05 + 0.15 * rng.random()
        b = np.exp(-((x - center)**2) / (2 * width**2))
        b = np.maximum(b, 0)
        ss = b.sum()
        if ss > 0:
            b /= ss
        basis.append(b)
    # Cosine modes
    for b_i in range(3):
        freq = b_i + 2
        b = np.abs(np.cos(np.pi * freq * x)) ** 2 + 0.1
        b /= b.sum()
        basis.append(b)
    
    basis = np.array(basis)
    m = len(basis)
    
    # Initialize alpha
    alpha = np.ones(m) / m
    
    # Optimize alpha using gradient descent
    t0 = time.monotonic()
    lr = 0.1
    best_a = (alpha @ basis) / (alpha @ basis).sum()
    best_score = _fast_autocorr(best_a).max()
    
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        a_cur = alpha @ basis
        s_cur = a_cur.sum()
        if s_cur < 1e-15:
            break
        a_cur = a_cur / s_cur
        b = _fast_autocorr(a_cur)
        k_star = int(np.argmax(b))
        
        # Gradient w.r.t. a
        grad_a = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            if 0 <= j1 < n:
                grad_a[i] += 2 * a_cur[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad_a[i] += 2 * a_cur[j2]
        
        # Gradient w.r.t. alpha: grad_alpha[k] = basis[k] @ grad_a / s_cur
        grad_alpha = basis @ grad_a / s_cur
        # Subtract mean to handle the normalization constraint
        grad_alpha -= np.mean(grad_alpha)
        
        alpha -= lr * grad_alpha
        alpha = np.maximum(alpha, 0)
        alpha_sum = alpha.sum()
        if alpha_sum < 1e-15:
            alpha = np.ones(m) / m
        else:
            alpha /= alpha_sum
        
        a_new = alpha @ basis
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        score = _fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        # Adaptive learning rate
        if it % 1000 == 0:
            lr *= 0.95
    
    # Final: return best_a scaled to proper norm
    norm = np.sqrt(2.0 * len(best_a))
    return best_a * norm / best_a.sum(), best_score
