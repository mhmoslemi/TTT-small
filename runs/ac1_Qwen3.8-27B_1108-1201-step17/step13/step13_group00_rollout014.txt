def _pair_cd(a0, time_limit, rng, verbose=True):

        n_passes += 1
        improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < 1e-10:
                continue

            g = np.zeros(size)
            # Positive part: adding to position j
            # Negative part: removing from position i
            # g[k] = a[k-i] + a[i-k] (coefficients in conv when mass at i)
            for k in range(max(0, i), min(size, i + n)):
                g[k] += a[k - i]
            for k in range(max(0, n - 1 - i), min(size, n + i)):

                if 0 <= j_val < n:
                    g[k] += a[j_val]

            # Try top candidate j's based on reducing the peak
            max_k = np.argmax(conv)
            candidate_js = []
            # j's that reduce the peak: we want a[k*-j] + a[j-k*] to be small
            peak_contrib = np.zeros(n)
            for j in range(n):
                v = 0.0
                k1 = max_k - j
                if 0 <= k1 < n:
                    v += a[k1]
                k2 = j - max_k
                if 0 <= k2 < n:
                    v += a[k2]
                peak_contrib[j] = v

            # Pick j's with smallest peak contribution (excluding i)
            sorted_j = np.argsort(peak_contrib)
            for jj in sorted_j:
                if jj != i:
                    candidate_js.append(int(jj))
                if len(candidate_js) >= 5:
                    break

            # Also try j = i±1, i±2 for local moves
            for offset in [1, -1, 2, -2]:
                jj = i + offset
                if 0 <= jj < n and jj != i and jj not in candidate_js:

            for j in candidate_js:
                if j < 0 or j >= n or j == i:
                    continue
                if a[i] < 1e-10:
                    break

                # Compute the change vector for transferring delta from i to j
                dv = np.zeros(size)
                # Remove delta from i: change in conv is -delta * (contribution of i)
                for k in range(max(0, i), min(size, i + n)):
                    dv[k] -= a[k - i]
                for k in range(max(0, n - 1 - i), min(size, n + i)):
                    j_val = n - 1 - k
                    if 0 <= j_val < n:

                # Add delta to j: change in conv is +delta * (contribution of j)
                for k in range(max(0, j), min(size, j + n)):
                    dv[k] += a[k - j]
                for k in range(max(0, n - 1 - j), min(size, n + j)):
                    j_val = n - 1 - k
                    if 0 <= j_val < n:
                        dv[k] += a[j_val]

                # Find best delta in [0, a[i]]
                best_d = 0.0
                best_val = conv.max()
                # Try a few delta values (geometric spacing)
                n_samples = 10
                deltas = np.geomspace(1e-6 * a[i], a[i], n_samples)
                for d in deltas:
                    new_conv = conv + d * dv
                    # Second-order correction at k=2i (reducing) and k=2j (increasing)

                    a[i] -= best_d
                    a[j] += best_d
                    a = _normalize_sqrt_2n(a)
                    conv = _fast_autocorr(a)
                    cur = conv.max()
                    if cur < best_score - 1e-10:
                        best_score = cur
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                    else:
                        # Revert
                        a[j] -= best_d
                        a[i] += best_d
                        a = _normalize_sqrt_2n(a)
                        conv = _fast_autocorr(a)

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _structured_init(n, rng):
    """Generate a structured initialization."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def _evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: finds best sequence within time budget."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if parent is None:
        n_base = 1600
        best_seq = _structured_init(n_base, rng)
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: Main LP Nesterov on parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 480.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = _nesterov_lp(a_norm, t1, solver, verbose=True)
        _try_update(a_out, f"lp_main_n{n_base}")

    # === Phase 2: Iterated local search (perturb + re-optimize) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Iterated Local Search n={n_best} (budget={remaining:.0f}s) ===")
    t2_total = min(remaining * 0.25, 350.0)
    if t2_total > 30:
        n_trials = 5
        t_per_trial = t2_total / n_trials
        epsilons = [0.05, 0.1, 0.15, 0.25, 0.4]

        for trial in range(n_trials):
            if time.monotonic() - t_global > time_budget - 60:
                break
            eps = epsilons[trial % len(epsilons)]
            noise = rng.random(n_best)
            noise = noise / noise.sum()
            a_pert = (1.0 - eps) * best_seq + eps * noise
            a_pert = np.maximum(a_pert, 0.0)
            s = a_pert.sum()
            if s < 1e-10:
                continue
            a_pert = a_pert / s
            a_norm = a_pert * np.sqrt(2.0 * n_best) / a_pert.sum()
            t_trial = min(t_per_trial, time_budget - (time.monotonic() - t_global) - 60)
            if t_trial < 10:
                break
            a_out, _ = _nesterov_lp(a_norm, t_trial, solver, verbose=False)
            _try_update(a_out, f"ils_{trial}_eps{eps}")

    # === Phase 3: Different structured initializations ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Structured Inits (budget={remaining:.0f}s) ===")
    t3_total = min(remaining * 0.15, 200.0)
    if t3_total > 30:
        n_inits = 3
        t_per = t3_total / n_inits
        for ii in range(n_inits):
            if time.monotonic() - t_global > time_budget - 60:
                break
            a_init = _structured_init(n_best, rng)
            a_norm = a_init * np.sqrt(2.0 * n_best) / a_init.sum()
            t_trial = min(t_per, time_budget - (time.monotonic() - t_global) - 60)
            if t_trial < 15:
                break
            a_out, _ = _nesterov_lp(a_norm, t_trial, solver, verbose=False)
            _try_update(a_out, f"struct_{ii}_n{n_best}")

    # === Phase 4: Pair CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Pair CD n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 180.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _pair_cd(a_norm, t4, rng, verbose=True)
        _try_update(a_out, f"pcd_n{n_best}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-Length (budget={remaining:.0f}s) ===")
    t5_total = min(remaining * 0.15, 200.0)
    if t5_total > 30:
        # Try longer
        n_long = n_best + 400
        t5a = min(t5_total * 0.6, time_budget - (time.monotonic() - t_global) - 60)
        if t5a > 20 and n_long <= 4000:
            init_long = _interpolate_seq(best_seq, n_long)
            a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
            a_out, _ = _nesterov_lp(a_norm, t5a, solver, verbose=False)
            _try_update(a_out, f"long_n{n_long}")

        # Try shorter
        n_short = max(800, n_best - 400)
        if n_short != n_best:
            t5b = min(t5_total * 0.4, time_budget - (time.monotonic() - t_global) - 60)
            if t5b > 20:
                init_short = _interpolate_seq(best_seq, n_short)
                a_norm = init_short * np.sqrt(2.0 * n_short) / init_short.sum()
                a_out, _ = _nesterov_lp(a_norm, t5b, solver, verbose=False)
                _try_update(a_out, f"short_n{n_short}")

    # === Phase 6: Final LP Nesterov polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final Polish n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.5, 200.0)
    if t6 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _nesterov_lp(a_norm, t6, solver, verbose=True)
        _try_update(a_out, f"final_polish_n{n_best}")

    # === Phase 7: Final Pair CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.4, 100.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _pair_cd(a_norm, t7, rng, verbose=True)
        _try_update(a_out, f"final_pcd_n{n_best}")

    # Final check
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
