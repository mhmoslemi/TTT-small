Now let me think about the peak-attacking descent more carefully.

The idea is: given a sequence a, find the peak position k*. The value at k* is:
conv[k*] = sum_{j} a[j] * a[k*-j] (with bounds)

To reduce this, we want to decrease some a[j] where a[k*-j] is large. But we also need to account for the effect on other convolution values.

A simple approach:
1. Find k* = argmax conv.
2. Find j* that maximizes a[j*] * a[k* - j*] (the most contributing term).
3. Move a small amount of mass from j* to some position j' that minimizes the increase in other peaks.
4. Check if the overall max decreased.

For step 3, a good choice for j' would be a position where:
- a[j'] is already small (so adding mass doesn't create a new peak).
- j' is far from all current peak positions (so it doesn't contribute to existing peaks).

                # Distance from each peak
                peak_badness = np.zeros(n)

                    for pos in range(max(0, k2-n+1), min(n, k2+1)):

                    break

def bregman_optimize(a0, time_limit, verbose=False):
    """Multiplicative (Bregman) updates for max-autocorrelation minimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    # Ensure strictly positive for multiplicative updates
    a = a + 1e-10
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    eta = 0.01  # Initial step size
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        k_star = int(np.argmax(b))
        
        # Subgradient of max at k_star
        # grad[i] = a[k_star - i] + a[i - k_star] (with bounds)
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            if 0 <= j1 < n:
                grad[i] += a[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += a[j2]
        
        # Multiplicative update
        # a[i] *= exp(-eta * grad[i] / max(grad))
        grad_norm = grad.max()
        if grad_norm < 1e-15:
            break
        update = np.exp(-eta * grad / grad_norm)
        a = a * update
        a = np.maximum(a, 1e-15)

    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:

            prob.solve(solver=s)

                _global_solver = s
                print(f"  LP solver: {s}")

        except Exception:

    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    dual_val = None
    try:
        dual_val = np.asarray(prob.constraints[0].dual_value, dtype=np.float64)
        if dual_val is not None:
            dual_val = np.maximum(dual_val, 0.0)
    except Exception:
        pass
    return b_val, dual_val


def nesterov_lp_v2(a0, time_limit, solver, verbose=True, use_adaptive_constriction=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0
    peak_weights = np.ones(2 * n - 1)  # For dual-informed weighting

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

            break

        # Adaptive constriction: start looser, tighten over time
        if use_adaptive_constriction:
            constriction = min(1.0 + 0.5 / (1.0 + t / 100.0), 1.5)
            M_eff = M_curr * constriction
        else:
            M_eff = M_curr

        b, dual = lp_best_response(y, M_eff, solver)
        if b is None:

        b_sum = b.sum()
        if b_sum < 1e-15:

        n_iters += 1

        # Track dual for peak identification

            decay = 0.95
            peak_weights = decay * peak_weights + (1 - decay) * (dual + 1e-10) / (dual.max() + 1e-10)

        # Step size with early boost
        alpha = 2.0 / (t + 2.0)
        if t < 30:
            alpha = min(alpha * 1.3, 1.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:

                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:

            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        n_avg += 1
        if n_avg % 5 == 0:

            s_avg = avg_check.sum()
            if s_avg > 1e-15:

def bregman_optimize(a0, time_limit, verbose=False):
    """Multiplicative Bregman updates targeting max autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    a = a + 1e-8  # Ensure strictly positive

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = 0.05
    decay = 0.997

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        k_star = int(np.argmax(b))

        # Subgradient: derivative of conv[k_star] w.r.t. a
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            j = k_star - i
            grad[i] += a[j]
            grad[j] += a[i]
            if i == j:
                grad[i] -= a[j]  # Avoid double counting diagonal

        # For non-diagonal terms, each pair (i, k_star-i) with i != k_star-i
        # contributes a[i]*a[k_star-i] + a[k_star-i]*a[i] = 2*a[i]*a[k_star-i]
        # Derivative w.r.t. a[i] is a[k_star-i], w.r.t. a[k_star-i] is a[i]
        # So the above is correct (each gets one copy)

        grad_norm = grad.max()
        if grad_norm < 1e-15:
            break

        # Multiplicative update
        update = np.exp(-eta * grad / grad_norm)
        a = a * update
        a = np.maximum(a, 1e-12)
        s = a.sum()
        if s < 1e-10:
            break
        a = a * norm / s

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        eta *= decay
        if eta < 1e-5:
            break

        if verbose and t % 500 == 0:
            print(f"    Bregman t={t}: best={best_score:.8f}, eta={eta:.6f}")

    return best_a, best_score


def peak_attacking_v2(a0, time_limit, peak_weights=None, seed=42):
    """Dual-informed peak-attacking coordinate descent."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()

    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]

        improved_this_round = False

        for k_peak in top_k:
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:5]]

            # Build badness mask: positions that contribute to any top peak
            peak_badness = np.zeros(n)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for pos in range(js2, je2 + 1):
                    other = k2 - pos
                    if 0 <= other < n:
                        peak_badness[pos] += a[other] * b[k2]

            for src in top_contrib_idx:
                if a[src] < 1e-10:
                    continue

                # Score destinations: low badness, low current value
                dst_score = -(peak_badness + 0.05 * a)
                dst_score[src] = -np.inf

                # Use peak_weights to bias toward less "important" positions
                if peak_weights is not None:
                    # Map peak_weights (size 2n-1) to position weights
                    pos_weight = np.zeros(n)
                    for i in range(n):
                        # Position i contributes to卷积 positions i (as a[i]*a[0]) to i+n-1
                        # Simplified: just use average contribution
                        lo = max(0, i)
                        hi = min(2 * n - 2, i + n - 1)
                        if hi >= lo:
                            pos_weight[i] = np.mean(peak_weights[lo:hi + 1])
                    dst_score -= 0.1 * pos_weight

                candidates = np.argsort(dst_score)[::-1]
                candidates = candidates[dst_score[candidates] > -np.inf][:max(3, n // 50)]

                for dst in candidates[:5]:
                    if dst == src:
                        continue
                    for frac in [0.03, 0.07, 0.015, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-12:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-14:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved_this_round = True
                            stall = 0
                            break
                    if improved_this_round:
                        break
                if improved_this_round:
                    break
            if improved_this_round:
                break

        if not improved_this_round:
            stall += 1
            # Random kicks to escape
            if stall % 5 == 0 or stall > 30:
                n_kicks = 30 if stall < 50 else 60
                for _ in range(n_kicks):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.005, 0.08)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break

            if stall > 300:
                break

    return best_a, best_score


def slope_descent(a0, time_limit, seed=42):
    """Slope-based descent: move mass along the steepest descent direction."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_star = int(np.argmax(b))

        # Compute full gradient of max (at k_star)
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            j = k_star - i
            grad[i] += a[j]
            if i != j:
                grad[j] += a[i]

        # Find direction of steepest decrease projected onto simplex
        # We want to decrease a where grad is high, increase where grad is low
        i_high = int(np.argmax(grad * a))  # Most expensive to keep
        i_low = int(np.argmin(grad))  # Cheapest to increase

        if a[i_high] < 1e-10 or i_high == i_low:
            i_high = int(np.argmax(a))
            i_low = int(np.argmin(a))

        # Line search along this direction
        improved = False
        for frac in [0.01, 0.005, 0.02, 0.002, 0.03]:
            amt = a[i_high] * frac
            a_trial = a.copy()
            a_trial[i_high] -= amt
            a_trial[i_low] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                continue
            a_trial = a_trial * norm / s_t
            score = fast_autocorr(a_trial).max()
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break

        if not improved:
            # Try random pairs
            found = False
            for _ in range(20):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.002, 0.04)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    found = True
                    break
            if not found:
                break

    return best_a, best_score


def structured_initialization(n, variant=0):
    """Create structured initializations with different properties."""
    if variant == 0:
        # Cosine bell
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif variant == 1:
        # Raised cosine with frequency
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 2:
        # Multi-frequency
        x = np.linspace(0, 1, n)
        a = 0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(10 * np.pi * x) + 0.1 * np.cos(20 * np.pi * x)
    elif variant == 3:
        # Gaussian mixture
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.2) ** 2) + np.exp(-20 * (x - 0.5) ** 2) + np.exp(-20 * (x - 0.8) ** 2)
    elif variant == 4:
        # Linear ramp (asymmetric)
        x = np.linspace(0, 1, n)
        a = 0.5 + x
    else:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** 0.5
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def ensemble_search(a0, time_limit, solver, n_variants=3):
    """Run multiple Nesterov trajectories from perturbed starts."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    rng = np.random.default_rng(999)
    best_a = a0.copy()
    best_score = fast_autocorr(a0).max()
    t0 = time.monotonic()

    per_variant_time = time_limit / n_variants

    for v in range(n_variants):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        variant_time = min(remaining / (n_variants - v), per_variant_time)
        if variant_time < 10:
            break

        # Create perturbed initialization
        if v == 0:
            init = a0.copy()
        elif v == 1:
            # Add small random noise
            noise = rng.normal(0, 0.01, n)
            init = a0 + noise
            init = np.maximum(init, 0)
            init = init * norm / init.sum()
        else:
            # Structured blend
            struct = structured_initialization(n, variant=v % 5)
            struct = struct * norm / struct.sum()
            init = 0.7 * a0 + 0.3 * struct
            init = np.maximum(init, 0)
            init = init * norm / init.sum()

        a_out, sc_out, _ = nesterov_lp_v2(init, variant_time, solver, verbose=False)
        if sc_out < best_score:
            best_score = sc_out
            best_a = a_out.copy()
            print(f"    Ensemble variant {v}: {sc_out:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main optimization function."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1755.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load previous best
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 0: Quick multi-length scan with LP-Nesterov ===
    print("\n=== Phase 0: Multi-length LP scan ===")
    scan_lengths = [1600, 2000, 2400]
    scan_results = {}
    scan_budget = time_budget * 0.20
    scan_t0 = time.monotonic()

    for n_scan in scan_lengths:
        if time.monotonic() - scan_t0 > scan_budget:
            break
        remaining_scan = scan_budget - (time.monotonic() - scan_t0)
        if remaining_scan < 15:
            break

        if best_seq is not None:
            init_seq = interpolate_sequence(best_seq, n_scan)
        else:
            init_seq = structured_initialization(n_scan, 0)

        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        per_scan = min(remaining_scan / max(1, len(scan_lengths) - scan_lengths.index(n_scan)), 100.0)
        a_out, sc_out, _ = nesterov_lp_v2(a_norm, per_scan, solver, verbose=False)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: max_b={sc_out:.8f}")

    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        do_update(scan_results[best_scan_n][0], f"scan_n{best_scan_n}")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_initialization(n_base, 0)
        best_score = evaluate_safe(best_seq)
    else:
        n_base = len(best_seq)

    # === Phase 1: Deep LP-Nesterov at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: Deep LP n={n_best} (budget={remaining * 0.35:.0f}s) ===")
    t1 = remaining * 0.35

    a_in = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _, peak_w = nesterov_lp_v2(a_in, t1, solver, verbose=True, use_adaptive_constriction=True)
    do_update(a_out, f"lp_deep_n{n_best}")

    # === Phase 2: Ensemble search (different trajectories) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Ensemble n={n_best} (budget={remaining * 0.15:.0f}s) ===")
    t2 = remaining * 0.15
    if t2 > 30:
        a_in = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = ensemble_search(a_in, t2, solver, n_variants=3)
        do_update(a_out, "ensemble")

    # === Phase 3: Bregman optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Bregman n={n_best} (budget={remaining * 0.12:.0f}s) ===")
    t3 = remaining * 0.12
    if t3 > 20:
        a_in = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = bregman_optimize(a_in, t3, verbose=False)
        do_update(a_out, f"bregman_n{n_best}")

    # === Phase 4: Peak-attacking with dual info ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Peak attack n={n_best} (budget={remaining * 0.18:.0f}s) ===")
    t4 = remaining * 0.18
    if t4 > 20:
        a_in = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_attacking_v2(a_in, t4, peak_weights=peak_w, seed=12345)
        do_update(a_out, f"peak_attack_n{n_best}")

    # === Phase 5: Try a longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_best + 400
    print(f"\n=== Phase 5: LP n={n_long} (budget={remaining * 0.15:.0f}s) ===")
    t5 = remaining * 0.15
    if t5 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _, _ = nesterov_lp_v2(a_norm, t5, solver, verbose=True)
        do_update(a_out, f"lp_long_n{n_long}")

    # === Phase 6: Slope descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Slope descent n={n_best} (budget={remaining * 0.10:.0f}s) ===")
    t6 = remaining * 0.10
    if t6 > 15:
        a_in = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = slope_descent(a_in, t6, seed=777)
        do_update(a_out, f"slope_n{n_best}")

    # === Phase 7: Final peak-attacking polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_best} (budget={remaining * 0.60:.0f}s) ===")
    t7 = remaining * 0.60
    if t7 > 10:
        a_in = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_attacking_v2(a_in, t7, peak_weights=None, seed=99999)
        do_update(a_out, "final_polish")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
