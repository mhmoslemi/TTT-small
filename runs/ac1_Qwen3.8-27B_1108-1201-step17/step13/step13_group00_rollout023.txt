Key insight for fast pair CD:
- We want to find pairs (i, j) and a delta > 0 such that moving delta from a[i] to a[j] reduces the max autocorrelation.
- The change in autocorrelation at lag k is: delta * (a[j+k-i] + a[j+i-k] - a[i+k-j] - a[i+j-k]) (with appropriate bounds)
- For the step to be beneficial, we need this to reduce the peak.

A faster approach:
1. Compute the autocorrelation and find the peak k*.
2. For the peak k*, the subgradient tells us which "direction" reduces it.
3. Specifically, we want to decrease a[i] for large subgradient[i] and increase a[j] for small subgradient[j].
4. Try the (i, j) pair with the largest "gap" in subgradient.

This is essentially what FW does, but with a "pair move" instead of a "full move to a vertex".

1. Compute conv and find peak k*
2. Compute subgradient at k*: grad[i] = 2*(a[k*-i] + a[i-k*])
3. Find i_max = argmax(grad) and j_min = argmin(grad)
4. Find optimal delta to move from i_max to j_min (binary search or closed form)
5. Apply the move
6. Repeat

This is O(n log n) per step (dominated by the FFT for the autocorrelation), and should be much faster than the previous pair CD.

```python
    N_grad = 1 << (3 * n - 1).bit_length()

        elapsed = time.monotonic() - t0

        if remaining < 2:
            break

        y = a.copy()

        # Allocate proportional time to each stage
        stage_time = remaining / (n_stages - si) * 0.9
        n_steps = int(min(stage_time * 20000, 200000))

            it += 1

            # Clip to prevent overflow

            if w_sum < 1e-30:

                w[int(np.argmax(conv))] = 1.0
            else:

            W_fft = rfft(w, N_grad)

            convolution = irfft(A_fft_g * W_fft, N_grad)[:n]
            grad = 2.0 * (cross_corr + convolution) / T

            if s < 1e-15:

            else:

            y = y_new
            a = a_new

                # Actually let's compute fresh

            if fresh_score < best_score:

    fresh_score = fast_autocorr(a).max()
    if fresh_score < best_score:
        best_score = fresh_score
        best_a = a.copy()
```

```python
# a[k-j] is valid when 0 <= k-j < n, i.e., j <= k < j+n
# a[j-k] is valid when 0 <= j-k < n, i.e., j-n+1 <= k <= j
# a[k-i] is valid when 0 <= k-i < n, i.e., i <= k < i+n
# a[i-k] is valid when 0 <= i-k < n, i.e., i-n+1 <= k <= i

k_vals = np.arange(size)
g_change = np.zeros(size)
# a[k-j]
mask1 = (k_vals >= j) & (k_vals - j < n)
g_change[mask1] += a[k_vals[mask1] - j]
# a[j-k]
mask2 = (k_vals <= j) & (j - k_vals < n)
g_change[mask2] += a[j - k_vals[mask2]]
# -a[k-i]
mask3 = (k_vals >= i) & (k_vals - i < n)
g_change[mask3] -= a[k_vals[mask3] - i]
# -a[i-k]
mask4 = (k_vals <= i) & (i - k_vals < n)
g_change[mask4] -= a[i - k_vals[mask4]]
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LSE gradient descent with Nesterov acceleration (O(n log n) per step via FFT)
2. Gradient-guided pair coordinate descent with vectorized delta search
3. Temperature annealing and multi-stage optimization
4. Multi-length exploration and perturbation-based restarts
Key improvement: O(n log n) gradient computation enables orders of magnitude more iterations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def project_nonneg_sum(a, target_sum):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target_sum / len(a)
    return a * target_sum / s


def lse_gradient_nesterov(a0, time_limit, T_start=30.0, T_end=0.2, n_stages=15, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    size = 2 * n - 1
    N_ac = 1 << (size - 1).bit_length()
    N_grad = 1 << (3 * n - 1).bit_length()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 2:
            break

        stage_time = remaining / (n_stages - si) * 0.85
        n_steps = int(min(stage_time * 25000, 500000))
        n_steps = max(n_steps, 100)

        y = a.copy()
        x_prev = a.copy()

        for it in range(1, n_steps + 1):
            if time.monotonic() - t0 > time_limit:
                break

            # Compute autocorrelation at y (Nesterov lookahead point)
            A_y = rfft(y, N_ac)
            conv = irfft(A_y * A_y, N_ac)[:size]
            max_conv = conv.max()
            if max_conv < 1e-14:
                break

            # LSE weights (softmax)
            shifted = (conv - max_conv) / T
            shifted = np.clip(shifted, -60.0, 60.0)
            exp_w = np.exp(shifted)
            w_sum = exp_w.sum()
            if w_sum < 1e-30:
                w = np.zeros(size)
                w[int(np.argmax(conv))] = 1.0
            else:
                w = exp_w / w_sum

            # Gradient via FFT
            W_fft = rfft(w, N_grad)
            A_y_fft = rfft(y, N_grad)
            cross_corr = irfft(W_fft * np.conj(A_y_fft), N_grad)[:n]
            convolution = irfft(A_y_fft * W_fft, N_grad)[:n]
            grad = 2.0 * (cross_corr + convolution) / T

            # Adaptive step size
            grad_norm = np.abs(grad).max()
            if grad_norm < 1e-15:
                break
            step = 2.0 / (grad_norm * 0.1 + T / max_conv)
            # Cap step to avoid divergence
            step = min(step, 1.0 / max(T / max_conv, 1e-10) * 0.5)

            # Nesterov update
            x_new = y - step * grad
            x_new = project_nonneg_sum(x_new, norm)

            theta = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
            momentum = (theta - 1.0) / theta
            y_new = x_new + momentum * (x_new - x_prev)
            y_new = project_nonneg_sum(y_new, norm)

            x_prev = x_new
            y = y_new
            a = x_new

            # Track best periodically
            if it % 500 == 0 or it == n_steps:
                score = fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        if verbose:
            fresh_score = fast_autocorr(a).max()
            if fresh_score < best_score:
                best_score = fresh_score
                best_a = a.copy()
            print(f"    LSE T={T:.3f}: best={best_score:.8f} ({time.monotonic()-t0:.0f}s, {n_steps} steps)")

    fresh_score = fast_autocorr(a).max()
    if fresh_score < best_score:
        best_score = fresh_score
        best_a = a.copy()
    return best_a, best_score


def gradient_guided_pair_cd(a0, time_limit, rng=None, verbose=True):
    n = len(a0)
    if rng is None:
        rng = np.random.default_rng(42)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    k_vals = np.arange(size)

    no_improve = 0
    total_moves = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_autocorr(a)
        peak_val = conv.max()
        k_star = int(np.argmax(conv))

        # Subgradient at peak
        grad = np.zeros(n)
        # grad[i] = 2 * (a[k_star - i] + a[i - k_star]) with bounds
        idx1 = k_star - np.arange(n)
        mask1 = (idx1 >= 0) & (idx1 < n)
        grad[mask1] += a[idx1[mask1]]
        idx2 = np.arange(n) - k_star
        mask2 = (idx2 >= 0) & (idx2 < n)
        grad[mask2] += a[idx2[mask2]]
        grad *= 2.0

        # Try several candidate pairs
        sorted_idx = np.argsort(-grad)
        candidates = []
        # Top argmax vs bottom argmin
        candidates.append((sorted_idx[0], sorted_idx[-1]))
        candidates.append((sorted_idx[0], sorted_idx[-2]))
        candidates.append((sorted_idx[1], sorted_idx[-1]))
        # Some random pairs
        for _ in range(4):
            ri = int(rng.integers(n))
            rj = int(rng.integers(n))
            if ri != rj:
                candidates.append((ri, rj))

        moved = False
        for (i_best, j_best) in candidates:
            if time.monotonic() - t0 > time_limit:
                break
            if i_best == j_best:
                continue
            ai = a[i_best]
            if ai < 1e-12:
                continue

            # Compute change vector: g_change[k] = a[k-j] + a[j-k] - a[k-i] - a[i-k]
            g_change = np.zeros(size)
            m1 = (k_vals >= j_best) & (k_vals - j_best < n)
            g_change[m1] += a[k_vals[m1] - j_best]
            m2 = (k_vals <= j_best) & (j_best - k_vals < n)
            g_change[m2] += a[j_best - k_vals[m2]]
            m3 = (k_vals >= i_best) & (k_vals - i_best < n)
            g_change[m3] -= a[k_vals[m3] - i_best]
            m4 = (k_vals <= i_best) & (i_best - k_vals < n)
            g_change[m4] -= a[i_best - k_vals[m4]]

            # Find best delta in (0, ai] minimizing max(conv + d*g_change)
            # Use ternary search on the convex function f(d) = max_k(conv[k] + d*g_change[k])
            lo, hi = 0.0, ai
            for _ in range(25):
                m1d = lo + (hi - lo) / 3.0
                m2d = hi - (hi - lo) / 3.0
                f1 = (conv + m1d * g_change).max()
                f2 = (conv + m2d * g_change).max()
                if f1 < f2:
                    hi = m2d
                else:
                    lo = m1d
            best_d = (lo + hi) / 2.0

            # Also check d=ai (boundary)
            f_boundary = (conv + ai * g_change).max()
            f_ternary = (conv + best_d * g_change).max()
            if f_boundary < f_ternary:
                best_d = ai

            if best_d > 1e-12:
                new_max = (conv + best_d * g_change).max()
                # Add second-order correction for the diagonal terms
                # At k=2*i: (ai-d)^2 = ai^2 - 2*ai*d + d^2
                if 2 * i_best < size:
                    corr_2i = conv[2 * i_best] - 2 * ai * best_d + best_d * best_d
                    new_max = max(new_max, corr_2i)
                # At k=2*j: (aj+d)^2 = aj^2 + 2*aj*d + d^2
                if 2 * j_best < size:
                    corr_2j = conv[2 * j_best] + 2 * a[j_best] * best_d + best_d * best_d
                    new_max = max(new_max, corr_2j)

                if new_max < peak_val - 1e-12:
                    a[i_best] -= best_d
                    a[j_best] += best_d
                    a = project_nonneg_sum(a, norm)
                    total_moves += 1
                    moved = True
                    # Verify with fresh computation
                    new_conv = fast_autocorr(a)
                    actual_max = new_conv.max()
                    if actual_max < best_score - 1e-14:
                        best_score = actual_max
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        # Revert
                        a[j_best] -= best_d
                        a[i_best] += best_d
                        a = project_nonneg_sum(a, norm)
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if not moved and no_improve > 200:
            # Random perturbation to escape local optimum
            n_perturb = max(1, n // 100)
            indices = rng.choice(n, size=n_perturb, replace=False)
            for idx in indices:
                a[idx] *= (1.0 + 0.05 * rng.standard_normal())
            a = project_nonneg_sum(a, norm)
            no_improve = 0

        if no_improve > 2000:
            break

        if verbose and total_moves % 5000 == 0:
            print(f"    PairCD: score={best_score:.8f} ({time.monotonic()-t0:.0f}s, {total_moves} moves)")

    return best_a, best_score


def fw_subgradient_fast(a0, time_limit, n_iters=500000, verbose=False):
    """Fast FW with subgradient - no LP needed."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    N_ac = 1 << (size - 1).bit_length()

    y = a.copy()
    x_prev = a.copy()

    for t in range(1, n_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Autocorrelation at y
        A_y = rfft(y, N_ac)
        conv = irfft(A_y * A_y, N_ac)[:size]
        k_star = int(np.argmax(conv))

        # Subgradient at k_star
        grad = np.zeros(n)
        idx1 = k_star - np.arange(n)
        mask1 = (idx1 >= 0) & (idx1 < n)
        grad[mask1] += y[idx1[mask1]]
        idx2 = np.arange(n) - k_star
        mask2 = (idx2 >= 0) & (idx2 < n)
        grad[mask2] += y[idx2[mask2]]
        grad *= 2.0

        # FW: move to vertex with minimum gradient
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm

        alpha = 2.0 / (t + 2.0)
        x_new = (1.0 - alpha) * y + alpha * v
        x_new = project_nonneg_sum(x_new, norm)

        theta = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (theta - 1.0) / theta
        y_new = x_new + momentum * (x_new - x_prev)
        y_new = project_nonneg_sum(y_new, norm)

        x_prev = x_new
        y = y_new
        a = x_new

        if t % 1000 == 0:
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    score = fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    return best_a, best_score


def structured_init(n, rng):
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 0.4 + 0.3 * np.sin(2 * np.pi * x) ** 2 + 0.2 * np.cos(4 * np.pi * x) ** 2
    else:
        x = np.linspace(0.0, 1.0, n)
        a = np.ones(n) * 0.5 + 0.3 * np.sin(np.pi * x) + 0.2 * np.sin(3 * np.pi * x) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()

        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

        do_try_update(a_out, f"lse_n{n3}")

    # Phase 7: Final polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.5, 200.0)
    if t7 > 10:
        # LSE with very low temperature
        t7a = t7 * 0.5
        print(f"\n=== Phase 7a: LSE low-T n={n_best} (budget={t7a:.0f}s) ===")
        a_out, _ = lse_gradient_nesterov(best_seq, t7a, T_start=5.0, T_end=0.05, n_stages=10, verbose=True)
        do_try_update(a_out, f"lse_lowT_n{n_best}")

        # Pair CD
        remaining = time_budget - (time.monotonic() - t_global)
        n_best = len(best_seq)
        t7b = min(remaining * 0.5, 150.0)
        if t7b > 5:
            print(f"\n=== Phase 7b: Final Pair CD n={n_best} (budget={t7b:.0f}s) ===")
            a_out, _ = gradient_guided_pair_cd(best_seq, t7b, rng, verbose=True)
            do_try_update(a_out, f"pcd_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
