def build_conv_matrix(a):
    """Build the constraint matrix for the LP."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, M, solver):
    """Solve the LP to find the best response."""
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def compute_lse_gradient(a, w):
    """Compute gradient of weighted autocorrelation."""
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_autocorr(a):

    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]

                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break
            elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                a = a_trial
                improved = True
                break

        if not improved:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.03, 0.15)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] = max(0.0, a[i] - amt)
                a_trial[j] += amt
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                    elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                        a = a_trial

            # Periodic larger perturbation
            if rng.random() < 0.01:
                idx_pert = rng.choice(n, size=max(1, n // 100), replace=False)
                a_trial = a.copy()
                for idx in idx_pert:
                    j2 = rng.integers(n)
                    amt2 = a[idx] * rng.uniform(0.1, 0.5)
                    a_trial[idx] -= amt2
                    a_trial[j2] += amt2
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score * 1.005:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    # Fallback
    if best_seq is None or not np.isfinite(best_score):
        n_init = 1600
        x = np.linspace(0.0, 1.0, n_init)
        best_seq = np.sin(np.pi * x) ** 2.0
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq)

    # Phase 0: Quick short-length optimization for diverse initialization
    # Optimize at half length to find a potentially different basin
    n_short = max(400, n_base // 2)
    remaining = time_budget - (time.monotonic() - t_global)
    t0_phase = min(remaining * 0.06, 90.0)
    if t0_phase > 20:
        print(f"\n=== Phase 0: Short-length LP n={n_short} (budget={t0_phase:.0f}s) ===")
        init_short = interpolate_sequence(best_seq, n_short)
        a_norm = init_short * np.sqrt(2.0 * n_short) / init_short.sum()
        a_out, sc_short = nesterov_lp_optimize(a_norm, t0_phase, solver, verbose=False)
        # Interpolate back and use as alternate initialization
        if a_out is not None:
            a_interp = interpolate_sequence(a_out, n_base)
            a_norm = a_interp * np.sqrt(2.0 * n_base) / a_interp.sum()
            # Quick LSE to settle
            t_settle = min(30.0, time_budget - (time.monotonic() - t_global))
            if t_settle > 10:
                a_settled, _ = lse_gradient_descent(a_norm, t_settle, verbose=False)
                do_try_update(a_settled, f"short_interp_n{n_base}")

    # Phase 1: Main LP Nesterov at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.32, 520.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_base}")

    # Phase 2: LSE gradient descent
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LSE-GD n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t2, verbose=True)
        do_try_update(a_out, f"lse_n{n_cur}")

    # Phase 3: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: PeakShave n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.08, 100.0)

    t4 = min(remaining * 0.18, 270.0)

        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n2}")
        # Polish
        remaining = time_budget - (time.monotonic() - t_global)
        t4b = min(remaining * 0.06, 60.0)
        if t4b > 10:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = lse_gradient_descent(a_norm, t4b, verbose=False)
            do_try_update(a_out, f"lse_polish_n{n_cur}")

    # Phase 5: Even longer
    n3 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP Nesterov n={n3} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 170.0)

        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n3}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final PeakShave n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 130.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t6, seed=555, verbose=True)

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LSE n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.10, 100.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t7, verbose=True)

    remaining = time_budget - (time.monotonic() - t_global)
