import time
from scipy.fft import rfft, irfft

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out

def _lse_grad(y, T, n):
    """Gradient of LSE-smoothed max autocorrelation."""
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def _lse_value(y, T, n):
    """LSE-smoothed max autocorrelation value."""
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    return T * np.log(np.sum(np.exp(shifted)))

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def _lse_grad(y, T, n):
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def _lse_fista(a0, T_schedule, iters_per_stage, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    L_init = n / 50.0

    for stage_i, T in enumerate(T_schedule):
        L = max(L_init / (T * 5.0), n / (T * 3.0))
        t_fista = 1
        staging = a.copy()
        max_iters = int(iters_per_stage * (T_schedule[0] / T) ** 0.3)
        max_iters = max(max_iters, 3000)

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score

            grad = _lse_grad(y, T, n)
            step = 1.0 / L
            z = y - step * grad
            a_new = _proj_simplex(z)

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()

            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1

            if it % 200 == 0:
                sc = _fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = _fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA stage={stage_i} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

def _build_conv_matrix(a, n):
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j: j + n, j] = a
    return T


def _lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def _peak_shave(a, time_limit, seed=42):
    """Targeted peak reduction via mass redistribution with multiple step sizes."""
    n = len(a)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2: je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            if len(candidates) == 0:
                break

            candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 5)]
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.0005, 0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = _fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            for _ in range(3):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-10:
                    frac = rng.uniform(0.003, 0.05)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = _fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0
                                break

            if stall > 300:
                break

    return best_a, best_score


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def _subspace_polish(a, time_limit, solver, top_pairs=50):
    """Optimize only the top-contributing variables for each peak."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, _fast_autocorr(a).max()
    a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:5]
        active_indices = set()
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            top_local = idx[np.argsort(contrib)[::-1]][:10]
            active_indices.update(top_local.tolist())

        if len(active_indices) == 0:
            break
        active_list = sorted(active_indices)
        a_sub = a[active_list].copy()

        # Try small perturbations in the active subspace
        rng = np.random.default_rng(int(time.monotonic() * 1000))
        improved = False
        for _ in range(200):
            a_trial = a.copy()
            direction = rng.standard_normal(len(active_list))
            step = rng.uniform(0.001, 0.02) * a[active_list].std()
            a_trial[active_list] = a_sub + step * direction
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t
            score = _fast_autocorr(a_trial).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_trial.copy()
                a = a_trial.copy()
                improved = True
                break

        if not improved:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: LP-centric with adaptive perturbation, multi-length, peak shaving."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 100.0, 1680.0)
    rng = np.random.default_rng(seed)

    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = _evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = _evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1200
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.2
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LSE-FISTA warm start (fast) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    if t1 > 10:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        T_sched = [15.0, 4.0, 1.0, 0.3, 0.08]
        a_out, _ = _lse_fista(best_seq, T_sched, 10000, t1, verbose=True)
        _try_update(a_out, f"lse_n{n_base}")

    # === Phase 2: LP-Nesterov with adaptive perturbation (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LP-Nest+Perturb n={n_cur} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _lp_nesterov_adaptive(a_norm, t2, solver, verbose=True,
                                          perturb_every=30, perturb_scale=0.08)
        _try_update(a_out, f"lp_n{n_cur}")

    # === Phase 3: Perturb + short LP (escape local minima) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 60:
        t3 = min(remaining * 0.12, 120.0)
        if t3 > 20:
            print(f"\n=== Phase 3: Perturb + LP n={n_cur} (budget={t3:.0f}s) ===")
            a_base = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            best_pert_score = _evaluate_safe(a_base)
            for pert_seed in [111, 222, 333]:
                t_remaining = time_budget - (time.monotonic() - t_global)
                if t_remaining < 30:
                    break
                rng_p = np.random.default_rng(pert_seed)
                noise_scale = 0.15
                noise = rng_p.standard_normal(n_cur) * noise_scale
                a_pert = a_base + noise
                a_pert = np.maximum(a_pert, 0.0)
                s_p = a_pert.sum()
                if s_p < 1e-10:
                    continue
                a_pert = a_pert * np.sqrt(2.0 * n_cur) / s_p
                t_sub = min(t_remaining * 0.3, 40.0)
                if t_sub < 10:
                    break
                a_out, _ = _lp_nesterov_adaptive(a_pert, t_sub, solver, verbose=False,
                                                  perturb_every=50, perturb_scale=0.05)
                _try_update(a_out, f"perturb_{pert_seed}_n{n_cur}")

    # === Phase 4: Expand length + optimize ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 80:
        n2 = n_cur + 400
        t4 = min(remaining * 0.20, 300.0)
        if t4 > 30:
            print(f"\n=== Phase 4: Expand to n={n2} (budget={t4:.0f}s) ===")
            init_long = _interpolate_sequence(best_seq, n2)
            # Quick LSE warm start at new length
            t4a = min(t4 * 0.3, 60.0)
            if t4a > 10:
                a_warm, _ = _lse_fista(init_long, [5.0, 1.5, 0.4], 5000, t4a, verbose=False)
                _try_update(a_warm, f"lse_expand_n{n2}")
            # LP at new length
            t4b = min(t4 - t4a, 250.0)
            if t4b > 20:
                n_best = len(best_seq)
                a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
                if n_best != n2:
                    a_norm = _interpolate_sequence(best_seq, n2) * np.sqrt(2.0 * n2) / n2
                a_out, _ = _lp_nesterov_adaptive(a_norm, t4b, solver, verbose=True,
                                                  perturb_every=30, perturb_scale=0.06)
                _try_update(a_out, f"lp_expand_n{n2}")

    # === Phase 5: LP polish at current best (with subspace) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        t5 = min(remaining * 0.20, 250.0)
        if t5 > 20:
            print(f"\n=== Phase 5: LP polish n={n_best} (budget={t5:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = _lp_nesterov_adaptive(a_norm, t5 * 0.7, solver, verbose=True,
                                              perturb_every=40, perturb_scale=0.03)
            _try_update(a_out, f"lp_polish_n{n_best}")
            # Subspace polish
            t5b = t5 * 0.3
            if t5b > 10:
                a_out2, _ = _subspace_polish(a_out if a_out is not None else a_norm, t5b, solver)
                _try_update(a_out2, f"subspace_n{n_best}")

    # === Phase 6: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t6 = min(remaining * 0.25, 120.0)
        if t6 > 10:
            print(f"\n=== Phase 6: Peak shave n={n_best} (budget={t6:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = _peak_shave(a_norm, t6, seed=99999)
            _try_update(a_out, "peak_shave")

    # === Phase 7: Final LP micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t7 = min(remaining * 0.30, 150.0)
        if t7 > 15:
            print(f"\n=== Phase 7: Final LP n={n_best} (budget={t7:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = _lp_nesterov_adaptive(a_norm, t7, solver, verbose=True,
                                              perturb_every=60, perturb_scale=0.02)
            _try_update(a_out, "lp_final")

    # === Phase 8: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 10:
        t8 = min(remaining * 0.5, 60.0)
        if t8 > 5:
            print(f"\n=== Phase 8: Final peak shave n={n_best} (budget={t8:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = _peak_shave(a_norm, t8, seed=77777)
            _try_update(a_out, "peak_shave_final")

    # Final verification
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
