import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 0:
            result = result / s * target_sum
    return result


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b)_k = (a_fixed*b)_k."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Iteratively flatten power spectrum."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = internal_score(a)
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    return best_a


def compute_smooth_subgrad(a, T_smooth=10.0):
    """Compute smoothed subgradient of max_k (a*a)_k."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T_smooth * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient: grad_j = 2 * sum_k w_k * a_{k-j}
    fft_size = 1 << (3 * n - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w

def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving with quadratic approximation."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / max(s, 1e-15)
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:15]
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            # Find the contributor with largest product
            src_local = np.argmax(contrib)
            src = idx[src_local]
            
            # Find destination (low-contribution area)
            peak_areas = set()
            for k2 in top_k[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))
            
            candidates = [j for j in range(n) if j not in peak_areas and a[j] < np.median(a)]
            if not candidates:
                candidates = [j for j in range(n) if j != src and a[j] < np.median(a)]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            
            dst = candidates[rng.integers(len(candidates))]
            
            # Try multiple transfer fractions
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 300:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

Now let me think about potential issues with the Nesterov subgradient method:
- The learning rate needs to be chosen carefully. Too large → divergence, too small → slow.
- The temperature T_smooth controls the smoothness. High T → smooth (all peaks weighted equally), low T → sharp (only max peak matters).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Spectral shaping for fast initialization
2. Nesterov-accelerated projective subgradient (O(n log n) per iter, fast)
3. Nesterov-accelerated LP with adaptive active-set reduction (accurate polishing)
4. Peak shaving coordinate descent (final refinement)
Key improvements: active-set LP for 10-50x faster solves, Nesterov subgradient
as a fast pre-optimizer, and more aggressive peak shaving.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0} using bisection on threshold."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_mask = v_sorted * np.arange(1, n + 1) > (cssv - target_sum)
    rho_idx = np.nonzero(rho_mask)[0]
    if len(rho_idx) == 0:
        return np.full(n, target_sum / n)
    rho = rho_idx[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 0:
            result = result / s * target_sum
    return result


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b)_k = (a_fixed*b)_k."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    return best_a


def compute_smooth_subgrad(a, T_smooth):
    """Compute smoothed subgradient of max_k (a*a)_k using softmax weights."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T_smooth * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    # grad_i = 2 * sum_m w_{i+m} * a_m = 2 * (a_rev * w)[n-1 + i]
    fft_size = 1 << (3 * n - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def nesterov_subgrad(a0, time_limit, T_start=80.0, T_end=0.5, n_stages=12):
    """Nesterov-accelerated projective subgradient with LSE smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / max(s, 1e-15)
    best_a = a.copy()
    best_score = internal_score(a)

    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T_sm in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Adaptive iterations per stage
        stage_iters = max(50, int(remaining * 40.0 / (n_stages - si)))

        a_prev = a.copy()
        avg_a = np.zeros(n)
        n_avg = 0

        for t in range(1, stage_iters + 1):
            if time.monotonic() - t0 > time_limit:
                break

            # Nesterov extrapolation: y = a + beta_t * (a - a_prev)
            if t == 1:
                y = a.copy()
            else:
                beta_t = (t - 1) / (t + 2.0)
                y = a + beta_t * (a - a_prev)
                y = np.maximum(y, 0)
                s_y = y.sum()
                if s_y > 1e-15:
                    y = y / s_y
                else:
                    y = a.copy()

            # Compute subgradient at y
            grad, w = compute_smooth_subgrad(y, T_sm)

            # Adaptive learning rate
            gnorm = np.sqrt(np.dot(grad, grad))
            if gnorm < 1e-15:
                gnorm = 1e-15
            eta_t = 1.0 / (np.sqrt(t + 1.0) * gnorm)
            eta_t = min(eta_t, 0.05)

            # Projected update
            a_new = y - eta_t * grad
            a_new = project_simplex(a_new)

            a_prev = a
            a = a_new

            # Track best
            score = internal_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()

            # Time average
            avg_a += a
            n_avg += 1
            if n_avg % 100 == 0:
                avg_check = avg_a / n_avg
                score_avg = internal_score(avg_check)
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        # Reset to best for next stage
        a = best_a.copy()
        if si % 4 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad stage {si+1}/{n_stages} (T={T_sm:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

        s = y_new.sum()
        if s < 1e-15:

        else:

            rng = np.random.default_rng(t)

            y = np.maximum(y, 0)
            y = y / max(y.sum(), 1e-15)
            stall_count = 0

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NL t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving with adaptive step sizes."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / max(s, 1e-15)
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:12]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Build set of "hot" indices (near peaks)
            hot_set = set()
            for k2 in top_k[:6]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                hot_set.update(range(js2, je2 + 1))

            # Destination: prefer cold, low-valued positions
            med_val = np.median(a)
            candidates = [j for j in range(n) if j not in hot_set and a[j] < med_val]
            if not candidates:
                candidates = [j for j in range(n) if j != src and a[j] < med_val * 2]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            if not candidates:
                break

            dst = candidates[rng.integers(len(candidates))]

            # Try progressively larger transfers
            for frac in [0.001, 0.003, 0.005, 0.008, 0.012, 0.02, 0.03]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.001, 0.008)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 400:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
