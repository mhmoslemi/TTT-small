- **Trigonometric family**: a[j] = 1 + sum_k c_k * cos(2πk*j/(n-1)). 
- **Spline family**: Piecewise polynomial.

The polynomial family is interesting because:
- For a[j] = (j+1)*(n-j)/(n*(n+1)/2) (triangular), the convolution is cubic.
- For a[j] = (1 - (2j/(n-1) - 1)^2) (parabolic), we get a specific convolution shape.

The advantage of parametric families is that we reduce the dimensionality, making optimization easier. The disadvantage is that we might miss the optimal solution.

A hybrid approach: use the parametric family for a rough optimization, then use the full coordinate optimization for polishing.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_z_objective(z, T):
    """Objective in z-space (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    c = fast_autocorr(a)
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    lse = (Tc_max + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()
    
    # grad_a[j] = 2 * sum_{k: valid} w[k] * a[k-j]
    grad_a = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-15:
            continue
        js = max(0, k - (n - 1))
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * wk * a[k - idx]
    
    # grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    obj = n * lse  # Since objective is 2*n*max_c, and LSE approximates max_c, obj ≈ 2*n*lse
    # Actually the true objective is 2*n*max_c/(sum_a)^2. With sum_a=1 (softmax), it's 2*n*max_c.
    # LSE approximates max_c, so obj ≈ 2*n*lse.
    return 2.0 * n * lse, grad_z


def lbfgs_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=20):
    """
    L-BFGS optimization on the LSE approximation with multi-stage temperature annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        z = np.log(np.maximum(a, 1e-30))
        
        # Define objective and gradient for L-BFGS
        def fun_and_grad(z_vec):
            return lse_z_objective(z_vec, T)
        
        try:
            result = minimize(
                fun_and_grad,
                z,
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': max(20, int(remaining * 3)), 'maxfun': max(50, int(remaining * 5)), 'gtol': 1e-8}
            )
            if result.success or result.fun < best_score:
                a_new = softmax_fn(result.x)
                score = 2.0 * n * fast_autocorr(a_new).max()
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                    a = a_new.copy()
                else:
                    a = a_new.copy()  # Still use the LSE-optimal point as starting point
        except Exception as e:
            print(f"      L-BFGS stage {si} error: {e}")
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE-LBFGS stage {si}/{n_stages} (T={T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def gradient_descent_lse(a0, time_limit, T_start=30.0, T_end=0.3, n_stages=15, lr=0.01):
    """
    Plain gradient descent with momentum on LSE approximation.
    Faster per iteration than L-BFGS for large n.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        z = np.log(np.maximum(a, 1e-30))
        v = np.zeros(n)  # momentum
        beta = 0.9
        current_lr = lr * (0.7 ** si)
        
        max_iters = max(50, int(remaining * 10))
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit - 2:
                break
            
            obj, grad_z = lse_z_objective(z, T)
            v = beta * v + (1 - beta) * grad_z
            z = z - current_lr * v
            
            if (it + 1) % 20 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()
        
        a_stage = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a_stage).max()
        if score < best_score:
            best_score = score
            best_a = a_stage.copy()
            a = a_stage.copy()
    
    return best_a, best_score


def build_toeplitz(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP, cp.MOSEK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Constraint pool: list of (y_snapshot, toeplitz) to reuse
    constraint_pool = []
    MAX_POOL = 5
    
    avg_a = np.zeros(n)
    n_avg = 0
    stall = 0
    warm_b = None
    last_print = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Build Toeplitz from current y
        T = build_toeplitz(y)
        
        # Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0
        b_var = cp.Variable(n, nonneg=True)
        if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
            b_var = cp.Variable(n, nonneg=True, value=warm_b)
        M_var = cp.Variable(value=1.0 / n)
        
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        solved = False
        for sol in [solver, cp.SCS, cp.ECOS, cp.CLARABEL]:
            try:
                prob.solve(solver=sol)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved:
            warm_b = None
            if t % 100 == 0:
                print(f"      LP failed at iter {t}")
            continue
        
        b_star = np.asarray(b_var.value, dtype=np.float64)
        b_star = np.maximum(b_star, 0)
        s_b = b_star.sum()
        if s_b < 1e-15:
            continue
        b_star = b_star / s_b
        warm_b = b_star
        
        M_val = float(M_var.value)
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new
        
        y = y_new
        a = a_new
        
        # Track averages
        avg_a += a
        n_avg += 1
        
        # Check scores
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        
        if n_avg % 20 == 0 and n_avg >= 40:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall = 0
        
        # Restart if stalled
        if stall > 300 and t > 50:
            a = best_a.copy()
            y = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            stall = 0
            warm_b = None
            # Add small perturbation
            rng = np.random.default_rng(t)
            noise = rng.normal(0, 0.005, n)
            y = y + noise * y
            y = np.maximum(y, 0)
            y = y / y.sum()
        
        if t - last_print > 1000:
            elapsed = time.monotonic() - t0
            print(f"      LP-v2 t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")
            last_print = t
    
    return best_a, best_score


def polynomial_param_optimize(a0, time_limit):
    """
    Fit a polynomial to the best sequence and optimize the polynomial coefficients
    using L-BFGS on the LSE objective.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    t0 = time.monotonic()
    
    # Try degrees 3, 4, 5, 6
    for degree in [3, 4, 5]:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Initial polynomial coefficients from fitting
        x = np.linspace(0, 1, n)
        coeffs = np.polyfit(x, a, degree)
        
        def poly_obj(coeffs_vec):
            # coeffs_vec contains the polynomial coefficients
            poly = np.poly1d(coeffs_vec)
            a_poly = poly(x)
            a_poly = np.maximum(a_poly, 0)
            s_poly = a_poly.sum()
            if s_poly < 1e-10:
                return 1e10, np.zeros(degree + 1)
            a_poly = a_poly / s_poly
            # LSE with moderate temperature
            T = 20.0
            c = fast_autocorr(a_poly)
            Tc = T * c
            Tc_max = Tc.max()
            exp_Tc = np.exp(Tc - Tc_max)
            lse = (Tc_max + np.log(exp_Tc.sum())) / T
            obj = 2.0 * n * lse
            return obj
        
        try:
            result = minimize(poly_obj, coeffs, method='Nelder-Mead',
                            options={'maxiter': 5000, 'xatol': 1e-8, 'fatol': 1e-10})
            if result.fun < 1e10:
                poly = np.poly1d(result.x)
                a_poly = poly(x)
                a_poly = np.maximum(a_poly, 0)
                s_poly = a_poly.sum()
                if s_poly > 1e-10:
                    a_poly = a_poly / s_poly
                    score = 2.0 * n * fast_autocorr(a_poly).max()
                    print(f"      Poly degree {degree}: score={score:.6f}")
                    if score < 2.0:
                        return a_poly, score
        except Exception as e:
            print(f"      Poly degree {degree} error: {e}")
    
    return a, 2.0 * n * fast_autocorr(a).max()


def coordinate_descent_polish(a0, time_limit, seed=42):
    """
    Coordinate descent: optimize pairs of coordinates at a time.
    For a fixed pair (i, j) with a[i]=p, a[j]=q, the convolution values
    that depend on them are: c[k] for k in {i, i+1, ..., i+j, j+i, ...} 
    Actually, changing a[i] and a[j] affects c[k] for all k where i or j appears.
    
    Simpler approach: move mass between two coordinates and check if max decreases.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 3:
        c = fast_autocorr(a)
        cmax = c.max()
        
        # Find top peaks
        top_peaks = np.argsort(c)[::-1][:20]
        
        # Build danger set
        danger = np.zeros(n, dtype=bool)
        for k in top_peaks[:10]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            danger[js:je + 1] = True
        
        # Try moving mass from dangerous to non-dangerous positions
        improved = False
        danger_indices = np.where(danger)[0]
        safe_indices = np.where(~danger)[0]
        
        if len(danger_indices) == 0 or len(secure_indices) == 0:
            break
        
        # Sort dangerous by value (highest first), safe by value (lowest first)
        danger_sorted = danger_indices[np.argsort(a[danger_indices])[::-1]]
        safe_sorted = safe_indices[np.argsort(a[safe_indices])]
        
        for src in danger_sorted[:20]:
            for dst in safe_sorted[:10]:
                if src == dst:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < cmax * (1 - 1e-10):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall % 20 == 0:
                # Random perturbation
                n_pert = rng.integers(2, 8)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.02)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 300:
                break
    
    return best_a, best_score
