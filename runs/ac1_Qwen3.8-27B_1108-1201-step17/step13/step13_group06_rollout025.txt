import numpy as np

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def build_toeplitz(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP, cp.MOSEK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(y, T, solver, warm_b=None):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(y)
    b_val = None
    M_val = None
    
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver] + [s for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP] if s != solver]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_out = np.asarray(b_var.value, dtype=np.float64)
    b_out = np.maximum(b_out, 0)
    s_sum = b_out.sum()
    if s_sum < 1e-15:
        return None, None
    b_out = b_out / s_sum
    return b_out, float(M_var.value)

def peak_equalize(a0, time_limit, m_peaks=5, seed=42):
    """
    Peak equalization: try to reduce the maximum autocorrelation peak
    by moving mass from contributing positions to non-contributing ones.
    Uses gradient information from the active peaks.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 3:
        c = fast_autocorr(a)
        cmax = c.max()
        
        # Find top m peaks
        top_idx = np.argsort(c)[::-1][:m_peaks]
        top_vals = c[top_idx]
        
        # Compute gradient of each peak with respect to a
        # dc[k]/da[i] = a[k-i] (if 0 <= k-i < n)
        # For the combined objective max(c[top]), use the max peak's gradient
        # or a weighted combination
        
        # Weighted gradient (emphasize the highest peak)
        weights = np.exp(top_vals / (cmax * 0.1))  # softmax-like weighting
        weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for wi, ki in zip(weights, top_idx):
            ki = int(ki)
            for i in range(n):
                j = ki - i
                if 0 <= j < n:
                    grad[i] += wi * a[j]
        
        # We want to move mass from high-grad positions to low-grad positions
        # This reduces the weighted peak
        # Sort by gradient
        order = np.argsort(grad)
        
        # Try moving mass from high-grad to low-grad positions
        improved = False
        n_high = max(3, n // 20)
        n_low = max(3, n // 20)
        high_positions = order[-n_high:]  # highest gradient
        low_positions = order[:n_low]       # lowest gradient
        
        for src in high_positions[:5]:
            for dst in low_positions[:5]:
                if src == dst:
                    continue
                if a[src] < 1e-12:
                    continue
                
                # Try different step sizes
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_cmax = fast_autocorr(a_trial).max()
                    if new_cmax < cmax * (1 - 1e-10):
                        a = a_trial
                        new_score = 2.0 * n * new_cmax
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            if stall % 20 == 0:
                a_trial = a.copy()
                n_pert = rng.integers(2, 6)
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.02)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 300:
                break
    
    return best_a, best_score


def spectral_refine(a0, time_limit, alpha=0.05, seed=123):
    """
    Spectral domain refinement: flatten the power spectrum in frequency domain,
    then project back to non-negative simplex.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    current_alpha = alpha
    while time.monotonic() - t0 < time_limit - 3:
        # Compute autocorrelation
        c = fast_autocorr(a)
        cmax = c.max()
        
        # Compute DFT (zero-padded for linear convolution)
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        
        # Flatten the spectrum: reduce high-power bins
        # P_new = P / (P + alpha * P.max())
        P_flat = P / (P + current_alpha * P.max())
        
        # Reconstruct maintaining phase
        phase_A = np.angle(A)
        A_new = np.sqrt(P_flat) * np.exp(1j * phase_A)
        A_new[0] = np.abs(A_new[0])  # Keep DC positive
        
        # Inverse DFT
        a_new = irfft(A_new, fft_size)[:n]
        
        # Project to non-negative, sum=1
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-12:
            # Try with less alpha
            current_alpha *= 0.5
            if current_alpha < 1e-6:
                break
            continue
        a_new = a_new / sn
        
        # Evaluate
        new_cmax = fast_autocorr(a_new).max()
        new_score = 2.0 * n * new_cmax
        
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Reduce alpha gradually
            current_alpha *= 0.95
        else:
            # Try a blend
            for t_blend in [0.1, 0.05, 0.02]:
                a_blend = (1 - t_blend) * a + t_blend * a_new
                a_blend = np.maximum(a_blend, 0)
                sb = a_blend.sum()
                if sb < 1e-12:
                    continue
                a_blend = a_blend / sb
                sb_score = 2.0 * n * fast_autocorr(a_blend).max()
                if sb_score < best_score:
                    best_score = sb_score
                    best_a = a_blend.copy()
                    a = a_blend
                    stall = 0
                    break
            else:
                stall += 1
                current_alpha *= 0.7
                if current_alpha < 1e-8:
                    # Random restart
                    noise = rng.normal(0, 0.005, n)
                    a = a * (1 + noise)
                    a = np.maximum(a, 0)
                    sn = a.sum()
                    if sn > 1e-10:
                        a = a / sn
                    current_alpha = alpha
            
            if stall > 200:
                break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def u_shape_init(n, sharpness=2.0):
    """Create a U-shaped initialization (high at edges, low in center)."""
    x = np.linspace(0, 1, n)
    a = np.exp(-sharpness * np.sin(np.pi * x) ** 2)
    # This is high at x=0 and x=1, low at x=0.5
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def edge_concentrated_init(n, width=0.1):
    """Concentrate mass at the edges."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0) / width) ** 2) + np.exp(-((x - 1) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def bimodal_init(n, center1=0.2, center2=0.8, width=0.05):
    """Two Gaussian bumps."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center1) / width) ** 2) + np.exp(-((x - center2) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    
    # U-shape with different sharpness
    for sh in [1.0, 2.0, 4.0, 8.0]:
        inits.append(u_shape_init(n, sh))
    
    # Edge concentrated
    for w in [0.05, 0.1, 0.2, 0.3]:
        inits.append(edge_concentrated_init(n, w))
    
    # Bimodal
    for c1, c2, w in [(0.2, 0.8, 0.05), (0.15, 0.85, 0.1), (0.3, 0.7, 0.15)]:
        inits.append(bimodal_init(n, c1, c2, w))
    
    # Cosine-based
    x = np.linspace(0, 1, n)
    for freq in [2, 3, 4]:
        a = 0.5 + 0.5 * np.cos(freq * np.pi * x)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            inits.append(a / s)
    
    # Power law at edges
    x = np.linspace(0, 1, n)
    for p in [1.0, 2.0, 4.0]:
        a = x ** p + (1 - x) ** p
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            inits.append(a / s)
    
    return inits
