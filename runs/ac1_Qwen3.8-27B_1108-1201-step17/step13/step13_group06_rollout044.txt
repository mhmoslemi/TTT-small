def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def build_toeplitz(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(y, T, solver, warm_b=None):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    n = len(y)
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.CLARABEL, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def lse_objective_grad(a, T_temp):
    """Compute logsumexp objective and gradient for a."""
    n = len(a)
    c = fast_autocorr(a)
    max_c = c.max()
    exp_c = np.exp((c - max_c) / T_temp)
    lse_val = max_c + T_temp * np.log(exp_c.sum())
    w = exp_c / exp_c.sum()

    grad_a = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-15:
            continue
        js = max(0, k - (n - 1))
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * wk * a[k - idx]

    return lse_val, grad_a


def lse_gradient_descent(a0, time_limit, T_start=200.0, T_end=0.1, n_stages=40):
    """Full LSE gradient descent with multi-stage annealing and Adam."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        base_lr = 5e-3 * min(1.0, 80.0 / n)
        lr = base_lr * (0.9 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_it = max(30, int(min(remaining, 80.0) * 3))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 5:
                break

            a = softmax_fn(z)
            _, grad_a = lse_objective_grad(a, stage_T)

            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 8 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    if (it + 1) % 40 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))
                        m_z = np.zeros(n)
                        v_z = np.zeros(n)
                        t_adam = 0

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_directed_search(a0, time_limit, seed=42):
    """
    Peak-directed stochastic local search.
    Moves mass from positions contributing to peaks toward safer positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        thr = bmax * 0.985
        top_peaks = np.where(b >= thr)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:10]

        # Build importance map
        importance = np.zeros(n)
        for k in top_peaks[:20]:
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            importance[idx] += a[idx] * a[k - idx]

        imax = importance.max()
        if imax < 1e-15:
            break
        importance = importance / imax

        # Find hot (high importance) and cold (low importance) positions
        hot_thresh = importance > 0.3
        hot_idx = np.where(hot_thresh)[0]
        cold_idx = np.where(~hot_thresh)[0]

        if len(hot_idx) == 0 or len(cold_idx) == 0:
            stall += 1
            if stall > 200:
                break
            continue

        # Try multiple moves
        found_improvement = False
        n_tries = min(50, len(hot_idx) * len(cold_idx) // 10)
        for _ in range(n_tries):
            src = hot_idx[rng.integers(len(hot_idx))]
            dst = cold_idx[rng.integers(len(cold_idx))]
            if src == dst:
                continue
            frac = rng.uniform(0.001, 0.05)
            amt = a[src] * frac
            if amt < 1e-12 or a[src] < 1e-10:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-10:
                continue
            a_trial = a_trial / st
            new_max = fast_autocorr(a_trial).max()
            if new_max < bmax * (1 - 1e-7):
                a = a_trial
                new_score = 2.0 * n * new_max
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                found_improvement = True
                stall = 0
                break

        if not found_improvement:
            stall += 1
            if stall % 200 == 0 and stall > 0:
                # Random perturbation
                n_pert = rng.integers(1, 5)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.04)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 1000:
                break

        iter_count += 1
        if iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PSD iter={iter_count} ({elapsed:.0f}s): best={best_score:.6f}, stall={stall}")

    return best_a, best_score


def coordinate_polish(a0, time_limit):
    """
    Coordinate-wise optimization: for each position, find optimal value
    by binary search on the 1D problem.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Precompute base convolution (with a[i]=0 for each i is expensive,
    # so we use a different approach: try small perturbations)
    order = np.random.default_rng(123).permutation(n)

    for i in order:
        if time.monotonic() - t0 > time_limit - 2:
            break

        a_save = a[i]
        if a_save < 1e-12:
            continue

        # Binary search for best a[i] in [0, a_save * 3]
        lo = 0.0
        hi = a_save * 3.0
        best_val = a_save
        best_local = best_score

        for _ in range(12):
            mid = (lo + hi) / 2.0
            a_trial = a.copy()
            a_trial[i] = mid
            st = a_trial.sum()
            if st < 1e-10:
                lo = mid
                continue
            a_trial = a_trial / st
            sc = 2.0 * n * fast_autocorr(a_trial).max()
            if sc < best_local:
                best_local = sc
                best_val = mid
            # Determine direction: is the function decreasing or increasing at mid?
            mid2 = (mid + hi) / 2.0 if mid < hi else mid
            if mid2 > mid + 1e-15:
                a_trial2 = a.copy()
                a_trial2[i] = mid2
                st2 = a_trial2.sum()
                if st2 > 1e-10:
                    a_trial2 = a_trial2 / st2
                    sc2 = 2.0 * n * fast_autocorr(a_trial2).max()
                    if sc2 > sc:
                        hi = mid2
                    else:
                        lo = mid
            else:
                hi = mid
                break

        a[i] = best_val
        st = a.sum()
        if st > 1e-10:
            a = a / st
            sc = 2.0 * n * fast_autocorr(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multiplicative_weights_refine(a0, time_limit, eta=0.005, seed=999):
    """Multiplicative weights refinement targeting peak contributions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    current_eta = eta
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.99
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]

        # Compute per-position contribution to top peaks
        importance = np.zeros(n)
        for k in top_peaks[:15]:
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            importance[idx] += a[idx] * a[k - idx]

        imax = importance.max()
        if imax < 1e-15:
            break
        importance = importance / imax

        # Multiplicative update
        updates = 1.0 - current_eta * importance
        updates = np.maximum(updates, 0.01)
        updates = updates / updates.mean()
        a_new = a * updates
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        score_new = 2.0 * n * fast_autocorr(a_new).max()
        if score_new < best_score:
            a = a_new
            best_score = score_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 30 == 0:
                current_eta *= 0.7
                if current_eta < 1e-7:
                    current_eta = eta
                    a = best_a.copy()
                    noise = rng.normal(0, 0.005, n)
                    a = a * (1 + noise)
                    a = np.maximum(a, 0)
                    sn = a.sum()
                    if sn > 1e-10:
                        a = a / sn

            if stall > 300:
                break

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization combining LP game, LSE gradient, peak-directed
    search, multiplicative weights, multi-length exploration, and coordinate polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(1800.0) - 120.0, 1680.0)

    solver = find_solver()
    print(f"LP solver: {solver}, budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
        else:
            parent = None
    except NameError:
        parent = None
    except Exception:
        parent = None

    if best_seq is None:
        n_init = 1600
        x = np.linspace(0, 1, n_init)
        best_seq = np.exp(-3 * (x - 0.5) ** 2)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Generated initial: n={len(best_seq)}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov game (primary optimizer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ===")
    t1 = min(remaining * 0.38, 600.0)
    if t1 > 60:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: LSE gradient descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 2: LSE gradient n={n_base} ===")
    t2 = min(remaining * 0.18, 280.0)
    if t2 > 40:
        a_out, _ = lse_gradient_descent(best_seq, t2, T_start=150.0, T_end=0.05, n_stages=35)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 3: Peak-directed stochastic local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 3: Peak-directed search n={n_base} ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_out, _ = peak_directed_search(best_seq, t3, seed=42)
        try_update(a_out, f"psd_n{n_base}")

    # === Phase 4: Multiplicative weights refine ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 4: Mult. weights n={n_base} ===")
    t4 = min(remaining * 0.08, 120.0)
    if t4 > 20:
        a_out, _ = multiplicative_weights_refine(best_seq, t4, eta=0.003, seed=555)
        try_update(a_out, f"mw_n{n_base}")

    # === Phase 5: Try longer lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        n_long = n_base + 200
        print(f"=== Phase 5: LP Nesterov n={n_long} ===")
        t5 = min(remaining * 0.15, 200.0)
        if t5 > 50:
            init_long = interpolate_sequence(best_seq, n_long)
            a_out, _ = nesterov_lp_optimize(init_long, t5, solver)
            try_update(a_out, f"lp_n{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            n_long2 = n_base + 400
            print(f"=== Phase 5b: LSE n={n_long2} ===")
            t5b = min(remaining * 0.12, 150.0)
            if t5b > 30:
                init_long2 = interpolate_sequence(best_seq, n_long2)
                a_out, _ = lse_gradient_descent(init_long2, t5b, T_start=80.0, T_end=0.5, n_stages=20)
                try_update(a_out, f"lse_n{n_long2}")

    # === Phase 6: LSE at slightly shorter length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_short = n_base - 100
        if n_short > 100:
            print(f"=== Phase 6: LSE n={n_short} ===")
            t6 = min(remaining * 0.10, 120.0)
            if t6 > 25:
                init_short = interpolate_sequence(best_seq, n_short)
                a_out, _ = lse_gradient_descent(init_short, t6, T_start=50.0, T_end=0.5, n_stages=15)
                try_update(a_out, f"lse_n{n_short}")

    # === Phase 7: Coordinate polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"=== Phase 7: Coordinate polish n={len(best_seq)} ===")
        t7 = min(remaining * 0.20, 150.0)
        if t7 > 30:
            a_out, _ = coordinate_polish(best_seq, t7)
            try_update(a_out, "coord_polish")

    # === Phase 8: Final LSE polish (very low temp) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"=== Phase 8: Final LSE polish ===")
        t8 = min(remaining * 0.25, 100.0)
        if t8 > 20:
            a_out, _ = lse_gradient_descent(best_seq, t8, T_start=10.0, T_end=0.01, n_stages=15)
            try_update(a_out, "final_lse")

    # === Phase 9: Final peak-directed search ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"=== Phase 9: Final peak-directed ===")
        t9 = min(remaining * 0.30, 60.0)
        if t9 > 10:
            a_out, _ = peak_directed_search(best_seq, t9, seed=777)
            try_update(a_out, "peak_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
