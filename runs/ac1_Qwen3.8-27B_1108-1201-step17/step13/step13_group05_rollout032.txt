"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth-max Nesterov (O(n log n) per step via FFT-based gradients)
2. Multiplicative Weights game (O(n) per step, no LP solver needed)
3. LP best-response with Nesterov acceleration (for polishing)
4. Aggressive coordinate descent with shrinkage
5. Multi-length optimization (1600 -> 2400 -> 3200 -> 4800)
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_conv(a, b):
    """Compute linear convolution of a and b using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    return fast_conv(a, a)

def compute_gradient_smooth_max(a, p):
    """Compute gradient of (1/beta)*log(sum exp(beta*(a*a))) at a, given softmax weights p."""
    n = len(a)
    # grad_j = 2 * sum_k p_k * a_{k-j} (with a_m = 0 outside [0,n-1])
    # = 2 * sum_m a_m * p_{j+m}
    # Using FFT: grad = 2 * c[n-1:n-1+n] where c = ifft(fft(a_rev) * fft(p))
    fft_size = 1 << (2 * n - 1).bit_length()
    a_rev = np.zeros(n)
    a_rev[:] = a[::-1]
    A_rev = rfft(a_rev, fft_size)
    P = rfft(p, fft_size)
    c = irfft(A_rev * P, fft_size)
    return 2.0 * c[n-1:n-1+n]

def proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = u[np.max(np.where(cond)[0])]
    return np.maximum(v - rho, 0.0)

def smooth_max_nesterov(a0, time_limit, beta_start=20.0, beta_end=200.0, verbose=True):
    """Nesterov-accelerated gradient descent on smooth max approximation."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    v = a.copy()
    x_prev = a.copy()
    
    # Precompute FFT size
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta
        beta = min(beta_start + t * 0.1, beta_end)
        
        # Compute autocorrelation
        ac = fast_autocorr(v)
        
        # Softmax weights
        shifted = beta * ac - beta * ac.max()
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.ones(len(ac)) / len(ac)
        else:
            p = p / p_sum
        
        # Gradient
        grad = compute_gradient_smooth_max(v, p)
        
        # Step size (use Lipschitz estimate)
        # f_beta is Lipschitz with constant ~ beta * ||a||_inf
        L = beta * (np.max(ac) + 1e-10)
        step = 1.0 / L
        
        # Gradient step with projection
        v_new = v - step * grad
        v_new = proj_simplex(v_new, c)
        
        # Nesterov momentum
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t)) / 2.0
        t_prev = (1.0 + np.sqrt(1.0 + 4.0 * (t - 1))) / 2.0 if t > 1 else 1.0
        x_new = v_new
        v = x_new + (t_prev - 1.0) / t_k * (x_new - x_prev)
        v = np.maximum(v, 0.0)
        s_v = v.sum()
        if s_v < 1e-15:
            v = x_new.copy()
        else:
            v = v * c / s_v
        x_prev = x_new
        
        # Track best
        M = fast_autocorr(x_new).max()
        if M < best_m:
            best_m = M
            best_a = x_new.copy()
        
        M2 = fast_autocorr(v).max()
        if M2 < best_m:
            best_m = M2
            best_a = v.copy()
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    SmoothMax t={t} ({elapsed:.1f}s): J={J:.6f} beta={beta:.1f}")
    
    return best_a, best_m

def mw_game_step(a, M, n_inner=500, eta=1.0):
    """Multiplicative weights approximation to LP best response."""
    n = len(a)
    # Solve: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0
    # Normalize: work with b' = b/M, so max sum(b') s.t. (a*b')_k <= 1
    # MW: b'_j starts uniform, update based on most violated constraint
    b = np.ones(n) / n
    
    for t in range(n_inner):
        ab = fast_conv(a, b)
        # Find most violated constraint
        k_star = int(np.argmax(ab))
        viol = ab[k_star] / max(M, 1e-30)
        if viol < 1.0 + 1e-10:
            break
        # Update: b_j *= exp(-eta * a_{k_star - j} / M)
        row = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            row[j] = a[k_star - j]
        b *= np.exp(-eta * row / max(M, 1e-30))
        b = np.maximum(b, 1e-30)
        b_sum = b.sum()
        if b_sum < 1e-30:
            b = np.ones(n) / n
        # Don't normalize - we want to track the actual sum
    
    return b

def mw_nesterov(a0, time_limit, n_inner=300, eta=1.0, verbose=True):
    """Nesterov-accelerated multiplicative weights game."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        
        # MW best response
        b = mw_game_step(y, M_curr, n_inner=n_inner, eta=eta)
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * c / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c / s_y
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * c / s_x
        
        y = y_new
        x = x_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_m:
            best_m = score_y
            best_a = y.copy()
        
        score_x = fast_autocorr(x).max()
        if score_x < best_m:
            best_m = score_x
            best_a = x.copy()
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    MW t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def lp_best_response_fast(a, M):
    """Solve LP best response using scipy linprog (faster than cvxpy for this structure)."""
    from scipy.optimize import linprog
    n = len(a)
    
    # max sum(b) s.t. (a*b)_k <= M for all k, b >= 0
    # In linprog form: min -sum(b) s.t. (a*b)_k <= M, b >= 0
    
    # Build constraint matrix A (2n-1 x n) where A[k,j] = a[k-j] if 0<=k-j<n
    # This is a Toeplitz matrix
    # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    
    # For scipy linprog, we need A_ub @ x <= b_ub
    # A_ub is (2n-1) x n
    # This matrix is sparse (each row has at most min(n, k+1, 2n-1-k) nonzeros)
    
    # For n=1600, this is a 3199x1600 matrix with ~n^2/2 nonzeros
    # That's about 1.3M nonzeros, which might be too large for dense linprog
    
    # Let's use the structure: each row k has nonzeros at positions j where 0<=k-j<n
    # i.e., j in [max(0,k-n+1), min(k, n-1)]
    
    # For large n, let's just use the sparse matrix
    from scipy.sparse import lil_matrix, csr_matrix
    
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A[k, j] = a[k - j]
    
    A_csr = A.tocsr()
    b_ub = M * np.ones(2 * n - 1)
    c_obj = -np.ones(n)  # minimize -sum(b)
    
    res = linprog(c_obj, A_ub=A_csr, b_ub=b_ub, method='highs', bounds=[(0, None)] * n)
    
    if res.status != 0:
        return None
    
    b_val = np.maximum(res.x, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val
