```python
def convolve_matrix(f):
    """Construct the (2n-1) x n matrix C such that C @ g = convolve(f, g)."""
    n = len(f)
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                C[k, j] = f[i]
    return C
```

```python
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def convolve_matrix(f):
    """Construct (2n-1) x n matrix C such that C @ g = convolve(f, g)."""
    n = len(f)
    C = np.zeros((2*n-1, n))

    for k in range(2*n-1):

        j_max = min(n-1, k)
        for j in range(j_min, j_max + 1):
            C[k, j] = f[k - j]
    return C

def lp_step(f, M, solver='CLARABEL'):

    prob = cp.Problem(objective, constraints)
    prob.solve(solver=solver)
    if g.value is None:
        return None
    return np.maximum(g.value, 0)

        return f
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative peak shaving,
smooth gradient refinement, multiplicative weight updates, and multi-length search.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.argmax(~cond))
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def convolve_matrix(f):
    """Construct (2n-1) x n matrix C such that C @ g = convolve(f, g)."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            C[k, j] = f[k - j]
    return C


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_smooth_conv(a, w):
    """Gradient of weighted sum of convolution w.r.t. a."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def lp_peak_shave(f, n_iters=50, t_start=0.15, t_min=0.01, deadline=None):
    """LP-based iterative peak shaving."""
    n = len(f)
    f = np.maximum(f.astype(np.float64), 0.0)
    S = f.sum()
    if S < 1e-10:
        return f
    f = f / S

    best_f = f.copy()
    best_peak = fast_convolve(f).max()
    t = t_start

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Construct constraint matrix
        C = convolve_matrix(f)

        # Solve LP: max sum(g) s.t. C @ g <= M, g >= 0
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=C, b_ub=np.full(2*n-1, M),
                      bounds=[(0, None)] * n, method='highs')

        if res.status != 0:
            break

        g = np.maximum(res.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break

        # Scale g to sum 1
        g = g / g_sum

        # Update
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        S_new = f_new.sum()
        if S_new < 1e-10:
            break
        f_new = f_new / S_new

        peak_new = fast_convolve(f_new).max()

        if peak_new < best_peak:
            best_peak = peak_new
            best_f = f_new.copy()

        f = f_new

        # Adapt step size
        if peak_new >= M * (1 - 0.5 * t):
            t = max(t_min, t * 0.7)
        else:
            t = min(0.5, t * 1.3)

    return best_f


def smooth_gradient_optimize(a0, beta, n_iter=200, lr=0.05):
    """Projected gradient descent with smooth max."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()

    best_obj = float('inf')
    best_a = a.copy()

    for it in range(n_iter):
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        grad_a = compute_grad_smooth_conv(a, w)
        norm_grad = np.linalg.norm(grad_a)
        if norm_grad < 1e-15:
            break

        # Backtracking line search
        t = lr
        for _ in range(10):
            a_new = project_simplex(a - t * grad_a)
            conv_new = fast_convolve(a_new)
            sm_new, _ = smooth_max_and_weights(conv_new, beta)
            obj_new = 2.0 * n * sm_new
            if obj_new <= obj + 1e-10:
                break
            t *= 0.5

        a = a_new
        lr *= 0.999

    return best_a, best_obj


def multiplicative_weight_update(a, n_iters=50, eta=0.5):
    """Multiplicative weight (exponential) update to reduce peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_peak = fast_convolve(a).max()

    for it in range(n_iters):
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))

        # Cost for each position: its partner at the peak lag
        cost = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                cost[i] = a[j]
            else:
                cost[i] = 0.0

        # Exponential update: reduce high-cost positions
        updates = np.exp(-eta * cost / max(cost.max(), 1e-15))
        a = a * updates
        a = np.maximum(a, 1e-15)
        S = a.sum()
        if S > 1e-15:
            a = a / S

        peak = fast_convolve(a).max()
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # Decay eta
        eta *= 0.98

    return best_a


def flat_spectrum_init(n, rng):
    """Initialize with approximately flat power spectrum."""
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = 1.0
    phases = rng.uniform(0, 2 * np.pi, fft_size)
    for k in range(1, fft_size // 2 + 1):
        A[k] = np.exp(1j * phases[k])
        if k < fft_size - k:
            A[fft_size - k] = np.exp(-1j * phases[k])
    a = np.real(ifft(A, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def generate_initializations(n, parent, rng):

        g /= g.sum()
        inits['gauss_%s' % str(sf)] = g

    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
```

```python
        grad = np.zeros(n)

            best_a = a_new.copy()
        
        a = a_new
        lr_current *= 0.999
    
    return best_a, best_peak
```
