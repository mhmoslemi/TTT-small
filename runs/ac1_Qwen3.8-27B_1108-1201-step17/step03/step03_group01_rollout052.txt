import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_correlate_fft(w, a):
    """
    Compute corr_i = sum_k w_{k+i} * a_k for i = 0, ..., n-1
    where w has length 2n-1 and a has length n.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    # Pad to sufficient length for linear cross-correlation
    total_len = len(w) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    # ifft(FFT(w) * conj(FFT(a)))_i = sum_j w_j * a_{j-i} (linear)
    # = sum_k w_{k+i} * a_k (by substitution j = k+i)
    corr = np.fft.irfft(W * np.conj(A), fft_size)
    return corr[:n]


def lp_gradient_smooth(a, p=200):
    """
    Compute smooth gradient of (sum_k b_k^p)^{1/p} where b = conv(a,a).
    Returns (gradient, b, F_p_value).
    """
    n = len(a)
    b = fast_convolve(a)
    b_pos = np.maximum(b, 1e-30)
    
    # Stable log-sum-exp computation
    log_b = np.log(b_pos)
    p_log_b = p * log_b
    max_p_log_b = p_log_b.max()
    log_sum_exp = max_p_log_b + np.log(np.sum(np.exp(p_log_b - max_p_log_b)))
    
    # w_k = b_k^p / sum b_j^p = exp(p*log(b_k) - log_sum_exp)
    log_w = p_log_b - log_sum_exp
    w = np.exp(log_w)
    
    # F_p = (sum b_k^p)^{1/p} = exp(log_sum_exp / p)
    F_p = np.exp(log_sum_exp / p)
    
    # Gradient: dF_p/da_i = 2 * F_p^{1-p} * sum_k b_k^{p-1} * a_{k-i}
    # = 2 * F_p * sum_k (b_k^p / sum b_j^p) * (b_k^{p-1} / b_k^p) * a_{k-i}
    # = 2 * F_p * sum_k w_k * (1/b_k) * a_{k-i}

    # So grad_i = 2 * F_p * sum_k (w_k / b_k) * a_{k-i}

def lp_optimize(a0, time_limit, verbose=False):
    """
    LP-based iterative optimization with line search and perturbations.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    b = fast_convolve(a)
    M = b.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 80
    perturb_epoch = 0
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break
        
        # Solve LP
        M_lp = M * (1.0 + 1e-5)
        g, g_sum, dual = solve_lp_direction(a, M_lp)
        
        if g is None or g_sum <= 1.0 + 1e-8:
            no_improve += 1
        else:
            d = g - a
            if np.linalg.norm(d) < 1e-12:
                no_improve += 1
            else:
                t_best, obj_best = line_search(a, d, n, best_obj)
                
                if t_best > 0 and obj_best < best_obj - 1e-14:
                    a = a + t_best * d
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                    b = fast_convolve(a)
                    M = b.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
        
        iteration += 1
        
        # Stagnation handling
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 31 + perturb_epoch * 101 + 7)
            scale = 0.003 + 0.004 * min(perturb_epoch, 8)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            b = fast_convolve(a)
            M = b.max()
            no_improve = 0
            perturb_epoch += 1
            max_no_improve = min(250, 80 + 30 * perturb_epoch)
            
            if perturb_epoch > 40:
                break
    
    return best_a, best_obj


def smooth_optimize(a0, time_limit, p_init=100, p_max=5000, lr_init=0.008, 
                    momentum_init=0.9, verbose=False):
    """
    Smooth Lp-based optimization with momentum and adaptive parameters.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    lr = lr_init
    momentum = momentum_init
    p = p_init
    
    b = fast_convolve(a)
    best_obj = 2.0 * n * b.max()
    best_a = a.copy()
    no_improve = 0
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 300:
            # Perturb
            rng = np.random.default_rng(int((time.monotonic() - t0) * 100) % 10**9)
            a = best_a + rng.standard_normal(n) * 0.005
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            b = fast_convolve(a)
            no_improve = 0
            v = np.zeros(n)
            lr = lr_init * 0.5
            if no_improve > 600:
                break
        
        # Compute smooth gradient
        b = fast_convolve(a)
        b_pos = np.maximum(b, 1e-30)
        
        # Stable Lp weights
        log_b = np.log(b_pos)
        p_log_b = p * log_b
        max_val = p_log_b.max()
        exp_shifted = np.exp(p_log_b - max_val)
        sum_exp = exp_shifted.sum()
        if sum_exp < 1e-300:
            w = np.ones(len(b)) / len(b)
        else:
            w = exp_shifted / sum_exp
        
        F_p = np.exp((max_val + np.log(sum_exp)) / p)
        
        w_over_b = w / b_pos
        corr = cross_correlate_fft(w_over_b, a)
        grad = 2.0 * F_p * corr
        
        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
            b = fast_convolve(a)
            cur_obj = 2.0 * n * b.max()
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            a = best_a.copy()
            v = np.zeros(n)
            no_improve += 1
        
        # Adaptive parameters
        lr *= 0.9997
        p = min(p_max, p * 1.005)
        momentum = min(0.97, momentum + 0.0001)
        
        # Periodically boost learning rate
        if no_improve % 50 == 49 and no_improve > 0:
            lr = max(lr, lr_init * 0.3)
    
    return best_a, best_obj


def dual_guided_optimize(a0, time_limit, verbose=False):
    """
    Uses LP dual variables to construct weighted smooth objectives.
    Alternates between LP direction finding and QP-like subproblems.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    b = fast_convolve(a)
    best_obj = 2.0 * n * b.max()
    best_a = a.copy()
    no_improve = 0
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 100:
            rng = np.random.default_rng(int((time.monotonic() - t0) * 100) % 10**9)
            a = best_a + rng.standard_normal(n) * 0.004
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            b = fast_convolve(a)
            no_improve = 0
            if no_improve > 300:
                break
        
        # Get LP solution and dual
        M = b.max()
        g, g_sum, dual = solve_lp_direction(a, M * 1.0001)
        
        if g is not None and g_sum > 1.0 + 1e-8:
            d = g - a
            if np.linalg.norm(d) > 1e-10:
                t_best, obj_best = line_search(a, d, n, best_obj, max_candidates=8)
                if t_best > 0:
                    a = a + t_best * d
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                    b = fast_convolve(a)
                    if 2.0 * n * b.max() < best_obj:
                        best_obj = 2.0 * n * b.max()
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Every 20 iterations, do a smooth step
        if no_improve % 20 == 19 and no_improve > 0:
            # Use Lp with moderate p for a few steps
            for _ in range(5):
                b_pos = np.maximum(b, 1e-30)
                log_b = np.log(b_pos)
                p_log_b = 100.0 * log_b
                max_val = p_log_b.max()
                w = np.exp(p_log_b - max_val)
                w /= w.sum()
                F_p = np.exp((max_val + np.log(w.sum())) / 100.0)
                w_over_b = w / b_pos
                corr = cross_correlate_fft(w_over_b, a)
                grad = 2.0 * F_p * corr
                a = a - 0.005 * grad
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                b = fast_convolve(a)
                if 2.0 * n * b.max() < best_obj:
                    best_obj = 2.0 * n * b.max()
                    best_a = a.copy()
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()
    inits['ramp_rev'] = inits['ramp'][::-1].copy()
    
    # Triangle
    tri = np.minimum(x, n - 1 - x).astype(np.float64) + 1.0
    inits['triangle'] = tri / tri.sum()
    
    # Bimodal (edges)
    bi = np.ones(n)
    mid = n // 2
    bi[mid - n//8:mid + n//8] = 0.3
    inits['bimodal'] = bi / bi.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n-1)) / (0.2 * n)) ** 2)
    inits['gaussian'] = g / g.sum()
    
    # Sine
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits['sine'] = sine / sine.sum()
    
    # Cosine bell
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cb[0] = cb[-1] = 0.1
    inits['cosine'] = cb / cb.sum()
    
    # Power law
    pw = (x + 1.0) ** 0.3
    inits['power'] = pw / pw.sum()
    
    # Decaying exponential
    exp_dec = np.exp(-x / (n * 0.3))
    inits['exponential'] = exp_dec / exp_dec.sum()
    
    # Random smooth
    freqs = rng.integers(1, 5, size=5)
    amps = rng.random(5)
    phases = rng.random(5) * 2 * np.pi
    smooth = np.ones(n)
    for f, amp, ph in zip(freqs, amps, phases):
        smooth += amp * np.sin(2 * np.pi * f * x / n + ph)
    smooth = np.maximum(smooth, 0.01)
    inits['smooth_random'] = smooth / smooth.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent.astype(np.float64), 0.0)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
            if len(p) == n:
                inits['parent'] = p
            else:
                # Resample
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0.0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_resample'] = pp / pps
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: LP optimization, smooth Lp optimization,
    dual-guided optimization, and adaptive restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent sequence: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass
    
    def update_best(a, label=""):
        nonlocal best_sequence, best_score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  NEW BEST {label}: {score:.8f} (n={len(a)})")
            return True
        return False
    
    # === Phase 1: Quick LP scan across lengths ===
    print("=== Phase 1: LP Scan ===")
    t_start = time.monotonic()
    phase1_budget = min(400.0, (deadline - time.monotonic()) * 0.25)
    phase1_deadline = t_start + phase1_budget
    
    lengths = [256, 384, 512, 640]
    scan_results = []
    
    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break
        
        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'parent_resample', 'uniform', 'bimodal', 'sine', 
                    'smooth_random', 'random', 'triangle', 'cosine']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:4]
        
        t_per_init = max(10.0, (phase1_budget - 20.0) / (len(lengths) * 4))
        
        for name in init_names:
            if time.monotonic() > phase1_deadline - 15:
                break
            t_alloc = min(t_per_init, phase1_deadline - time.monotonic() - 10)
            if t_alloc < 5:
                break
            
            a0 = inits[name]
            a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc)
            update_best(a_opt, f"P1 n={n} {name}")
            scan_results.append((obj_opt, n, name, a_opt.copy()))
    
    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    
    # === Phase 2: Deep LP optimization ===
    print("=== Phase 2: Deep LP ===")
    t_start = time.monotonic()
    
    # Select top candidates (unique by n)
    top_configs = []
    seen_n = set()
    for obj, n, name, a_opt in scan_results:
        if n not in seen_n:
            top_configs.append((n, a_opt))
            seen_n.add(n)
        if len(top_configs) >= 3:
            break
    
    if best_sequence is not None and len(best_sequence) not in seen_n:
        top_configs.append((len(best_sequence), best_sequence.copy()))
    
    remaining_for_deep = deadline - time.monotonic() - 200.0
    t_per_deep = max(30.0, remaining_for_deep * 0.6 / max(1, len(top_configs)))
    
    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 180:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 150)
        if t_alloc < 10:
            break
        
        a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc)
        update_best(a_opt, f"P2 deep LP {i} (n={n})")
    
    print(f"Phase 2 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    
    # === Phase 3: Smooth Lp optimization ===
    print("=== Phase 3: Smooth Lp ===")
    t_start = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 100.0
        if remaining > 30:
            n_best = len(best_sequence)
            
            sg_configs = [
                {'p_init': 100, 'p_max': 3000, 'lr_init': 0.006, 'momentum_init': 0.92},
                {'p_init': 200, 'p_max': 5000, 'lr_init': 0.004, 'momentum_init': 0.95},
                {'p_init': 50, 'p_max': 2000, 'lr_init': 0.01, 'momentum_init': 0.88},
            ]
            
            t_per_sg = remaining * 0.7 / len(sg_configs)
            
            for ci, cfg in enumerate(sg_configs):
                if time.monotonic() > deadline - 80:
                    break
                t_alloc = min(t_per_sg, deadline - time.monotonic() - 60)
                if t_alloc < 8:
                    break
                
                a_opt, obj_opt = smooth_optimize(best_sequence, time_limit=t_alloc, **cfg)
                update_best(a_opt, f"P3 smooth {ci}")
    
    print(f"Phase 3 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    
    # === Phase 4: Dual-guided optimization ===
    print("=== Phase 4: Dual-guided ===")
    t_start = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 80.0
        if remaining > 20:
            t_alloc = min(remaining * 0.6, deadline - time.monotonic() - 60)
            if t_alloc > 10:
                a_opt, obj_opt = dual_guided_optimize(best_sequence, time_limit=t_alloc)
                update_best(a_opt, "P4 dual-guided")
    
    print(f"Phase 4 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    
    # === Phase 5: Final polish - alternate LP and smooth ===
    print("=== Phase 5: Final Polish ===")
    t_start = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 15:
            # Alternate between LP and smooth optimization
            a = best_sequence.copy()
            n = len(a)
            n_rounds = max(2, int(remaining / 20))
            t_per_round = remaining / n_rounds
            
            for ri in range(n_rounds):
                if time.monotonic() > deadline - 5:
                    break
                t_alloc = min(t_per_round, deadline - time.monotonic() - 5)
                if t_alloc < 3:
                    break
                
                if ri % 2 == 0:
                    a_opt, _ = lp_optimize(a, time_limit=t_alloc * 0.7)
                else:
                    a_opt, _ = smooth_optimize(a, time_limit=t_alloc * 0.7,
                                               p_init=500, p_max=5000,
                                               lr_init=0.003, momentum_init=0.95)
                a = a_opt
                score = float(evaluate_sequence(a.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a.copy()
                    print(f"  P5 round {ri}: {score:.8f}")
    
    print(f"Phase 5 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    
    # === Phase 6: Final perturb and quick optimize ===
    print("=== Phase 6: Perturb & Optimize ===")
    t_start = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 10:
            n_best = len(best_sequence)
            n_perturb = max(1, int(remaining / 12))
            t_per = remaining / n_perturb
            
            for pi in range(n_perturb):
                if time.monotonic() > deadline - 3:
                    break
                if t_per < 4:
                    break
                
                scale = 0.002 * (1 + pi * 0.5)
                noise = rng.standard_normal(n_best) * scale
                a_pert = best_sequence + noise
                a_pert = np.maximum(a_pert, 0.0)
                s = a_pert.sum()
                if s > 1e-15:
                    a_pert = a_pert / s
                
                t_alloc = min(t_per, deadline - time.monotonic() - 2)
                if t_alloc < 2:
                    break
                
                # Quick smooth optimization
                a_opt, _ = smooth_optimize(a_pert, time_limit=t_alloc,
                                          p_init=300, p_max=3000,
                                          lr_init=0.005, momentum_init=0.93)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P6 perturb {pi}: {score:.8f}")
    
    print(f"Phase 6 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Should not happen, but fallback
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
