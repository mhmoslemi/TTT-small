    Returns new value of a[j] and updated conv."""

    # conv[k] = base[k] + contribution from a[j]
    # For k with 0 <= k-j < n and k != 2j: contribution = 2*old*a[k-j]
    # For k = 2j: contribution = old^2

    # Linear terms: for k in [j, j+n-1] (i.e., 0 <= k-j <= n-1), k != 2j
    #   slope = 2*a[k-j], intercept = conv[k] - 2*old*a[k-j]
    # Quadratic term: k = 2j (if 0 <= 2j <= 2n-2)
    #   value = conv[2j] - old^2 + t^2

    def F(t):
        # Compute max_k f_k(t)
        vals = conv.copy()
        # Linear updates
        k_range = np.arange(j, min(2*n-1, j+n))
        mask = k_range != 2*j

        # Quadratic update
        if 0 <= 2*j <= 2*n-2:
            vals[2*j] = conv[2*j] + t*t - old*old
        return vals.max()
    
    # Simple approach: try a few t values and pick the best
    # Then refine with golden section
    candidates = [0.0, old * 0.5, old * 0.75, old, old * 1.25, old * 1.5, old * 2.0]
    best_t = min(candidates, key=lambda t: F(t))
    
    # Golden section refinement around best_t
    if best_t == 0.0:
        lo, hi = 0.0, candidates[1] if len(candidates) > 1 else 1.0
    else:
        lo = max(0.0, best_t * 0.5)
        hi = best_t * 1.5

    gr = (np.sqrt(5) - 1) / 2
    a_gs = lo + (1-gr)*(hi-lo)
    b_gs = lo + gr*(hi-lo)

    fb = F(b_gs)

            hi = b_gs
            b_gs = a_gs
            fb = fa
            a_gs = lo + (1-gr)*(hi-lo)
            fa = F(a_gs)
        else:
            lo = a_gs
            a_gs = b_gs
            fa = fb
            b_gs = lo + gr*(hi-lo)
            fb = F(b_gs)

    F_star = F(t_star)
    F_old = F(old)

        a[j] = t_star
        for k in range(j, min(2*n-1, j+n)):

                conv[k] += 2*(t_star - old)*a[k-j]
        if 0 <= 2*j <= 2*n-2:
            conv[2*j] += t_star*t_star - old*old
        # Renormalize... actually let's not renormalize inside the loop
        # The sum constraint will be handled outside
    
    return t_star

refinement with hierarchical constraint bounds, backtracking line search,

Key improvements over prior approach:

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - np.arange(j_start, j_end + 1)]
    return A


def objective_normalized(a):
    """Compute 2*n*max(conv) for normalized (sum=1) sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def lp_hierarchical(a, M, peak_lags, delta=0.01, slack=0.05):
    """
    LP: max sum(b) s.t. b >= 0,
        conv(a,b)[k] <= M*(1-delta) for k in peak_lags,
        conv(a,b)[k] <= M + slack*M for others.
    """
    n = len(a)
    A_mat = build_conv_matrix(a)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    b_ub = np.full(2 * n - 1, M + slack * M)
    for k in peak_lags:
        b_ub[k] = M * (1.0 - delta)

    res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def lp_min_peak_direct(a, peak_lags, num_extra=20):
    """
    LP: min t s.t. sum(b)=1, b>=0,
        conv(a,b)[k] <= t for k in peak_lags,
        conv(a,b)[k] <= M*(1+0.1) for others.
    """
    n = len(a)
    conv_a = fast_convolve(a)
    M = conv_a.max()
    A_mat = build_conv_matrix(a)
    num_rows = 2 * n - 1

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A_mat
    A_ub[:, n] = -1.0

    # For non-peak lags, use a looser bound
    b_ub = np.full(num_rows, M * 1.3)
    # For peak lags, the constraint is conv(a,b)[k] - t <= 0 (handled by t column)
    # Actually all constraints use the -t column. For non-peaks we can use a tighter fixed bound.
    non_peak = np.ones(num_rows, dtype=bool)
    for k in peak_lags:
        non_peak[k] = False
    b_ub[non_peak] = M * 1.1  # tighter for non-peaks since they're already low

    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    if res.status == 0:

def line_search(a, direction, best_obj, max_iter=50):
    """Backtracking line search along a + t*direction (with simplex projection)."""
    n = len(a)
    dir_norm = np.linalg.norm(direction)
    if dir_norm < 1e-12:
        return a, best_obj

    direction = direction / dir_norm
    # Find initial step

        obj = 2.0 * n * conv.max()
        if obj < best_obj:
            return a_try, obj
        t *= 0.7
        if t < 0.001:
            break
    return a, best_obj

                if s_new < 1e-15:

def softmax_gradient(a, tau):

    scaled = tau * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp

    grad = np.zeros(n)
    for j in range(n):
        k_lo = j
        k_hi = min(2 * n - 2, j + n - 1)
        if k_lo <= k_hi:
            grad[j] = (n / tau) * np.dot(p[k_lo:k_hi + 1], a[k_lo - j:k_hi - j + 1])
    return grad


def smoothed_grad_descent(a0, time_limit, tau_init=20.0, lr_init=0.01,
                          momentum=0.9, max_iters=10000):
    """Smoothed gradient descent with momentum and adaptive temperature."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    tau = tau_init
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        grad = softmax_gradient(a, tau)
        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            lr *= 0.7
            no_improve = 0
        if no_improve > 2000:
            break

        # Cool down temperature
        if it % 500 == 0 and tau < 200:
            tau *= 1.5

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n (all sum to 1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum > 1e-10:
            inits[f'cos_{freq}'] = a / a_sum

    # Bimodal
    g1 = np.exp(-0.5 * (x / (0.15 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - n) / (0.15 * n)) ** 2)
    bim = g1 + g2
    bim /= bim.sum()
    inits['bimodal'] = bim

    # Truncated Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.3
    alt /= alt.sum()
    inits['alternating'] = alt

    # Superposition of parent and random (if parent exists)
    if parent is not None and len(parent) > 0:
        pp = inits.get('parent', None)
        if pp is not None:
            for mix in [0.3, 0.7]:
                mixed = mix * pp + (1 - mix) * inits['random']
                mixed = project_simplex(mixed)
                inits[f'parent_mix_{mix}'] = mixed

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search combining LP refinement,
    coordinate descent, smoothed gradient, and diversification.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return best_score
        a_list = [float(max(0.0, x)) for x in a]
        score = evaluate_sequence(a_list)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = np.array(a_list, dtype=np.float64).copy()
            print(f"  New best: {score:.6f} (n={len(a_list)})")
        return score

    # Phase 1: LP-based exploration with multiple initializations and lengths
    print("=== Phase 1: LP Exploration ===")
    t1 = time.monotonic()

    lengths = [512]
    if parent is not None:
        pl = len(parent)
        if pl != 512:
            lengths.insert(0, pl)
    # Add a few nearby lengths
    for L in [500, 520, 480, 544]:
        if L not in lengths:
            lengths.append(L)

    lengths = [L for L in lengths if 64 <= L <= 2048]

    for n in lengths:
        remain = deadline - time.monotonic() - 500
        if remain < 60:
            break

        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'parent_mix_0.7', 'uniform', 'bimodal',
                    'random', 'cos_1', 'gauss']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:5]

        time_per_init = max(20.0, remain / (len(lengths) * len(init_names) * 3))

        for name in init_names:
            remain = deadline - time.monotonic() - 400
            if remain < 10:
                break
            t_alloc = min(time_per_init, remain / 2.0)
            if t_alloc < 8:
                break

            a0 = inits[name]
            # Try multiple LP methods
            for method in ['hierarchical', 'min_peak', 'basic']:
                t_single = t_alloc / 3.0
                if t_single < 3:
                    break
                a_opt, _ = lp_iterative_refined(
                    a0, time_limit=t_single, method=method, ls=True)
                try_update(a_opt)

    print(f"Phase 1 done: best={best_score:.6f}, elapsed={time.monotonic()-t1:.0f}s")

    # Phase 2: Deep refinement on best
    print("=== Phase 2: Deep Refinement ===")
    t2 = time.monotonic()

    if best_sequence is not None:
        phase2_deadline = deadline - 400
        n_best = len(best_sequence)
        a_start = best_sequence / best_sequence.sum()

        # Multiple rounds with different parameters
        for round_idx in range(6):
            if time.monotonic() > phase2_deadline:
                break
            t_alloc = min(60.0, phase2_deadline - time.monotonic() - 5)
            if t_alloc < 5:
                break

            for method in ['hierarchical', 'min_peak']:
                if time.monotonic() > phase2_deadline:
                    break
                t_single = t_alloc / 2.0
                if t_single < 3:
                    break
                a_opt, _ = lp_iterative_refined(
                    a_start, time_limit=t_single, method=method, ls=True)
                if a_opt is not None:
                    try_update(a_opt)
                    if best_sequence is not None:
                        a_start = best_sequence / best_sequence.sum()

    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t2:.0f}s")

    # Phase 3: Coordinate descent polish
    print("=== Phase 3: Coordinate Descent ===")
    t3 = time.monotonic()

    if best_sequence is not None:
        phase3_deadline = deadline - 200
        a_start = best_sequence / best_sequence.sum()
        n_best = len(a_start)

        # Run coordinate descent for bounded time
        t_cd = min(60.0, phase3_deadline - time.monotonic() - 5)
        if t_cd > 5:
            # Run sweeps until time runs out
            n_sweeps_per_call = 5
            while time.monotonic() < phase3_deadline - 5:
                t_sweep = min(20.0, phase3_deadline - time.monotonic() - 5)
                if t_sweep < 2:
                    break
                a_opt = coord_descent_polish(a_start, n_sweeps=n_sweeps_per_call, rng=rng)
                try_update(a_opt)
                if best_sequence is not None:
                    a_start = best_sequence / best_sequence.sum()

    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3:.0f}s")

    # Phase 4: Smoothed gradient
    print("=== Phase 4: Smoothed Gradient ===")
    t4 = time.monotonic()

    if best_sequence is not None:
        phase4_deadline = deadline - 100
        a_start = best_sequence / best_sequence.sum()
        n_best = len(a_start)

        for tau_val in [30.0, 50.0, 80.0]:
            if time.monotonic() > phase4_deadline:
                break
            t_alloc = min(40.0, phase4_deadline - time.monotonic() - 5)
            if t_alloc < 5:
                break
            a_opt, _ = smoothed_grad_descent(
                a_start, time_limit=t_alloc, tau_init=tau_val,
                lr_init=0.008, momentum=0.92)
            try_update(a_opt)
            if best_sequence is not None:
                a_start = best_sequence / best_sequence.sum()

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t4:.0f}s")

    # Phase 5: Perturbation + re-optimization (diversification)
    print("=== Phase 5: Diversification ===")
    t5 = time.monotonic()

    if best_sequence is not None:
        phase5_deadline = deadline - 30
        n_best = len(best_sequence)
        a_base = best_sequence / best_sequence.sum()

        for scale in [0.003, 0.01, 0.02, 0.005, 0.015]:
            if time.monotonic() > phase5_deadline:
                break
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(a_base + noise)
            t_alloc = min(30.0, phase5_deadline - time.monotonic() - 5)
            if t_alloc < 5:
                break
            for method in ['hierarchical', 'basic']:
                if time.monotonic() > phase5_deadline:
                    break
                t_single = t_alloc / 2.0
                if t_single < 3:
                    break
                a_opt, _ = lp_iterative_refined(
                    a_pert, time_limit=t_single, method=method, ls=True)
                try_update(a_opt)

        # Also try different lengths for the best "shape"
        if time.monotonic() < phase5_deadline - 30:
            for L in [500, 520, 488, 528]:
                if time.monotonic() > phase5_deadline - 10:
                    break
                # Interpolate best to new length
                a_interp = np.interp(np.linspace(0, 1, L),
                                     np.linspace(0, 1, n_best), a_base)
                a_interp = project_simplex(a_interp)
                t_alloc = min(25.0, phase5_deadline - time.monotonic() - 10)
                if t_alloc < 5:
                    break
                a_opt, _ = lp_iterative_refined(
                    a_interp, time_limit=t_alloc, method='hierarchical', ls=True)
                try_update(a_opt)

    print(f"Phase 5 done: best={best_score:.6f}, elapsed={time.monotonic()-t5:.0f}s")

    # Phase 6: Final fine-tuning
    print("=== Phase 6: Final Polish ===")
    t6 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_start = best_sequence / best_sequence.sum()
            # Very careful LP with tiny steps
            a_opt, _ = lp_iterative_refined(
                a_start, time_limit=min(remaining, 60.0),
                method='min_peak', ls=True)
            try_update(a_opt)

            if best_sequence is not None and time.monotonic() < deadline - 5:
                a_start = best_sequence / best_sequence.sum()
                a_opt = coord_descent_polish(a_start, n_sweeps=10, rng=rng)
                try_update(a_opt)

    print(f"Phase 6 done: best={best_score:.6f}, elapsed={time.monotonic()-t6:.0f}s")

    # Final fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512

    # Final verification
    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
