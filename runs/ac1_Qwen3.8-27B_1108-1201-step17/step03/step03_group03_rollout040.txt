    n = len(v)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS with soft-max annealing
(softmax parameterization), LP-based directional refinement, and subgradient polish.
Multi-length, multi-start search with time-budgeted phases.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_max_val(c, tau):
    """Soft maximum with temperature tau."""
    c_max = c.max()
    shifted = tau * (c - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    return c_max + np.log(np.sum(np.exp(shifted))) / tau


def compute_grad_a(a, w, n):
    """Compute grad_a[j] = 2n * sum_m a[m] * w[j+m] via FFT cross-correlation."""
    size_total = 3 * n - 2  # n + (2n-1) - 1
    fft_size = 2 ** (size_total - 1).bit_length()
    A_fft = np.fft.rfft(a, fft_size)
    W_fft = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(np.conj(A_fft) * W_fft, fft_size)[:n]
    return 2.0 * n * cross


def lgbfgs_optimize(a_init, time_limit, tau_start=10.0, tau_end=2000.0,
                    num_stages=15, maxiter_stage=300):
    """L-BFGS with softmax parameterization and soft-max annealing."""
    n = len(a_init)
    t0 = time.monotonic()
    
    a = np.maximum(a_init, 1e-12)
    a = a / a.sum()
    theta = np.log(a + 1e-300)
    
    best_a = a.copy()
    best_true_obj = 2.0 * n * fast_convolve(a).max()
    
    for stage in range(num_stages):
        if time.monotonic() - t0 > time_limit:
            break
        frac = stage / max(1, num_stages - 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        stage_t0 = time.monotonic()
        stage_limit = min(time_limit - (stage_t0 - t0), 60.0)
        
        def obj_and_grad(theta):
            ts = theta - theta.max()
            ts = np.clip(ts, -50, 50)
            a = np.exp(ts)
            s = a.sum()
            if s < 1e-300:
                return 1e10, np.zeros_like(theta)
            a = a / s
            
            c = fast_convolve(a)
            obj = 2.0 * n * soft_max_val(c, tau)
            
            c_max = c.max()
            shifted = np.clip(tau * (c - c_max), -50, 50)
            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum < 1e-300:
                return obj, np.zeros_like(theta)
            w = w / w_sum
            
            grad_a = compute_grad_a(a, w, n)
            ga_dot_a = np.dot(grad_a, a)
            grad_theta = a * (grad_a - ga_dot_a)
            
            return obj, grad_theta
        
        try:
            result = minimize(obj_and_grad, theta, method='L-BFGS-B',
                            jac=True, options={'maxiter': maxiter_stage,
                                             'gtol': 1e-10, 'maxfun': 10000,
                                             'maxls': 50})
            if result.fun < 1e10:
                theta = result.x
        except Exception:
            pass
        
        # Evaluate true objective
        ts = theta - theta.max()
        ts = np.clip(ts, -50, 50)
        a_new = np.exp(ts)
        a_new = a_new / a_new.sum()
        c = fast_convolve(a_new)
        true_obj = 2.0 * n * c.max()
        
        if true_obj < best_true_obj:
            best_true_obj = true_obj
            best_a = a_new.copy()
    
    return best_a, best_true_obj


def build_lp_matrix(n):
    """Build the LP constraint matrix A_ub where (A_ub @ g)_k = (f*g)_k."""
    # This is sparse in structure; for efficiency, we'll build it as a dense matrix
    # A_ub has shape (2n-1, n)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = 1.0  # Will be multiplied by f[idx] later
    return A


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f_init, time_limit, blend_init=0.4, blend_min=0.005,
                blend_halve=10, max_iters=500):
    """LP-based directional optimization."""
    a = np.maximum(f_init, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = conv.max()
        g = solve_lp_direction(a, M, n)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Sparsity check
        if g.max() > 0.4:
            uniform = np.ones(n) / n
            g = 0.7 * g + 0.3 * uniform
        
        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall >= blend_halve:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min * 1.1:
                break
    
    return best_a, best_obj


def subgrad_optimize(a0, time_limit, lr_init=0.01, momentum=0.92,
                     top_k=30, temp=0.001):
    """Subgradient descent with momentum on the simplex."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break
        
        scaled = (peak_vals - pmax) / (temp + 0.01 * pmax)
        w = np.exp(scaled)
        w /= w.sum()
        
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = w[idx] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]
        
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            rng = np.random.default_rng(no_improve)
            a = project_simplex(a + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(a)
            v *= 0.5
            no_improve = 0
            lr = max(0.002, lr * 0.8)
        
        lr *= 0.999
        temp *= 0.998
    
    return best_a, best_obj


def resample_to_length(a, new_n):
    """Resample sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS with softmax parameterization and soft-max objective (annealing)
2. LP-based directional optimization
3. Subgradient refinement with momentum
4. Multi-length, multi-start search with time-budgeted phases.
Key innovation: correct 4n factor in gradient, aggressive soft-max annealing,
and efficient FFT-based cross-correlation for gradients.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_max_val(c, tau):
    """Soft maximum with temperature tau (numerically stable)."""
    c_max = c.max()
    shifted = tau * (c - c_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / tau


def compute_grad_a(a, w, n):
    """Compute grad_a[i] = 4n * sum_m a[m] * w[m+i] via FFT cross-correlation."""
    size_total = 3 * n - 2
    fft_size = 2 ** (size_total - 1).bit_length()
    A_fft = np.fft.rfft(a, fft_size)
    W_fft = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(np.conj(A_fft) * W_fft, fft_size)[:n]
    return 4.0 * n * cross


def lgbfgs_optimize(a_init, time_limit, tau_start=10.0, tau_end=3000.0,
                    num_stages=15, maxiter_stage=300, maxfun_stage=5000):
    """L-BFGS with softmax parameterization and soft-max annealing."""
    n = len(a_init)
    t0 = time.monotonic()

    a = np.maximum(a_init, 1e-12)
    a = a / a.sum()
    theta = np.log(a + 1e-300)

    best_a = a.copy()
    best_true_obj = 2.0 * n * fast_convolve(a).max()

    for stage in range(num_stages):
        if time.monotonic() - t0 > time_limit:
            break
        frac = stage / max(1, num_stages - 1)
        tau = tau_start * (tau_end / tau_start) ** frac

        def obj_and_grad(th):
            ts = th - th.max()
            ts = np.clip(ts, -40.0, 40.0)
            a_s = np.exp(ts)
            s = a_s.sum()
            if s < 1e-300:
                return 1e10, np.zeros_like(th)
            a_s = a_s / s

            c = fast_convolve(a_s)
            obj = 2.0 * n * soft_max_val(c, tau)

            c_max = c.max()
            shifted = np.clip(tau * (c - c_max), -50.0, 50.0)
            w_s = np.exp(shifted)
            w_sum = w_s.sum()
            if w_sum < 1e-300:
                return obj, np.zeros_like(th)
            w_s = w_s / w_sum

            grad_a = compute_grad_a(a_s, w_s, n)
            ga_dot_a = np.dot(grad_a, a_s)
            grad_theta = a_s * (grad_a - ga_dot_a)

            return obj, grad_theta

        try:
            result = minimize(obj_and_grad, theta, method='L-BFGS-B',
                              jac=True, options={'maxiter': maxiter_stage,
                                                 'gtol': 1e-12,
                                                 'maxfun': maxfun_stage,
                                                 'maxls': 30})
            if result.fun < 1e10:
                theta = result.x.copy()
        except Exception:
            pass

        # Evaluate true objective
        ts = theta - theta.max()
        ts = np.clip(ts, -40.0, 40.0)
        a_new = np.exp(ts)
        a_new = a_new / a_new.sum()
        c = fast_convolve(a_new)
        true_obj = 2.0 * n * c.max()

        if true_obj < best_true_obj:
            best_true_obj = true_obj
            best_a = a_new.copy()

    return best_a, best_true_obj


def build_lp_matrix(n):
    """Pre-build the LP constraint matrix structure."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = 1.0
    return A


def solve_lp_direction(f, M, n, A_struct=None):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    if A_struct is None:
        A_ub = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                A_ub[k, j] = f[k - j]
    else:
        A_ub = A_struct.copy()
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                A_ub[k, j] *= f[k - j]

    b_ub = np.full(2 * n - 1, M)
    c_vec = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f_init, time_limit, blend_init=0.35, blend_min=0.003,
                blend_halve=12, max_iters=300):
    """LP-based directional optimization on the simplex."""
    a = np.maximum(f_init, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    uniform = np.ones(n) / n

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = conv.max()
        g = solve_lp_direction(a, M, n)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if g.max() > 0.35:
            g = 0.7 * g + 0.3 * uniform

        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall >= blend_halve:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend <= blend_min * 1.05:
                break

    return best_a, best_obj


def subgrad_optimize(a0, time_limit, lr_init=0.008, momentum=0.92,
                     top_k=30, temp=0.001):
    """Subgradient descent with momentum on the simplex."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break

        scaled = (peak_vals - pmax) / (temp + 0.02 * pmax)
        w_s = np.exp(scaled)
        w_s /= w_s.sum()

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w_s[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = w_s[idx] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            rng_local = np.random.default_rng(no_improve * 7 + 3)
            a = project_simplex(a + rng_local.standard_normal(n) * 0.004)
            conv = fast_convolve(a)
            v *= 0.4
            no_improve = 0
            lr = max(0.001, lr * 0.7)

        lr *= 0.998
        temp *= 0.997

    return best_a, best_obj


def resample_to_length(a, new_n):
    """Resample sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def symmetric_enforce(a):
    """Make a sequence symmetric about its center."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym
