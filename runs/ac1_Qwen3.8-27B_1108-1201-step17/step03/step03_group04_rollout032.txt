"""
Improved LP-based Frank-Wolfe with active-set LP oracle, multi-length scanning,
peak-targeted local optimization, and golden-section line search to minimize
2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k] for b >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle_full(a, M, active_indices=None):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M (on active indices), b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    
    if active_indices is not None:
        # Only use constraints at active indices
        A_sub = A[active_indices, :]
        b_ub = np.full(len(active_indices), M)
    else:
        A_sub = A
        b_ub = np.full(2 * n - 1, M)
    
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_sub, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': 8.0})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def golden_section_search(a, d, f_eval, tol=1e-5, max_iter=30):
    """Golden section search for t in [0, 1] minimizing f(a + t*d)."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~ 0.618
    
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d_ = lo + gr * (hi - lo)
    fc = f_eval(c)
    fd = f_eval(d_)
    
    for _ in range(max_iter):
        if fc < fd:
            hi = d_
            d_ = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f_eval(c)
        else:
            lo = c
            c = d_
            fc = fd
            d_ = lo + gr * (hi - lo)
            fd = f_eval(d_)
        if hi - lo < tol:
            break
    
    t_best = (lo + hi) / 2
    return t_best


def compute_obj_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for normalized sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def fpw_optimize(a0, time_limit, verbose=True, verbose_every=100):
    """Improved Frank-Wolfe with active-set LP and golden section search."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 150

    # Determine active set: top K convolution indices
    top_k = min(20, len(conv))
    
    while time.monotonic() - t0 < time_limit:
        # Update active set every 50 iterations
        if it % 50 == 0:
            conv = fast_convolve(a)
            M = conv.max()
            active_idx = np.argpartition(conv, -top_k)[-top_k:]
            # Always include the global max
            max_idx = np.argmax(conv)
            if max_idx not in active_idx:
                active_idx = np.append(active_idx, max_idx)

        # Solve LP with active set constraints
        b = lp_oracle_full(a, M, active_idx)
        if b is None:
            # Try full constraints
            b = lp_oracle_full(a, M, None)
            if b is None:
                break
        
        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb

        # Check if stuck
        if np.allclose(a, b, atol=1e-10):
            no_improve += 1
        else:
            # Golden section line search
            def f_of_t(t):
                candidate = (1 - t) * a + t * b
                if candidate.sum() < 1e-15:
                    return np.inf
                candidate = candidate / candidate.sum()
                conv_c = fast_convolve(candidate)
                return 2.0 * n * conv_c.max()
            
            t_best = golden_section_search(a, b, f_of_t)
            
            if t_best < 0.01:
                no_improve += 1
            else:
                a_new = (1 - t_best) * a + t_best * b
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new / s_new
                conv_new = fast_convolve(a_new)
                cur_obj = 2.0 * n * conv_new.max()
                if cur_obj < best_obj - 1e-12:
                    best_obj = cur_obj
                    best_a = a_new.copy()
                    a = a_new
                    no_improve = 0
                else:
                    no_improve += 1

        # Stuck handling
        if no_improve >= max_stall:
            rng = np.random.default_rng(it * 7 + 3)
            noise = rng.standard_normal(n) * 0.005
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max_stall * 1.5)

        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    FPW it={it}, obj={best_obj:.6f}, M={M:.4e}, elapsed={elapsed:.1f}s")

def peak_targeted_optimize(a0, time_limit, verbose=True):

def swap_heuristic(a0, time_limit, verbose=True):
    """Try pairwise swaps to reduce the peak."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    while time.monotonic() - t0 < time_limit:
        # Try random swaps
        for _ in range(200):
            if time.monotonic() > t0 + time_limit:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            
            # Try swapping
            a[j], a[i] = a[i].copy(), a[j].copy()
            # Re-normalize (swapping doesn't change sum, but let's be safe)
            s = a.sum()
            if s < 1e-15:
                a[j], a[i] = a[i].copy(), a[j].copy()
                continue
            
            conv = fast_convolve(a)
            obj = 2.0 * n * conv.max() / (s * s)
            
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
            else:
                # Revert
                a[j], a[i] = a[i].copy(), a[j].copy()
    
    return best_a, best_obj


def continuous_tilt(a0, time_limit, verbose=True):

    n = len(a0)

    best_obj = 2.0 * n * conv.max()

    t0 = time.monotonic()
    
    # Try tilts in several directions
    directions = []
    # Linear tilt
    x = np.linspace(0, 1, n)
    directions.append(x - x.mean())
    # Quadratic
    directions.append((x - x.mean())**2 - ((x - x.mean())**2).mean())
    # Edge boost
    edge = np.zeros(n)
    edge[:n//8] = 1
    edge[-n//8:] = 1
    directions.append(edge - edge.mean())
    # Center boost
    center = np.exp(-0.5 * ((x - 0.5) / 0.3)**2)
    directions.append(center - center.mean())
    
    rng = np.random.default_rng(42)
    for _ in range(5):
        directions.append(rng.standard_normal(n))
        directions[-1] -= directions[-1].mean()
    
    for d in directions:
        d = d / (np.max(np.abs(d)) + 1e-15)
        
        # Search over tilt magnitude
        for eps in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            for sign in [1, -1]:

                a_tilt = np.maximum(a_tilt, 0)
                s = a_tilt.sum()
                if s < 1e-15:
                    continue
                a_tilt = a_tilt / s
                conv_t = fast_convolve(a_tilt)
                obj = 2.0 * n * conv_t.max() / (s * s)
                # Wait, s is already 1 after normalization
                obj = 2.0 * n * conv_t.max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a_tilt.copy()
                    a = a_tilt.copy()
                    if verbose:

def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    
    x = np.arange(n, dtype=np.float64)
    inits['uniform'] = np.ones(n) / n
    
    # Monotone increasing
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp.copy()
    
    # Monotone decreasing
    inits['ramp_rev'] = ramp[::-1].copy()
    
    # Edge-concentrated
    edge_w = max(1, n // 8)
    ec = np.zeros(n)
    ec[:edge_w] = 1.0
    ec[-edge_w:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec.copy()
    
    # Asymmetric edge (left heavy)
    ec2 = np.zeros(n)
    ec2[:n//4] = 2.0
    ec2[n//4:n//2] = 1.0
    ec2[n//2:] = 0.5
    ec2 /= ec2.sum()
    inits['left_heavy'] = ec2.copy()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g.copy()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Power law
    pw = np.arange(1, n + 1, dtype=np.float64) ** 0.5
    pw /= pw.sum()
    inits['power'] = pw.copy()
    inits['power_rev'] = pw[::-1].copy()
    
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps
    
    return inits

def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)

    return np.fft.irfft(A * B, fft_size)[:size]

    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M, active_indices=None, time_limit=8.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M (on active or all), b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)

    if active_indices is not None:
        A_sub = A[active_indices, :]
        b_ub = np.full(len(active_indices), M)
    else:
        A_sub = A
        b_ub = np.full(2 * n - 1, M)

    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_sub, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def golden_section_search(f_eval, lo=0.0, hi=1.0, max_iter=30):
    """Golden section search minimizing f_eval(t) on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f_eval(c)
    fd = f_eval(d)
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f_eval(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f_eval(d)
        if hi - lo < 1e-6:
            break
    return (lo + hi) / 2.0


def fpw_optimize(a0, time_limit, verbose=True, verbose_every=100):
    """Frank-Wolfe with adaptive active-set LP and golden section search."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 120
    top_k = min(15, len(conv))

    while time.monotonic() - t0 < time_limit:
        # Update active set periodically
        if it % 40 == 0:
            conv = fast_convolve(a)
            M = conv.max()
            active_idx = np.argpartition(conv, -top_k)[-top_k:]
            max_idx = np.argmax(conv)
            if max_idx not in active_idx:
                active_idx = np.append(active_idx, max_idx)

        # Solve LP
        b = lp_oracle(a, M, active_idx, time_limit=6.0)
        if b is None:
            b = lp_oracle(a, M, None, time_limit=10.0)
            if b is None:
                break

        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb

        if np.allclose(a, b, atol=1e-10):
            no_improve += 1
        else:
            # Golden section line search
            def f_of_t(t):
                candidate = (1.0 - t) * a + t * b
                sc = candidate.sum()
                if sc < 1e-15:
                    return np.inf
                candidate = candidate / sc
                conv_c = fast_convolve(candidate)
                return 2.0 * n * conv_c.max()

            t_best = golden_section_search(f_of_t, max_iter=25)

            if t_best < 0.01:
                no_improve += 1
            else:
                a_new = (1.0 - t_best) * a + t_best * b
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new / s_new
                conv_new = fast_convolve(a_new)
                cur_obj = 2.0 * n * conv_new.max()
                if cur_obj < best_obj - 1e-12:
                    best_obj = cur_obj
                    best_a = a_new.copy()
                    a = a_new
                    no_improve = 0
                else:
                    no_improve += 1

        if no_improve >= max_stall:
            rng = np.random.default_rng(it * 13 + 7)
            noise = rng.standard_normal(n) * 0.004
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max_stall * 1.4)

        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    FPW it={it}, obj={best_obj:.6f}, M={M:.4e}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def peak_gradient_polish(a0, time_limit, verbose=True):
    """Smooth gradient descent targeting the peak with adaptive temperature."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    temp = 0.05
    lr = 0.003
    momentum = 0.92
    v = np.zeros(n)
    top_k = min(25, len(conv))

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)

        # Get top peaks
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

        # Compute smooth gradient (logsumexp approximation)
        shifted = peak_vals / temp - peak_vals.max() / temp
        weights = np.exp(shifted)
        weights /= weights.sum()

        g = np.zeros(n)
        for wi, ki in zip(weights, peaks):
            if wi < 1e-12:
                continue
            k = int(ki)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            g[idx] += 2.0 * wi * a[k - idx]

        g /= temp
        # Project onto tangent space of simplex
        g = g - np.mean(g)

        v = momentum * v - lr * g
        a_new = np.maximum(a + v, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new
        a = a_new

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 500 == 0:
                print(f"    PeakGrad it={it}, obj={best_obj:.6f}, elapsed={time.monotonic()-t0:.1f}s")

        # Adaptive schedule
        temp *= 0.997
        lr *= 0.9997
        momentum = min(0.98, momentum + 0.00001)

        it += 1

    return best_a, best_obj


def tilt_search(a0, time_limit=30.0, verbose=True):
    """Search over structured perturbations (tilts) to find improvements."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    x = np.linspace(0.0, 1.0, n)
    directions = []

    # Linear tilt
    d1 = x - x.mean()
    directions.append(d1 / (np.max(np.abs(d1)) + 1e-15))

    # Quadratic
    d2 = (x - x.mean()) ** 2
    d2 = d2 - d2.mean()
    directions.append(d2 / (np.max(np.abs(d2)) + 1e-15))

    # Cubic
    d3 = (x - x.mean()) ** 3
    d3 = d3 - d3.mean()
    directions.append(d3 / (np.max(np.abs(d3)) + 1e-15))

    # Edge boost
    edge = np.zeros(n)
    edge[:n // 10] = 1.0
    edge[-n // 10:] = 1.0
    edge = edge - edge.mean()
    directions.append(edge / (np.max(np.abs(edge)) + 1e-15))

    # Center boost
    center = np.exp(-0.5 * ((x - 0.5) / 0.25) ** 2)
    center = center - center.mean()
    directions.append(center / (np.max(np.abs(center)) + 1e-15))

    # Random directions
    rng = np.random.default_rng(99)
    for _ in range(8):
        dr = rng.standard_normal(n)
        dr = dr - dr.mean()
        directions.append(dr / (np.max(np.abs(dr)) + 1e-15))

    eps_values = [0.001, 0.002, 0.003, 0.005, 0.008, 0.012, 0.02, 0.03]

    for d in directions:
        if time.monotonic() - t0 > time_limit:
            break
        for eps in eps_values:
            for sign in [1.0, -1.0]:
                a_tilt = a + sign * eps * d
                a_tilt = np.maximum(a_tilt, 0.0)
                s = a_tilt.sum()
                if s < 1e-15:
                    continue
                a_tilt = a_tilt / s
                conv_t = fast_convolve(a_tilt)
                obj = 2.0 * n * conv_t.max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a_tilt.copy()
                    a = a_tilt.copy()

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit=60.0, verbose=True):
    """Cycle through coordinates, optimizing each one given the others."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            # Save current value
            old_val = a[i]
            if old_val < 1e-15:
                # Try setting to small positive
                for trial in [0.001 / n, 0.005 / n, 0.01 / n]:
                    a[i] = trial
                    s = a.sum()
                    a_norm = a / s
                    conv_c = fast_convolve(a_norm)
                    obj = 2.0 * n * conv_c.max()
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_a = a_norm.copy()
                        a = a_norm.copy()
                        old_val = a[i]
                        improved = True
                        break
                if a[i] != old_val:
                    continue

            # Try a few multipliers of the current value
            for mult in [0.5, 0.7, 1.3, 1.7, 2.0]:
                a[i] = old_val * mult
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    continue
                a_norm = a / s
                conv_c = fast_convolve(a_norm)
                obj = 2.0 * n * conv_c.max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a_norm.copy()
                    a = a_norm.copy()
                    old_val = a[i]
                    improved = True

        if not improved and verbose:
            print(f"    CoordDescent: no improvement, obj={best_obj:.6f}, elapsed={time.monotonic()-t0:.1f}s")
            break

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp.copy()
    inits['ramp_rev'] = ramp[::-1].copy()

    edge_w = max(1, n // 8)
    ec = np.zeros(n)
    ec[:edge_w] = 1.0
    ec[-edge_w:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec.copy()

    ec2 = np.zeros(n)
    ec2[:n // 4] = 2.0
    ec2[n // 4:n // 2] = 1.0
    ec2[n // 2:] = 0.3
    ec2 /= ec2.sum()
    inits['left_heavy'] = ec2.copy()

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g.copy()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    pw = np.arange(1, n + 1, dtype=np.float64) ** 0.7
    pw /= pw.sum()
    inits['power'] = pw.copy()

    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: LP-Frank-Wolfe with active-set oracle, peak-targeted
    smooth gradient, coordinate descent polish, tilt search, and restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}, score={parent_score:.6f}")
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
    except Exception as e:
        print(f"No parent available: {e}")

    # === Phase 1: LP-FW on parent ===
    print("=== Phase 1: LP-Frank-Wolfe on parent ===")
    n_parent = len(parent) if parent is not None else 384
    if time.monotonic() < deadline - 500:
        a0 = parent.copy() / parent.sum() if parent is not None else np.ones(n_parent) / n_parent
        t_alloc = min(400.0, deadline - time.monotonic() - 400.0)
        if t_alloc > 20:
            print(f"  n={n_parent}, t_alloc={t_alloc:.0f}s")
            a_opt, obj_opt = fpw_optimize(a0, time_limit=t_alloc, verbose=True, verbose_every=200)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 1 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 2: Peak gradient polish ===
    if best_sequence is not None and time.monotonic() < deadline - 300:
        print("=== Phase 2: Peak Gradient Polish ===")
        t_alloc = min(100.0, deadline - time.monotonic() - 250.0)
        if t_alloc > 10:
            a_opt, obj_opt = peak_gradient_polish(best_sequence, time_limit=t_alloc, verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 2 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 3: Coordinate descent ===
    if best_sequence is not None and time.monotonic() < deadline - 250:
        print("=== Phase 3: Coordinate Descent ===")
        t_alloc = min(60.0, deadline - time.monotonic() - 200.0)
        if t_alloc > 10:
            a_opt, obj_opt = coordinate_descent_polish(best_sequence, time_limit=t_alloc, verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 3 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 4: Tilt search ===
    if best_sequence is not None and time.monotonic() < deadline - 200:
        print("=== Phase 4: Tilt Search ===")
        a_opt, obj_opt = tilt_search(best_sequence, time_limit=30.0, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            print(f"  Phase 4 score={score:.6f}")
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()

    # === Phase 5: Deep LP-FW ===
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print("=== Phase 5: Deep LP-FW ===")
        t_alloc = deadline - time.monotonic() - 60.0
        if t_alloc > 20:
            a_opt, obj_opt = fpw_optimize(best_sequence, time_limit=t_alloc,
                                          verbose=True, verbose_every=200)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 5 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 6: Peak gradient (again) ===
    if best_sequence is not None and time.monotonic() < deadline - 40:
        print("=== Phase 6: Peak Gradient (round 2) ===")
        t_alloc = min(50.0, deadline - time.monotonic() - 20.0)
        if t_alloc > 10:
            a_opt, obj_opt = peak_gradient_polish(best_sequence, time_limit=t_alloc, verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 6 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 7: Perturbation restarts ===
    if best_sequence is not None and time.monotonic() < deadline - 20:
        print("=== Phase 7: Perturbation Restarts ===")
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 15.0
        n_restarts = max(1, int(remaining // 20))
        n_restarts = min(n_restarts, 5)

        for i in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break
            scale = 0.0015 * (1.0 + i * 0.5)
            noise = rng.standard_normal(n_best) * scale
            a_pert = np.maximum(best_sequence + noise, 0.0)
            s_pert = a_pert.sum()
            if s_pert < 1e-10:
                break
            a_pert = a_pert / s_pert
            t_alloc = min(15.0, deadline - time.monotonic() - 10.0)
            if t_alloc < 5:
                break
            a_opt, obj_opt = fpw_optimize(a_pert, time_limit=t_alloc, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Restart {i}: score={score:.6f} (improved!)")

    # === Bonus: try a different length if time permits ===
    if best_sequence is not None and time.monotonic() < deadline - 60:
        alt_n = len(best_sequence) + 32
        print(f"=== Bonus: Try n={alt_n} ===")
        inits = generate_inits(alt_n, best_sequence, rng)
        for name in ['parent_interp', 'uniform', 'ramp']:
            if time.monotonic() > deadline - 30:
                break
            if name not in inits:
                continue
            a0 = inits[name]
            t_alloc = min(25.0, deadline - time.monotonic() - 20.0)
            if t_alloc < 10:
                break
            a_opt, obj_opt = fpw_optimize(a0, time_limit=t_alloc, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  n={alt_n}, {name}: score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
