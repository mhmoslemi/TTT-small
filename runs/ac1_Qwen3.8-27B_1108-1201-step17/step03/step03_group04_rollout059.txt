def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

    n = len(a)

def softmax_grad(a, beta=20.0):
    """Compute gradient of softmax surrogate using FFT."""
    n = len(a)
    conv = fast_convolve(a)
    # Softmax weights (numerically stable)
    shifted = beta * (conv - conv.max())
    # Clip for numerical stability
    shifted = np.maximum(shifted, -50.0)
    w = np.exp(shifted)

        w[np.argmax(conv)] = 1.0

    grad = 2.0 * cross[n-1 : 2*n-1]
    return grad, conv.max()

def simplex_project_exact(v):
    """Exact projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

"""
FFT-based gradient descent with adaptive softmax surrogate, parametric search,
and multi-start strategy to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses momentum, adaptive learning rate, and periodic perturbation to escape local minima.
"""

import numpy as np
import time


def fast_convolve(a, b=None):
    """Linear convolution via FFT. Returns full convolution."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_obj_normalized(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2. Returns inf if sum too small."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def local_obj(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_gradient(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*conv_k)) w.r.t. a.
    Uses FFT for efficiency. Returns (grad, conv_max).
    """
    n = len(a)
    conv = fast_convolve(a)
    # Numerically stable softmax
    shifted = beta * (conv - conv.max())
    shifted[shifted < -50] = -50
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # grad[j] = 2 * sum_k w[k] * a[k-j] = 2 * cross[n-1+j]
    # where cross = conv(w, a_flipped)
    a_flip = a[::-1]
    cross = fast_convolve(w, a_flip)
    grad = 2.0 * cross[n - 1:2 * n - 1].copy()
    return grad, conv.max()


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v_clip = np.maximum(v, 0.0)
    s = v_clip.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return v_clip / s


def gradient_descent_opt(a0, time_limit, beta_init=15.0, beta_max=80.0,
                         momentum=0.85, lr_init=0.0005, verbose=False):
    """
    Projected gradient descent with momentum on softmax surrogate.
    Adaptive beta and learning rate. Returns (best_a, best_obj).
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = local_obj(a)
    best_a = a.copy()

    v = np.zeros(n)
    lr = lr_init
    beta = beta_init
    t0 = time.monotonic()
    it = 0
    stall = 0
    best_stall = 0

    while time.monotonic() - t0 < time_limit:
        grad, cur_max = softmax_gradient(a, beta)
        v = momentum * v - lr * grad
        a_new = a + v
        a = project_simplex(a_new)

        conv = fast_convolve(a)
        cur_obj = local_obj(a)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
            # Periodically increase beta
            if it > 50:
                beta = min(beta_max, beta * 1.002)
        else:
            stall += 1

        # Adaptive learning rate
        if stall > 30:
            lr *= 0.85
            v *= 0.3
            stall = 0
        if stall > 150:
            # Perturb to escape local minimum
            rng = np.random.default_rng(12345 + it)
            noise = rng.standard_normal(n) * 0.008
            a = a + noise
            a = project_simplex(a)
            v = np.zeros(n)
            lr = max(lr_init * 0.3, lr * 0.5)
            beta = max(beta_init, beta * 0.7)
            stall = 0
            best_stall += 1
            if best_stall > 5:
                break

        it += 1
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GD it={it}, obj={best_obj:.6f}, beta={beta:.1f}, lr={lr:.2e}, t={elapsed:.1f}s")

    return best_a, best_obj


def parametric_search(n, time_limit, verbose=False):
    """
    Search over parametric families: sum of Gaussian bumps.
    a[j] = sum_i w_i * exp(-(x_i)^2 / (2*s_i^2)) where x_i = j - c_i
    Optimize (w_i, c_i, s_i) via Nelder-Mead.
    """
    from scipy.optimize import minimize

    best_a = np.ones(n) / n
    best_obj = local_obj(best_a)

    x_grid = np.arange(n, dtype=np.float64)

    def eval_params(params):
        m = len(params) // 3
        a = np.zeros(n)
        for i in range(m):
            w_i = params[3 * i]
            c_i = params[3 * i + 1]
            s_i = params[3 * i + 2] + 0.1
            a += w_i * np.exp(-0.5 * ((x_grid - c_i) / s_i) ** 2)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return local_obj(np.ones(n) / n)
        a = a / s
        return local_obj(a)

    def eval_params_save(params):
        m = len(params) // 3
        a = np.zeros(n)
        for i in range(m):
            w_i = params[3 * i]
            c_i = params[3 * i + 1]
            s_i = params[3 * i + 2] + 0.1
            a += w_i * np.exp(-0.5 * ((x_grid - c_i) / s_i) ** 2)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return local_obj(np.ones(n) / n), np.ones(n) / n
        a = a / s
        return local_obj(a), a

    rng = np.random.default_rng(42)
    t0 = time.monotonic()

    for m_bumps in [2, 3, 4]:
        if time.monotonic() - t0 > time_limit:
            break
        for trial in range(5):
            if time.monotonic() - t0 > time_limit - 10:
                break
            # Initialize params
            params = np.zeros(3 * m_bumps)
            for i in range(m_bumps):
                params[3 * i] = 1.0 / m_bumps  # weight
                params[3 * i + 1] = (i + 0.5) * n / m_bumps + rng.standard_normal() * n * 0.05  # center
                params[3 * i + 2] = n * (0.1 + 0.15 * rng.random())  # width

            # Nelder-Mead
            try:
                res = minimize(eval_params, params, method='Nelder-Mead',
                               options={'maxiter': 2000, 'xatol': 1e-6, 'fatol': 1e-8,
                                        'adaptive': True})
                if res.fun < best_obj:
                    obj, a_opt = eval_params_save(res.x)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a_opt
            except Exception:
                pass

        if verbose:
            print(f"    Parametric m={m_bumps}: obj={best_obj:.6f}")

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Edge-concentrated (two bumps at edges)
    edge_w = max(1, n // 6)
    ec = np.zeros(n)
    ec[:edge_w] = 1.0
    ec[-edge_w:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec.copy()

    # Single edge
    se = np.zeros(n)
    se[:int(n * 0.3)] = 1.0
    se /= se.sum()
    inits['single_edge'] = se.copy()

    # Gaussian
    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g.copy()

    # Bimodal
    bw = n // 5
    g1 = np.exp(-0.5 * ((x - bw) / (0.15 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - (n - 1 - bw)) / (0.15 * n)) ** 2)
    bim = g1 + g2
    bim /= bim.sum()
    inits['bimodal'] = bim.copy()

    # Triangular with edge emphasis
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri = 1.0 / tri  # invert to emphasize edges
    tri /= tri.sum()
    inits['tri_inv'] = tri.copy()

    # Power law
    d = np.abs(x - 0.5 * (n - 1))
    pl = 1.0 / (1.0 + d * 0.05) ** 1.5
    pl /= pl.sum()
    inits['powerlaw'] = pl.copy()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Sparse random
    sparse = rng.random(n)
    sparse[sparse < 0.6] = 0.0
    if sparse.sum() > 1e-10:
        sparse /= sparse.sum()
    else:
        sparse = rng.dirichlet(np.ones(n))
    inits['sparse'] = sparse.copy()

    # Parent
    if parent is not None:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps

    return inits


def lp_direction(a, M, max_iter=200):
    """
    Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns best direction b (normalized) or None.
    Uses scipy linprog with time limit.
    """
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    n = len(a)
    # Build constraint matrix: A[b] = conv(a, b)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 3.0})
        if res.success:
            b = np.maximum(res.x, 0.0)
            sb = b.sum()
            if sb > 1e-15:
                b = b / sb
                return b
    except Exception:
        pass
    return None


def fw_lp_step(a, n_steps=3):
    """A few LP-FW iterations to escape local minima."""
    n = len(a)
    a = a.copy()
    conv = fast_convolve(a)
    M = conv.max()

    for _ in range(n_steps):
        b = lp_direction(a, M)
        if b is None:
            break
        # Line search
        best_t = 0
        best_obj = local_obj(a)
        for t in [0.3, 0.5, 0.7, 0.9]:
            candidate = (1 - t) * a + t * b
            obj = local_obj(candidate)
            if obj < best_obj:
                best_obj = obj
                best_t = t
        if best_t == 0:
            break
        a = (1 - best_t) * a + best_t * b
        conv = fast_convolve(a)
        M = conv.max()

    a = project_simplex(a)
    return a, local_obj(a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining FFT gradient descent, parametric search,
    and LP-FW steps to minimize 2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}")
    except Exception:
        pass

    # === Phase 1: Multi-length gradient descent ===
    print("=== Phase 1: Multi-start Gradient Descent ===")
    lengths = [256, 384, 512, 768]
    inits_to_try = ['edge', 'bimodal', 'uniform', 'parent_interp', 'parent',
                    'single_edge', 'gauss', 'tri_inv', 'random', 'powerlaw']

    for n in lengths:
        if time.monotonic() > deadline - 240:
            break
        inits = generate_inits(n, parent, rng)
        available = [x for x in inits_to_try if x in inits]

        remaining = deadline - time.monotonic() - 180
        t_per = max(15.0, min(60.0, remaining / (len(lengths) * 3)))

        for name in available[:5]:
            if time.monotonic() > deadline - 60:
                break
            t_alloc = max(10.0, min(50.0, (deadline - time.monotonic() - 60) / 2))
            if t_alloc < 5:
                break

            a0 = inits[name]
            print(f"  n={n}, init={name}, t={t_alloc:.0f}s")
            a_opt, obj_opt = gradient_descent_opt(
                a0, time_limit=t_alloc, beta_init=12.0, beta_max=60.0,
                momentum=0.82, lr_init=0.0004, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"    score={score:.6f}, local={obj_opt:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    print(f"After Phase 1: best={best_score:.6f}")

    # === Phase 2: Parametric search on best length ===
    if best_sequence is not None:
        n_best = len(best_sequence)
        print(f"=== Phase 2: Parametric Search (n={n_best}) ===")
        remaining = deadline - time.monotonic() - 120
        if remaining > 20:
            t_param = min(remaining * 0.3, 90.0)
            a_opt, obj_opt = parametric_search(n_best, time_limit=t_param, verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Parametric score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 3: Deep gradient descent on best ===
    if best_sequence is not None:
        print("=== Phase 3: Deep Optimization ===")
        remaining = deadline - time.monotonic() - 60
        if remaining > 10:
            # High beta, small LR for fine-tuning
            t_deep = remaining * 0.5
            a_opt, obj_opt = gradient_descent_opt(
                best_sequence, time_limit=t_deep, beta_init=40.0, beta_max=100.0,
                momentum=0.92, lr_init=0.0001, verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Deep GD score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 4: LP-FW escape attempts ===
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print("=== Phase 4: LP-FW Escape ===")
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 20
        n_attempts = max(1, int(remaining // 15))
        n_attempts = min(n_attempts, 8)

        for i in range(n_attempts):
            if time.monotonic() > deadline - 10:
                break
            # Slightly perturb then LP-FW
            scale = 0.002 * (1 + i * 0.5)
            noise = rng.standard_normal(n_best) * scale
            a_pert = np.maximum(best_sequence + noise, 0.0)
            a_pert = a_pert / a_pert.sum()
            t_alloc = min(10.0, deadline - time.monotonic() - 8)
            if t_alloc < 3:
                break
            a_opt, obj_opt = fw_lp_step(a_pert, n_steps=3)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  LP-FW {i}: score={score:.6f} (improved!)")

    # === Phase 5: Final polish ===
    if best_sequence is not None and time.monotonic() < deadline - 15:
        print("=== Phase 5: Final Polish ===")
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            a_opt, obj_opt = gradient_descent_opt(
                best_sequence, time_limit=remaining, beta_init=80.0, beta_max=120.0,
                momentum=0.95, lr_init=0.00003, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Final polish score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Should not happen, but safety
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
