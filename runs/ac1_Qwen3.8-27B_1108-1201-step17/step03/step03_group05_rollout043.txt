"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using spectral flattening (iterative
magnitude compression in frequency domain), combined with subgradient descent
on the peak convolution, and coordinate-based polishing.
Explores multiple sequence lengths and initializations.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_flatten(a0, time_limit, seed=0, alpha_starts=None, gamma_range=None):
    """
    Iterative spectral flattening to reduce peak autocorrelation.
    
    Works by compressing the magnitude spectrum (making it flatter) and
    projecting back to the non-negative simplex.
    """
    if alpha_starts is None:
        alpha_starts = [0.3, 0.5, 0.7, 0.85]
    if gamma_range is None:
        gamma_range = (0.3, 0.8)
    
    rng = np.random.default_rng(seed)
    n = len(a0)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = None
    best_obj = float('inf')
    t0 = time.monotonic()
    
    for alpha in alpha_starts:
        if time.monotonic() - t0 > time_limit:
            break
        
        a = np.maximum(a0.astype(np.float64), 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        
        current_obj = fast_convolve(a).max()
        no_improve = 0
        gamma = gamma_range[1]
        
        while time.monotonic() - t0 < time_limit:
            time_per_alpha = time_limit / len(alpha_starts)
            if (time.monotonic() - t0) - (alpha_starts.index(alpha) * time_per_alpha) > time_per_alpha + 5:
                break
            
            A = np.fft.fft(a, fft_size)
            power = np.abs(A) ** 2
            mean_power = power.mean() + 1e-30
            
            # Compress magnitude
            phase = A / (np.abs(A) + 1e-30)
            new_mag = (power / mean_power) ** alpha
            new_A = new_mag * phase
            new_a = np.fft.ifft(new_A).real[:n]
            
            new_a = np.maximum(new_a, 0.0)
            s_new = new_a.sum()
            if s_new < 1e-15:
                # Fallback: less aggressive
                new_mag = (power / mean_power) ** (alpha * 0.5)
                new_A = new_mag * phase
                new_a = np.fft.ifft(new_A).real[:n]
                new_a = np.maximum(new_a, 0.0)
                s_new = new_a.sum()
                if s_new < 1e-15:
                    break
                new_a = new_a / s_new
            
            # Mix
            new_a = (1 - gamma) * a + gamma * (new_a / s_new)
            new_a = np.maximum(new_a, 0.0)
            new_a = new_a / new_a.sum()
            
            conv_new = fast_convolve(new_a)
            obj_new = conv_new.max()
            
            if obj_new < current_obj - 1e-12:
                current_obj = obj_new
                a = new_a
                no_improve = 0
                # Increase gamma slowly
                gamma = min(1.0, gamma * 1.01)
            else:
                no_improve += 1
                # Decrease gamma
                gamma = max(gamma_range[0], gamma * 0.95)
                
                if no_improve > 200:
                    # Add spectral noise
                    noise = rng.standard_normal(fft_size) * 0.01
                    A_noisy = A + noise

                    a_noisy = np.maximum(a_noisy, 0.0)
                    s_noisy = a_noisy.sum()
                    if s_noisy > 1e-10:
                        a = a_noisy / s_noisy
                    no_improve = 0
            
            if a is not None and current_obj < best_obj:
                best_obj = current_obj
                best_a = a.copy()
        
        # Check if we should continue
        time_remaining = time_limit - (time.monotonic() - t0)
        if time_remaining < 2:
            break
    
    if best_a is None:

        best_obj = fast_convolve(best_a).max()

def subgradient_peak_reduction(a0, time_limit, seed=0):
    """
    Subgradient method targeting the peak lags of the convolution.
    At each step, identify which lags are at the peak and push mass away
    from positions that contribute to those peaks.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Find peak lags (within 5% of max)
        threshold = max_val * 0.95
        peak_lags = np.where(conv >= threshold)[0]
        
        if len(peak_lags) == 0:
            break
        
        # Compute subgradient: for each peak lag k, the gradient of conv[k] w.r.t. a_i
        # conv[k] = sum_i a[i] * a[k-i] (for valid i)
        # d(conv[k])/d(a[j]) = a[k-j] + a[j-k] (accounting for both terms)
        # Actually: conv[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i]*a[k-i]
        # d(conv[k])/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
        # But since conv is symmetric (conv[k] = conv[2n-2-k]), we can simplify
        
        # Build gradient
        grad = np.zeros(n)
        for k in peak_lags:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                # d/d(a[i]) of a[i]*a[j] = a[j]
                # d/d(a[j]) of a[i]*a[j] = a[i]
                # But if i == j, it's 2*a[i]
                grad[i] += a[j]
                if j != i:
                    grad[j] += a[i]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm
        
        # Update: move away from peak-contributing positions
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            lr = min(0.05, lr * 1.02)
        else:
            no_improve += 1
            lr = max(0.001, lr * 0.95)
            if no_improve > 300:
                # Random restart from current best
                noise = rng.standard_normal(n) * 0.02
                a = project_simplex(best_a + noise)
                lr = 0.01
                no_improve = 0
        
        # Adaptive: if too many peak lags, be more conservative
        if len(peak_lags) > n // 2:
            lr *= 0.5
    
    return best_a, best_obj


def coordinate_descent_peaks(a0, time_limit, seed=0):
    """
    Coordinate descent: for each element, try to find the optimal value
    that reduces the peak convolution.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_val = conv.max()
        peak_lags = np.where(conv >= max_val * 0.98)[0]
        
        improved = False
        # Try adjusting elements that contribute to peaks
        for k in peak_lags[:10]:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            
            for i in range(i_start, i_end + 1):
                j = k - i
                if i == j:
                    continue
                # Try reducing a[i] and adding to a random position
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    amt = a[i] * frac
                    if amt < 1e-8:
                        break
                    # Find a good target: position far from peak contributions
                    # Try several targets
                    for target in [rng.integers(0, n), (i + n // 2) % n, (i + n // 3) % n]:
                        if target == i or target == j:
                            continue
                        a_try = a.copy()
                        a_try[i] -= amt
                        a_try[target] += amt
                        conv_try = fast_convolve(a_try)
                        if conv_try.max() < best_obj - 1e-10:
                            best_obj = conv_try.max()
                            best_a = a_try.copy()
                            a = a_try.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Global random perturbation
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                for frac in [0.03, 0.05, 0.1, 0.15]:
                    amt = min(a[i], a[j]) * frac
                    if amt < 1e-8:
                        break
                    a_try = a.copy()
                    if a[i] >= a[j]:
                        a_try[i] -= amt
                        a_try[j] += amt
                    else:
                        a_try[j] -= amt
                        a_try[i] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj - 1e-10:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()

def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Various shaped initializations
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    # Exponential decay
    exp_dec = np.exp(-x / (n * 0.3))
    exp_dec /= exp_dec.sum()
    inits['exp_dec'] = exp_dec
    
    # Gaussian (centered)
    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{int(sf*100)}'] = g
    
    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(2, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    cos_w = np.maximum(cos_w, 1e-8)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w
    
    # Kaiser-like
    for beta_k in [3, 5, 8]:
        r = x / (n - 1) * 2 - 1
        from scipy.special import i0
        kaiser = i0(beta_k * np.sqrt(np.maximum(1 - r**2, 0)))
        kaiser = np.maximum(kaiser, 1e-8)
        kaiser /= kaiser.sum()
        inits[f'kaiser_{beta_k}'] = kaiser
    
    # Random
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand3'] = rng.dirichlet(np.ones(n) * 3.0)
    
    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
    
    return inits


def smoothed_gradient_descent(a0, time_limit, seed=0, beta_init=50.0, beta_end=500.0,
                               lr_init=0.008, momentum=0.93):
    """
    Smoothed (log-sum-exp) gradient descent with Nesterov momentum.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac
        lr = lr_init * (1.0 - 0.6 * frac)
        lr = max(0.0002, lr)
        
        # Nesterov
        y = a + momentum * v
        conv_y = fast_convolve(y)
        
        # Smoothed gradient
        scaled = beta * conv_y
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv_y))
            p[np.argmax(conv_y)] = 1.0
        else:
            p /= p_sum
        
        w = conv_y * p
        
        # Cross-correlation for gradient
        fft_size = 1 << (3 * n - 1).bit_length()
        A = np.fft.rfft(y, fft_size)
        W = np.fft.rfft(w, fft_size)
        cross = np.fft.irfft(A * np.conj(W), fft_size)
        grad = 2.0 * cross[:n]
        
        v = momentum * v - lr * grad
        a = project_simplex(a + v)
        
        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 250:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            v *= 0.3
            no_improve = 0
    
    return best_a, best_obj

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using spectral flattening (iterative
magnitude compression in frequency domain), corrected subgradient descent on
peak lags, smoothed gradient descent with Nesterov momentum, and coordinate
polish. Explores multiple sequence lengths (up to 1024) with diverse
initializations and adaptive method switching.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_flatten(a0, time_limit, seed=0, alpha_starts=None, gamma_init=0.5):
    """
    Iterative spectral flattening: compress magnitude spectrum to reduce
    peak auto-convolution. Projects back to non-negative simplex each step.
    """
    if alpha_starts is None:
        alpha_starts = [0.3, 0.5, 0.7, 0.85]

    rng = np.random.default_rng(seed)
    n = len(a0)
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = None
    best_obj = float('inf')
    t0 = time.monotonic()

    for alpha in alpha_starts:
        time_per_alpha = time_limit / len(alpha_starts)
        t_alpha_start = time.monotonic()

        a = np.maximum(a0.astype(np.float64), 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

        current_obj = fast_convolve(a).max()
        no_improve = 0
        gamma = gamma_init

        while time.monotonic() - t_alpha_start < time_per_alpha + 2:
            if time.monotonic() - t0 > time_limit:
                break

            A = np.fft.fft(a, fft_size)
            power = np.abs(A) ** 2
            mean_power = power.mean() + 1e-30

            # Compress magnitude spectrum
            phase = A / (np.abs(A) + 1e-30)
            new_mag = (power / mean_power) ** alpha
            new_A = new_mag * phase
            new_a = np.fft.ifft(new_A).real[:n]

            new_a = np.maximum(new_a, 0.0)
            s_new = new_a.sum()
            if s_new < 1e-15:
                new_mag = (power / mean_power) ** (alpha * 0.4)
                new_A = new_mag * phase
                new_a = np.fft.ifft(new_A).real[:n]
                new_a = np.maximum(new_a, 0.0)
                s_new = new_a.sum()
                if s_new < 1e-15:
                    break
                new_a = new_a / s_new

            # Mix for stability
            mixed = (1.0 - gamma) * a + gamma * (new_a / s_new)
            mixed = np.maximum(mixed, 0.0)
            s_mixed = mixed.sum()
            if s_mixed < 1e-15:
                break
            mixed = mixed / s_mixed

            conv_new = fast_convolve(mixed)
            obj_new = conv_new.max()

            if obj_new < current_obj - 1e-12:
                current_obj = obj_new
                a = mixed
                no_improve = 0
                gamma = min(0.95, gamma * 1.005)
            else:
                no_improve += 1
                gamma = max(0.15, gamma * 0.97)
                if no_improve > 150:
                    # Spectral noise perturbation
                    noise_spec = (rng.standard_normal(fft_size) +
                                  1j * rng.standard_normal(fft_size)) * 0.005
                    A_noisy = A + noise_spec
                    a_noisy = np.fft.ifft(A_noisy).real[:n]
                    a_noisy = np.maximum(a_noisy, 0.0)
                    s_noisy = a_noisy.sum()
                    if s_noisy > 1e-10:
                        a = a_noisy / s_noisy
                        current_obj = fast_convolve(a).max()
                    no_improve = 0
                    gamma = 0.5

            if current_obj < best_obj:
                best_obj = current_obj
                best_a = a.copy()

    if best_a is None:
        best_a = a0.copy()
        best_obj = fast_convolve(best_a).max()

    return best_a, best_obj


def subgradient_peak(a0, time_limit, seed=0):
    """
    Subgradient method on the max of convolution. Computes exact subgradient
    by summing gradients over all active (peak) lags.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = 0.008
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_val = conv.max()
        threshold = max_val * 0.97
        peak_lags = np.where(conv >= threshold)[0]

        if len(peak_lags) == 0:
            break

        # Compute subgradient: grad[j] = 2 * sum_{k in peaks, 0<=k-j<n} a[k-j]
        grad = np.zeros(n)
        for k in peak_lags:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                grad[j] += 2.0 * a[k - j]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            lr = min(0.03, lr * 1.01)
        else:
            no_improve += 1
            lr = max(0.0005, lr * 0.96)
            if no_improve > 200:
                noise = rng.standard_normal(n) * 0.015
                a = project_simplex(best_a + noise)
                lr = 0.006
                no_improve = 0

    return best_a, best_obj


def lse_gradient_descent(a0, time_limit, seed=0, beta_init=40.0, beta_end=400.0,
                         lr_init=0.007, momentum=0.93):
    """
    Smoothed (log-sum-exp) Nesterov gradient descent on the auto-convolution peak.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac
        lr = max(0.0002, lr_init * (1.0 - 0.65 * frac))

        # Nesterov lookahead
        y = a + momentum * v
        conv_y = fast_convolve(y)

        # Log-sum-exp smoothed gradient
        scaled = beta * conv_y
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv_y))
            p[np.argmax(conv_y)] = 1.0
        else:
            p /= p_sum

        w = conv_y * p

        # Cross-correlation for gradient: grad[i] = 2 * sum_m y[m] * w[m+i]
        fft_size = 1 << (3 * n - 1).bit_length()
        A = np.fft.rfft(y, fft_size)
        W = np.fft.rfft(w, fft_size)
        cross = np.fft.irfft(A * np.conj(W), fft_size)
        grad = 2.0 * cross[:n]

        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 200:
            noise = rng.standard_normal(n) * 0.004
            a = project_simplex(a + noise)
            v *= 0.2
            no_improve = 0

    return best_a, best_obj


def coordinate_polish(a0, time_limit, seed=0):
    """
    Targeted coordinate descent: identify peak-contributing pairs and
    transfer mass to reduce the maximum convolution value.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_val = conv.max()
        threshold = max_val * 0.98
        peak_lags = np.where(conv >= threshold)[0]

        if len(peak_lags) == 0:
            break

        improved = False

        # For top peak lags, find contributing pairs and try to reduce them
        for k in peak_lags[:8]:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)

            # Find the pair with maximum product
            if j_end < j_start:
                continue
            products = np.array([a[j] * a[k - j] for j in range(j_start, j_end + 1)])
            if len(products) == 0:
                continue

            # Try reducing the largest contributor
            for attempt in range(3):
                idx = rng.choice(len(products), size=min(3, len(products)), replace=False)
                for ii in idx:
                    j = j_start + int(ii)
                    partner = k - j
                    if j == partner:
                        continue
                    # Reduce a[j], add to a random distant position
                    for frac in [0.05, 0.1, 0.2]:
                        amt = a[j] * frac
                        if amt < 1e-9:
                            break
                        for target in [rng.integers(0, n), (j + n // 2) % n,
                                       (j + n // 3) % n, (j + 2 * n // 3) % n]:
                            if target == j or target == partner:
                                continue
                            a_try = a.copy()
                            a_try[j] -= amt
                            a_try[int(target)] += amt
                            conv_try = fast_convolve(a_try)
                            if conv_try.max() < best_obj - 1e-10:
                                best_obj = conv_try.max()
                                best_a = a_try.copy()
                                a = a_try.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Global random pair move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 1e-6:
                for frac in [0.03, 0.05, 0.08, 0.12]:
                    amt = a[i] * frac
                    if amt < 1e-9:
                        break
                    a_try = a.copy()
                    a_try[i] -= amt
                    a_try[j] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj - 1e-10:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        break

    return best_a, best_obj


def gen_initializations(n, parent, rng):
    """Generate diverse initializations of length n (normalized to sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Gaussian shapes
    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{int(sf * 100)}'] = g

    # Exponential
    exp_d = np.exp(-x / (n * 0.4))
    exp_d /= exp_d.sum()
    inits['exp'] = exp_d

    # Bimodal
    bi = np.zeros(n)
    edge = max(2, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Cosine
    cos_w = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(n - 1, 1)))
    cos_w = np.maximum(cos_w, 1e-8)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w

    # Log-cosine (sharper edges)
    logcos = np.log(np.maximum(cos_w, 1e-6)) + 1.0
    logcos = np.maximum(logcos, 0.0)
    ls = logcos.sum()
    if ls > 1e-10:
        inits['logcos'] = logcos / ls

    # Random Dirichlet
    inits['rand_a'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand_b'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand_c'] = rng.dirichlet(np.ones(n) * 3.0)

    # Parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: spectral flattening + subgradient + LSE gradient +
    coordinate polish. Explores lengths 192-1024 with diverse initializations.
    """
    global height_sequence_1

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = evaluate_sequence(p.tolist())
            print(f"Parent: n={len(p)}, sum={p.sum():.2f}, score={parent_score:.6f}",
                  flush=True)
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
    except Exception as e:
        print(f"No parent: {e}", flush=True)

    # === Phase 1: Broad scan with spectral + gradient ===
    print("=== Phase 1: Broad Scan ===", flush=True)
    lengths = [192, 256, 320, 384, 512, 640, 768, 1024]

    phase1_budget = 450.0
    t_p1 = time.monotonic()
    scan_results = []

    for n in lengths:
        if time.monotonic() > deadline - 500:
            break
        if (time.monotonic() - t_p1) > phase1_budget:
            break

        inits = gen_initializations(n, parent, rng)
        # Priority: parent > structured > random
        priority = ['parent', 'parent_ds', 'parent_up', 'gauss_25', 'gauss_35',
                    'cosine', 'logcos', 'exp', 'uniform', 'rand_a']
        names = [nm for nm in priority if nm in inits][:5]

        time_per_n = phase1_budget / len(lengths)
        per_init = max(4.0, time_per_n / max(1, len(names)))

        for name in names:
            if time.monotonic() > deadline - 480:
                break
            t_alloc = min(per_init, 30.0)
            if t_alloc < 3:
                break

            a0 = inits[name]

            # Spectral flattening
            a_sp, _ = spectral_flatten(a0, t_alloc * 0.6,
                                       seed=rng.integers(0, 100000),
                                       alpha_starts=[0.4, 0.6, 0.8],
                                       gamma_init=0.5)
            score_sp = float(evaluate_sequence(a_sp.tolist()))

            # Quick gradient on spectral result
            a_gd, _ = lse_gradient_descent(a_sp, t_alloc * 0.4,
                                           seed=rng.integers(0, 100000),
                                           beta_init=30, beta_end=200,
                                           lr_init=0.008, momentum=0.92)
            score_gd = float(evaluate_sequence(a_gd.tolist()))

            best_local = a_sp if score_sp <= score_gd else a_gd
            best_local_score = min(score_sp, score_gd)

            if np.isfinite(best_local_score):
                scan_results.append((best_local_score, n, name, best_local.copy()))
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_sequence = best_local.copy()
                    print(f"  n={n}, {name}: {best_local_score:.6f}", flush=True)

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}", flush=True)

    # === Phase 2: Deep optimization on top candidates ===
    print("=== Phase 2: Deep Optimization ===", flush=True)

    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, a_opt))
            seen.add(key)
        if len(top_configs) >= 5:
            break

    # Always include current best
    if best_sequence is not None:
        top_configs.insert(0, (len(best_sequence), best_sequence.copy()))

    # Add a random restart at best length
    if best_sequence is not None:
        bn = len(best_sequence)
        top_configs.append((bn, rng.dirichlet(np.ones(bn) * 0.5)))

    remaining = deadline - time.monotonic() - 180.0
    n_configs = max(1, len(top_configs))
    per_config = max(15.0, remaining / n_configs)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 130:
            break
        t_alloc = min(per_config, 220.0)
        if t_alloc < 10:
            break

        print(f"  Config {i}: n={n}", flush=True)
        cur = a0.copy()

        # 1. Spectral (30%)
        t1 = t_alloc * 0.30
        if t1 > 5:
            cur, _ = spectral_flatten(cur, t1, seed=rng.integers(0, 100000),
                                       alpha_starts=[0.35, 0.55, 0.75],
                                       gamma_init=0.4)
            s1 = float(evaluate_sequence(cur.tolist()))
            if np.isfinite(s1) and s1 < best_score:
                best_score = s1
                best_sequence = cur.copy()
                print(f"    Spectral: {s1:.6f}", flush=True)

        # 2. Subgradient (25%)
        t2 = t_alloc * 0.25
        if t2 > 5:
            cur, _ = subgradient_peak(cur, t2, seed=rng.integers(0, 100000))
            s2 = float(evaluate_sequence(cur.tolist()))
            if np.isfinite(s2) and s2 < best_score:
                best_score = s2
                best_sequence = cur.copy()
                print(f"    SubGrad: {s2:.6f}", flush=True)

        # 3. LSE Gradient (25%)
        t3 = t_alloc * 0.25
        if t3 > 5:
            cur, _ = lse_gradient_descent(cur, t3, seed=rng.integers(0, 100000),
                                           beta_init=50, beta_end=400,
                                           lr_init=0.006, momentum=0.94)
            s3 = float(evaluate_sequence(cur.tolist()))
            if np.isfinite(s3) and s3 < best_score:
                best_score = s3
                best_sequence = cur.copy()
                print(f"    GradDesc: {s3:.6f}", flush=True)

        # 4. Spectral again (20%)
        t4 = t_alloc * 0.20
        if t4 > 5:
            cur, _ = spectral_flatten(cur, t4, seed=rng.integers(0, 100000),
                                       alpha_starts=[0.5, 0.7, 0.9],
                                       gamma_init=0.3)
            s4 = float(evaluate_sequence(cur.tolist()))
            if np.isfinite(s4) and s4 < best_score:
                best_score = s4
                best_sequence = cur.copy()
                print(f"    Spec2: {s4:.6f}", flush=True)

    # === Phase 3: Polish ===
    print("=== Phase 3: Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0

        # Coordinate descent
        t_cd = remaining * 0.35
        if t_cd > 8:
            a_cd, _ = coordinate_polish(best_sequence, t_cd, seed=777)
            s_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(s_cd) and s_cd < best_score:
                best_score = s_cd
                best_sequence = a_cd.copy()
                print(f"  Coord: {s_cd:.6f}", flush=True)

        # Fine spectral
        remaining = deadline - time.monotonic() - 10.0
        t_fs = remaining * 0.30
        if t_fs > 5:
            a_fs, _ = spectral_flatten(best_sequence, t_fs, seed=999,
                                        alpha_starts=[0.4, 0.6, 0.8, 0.95],
                                        gamma_init=0.3)
            s_fs = float(evaluate_sequence(a_fs.tolist()))
            if np.isfinite(s_fs) and s_fs < best_score:
                best_score = s_fs
                best_sequence = a_fs.copy()
                print(f"  FineSpec: {s_fs:.6f}", flush=True)

        # Subgradient final
        remaining = deadline - time.monotonic() - 8.0
        t_sg = remaining * 0.35
        if t_sg > 5:
            a_sg, _ = subgradient_peak(best_sequence, t_sg, seed=555)
            s_sg = float(evaluate_sequence(a_sg.tolist()))
            if np.isfinite(s_sg) and s_sg < best_score:
                best_score = s_sg
                best_sequence = a_sg.copy()
                print(f"  FinalSub: {s_sg:.6f}", flush=True)

        # Final coordinate
        remaining = deadline - time.monotonic() - 3.0
        if remaining > 3:
            a_cf, _ = coordinate_polish(best_sequence, remaining, seed=111)
            s_cf = float(evaluate_sequence(a_cf.tolist()))
            if np.isfinite(s_cf) and s_cf < best_score:
                best_score = s_cf
                best_sequence = a_cf.copy()
                print(f"  FinalCoord: {s_cf:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Critical fallback
        best_sequence = np.ones(192) / 192
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
