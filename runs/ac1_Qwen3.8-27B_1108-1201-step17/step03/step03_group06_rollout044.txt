def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

        # This is the cross-correlation evaluated at lag i

        A = np.fft.fft(a, fft_size)

        term2 = np.fft.ifft(AI * A).real[:n]
        
        subgrad = (2.0 / len(active)) * (term1 + term2)

    return best_a, best_obj

Alternatively, I can use the quadratic model:
`conv(a_new, a_new) ≈ (1-t)^2 * conv(a,a) + 2t(1-t) * conv(a, b/S) + t^2 * conv(b/S, b/S)`

The max of this is approximately:
`(1-t)^2 * M + 2t(1-t) * (M/S) + t^2 * M_b`

where `M_b = max(conv(b/S, b/S))`.

The optimal `t` minimizes this quadratic:
`f(t) = M(1-t)^2 + 2(M/S)t(1-t) + M_b * t^2`
`= M(1-2t+t^2) + 2(M/S)(t-t^2) + M_b * t^2`
`= M - 2Mt + Mt^2 + 2Mt/S - 2Mt^2/S + M_b * t^2`
`= M + t(-2M + 2M/S) + t^2(M - 2M/S + M_b)`

`f'(t) = -2M + 2M/S + t*2(M - 2M/S + M_b) = 0`
`t = (2M - 2M/S) / (2(M - 2M/S + M_b))`
`= (M - M/S) / (M - 2M/S + M_b)`
`= M(S-1)/S / ((M(S-2) + M_b*S)/S)`
`= M(S-1) / (M(S-2) + M_b*S)`

Let me simplify: let `r = M_b / M` (ratio of b's max convolution to a's).
`f'(t) = 0` gives `t = (1 - 1/S) / (1 - 2/S + r)`
`= (S-1) / (S-2+r*S)`

If `S = 2` and `r = 1`: `t = 1 / (0+2) = 0.5`
If `S = 3` and `r = 1`: `t = 2 / (1+3) = 0.5`
If `S = 10` and `r = 0.5`: `t = 9 / (8+5) = 9/13 ≈ 0.69`

So the optimal step is typically in [0.3, 0.8]. I'll clamp it to [0.05, 1.0].

"""
Improved optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive active-set LP, mirror descent, and smooth gradient methods.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp_matrix_a_rows(a, indices):
    n = len(a)
    rows = np.zeros((len(indices), n))
    for row_i, k in enumerate(indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min > i_max:
            continue
        idx = np.arange(i_min, i_max + 1)
        rows[row_i, idx] = a[k - idx]
    return rows


def solve_lp(a, M, active_idx):
    n = len(a)
    A_ub = build_lp_matrix_a_rows(a, active_idx)
    b_ub = np.full(len(active_idx), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def mirror_descent_step(a, conv, eta):
    n = len(a)
    M = conv.max()
    threshold = M * 0.9
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    # Compute subgradient
    active_ind = np.zeros(2*n-1)
    active_ind[active] = 1.0
    
    fft_size = 1 << (2*n).bit_length()
    AI = np.fft.fft(active_ind, fft_size)
    A = np.fft.fft(a, fft_size)
    
    term1 = np.fft.ifft(A * np.conj(AI)).real[:n]
    term2 = np.fft.ifft(AI * A).real[:n]
    
    subgrad = (2.0 / len(active)) * (term1 + term2)
    
    # Mirror descent update
    log_a = np.log(a + 1e-30)
    log_a -= eta * subgrad
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    a_new = a_new / a_new.sum()
    return a_new
