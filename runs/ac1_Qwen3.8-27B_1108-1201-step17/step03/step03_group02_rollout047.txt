"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using aggressive LP-based iterative
refinement with line search, active-set detection, multi-length exploration,
and smooth optimization polishing. Key improvements over previous approach:
- Line search over blend parameter instead of fixed step

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = a[k - j_start:k - j_end + 1][::-1]
    return A


def solve_min_peak_lp(a, M_target=None):
    """
    Solve: min M s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M for all k.
    Returns (b, M_opt, dual_vars) or (None, None, None) on failure.
    """
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        b = np.maximum(res.x[:n], 0)
        M_opt = max(0.0, res.x[n])
        duals = -res.ineqlin.marginals if res.ineqlin.marginals is not None else None
        return b, M_opt, duals
    return None, None, None


def solve_max_sum_lp(a, M):
    """
    Solve: max sum(b) s.t. b>=0, conv(a,b)[k] <= M for all k.
    Returns b (normalized to sum=1) or None.
    """
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        b = np.maximum(res.x, 0)
        s = b.sum()
        if s > 1e-15:
            return b / s
    return None


def lp_iteration(a, method='min_peak'):
    """
    Perform one LP-based iteration. Returns (b, objective_info).
    a should be normalized to sum=1.
    """
    conv = fast_convolve(a)
    M_current = conv.max()

    if method == 'min_peak':
        b, M_new, duals = solve_min_peak_lp(a)
        if b is None:
            return None, M_current
        return b, M_new
    else:
        b = solve_max_sum_lp(a, M_current)
        if b is None:
            return None, M_current
        return b, None


def line_search_blend(a, b, n, num_trials=8):
    """
    Find best blend factor t in a_new = (1-t)*a + t*b by evaluating objective.
    Returns (a_new, best_t, best_obj).
    """
    conv_a = fast_convolve(a)
    obj_a = 2.0 * n * conv_a.max()  # sum=1 so denominator is 1

    best_obj = obj_a
    best_t = 0.0
    best_a = a.copy()

    # Try logarithmically spaced trial values
    ts = np.array([0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0])
    for t in ts:
        a_new = (1.0 - t) * a + t * b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_t = t
            best_a = a_new.copy()

    return best_a, best_t, best_obj


def run_lp_search(a0, time_limit, method='min_peak', max_iters=20000):
    """
    Run LP-based iterative refinement with line search.
    Returns (best_a, best_obj).
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    current_method = method
    consecutive_fail = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b, M_info = lp_iteration(a, method=current_method)
        if b is None:
            consecutive_fail += 1
            if consecutive_fail > 5:
                current_method = 'max_sum' if current_method == 'min_peak' else 'min_peak'
                consecutive_fail = 0
            continue
        consecutive_fail = 0

        a_new, t_used, obj_new = line_search_blend(a, b, n)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # If line search didn't improve, try a very small step
            if t_used == 0.0:
                a_new = (1.0 - 0.01) * a + 0.01 * b
                a_new = np.maximum(a_new, 0.0)
                s2 = a_new.sum()
                if s2 > 1e-15:
                    a_new = a_new / s2
                    conv_new = fast_convolve(a_new)
                    obj_new = 2.0 * n * conv_new.max()
                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        no_improve = 0
                        a = a_new
                        continue

        a = a_new
        conv = fast_convolve(a)

        # Periodically switch methods
        if no_improve > 200:
            current_method = 'max_sum' if current_method == 'min_peak' else 'min_peak'
            no_improve = 0
        if no_improve > 1000:
            break

    return best_a, best_obj


def smooth_objective_and_grad(a, temperature):
    """
    Compute log-sum-exp smoothed objective and gradient.
    Objective (with sum(a)=1): 2*n * (1/t) * log(sum(exp(t * conv(a,a))))
    """
    n = len(a)
    conv = fast_convolve(a)
    scaled = temperature * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    obj = 2.0 * n * (1.0 / temperature) * (np.log(sum_exp) + temperature * conv.max() - scaled.max())
    # Actually let's just compute it directly:
    obj = 2.0 * n / temperature * np.log(np.exp(temperature * conv).sum()) if temperature * conv.max() < 500 else 2.0 * n * conv.max()

    # Recompute properly
    scaled2 = temperature * conv
    max_scaled = scaled2.max()
    exp_vals2 = np.exp(scaled2 - max_scaled)
    sum_exp2 = exp_vals2.sum()
    obj = 2.0 * n / temperature * (max_scaled + np.log(sum_exp2))

    p = exp_vals2 / sum_exp2
    grad_a = np.zeros(n)
    for j in range(n):
        k_lo = j
        k_hi = min(2 * n - 2, j + n - 1)
        if k_lo <= k_hi:
            grad_a[j] = (2.0 * n) * np.dot(p[k_lo:k_hi + 1], a[k_lo - j:k_hi - j + 1])
    return obj, grad_a


def smooth_gradient_descent(a0, time_limit, lr_init=0.002, momentum=0.9,
                            temp_init=50.0, temp_end=500.0):
    """Projected gradient descent on smoothed objective."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    total_iters = 5000

    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Schedule temperature
        progress = it / total_iters
        temp = temp_init * (temp_end / temp_init) ** progress

        _, grad = smooth_objective_and_grad(a, temp)
        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:

            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            lr *= 0.5
            no_improve = 0
            v *= 0.5
        if no_improve > 3000:

    return best_a, best_obj


def subgradient_refine(a0, time_limit, lr_init=0.02, momentum=0.9, top_k=50,
                       temp=0.001):
    """Projected subgradient with focused peaks."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    rng = np.random.default_rng(999)
    perturb_every = 2000

    while time.monotonic() - t0 < time_limit:
        # Compute subgradient from top peaks
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))
        peak_vals = conv[peaks]
        scaled_pv = peak_vals / temp
        scaled_pv -= scaled_pv.max()
        w = np.exp(scaled_pv)
        w /= w.sum()

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            perturb_every = int(perturb_every * 1.4)
            lr = max(lr_init * 0.2, lr * 0.4)

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit, max_rounds=500):
    """
    Coordinate descent: for each coordinate, find the optimal value
    that minimizes the objective (given other coordinates fixed).
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for round_num in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        # Process coordinates in random order
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-15:
                continue

            # conv(a,a) changes when a[j] changes:
            # conv[k] depends on a[j] through terms a[j]*a[k-j] and a[k-j]*a[j]
            # So d(conv[k])/d(a[j]) = a[k-j] + a[j-k] (where valid)
            # For the peak, we want to find a[j] that minimizes max(conv)

            # Binary search for optimal a[j] in [0, old_val * 3]
            # Actually, let's try a few candidate values
            candidates = [old_val * 0.5, old_val * 0.7, old_val * 0.85,
                          old_val, old_val * 1.15, old_val * 1.3]
            candidates = [max(0.0, c) for c in candidates]

            best_local_obj = best_obj  # We want to not make things worse
            best_local_val = old_val

            for cv in candidates:
                if cv == old_val:
                    continue
                a[j] = cv
                # Adjust to maintain sum=1 (approximately)
                # Actually for coordinate descent on simplex, we should
                # redistribute the difference
                diff = old_val - cv
                if abs(diff) < 1e-15:
                    a[j] = old_val
                    continue
                # Redistribute diff to other coordinates proportionally
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                others = a[mask]
                if others.sum() < 1e-15:
                    a[j] = old_val
                    continue
                a[mask] = others + diff * others / others.sum()
                a[np.where(a < 0)[0]] = 0
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a / s_new
                    conv_new = fast_convolve(a)
                    obj_new = 2.0 * n * conv_new.max()
                    if obj_new < best_local_obj - 1e-14:
                        best_local_obj = obj_new
                        best_local_val = cv
                        if obj_new < best_obj - 1e-12:
                            best_obj = obj_new
                            best_a = a.copy()
                a[j] = old_val  # restore
            # Actually this approach is too slow (O(n * candidates * conv_time) per round)
            # Let's break after a few rounds
        if round_num > 50:
            break

    return best_a, best_obj


def interpolate_sequence(parent, new_n):
    """Interpolate a sequence to a different length."""
    if len(parent) == new_n:
        return parent.copy()
    old_n = len(parent)
    if old_n == 1:
        return np.full(new_n, parent[0])
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, parent)

        # Run multiple rounds with alternating methods
        for method in ['min_peak', 'max_sum', 'min_peak', 'min_peak']:
            if time.monotonic() > phase2_deadline:
                break
            t_alloc = min(120.0, phase2_deadline - time.monotonic() - 10)
            if t_alloc < 10:
                break
            print(f"  Running {method} for {t_alloc:.0f}s...")
            a_opt, obj = run_lp_search(best_sequence, time_limit=t_alloc,
                                       method=method, max_iters=50000)
            try_update(a_opt)

    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t2:.0f}s")

    # Phase 3: Perturbation and re-optimization
    print("\n=== Phase 3: Perturbation ===")
    t3 = time.monotonic()

    if best_sequence is not None:
        phase3_deadline = deadline - 250
        n_best = len(best_sequence)
        for scale in [0.001, 0.005, 0.002, 0.01]:
            if time.monotonic() > phase3_deadline:
                break
            for trial in range(3):
                if time.monotonic() > phase3_deadline:
                    break
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(best_sequence + noise)
                t_alloc = min(30.0, phase3_deadline - time.monotonic())
                if t_alloc < 5:
                    break
                a_opt, _ = run_lp_search(a_pert, time_limit=t_alloc,
                                         method='min_peak', max_iters=5000)
                try_update(a_opt)

    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3:.0f}s")

    # Phase 4: Smooth gradient polishing
    print("\n=== Phase 4: Smooth Gradient Polish ===")
    t4 = time.monotonic()

    if best_sequence is not None:
        phase4_deadline = deadline - 120
        for temp_init in [30.0, 80.0]:
            if time.monotonic() > phase4_deadline:
                break
            t_alloc = min(60.0, phase4_deadline - time.monotonic() - 5)
            if t_alloc < 10:
                break
            a_opt, _ = smooth_gradient_descent(
                best_sequence, time_limit=t_alloc,
                lr_init=0.001, momentum=0.92,
                temp_init=temp_init, temp_end=300.0)
            try_update(a_opt)

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t4:.0f}s")

    # Phase 5: Subgradient refinement
    print("\n=== Phase 5: Subgradient Refinement ===")
    t5 = time.monotonic()

    if best_sequence is not None:
        phase5_deadline = deadline - 60
        n_best = len(best_sequence)
        for lr in [0.01, 0.02, 0.005]:
            if time.monotonic() > phase5_deadline:
                break
            t_alloc = min(30.0, phase5_deadline - time.monotonic() - 5)
            if t_alloc < 5:
                break
            a_opt, _ = subgradient_refine(
                best_sequence, time_limit=t_alloc,
                lr_init=lr, momentum=0.92,
                top_k=max(20, n_best // 16), temp=0.0005)
            try_update(a_opt)

    print(f"Phase 5 done: best={best_score:.6f}, elapsed={time.monotonic()-t5:.0f}s")

    # Phase 6: Final fine LP iterations
    print("\n=== Phase 6: Final Fine LP ===")
    t6 = time.monotonic()

    if best_sequence is not None:
        while time.monotonic() < deadline - 10:
            t_alloc = min(20.0, deadline - time.monotonic() - 5)
            if t_alloc < 3:
                break
            a_opt, _ = run_lp_search(best_sequence, time_limit=t_alloc,
                                     method='min_peak', max_iters=10000)
            old_best = best_score
            try_update(a_opt)
            if best_score >= old_best - 1e-10:
                # Also try max_sum
                a_opt2, _ = run_lp_search(best_sequence, time_limit=t_alloc,
                                          method='max_sum', max_iters=10000)
                try_update(a_opt2)

    print(f"Phase 6 done: best={best_score:.6f}, elapsed={time.monotonic()-t6:.0f}s")

    # Final evaluation
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
