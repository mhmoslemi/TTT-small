```
def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0: return np.zeros(0)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    c = np.fft.irfft(A * A, fsize)[:size]
    # clean tiny imaginary and negative
    c = np.where(c > 0.0, c, 0.0)
    return c
```

```
def project_simplex(v):
    v=np.asarray(v,float)
    n=v.size
    if n==0: return v
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    cond = u * np.arange(1,n+1) > (cssv - 1.0)
    if not cond.any():
       return np.ones(n)/n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0)/(rho+1.0)
    return np.maximum(v - theta, 0.0)
```

```python
"""
Searches for non-negative step-function heights minimizing

    2*n*max_k conv(a,a)_k / (sum(a))^2

using a hybrid of:
  - structured initializations (endpoint-singular, cosine, Gaussian, anti-palindromic),
  - projected subgradient methods on softened peak convolutions,
  - L-BFGS-B polishing in log-softmax variables using p-norm/log-sum smoothing,
  - LP-based refinement (minimize max(a*b) with sum b=1, b>=0).

The final answer is always the best sequence observed under evaluate_sequence.
"""

import time
import numpy as np
from scipy.optimize import linprog, minimize


class LfgTimeStop(Exception):
    pass


# Global state for L-BFGS callback (avoids nested functions/closures).
_LFG_P = 2.0
_LFG_N = 1
_LFG_DEADLINE = 0.0
_LFG_STOP = False
_LFG_LAST_X = None


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    a = np.maximum(a, 0.0)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    c = np.fft.irfft(A * A, fsize)[:size]
    c = np.where(c > 0.0, c, 0.0)
    return c


def local_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-15:
        return np.inf
    a = a / s
    c = fast_convolve(a)
    if c.size == 0:
        return np.inf
    return float(2.0 * a.size * float(np.max(c)))


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n, dtype=np.float64) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def corr_gradient(a, w):
    """
    Gradient term:
        g[j] = 2 * sum_m w[j+m] * a[m]
    where valid indices are handled.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    L = 2 * n - 1
    if n == 0 or L <= 0:
        return np.zeros_like(a)
    m = L + n - 1
    fsize = 1 << (m - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    Wr = np.fft.rfft(w[::-1], fsize)
    d = np.fft.irfft(A * Wr, fsize)
    return 2.0 * d[n - 1:L][::-1]


def softmax_from_x(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    n = x.size
    if n == 0:
        return x
    x = np.where(np.isfinite(x), x, 0.0)
    mx = float(np.max(x))
    e = np.exp(x - mx)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / n
    return e / s


def softmax_jacobian(g, a):
    a = np.asarray(a, dtype=np.float64).ravel()
    g = np.asarray(g, dtype=np.float64).ravel()
    return a * (g - float(np.dot(g, a)))


def obj_grad_p(a, p):
    """
    Smooth objective F_p(a) = (sum_k c_k^p)^(1/p), c = conv(a,a).
    Returns (F_p, grad_a, max_c).
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = fast_convolve(a)
    if c.size == 0:
        return 0.0, np.zeros(n, dtype=np.float64), 0.0
    cmax = float(np.max(c))
    if cmax <= 1e-300:
        return 0.0, np.zeros(n, dtype=np.float64), 0.0

    z = c / cmax
    p = float(p)

    if p <= 24.0:
        zp = z ** p
        sp = float(np.sum(zp))
        if sp <= 1e-300 or not np.isfinite(sp):
            d = np.zeros_like(z, dtype=np.float64)
            d[int(np.argmax(z))] = 1.0
            F = cmax
        else:
            F = cmax * (sp ** (1.0 / p))
            d = (sp ** ((1.0 - p) / p)) * (z ** (p - 1.0))
    else:
        zsafe = np.maximum(z, 1e-300)
        logz = np.log(zsafe)
        plog = p * logz
        maxp = float(np.max(plog))
        if not np.isfinite(maxp):
            d = np.zeros_like(z, dtype=np.float64)
            d[int(np.argmax(z))] = 1.0
            F = cmax
        else:
            exps = np.exp(plog - maxp)
            sumexp = float(np.sum(exps))
            if sumexp <= 0.0 or not np.isfinite(sumexp):
                d = np.zeros_like(z, dtype=np.float64)
                d[int(np.argmax(z))] = 1.0
                F = cmax
            else:
                log_sp = maxp + float(np.log(sumexp))
                F = cmax * float(np.exp(log_sp / p))
                log_d = (p - 1.0) * logz + ((1.0 - p) / p) * log_sp
                d = np.exp(np.clip(log_d, -745.0, 700.0))
                d = np.where(np.isfinite(d), d, 0.0)

    d = np.maximum(d, 0.0)
    g = corr_gradient(a, d)
    return float(F), g, cmax


def _lbfg_fun(x):
    global _LFG_LAST_X, _LFG_STOP
    arr = np.asarray(x, dtype=np.float64).ravel()
    _LFG_LAST_X = arr.copy()
    if _LFG_STOP or time.monotonic() > _LFG_DEADLINE:
        _LFG_STOP = True
        raise LfgTimeStop()
    a = softmax_from_x(arr)
    F, _, _ = obj_grad_p(a, _LFG_P)
    return F


def _lbfg_grad(x):
    global _LFG_LAST_X, _LFG_STOP
    arr = np.asarray(x, dtype=np.float64).ravel()
    _LFG_LAST_X = arr.copy()
    if _LFG_STOP or time.monotonic() > _LFG_DEADLINE:
        _LFG_STOP = True
        raise LfgTimeStop()
    a = softmax_from_x(arr)
    _, g, _ = obj_grad_p(a, _LFG_P)
    return softmax_jacobian(g, a)


def _lbfg_cb(xk):
    global _LFG_STOP
    if _LFG_STOP or time.monotonic() > _LFG_DEADLINE:
        _LFG_STOP = True
    return bool(_LFG_STOP)


def optimize_lbfgs(a0, p, time_limit, maxiter=800):
    global _LFG_P, _LFG_N, _LFG_DEADLINE, _LFG_STOP, _LFG_LAST_X

    a = project_simplex(np.asarray(a0, dtype=np.float64).ravel())
    n = a.size
    if n == 0 or time_limit <= 0.0:
        return a

    _LFG_P = float(p)
    _LFG_N = int(n)
    _LFG_DEADLINE = time.monotonic() + max(0.1, float(time_limit))
    _LFG_STOP = False
    _LFG_LAST_X = None

    x0 = np.log(np.maximum(a, 1e-300))
    x0 -= float(np.max(x0))
    _LFG_LAST_X = x0.copy()

    bounds = [(-30.0, 30.0)] * n
    try:
        res = minimize(
            _lbfg_fun,
            x0,
            jac=_lbfg_grad,
            method="L-BFGS-B",
            bounds=bounds,
            callback=_lbfg_cb,
            options={
                "maxiter": int(maxiter),
                "ftol": 1e-12,
                "gtol": 1e-8,
                "maxls": 20,
            },
        )
        if _LFG_LAST_X is not None:
            xopt = _LFG_LAST_X
        else:
            xopt = x0
    except LfgTimeStop:
        xopt = _LFG_LAST_X if _LFG_LAST_X is not None else x0
    except Exception:
        xopt = _LFG_LAST_X if _LFG_LAST_X is not None else x0

    _LFG_STOP = False
    a = softmax_from_x(xopt)
    a = project_simplex(a)
    return a


def optimize_subgrad(a0, time_limit, mode="soft", temp_frac=0.02, top_k=30,
                     lr_init=None, mom=0.88):
    a = project_simplex(np.asarray(a0, dtype=np.float64).ravel())
    n = a.size
    L = 2 * n - 1
    if n == 0 or time_limit <= 0.0 or L <= 0:
        return a

    if lr_init is None:
        lr_init = max(0.02, 1.0 / np.sqrt(float(n)))
    lr = float(lr_init)
    v = np.zeros(n, dtype=np.float64)
    best = a.copy()
    best_local = local_score(a)
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(20240517)

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        M = float(np.max(c)) if c.size else 0.0
        cur = 2.0 * n * M
        if cur < best_local:
            best_local = cur
            best = a.copy()

        if mode == "top":
            k = int(top_k)
            if k <= 0:
                k = 10
            k = min(k, L)
            if k == 1:
                w = np.zeros(L, dtype=np.float64)
                w[int(np.argmax(c))] = 1.0
            elif k >= L:
                T = max(float(temp_frac) * M, 1e-15)
                scaled = c / T
                scaled -= float(np.max(scaled))
                ew = np.exp(scaled)
                s = float(np.sum(ew))
                w = ew / s if s > 0 else (np.zeros(L, dtype=np.float64).__setitem__(int(np.argmax(c)), 1.0) or Ow)
            else:
                peaks = np.argpartition(c, -k)[-k:]
                vals = c[peaks]
                T = max(float(temp_frac) * M, 1e-15)
                scaled = vals / T
                scaled -= float(np.max(scaled))
                ew = np.exp(scaled)
                s = float(np.sum(ew))
                if s <= 0.0:
                    w_top = np.zeros_like(ew)
                    w_top[int(np.argmax(ew))] = 1.0
                else:
                    w_top = ew / s
                w = np.zeros(L, dtype=np.float64)
                w[peaks] = w_top
        else:
            T = max(float(temp_frac) * M, 1e-15)
            scaled = c / T
            scaled -= float(np.max(scaled))
            ew = np.exp(scaled)
            s = float(np.sum(ew))
            if s <= 0.0 or not np.isfinite(s):
                w = np.zeros(L, dtype=np.float64)
                w[int(np.argmax(c))] = 1.0
            else:
                w = ew / s

        g = corr_gradient(a, w)
        old_v = v.copy()
        v = mom * old_v - lr * g
        an = project_simplex(a + v)
        cn = fast_convolve(an)
        new = 2.0 * n * (float(np.max(cn)) if cn.size else 0.0)

        if new < cur - 1e-12 or new < best_local - 1e-8:
            a = an
            no_improve = 0
            if new < best_local:
                best_local = new
                best = a.copy()
            lr = min(0.30, lr * 1.04)
        else:
            v = old_v
            no_improve += 1
            lr *= 0.72
            if no_improve > 35:
                a = project_simplex(best + rng.standard_normal(n) * 0.002)
                v = np.zeros(n, dtype=np.float64)
                lr = float(lr_init)
                no_improve = 0

    cur = local_score(a)
    if cur < best_local:
        best = a.copy()
    return best


def build_conv_matrix(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for k in range(L):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            start = k - je
            stop = k - js + 1
            A[k, js:je + 1] = a[start:stop][::-1]
    return A


def lp_min_peak_A(A, hi_time_limit=None):
    n = int(A.shape[1])
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    A_ub = np.zeros((A.shape[0], n + 1), dtype=np.float64)
    A_ub[:, :n] = A
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(A.shape[0], dtype=np.float64)

    bounds = [(0.0, None)] * n + [(0.0, None)]
    opts = {}
    if hi_time_limit is not None and hi_time_limit > 0.0:
        opts["time_limit"] = float(max(1.0, float(hi_time_limit)))

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options=opts,
        )
    except Exception:
        return None, None

    if res.status != 0:
        return None, None

    b = np.maximum(res.x[:n], 0.0)
    s = float(np.sum(b))
    if s <= 1e-15:
        return None, None
    b = b / s
    return b, float(res.x[n])


def optimize_lp_refine(a0, time_limit, max_iters=120):
    a = project_simplex(np.asarray(a0, dtype=np.float64).ravel())
    n = a.size
    if n == 0 or time_limit <= 0.0:
        return a

    best = a.copy()
    best_local = local_score(a)
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(91827)

    for _ in range(int(max_iters)):
        if time.monotonic() - t0 >= time_limit:
            break

        remaining = time_limit - (time.monotonic() - t0)
        if remaining <= 1.0:
            break

        A = build_conv_matrix(a)
        b, _ = lp_min_peak_A(A, hi_time_limit=min(30.0, remaining * 0.8))
        if b is None:
            break

        candidates = [a]
        for t in (0.05, 0.10, 0.25, 0.50, 0.75, 1.00):
            cand = (1.0 - t) * a + t * b
            cand = project_simplex(cand)
            candidates.append(cand)

        cur_local = local_score(a)
        best_c = a
        best_c_local = cur_local
        for cand in candidates:
            sc = local_score(cand)
            if sc < best_c_local:
                best_c = cand
                best_c_local = sc

        if best_c_local < best_local - 1e-9:
            a = best_c
            best = a.copy()
            best_local = best_c_local
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                pert = rng.standard_normal(n) * 0.003
                a = project_simplex(a + pert)
                no_improve = 0
            if no_improve > 80:
                break

    cur = local_score(a)
    if cur < best_local:
        best = a.copy()
    return best


def add_candidate(cands, arr):
    if arr is None:
        return
    a = np.asarray(arr, dtype=np.float64).ravel()
    if a.size == 0:
        return
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s > 1e-15:
        cands.append(a / s)


def resample_sequence(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if a.size == 0 or s <= 1e-15:
        return np.ones(n, dtype=np.float64) / max(1, n)
    a = a / s
    if a.size == n:
        return a
    if a.size == 1:
        return np.full(n, float(a[0]), dtype=np.float64)

    out = np.interp(
        np.linspace(0.0, 1.0, n),
        np.linspace(0.0, 1.0, a.size),
        a,
    )
    out = np.maximum(out, 0.0)
    ts = float(np.sum(out))
    if ts <= 1e-15:
        return np.ones(n, dtype=np.float64) / n
    return out / ts


def get_parent_sequence():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        return None
    if p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 1e-8:
        return None
    return p / s


def generate_initializations(n, parent, rng):
    cands = []
    n = int(n)
    if n <= 0:
        return cands

    x = np.arange(n, dtype=np.float64)
    add_candidate(cands, np.ones(n, dtype=np.float64))

    if parent is not None:
        p = np.asarray(parent, dtype=np.float64).ravel()
        if p.size > 0:
            if p.size == n:
                add_candidate(cands, p)
            else:
                add_candidate(cands, resample_sequence(p, n))

    if n > 1:
        t = x / float(n - 1)
    else:
        t = np.array([0.5], dtype=np.float64)

    # Endpoint-singular power laws.
    if n > 1:
        left = x + 1.0
        right = float(n) - x
        for al in (0.05, 0.10, 0.15, 0.20, 0.25, 0.30, 0.35, 0.40,
                   0.45, 0.50, 0.60, 0.70, 0.80, 0.90, 1.00, 1.15, 1.30):
            add_candidate(cands, left ** (-al) * right ** (-al))

        for al in (0.20, 0.45, 0.70, 1.00, 1.30):
            for be in (0.20, 0.45, 0.70, 1.00, 1.30):
                add_candidate(cands, left ** (-al) * right ** (-be))

        # Geometric endpoint decay.
        for r in (0.60, 0.70, 0.80, 0.85, 0.90, 0.95):
            add_candidate(cands, r ** x + r ** (n - 1 - x))

    # Polynomial / cosine shapes that raise endpoints and lower center.
    for g in (0.5, 1.0, 2.0, 4.0, 8.0):
        add_candidate(cands, 1.0 + g * (2.0 * t - 1.0) ** 2)

    phase = 2.0 * np.pi * t
    for amp in (0.20, 0.40, 0.60, 0.80, 0.95):
        add_candidate(cands, 1.0 + amp * np.cos(phase))
        add_candidate(cands, 1.0 + amp * np.cos(2.0 * phase))
        add_candidate(cands, 1.0 + amp * np.cos(np.pi * t + 0.25 * np.pi))

    # End Gaussians and one/two bumps.
    for w in (0.03, 0.06, 0.10, 0.15, 0.22):
        e0 = np.exp(-0.5 * (x / max(1.0, w * float(n))) ** 2)
        e1 = np.exp(-0.5 * ((float(n - 1) - x) / max(1.0, w * float(n))) ** 2)
        add_candidate(cands, e0)
        add_candidate(cands, e1)
        add_candidate(cands, e0 + e1)
        add_candidate(cands, 0.7 * e0 + 0.3 * e1)
        add_candidate(cands, 0.3 * e0 + 0.7 * e1)

    # Random Dirichlet bases.
    try:
        for alpha in (0.3, 0.7, 1.0, 2.0):
            add_candidate(cands, rng.dirichlet(np.full(n, alpha)))
    except Exception:
        pass

    # Upsampled short random spines.
    try:
        for m in (16, 32):
            spine = np.abs(rng.standard_normal(m)) + 0.1
            add_candidate(cands, resample_sequence(spine, n))
    except Exception:
        pass

    # Anti-palindromic binary modulation, clipped to nonnegative.
    try:
        half = n // 2
        for amp in (0.5, 0.8, 0.95):
            first = rng.choice([-1.0, 1.0], size=half)
            if n % 2 == 0:
                b = np.concatenate((first, -first[::-1]))
            else:
                b = np.concatenate((first, np.array([0.0]), -first[::-1]))
            a = 0.5 * (1.0 + amp * b)
            add_candidate(cands, a)
            # slight smoothing
            k = np.ones(5) / 5.0
            a2 = np.convolve(a, k, mode="same")
            add_candidate(cands, a2)
    except Exception:
        pass

    return cands

    return item[0]

def strong_optimize(a0, budget, allow_lp=True):
    a = project_simplex(np.asarray(a0, dtype=np.float64).ravel())
    best = a.copy()
    best_local = local_score(a)
    n = a.size
    if n == 0 or budget <= 0.0:
        return best

    # (method, param1, param2, weight)
    # sub soft: param1=temp_frac, param2=0
    # sub top:  param1=unused, param2=top_k
    # lbfgs:    param1=p
    # lp:       none
    tasks = [
        ("sub", 0.05, 0, 0.8),
        ("sub", 0.02, 0, 1.2),
        ("sub", 0.005, 0, 1.0),
        ("sub", 0.0, 10, 0.8),
        ("sub", 0.0, 30, 1.2),
        ("sub", 0.0, 80, 0.8),
        ("lbfgs", 8.0, 0, 1.2),
        ("lbfgs", 16.0, 0, 1.5),
        ("lbfgs", 32.0, 0, 1.0),
        ("lbfgs", 64.0, 0, 0.8),
    ]
    if allow_lp and 64 <= n <= 1024:
        tasks.append(("lp", 0.0, 0, 0.8))

    weights = [float(t[3]) for t in tasks]
    deadline = time.monotonic() + max(0.1, float(budget))
    ran = False

    for idx, (meth, p1, p2, w) in enumerate(tasks):
        rem = deadline - time.monotonic()
        if rem <= 1.0:
            break
        sum_rem = float(sum(weights[idx:]))
        if sum_rem <= 0.0:
            break
        t_alloc = rem * float(w) / sum_rem
        if t_alloc < 2.5:
            continue

        ran = True
        if meth == "sub":
            if int(p2) == 0:
                a = optimize_subgrad(a, t_alloc, mode="soft", temp_frac=float(p1))
            else:
                a = optimize_subgrad(a, t_alloc, mode="top", temp_frac=0.02, top_k=int(p2))
        elif meth == "lbfgs":
            a = optimize_lbfgs(a, p1, t_alloc, maxiter=1000)
        elif meth == "lp":
            a = optimize_lp_refine(a, t_alloc, max_iters=140)
        else:
            continue

        sc = local_score(a)
        if sc < best_local:
            best_local = sc
            best = a.copy()
        else:
            # Use a small deterministic perturbation sometimes to avoid repeated
            # stagnation from the exact same starting point.
            if idx % 2 == 1 and n > 1:
                y = np.arange(n, dtype=np.float64)
                pert = 0.002 * np.sin((2.0 * np.pi * (idx + 1.0)) * y / float(n))
                perturb = project_simplex(best + pert)
                pe = local_score(perturb)
                # keep perturb only if it is locally at least as good; otherwise next
                # method starts from best.
                if pe <= best_local + 1e-8:
                    a = perturb
                else:
                    a = best.copy()
            else:
                a = best.copy()

    if not ran:
        rem = deadline - time.monotonic()
        if rem > 1.0:
            a = optimize_subgrad(a, rem, mode="soft", temp_frac=0.02)
            sc = local_score(a)
            if sc < best_local:
                best_local = sc
                best = a.copy()

    cur = local_score(a)
    if cur < best_local:
        best = a.copy()
    return best


def update_best_sequence(a, best):
    if a is None:
        return
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-15:
        return
    a = a / s
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        score = np.inf
    if np.isfinite(score) and score < best[1]:
        best[1] = score
        best[0] = a.copy()
        print(f"New best score={score:.6f} n={a.size}")


def propose_candidate(seed=42, budget_s=1800):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    horizon = max(60.0, float(budget_s) - 45.0)
    deadline = t_start + horizon

    best = [None, float("inf")]

    parent = get_parent_sequence()
    if parent is not None:
        print(f"Parent sequence available: n={parent.size}")
        update_best_sequence(parent, best)

    lengths = []
    if parent is not None:
        lengths.append(int(parent.size))
    lengths.extend([256, 384, 512, 768, 1024])

    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen and 64 <= L <= 1536:
            seen.add(L)
            unique_lengths.append(L)
    lengths = unique_lengths

    quick_time = max(30.0, min(240.0, 0.18 * horizon))
    quick_deadline = t_start + quick_time
    quick_cands = []

    print("=== Quick initialization/search ===")
    for n in lengths:
        if time.monotonic() > quick_deadline:
            break

        cands = generate_initializations(n, parent, rng)
        if not cands:
            continue

        scored = []
        for a in cands:
            sc = local_score(a)
            if np.isfinite(sc):
                scored.append((sc, a))

        if not scored:
            continue

        scored.sort(key=_score_key)
        top = scored[:5]
        print(f"  n={n}: best local quick={top[0][0]:.6f}")

        for sc, a in top:
            t_rem = quick_deadline - time.monotonic()
            if t_rem < 5.0:
                break
            t_alloc = min(18.0, max(4.0, t_rem / 4.0))
            a2 = optimize_subgrad(a, t_alloc, mode="soft", temp_frac=0.02)
            sc2 = local_score(a2)
            if sc2 < sc:
                a = a2
                sc = sc2

            quick_cands.append((sc, a))
            update_best_sequence(a, best)

        if top:
            update_best_sequence(top[0][1], best)

    # Build a small set of candidates for deep polishing.
    deep_candidates = []
    if best[0] is not None:
        deep_candidates.append(best[0].copy())
    if parent is not None:
        deep_candidates.append(parent.copy())
    for _, a in quick_cands:
        deep_candidates.append(a)

    # Include upsampled versions of current best to larger lengths.
    if best[0] is not None:
        for n in (512, 768, 1024):
            deep_candidates.append(resample_sequence(best[0], n))
    if parent is not None:
        for n in (768, 1024):
            deep_candidates.append(resample_sequence(parent, n))

    deep_scored = []
    for a in deep_candidates:
        sc = local_score(a)
        if np.isfinite(sc):
            deep_scored.append((sc, a))
    deep_scored.sort(key=_score_key)

    # Keep top four by local score, but force the current true best in first.
    deep_candidates = []
    if best[0] is not None:
        deep_candidates.append(best[0].copy())
    for _, a in deep_scored[:8]:
        if len(deep_candidates) >= 4:
            break
        # Very simple duplicate avoidance by length and local score range.
        ok = True
        for b in deep_candidates:
            if b.size != a.size and np.linalg.norm(a - resample_sequence(b, a.size)) < 1e-6 / max(1, a.size):
                ok = False
                break
        if ok:
            deep_candidates.append(a)

    if not deep_candidates:
        deep_candidates.append(np.ones(512, dtype=np.float64) / 512.0)

    deep_deadline = deadline - 90.0
    weights = [3.0, 2.0, 1.0, 1.0]

    print("=== Deep polishing ===")
    for idx, a in enumerate(deep_candidates):
        remaining = deep_deadline - time.monotonic()
        if remaining <= 20.0:
            break
        sum_rem = float(sum(weights[idx:]))
        t_alloc = remaining * float(weights[idx]) / max(1e-12, sum_rem)
        t_alloc = max(25.0, t_alloc)
        allow_lp = (len(a) <= 1024)
        a2 = strong_optimize(a, t_alloc, allow_lp=allow_lp)
        update_best_sequence(a2, best)
        print(f"  deep idx={idx} n={len(a2)} local={local_score(a2):.6f} true={best[1]:.6f}")

    # Final perturbations around the best true sequence.
    if best[0] is not None:
        final_deadline = deadline - 12.0
        scales = [0.003, 0.001, 0.008, 0.0007, 0.015, 0.005]
        print("=== Final basin perturbations ===")
        for i, scale in enumerate(scales):
            rem = final_deadline - time.monotonic()
            if rem < 8.0:
                break
            t_alloc = min(45.0, rem / max(1, len(scales) - i))
            if t_alloc < 5.0:
                break

            base = project_simplex(best[0] + rng.standard_normal(best[0].size) * scale)
            base = optimize_subgrad(base, max(3.0, 0.45 * t_alloc), mode="top", top_k=10)
            base = optimize_lbfgs(base, 16.0, max(3.0, 0.45 * t_alloc), maxiter=700)
            update_best_sequence(base, best)

    # Final safety: use best if available, otherwise uniform.
    if best[0] is None:
        fallback = np.ones(512, dtype=np.float64) / 512.0
        best[0] = fallback
        try:
            best[1] = float(evaluate_sequence(fallback.tolist()))
        except Exception:
            best[1] = np.inf

    out = np.asarray(best[0], dtype=np.float64).ravel()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-15:
        out = out / s
    else:
        out = np.ones(512, dtype=np.float64) / 512.0

    print(f"FINAL: n={out.size}, score={best[1]:.6f}")
    return [float(x) for x in out]
```
