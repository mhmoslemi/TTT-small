Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimizer with
log-sum-exp smoothed objective, temperature annealing, and
multi-restart strategy with diverse initializations.

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad(a, p, n):
    """Compute gradient base: grad_base[j] = sum_k p[k] * a[k-j]"""
    rev_a = a[::-1]
    conv_len = len(rev_a) + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    R = np.fft.rfft(rev_a, fft_size)
    P = np.fft.rfft(p, fft_size)
    corr = np.fft.irfft(R * P, fft_size)[:conv_len]
    return corr[n-1:2*n-1]

def smooth_obj_grad(a, t):
    n = len(a)
    conv = fast_convolve(a)
    scaled = t * conv
    max_scaled = scaled.max()
    exp_vals = np.exp(scaled - max_scaled)
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp
    obj = (2.0 * n / t) * (max_scaled + np.log(sum_exp))
    grad_base = compute_grad(a, p, n)
    grad = (2.0 * n / t) * grad_base
    return obj, grad, conv, p

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, p, n):
    """Compute gradient of log-sum-exp smoothed objective."""
    rev_a = a[::-1]
    conv_len = n + (2*n - 1) - 1  # = 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    R = np.fft.rfft(rev_a, fft_size)
    P = np.fft.rfft(p, fft_size)
    corr = np.fft.irfft(R * P, fft_size)[:conv_len]
    return corr[n-1:2*n-1]

def smooth_obj_and_grad(a, t):
    """Compute smoothed objective, gradient, and convolution."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = t * conv
    max_s = scaled.max()
    exps = np.exp(scaled - max_s)
    sum_exps = exps.sum()
    p = exps / sum_exps
    obj = (2.0 * n / t) * (max_s + np.log(sum_exps))
    grad_base = compute_gradient(a, p, n)
    grad = (2.0 * n / t) * grad_base
    return obj, grad, conv

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimizer with
log-sum-exp smoothed objective, exponential temperature annealing,
mirror descent, multi-restart with diverse initializations,
and exploitation of multiple sequence lengths.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, p, n):
    """Compute grad_base[j] = sum_k p[k]*a[k-j] via FFT cross-correlation."""
    rev_a = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    R = np.fft.rfft(rev_a, fft_size)
    P = np.fft.rfft(p, fft_size)
    corr = np.fft.irfft(R * P, fft_size)[:conv_len]
    return corr[n - 1:2 * n - 1]


def smooth_obj_and_grad(a, t):
    """Compute log-sum-exp smoothed objective and its gradient."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = t * conv
    max_s = scaled.max()
    exps = np.exp(scaled - max_s)
    sum_exps = exps.sum()
    p = exps / sum_exps
    grad_base = compute_gradient(a, p, n)
    grad = (2.0 * n / t) * grad_base
    return conv, grad


def adam_search(a0, time_limit, t_start=300.0, t_end=30000.0,
                lr_start=0.04, lr_end=0.0008, beta1=0.9, beta2=0.999,
                shake_every=1200, shake_scale=0.004, seed=0):
    """Adam optimization with exponential temperature annealing."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    step_count = 0

    conv, _ = smooth_obj_and_grad(a, t_start)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    no_improve = 0
    next_shake = shake_every

    while time.monotonic() - t0 < time_limit:
        step_count += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        t = t_start * (t_end / t_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        conv, grad = smooth_obj_and_grad(a, t)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        bc1 = 1.0 - beta1 ** step_count
        bc2 = 1.0 - beta2 ** step_count
        m_hat = m / bc1
        v_hat = v / bc2

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_simplex(a_new)

        new_conv = fast_convolve(a_new)
        true_obj = 2.0 * n * new_conv.max()

        if true_obj < best_obj - 1e-9:
            best_obj = true_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve >= next_shake:
            noise = rng.standard_normal(n) * shake_scale
            a = project_simplex(a + noise)
            m *= 0.3
            v *= 0.3
            no_improve = 0
            next_shake = int(next_shake * 1.15)

    return best_a, best_obj


def mirror_search(a0, time_limit, t_start=300.0, t_end=30000.0,
                  lr_start=0.015, lr_end=0.0005, seed=0):
    """Mirror descent (multiplicative weights) with temperature annealing."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    conv, _ = smooth_obj_and_grad(a, t_start)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    no_improve = 0
    shake_count = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        t = t_start * (t_end / t_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        conv, grad = smooth_obj_and_grad(a, t)

        # Multiplicative weights update
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - lr * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        new_conv = fast_convolve(a_new)
        true_obj = 2.0 * n * new_conv.max()

        if true_obj < best_obj - 1e-9:
            best_obj = true_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 3000:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            no_improve = 0
            shake_count += 1
            if shake_count > 25:
                break
            if shake_count > 15:
                break

    return best_a, best_obj


def hybrid_search(a0, time_limit, seed=0):
    """Alternate between Adam and mirror descent steps."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    step_count = 0

    conv, _ = smooth_obj_and_grad(a, 300.0)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    no_improve = 0
    use_adam = True
    adam_count = 0
    mirror_count = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        t = 300.0 * (30000.0 / 300.0) ** progress

        conv, grad = smooth_obj_and_grad(a, t)

        if use_adam:
            lr = 0.04 * (0.0008 / 0.04) ** progress
            step_count += 1
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * (grad * grad)
            bc1 = 1.0 - 0.9 ** step_count
            bc2 = 1.0 - 0.999 ** step_count
            a_new = a - lr * (m / bc1) / (np.sqrt(v / bc2) + 1e-8)
            a_new = project_simplex(a_new)
            adam_count += 1
            if adam_count >= 500:
                use_adam = False
                adam_count = 0
        else:
            lr_m = 0.015 * (0.0005 / 0.015) ** progress
            log_a = np.log(a + 1e-30)
            log_a_new = log_a - lr_m * grad
            log_a_new -= log_a_new.max()
            a_new = np.exp(log_a_new)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            mirror_count += 1
            if mirror_count >= 300:
                use_adam = True
                mirror_count = 0

        new_conv = fast_convolve(a_new)
        true_obj = 2.0 * n * new_conv.max()

        if true_obj < best_obj - 1e-9:
            best_obj = true_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 2000:
            noise = rng.standard_normal(n) * 0.004
            a = project_simplex(a + noise)
            m *= 0.5
            v *= 0.5
            no_improve = 0

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)

    if parent is not None:
        if len(parent) >= n:
            p = np.maximum(parent[:n], 0)
        else:
            p = np.interp(np.linspace(0, 1, n),
                         np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
        ps = p.sum()
        if ps > 1e-10:
            inits.append(p / ps)

    for freq in [1, 2, 3, 5, 7]:
        a = 1.0 + 0.85 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0)
        if a.sum() > 1e-10:
            inits.append(a / a.sum())

    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    g2 = (np.exp(-0.5 * ((x - 0.2 * n) / (0.12 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.8 * n) / (0.12 * n)) ** 2))
    inits.append(g2 / g2.sum())

    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 0.05))

    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    ramp2 = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp2 / ramp2.sum())

    for npeaks in [3, 5, 8]:
        peaks = rng.choice(n, size=npeaks, replace=False)
        a = np.zeros(n)
        for pk in peaks:
            width = max(1, n // (npeaks * 5))
            a[max(0, pk - width):pk + width + 1] = 1.0
        a = np.maximum(a, 0)
        if a.sum() > 1e-10:
            inits.append(a / a.sum())

    # Triangle
    tri = 1.0 - 2.0 * np.abs(x / n - 0.5)
    tri = np.maximum(tri, 0)
    if tri.sum() > 1e-10:
        inits.append(tri / tri.sum())

    # W-shaped
    w = 1.0 - 4.0 * np.abs(np.abs(x / n - 0.5) - 0.25)
    w = np.maximum(w, 0)
    if w.sum() > 1e-10:
        inits.append(w / w.sum())

    return inits

        if np.isfinite(score) and score < best_score:

            best_sequence = a.copy()

        return best_score

    # Phase 1: Broad exploration with multiple lengths and methods
    print("=== Phase 1: Broad Exploration ===")
    t1_start = time.monotonic()

    lengths_to_try = [512]
    if parent is not None:
        pl = len(parent)
        if pl not in lengths_to_try and 64 <= pl <= 4096:
            lengths_to_try.insert(0, pl)
    for L in [256, 1024, 2048]:
        if L not in lengths_to_try:
            lengths_to_try.append(L)

    lengths_to_try = [L for L in lengths_to_try if 64 <= L <= 4096]
    lengths_to_try.sort()

    phase1_deadline = deadline - 400
    total_runs = 0
    for n in lengths_to_try:
        total_runs += 5
    time_per_run = max(8.0, (phase1_deadline - time.monotonic()) / max(1, total_runs))

    run_id = 0
    for n in lengths_to_try:
        inits = generate_inits(n, parent, rng)
        n_inits = len(inits)

        for trial in range(5):
            if time.monotonic() > phase1_deadline:
                break
            t_alloc = min(time_per_run * 1.5, phase1_deadline - time.monotonic() - 10)
            if t_alloc < 5:
                break

            idx = (trial * 3 + run_id) % n_inits
            a0 = inits[idx]

            method_choice = (trial + run_id) % 4
            if method_choice == 0:
                a_opt, _ = adam_search(a0, t_alloc,
                                        t_start=200, t_end=20000,
                                        lr_start=0.05, lr_end=0.001,
                                        seed=run_id)
            elif method_choice == 1:
                a_opt, _ = adam_search(a0, t_alloc,
                                        t_start=500, t_end=80000,
                                        lr_start=0.02, lr_end=0.0005,
                                        shake_every=800, shake_scale=0.006,
