"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the key insight that fixing a and optimizing g over the simplex with

import numpy as np
import time
from scipy.optimize import linprog


def autocorr(a):
    """Compute full linear auto-correlation (same as conv(a,a) for real a)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_corr(a, b):
    """Compute full linear cross-correlation a * b."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def obj_value(a):
    """Compute objective value for sequence a (assumes sum(a) > 0)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    ac = autocorr(a)
    return 2.0 * n * ac.max() / (s * s)


def lp_find_direction(a):
    """
    Solve LP: minimize t subject to (a*g)_k <= t for all k, sum(g)=1, g>=0.
    Returns (g, t) where g is the optimal direction and t is the optimal peak.
    """
    n = len(a)
    num_constraints = 2 * n - 1
    
    # Build constraint matrix: A @ g <= t => A @ g - t <= 0
    # A[k, j] = a[k - j] if 0 <= k-j < n, else 0
    # Actually (a*g)_k = sum_j a_j * g[k-j] ... wait let me be careful.
    # conv(a,g)_k = sum_j a_j * g_{k-j} where the sum is over valid j.
    # So (a*g)_k = sum_{j: 0<=j<n, 0<=k-j<n} a_j * g_{k-j}
    # Let i = k-j, so j = k-i:
    # (a*g)_k = sum_i a_{k-i} * g_i where 0<=k-i<n and 0<=i<n
    # = sum_i g_i * a_{k-i}
    
    # So the constraint matrix is A[k, i] = a[k-i] if 0 <= k-i < n
    
    # This is a banded matrix (Toeplitz-like)
    A_mat = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        # i ranges from max(0, k-n+1) to min(n-1, k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A_mat[k, i] = a[k - i]
    
    # LP: minimize t (last variable)
    # Variables: [g_0, ..., g_{n-1}, t]
    # Constraints: A_mat @ g - t <= 0  =>  [A_mat, -1] @ [g, t] <= 0
    # Equality: sum(g) = 1  =>  [1, 1, ..., 1, 0] @ [g, t] = 1
    # Bounds: g_i >= 0, t free
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.hstack([A_mat, -np.ones((num_constraints, 1))])
    b_ub = np.zeros(num_constraints)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        g = result.x[:n]
        t_val = result.x[-1]
        # Ensure g is valid
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs > 1e-15:
            g = g / gs
        return g, t_val
    return a.copy(), np.inf


def build_conv_matrix_fast(a):
    """Build the convolution matrix A where A[k,i] = a[k-i] if valid."""
    n = len(a)
    num_constraints = 2 * n - 1
    A_mat = np.zeros((num_constraints, n))
    for k in range(num_constraints):

        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        A_mat[k, idx] = a[k - idx]
    return A_mat


def line_search(a, g, max_iter=30):
    """
    Line search: minimize max(conv((1-s)*a + s*g, (1-s)*a + s*g)) over s in [0, 1].

    """
    n = len(a)
    best_s = 0.0
    ac = autocorr(a)
    best_val = ac.max()
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    
    lo, hi = 0.0, 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    
    for _ in range(max_iter):
        a1 = (1 - c1) * a + c1 * g
        a2 = (1 - c2) * a + c2 * g
        v1 = autocorr(a1).max()
        v2 = autocorr(a2).max()
        
        if v1 < v2:
            hi = c2
            c2 = c1
            v2 = v1
            c1 = hi - gr * (hi - lo)
            if c1 == c2:
                c1 = hi - 1e-10
                v1 = autocorr((1-c1)*a + c1*g).max()
        else:
            lo = c1
            c1 = c2
            v1 = v2

        # Track best

        if v1 < best_val:
            best_val = v1
            best_s = c1

        a = a / s
    n = len(a)

                    a = best_a.copy()

def subgradient_polish(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30):
    """
    Subgradient descent for polishing. Minimizes max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    ac = autocorr(a)
    best_val = ac.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Find top-k peaks
        peaks = np.argpartition(ac, -top_k)[-top_k:]
        peak_vals = ac[peaks]
        
        # Softmax weights on peaks
        scaled = peak_vals / 0.0002
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(top_k)
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        
        # Subgradient: for peak at position k, grad_i = 2 * a[k-i] (projected)
        sg = np.zeros(n)
        for i in range(top_k):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * 2.0 * a[k - idx]
        
        # Remove mean to maintain sum=1 (project onto tangent space)
        sg = sg - sg.mean()
        
        v = momentum * v - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
        
        ac = autocorr(a)
        cur_val = ac.max()
        
        if cur_val < best_val * (1 - 1e-13):
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9995
        momentum = min(0.97, momentum + 0.0002)
        
        if no_improve >= 500:
            break
    
    return best_a, 2.0 * n * best_val

"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the key insight that fixing a and solving min_t {t : (a*g)_k <= t, sum(g)=1, g>=0}
gives a descent direction. Combined with golden-section line search, multiple n values,
subgradient polishing, and perturbation-based restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog


def autocorr(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """
    Build matrix C where C[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    So that C @ g = conv(a, g).
    """
    n = len(a)
    num_rows = 2 * n - 1
    C = np.zeros((num_rows, n))
    for j in range(n):
        # C[k, j] = a[k-j] for valid k-j
        # k-j >= 0 => k >= j
        # k-j < n => k < j + n
        k_start = j
        k_end = min(j + n, num_rows)
        C[k_start:k_end, j] = a[:k_end - k_start]
    return C


def lp_direction(a):
    """
    Solve LP: minimize t s.t. conv(a,g)_k <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt).
    """
    n = len(a)
    num_constraints = 2 * n - 1

    C = build_conv_matrix(a)

    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Inequality: C @ g - t <= 0
    A_ub = np.hstack([C, -np.ones((num_constraints, 1))])
    b_ub = np.zeros(num_constraints)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(-np.inf, np.inf)]

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 30})

    if result.success:
        g = result.x[:n]
        t_val = result.x[-1]
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs > 1e-15:
            g = g / gs
        return g, t_val
    else:
        return a.copy(), np.inf


def golden_section_line_search(a, g, iterations=40):
    """
    find s in [0,1] minimizing max(conv((1-s)*a + s*g, (1-s)*a + s*g)).
    The function is convex in s (max of quadratics).
    """
    n = len(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    lo = 0.0
    hi = 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)

    a1 = (1.0 - c1) * a + c1 * g
    v1 = autocorr(a1).max()
    a2 = (1.0 - c2) * a + c2 * g
    v2 = autocorr(a2).max()

    best_s = 0.0
    best_v = min(v1, v2, autocorr(a).max())
    if v1 <= best_v:
        best_s, best_v = c1, v1
    if v2 <= best_v:
        best_s, best_v = c2, v2

    for _ in range(iterations):
        if hi - lo < 1e-12:
            break
        if v1 < v2:
            hi = c2
            c2 = c1
            v2 = v1
            c1 = hi - gr * (hi - lo)
            a1 = (1.0 - c1) * a + c1 * g
            v1 = autocorr(a1).max()
        else:
            lo = c1
            c1 = c2
            v1 = v2
            c2 = lo + gr * (hi - lo)
            a2 = (1.0 - c2) * a + c2 * g
            v2 = autocorr(a2).max()

        if v1 < best_v:
            best_v = v1
            best_s = c1
        if v2 < best_v:
            best_v = v2
            best_s = c2

    return best_s, best_v


def lp_optimize(a0, time_limit, verbose=False, max_perturb=60):
    """
    LP-based iterative optimization.
    a0: initial sequence (normalized to sum=1, nonnegative).
    Returns (best_a, best_objective).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    ac = autocorr(a)
    best_peak = ac.max()
    best_a = a.copy()
    best_obj = 2.0 * n * best_peak

    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 150
    perturb_count = 0
    lp_fail = 0

    while time.monotonic() - t0 < time_limit:
        # Solve LP for direction
        g, t_lp = lp_direction(a)

        if not np.isfinite(t_lp) or t_lp >= best_peak * (1.0 - 1e-10):
            no_improve += 1
            lp_fail += 1
        else:
            lp_fail = 0
            # Line search along a -> g
            s_best, v_best = golden_section_line_search(a, g, iterations=35)

            if v_best < best_peak * (1.0 - 1e-12):
                a_new = (1.0 - s_best) * a + s_best * g
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a = a_new / s_new
                else:
                    a = best_a.copy()

                ac = autocorr(a)
                cur_peak = ac.max()
                if cur_peak < best_peak:
                    best_peak = cur_peak
                    best_obj = 2.0 * n * best_peak
                    best_a = a.copy()
                    no_improve = 0
                    if verbose and perturb_count % 10 == 0:
                        elapsed = time.monotonic() - t0
                        print(f"    LP n={n}: peak={best_peak:.8f}, obj={best_obj:.6f}, t={elapsed:.0f}s")
            else:
                no_improve += 1

        # Stagnation: perturb
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(hash(int(time.monotonic() * 1e6)) % (2**31))
            scale = 0.002 + 0.004 * min(perturb_count, 8)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(400, 150 + 40 * perturb_count)

            if perturb_count > max_perturb:
                break

        # LP repeatedly failing: bigger perturbation
        if lp_fail > 80:
            rng = np.random.default_rng(hash(int(time.monotonic() * 7e5)) % (2**31))
            noise = rng.standard_normal(n) * 0.01
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            lp_fail = 0
            no_improve = 0

    return best_a, best_obj


def subgradient_optimize(a0, time_limit, lr_init=0.008, momentum=0.92, top_k=40, temp_scale=0.0003):
    """
    Projected subgradient descent with momentum.
    Minimizes max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    ac = autocorr(a)
    best_peak = ac.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    temp = temp_scale * best_peak
    batch_size = 50

    while time.monotonic() - t0 < time_limit:
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit:
                break

            # Top-k peaks with softmax weighting
            if top_k < len(ac):
                peaks = np.argpartition(ac, -top_k)[-top_k:]
            else:
                peaks = np.arange(len(ac))
            peak_vals = ac[peaks]

            scaled = peak_vals / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(peak_vals)] = 1.0
            else:
                w /= w_sum

            # Subgradient of max(conv(a,a)) w.r.t. a_i
            # d/d a_i [conv(a,a)_k] = a[k-i] + a[i] (counting both terms in the sum)
            # More precisely: conv(a,a)_k = sum_j a_j * a_{k-j}
            # d/d a_i: term j=i gives a_{k-i}, term k-j=i gives a_i
            # So grad_i = a_{k-i} + a_i (if both indices valid)
            # For the weighted sum: grad_i = sum_k w_k * (a[k-i] + a[i])
            # = sum_k w_k * a[k-i] + (sum_k w_k) * a[i]
            # = sum_k w_k * a[k-i] + a[i] (since sum w_k = 1)

            sg = np.zeros(n)
            for ii in range(len(peaks)):
                if w[ii] < 1e-14:
                    continue
                k = int(peaks[ii])
                # Contribution from a[k-i] term:
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                idx = np.arange(j_start, j_end + 1)
                sg[idx] += w[ii] * a[k - idx]
            # Add a[i] term (sum_k w_k * a[i] = a[i])
            sg += a

            # Project onto tangent space of sum(a)=1: subtract mean
            sg = sg - sg.mean()

            v = momentum * v - lr * sg
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
            else:
                a = best_a.copy()

            ac = autocorr(a)
            cur_peak = ac.max()

            if cur_peak < best_peak * (1.0 - 1e-13):
                best_peak = cur_peak
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        lr *= 0.999
        temp *= 1.003
        momentum = min(0.97, momentum + 0.0002)

        if no_improve >= 400:
            break

    return best_a, 2.0 * n * best_peak


def multiplicative_update(a0, time_limit, lr_init=0.05, top_k=25):
    """
    Mirror descent (multiplicative weight update) for peak reduction.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    ac = autocorr(a)
    best_peak = ac.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    temp = 0.001 * best_peak

    while time.monotonic() - t0 < time_limit:
        if top_k < len(ac):
            peaks = np.argpartition(ac, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(ac))
        peak_vals = ac[peaks]

        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 1e-15:
            w /= w_sum
        else:
            w = np.zeros(len(peaks))
            w[0] = 1.0

        # Pressure at each position: how much each a_i contributes to peaks
        pressure = np.zeros(n)
        for ii in range(len(peaks)):
            if w[ii] < 1e-14:
                continue
            k = int(peaks[ii])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            pressure[idx] += w[ii] * a[k - idx]
        # Also add the "other side" contribution
        for ii in range(len(peaks)):
            if w[ii] < 1e-14:
                continue
            k = int(peaks[ii])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            pressure[idx] += w[ii] * a[idx]  # a[i] term

        # Normalize
        p_max = pressure.max()
        if p_max > 1e-15:
            pressure /= p_max

        # Multiplicative update
        a_new = a * np.exp(-lr * pressure)
        a_new = a_new + 1e-12 / n  # floor to avoid zeros
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        ac = autocorr(a)
        cur_peak = ac.max()

        if cur_peak < best_peak * (1.0 - 1e-13):
            best_peak = cur_peak
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.997
        temp *= 1.005

        if no_improve >= 300:
            rng = np.random.default_rng(hash(int(time.monotonic() * 3e5)) % (2**31))
            a = best_a + rng.standard_normal(n) * 0.004
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            no_improve = 0

    return best_a, 2.0 * n * best_peak


def make_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    # Sine
    si = np.sin(np.pi * (x + 0.5) / n) + 0.05
    inits['sine'] = si / si.sum()

    # Bimodal (peaks at edges)
    bi = np.minimum(x + 1, n - x) ** 0.5 + 0.1
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2) + 0.01
    inits['gaussian'] = g / g.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
            if len(p) == n:
                inits['parent'] = p.copy()
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_ds'] = pp / pps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_up'] = pp / pps

    return inits

                elif pi < 5:
                    print(f"  Perturb {pi}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
