Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a combination of:

import numpy as np

from scipy.optimize import minimize, linprog
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def exact_score(a):
    """Compute the score directly."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_direction(a, M):
    """
    Solve the LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k.
    Returns the optimal b (length n, non-negative).
    """
    n = len(a)
    # Constraint matrix: C[k, j] = a[j] if max(0, k-n+1) <= j <= min(n-1, k), else 0
    # Number of constraints: 2n-1
    n_constraints = 2 * n - 1
    
    # Build the constraint matrix (sparse would be better but for n<=1024 dense is OK)
    C = np.zeros((n_constraints, n))
    for k in range(n_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        C[k, j_start:j_end+1] = a[j_start:j_end+1]
    
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    # Constraints: C @ b <= M (all same M)
    res = linprog(c, A_ub=C, b_ub=np.full(n_constraints, M), bounds=bounds,
                  method='highs', options={'presolve': True})
    
    if res.success:
        return res.x
    else:
        # Fallback: return uniform
        return np.ones(n) / n


def topk_gradient(a, K=10):
    """
    Compute gradient of sum of top-K peaks of conv(a,a).
    Returns (objective_value, gradient)
    """
    n = len(a)
    conv = fast_convolve(a)
    topk_idx = np.argsort(conv)[::-1][:K]
    obj = conv[topk_idx].sum()
    
    # Gradient: grad[i] = sum_{k in topk} 2*a[k-i] if 0 <= k-i < n
    grad = np.zeros(n)
    for k in topk_idx:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    
    return obj, grad

        j_max = min(n - 1, k)

def smooth_max(x, beta):
    """Log-sum-exp approximation of max."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    result = x_max + np.log(np.exp(shifted).sum()) / beta
    return result


def smooth_max_weighted(x, weights, beta):
    """Weighted log-sum-exp."""
    x_max = (x * weights).max()
    shifted = beta * (x * weights - x_max)
    shifted = np.clip(shifted, -100, 100)
    result = x_max + np.log(np.exp(shifted).sum()) / beta
    return result


def optimize_with_lbfgsb_topk(a0, K=20, maxiter=300):
    """L-BFGS-B optimization minimizing sum of top-K peaks (via softmax reparam)."""
    n = len(a0)
    a0_safe = np.maximum(a0, 1e-30)
    a0_safe = a0_safe / a0_safe.sum()
    x0 = np.log(a0_safe)
    
    def obj(x):
        x_shifted = x - x.max()
        a = np.exp(x_shifted)
        a = a / a.sum()
        conv = fast_convolve(a)
        # Use top-K sum as proxy (smoothed)
        beta = 5.0 / K
        topk = np.sort(conv)[::-1][:K]
        return 2.0 * n * (topk.sum() / K)  # Average of top K
    
    def jac(x):
        x_shifted = x - x.max()
        a = np.exp(x_shifted)
        a = a / a.sum()
        conv = fast_convolve(a)
        topk_idx = np.argsort(conv)[::-1][:K]
        
        # Gradient of sum of top-K w.r.t. a
        grad_a = np.zeros(n)
        for k in topk_idx:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                i = k - j
                grad_a[i] += 2.0 * a[j]
        
        grad_a /= K  # Because we average
        
        # Chain rule through softmax
        lagrange = np.dot(a, grad_a)
        return a * (grad_a - lagrange) * 2.0 * n / (a.sum()**2)
        # Wait, the objective is 2*n*max_c/s^2, and s=1 after normalization...
        # Actually with softmax, a always sums to 1, so the objective is just 2*n*topk_mean
        # And the gradient w.r.t. x is: 2*n/K * d(sum topk)/d(a) * da/dx
        # da/dx for softmax: da_i/dx_j = a_i*(delta_ij - a_j)
        # So grad_x = 2*n/K * (a * (grad_a - dot(a, grad_a)))
        
        # Let me redo
        return 2.0 * n / K * a * (grad_a - np.dot(a, grad_a))
    
    result = minimize(obj, x0, jac=jac, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxls': 50})
    
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, exact_score(a_opt)


def optimize_lp_iterative(a0, n_iter=500, t_init=0.1, t_min=0.01):
    """
    Iterative LP-based optimization from the state of the art.
    At each step: find LP direction, mix, renormalize.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-15).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = exact_score(a)
    
    t = t_init
    for it in range(n_iter):
        conv = fast_convolve(a)
        M = conv.max()
        
        # Solve LP
        b = lp_direction(a, M)
        b = np.maximum(b, 0)
        s = b.sum()
        if s < 1e-15:
            break
        b = b / s  # Normalize b to sum to 1
        
        # Mix
        a_new = (1 - t) * a + t * b
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        new_score = exact_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step size
            t *= 0.5
            if t < t_min:
                t = t_min
                a = a_new  # Accept anyway to keep moving
        
        # Decay t
        t = max(t_min, t_init * (0.995 ** it))
    
    return best_a, best_score


def optimize_topk_pg(a0, K=20, n_iter=500, lr=0.02):
    """Projected gradient descent minimizing sum of top-K peaks."""
    n = len(a0)
    a = np.maximum(a0, 1e-15).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = exact_score(a)
    
    for it in range(n_iter):
        conv = fast_convolve(a)
        topk_idx = np.argsort(conv)[::-1][:K]
        obj = conv[topk_idx].sum() / K
        
        # Compute gradient
        grad = np.zeros(n)
        for k in topk_idx:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                i = k - j
                grad[i] += 2.0 * a[j]
        grad /= K

        a_new = project_simplex(a - lr * grad)
        
        new_score = exact_score(a_new)

            best_a = a_new.copy()

        lr *= 0.999

def sinusoidal_init(n, rng):
    """Initialize with sinusoidal patterns."""
    x = np.arange(n)
    a = np.ones(n)
    n_harmonics = rng.integers(1, 5)
    for _ in range(n_harmonics):
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a *= (1.0 + 0.3 * np.sin(2 * np.pi * freq * x / n + phase))
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def window_init(n, window_type='kaiser', rng=None):
    """Initialize with a standard window function."""
    x = np.linspace(0, 1, n)
    if window_type == 'kaiser':
        from scipy.signal import get_window
        a = get_window('kaiser', n, beta=5.0)
    elif window_type == 'hann':
        a = np.hanning(n)
    elif window_type == 'blackman':
        a = np.blackman(n)
    elif window_type == 'nuttall':
        a = np.blackman(n)  # scipy.signal.windows.nuttall
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def hybrid_optimize(a0, budget_s):

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using LP-based direction finding,
top-K peak gradient descent, and L-BFGS-B fine-tuning with multi-length exploration.
Key innovation: iterative LP direction finding (from the state-of-the-art LP approach)
combined with targeted peak reduction and adaptive multi-phase search.

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def exact_score(a):
    """Compute the exact score: 2n*max(conv)/(sum)^2."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v[::1])[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a):
    """Build constraint matrix for LP: C[k,j] = a[j] if 0 <= k-j < n, else 0."""
    n = len(a)
    n_constraints = 2 * n - 1
    C = np.zeros((n_constraints, n))
    for k in range(n_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        C[k, j_start:j_end + 1] = a[j_start:j_end + 1]
    return C


def lp_direction(a, M):
    """
    Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k.
    Returns optimal b (length n, non-negative, sum may not equal 1).
    """
    n = len(a)
    C = build_lp_matrix(a)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    rhs = np.full(2 * n - 1, M)

    try:
        res = linprog(c, A_ub=C, b_ub=rhs, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            return res.x
        else:
            return None
    except Exception:
        return None


def lp_direction_weighted(a, use_actual_rhs=False):
    """
    LP direction with option to use actual conv values as RHS (tighter constraints).
    """
    n = len(a)
    conv = fast_convolve(a)
    if use_actual_rhs:
        rhs = conv.copy()
        # Add small epsilon to avoid infeasibility
        rhs = rhs + 1e-10
    else:
        M = conv.max()
        rhs = np.full(2 * n - 1, M)

    C = build_lp_matrix(a)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=C, b_ub=rhs, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            return res.x
        else:
            return None
    except Exception:
        return None


def topk_gradient(a, K=10):
    """
    Compute gradient of sum of top-K convolution peaks.
    Returns (sum_of_topk, gradient_vector).
    """
    n = len(a)
    conv = fast_convolve(a)
    topk_idx = np.argsort(conv)[::-1][:K]
    obj = conv[topk_idx].sum()

    grad = np.zeros(n)
    for k in topk_idx:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            i = k - j
            grad[i] += 2.0 * a[j]

    return obj, grad


def smooth_max(x, beta):
    """Log-sum-exp approximation of max."""
    x_max = x.max()
    shifted = np.clip(beta * (x - x_max), -100, 100)
    return x_max + np.log(np.exp(shifted).sum()) / beta


def window_init(n, window_type='kaiser'):
    """Initialize with a standard window function."""
    if window_type == 'kaiser':
        from scipy.signal import get_window
        a = np.asarray(get_window('kaiser', n, beta=5.0))
    elif window_type == 'hann':
        a = np.hanning(n)
    elif window_type == 'blackman':
        a = np.blackman(n)
    elif window_type == 'flat_top':
        # Flat-top window coefficients
        a = (0.21557895 - 0.41663158 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
             + 0.277263158 * np.cos(4 * np.pi * np.arange(n) / (n - 1))
             - 0.083578947 * np.cos(6 * np.pi * np.arange(n) / (n - 1))
             + 0.006947368 * np.cos(8 * np.pi * np.arange(n) / (n - 1)))
    else:
        a = np.ones(n)
    a = np.abs(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def sinusoidal_init(n, rng):
    """Initialize with sinusoidal modulation."""
    x = np.arange(n)
    a = np.ones(n)
    n_harmonics = rng.integers(1, 4)
    for _ in range(n_harmonics):
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        a *= (1.0 + amp * np.sin(2 * np.pi * freq * x / n + phase))
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def bipartite_init(n, rng):
    """Initialize with two clusters of mass."""
    a = np.zeros(n)
    n_clusters = rng.integers(2, 5)
    widths = rng.uniform(0.05, 0.15, n_clusters)
    centers = rng.uniform(0.1, 0.9, n_clusters)
    for c, w in zip(centers, widths):
        a += np.exp(-0.5 * ((x - c) / w) ** 2) if False else np.exp(-0.5 * ((np.arange(n) / n - c) / w) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def lbfgsb_smooth_obj(x, n, beta):
    """Objective for L-BFGS-B with log-sum-exp smoothing."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm = smooth_max(conv, beta)
    return 2.0 * n * sm


def lbfgsb_smooth_jac(x, n, beta):
    """Jacobian for L-BFGS-B with log-sum-exp smoothing."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)

    # Softmax weights for the smooth max
    shifted = beta * (conv - conv.max())
    shifted = np.clip(shifted, -100, 100)
    w = np.exp(shifted)
    w = w / w.sum()

    # Gradient of smooth max w.r.t. a
    n_conv = len(conv)
    fft_size = 1 << (n_conv + n - 1 + 63).bit_length()
    W = fft(w, fft_size)
    A = fft(a, fft_size)
    grad_a = np.real(ifft(np.conj(W) * A, fft_size))[:n] * 2.0

    # Chain rule through softmax: da_i/dx_j = a_i*(delta_ij - a_j)
    lagrange = np.dot(a, grad_a)
    return 2.0 * n * a * (grad_a - lagrange)


def optimize_lbfgsb(a0, beta, maxiter=300):
    """L-BFGS-B optimization with softmax reparametrization and smooth max."""
    n = len(a0)
    a0_safe = np.maximum(a0, 1e-30)
    a0_safe = a0_safe / a0_safe.sum()
    x0 = np.log(a0_safe)

    try:
        result = minimize(lbfgsb_smooth_obj, x0, jac=lbfgsb_smooth_jac,
                          args=(n, beta), method='L-BFGS-B',
                          options={'maxiter': maxiter, 'maxls': 50})
        x_opt = result.x
    except Exception:
        x_opt = x0

    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, exact_score(a_opt)


def optimize_topk_lbfgsb(a0, K=10, maxiter=200):
    """L-BFGS-B minimizing sum of top-K peaks."""
    n = len(a0)
    a0_safe = np.maximum(a0, 1e-30)
    a0_safe = a0_safe / a0_safe.sum()
    x0 = np.log(a0_safe)

    def obj(x):
        x_shifted = x - x.max()
        a = np.exp(x_shifted)
        a = a / a.sum()
        conv = fast_convolve(a)
        topk = np.sort(conv)[::-1][:K]
        return 2.0 * n * topk.sum() / K

    def jac(x):
        x_shifted = x - x.max()
        a = np.exp(x_shifted)
        a = a / a.sum()
        conv = fast_convolve(a)
        topk_idx = np.argsort(conv)[::-1][:K]

        grad_a = np.zeros(n)
        for k in topk_idx:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                i = k - j
                grad_a[i] += 2.0 * a[j]
        grad_a /= K

        lagrange = np.dot(a, grad_a)
        return 2.0 * n / K * a * (grad_a - lagrange)

    try:
        result = minimize(obj, x0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'maxls': 50})
        x_opt = result.x
    except Exception:
        x_opt = x0

    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, exact_score(a_opt)


def hybrid_optimize(a0, budget_s, rng=None):
    """
    Multi-strategy hybrid optimization:
    1. LP-based direction finding
    2. Top-K projected gradient
    3. L-BFGS-B fine-tuning
    """
    n = len(a0)
    a = np.maximum(a0, 1e-15).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = exact_score(a)
    t_start = time.monotonic()

    if rng is None:
        rng = np.random.default_rng(0)

    # --- Phase A: LP iterations ---
    lp_budget = budget_s * 0.35
    lp_count = 0
    while time.monotonic() - t_start < lp_budget:
        conv = fast_convolve(a)
        M = conv.max()

        b = lp_direction(a, M)
        if b is None:
            # Fallback: try weighted version
            b = lp_direction_weighted(a, use_actual_rhs=True)
            if b is None:
                break

        b = np.maximum(b, 0)
        s = b.sum()
        if s < 1e-15:
            break
        b = b / s

        t = 0.05 * (0.995 ** lp_count) + 0.01
        a_new = (1 - t) * a + t * b
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        new_score = exact_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
        a = a_new
        lp_count += 1

    print("    LP: %d iters, score=%.6f" % (lp_count, best_score))

    # --- Phase B: Top-K projected gradient ---
    pg_budget = budget_s * 0.35
    pg_start = time.monotonic() - t_start
    lr = 0.03
    K = max(5, n // 32)
    K = min(K, 30)
    pg_count = 0

    while time.monotonic() - t_start - pg_start < pg_budget and time.monotonic() - t_start < budget_s * 0.75:
        conv = fast_convolve(a)
        topk_idx = np.argsort(conv)[::-1][:K]

        grad = np.zeros(n)
        for k in topk_idx:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                i = k - j
                grad[i] += 2.0 * a[j]
        grad /= K

        a_new = project_simplex(a - lr * grad)
        new_score = exact_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
        a = a_new
        lr *= 0.999
        pg_count += 1

    print("    PG: %d iters, score=%.6f" % (pg_count, best_score))

    # --- Phase C: L-BFGS-B fine-tuning ---
    remaining = budget_s - (time.monotonic() - t_start)
    if remaining > 15:
        maxiter = int(min(200, remaining * 8))
        a_opt, score = optimize_topk_lbfgsb(a, K=8, maxiter=maxiter)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print("    L-BFGS-B topK: %.6f" % score)

        remaining = budget_s - (time.monotonic() - t_start)
        if remaining > 15:
            maxiter = int(min(200, remaining * 8))
            a_opt, score = optimize_lbfgsb(a, beta=10.0, maxiter=maxiter)
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print("    L-BFGS-B smooth: %.6f" % score)

    return best_a, best_score


def multi_start_search(n, parent, rng, budget_s):
    """Multi-start optimization for a given length n."""
    inits = []

    if parent is not None:
        if len(parent) == n:
            inits.append(('parent', parent.copy()))
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits.append(('parent_interp', pp / ps))

    for wt in ['kaiser', 'hann', 'blackman', 'flat_top']:
        inits.append((wt, window_init(n, wt)))

    inits.append(('sine1', sinusoidal_init(n, rng)))
    inits.append(('sine2', sinusoidal_init(n, rng)))
    inits.append(('bipartite', bipartite_init(n, rng)))
    inits.append(('random', rng.dirichlet(np.ones(n) * 0.5)))

    # Quick score all initializations
    scored = []
    for name, a0 in inits:
        s = a0.sum()
        if s > 1e-10:
            a0 = a0 / s
        score = exact_score(a0)
        scored.append((score, name, a0))

    scored.sort(key=lambda x: x[0])

    # Optimize the top 3 initializations
    best_a = None
    best_score = float('inf')
    n_top = min(3, len(scored))
    time_per = budget_s / n_top

    for i in range(n_top):
        if time.monotonic() - t_start_global > deadline_global - 50:
            break
        score, name, a0 = scored[i]
        t_alloc = min(time_per * (1.5 - 0.25 * i), deadline_global - time.monotonic() - 50)
        if t_alloc < 15:
            break
        print("      Init '%s' (score=%.6f), budget=%.0fs" % (name, score, t_alloc))
        a_opt, opt_score = hybrid_optimize(a0, t_alloc, rng)
        if opt_score < best_score:
            best_score = opt_score
            best_a = a_opt.copy()

    return best_a, best_score

        192, 256, 320, 384, 512
    ]))
    lengths_to_try = [l for l in lengths_to_try if 64 <= l <= 768]
    lengths_to_try = sorted(lengths_to_try)[:5]

    print("Lengths to try: %s" % lengths_to_try)

    for n in lengths_to_try:
        if time.monotonic() > phase1_deadline - 20:
            break

        # Quick evaluation of a few initializations
        inits = []
        if parent is not None:
            if len(parent) == n:
                inits.append(parent.copy())
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
                pp = np.maximum(pp, 0)
                ps = pp.sum()
                if ps > 1e-10:
                    inits.append(pp / ps)
        inits.append(window_init(n, 'kaiser'))
        inits.append(window_init(n, 'blackman'))
        inits.append(sinusoidal_init(n, rng))

        for a0 in inits:
            if time.monotonic() > phase1_deadline - 10:
                break
            try_candidate(a0)

    print("Phase 1 best: %.6f (n=%d)" % (best_score, len(best_sequence) if best_sequence is not None else 0))

    # === Phase 2: Deep optimization on promising lengths ===
    print("=== Phase 2: Deep Optimization ===")
    phase2_deadline = deadline_global - 120.0

    if best_sequence is not None:
        best_n = len(best_sequence)
    else:
        best_n = 256

    # Focus lengths: best and neighbors
    focus_lengths = sorted(set([
        best_n, best_n + 16, best_n - 16, best_n + 32, best_n - 32
    ]))
    focus_lengths = [l for l in focus_lengths if 64 <= l <= 768]
    focus_lengths = sorted(focus_lengths)[:3]

    print("Focus lengths: %s" % focus_lengths)

    time_per_length = (phase2_deadline - time.monotonic()) / max(1, len(focus_lengths))

    for n in focus_lengths:
        if time.monotonic() > phase2_deadline - 30:
            break
        t_alloc = min(time_per_length * 1.2, phase2_deadline - time.monotonic() - 30)
        if t_alloc < 40:
            break

        print("  Optimizing n=%d (budget=%.0fs)" % (n, t_alloc))
        a_opt, score = multi_start_search(n, parent if parent is not None else best_sequence,
                                           rng, t_alloc)
        if a_opt is not None:
            try_candidate(a_opt)

    print("Phase 2 best: %.6f" % best_score)

    # === Phase 3: Intensive fine-tuning ===
    print("=== Phase 3: Fine-tuning ===")
    if best_sequence is not None:
        n_best = len(best_sequence)

        # 3a: More LP iterations on the best
        remaining = deadline_global - time.monotonic() - 30
        if remaining > 40:
            t_alloc = min(80.0, remaining * 0.4)
            print("  LP fine-tune (budget=%.0fs)" % t_alloc)

            a = best_sequence.copy()
            for it in range(1000):
                if time.monotonic() > t_start_global + t_alloc + (deadline_global - t_start_global - remaining - 30):
                    break
                if time.monotonic() > deadline_global - 40:
                    break
                conv = fast_convolve(a)
                M = conv.max()
                b = lp_direction(a, M)
                if b is None:
                    break
                b = np.maximum(b, 0)
                s = b.sum()
                if s < 1e-15:
                    break
                b = b / s

                t = 0.02
                a_new = (1 - t) * a + t * b
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s < 1e-15:
                    break
                a_new = a_new / s

                new_score = exact_score(a_new)
                if new_score < best_score:
                    best_score = new_score
                    best_sequence = a_new.copy()
                a = a_new

            print("  After LP: %.6f" % best_score)

        # 3b: L-BFGS-B with different parameters
        remaining = deadline_global - time.monotonic() - 25
        if remaining > 20:
            for beta, K in [(15.0, None), (8.0, None), (None, 5), (None, 8)]:
                if time.monotonic() > deadline_global - 30:
                    break
                t_alloc = min(50.0, (deadline_global - time.monotonic() - 25) / 2)
                if t_alloc < 10:
                    break
                maxiter = int(min(300, t_alloc * 10))
                if beta is not None:
                    a_opt, score = optimize_lbfgsb(best_sequence, beta=beta, maxiter=maxiter)
                else:
                    a_opt, score = optimize_topk_lbfgsb(best_sequence, K=K, maxiter=maxiter)
                try_candidate(a_opt)

        # 3c: Perturbation + local search
        remaining = deadline_global - time.monotonic() - 20
        if remaining > 40:
            n_pert = min(6, int(remaining // 15))
            print("  Perturbation search (%d attempts)" % n_pert)
            for i in range(n_pert):
                if time.monotonic() > deadline_global - 30:
                    break
                scale = 0.005 + 0.003 * i
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(best_sequence + noise)

                t_alloc = max(15.0, (deadline_global - time.monotonic() - 25) / (n_pert - i))
                a_opt, score = hybrid_optimize(a_pert, min(t_alloc, 40.0), rng)
                try_candidate(a_opt)

        # 3d: Try a different length
        remaining = deadline_global - time.monotonic() - 15
        if remaining > 50:
            alt_n = n_best + 16 if n_best < 512 else n_best - 16
            if 64 <= alt_n <= 768:
                print("  Trying n=%d" % alt_n)
                if best_sequence is not None:
                    a0 = np.interp(np.linspace(0, 1, alt_n),
                                   np.linspace(0, 1, n_best), best_sequence)
                    a0 = np.maximum(a0, 0)
                    a0 = a0 / a0.sum()
                else:
                    a0 = window_init(alt_n, 'kaiser')
                t_alloc = min(60.0, remaining - 20)
                if t_alloc > 20:
                    a_opt, score = hybrid_optimize(a0, t_alloc, rng)
                    try_candidate(a_opt)

    # Final verification with the actual evaluator
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print("\nFINAL: score=%.6f, n=%d" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
