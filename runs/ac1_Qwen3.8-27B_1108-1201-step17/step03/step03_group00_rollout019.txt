import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M, n_constraints=None):
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix A_ub where A_ub[k, j] = f[k-j] if valid
    if n_constraints is None:
        n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n))
    for j in range(n):
        lo = max(0, j)
        hi = min(n_constraints - 1, j + n - 1)
        if lo <= hi:
            A_ub[lo:hi+1, j] = f[:hi-lo+1]
    b_ub = np.full(n_constraints, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def solve_lp_sparse(f, M):
    """Solve LP with sparse constraint matrix for efficiency."""
    from scipy.sparse import csr_matrix
    n = len(f)
    c = -np.ones(n)
    n_constraints = 2 * n - 1
    # Build sparse matrix
    rows = []
    cols = []
    vals = []

        for k in range(max(0, j), min(2*n-1, j+n)):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
    A_ub = csr_matrix((vals, (rows, cols)), shape=(n_constraints, n))
    b_ub = np.full(n_constraints, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def anti_peak_move(f, rng, strength=0.05):
    """Move mass away from positions contributing to the convolution peak."""
    n = len(f)
    conv = fast_convolve(f)
    k_max = np.argmax(conv)
    
    # Find positions contributing to conv[k_max]
    j_lo = max(0, k_max - n + 1)
    j_hi = min(n - 1, k_max)
    
    # Move mass from these positions to "safe" positions (those not contributing to top peaks)
    # Safe positions are those where the convolution value is below average
    conv_mean = conv.mean()
    safe_positions = np.where(conv < conv_mean * 1.2)[0]
    
    # Pick a random pair: one from peak contributors, one from safe
    contributors = np.arange(j_lo, j_hi + 1)
    if len(contributors) == 0 or len(safe_positions) == 0:
        return f
    
    i = contributors[rng.integers(len(contributors))]
    j = safe_positions[rng.integers(len(safe_positions))]
    
    if i == j:
        return f
    
    # Transfer mass
    amount = min(f[i], strength / n)
    f_new = f.copy()
    f_new[i] -= amount
    f_new[j] += amount
    f_new = np.maximum(f_new, 0.0)
    f_new = f_new / f_new.sum()
    
    return f_new

1. LP-based Frank-Wolfe for global optimization
2. Lp-norm projected gradient descent (FFT-accelerated) for smooth refinement

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_sparse(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    n_constraints = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(n_constraints - 1, j + n - 1)
        for k in range(k_start, k_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A_ub = csr_matrix((vals, (rows, cols)), shape=(n_constraints, n))
    b_ub = np.full(n_constraints, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

            m_exp = min(m_exp * 1.07, 5.0)

        if no_improve > 45:
            scale = 0.003 + 0.004 * min(no_improve // 20, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def lp_gradient_optimize(f0, time_limit, p=30, lr=0.02, mom=0.9, seed=0):
    """Optimize using Lp-norm of convolution as smooth surrogate with momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    current_lr = lr
    v = np.zeros(n)
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        # Compute Lp-subgradient
        conv_clipped = np.maximum(conv, 1e-30)
        weights = conv_clipped ** p
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.zeros(conv_len)
            weights[conv.argmax()] = 1.0
        else:
            weights /= wsum

        # Gradient via FFT cross-correlation
        f_rev = f[::-1]
        fft_f_rev = np.fft.rfft(f_rev, fft_size)
        fft_w = np.fft.rfft(weights, fft_size)
        corr = np.fft.irfft(fft_f_rev * fft_w, fft_size)
        grad = 2.0 * corr[n - 1:n - 1 + n]

        # Momentum update
        v = mom * v - current_lr * grad
        f_new = f + v
        f_new = project_simplex(f_new)

        new_conv = fast_convolve(f_new)
        new_obj = new_conv.max()

        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            current_lr = min(current_lr * 1.01, lr * 2)
        else:
            no_improve += 1
            # Retry with halved step
            f_new2 = f - 0.3 * grad
            f_new2 = project_simplex(f_new2)
            new_conv2 = fast_convolve(f_new2)
            if new_conv2.max() < best_obj - 1e-14:
                best_obj = new_conv2.max()
                best_f = f_new2.copy()
                f = f_new2
                no_improve = 0
            current_lr *= 0.998

        if no_improve > 300:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            no_improve = 0
            current_lr = max(lr * 0.6, 0.002)

    return best_f, best_obj


def anti_peak_diversify(f, rng, strength=0.08):
    """Move mass away from convolution peak contributors toward low-conv positions."""
    n = len(f)
    conv = fast_convolve(f)
    k_max = np.argmax(conv)

    j_lo = max(0, k_max - n + 1)
    j_hi = min(n - 1, k_max)
    contributors = np.arange(j_lo, j_hi + 1)

    conc = conv.max() / (conv.mean() + 1e-30)
    if conc < 1.5:
        return f

    # Find low-convolution target positions
    target_threshold = np.median(conv)
    targets = np.where(conv < target_threshold)[0]
    if len(targets) == 0:
        return f

    f_new = f.copy()
    # Move mass from multiple contributors to a target
    n_moves = min(5, len(contributors))
    for _ in range(n_moves):
        i = contributors[rng.integers(len(contributors))]
        j = targets[rng.integers(len(targets))]
        if i == j:
            continue
        amount = min(f_new[i] * 0.1, strength / n)
        f_new[i] -= amount
        f_new[j] += amount

    f_new = np.maximum(f_new, 0.0)
    fs = f_new.sum()
    if fs < 1e-15:
        return f
    return f_new / fs


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.4))

    x = np.arange(n, dtype=np.float64)

    # Raised cosine (tapered)
    rc = 0.5 * (1 + np.cos(np.pi * 2 * x / (n - 1) - np.pi))
    if rc.sum() > 0:
        inits.append(rc / rc.sum())

    # Gaussian center
    g = np.exp(-4.0 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits.append(g / g.sum())

    # Two-peaked
    two = np.exp(-15 * ((x - 0.25 * (n - 1)) / (0.2 * n)) ** 2) + \
          np.exp(-15 * ((x - 0.75 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(two / two.sum())

    # Cosine modulated
    cmod = (1.0 + 0.5 * np.cos(2 * np.pi * x / n)) * np.exp(-2.0 * (x / n - 0.5) ** 2)
    if cmod.sum() > 0:
        inits.append(cmod / cmod.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Noisy parent
        pp = px + rng.standard_normal(n) * 0.02
        pp = np.maximum(pp, 0)
        if pp.sum() > 1e-10:
            inits.append(pp / pp.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining LP-based Frank-Wolfe,
    Lp-norm gradient descent, structured diversification, and length extension.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if len(a) == 0:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence((parent * parent.sum()).tolist())
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Exploration (~350s) ===
    print("=== Phase 1: Exploration ===")
    t1 = time.monotonic()

    if parent is not None:
        t_alloc = min(150, deadline - time.monotonic() - 700)
        if t_alloc > 20:
            print(f"  FW on parent (n={len(parent)}), t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(parent, time_limit=t_alloc, seed=seed)
            try_seq(a_opt, "parent_fw")

            # Lp gradient on parent
            t_alloc = min(80, deadline - time.monotonic() - 600)
            if t_alloc > 15:
                a_opt, _ = lp_gradient_optimize(parent, time_limit=t_alloc, p=25, lr=0.02, mom=0.85, seed=seed+1)
                try_seq(a_opt, "parent_lp")

    # Try other lengths
    other_lengths = [384, 256, 512, 192, 640]
    if parent is not None:
        other_lengths = [l for l in other_lengths if l != len(parent)]

    for n in other_lengths[:4]:
        if time.monotonic() > deadline - 550:
            break
        t_alloc = min(50, deadline - time.monotonic() - 500)
        if t_alloc < 15:
            break
        f_init = np.ones(n) / n
        a_opt, _ = fw_lp_optimize(f_init, time_limit=t_alloc, seed=seed + n)
        try_seq(a_opt, f"n{n}_uni")

    # Try a non-uniform init on parent length
    if parent is not None and time.monotonic() < deadline - 500:
        n_p = len(parent)
        inits = generate_inits(n_p, parent, rng)
        for i, finit in enumerate(inits[1:4]):
            t_alloc = min(40, deadline - time.monotonic() - 450)
            if t_alloc < 10:
                break
            a_opt, _ = fw_lp_optimize(finit, time_limit=t_alloc, seed=seed + 100 + i)
            try_seq(a_opt, f"n{n_p}_init{i}")

    print(f"  Phase 1 done in {time.monotonic()-t1:.0f}s, best={best_score:.6f}")

    # === Phase 2: Deep optimization (~900s) ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    t2 = time.monotonic()
    iter_count = 0

    if best_seq is not None:
        while time.monotonic() < deadline - 250:
            iter_count += 1
            n_best = len(best_seq)

            # Alternate optimizers
            if iter_count % 3 == 1:
                # FW-LP
                t_alloc = min(100, (deadline - time.monotonic() - 200) // 3)
                if t_alloc < 15:
                    break
                a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, seed=seed + iter_count)
                try_seq(a_opt, f"fw_{iter_count}")

            elif iter_count % 3 == 2:
                # Lp-gradient with varying p
                p_val = [20, 35, 50][iter_count % 3]
                lr_val = [0.02, 0.015, 0.01][iter_count % 3]
                mom_val = [0.85, 0.9, 0.92][iter_count % 3]
                t_alloc = min(80, (deadline - time.monotonic() - 150) // 3)
                if t_alloc < 15:
                    break
                a_opt, _ = lp_gradient_optimize(best_seq, time_limit=t_alloc,
                                                p=p_val, lr=lr_val, mom=mom_val,
                                                seed=seed + iter_count)
                try_seq(a_opt, f"lp_{iter_count}")

            else:
                # Diversify then optimize
                a_div = anti_peak_diversify(best_seq, rng, strength=0.06 + 0.02 * (iter_count % 4))
                t_alloc = min(60, (deadline - time.monotonic() - 100) // 3)
                if t_alloc < 10:
                    break
                a_opt, _ = fw_lp_optimize(a_div, time_limit=t_alloc, seed=seed + iter_count)
                try_seq(a_opt, f"div_{iter_count}")

            # Occasionally try a random restart from best
            if iter_count % 5 == 0 and time.monotonic() < deadline - 200:
                noise = rng.standard_normal(n_best) * (0.005 + 0.003 * (iter_count % 3))
                f_pert = project_simplex(best_seq + noise)
                t_alloc = min(40, deadline - time.monotonic() - 100)
                if t_alloc > 10:
                    a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc, seed=seed + 1000 + iter_count)
                    try_seq(a_opt, f"restart_{iter_count}")

    print(f"  Phase 2 done in {time.monotonic()-t2:.0f}s, best={best_score:.6f}")

    # === Phase 3: Length extension (~200s) ===
    print(f"\n=== Phase 3: Length Extension ===")

    if best_seq is not None:
        n_best = len(best_seq)
        candidate_lengths = sorted(set([n_best + 64, n_best + 128, n_best * 2, n_best + 256]))
        candidate_lengths = [l for l in candidate_lengths if l <= 2048]

        for n_new in candidate_lengths[:2]:
            if time.monotonic() > deadline - 100:
                break
            f_ext = np.interp(
                np.linspace(0, 1, n_new),
                np.linspace(0, 1, n_best),
                best_seq
            )
            f_ext = np.maximum(f_ext, 0)
            fes = f_ext.sum()
            if fes < 1e-10:
                continue
            f_ext = f_ext / fes

            t_alloc = min(70, deadline - time.monotonic() - 60)
            if t_alloc < 15:
                break
            a_opt, _ = fw_lp_optimize(f_ext, time_limit=t_alloc, seed=seed + 5000)
            try_seq(a_opt, f"ext_{n_new}")

            # Lp polish on extended
            t_alloc = min(40, deadline - time.monotonic() - 50)
            if t_alloc > 10:
                a_opt, _ = lp_gradient_optimize(f_ext, time_limit=t_alloc, p=25, lr=0.015, mom=0.9, seed=seed + 5001)
                try_seq(a_opt, f"ext_lp_{n_new}")

    # === Phase 4: Final polishing (~150s) ===
    print(f"\n=== Phase 4: Final Polish ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Fine Lp with high p
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            a_opt, _ = lp_gradient_optimize(best_seq, time_limit=min(rem, 50),
                                            p=60, lr=0.008, mom=0.93, seed=seed + 9000)
            try_seq(a_opt, "lp_final")

        # FW on the result
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(rem, 35), seed=seed + 9001)
            try_seq(a_opt, "fw_final")

        # Very small perturbations
        for s in range(4):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.0015
            f_p = project_simplex(best_seq + noise)
            t_alloc = min(rem, 15)
            a_opt, _ = fw_lp_optimize(f_p, time_limit=t_alloc, seed=seed + 9100 + s)
            try_seq(a_opt, f"micro_{s}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
