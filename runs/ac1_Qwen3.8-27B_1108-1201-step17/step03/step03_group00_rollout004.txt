def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def conv_subgradient(f, conv, top_k, temp):
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        sg[idx] += w[i] * 2.0 * f[k - idx]
    return sg

def adam_descent(f0, time_limit, top_k=50, init_temp=0.0005, lr=0.005, verbose=False):
    f = f0.astype(np.float64).copy()
    n = len(f)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.75, 0.999, 1e-8
    t = 0
    temp = init_temp
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        sg = conv_subgradient(f, conv, top_k, temp)
        
        m = b1 * m + (1 - b1) * sg
        v = b2 * v + (1 - b2) * (sg * sg)
        mhat = m / (1.0 - b1**t)
        vhat = v / (1.0 - b2**t)
        
        lr_t = lr / (1.0 + 0.0005 * t)
        f_new = f - lr_t * mhat / (np.sqrt(vhat) + eps)
        f_new = project_simplex(f_new)
        f = f_new
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        temp *= 1.001
        # Also periodically reset temperature
        if t % 500 == 0:
            temp = max(temp, 0.0005)
        
        if no_improve > 300:

            scale = 0.005 + 0.01 * min(no_improve / 500.0, 1.0)

            v = np.zeros(n)
            t = 0
            temp = init_temp
            lr = lr * 0.8
            no_improve = 0
            conv = fast_convolve(f)
            if verbose:
                print(f"  Restart at iter {iter_count}, best={best_obj:.10f}")
        
        iter_count += 1
    
    return best_f, best_obj

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent: try adjusting one coordinate at a time."""
    f = f0.astype(np.float64).copy()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Find the argmax of convolution
        k_star = int(np.argmax(conv))
        
        # For this k, the gradient w.r.t. a[i] is 2*a[k-i] for i in valid range
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        
        # Try to reduce a[i] for the indices that contribute most to conv[k_star]
        # The contribution of a[i] to conv[k_star] is 2*a[i]*a[k-i]
        # We want to reduce the total while maintaining sum=1
        
        # Simple approach: find the index with the largest a[i]*a[k-i] product
        # and try to reduce it slightly
        idx = np.arange(lo, hi + 1)
        products = f[idx] * f[k_star - idx]
        
        # Try reducing the largest contributor and increasing a random other index
        i_max = idx[np.argmax(products)]
        
        # Try several perturbations
        improved = False
        for delta in [0.001, 0.005, 0.01, 0.02]:
            f_try = f.copy()
            # Reduce a[i_max] by delta, distribute to random positions
            if f[i_max] > delta:
                f_try[i_max] -= delta
                # Add to a random position (or spread evenly)
                other_idx = np.random.randint(n)
                if other_idx != i_max:
                    f_try[other_idx] += delta
                    new_obj = fast_convolve(f_try).max()
                    if new_obj < best_obj - 1e-12:
                        f = f_try
                        best_obj = new_obj
                        best_f = f.copy()
                        conv = fast_convolve(f)
                        improved = True
                        break
        
        if not improved:
            no_improve += 1
            if no_improve > 5000:
                break
        else:
            no_improve = 0
    
    return best_f, best_obj

def solve_lp_direction(f, M):
    """Solve LP to find a direction that keeps convolution bounded."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def fw_lp_step(f0, time_limit):
    """Frank-Wolfe with LP oracle, simplified version."""
    f = f0.astype(np.float64).copy()
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        g = solve_lp_direction(f, M * m_exp)
        if g is None:
            m_exp *= 1.2
            if m_exp > 5.0:
                m_exp = 1.0
            continue
        
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs
        
        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.3, 0.5, 0.7, 0.9]:
            ft = (1 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = (1 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-12:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)
        
        if no_improve > 80:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj

def generate_initialization(n, parent=None, rng=None):

    # Gaussian

        if ps > 1e-10:
