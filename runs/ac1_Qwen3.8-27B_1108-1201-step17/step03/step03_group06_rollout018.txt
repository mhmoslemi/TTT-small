"""
Convex optimization for minimum peak auto-convolution sequences.
Uses Nesterov-accelerated smoothed gradient on the simplex with adaptive
log-sum-exp, coordinate descent polishing, and multi-length exploration.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, beta):
    """Compute gradient of log-sum-exp(beta * conv(a,a)) approximation of max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # Gradient of sum_k w[k] * conv(a,a)[k] w.r.t. a[i]
    # = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
    # First term: sum_k w[k] * a[k-i] = cross-correlation
    # Second term: sum_k w[k] * a[i-k] = convolution of w and a
    size = 2 * n
    fft_size = 1 << (size - 1).bit_length()

    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a

    W = np.fft.fft(w_padded)
    A = np.fft.fft(a_padded)

    # cross[i] = sum_k w[k] * a[k-i] = ifft(fft(a) * conj(fft(w)))[i]
    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    # conv_wa[i] = sum_k w[k] * a[i-k] = ifft(fft(a) * fft(w))[i]
    conv_wa = np.fft.ifft(A * W).real[:n]

    grad = cross + conv_wa
    return grad, conv


def compute_subgradient(a):
    """Compute exact subgradient of max(conv(a,a)) using active set."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    tol = 1e-8 * M
    active = np.where(conv >= M - tol)[0]

    # For each active k, gradient w.r.t. a[i] is a[k-i] + a[i-k] (with bounds)
    grad = np.zeros(n)
    for k in active:
        # a[k-i] valid when 0 <= k-i < n, i.e., max(0,k-n+1) <= i <= min(n-1,k)
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        vals = a[k - idx]
        grad[idx] += vals
        # For the second term a[i-k], valid when 0 <= i-k < n, i.e., k <= i <= k+n-1
        # But i < n, so: max(0,k) <= i <= min(n-1, k+n-1) = n-1 (since k < 2n)
        j_min = max(0, k)
        j_max = min(n - 1, k + n - 1)
        if j_min <= j_max:
            jdx = np.arange(j_min, j_max + 1)
            jvals = a[jdx - k]
            grad[jdx] += jvals
        # Fix double counting when k-i = i (i.e., k = 2i)
        # Already handled correctly since both terms contribute a[i] to grad[i]

    return grad / len(active), conv, M


def nesterov_smoothed(a0, time_limit, beta_init=200.0, beta_final=5000.0,
                       lr_scale=0.8, seed=0):
    """Nesterov-accelerated gradient on smoothed objective on the simplex."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    total_iters = 0
    beta = beta_init
    st = a.copy()  # Nesterov point
    v = np.zeros(n)

    no_improve = 0
    max_no_improve = 150

    while time.monotonic() - t0 < time_limit:
        # Nesterov: evaluate gradient at st
        grad, conv_st = compute_smooth_gradient(st, beta)
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        # Step size: approximately 1/L where L ~ beta * max(conv) * n / (n-1)
        # Use conservative estimate
        lr = lr_scale / (beta * best_obj * 1.5 + 1e-10)
        lr = min(lr, 0.5)

        # Momentum
        mu = (1.0 + np.sqrt(1.0 + 4.0 * (total_iters + 1) ** 2)) / 2.0
        v = ((total_iters) / mu) * v - lr * grad
        st = project_simplex(a + v)
        # Inertial step
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (total_iters + 1) ** 2)) / 2.0
        a_new = st + ((total_iters + 1) / t_k) * (st - a)
        # Actually, standard Nesterov: a_new = project(st + mom*(st - a))
        # Let me use a simpler momentum scheme for stability on the simplex
        a_new = project_simplex(st + 0.9 * (st - a))

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        total_iters += 1

        # Adaptive beta: increase beta to sharpen the max approximation
        if total_iters % 50 == 0:
            progress = min(1.0, (time.monotonic() - t0) / time_limit)
            beta = beta_init * (beta_final / beta_init) ** progress

        # Stuck detection: perturb and restart
        if no_improve >= max_no_improve:
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            st = a.copy()
            v = np.zeros(n)
            beta = max(beta_init, beta * 0.5)
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.1)

    conv_final = fast_convolve(best_a)
    return best_a, conv_final.max()


def subgradient_descent(a0, time_limit, max_active=50, seed=0):
    """Pure subgradient method with averaging for exact max."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    t = 0
    a_avg = a.copy()
    t_sum = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        grad, conv, M = compute_subgradient(a)
        # Truncate to top active set if too large
        # (already handled in compute_subgradient by tol)

        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        # Polyak-type step size
        lr = 2.0 / (t + 5) * best_obj * 0.5
        lr = min(lr, 0.3)

        a_new = project_simplex(a - lr * grad)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        # Running average
        t += 1
        t_sum += 1
        a_avg = a_avg + (1.0 / t_sum) * (a_new - a_avg)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Check average
        conv_avg = fast_convolve(a_avg)
        if conv_avg.max() < best_obj:
            best_obj = conv_avg.max()
            best_a = a_avg.copy()
            no_improve = 0

        a = a_new

        if no_improve >= 200:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            a_avg = a.copy()
            no_improve = 0

    conv_final = fast_convolve(best_a)
    return best_a, conv_final.max()


def coordinate_descent(a0, time_limit, seed=0):
    """Pairwise coordinate descent on the simplex."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Random pairs
        num_pairs = min(n * 2, 2000)
        pairs = []
        for _ in range(num_pairs):
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                pairs.append((i, j))

        for (i, j) in pairs:
            if time.monotonic() - t0 > time_limit:
                break
            # Optimize a[i] and a[j] with a[i]+a[j] = const
            c_val = a[i] + a[j]
            if c_val < 1e-15:
                continue

            # Bisection on a[i] in [0, c_val]
            best_ai = a[i]
            best_local = best_obj

            # Evaluate at a few points
            fracs = [0.0, 0.1, 0.25, 0.4, 0.5, 0.6, 0.75, 0.9, 1.0]
            for f in fracs:
                ai_test = f * c_val
                aj_test = c_val - ai_test
                a[i] = ai_test
                a[j] = aj_test
                # Only need to check affected convolution entries
                # Entries affected: k where k-i or i-k is in valid range, or k-j or j-k
                # For speed, just compute full conv (n is moderate)
                conv_test = fast_convolve(a)
                obj_test = conv_test.max()
                if obj_test < best_local:
                    best_local = obj_test
                    best_ai = ai_test

            # Refine around best
            if 0 < best_ai < c_val:
                lo = max(0, best_ai - 0.05 * c_val)
                hi = min(c_val, best_ai + 0.05 * c_val)
                for _ in range(10):
                    mid = (lo + hi) / 2
                    a[i] = mid
                    a[j] = c_val - mid
                    conv_test = fast_convolve(a)
                    obj_test = conv_test.max()
                    if obj_test < best_local:
                        best_local = obj_test
                        best_ai = mid
                        lo = mid
                    else:
                        hi = mid
                a[i] = best_ai
                a[j] = c_val - best_ai
            elif best_ai == 0.0:
                a[i] = 0.0
                a[j] = c_val
            else:
                a[i] = c_val
                a[j] = 0.0

            if best_local < best_obj - 1e-14:
                best_obj = best_local
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

    conv_final = fast_convolve(best_a)
    return best_a, conv_final.max()


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Bimodal (mass at edges)
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Raised cosine
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) if n > 1 else np.ones(1)
    cos_w = np.maximum(cos_w, 0.01)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss_center'] = g

    # Gaussian wide
    g2 = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.4 * n)) ** 2)
    g2 /= g2.sum()
    inits['gauss_wide'] = g2

    # Triangular (peak at center)
    tri = 1.0 - 2.0 * np.abs(x - 0.5 * (n - 1)) / (n - 1) if n > 1 else np.ones(1)
    tri = np.maximum(tri, 0.01)
    tri /= tri.sum()
    inits['triangle_c'] = tri

    # Random (spiky)
    inits['random_spiky'] = rng.dirichlet(np.ones(n) * 0.1)
    # Random (smooth)
    inits['random_smooth'] = rng.dirichlet(np.ones(n) * 1.0)

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def iterative_refinement(a0, time_limit, method='nesterov', seed=0):
    """Iteratively alternate between different optimization methods."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    phase_time = min(30.0, time_limit / 5)

    methods = ['nesterov', 'subgrad', 'nesterov', 'subgrad', 'nesterov',
               'coord', 'nesterov', 'coord']
    method_idx = 0

    while time.monotonic() - t0 < time_limit - 5:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < phase_time:
            break

        m = methods[method_idx % len(methods)]
        t_alloc = min(phase_time, remaining - 5)
        t_alloc = max(10.0, t_alloc)

        if m == 'nesterov':
            # Use progressively higher beta
            beta_mult = 1 + method_idx * 0.5
            a_opt, obj_opt = nesterov_smoothed(
                a, time_limit=t_alloc,
                beta_init=200 * beta_mult,
                beta_final=min(10000, 5000 * beta_mult),
                seed=seed + method_idx)
        elif m == 'subgrad':
            a_opt, obj_opt = subgradient_descent(
                a, time_limit=t_alloc, seed=seed + method_idx + 100)
        elif m == 'coord':
            a_opt, obj_opt = coordinate_descent(
                a, time_limit=t_alloc, seed=seed + method_idx + 200)
        else:
            a_opt, obj_opt = a, best_obj

        if obj_opt < best_obj - 1e-13:
            best_obj = obj_opt
            best_a = a_opt.copy()
            a = a_opt.copy()

        method_idx += 1

    conv_final = fast_convolve(best_a)
    return best_a, conv_final.max()


def fine_perturbation_search(a0, time_limit, seed=0):
    """Very fine local search with structured perturbations."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()

    # Target the peak positions for reduction
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        peak_idx = np.argmax(conv)

        # Identify which a[i] contribute most to the peak
        i_min = max(0, peak_idx - n + 1)
        i_max = min(n - 1, peak_idx)
        contributors = np.arange(i_min, i_max + 1)
        contrib_vals = a[contributors] * a[peak_idx - contributors]

        # Reduce the largest contributors slightly, redistribute mass
        if len(contributors) > 2:
            top_contrib = np.argsort(contrib_vals)[::-1][:min(5, len(contributors))]
            # Move small amount from top contributors to non-contributors
            non_contrib = np.setdiff1d(np.arange(n), contributors)
            if len(non_contrib) > 0:
                eps = 0.001 * M / (a[contributors[top_contrib[0]]] + 1e-15)
                eps = min(eps, 0.01)

                for tc in top_contrib:
                    idx = contributors[tc]
                    a[idx] -= eps
                    # Distribute to random non-contributor
                    target = non_contrib[rng.integers(len(non_contrib))]
                    a[target] += eps

                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s

                conv_new = fast_convolve(a)
                if conv_new.max() < best_obj - 1e-14:
                    best_obj = conv_new.max()
                    best_a = a.copy()
                else:
                    # Revert
                    a = best_a.copy()

        # Also try random fine perturbations
        for scale in [0.0005, 0.001, 0.002, 0.005]:
            noise = rng.standard_normal(n) * scale
            a_pert = project_simplex(a + noise)
            conv_pert = fast_convolve(a_pert)
            if conv_pert.max() < best_obj - 1e-14:
                best_obj = conv_pert.max()
                best_a = a_pert.copy()
                a = a_pert.copy()

    conv_final = fast_convolve(best_a)
    return best_a, conv_final.max()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization for minimum peak auto-convolution sequences.
    Combines Nesterov-accelerated smoothed gradient, subgradient, coordinate descent,
    and fine perturbation search across multiple lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: length={len(p)}, sum={p.sum():.2f}")
    except NameError:
        pass
    except Exception:
        pass

    def try_sequence(a):
        """Evaluate and update best if improved."""
        nonlocal best_sequence, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  New best: score={score:.8f}, n={len(a)}")

    # ========== Phase 1: Quick Exploration (150s) ==========
    print("=== Phase 1: Quick Exploration ===")
    t_phase1 = time.monotonic()
    phase1_time = 150.0

    lengths = [280, 300, 320, 340, 360, 384, 400, 448, 512]

    # Prioritize length 320 (previous best)
    lengths_priority = [320] + [l for l in lengths if l != 320]

    for n in lengths_priority:
        if time.monotonic() + 10 > t_phase1 + phase1_time:
            break

        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())

        # Priority order
        priority = ['parent', 'parent_ds', 'parent_up', 'bimodal',
                    'random_spiky', 'cosine', 'gauss_center', 'uniform']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:6]

        time_remaining = t_phase1 + phase1_time - time.monotonic()
        time_per = max(5.0, time_remaining / (len(lengths_priority) - lengths_priority.index(n) + 1)
                       / max(1, len(init_names)))

        for name in init_names:
            if time.monotonic() + 5 > t_phase1 + phase1_time:
                break
            t_alloc = min(time_per, t_phase1 + phase1_time - time.monotonic() - 5)
            if t_alloc < 4:
                break

            a0 = inits[name].copy()
            # Quick optimization
            a_opt, obj_opt = nesterov_smoothed(
                a0, time_limit=t_alloc,
                beta_init=300, beta_final=3000,
                seed=n + hash(name) % 997)

            try_sequence(a_opt)

    print(f"Phase 1 done. Best so far: {best_score:.8f}")

    # ========== Phase 2: Deep Nesterov Optimization (600s) ==========
    print("=== Phase 2: Deep Nesterov Optimization ===")
    phase2_start = time.monotonic()
    phase2_time = 600.0

    if best_sequence is None:
        best_sequence = np.ones(320) / 320
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    n_best = len(best_sequence)

    # Multiple runs with different parameters
    nesterov_configs = [
        {'beta_init': 150, 'beta_final': 4000, 'lr_scale': 0.9},
        {'beta_init': 300, 'beta_final': 6000, 'lr_scale': 0.7},
        {'beta_init': 500, 'beta_final': 8000, 'lr_scale': 0.6},
        {'beta_init': 100, 'beta_final': 2000, 'lr_scale': 1.0},
        {'beta_init': 400, 'beta_final': 10000, 'lr_scale': 0.5},
        {'beta_init': 200, 'beta_final': 5000, 'lr_scale': 0.8},
    ]

    # Starting points
    starts = [best_sequence.copy()]
    # Add perturbed versions
    for i in range(4):
        noise = rng.standard_normal(n_best) * 0.02
        starts.append(project_simplex(best_sequence + noise))
    # Different initializations at best length
    inits_best = generate_initializations(n_best, parent, rng)
    for extra in ['bimodal', 'random_spiky', 'cosine']:
        if extra in inits_best:
            starts.append(inits_best[extra].copy())

    total_runs = len(starts) * len(nesterov_configs)
    time_per_run = phase2_time / max(1, total_runs)
    time_per_run = max(15.0, time_per_run)

    run_count = 0
    for i, a_start in enumerate(starts):
        for j, cfg in enumerate(nesterov_configs):
            if time.monotonic() + 10 > phase2_start + phase2_time:
                break
            t_alloc = min(time_per_run, phase2_start + phase2_time - time.monotonic() - 10)
            if t_alloc < 10:
                break

            a_opt, obj_opt = nesterov_smoothed(
                a_start, time_limit=t_alloc,
                beta_init=cfg['beta_init'], beta_final=cfg['beta_final'],
                lr_scale=cfg['lr_scale'], seed=run_count * 13 + 7)

            try_sequence(a_opt)
            run_count += 1

    print(f"Phase 2 done. Best so far: {best_score:.8f}")

    # ========== Phase 3: Subgradient + Coordinate Descent (350s) ==========
    print("=== Phase 3: Subgradient + Coord Descent ===")
    phase3_start = time.monotonic()
    phase3_time = 350.0

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Subgradient runs
        t_alloc = min(100.0, phase3_time * 0.3)
        if t_alloc > 15:
            a_opt, obj_opt = subgradient_descent(
                best_sequence, time_limit=t_alloc, seed=301)
            try_sequence(a_opt)

        # More Nesterov from current best
        t_alloc = min(100.0, phase3_time * 0.3)
        if t_alloc > 15:
            a_opt, obj_opt = nesterov_smoothed(
                best_sequence, time_limit=t_alloc,
                beta_init=2000, beta_final=15000,
                lr_scale=0.5, seed=401)
            try_sequence(a_opt)

        # Coordinate descent
        t_alloc = min(100.0, phase3_time * 0.3)
        if t_alloc > 15:
            a_opt, obj_opt = coordinate_descent(
                best_sequence, time_limit=t_alloc, seed=501)
            try_sequence(a_opt)

        # Another round of Nesterov from best
        remaining = phase3_start + phase3_time - time.monotonic() - 10
        if remaining > 15:
            a_opt, obj_opt = nesterov_smoothed(
                best_sequence, time_limit=remaining,
                beta_init=3000, beta_final=20000,
                lr_scale=0.4, seed=601)
            try_sequence(a_opt)

    print(f"Phase 3 done. Best so far: {best_score:.8f}")

    # ========== Phase 4: Iterative Refinement (200s) ==========
    print("=== Phase 4: Iterative Refinement ===")
    phase4_start = time.monotonic()
    phase4_time = 200.0

    if best_sequence is not None:
        t_alloc = min(phase4_time, deadline - time.monotonic() - 30)
        if t_alloc > 30:
            a_opt, obj_opt = iterative_refinement(
                best_sequence, time_limit=t_alloc, seed=701)
            try_sequence(a_opt)

    print(f"Phase 4 done. Best so far: {best_score:.8f}")

    # ========== Phase 5: Fine Perturbation Search (remaining) ==========
    print("=== Phase 5: Fine Perturbation Search ===")
    phase5_start = time.monotonic()
    phase5_time = max(10.0, deadline - time.monotonic() - 10)

    if best_sequence is not None:
        a_opt, obj_opt = fine_perturbation_search(
            best_sequence, time_limit=phase5_time, seed=801)
        try_sequence(a_opt)

    # Final Nesterov polish with very high beta
    remaining = deadline - time.monotonic() - 10
    if best_sequence is not None and remaining > 15:
        a_opt, obj_opt = nesterov_smoothed(
            best_sequence, time_limit=min(remaining, 60),
            beta_init=5000, beta_final=30000,
            lr_scale=0.3, seed=901)
        try_sequence(a_opt)

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(320) / 320

    # Ensure all values are in valid range
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(320) / 320
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {final_score:.8f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
