```python
import numpy as np
import time

def fft_convolve(a):
    """Compute linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    rho_final = np.max(rho[cond]) if cond.any() else 1
    theta = (cssv[rho_final - 1] - 1) / rho_final
    return np.maximum(v - theta, 0)

def subgradient_conv_max(a):
    """Compute subgradient of max(conv(a)) at a."""
    n = len(a)
    conv = fft_convolve(a)
    k_star = np.argmax(conv)
    g = np.zeros(n)
    # term1: a[k*-j] for valid j
    j_min1 = max(0, k_star - n + 1)
    j_max1 = min(k_star, n - 1)
    if j_min1 <= j_max1:
        idx = np.arange(j_min1, j_max1 + 1)
        g[idx] += 2.0 * a[k_star - idx]
    # term2: a[j-k*] for valid j (only when k* < n)
    if k_star < n:
        j_min2 = k_star
        j_max2 = n - 1
        if j_min2 <= j_max2:
            idx = np.arange(j_min2, j_max2 + 1)
            g[idx] += 2.0 * a[idx - k_star]
    return g * (2.0 * n), k_star

def fista_optimize(a0, max_iter=10000, lr=None, tol=1e-10):
    """FISTA optimization for min 2n*max(conv) s.t. a in simplex."""
    n = len(a0)
    a = project_simplex(a0)
    y = a.copy()
    t = 1.0
    
    if lr is None:
        # Estimate Lipschitz constant
        # The gradient of 2n*conv[k] has norm bounded by ~4n*||a||_inf*n ≈ 4n^2 (rough)
        # Use a conservative estimate
        lr = 1.0 / (4.0 * n)
    
    best_obj = np.inf
    best_a = a.copy()
    
    for it in range(max_iter):
        # Compute subgradient at y
        g, k_star = subgradient_conv_max(y)
        
        # FISTA update
        a_new = project_simplex(y - lr * g)
        
        # Compute objective
        conv = fft_convolve(a_new)
        obj = 2.0 * n * conv.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        if abs(obj - 2.0 * n * fft_convolve(a).max()) < tol and it > 100:
            break
        
        # Momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        t = t_new
        
        # Decay learning rate
        lr *= 0.9999
    
    return best_a, best_obj
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FISTA with exact subgradient,
adaptive momentum, frequency-domain initializations, and multi-length search.
"""

import numpy as np
import time


def fft_convolve(a):
    """Compute linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_max = rho[cond][-1]
    theta = (cssv[rho_max - 1] - 1.0) / rho_max
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, k_star):
    """Compute subgradient of max(conv) at a, given argmax k_star."""
    n = len(a)
    g = np.zeros(n)
    # term1: 2*a[k*-j] for valid j
    j_min1 = max(0, k_star - n + 1)
    j_max1 = min(k_star, n - 1)
    if j_min1 <= j_max1:
        idx = np.arange(j_min1, j_max1 + 1)
        g[idx] += 2.0 * a[k_star - idx]
    # term2: 2*a[j-k*] for valid j (only when k* < n)
    if k_star < n:
        j_min2 = k_star
        j_max2 = n - 1
        if j_min2 <= j_max2:
            idx = np.arange(j_min2, j_max2 + 1)
            g[idx] += 2.0 * a[idx - k_star]
    return g


def fista_optimize(a0, max_iter=50000, init_lr=None, time_limit=None, t_start=None):
    """FISTA with exact subgradient and adaptive momentum."""
    n = len(a0)
    a = project_simplex(a0.astype(np.float64))
    y = a.copy()
    t_val = 1.0
    
    if init_lr is None:
        init_lr = 1.0 / (4.0 * n)
    lr = init_lr
    
    best_obj = np.inf
    best_a = a.copy()
    
    for it in range(max_iter):
        if time_limit is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit:
                break
        
        # Compute subgradient at y
        conv_y = fft_convolve(y)
        k_star = np.argmax(conv_y)
        g = compute_subgradient(y, k_star) * (2.0 * n)
        
        # Gradient step + projection
        a_new = project_simplex(y - lr * g)
        
        # Compute objective at a_new
        conv_new = fft_convolve(a_new)
        obj = 2.0 * n * conv_new.max()
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        # Adaptive momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        momentum = (t_val - 1.0) / t_new
        # Damp momentum if objective increases
        if obj > best_obj * 1.001:
            momentum *= 0.5
        y = a_new + momentum * (a_new - a)
        y = project_simplex(y)  # Keep y in simplex
        t_val = t_new
        
        # Very slow decay of lr
        lr *= 0.99999
        
        # Convergence check
        if it % 100 == 0 and obj - best_obj < 1e-12:
            break
    
    return best_a, best_obj


def smooth_obj_and_grad(a, beta):
    """Compute smoothed objective and gradient using log-sum-exp."""
    n = len(a)
    conv = fft_convolve(a)
    # Log-sum-exp for smooth max
    c = conv.max()
    shifted = beta * (conv - c)
    shifted = np.clip(shifted, -50, 50)
    exp_s = np.exp(shifted)
    sm = c + np.log(exp_s.sum()) / beta
    w = exp_s / exp_s.sum()
    obj = 2.0 * n * sm
    # Gradient: 2n * sum_k w[k] * grad(conv[k])
    grad = np.zeros(n)
    # Efficient: grad[j] = 2n * sum_k w[k] * (2*a[k-j] + 2*a[j-k])
    # = 4n * (cross-correlation of w with a, both directions)
    # Use FFT for efficiency
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    grad = np.fft.irfft(np.conjugate(np.fft.fft(w, N)) * np.fft.fft(a, N), N)[:n] * (4.0 * n)
    return obj, grad, sm


def fista_smooth(a0, beta, max_iter=10000, init_lr=None, time_limit=None, t_start=None):
    """FISTA with smoothed objective."""
    n = len(a0)
    a = project_simplex(a0.astype(np.float64))
    y = a.copy()
    t_val = 1.0
    
    if init_lr is None:
        init_lr = 1.0 / (4.0 * n + beta)
    lr = init_lr
    
    best_obj = np.inf
    best_a = a.copy()
    
    for it in range(max_iter):
        if time_limit is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit:
                break
        
        obj, grad, _ = smooth_obj_and_grad(y, beta)
        
        a_new = project_simplex(y - lr * grad)
        
        # Evaluate true objective
        conv_new = fft_convolve(a_new)
        true_obj = 2.0 * n * conv_new.max()
        
        if true_obj < best_obj:
            best_obj = true_obj
            best_a = a_new.copy()
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        momentum = (t_val - 1.0) / t_new
        y = a_new + momentum * (a_new - a)
        y = project_simplex(y)
        t_val = t_new
        lr *= 0.9999
    
    return best_a, best_obj


def init_flat_spectrum(n, rng):
    """Initialize with flat power spectrum (random phases)."""
    N = 1 << (n - 1).bit_length()
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    A = np.zeros(N, dtype=complex)
    for k in range(N // 2 + 1):
        A[k] = np.exp(1j * phases[k])
        if k < N - k:
            A[N - k] = np.exp(-1j * phases[k])
    a = np.fft.ifft(A).real[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def init_from_parent(parent, n):
    """Resample parent to length n."""
    if parent is None or len(parent) == 0:
        return None
    parent = np.maximum(parent, 0)
    s = parent.sum()
    if s < 1e-10:
        return None
    x_new = np.linspace(0, 1, n)
    x_old = np.linspace(0, 1, len(parent))
    a = np.interp(x_new, x_old, parent / s)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    return a


def peak_shave(a, n_shaves=10, factor=0.9):
    """Reduce the peak of the autocorrelation."""
    n = len(a)
    a = a.copy()
    for _ in range(n_shaves):
        conv = fft_convolve(a)
        k_star = np.argmax(conv)
        peak_val = conv[k_star]
        mean_val = conv.mean()
        if peak_val <= mean_val * 1.01:
            break
        # Reduce elements that contribute to the peak
        j_min = max(0, k_star - n + 1)
        j_max = min(k_star, n - 1)
        # The contribution of a[j] to conv[k_star] is proportional to a[k_star-j]
        contrib = np.zeros(n)
        for j in range(j_min, j_max + 1):
            contrib[j] = a[k_star - j]
            if k_star < n and 0 <= j - k_star < n:
                contrib[j] += a[j - k_star]
        total = contrib.sum()
        if total > 1e-15:
            reduction = (peak_val - mean_val) * (1 - factor)
            a[:] -= reduction * contrib / total
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a /= s
    return a
