def targeted_peak_reduction(a, n_peaks=3, n_trials=10):
    """
    For each of the top n_peaks peaks, try targeted reductions.
    Returns the best a found.
    """
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()
    
    # Get top peaks
    peak_idxs = np.argsort(conv)[-n_peaks:]
    
    best_a = a.copy()
    best_obj = max_val
    
    for k in peak_idxs:
        k = int(k)
        # Elements contributing: a[i] * a[k-i] for valid i
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        
        if i_max < i_min:
            continue
        
        # Contributions
        contribs = np.array([a[i] * a[k - i] for i in range(i_min, i_max + 1)])
        
        if contribs.sum() < 1e-15:
            continue
        
        # For the top contributing pair, try moving mass
        for trial in range(n_trials):
            # Pick the pair with max contribution (or a random one)
            if trial == 0:
                idx = i_min + np.argmax(contribs)
            else:
                idx = i_min + rng.integers(0, i_max - i_min + 1)
            
            partner = k - idx
            
            if idx == partner:
                # a[idx]^2 term; can only reduce a[idx]
                for frac in [0.01, 0.02, 0.05]:
                    amt = a[idx] * frac
                    if amt < 1e-10:
                        break
                    # Move to a random position far from both idx and partner
                    j = rng.integers(0, n)
                    a_try = a.copy()
                    a_try[idx] -= amt
                    a_try[j] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
            else:
                # Can reduce either a[idx] or a[partner]
                for elem in [idx, partner]:
                    for frac in [0.01, 0.02, 0.05, 0.1]:
                        amt = a[elem] * frac
                        if amt < 1e-10:
                            break
                        j = rng.integers(0, n)
                        if j == elem:
                            continue
                        a_try = a.copy()
                        a_try[elem] -= amt
                        a_try[j] += amt
                        conv_try = fast_convolve(a_try)
                        if conv_try.max() < best_obj:
                            best_obj = conv_try.max()
                            best_a = a_try.copy()
    
    return best_a, best_obj

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using Safe Mass Redistribution (SMR):
a greedy algorithm that identifies the safest positions to add mass (maximizing
the allowable increase before any convolution peak exceeds the current max),
and transfers mass from peak-contributing positions to those safest targets.
Combined with smoothed gradient descent for global moves and multi-phase diverse search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_delta_max(a, conv, max_conv):
    """
    For each position j, compute the maximum safe delta we can add to a[j]
    without any conv[k] exceeding max_conv.
    
    delta_max[j] = min over m of (max_conv - conv[j+m]) / (2*a[m])
    where m ranges over valid indices with a[m] > 0.
    """
    n = len(a)
    delta_max = np.full(n, 1.0)  # cap at 1.0
    
    active_m = np.where(a > 1e-14)[0]
    for m in active_m:
        j_max = min(n - 1, 2 * n - 2 - m)
        if j_max < 0:
            continue
        constraint = (max_conv - conv[m:m + j_max + 1]) / (2.0 * a[m])
        # delta_max[j] = min(delta_max[j], constraint[j])
        np.minimum(delta_max[:j_max + 1], constraint, out=delta_max[:j_max + 1])
    
    delta_max = np.maximum(delta_max, 0.0)
    return delta_max

    Iteratively transfers mass from peak-contributing positions to the safest targets.

        if rng.random() < random_prob:
            source = int(rng.integers(0, n))
            if a[source] < 1e-10:
                source = int(np.argmax(a))
        else:
            # Pick element contributing most to the current peak
            peak_k = int(np.argmax(conv))
            i_min = max(0, peak_k - n + 1)
            i_max = min(n - 1, peak_k)
            if i_max >= i_min:
                contribs = np.array([a[i] * a[peak_k - i] for i in range(i_min, i_max + 1)])
                if contribs.sum() > 1e-15:
                    source = i_min + int(np.argmax(contribs))
                else:
                    source = int(np.argmax(a))
            else:
                source = int(np.argmax(a))
        
        if a[source] < 1e-12:
            continue
        
        # Select target from candidates
        # Score: prefer large dmax and small current value
        cand_dmax = dmax[candidates]
        cand_val = a[candidates]
        # Normalize for comparison
        score = cand_dmax / (cand_val + 1e-10)
        # Also penalize positions very close to source (less effective for peak reduction)
        dist_penalty = 1.0 / (1.0 + 0.5 * np.abs(candidates.astype(float) - source) / n)
        score = score * dist_penalty
        
        target = candidates[int(np.argmax(score))]
        
        # Determine transfer amount
        amount = min(a[source] * 0.8, dmax[target] * 0.95)
        if amount < 1e-12:
            amount = a[source] * 0.1
            if amount < 1e-12:
                continue
        
        # Perform transfer
        a[source] -= amount
        a[target] += amount
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-15:
            break
        a = a / a_sum
        
        # Recompute convolution
        conv = fast_convolve(a)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # If stuck for too long, do a perturbation
        if no_improve > stagnation_limit:
            a = project_simplex(a + rng.standard_normal(n) * 0.01)
            conv = fast_convolve(a)
            if conv.max() < best_obj:
                best_obj = conv.max()
                best_a = a.copy()
            no_improve = 0
            stagnation_limit = int(stagnation_limit * 1.4)
    
    return best_a, best_obj


def smoothed_gradient_descent(a0, time_limit, seed=0, beta=50.0, lr=0.01, momentum=0.9):
    """
    Nesterov accelerated projected gradient descent on log-sum-exp smoothed objective.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        cur_beta = beta * (1.0 + 2.0 * frac)  # increase beta over time
        cur_lr = lr * (1.0 - 0.6 * frac)
        
        # Nesterov lookahead
        y = a + momentum * v
        conv_y = fast_convolve(y)
        
        # Smoothed gradient
        scaled = cur_beta * conv_y
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv_y))
            p[np.argmax(conv_y)] = 1.0
        else:
            p /= p_sum
        
        w = conv_y * p
        fft_size = 1 << (3 * n - 1).bit_length()
        A = np.fft.rfft(y, fft_size)
        W = np.fft.rfft(w, fft_size)
        cross = np.fft.irfft(A * np.conj(W), fft_size)
        grad = 2.0 * cross[:n]
        
        # Update
        v = momentum * v - cur_lr * grad
        a = project_simplex(a + v)
        
        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 300:
            noise_scale = 0.003 * (1.0 + 0.5 * np.sin(total_iters * 0.01))
            a = project_simplex(a + rng.standard_normal(n) * noise_scale)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
    
    return best_a, best_obj


def lp_direction_find(a, M, n, time_limit=30):
    """
    Solve LP: max sum(b_j) s.t. b_j >= 0, (a*b)[k] <= M for all k.
    Returns b normalized to sum=1, or None.
    """
    c_obj = -np.ones(n)
    
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    
    if not result.success or result.x is None:
        return None
    
    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    
    return b / b_sum


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()
    
    tri = 1.0 / (1.0 + 0.5 * ((x - n / 2) / (n / 4)) ** 2)
    tri /= tri.sum()
    inits['bell'] = tri
    
    for sf in [0.2, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{int(sf * 100)}'] = g
    
    bi = np.zeros(n)
    edge = max(2, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / max(n - 1, 1)))
    cos_w = np.maximum(cos_w, 1e-6)
    cos_w /= cos_w.sum()
    inits['raised_cos'] = cos_w
    
    g_off = np.exp(-0.5 * ((x - 0.3 * n) / (0.2 * n)) ** 2)
    g_off /= g_off.sum()
    inits['gauss_off'] = g_off
    
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 2.0)
    inits['rand3'] = rng.dirichlet(np.ones(n) * 0.2)
    
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining Safe Mass Redistribution (SMR),
    smoothed gradient descent, and LP direction finding.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = evaluate_sequence(p.tolist())
            print(f"Parent: n={len(p)}, sum={p.sum():.2f}, score={parent_score:.6f}", flush=True)
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
    except Exception as e:
        print(f"No parent: {e}", flush=True)
    
    # Phase 1: Quick scan with SMR
    print("=== Phase 1: Quick SMR Scan ===", flush=True)
    lengths = [256, 320, 384, 512, 640, 768]
    
    scan_results = []
    t_phase1 = time.monotonic()
    phase1_budget = 300.0
    
    for n in lengths:
        if time.monotonic() > deadline - 400:
            break
        
        inits = generate_initializations(n, parent, rng)
        names = list(inits.keys())
        
        # Priority
        priority = ['parent', 'parent_ds', 'parent_up', 'bell', 'gauss_35',
                    'raised_cos', 'uniform', 'rand1', 'gauss_20', 'bimodal']
        names = [nm for nm in priority if nm in names] + \
                [nm for nm in names if nm not in priority]
        names = names[:5]
        
        time_rem = phase1_budget - (time.monotonic() - t_phase1)
        per_init = max(8.0, time_rem / max(1, len(names) * len(lengths)))
        
        for name in names:
            if time.monotonic() > deadline - 350:
                break
            t_alloc = min(per_init, 40.0)
            if t_alloc < 5:
                break
            
            a0 = inits[name]
            a_opt, obj_opt = safe_redistribute(a0, t_alloc, seed=int(rng.integers(0, 100000)))
            
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}", flush=True)
    
    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}", flush=True)
    
    # Phase 2: Deep optimization
    print("=== Phase 2: Deep Optimization ===", flush=True)
    
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = n
        if key not in seen:
            top_configs.append((n, a_opt))
            seen.add(key)
        if len(top_configs) >= 4:
            break
    
    if best_sequence is not None:
        top_configs.insert(0, (len(best_sequence), best_sequence.copy()))
    
    # Add random restarts at best length
    best_n = len(best_sequence) if best_sequence is not None else 384
    top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 0.3)))
    top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 1.0)))
    top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 0.7)))
    
    remaining = deadline - time.monotonic() - 120.0
    n_configs = len(top_configs)
    per_config = max(20.0, remaining / n_configs)
    
    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 100:
            break
        t_alloc = min(per_config, 300.0)
        if t_alloc < 10:
            break
        
        print(f"  Config {i}: n={n}, alloc={t_alloc:.0f}s", flush=True)
        
        # Sub-phase A: SMR (60% of allocation)
        t_a = t_alloc * 0.55
        a_current = a0.copy()
        a_s, obj_s = safe_redistribute(a_current, t_a, seed=int(rng.integers(0, 10**6)))
        score_s = float(evaluate_sequence(a_s.tolist()))
        if np.isfinite(score_s) and score_s < best_score:
            best_score = score_s
            best_sequence = a_s.copy()
            print(f"    SMR: {score_s:.6f}", flush=True)
        a_current = a_s
        
        # Sub-phase B: Smoothed gradient (25%)
        t_b = t_alloc * 0.25
        if t_b > 5:
            a_g, obj_g = smoothed_gradient_descent(
                a_current, t_b, seed=int(rng.integers(0, 10**6)),
                beta=40.0, lr=0.008, momentum=0.92)
            score_g = float(evaluate_sequence(a_g.tolist()))
            if np.isfinite(score_g) and score_g < best_score:
                best_score = score_g
                best_sequence = a_g.copy()
                print(f"    Grad: {score_g:.6f}", flush=True)
            a_current = a_g
        
        # Sub-phase C: SMR again on the result (20%)
        t_c = t_alloc * 0.20
        if t_c > 5:
            a_sc, obj_sc = safe_redistribute(a_current, t_c, seed=int(rng.integers(0, 10**6)))
            score_sc = float(evaluate_sequence(a_sc.tolist()))
            if np.isfinite(score_sc) and score_sc < best_score:
                best_score = score_sc
                best_sequence = a_sc.copy()
                print(f"    SMR2: {score_sc:.6f}", flush=True)
    
    # Phase 3: Final polish
    print("=== Phase 3: Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0
        n_best = len(best_sequence)
        
        # Extended SMR with different random seeds
        t_smr = remaining * 0.4
        if t_smr > 10:
            a_p, obj_p = safe_redistribute(best_sequence, t_smr, seed=7777)
            score_p = float(evaluate_sequence(a_p.tolist()))
            if np.isfinite(score_p) and score_p < best_score:
                best_score = score_p
                best_sequence = a_p.copy()
                print(f"  SMR_polish: {best_score:.6f}", flush=True)
        
        # LP direction
        remaining = deadline - time.monotonic() - 15.0
        t_lp = remaining * 0.3
        if t_lp > 15 and n_best <= 512:
            a_lp_input = best_sequence.copy()
            s_lp = a_lp_input.sum()
            a_lp_input = a_lp_input / s_lp
            conv_lp = fast_convolve(a_lp_input)
            M_lp = conv_lp.max()
            
            improved = False
            for M_factor in [1.0, 1.001, 1.005, 1.01, 1.02]:
                b = lp_direction_find(a_lp_input, M_lp * M_factor, n_best,
                                      time_limit=min(t_lp * 0.3, 60))
                if b is not None:
                    for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
                        a_try = (1 - t) * a_lp_input + t * b
                        a_try = np.maximum(a_try, 0)
                        a_try = a_try / a_try.sum()
                        conv_try = fast_convolve(a_try)
                        obj_try = conv_try.max()
                        if obj_try < M_lp:
                            a_lp_input = a_try
                            M_lp = obj_try
                            improved = True
                            break
                    if improved:
                        break
            
            if improved:
                final_a = a_lp_input * s_lp  # restore original scale
                score_lp = float(evaluate_sequence(final_a.tolist()))
                if np.isfinite(score_lp) and score_lp < best_score:
                    best_score = score_lp
                    best_sequence = final_a.copy()
                    print(f"  LP: {best_score:.6f}", flush=True)
        
        # Final SMR polish
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 5:
            a_f, obj_f = safe_redistribute(best_sequence, remaining, seed=4242)
            score_f = float(evaluate_sequence(a_f.tolist()))
            if np.isfinite(score_f) and score_f < best_score:
                best_score = score_f
                best_sequence = a_f.copy()
                print(f"  FinalSMR: {best_score:.6f}", flush=True)
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
