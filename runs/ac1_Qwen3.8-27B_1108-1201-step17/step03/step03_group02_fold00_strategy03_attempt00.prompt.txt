def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""

        j_end = min(n - 1, k)

            A[k, j] = a[k - j]
    return A

def subgradient_multi_peak(a, conv, top_k, temperature):
    """Weighted subgradient over top-k convolution peaks."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

    if temperature > 1e-15:
        scaled = peak_vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        wi = w[i] * 2.0
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += wi * a[k - idx]
    return sg

def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    for freq in [1, 2, 3]:
        a = 1.0 + 0.85 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0)
        a /= a.sum()
        inits[f'cos_{freq}'] = a

    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    inits['random'] = rng.dirichlet(np.ones(n) * 0.4)

    for num_peaks in [3, 5]:
        peaks = rng.choice(n, size=num_peaks, replace=False)
        a = np.zeros(n)
        for p in peaks:
            a[max(0, p - 2):p + 3] = 1.0
        a = np.maximum(a, 0)
        a /= a.sum()
        inits[f'peaks_{num_peaks}'] = a

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    return inits
