import numpy as np
import time
from scipy.fft import fft, ifft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def smooth_max_and_weights(x, k=10, beta=50):
    """Compute a weighted sum of top-k peaks."""
    n = len(x)
    # Get top-k indices
    if k >= n:
        w = np.ones(n) / n
        return x @ w, w
    # Smooth top-k using softmax with temperature
    x_max = x.max()
    shifted = x - x_max
    weights = np.exp(shifted * beta)
    weights = weights / weights.sum()
    # Focus on top-k by increasing beta
    obj = x @ weights
    return obj, weights

def compute_grad_conv(a, w):
    """Gradient of sum_k w[k] * conv[k] w.r.t. a."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad

def adam_optimize(a0, beta_schedule, k_schedule, n_iter, lr=0.01, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """Adam optimizer on the simplex."""
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    
    best_obj = float('inf')
    best_a = a.copy()
    
    for it in range(n_iter):
        # Determine current beta and k
        frac = it / max(1, n_iter - 1)
        beta = beta_schedule if isinstance(beta_schedule, float) else beta_schedule[0] + (beta_schedule[1] - beta_schedule[0]) * frac
        k = int(k_schedule) if isinstance(k_schedule, int) else int(k_schedule[0] + (k_schedule[1] - k_schedule[0]) * frac)
        k = min(k, 2*n-1)
        
        conv = fast_convolve(a)
        obj, w = smooth_max_and_weights(conv, k=k, beta=beta)
        obj_val = 2.0 * n * obj
        
        if obj_val < best_obj:
            best_obj = obj_val
            best_a = a.copy()
        
        grad_a = compute_grad_conv(a, w)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_a
        v = beta2 * v + (1 - beta2) * grad_a**2
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = project_simplex(a_new)
        
        # Check improvement
        conv_new = fast_convolve(a_new)
        _, w_new = smooth_max_and_weights(conv_new, k=k, beta=beta)
        obj_new = 2.0 * n * (conv_new @ w_new)
        
        if obj_new < obj_val * 1.001:
            a = a_new
        else:
            # Try smaller step
            a_new = a - lr * 0.1 * m_hat / (np.sqrt(v_hat) + epsilon)
            a_new = project_simplex(a_new)
            a = a_new
        
        # Decay learning rate
        lr *= 0.9999
    
    return best_a, best_obj

L-BFGS-B polishing. Key innovations over previous approach:

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def topk_weighted_max(x, k, beta):
    """Compute smooth approximation to sum of top-k values, and weights."""
    n = len(x)
    k = min(k, n)
    if k == n:
        w = np.ones(n) / n
        return x @ w, w
    # Use softmax with high temperature to focus on top-k
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    if total < 1e-30:
        w = np.zeros(n)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_s / total
    obj = x @ w
    return obj, w


def compute_gradient(a, w):
    """Gradient of sum_k w[k]*conv[k] w.r.t. a, using FFT."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def upsample_sequence(a, new_n):
    """Upsample a sequence to a new length using interpolation and re-normalize."""
    if len(a) == new_n:
        return a.copy()
    if len(a) > new_n:
        # Downsample
        x_old = np.linspace(0, 1, len(a))
        x_new = np.linspace(0, 1, new_n)
        a_new = np.interp(x_new, x_old, a)
    else:
        # Upsample
        x_old = np.linspace(0, 1, len(a))
        x_new = np.linspace(0, 1, new_n)
        a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def flat_spectrum_init(n, rng, n_trials=50):
    """Initialize with approximately flat power spectrum via random phases."""
    fft_size = 1 << (n - 1).bit_length()
    best_a = np.ones(n) / n
    best_obj = float('inf')

    for _ in range(n_trials):
        A = np.zeros(fft_size, dtype=complex)
        A[0] = float(np.sqrt(n))
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2)
        for k_idx in range(1, fft_size // 2):
            A[k_idx] = np.exp(1j * phases[k_idx - 1])
            A[fft_size - k_idx] = np.exp(-1j * phases[k_idx - 1])
        if fft_size % 2 == 0:
            A[fft_size // 2] = np.exp(1j * phases[-1])

        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            conv = fast_convolve(a)
            obj = 2.0 * n * conv.max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a


def random_phase_optimize(n, rng, n_trials=30, iters_per_trial=100, lr=0.1):
    """Optimize phases in frequency domain to minimize peak."""
    fft_size = 1 << (n - 1).bit_length()
    best_a = np.ones(n) / n
    best_obj = float('inf')

    for trial in range(n_trials):
        # Random initial phases
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2)
        
        for _ in range(iters_per_trial):
            # Build spectrum from phases
            A = np.zeros(fft_size, dtype=complex)
            A[0] = float(np.sqrt(n))
            for k_idx in range(1, fft_size // 2):
                A[k_idx] = np.exp(1j * phases[k_idx - 1])
                A[fft_size - k_idx] = np.exp(-1j * phases[k_idx - 1])
            if fft_size % 2 == 0:
                A[fft_size // 2] = np.exp(1j * phases[-1])

            a = np.real(ifft(A, fft_size))[:n]
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-10:
                phases += rng.uniform(-0.5, 0.5, len(phases))
                continue
            a /= s

            conv = fast_convolve(a)
            obj = 2.0 * n * conv.max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

            # Gradient-based phase update (simplified)
            # Perturb phases slightly
            phases += rng.uniform(-lr, lr, len(phases))

    return best_a


def adam_optimize(a0, n_iter=2000, lr_init=0.05, beta1=0.9, beta2=0.999,
                  epsilon=1e-8, k_schedule=(1, 15), beta_schedule=(200.0, 30.0)):
    """Adam optimizer on simplex with adaptive top-k peak objective."""
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()

    m = np.zeros(n)
    v = np.zeros(n)

    best_obj = float('inf')
    best_a = a.copy()

    for it in range(n_iter):
        frac = it / max(1, n_iter - 1)

        # Cosine annealing for learning rate
        lr = lr_init * 0.5 * (1 + np.cos(np.pi * frac))
        lr = max(lr, lr_init * 0.01)

        # Adaptive k and beta
        k = int(k_schedule[0] + (k_schedule[1] - k_schedule[0]) * frac)
        k = min(k, 2 * n - 1)
        beta = beta_schedule[0] + (beta_schedule[1] - beta_schedule[0]) * frac

        conv = fast_convolve(a)
        obj, w = topk_weighted_max(conv, k, beta)
        obj_val = 2.0 * n * obj

        if obj_val < best_obj:
            best_obj = obj_val
            best_a = a.copy()

        # Compute gradient
        grad_a = compute_gradient(a, w)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_a
        v = beta2 * v + (1 - beta2) * (grad_a ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = project_simplex(a_new)

        # Verify improvement (with tolerance)
        conv_new = fast_convolve(a_new)
        _, w_new = topk_weighted_max(conv_new, k, beta)
        obj_new = 2.0 * n * (conv_new @ w_new)

        if obj_new <= obj_val * (1 + 1e-6) or obj_new < best_obj:
            a = a_new
        else:
            # Halve step
            a_new = a - lr * 0.5 * m_hat / (np.sqrt(v_hat) + epsilon)
            a_new = project_simplex(a_new)
            a = a_new

    return best_a, best_obj


def lbfgsb_optimize(a0, beta=50.0, maxiter=300):
    """L-BFGS-B with softmax reparametrization."""
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)

    def obj_func(x):
        x_shift = x - x.max()
        aa = np.exp(x_shift)
        aa = aa / aa.sum()
        conv = fast_convolve(aa)
        sm, _ = topk_weighted_max(conv, min(20, 2 * n - 1), beta)
        return 2.0 * n * sm

    def jac_func(x):
        x_shift = x - x.max()
        aa = np.exp(x_shift)
        aa = aa / aa.sum()
        conv = fast_convolve(aa)
        sm, w = topk_weighted_max(conv, min(20, 2 * n - 1), beta)
        grad_a = compute_gradient(aa, w)
        lagrange = np.dot(aa, grad_a)
        return aa * (grad_a - lagrange)

    result = minimize(obj_func, x0, jac=jac_func, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxls': 50})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, result.fun


def peak_balance_optimize(a0, n_steps=500, k=5, lr=0.01):
    """Peak balancing: iteratively equalize the top-k peaks."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()

    best_obj = float('inf')
    best_a = a.copy()

    for step in range(n_steps):
        conv = fast_convolve(a)
        obj_val = 2.0 * n * conv.max()

        if obj_val < best_obj:
            best_obj = obj_val
            best_a = a.copy()

        # Get top-k peaks
        k_cur = min(k, 2 * n - 1)
        top_indices = np.argsort(conv)[-k_cur:]
        top_vals = conv[top_indices]
        mu = top_vals.mean()

        # Build weight vector: positive weight for peaks below mean (want to increase)
        # negative weight for peaks above mean (want to decrease)
        w = np.zeros(2 * n - 1)
        for j in top_indices:
            w[j] = -(conv[j] - mu)  # negative for peaks above mean

        # Compute gradient
        grad = compute_gradient(a, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Step
        lr_cur = lr * (1.0 - step / n_steps) + 0.001
        a_new = a - lr_cur * grad / grad_norm
        a_new = project_simplex(a_new)

        # Check
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        if obj_new <= obj_val * 1.001:
            a = a_new
        else:
            a_new = a - lr_cur * 0.5 * grad / grad_norm
            a_new = project_simplex(a_new)
            a = a_new

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Edge-concentrated
    edge_w = np.exp(-((x - n / 2) / (n * 0.3)) ** 2)
    edge_w = edge_w / edge_w.sum()
    inits['edge'] = edge_w

    # Bimodal
    bi = np.exp(-((x - n * 0.2) / (n * 0.1)) ** 2) + np.exp(-((x - n * 0.8) / (n * 0.1)) ** 2)
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Gaussian center
    g = np.exp(-((x - n / 2) / (n * 0.25)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Flat spectrum
    inits['flat_spec'] = flat_spectrum_init(n, rng, n_trials=20)

    # Parent-based
    if parent is not None and len(parent) > 0:
        pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                       np.maximum(parent, 0))
        pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    return inits

            a_opt, _ = lbfgsb_optimize(best_sequence, beta=30.0,
                                       maxiter=int(min(300, remaining * 10)))
            score_and_update(a_opt)

        # Random perturbations + re-optimize
        remaining = deadline - time.monotonic() - 15
        for _ in range(8):
            remaining = deadline - time.monotonic() - 15
            if remaining < 10:
                break
            scale = 0.005 + 0.005 * rng.uniform()
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_sequence + noise)
            n_it = int(min(1000, remaining * 3))
            a_opt, _ = adam_optimize(a_pert, n_iter=n_it, lr_init=0.02,
                                     k_schedule=(2, 8), beta_schedule=(100.0, 30.0))
            score = score_and_update(a_opt)
            if score < best_score:
                best_sequence = a_opt

    # === Phase 4: Try different lengths one more time ===
    print("=== Phase 4: Length exploration ===")

    remaining = deadline - time.monotonic() - 10
    if remaining > 40 and best_sequence is not None:
        # Try n=192 and n=320
        for alt_n in [192, 320, 384]:
            remaining = deadline - time.monotonic() - 10
            if remaining < 20:
                break
            a0 = upsample_sequence(best_sequence, alt_n)
            n_it = int(min(2000, remaining * 5))
            a_opt, _ = adam_optimize(a0, n_iter=n_it, lr_init=0.03,
                                     k_schedule=(2, 10), beta_schedule=(150.0, 30.0))
            score_and_update(a_opt)

    # Final verification
    if best_sequence is None:
        # Fallback
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL: score=%.6f, n=%d" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
