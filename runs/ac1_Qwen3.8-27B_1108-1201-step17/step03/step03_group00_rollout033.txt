Multi-phase approach: explore diverse initializations, then exploit the best.

import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def score_from_conv(conv, n):
    """Compute score from auto-convolution (assumes a >= 0)."""
    return 2.0 * n * conv.max()

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0. Returns g or None."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except:
        pass
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using incremental coordinate descent
with exact line search, simulated annealing for escape, and LP-based Frank-Wolfe
directions. Multi-phase: explore diverse initializations, then exploit the best.
Key innovation: O(n) incremental convolution updates enable rapid coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def coord_descent_pass(f, conv, order, n_deltas=12):
    """One pass of coordinate descent with incremental convolution updates.
    Returns updated f and conv."""
    n = len(f)
    for i in order:
        a_i = f[i]
        if a_i < 1e-15:
            continue

        # Candidate delta values (fraction of a_i)
        fracs = np.array([-0.9, -0.7, -0.5, -0.3, -0.15, -0.05,
                          0.05, 0.15, 0.3, 0.5, 0.7, 0.9, 1.5, 2.0])
        deltas = fracs * a_i

        best_delta = 0.0
        best_val = conv.max()

        # Precompute: which k positions are affected
        # k in [i, i+n-1] where f[k-i] > 0
        # f[k-i] for k-i in [0, n-1] is just f[0], f[1], ..., f[n-1]
        
        for delta in deltas:
            new_ai = a_i + delta
            if new_ai < -1e-15:
                continue
            new_ai = max(0.0, new_ai)
            delta = new_ai - a_i
            if abs(delta) < 1e-15:
                continue

            one_plus_d = 1.0 + delta
            if one_plus_d < 1e-10:
                continue

            scale = 1.0 / (one_plus_d * one_plus_d)

            # Compute new max of convolution
            # For k not affected (f[k-i]=0): new_c[k] = c[k] * scale
            # For k affected: new_c[k] = (c[k] + 2*delta*f[k-i] + delta^2*[k==2i]) * scale

            # The max over unaffected positions:
            # We need max of c[k] for k where k-i not in [0, n-1], i.e., k < i or k > i+n-1
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            if k_lo > 0:
                max_unaff_left = conv[:k_lo].max()
            else:
                max_unaff_left = 0.0
            if k_hi < 2 * n - 2:
                max_unaff_right = conv[k_hi + 1:].max()
            else:
                max_unaff_right = 0.0
            max_unaff = max(max_unaff_left, max_unaff_right)

            # For affected positions k in [k_lo, k_hi]:
            # new_c[k] = (c[k] + 2*delta*f[k-k_lo] + delta^2*[k==2i]) * scale
            # f[k-k_lo] = f[k-i]
            affected_idx = np.arange(k_lo, k_hi + 1)
            affected_f = f[affected_idx - i]
            new_affected = conv[affected_idx] + 2.0 * delta * affected_f
            if 2 * i >= k_lo and 2 * i <= k_hi:
                new_affected[2 * i - k_lo] += delta * delta
            max_aff = (new_affected * scale).max()

            max_unaff_val = max_unaff * scale
            total_max = max(max_unaff_val, max_aff)

            if total_max < best_val:
                best_val = total_max
                best_delta = delta
                best_new_ai = new_ai

        if best_delta != 0.0:
            # Apply the update
            new_f = f / (1.0 + best_delta)
            new_f[i] = (f[i] + best_delta) / (1.0 + best_delta)
            new_conv = fast_convolve(new_f)
            f = new_f
            conv = new_conv

    return f, conv


def optimize_coord_descent(f0, time_limit, rng_seed=42):
    """Run coordinate descent for a given time limit."""
    n = len(f0)
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    no_improve = 0
    pass_num = 0

    while time.monotonic() - t0 < time_limit:
        pass_num += 1
        order = rng.permutation(n)
        f, conv = coord_descent_pass(f, conv, order)

        current_obj = conv.max()
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic annealing restart
        if no_improve > 50:
            noise_scale = 0.01 + 0.02 * min(no_improve // 50, 5)
            f_noisy = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            f = f_noisy
            conv = fast_convolve(f)
            no_improve = 0

    return best_f, best_obj


def optimize_with_anneal(f0, time_limit, rng_seed=99):
    """Coordinate descent with periodic simulated annealing."""
    n = len(f0)
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    temperature = 0.05
    cool_rate = 0.999

    while time.monotonic() - t0 < time_limit:
        # Do a few passes of coordinate descent
        sub_time = min(2.0, time_limit - (time.monotonic() - t0))
        if sub_time < 0.5:
            break
        f, conv = coord_descent_pass(f, conv, rng.permutation(n))

        current_obj = conv.max()
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_f = f.copy()

        # Annealing step: propose a perturbation
        if rng.random() < 0.3:
            noise_scale = temperature * (1.0 + 0.5 * np.sin(time.monotonic()))
            f_prop = project_simplex(f + rng.standard_normal(n) * noise_scale)
            conv_prop = fast_convolve(f_prop)
            obj_prop = conv_prop.max()

            # Metropolis criterion
            if obj_prop < best_obj or rng.random() < np.exp(-(obj_prop - best_obj) / max(temperature, 1e-8)):
                f = f_prop
                conv = conv_prop

        temperature *= cool_rate
        if temperature < 0.001:
            temperature = 0.001

    return best_f, best_obj


def optimize_fw_lp(f0, time_limit, rng_seed=7):
    """Frank-Wolfe with LP oracle (from previous approach, adapted)."""
    n = len(f0)
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    no_improve = 0
    m_exp = 1.0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp)
        if g is None:
            m_exp = min(m_exp * 1.2, 5.0)
            if m_exp >= 5.0:
                # Fall back to coordinate descent
                f, conv = coord_descent_pass(f, fast_convolve(f), rng.permutation(n))
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            else:
                m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)

        # Switch to coordinate descent periodically
        if it % 50 == 0:
            f, conv = coord_descent_pass(f, fast_convolve(f), rng.permutation(n))
            M = conv.max()

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    # Exponential decay
    for alpha in [1.0, 2.0, 3.0]:
        e = np.exp(-alpha * x / n)
        inits.append(e / e.sum())

    # Cosine
    c = np.cos(np.pi * x / (2 * n)) ** 2
    inits.append(c / c.sum())

    # Linear ramp
    r = np.linspace(0.1, 1.0, n)
    inits.append(r / r.sum())

    # Bimodal
    b = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits.append(b / b.sum())

    # Raised cosine with frequency
    for freq in [1, 2, 3]:
        rc = 0.5 * (1 + np.cos(2 * np.pi * freq * x / n))
        rc = rc + 0.1
        inits.append(rc / rc.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(
