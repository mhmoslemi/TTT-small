def smooth_fd(a_init, time_limit, alpha0=5, alpha_growth=1.005, alpha_max=50, lr=0.005):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    alpha = alpha0
    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum coefficient
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        
        # Softmax weights
        scaled = alpha * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Gradient
        grad = 2.0 * np.correlate(w, a, mode='valid')
        
        # Momentum update
        v = mu * v - lr * grad
        a = project_simplex(a + v)
        
        # Anneal alpha
        alpha = min(alpha * alpha_growth, alpha_max)
        
        # Track best
        obj = fast_convolve(a).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 200:
            a = project_simplex(best_a + np.random.randn(n) * 0.002)
            v = np.zeros(n)
            no_improve = 0
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth gradient descent
with log-sum-exp approximation, annealing, simplex projection, and
LP-based Frank-Wolfe polishing. Key innovation: ramp-shaped initialization
which exploits the asymmetry of linear convolution to achieve ~4/3 objective,
far below the 2.0 lower bound for symmetric sequences.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, alpha):
    """Compute gradient of smooth objective (1/alpha)*log(sum exp(alpha*conv(a,a)))."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = alpha * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    # grad_i = 2 * sum_j a_j * w_{i+j} for j=0..n-1, i=0..n-1
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad, conv

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f*g)_k <= M for k=0..2n-2
    # (f*g)_k = sum_j f_{k-j} * g_j (with valid indices)
    # For fixed k, the constraint is: sum_{j=max(0,k-n+1)}^{min(k,n-1)} f_{k-j} * g_j <= M
    # We can use scipy's linprog with matrix form.
    # A_ub[g] <= b_ub, where A_ub[k, j] = f[k-j] if valid, else 0.
    
    # To avoid building a huge sparse matrix explicitly, use the structure.
    # Actually for n=384, the matrix is 767 x 384, which is manageable.
    A_ub = np.zeros((2*n-1, n))
    for j in range(n):
        # g_j appears in conv_k for k = j, j+1, ..., j+n-1
        # conv_k includes f[k-j] * g_j
        # So A_ub[k, j] = f[k-j] for k in [j, j+n-1]
        A_ub[j:j+n, j] = f[:n]  # f[0], f[1], ..., f[n-1]
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None
