```python
        return conv, 2.0 * n * conv.max() / (a.sum()**2)
    
    conv_new = conv.copy()
    # Update b_k for all k
    # For k where 0 <= k-i <= n-1 (i.e., k in [i, i+n-1]):
    lo = max(0, i)
    hi = min(2*n-2, i + n - 1)
    # b_k changes by 2*delta*a[k-i] for k != 2i, and by delta*(ai_new+ai_old) for k = 2i

        if k == 2*i:

        else:
            j = k - i
            if 0 <= j < n:
                conv_new[k] += 2.0 * delta * a[j]
    
    s_new = a.sum() - ai_old + ai_new
    obj = 2.0 * n * conv_new.max() / (s_new * s_new)
    return conv_new, obj
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov accelerated subgradient
with soft-max peak weighting, simplex projection, and coordinate descent polishing.
Key innovations: (1) Nesterov momentum for faster subgradient convergence,
(2) GPU-like FFT-accelerated subgradient computation, (3) exact 1D coordinate
descent with O(n) per-evaluation cost via incremental convolution updates,
(4) multi-scale initialization with structured random restarts.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_objective(a):
    """Returns (objective_value, convolution_array)."""
    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf'), conv
    return 2.0 * n * conv.max() / (s * s), conv

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cv[rho[-1]] - 1.0) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient_fft(a, conv, temp=0.001):
    """Compute soft-max subgradient of max(conv) w.r.t. a using FFT."""
    n = len(a)
    # Soft-max weights
    scaled = conv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    # g[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is cross-correlation of w and a at lags 0..n-1
    # Use FFT: ifft(conj(fft(w, N)) * fft(a, N))
    N = 1 << (len(w) + n - 1).bit_length()
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    corr = np.fft.ifft(np.conj(W) * A)[:n]
    return 2.0 * np.real(corr)

def accelerated_subgradient(a0, time_limit, temp_init=0.002, temp_min=0.00005,
                            lr_init=0.03, beta_init=0.85, beta_end=0.98):
    """Nesterov accelerated subgradient with simplex projection."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    
    obj, conv = compute_objective(a)
    best_obj = obj
    best_a = a.copy()
    
    t0 = time.time()
    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    beta = beta_init
    total_iters = 0
    no_improve = 0
    max_no_improve = 800
    
    while time.time() - t0 < time_limit:
        sg = compute_subgradient_fft(a, conv, temp)
        
        # Nesterov momentum
        beta = min(beta_end, beta + 0.0001)
        v = beta * v - lr * sg
        a_new = a + v
        a_new = project_simplex(a_new)
        
        obj_new, conv_new = compute_objective(a_new)
        total_iters += 1
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        conv = conv_new
        
        # Adaptive LR
        if no_improve > 200:
            lr *= 0.85
            no_improve = 0
        elif no_improve > 50:
            lr *= 0.95
        
        # Temperature annealing
        progress = min(1.0, total_iters / 5000.0)
        temp = temp_init * (temp_min / temp_init) ** progress
        temp = max(temp, temp_min)
        
        if no_improve > max_no_improve:
            break
        if lr < 1e-8:
            break
    
    return best_a, best_obj

    """Polish using coordinate descent with incremental convolution updates."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    
    obj, conv = compute_objective(a)
    best_obj = obj
    best_a = a.copy()
    
    t0 = time.time()
    
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        sweep_improved = False
        
        for i in range(n):
            if time.time() - t0 > time_limit:
                break
            
            ai = a[i]
            if ai < 1e-15:
                continue

            hi = min(2.5 * ai + 0.001, S * 0.5)

            best_val = compute_objective(a)[0]  # Current value
            best_ai = ai

                delta = x - ai
                if abs(delta) < 1e-15:
                    val = best_val
                else:

                    # b_k changes by 2*delta*a[k-i] for k != 2i
                    # b_{2i} changes by x^2 - ai^2
                    new_conv = conv.copy()
                    k_range = range(max(0, i), min(2*n-2, i+n-1) + 1)
                    for k in k_range:

                        else:

                    S_new = S + delta
                    return 2.0 * n * new_conv.max() / (S_new * S_new)
                
                # Golden section
                a_g = glo + (1-gr)*(ghi-glo)

                        a_g = b_g

                    best_obj = obj
                    best_a = a.copy()
                sweep_improved = True
        
        if not sweep_improved:
            break
    
    return best_a, best_obj
```

```python
def coord_eval_vec(a, i, x, ai, conv, n, S):
    """Vectorized evaluation of objective when a[i] changes from ai to x."""
    delta = x - ai
    if abs(delta) < 1e-15:
        return 2.0 * n * conv.max() / (S * S)
    
    k_lo = max(0, i)
    k_hi = min(2*n-2, i+n-1)
    k_vals = np.arange(k_lo, k_hi+1)
    j_vals = k_vals - i  # all in [0, n-1]
    
    delta_conv = 2.0 * delta * a[j_vals]
    mask_2i = (k_vals == 2*i)
    if np.any(mask_2i):
        delta_conv[mask_2i] = x*x - ai*ai
    
    # New max
    old_max = conv.max()
    new_max = old_max
    # Check if the max is in the changed region
    changed_max = np.max(conv[k_vals] + delta_conv)
    if k_lo > 0:
        unchanged_max_left = conv[:k_lo].max() if k_lo > 0 else 0
    else:
        unchanged_max_left = 0
    if k_hi < 2*n-2:
        unchanged_max_right = conv[k_hi+1:].max() if k_hi+1 < 2*n-1 else 0
    else:
        unchanged_max_right = 0
    new_max = max(changed_max, unchanged_max_left, unchanged_max_right)
    
    S_new = S + delta
    return 2.0 * n * new_max / (S_new * S_new)
```
