    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, t):
    """
    Compute gradient of soft_max(conv(a,a), t) w.r.t. a.
    Returns (gradient, conv, soft_max_value).
    """
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / t
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    t_sq_sum = exp_scaled.sum()
    w = exp_scaled / t_sq_sum
    sm_val = t * (np.log(t_sq_sum) + scaled.max())  # = t * log(sum(exp(conv/t)))
    
    # Gradient: g[k] = 2 * sum_m a[m] * w[m+k]
    # Compute via direct summation (O(n^2), fine for n=384)
    g = np.zeros(n)
    for k in range(n):
        g[k] = 2.0 * np.dot(a, w[k:k + n])
    
    return g, conv, sm_val, w

def nesterov_smooth_descent(a0, time_limit, t_start=1.0, t_end=0.005, 
                              lr_start=0.002, verbose=False):
    """
    Nesterov accelerated gradient descent on soft_max(conv(a,a), t)
    with projection onto simplex and adaptive temperature cooling.
    """
    a = a0.copy()
    n = len(a)
    
    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum parameter
    
    t = t_start
    lr = lr_start
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        y = a + mu * v
        
        # Compute gradient at y
        g, conv, sm_val, w = smooth_gradient(y, t)
        
        # Update
        v = mu * v + g  # Hmm, standard Nesterov is:
        # y = a + mu*v
        # a_new = proj(a - lr * grad(y))
        # v_new = a_new - a
        # Let me use the standard form:
        
        # Actually, the standard Nesterov (accelerated gradient) is:
        # y_k = x_k + (beta_k - 1)/beta_{k+1} * (x_k - x_{k-1})
        # x_{k+1} = proj(x_k - lr_k * grad(y_k))
        # Or equivalently with momentum:
        # v_{k+1} = beta * v_k + grad(x_k)
        # x_{k+1} = proj(x_k - lr * v_{k+1})
        
        # Let me use the simpler "heavy ball" momentum:
        v = mu * v + g
        a_new = project_simplex(a - lr * v)
        
        # Check improvement
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"      iter {it}: obj={best_obj:.10f}, t={t:.6f}, lr={lr:.6f}")
        else:
            no_improve += 1
        
        a = a_new
        
        # Adapt learning rate
        lr *= 0.9997
        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
        
        # Cool temperature
        t *= 0.9992
        if t < t_end:
            t = t_end
        
        # Periodic restart
        if it % 5000 == 0:
            v = np.zeros(n)
            mu = min(mu + 0.01, 0.95)
    
    return best_a, best_obj


def subgradient_descent(a0, time_limit, top_k=30, weight_temp_start=0.05,
                         lr=0.001, mom=0.93, verbose=False):
    """
    Subgradient descent with multi-peak weighted subgradients.
    """
    a = a0.copy()
    n = len(a)
    v = np.zeros(n)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    wt = weight_temp_start
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        sg, conv = subgradient(a, top_k=top_k, weight_temp=wt)
        
        v = mom * v - lr * sg
        a_new = project_simplex(a + v)
        
        # Re-normalize (should already be on simplex from projection)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 1000 == 0:
                print(f"      SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        a = a_new
        lr *= 0.9998
        wt *= 1.002
        
        if no_improve > 500:
            # Random restart
            rng = np.random
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            lr = lr * 2
            no_improve = 0
    
    return best_a, best_obj

def fast_convolve(a):
    """Auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
