import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def coord_min_1d(v, a, i, n, conv):
    """
    Minimize max_k conv_new[k] over x = a_i (new value), keeping all else fixed.
    Returns the optimal x >= 0.
    """
    # Gather the coefficients
    # conv_new[k] for k != 2i: linear in x with slope a_{k-i}
    # conv_new[2i]: quadratic x^2 + (conv[2i] - v^2)
    
    c_2i = None
    if 0 <= 2*i < 2*n - 1:
        c_2i = conv[2*i] - v*v  # constant part of the quadratic
    
    # Linear terms: for each valid k (k != 2i, 0 <= k-i < n, 0 <= k < 2n-1)
    # conv_new[k] = conv[k] - v*a_{k-i} + x*a_{k-i}
    # = (conv[k] - v*a_{k-i}) + x * a_{k-i}
    
    # Gather (slope, intercept) pairs
    slopes = []
    intercepts = []
    k_start = max(0, i)  # k-i >= 0 => k >= i
    k_end = min(2*n-2, i + n - 1)  # k-i < n => k < i+n, and k < 2n-1
    
    for k in range(k_start, k_end + 1):
        j = k - i  # = k-i, the other index
        if j == i:
            continue  # this is the k=2i case
        if 0 <= j < n:
            slopes.append(a[j])
            intercepts.append(conv[k] - v * a[j])
    
    slopes = np.array(slopes)
    intercepts = np.array(intercepts)
    
    # Also include the "constant" peaks (k where k-i is not in [0,n-1] and k != 2i)
    # These don't depend on x, so they give a constant lower bound
    const_peaks = []

        if not (0 <= j < n) or j == i:
            const_peaks.append(conv[k])
    const_max = max(const_peaks) if const_peaks else 0
    
    # Now minimize: max(max_l(slope_l * x + intercept_l), x^2 + c_2i if c_2i is not None, const_max)
    
    # The function is convex in x (max of affine and quadratic functions)
    # Find the minimum via bisection on the subgradient

    # The minimum is where the subgradient contains 0
    
    # Practical approach: ternary search or golden section on [0, upper_bound]
    
    # Upper bound: the current value v is feasible, so x=0 or x=v are bounds
    # Actually x can be anything >= 0. Let's find a reasonable upper bound.

        if c_2i is not None:

        m = max(m, const_max)
        return m

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using:
1. Cyclic/Randomized coordinate descent with exact 1D minimization (convex subproblem per coord)
2. Smoothed log-sum-exp gradient descent with Nesterov momentum
3. Frequency-domain PHO iterations for peak reduction
4. Multi-phase diverse search with structured initializations and annealing restarts

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coord_1d_min(v, a, i, n, conv, c2i_offset):
    """
    Solve the 1D subproblem: minimize max(conv_new) over x = a_i >= 0.
    Uses golden section search on the convex function.
    """
    # Gather affected k values
    k_lo = i
    k_hi = min(i + n - 1, 2 * n - 2)

    slopes = []
    intercepts = []
    quad_const = None

    for k in range(k_lo, k_hi + 1):
        j = k - i
        if j == i:
            # Quadratic case: k = 2i
            quad_const = conv[k] - v * v
        else:
            slopes.append(a[j])
            intercepts.append(conv[k] - v * a[j])

    slopes = np.array(slopes) if slopes else np.array([0.0])
    intercepts = np.array(intercepts) if intercepts else np.array([0.0])

    # Constant peaks (unaffected k)
    affected_set = set(range(k_lo, k_hi + 1))
    const_max = 0.0
    for k in range(2 * n - 1):
        if k not in affected_set:
            if conv[k] > const_max:
                const_max = conv[k]

    # Evaluate f(x)
    def f(x):
        # Linear part
        linear_vals = slopes * x + intercepts
        m = linear_vals.max()
        if quad_const is not None:
            m = max(m, x * x + quad_const)
        m = max(m, const_max)
        return m

    # Current value
    f_v = f(v)

    # Upper bound
    ub = max(v + 1.0, 1.0)
    while f(ub) < f_v and ub < 1e6:
        ub *= 2.0
    ub = min(ub, 1e4)

    # Golden section search on [0, ub]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = ub
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)

    for _ in range(40):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)

    x_opt = (lo + hi) / 2.0
    return max(0.0, x_opt)


def coordinate_descent(a0, time_limit, seed=0, max_sweeps=100000):
    """
    Cyclic coordinate descent with exact 1D minimization.
    Randomized coordinate order with occasional full sweeps.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    if n == 0:
        return a0.copy(), float('inf')

    conv = fast_convolve(a)
    best_a = a.copy()
    best_f = conv.max()
    t0 = time.monotonic()
    sweep_count = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweep_count += 1
        f_before = conv.max()

        # Randomized order
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            v = a[i]
            if v < 1e-15:
                # Try turning it on slightly
                x_opt = coord_1d_min(0.0, a, i, n, conv, 0)
                if x_opt > 1e-10:
                    a[i] = x_opt
                    # Update conv incrementally is complex; recompute
                    conv = fast_convolve(a)
                continue

            x_opt = coord_1d_min(v, a, i, n, conv, 0)
            if x_opt != v:
                a[i] = x_opt
                conv = fast_convolve(a)

        f_after = conv.max()
        if f_after < best_f - 1e-12:
            best_f = f_after
            best_a = a.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1

        # Annealing restart
        if no_improve_sweeps >= 50:
            noise_scale = 0.005 + 0.01 * min(no_improve_sweeps / 100.0, 1.0)
            noise = rng.standard_normal(n) * noise_scale
            a = np.maximum(a + noise, 0.0)
            conv = fast_convolve(a)
            no_improve_sweeps = 0

    return best_a, best_f


def coordinate_descent_incremental(a0, time_limit, seed=0):
    """
    Coordinate descent with incremental convolution update for speed.
    When a[i] changes from v to x, update conv in O(n) instead of O(n log n).
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    conv = fast_convolve(a)
    best_a = a.copy()
    best_f = conv.max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            v = a[i]
            x_opt = coord_1d_min(v, a, i, n, conv, 0)

            if abs(x_opt - v) > 1e-12:
                # Update conv incrementally
                delta = x_opt - v
                k_lo = i
                k_hi = min(i + n - 1, 2 * n - 2)

                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j == i:
                        # Quadratic: old v^2, new x^2
                        conv[k] = conv[k] - v * v + x_opt * x_opt
                    else:
                        conv[k] = conv[k] + delta * a[j]

                a[i] = x_opt
                sweep_improved = True

        f_cur = conv.max()
        if f_cur < best_f - 1e-12:
            best_f = f_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 80:
            noise_scale = 0.003 + 0.005 * min(no_improve / 150.0, 1.0)
            idx = rng.integers(0, n, size=max(3, n // 20))
            a[idx] = np.maximum(a[idx] + rng.standard_normal(len(idx)) * noise_scale, 0.0)
            conv = fast_convolve(a)
            no_improve = 0

    return best_a, best_f


def smoothed_gradient_optimize(a0, time_limit, beta_init=30.0, beta_end=300.0,
                                lr_init=0.015, momentum=0.92, seed=0):
    """Log-sum-exp smoothed gradient descent with Nesterov momentum."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    v = np.zeros(n)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    perturb_every = 400

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac
        lr = lr_init * (1.0 - 0.7 * frac)
        lr = max(lr, 0.0003)

        y = a + momentum * v
        conv_y = fast_convolve(y)
        scaled = beta * conv_y
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv_y))
            p[np.argmax(conv_y)] = 1.0
        else:
            p /= p_sum

        w = conv_y * p
        fft_size = 1 << (3 * n - 1).bit_length()
        A = np.fft.rfft(y, fft_size)
        W = np.fft.rfft(w, fft_size)
        cross = np.fft.irfft(A * np.conj(W), fft_size)
        grad = 2.0 * cross[:n]

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        conv_new = fast_convolve(a_new)
        cur_obj = conv_new.max()
        a = a_new

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= perturb_every:
            noise_scale = 0.003
            noise = rng.standard_normal(n) * noise_scale
            a = project_simplex(a + noise)
            v *= 0.3
            no_improve = 0
            perturb_every = int(perturb_every * 1.3)

    return best_a, best_obj


def pho_peak_reduction(a0, time_limit, seed=0):
    """
    PHO-inspired peak reduction: iteratively cap convolution peaks
    and invert through frequency domain.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_f = fast_convolve(a).max()
    t0 = time.monotonic()
    target_ratio = 1.2

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        f_val = conv.max()
        mean_conv = conv.sum() / len(conv)

        if f_val < best_f:
            best_f = f_val
            best_a = a.copy()

        threshold = target_ratio * mean_conv
        if f_val <= threshold * 1.01:
            # Try tighter threshold
            target_ratio = max(1.05, target_ratio * 0.95)
            if target_ratio <= 1.05:
                break

        # Create target
        conv_target = np.minimum(conv, threshold)

        # Invert via frequency domain
        A = np.fft.rfft(a, fft_size)
        C = np.fft.rfft(conv_target, fft_size)
        C_safe = np.maximum(C, 1e-10)
        phase_A = A / (np.abs(A) + 1e-30)
        A_new = np.sqrt(C_safe) * phase_A
        a_new_full = np.fft.irfft(A_new, fft_size)
        a_new = np.maximum(a_new_full[:n], 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum

        conv_new = fast_convolve(a_new)
        if conv_new.max() < f_val:
            a = a_new
            target_ratio = max(1.05, target_ratio * 0.98)
        else:
            # Blend
            improved = False
            for t in [0.2, 0.4, 0.6, 0.8]:
                a_blend = (1 - t) * a + t * a_new
                a_blend = project_simplex(a_blend)
                if fast_convolve(a_blend).max() < f_val:
                    a = a_blend
                    improved = True
                    break
            if not improved:
                target_ratio = min(2.0, target_ratio * 1.1)
                if target_ratio >= 2.0:
                    idx = rng.integers(0, n, size=max(5, n // 10))
                    a[idx] = np.maximum(a[idx] + rng.standard_normal(len(idx)) * 0.01, 0.0)
                    target_ratio = 1.2

    return best_a, best_f


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Bimodal (U-shape) - strong candidate
    for edge_frac in [0.15, 0.25, 0.35]:
        edge = max(2, int(n * edge_frac / 2))
        bi = np.zeros(n)
        bi[:edge] = 1.0
        bi[-edge:] = 1.0
        # Smooth edges
        for i in range(edge):
            fade = (i + 1) / (edge + 1)
            bi[i] *= (0.3 + 0.7 * fade)
            bi[n - 1 - i] *= (0.3 + 0.7 * fade)
        bi = np.maximum(bi, 0.01)
        bi /= bi.sum()
        inits[f'bimodal_{int(edge_frac*100)}'] = bi

    # Smooth U-shape
    u_shape = 1.0 / (1.0 + 5.0 * np.exp(-((x - n/2) / (n/6))**2))
    u_shape = np.maximum(u_shape, 0.01)
    u_shape /= u_shape.sum()
    inits['u_smooth'] = u_shape

    # Raised cosine (inverted - dip in middle)
    cos_w = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    cos_w = np.maximum(cos_w, 0.05)
    cos_w /= cos_w.sum()
    inits['inv_cos'] = cos_w

    # Gaussian (centered)
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss_center'] = g

    # Off-center Gaussian
    g_off = np.exp(-0.5 * ((x - 0.3 * n) / (0.2 * n)) ** 2)
    g_off /= g_off.sum()
    inits['gauss_off'] = g_off

    # Bimodal Gaussian (two bumps)
    bump = max(2, n // 8)
    g_bi = np.exp(-0.5 * ((x - bump) / (n/10))**2) + np.exp(-0.5 * ((x - (n-1-bump)) / (n/10))**2)
    g_bi /= g_bi.sum()
    inits['gauss_bimodal'] = g_bi

    # Chirp-magnitude inspired
    freq = np.linspace(0.1, 0.4, n)
    chirp = np.abs(np.exp(1j * np.pi * (x / n) ** 2))
    # Use magnitude of modulated signal
    mod = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x)
    mod = np.maximum(mod, 0.01)
    mod /= mod.sum()
    inits['modulated'] = mod

    # Random
    inits['rand_beta'] = rng.beta(0.5, 0.5, n)
    inits['rand_beta'] /= inits['rand_beta'].sum()
    inits['rand_dirichlet'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand_spiky'] = rng.dirichlet(np.ones(n) * 0.1)

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def multi_scale_optimize(a0, time_limit, seed=0):
    """
    Multi-scale: coarsen, optimize, then refine.
    """
    n = len(a0)
    if n <= 32:
        return coordinate_descent_incremental(a0, time_limit, seed)

    # Coarsen by factor 4
    factor = 4
    n_coarse = n // factor
    if n_coarse < 8:
        return coordinate_descent_incremental(a0, time_limit, seed)

    # Downsample
    a_coarse = np.zeros(n_coarse)
    for i in range(n_coarse):
        a_coarse[i] = a0[i*factor:(i+1)*factor].sum()
    a_coarse = np.maximum(a_coarse, 0)
    a_coarse /= a_coarse.sum()

    # Optimize coarse
    t_coarse = time_limit * 0.3
    a_c_opt, _ = coordinate_descent_incremental(a_coarse, t_coarse, seed)

    # Upsample
    a_up = np.zeros(n)
    for i in range(n_coarse):
        a_up[i*factor:(i+1)*factor] = a_c_opt[i]
    a_up = np.maximum(a_up, 0)
    s = a_up.sum()
    if s > 1e-10:
        a_up /= s

    # Blend with original
    a_blend = 0.5 * a_up + 0.5 * a0
    a_blend = project_simplex(a_blend)

    # Fine optimization
    t_fine = time_limit - (time.monotonic() - (time.monotonic() - time_limit))  # remaining
    t_fine = time_limit * 0.7
    return coordinate_descent_incremental(a_blend, t_fine, seed + 1)

                best_sequence = a_fup.copy()
                print(f"    FollowUp CCD: {score_fup:.6f}", flush=True)

    # Phase 3: Polish
    print("=== Phase 3: Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0

        # Long CCD
        t_long_ccd = remaining * 0.5
        if t_long_ccd > 10:
            a_pol, _ = coordinate_descent_incremental(best_sequence, t_long_ccd, seed=42)
            score_pol = float(evaluate_sequence(a_pol.tolist()))
            if np.isfinite(score_pol) and score_pol < best_score:
                best_score = score_pol
                best_sequence = a_pol.copy()
                print(f"  Polish CCD: {best_score:.6f}", flush=True)

        # Fine gradient
        remaining = deadline - time.monotonic() - 10.0
        t_fg = remaining * 0.3
        if t_fg > 5:
            a_fg, _ = smoothed_gradient_optimize(best_sequence, t_fg,
                                                  beta_init=200, beta_end=800,
                                                  lr_init=0.003, momentum=0.96,
                                                  seed=777)
            score_fg = float(evaluate_sequence(a_fg.tolist()))
            if np.isfinite(score_fg) and score_fg < best_score:
                best_score = score_fg
                best_sequence = a_fg.copy()
                print(f"  FineGrad: {best_score:.6f}", flush=True)

        # Final CCD
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 5:
            a_fin, _ = coordinate_descent_incremental(best_sequence, remaining, seed=999)
            score_fin = float(evaluate_sequence(a_fin.tolist()))
            if np.isfinite(score_fin) and score_fin < best_score:
                best_score = score_fin
                best_sequence = a_fin.copy()
                print(f"  FinalCCD: {best_score:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
