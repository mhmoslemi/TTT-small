Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-based iterative
refinement with hinge-penalty gradient polishing, adaptive step sizes, and

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end+1] = a[k - np.arange(j_start, j_end+1)]
    return A

def lp_min_peak(a, M_init=None):
    """Solve: min M s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None

def lp_max_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None

def hinge_gradient(a, t, n):
    """Compute hinge-penalty gradient efficiently using FFT."""
    conv = fast_convolve(a)
    h = np.maximum(conv - t, 0)
    if h.max() < 1e-15:
        return np.zeros(n), 0.0
    a_rev = a[::-1]
    fft_size = 1 << (len(h) + n - 1 - 1).bit_length()
    H = np.fft.rfft(h, fft_size)
    Br = np.fft.rfft(a_rev, fft_size)
    grad_raw = np.fft.irfft(H * Br, fft_size)[:len(h) + n - 1]
    grad = 2.0 * n * grad_raw[n-1 : 2*n-1]
    obj = 2.0 * n * h.sum()
    return grad, obj
