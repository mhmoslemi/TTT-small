import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp(a, M, C=None):
    """Solve LP: maximize sum(g) s.t. C @ g <= M, g >= 0."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    
    if result.success and result.x is not None:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g, g_sum
    return None, 0.0

        a_try = np.maximum(a_try, 0.0)

        a_try = a_try / s
        conv = fast_convolve(a_try)
        return 2.0 * n * conv.max()

    f_lo = best_obj
    f_hi = obj_at(hi)

    if f_hi > f_lo:
        # Try smaller steps
        hi = 0.5
        f_hi = obj_at(hi)
        if f_hi > f_lo:
            hi = 0.1
            f_hi = obj_at(hi)
            if f_hi > f_lo:
                hi = 0.01
                f_hi = obj_at(hi)

    c_pt = hi - phi * (hi - lo)
    d_pt = lo + phi * (hi - lo)
    f_c = obj_at(c_pt)

            f_d = f_c

            c_pt = d_pt
            f_c = f_d

        if f_c < best_f:

        if f_d < best_f:

        C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'presolve': True, 'dual_feasibility_tolerance': 1e-10,
               'primal_feasibility_tolerance': 1e-10, 'time_limit': 30}
    result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs', options=options)
    if result.success and result.x is not None:
        g = result.x
        g_sum = g.sum()
        duals = None
        if result.ineqlin is not None and result.ineqlin.marginals is not None:
            duals = result.ineqlin.marginals
        if g_sum > 1e-10:
            return g, g_sum, duals
    return None, 0.0, None


def obj_and_conv(a, n):
    """Compute objective value and convolution for normalized a."""
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf'), conv
    val = 2.0 * n * conv.max() / (s * s)
    return val, conv


def line_search(a, d, n, curr_obj, max_evals=25):
    """Adaptive line search with geometric + refinement."""
    best_t = 0.0
    best_obj = curr_obj
    evals = 0

    # Geometric candidates
    candidates = [0.8, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04, 0.02, 0.01]
    for t in candidates:
        if evals >= max_evals:
            break
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            evals += 1
            continue
        a_try = a_try / s
        obj_try, _ = obj_and_conv(a_try, n)
        evals += 1
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t

    # Refinement around best
    if 0.01 < best_t < 0.99 and evals < max_evals:
        step = 0.04
        for _ in range(4):
            improved = False
            for dt in [-step, step]:
                t_try = best_t + dt
                if t_try <= 0 or t_try >= 1.0 or evals >= max_evals:
                    continue
                a_try = a + t_try * d
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                obj_try, _ = obj_and_conv(a_try, n)
                evals += 1
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_t = t_try
                    improved = True
            step *= 0.5
            if not improved:
                break

    return best_t, best_obj


def lp_optimize_improved(a0, time_limit, verbose=False):
    """Improved LP-based optimization with adaptive constraints and peak targeting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    _, conv = obj_and_conv(a, n)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 200
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        C = build_conv_matrix(a)

        # Adaptive constraint: tighten slightly for more aggressive direction
        tighten = 1.0 - 0.001 * min(iteration, 100)
        M_lp = M * max(0.995, tighten)

        g, g_sum, duals = solve_lp_direction(a, M_lp, C)

        if g is None or g_sum <= 1.0 + 1e-8:
            no_improve += 1
        else:
            d = g / g_sum - a
            norm_d = np.linalg.norm(d)

            if norm_d < 1e-14:
                no_improve += 1
            else:
                t_best, obj_best = line_search(a, d, n, best_obj)

                if obj_best < best_obj - 1e-13:
                    a_new = a + t_best * d
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a = a_new / s_new
                    else:
                        a = best_a.copy()

                    _, conv = obj_and_conv(a, n)
                    M = conv.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0

                    if verbose and iteration % 200 == 0:
                        print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={elapsed:.1f}s")
                else:
                    no_improve += 1

        iteration += 1

        # Stagnation handling with peak-targeted perturbation
        if no_improve >= max_no_improve:
            _, conv = obj_and_conv(best_a, n)
            # Find top peaks
            top_k = min(10, 2 * n - 1)
            peak_idxs = np.argpartition(conv, -top_k)[-top_k:]

            # Identify elements contributing most to peaks
            contrib = np.zeros(n)
            for k in peak_idxs:
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                idx = np.arange(j_start, j_end + 1)
                contrib[idx] += conv[k] * best_a[k - idx]

            # Perturb: reduce high-contributing elements, boost others
            rng = np.random.default_rng(iteration * 7 + 13)
            scale = 0.003 + 0.005 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            # Extra reduction at high-contribution elements
            noise -= 0.5 * scale * contrib / (contrib.max() + 1e-15)

            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            _, conv = obj_and_conv(a, n)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(500, 200 + 50 * perturb_count)

            if perturb_count > 50:
                break

    return best_a, best_obj


def subgradient_optimize(a0, time_limit, lr_init=0.012, momentum=0.92, top_k=60, temp=0.00015):
    """Projected subgradient with softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    _, conv = obj_and_conv(a, n)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    batch = 20

    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break

            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]
            scaled = peak_vals / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(peak_vals)] = 1.0
            else:
                w /= w_sum

            sg = np.zeros(n)
            for i in range(len(peaks)):
                if w[i] < 1e-12:
                    continue
                k = int(peaks[i])
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                idx = np.arange(j_start, j_end + 1)
                sg[idx] += w[i] * 2.0 * a[k - idx]

            v = momentum * v - lr * sg
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
            else:
                a = best_a.copy()

            _, conv = obj_and_conv(a, n)
            cur_obj = 2.0 * n * conv.max()

            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        lr *= 0.998
        temp *= 1.003
        momentum = min(0.97, momentum + 0.0002)

        if no_improve >= 400:
            break

    return best_a, best_obj


def generate_diverse_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['random2'] = rng.dirichlet(np.ones(n) * 1.0)

    ramp = x + 1.0
    inits['ramp'] = ramp / ramp.sum()

    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    # Bimodal (concentrated at edges)
    bi = np.zeros(n)
    edge = max(1, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Sine
    inits['sine'] = (np.sin(np.pi * (x + 1) / (n + 1)) + 0.05)
    inits['sine'] = inits['sine'] / inits['sine'].sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_ds'] = pp / pps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_up'] = pp / pps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: broad LP scan across lengths, deep LP optimization,
    subgradient polishing, and perturbation-based restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    if parent is not None:
        parent_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(parent_score):
            best_score = parent_score
            best_sequence = parent.copy()
            print(f"Parent score: {parent_score:.6f}")

    # Phase 1: Quick LP scan across multiple lengths
    print("=== Phase 1: Multi-length LP Scan ===")
    t_start = time.monotonic()
    phase1_limit = min(400.0, (deadline - time.monotonic()) * 0.25)
    phase1_deadline = t_start + phase1_limit

    lengths = [384, 448, 512, 576, 640, 768]
    scan_results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break

        inits = generate_diverse_initializations(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'parent_ds', 'parent_up', 'uniform', 'random', 'sine', 'gaussian']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:4]

        t_per_init = max(10.0, (phase1_limit - 20.0) / (len(lengths) * 4))

        for name in init_names:
            if time.monotonic() > phase1_deadline - 15:
                break
            t_alloc = min(t_per_init, phase1_deadline - time.monotonic() - 10)
            if t_alloc < 5:
                break

            a0 = inits[name]
            a_opt, obj_opt = lp_optimize_improved(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))

            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # Phase 2: Deep LP optimization on top candidates
    print("=== Phase 2: Deep LP Optimization ===")
    t_start2 = time.monotonic()
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, a_opt))
            seen.add(key)
        if len(top_configs) >= 3:
            break

    if best_sequence is not None:
        top_configs.append((len(best_sequence), best_sequence.copy()))

    # Deduplicate by length
    final_configs = []
    seen_lens = set()
    for n, a0 in top_configs:
        if n not in seen_lens:
            final_configs.append((n, a0))
            seen_lens.add(n)

    remaining = deadline - time.monotonic() - 200.0
    n_candidates = max(1, len(final_configs))
    t_per_deep = max(30.0, remaining * 0.45 / n_candidates)

    for i, (n, a0) in enumerate(final_configs):
        if time.monotonic() > deadline - 180:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 150)
        if t_alloc < 15:
            break

        a_opt, obj_opt = lp_optimize_improved(a0, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep LP {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # Phase 3: Subgradient polishing with multiple configs
    print("=== Phase 3: Subgradient Polish ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 100.0
        if remaining > 30:
            sg_configs = [
                {'lr_init': 0.008, 'momentum': 0.93, 'top_k': 50, 'temp': 0.00012},
                {'lr_init': 0.015, 'momentum': 0.90, 'top_k': 80, 'temp': 0.0002},
                {'lr_init': 0.005, 'momentum': 0.95, 'top_k': 30, 'temp': 0.00008},
            ]

            t_per_sg = remaining / (len(sg_configs) + 2)

            for ci, cfg in enumerate(sg_configs):
                if time.monotonic() > deadline - 60:
                    break
                t_alloc = min(t_per_sg, deadline - time.monotonic() - 40)
                if t_alloc < 10:
                    break

                a_opt, obj_opt = subgradient_optimize(best_sequence, time_limit=t_alloc, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  SG config {ci}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # Phase 4: Final LP polish + perturbation restarts
    print("=== Phase 4: Final Polish ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 30:
            # Multiple perturbation + LP cycles
            n_best = len(best_sequence)
            n_attempts = max(1, int(remaining / 40))
            t_per_attempt = remaining / n_attempts

            for pi in range(n_attempts):
                if time.monotonic() > deadline - 25:
                    break
                t_alloc = min(t_per_attempt - 5, deadline - time.monotonic() - 20)
                if t_alloc < 10:
                    break

                # Different perturbation strategies
                if pi % 3 == 0:
                    # Small Gaussian perturbation
                    scale = 0.002 + 0.001 * pi
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = best_sequence + noise
                elif pi % 3 == 1:
                    # Peak-targeted perturbation
                    _, conv = obj_and_conv(best_sequence, n_best)
                    top_peaks = np.argpartition(conv, -5)[-5:]
                    a_pert = best_sequence.copy()
                    for k in top_peaks:
                        j_start = max(0, k - n_best + 1)
                        j_end = min(n_best - 1, k)
                        a_pert[j_start:j_end+1] *= 0.98
                    a_pert = a_pert + rng.standard_normal(n_best) * 0.001
                else:
                    # Blend with a different initialization
                    alt_inits = generate_diverse_initializations(n_best, best_sequence, rng)
                    alt_name = list(alt_inits.keys())[pi % len(alt_inits)]
                    a_pert = 0.9 * best_sequence + 0.1 * alt_inits[alt_name]

                a_pert = np.maximum(a_pert, 0.0)
                s = a_pert.sum()
                if s > 1e-15:
                    a_pert = a_pert / s

                a_opt, obj_opt = lp_optimize_improved(a_pert, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Perturb {pi}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Final LP polish on best
    if best_sequence is not None and time.monotonic() < deadline - 10:
        t_final = min(20.0, deadline - time.monotonic() - 5)
        if t_final > 5:
            a_opt, obj_opt = lp_optimize_improved(best_sequence, time_limit=t_final)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: {score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
