import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    # For each k, conv(a,b)[k] = sum_{j} a[k-j] * b[j] where 0<=j<n and 0<=k-j<n
    # So (A @ b)[k] = sum_j A[k,j] * b[j] where A[k,j] = a[k-j] if 0<=k-j<n else 0
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    return A

def lp_min_peak_direction(a):
    """Solve: min t s.t. conv(a,b)[k] <= t for all k, sum(b)=1, b>=0.
    Returns (b_opt, t_opt)."""
    n = len(a)
    num_rows = 2 * n - 1
    A = build_conv_matrix(a)
    # Variables: b_0,...,b_{n-1}, t
    # min t
    # s.t. A @ b - t <= 0 (i.e., conv(a,b)[k] <= t)
    #      sum(b) = 1
    #      b >= 0
    #      t free
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # A@b - t <= 0
    b_ub = np.zeros(num_rows)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        b_opt = np.maximum(res.x[:n], 0)
        t_opt = res.x[n]
        return b_opt, t_opt
    return None, None

def lp_fixed_point_iteration(a0, time_limit, max_iters=10000):
    """Iterate a -> argmin_b max_k conv(a,b) until convergence or time limit."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    prev_t = conv.max()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b, new_t = lp_min_peak_direction(a)
        if b is None:
            break
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum
        
        # Check convergence
        if np.linalg.norm(b - a) < 1e-10:
            break
        
        # Update
        a = b
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Check if the LP minimum peak is achieved at the current a
        # (i.e., we're at a fixed point)
        if new_t >= conv.max() * (1 - 1e-8):
            # We might be at a fixed point
            if abs(new_t - conv.max()) < 1e-8 * conv.max():
                break
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fixed-point iteration
with monotone sequence initializations and adaptive refinement.
Key insight: monotone sequences minimize central autocorrelation by rearrangement,
giving objectives ~4/3 vs 2 for uniform.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def lp_min_peak(a):
    """Solve: min t s.t. conv(a,b)[k] <= t for all k, sum(b)=1, b>=0.
    Returns (b, t)."""
    n = len(a)
    num_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None

def lp_max_sum(a, M):
    """Solve: max sum(b) s.t. conv(a,b)[k] <= M for all k, b>=0.
    Returns b."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def compute_obj(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (a.sum()**2)

def lp_refine(a0, time_limit, max_iters=50000):
    """LP-based refinement using min-peak direction with adaptive step."""
    a = normalize(a0)
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    method = 'min_peak'
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        if method == 'min_peak':
            b, t_lp = lp_min_peak(a)
            if b is None:
                method = 'max_sum'
                continue
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum
        else:
            M = conv.max()
            b = lp_max_sum(a, M)
            if b is None:
                method = 'min_peak'
                continue
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum
        
        if np.linalg.norm(b - a, ord=np.inf) < 1e-10:
            break
        
        # Try full step
        a_new = b
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            a = a_new
            conv = conv_new
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            # Try partial steps
            improved = False
            for step in [0.5, 0.3, 0.1, 0.05]:
                a_try = (1 - step) * a + step * b
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    a = a_try
                    conv = conv_try
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve = 0
                    improved = True
                    break
            if not improved:
                no_improve += 1
                if no_improve > 30:
                    method = 'max_sum' if method == 'min_peak' else 'min_peak'
                    no_improve = 0
                elif no_improve > 100:
                    # Perturb and try again
                    rng = np.random.default_rng(it)
                    a = project_simplex(a + rng.standard_normal(n) * 0.005)
                    conv = fast_convolve(a)
                    no_improve = 0
                if no_improve > 500:
                    break
    
    return best_a, best_obj

def subgradient_polish(a0, time_limit, lr_init=0.02, top_k=50):
    """Subgradient polish on the current best."""
    a = normalize(a0)
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    momentum = 0.9
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Find top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        # Compute subgradient
        sg = np.zeros(n)
        for k in peaks:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += 2.0 * a[k - j]
        
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.999
        if no_improve > 500:
            lr *= 0.5
            v *= 0.1
            no_improve = 0
        if no_improve > 5000:
            break
    
    return best_a, best_obj
