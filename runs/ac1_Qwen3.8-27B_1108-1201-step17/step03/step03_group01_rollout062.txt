def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Sine-based
    for freq in [1, 2, 3]:
        s = np.sin(np.pi * freq * x / n) + 1.5
        inits[f'sine_{freq}'] = s / s.sum()

    # Cosine bell (concentrated in middle)
    for width in [0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (width * n)) ** 2)
        inits[f'gauss_{width}'] = g / g.sum()

    # Bimodal
    for edge_frac in [0.15, 0.25, 0.35]:
        bi = np.zeros(n)
        edge = max(1, int(n * edge_frac))
        bi[:edge] = 1.0
        bi[-edge:] = 1.0
        inits[f'bimodal_{edge_frac}'] = bi / bi.sum()

    # Power law
    for pw in [0.5, 1.0, 2.0]:
        p = (x + 1.0) ** pw
        inits[f'power_{pw}'] = p / p.sum()
        inits[f'power_{pw}_rev'] = p[::-1] / p.sum()

    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                pp = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    pp = pp / pps
                else:
                    pp = np.ones(n) / n
            inits['parent'] = pp

    return inits


def kkt_gap(a, c, peaks, lam):
    """Compute KKT gap: how much the subgradient deviates from constant on support."""
    g = compute_subgradient(a, peaks, lam)
    support = a > 1e-10
    if support.sum() < 2:
        return 0.0
    g_support = g[support]
    return (g_support.max() - g_support.min()) / (abs(g_support).mean() + 1e-30)

    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    # Determine working length
    if parent is not None:
        n_work = len(parent)
    else:
        n_work = 512

    print(f"Working length: {n_work}")

    # === Phase 1: Quick mirror descent sweep (300s) ===
    print("=== Phase 1: Mirror Descent Sweep ===")
    t_start = time.monotonic()
    phase1_limit = 350.0
    phase1_deadline = t_start + phase1_limit

    inits = generate_diverse_inits(n_work, parent, rng)
    # Prioritize parent, then diverse patterns
    priority = ['parent', 'sine_1', 'sine_2', 'gauss_0.3', 'bimodal_0.25',
                'triangle', 'power_1.0', 'uniform', 'random']
    init_names = [nm for nm in priority if nm in inits] + \
                 [nm for nm in inits if nm not in priority]
    init_names = init_names[:6]

    t_per_init = max(15.0, (phase1_limit - 10.0) / len(init_names))

    phase1_results = []
    for name in init_names:
        if time.monotonic() > phase1_deadline - 10:
            break
        t_alloc = min(t_per_init, phase1_deadline - time.monotonic() - 5)
        if t_alloc < 5:
            break

        a0 = inits[name]
        a_opt, obj_opt = mirror_descent_optimize(a0, time_limit=t_alloc,
                                                 initial_eta=0.8, tol_frac=0.02,
                                                 max_peaks=30)
        score = float(evaluate_sequence(a_opt.tolist()))
        phase1_results.append((score, n_work, name, a_opt.copy()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  MD {name}: {score:.8f}")

    phase1_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.8f}")

    # === Phase 2: LP deep optimization (400s) ===
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()
    phase2_limit = 400.0
    phase2_deadline = t_start2 + phase2_limit

    # Take top candidates from Phase 1 + parent
    lp_candidates = []
    seen_inits = set()
    for score, n, name, a_opt in phase1_results:
        if name not in seen_inits:
            lp_candidates.append((n, a_opt, name))
            seen_inits.add(name)
        if len(lp_candidates) >= 3:
            break

    if parent is not None:
        pp = parent.copy()
        ps = pp.sum()
        if ps > 1e-10:
            pp = pp / ps
            lp_candidates.append((n_work, pp, 'parent_lp'))

    t_per_lp = max(30.0, (phase2_limit - 20.0) / max(1, len(lp_candidates)))

    for i, (n_c, a0, name) in enumerate(lp_candidates):
        if time.monotonic() > phase2_deadline - 10:
            break
        t_alloc = min(t_per_lp, phase2_deadline - time.monotonic() - 5)
        if t_alloc < 10:
            break

        a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  LP {name} (n={n_c}): {score:.8f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.8f}")

    # === Phase 3: Alternating refinement (700s) ===
    print("=== Phase 3: Alternating Refinement ===")
    t_start3 = time.monotonic()
    phase3_limit = 700.0
    phase3_deadline = t_start3 + phase3_limit

    if best_sequence is None:
        best_sequence = np.ones(n_work) / n_work
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    a_cur = best_sequence.copy()
    a_cur = np.maximum(a_cur, 0)
    sc = a_cur.sum()
    if sc > 1e-15:
        a_cur = a_cur / sc

    # Alternate between LP and MD with perturbations
    no_global_improve = 0
    perturb_scale = 0.005
    cycle = 0
    t_per_cycle = max(5.0, phase3_limit / 120)

    while time.monotonic() < phase3_deadline and no_global_improve < 40:
        cycle += 1
        elapsed_total = time.monotonic() - time.monotonic() + (time.monotonic() - t_start3)
        if time.monotonic() > phase3_deadline - 5:
            break

        # Apply perturbation
        if cycle > 1:
            # Mix of Fourier and random perturbations
            if cycle % 3 == 0:
                freq = rng.integers(1, n_work // 3 + 1)
                phase = rng.uniform(0, 2 * np.pi)
                pert = perturb_scale * np.sin(2 * np.pi * freq * np.arange(n_work) / n_work + phase)
            elif cycle % 3 == 1:
                pert = rng.standard_normal(n_work) * perturb_scale
            else:
                # Localized perturbation
                center = rng.integers(n_work // 4, 3 * n_work // 4)
                width = rng.integers(5, n_work // 5 + 1)
                pos = (np.arange(n_work) - center) ** 2 / (2.0 * width ** 2)
                pert = perturb_scale * np.exp(-pos) * rng.standard_normal(n_work)

            a_pert = a_cur + pert
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp > 1e-15:
                a_cur = a_pert / sp
            perturb_scale = min(0.05, perturb_scale * 1.02)

        # Optimize: alternate LP and MD
        t_alloc = min(t_per_cycle, phase3_deadline - time.monotonic() - 5)
        if t_alloc < 3:
            break

        if cycle % 4 in [0, 1]:
            # LP-based
            a_opt, obj_opt = lp_optimize(a_cur, time_limit=t_alloc * 0.7)
            if np.isfinite(obj_opt) and obj_opt < 2.0 * n_work:
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score - 1e-10:
                    best_score = score
                    best_sequence = a_opt.copy()
                    a_cur = a_opt.copy()
                    no_global_improve = 0
                    perturb_scale = 0.003
                    print(f"  Cycle {cycle} LP: {score:.8f}")
                else:
                    no_global_improve += 1
        else:
            # Mirror descent
            a_opt, obj_opt = mirror_descent_optimize(a_cur, time_limit=t_alloc,
                                                     initial_eta=0.5 + 0.1 * (cycle % 5),
                                                     tol_frac=0.005 + 0.005 * (cycle % 3),
                                                     max_peaks=50)
            if np.isfinite(obj_opt):
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score - 1e-10:
                    best_score = score
                    best_sequence = a_opt.copy()
                    a_cur = a_opt.copy()
                    no_global_improve = 0
                    perturb_scale = 0.003
                    print(f"  Cycle {cycle} MD: {score:.8f}")
                else:
                    no_global_improve += 1

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.8f}")

    # === Phase 4: Final polish (remaining time) ===
    print("=== Phase 4: Final Polish ===")
    t_start4 = time.monotonic()

    if best_sequence is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 5
        # Final LP polish with more time
        t_lp = remaining * 0.4
        if t_lp > 10:
            a_opt, _ = lp_optimize(best_sequence, time_limit=t_lp)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: {score:.8f}")

        # Final MD polish
        remaining = deadline - time.monotonic() - 5
        t_md = remaining * 0.4
        if t_md > 5:
            a_opt, _ = mirror_descent_optimize(best_sequence, time_limit=t_md,
                                               initial_eta=0.3, tol_frac=0.001,
                                               max_peaks=80)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final MD: {score:.8f}")

        # Try a few more perturbations + quick MD
        remaining = deadline - time.monotonic() - 5
        n_tries = max(1, int(remaining / 8))
        for i in range(n_tries):
            if time.monotonic() > deadline - 3:
                break
            rng2 = np.random.default_rng(1000 + i)
            scale = 0.002 * (1 + 0.5 * i)
            pert = rng2.standard_normal(len(best_sequence)) * scale
            a_pert = best_sequence + pert
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp < 1e-15:
                continue
            a_pert /= sp
            t_alloc = min(5.0, deadline - time.monotonic() - 2)
            if t_alloc < 2:
                break
            a_opt, _ = mirror_descent_optimize(a_pert, time_limit=t_alloc,
                                               initial_eta=0.4, tol_frac=0.003)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final perturb {i}: {score:.8f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.8f}")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(n_work) / n_work

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
        # Make sure we return the right sequence
    else:
        best_sequence = np.ones(n_work) / n_work
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
