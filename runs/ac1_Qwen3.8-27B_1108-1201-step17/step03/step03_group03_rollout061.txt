momentum, simulated annealing, and coordinate descent polishing.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_norm(a):
    """Objective assuming sum(a)=1: returns 2*n*max(conv(a,a))."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def subgrad_direct(a, conv, n, rel_tol=1e-8):
    """Compute subgradient of max(conv) w.r.t. a using direct method."""
    pmax = conv.max()
    if pmax < 1e-18:
        return np.zeros(n)
    threshold = pmax - rel_tol * pmax
    active_idx = np.where(conv >= threshold)[0]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(conv)])
    
    sg = np.zeros(n)
    for k in active_idx:
        if k < n:
            sg[:k+1] += a[k::-1]
        else:
            sg[k-n+1:] += a[n-1:k-n:-1]
    return sg / len(active_idx)

Mirror descent with negative entropy:

The update is multiplicative: if `sg[j] > 0` (this coordinate contributes to the peak), we decrease `a[j]`. If `sg[j] < 0` (unlikely since `a >= 0` and `sg` is a sum of products of non-negative values), we increase it.

In mirror descent, the update is:
`a_{k+1} = argmin_a { lr * <sg, a> + D(a || a_k) }`

Where `D(a || a_k) = sum_j a[j] * log(a[j]/a_k[j]) - a[j] + a_k[j]` is the KL divergence.

The solution is: `a_{k+1}[j] = a_k[j] * exp(-lr * sg[j]) / Z`

But since `sg[j] >= 0` for all `j` (because it's a sum of non-negative terms), this would only decrease all coordinates, and the normalization `Z` would scale them up. The net effect is that coordinates with larger `sg[j]` decrease more relative to others.

This is actually correct! The normalization ensures `sum(a_{k+1}) = 1`, and the relative changes are `a_{k+1}[j] / a_k[j] = exp(-lr * sg[j]) / Z`. So coordinates with high `sg[j]` get relatively smaller, and coordinates with low `sg[j]` get relatively larger (or decrease less).

import numpy as np
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_val(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()


def subgrad_compute(a, conv, n, rel_tol=1e-8):
    pmax = conv.max()
    if pmax < 1e-18:
        return np.zeros(n)
    threshold = pmax * (1.0 - rel_tol)
    active_idx = np.where(conv >= threshold)[0]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(conv)])
    sg = np.zeros(n)
    for k in active_idx:
        if k < n:
            sg[:k+1] += a[k::-1]
        else:
            sg[k-n+1:] += a[n-1:k-n:-1]
    return sg / len(active_idx)


def mirror_descent(a_init, time_limit, lr_init=0.1, momentum=0.9,
                   decay=0.999, pert_interval=100, pert_size=0.01, seed=0):
    """Mirror descent (multiplicative updates) with momentum."""
    a = np.maximum(a_init, 1e-15).astype(np.float64).copy()
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)  # momentum on the log-scale
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        sg = subgrad_compute(a, conv, n)
        
        # Momentum on the gradient
        v = momentum * v - lr * sg
        
        # Multiplicative update
        a_new = a * np.exp(v)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            a = a_new
            conv = conv_new
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= pert_interval:
            # Add random perturbation on the log scale
            log_a = np.log(np.maximum(a, 1e-15))
            log_a += rng.standard_normal(n) * pert_size
            a = np.exp(log_a)
            a = a / a.sum()
            conv = fast_convolve(a)
            v = np.zeros(n)
            no_improve = 0
            lr = max(0.001, lr * 0.5)
        
        lr *= decay
    
    return best_a, best_obj
