"""
Improved LP-based Frank-Wolfe with top-T constraint oracle, peak-directed
subgradient polishing, multi-length scanning, and adaptive exploration.
Key improvements over previous: (1) top-T LP oracle for speed, (2) peak-directed
subgradient with adaptive temperature, (3) better initialization from parent
with structured perturbations, (4) adaptive length selection.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_objective_normalized(a):
    """Compute 2*n*max(conv) for sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix_rows(a, top_indices):
    """Build sparse constraint matrix only for specified lag indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(top_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(i)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(top_indices), n))


def lp_oracle_topT(a, M, top_indices):
    """Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for k in top_indices, b >= 0."""
    n = len(a)
    T = len(top_indices)
    A = build_conv_matrix_rows(a, top_indices)
    c_obj = -np.ones(n)
    b_ub = np.full(T, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                      options={'presolve': True, 'time_limit': 3.0})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_oracle_full(a, M):
    """Solve LP with all constraints (fallback)."""
    n = len(a)
    top_indices = list(range(2 * n - 1))
    return lp_oracle_topT(a, M, top_indices)


def fpw_optimize(a0, time_limit, verbose=False, topT_init=8, topT_max=40):
    """Frank-Wolfe iterations with adaptive top-T LP oracle."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 100
    current_topT = topT_init

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Get top peaks
        topT = min(current_topT, 2 * n - 1)
        peaks = np.argpartition(conv, -topT)[-topT:]
        peaks = peaks[np.argsort(-conv[peaks])]

        # Solve LP oracle
        b = lp_oracle_topT(a, M, peaks.tolist())
        if b is None:
            # Fallback: try with fewer constraints
            peaks_half = peaks[:max(3, len(peaks) // 2)]
            b = lp_oracle_topT(a, M, peaks_half.tolist())
            if b is None:
                break

        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb

        # Check if stuck
        if np.allclose(a, b, atol=1e-12):
            no_improve += 1
        else:
            # Line search
            best_t = 0.0
            best_local = compute_objective_normalized(a)
            for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidate = (1 - t) * a + t * b
                obj_val = compute_objective_normalized(candidate)
                if obj_val < best_local:
                    best_local = obj_val
                    best_t = t

            if best_t < 0.01:
                no_improve += 1
            else:
                a = (1 - best_t) * a + best_t * b
                conv = fast_convolve(a)
                new_M = conv.max()
                cur_obj = 2.0 * n * new_M
                if cur_obj < best_obj - 1e-14:
                    best_obj = cur_obj
                    best_a = a.copy()
                    M = new_M
                    no_improve = 0
                else:
                    no_improve += 1

        # Adapt topT: increase over time for more constraints
        if it % 100 == 0:
            current_topT = min(topT_max, topT_init + it // 50)

        # Stuck handling: noise perturbation
        if no_improve >= max_stall:
            rng = np.random.default_rng(it + 12345)
            noise = rng.standard_normal(n) * 0.002
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max_stall * 1.5)
            current_topT = topT_init  # Reset to fewer constraints for exploration

        it += 1
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FPW it={it}, obj={best_obj:.6f}, M={M:.4e}, topT={current_topT}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def peak_directed_subgradient(a0, time_limit, lr_init=0.004, momentum=0.93):
    """Subgradient descent focusing on top peaks with adaptive weighting."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        # Get top peaks with soft weighting
        top_k = min(20, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        
        # Soft attention: weight peaks by exponential of their normalized values
        temp = max(peak_vals.max() * 0.08, 1e-15)
        scaled = peak_vals / temp
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Compute subgradient: gradient of weighted sum of peaks
        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            # d/dx of sum_j x[j]*x[k-j] at k: 2 * sum over pairs
            # For the k-th lag: grad = 2 * [x[k-j] for valid j] (symmetrized)
            grad_part = 2.0 * a[k - idx].copy()
            g[idx] += w[i] * grad_part

        v = momentum * v - lr * g
        a_new = np.maximum(a + v, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()

        lr *= 0.9993
        it += 1

    return best_a, best_obj


def directed_peak_reduction(a0, time_limit, n_inner=500):
    """Iteratively reduce the single largest peak by solving 1D subproblems."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        # Find the peak lag
        k_star = int(np.argmax(conv))
        
        # Try to reduce conv[a,a][k_star] by adjusting pairs (j, k_star-j)
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        
        improved = False
        for j in range(j_start, j_end + 1):
            k2 = k_star - j
            # Redistribute between a[j] and a[k2] keeping their sum constant
            s_pair = a[j] + a[k2]
            if s_pair < 1e-15:
                continue
            
            best_ratio = a[j] / s_pair
            best_val = conv[k_star]
            
            # Try different ratios
            for ratio in [best_ratio + 0.05, best_ratio - 0.05, best_ratio + 0.1,
                         best_ratio - 0.1, 0.0, 1.0, 0.5]:
                if ratio < -0.01 or ratio > 1.01:
                    continue
                ratio = np.clip(ratio, 0.0, 1.0)
                a[j] = ratio * s_pair
                a[k2] = (1 - ratio) * s_pair
                
                # Quick check: recompute conv at k_star only (approximate)
                # Full recompute is expensive; just check the affected lag
                new_val = 0.0
                for j2 in range(max(0, k_star - n + 1), min(n - 1, k_star) + 1):
                    new_val += a[j2] * a[k_star - j2]
                
                if new_val < best_val - 1e-15:
                    best_val = new_val
                    best_ratio = ratio
                else:
                    a[j] = a[j]  # restore
                    a[k2] = a[k2]
            
            # Apply best ratio
            a[j] = best_ratio * s_pair
            a[k2] = (1 - best_ratio) * s_pair

        if not improved:
            break

        # Full recompute
        conv = fast_convolve(a)
        s = a.sum()
        if s > 1e-15:
            a = a / s
            conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()

        it += 1

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    inits['uniform'] = np.ones(n) / n

    # Edge-concentrated (various widths)
    for frac in [0.2, 0.3, 0.5]:
        edge = max(1, int(n * frac))
        ec = np.zeros(n)
        ec[:edge] = 1.0
        ec[-edge:] = 1.0
        ec /= ec.sum()
        inits[f'edge_{frac}'] = ec.copy()

    # Smooth edge concentration (exponential falloff from edges)
    x = np.arange(n, dtype=np.float64)
    for sigma_factor in [0.2, 0.3, 0.5]:
        dist = np.minimum(x, n - 1 - x)
        env = np.exp(-(dist / (sigma_factor * n)) ** 0.5)
        env /= env.sum()
        inits[f'edge_falloff_{sigma_factor}'] = env.copy()

    # Triangle (peaked in center - opposite of what we want, but useful)
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri.copy()

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g.copy()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Random sparse
    sparse = rng.random(n)
    sparse[sparse < 0.4] = 0.0
    if sparse.sum() < 1e-10:
        sparse = rng.dirichlet(np.ones(n))
    else:
        sparse /= sparse.sum()
    inits['sparse'] = sparse.copy()

    # Oscillatory edge pattern
    for freq in [3.0, 5.0, 8.0]:
        osc = np.ones(n) * 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n)
        osc = np.maximum(osc, 0.01)
        # Multiply by edge envelope
        env = np.minimum(x + 1, n - x)
        osc *= env
        osc /= osc.sum()
        inits[f'osc_{freq}'] = osc.copy()

    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps

    return inits
