from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft

def polyak_subgrad(a0, time_limit, lb_factor=1.0, n_active=50, momentum=0.5, seed=42):
    """
    Polyak subgradient method with momentum.
    a0: initial point on simplex (sum=1, non-negative)
    lb_factor: multiply the theoretical lower bound by this (should be <= 1)
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    # Lower bound on max(conv)
    lb = 1.0 / (2 * n - 1) * lb_factor
    
    conv = fast_convolve(a)
    best_f = conv.max()
    best_a = a.copy()
    v = np.zeros(n)
    
    t0 = time.monotonic()
    iters = 0
    step_scale = 1.0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        f_cur = conv.max()
        
        if f_cur < best_f - 1e-12:
            best_f = f_cur
            best_a = a.copy()
        
        # Compute subgradient
        g = subgrad_max_conv(a, conv, n_active)
        g_norm_sq = np.dot(g, g)
        
        if g_norm_sq < 1e-30:
            break
        
        # Polyak step size
        if f_cur > lb + 1e-10:
            eta = (f_cur - lb) / g_norm_sq * step_scale
        else:
            eta = 0.01 / np.sqrt(g_norm_sq) * step_scale
        
        # Cap the step
        eta = min(eta, 1.0 / np.sqrt(g_norm_sq + 1e-10))
        
        # Momentum update
        v = momentum * v - eta * g
        a_new = project_simplex(a + v)
        
        # Check if step improved
        conv_new = fast_convolve(a_new)
        f_new = conv_new.max()
        
        if f_new > f_cur * 1.05 + 1e-8:
            # Step was too large, reduce it
            step_scale *= 0.7
            v *= 0.5
            a_new = project_simplex(a - 0.1 * eta * g)
            conv_new = fast_convolve(a_new)
            f_new = conv_new.max()
            if f_new > f_cur:
                # Still bad, just decay
                a_new = a.copy()
        
        a = a_new
        iters += 1
        
        if iters % 5000 == 0:
            print(f"  PolySub: iter={iters}, f={f_cur:.8f}, best={best_f:.8f}", flush=True)
    
    return best_a, best_f


def lse_optimize(a0, time_limit, beta_start=50, beta_end=500, lr=0.01, 
                 momentum=0.9, seed=42):
    """Log-sum-exp gradient descent with Nesterov momentum."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_f = conv.max()
    best_a = a.copy()
    v = np.zeros(n)
    
    t0 = time.monotonic()
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_start + (beta_end - beta_start) * frac
        lr_cur = lr * (1.0 - 0.5 * frac)
        
        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        conv_y = fast_convolve(y)
        grad = lse_gradient(y, conv_y, beta)
        
        v = momentum * v - lr_cur * grad
        a = project_simplex(a + v)
        
        conv = fast_convolve(a)
        f_cur = conv.max()
        iters += 1
        
        if f_cur < best_f - 1e-12:
            best_f = f_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(a + noise)
            v *= 0.3
            no_improve = 0
    
    return best_a, best_f

    n = len(a0)

    while time.monotonic() - t0 < time_limit:
        def obj_and_grad(x):
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-15:
                return 1e10, np.zeros(n)
            x = x / s
            conv = fast_convolve(x)
            scaled = beta * conv
            scaled -= scaled.max()
            val = conv.max() + np.log(np.exp(scaled).sum()) / beta
            grad = lse_gradient(x, conv, beta)
            return val, grad
        
        # Use L-BFGS-B with bounds
        bounds = [(0, None)] * n
        
        try:
            result = minimize(
                obj_and_grad, a0.copy(),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': max_iter, 'gtol': 1e-8, 'maxls': 50}
            )
        except Exception:
            break
        
        x_opt = np.maximum(result.x, 0)
        s_opt = x_opt.sum()
        if s_opt < 1e-15:
            break
        x_opt = x_opt / s_opt
        
        conv_opt = fast_convolve(x_opt)
        f_opt = conv_opt.max()
        
        if f_opt < best_f - 1e-12:
            best_f = f_opt
            best_a = x_opt.copy()
            a0 = x_opt.copy()
        else:
            # Try with higher beta
            beta *= 1.5
            if beta > 5000:
                break
    
    return best_a, best_f
