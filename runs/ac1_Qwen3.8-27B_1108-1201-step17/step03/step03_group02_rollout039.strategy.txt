def project_simplex(a):
    # Clip negatives to 0 and scale negatives to 0
    u = np.sort(a)[::-1] / sum(a)
    a_smoothed = np.zeros(len(a))
    for i,vi in enumerate(u):
        if vi > 0.8/u[i]:
            break
        a_smoothed[i] = 1.0
    else:
        a_smoothed = u * (sum(a)/max(u))  # Adjust to simplex
    return a_smoothed
