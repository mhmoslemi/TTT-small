Multi-method optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines log-sum-exp gradient descent, active-set subgradient, coordinate descent,
and LP-based direction finding with diverse initializations and adaptive time allocation.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    Gradient of (1/beta) * log(sum(exp(beta * conv))) w.r.t. a.
    This approximates grad of max(conv) as beta -> inf.

    scaled -= scaled.max()

        p[np.argmax(conv)] = 1.0

"""
Multi-method optimization minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothed gradient descent with Nesterov momentum,
adaptive beta scheduling, active-set subgradient refinement,
coordinate descent polish, LP-based direction finding,
and diverse initializations across multiple sequence lengths.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(a, conv, beta):
    """
    Gradient of (1/beta)*log(sum(exp(beta*conv))) w.r.t. a.
    Uses FFT-based cross-correlation for O(n log n) computation.
    """
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum

    # grad[i] = 2 * sum_j a[j] * p[i+j] for j=0..n-1, i+j < 2n-1
    # Use: conv(a, p_rev)[n-1 : 2n-1][::-1] where p_rev = p[::-1]
    fft_size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    Pr = np.fft.rfft(p[::-1], fft_size)
    conv_ap = np.fft.irfft(A * Pr, fft_size)
    g = conv_ap[n - 1:2 * n - 1][::-1]
    return 2.0 * g


def active_set_subgrad(a, conv, K=5):
    """
    Subgradient using top-K peaks with uniform weighting.
    Returns the weighted sum of gradients for the active peaks.
    """
    n = len(a)
    # Find top-K indices
    if K >= len(conv):
        top_k = np.arange(len(conv))
    else:
        top_k = np.argpartition(conv, -K)[-K:]

    # Weight by how close to max (exponential weighting for emphasis)
    vals = conv[top_k]
    weights = np.exp(10.0 * (vals - vals.max()))
    weights /= weights.sum()

    # Compute weighted gradient
    # grad for peak k: grad_k[i] = 2*a[k-i] if 0 <= k-i < n
    # We compute this via FFT: for each k, gradient is 2*shifted_reversed(a)
    # Sum over k with weights: sum_k w_k * 2 * a_shifted(k)

    # Efficient: sum_k w_k * a[k-i] = cross-correlation of w и a
    # Let W = weights placed at positions top_k in a length 2n-1 array
    W_full = np.zeros(2 * n - 1)
    W_full[top_k] = weights

    # We want g[i] = sum_k W_full[k] * a[k-i] for valid k-i
    # = conv(a, W_full[::-1])[n-1 : 2n-1][::-1] (same trick as before)
    fft_size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    Wr = np.fft.rfft(W_full[::-1], fft_size)
    conv_aw = np.fft.irfft(A * Wr, fft_size)
    g = conv_aw[n - 1:2 * n - 1][::-1]

    return 2.0 * g


def coordinate_descent_polish(a0, time_limit, seed=0):
    """
    Coordinate descent: for each coordinate, try small perturbations
    and keep the best. Uses fast conv update.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    order = list(range(n))

    while time.monotonic() - t0 < time_limit:
        np.random.default_rng(int(time.monotonic() * 1e6) % (2**31)).shuffle(order)
        improved_this_pass = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Try different fractions of mass to move
            for frac in [0.02, 0.05, 0.1, 0.2, 0.5]:
                # Move mass from i to a random other position
                j = int(rng.integers(0, n))
                if j == i:
                    continue

                amt = a[i] * frac
                if amt < 1e-10:
                    continue

                # Quick update: only n values of conv change
                a_try = a.copy()
                a_try[i] -= amt
                a_try[j] += amt
                if a_try[i] < -1e-12 or a_try[j] < -1e-12:
                    continue

                # Compute new conv efficiently (only affected lags change)
                # For position i: affects conv[k] for k in [i, i+n-1]
                # For position j: affects conv[k] for k in [j, j+n-1]
                new_conv = conv.copy()

                # Undo old contributions of a[i] and a[j], add new
                # Old a[i] contribution to conv[k]: 2*old_ai*a[k-i] (for valid k-i)
                # Actually, let's just recompute the full conv for simplicity
                # when n is not too large
                new_conv = fast_convolve(a_try)
                new_obj = new_conv.max()

                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_a = a_try.copy()
                    a = a_try.copy()
                    conv = new_conv
                    improved_this_pass = True
                    break

        if not improved_this_pass:
            # Try random pair swaps with smaller amounts
            for _ in range(n // 4):
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                for frac in [0.01, 0.03, 0.08]:
                    amt = a[i] * frac
                    if amt < 1e-12:
                        continue
                    a_try = a.copy()
                    a_try[i] -= amt
                    a_try[j] += amt
                    if a_try[i] < -1e-12:
                        continue
                    new_conv = fast_convolve(a_try)
                    if new_conv.max() < best_obj - 1e-12:
                        best_obj = new_conv.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        conv = new_conv
                        break

    return best_a, best_obj


def lp_direction(a, M, n):
    """
    Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k.
    Returns b normalized to sum=1, or None.
    """
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 20})
    except Exception:
        return None

    if not result.success or result.x is None:
        return None

    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    return b / b_sum


def optimize_lp_phase(a0, time_limit, seed=42):
    """LP-based direction finding to reduce max convolution."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    current_a = a.copy()

    t0 = time.monotonic()
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(current_a)
        M = conv_cur.max()

        b = None
        for M_factor in [1.0, 1.001, 1.005, 1.01, 1.02]:
            b = lp_direction(current_a, M * M_factor, n)
            if b is not None:
                break

        if b is None:
            noise = rng.standard_normal(n) * 0.02
            current_a = project_simplex(current_a + noise)
            stuck += 1
            if stuck > 15:
                break
            continue

        stuck = 0
        improved = False
        for t in [0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03]:
            a_try = (1 - t) * current_a + t * b
            a_try = np.maximum(a_try, 0)
            ss = a_try.sum()
            if ss < 1e-10:
                continue
            a_try /= ss
            conv_try = fast_convolve(a_try)
            obj_try = conv_try.max()
            if obj_try < best_obj - 1e-10:
                best_obj = obj_try
                best_a = a_try.copy()
                current_a = a_try.copy()
                improved = True
                break
            elif obj_try < M * 0.998:
                current_a = a_try.copy()
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                improved = True
                break

        if not improved:
            # Small step
            for t in [0.02, 0.03, 0.05]:
                a_try = (1 - t) * current_a + t * b
                a_try = np.maximum(a_try, 0)
                ss = a_try.sum()
                if ss < 1e-10:
                    continue
                a_try /= ss
                conv_try = fast_convolve(a_try)
                if conv_try.max() < M * 0.9999:
                    current_a = a_try.copy()
                    if conv_try.max() < best_obj:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                    break

    return best_a, best_obj


def optimize_gradient(a0, time_limit, beta_init=30.0, beta_end=300.0,
                      lr_init=0.012, momentum=0.92, seed=0,
                      perturb_every=400, perturb_scale=0.004,
                      use_active_set=False, active_K=5):
    """
    Nesterov accelerated projected gradient on smoothed/active-set objective.
    Input: a0 on simplex. Returns (best_a, best_max_conv).
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys

        conv_y = fast_convolve(y)

        if use_active_set and beta > 50:
            grad = active_set_subgrad(y, conv_y, K=active_K)
        else:
            grad = compute_smoothed_grad(y, conv_y, beta)

        # Momentum update
        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = conv.max()
        iters += 1

        if cur_obj < best_obj - 1e-11:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic perturbation
        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            perturb_every = int(perturb_every * 1.3)
            lr = max(0.0003, lr * 0.6)

        # Adaptive LR
        lr = max(0.0002, lr_init * (1.0 - 0.75 * frac))

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    rc = np.maximum(rc, 1e-8)
    rc /= rc.sum()
    inits['raised_cos'] = rc

    # Blackman window
    bw = np.blackman(n)
    bw = np.maximum(bw, 1e-8)
    bw /= bw.sum()
    inits['blackman'] = bw

    # Hamming
    hm = np.hamming(n)
    hm = np.maximum(hm, 1e-8)
    hm /= hm.sum()
    inits['hamming'] = hm

    # Gaussian (wide)
    g1 = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
    g1 /= g1.sum()
    inits['gauss_wide'] = g1

    # Gaussian (narrow)
    g2 = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    g2 /= g2.sum()
    inits['gauss_narrow'] = g2

    # Bimodal
    bi = np.zeros(n)
    edge = max(2, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Inverted raised cosine (U-shaped)
    inv_rc = 1.0 - 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    inv_rc = np.maximum(inv_rc, 1e-6)
    inv_rc /= inv_rc.sum()
    inits['u_shaped'] = inv_rc

    # Random Dirichlet
    inits['dir_0.3'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['dir_1.0'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['dir_3.0'] = rng.dirichlet(np.ones(n) * 3.0)

    # Frequency-based: inverse of flat spectrum (sinc-like, clipped)
    freq = np.fft.rfftfreq(n)
    target_mag = 1.0 / (1.0 + 5.0 * np.abs(freq) * n)
    phase = rng.uniform(0, 2 * np.pi, len(target_mag))
    target_phase = np.exp(1j * phase)
    spectrum = target_mag * target_phase
    a_freq = np.fft.irfft(spectrum * n, n)
    a_freq = np.maximum(a_freq, 0)
    fs = a_freq.sum()
    if fs > 1e-10:
        inits['freq_flat'] = a_freq / fs

    # Parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_crop'] = pp / ps
        if len(parent) > 0:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps

    return inits
