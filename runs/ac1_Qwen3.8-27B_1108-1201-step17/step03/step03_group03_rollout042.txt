def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_grad_descent(a_init, time_limit, t0_temp=1.0, t_end_temp=100.0, 
                          lr_init=0.01, momentum=0.9, top_k=50, nesterov=True):
    """
    Smoothed gradient descent with log-sum-exp approximation of max.
    Uses Nesterov acceleration and adaptive temperature.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    # Pre-compute FFT size
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    
    v = np.zeros(n)  # momentum
    y = a.copy()     # Nesterov lookahead
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t_start = time.monotonic()
    total_time = time_limit
    lr = lr_init
    it = 0
    
    while time.monotonic() - t_start < total_time:
        it += 1
        # Adaptive temperature: start mild, increase to make it sharper
        progress = min(1.0, (time.monotonic() - t_start) / total_time)
        temp = t0_temp + (t_end_temp - t0_temp) * progress
        
        # Nesterov step
        if nesterov:
            y = project_simplex(a + momentum * (a - best_a + 0.0))  
            # Actually standard Nesterov: y = a + momentum * (a - prev_a)
            # Let me use a simpler momentum approach
            v = momentum * v - lr * 0  # placeholder
            y = project_simplex(a + v)
        
        # Compute gradient at y (or at a for simplicity)
        conv_y = fast_convolve(y)
        
        # Softmax weights for top peaks
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        peak_vals = conv_y[peaks]
        max_peak = peak_vals.max()
        if max_peak < 1e-15:
            break
        
        # Softmax with temperature
        scaled = (peak_vals - max_peak) / temp
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        
        # Gradient: grad_j = 2n * sum_k w_k * a[k-j] (for active peaks k)
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            wi = w[idx]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * y[k - j]
        
        # Update
        v = momentum * v - lr * sg
        a_new = project_simplex(y + v)
        
        conv_new = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv_new.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
        
        a = a_new
        lr *= 0.9995
    
    return best_a, best_obj


def build_lp_pairs(n):
    """Pre-build index pairs for LP constraint matrix."""
    pairs = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            pairs.append((k, j, k - j))
    return pairs


def solve_lp_direction(f, M, pairs, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for k, j, idx in pairs:
        A_ub[k, j] = f[idx]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def optimize_lp(f_init, time_limit, blend_init=0.3, blend_min=0.003,
                blend_halve_every=15, sparsity_blend=0.2):
    """LP-based directional optimization with adaptive blending."""
    a = np.maximum(f_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    pairs = build_lp_pairs(n)
    uniform = np.ones(n) / n
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        
        g = solve_lp_direction(a, M, pairs, n)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # If LP solution is too sparse, blend with uniform
        max_g = g.max()
        if max_g > 0.5:
            g = (1.0 - sparsity_blend) * g + sparsity_blend * uniform
        
        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall >= blend_halve_every:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min:
                break
        
        if stall > blend_halve_every * 4:
            rng_local = np.random.default_rng(hash(tuple(a[:5].__round__(6))) % 2**31)
            a = project_simplex(a + rng_local.standard_normal(n) * 0.008)
            conv = fast_convolve(a)
            stall = 0
    
    return best_a, best_obj


def coordinate_descent_polish(a_in, max_rounds=50, n_steps=3):
    """Coordinate descent: try moving mass between pairs of coordinates."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    for rnd in range(max_rounds):
        improved = False
        for i in range(0, n, max(1, n // 64)):  # Sample coordinates for speed
            if a[i] < 1e-8:
                continue
            # Find a few candidate targets
            for _ in range(n_steps):
                j = np.random.randint(n)
                while j == i:
                    j = np.random.randint(n)
                delta = a[i] * 0.05
                if delta < 1e-9:
                    continue
                a_test = a.copy()
                a_test[i] -= delta
                a_test[j] += delta
                a_test = np.maximum(a_test, 0.0)
                st = a_test.sum()
                if st < 1e-15:
                    continue
                a_test /= st
                conv_test = fast_convolve(a_test)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-13:
                    a = a_test
                    best_obj = obj_test
                    conv = conv_test
                    improved = True
        if not improved:
            break
    
    return a


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Triangle
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_w = np.maximum(cos_w, 0.001)
    inits['cosine'] = cos_w / cos_w.sum()
    
    # Sine (zero at edges)
    sine = np.sin(np.pi * (x + 0.5) / n)
    sine = np.maximum(sine, 0.001)
    inits['sine'] = sine / sine.sum()
    
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent'] = p / ps
        elif len(parent) > n:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                          np.maximum(parent, 0))
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_ds'] = p / ps
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                          np.maximum(parent, 0))
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_up'] = p / ps
    
    return inits

c_2 = a1*a1 = a1^2. dc_2/da_1 = 2*a1. Formula: 2*a_{2-1} = 2*a1. ✓

So the gradient of `c_k` with respect to `a_j` is `2*a_{k-j}` (where a_m = 0 for m outside [0,n-1]).

Therefore, the gradient of `f(a) = 2n * max_k c_k` (at a point where the max is unique at k*) is:
`df/da_j = 2n * 2 * a_{k*-j} = 4n * a_{k*-j}`

For the smoothed version with temperature t:

`df_t/da_j = (2n/t) * sum_m (t*exp(t*c_m)/sum exp(t*c)) * 2*a_{m-j}`
`= 4n * sum_m w_m * a_{m-j}`

So the gradient is `4n * (w ⋆ a)[j]` = `4n * IFFT(FFT(w_padded) * conj(FFT(a_padded)))`.

import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(c, a, temp, fft_size):
    """Compute gradient of smoothed objective using FFT-based cross-correlation."""
    n = len(a)
    # softmax weights on the convolution peaks
    c_max = c.max()
    scaled = (c - c_max) / temp
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # grad_j = 4n * sum_k w_k * a[k-j] = 4n * cross_correlate(w, a)[j]
    # cross_correlate(w, a)[j] = sum_k w[k] * a[k-j]
    # = IFFT(FFT(w) * conj(FFT(a)))
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    grad_full = np.fft.irfft(W * np.conj(A), fft_size)
    grad = 4.0 * n * grad_full[:n]
    return grad


def smoothed_grad_descent(a_init, time_limit, temp_init=0.5, temp_end=50.0,
                          lr_init=0.005, momentum=0.9, nesterov=True):
    """
    Smoothed gradient descent with log-sum-exp, adaptive step size, and Nesterov momentum.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    
    v = np.zeros(n)
    y = a.copy()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t_start = time.monotonic()
    lr = lr_init
    it = 0
    no_improve = 0
    
    while time.monotonic() - t_start < time_limit:
        it += 1
        elapsed = time.monotonic() - t_start
        progress = min(1.0, elapsed / time_limit)
        temp = temp_init + (temp_end - temp_init) * (progress ** 0.7)
        
        # Nesterov lookahead
        if nesterov and it > 1:
            y = project_simplex(a + momentum * v)
        else:
            y = a.copy()
        
        # Compute gradient at y
        conv_y = fast_convolve(y)
        max_peak = conv_y.max()
        if max_peak < 1e-15:
            break
        
        grad = compute_gradient(conv_y, y, temp, fft_size)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Adaptive step size
        if it <= 1:
            eta = lr
        else:
            eta = lr
        
        # Gradient step with projection
        v = momentum * v
        a_new = project_simplex(y - eta * grad)
        
        conv_new = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv_new.max()
        
        # Adaptive learning rate
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
            lr = min(lr * 1.05, 0.05)
        else:
            no_improve += 1
            if no_improve > 50:
                lr = lr * 0.7
                no_improve = 0
                if lr < 1e-5:
                    break
        
        a = a_new
        lr *= 0.9998
    
    # Final comparison
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    if cur_obj < best_obj:
        best_obj = cur_obj
        best_a = a.copy()
    
    return best_a, best_obj

def optimize_lp(f_init, time_limit, blend_init=0.2, blend_min=0.002,
                blend_halve_every=12, sparsity_blend=0.15, constraint_frac=0.3):
    """
    LP-based directional optimization with adaptive blending and constraint reduction.
    """
    a = np.maximum(f_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    
    # Pre-build full constraint matrix for smaller n, or use reduced version
    if n <= 1536:
        A_ub_full = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                A_ub_full[k, j] = a[k - j]
        use_full = True
    else:
        use_full = False
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        
        if use_full:
            A_ub = A_ub_full
            b_ub = np.full(2 * n - 1, M)
        else:
            # Use only constraints where c_k > (1-constraint_frac) * M
            threshold = M * (1.0 - constraint_frac)
            active_k = np.where(conv >= threshold)[0]
            A_ub = np.zeros((len(active_k), n))
            for idx, k in enumerate(active_k):
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                for j in range(j_start, j_end + 1):
                    A_ub[idx, j] = a[k - j]
            b_ub = np.full(len(active_k), M)
        
        c_obj = -np.ones(n)
        bounds = [(0, None)] * n
        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                             method='highs', options={'presolve': True})
            if result.success:
                g = np.maximum(result.x, 0.0)
            else:
                break
        except Exception:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        max_g = g.max()
        if max_g > 0.4:
            uniform = np.ones(n) / n
            g = (1.0 - sparsity_blend) * g + sparsity_blend * uniform
        
        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall >= blend_halve_every:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min:
                break
        
        if stall > blend_halve_every * 5:
            rng_local = np.random.default_rng(12345 + int(stall))
            a = project_simplex(a + rng_local.standard_normal(n) * 0.005)
            conv = fast_convolve(a)
            stall = 0
        
        # Rebuild constraint matrix periodically (since a changed)
        if use_full and stall == 0:
            for k in range(2 * n - 1):
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                for j in range(j_start, j_end + 1):
                    A_ub_full[k, j] = a[k - j]
    
    return best_a, best_obj


def coordinate_descent_polish(a_in, max_rounds=40, n_samples=8):
    """Coordinate descent: try moving mass between sampled pairs of coordinates."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    rng = np.random.default_rng(99)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    for rnd in range(max_rounds):
        improved = False
        for _ in range(n * n_samples):
            i = rng.integers(n)
            if a[i] < 1e-9:
                continue
            j = rng.integers(n)
            if j == i:
                continue
            for frac in [0.03, 0.08, 0.15]:
                delta = a[i] * frac
                if delta < 1e-10:
                    continue
                a_test = a.copy()
                a_test[i] -= delta
                a_test[j] += delta
                a_test = np.maximum(a_test, 0.0)
                st = a_test.sum()
                if st < 1e-15:
                    continue
                a_test /= st
                conv_test = fast_convolve(a_test)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-13:
                    a = a_test
                    best_obj = obj_test
                    conv = conv_test
                    improved = True
        if not improved:
            break
    
    return a


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations including asymmetric shapes."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Ramp (asymmetric - reduces middle peak)
    ramp = x + 1.0
    inits['ramp'] = ramp / ramp.sum()
    
    # Reverse ramp
    rramp = n - x
    inits['rramp'] = rramp / rramp.sum()
    
    # Sigmoid (asymmetric smooth transition)
    for alpha in [2.0, 4.0, 6.0]:
        sig = 1.0 / (1.0 + np.exp(-alpha * (x / n - 0.5)))
        sig = np.maximum(sig, 1e-6)
        inits[f'sig{alpha}'] = sig / sig.sum()
    
    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.2 * n) / (0.08 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.8 * n) / (0.08 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()
    
    # Cosine window (smooth, zero at edges)
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) if n > 1 else np.array([1.0])
    cos_w = np.maximum(cos_w, 0.001)
    inits['cosine'] = cos_w / cos_w.sum()
    
    # Sine (zero at edges)
    sine = np.sin(np.pi * (x + 0.5) / n)
    sine = np.maximum(sine, 0.001)
    inits['sine'] = sine / sine.sum()
    
    # Triangle
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Power law
    pow_l = (x + 1.0) ** 2
    inits['pow2'] = pow_l / pow_l.sum()
    
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent'] = p / ps
        elif len(parent) > n:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                          np.maximum(parent, 0))
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_ds'] = p / ps
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                          np.maximum(parent, 0))
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_up'] = p / ps
    
    return inits

                a_g, _ = smoothed_grad_descent(best_sequence, t_g,
                                               temp_init=1.0, temp_end=60.0,
                                               lr_init=0.002, momentum=0.92,
                                               nesterov=True)
                score_g = float(evaluate_sequence(a_g.tolist()))
                if np.isfinite(score_g) and score_g < best_score:
                    best_score = score_g
                    best_sequence = a_g.copy()
                    print(f"  Cycle {cycle} Grad: {score_g:.6f}")
            
            # LP refinement (only if n is manageable)
            t_l = t_cyc * 0.5
            if t_l > 8 and n_b <= 2048:
                a_l, _ = optimize_lp(best_sequence, t_l,
                                     blend_init=0.06, blend_min=0.001,
                                     blend_halve_every=15, constraint_frac=0.15)
                score_l = float(evaluate_sequence(a_l.tolist()))
                if np.isfinite(score_l) and score_l < best_score:
                    best_score = score_l
                    best_sequence = a_l.copy()
                    print(f"  Cycle {cycle} LP: {score_l:.6f}")
            
            # Remaining time for gradient
            t_rem = deadline - time.monotonic() - 90
            if t_rem > 5:
                a_r, _ = smoothed_grad_descent(best_sequence, min(t_rem * 0.5, 30.0),
                                               temp_init=5.0, temp_end=100.0,
                                               lr_init=0.001, momentum=0.93,
                                               nesterov=True)
                score_r = float(evaluate_sequence(a_r.tolist()))
                if np.isfinite(score_r) and score_r < best_score:
                    best_score = score_r
                    best_sequence = a_r.copy()
                    print(f"  Cycle {cycle} Refine: {score_r:.6f}")
    
    # Phase 4: Final polish
    print("=== Phase 4: Final Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 15:
        n_b = len(best_sequence)
        
        # Coordinate descent
        if time.monotonic() < deadline - 20:
            t_cd = min(60.0, deadline - time.monotonic() - 15)
            a_cd = coordinate_descent_polish(best_sequence, max_rounds=30, n_samples=6)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  CD: {score_cd:.6f}")
        
        # Small perturbations + gradient
        if time.monotonic() < deadline - 10:
            for ps in [0.001, 0.003, 0.008]:
                if time.monotonic() > deadline - 8:
                    break
                for _ in range(2):
                    if time.monotonic() > deadline - 6:
                        break
                    noise = rng.standard_normal(n_b) * ps
                    a_p = project_simplex(best_sequence + noise)
                    t_p = min(3.0, deadline - time.monotonic() - 5)
                    if t_p > 1:
                        a_opt, _ = smoothed_grad_descent(a_p, t_p,
                                                         temp_init=5.0, temp_end=80.0,
                                                         lr_init=0.001, momentum=0.9,
                                                         nesterov=True)
                        score = float(evaluate_sequence(a_opt.tolist()))
                        if np.isfinite(score) and score < best_score:
                            best_score = score
                            best_sequence = a_opt.copy()
                            print(f"  Perturb({ps}): {score:.6f}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
    
    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
