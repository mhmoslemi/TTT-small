```python
def lp_step(f, M, n):
    """Solve LP: max sum(b) s.t. b >= 0, (f*b)[k] <= M for all k."""
    # f * b is linear in b (f is fixed)
    # (f*b)[k] = sum_j f[j] * b[k-j] (valid indices)
    # For each k, this is a linear function of b.
    
    # Variable: b[0], ..., b[n-1]
    # Objective: maximize sum(b) -> minimize -sum(b)
    # Constraints: (f*b)[k] <= M for k = 0, ..., 2n-2
    #              b[i] >= 0
    
    # Build constraint matrix A_ub such that A_ub @ b <= M
    # (f*b)[k] = sum_j f[j] * b[k-j] where 0 <= j < n and 0 <= k-j < n
    # So for fixed k, the coefficient of b[i] is f[k-i] if 0 <= k-i < n, else 0.
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] = f[j]
    
    c = -np.ones(n)  # minimize -sum(b)
    b_ub = np.full(2*n-1, M)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if result.success:
        return result.x
    else:
        return f.copy()  # fallback
```

```python
import time
from scipy.optimize import minimize, linprog
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    w = (u > (cssv - 1.0) / np.arange(1, n + 1))

    rho = np.argmax(~w)
    theta = (cssv[rho] - 1.0) / (rho + 1)

    x_max = x.max()
    shifted = beta * (x - x_max)

        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_s / total
    sm = x_max + np.log(total) / beta

    w[argmax_set] = 1.0 / len(argmax_set)
    grad = conv_grad_smooth(a, w)
    return grad, max_val, argmax_set


def solve_lp_max_sum(f, M, n):
    """Solve LP: max sum(b) s.t. (f*b)[k] <= M, b >= 0.
    f is the fixed sequence, M is the peak bound."""
    # (f*b)[k] = sum_j f[j]*b[k-j] (valid indices)
    # Build constraint matrix: A_ub[k, i] = f[k-i] if 0 <= k-i < n
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            A_ub[k, i_min:i_max + 1] = f[k - i_min:k - i_max + 1][::-1]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                         bounds=[(0, None)] * n, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return f.copy()


def solve_lp_min_peak(f, n):
    """Solve LP: min M s.t. (f*b)[k] <= M, sum(b)=1, b >= 0."""
    # Variables: b[0],...,b[n-1], M
    # Objective: minimize M -> c = [0,...,0,1]
    # Constraints: (f*b)[k] - M <= 0 for each k
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            A_ub[k, i_min:i_max + 1] = f[k - i_min:k - i_max + 1][::-1]
        A_ub[k, n] = -1.0
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize M
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=np.zeros(2 * n - 1),
                         A_eq=A_eq, b_eq=b_eq,
                         bounds=[(0, None)] * n + [(0, None)],
                         method='highs')
        if result.success:
            return result.x[:n], result.x[n]
    except Exception:
        pass
    return f.copy(), fast_convolve(f).max()


def lp_iteration(a, t=0.1):
    """One step of the LP-based iteration from the paper.
    Returns updated sequence."""
    n = len(a)
    c = fast_convolve(a)
    M = c.max()
    
    # Solve LP: max sum(b) s.t. f*b peak <= M, b >= 0
    b = solve_lp_max_sum(a, M, n)
    sum_b = b.sum()
    if sum_b < 1e-10:
        return a.copy()
    
    # Rescale to match sum(a)
    g = b * (a.sum() / sum_b)
    
    # Convex combination
    new_a = (1 - t) * a + t * g
    new_a = np.maximum(new_a, 0)
    s = new_a.sum()
    if s > 1e-15:
        new_a /= s
    else:
        new_a = a.copy()
    return new_a


def lbfgsb_smooth(a0, beta, maxiter=200):
    """L-BFGS-B optimization with log-sum-exp smoothing."""
    n = len(a0)
    
    def obj(x):
        x_shifted = x - x.max()
        a = np.exp(x_shifted)
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, _ = smooth_max_and_weights(conv, beta)
        return 2.0 * n * sm
    
    def jac(x):
        x_shifted = x - x.max()
        a = np.exp(x_shifted)
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        grad_a = conv_grad_smooth(a, w)
        lagrange = np.dot(a, grad_a)
        return a * (grad_a - lagrange)
    
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)
    
    try:
        result = minimize(obj, x0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'maxls': 40, 'gtol': 1e-8})
        x_opt = result.x
        a_opt = np.exp(x_opt - x_opt.max())
        a_opt = a_opt / a_opt.sum()
        return a_opt
    except Exception:
        return a0.copy()


def subgradient_descent(a0, n_steps=1000, lr=0.01):
    """Projected subgradient descent on max(conv(a,a))."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    for step in range(n_steps):
        grad, max_val, _ = subgradient_max_conv(a)
        obj = 2.0 * n * max_val
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Adaptive learning rate
        alpha = lr / (1.0 + 0.01 * step)
        a_new = project_simplex(a - alpha * grad)
        a = a_new
    
    return best_a


def flat_spectrum_init(n, rng):
    """Initialize with approximately flat power spectrum."""
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = 1.0
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    for k in range(1, fft_size // 2 + 1):
        A[k] = np.exp(1j * phases[k])
    for k in range(1, min(n, fft_size // 2)):
        A[fft_size - k] = np.exp(-1j * phases[k])
    a = np.real(ifft(A, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    inits['uniform'] = np.ones(n) / n
    
    # From parent
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = pp / ps
            else:
                pp2 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), pp)
                pp2 = np.maximum(pp2, 0)
                ps2 = pp2.sum()
                if ps2 > 1e-10:
                    inits['parent_interp'] = pp2 / ps2
    
    inits['flat_spec'] = flat_spectrum_init(n, rng)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 3)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Gaussian center
    x = np.arange(n, dtype=np.float64)
    for sf in [0.2, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%s' % str(sf)] = g
    
    return inits


def score_sequence(a):
    """Compute the exact objective locally."""
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    c = fast_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search function combining LP, subgradient, and L-BFGS-B methods."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent: length=%d, local_score=%.6f" % (len(p), score_sequence(p)))
    except Exception:
        pass
    
    def update_best(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return
        score = score_sequence(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            elapsed = time.monotonic() - t_start
            print("  NEW BEST: score=%.6f, n=%d (t=%.0fs)" % (score, len(a), elapsed))
    
    # ============ PHASE 1: Quick exploration ============
    print("=== Phase 1: Exploration ===")
    p1_deadline = t_start + 300.0
    
    # Try several lengths with the best initializations
    if parent is not None:
        parent_n = len(parent)
        lengths = sorted(set([l for l in [parent_n // 2, parent_n, parent_n * 2,
                                           128, 192, 256, 384, 512]
                              if 64 <= l <= 4096]))
    else:
        lengths = [128, 256, 384, 512]
    
    for n in lengths:
        if time.monotonic() > p1_deadline - 20:
            break
        
        inits = generate_inits(n, parent, rng)
        init_keys = list(inits.keys())
        # Priority
        priority = ['parent', 'parent_interp', 'flat_spec', 'uniform', 'gauss_0.35', 'random']
        init_keys = [k for k in priority if k in inits] + \
                    [k for k in init_keys if k not in priority]
        
        for name in init_keys[:4]:
            if time.monotonic() > p1_deadline - 10:
                break
            a0 = inits[name]
            t_alloc = min(40.0, p1_deadline - time.monotonic() - 5)
            if t_alloc < 5:
                break
            
            # Quick LP iterations
            a_cur = a0.copy()
            n_lp_steps = int(min(200, t_alloc * 30))
            for i in range(n_lp_steps):
                if time.monotonic() > p1_deadline - 5:
                    break
                a_cur = lp_iteration(a_cur, t=0.15)
            update_best(a_cur)
            
            # Quick subgradient
            if time.monotonic() < p1_deadline - 10:
                a_sg = subgradient_descent(a_cur, n_steps=int(min(200, t_alloc * 20)), lr=0.02)
                update_best(a_sg)
    
    print("Phase 1 done. Best: %.6f" % best_score)
    
    # ============ PHASE 2: Deep optimization ============
    print("=== Phase 2: Deep Optimization ===")
    p2_deadline = deadline - 200.0
    
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = score_sequence(best_sequence)
    
    # Work primarily on the best length, but occasionally try others
    best_n = len(best_sequence)
    alt_lengths = [l for l in [best_n * 2, best_n // 2, best_n + 64, best_n - 64]
                   if 64 <= l <= 4096 and l != best_n]
    
    # Main optimization loop on best sequence
    a = best_sequence / best_sequence.sum()
    n = best_n
    method_cycle = 0
    plateau_count = 0
    current_obj = score_sequence(a)
    
    while time.monotonic() < p2_deadline:
        elapsed = time.monotonic() - t_start
        remaining = p2_deadline - time.monotonic()
        if remaining < 5:
            break
        
        method_cycle += 1
        method = method_cycle % 10
        
        if method < 4:
            # LP iteration (4 out of 10 cycles)
            n_steps = int(min(200, remaining * 5))
            t_param = 0.12
            for _ in range(n_steps):
                if time.monotonic() > p2_deadline - 2:
                    break
                a = lp_iteration(a, t=t_param)
            new_obj = score_sequence(a)
            if new_obj < current_obj - 1e-9:
                plateau_count = 0
                current_obj = new_obj
            else:
                plateau_count += 1
            
        elif method < 7:
            # Subgradient descent (3 out of 10)
            n_steps = int(min(500, remaining * 30))
            lr = max(0.001, 0.02 / (1 + plateau_count * 0.5))
            a = subgradient_descent(a, n_steps=n_steps, lr=lr)
            new_obj = score_sequence(a)
            if new_obj < current_obj - 1e-9:
                plateau_count = 0
                current_obj = new_obj
            else:
                plateau_count += 1
                
        else:
            # L-BFGS-B polish (3 out of 10)
            beta = max(5.0, 50.0 / (1 + plateau_count))
            maxiter = int(min(150, remaining * 10))
            if maxiter > 10:
                a = lbfgsb_smooth(a, beta=beta, maxiter=maxiter)
                new_obj = score_sequence(a)
                if new_obj < current_obj - 1e-9:
                    plateau_count = 0
                    current_obj = new_obj
                else:
                    plateau_count += 1
        
        update_best(a)
        
        # If plateaued for a while, try a perturbation
        if plateau_count > 50 and method_cycle % 10 == 0:
            scale = 0.01 + 0.02 * rng.random()
            noise = rng.standard_normal(n) * scale
            a_pert = project_simplex(a + noise)
            new_obj = score_sequence(a_pert)
            if new_obj < current_obj:
                a = a_pert
                current_obj = new_obj
                plateau_count = 0
            else:
                # Try LP from perturbed
                a_pert2 = lp_iteration(a_pert, t=0.2)
                new_obj2 = score_sequence(a_pert2)
                if new_obj2 < current_obj:
                    a = a_pert2
                    current_obj = new_obj2
                    plateau_count = 0
        
        # Periodically try an alternative length
        if method_cycle % 50 == 0 and alt_lengths and remaining > 60:
            alt_n = alt_lengths[(method_cycle // 50) % len(alt_lengths)]
            if alt_n != n:
                # Interpolate best to alt length
                a_alt = np.interp(np.linspace(0, 1, alt_n),
                                  np.linspace(0, 1, n), a)
                a_alt = np.maximum(a_alt, 0)
                a_alt = a_alt / a_alt.sum()
                # Quick optimization
                for _ in range(50):
                    a_alt = lp_iteration(a_alt, t=0.15)
                a_alt = subgradient_descent(a_alt, n_steps=100, lr=0.01)
                alt_score = score_sequence(a_alt)
                if alt_score < best_score:
                    best_score = alt_score
                    best_sequence = a_alt.copy()
                    print("  Alt length %d improved: %.6f" % (alt_n, alt_score))
                elif alt_score < best_score + 0.05:
                    # Keep it as a secondary best for later
                    pass
    
    update_best(a)
    print("Phase 2 done. Best: %.6f" % best_score)
    
    # ============ PHASE 3: Final polishing ============
    print("=== Phase 3: Polishing ===")
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
    
    a = best_sequence / best_sequence.sum()
    n = len(a)
    
    # Multi-beta L-BFGS-B refinement
    for beta in [30.0, 15.0, 8.0, 4.0, 2.0, 1.0]:
        if time.monotonic() > deadline - 5:
            break
        remaining = deadline - time.monotonic() - 3
        if remaining < 5:
            break
        maxiter = int(min(300, remaining * 15))
        if maxiter > 20:
            a = lbfgsb_smooth(a, beta=beta, maxiter=maxiter)
            update_best(a)
    
    # Alternating LP + subgradient with decreasing step
    for i in range(20):
        if time.monotonic() > deadline - 5:
            break
        a = lp_iteration(a, t=0.05)
        a = subgradient_descent(a, n_steps=50, lr=0.005 / (1 + i * 0.1))
        update_best(a)
    
    # Final random perturbations
    for _ in range(10):
        if time.monotonic() > deadline - 10:
            break
        scale = 0.003 * rng.random()
        noise = rng.standard_normal(n) * scale
        a_pert = project_simplex(a + noise)
        a_pert = lbfgsb_smooth(a_pert, beta=20.0, maxiter=50)
        update_best(a_pert)
    
    # Verify with evaluate_sequence
    if best_sequence is not None:
        final_score = float(evaluate_sequence(best_sequence.tolist()))
        if np.isfinite(final_score):
            print("Final verified score: %.6f (n=%d)" % (final_score, len(best_sequence)))
        else:
            # Fallback
            best_sequence = np.ones(128) / 128
            print("Fallback to uniform 128. Score: %.6f" % evaluate_sequence(best_sequence.tolist()))
    else:
        best_sequence = np.ones(128) / 128
    
    print("FINAL: score=%.6f, n=%d" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
```
