from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights with numerical stability."""
    x_max = x.max()

    return sm, w


def compute_grad_smooth_conv(a, w):
    """Gradient of weighted sum of conv_lags w.r.t. a."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad

    return a_opt, result.fun


def optimize_pg_smooth(a0, beta, lr=0.05, n_iter=500):
    """Projected gradient descent on simplex with smooth max objective."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_obj = float('inf')
    best_a = a.copy()
    for it in range(n_iter):
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        if obj < 1e-10:
            break
        grad_a = compute_grad_smooth_conv(a, w)
        norm_grad = np.linalg.norm(grad_a)
        if norm_grad < 1e-15:
            break
        t = lr
        a_new = a
        for _ in range(12):
            a_new = project_simplex(a - t * grad_a)
            conv_new = fast_convolve(a_new)
            sm_new, _ = smooth_max_and_weights(conv_new, beta)
            obj_new = 2.0 * n * sm_new
            if obj_new <= obj + 1e-10:
                break
            t *= 0.5
        a = a_new
        lr *= 0.999
    return best_a, best_obj


def flat_spectrum_init(n, rng):
    """Initialize with approximately flat power spectrum, non-negative."""
    fft_size = 1 << (n - 1).bit_length()
    best_a = np.ones(n) / n
    best_obj = float('inf')
    for trial in range(30):
        A = np.zeros(fft_size, dtype=complex)
        A[0] = np.sqrt(n)
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        for k in range(1, fft_size // 2 + 1):
            A[k] = np.exp(1j * phases[k])
            if k < fft_size - k:
                A[fft_size - k] = np.exp(-1j * phases[k])
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
            conv = fast_convolve(a)
            obj = 2.0 * n * conv.max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    return best_a


def golden_mean_init(n, rng):
    """Initialize using a golden-angle-based pattern for good spectral properties."""
    best_a = np.ones(n) / n
    best_obj = float('inf')
    for trial in range(15):
        fft_size = 1 << (n - 1).bit_length()
        A = np.zeros(fft_size, dtype=complex)
        A[0] = np.sqrt(n)
        for k in range(1, fft_size):
            if k < fft_size - k:
                phase = rng.uniform(0, 2 * np.pi)
                A[k] = np.exp(1j * phase)
                A[fft_size - k] = np.exp(-1j * phase)
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
            conv = fast_convolve(a)
            obj = 2.0 * n * conv.max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    return best_a


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uniform'] = np.ones(n) / n

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()

    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    for sf in [0.1, 0.2, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%s' % str(sf)] = g

    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['flat_spec'] = flat_spectrum_init(n, rng)
    inits['golden'] = golden_mean_init(n, rng)

    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0)
        ps = pp.sum()
        if ps > 1e-10:
            pp = pp / ps
            if len(pp) == n:
                inits['parent'] = pp
            else:
                inits['parent_re'] = np.interp(
                    np.linspace(0, 1, n), np.linspace(0, 1, len(pp)), pp)
    return inits


def optimize_betaschedule(a0, betas, maxiter_per_beta=200):
    """Run L-BFGS-B with a schedule of decreasing beta values."""
    n = len(a0)
    a_cur = a0.copy()
    best_a = a0.copy()
    best_obj = 2.0 * n * fast_convolve(a0).max()
    for beta in betas:
        a_opt, obj = optimize_lbfgsb(a_cur, beta=beta, maxiter=maxiter_per_beta)
        true_obj = 2.0 * n * fast_convolve(a_opt).max()
        if true_obj < best_obj:
            best_obj = true_obj
            best_a = a_opt.copy()
        a_cur = a_opt
    return best_a, best_obj


def polish_low_beta(a0, time_budget=60.0):
    """Final polish with very low beta values."""
    n = len(a0)
    a_cur = a0.copy()
    best_a = a0.copy()
    best_obj = 2.0 * n * fast_convolve(a0).max()
    start = time.monotonic()
    for beta in [8.0, 5.0, 3.0, 2.0, 1.5, 1.0]:
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 5:
            break
        remaining = time_budget - elapsed - 3
        if remaining < 3:
            break
        maxiter = int(min(300, max(50, remaining * 25)))
        a_opt, _ = optimize_lbfgsb(a_cur, beta=beta, maxiter=maxiter)
        true_obj = 2.0 * n * fast_convolve(a_opt).max()
        if true_obj < best_obj:
            best_obj = true_obj
            best_a = a_opt.copy()
        a_cur = a_opt
    return best_a, best_obj


def propose_candidate(budget_s=1800, **kwargs):
    """
    Multi-phase search: quick PG scan, deep L-BFGS-B with beta schedule,
    and low-beta polishing. Uses parent sequence and frequency-domain inits.
    """
    rng = np.random.default_rng(12345)
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)

    best_sequence = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent: len=%d, sum=%.2f" % (len(p), p.sum()))
    except Exception:
        pass

    def score_and_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return float('inf')
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: %.8f, n=%d (t=%.0fs)" % (score, len(a), time.monotonic() - t_start))
        return score if np.isfinite(score) else float('inf')

    # Phase 1: Quick scan (200s)
    print("=== Phase 1: Quick Scan ===")
    lengths = [128, 192, 256, 320, 384, 512]
    phase1_end = t_start + 200.0
    scan_results = []

    for n in lengths:
        if time.monotonic() > phase1_end - 20:
            break
        inits = generate_initializations(n, parent, rng)
        priority = ['parent', 'parent_re', 'flat_spec', 'golden', 'uniform', 'gauss_0.2', 'random']
        names = [nm for nm in priority if nm in inits]
        names = names[:4]

        for name in names:
            if time.monotonic() > phase1_end - 10:
                break
            a0 = inits[name]
            a_opt, _ = optimize_pg_smooth(a0, beta=60.0, lr=0.1, n_iter=200)
            obj = 2.0 * n * fast_convolve(a_opt).max()
            scan_results.append((obj, n, name, a_opt.copy()))

    scan_results.sort(key=lambda x: x[0])
    print("Phase 1 done. Best obj=%.6f" % (scan_results[0][0] if scan_results else 999))

    # Phase 2: Deep L-BFGS-B (1300s)
    print("=== Phase 2: Deep L-BFGS-B ===")
    phase2_end = t_start + 1550.0

    # Determine target lengths
    target_lengths = [256, 320, 192]
    if parent is not None:
        pn = len(parent)
        if pn not in target_lengths:
            target_lengths.insert(0, pn)

    candidates = []
    for obj, n, name, a in scan_results[:6]:
        candidates.append((n, a.copy()))

    if parent is not None:
        pn = len(parent)
        p_norm = np.maximum(parent, 0)
        p_norm = p_norm / p_norm.sum()
        candidates.append((pn, p_norm))

    # Ensure we have candidates for key lengths
    for n in [256, 384]:
        if not any(c[0] == n for c in candidates):
            inits = generate_initializations(n, parent, rng)
            a0 = inits.get('flat_spec', inits['uniform'])
            candidates.append((n, a0))

    seen_lengths = set()
    for n, a0 in candidates:
        if time.monotonic() > phase2_end - 60:
            break
        if n in seen_lengths:
            continue
        seen_lengths.add(n)

        remaining = phase2_end - time.monotonic() - 30
        if remaining < 20:
            break
        time_per_candidate = min(300.0, remaining / max(1, len(candidates) - list(seen_lengths).index(n) if n in list(seen_lengths) else 1))
        time_per_candidate = min(350.0, remaining / 2)

        print("  Optimizing n=%d (t=%.0fs remaining)" % (n, remaining))
        betas = [100.0, 50.0, 25.0, 12.0, 6.0]
        a_opt, _ = optimize_betaschedule(a0, betas, maxiter_per_beta=200)
        score_and_update(a_opt)

        # Also try from a different init at this length
        if time.monotonic() < phase2_end - 100:
            inits = generate_initializations(n, parent, rng)
            init_keys = [k for k in inits.keys() if k not in ['parent', 'parent_re']]
            if 'golden' in init_keys:
                a2 = inits['golden']
                betas2 = [80.0, 40.0, 20.0, 10.0]
                a_opt2, _ = optimize_betaschedule(a2, betas2, maxiter_per_beta=150)
                score_and_update(a_opt2)

    print("Phase 2 done. Best: %.8f" % best_score)

    # Phase 3: Polish (remaining time)
    print("=== Phase 3: Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 30:
            print("  Polishing with low beta (t=%.0fs)" % remaining)
            a_polished, _ = polish_low_beta(best_sequence / best_sequence.sum(), time_budget=remaining * 0.5)
            score_and_update(a_polished)

        remaining = deadline - time.monotonic() - 10.0
        if remaining > 30:
            # Random perturbations + re-optimize
            n_best = len(best_sequence)
            for attempt in range(4):
                if time.monotonic() > deadline - 20:
                    break
                scale = 0.005 + 0.003 * rng.uniform()
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(best_sequence / best_sequence.sum() + noise)
                t_alloc = min(60.0, (deadline - time.monotonic() - 15) / 2)
                if t_alloc < 10:
                    break
                betas_p = [20.0, 10.0, 5.0]
                a_opt, _ = optimize_betaschedule(a_pert, betas_p, maxiter_per_beta=100)
                score_and_update(a_opt)

        remaining = deadline - time.monotonic() - 10.0
        if remaining > 20:
            # Final ultra-low beta polish
            a_final, _ = polish_low_beta(best_sequence / best_sequence.sum(), time_budget=remaining * 0.6)
            score_and_update(a_final)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
        best_sequence = np.maximum(best_sequence, 0.0)

    print("\nFINAL: score=%.8f, n=%d" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
