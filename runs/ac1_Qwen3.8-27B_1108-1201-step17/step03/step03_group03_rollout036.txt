"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed (log-sum-exp) 
projected gradient descent with temperature annealing, combined with 
LP-based directional refinement and coordinate descent polish.
The key innovation is the use of a smooth approximation of the max 
function with adaptive temperature, providing better gradient directions
than hard subgradients and enabling escape from local minima.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_pairs(n):
    """Pre-build index pairs for LP constraint matrix."""
    pairs = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            pairs.append((k, j, k - j))
    return pairs


def solve_lp_direction(f, M, pairs, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for k, j, idx in pairs:
        A_ub[k, j] = f[idx]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                   'time_limit': 10})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def compute_smoothed_grad(a, w):
    """Compute gradient of LSE(tau * conv(a,a)) wrt a.
    grad[j] = sum_k w[k] * a[k-j]
    where w = softmax(tau * conv)
    """
    n = len(a)
    size = 1
    while size < 2 * n:
        size *= 2
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    R = np.fft.irfft(W * np.conj(A), size)
    return R[:n]


def smoothed_gradient_descent(a0, time_limit, tau_init=5.0, tau_end=500.0,
                                lr_init=0.02, momentum=0.85, 
                                anneal_rate=1.05, batch_topk=0):
    """Projected gradient descent on LSE-smoothed objective with annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    tau = tau_init
    lr = lr_init
    v = np.zeros(n)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute softmax weights
        conv_max = conv.max()
        if conv_max < 1e-15:
            break
        
        # Temperature-scaled softmax
        scaled = tau * conv
        scaled -= scaled.max()  # numerical stability
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient
        grad = compute_smoothed_grad(a, w)
        
        # Momentum update
        v = momentum * v - lr * grad
        a = project_simplex(a + v)
        
        # Recompute convolution
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Anneal temperature
        tau = min(tau * anneal_rate, tau_end)
        
        # Decay learning rate
        lr = max(0.001, lr * 0.999)
        
        # Restart if stuck
        if no_improve > 300:
            noise = np.random.default_rng(hash(tuple(np.round(a[:8], 4))) % (2**31)).standard_normal(n) * 0.01
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
        
        # If tau is very high, switch to steeper annealing
        if tau > tau_end * 0.5:
            lr = max(0.0005, lr * 0.99)
    
    return best_a, best_obj


def optimize_lp(f_init, time_limit, blend_init=0.4, blend_min=0.005,
                blend_halve_every=12):
    """LP-based directional optimization with adaptive blending."""
    a = np.maximum(f_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    pairs = build_lp_pairs(n)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init
    stall = 0

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break

        g = solve_lp_direction(a, M, pairs, n)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall >= blend_halve_every:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min:
                break

        if stall > blend_halve_every * 4:
            rng_local = np.random.default_rng(hash(tuple(np.round(a[:8], 6))) % (2**31))
            a = project_simplex(a + rng_local.standard_normal(n) * 0.01)
            conv = fast_convolve(a)
            stall = 0

    return best_a, best_obj


def subgrad_refine(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30):
    """Subgradient descent with momentum on the simplex (top-k peaks)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break

        # Weight peaks by their values (softer than hard max)
        w = peak_vals / (peak_vals.max() + 1e-10)
        w = w ** 2  # emphasize larger peaks
        w /= w.sum()

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = w[idx] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            a = project_simplex(a + np.random.default_rng().standard_normal(n) * 0.005)
            conv = fast_convolve(a)
            v *= 0.5
            no_improve = 0

        lr *= 0.9995

    return best_a, best_obj


def coordinate_descent_polish(a_in, max_rounds=50):
    """Coordinate descent: try small mass transfers between coordinates."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()

    for rnd in range(max_rounds):
        improved = False
        # Sort indices by value to prioritize moving mass from high to low
        idx_order = np.argsort(a)
        for i in range(n):
            if a[i] < 1e-8:
                continue
            # Try a few target positions
            for j in [idx_order[0], idx_order[n//4], idx_order[n//2], idx_order[3*n//4]]:
                if j == i:
                    continue
                delta = a[i] * 0.03
                if delta < 1e-9:
                    continue
                a_test = a.copy()
                a_test[i] = max(0, a_test[i] - delta)
                a_test[j] += delta
                a_test = np.maximum(a_test, 0.0)
                st = a_test.sum()
                if st < 1e-15:
                    continue
                a_test /= st
                conv_test = fast_convolve(a_test)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-13:
                    a = a_test
                    best_obj = obj_test
                    conv = conv_test
                    improved = True
        if not improved:
            break

    return a


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    # Edge-weighted
    edge = np.zeros(n)
    e = max(1, n // 5)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()

    # Gaussian (centered)
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Random (Dirichlet)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Bimodal
    g2 = (np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2))
    inits['bimodal'] = g2 / g2.sum()

    # Inverse triangle (peak at edges, dip in center)
    inv_tri = np.maximum(n - np.minimum(x + 1.0, n - x), 0.01)
    inits['inv_triangle'] = inv_tri / inv_tri.sum()

    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_w = np.maximum(cos_w, 0.01)
    inits['cosine'] = cos_w / cos_w.sum()

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10 and len(parent) > 0:
            if len(parent) == n:
                inits['parent'] = p / ps
            elif len(parent) > n:
                p_res = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), p)
                p_res = np.maximum(p_res, 0)
                prs = p_res.sum()
                if prs > 1e-10:
                    inits['parent_ds'] = p_res / prs
            else:
                p_res = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), p)
                p_res = np.maximum(p_res, 0)
                prs = p_res.sum()
                if prs > 1e-10:
                    inits['parent_up'] = p_res / prs

    return inits

    # We need to pad w with zeros to handle this

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using temperature-annealed smoothed
(log-sum-exp) projected gradient descent, LP-based directional refinement,
spectral flattening initialization, and coordinate descent polish.
Key innovations: smooth approximations of max via softmax with annealing for
better exploration, multiplicative weight updates as alternative projection,
and iterative spectral projection for structured initializations.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def multiplicative_project(v, current_a):
    """Mirror-descent style update: a_new = a * exp(-eta*v) normalized."""
    # Already normalized by caller
    return np.maximum(current_a, 1e-15)


def build_lp_pairs(n):
    """Pre-build index pairs for LP constraint matrix."""
    pairs = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            pairs.append((k, j, k - j))
    return pairs


def solve_lp_direction(f, M, pairs, n, time_limit=8.0):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for k, j, idx in pairs:
        A_ub[k, j] = f[idx]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs',
                         options={'presolve': True, 'time_limit': time_limit})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def compute_smoothed_grad(a, w):
    """Compute gradient of LSE(tau * conv(a,a)) wrt a.
    grad[j] = sum_k w[k] * a[k-j]
    Uses FFT-based convolution with reversed a.
    """
    n = len(a)
    a_rev = a[::-1]
    size = 1
    while size < 3 * n:
        size *= 2
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(W * A, size)
    return conv[n - 1:2 * n - 1]


def smoothed_gradient_descent(a0, time_limit, tau_init=5.0, tau_end=500.0,
                               lr_init=0.02, momentum=0.85,
                               anneal_rate=1.05, use_mw=False):
    """Projected gradient descent on LSE-smoothed objective with annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    tau = tau_init
    lr = lr_init
    v = np.zeros(n)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    rng_local = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        conv_max = conv.max()
        if conv_max < 1e-15:
            break

        # Compute softmax weights (temperature-scaled)
        scaled = tau * (conv - conv_max)  # subtract max for numerical stability
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum

        # Compute gradient
        grad = compute_smoothed_grad(a, w)

        # Momentum update
        if use_mw:
            # Multiplicative weights style
            a_new = a * np.exp(-lr * grad)
            a_new = np.maximum(a_new, 0)
            ns = a_new.sum()
            if ns < 1e-15:
                a_new = a
            else:
                a = a_new / ns
        else:
            v = momentum * v - lr * grad
            a = project_simplex(a + v)

        # Recompute convolution
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Anneal temperature
        tau = min(tau * anneal_rate, tau_end)

        # Decay learning rate
        lr = max(0.001, lr * 0.999)

        # Restart if stuck
        if no_improve > 200:
            noise = rng_local.standard_normal(n) * 0.01
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0

        # Adaptive: if very high tau, sharpen more
        if tau >= tau_end * 0.8:
            lr = max(0.0005, lr * 0.995)

    return best_a, best_obj


def optimize_lp(f_init, time_limit, blend_init=0.4, blend_min=0.005,
                blend_halve_every=12):
    """LP-based directional optimization with adaptive blending."""
    a = np.maximum(f_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    pairs = build_lp_pairs(n)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    rng_local = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break

        g = solve_lp_direction(a, M, pairs, n, time_limit=5.0)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall >= blend_halve_every:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min:
                break

        if stall > blend_halve_every * 4:
            a = project_simplex(a + rng_local.standard_normal(n) * 0.008)
            conv = fast_convolve(a)
            stall = 0

    return best_a, best_obj


def subgrad_refine(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30):
    """Subgradient descent with momentum on the simplex (top-k peaks)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break

        # Softer weighting: emphasize peaks that are close to max
        w = peak_vals / (pmax + 1e-15)
        w = w ** 3  # cubic to emphasize larger peaks
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = w[idx] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            a = project_simplex(a + np.random.default_rng().standard_normal(n) * 0.004)
            conv = fast_convolve(a)
            v *= 0.4
            no_improve = 0

        lr *= 0.999

    return best_a, best_obj


def coordinate_descent_polish(a_in, max_rounds=40):
    """Coordinate descent: small mass transfers between coordinates."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()

    for rnd in range(max_rounds):
        improved = False
        idx_order = np.argsort(a)
        for i in range(n):
            if a[i] < 1e-9:
                continue
            # Try transferring to a few low-value positions
            for j in [idx_order[0], idx_order[n // 4], idx_order[n // 2],
                      idx_order[3 * n // 4], idx_order[n - 1]]:
                if j == i:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    delta = a[i] * frac
                    if delta < 1e-10:
                        continue
                    a_test = a.copy()
                    a_test[i] = max(0.0, a_test[i] - delta)
                    a_test[j] += delta
                    st = a_test.sum()
                    if st < 1e-15:
                        continue
                    a_test /= st
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_obj - 1e-13:
                        a = a_test
                        best_obj = obj_test
                        conv = conv_test
                        improved = True
                        break
        if not improved:
            break

    return a


def spectral_flatten_init(n, rng, iterations=30):
    """Initialize by iteratively flattening the DFT magnitude."""
    a = rng.dirichlet(np.ones(n) * 0.3)
    for _ in range(iterations):
        A = np.fft.rfft(a, n)
        mag = np.abs(A)
        mean_mag = mag.mean()
        if mean_mag < 1e-15:
            break
        phase = A / (mag + 1e-30)
        A_flat = mean_mag * phase
        a_new = np.fft.irfft(A_flat, n)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            # Fallback: use the pre-projection real part
            a_new = np.maximum(a_new + 0.01, 0)
            s = a_new.sum()
            if s < 1e-10:
                break
        a = a_new / s
    return a
