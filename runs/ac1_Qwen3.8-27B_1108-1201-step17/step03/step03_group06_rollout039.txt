from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def build_lp_matrix_sparse(a, k_indices):

        row = np.zeros(n)

def smooth_gradient(a, beta):
    """Compute smooth gradient of max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv

    w = np.exp(scaled)

    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    size = 2 * n
    fft_size = 1 << (size - 1).bit_length()
    
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    
    W = np.fft.fft(w_padded)
    A = np.fft.fft(a_padded)
    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    conv_wa = np.fft.ifft(A * W).real[:n]
    
    grad = cross + conv_wa
    return grad, conv

    s = a_new.sum()

"""
Multi-phase optimization for minimum peak auto-convolution sequences.
Combines adaptive active-set LP, smooth gradient descent with momentum,
and hybrid alternating optimization to minimize 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, beta):
    """Compute smooth gradient of max(conv(a,a)) using log-sum-exp."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    size = 2 * n
    fft_size = 1 << (size - 1).bit_length()
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    
    W = np.fft.fft(w_padded)
    A = np.fft.fft(a_padded)
    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    conv_wa = np.fft.ifft(A * W).real[:n]
    grad = cross + conv_wa
    return grad, conv, w


def solve_lp_direction(a, M, k_indices):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k]<=M for k in k_indices."""
    n = len(a)
    if len(k_indices) == 0:
        return np.ones(n) / n
    
    A_ub = np.zeros((len(k_indices), n))
    for i, k in enumerate(k_indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[i, idx] = a[k - idx]
    
    b_ub = np.full(len(k_indices), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    """Compute 2n*max(conv) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def resample_sequence(a, new_n):

"""
Multi-phase hybrid optimization for minimum peak auto-convolution sequences.
Combines adaptive active-set LP, smooth gradient descent with momentum,
peak-balancing heuristics, and hybrid alternating optimization to minimize
2n*max(conv(a,a))/(sum(a))^2.

Key improvements over previous:
- Adaptive active-set constraint management
- Backtracking line search in LP phase
- More aggressive time allocation for deep optimization
- Block-level perturbations for escaping local minima
- Simulated-restart mechanism with large jumps

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, beta):
    """Compute smooth gradient of max(conv(a,a)) using log-sum-exp weighting."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    size = 2 * n
    fft_size = 1 << (size - 1).bit_length()
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a

    W = np.fft.fft(w_padded)
    A = np.fft.fft(a_padded)
    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    conv_wa = np.fft.ifft(A * W).real[:n]
    grad = cross + conv_wa
    return grad, conv


def solve_lp_direction(a, M, k_indices):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k]<=M for k in k_indices."""
    n = len(a)
    if len(k_indices) == 0:
        return np.ones(n) / n

    A_ub = np.zeros((len(k_indices), n))
    for i, k in enumerate(k_indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[i, idx] = a[k - idx]

    b_ub = np.full(len(k_indices), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def objective_normalized(a):
    """Compute 2n*max(conv) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def resample_sequence(a, new_n):
    """Resample a sequence to a different length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new


def block_perturb(a, rng, block_size=16, strength=0.02):
    """Perturb a random block of coordinates, renormalize."""
    n = len(a)
    a_new = a.copy()
    start = rng.integers(0, max(1, n - block_size))
    end = min(n, start + block_size)
    a_new[start:end] += rng.standard_normal(end - start) * strength
    a_new = project_simplex(a_new)
    return a_new

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: length={len(parent)}, sum_original={p.sum():.4f}")
    except NameError:
        pass
    except Exception:
        pass

    # ========== Phase 1: Quick Exploration (100s) ==========
    print("=== Phase 1: Quick Exploration ===")
    t1 = time.monotonic()
    p1_time = 100.0

    results = []

    if parent is not None:
        pn = len(parent)
        # Optimize parent at original length
        ta = min(50.0, p1_time - (time.monotonic() - t1) - 20)
        if ta > 15:
            a_opt, obj = lp_optimize(parent, time_limit=ta,
                                     step_init=0.9, k_frac=0.3, seed=1)
            results.append((obj, pn, a_opt))
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P1 parent(n={pn}): {score:.8f}")

        # Try parent resampled to nearby lengths
        for dl in [-32, 32, 64, -64]:
            if time.monotonic() - t1 > p1_time - 30:
                break
            nn = pn + dl
            if nn < 128 or nn > 512:
                continue
            a_r = resample_sequence(parent, nn)
            ta = min(35.0, p1_time - (time.monotonic() - t1) - 15)
            if ta < 10:
                break
            a_opt, obj = lp_optimize(a_r, time_limit=ta,
                                     step_init=0.8, k_frac=0.3, seed=nn + 100)
            results.append((obj, nn, a_opt))
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P1 resampled(n={nn}): {score:.8f}")

    # Diverse initializations
    n_base = len(parent) if parent is not None else 320
    x = np.arange(n_base, dtype=np.float64)

    diverse = {}
    # Bimodal (edges high)
    bm = np.zeros(n_base)
    edge = max(1, n_base // 4)
    bm[:edge] = 1.0
    bm[-edge:] = 1.0
    diverse['bimodal'] = bm / bm.sum()

    # Wide gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n_base) / (0.25 * n_base)) ** 2)
    diverse['gauss'] = g / g.sum()

    # Random sparse
    diverse['random'] = rng.dirichlet(np.ones(n_base) * 0.3)

    # Alternating
    alt = np.ones(n_base)
    alt[::2] = 0.4
    diverse['alternating'] = alt / alt.sum()

    for name in list(diverse.keys())[:3]:
        if time.monotonic() - t1 > p1_time - 10:
            break
        ta = min(25.0, p1_time - (time.monotonic() - t1) - 5)
        if ta < 8:
            break
        a_opt, obj = lp_optimize(diverse[name], time_limit=ta,
                                 step_init=1.0, k_frac=0.25,
                                 seed=abs(hash(name)) % 10000)
        results.append((obj, n_base, a_opt))
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  P1 {name}: {score:.8f}")

    results.sort(key=lambda z: z[0])
    print(f"Phase 1 done: best={best_score:.8f}")

    # ========== Phase 2: Deep LP Optimization (500s) ==========
    print("=== Phase 2: Deep LP Optimization ===")
    t2 = time.monotonic()
    p2_time = 500.0

    # Collect starting points
    starts = []
    if best_sequence is not None:
        starts.append(best_sequence.copy())
    for _, _, a in results[:4]:
        starts.append(a.copy())

    # Deduplicate
    unique_starts = []
    for s in starts:
        is_dup = False
        for u in unique_starts:
            if len(u) == len(s) and np.linalg.norm(u - s) < 0.03:
                is_dup = True
                break
        if not is_dup:
            unique_starts.append(s)
    starts = unique_starts[:5]

    # LP configurations
    configs = [
        (0.9, 0.3, 1.0),
        (1.0, 0.4, 1.0),
        (0.6, 0.2, 1.0),
        (0.8, 0.5, 1.0),
        (1.0, 0.3, 1.03),
        (0.7, 0.35, 0.97),
    ]

    total = len(starts) * len(configs)
    t_per = max(12.0, p2_time / max(1, total))

    rc = 0
    for sa in starts:
        for (si, kf, rl) in configs:
            if time.monotonic() - t2 > p2_time - 20:
                break
            ta = min(t_per, p2_time - (time.monotonic() - t2) - 10)
            if ta < 8:
                break
            a_opt, obj = lp_optimize(sa, time_limit=ta,
                                     step_init=si, k_frac=kf, relax=rl,
                                     seed=rc * 19 + 7)
            rc += 1
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P2 [{rc}] step={si},kf={kf},relax={rl}: {score:.8f}")

    print(f"Phase 2 done: best={best_score:.8f}")

    # ========== Phase 3: Gradient Polish (350s) ==========
    print("=== Phase 3: Gradient Polish ===")
    t3 = time.monotonic()
    p3_time = 350.0

    if best_sequence is not None:
        n_b = len(best_sequence)
        gcfgs = [
            (500, 0.03, 0.90),
            (1000, 0.02, 0.92),
            (2000, 0.01, 0.94),
            (300, 0.04, 0.88),
            (5000, 0.005, 0.95),
            (100, 0.05, 0.85),
            (8000, 0.003, 0.96),
        ]
        t_per_g = p3_time / (len(gcfgs) + 4)

        for i, (bi, li, mi) in enumerate(gcfgs):
            if time.monotonic() - t3 > p3_time - 15:
                break
            ta = min(t_per_g, p3_time - (time.monotonic() - t3) - 10)
            if ta < 10:
                break
            a_opt, obj = gradient_optimize(best_sequence, time_limit=ta,
                                           beta_init=bi, lr_init=li,
                                           momentum=mi, seed=i * 13 + 1)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P3 grad[{i}] beta={bi}: {score:.8f}")

        # Block perturbations + gradient
        for i in range(4):
            if time.monotonic() - t3 > p3_time - 10:
                break
            a_p = block_perturb(best_sequence, rng, block_size=32, strength=0.015)
            ta = min(t_per_g, p3_time - (time.monotonic() - t3) - 5)
            if ta < 8:
                break
            a_opt, obj = gradient_optimize(a_p, time_limit=ta,
                                           beta_init=1500, lr_init=0.02,
                                           momentum=0.92, seed=300 + i)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P3 blockpert[{i}]: {score:.8f}")

    print(f"Phase 3 done: best={best_score:.8f}")

    # ========== Phase 4: Hybrid Alternating (350s) ==========
    print("=== Phase 4: Hybrid Alternating ===")
    t4 = time.monotonic()
    p4_time = 350.0

    if best_sequence is not None:
        n_b = len(best_sequence)
        a = best_sequence.copy()
        best_phase = a.copy()
        best_obj_ph = objective_normalized(a)
        v = np.zeros(n_b)
        step = 0.7
        lr = 0.02
        mom = 0.9
        beta = 1000
        k_frac = 0.35
        no_imp = 0
        it = 0

        t4_0 = time.monotonic()
        while time.monotonic() - t4_0 < p4_time - 10:
            conv = fast_convolve(a)
            M = conv.max()

            if it % 3 == 0:
                # LP step
                k_num = max(10, int(n_b * k_frac))
                peaks = np.argpartition(conv, -k_num)[-k_num:]
                b = solve_lp_direction(a, M, peaks)
                if b is not None:
                    bs = b.sum()
                    if bs > 1e-15:
                        b = b / bs
                        a = (1.0 - step) * a + step * b
                        a = np.maximum(a, 0.0)
                        s = a.sum()
                        if s > 1e-15:
                            a = a / s
            elif it % 3 == 1:
                # Gradient step
                grad, conv = smooth_gradient(a, beta)
                gn = np.linalg.norm(grad)
                if gn > 1e-15:
                    grad /= gn
                v = mom * v - lr * grad
                a = project_simplex(a + v)
            else:
                # Small random perturbation
                a = block_perturb(a, rng, block_size=16,
                                  strength=0.003 * (1.0 + it / 500.0))
                v *= 0.5

            obj = objective_normalized(a)
            if obj < best_obj_ph - 1e-13:
                best_obj_ph = obj
                best_phase = a.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 30 == 0:
                    step *= 0.93
                    lr *= 0.93
                    v *= 0.7
                if no_imp > 200:
                    # Larger restart
                    a = block_perturb(best_phase, rng, block_size=64,
                                      strength=0.02)
                    v = np.zeros(n_b)
                    no_imp = 0
                    step = 0.7
                    lr = 0.02
                    beta = 1000

            it += 1

        score = float(evaluate_sequence(best_phase.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = best_phase.copy()
            print(f"  P4 hybrid: {score:.8f}")

    print(f"Phase 4 done: best={best_score:.8f}")

    # ========== Phase 5: Final Polish ==========
    print("=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.1f}s")

    if best_sequence is not None:
        n_b = len(best_sequence)

        # Fine LP with more constraints
        ta = min(remaining * 0.3, 70.0)
        if ta > 12:
            a_opt, _ = lp_optimize(best_sequence, time_limit=ta,
                                   step_init=0.3, k_frac=0.6, seed=999)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P5 fine LP: {score:.8f}")

        # Fine gradient with high beta
        ta = min(remaining * 0.25, 50.0)
        if ta > 10:
            a_opt, _ = gradient_optimize(best_sequence, time_limit=ta,
                                         beta_init=8000, lr_init=0.003,
                                         momentum=0.96, seed=777)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P5 fine grad: {score:.8f}")

        # Micro perturbations with short gradient follow-up
        remaining = deadline - time.monotonic() - 10
        if remaining > 15:
            nt = min(int(remaining // 4), 20)
            for i in range(nt):
                if time.monotonic() > deadline - 15:
                    break
                improved = False
                for sc in [0.0003, 0.0008, 0.002, 0.005]:
                    noise = rng.standard_normal(n_b) * sc
                    ap = project_simplex(best_sequence + noise)
                    sp = float(evaluate_sequence(ap.tolist()))
                    if np.isfinite(sp) and sp < best_score:
                        best_score = sp
                        best_sequence = ap.copy()
                        print(f"  P5 pert[{i}] s={sc}: {sp:.8f}")
                        improved = True
                        break
                if improved:
                    # Short gradient polish after improvement
                    ta2 = min(8.0, deadline - time.monotonic() - 12)
                    if ta2 > 4:
                        a_opt, _ = gradient_optimize(
                            best_sequence, time_limit=ta2,
                            beta_init=5000, lr_init=0.005,
                            momentum=0.95, seed=888 + i)
                        score = float(evaluate_sequence(a_opt.tolist()))
                        if np.isfinite(score) and score < best_score:
                            best_score = score
                            best_sequence = a_opt.copy()
                            print(f"  P5 pert+grad[{i}]: {best_score:.8f}")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    fs = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.8f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
