"""

Key improvements:
1. Fixed LSE gradient (previous had erroneous conv-weighting)
2. Top-k peak targeting for focused optimization
3. Adaptive beta annealing with Nesterov momentum

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient_correct(a, conv, beta, top_k=None):
    """
    Correct gradient of (1/beta)*logsumexp(beta*conv) w.r.t. a.
    Uses top_k peaks if specified for focused optimization.
    """
    n = len(a)
    c = conv.copy()
    if top_k is not None and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_masked = np.zeros_like(c)
        c_masked[idx] = c[idx]
        c = c_masked

    scaled = beta * c
    scaled_max = scaled.max()
    if scaled_max < -50:
        p = np.zeros(len(c))
        p[np.argmax(conv)] = 1.0
    else:
        scaled = scaled - scaled_max
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(c))
            p[np.argmax(conv)] = 1.0
        else:
            p /= p_sum

    # Compute cross-correlation: corr[i] = sum_k a[k] * p[k+i]
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(A * np.conj(P), fft_size)

    grad = 2.0 * cross[:n]

    # Subtract diagonal correction: p[2i]*a[i] where 2i < 2n-1
    i_arr = np.arange(n)
    valid = 2 * i_arr < len(p)
    if valid.any():
        grad[valid] -= p[2 * i_arr[valid]] * a[i_arr[valid]]

    return grad


def subgradient_topk(a, conv, k=5):
    """Exact subgradient targeting only top-k peaks (average of their gradients)."""
    n = len(a)
    top_idx = np.argpartition(conv, -k)[-k:]
    p = np.zeros(len(conv))
    p[top_idx] = 1.0 / len(top_idx)

    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(A * np.conj(P), fft_size)

    grad = 2.0 * cross[:n]
    i_arr = np.arange(n)
    valid = 2 * i_arr < len(p)
    if valid.any():
        grad[valid] -= p[2 * i_arr[valid]] * a[i_arr[valid]]

    return grad


def lp_direction(a, M, n):
    """
    Solve LP: max sum(b_j) s.t. b_j >= 0, (a*b)[k] <= M for all k.
    Returns b normalized to sum=1, or None.
    """
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 60})
    except Exception:
        return None

    if not result.success or result.x is None:
        return None

    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None

    return b / b_sum


def optimize_nesterov(a0, time_limit, beta_init=20.0, beta_end=300.0,
                      lr_init=0.02, momentum=0.9, top_k_init=None,
                      top_k_end=None, perturb_every=300, perturb_scale=0.005,
                      seed=0):
    """
    Nesterov accelerated projected gradient descent with corrected LSE gradient.
    Input: a0 on simplex (sum=1, non-negative).
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    if top_k_init is None:
        top_k_init = n
    if top_k_end is None:
        top_k_end = max(3, n // 20)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * (frac ** 0.7)
        top_k = int(top_k_init + (top_k_end - top_k_init) * frac)
        top_k = max(3, top_k)

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = a.copy()
        else:
            y = y / y_sum

        conv_y = fast_convolve(y)
        grad = lse_gradient_correct(y, conv_y, beta, top_k=top_k)

        # Adaptive learning rate
        lr = lr_init * (1.0 + 0.5 * np.cos(np.pi * frac)) * (1.0 - 0.5 * frac)
        lr = max(0.0005, lr)

        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a = project_simplex(a_new)

        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic perturbation to escape local minima
        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * perturb_scale * (0.5 + frac)
            a = project_simplex(a + noise)
            v *= 0.2
            no_improve = 0
            perturb_every = int(perturb_every * 1.3)

        # Occasionally use exact subgradient
        if total_iters % 50 == 0 and frac > 0.3:
            sg = subgradient_topk(a, conv, k=max(3, min(10, n // 30)))
            a = project_simplex(a - lr * 0.5 * sg)
            conv = fast_convolve(a)
            if conv.max() < best_obj:
                best_obj = conv.max()
                best_a = a.copy()

    return best_a, best_obj


def optimize_lp_phase(a0, time_limit, seed=42):
    """LP-based direction finding phase with multiple M factors."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    current_a = a.copy()

    t0 = time.monotonic()
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(current_a)
        M = conv_cur.max()

        b = None
        for M_factor in [1.0, 1.002, 1.005, 1.01, 1.02]:
            b = lp_direction(current_a, M * M_factor, n)
            if b is not None:
                break

        if b is None:
            noise = rng.standard_normal(n) * 0.008
            current_a = project_simplex(current_a + noise)
            stuck += 1
            if stuck > 5:
                break
            continue

        stuck = 0

        improved = False
        for t in [0.3, 0.2, 0.15, 0.1, 0.05, 0.02]:
            a_try = (1 - t) * current_a + t * b
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = conv_try.max()

            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                current_a = a_try.copy()
                improved = True
                break
            elif obj_try < M * 0.998:
                current_a = a_try.copy()
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                improved = True
                break

        if not improved:
            # Try very small steps
            for t in [0.01, 0.005]:
                a_try = (1 - t) * current_a + t * b
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                if conv_try.max() < M * 0.9999:
                    current_a = a_try.copy()
                    if conv_try.max() < best_obj:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                    break

    return best_a, best_obj


def pairwise_descent(a0, time_limit, seed=0):
    """
    Pairwise coordinate descent: transfer mass between pairs to reduce max peak.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(a)
        max_val = conv_cur.max()

        # Get top peaks
        top_k = min(5, 2 * n - 1)
        peak_indices = np.argpartition(conv_cur, -top_k)[-top_k:]

        improved = False
        for k in peak_indices:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue

            # Find contributing pairs sorted by contribution
            pairs = []
            for i in range(i_start, i_end + 1):
                j = k - i
                pairs.append((a[i] * a[j], i, j))
            pairs.sort(reverse=True)

            # Try reducing top contributor
            for contrib, i, j in pairs[:3]:
                if contrib < 1e-12:
                    continue
                # Try moving mass from the larger of a[i], a[j] to a random position
                src = i if a[i] >= a[j] else j
                for _ in range(2):
                    dst = int(rng.integers(0, n))
                    if dst == src or dst == (i if src == j else j):
                        continue
                    for frac in [0.1, 0.05, 0.02]:
                        amt = a[src] * frac
                        if amt < 1e-10:
                            break
                        a_try = a.copy()
                        a_try[src] -= amt
                        a_try[dst] += amt
                        conv_try = fast_convolve(a_try)
                        if conv_try.max() < best_obj - 1e-12:
                            best_obj = conv_try.max()
                            best_a = a_try.copy()
                            a = a_try.copy()
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break

        iters += 1

        # Random exploration every few iterations
        if not improved and iters % 100 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                for frac in [0.05, 0.02, 0.01]:
                    amt = a[i] * frac
                    if amt < 1e-10:
                        break
                    a_try = a.copy()
                    a_try[i] -= amt
                    a_try[j] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj - 1e-12:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        break

    return best_a, best_obj


def make_bimodal_init(n, alpha, rng):
    """Create a U-shaped (bimodal) initialization with parameter alpha."""
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    # U-shaped: more mass at edges
    a = 1.0 + alpha * (2 * x - 1) ** 2
    a = np.maximum(a, 0.01)
    # Add small random perturbation
    a += rng.standard_normal(n) * 0.01
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def make_edge_conc_init(n, edge_width, rng):
    """Initialize with concentration at edges."""
    a = np.zeros(n)
    ew = max(2, int(edge_width * n))
    # Smooth edge concentration
    for i in range(n):
        d_left = min(i, n - 1 - i)
        a[i] = np.exp(-0.5 * (d_left / ew) ** 2)
    a += rng.standard_normal(n) * 0.02
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses corrected gradients, LP direction finding, and pairwise descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_sequence = None
    best_score = float('inf')

    # Try to get parent
    parent = None
    parent_n = 384
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_n = len(parent)
            parent_score = evaluate_sequence(p.tolist())
            print(f"Parent: n={parent_n}, score={parent_score:.6f}", flush=True)
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
    except Exception as e:
        print(f"No parent available: {e}", flush=True)

    # Phase 1: Quick scan over lengths and initializations
    print("=== Phase 1: Quick Scan ===", flush=True)
    t_p1_start = time.monotonic()
    p1_budget = 120.0

    # Lengths to try
    if parent is not None:
        lengths = sorted(set([max(64, parent_n // 2), parent_n,
                              parent_n + 64, parent_n + 128,
                              max(64, parent_n - 64)]))
    else:
        lengths = [192, 256, 320, 384, 448, 512]

    scan_results = []

    for n in lengths:
        if time.monotonic() > deadline - 250:
            break
        if n < 32:
            continue

        # Generate several initializations
        inits = {}

        if parent is not None:
            if len(parent) == n:
                inits['parent'] = parent.copy()
            elif len(parent) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
                pp = np.maximum(pp, 0)
                ps = pp.sum()
                if ps > 1e-10:
                    inits['parent_ds'] = pp / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
                pp = np.maximum(pp, 0)
                ps = pp.sum()
                if ps > 1e-10:
                    inits['parent_up'] = pp / ps

        # U-shaped / bimodal
        for alpha in [0.5, 1.0, 2.0, 4.0, 8.0]:
            inits[f'u_{alpha}'] = make_bimodal_init(n, alpha, rng)

        # Edge concentration
        for ew_frac in [0.1, 0.2, 0.3]:
            inits[f'edge_{int(ew_frac*100)}'] = make_edge_conc_init(n, ew_frac, rng)

        # Uniform
        inits['uniform'] = np.ones(n) / n

        # Random Dirichlet
        inits['rand1'] = rng.dirichlet(np.ones(n) * 0.5)
        inits['rand2'] = rng.dirichlet(np.ones(n) * 2.0)

        names = list(inits.keys())
        # Prioritize parent and U-shaped
        priority = ['parent', 'parent_ds', 'parent_up', 'u_2.0', 'u_4.0',
                    'u_8.0', 'edge_20', 'edge_30', 'uniform', 'rand1']
        names = [nm for nm in priority if nm in names] + \
                [nm for nm in names if nm not in priority]
        names = names[:8]

        time_rem = p1_budget - (time.monotonic() - t_p1_start)
        per_init = max(4.0, time_rem / max(1, len(names)))

        for name in names:
            if time.monotonic() > deadline - 220:
                break
            t_alloc = min(per_init, 15.0)
            if t_alloc < 3:
                break

            a0 = inits[name]
            a_opt, obj_opt = optimize_nesterov(
                a0, t_alloc, beta_init=15, beta_end=80,
                lr_init=0.015, momentum=0.88,
                top_k_init=n, top_k_end=max(5, n // 10),
                perturb_every=200, perturb_scale=0.004,
                seed=int(rng.integers(0, 100000)))

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}", flush=True)

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}", flush=True)

    # Phase 2: Deep optimization
    print("=== Phase 2: Deep Optimization ===", flush=True)

    # Select top configs
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = n
        if key not in seen:
            top_configs.append((n, a_opt))
            seen.add(key)
        if len(top_configs) >= 3:
            break

    # Always include current best
    if best_sequence is not None:
        top_configs.insert(0, (len(best_sequence), best_sequence.copy()))

    # Add a couple random restarts at best length
    bn = len(best_sequence) if best_sequence is not None else parent_n
    top_configs.append((bn, make_bimodal_init(bn, 3.0, rng)))
    top_configs.append((bn, make_edge_conc_init(bn, 0.25, rng)))

    remaining = deadline - time.monotonic() - 100.0
    n_configs = len(top_configs)
    per_config = max(20.0, remaining / n_configs)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 80:
            break
        t_alloc = min(per_config, 300.0)
        if t_alloc < 10:
            break

        print(f"  Config {i}: n={n}, time={t_alloc:.0f}s", flush=True)

        # Sub-phase A: Nesterov gradient (50%)
        t_a = t_alloc * 0.45
        a_a, obj_a = optimize_nesterov(
            a0, t_a, beta_init=30, beta_end=400,
            lr_init=0.012, momentum=0.92,
            top_k_init=n, top_k_end=max(3, n // 30),
            perturb_every=250, perturb_scale=0.003,
            seed=int(rng.integers(0, 1000000)))

        score_a = float(evaluate_sequence(a_a.tolist()))
        if np.isfinite(score_a) and score_a < best_score:
            best_score = score_a
            best_sequence = a_a.copy()
            print(f"    Nesterov: {score_a:.6f}", flush=True)

        # Sub-phase B: LP direction (30%)
        t_b = t_alloc * 0.30
        if t_b > 8:
            a_b, obj_b = optimize_lp_phase(a_a, t_b, seed=int(rng.integers(0, 1000000)))
            score_b = float(evaluate_sequence(a_b.tolist()))
            if np.isfinite(score_b) and score_b < best_score:
                best_score = score_b
                best_sequence = a_b.copy()
                print(f"    LP: {score_b:.6f}", flush=True)
            a_a = a_b

        # Sub-phase C: Fine Nesterov (25%)
        t_c = t_alloc * 0.25
        if t_c > 5:
            a_c, obj_c = optimize_nesterov(
                a_a, t_c, beta_init=200, beta_end=1000,
                lr_init=0.004, momentum=0.95,
                top_k_init=max(10, n // 10), top_k_end=max(3, n // 50),
                perturb_every=150, perturb_scale=0.001,
                seed=int(rng.integers(0, 1000000)))

            score_c = float(evaluate_sequence(a_c.tolist()))
            if np.isfinite(score_c) and score_c < best_score:
                best_score = score_c
                best_sequence = a_c.copy()
                print(f"    Fine: {score_c:.6f}", flush=True)

    # Phase 3: Polish
    print("=== Phase 3: Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0

        # Pairwise descent
        t_pw = remaining * 0.30
        if t_pw > 10:
            a_pw, obj_pw = pairwise_descent(best_sequence, t_pw, seed=12345)
            score_pw = float(evaluate_sequence(a_pw.tolist()))
            if np.isfinite(score_pw) and score_pw < best_score:
                best_score = score_pw
                best_sequence = a_pw.copy()
                print(f"  Pairwise: {best_score:.6f}", flush=True)

        # Fine gradient
        remaining = deadline - time.monotonic() - 20.0
        t_fg = remaining * 0.35
        if t_fg > 5:
            a_fg, obj_fg = optimize_nesterov(
                best_sequence, t_fg, beta_init=500, beta_end=2000,
                lr_init=0.001, momentum=0.97,
                top_k_init=10, top_k_end=3,
                perturb_every=100, perturb_scale=0.0005,
                seed=77777)
            score_fg = float(evaluate_sequence(a_fg.tolist()))
            if np.isfinite(score_fg) and score_fg < best_score:
                best_score = score_fg
                best_sequence = a_fg.copy()
                print(f"  FineGrad: {best_score:.6f}", flush=True)

        # Final LP
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 15:
            a_lp, obj_lp = optimize_lp_phase(best_sequence, remaining * 0.7, seed=55555)
            score_lp = float(evaluate_sequence(a_lp.tolist()))
            if np.isfinite(score_lp) and score_lp < best_score:
                best_score = score_lp
                best_sequence = a_lp.copy()
                print(f"  LP Final: {best_score:.6f}", flush=True)

        # Final pairwise
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 8:
            a_fp, obj_fp = pairwise_descent(best_sequence, remaining, seed=99999)
            score_fp = float(evaluate_sequence(a_fp.tolist()))
            if np.isfinite(score_fp) and score_fp < best_score:
                best_score = score_fp
                best_sequence = a_fp.copy()
                print(f"  FinalPair: {best_score:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Should not happen, but fallback
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}", flush=True)
    return [float(x) for x in best_sequence]
