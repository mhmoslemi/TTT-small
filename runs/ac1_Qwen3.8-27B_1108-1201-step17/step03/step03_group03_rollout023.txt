```python
    active = np.nonzero(w)[0]
    for k in active:
        wk = w[k]
        if wk == 0:
            continue
        # db_k/da_i for each i
        # i ranges from 0 to n-1
        # k-i valid: 0 <= k-i < n => max(0, k-n+1) <= i <= min(n-1, k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # a_{k-i} for i in [i_start, i_end]: this is a[k-i_start] to a[k-i_end]
        # i.e., a[k : i_start-1 : -1]... let me just index directly
        for i in range(i_start, i_end + 1):
            ki = k - i
            grad[i] += wk * a[ki]
            # Second term: a_{i-k}, valid if 0 <= i-k < n => k <= i <= k+n-1
            ik = i - k
            if 0 <= ik < n:
                grad[i] += wk * a[ik]
            # Diagonal correction
            if k == 2 * i:
                grad[i] -= wk * a[i]
    return grad
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated subgradient
descent with adaptive step size, softmax-weighted active set, and local pair 
optimization polishing. Explores multiple initializations and sequence lengths.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, w):
    """Compute subgradient of sum_k w_k * (a*a)_k w.r.t. a.
    
    Parameters:
    - a: array of length n (normalized, sum=1, a>=0)
    - w: array of length 2n-1 (weights, sum=1, w>=0)
    
    Returns:
    - gradient: array of length n
    """
    n = len(a)
    grad = np.zeros(n)
    active = np.nonzero(w)[0]
    for k in active:
        wk = w[k]
        if abs(wk) < 1e-15:
            continue
        # For lag k, db_k/da_i = a_{k-i} (if valid) + a_{i-k} (if valid) - a_i (if k=2i)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            ki = k - i
            grad[i] += wk * a[ki]
            ik = i - k
            if 0 <= ik < n:
                grad[i] += wk * a[ik]
            if k == 2 * i:
                grad[i] -= wk * a[i]
    return grad


def get_active_weights(b, top_k=30, threshold_frac=0.95):
    """Get softmax-weighted active set weights."""
    max_b = b.max()
    if max_b < 1e-15:
        return np.ones(len(b)) / len(b)
    
    # Include all lags above threshold_frac * max_b
    threshold = threshold_frac * max_b
    active_mask = b >= threshold
    
    # Also include top_k for sure
    if active_mask.sum() < top_k:
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        active_mask[top_idx] = True
    
    # Softmax weighting over active set
    b_active = b[active_mask].copy()
    # Use temperature for smoothing
    T = max_b * 0.05  # 5% temperature
    b_active -= b_active.max()
    w_active = np.exp(b_active / T)
    w_sum = w_active.sum()
    if w_sum < 1e-15:
        w_active = np.zeros(len(b_active))
        w_active[np.argmax(b_active)] = 1.0
    else:
        w_active /= w_sum
    
    w = np.zeros(len(b))
    w[active_mask] = w_active
    return w


def nesterov_subgrad(a0, time_limit, lr_init=0.01, momentum_init=0.9,
                      top_k=30, threshold_frac=0.9, temperature=0.05,
                      adapt_lr=True):
    """Nesterov-accelerated subgradient descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    y = a.copy()  # Nesterov extrapolation point
    v = np.zeros(n)  # momentum
    conv = fast_convolve(y)
    obj = 2.0 * n * conv.max()
    best_obj = obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    momentum = momentum_init
    lr = lr_init
    grad_sq_sum = lr * lr  # for standard step size rule
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute gradient at y
        w = get_active_weights(conv, top_k=top_k, threshold_frac=threshold_frac)
        g = compute_subgradient(y, w)
        
        # Standard subgradient step size
        if adapt_lr:
            g_norm_sq = np.dot(g, g)
            if g_norm_sq > 1e-15:
                grad_sq_sum += g_norm_sq
                lr = 1.0 / np.sqrt(grad_sq_sum)
                # Scale to be in a reasonable range
                lr = min(lr, 0.1)
                lr = max(lr, 0.001)
        
        # Nesterov update
        v = momentum * v - lr * g
        y_new = project_simplex(a + v)
        
        # For Nesterov, we alternate: update y, then set a = y_new
        # But for constrained problems, a simpler approach works:
        a = y_new
        y = a.copy()  # Simplified: no extrapolation (just momentum)
        
        # Actually, proper Nesterov:
        # y_{k+1} = (1 + beta_k/beta_{k-1}) * x_{k+1} - (beta_k/beta_{k-1}) * x_k
        # For simplicity, use: y = a + momentum * (a - a_prev)
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Decay momentum slightly
        if it % 100 == 0:
            momentum = max(0.8, momentum * 0.999)
        
        # Perturb if stuck
        if no_improve > 500:
            noise = np.random.default_rng().standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.1
            no_improve = 0
            grad_sq_sum = lr * lr  # Reset step size
    
    return best_a, best_obj


def pair_optimize(a, max_rounds=20):
    """Local optimization: for each pair (i,j), find optimal mass split."""
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    for rnd in range(max_rounds):
        improved = False
        # Sample a subset of pairs for efficiency
        rng_local = np.random.default_rng(rnd * 7919 + 13)
        
        for i in range(n):
            if a[i] < 1e-8:
                continue
            # Try moving mass to a few random other positions
            candidates = rng_local.choice(n, size=min(10, n-1), replace=False)
            for j in candidates:
                if j == i:
                    continue
                # Binary search / golden section for optimal transfer
                # Move delta from i to j
                best_delta = 0
                best_local_obj = best_obj
                # Try a few deltas
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5]:
                    delta = a[i] * frac
                    if delta < 1e-10:
                        continue
                    a_test = a.copy()
                    a_test[i] = a[i] - delta
                    a_test[j] = a[j] + delta
                    if a_test[i] < 0:
                        continue
                    a_test = np.maximum(a_test, 0)
                    st = a_test.sum()
                    a_test /= st
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_local_obj - 1e-14:
                        best_local_obj = obj_test
                        best_delta = delta
                        best_a_test = a_test.copy()
                
                if best_delta > 0:
                    a = best_a_test
                    conv = fast_convolve(a)
                    if 2.0 * n * conv.max() < best_obj:
                        best_obj = 2.0 * n * conv.max()
                    improved = True
        
        if not improved:
            break
    
    return a


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k.
    Uses scipy's linprog."""
    from scipy.optimize import linprog
    
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 10})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def optimize_lp_blend(f_init, time_limit, blend_init=0.3, blend_min=0.01):
    """LP-based directional optimization with adaptive blending."""
    a = np.maximum(f_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    uniform = np.ones(n) / n
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        
        g = lp_direction(a, M, n)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Blend
        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall >= 10:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min * 1.5:
                break
    
    return best_a, best_obj


def gen_initializations(n, parent=None, rng=None):
    """Generate diverse initializations for length n."""
    if rng is None:
        rng = np.random.default_rng(123)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Triangle
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # Gaussian (centered)
    g = np.exp(-0.5 * ((x - mid) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
          np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()
    
    # Edge-concentrated
    edge = np.zeros(n)
    e = max(2, n // 8)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()
    
    # Sinc-like (for low autocorrelation sidelobes)
    sinc = np.sinc((x - mid) / (n * 0.15))
    sinc = np.abs(sinc)
    sinc = np.maximum(sinc, 0.001)
    inits['sinc'] = sinc / sinc.sum()
    
    # Parent
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent'] = p / ps
        elif len(parent) > n:
            p = np.interp(x, np.linspace(0, len(parent)-1, n), np.maximum(parent, 0))
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_ds'] = p / ps
        else:
            p = np.interp(x, np.linspace(0, len(parent)-1, n), np.maximum(parent, 0))
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_up'] = p / ps
    
    return inits
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated subgradient
descent with softmax-weighted active sets, LP directional refinement, and
aggressive local pair optimization. Key improvements over previous approach:

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, w):
    """Compute subgradient of sum_k w_k * (a*a)_k w.r.t. a.
    
    Parameters:
    - a: array of length n (sum=1, a>=0)
    - w: array of length 2n-1 (weights, sum=1, w>=0)
    
    Returns:
    - gradient: array of length n
    """
    n = len(a)
    grad = np.zeros(n)
    active = np.nonzero(w)[0]
    
    for k in active:
        wk = w[k]
        if abs(wk) < 1e-15:
            continue
        # Term 1: a_{k-i} for i where 0 <= k-i < n
        i1_start = max(0, k - n + 1)
        i1_end = min(n - 1, k)
        if i1_start <= i1_end:
            idx1 = np.arange(i1_start, i1_end + 1)
            grad[idx1] += wk * a[k - idx1]
        
        # Term 2: a_{i-k} for i where 0 <= i-k < n
        i2_start = max(0, k)
        i2_end = min(n - 1, k + n - 1)
        if i2_start <= i2_end:
            idx2 = np.arange(i2_start, i2_end + 1)
            grad[idx2] += wk * a[idx2 - k]
        
        # Diagonal correction: when k = 2i, we double-counted
        if k % 2 == 0:
            i_diag = k // 2
            if 0 <= i_diag < n:
                grad[i_diag] -= wk * a[i_diag]
    
    return grad


def get_active_weights(b, top_k=40, threshold_frac=0.85, temperature=0.05):
    """Get softmax-weighted active set weights."""
    max_b = b.max()
    if max_b < 1e-15:
        return np.ones(len(b)) / len(b)
    
    threshold = threshold_frac * max_b
    active_mask = b >= threshold
    
    # Ensure top_k are included
    if active_mask.sum() < top_k:
        top_idx = np.argpartition(b, -min(top_k, len(b)))[-min(top_k, len(b)):]
        active_mask[top_idx] = True
    
    b_active = b[active_mask].copy()
    T = temperature * max_b
    b_active -= b_active.max()
    w_active = np.exp(b_active / T)
    w_sum = w_active.sum()
    if w_sum < 1e-30:
        w_active = np.zeros(len(b_active))
        w_active[np.argmax(b_active)] = 1.0
    else:
        w_active /= w_sum
    
    w = np.zeros(len(b))
    w[active_mask] = w_active
    return w


def nesterov_subgrad(a0, time_limit, lr_init=0.015, momentum_init=0.9,

    a = np.maximum(a0.astype(np.float64), 0.0)

    v = np.zeros(n)

    conv = fast_convolve(a)

            lr = lr_init
        elif use_adaptive_lr:
            grad_norm_sq_sum += g_norm_sq
            lr = 1.0 / np.sqrt(grad_norm_sq_sum)

        a = a_new

def lp_direction_solve(f, M, n, time_limit_lp=5.0):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    from scipy.optimize import linprog
    
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs',
                         options={'presolve': True, 'time_limit': time_limit_lp})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None

    while time.monotonic() - t0 < time_limit:
        improved = False
        rng_local = np.random.default_rng(int((time.monotonic() - t0) * 10000) % (2**31))
        
        # Process coordinates in random order
        order = rng_local.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-7:
                continue
            
            targets = rng_local.choice(n, size=min(max_targets, n - 1), replace=False)
            for j in targets:
                if j == i:
                    continue
                for frac in fracs:
                    delta = a[i] * frac
                    if delta < 1e-9:
                        continue
                    a_test = a.copy()
                    a_test[i] = a[i] - delta
                    a_test[j] = a[j] + delta
                    if a_test[i] < -1e-12:
                        continue
                    a_test = np.maximum(a_test, 0.0)
                    a_test /= a_test.sum()
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_obj - 1e-14:
                        a = a_test
                        best_obj = obj_test
                        conv = conv_test
                        improved = True
                        break  # Move to next j
        
        if improved:
            no_improve_rounds = 0
        else:
            no_improve_rounds += 1
            if no_improve_rounds >= 5:
                break
    
    return a


def local_triple_search(a_in, time_limit, max_triples=300):
    """Local search: try moving mass from one coordinate to split between two."""
    a = np.maximum(a_in.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    t0 = time.monotonic()
    
    rng_local = np.random.default_rng(42)
    attempts = 0
    
    while time.monotonic() - t0 < time_limit and attempts < max_triples:
        attempts += 1
        # Pick a coordinate with significant mass
        if a.max() < 1e-5:
            break
        i = rng_local.choice(n, p=a / a.sum())
        if a[i] < 1e-6:
            continue
        
        # Pick two target coordinates
        j1, j2 = rng_local.choice(n, size=2, replace=False)
        if j1 == i or j2 == i:
            continue
        
        for frac_total in [0.05, 0.1, 0.2]:
            delta = a[i] * frac_total
            if delta < 1e-8:
                continue
            for split in [0.3, 0.5, 0.7]:
                d1 = delta * split
                d2 = delta - d1
                a_test = a.copy()
                a_test[i] -= delta
                a_test[j1] += d1
                a_test[j2] += d2
                if a_test[i] < -1e-12:
                    continue
                a_test = np.maximum(a_test, 0.0)
                a_test /= a_test.sum()
                conv_test = fast_convolve(a_test)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-14:
                    a = a_test
                    best_obj = obj_test
                    conv = conv_test
                    break
    
    return a


def gen_initializations(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(99)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    
    inits['uniform'] = np.ones(n) / n
    
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    g = np.exp(-0.5 * ((x - mid) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    g2 = (np.exp(-0.5 * ((x - 0.25 * n) / (0.12 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.75 * n) / (0.12 * n)) ** 2))
    inits['bimodal'] = g2 / g2.sum()
    
    edge = np.zeros(n)
    e = max(2, n // 8)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()
    
    # Flat-top with tapered edges (like a window function)
    w = np.ones(n)
    taper = max(1, n // 16)
    for i in range(taper):
        val = 0.5 * (1 + np.cos(np.pi * i / taper))
        w[i] = val
        w[n - 1 - i] = val
    inits['window'] = w / w.sum()
    
    # Parent
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent'] = p / ps
        elif len(parent) > n:
            p = np.interp(x, np.linspace(0, len(parent) - 1, n), np.maximum(parent, 0))
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_ds'] = p / ps
        else:
            p = np.interp(x, np.linspace(0, len(parent) - 1, n), np.maximum(parent, 0))
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_up'] = p / ps
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: Nesterov subgradient -> LP directional -> 
    local pair/triple search -> final polish. Uses adaptive step sizes and
    softmax-weighted active sets for precise convergence.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
    except Exception:
        pass
    
    n = 512
    
    # Phase 1: Nesterov subgradient - primary optimization from parent
    print("=== Phase 1: Nesterov Subgradient (from parent) ===")
    inits = gen_initializations(n, parent, rng)
    
    # Start primarily from parent
    phase1_candidates = []
    if 'parent' in inits:
        phase1_candidates.append(('parent', inits['parent']))
    elif 'parent_ds' in inits:
        phase1_candidates.append(('parent_ds', inits['parent_ds']))
    elif 'parent_up' in inits:
        phase1_candidates.append(('parent_up', inits['parent_up']))
    
    # Add a few diverse inits
    for nm in ['uniform', 'bimodal', 'gauss', 'window', 'random']:
        if nm in inits and nm not in [c[0] for c in phase1_candidates]:
            phase1_candidates.append((nm, inits[nm]))
    
    phase1_time = min(700.0, (deadline - time.monotonic()) * 0.4)
    # Allocate more time to parent
    t_parent = phase1_time * 0.5
    t_other = phase1_time * 0.5 / max(1, len(phase1_candidates) - 1)
    
    results = []
    for idx, (name, a0) in enumerate(phase1_candidates):
        if time.monotonic() > deadline - (phase1_time - t_parent):
            break
        if idx == 0:
            t_alloc = min(t_parent, deadline - time.monotonic() - 400)
        else:
            t_alloc = min(t_other, deadline - time.monotonic() - 300)
        if t_alloc < 8:
            break
        
        a_opt, obj_opt = nesterov_subgrad(a0, t_alloc, lr_init=0.02,
                                           momentum_init=0.9, top_k=50,
                                           threshold_frac=0.8,
                                           temperature=0.04,
                                           use_adaptive_lr=True,
                                           perturb_stall=500)
        score = float(evaluate_sequence(a_opt.tolist()))
        results.append((score, name, a_opt.copy()))
        if score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
        print(f"  {name}: score={score:.6f}")
    
    results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")
    
    # Phase 2: LP directional refinement
    print("=== Phase 2: LP Directional Refinement ===")
    top_list = [a for _, _, a in results[:3]]
    if best_sequence is not None:
        top_list.append(best_sequence.copy())
    
    remaining = deadline - time.monotonic() - 350.0
    t_per_lp = max(15.0, remaining / max(1, len(top_list)))
    
    for idx, a0 in enumerate(top_list):
        if time.monotonic() > deadline - 300:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 250)
        if t_alloc < 10:
            break
        
        a_opt, obj_opt = optimize_lp_directional(a0, t_alloc,
                                                  blend_init=0.2,
                                                  blend_min=0.005)
        # Follow with subgradient
        t_ns = min(15.0, t_alloc * 0.3)
        a_opt2, obj2 = nesterov_subgrad(a_opt, t_ns, lr_init=0.008,
                                         momentum_init=0.92, top_k=50,
                                         threshold_frac=0.75,
                                         temperature=0.03)
        if obj2 < obj_opt:
            a_opt, obj_opt = a_opt2, obj2
        
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  LP+SG cand {idx}: {score:.6f}")
    
    # Phase 3: Local pair search
    print("=== Phase 3: Local Pair Search ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 150.0
        t_pair = min(150.0, remaining * 0.6)
        
        a_pa = local_pair_search(best_sequence, t_pair,
                                 max_targets=6,
                                 fracs=[0.02, 0.05, 0.1, 0.15])
        score_pa = float(evaluate_sequence(a_pa.tolist()))
        if np.isfinite(score_pa) and score_pa < best_score:
            best_score = score_pa
            best_sequence = a_pa.copy()
            print(f"  Pair search: {score_pa:.6f}")
        
        # Triple search
        t_triple = min(50.0, remaining * 0.2)
        a_tr = local_triple_search(a_pa if score_pa < float('inf') else best_sequence,
                                   t_triple, max_triples=500)
        score_tr = float(evaluate_sequence(a_tr.tolist()))
        if np.isfinite(score_tr) and score_tr < best_score:
            best_score = score_tr
            best_sequence = a_tr.copy()
            print(f"  Triple search: {score_tr:.6f}")
    
    # Phase 4: Final Nesterov polish (very fine parameters)
    print("=== Phase 4: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 80.0
        for cycle in range(3):
            if time.monotonic() > deadline - 60:
                break
            t_cyc = min(remaining / (3 - cycle), 40.0)
            if t_cyc < 5:
                break
            
            a_opt, obj_opt = nesterov_subgrad(best_sequence, t_cyc,
                                              lr_init=0.004,
                                              momentum_init=0.94,
                                              top_k=60,
                                              threshold_frac=0.7,
                                              temperature=0.02,
                                              perturb_stall=600)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish {cycle}: {score:.6f}")
            
            # Quick pair polish
            a_p2 = local_pair_search(best_sequence, 5.0,
                                     max_targets=4,
                                     fracs=[0.02, 0.05])
            score2 = float(evaluate_sequence(a_p2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best_score = score2
                best_sequence = a_p2.copy()
                print(f"  Pair polish {cycle}: {score2:.6f}")
    
    # Phase 5: Random restarts with small perturbations from best
    if time.monotonic() < deadline - 30 and best_sequence is not None:
        print("=== Phase 5: Random Restarts ===")
        n_b = len(best_sequence)
        t_restarts = deadline - time.monotonic() - 25
        n_restarts = min(6, max(1, int(t_restarts / 20)))
        
        for r in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            t_r = min(15.0, t_restarts / (n_restarts - r))
            if t_r < 5:
                break
            
            noise = rng.standard_normal(n_b) * (0.003 + 0.002 * r)
            a_pert = project_simplex(best_sequence + noise)
            a_opt, obj_opt = nesterov_subgrad(a_pert, t_r, lr_init=0.008,
                                              momentum_init=0.9, top_k=40,
                                              threshold_frac=0.8,
                                              temperature=0.03)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Restart {r}: {score:.6f}")
    
    # Final validation
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(512) / 512
        final_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
