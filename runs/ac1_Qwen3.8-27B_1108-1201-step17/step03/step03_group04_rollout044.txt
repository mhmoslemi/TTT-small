```python
def compute_subgradient(a, conv, top_k=15, temperature=0.05):
    n = len(a)
    # Find top-k peaks
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    
    # Softmax weights
    temp = max(peak_vals.max() * temperature, 1e-15)
    weights = np.exp((peak_vals - peak_vals.max()) / temp)
    weights /= weights.sum()
    
    # Compute subgradient
    grad = np.zeros(n)
    for i, k in enumerate(peaks):
        if weights[i] < 1e-10:
            continue
        # grad_j += 2 * weights[i] * a[k-j] for valid j
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            grad[j] += 2.0 * weights[i] * a[k - j]
    return grad
```

Hybrid subgradient + LP-boosted optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses accelerated entropy-regularized gradient with multi-peak softmax weighting,
multi-scale initialization, periodic LP kicks, and perturbation restarts.
"""

import numpy as np
import time


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_conv_max(a):
    """Compute max of autocorrelation."""
    conv = fast_convolve(a)
    return conv.max()


def compute_obj_local(a):
    """Compute 2*n*max(conv) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * compute_conv_max(a)


def project_simplex(v):
    """Project onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgradient(a, conv, top_k=20, temperature=0.08):
    """Compute weighted subgradient using top-k convolution peaks."""
    n = len(a)
    actual_k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -actual_k)[-actual_k:]
    peak_vals = conv[peaks]

    temp = max(peak_vals.max() * temperature, 1e-15)
    weights = np.exp((peak_vals - peak_vals.max()) / temp)
    weights /= weights.sum()

    grad = np.zeros(n)
    for i in range(actual_k):
        if weights[i] < 1e-12:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            grad[j_start:j_end + 1] += 2.0 * weights[i] * a[k - np.arange(j_start, j_end + 1)]
    return grad


def entropy_gradient_step(a, conv, lr, top_k, temperature):
    """One step of entropy-regularized gradient descent on simplex."""
    g = compute_subgradient(a, conv, top_k, temperature)
    new_a = np.exp(-lr * g)
    s = new_a.sum()
    if s < 1e-30:
        return a
    return new_a / s


def lp_oracle_fast(a, M, time_limit=3.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns None on failure."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def accelerate_optimize(a0, time_limit, lr_init=0.1, momentum=0.9, top_k_init=20,
                        temp_init=0.08, verbose=False, verbose_every=500):
    """
    Accelerated subgradient optimization with adaptive parameters.
    Works on simplex (sum(a)=1).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    v = np.zeros(n)  # momentum
    lr = lr_init
    temp = temp_init
    top_k = top_k_init
    no_improve = 0
    max_stall = 300

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        # Compute subgradient
        g = compute_subgradient(a, conv, top_k, temp)

        # Nesterov momentum step
        v = momentum * v - lr * g
        a_candidate = np.maximum(a + v, 0.0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            a_candidate = np.ones(n) / n
        else:
            a_candidate = a_candidate / sc

        # Evaluate
        conv_new = fast_convolve(a_candidate)
        M_new = conv_new.max()
        obj_new = 2.0 * n * M_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_candidate.copy()
            no_improve = 0
            a = a_candidate
            conv = conv_new
            M = M_new
        else:
            no_improve += 1
            # Try smaller step
            v = momentum * 0.5 * v
            a_candidate = np.maximum(a + v, 0.0)
            sc = a_candidate.sum()
            if sc < 1e-15:
                a_candidate = a  # stay
            else:
                a_candidate = a_candidate / sc
                conv_new = fast_convolve(a_candidate)
                M_new = conv_new.max()
                obj_new = 2.0 * n * M_new
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_candidate.copy()
                    no_improve = 0
                    a = a_candidate
                    conv = conv_new
                    M = M_new

        # Adaptive parameters
        if it % 100 == 0:
            lr *= 0.995
            temp = max(0.02, temp_init * (0.999 ** it))
            top_k = min(40, top_k_init + it // 200)

        # Stall handling: perturbation
        if no_improve >= max_stall:
            rng = np.random.default_rng(it + 12345)
            scale = 0.002 * (1 + 0.5 * (no_improve // max_stall))
            noise = rng.standard_normal(n) * scale
            a_new = np.maximum(a + noise, 0.0)
            a_new = a_new / a_new.sum()
            conv = fast_convolve(a_new)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a_new.copy()
            a = a_new
            v = np.zeros(n)
            no_improve = 0
            max_stall = int(max_stall * 1.5)
            lr = max(0.001, lr * 1.5)

        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"      ACCEL it={it}, obj={best_obj:.7f}, lr={lr:.5f}, "
                  f"temp={temp:.4f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def lp_boost(a0, time_limit, n_kicks=5, verbose=False):
    """Apply LP oracle kicks to improve the solution."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    time_per_kick = time_limit / max(1, n_kicks)

    for kick in range(n_kicks):
        if time.monotonic() - t0 > time_limit:
            break
        b = lp_oracle_fast(a, M, time_limit=min(3.0, time_per_kick * 0.5))
        if b is None:
            break
        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb

        # Line search
        best_t = 0.0
        best_obj_ls = 2.0 * n * conv.max()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            candidate = (1 - t) * a + t * b
            conv_c = fast_convolve(candidate)
            obj_c = 2.0 * n * conv_c.max()
            if obj_c < best_obj_ls:
                best_obj_ls = obj_c
                best_t = t

        if best_t > 0:
            a = (1 - best_t) * a + best_t * b
            conv = fast_convolve(a)
            M = conv.max()
            cur_obj = 2.0 * n * M
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"      LP kick {kick}: obj={best_obj:.7f}, t={best_t}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # U-shape (high at edges)
    u = 1.0 + 0.8 * np.cos(np.pi * x / (n - 1))
    inits['u_shape'] = u / u.sum()

    # W-shape
    w = 1.0 + 0.5 * np.cos(2 * np.pi * x / (n - 1)) + 0.3 * np.cos(4 * np.pi * x / (n - 1))
    w = np.maximum(w, 0.01)
    inits['w_shape'] = w / w.sum()

    # Edge concentrated (top 40%)
    ec = np.ones(n)
    edge_width = max(1, int(0.4 * n))
    ec[:edge_width] = 1.5
    ec[-edge_width:] = 1.5
    inits['edge_40'] = ec / ec.sum()

    # Edge concentrated (top 25%)
    edge_width = max(1, int(0.25 * n))
    ec2 = np.ones(n)
    ec2[:edge_width] = 2.0
    ec2[-edge_width:] = 2.0
    inits['edge_25'] = ec2 / ec2.sum()

    # Gaussian (center)
    g = np.exp(-0.5 * ((x - mid) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Two Gaussians (edges)
    g2 = np.exp(-0.5 * (x / (0.15 * n)) ** 2) + np.exp(-0.5 * ((x - n + 1) / (0.15 * n)) ** 2)
    g2 = np.maximum(g2, 1e-10)
    inits['two_gauss'] = g2 / g2.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 1.0)

    # Random with edge bias
    edge_bias = np.ones(n) * 0.3
    edge_width = max(1, n // 4)
    edge_bias[:edge_width] = 1.0
    edge_bias[-edge_width:] = 1.0
    rb = rng.dirichlet(edge_bias)
    inits['rand_edge'] = rb

    # Parent interpolation
    if parent is not None:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(x, np.linspace(0, n - 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps

    return inits

    Multi-phase hybrid optimization:
    Phase 1: Quick scan of initializations at multiple lengths
    Phase 2: Deep subgradient optimization on best candidate
    Phase 3: LP-boost refinement
    Phase 4: Perturbation restarts and final polish
    """

    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)

            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent score: {parent_score:.6f}")
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = parent.copy()
    except Exception:
        pass

    # === Phase 1: Scan initializations ===
    print("=== Phase 1: Initialization Scan ===")
    lengths_to_try = [384, 512, 256]
    # Prioritize parent length
    if parent is not None:
        pl = len(parent)
        if pl not in lengths_to_try:
            lengths_to_try.insert(0, pl)
        else:
            lengths_to_try.remove(pl)
            lengths_to_try.insert(0, pl)

    t0 = time.monotonic()
    for n in lengths_to_try:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic() - 60
        inits = generate_inits(n, parent, rng)
        init_names = list(inits.keys())

        # Prioritize
        priority = ['parent', 'parent_interp', 'u_shape', 'w_shape',
                    'edge_25', 'edge_40', 'two_gauss', 'gauss',
                    'uniform', 'rand_edge', 'random']
        init_names.sort(key=lambda x: priority.index(x) if x in priority else 99)

        # Quick evaluation of each init
        for name in init_names[:8]:
            if time.monotonic() > deadline - 90:
                break
            a0 = inits[name]
            s = a0.sum()
            if s < 1e-10:
                continue
            a0 = a0 / s
            score = float(evaluate_sequence(a0.tolist()))
            print(f"  n={n}, init={name}, score={score:.6f}")

            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a0.copy()

        # Quick optimization of best init for this length
        if time.monotonic() < deadline - 60:
            best_init_for_n = None
            best_init_score = float('inf')
            for name in init_names[:6]:
                a0 = inits[name]
                s = a0.sum()
                if s < 1e-10:
                    continue
                a0 = a0 / s
                local_obj = compute_obj_local(a0)
                if local_obj < best_init_score:
                    best_init_score = local_obj
                    best_init_for_n = a0.copy()

            if best_init_for_n is not None:
                t_quick = min(30.0, (deadline - time.monotonic() - 60) / max(1, len(lengths_to_try) - lengths_to_try.index(n)))
                if t_quick > 5:
                    a_opt, obj_opt = accelerate_optimize(best_init_for_n, time_limit=t_quick,
                                                         lr_init=0.05, momentum=0.85,
                                                         top_k_init=15, temp_init=0.1)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    print(f"  n={n}, optimized, score={score:.6f}, local={obj_opt:.6f}")
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()

    print(f"After Phase 1: best={best_score:.6f}, n={len(best_sequence) if best_sequence is not None else 0}")

    if best_sequence is None:
        best_sequence = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # === Phase 2: Deep subgradient optimization ===
    print("=== Phase 2: Deep Optimization ===")
    remaining = deadline - time.monotonic() - 120.0
    t_deep = remaining * 0.45
    if t_deep > 10:
        n_cur = len(best_sequence)
        a_opt, obj_opt = accelerate_optimize(best_sequence, time_limit=t_deep,

                                             top_k_init=25, temp_init=0.06,

        score = float(evaluate_sequence(a_opt.tolist()))
        print(f"  Deep score={score:.6f}, local={obj_opt:.6f}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()

    # === Phase 3: LP-boost refinement ===
    print("=== Phase 3: LP-Boost ===")
    remaining = deadline - time.monotonic() - 60.0
    if remaining > 15:
        t_lp = remaining * 0.3
        n_cur = len(best_sequence)
        if n_cur <= 512:  # LP is expensive for large n
            a_opt, obj_opt = lp_boost(best_sequence, time_limit=t_lp,
                                      n_kicks=max(3, int(t_lp / 5)), verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            print(f"  LP-boost score={score:.6f}")
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
        else:
            # For large n, do a few quick LP kicks
            a_opt, obj_opt = lp_boost(best_sequence, time_limit=min(t_lp, 30),
                                      n_kicks=2, verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            print(f"  LP-quick score={score:.6f}")
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()

    # === Phase 4: Subgradient polish after LP ===
    print("=== Phase 4: Post-LP Polish ===")
    remaining = deadline - time.monotonic() - 40.0
    if remaining > 10:
        t_polish = remaining * 0.5
        a_opt, obj_opt = accelerate_optimize(best_sequence, time_limit=t_polish,
                                             lr_init=0.01, momentum=0.92,
                                             top_k_init=30, temp_init=0.04)
        score = float(evaluate_sequence(a_opt.tolist()))
        print(f"  Polish score={score:.6f}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()

    # === Phase 5: Perturbation restarts ===
    print("=== Phase 5: Perturbation Restarts ===")
    remaining = deadline - time.monotonic() - 20.0
    n_cur = len(best_sequence)
    n_restarts = max(1, int(remaining // 12))
    n_restarts = min(n_restarts, 6)

    for i in range(n_restarts):
        if time.monotonic() > deadline - 15:
            break
        scale = 0.003 * (1 + i * 0.3)
        noise = rng.standard_normal(n_cur) * scale
        a_pert = np.maximum(best_sequence + noise, 0.0)
        a_pert = a_pert / a_pert.sum()
        t_alloc = min(10.0, deadline - time.monotonic() - 12)
        if t_alloc < 3:
            break
        a_opt, obj_opt = accelerate_optimize(a_pert, time_limit=t_alloc,
                                             lr_init=0.02, momentum=0.9,
                                             top_k_init=25, temp_init=0.05)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Restart {i}: score={score:.6f} (IMPROVED!)")
            else:
                print(f"  Restart {i}: score={score:.6f}")

    # === Phase 6: Final micro-optimize ===
    if time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 3.0
        if remaining > 3:
            a_opt, obj_opt = accelerate_optimize(best_sequence, time_limit=remaining,
                                                 lr_init=0.005, momentum=0.93,
                                                 top_k_init=35, temp_init=0.03)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final micro: score={score:.6f}")

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
