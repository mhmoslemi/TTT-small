def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    # For each output k, the non-zero entries are at positions j where a[k-j] != 0
    # But for sparsity, we include all possible positions
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

The linear cross-correlation of x (length p) and y (length q) is:

This can be computed as:
1. Zero-pad x to length N >= p + q - 1
2. Zero-pad y to length N
3. Compute ifft(FFT(x) * conj(FFT(y)))
4. Take the first p + q - 1 values

Actually, the standard formula is:
Circular cross-correlation: (x ⊛_c y)[k] = sum_j x[(j+k) mod N] * conj(y[j])

If y is real: (x ⊛_c y)[k] = sum_j x[(j+k) mod N] * y[j]

In frequency domain: DFT(x ⊛_c y) = DFT(x) * conj(DFT(y))

For linear cross-correlation, we zero-pad to avoid circular aliasing.

So: ifft(FFT(x_padded) * conj(FFT(y_padded))) gives the linear cross-correlation.

```python
"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines soft-max smoothed gradient descent with LP-based Frank-Wolfe
and coordinate descent polishing, exploring multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def softmax_compute_grad(a, beta):
    """
    Compute gradient of g(a) = log(sum_k exp(beta * C_k)) where C = conv(a,a).
    Returns (gradient, C, g_value).
    """
    n = len(a)
    C = fast_convolve(a)
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    Z = exp_C.sum()
    w = exp_C / Z
    g_val = np.log(Z) + beta * max_C
    
    # grad[i] = 2*beta * sum_{j=0}^{n-1} w[i+j] * a[j]
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    xcorr = np.fft.irfft(W * np.conj(A), fft_size)[:size]
    grad = 2.0 * beta * xcorr[:n]
    
    return grad, C, g_val


def softmax_momentum_descent(a0, time_limit, beta_schedule=None, verbose=True):
    """
    Momentum-based gradient descent on the soft-max smoothed objective,
    with beta annealing.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    C = fast_convolve(a)
    best_obj = 2.0 * n * C.max()
    best_a = a.copy()
    
    if beta_schedule is None:
        beta_schedule = [5.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]
    
    t0 = time.monotonic()
    
    for phase, beta in enumerate(beta_schedule):
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 1.0:
            break
        
        # Allocate time: later phases get more time
        n_phases_left = len(beta_schedule) - phase
        phase_time = remaining * (2.0 / (n_phases_left + 1))
        # Cap early phases
        phase_time = min(phase_time, remaining * 0.4)
        
        if verbose:
            print(f"  SM Phase {phase}: beta={beta:.0f}, time={phase_time:.0f}s")
        
        # Momentum parameters
        v = np.zeros(n)
        momentum = 0.9
        lr = 1.0 / (beta * n)  # Conservative initial lr
        
        phase_start = time.monotonic()
        it = 0
        no_improve = 0
        
        while time.monotonic() - phase_start < phase_time:
            # Compute gradient
            grad, C, g_val = softmax_compute_grad(a, beta)
            
            # Project to simplex tangent space
            grad = grad - grad.mean()
            
            # Momentum update
            v = momentum * v - lr * grad
            a_new = np.maximum(a + v, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = np.ones(n) / n
            else:
                a_new = a_new / s_new
            
            # Evaluate
            C_new = fast_convolve(a_new)
            max_new = C_new.max()
            obj_new = 2.0 * n * max_new
            
            if obj_new <= best_obj * 1.01 + 1e-10:
                a = a_new
                if obj_new < best_obj - 1e-15:
                    best_obj = obj_new
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                # Step too large, reduce
                v = 0.5 * v
                lr *= 0.5
                no_improve += 1
            
            # Adaptive momentum: reduce when stuck
            if no_improve > 50:
                momentum = max(0.5, momentum * 0.95)
                v = 0.5 * v
                no_improve = 0
            
            lr *= 0.9995
            it += 1
        
        if verbose:
            C_final = fast_convolve(a)
            print(f"    Phase {phase} done: obj={2.0*n*C_final.max():.6f}, best={best_obj:.6f}, iters={it}")
    
    return best_a, best_obj


def lp_oracle(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None
