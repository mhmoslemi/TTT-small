def fast_convolve(a):

    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
with golden-section line search, coordinate descent polishing, and multi-length
exploration. Key improvements over fixed-step LP: optimal step size via 1D
convex optimization, dual LP formulations, and aggressive polynomial-time
convergence exploitation.

import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a[k_start - j:k_end - j]
    return A


def line_search_peak(a, b, t_max=1.0, n_evals=28):
    """Golden section search to minimize max(conv((1-t)*a+t*b, ...)) over t in [0, t_max]."""
    def f(t):
        a_new = (1.0 - t) * a + t * b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            return 1e10
        a_new = a_new / s
        conv = fast_convolve(a_new)
        return conv.max()

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = t_max
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)

    for _ in range(n_evals):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)

    t_opt = (lo + hi) / 2.0
    return t_opt, f(t_opt)


def lp_max_sum(a, M, time_limit_sec=10):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns normalized b or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit_sec})
    if res.status == 0:
        b = np.maximum(res.x, 0.0)
        s = b.sum()
        if s > 1e-15:
            return b / s
    return None

def subgradient_refine(a, time_limit=30.0, top_k=20):
    """Subgradient method with momentum for peak minimization."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    lr = 0.05
    momentum = 0.9
    no_improve = 0
    rng = np.random.default_rng(77)

    while time.monotonic() - t0 < time_limit:
        # Subgradient: pick top-k peaks and average their gradients
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))

        sg = np.zeros(n)
        for k in peaks:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += a[k - idx]
        sg *= 2.0 / len(peaks)

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            lr *= 0.7
            no_improve = 0
            if lr < 0.001:
                # Perturb
                a = project_simplex(a + rng.standard_normal(n) * 0.01)
                v *= 0.1
                lr = 0.03
        if no_improve > 2000:
            break

        lr *= 0.9995

    return best_a

            inits[f'cos_{freq}'] = a / ss

    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Triangular
    tri = 1.0 - np.abs(x / n - 0.5) * 2.0
    tri = np.maximum(tri, 0.0)
    inits['tri'] = tri / tri.sum()

    return inits

    best_score = float('inf')

    try:

        pass

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return best_score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  New best: {score:.6f} (n={len(a)})")
        return score

    print("=== Phase 1: LP + Line Search (primary length) ===")

    else:
        primary_n = 512
