Let me also think about what makes the LP approach work well and how to improve it.

The LP oracle solves: max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0.

The key insight is that at the optimum, b will be concentrated where a has small values (to keep conv(a,b) small). This makes b a "dual" of a in some sense.

One improvement: instead of always using the same M, we can use a slightly larger M (M * (1 + epsilon)) to give the LP more freedom, which might lead to better directions.

Another improvement: use a "warm start" for the LP by using the previous b as the initial guess.

Let me also think about the **active set approach**:
1. At the current point a, identify the set S of "active" convolution positions (where conv(a,a)[k] is close to the max).
2. Solve a smaller optimization problem that only considers these constraints.
3. This is like a reduced LP.

For the reduced LP: max sum(b) s.t. conv(a,b)[k] <= M for k in S, b >= 0.
This has |S| constraints instead of 2n-1, which is much faster.

If |S| is small (say 5-20), this LP is very fast.

Let me now code up the full solution:

```python
"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines Nesterov-accelerated multi-peak subgradient, reduced LP active-set
refinement, and spectral reparameterization for fast convergence to near-optimal
non-negative sequences with low autocorrelation peaks.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_convolve(a, b=None):
    n = len(a)
    m = len(b) if b is not None else n
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b if b is not None else a, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()

def nesterov_subgradient(a0, time_limit, n_peaks=30, lr_init=0.005, 
                          momentum_init=0.9, verbose=False):
    """Nesterov-accelerated subgradient with multi-peak softmax."""
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    lr = lr_init
    momentum = momentum_init
    t0 = time.monotonic()
    it = 0
    prev_best = best_obj
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        M = c.max()
        obj = 2.0 * n * M
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Multi-peak subgradient
        k_peaks = min(n_peaks, 2 * n - 1)
        top_idx = np.argpartition(c, -k_peaks)[-k_peaks:]
        
        # Softmax weights with adaptive temperature
        progress = min(1.0, it / max(1, int(time_limit / 0.005)))
        temp = M * (0.05 + 0.2 * (1.0 - progress))
        w = np.exp(c[top_idx] / temp)
        w = w / w.sum()
        
        # Compute subgradient
        g = np.zeros(n)
        for i in range(len(top_idx)):
            if w[i] < 1e-12:
                continue
            k = int(top_idx[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            j = np.arange(j_start, j_end + 1)
            u = k - j
            g[j] += 2.0 * w[i] * a[u]
        
        # Nesterov update
        v = momentum * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
            v = np.zeros(n)
        else:
            a_new = a_new / s
        a = a_new
        
        # Adaptive parameters
        if it > 0 and it % 200 == 0:
            lr *= 0.995
            if M > prev_best * 1.02:
                # Restart momentum
                v = np.zeros(n)
                momentum = max(0.5, momentum * 0.9)
            else:
                momentum = min(0.99, momentum * 1.002)
            prev_best = M
        
        it += 1
        if verbose and it % 500 == 0:
            print(f"      NS it={it}, obj={obj:.6f}, lr={lr:.6f}, mom={momentum:.4f}")
    
    return best_a, best_obj

def reduced_lp_oracle(a, M, active_set=None, max_active=30):
    """LP: max sum(b) s.t. conv(a,b)[k] <= M for k in active_set, b >= 0."""
    n = len(a)
    if active_set is None:
        c = fast_convolve(a)
        threshold = M * 0.98
        active_set = np.where(c >= threshold)[0]
        if len(active_set) > max_active:
            # Keep only the top max_active
            top = np.argpartition(c, -max_active)[-max_active:]
            active_set = top
    
    # Build sparse matrix for active constraints
    rows = []
    cols = []
    vals = []
    for ii, k in enumerate(active_set):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(ii)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_set), n))
    c_obj = -np.ones(n)
    b_ub = np.full(len(active_set), M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': 10.0})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def full_lp_oracle(a, M):
    """LP with all constraints (slower but more accurate)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': 20.0})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def fw_iteration(a0, time_limit, use_full_lp=False, max_active=30, verbose=False):
    """Frank-Wolfe with LP oracle (reduced or full)."""
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    c = fast_convolve(a)
    M = c.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        M = c.max()
        
        if use_full_lp:
            b = full_lp_oracle(a, M)
        else:
            b = reduced_lp_oracle(a, M, max_active=max_active)
        
        if b is None:
            break
        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb
        
        if np.allclose(a, b, atol=1e-10):
            stall += 1
        else:
            # Line search
            best_t = 0.0
            best_o = 2.0 * n * M
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                cand = (1 - t) * a + t * b
                o = compute_obj(cand)
                if o < best_o:
                    best_o = o
                    best_t = t
            
            if best_t > 0.01:
                a = (1 - best_t) * a + best_t * b
                if best_o < best_obj - 1e-13:
                    best_obj = best_o
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        
        if stall >= 50:
            # Perturb and reset
            rng = np.random.default_rng(it)
            noise = rng.standard_normal(n) * 0.005
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            stall = 0
        
        it += 1
        if verbose and it % 100 == 0:
            print(f"      FW it={it}, obj={best_obj:.6f}, M={M:.6e}")
    
    return best_a, best_obj

def taper_init(n, alpha=2.0, beta=2.0):
    """Create a tapered initialization (power-law edges)."""
    x = np.arange(n, dtype=np.float64)
    # Power-law taper on both ends
    edge = max(1, n // 4)
    left = np.minimum(1.0, (x + 1) / edge) ** alpha
    right = np.minimum(1.0, (n - x) / edge) ** beta
    a = left * right
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    return a

def alternating_minimization(a0, time_limit, n_peaks=20, verbose=False):
    """
    Alternating: fix the "shape" and optimize the peak-constraining subset.
    Uses a projected gradient approach where we alternate between
    reducing peak contributions and maintaining sum.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    a = a / a.sum()
    
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    lr = 0.01
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        M = c.max()
        obj = 2.0 * n * M
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        
        # Find top peaks and compute "pressure" at each position
        k_peaks = min(n_peaks, 2 * n - 1)
        top_idx = np.argpartition(c, -k_peaks)[-k_peaks:]
        
        temp = M * 0.1
        w = np.exp(c[top_idx] / temp)
        w = w / w.sum()
        
        # Compute pressure (how much each position contributes to peaks)
        pressure = np.zeros(n)
        for i in range(len(top_idx)):
            if w[i] < 1e-12:
                continue
            k = int(top_idx[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            j = np.arange(j_start, j_end + 1)
            pressure[j] += w[i] * a[k - j]
        
        # Reduce high-pressure positions, increase low-pressure
        # Use a "water-filling" style update
        a_new = a.copy()
        # Subtract pressure (proportionally to a to maintain non-negativity)
        a_new -= lr * pressure * (a / max(a.max(), 1e-15))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        if it % 500 == 0 and it > 0:
            lr *= 0.98
        
        it += 1
    
    return best_a, best_obj
