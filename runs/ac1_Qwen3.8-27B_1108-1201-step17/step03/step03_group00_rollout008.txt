import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(a, conv, beta):
    """Compute gradient of log-sum-exp smoothed objective.
    L(a) = logsumexp(beta * conv) / beta
    grad_j = sum_k p_k * 2 * a[k-j] for valid k-j
    Uses FFT for efficiency."""
    n = len(a)
    # Compute softmax weights
    conv_shifted = conv - conv.max()
    exp_conv = np.exp(beta * conv_shifted)
    exp_sum = exp_conv.sum()
    if exp_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_conv / exp_sum

    # grad[j] = 2 * sum_{i=0}^{n-1} a[i] * w[j+i] for j=0,...,n-1
    # This is cross-correlation of a and w at lags 0..n-1
    # Use FFT: conv(a, w_flipped)[2n-2-j] gives the answer
    w_flipped = w[::-1]
    fft_size = 1 << (3 * n - 2 + 31).bit_length()
    A_fft = np.fft.rfft(a, fft_size)
    W_fft = np.fft.rfft(w_flipped, fft_size)
    conv_full = np.fft.irfft(A_fft * W_fft, fft_size)
    # grad[j] = 2 * conv_full[2n-2-j] for j = 0, ..., n-1
    grad = 2.0 * conv_full[2 * n - 2 - np.arange(n)][::-1]
    return grad

def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g or None."""
    n = len(f)
    if n > 800:
        return None  # Too large for efficient LP
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def generate_parametric_inits(n, rng):
    """Generate diverse parametric initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xc = n / 2.0  # center

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangular (ramp up, ramp down)
    tri = np.minimum(x + 1, n - x)
    inits.append(tri / tri.sum())

    # Quadratic
    quad = np.minimum((x + 1) ** 2, (n - x) ** 2)
    inits.append(quad / quad.sum())

    # Gaussian
    gauss = np.exp(-0.5 * ((x - xc) / (0.25 * n)) ** 2)
    inits.append(gauss / gauss.sum())

    # Raised cosine
    cos_init = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_init = cos_init + 0.1
    inits.append(cos_init / cos_init.sum())

    # Edge-concentrated (inverse of triangular)
    edge = 1.0 / (np.minimum(x + 1, n - x) + 0.5)
    inits.append(edge / edge.sum())

    # Random (Dirichlet)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Two-bump
    bump1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2)
    bump2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2)
    inits.append((bump1 + bump2) / (bump1 + bump2).sum())

    # Spectral: |FFT of random phase|
    freq = np.fft.rfftfreq(n)
    mask = np.ones(len(freq))
    mask[0] = 0  # remove DC
    phase = rng.uniform(0, 2 * np.pi, len(freq))
    spec = mask * np.exp(1j * phase)
    sig = np.fft.irfft(spec * n, n)
    sig = np.abs(sig) + 1e-10
    inits.append(sig / sig.sum())

    return inits

    # Parametric inits at parent length
    for i, init in enumerate(generate_parametric_inits(n_parent, rng)):
        candidates.append((f'param_{i}', init))

    # Evaluate all candidates
    for label, cand in candidates:
        try_seq(cand, label)

    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Smoothed mirror descent on best ===
    print(f"\n=== Phase 2: Smoothed Mirror Descent ===")
    t_phase2 = min(400, deadline - time.monotonic() - 600)
    if t_phase2 > 30 and best_seq is not None:
        a_opt, obj = smoothed_mirror_descent(best_seq, time_limit=t_phase2,
                                             beta_start=100, beta_end=50000,
                                             lr_init=0.03, lr_decay=0.9997,
                                             mom_init=0.85, verbose=True)
        try_seq(a_opt, "SMD_p2")

    # Also try on parent with different params
    if parent is not None and time.monotonic() < deadline - 500:
        t_alloc = min(150, deadline - time.monotonic() - 500)
        if t_alloc > 20:
            a_opt, obj = smoothed_mirror_descent(parent, time_limit=t_alloc,
                                                 beta_start=50, beta_end=100000,
                                                 lr_init=0.05, lr_decay=0.999,
                                                 mom_init=0.8, verbose=True)
            try_seq(a_opt, "SMD_parent2")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Try different lengths ===
    print(f"\n=== Phase 3: Multi-length Exploration ===")
    t_phase3_end = time.monotonic() + min(300, deadline - time.monotonic() - 400)

    lengths_to_try = [384, 512, 256, 640, 448]
    if parent is not None:
        if len(parent) not in lengths_to_try:
            lengths_to_try.insert(0, len(parent))

    for n_try in lengths_to_try:
        if time.monotonic() > t_phase3_end:
            break
        if n_try == (len(parent) if parent is not None else -1):
            continue  # Already tried

        inits = generate_parametric_inits(n_try, rng)
        # Pick a few promising inits
        for i, init in enumerate(inits[:4]):
            if time.monotonic() > t_phase3_end - 60:
                break
            t_alloc = min(60, t_phase3_end - time.monotonic() - 60)
            if t_alloc < 10:
                break
            a_opt, _ = smoothed_mirror_descent(init, time_limit=t_alloc,
                                               beta_start=100, beta_end=50000,
                                               lr_init=0.02, lr_decay=0.9995,
                                               mom_init=0.85)
            try_seq(a_opt, f"n{n_try}_i{i}")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Deep optimization with LP-FW on best ===
    print(f"\n=== Phase 4: LP Frank-Wolfe Deep Optimization ===")
    if best_seq is not None:
        n_best = len(best_seq)
        t_phase4_end = time.monotonic() + min(300, deadline - time.monotonic() - 150)

        # LP-FW on best
        t_alloc = min(120, t_phase4_end - time.monotonic() - 100)
        if t_alloc > 20:
            a_opt, _ = lp_frank_wolfe(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "LPFW_best")

        # Perturbed LP-FW
        if time.monotonic() < t_phase4_end - 50:
            noise = rng.standard_normal(n_best) * 0.01
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(80, t_phase4_end - time.monotonic() - 50)
            if t_alloc > 20:
                a_opt, _ = lp_frank_wolfe(f_pert, time_limit=t_alloc)
                try_seq(a_opt, "LPFW_pert")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Peak-directed polishing ===
    print(f"\n=== Phase 5: Peak-Directed Polish ===")
    if best_seq is not None:
        t_phase5_end = time.monotonic() + min(150, deadline - time.monotonic() - 60)

        # SMD with high beta (sharpening)
        t_alloc = min(80, t_phase5_end - time.monotonic() - 60)
        if t_alloc > 15:
            a_opt, _ = smoothed_mirror_descent(best_seq, time_limit=t_alloc,
                                               beta_start=5000, beta_end=200000,
                                               lr_init=0.005, lr_decay=0.999,
                                               mom_init=0.9, verbose=True)
            try_seq(a_opt, "SMD_sharp")

        # Peak-directed
        t_alloc = min(60, t_phase5_end - time.monotonic() - 20)
        if t_alloc > 10:
            a_opt, _ = peak_directed_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_polish")

    print(f"  Phase 5 best: {best_score:.8f}")

    # === Phase 6: Final fine-tuning ===
    print(f"\n=== Phase 6: Final Fine-Tuning ===")
    if best_seq is not None:
        n_best = len(best_seq)
        t_phase6_end = deadline - 10

        # Very fine SMD
        t_alloc = min(60, t_phase6_end - time.monotonic() - 30)
        if t_alloc > 10:
            a_opt, _ = smoothed_mirror_descent(best_seq, time_limit=t_alloc,
                                               beta_start=20000, beta_end=500000,
                                               lr_init=0.002, lr_decay=0.998,
                                               mom_init=0.93)
            try_seq(a_opt, "SMD_fine")

        # Tiny perturbations + quick SMD
        for p in range(5):
            rem = t_phase6_end - time.monotonic() - 5
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.002
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(20, rem - 5)
            if t_alloc < 8:
                break
            a_opt, _ = smoothed_mirror_descent(f_pert, time_limit=t_alloc,
                                               beta_start=10000, beta_end=100000,
                                               lr_init=0.003, lr_decay=0.998,
                                               mom_init=0.9)
            try_seq(a_opt, f"fine_pert_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
