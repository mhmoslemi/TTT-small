"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Diverse initializations (bimodal, edge-concentrated, flat-spectrum, random)
2. Riemannian projected gradient with Nesterov momentum on the simplex
3. L-BFGS-B with softmax reparametrization and adaptive temperature
4. Problem-specific peak-shaving heuristics
5. Multi-length exploration with time-budgeted phases
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max (log-sum-exp) and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_smooth_conv(a, w):
    """Gradient of sum_k w[k]*conv[k] w.r.t. a via FFT."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    # The gradient is: sum_k w[k]*(a[k-i] + a[i-k]) = 2*cross_correlation(w, a)
    # cross_correlation(w, a)[i] = sum_k w[k] * a[k-i] (circular)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def compute_grad_conv_at_lag(a, k):
    """Exact gradient of conv[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        gi = 0.0
        if 0 <= k - i < n:
            gi += a[k - i]
        if 0 <= i - k < n:
            gi += a[i - k]
        grad[i] = gi
    return grad


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_obj(x, n, beta):
    """Objective for L-BFGS-B (softmax parametrization)."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def _softmax_jac(x, n, beta):
    """Jacobian for L-BFGS-B (softmax parametrization)."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_grad_smooth_conv(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def optimize_lbfgsb(a0, beta, maxiter=300):
    """Optimize using L-BFGS-B with softmax reparametrization."""
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)
    result = minimize(_softmax_obj, x0, jac=_softmax_jac, args=(n, beta),
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 30})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt


def optimize_pg_nesterov(a0, beta, lr=0.05, n_iter=500, momentum=0.9):
    """Projected gradient descent with Nesterov momentum."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    v = np.zeros(n)
    best_obj = float('inf')
    best_a = a.copy()

    for it in range(n_iter):
        # Nesterov lookahead
        a_look = a - v
        a_look = np.maximum(a_look, 0)
        s = a_look.sum()
        if s > 1e-15:
            a_look = a_look / s
        else:
            a_look = a.copy()

        conv = fast_convolve(a_look)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm

        if obj < best_obj:
            best_obj = obj
            best_a = a_look.copy()

        if sm < 1.0 / (2.0 * n) * 1.01:
            break

        grad_a = compute_grad_smooth_conv(a_look, w)

        # Gradient of the objective on the simplex (project out the mean)
        grad_proj = grad_a - np.mean(grad_a)

        # Adaptive learning rate
        norm_grad = np.linalg.norm(grad_proj)
        if norm_grad < 1e-15:
            break

        t = lr * (n / (norm_grad + 1e-10)) * 0.01
        t = min(t, 0.5)

        # Update with Nesterov
        v_new = momentum * v - t * grad_proj
        a_new = project_simplex(a - v_new)
        v = v_new

        # Decay
        lr *= 0.999

    return best_a, best_obj


def optimize_pg_exact(a0, n_iter=2000, lr=0.1):
    """Projected gradient using exact subgradient of max (at argmax lag)."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_obj = float('inf')
    best_a = a.copy()

    for it in range(n_iter):
        conv = fast_convolve(a)
        max_k = int(np.argmax(conv))
        max_val = conv[max_k]
        obj = 2.0 * n * max_val

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Exact subgradient at the peak lag
        grad = compute_grad_conv_at_lag(a, max_k)
        grad_proj = grad - np.mean(grad)

        norm_grad = np.linalg.norm(grad_proj)
        if norm_grad < 1e-15:
            break

        t = lr / (norm_grad + 1e-10)
        t = min(t, 0.1)

        a = project_simplex(a - t * grad_proj)
        lr *= 0.9995

    return best_a, best_obj


def peak_shave(a, n_steps=50, alpha=0.1):
    """Problem-specific: reduce elements contributing most to the current peak."""
    n = len(a)
    a = a.copy()
    for _ in range(n_steps):
        conv = fast_convolve(a)
        # Find top few peaks
        top_k = np.argsort(conv)[::-1][:5]
        for k in top_k:
            if conv[k] < conv.max() * 0.9:
                break
            # Sensitivity of a[i] to conv[k] is a[k-i] (if in range)
            sens = np.zeros(n)
            for i in range(n):
                if 0 <= k - i < n:
                    sens[i] += a[k - i]
                if 0 <= i - k < n:
                    sens[i] += a[i - k]
            # Reduce high-sensitivity elements, keep low-sensitivity
            total_sens = sens.sum()
            if total_sens < 1e-15:
                break
            # Distribute reduction proportional to sensitivity
            reduction = alpha * sens / total_sens * a * (conv[k] / max(conv.max(), 1e-10))
            a = np.maximum(a - reduction, 0.0)
            s = a.sum()
            if s > 1e-10:
                a /= s
            else:
                a = np.ones(n) / n
                return a
    return a


def water_filling_refine(a, n_iter=100):
    """Iteratively equalize the top convolution peaks."""
    n = len(a)
    a = a.copy()
    for it in range(n_iter):
        conv = fast_convolve(a)
        max_val = conv.max()
        mean_val = conv.mean()
        if max_val / max(mean_val, 1e-15) < 1.05:
            break
        # Find all "hot" lags (above 90% of max)
        hot_threshold = conv.max() * 0.9
        hot_lags = np.where(conv >= hot_threshold)[0]
        if len(hot_lags) == 0:
            break
        # Build sensitivity matrix for hot lags
        total_sens = np.zeros(n)
        for k in hot_lags:
            for i in range(n):
                if 0 <= k - i < n:
                    total_sens[i] += a[k - i] / max(len(hot_lags), 1)
                if 0 <= i - k < n:
                    total_sens[i] += a[i - k] / max(len(hot_lags), 1)
        s = total_sens.sum()
        if s < 1e-15:
            break
        # Reduce proportionally to total sensitivity
        lr = 0.05
        a = np.maximum(a - lr * total_sens / s * a, 0.0)
        ss = a.sum()
        if ss > 1e-10:
            a /= ss
        else:
            break
    return a


def bimodal_init(n, sharpness=2.0):
    """Bimodal (edge-concentrated) initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-sharpness * (2 * x - 1) ** 2 * n / 4)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def edge_init(n, power=2.0):
    """Edge-weighted initialization: a[i] proportional to (dist from center)^power."""
    x = np.abs(np.arange(n) - (n - 1) / 2.0) / ((n - 1) / 2.0 + 1e-10)
    a = x ** power
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def flat_spectrum_init(n, rng):
    """Initialize with random phases for approximately flat power spectrum."""
    fft_size = 1 << (n - 1).bit_length()
    best_a = np.ones(n) / n
    best_obj = float('inf')
    for trial in range(30):
        A = np.zeros(fft_size, dtype=complex)
        A[0] = np.sqrt(n)
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2)
        for k in range(1, fft_size // 2):
            A[k] = np.exp(1j * phases[k - 1])
            A[fft_size - k] = np.exp(-1j * phases[k - 1])
        if fft_size % 2 == 0:
            A[fft_size // 2] = 1.0
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            conv = fast_convolve(a)
            obj = 2.0 * n * conv.max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    return best_a


def cosine_taper_init(n, rng, alpha=0.5):
    """Cosine taper blended with random."""
    x = np.cos(np.pi * np.arange(n) / (n - 1 + 1e-10))
    a = np.abs(x) ** (1.0 / max(alpha, 0.01))
    # Blend with random for diversity
    r = rng.dirichlet(np.ones(n) * 0.3)
    a = alpha * a / a.sum() + (1 - alpha) * r
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    inits['uniform'] = np.ones(n) / n
    inits['bimodal'] = bimodal_init(n, 2.0)
    inits['bimodal_sharp'] = bimodal_init(n, 4.0)
    inits['edge2'] = edge_init(n, 2.0)
    inits['edge4'] = edge_init(n, 4.0)
    inits['flat_spec'] = flat_spectrum_init(n, rng)
    inits['cos_taper'] = cosine_taper_init(n, rng, 0.5)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.1)

    # Triangular (inverted: low at edges, high in middle) - for comparison
    x = np.arange(n, dtype=np.float64)
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['tri_center'] = tri / tri.sum()

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()

    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_resized'] = pp / ps
    return inits


def score_and_track(a, best_state):
    """Evaluate and update best state. Returns score."""
    if a is None:
        return float('inf')
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_state['score']:
        best_state['score'] = score
        best_state['seq'] = a.copy()
        best_state['n'] = len(a)
    return score if np.isfinite(score) else float('inf')


def anneal_beta_optimize(a0, betas, maxiter_per_beta=200, time_budget=60.0):
    """Multi-beta L-BFGS-B with annealing."""
    t_start = time.monotonic()
    a = a0.copy()
    for beta in betas:
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 3:
            break
        mi = int(min(maxiter_per_beta, max(30, remaining * 20)))
        a = optimize_lbfgsb(a, beta, maxiter=mi)
    return a


def nesterov_then_lbfgsb(a0, time_budget=60.0):
    """Hybrid: Nesterov PG then L-BFGS-B polish."""
    t_start = time.monotonic()
    elapsed_check = lambda: time.monotonic() - t_start

    # Phase A: Nesterov PG with medium beta
    t_alloc_a = time_budget * 0.5
    n_iter_a = int(min(2000, t_alloc_a * 100))
    a1, _ = optimize_pg_nesterov(a0, beta=30.0, lr=0.08, n_iter=n_iter_a)

    # Phase B: L-BFGS-B with decreasing beta
    remaining = time_budget - elapsed_check()
    if remaining > 5:
        betas = [20.0, 10.0, 5.0, 3.0]
        t_per = remaining / len(betas)
        a_final = a1.copy()
        for beta in betas:
            if t_per < 2:
                break
            mi = int(min(200, max(20, t_per * 25)))
            a_final = optimize_lbfgsb(a_final, beta, maxiter=mi)
        a1 = a_final

    return a1

        alt_lengths = [l for l in [n_cur // 2, n_cur * 2, n_cur + 64, n_cur - 64]
                       if 64 <= l <= 2048 and l != n_cur]
        t4_end = deadline - 30.0

        for alt_n in alt_lengths[:3]:
            if time.monotonic() > t4_end - 30:
                break
            # Resize best to alt_n
            a_resized = np.interp(np.linspace(0, 1, alt_n),
                                  np.linspace(0, 1, n_cur), best_state['seq'])
            a_resized = np.maximum(a_resized, 0)
            sr = a_resized.sum()
            if sr > 1e-10:
                a_resized /= sr
            # Optimize
            a_opt = nesterov_then_lbfgsb(a_resized, time_budget=min(40.0, t4_end - time.monotonic() - 10))
            score_and_track(a_opt, best_state)

    # === Phase 5: Final polish ===
    print("\n=== Phase 5: Final Polish ===")
    if best_state['seq'] is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            a_final = best_state['seq'] / best_state['seq'].sum()
            # Low-beta L-BFGS-B for precise optimization
            for beta in [2.0, 1.0, 0.5]:
                if time.monotonic() > deadline - 10:
                    break
                a_final = optimize_lbfgsb(a_final, beta, maxiter=int(min(200, remaining * 10)))
                score_and_track(a_final, best_state)
            # Final exact subgradient steps
            if time.monotonic() < deadline - 5:
                a_exact, _ = optimize_pg_exact(a_final, n_iter=500, lr=0.05)
                score_and_track(a_exact, best_state)

    # Final verification and return
    if best_state['seq'] is None:
        best_state['seq'] = np.ones(128) / 128

    final_seq = np.maximum(best_state['seq'], 0.0)
    final_score = float(evaluate_sequence(final_seq.tolist()))
    if np.isfinite(final_score):
        best_state['score'] = final_score

    print("\nFINAL: score=%.6f, n=%d, time=%.0fs" % (
        best_state['score'], len(final_seq), time.monotonic() - t_start))

    return [float(x) for x in final_seq]
