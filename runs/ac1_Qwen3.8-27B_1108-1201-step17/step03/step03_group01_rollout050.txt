optimization with parabolic line search, adaptive margins, multi-length search,
and structured perturbation strategies.

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def convolve_xy(x, y):
    """Linear cross-convolution via FFT."""
    n = len(x)
    m = len(y)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    X = np.fft.rfft(x, fft_size)
    Y = np.fft.rfft(y, fft_size)
    return np.fft.irfft(X * Y, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C

def solve_lp_direction(a, M, A_ub=None):
    """
    Solve LP: maximize sum(g) s.t. A_ub @ g <= M, g >= 0.
    Returns (g, g_sum).
    """
    n = len(a)
    if A_ub is None:
        A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g, g_sum
    return None, 0.0

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-based iterative
optimization with adaptive margins, parabolic line search, smoothed gradient

Key improvements over previous: tighter adaptive LP margins, dual-informed

import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp(a, M, A_ub):
    """
    Solve LP: maximize sum(g) s.t. A_ub @ g <= M, g >= 0.
    Returns (g_raw, g_sum) or (None, 0).
    """
    n = len(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g, g_sum
    return None, 0.0


def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) for a (assumed normalized, sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def line_search(a, d, best_obj, n, max_evals=25):
    """
    Adaptive line search with parabolic refinement.
    Returns (best_t, best_obj).

    best_t = 0.0

    candidates = [0.5, 0.3, 0.7, 0.15, 0.85, 0.1, 0.2, 0.4, 0.6, 0.8, 0.95, 0.05]

        if evals >= max_evals:

        if s < 1e-15:
            evals += 1

        obj_try, _ = compute_obj(a_try)
        evals += 1
        points.append((t, obj_try))

            best_obj_ls = obj_try

    if len(points) >= 3 and evals < max_evals:

        # Try parabola on top 3
        for start in range(max(0, len(points_sorted) - 3), min(2, len(points_sorted))):
            if evals >= max_evals:
                break
            p1 = points_sorted[start]
            p2 = points_sorted[start + 1]
            p3 = points_sorted[start + 2]
            ts = np.array([p1[0], p2[0], p3[0]])
            fs = np.array([p1[1], p2[1], p3[1]])

            j_start = max(0, peak_lag - n + 1)
            j_end = min(n - 1, peak_lag)
            n_perturb = min(30, max(5, (j_end - j_start + 1) // 3))
            idx_perturb = rng.choice(range(j_start, j_end + 1), size=n_perturb, replace=False)

            a = best_a.copy()
            a[idx_perturb] -= scale * 0.4

            # Boost low-valued lags
            low_lags = np.argsort(conv)[:10]
            for ll in low_lags:
                ll = int(ll)
                js = max(0, ll - n + 1)
                je = min(n - 1, ll)
                if je > js:
                    idx = rng.randint(js, je + 1)
                    a[idx] += scale * 0.3

            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a + rng.standard_normal(n) * scale
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                else:
                    a = best_a.copy()

            _, conv = compute_obj(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(250, 80 + 20 * perturb_count)

            if perturb_count > 40:
                break

    return best_a, best_obj


def smooth_gradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.95, tau=0.0005):
    """
    Smoothed (log-sum-exp) gradient descent with momentum.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    _, conv = compute_obj(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    lr = lr_init

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        # Compute softmax weights on autocorrelation
        conv_shifted = conv / tau - (conv.max() / tau)
        w = np.exp(conv_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum

        # Gradient: 4n * sum_k w_k * a_{k-j}
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += wk * a[k - idx]
        grad *= 4.0 * n

        # Momentum step
        v = momentum * v - lr * grad
        a_new = np.maximum(a + v, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        _, conv = compute_obj(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997
        tau *= 1.0005
        momentum = min(0.98, momentum + 0.0001)

        if no_improve >= 1000:
            break

    return best_a, best_obj


def make_initialization(n, parent=None, rng=None):
    """Generate a reasonable initialization of length n."""
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
            if len(p) == n:
                return p.copy()
            else:
                a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                a = np.maximum(a, 0)
                a = a / a.sum()
                return a

    if rng is None:
        rng = np.random.default_rng(0)

    # Mix of uniform and random
    alpha = 0.3
    uniform = np.ones(n) / n
    random_v = rng.dirichlet(np.ones(n) * 0.3)
    return alpha * uniform + (1 - alpha) * random_v


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: LP optimization with adaptive margins, multi-length
    exploration, smoothed gradient polishing, and final perturbations.
    """
    global height_sequence_1

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    def try_update(a_candidate):
        """Evaluate and update best if improved. Returns score or inf."""
        if a_candidate is None:
            return float('inf')
        try:
            score = float(evaluate_sequence(a_candidate.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score):
            return score
        return float('inf')

    # === Phase 1: LP from parent ===
    print("=== Phase 1: LP from parent ===")
    t_start = time.monotonic()

    if parent is not None:
        t_alloc = min(500.0, (deadline - time.monotonic()) * 0.35)
        if t_alloc > 30:
            a_opt, obj_opt = lp_optimize(parent, time_limit=t_alloc, tight_margin=1.0001, verbose=True)
            score = try_update(a_opt)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Phase 1: {score:.6f}")

    if best_sequence is None:
        n = 512
        a0 = make_initialization(n, parent, rng)
        t_alloc = min(300.0, (deadline - time.monotonic()) * 0.3)
        a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc, tight_margin=1.0001)
        score = try_update(a_opt)
        if np.isfinite(score):
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Phase 1 (fallback): {score:.6f}")

    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # === Phase 2: Multi-n exploration ===
    print("=== Phase 2: Multi-n exploration ===")
    t_start = time.monotonic()

    n_to_try = [256, 384, 512, 640, 1024]
    if parent is not None:
        n_to_try = [x for x in n_to_try if x != len(parent)]
        n_to_try.sort(key=lambda x: abs(x - len(parent)))
    else:
        n_to_try.sort()

    remaining = deadline - time.monotonic() - 120.0
    t_per_n = max(25.0, remaining * 0.3 / max(1, min(4, len(n_to_try))))

    for n in n_to_try[:4]:
        if time.monotonic() > deadline - 100:
            break
        t_alloc = min(t_per_n, deadline - time.monotonic() - 60)
        if t_alloc < 15:
            break

        a0 = make_initialization(n, parent, rng)
        a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc, tight_margin=1.0001)
        score = try_update(a_opt)
        if np.isfinite(score):
            print(f"  n={n}: {score:.6f}")
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()

    print(f"Phase 2 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # === Phase 3: Deep LP on best with tighter margin ===
    print("=== Phase 3: Deep LP (tighter) ===")
    t_start = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 80.0
        if remaining > 30:
            t_alloc = remaining * 0.5
            a_opt, obj_opt = lp_optimize(best_sequence, time_limit=t_alloc, tight_margin=1.00005, verbose=True)
            score = try_update(a_opt)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep LP: {score:.6f}")

            if time.monotonic() < deadline - 30:
                t_alloc2 = min(80.0, deadline - time.monotonic() - 20)
                if t_alloc2 > 10:
                    a_opt, obj_opt = lp_optimize(best_sequence, time_limit=t_alloc2, tight_margin=1.00001)
                    score = try_update(a_opt)
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Deep LP2: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # === Phase 4: Smoothed gradient polishing ===
    print("=== Phase 4: Smooth Gradient Polish ===")
    t_start = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 20:
            sg_configs = [
                {'lr_init': 0.004, 'momentum': 0.95, 'tau': 0.0003},
                {'lr_init': 0.008, 'momentum': 0.93, 'tau': 0.001},
            ]
            t_per_sg = remaining / (len(sg_configs) + 1)

            for ci, cfg in enumerate(sg_configs):
                if time.monotonic() > deadline - 15:
                    break
                t_alloc = min(t_per_sg, deadline - time.monotonic() - 10)
                if t_alloc < 8:
                    break

                a_opt, obj_opt = smooth_gradient_optimize(best_sequence, time_limit=t_alloc, **cfg)
                score = try_update(a_opt)
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Smooth grad {ci}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # === Phase 5: Final perturbations + quick LP ===
    print("=== Phase 5: Final Perturbations ===")
    t_start = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 15:
            n_best = len(best_sequence)
            n_perturb = max(1, int(remaining / 25))
            t_per_perturb = remaining / n_perturb

            for pi in range(n_perturb):
                if time.monotonic() > deadline - 5:
                    break
                if t_per_perturb < 5:
                    break

                scale = 0.001 + 0.002 * (pi + 1)
                noise = rng.standard_normal(n_best) * scale
                a_pert = np.maximum(best_sequence + noise, 0.0)
                s = a_pert.sum()
                if s > 1e-15:
                    a_pert = a_pert / s
                else:
                    continue

                t_alloc = min(t_per_perturb, deadline - time.monotonic() - 3)
                if t_alloc < 4:
                    break

                a_opt, obj_opt = lp_optimize(a_pert, time_limit=t_alloc, tight_margin=1.0001)
                score = try_update(a_opt)
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Perturb {pi}: {score:.6f}")

    print(f"Phase 5 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"Fallback uniform: {best_score:.6f}")

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
