def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (s * s)


def active_set_lp(f, M, n, active_indices=None, tol=1e-9):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for active k only.
    If active_indices is None, use all k where conv is near max.
    """
    conv = fast_convolve(f)
    if active_indices is None:
        # Use constraints where conv is within a small tolerance of max
        threshold = M - tol * M
        active_indices = np.where(conv >= threshold)[0]
    
    if len(active_indices) == 0:
        active_indices = np.array([np.argmax(conv)])
    
    # Build constraint matrix for active constraints only
    n_constraints = len(active_indices)
    A_ub = np.zeros((n_constraints, n))
    for idx, k in enumerate(active_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[idx, j] = f[k - j]
    
    b_ub = np.full(n_constraints, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 10})
        if result.success:
            return np.maximum(result.x, 0.0), active_indices
    except Exception:
        pass
    return None, None

def simulated_annealing(a_init, time_limit, T_init=0.1, T_min=0.0001,
                        cooling_rate=0.9995, max_steps_per_temp=5):
    """
    Simulated annealing on the simplex.
    Moves: pick random pair (i,j), transfer mass from one to the other.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    T = T_init
    np.random.seed(int(time.monotonic() * 1000) % 100000)
    
    while T > T_min and time.monotonic() - t0 < time_limit:
        for step in range(max_steps_per_temp):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick a move: transfer mass between i and j
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            
            # Transfer fraction of min(a[i], a[j])
            frac = np.random.uniform(0.01, 0.2)
            delta = frac * min(a[i], a[j])
            if delta < 1e-10:
                continue
            
            # Determine direction: transfer from larger to smaller or random
            if np.random.random() < 0.5:
                # Transfer from i to j
                if a[i] < delta:
                    continue
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
            else:
                # Transfer from j to i
                if a[j] < delta:
                    continue
                a_new = a.copy()
                a_new[j] -= delta
                a_new[i] += delta
            
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            delta_obj = obj_new - 2.0 * n * conv.max()
            
            if delta_obj < 0 or np.random.random() < np.exp(-delta_obj / T):
                a = a_new
                conv = conv_new
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()
        
        T *= cooling_rate
    
    return best_a, best_obj


def subgradient_descent(a0, time_limit, lr_init=0.02, momentum=0.9, top_k=50):
    """Subgradient descent with momentum on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pmax = conv[peaks].max()
        if pmax < 1e-15:
            break
        
        # Subgradient: for each peak k, the gradient wrt a_j is a_{k-j}
        sg = np.zeros(n)
        for k in peaks:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += a[k - j]
        
        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm > 1e-15:
            sg = sg / sg_norm
        
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            noise = np.random.randn(n) * 0.01
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.5
            no_improve = 0
            lr = max(0.002, lr * 0.5)
        
        lr *= 0.9995
    
    return best_a, best_obj


def coordinate_descent_polish(a_in, max_rounds=50, n_try=5):
    """
    Coordinate descent: try moving mass between pairs of coordinates.
    For each pair, try several transfer amounts and pick the best.
    """
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    for rnd in range(max_rounds):
        improved = False
        # Random subset of pairs for efficiency
        pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        np.random.shuffle(pairs)
        pairs = pairs[:min(5000, len(pairs))]
        
        for i, j in pairs:
            if a[i] < 1e-8 and a[j] < 1e-8:
                continue
            
            best_delta = 0
            best_local_obj = best_obj
            
            # Try several transfer amounts
            max_transfer = max(a[i], a[j]) * 0.5
            if max_transfer < 1e-10:
                continue
            
            for frac in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                delta = frac * max_transfer
                
                # Try i -> j
                if a[i] >= delta:
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    a_test = a_test / a_test.sum()
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_local_obj:
                        best_local_obj = obj_test
                        best_delta = delta  # positive means i->j
                
                # Try j -> i
                if a[j] >= delta:
                    a_test = a.copy()
                    a_test[j] -= delta
                    a_test[i] += delta
                    a_test = a_test / a_test.sum()
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_local_obj:
                        best_local_obj = obj_test
                        best_delta = -delta  # negative means j->i
            
            if best_delta != 0 and best_local_obj < best_obj - 1e-13:
                if best_delta > 0:
                    a[i] -= best_delta
                    a[j] += best_delta
                else:
                    a[j] -= best_delta
                    a[i] += best_delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                conv = fast_convolve(a)

            break
    
    return a


def gen_inits(n, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Triangle
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # Edge concentration
    edge = np.zeros(n)
    e = max(1, n // 6)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    # Random
    inits['random'] = np.random.dirichlet(np.ones(n) * 0.3)
    
    # Bimodal
    g2 = (np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2))
    inits['bimodal'] = g2 / g2.sum()
    
    # Log-spaced
    log_spaced = np.exp(np.linspace(0, 2, n))
    inits['logspace'] = log_spaced / log_spaced.sum()
    
    # Power law
    power = np.power(x + 1, 2)
    inits['power'] = power / power.sum()
    
    # Cosine
    cos_seq = np.cos(np.pi * x / n) ** 2 + 0.01
    inits['cosine'] = cos_seq / cos_seq.sum()
    
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent'] = p / ps
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                          np.maximum(parent, 0))
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_resample'] = p / ps
    
    return inits

def active_set_lp(f, M, n, tol_frac=0.02, time_limit_lp=8.0):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for near-peak k.
    """
    conv = fast_convolve(f)
    threshold = M * (1.0 - tol_frac)
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])

    n_c = len(active)
    A_ub = np.zeros((n_c, n))
    for idx, k in enumerate(active):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[idx, j] = f[k - j]

    b_ub = np.full(n_c, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                   'time_limit': time_limit_lp})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None

def subgradient_descent(a0, time_limit, lr_init=0.015, momentum=0.9, top_k=40):
    """Subgradient descent with momentum on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pmax = conv[peaks].max()
        if pmax < 1e-15:
            break

        sg = np.zeros(n)
        for k in peaks:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += a[k - j]

        sg_norm = np.linalg.norm(sg)
        if sg_norm > 1e-15:
            sg = sg / sg_norm

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 250:
            a = project_simplex(a + np.random.randn(n) * 0.008)
            conv = fast_convolve(a)
            v *= 0.5
            no_improve = 0
            lr = max(0.002, lr * 0.6)

        lr *= 0.9996

    return best_a, best_obj

                inits['parent'] = p / ps
        else:
            p = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)),
                          np.maximum(parent, 0))
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_resample'] = p / ps

    return inits


def try_refine(a_in, time_budget):
    """Try multiple refinement methods and return the best."""
    n = len(a_in)
    results = []
    t0 = time.monotonic()

    # Targeted CD
    t_cd = min(time_budget * 0.3, 30.0)
    if t_cd > 2:
        a_cd = targeted_cd(a_in, max_rounds=60, n_top_peaks=3,
                           n_sources=12, n_targets=12)
        conv_cd = fast_convolve(a_cd)
        results.append((2.0 * n * conv_cd.max(), a_cd))

    # Pairwise CD
    t_rw = min(time_budget * 0.25, 20.0)
    if t_rw > 2 and time.monotonic() - t0 < time_budget * 0.3:
        a_rw = pairwise_cd(a_cd if results else a_in,
                           max_rounds=20, n_pairs=1500)
        conv_rw = fast_convolve(a_rw)
        results.append((2.0 * n * conv_rw.max(), a_rw))

    # Subgradient
    t_sg = min(time_budget * 0.25, 15.0)
    if t_sg > 2 and time.monotonic() - t0 < time_budget * 0.6:
        best_so_far = min(results, key=lambda x: x[0])[1] if results else a_in
        a_sg, obj_sg = subgradient_descent(best_so_far, t_sg,
                                           lr_init=0.008, momentum=0.92, top_k=50)
        results.append((obj_sg, a_sg))

    # LP
    t_lp = min(time_budget * 0.2, 15.0)
    if t_lp > 2 and time.monotonic() - t0 < time_budget * 0.8:
        best_so_far = min(results, key=lambda x: x[0])[1] if results else a_in
        a_lp, obj_lp = optimize_lp_direction(best_so_far, t_lp,
                                             blend_init=0.2, tol_frac=0.03)
        results.append((obj_lp, a_lp))

    if not results:
        return a_in
    return min(results, key=lambda x: x[0])[1]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: subgradient scan -> LP refinement -> targeted CD
    -> pairwise CD -> final polish. Explores multiple lengths.
    """
    np.random.seed(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    def update_best(a_opt, label):
        nonlocal best_sequence, best_score
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  {label}: score={score:.6f}, n={len(a_opt)}")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p.all()) and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
    except Exception:
        pass

    # Phase 1: Quick scan
    print("=== Phase 1: Quick Scan ===")
    lengths = [384, 512, 640, 768]
    results = []
    phase1_end = time.monotonic() + min(250.0, (deadline - time.monotonic()) * 0.22)

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        inits = gen_inits(n, parent)
        prio = ['parent', 'parent_resample', 'uniform', 'triangle',
                'bimodal', 'u_shape', 'random', 'edge', 'cosine', 'gauss']
        names = [nm for nm in prio if nm in inits][:4]
        t_per = max(8.0, 180.0 / (len(lengths) * 4))

        for name in names:
            if time.monotonic() > phase1_end - 5:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 3)
            if t_alloc < 4:
                break
            a0 = inits[name]
            a_opt, obj_opt = subgradient_descent(a0, t_alloc, lr_init=0.02,
                                                 momentum=0.9, top_k=30)
            results.append((obj_opt, n, name, a_opt))
            sc = update_best(a_opt, f"SG n={n} {name}")
            if sc < 1.51:
                print(f"    ** Promising: {sc:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: Deep refinement on top candidates
    print("=== Phase 2: Deep Refinement ===")
    top_list = []
    seen = set()
    for obj_opt, n, name, a_opt in results[:8]:
        key = (n, name)
        if key not in seen:
            top_list.append((n, a_opt.copy()))
            seen.add(key)
        if len(top_list) >= 5:
            break
    if best_sequence is not None:
        top_list.append((len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 250.0
    t_per_cand = max(25.0, remaining / max(1, len(top_list)))

    for n, a0 in top_list:
        if time.monotonic() > deadline - 200:
            break
        t_alloc = min(t_per_cand, deadline - time.monotonic() - 150)
        if t_alloc < 10:
            break

        a_opt = try_refine(a0, t_alloc)
        sc = update_best(a_opt, f"Refine n={n}")

    # Phase 3: Focused local search on best
    print("=== Phase 3: Focused Local Search ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 80.0
        n_best = len(best_sequence)

        # Multiple targeted CD rounds with different parameters
        for cd_round in range(3):
            if time.monotonic() > deadline - 60:
                break
            t_cd = min(remaining * 0.2, 40.0)
            n_top = 3 + cd_round
            n_src = 12 + cd_round * 4
            a_cd = targeted_cd(best_sequence, max_rounds=80,
                               n_top_peaks=n_top, n_sources=n_src, n_targets=n_src)
            sc = update_best(a_cd, f"TargetedCD {cd_round}")

        # Pairwise CD
        if time.monotonic() < deadline - 40:
            a_pw = pairwise_cd(best_sequence, max_rounds=30, n_pairs=3000)
            sc = update_best(a_pw, "PairwiseCD")

        # Subgradient refinement
        if time.monotonic() < deadline - 30:
            a_sg, obj_sg = subgradient_descent(best_sequence, 20.0,
                                              lr_init=0.005, momentum=0.93, top_k=60)
            sc = update_best(a_sg, "SG refine")

    # Phase 4: Final intensive polish
    print("=== Phase 4: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0

        for pol_round in range(4):
            if time.monotonic() > deadline - 15:
                break
            t_pol = min(remaining / (4 - pol_round), 35.0)
            if t_pol < 5:
                break

            a_opt = try_refine(best_sequence, t_pol)
            sc = update_best(a_opt, f"Polish {pol_round}")

            # Quick targeted CD
            if time.monotonic() < deadline - 12:
                a_cd = targeted_cd(best_sequence, max_rounds=40,
                                   n_top_peaks=4, n_sources=20, n_targets=20)
                sc = update_best(a_cd, f"Polish CD {pol_round}")

    # Phase 5: Try different lengths with the best as template
    print("=== Phase 5: Length Exploration ===")
    if best_sequence is not None and time.monotonic() < deadline - 30:
        n_best = len(best_sequence)
        for n_try in [n_best - 64, n_best + 64]:
            if n_try < 64 or n_try > 2048:
                continue
            if time.monotonic() > deadline - 20:
                break
            p = np.interp(np.linspace(0, 1, n_try),
                          np.linspace(0, 1, n_best), best_sequence)
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                p /= ps
                a_opt = try_refine(p, min(30.0, deadline - time.monotonic() - 15))
                sc = update_best(a_opt, f"Length {n_try}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
