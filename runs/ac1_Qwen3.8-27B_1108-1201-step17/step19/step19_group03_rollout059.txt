where a is a non-negative sequence of length n, and a*a is the full convolution (autocorrelation).

Key insights:
1. The objective is scale-invariant: if we scale a by c, both max(a*a) scales by c² and (sum(a))² scales by c², so they cancel.
2. So we can normalize to sum(a) = 1, and the objective becomes 2*n*max(a*a).
3. The minimum possible value of max(a*a) for a non-negative sequence with sum=1 is at least 1/n (by the pigeonhole principle / the fact that the DC component of the autocorrelation is 1, so max ≥ 1/n).
4. The theoretical lower bound is 2, achieved when the autocorrelation is perfectly flat (all values equal to 1/n). But for non-negative sequences, we can't achieve perfectly flat autocorrelation.

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a, k_indices):
    n = len(a)
    K = len(k_indices)
    T = np.zeros((K, n))
    for ii, k in enumerate(k_indices):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        for j in range(j_start, j_end):
            T[ii, j] = a[ki - j]
    return T


def lp_fw_step(a, M, constraint_indices, solver):
    n = len(a)
    K = len(constraint_indices)
    if K == 0:
        return None
    T = build_conv_matrix(a, constraint_indices)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum, b_val


def line_search_fw(a, g, M, max_evals=8):
    best_alpha = 0.0
    best_val = fast_autocorr(a).max()
    alphas = [2.0 / (min(best_val * 1000, 500) + 2), 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.8]
    alphas = [min(1.0, max(0.01, al)) for al in alphas]
    alphas = list(dict.fromkeys(alphas))[:max_evals]
    best_m = best_val
    for alpha in alphas:
        a_trial = (1.0 - alpha) * a + alpha * g
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t
        m_trial = fast_autocorr(a_trial).max()
        if m_trial < best_m:
            best_m = m_trial
            best_alpha = alpha
    return best_alpha, best_m


def lp_fw_optimize(a0, time_limit, solver, verbose=False, K_init=200, K_max=None,
                   adaptive_K=True, use_line_search=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    if K_max is None:
        K_max = min(2 * n - 1, 3000)
    K = K_init
    t_iter = 0
    support = [a.copy()]
    max_support = 15

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-15:
            break

        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]

        g_dir, b_raw = lp_fw_step(a, M_curr, binding_idx, solver)
        if g_dir is None:
            K = min(K + 50, K_max)
            continue

        if use_line_search:
            best_alpha, best_trial_m = line_search_fw(a, g_dir, M_curr)
        else:
            best_alpha = 2.0 / (t_iter + 2)
            a_trial = (1.0 - best_alpha) * a + best_alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            a_trial = a_trial / s_t
            best_trial_m = fast_autocorr(a_trial).max()

        a_new = (1.0 - best_alpha) * a + best_alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            if len(support) < max_support:
                support.append(g_dir.copy())
            else:
                support.pop(0)
                support.append(g_dir.copy())
        a = a_new
        t_iter += 1

        if adaptive_K and t_iter % 5 == 0:
            K = min(K + 20, K_max)

        # Away-step: try moving away from old support points
        if len(support) > 2 and t_iter % 3 == 0:
            for sp in support:
                away = a - 0.3 * sp
                away = np.maximum(away, 0.0)
                s_away = away.sum()
                if s_away < 1e-15:
                    continue
                away = away / s_away
                m_away = fast_autocorr(away).max()
                if m_away < best_m - 1e-15:
                    best_m = m_away
                    best_a = away.copy()
                    a = away

        if verbose and t_iter % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t_iter} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

    return best_a, best_m


def logsumexp_gradient_optimize(a0, time_limit, verbose=False, tau_init=5.0, tau_final=50.0,
                                eta_init=0.3, rng_seed=77):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1.0))
        tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(a)
        ac_max = ac.max()
        shifted = tau * (ac - ac_max)
        exp_ac = np.exp(shifted)
        p = exp_ac / exp_ac.sum()

        # grad[j] = sum_k p[k] * a[k-j] for valid k-j
        fft_size = 1 << (2 * n - 2).bit_length()
        P = np.fft.rfft(p, fft_size)
        A = np.fft.rfft(a, fft_size)
        # cross-correlation: grad = ifft(conj(P) * A) or similar
        # Actually: grad[j] = sum_k p[k] * a[k-j]
        # This is (p * a_flipped)[something]... let me think
        # sum_k p[k] a[k-j] = (p convolved with a)[j + n - 1]... 
        # Actually it's the cross-correlation of p and a
        # corr[j] = sum_k p[k] a[k-j] = ifft(conj(FFT(p)) * FFT(a))[j] (with proper indexing)
        cross = np.fft.irfft(np.conj(P) * A, fft_size)
        grad = cross[:n]

        # Project gradient to tangent space (sum = 0)
        grad_tan = grad - grad.mean()

        eta = eta_init / (1.0 + 0.1 * stall)
        a_trial = a - eta * grad_tan
        a_new = project_simplex(a_trial)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                eta *= 0.5
            if stall > 300:
                pert = rng.normal(0, 0.01, n) * best_a
                a_p = np.maximum(best_a + pert, 1e-10)
                a = a_p / a_p.sum()
                m_p = fast_autocorr(a).max()
                if m_p < best_m:
                    best_m = m_p
                    best_a = a.copy()
                stall = 100
        a = a_new

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LSE-Grad it={it} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}, stall={stall}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    n_top = 15

    while time.monotonic() - t0 < time_limit and stall < 80:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_top]

        # Compute sensitivity: how much does each coordinate affect the top peaks
        sens = np.zeros(n)
        for k in top_k[:8]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[ki - idx]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[ki - idx]
            max_contrib_idx = idx[np.argmax(contrib)]
            src_val = a[max_contrib_idx]
            if src_val < 1e-12:
                continue

            # Find low-sensitivity target
            low_sens = np.argsort(sens)[:5]
            for dst in low_sens:
                if dst == max_contrib_idx:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[max_contrib_idx] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def spectral_pocs(a0, time_limit, verbose=False, n_trials=5, rng_seed=0):
    n = len(a0)
    best_a = a0.copy()
    best_m = fast_autocorr(a0).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit - 5:
            break
        t_trial = min((time_limit - (time.monotonic() - t0)) / max(1, n_trials - trial), 30.0)
        t_start = time.monotonic()

        rng = np.random.default_rng(rng_seed + trial * 1000)
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        A_spec = np.exp(1j * phases)
        A_full = np.zeros(fft_size, dtype=np.complex128)
        A_full[:fft_size // 2 + 1] = A_spec
        for i in range(1, fft_size // 2):
            A_full[fft_size - i] = np.conj(A_spec[i])
        a = np.fft.irfft(A_full, fft_size)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s

        m_curr = fast_autocorr(a).max()

        n_pocs_iters = max(50, int(t_trial * 50))
        for it in range(n_pocs_iters):
            if time.monotonic() - t_start > t_trial:
                break
            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            alpha_blend = 0.05 + 0.15 * min(1.0, it / 100)
            new_mag = (1 - alpha_blend) * mag + alpha_blend * mag_mean
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            m_new = fast_autocorr(a_new).max()
            if m_new < m_curr:
                m_curr = m_new
                a = a_new

        if m_curr < best_m:
            best_m = m_curr
            best_a = a.copy()
            if verbose:
                print(f"    POCS trial {trial}: J={2.0*n*best_m:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_initial(n, style='ramp'):
    if style == 'ramp':
        a = np.arange(1, n + 1, dtype=np.float64)
    elif style == 'u_shape':
        x = np.linspace(-1, 1, n)
        a = 1.0 + 2.0 * x ** 2
    elif style == 'bimodal':
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.1) / 0.1) ** 2) + np.exp(-((x - 0.9) / 0.1) ** 2) + 0.01
    elif style == 'cosine':
        x = np.linspace(0, 1, n)
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 2
    elif style == 'sin2':
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    else:
        a = np.ones(n)
    return norm_sum1(a)

        sc = evaluate_safe(arr_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(arr)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        best_seq = make_initial(n_base, 'ramp')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Spectral POCS for diverse initialization ===
    print(f"\n=== Phase 1: Spectral POCS (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 120.0)
    if t1 > 20:
        a_pocs, _ = spectral_pocs(best_seq, t1, verbose=True, n_trials=5, rng_seed=42)
        try_update(a_pocs, "pocs")

    # === Phase 2: Log-sum-exp gradient for fast progress ===
    print(f"\n=== Phase 2: LogSumExp Gradient (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.12, 180.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = logsumexp_gradient_optimize(a_in, t2, verbose=True,
                                                tau_init=5.0, tau_final=60.0,
                                                eta_init=0.3, rng_seed=77)
        try_update(a_out, "lse_grad")

    # === Phase 3: Main LP-FW optimization ===
    print(f"\n=== Phase 3: LP-FW Main (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.3, 450.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        K_init = min(200, 2 * n0 - 1)
        K_max = min(2500, 2 * n0 - 1)
        a_out, _ = lp_fw_optimize(a_in, t3, solver, verbose=True,
                                   K_init=K_init, K_max=K_max,
                                   adaptive_K=True, use_line_search=True)
        try_update(a_out, "lp_main")

    # === Phase 4: LP-FW round 2 with more constraints ===
    print(f"\n=== Phase 4: LP-FW R2 (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.2, 300.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        K_init = min(500, 2 * n0 - 1)
        K_max = min(4000, 2 * n0 - 1)
        a_out, _ = lp_fw_optimize(a_in, t4, solver, verbose=True,
                                   K_init=K_init, K_max=K_max,
                                   adaptive_K=True, use_line_search=True)
        try_update(a_out, "lp_r2")

    # === Phase 5: Log-sum-exp gradient round 2 ===
    print(f"\n=== Phase 5: LSE-Grad R2 (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.1, 150.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = logsumexp_gradient_optimize(a_in, t5, verbose=True,
                                                tau_init=10.0, tau_final=100.0,
                                                eta_init=0.2, rng_seed=200)
        try_update(a_out, "lse_r2")

    # === Phase 6: Try different lengths ===
    print(f"\n=== Phase 6: Length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 2000]:
        new_n = n_best + delta
        if 1000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best][:3]

    for new_n in trial_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.4, 200.0)
        if t6a > 30:
            a_out, _ = lp_fw_optimize(a_long, t6a, solver, verbose=False,
                                       K_init=min(200, 2 * new_n - 1),
                                       K_max=min(2000, 2 * new_n - 1),
                                       adaptive_K=True, use_line_search=True)
            try_update(a_out, f"lp_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.3, 100.0)
            if t6b > 20:
                a_out, _ = logsumexp_gradient_optimize(
                    interpolate_sequence(best_seq, new_n), t6b, verbose=False,
                    tau_init=8.0, tau_final=80.0, eta_init=0.25, rng_seed=new_n % 500)
                try_update(a_out, f"lse_n{new_n}")

    # === Phase 7: LP-FW final on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: LP-FW Final (n={n_best}) ===")
    t7 = min(remaining * 0.25, 300.0)
    if t7 > 30:
        a_in = norm_sum1(best_seq)
        K_init = min(400, 2 * n_best - 1)
        K_max = min(5000, 2 * n_best - 1)
        a_out, _ = lp_fw_optimize(a_in, t7, solver, verbose=True,
                                   K_init=K_init, K_max=K_max,
                                   adaptive_K=True, use_line_search=True)
        try_update(a_out, "lp_final")

    # === Phase 8: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Coordinate Polish (n={n_best}) ===")
    t8 = min(remaining * 0.3, 200.0)
    if t8 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t8, verbose=True, rng_seed=456)
        try_update(a_out, "coord")

    # === Phase 9: Final combined pass ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final pass ===")
    if remaining > 60:
        t9a = min(remaining * 0.4, 200.0)
        if t9a > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_fw_optimize(a_in, t9a, solver, verbose=False,
                                       K_init=min(600, 2 * n_best - 1),
                                       K_max=min(6000, 2 * n_best - 1),
                                       adaptive_K=True, use_line_search=True)
            try_update(a_out, "lp_very_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t9b = min(remaining * 0.5, 100.0)
            if t9b > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_polish(a_in, t9b, verbose=False, rng_seed=789)
                try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = make_initial(n_fb, 'ramp')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
