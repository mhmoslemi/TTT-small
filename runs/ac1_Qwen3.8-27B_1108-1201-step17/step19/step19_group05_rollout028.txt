```python
def build_convolution_matrix(a, n):
    """Build the (2n-1) x n convolution matrix C where (Cb)_k = sum_j a[k-j] b_j."""
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # Column j has nonzeros at rows k where 0 <= k-j < n
        # i.e., j <= k <= j+n-1
        # And k ranges from 0 to 2n-2
        k_start = max(0, j)  # = j since j >= 0
        k_end = min(2*n-2, j+n-1)
        ks = range(k_start, k_end+1)
        # a[k-j] for k in ks
        rows.extend(ks)
        cols.extend([j]*(k_end-k_start+1))
        vals.extend(a[k-j] for k in ks)
    return coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
```
