def fast_conv(a):
    """Compute full convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

def fast_conv(a):

    if ws < 1e-30:

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss

    # === Phase 1: Multiplicative descent on parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: MD on parent (n={n_parent}, budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 30:
        a_out, sc_out = multiplicative_descent(best_seq, t1, n_active=60, lr0=0.04, beta0=5.0, verbose=True)
        check_update(a_out, f"md_p1_n{n_parent}")

    # === Phase 2: Targeted local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Local search (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 250.0)
    if t2 > 20:
        a_out, sc_out = targeted_local_search(best_seq, t2, verbose=True)
        check_update(a_out, f"ls_n{len(best_seq)}")

    # === Phase 3: Polish descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Polish (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 15:
        a_out, sc_out = polish_descent(best_seq, t3, verbose=False)
        check_update(a_out, f"polish_n{len(best_seq)}")

    # === Phase 4: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Length exploration (budget={remaining:.0f}s) ===")

    length_tries = []
    for delta in [-1000, -500, 500, 1000, 2000, 3000]:
        nn = n_best + delta
        if 2000 <= nn <= 10000 and nn != n_best:
            length_tries.append(nn)
    for rn in [3000, 4000, 5000, 6000, 7000, 8000]:
        if rn != n_best and 2000 <= rn <= 10000:
            length_tries.append(rn)
    length_tries = sorted(set(length_tries))

    for new_n in length_tries[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")

        a_interp = interpolate_seq(best_seq, new_n)

        t4a = min(remaining * 0.35, 150.0)
        if t4a > 20:
            a_out, _ = multiplicative_descent(a_interp, t4a, n_active=50, lr0=0.03, beta0=4.0, verbose=False)
            check_update(a_out, f"md_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t4b = min(remaining * 0.25, 80.0)
            if t4b > 15:
                a_out, _ = targeted_local_search(best_seq, t4b, verbose=False)
                check_update(a_out, f"ls_n{new_n}")

    # === Phase 5: Diverse restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Diverse restarts (budget={remaining:.0f}s) ===")
    rng = np.random.default_rng(777)

    n_restart = len(best_seq)
    n_restarts = min(4, int(remaining // 60))

    for ri in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break

        # Generate diverse init and blend with current best
        a_div = diverse_init(n_restart, rng)
        w = float(rng.uniform(0.1, 0.35))
        a_init = normalize((1 - w) * best_seq + w * a_div)

        t5 = min(remaining * 0.3, 120.0)
        if t5 > 15:
            a_out, _ = multiplicative_descent(a_init, t5, n_active=40,
                                              lr0=float(rng.uniform(0.02, 0.06)),
                                              beta0=float(rng.uniform(3.0, 6.0)),
                                              verbose=False)
            check_update(a_out, f"restart{ri}_n{n_restart}")

    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_best}, budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.40, 150.0)
    if t6 > 15:
        a_out, _ = polish_descent(best_seq, t6, verbose=False)
        check_update(a_out, f"final_polish_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t7 = min(remaining * 0.5, 80.0)
        if t7 > 10:
            a_out, _ = targeted_local_search(best_seq, t7, verbose=False)
            check_update(a_out, f"final_ls_n{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 5000
        best_seq = ideal_init(n_fb)
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))
        print(f"\nFallback: score={best_score:.6f}, n={n_fb}")

    return [float(x) for x in best_seq]
