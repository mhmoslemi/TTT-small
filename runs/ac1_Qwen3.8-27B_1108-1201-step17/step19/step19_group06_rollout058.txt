    """Compute score from autocorrelation (assumes sum=1)."""
    n = len(a)
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max())

def topk_softmax_value(ac, k=50, beta=10.0):
    """Compute soft-max of top-k autocorrelation values."""
    top_k = min(k, len(ac))
    top_vals = np.sort(ac)[::-1][:top_k]
    max_v = top_vals[0]
    expo = beta * (top_vals - max_v)
    expo = np.clip(expo, -50, 50)
    return max_v + np.log(np.sum(np.exp(expo))) / beta

def compute_lag_contribution(a, k, n):
    """For lag k, return list of (pair_contribution, i, j) sorted by contribution."""
    i_lo = max(0, k - n + 1)
    i_hi = min(n, k + 1)
    if i_lo >= i_hi:
        return np.array([]), np.array([], dtype=int), np.array([], dtype=int)
    idx = np.arange(i_lo, i_hi)
    j_idx = k - idx
    contrib = a[idx] * a[j_idx]
    order = np.argsort(contrib)[::-1]
    return contrib[order], idx[order], j_idx[order]

Now let me think about the main optimization loop. I want something that's:
1. Fast (can do many iterations in 1800s)
2. Effective (makes steady progress)
3. Diverse (explores different parts of the space)

I think the most promising approach is a **hybrid** that alternates between:
- **Smooth updates** (mirror descent / gradient-based): good for fine-tuning
- **Discrete updates** (coordinate moves): good for breaking symmetry and escaping local minima
- **Structured perturbations**: good for exploration

Let me write the full solution now. I'll focus on:

1. A better "peak shaving" that considers the impact on ALL lags, not just the target lag
2. Adaptive learning rate with backtracking
3. More aggressive length exploration
4. A "two-phase" approach: first find good structure (coarse), then fine-tune

Actually, let me reconsider the problem from scratch. The score is `2*n*max_b/sum_a^2`. Since the score is scale-invariant, WLOG sum_a = 1, and we minimize `2*n*max_b`.

The key question is: what's the best possible `max_b` for a non-negative sequence of length n with sum 1?

For the autocorrelation `b[k] = sum_i a[i]*a[k-i]`:
- `b[k] >= 0` for all k (since a >= 0)
- `sum_k b[k] = 1` (since sum a = 1)
- There are 2n-1 values of k

If we could make all `b[k] = 1/(2n-1)`, the score would be `2n/(2n-1) ≈ 1`.

But the constraint is that `b` must be the autocorrelation of a non-negative sequence. Not every non-negative `b` with sum 1 is achievable.

The key constraint is that `b` must be a valid autocorrelation function. For a non-negative sequence, the autocorrelation has additional structure beyond just being non-negative and summing to 1.

Specifically, the autocorrelation of a non-negative sequence is the full convolution of the sequence with itself, which means it's determined by n parameters (the sequence) but has 2n-1 degrees of freedom. So there are n-1 constraints that b must satisfy.

These constraints are essentially: the "even" and "odd" parts of b must be consistent with a single non-negative sequence. This is related to the "phase retrieval" problem.

For our purposes, the practical approach is to use optimization to find good sequences.

**Final algorithm plan:**

Phase 1 (fast, ~30s): 
- Quick parametric search + load parent
- Evaluate a few candidate lengths

Phase 2 (main optimization, ~1200s):
- Hybrid optimizer on the best starting sequence
- Alternates between:
  a. Mirror descent with adaptive top-K (smooth)
  b. Peak-shaving coordinate moves (discrete)
  c. Spectral equalization (global)
- With periodic restarts and exploration

Phase 3 (length exploration, ~300s):
- Try nearby lengths with interpolation
- Quick optimization on each

Phase 4 (final polish, ~270s):
- Fine-tune the best sequence found
- Very small step sizes, focused on the top few peaks

Let me now write the actual code. I'll make it different from the previous approach in these ways:

1. **Top-K equalization objective**: Instead of just minimizing the max, I minimize a soft-max of the top K peaks. This encourages the top peaks to be equal, which is a better "shaped" objective.

2. **Lag-aware coordinate moves**: When making a coordinate move, I compute the impact on ALL top lags (not just the target lag) and choose moves that reduce the worst case.

3. **Adaptive K**: The number of top lags to consider increases over time (from 10 to 200).

4. **Backtracking line search**: For the mirror descent steps, I use backtracking to ensure sufficient decrease.

5. **Spectral reweighting**: Instead of just flattening the spectrum, I use a "water-filling" approach to equalize the top autocorrelation values in the time domain.

6. **Multi-resolution**: Optimize at a coarser resolution first, then refine.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid of:
1. Top-K soft-max objective for peak equalization
2. Lag-aware coordinated moves (impact on all top lags)
3. Adaptive mirror descent with backtracking
4. Multi-resolution coarse-to-fine optimization
5. Aggressive length exploration with spectral interpolation
"""

import numpy as np
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def score_from_ac(a):
    n = len(a)
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max())


def smooth_max_top(ac, K, beta):
    """Smooth approximation of max of top-K values."""
    K = min(K, len(ac))
    top = np.sort(ac)[::-1][:K]
    mx = top[0]
    expo = beta * (top - mx)
    expo = np.clip(expo, -50, 50)
    return mx + np.log(np.sum(np.exp(expo))) / beta


def compute_grad_topk(a, ac, K, beta):
    """
    Compute gradient of smooth max of top-K autocorrelation values.
    Returns gradient vector of shape (n,).
    """
    n = len(a)
    K_eff = min(K, len(ac))
    top_idx = np.argsort(ac)[::-1][:K_eff]
    top_vals = ac[top_idx]
    
    mx = top_vals[0]
    expo = beta * (top_vals - mx)
    expo = np.clip(expo, -50, 50)
    w = np.exp(expo)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(K_eff) / K_eff
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n, k_int + 1)
        if i_lo >= i_hi:
            continue
        idx = np.arange(i_lo, i_hi)
        j_idx = k_int - idx
        valid = (j_idx >= 0) & (j_idx < n)
        grad[idx[valid]] += 2.0 * wi * a[j_idx[valid]]
    
    return grad


def mirror_descent_v2(a0, time_limit, lr_init=0.05, beta_init=5.0, K_init=30,
                      K_max=200, verbose=False, rng_seed=42):
    """
    Improved mirror descent with:
    - Top-K soft-max objective
    - Backtracking line search
    - Adaptive K
    - Momentum with adaptive rate
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = score_from_ac(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    a_prev = a.copy()
    momentum = 0.8
    lr = lr_init
    beta = beta_init
    K = K_init
    stall = 0
    iterations = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov-style extrapolation
        ratio = a / np.maximum(a_prev, 1e-30)
        ratio = np.clip(ratio, 0.01, 100.0)
        a_nest = a * np.power(ratio, momentum)
        a_nest = norm_sum1(a_nest)
        
        # Compute gradient
        ac = fast_autocorr(a_nest)
        grad = compute_grad_topk(a_nest, ac, K, beta)
        
        # Multiplicative update with backtracking
        found = False
        lr_try = lr
        for bt in range(8):
            log_a = np.log(np.maximum(a_nest, 1e-30)) - lr_try * grad
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()
            
            ac_new = fast_autocorr(a_new)
            score_new = ac_new.max()
            
            # Check if sufficient decrease (compared to current, not best)
            ac_curr = fast_autocorr(a)
            curr_max = ac_curr.max()
            
            if score_new < curr_max * (1 - 1e-8) or bt == 7:
                found = True
                break
            lr_try *= 0.5
        
        a_prev = a
        a = a_new
        iterations += 1
        
        # Track best
        current_score = score_from_ac(a)
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive parameters
        if iterations % 200 == 0:
            K = min(K + 5, K_max)
            # Slightly reduce learning rate over time
            lr = lr_init / (1.0 + 0.01 * np.sqrt(iterations))
        
        # Restart
        if stall > 3000:
            stall = 0
            # Informed perturbation: perturb positions contributing to top lags
            ac = fast_autocorr(best_a)
            top_lags = np.argsort(ac)[::-1][:20]
            n_pert = min(int(n * 0.1), 500)
            pert_positions = np.random.default_rng(stall).choice(n, size=n_pert, replace=False)
            
            pert = best_a.copy()
            pert[pert_positions] *= (1.0 + rng.normal(0, 0.1, n_pert))
            pert = norm_sum1(pert)
            
            w = float(rng.uniform(0.1, 0.3))
            a = norm_sum1((1 - w) * best_a + w * pert)
            a_prev = a.copy()
            lr *= 0.9
            beta = max(beta - 0.3, 2.0)
            
            if verbose:
                J = 2.0 * n * best_score
                print(f"      MD restart ({time.monotonic()-t0:.0f}s): J={J:.6f}, K={K}")
    
    if verbose:
        J = 2.0 * n * best_score
        print(f"      MD v2 done ({time.monotonic()-t0:.0f}s, {iterations} iters): J={J:.6f}")
    return best_a, best_score


def peak_shave(a0, time_limit, verbose=False, rng_seed=77):
    """
    Aggressive peak-shaving: directly reduce the max autocorrelation
    by making coordinated moves that consider impact on multiple lags.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    # Precompute which lags are "important"
    n_top_lags = 30
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        current_max = ac.max()
        
        if current_max < best_m - 1e-14:
            best_m = current_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                # Perturb and reset
                a = norm_sum1(best_a * (1.0 + rng.normal(0, 0.02, n)))
                stall = 0
                if stall > 50000:
                    break
        
        # Get top lags
        top_lags = np.argsort(ac)[::-1][:n_top_lags]
        
        # Focus on the single worst lag
        k_star = int(top_lags[0])
        
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n, k_star + 1)
        if i_lo >= i_hi:
            continue
        
        idx = np.arange(i_lo, i_hi)
        j_idx = k_star - idx
        contrib = a[idx] * a[j_idx]
        
        # Find top contributing pairs
        n_pairs = min(20, len(idx))
        pair_order = np.argsort(contrib)[::-1][:n_pairs]
        
        improved = False
        
        for pi in pair_order:
            if time.monotonic() - t0 > time_limit:
                break
            
            src = int(idx[pi])
            src_val = a[src]
            if src_val < 1e-14:
                continue
            
            # Compute "safety score" for each destination:
            # How much would moving mass to dst increase the top lags?
            dst_scores = np.zeros(n)
            for k2 in top_lags[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                if s2 < e2:
                    ii = np.arange(s2, e2)
                    jj = k2i - ii
                    # If we add mass to dst, how much does b[k2] increase?
                    # b[k2] increases by dst_mass * (a[k2-dst] + a[k2-dst] if dst != k2-dst)
                    # Actually: db[k2]/da[dst] = 2*a[k2-dst] if valid
                    m = dst
                    km = k2i - m
                    if 0 <= km < n:
                        dst_scores[m] += a[km]
            
            dst_scores[src] = np.inf  # Don't move to self
            
            # Also penalize positions that are high (to spread mass)
            dst_scores += 0.1 * a
            
            # Choose destination with lowest impact
            dst = int(np.argmin(dst_scores))
            
            if a[dst] > 0.9 * src_val:
                continue  # Don't move to a position with similar value
            
            # Try moving different fractions
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                ac_trial = fast_autocorr(a_trial)
                m_trial = ac_trial.max()
                
                if m_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved and stall > 1000:
            # Random small perturbation
            n_moves = int(rng.integers(10, 30))
            for _ in range(n_moves):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = float(rng.uniform(0.01, 0.05))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            if a.sum() > 1e-15:
                a = a / a.sum()
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"      Peak shave done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def spectral_equalize(a0, time_limit, verbose=False):
    """
    Spectral domain optimization: modify the spectrum to reduce autocorrelation peaks.
    Uses iterative projection: time-domain non-negativity + spectral flatness.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(55)
    
    fft_size = 1 << (2 * n - 2).bit_length()
    stall = 0
    lr = 0.1
    beta = 5.0
    
    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current spectrum
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A)**2
        
        # Target: suppress spectral peaks (which correspond to time-domain structure
        # that creates autocorrelation peaks)
        P_max = P.max()
        if P_max < 1e-30:
            break
        
        # Compute a "repulsion" force from spectral peaks
        P_norm = P / max(P_max, 1e-30)
        force = np.exp(beta * P_norm)
        force_mean = force.mean()
        
        # Damp peaks, boost valleys
        A_new = A * np.exp(-lr * (force - force_mean) / force_mean)
        
        # Inverse transform and enforce non-negativity
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = norm_sum1(a + rng.normal(0, 0.01, n) * a)
        else:
            a_new = a_new / s
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 2000:
                # Reheat
                a = norm_sum1(best_a + rng.normal(0, 0.005, n) * best_a)
                stall = 0
                lr *= 0.7
                beta = max(beta - 1.0, 1.0)
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"      Spectral done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def parametric_init(n):
    """Quick parametric search for good initializations."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    
    candidates = [np.ones(n) / n]
    
    # Power laws
    for p in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))
    
    # Gaussians at various positions
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sig in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            candidates.append(norm_sum1(np.exp(-((x - c)**2) / (2 * sig**2))))
    
    # Bimodal
    for sig in [0.05, 0.1, 0.15, 0.2]:
        for sep in [0.2, 0.3, 0.4, 0.5]:
            candidates.append(norm_sum1(
                np.exp(-((x - 0.5 + sep/2)**2) / (2*sig**2)) +
                np.exp(-((x - 0.5 - sep/2)**2) / (2*sig**2))))
    
    # Sine waves
    for f in [1, 2, 3, 5]:
        for phase in [0, np.pi/4, np.pi/2]:
            candidates.append(norm_sum1(np.sin(np.pi * f * x + phase)**2))
    
    # Exponential
    for r in [0.5, 1, 2, 5, 10]:
        candidates.append(norm_sum1(np.exp(-r * x)))
        candidates.append(norm_sum1(np.exp(r * x)))
    
    # T-shaped (high in middle, low at edges)
    for w in [0.1, 0.2, 0.3]:
        mask = (x > 0.5 - w/2) & (x < 0.5 + w/2)
        cand = np.where(mask, 1.0, 0.1)
        candidates.append(norm_sum1(cand))
    
    # Evaluate
    best_cand = None
    best_val = float('inf')
    for c in candidates:
        v = score_from_ac(c)
        if v < best_val:
            best_val = v
            best_cand = c.copy()
    
    return best_cand if best_cand is not None else np.ones(n) / n

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid of:
1. Top-K soft-max objective for peak equalization (multi-peak suppression)
2. Lag-aware coordinated moves with multi-lag impact scoring
3. Adaptive mirror descent with backtracking line search
4. Simulated-annealing style exploration with informed restarts
5. Wide length exploration with spectral interpolation
6. Coarse-to-fine multi-phase optimization
"""

import numpy as np
import time


def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely call the evaluation function."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def score_from_ac(a):
    """Compute score from autocorrelation (assumes sum=1)."""
    n = len(a)
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max())


def smooth_max_top(ac, K, beta):
    """Smooth approximation of max of top-K values."""
    K = min(K, len(ac))
    top = np.sort(ac)[::-1][:K]
    mx = top[0]
    expo = beta * (top - mx)
    expo = np.clip(expo, -50, 50)
    return mx + np.log(np.sum(np.exp(expo))) / beta


def compute_grad_topk(a, ac, K, beta):
    """
    Compute gradient of smooth max of top-K autocorrelation values.
    Returns gradient vector of shape (n,).
    """
    n = len(a)
    K_eff = min(K, len(ac))
    top_idx = np.argsort(ac)[::-1][:K_eff]
    top_vals = ac[top_idx]

    mx = top_vals[0]
    expo = beta * (top_vals - mx)
    expo = np.clip(expo, -50, 50)
    w = np.exp(expo)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(K_eff) / K_eff
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n, k_int + 1)
        if i_lo >= i_hi:
            continue
        idx = np.arange(i_lo, i_hi)
        j_idx = k_int - idx
        valid = (j_idx >= 0) & (j_idx < n)
        grad[idx[valid]] += 2.0 * wi * a[j_idx[valid]]

    return grad


def mirror_descent_v2(a0, time_limit, lr_init=0.05, beta_init=5.0, K_init=30,
                      K_max=200, verbose=False, rng_seed=42):

    Improved mirror descent with top-K soft-max objective,
    backtracking, adaptive K, and informed restarts.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    a_prev = a.copy()
    momentum = 0.8
    lr = lr_init
    beta = beta_init
    K = K_init
    stall = 0
    iterations = 0
    best_soft = smooth_max_top(fast_autocorr(a), K, beta)

    while time.monotonic() - t0 < time_limit:
        # Nesterov-style extrapolation in multiplicative space
        ratio = a / np.maximum(a_prev, 1e-30)
        ratio = np.clip(ratio, 0.01, 100.0)
        a_nest = a * np.power(ratio, momentum)
        a_nest = norm_sum1(a_nest)

        # Compute gradient
        ac = fast_autocorr(a_nest)
        grad = compute_grad_topk(a_nest, ac, K, beta)

        # Multiplicative update with backtracking
        lr_try = lr
        a_new = a_nest.copy()
        for bt in range(6):
            log_a = np.log(np.maximum(a_nest, 1e-30)) - lr_try * grad
            log_a -= log_a.max()
            candidate = np.exp(log_a)
            candidate = np.maximum(candidate, 1e-30)
            candidate = candidate / candidate.sum()

            ac_c = fast_autocorr(candidate)
            m_c = ac_c.max()
            m_curr = fast_autocorr(a).max()

            if m_c < m_curr * (1.0 - 1e-10) or bt == 5:
                a_new = candidate
                break
            lr_try *= 0.5

        a_prev = a
        a = a_new
        iterations += 1

        # Track best (use actual max, not soft-max)
        current_m = fast_autocorr(a).max()

def peak_shave_v2(a0, time_limit, verbose=False, rng_seed=77):
    """
    Aggressive peak-shaving with multi-lag impact scoring.
    For each move, evaluates impact on ALL top lags simultaneously.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    total_moves = 0

    n_top_lags = 25

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        current_max = ac.max()

        if current_max < best_m - 1e-14:
            best_m = current_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 8000:
                # Simulated annealing style perturbation
                temp = float(np.exp(-stall / 5000.0)) * 0.1 + 0.01
                a = norm_sum1(best_a * (1.0 + rng.normal(0, temp, n)))
                stall = 0
                if stall > 80000:
                    break

        top_lags = np.argsort(ac)[::-1][:n_top_lags]
        k_star = int(top_lags[0])

        i_lo = max(0, k_star - n + 1)
        i_hi = min(n, k_star + 1)
        if i_lo >= i_hi:
            continue

        idx = np.arange(i_lo, i_hi)
        j_idx = k_star - idx
        contrib = a[idx] * a[j_idx]

        n_pairs = min(15, len(idx))
        pair_order = np.argsort(contrib)[::-1][:n_pairs]

        improved = False

        for pi in pair_order:
            if time.monotonic() - t0 > time_limit:
                break

            src = int(idx[pi])
            src_val = a[src]
            if src_val < 1e-14:
                continue

            # Multi-lag impact score for each destination
            # dst_score[m] = estimated increase in max autocorrelation if we move mass to m
            dst_scores = np.zeros(n)
            n_check_lags = min(12, len(top_lags))
            for k2 in top_lags[:n_check_lags]:
                k2i = int(k2)
                # db[k2]/da[m] = 2*a[k2-m] if 0 <= k2-m < n
                km = k2i - np.arange(n)
                valid = (km >= 0) & (km < n)
                dst_scores += ac[k2i] * a[valid[np.arange(n)[valid]]] if False else 0

            # Simpler: for each destination m, compute sum over top lags of a[|k-m|]
            for k2 in top_lags[:n_check_lags]:
                k2i = int(k2)
                for m in range(0, n, max(1, n // 200)):
                    km = k2i - m
                    if 0 <= km < n:
                        dst_scores[m] += a[km]

            # Interpolate dst_scores for all positions
            coarse_positions = np.arange(0, n, max(1, n // 200))
            if len(coarse_positions) > 1:
                dst_scores = np.interp(np.arange(n), coarse_positions, dst_scores)

            dst_scores[src] = np.inf

            # Also penalize positions with high existing values
            dst_scores += 0.05 * a

            dst = int(np.argmin(dst_scores))

            if a[dst] > 0.95 * src_val:
                continue

            # Try transfers with valid fractions
            for frac in [0.03, 0.08, 0.15, 0.25, 0.4]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                ac_trial = fast_autocorr(a_trial)
                m_trial = ac_trial.max()

                if m_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    total_moves += 1
                    break

        if not improved and stall > 500:
            # Batch random perturbation
            n_moves = int(rng.integers(20, 60))
            for _ in range(n_moves):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = float(rng.uniform(0.005, 0.03))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt

    return best_a, best_m
