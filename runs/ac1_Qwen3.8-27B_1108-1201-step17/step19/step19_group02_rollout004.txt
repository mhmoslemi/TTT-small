def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_obj(a):

    return 2.0 * len(a) * ac.max()

def compute_subgradient(a, peak_positions, weights):
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_positions)):
        k = int(peak_positions[idx])
        w = weights[idx]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        a_idx_start = k - i_start
        a_idx_end = k - i_end
        g[i_start:i_end + 1] += 2.0 * w * a[a_idx_start:a_idx_end - 1:-1]
    return g


def subgradient_optimize(a0, time_limit, K0=5, K_max=80, c0=2.0, momentum=0.7,
                         restart_every=3000, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    vel = np.zeros(n)
    gamma = c0
    stall = 0
    cur_K = K0
    restart_count = 0
    step_count = 0

    while time.monotonic() - t0 < time_limit:
        step_count += 1

        y = x + momentum * vel
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        ac = fast_convolve(y)
        cur_K_eff = min(cur_K, len(ac) - 1)
        top_idx = np.argsort(ac)[::-1][:cur_K_eff]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals / top_vals.sum()

        g = compute_subgradient(y, top_idx, w)

        eta = gamma / (np.sqrt(step_count) + 1.0)
        x_new = y - eta * g
        x_new = np.maximum(x_new, 1e-30)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new

        vel = momentum * vel + (1.0 - momentum) * (x_new - x)
        x = x_new

        M_new = fast_convolve(x).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            restart_count += 1
            noise_level = max(0.03 / (1.0 + 0.15 * restart_count), 0.001)
            pert = rng.normal(0, noise_level, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r < 1e-15:
                x_r = best_a.copy()
            else:
                x_r = x_r / s_r
            x = x_r
            vel = np.zeros(n)
            cur_K = min(cur_K + 10, K_max)
            gamma = max(gamma * 0.85, 0.3)
            if verbose:
                J = 2.0 * n * best_m
                print("    SubGrad restart (t=" + str(round(time.monotonic() - t0, 1)) + "s): J=" + repr(J))

        if verbose and step_count % 5000 == 0:
            J = 2.0 * n * best_m
            print("    SubGrad step=" + str(step_count) + " (t=" + str(round(time.monotonic() - t0, 1)) + "s): J=" + repr(J) + ", K=" + str(cur_K))

    return best_a, best_m


def peak_redistribute(a0, time_limit, n_peaks=10, n_transfers=20, verbose=False):

def build_constraint_matrix_vec(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw(a0, time_limit, solver, initial_K=200, max_K=2000, add_per_iter=25,
          verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_convolve(y)
    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())

    no_improve = 0
    max_no_improve = 800
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: stagnation, stopping")
            break
        iter_count += 1

        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep = np.argsort(margins)[:max_K]
            active_set = sorted([active_set[i] for i in keep])

        K = len(active_set)
        if K == 0:
            break

        coeffs = build_constraint_matrix_vec(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = sorted(active_set[:initial_K])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)

        default_alpha = 2.0 / (iter_count + 2.0)
        y_new = (1.0 - default_alpha) * y + default_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if verbose and iter_count % 100 == 0:
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(iter_count) + " (t=" + str(round(time.monotonic() - t0, 1)) + "s): J=" + repr(J) + ", |S|=" + str(len(active_set)))

    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW done (t=" + str(round(time.monotonic() - t0, 1)) + "s): J=" + repr(J) + ", iters=" + str(iter_count))
    return best_a, best_m


def spectral_shaping(a0, time_limit, n_iter=200, verbose=False):
    """Try to flatten the power spectrum to reduce convolution peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_convolve(a)
        target = np.percentile(ac, 50)
        # Compute FFT
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        # Try to boost low-power frequencies
        power_clipped = np.minimum(power, target * n * 0.5)
        # Scale back
        ratio = power_clipped / np.maximum(power, 1e-30)
        A_shaped = A * np.sqrt(np.maximum(ratio, 0.01))
        a_new = np.fft.irfft(A_shaped, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
            M_new = fast_convolve(a_new).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a_new.copy()
            a = a_new

    return best_a, best_m


def parametric_init(n, time_limit, best_init, best_score):
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score

    candidates = []
    # Power laws (asymmetric)
    for p in [0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))
    # Two-sided
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        for beta in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            candidates.append(norm_sum1(a))
    # Gaussians
    for c in [0.15, 0.25, 0.5, 0.75, 0.85]:
        for sigma in [0.03, 0.05, 0.1, 0.15, 0.2, 0.3]:
            candidates.append(norm_sum1(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))))
    # Exponentials
    for rate in [0.5, 1, 2, 3, 5, 10, 20]:
        candidates.append(norm_sum1(np.exp(-rate * x)))
        candidates.append(norm_sum1(np.exp(rate * (x - 1.0))))
    # Cosine-based
    for freq in [1, 2, 3]:
        for ph in [0.0, np.pi / 4.0, np.pi / 2.0, 3.0 * np.pi / 4.0]:
            a = np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x + ph)) / 2.0, 0.001)
            candidates.append(norm_sum1(a))
    # Step functions
    for w in [0.1, 0.2, 0.3, 0.4]:
        a = np.ones(n)
        a[:int(w * n)] = 5.0
        candidates.append(norm_sum1(a))
        a2 = np.ones(n)
        a2[int(w * n):] = 5.0
        candidates.append(norm_sum1(a2))
    # Sine variants
    for p in [1, 2, 3, 4]:
        candidates.append(norm_sum1(np.sin(np.pi * x) ** p))
    # Combined shapes
    for w1 in [0.3, 0.5, 0.7]:
        candidates.append(norm_sum1(w1 * (i_arr + 1.0) ** 0.5 + (1 - w1) * (n - i_arr) ** 0.5))

    for a in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        v = compute_obj(a)
        if v < best_val:
            best_val = v
            best_a = a.copy()

    # Random search
    rng = np.random.default_rng(2024)
    for trial in range(300):
        if time.monotonic() - t0 > time_limit:
            break
        a = rng.random(n)
        a = a ** rng.uniform(0.2, 8.0)
        a = norm_sum1(a)
        v = compute_obj(a)
        if v < best_val:
            best_val = v
            best_a = a.copy()

    print("  Parametric: best J=" + repr(best_val))
    return best_a, best_val

        best_seq, best_score = try_update(best_seq, best_score, a_out, "subgrad2", t_global)

    # Phase 5: Peak redistribution
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Peak Redistribute (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        a_out, _ = peak_redistribute(norm_sum1(best_seq), t5,
                                     n_peaks=15, n_transfers=30, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakredist", t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    try_ns = [n_best - 500, n_best + 500, n_best + 1000]
    try_ns = [l for l in try_ns if 3000 <= l <= 10000]

    for new_n in try_ns[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print("\n=== Phase 6: Scale n=" + str(new_n) + " ===")
        a_long = interpolate_seq(best_seq, new_n)

        t6a = min(remaining * 0.40, 180.0)
        if t6a > 30:
            a_out, _ = subgradient_optimize(a_long, t6a,
                                            K0=8, K_max=50, c0=2.5, momentum=0.7,
                                            restart_every=3000, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "subgrad_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 9000:
            t6b = min(remaining * 0.35, 120.0)
            if t6b > 20:
                a_out, _ = lp_fw(norm_sum1(best_seq), t6b, solver,
                                 initial_K=250, max_K=2000, add_per_iter=30, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

    # Phase 7: Final LP-FW on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final LP-FW (n=" + str(n_best) + ") ===")
    if remaining > 60 and n_best <= 9000:
        t7 = min(remaining * 0.40, 250.0)
        if t7 > 30:
            a_out, _ = lp_fw(norm_sum1(best_seq), t7, solver,
                             initial_K=400, max_K=3000, add_per_iter=40, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # Phase 8: Final subgradient polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final SubGrad (n=" + str(n_best) + ") ===")
    if remaining > 30:
        t8 = min(remaining * 0.50, 150.0)
        if t8 > 15:
            a_out, _ = subgradient_optimize(norm_sum1(best_seq), t8,
                                            K0=20, K_max=80, c0=1.5, momentum=0.85,
                                            restart_every=2000, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "subgrad_final", t_global)

    # Phase 9: Peak redistribution final
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t9 = min(remaining * 0.60, 80.0)
        if t9 > 10:
            a_out, _ = peak_redistribute(norm_sum1(best_seq), t9,
                                         n_peaks=20, n_transfers=40, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peakredist_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
