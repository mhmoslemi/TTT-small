def mirror_descent_step(a, g, eta):
    log_a = np.log(np.maximum(a, 1e-30)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def accelerated_mirror_descent(a0, time_limit, top_k=80, temp_init=0.01, temp_final=0.1,
                                eta_init=0.3, momentum=0.9, restart_every=3000,
                                rng_seed=42, verbose=False):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    v = np.zeros(n)
    x = a.copy()
    it = 0
    stall = 0
    temperature = temp_init
    eta = eta_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / max(restart_every * 3, 1))
        temperature = temp_init + (temp_final - temp_init) * progress

        ac = fast_autocorr(x)
        g = compute_gradient_fast(x, ac, top_k, temperature)

        # Nesterov momentum
        v = momentum * v + g
        g_eff = g + momentum * v

        # Mirror descent step
        eta_t = eta / (1.0 + 0.01 * np.sqrt(it))
        a_new = mirror_descent_step(x, g_eff, eta_t)

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        x = a_new

        if stall > restart_every:
            # Restart with perturbation
            pert_scale = 0.02 + 0.03 * (stall / restart_every)
            pert = rng.normal(0, pert_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            v = np.zeros(n)
            stall = 0
            eta = eta_init * 0.5

        if it % 500 == 0:
            # Try to improve top_k and temperature
            if best_m < 0.25:
                top_k = min(top_k + 10, 300)
            temperature = min(temperature * 1.02, temp_final)

        if verbose and it % 2000 == 0:

            J = 2.0 * n * best_m
            print(f"    AMD it={it} ({elapsed:.0f}s): J={J:.6f}, stall={stall}, tau={temperature:.4f}")

    return best_a, best_m
