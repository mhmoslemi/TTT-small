def compute_gradient_softmax(x, beta, n_top=50):
    """
    Compute gradient of (2n/beta)*log(sum exp(beta*c[k])) w.r.t. x (log-space).
    Uses log-sum-exp approximation of max with top-n_top lags.
    """
    n = len(x)
    a = softmax_from_x(x)
    c = fast_autocorr(a)

    # Select top lags
    n_use = min(n_top, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:n_use]
    top_vals = c[top_idx]

    # Soft-max weights
    max_v = top_vals.max()
    expo = beta * (top_vals - max_v)
    expo = np.clip(expo, -200, 200)

    if w_sum < 1e-30:

            stall += 1

        # Adaptive beta
        if stall > 1000:
            beta_temp = min(beta_temp * 1.3, 2000.0)
            n_top = min(n_top + 5, 100)
        elif stall == 0 and beta_temp > beta_init:
            beta_temp = max(beta_temp * 0.9, beta_init)

        # Compute gradient
        grad_x, _ = compute_gradient_softmax(x, beta_temp, n_top)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * grad_x ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # Learning rate decay
        if t % 5000 == 0:
            lr *= 0.95

        # Periodic restart
        if stall > 5000:
            stall = 0
            pert_type = int(rng.integers(5))
            if pert_type == 0:
                x += rng.normal(0, 0.1, n)
            elif pert_type == 1:
                freqs = rng.integers(1, 20, size=3)
                for f in freqs:
                    x += 0.03 * np.sin(2 * np.pi * f * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
            elif pert_type == 2:
                x = np.log(np.maximum(best_a, 1e-15))
                x -= x.max()
                x += rng.normal(0, 0.05, n)
            elif pert_type == 3:
                a_pert = norm_sum1(best_a * (1.0 + rng.normal(0, 0.15, n)))
                x = np.log(np.maximum(a_pert, 1e-15))
                x -= x.max()
            else:
                # Symmetric perturbation
                half = n // 2
                noise = rng.normal(0, 0.08, half)
                x[:half] += noise
                x[half:] += noise[::-1] if n % 2 == 0 else noise[:-1][::-1]

            lr = max(lr * 0.7, 0.001)
            beta_temp = max(beta_temp * 0.5, beta_init)

    if verbose:
        print(f"      Adam done: score={best_score:.6f}, time={time.monotonic()-t0:.0f}s")
    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True, rng_seed=77):
    """
    Coordinate descent: for the peak lag, move mass from high-contributing
    positions to low-impact positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50000:
                break

        # Get peak lag
        k_peak = int(np.argmax(ac))

        # Compute contribution to peak at each position
        i_lo = max(0, k_peak - n + 1)
        i_hi = min(n, k_peak + 1)
        if i_lo >= i_hi:
            break
        idx = np.arange(i_lo, i_hi)
        j_idx = k_peak - idx
        contrib = a[idx] * a[j_idx]

        # Find top contributing source
        order = np.argsort(contrib)[::-1]

        improved = False
        for pi in order[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[pi])
            src_val = a[src]
            if src_val < 1e-14:
                continue

            # Compute impact at each destination on top few lags
            n_top = min(20, 2 * n - 1)
            top_lags = np.argsort(ac)[::-1][:n_top]

            scores = np.zeros(n)
            for k2 in top_lags[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                if s2 < e2:
                    ii = np.arange(s2, e2)
                    jj = k2i - ii
                    valid = (jj >= 0) & (jj < n)
                    scores[ii[valid]] += ac[k2i] * a[jj[valid]]

            scores[src] = np.inf
            dst = int(np.argmin(scores))

            if dst == src:
                continue

            # Try transfer
            for frac in [0.01, 0.03, 0.05, 0.1, 0.15]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved and stall > 500:
            # Random perturbation
            n_moves = int(rng.integers(3, 15))
            for _ in range(n_moves):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = float(rng.uniform(0.002, 0.02))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            m_new = fast_autocorr(a).max()
            if m_new < best_m:
                best_m = m_new
                best_a = a.copy()
                stall = 0

    if verbose:
        print(f"      Coord descent done: score={2.0*n*best_m:.6f}, time={time.monotonic()-t0:.0f}s")
    return best_a, best_m


def parametric_init(n, time_limit=10.0):
    """Quick parametric search for good initializations."""
    t0 = time.monotonic()
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)

    candidates = []

    # Uniform
    candidates.append(np.ones(n) / n)

    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))

    # Gaussians
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sig in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]:
            candidates.append(norm_sum1(np.exp(-((x - c) ** 2) / (2 * sig ** 2))))

    # Bimodal
    for sig in [0.05, 0.1, 0.15, 0.2]:
        candidates.append(norm_sum1(
            np.exp(-((x - 0.2) ** 2) / (2 * sig ** 2)) + np.exp(-((x - 0.8) ** 2) / (2 * sig ** 2))))
        candidates.append(norm_sum1(
            np.exp(-((x - 0.25) ** 2) / (2 * sig ** 2)) + np.exp(-((x - 0.75) ** 2) / (2 * sig ** 2))))

    # Sine
    for f in [1, 2, 3, 4]:
        candidates.append(norm_sum1(np.sin(np.pi * f * x) ** 2))

    # Exponential
    for r in [1, 2, 3, 5, 10]:
        candidates.append(norm_sum1(np.exp(-r * x)))
        candidates.append(norm_sum1(np.exp(r * x)))

    # Symmetric cosine
    candidates.append(norm_sum1(1.0 + np.cos(2 * np.pi * x)))
    candidates.append(norm_sum1(1.0 + 0.5 * np.cos(4 * np.pi * x)))

    # Evaluate
    best_cand = None
    best_val = float('inf')
    for c in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        v = score_from_ac(c)
        if v < best_val:
            best_val = v
            best_cand = c.copy()

    return best_cand if best_cand is not None else np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Multi-phase: init -> L-BFGS-B -> Adam -> coord descent -> length exploration -> final polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 5000
        best_seq = parametric_init(n0, time_limit=5.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = evaluate_safe(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")

    # Also try uniform baseline
    a_uni = np.ones(n0) / n0
    sc_uni = evaluate_safe(a_uni)
    if np.isfinite(sc_uni) and sc_uni < best_score:
        best_score = sc_uni
        best_seq = a_uni.copy()
        print(f"  Uniform: {sc_uni:.6f}")

    # Try parametric init at current length
    a_param = parametric_init(n0, time_limit=5.0)
    check_update(a_param, f"param_n{n0}")

    # === Phase 1: L-BFGS-B on parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: L-BFGS-B (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 350.0)
    if t1 > 30:
        a_out, _ = optimize_l_bfgs(best_seq, t1, beta_init=50.0, n_top=50, verbose=True)
        check_update(a_out, f"lbfgs_n{len(best_seq)}")

    # === Phase 2: Adam optimizer ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Adam (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 20:
        a_out, _ = optimize_adam(best_seq, t2, beta_init=30.0, lr_init=0.02, verbose=True, rng_seed=42)
        check_update(a_out, f"adam_n{len(best_seq)}")

    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coord descent (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 15:
        a_out, _ = coordinate_descent(best_seq, t3, verbose=True, rng_seed=123)
        check_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 4: Fine Adam (higher beta, smaller lr) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Fine Adam (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_out, _ = optimize_adam(best_seq, t4, beta_init=200.0, lr_init=0.005, verbose=False, rng_seed=99)
        check_update(a_out, f"fine_adam_n{len(best_seq)}")

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration (budget={remaining:.0f}s) ===")

    length_tries = []
    for delta in [-1000, -500, -200, 200, 500, 1000, 1500, 2000, 3000]:
        nn = n_best + delta
        if 2000 <= nn <= 10000 and nn != n_best:
            length_tries.append(nn)
    for rn in [3000, 4000, 5000, 6000, 7000, 8000]:
        if rn != n_best and 2000 <= rn <= 10000:
            length_tries.append(rn)
    length_tries = sorted(set(length_tries))

    for new_n in length_tries[:6]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")

        a_interp = interpolate_sequence(best_seq, new_n)

        # Quick L-BFGS-B on new length
        t5a = min(remaining * 0.3, 100.0)
        if t5a > 15:
            a_out, _ = optimize_l_bfgs(a_interp, t5a, beta_init=50.0, n_top=50, verbose=False)
            check_update(a_out, f"lbfgs_n{new_n}")

        # Adam on new length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t5b = min(remaining * 0.25, 80.0)
            if t5b > 10:
                a_out, _ = optimize_adam(best_seq if len(best_seq) == new_n else a_interp,
                                         t5b, beta_init=50.0, lr_init=0.01, verbose=False, rng_seed=new_n % 1000)
                check_update(a_out, f"adam_n{new_n}")

    # === Phase 6: Final polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_best}, budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.35, 150.0)
    if t6 > 15:
        a_out, _ = optimize_adam(best_seq, t6, beta_init=500.0, lr_init=0.003, verbose=False, rng_seed=777)
        check_update(a_out, f"final_adam_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t7 = min(remaining * 0.4, 100.0)
        if t7 > 10:
            a_out, _ = coordinate_descent(best_seq, t7, verbose=False, rng_seed=42)
            check_update(a_out, f"final_coord_n{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")

    return [float(x) for x in best_seq]
