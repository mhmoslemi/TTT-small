"""
Minimizes 2*n*max(autocorr(a))/(sum(a))^2 using:
1. Spectral shaping (frequency domain compression with non-negativity projection)
2. Nesterov accelerated projected gradient with FFT-based gradient and backtracking
3. LP-Frank-Wolfe with adaptive active set
4. Peak-targeted coordinate descent
5. Multi-start with diverse parametric initializations and evolutionary selection

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def objective_fast(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def simplex_project(z):
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def compute_gradient_fft(y, w, n):
    """Compute gradient of sum_k w[k]*conv(y,y)[k] w.r.t. y using FFT.
    w has length 2n-1 (the autocorrelation weights).
    grad[j] = conv(w, y_rev)[n-1+j] + conv(w, y_rev)[n-1-j]
    """
    conv_len = (2 * n - 1) + n - 1  # = 3n-2
    conv_size = 1 << (conv_len - 1).bit_length() if conv_len > 1 else 2
    Wr = np.fft.rfft(w, conv_size)
    Yr = np.fft.rfft(y[::-1], conv_size)
    conv_w_yr = np.fft.irfft(Wr * Yr, conv_size)
    # grad[j] = conv_w_yr[n-1+j] + conv_w_yr[n-1-j]
    idx1 = np.arange(n - 1, 2 * n - 1)  # n-1+j for j=0..n-1
    idx2 = np.arange(n - 1, -1, -1)     # n-1-j for j=0..n-1
    grad = conv_w_yr[idx1] + conv_w_yr[idx2]
    return grad

    """Iteratively flatten the power spectrum while maintaining non-negativity."""
    n = len(a)
    a = normalize(a.copy())
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    rng = np.random.default_rng(seed)
    
    for it in range(iters):
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        p_max = power.max()
        if p_max < 1e-20:
            break
        # Geometric mean as target
        log_power = np.log(power + 1e-30)
        target_log = log_power.mean()
        # Compress: new_log_power = target + alpha * (old_log - target)
        new_log_power = target_log + alpha * (log_power - target_log)
        new_power = np.exp(new_log_power)
        # Reconstruct with original phase
        phase = np.angle(A)
        new_A = np.sqrt(new_power) * (np.cos(phase) + 1j * np.sin(phase))
        a_new = np.fft.irfft(new_A, fft_size)[:n]
        # Add small noise to escape symmetries
        a_new += rng.normal(0, 1e-4, n)
        a = normalize(a_new)
    
    return a

            print("    Nest2 it=" + str(it) + " (" + str(round(elapsed,1)) + "s): J=" + repr(round(J,6)) + " beta=" + repr(round(beta,1)))
    
    return best_a, best_b_max

def lp_fw(a0, time_limit, verbose=True):
    """LP-Frank-Wolfe with adaptive active set management."""
    try:
        import cvxpy as cp
    except ImportError:
        return a0, fast_autocorr(a0).max()
    
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Find solver
    solver = None
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue
    if solver is None:
        return a0, fast_autocorr(a0).max()
    
    y = a.copy()
    ac_y = fast_autocorr(y)
    initial_K = min(300, 2 * n - 1)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 800
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: converged at iter " + str(t))
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-12:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Manage active set
        constraint_list = sorted(active_set)
        max_K = min(2000, 2 * n - 1)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K // 2])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violations
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-6)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:20]]
            for k in top_v:
                active_set.add(int(k))
        
        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = list(set([default_alpha, max(0.02, default_alpha * 0.5),
                           max(0.02, default_alpha * 1.5), 1.0 / (t + 1)]))
        alphas = [max(0.01, min(1.0, a)) for a in alphas]
        
        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y_new = y_new / s_y
        y = y_new
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW it=" + str(t) + " (" + str(round(elapsed,1)) + "s): J=" + repr(round(J,6)) + " |S|=" + str(len(active_set)))
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=True):
    """Peak-targeted coordinate descent with adaptive moves."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(314)
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:30]
        improved = False
        
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue
            
            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            src_val = a[src]
            partner_val = a[partner] if 0 <= partner < n else 0.0
            
            if src_val >= partner_val:
                reduce_idx = src
            else:
                reduce_idx = partner
            
            if a[reduce_idx] < 1e-12:
                continue
            
            # Compute pressure to find safe destination
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            
            # Pick lowest-value non-peak position
            cand_vals = a[candidates]
            dst = int(candidates[np.argmin(cand_vals)])
            if dst == reduce_idx:
                mask = candidates != reduce_idx
                if mask.any():
                    dst = int(candidates[mask][np.argmin(cand_vals[mask])])
                else:
                    continue
            
            for frac in [0.05, 0.1, 0.2, 0.3]:
                amt = a[reduce_idx] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random exploration
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (" + str(round(elapsed,1)) + "s): J=" + repr(round(J,6)) + " stall=" + str(stall))
    
    return best_a, best_m

def parametric_init(n, time_limit, best_init, best_score):
    """Try diverse parametric initializations and random starts."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score
    
    candidates = []
    # Power laws (asymmetric)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
        candidates.append(np.exp(p * (i_arr / n - 0.5)))
    # Symmetric U-shape
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p * ((n - 1) - i_arr) ** p)
    # Cosine windows
    for beta in [0.5, 1.0, 2.0, 5.0, 10.0]:
        candidates.append(0.54 + 0.46 * np.cos(2 * np.pi * x))
        candidates.append(np.abs(np.cos(np.pi * x)) ** beta)
    # Sine powers
    for p in [1, 2, 3, 4, 6, 8]:
        candidates.append(np.sin(np.pi * x) ** p)
    # Gaussian
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.05, 0.1, 0.15, 0.2, 0.3]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma ** 2)))
    # Bimodal
    for s in [0.1, 0.15, 0.2]:
        candidates.append(np.exp(-((x - 0.25) ** 2) / (2 * s**2)) + np.exp(-((x - 0.75) ** 2) / (2 * s**2)))
    # Combination
    candidates.append(np.exp(-2.0 * x) * (1.0 - np.exp(-2.0 * x)))
    candidates.append(np.power(np.sin(np.pi * x), 1.0) * np.exp(-3.0 * (x - 0.5)**2))
    
    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a = normalize(a_raw)
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()
    
    # Random with structure
    rng = np.random.default_rng(42)
    for _ in range(100):
        if time.monotonic() - t0 > time_limit:
            break
        shape = rng.uniform(0.3, 4.0)
        a_raw = (1.0 + x) ** (rng.uniform(-3, 3)) * (1.0 - x + 1e-3) ** (rng.uniform(-3, 3))
        a_raw *= np.exp(rng.normal(0, 0.3, n))
        a = normalize(a_raw)
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()
    
    return best_a, best_val

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, x)) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + repr(round(sc, 8)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed,1)) + "s)")
    return best_seq, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):

        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):

    # Final safety check
    result = np.asarray(best_seq, dtype=np.float64)
    result = np.clip(result, 0.0, 1000.0)
    return [float(x) for x in result]
