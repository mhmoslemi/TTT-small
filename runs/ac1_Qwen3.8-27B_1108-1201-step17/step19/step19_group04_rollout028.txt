def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s

def self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]

def conv_ab(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]

def corr_shift(w, a):
    w = np.asarray(w, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = w.size
    size = next_pow2(L + n - 1)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fw * fa.conj(), size)[:n]
    return c

    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    q = self_conv(a)
    return 2.0 * float(a.size) * float(np.max(q))

        if s <= 1e-15:
            break
        b = b / s
        q_new = self_conv(b)
        M_new = float(np.max(q_new))
        if M_new < M - 1e-15:
            a = b
            q = q_new
        else:
            break

    return a, float(np.max(q))


def soft_direction_best(a, q, max_dirs=6):
    """Try diverse softmax-mirror directions and return best line-search result."""
    a = np.asarray(a, dtype=np.float64)
    q = np.asarray(q, dtype=np.float64)
    n = a.size
    if n <= 1:
        return None, float(np.max(q))
    M_cur = float(np.max(q))

    best_new = None
    best_M = M_cur
    used = 0

    focus_opts = (2.0, 5.0, 12.0, 30.0)
    trust_opts = (0.2, 0.5, 1.0, 2.0)

    for focus in focus_opts:
        if used >= max_dirs:
            break
        w = weights_from_q(q, focus)
        c = corr_lambda_a(w, a)
        cmax = float(np.max(c))
        cmin = float(np.min(c))
        scale = cmax - cmin
        if scale <= 1e-15:
            continue
        for trust in trust_opts:
            if used >= max_dirs:
                break
            move = trust * (cmax - c) / scale
            move = np.clip(move, -50.0, 50.0)
            d = a * np.exp(move)
            d = normalize_vec(d)
            new, Mnew = line_search_direction(a, d, q, None)
            used += 1
            if Mnew < best_M - 1e-13:
                best_M = Mnew
                best_new = new

    if best_new is None:
        return None, M_cur
    return best_new, best_M


def smooth_polish_nesterov(a0, time_limit, rng_seed=11, verbose=False):
    """
    Nesterov-accelerated multiplicative gradient polish.
    """
    if time_limit <= 0.0:
        return normalize_vec(a0)
    t0 = time.monotonic()
    a = normalize_vec(a0)
    a_prev = a.copy()
    n = a.size
    if n <= 1:
        return a
    best = a.copy()
    best_score = objective_norm(a)
    trust = 0.40
    focus = 5.0
    fails = 0
    momentum = 0.0
    mom_rate = 0.01
    rng = np.random.default_rng(rng_seed)
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        q = self_conv(a)
        M = float(np.max(q))
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            fails = 0
        else:
            fails += 1

        if fails > 600:
            break
        if fails > 40 and fails % 25 == 0:
            focus = min(focus * 1.15, 80.0)
            trust = max(trust * 0.82, 0.02)
        if fails > 150 and fails % 150 == 0:
            scale_noise = 0.015 / (1.0 + 0.08 * (fails // 150))
            cand = best + scale_noise * rng.normal(size=best.size) * best
            cand = np.maximum(cand, 1e-14)
            cand = normalize_vec(cand)
            sc = objective_norm(cand)
            if np.isfinite(sc) and sc <= best_score + 1e-8:
                a = cand
                a_prev = cand.copy()
                momentum = 0.0
                fails = 0
                trust = 0.35
            else:
                a = best.copy()

        w = weights_from_q(q, focus)
        c = corr_lambda_a(w, a)
        cmax = float(np.max(c))
        cmin = float(np.min(c))
        scale = cmax - cmin
        if scale <= 1e-15:
            break

        move = trust * (cmax - c) / scale
        move = np.clip(move, -50.0, 50.0)
        b = a * np.exp(move)
        b = normalize_vec(b)

        # Nesterov: blend with previous iterate
        momentum = mom_rate * momentum
        a_yoyoyo = (1.0 + momentum) * b - momentum * a_prev
        a_yoyoyo = np.maximum(a_yoyoyo, 0.0)
        s_yoyoyo = float(a_yoyoyo.sum())
        if s_yoyoyo > 1e-15:
            a_yoyoyo = a_yoyoyo / s_yoyoyo
            sc_yoyoyo = objective_norm(a_yoyoyo)
            if sc_yoyoyo < objective_norm(b):
                a = a_yoyoyo
            else:
                a = b
        else:
            a = b
        a_prev = b.copy()

        sc = objective_norm(a)
        if sc < best_score - 1e-13:
            best = a.copy()
            best_score = sc
            a = a.copy()
            fails = 0
            trust = min(trust * 1.05, 5.0)
            mom_rate = min(mom_rate * 1.02, 0.4)
        elif sc < score - 1e-14:
            a = a.copy()
            fails = 0
            trust = min(trust * 1.08, 5.0)
            focus = max(focus * 0.98, 1.0)
        else:
            fails += 1
            trust = max(trust * 0.75, 0.02)
            focus = min(focus * 1.1, 80.0)
            mom_rate = max(mom_rate * 0.95, 0.005)

        iter_count += 1
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print("    smoothNest: it=%d t=%.1fs best=%.9f trust=%.3f focus=%.1f fails=%d"
                  % (iter_count, elapsed, best_score, trust, focus, fails))

    return best


def lp_fw_trust(a0, time_limit, verbose=True):
    """
    LP-based Frank-Wolfe with trust region and adaptive active set.
    """
    if time_limit <= 0.0:
        return normalize_vec(a0)
    t0 = time.monotonic()
    y = normalize_vec(a0)
    n = y.size
    if n <= 5:
        return y
    best = y.copy()
    best_score = objective_norm(y)
    no_improve = 0
    total_improve = 0
    last_print = t0
    iter_no = 0

    if n > 8000:
        initK = 200
        maxK = 600
    elif n > 5000:
        initK = 300
        maxK = 900
    else:
        initK = 400
        maxK = 1200

    K = initK
    use_lp = HAVE_LP
    lp_fails = 0
    trust_radius = 0.3
    consecutive_small = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining <= 2.0:
            break

        qy = self_conv(y)
        M_y = float(np.max(qy))
        score_y = 2.0 * n * M_y

        if score_y < best_score - 1e-12:
            best = y.copy()
            best_score = score_y

        if no_improve > 300:
            if time.monotonic() - t0 > 0.85 * time_limit:
                break
            y = best.copy()
            no_improve = 0
            trust_radius = 0.3

        order = np.argsort(qy)[::-1]
        Kcur = int(min(K, len(qy), 1800))
        act = order[:Kcur]
        q_active = qy[act]

        g_dir = None
        T = None

        if use_lp and Kcur >= 20:
            tstep = time.monotonic()
            g_dir, T = solve_lp_offset(y, act, q_active)
            spend = time.monotonic() - tstep
            if g_dir is None:
                lp_fails += 1
                if lp_fails >= 4:
                    use_lp = False
            else:
                lp_fails = 0
                if spend > 120.0:
                    use_lp = False
                elif spend > max(30.0, 0.15 * remaining):
                    K = max(100, K - 50)

        improved_this_iter = False

        if g_dir is not None:
            new, Mnew = line_search_direction(y, g_dir, qy, T)
            if Mnew < M_y - 1e-13:
                step_size = 1.0 - float(np.dot(new, y) / (float(np.dot(y, y)) + 1e-30))
                if step_size < 0.01:
                    consecutive_small += 1
                else:
                    consecutive_small = 0

                y = new
                no_improve = 0
                total_improve += 1
                improved_this_iter = True

                if consecutive_small > 10:
                    K = min(K + 30, maxK)
                    trust_radius = min(0.5, trust_radius * 1.1)
        else:
            new, Mnew = soft_direction_best(y, qy, max_dirs=8)
            if Mnew < M_y - 1e-13:
                y = new
                no_improve = 0
                total_improve += 1
                improved_this_iter = True
                K = min(K + 20, maxK)

        if not improved_this_iter:
            no_improve += 1
            if no_improve in (1, 3, 8, 20, 50):
                new2, M2 = soft_direction_best(y, qy, max_dirs=8)
                if M2 < M_y - 1e-13:
                    y = new2
                    no_improve = 0
                    total_improve += 1
                    improved_this_iter = True
            if not improved_this_iter:
                K = min(K + 25, maxK)
                if no_improve > 200:
                    # Try KKT-based update as escape
                    y_kkt, M_kkt = kkt_multiplicative_update(y, qy, trust=0.3, n_active=Kcur, n_iters=3)
                    sc_kkt = 2.0 * n * M_kkt
                    if sc_kkt < score_y - 1e-13:
                        y = y_kkt
                        no_improve = 0
                        total_improve += 1
                    else:
                        if no_improve > 350:
                            break

        # Adaptive K
        if improved_this_iter:
            if no_improve == 0 and total_improve % 20 == 0:
                K = max(100, K - 10)
        else:
            if no_improve % 30 == 0:
                K = min(K + 40, maxK)

        iter_no += 1
        now = time.monotonic()
        if verbose and (iter_no % 30 == 0 or (now - last_print) > 30.0):
            print("    LP-FW-TR: it=%d t=%.1fs best=%.9f cur=%.9f K=%d lp=%s noimp=%d improve=%d"
                  % (iter_no, now - t0, best_score, score_y, K, use_lp, no_improve, total_improve))
            last_print = now

    return best


def interpolate_sequence(f, new_n):
    f = normalize_vec(f)
    old_n = f.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return f.copy()
    if old_n <= 1:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f)
    return normalize_vec(g)


def make_parametric_starts(n, rng_seed=42):
    """Generate diverse parametric starting sequences of length n."""
    out = []
    x = np.arange(n, dtype=np.float64) / float(max(n - 1, 1))

    # Sinc^2 and variants
    for pw in (1.0, 1.5, 2.0, 3.0):
        out.append(normalize_vec(np.sin(np.pi * x) ** (2.0 * pw)))

    # Cosine bumps
    for pw in (0.5, 1.0, 1.5, 2.0, 3.0, 4.0):
        out.append(normalize_vec(np.cos(np.pi * (x - 0.5)) ** (2.0 * pw)))

    # Gaussian
    for sigma in (0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5):
        out.append(normalize_vec(np.exp(-((x - 0.5) ** 2) / (2.0 * sigma * sigma))))

    # Double Gaussian
    for c in (0.25, 0.3, 0.35):
        for sigma in (0.08, 0.12, 0.18, 0.25):
            seq = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)) + \
                  np.exp(-((x - (1.0 - c)) ** 2) / (2.0 * sigma * sigma))
            out.append(normalize_vec(seq))

    # Triple Gaussian
    for sigma in (0.08, 0.12, 0.2):
        seq = np.exp(-((x - 0.2) ** 2) / (2.0 * sigma * sigma)) + \
              np.exp(-((x - 0.5) ** 2) / (2.0 * sigma * sigma)) + \
              np.exp(-((x - 0.8) ** 2) / (2.0 * sigma * sigma))
        out.append(normalize_vec(seq))

    # Exponential decay
    for r in (1.0, 2.0, 3.0, 5.0, 8.0):
        out.append(normalize_vec(np.exp(-r * x) + np.exp(-r * (1.0 - x))))

    # Power functions
    for a_exp in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        for b_exp in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
            if a_exp + b_exp > 5.0:
                continue
            seq = ((x + 1e-8) ** a_exp) * (((1.0 - x) + 1e-8) ** b_exp)
            out.append(normalize_vec(seq))

    # Lorentzian (Cauchy)
    for gamma in (0.05, 0.1, 0.2, 0.3, 0.5):
        out.append(normalize_vec(1.0 / (1.0 + ((x - 0.5) / gamma) ** 2)))

    # Random smooth functions
    rng = np.random.default_rng(rng_seed)
    for _ in range(30):
        n_peaks = int(rng.integers(1, 5))
        seq = np.zeros(n, dtype=np.float64)
        for _ in range(n_peaks):
            ctr = rng.uniform(0.1, 0.9)
            wd = rng.uniform(0.04, 0.35)
            ht = rng.uniform(0.3, 1.5)
            seq += ht * np.exp(-((x - ctr) ** 2) / (2.0 * wd * wd))
        seq += 1e-8
        out.append(normalize_vec(seq))

    # Random Fourier
    for _ in range(15):
        seed_f = rng.integers(10000)
        rng_f = np.random.default_rng(seed_f)
        coeffs = []
        for freq in range(1, 8):
            amp = rng_f.uniform(0.1, 1.0) * (1.0 / freq)
            phase = rng_f.uniform(0, 2 * np.pi)
            coeffs.append((amp, freq, phase))
        seq = np.ones(n) * 2.0
        for amp, freq, phase in coeffs:
            seq += amp * np.cos(2.0 * np.pi * freq * x + phase)
        seq = np.maximum(seq, 1e-8)
        out.append(normalize_vec(seq))

    return out


def reduce_starts(cand_scores, max_kept=6):
    """Select diverse best starts."""
    remaining = list(cand_scores)
    kept = []
    for _ in range(max_kept):
        bi = -1
        bs = float('inf')
        for i, item in enumerate(remaining):
            sc = float(item[0])
            if np.isfinite(sc) and sc < bs:
                bs = sc
                bi = i
        if bi < 0:
            break
        seq = remaining[bi][1]
        ok = True
        for kseq in kept:
            if seq.size == kseq.size:
                dp = float(np.dot(seq, kseq))
                if dp > 0.997:
                    ok = False
                    break
        if ok:
            kept.append(seq.copy())
        remaining.pop(bi)
    return kept


def transform_search(a0, time_limit):
    """Try power, blend, and mirror transforms."""
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    best = a.copy()
    best_score = objective_norm(a)

    for g in (0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0, 5.0):
        if time.monotonic() - t0 > time_limit:
            break
        b = np.maximum(a, 1e-30) ** g
        b = normalize_vec(b)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    u = np.ones(n, dtype=np.float64) / float(n)
    for w in (0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5):
        if time.monotonic() - t0 > time_limit:
            break
        b = normalize_vec((1.0 - w) * a + w * u)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    rev = a[::-1].copy()
    for w in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7):
        if time.monotonic() - t0 > time_limit:
            break
        b = normalize_vec((1.0 - w) * a + w * rev)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    return best


def trim_search(a0, time_limit):
    """Try trimming edges."""
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n < 30 or time_limit <= 0.0:
        return a
    best = a.copy()
    best_score = objective_norm(a)
    max_side = min(n // 3, 5000)

    samples = [0, 1, 2, 3, 4, 5, 8, 12, 18, 27, 40, 60, 90, 130, 200, 300, 500,
               750, 1000, 1500, 2000, 4000]
    samples = [v for v in samples if v <= max_side]

    for l in samples:
        if time.monotonic() - t0 > time_limit:
            break
        if l >= n - 10:
            continue
        b = normalize_vec(a[l:])
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    for r in samples:
        if time.monotonic() - t0 > time_limit:
            break
        if r >= n - 10:
            continue
        b = normalize_vec(a[:-r])
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    # Fine search around best
    for dl in range(-20, 21, 2):
        for dr in range(-20, 21, 2):
            if time.monotonic() - t0 > time_limit:
                break
            l = dl
            r = dr
            if l < 0 or r < 0 or l + r >= n - 10:
                continue
            if l > 0 or r > 0:
                b = normalize_vec(a[l:max(0, n - r)])
                if b.size < 10:
                    continue
                sc = objective_norm(b)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best = b.copy()

    return best


def coordinate_polish(a0, time_limit, rng_seed=991):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 10 or time_limit <= 0.0:
        return a
    best = a.copy()
    best_score = objective_norm(a)
    stall = 0
    rng = np.random.default_rng(rng_seed)

    while time.monotonic() - t0 < time_limit and stall < 200:
        q = self_conv(a)
        n_top = min(20, len(q))
        top = np.argsort(q)[::-1][:n_top]

        # Compute influence scores
        scores = np.zeros(n, dtype=np.float64)
        for kk in top[:10]:
            k = int(kk)
            i0 = max(0, k - n + 1)
            i1 = min(n, k + 1)
            if i0 < i1:
                start = k - (i1 - 1)
                stop = k - i0 + 1
                scores[i0:i1] += a[start:stop][::-1]

        src = np.argsort(scores)[::-1][:15]
        dst = np.argsort(scores)[:15]

        improved = False
        for s_idx in src:
            if time.monotonic() - t0 > time_limit or improved:
                break
            d = int(s_idx)
            if a[d] <= 1e-15:
                continue
            for j in dst:
                if improved:
                    break
                j = int(j)
                if d == j:
                    continue
                v = float(a[d])
                if v <= 1e-15:
                    continue
                for frac in (0.02, 0.05, 0.1, 0.2, 0.3):
                    if time.monotonic() - t0 > time_limit:
                        break
                    amt = v * frac
                    if amt <= 1e-18:
                        continue
                    b = a.copy()
                    b[d] -= amt
                    b[j] += amt
                    b = np.maximum(b, 0.0)
                    ss = float(b.sum())
                    if ss <= 1e-15:
                        continue
                    b = b / ss
                    sc = objective_norm(b)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best = b.copy()
                        a = b
                        improved = True
                        break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    b = a.copy()
                    amt = a[i] * 0.03
                    b[i] -= amt
                    b[j] += amt
                    b = np.maximum(b, 0.0)
                    ss = b.sum()
                    if ss > 1e-15:
                        b = b / ss
                        sc = objective_norm(b)
                        if sc < best_score - 1e-13:
                            best_score = sc
                            best = b.copy()
                            a = b
                            stall = 0

    return best


def update_best(best_seq, best_score, a, label, t0):
    if a is None:
        return best_seq, best_score
    an = normalize_vec(a)
    if an.size == 0:
        return best_seq, best_score
    sc = objective_norm(an)
    if np.isfinite(sc) and sc < best_score - 1e-12:
        best_score = sc
        best_seq = an.copy()
        try:
            print("*** NEW BEST [%s]: %.9f n=%d t=%.1fs"
                  % (label, sc, an.size, time.monotonic() - t0))
        except Exception:
            pass
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search driver."""
    t_global = time.monotonic()
    total_budget = min(1800.0, float(budget_s)) - 100.0
    total_budget = max(60.0, total_budget)

    best_seq = None
    best_score = float('inf')

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if p.ndim != 1:
            p = p.ravel()
        if p.size > 0 and p.sum() > 1e-12:
            parent = p
            if parent.size > 10000:
                parent = interpolate_sequence(parent, 8000)
    except Exception:
        parent = None

    if parent is not None:
        best_seq = normalize_vec(parent)
        best_score = objective_norm(best_seq)
        print("Parent: n=%d objective=%.9f" % (best_seq.size, best_score))
    else:
        n_default = 6500
        x = np.linspace(0.0, 1.0, n_default)
        best_seq = normalize_vec(np.sin(np.pi * x) ** 2)
        best_score = objective_norm(best_seq)
        print("Default start: n=%d objective=%.9f" % (best_seq.size, best_score))

    # === Phase 1: Parametric sweep (30s) ===
    t_phase1 = min(30.0, 0.02 * total_budget)
    if t_phase1 > 5.0:
        n0 = int(best_seq.size)
        starts = make_parametric_starts(n0, rng_seed=123)
        # Also try at different lengths
        for nn in (n0 - 500, n0 + 500, n0 + 1000):
            if 3000 <= nn <= 12000:
                extra = make_parametric_starts(nn, rng_seed=456 + nn)
                starts.extend(extra)

        cand_scores = []
        for s in starts:
            s = normalize_vec(s)
            sc = objective_norm(s)
            if np.isfinite(sc):
                cand_scores.append((sc, s))

        if cand_scores:
            cand_scores.sort(key=lambda x: x[0])
            top_start = cand_scores[0][1]
            top_score = cand_scores[0][0]
            if top_score < best_score - 1e-12:
                best_score = top_score
                best_seq = top_start.copy()
                print("Phase1 best start: score=%.9f n=%d" % (best_score, best_seq.size))

        kept = reduce_starts(cand_scores, 6)
        if best_seq is not None:
            matched = False
            for k in kept:
                if k.size == best_seq.size and float(np.dot(k, best_seq)) > 0.998:
                    matched = True
                    break
            if not matched:
                kept.insert(0, best_seq.copy())
    else:
        kept = [best_seq.copy()]

    # === Phase 2: Smooth polish on diverse starts (80s) ===
    remaining = total_budget - (time.monotonic() - t_global)
    t_phase2 = min(80.0, 0.06 * remaining)
    if t_phase2 > 15.0:
        n_starts = min(4, len(kept))
        for idx in range(n_starts):
            remaining = total_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_lim = min(60.0, t_phase2 / max(n_starts - idx, 1) * 1.3)
            if t_lim < 10:
                continue
            out = smooth_polish_nesterov(kept[idx], t_lim, rng_seed=200 + 23 * idx, verbose=False)
            best_seq, best_score = update_best(best_seq, best_score, out,
                                               "smooth%d" % idx, t_global)

    # === Phase 3: Main LP-FW (bulk of time) ===
    remaining = total_budget - (time.monotonic() - t_global)
    t_phase3 = min(1250.0, 0.72 * remaining)
    if t_phase3 > 60.0:
        best_seq, best_score = update_best(
            best_seq, best_score,
            lp_fw_trust(best_seq, t_phase3, verbose=True),
            "LP_main_n%d" % best_seq.size,
            t_global)

    # === Phase 4: Second diverse start with LP ===
    remaining = total_budget - (time.monotonic() - t_global)
    if len(kept) >= 2 and remaining > 300:
        cand2 = None
        for k in kept[1:]:
            if k.size == best_seq.size and float(np.dot(k, best_seq)) > 0.995:
                continue
            cand2 = k
            break
        if cand2 is not None:
            t2 = min(280.0, 0.30 * remaining)
            if t2 > 50.0:
                out = lp_fw_trust(cand2, t2, verbose=False)
                best_seq, best_score = update_best(best_seq, best_score, out,
                                                   "LP_second", t_global)

    # === Phase 5: Nesterov polish ===
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 120:
        t3 = min(200.0, 0.25 * remaining)
        if t3 > 40:
            out = smooth_polish_nesterov(best_seq, t3, rng_seed=777, verbose=True)
            best_seq, best_score = update_best(best_seq, best_score, out,
                                               "nest_polish", t_global)

    # === Phase 6: Transform + Trim ===
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 80:
        t4 = min(60.0, 0.20 * remaining)
        if t4 > 15:
            out = transform_search(best_seq, t4)
            best_seq, best_score = update_best(best_seq, best_score, out,
                                               "transform", t_global)

    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 80:
        t5 = min(70.0, 0.22 * remaining)
        if t5 > 15:
            out = trim_search(best_seq, t5)
            best_seq, best_score = update_best(best_seq, best_score, out,
                                               "trim", t_global)

    # === Phase 7: Length exploration ===
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 200:
        cur_n = int(best_seq.size)
        for dn in (-500, -1000, 500, 1000, 2000):
            remaining = total_budget - (time.monotonic() - t_global)
            if remaining < 100:
                break
            nn = cur_n + dn
            if 3000 <= nn <= 12000 and nn != cur_n:
                a_long = interpolate_sequence(best_seq, nn)
                tt = min(100.0, 0.20 * remaining)
                if tt < 30:
                    continue
                out = smooth_polish_nesterov(a_long, tt, rng_seed=900 + nn % 997, verbose=False)
                best_seq, best_score = update_best(best_seq, best_score, out,
                                                   "scale_n%d" % nn, t_global)

    # === Phase 8: Coordinate polish ===
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 70:
        tc = min(120.0, 0.35 * remaining)
        if tc > 20:
            out = coordinate_polish(best_seq, tc, rng_seed=314159)
            best_seq, best_score = update_best(best_seq, best_score, out,
                                               "coord", t_global)

    # === Phase 9: Final short LP round ===
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 60:
        tf = min(80.0, 0.50 * remaining)
        if tf > 20:
            out = lp_fw_trust(best_seq, tf, verbose=False)
            best_seq, best_score = update_best(best_seq, best_score, out,
                                               "LP_final", t_global)

    # === Final transforms ===
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 30:
        tf = min(30.0, 0.7 * remaining)
        if tf > 8:
            out = transform_search(best_seq, tf)
            best_seq, best_score = update_best(best_seq, best_score, out,
                                               "tfinal", t_global)

    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 30:
        tr = min(30.0, 0.7 * remaining)
        if tr > 8:
            out = trim_search(best_seq, tr)
            best_seq, best_score = update_best(best_seq, best_score, out,
                                               "trimfinal", t_global)

    # === Validate and return ===
    if best_seq is None:
        x = np.linspace(0.0, 1.0, 6500)
        best_seq = normalize_vec(np.sin(np.pi * x) ** 2)

    out_arr = np.asarray(best_seq, dtype=np.float64)
    out_arr = np.where(np.isfinite(out_arr), out_arr, 0.0)
    out_arr = np.maximum(out_arr, 0.0)
    out_arr = np.minimum(out_arr, 1000.0)
    s = float(out_arr.sum())
    if s <= 0.01:
        x = np.linspace(0.0, 1.0, 6500)
        out_arr = np.sin(np.pi * x) ** 2
        s = float(out_arr.sum())
    out_arr = out_arr / s

    final_score = objective_norm(out_arr)
    try:
        official = evaluate_safe(out_arr)
        if np.isfinite(official):
            final_score = min(final_score, official)
    except Exception:
        pass

    elapsed = time.monotonic() - t_global
    print("FINAL: score=%.10f n=%d time=%.1fs" % (final_score, out_arr.size, elapsed))
    return [float(v) for v in out_arr]
