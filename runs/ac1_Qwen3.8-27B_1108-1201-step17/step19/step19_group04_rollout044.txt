def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s

def self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]

    b = np.asarray(b, dtype=np.float64)
    n = a.size

    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]

def corr_shift(w, a):
    w = np.asarray(w, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = w.size
    size = next_pow2(L + n - 1)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fw * fa.conj(), size)[:n]
    return np.maximum(c, 0.0)

def objective_norm(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    q = self_conv(a)
    return 2.0 * float(a.size) * float(np.max(q))

def evaluate_safe(a):
    try:
        x = [float(v) for v in np.asarray(a, dtype=np.float64)]
        val = evaluate_sequence(x)
        if np.isfinite(val):
            return float(val)
    except Exception:
        pass
    return float('inf')

def weights_softmax(q, temperature):
    """Compute softmax weights from autocorrelation values."""
    q = np.asarray(q, dtype=np.float64)
    m = float(q.max())
    if m <= 1e-15 or temperature <= 0:
        idx = np.argmax(q)
        w = np.zeros_like(q, dtype=np.float64)
        w[idx] = 1.0
        return w
    scaled = (q - m) / temperature
    scaled = np.clip(scaled, -50.0, 0.0)
    e = np.exp(scaled)
    s = float(e.sum())
    if s <= 1e-300:
        idx = np.argmax(q)
        w = np.zeros_like(q, dtype=np.float64)
        w[idx] = 1.0
        return w
    return e / s

def quad_max(alpha, q0, c, qd):
    a1 = 1.0 - alpha
    val = (a1 * a1) * q0 + (2.0 * alpha * a1) * c + (alpha * alpha) * qd
    return float(np.max(val))

def golden_alpha_max(q0, c, qd, a, b, iters=40):
    phi = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = b - phi * (b - a)
    d1 = a + phi * (b - a)
    f1 = quad_max(c1, q0, c, qd)
    f2 = quad_max(d1, q0, c, qd)
    for _ in range(iters):
        if f1 < f2:
            b = d1
            d1 = c1
            f2 = f1
            c1 = b - phi * (b - a)
            f1 = quad_max(c1, q0, c, qd)
        else:
            a = c1
            c1 = d1
            f1 = f2
            d1 = a + phi * (b - a)
            f2 = quad_max(d1, q0, c, qd)
    return 0.5 * (a + b)

def line_search_direction(a, d, q0, T=None):
    a = np.asarray(a, dtype=np.float64)
    d = np.asarray(d, dtype=np.float64)
    q0 = np.asarray(q0, dtype=np.float64)
    M0 = float(np.max(q0))
    dot = float(np.dot(a, d))
    if dot > 1.0 - 1e-7:
        return a.copy(), M0
    c = conv_ab(a, d)
    qd = self_conv(d)
    best_alpha = 0.0
    best_M = M0
    alphas = [1e-4, 5e-4, 1e-3, 2e-3, 5e-3, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 0.9, 1.0]
    if T is not None and np.isfinite(T) and T < 0.0 and M0 > 1e-15:
        t1 = min(0.30, max(5e-4, 0.5 * (-float(T)) / M0))
        alphas.append(max(1e-5, 0.5 * t1))
        alphas.append(min(0.95, 1.5 * t1))
    uniq = []
    seen = set()
    for al in alphas:
        al = float(min(1.0, max(0.0, al)))
        key = round(al, 12)
        if key not in seen:
            seen.add(key)
            uniq.append(al)
    for al in uniq:
        val = quad_max(al, q0, c, qd)
        if val < best_M:
            best_M = val
            best_alpha = al
    if best_M < M0 - 1e-13 and 0.0 < best_alpha < 1.0:
        al2 = golden_alpha_max(q0, c, qd, 0.0, 1.0, 35)
        val2 = quad_max(al2, q0, c, qd)
        if val2 < best_M:
            best_M = val2
            best_alpha = al2
    if best_alpha <= 0.0 or best_M >= M0 - 1e-14:
        return a.copy(), M0
    new = (1.0 - best_alpha) * a + best_alpha * d
    new = np.maximum(new, 0.0)
    s = float(new.sum())
    if s <= 1e-15:
        return a.copy(), M0
    new = new / s
    return new, best_M

def get_lp_artifacts(n):
    art = _LP_CACHE.get(n)
    if art is not None:
        return art
    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0
    Aeq = np.zeros((1, n + 1), dtype=np.float64)
    Aeq[0, :n] = 1.0
    bounds = [(0.0, np.inf)] * n + [(None, None)]
    art = (c, Aeq, bounds)
    if len(_LP_CACHE) > 6:
        _LP_CACHE.clear()
    _LP_CACHE[n] = art
    return art

def solve_lp_offset(y, active, q_active):
    if not HAVE_LP:
        return None, None
    y = np.asarray(y, dtype=np.float64)
    active = np.asarray(active, dtype=int)
    q_active = np.asarray(q_active, dtype=np.float64)
    n = y.size
    K = active.size
    if K <= 0:
        return None, None
    if K * (n + 1) > 12_000_000:
        return None, None
    try:
        A = np.zeros((K, n + 1), dtype=np.float64)
    except Exception:
        return None, None
    yy = 2.0 * y
    for r, k in enumerate(active):
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            A[r, i0:i1] = yy[start:stop][::-1]
    A[:, n] = -1.0
    b = np.maximum(q_active, 0.0)
    c, Aeq, bounds = get_lp_artifacts(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq,
                      b_eq=np.array([1.0], dtype=np.float64),
                      bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception:
        return None, None
    status = getattr(res, 'status', None)
    success = getattr(res, 'success', False)
    x = getattr(res, 'x', None)
    if (status != 0 or not success) or x is None:
        return None, None
    g = np.asarray(x[:n], dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-10:
        return None, None
    g = g / s
    T = float(x[n])
    if not np.isfinite(T):
        T = 0.0
    return g, T

def smooth_mirror_descent(a0, time_limit, rng_seed=42, verbose=False):
    """
    Smooth mirror descent with geometric temperature cooling and momentum.
    """
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 1:
        return a
    
    best = a.copy()
    best_score = objective_norm(a)
    
    # Temperature schedule: geometric cooling
    t_start = 2.0
    t_end = 0.05
    cool_rate = 0.995  # multiply temperature by this each iteration
    
    trust = 0.3
    momentum = 0.0
    prev_move = None
    rng = np.random.default_rng(rng_seed)
    
    fails = 0
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        q = self_conv(a)
        M = float(np.max(q))
        score = 2.0 * n * M
        
        if score < best_score - 1e-13:
            best = a.copy()
            best_score = score
            fails = 0
            trust = min(trust * 1.03, 3.0)
        else:
            fails += 1
        
        if fails > 600:
            # Restart with noise
            a = best.copy()
            noise = 0.01 * rng.normal(size=n)
            a = a + noise * a
            a = np.maximum(a, 1e-12)
            a = normalize_vec(a)
            fails = 0
            trust = 0.2
        if fails > 50 and fails % 25 == 0:
            trust = max(trust * 0.85, 0.02)
        
        # Current temperature
        temp = max(t_start * (cool_rate ** iters), t_end)
        
        # Compute softmax weights
        w = weights_softmax(q, temp)
        
        # Compute gradient-like quantity (correlation)
        c = corr_shift(w, a)
        cmax = float(np.max(c))
        cmin = float(np.min(c))
        scale = cmax - cmin
        if scale <= 1e-15:
            break
        
        # Compute move with momentum
        raw_move = trust * (cmax - c) / scale
        raw_move = np.clip(raw_move, -60.0, 60.0)
        
        if prev_move is not None and momentum > 0:
            raw_move = (1.0 - momentum) * raw_move + momentum * prev_move
        prev_move = raw_move.copy()
        
        b = a * np.exp(raw_move)
        b = normalize_vec(b)
        
        Mb = float(np.max(self_conv(b)))
        score_b = 2.0 * n * Mb
        
        if score_b < best_score - 1e-14:
            best = b.copy()
            best_score = score_b
            a = b
            fails = 0
            trust = min(trust * 1.05, 4.0)
            momentum = min(momentum + 0.01, 0.4)
        elif score_b < score - 1e-15:
            a = b
            fails = max(0, fails - 2)
            trust = min(trust * 1.02, 3.0)
            momentum = min(momentum + 0.005, 0.4)
        else:
            a = b  # Still move (simulated annealing style)
            trust = max(trust * 0.92, 0.02)
            momentum = max(momentum - 0.01, 0.0)
        
        iters += 1
        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    SMooth: it=%d t=%.1fs score=%.10f temp=%.4f trust=%.4f fail=%d"
                  % (iters, elapsed, best_score, temp, trust, fails))
    
    return best

def fw_lp_search(a0, time_limit, verbose=True):
    """
    Frank-Wolfe with LP subproblem and stratified active set.
    """
    if time_limit <= 0.0:
        return normalize_vec(a0)
    
    t0 = time.monotonic()
    y = normalize_vec(a0)
    n = y.size
    if n <= 5:
        return y
    
    best = y.copy()
    best_score = objective_norm(y)
    
    # Active set size
    if n > 8000:
        initK = 200
        maxK = 600
    elif n > 5000:
        initK = 300
        maxK = 800
    else:
        initK = 400
        maxK = 1200
    
    K = initK
    use_lp = HAVE_LP
    lp_fails = 0
    no_improve = 0
    iter_no = 0
    last_print = t0
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining <= 0.0:
            break
        
        qy = self_conv(y)
        M_y = float(np.max(qy))
        score_y = 2.0 * n * M_y
        
        if score_y < best_score - 1e-13:
            best = y.copy()
            best_score = score_y
        
        if no_improve > 300 and (time.monotonic() - last_print) > 30.0:
            if time.monotonic() - t0 > 0.85 * time_limit:
                break
            y = best.copy()
            no_improve = 0
            K = initK
        
        # Stratified active set selection
        Kcur = int(min(K, len(qy), 1500))
        # Mix: top Kcur//2 by value + stratified sample of remaining
        order = np.argsort(qy)[::-1]
        top_k = order[:Kcur // 2]
        
        # Stratified: divide lags into bins, take max in each bin
        num_bins = Kcur // 2
        if num_bins > 0:
            bin_edges = np.linspace(0, len(qy), num_bins + 1).astype(int)
            stratified = []
            for bi in range(num_bins):
                lo = bin_edges[bi]
                hi = bin_edges[bi + 1]
                if hi > lo:
                    bi_idx = lo + np.argmax(qy[lo:hi])
                    stratified.append(bi_idx)
            act = np.unique(np.concatenate([top_k, np.array(stratified, dtype=int)]))
        else:
            act = top_k
        
        Kcur = len(act)
        q_active = qy[act]
        
        g_dir = None
        T = None
        
        if use_lp and Kcur >= 20:
            tstep = time.monotonic()
            g_dir, T = solve_lp_offset(y, act, q_active)
            spend = time.monotonic() - tstep
            
            if g_dir is None:
                lp_fails += 1
                if lp_fails >= 4:
                    use_lp = False
            else:
                lp_fails = 0
                if spend > max(60.0, 0.1 * remaining):
                    use_lp = False
                    K = max(100, K - 50)
        
        if g_dir is not None:
            new, Mnew = line_search_direction(y, g_dir, qy, T)
            if Mnew < M_y - 1e-14:
                y = new
                no_improve = 0
                K = min(K + 20, maxK)
            else:
                no_improve += 1
                K = min(K + 30, maxK)
        else:
            # Fallback: smooth direction
            w = weights_softmax(qy, 1.0)
            c = corr_shift(w, y)
            cmax = float(np.max(c))
            cmin = float(np.min(c))
            scale = cmax - cmin
            if scale > 1e-15:
                move = 0.5 * (cmax - c) / scale
                move = np.clip(move, -50.0, 50.0)
                d = y * np.exp(move)
                d = normalize_vec(d)
                new, Mnew = line_search_direction(y, d, qy, None)
                if Mnew < M_y - 1e-14:
                    y = new
                    no_improve = 0
                else:
                    no_improve += 1
                    K = min(K + 30, maxK)
            else:
                no_improve += 1
        
        if no_improve > 200:
            break
        
        iter_no += 1
        now = time.monotonic()
        if verbose and (iter_no % 50 == 0 or (now - last_print) > 30.0):
            print("    FW-LP: it=%d t=%.1fs best=%.10f cur=%.10f K=%d lp=%s noimp=%d"
                  % (iter_no, now - t0, best_score, score_y, K, use_lp, no_improve))
            last_print = now
    
    return best

def coordinate_polish_extreme(a0, time_limit, rng_seed=777, verbose=False):
    """
    Aggressive coordinate descent: try many pairs, small steps, multiple rounds.
    """
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 10 or time_limit <= 0.0:
        return a
    
    best = a.copy()
    best_score = objective_norm(a)
    stall = 0
    rng = np.random.default_rng(rng_seed)
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        q = self_conv(a)
        M_cur = float(np.max(q))
        
        # Find top contributing lags
        top_lags = np.argsort(q)[::-1][:min(20, len(q))]
        
        # Score each coordinate by its contribution to peak lags
        scores = np.zeros(n, dtype=np.float64)
        for kk in top_lags[:10]:
            k = int(kk)
            i0 = max(0, k - n + 1)
            i1 = min(n, k + 1)
            if i0 < i1:
                start = k - (i1 - 1)
                stop = k - i0 + 1
                scores[i0:i1] += a[start:stop][::-1]
        
        # Source: high-score coordinates (contributing to peaks)
        # Destination: low-score coordinates (not contributing)
        src = np.argsort(scores)[::-1][:25]
        dst = np.argsort(scores)[:25]
        
        improved = False
        for s_idx in range(len(src)):
            if time.monotonic() - t0 > time_limit:
                break
            if improved:
                break
            d = int(src[s_idx])
            if a[d] <= 1e-15:
                continue
            for j_idx in range(len(dst)):
                j = int(dst[j_idx])
                if d == j:
                    continue
                v = float(a[d])
                if v <= 1e-15:
                    continue
                for frac in (0.01, 0.03, 0.05, 0.1, 0.15):
                    if time.monotonic() - t0 > time_limit:
                        break
                    amt = v * frac
                    if amt <= 1e-18:
                        continue
                    b = a.copy()
                    b[d] -= amt
                    b[j] += amt
                    b = np.maximum(b, 0.0)
                    ss = float(b.sum())
                    if ss <= 1e-15:
                        continue
                    b = b / ss
                    sc = objective_norm(b)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best = b.copy()
                        a = b
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall % 50 == 0:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    b = a.copy()
                    amt = a[i] * 0.03
                    b[i] -= amt
                    b[j] += amt
                    b = np.maximum(b, 0.0)
                    ss = float(b.sum())
                    if ss > 1e-15:
                        b = b / ss
                        sc = objective_norm(b)
                        if sc < best_score - 1e-14:
                            best_score = sc
                            best = b.copy()
                            a = b
                            stall = 0
    
    if verbose:
        print("    CoordPolish: final score=%.10f stall=%d" % (best_score, stall))
    return best

def interpolate_sequence(f, new_n):
    f = normalize_vec(f)
    old_n = f.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return f.copy()
    if old_n <= 1:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f)
    return normalize_vec(g)

def make_diverse_starts(n, parent):
    out = []
    if parent is None:
        p = np.ones(n, dtype=np.float64) / float(n)
    else:
        p = normalize_vec(parent)
        if p.size != n:
            p = interpolate_sequence(p, n)
    
    out.append(p)
    
    x = np.arange(n, dtype=np.float64) / float(max(n - 1, 1))
    
    # Various parametric shapes
    for alph in (0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        out.append(normalize_vec((x + 1e-8) ** alph))
        out.append(normalize_vec(((1.0 - x) + 1e-8) ** alph))
    
    for a_e, b_e in [(0.5, 0.5), (1.0, 1.0), (2.0, 2.0), (0.5, 2.0), (2.0, 0.5)]:
        out.append(normalize_vec(((x + 1e-8) ** a_e) * (((1.0 - x) + 1e-8) ** b_e)))
    
    for r in (0.5, 1.0, 2.0, 3.0, 5.0):
        out.append(normalize_vec(np.exp(-r * x) + np.exp(-r * (1.0 - x))))
    
    for ctr in (0.2, 0.3, 0.5, 0.7, 0.8):
        for wd in (0.05, 0.1, 0.15, 0.2, 0.3):
            seq = np.exp(-((x - ctr) ** 2) / (2.0 * wd * wd))
            out.append(normalize_vec(seq))
    
    # Sine-based
    for f_mult in (1, 2, 3):
        out.append(normalize_vec(np.sin(np.pi * f_mult * x) ** 2))
    
    # Blend with parent
    for w in (0.2, 0.4, 0.6, 0.8):
        out.append(normalize_vec((1.0 - w) * p + w * np.ones(n) / n))
    
    # Random smooth
    rng = np.random.default_rng(12345)
    for _ in range(15):
        seq = np.zeros(n, dtype=np.float64)
        for _ in range(int(rng.integers(1, 5))):
            ctr = rng.uniform(0.05, 0.95)
            wd = rng.uniform(0.03, 0.4)
            ht = rng.uniform(0.2, 1.0)
            seq += ht * np.exp(-((x - ctr) ** 2) / (2.0 * wd * wd))
        seq += 1e-8
        out.append(normalize_vec(seq))
    
    return out

def reduce_starts(cand_scores, max_kept=6):
    remaining = list(cand_scores)
    kept = []
    for _ in range(max_kept):
        bi = -1
        bs = float('inf')
        for i, item in enumerate(remaining):
            sc = float(item[0])
            if np.isfinite(sc) and sc < bs:
                bs = sc
                bi = i
        if bi < 0:
            break
        seq = remaining[bi][1]
        ok = True
        for kseq in kept:
            if seq.size == kseq.size:
                dp = float(np.dot(seq, kseq))
                if dp > 0.99:
                    ok = False
                    break
        if ok:
            kept.append(seq.copy())
        remaining.pop(bi)
    return kept

def try_length_change(best_seq, best_score, target_n, time_budget, t_global):
    """Try interpolating to a different length and polishing."""
    a = interpolate_sequence(best_seq, target_n)
    t0 = time.monotonic()
    out = smooth_mirror_descent(a, min(time_budget, 60.0), rng_seed=target_n % 10000, verbose=False)
    sc = objective_norm(out)
    if np.isfinite(sc) and sc < best_score - 1e-12:
        try:
            print("  Length change to %d: score=%.10f" % (target_n, sc))
        except Exception:
            pass
        best_seq = out
        best_score = sc
    return best_seq, best_score
