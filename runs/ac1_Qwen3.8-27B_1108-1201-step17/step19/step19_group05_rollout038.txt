```python
"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated mirror descent
with soft-max subgradient, spectral flattening, and coordinate descent polishing.
Key improvement over previous: O(n log n) gradient via cross-correlation FFT,
enabling use of all lags (not just top peaks) for a much better gradient signal.
"""

import numpy as np
import time


def fast_conv(a):
    """Full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_gradient_fft(a, C, tau):
    """
    Compute soft-max subgradient of max(C) using FFT-based cross-correlation.
    grad[i] = 2 * sum_k w[k] * a[k-i] (valid range) = 2 * (w cross-corr a)[i]
    """
    n = len(a)
    max_C = C.max()
    if max_C < 1e-30:
        return np.zeros(n)

    # Soft-max weights (numerically stable)
    log_w = tau * C / max_C
    log_w -= log_w.max()
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum

    # Cross-correlation via FFT: (w ⋆ a)[i] = conv(w, a_flip)[i + n - 1]
    a_flip = a[::-1].copy()
    corr_size = (2 * n - 1) + (n - 1)  # 3n - 2
    fft_size = 1 << (corr_size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    AF = np.fft.rfft(a_flip, fft_size)
    conv_result = np.fft.irfft(W * AF, fft_size)
    grad = 2.0 * conv_result[n - 1:2 * n - 2]
    return grad


def make_initialization(n, mode, seed=42):
    """Create various initializations."""
    rng = np.random.default_rng(seed)
    if mode == "constant":
        a = np.ones(n)
    elif mode == "ramp1":
        a = np.arange(1, n + 1, dtype=np.float64)
    elif mode == "ramp05":
        a = np.arange(1, n + 1, dtype=np.float64) ** 0.5
    elif mode == "ramp15":
        a = np.arange(1, n + 1, dtype=np.float64) ** 1.5
    elif mode == "chirp":
        x = np.arange(n, dtype=np.float64)
        a = 1.0 + np.cos(2.0 * np.pi * x * x / n)
    elif mode == "chirp2":
        x = np.arange(n, dtype=np.float64)
        a = np.cos(np.pi * x * x / n) ** 2 + 1.0
    elif mode == "gaussian_win":
        x = np.linspace(-3, 3, n)
        a = np.exp(-0.5 * x ** 2)
    elif mode == "random_exp":
        a = rng.exponential(1.0, n)
    elif mode == "random_cos":
        x = np.arange(n, dtype=np.float64)
        phases = rng.uniform(0, 2 * np.pi, 5)
        freqs = rng.choice(np.arange(1, n // 2), size=5, replace=False)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += np.cos(2.0 * np.pi * f * x / n + p)
        a = np.abs(a) + 0.1
    elif mode == "tapered":
        x = np.arange(n, dtype=np.float64)
        a = np.abs(np.sin(np.pi * x / n)) ** 0.5
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def mirror_descent_fft(a0, time_limit, tau0=5.0, tau1=50.0, lr0=1.0,
                       seed=42, verbose=False, momentum=True,
                       restart_every=5000, restart_count_max=20):
    """Mirror descent with FFT-based soft-max gradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    n_restarts = 0

    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1

        # Adaptive tau
        tau = tau0 + (tau1 - tau0) * min(1.0, t / 5000.0)

        # Nesterov momentum
        if momentum and t > 1:
            beta = (t - 1.0) / (t + 2.0)
            y = x + beta * (x - x_prev)
            y = np.maximum(y, 1e-20)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            else:
                y = x.copy()
        else:
            y = x.copy()

        # Compute convolution and gradient
        C = fast_conv(y)
        g = compute_gradient_fft(y, C, tau)

        # Step size
        eta = lr0 / (np.sqrt(t) + 1.0)

        # Multiplicative (mirror) update
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Track best
        M_new = C.max()
        M_x = fast_conv(x).max()
        M_eff = min(M_new, M_x)
        if M_eff < best_m - 1e-14:
            best_m = M_eff
            best_a = (x if M_x <= M_new else y).copy()
            stall = 0
        else:
            stall += 1

        # Restart
        if stall > restart_every and n_restarts < restart_count_max:
            stall = 0
            n_restarts += 1
            strength = min(0.15, 0.03 + 0.02 * min(n_restarts, 5))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-15)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                print(f"      MD restart {n_restarts} t={t}: J={2*n*best_m:.6f}")

        # Time batch check (process multiple steps per time check for speed)
        if t % 200 == 0 and time.monotonic() - t0 > time_limit:
            break

        if verbose and t % 2000 == 0:
            print(f"      MD t={t} ({time.monotonic()-t0:.0f}s): "
                  f"J={2*n*best_m:.6f}, tau={tau:.1f}, stall={stall}")

    # Final check
    M_final = fast_conv(x).max()
    if M_final < best_m:
        best_m = M_final
        best_a = x.copy()

    return best_a, best_m


def spectral_flatten_fft(a0, time_limit, alpha_start=0.2, alpha_end=0.8,
                         seed=42, verbose=False):
    """Gerchberg-Saxton spectral flattening with adaptive alpha."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_mean = P.mean()
        if P_max < 1e-30:
            break

        # Adaptive alpha
        frac = min(1.0, it / 2000.0)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # Flatten spectrum
        P_new = P + alpha * (P_mean - P)
        P_new = np.maximum(P_new, 1e-30)

        # Update magnitude
        mag_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        mag_ratio = np.clip(mag_ratio, 0.3, 3.0)
        A_new = A * mag_ratio
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Soft blend
            blend = 0.15
            a = (1 - blend) * a + blend * a_new
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a
            M_curr = fast_conv(a).max()
            if M_curr < best_m:
                best_m = M_curr
                best_a = a.copy()

        if verbose and it % 500 == 0:
            print(f"      Spec it={it} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m


def coordinate_descent_fast(a0, time_limit, seed=42, verbose=False):
    """Coordinate descent: move mass from high-contribution positions to low-cost ones."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 50:
        C = fast_conv(a)
        k_max = int(np.argmax(C))

        # Find the pair (i, j) with i+j = k_max that has max product
        j_lo = max(0, k_max - n + 1)
        j_hi = min(n - 1, k_max)
        if j_lo > j_hi:
            stall += 1
            continue
        idx = np.arange(j_lo, j_hi + 1)
        contrib = a[idx] * a[k_max - idx]
        src = idx[np.argmax(contrib)]

        if a[src] < 1e-14:
            stall += 1
            continue

        # Find low-cost destination: minimize impact on top peaks
        n_top = min(8, n)
        top_ks = np.argpartition(C, -n_top)[-n_top:]
        costs = np.zeros(n)
        for k2 in top_ks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            if lo2 <= hi2:
                valid = np.arange(lo2, hi2 + 1)
                costs[valid] += a[k2i - valid]

        # Destinations sorted by cost (prefer low cost and low value)
        score = costs * 0.05 + a
        sorted_idx = np.argsort(score)

        moved = False
        for frac in [0.1, 0.05, 0.2, 0.3, 0.5]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            for dst in sorted_idx[:5]:
                if dst == src:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_t = fast_conv(a_trial).max()
                if M_t < best_m - 1e-14:
                    a = a_trial
                    best_m = M_t
                    best_a = a.copy()
                    moved = True
                    stall = 0
                    break
            if moved:
                break

        if not moved:
            stall += 1
            # Random perturbation
            n_pert = min(3, n)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.02, 0.08)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

    return best_a, best_m


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss


def safe_eval(a):
    """Safely compute the objective value."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """
    Main search function. Uses FFT-accelerated mirror descent with soft-max
    gradient, spectral flattening, and coordinate descent to minimize J.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = safe_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = safe_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Generate initial candidates with diverse structures
    candidates = []

    # Parent (if available)
    if parent is not None:
        p_norm = parent / parent.sum()
        candidates.append((p_norm, "parent"))

    # Various initializations at different lengths
    for n_try in [4000, 5000, 6000]:
        for mode in ["chirp", "ramp1", "chirp2", "gaussian_win", "tapered"]:
            a = make_initialization(n_try, mode, seed=42 + n_try + hash(mode) % 100)
            candidates.append((a, f"{mode}_n{n_try}"))

    # Random cos-based
    for n_try in [5000, 6000]:
        a = make_initialization(n_try, "random_cos", seed=123)
        candidates.append((a, f"randcos_n{n_try}"))

    # Evaluate all initial candidates
    print(f"Initial candidates: {len(candidates)}")
    scored = []
    for a, label in candidates:
        sc = safe_eval(a)
        if np.isfinite(sc):
            scored.append((sc, a, label))
            print(f"  Init [{label}]: {sc:.6f}")
        else:
            sc = safe_eval(a)
            scored.append((sc, a, label))

    scored.sort(key=lambda x: x[0])
    print(f"Top candidate: {scored[0][2]} with score {scored[0][0]:.6f}")

    # Phase 1: Mirror descent on top candidates
    print(f"\n=== Phase 1: Mirror Descent (FFT gradient) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_top = min(8, len(scored))
    # Allocate more time to better candidates
    for i in range(n_top):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget:
            break
        # Time allocation: give more to top candidates
        if i < 3:
            t_avail = min(remaining * 0.15, 200.0)
        elif i < 5:
            t_avail = min(remaining * 0.10, 120.0)
        else:
            t_avail = min(remaining * 0.06, 60.0)

        remaining = time_budget - elapsed
        if remaining < 15:
            break
        t_avail = min(t_avail, remaining)

        sc0, a0, label = scored[i]
        print(f"  MD [{label}] (t={t_avail:.0f}s, init={sc0:.6f})...")
        a_out, _ = mirror_descent_fft(a0, t_avail, tau0=5.0, tau1=80.0,
                                       lr0=1.5, seed=42 + i, verbose=False,
                                       restart_every=4000, restart_count_max=15)
        try_update(a_out, f"md_{label}")

    # Phase 2: Spectral flattening on current best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Spectral Flattening ===")
    if remaining > 60 and best_seq is not None:
        t2 = min(remaining * 0.12, 150.0)
        a_in = best_seq / best_seq.sum()
        a_out, _ = spectral_flatten_fft(a_in, t2, verbose=False)
        try_update(a_out, "spectral")

        # MD after spectral
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t2b = min(remaining * 0.10, 120.0)
            a_in2 = a_out / max(a_out.sum(), 1e-15)
            a_out2, _ = mirror_descent_fft(a_in2, t2b, tau0=10.0, tau1=100.0,
                                            lr0=1.0, seed=99, verbose=False,
                                            restart_every=5000, restart_count_max=10)
            try_update(a_out2, "md_post_spec")

    # Phase 3: Aggressive MD with different params
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Aggressive Mirror Descent ===")
    if remaining > 60 and best_seq is not None:
        t3 = min(remaining * 0.18, 250.0)
        a_in = best_seq / best_seq.sum()
        a_out, _ = mirror_descent_fft(a_in, t3, tau0=20.0, tau1=200.0,
                                       lr0=2.5, seed=7, verbose=True,
                                       restart_every=3000, restart_count_max=20)
        try_update(a_out, "md_agg")

    # Phase 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coordinate Descent ===")
    if remaining > 30 and best_seq is not None:
        t4 = min(remaining * 0.10, 120.0)
        a_in = best_seq / best_seq.sum()
        a_out, _ = coordinate_descent_fast(a_in, t4, seed=42, verbose=False)
        try_update(a_out, "coord")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t4b = min(remaining * 0.08, 80.0)
            a_out, _ = coordinate_descent_fast(a_in, t4b, seed=777, verbose=False)
            try_update(a_out, "coord2")

    # Phase 5: Multi-length interpolation
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length ===")
    if best_seq is not None and remaining > 60:
        n_best = len(best_seq)
        for new_n in [n_best + 1000, n_best + 2000, max(3000, n_best - 1000),
                      n_best + 3000]:
            if new_n <= 0 or new_n > 12000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            a_interp = interpolate_to_length(best_seq, new_n)
            try_update(a_interp, f"interp_n{new_n}")

            t5 = min(remaining * 0.20, 100.0)
            if t5 > 15:
                a_out, _ = mirror_descent_fft(a_interp, t5, tau0=5.0, tau1=60.0,
                                               lr0=1.5, seed=100 + new_n % 100,
                                               verbose=False, restart_every=5000,
                                               restart_count_max=10)
                try_update(a_out, f"md_n{new_n}")

    # Phase 6: Final combined polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final Polish ===")
    if remaining > 30 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        # Spectral + MD combination
        t6a = min(remaining * 0.25, 80.0)
        t6b = min(remaining * 0.35, 120.0)
        if t6a > 10:
            a_out, _ = spectral_flatten_fft(a_in, t6a, verbose=False)
            try_update(a_out, "spec_final")
            a_in = a_out / max(a_out.sum(), 1e-15)
        remaining = time_budget - (time.monotonic() - t_global)
        if t6b > 15 and remaining > 20:
            t6b = min(t6b, remaining - 10)
            a_out, _ = mirror_descent_fft(a_in, t6b, tau0=30.0, tau1=300.0,
                                           lr0=2.0, seed=555, verbose=True,
                                           restart_every=4000, restart_count_max=15)
            try_update(a_out, "md_final")

    # Phase 7: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15 and best_seq is not None:
        t7 = min(remaining * 0.30, 60.0)
        a_in = best_seq / best_seq.sum()
        a_out, _ = coordinate_descent_fast(a_in, t7, seed=12345, verbose=False)
        try_update(a_out, "coord_final")

    # Final validation
    if best_seq is not None:
        fs = safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        a = make_initialization(5000, "chirp", seed=0)
        best_seq = a
        best_score = safe_eval(a)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
```
