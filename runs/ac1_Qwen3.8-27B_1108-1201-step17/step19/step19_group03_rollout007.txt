def solve_lp_fw_direction(a, K, M, solver):
    n = len(a)
    ac = fast_autocorr(a)
    if K >= 2 * n - 1:
        binding_idx = np.arange(2 * n - 1)
    else:
        binding_idx = np.argsort(ac)[::-1][:K]

    K_sel = len(binding_idx)
    T_sel = np.zeros((K_sel, n))
    for ii, k in enumerate(binding_idx):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        for j in range(j_start, j_end):
            T_sel[ii, j] = a[ki - j]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_sel @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

def compute_soft_subgradient(a, ac, top_k=60, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

            best_dst = -1
            for dst in low_cand[:5]:
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] = max(0, a_trial[src] - amt)
                    a_trial[dst] += amt
                    a_trial = a_trial / max(a_trial.sum(), 1e-15)
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_trial_m:
                        best_trial_m = M_trial
                        best_frac = amt
                        best_dst = dst

            if best_dst >= 0:
                a[src] = max(0, a[src] - best_frac)
                a[best_dst] += best_frac
                a = a / a.sum()
                best_m = best_trial_m
                best_a = a.copy()
                improved = True

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] = max(0, a[i] - amt)
                a[j] += amt
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_good_init(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    # Add some spectral shaping
    for freq in range(2, 15):
        coeff = rng.normal(0, 0.02 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        best_seq = make_good_init(n_base, seed=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Quick POCS (60s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 15:
        print(f"\n=== Phase 1: POCS (n={n0}, t={t1:.0f}s) ===")
        a_out, _ = pocs_optimize(best_seq, t1, rng_seed=7)
        try_update(a_out, "pocs")

    # === Phase 2: Main Nesterov LP-FW (500s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 530.0)
    if t2 > 50:
        n_best = len(best_seq)
        print(f"\n=== Phase 2: NLP-FW (n={n_best}, t={t2:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        K_init = min(100, 2 * n_best - 1)
        K_max = min(3000, 2 * n_best - 1)
        a_out, _ = nesterov_lp_fw(a_in, t2, solver, K_init=K_init, K_max=K_max,
                                    momentum=0.5, verbose=True, rng_seed=0)
        try_update(a_out, "nlpfw_main")

    # === Phase 3: AMD (200s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.14, 200.0)
    if t3 > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 3: AMD (n={n_best}, t={t3:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t3, top_k=80, beta=6.0, c0=5.0,
                                 restart_every=3000, verbose=False, rng_seed=42)
        try_update(a_out, "amd")

    # === Phase 4: Second NLP-FW with different params (300s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 50:
        n_best = len(best_seq)
        print(f"\n=== Phase 4: NLP-FW r2 (n={n_best}, t={t4:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        K_init = min(200, 2 * n_best - 1)
        K_max = min(5000, 2 * n_best - 1)
        a_out, _ = nesterov_lp_fw(a_in, t4, solver, K_init=K_init, K_max=K_max,
                                    momentum=0.3, verbose=True, rng_seed=100)
        try_update(a_out, "nlpfw_r2")

    # === Phase 5: Multi-n exploration (150s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-n (t={remaining:.0f}s available) ===")
    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 1500, -1500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 8000 and new_n != n_best:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))[:3]

    for new_n in trial_lengths:

        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        t5a = min(remaining * 0.4, 100.0)
        if t5a > 20:
            K_init = min(100, 2 * new_n - 1)
            K_max = min(2000, 2 * new_n - 1)
            a_out, _ = nesterov_lp_fw(a_long, t5a, solver, K_init=K_init, K_max=K_max,
                                        momentum=0.5, verbose=False, rng_seed=new_n % 500)
            try_update(a_out, f"nlpfw_n{new_n}")

    # === Phase 6: Final NLP-FW on best (200s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.20, 200.0)
    if t6 > 40:
        print(f"\n=== Phase 6: NLP-FW final (n={n_best}, t={t6:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        K_init = min(300, 2 * n_best - 1)

                                    momentum=0.4, verbose=True, rng_seed=999)
        try_update(a_out, "nlpfw_final")

    # === Phase 7: Coordinate polish (120s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.15, 120.0)
    if t7 > 20:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Coord polish (n={n_best}, t={t7:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t7, verbose=False, rng_seed=789)
        try_update(a_out, "coord")

    # === Phase 8: AMD final (80s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.20, 80.0)
    if t8 > 20:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: AMD final (n={n_best}, t={t8:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t8, top_k=100, beta=8.0, c0=6.0,
                                 restart_every=2000, verbose=False, rng_seed=555)
        try_update(a_out, "amd_final")

    # === Phase 9: Final NLP-FW squeeze (remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        t9 = remaining - 10
        print(f"\n=== Phase 9: Final squeeze (n={n_best}, t={t9:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        K_init = min(500, 2 * n_best - 1)
        K_max = min(2 * n_best - 1, 8000)
        a_out, _ = nesterov_lp_fw(a_in, t9, solver, K_init=K_init, K_max=K_max,
                                    momentum=0.3, verbose=True, rng_seed=2024)
        try_update(a_out, "final_squeeze")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        best_seq = make_good_init(n_fb, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
