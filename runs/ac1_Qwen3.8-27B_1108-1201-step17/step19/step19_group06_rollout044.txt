import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum(a, target=1.0):
    """Clip to non-negative and scale to target sum."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, target / len(a))
    return a * (target / s)


def score_from_ac(a):

    n = len(a)
    ac = fast_autocorr(a)

    return float(2.0 * n * ac.max() / (S * S))

def compute_topk_grad(a, ac, top_k, beta):
    """Compute weighted gradient over top autocorrelation lags."""
    n = len(a)
    k_alarm = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_alarm]
    top_vals = ac[top_idx]
    max_v = top_vals.max()
    expo = beta * (top_vals - max_v)
    expo = np.clip(expo, -50.0, 50.0)
    weights = np.exp(expo)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_alarm) / k_alarm
    else:
        weights = weights / w_sum
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n, k_int + 1)
        if i_lo >= i_hi:
            continue
        idx = np.arange(i_lo, i_hi)
        j_idx = k_int - idx
        valid = (j_idx >= 0) & (j_idx < n)
        grad[idx[valid]] += 2.0 * w * a[j_idx[valid]]
    return grad

        improved = False

        for k_pos in top_lags[:12]:
            if time.monotonic() - t0 > time_limit * 0.95:
                break
            k_int = int(k_pos)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo >= i_hi:
                continue
            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            n_pairs = min(8, len(idx))
            pair_order = np.argsort(contrib)[::-1][:n_pairs]

            for pi in pair_order:
                if time.monotonic() - t0 > time_limit * 0.95:
                    break
                src = int(idx[pi])
                src_val = a[src]
                if src_val < 1e-14:
                    continue

                # Find best destination using impact score
                scores = np.zeros(n)
                for k2 in top_lags[:6]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n, k2i + 1)
                    if s2 < e2:
                        ii = np.arange(s2, e2)
                        jj = k2i - ii
                        scores[ii] += a[jj] * ac[k2i]
                        scores[jj] += a[ii] * ac[k2i]
                scores[src] = np.inf
                dst = int(np.argmin(scores))

                if dst == src:
                    continue

                for frac in [0.02, 0.05, 0.1, 0.15, 0.2]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            if stall > 500:
                n_moves = int(rng.integers(5, 15))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = float(rng.uniform(0.005, 0.02))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    m_new = fast_autocorr(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                        stall = 0

    return best_a, best_m


def interpolate_seq(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def propose_candidate(budget_s=1800.0):
    """
    Multi-phase optimization: baseline eval -> mirror descent -> coord descent
    -> length exploration -> final polish. Always returns a valid sequence.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            p_score = evaluate_safe(p_norm)
            if np.isfinite(p_score):
                best_seq = p_norm.copy()
                best_score = p_score
                print(f"Parent loaded: n={len(p_norm)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Try uniform baseline
    if best_seq is not None:
        n_uni = len(best_seq)
    else:
        n_uni = 5000

    a_uni = np.ones(n_uni) / n_uni
    sc_uni = evaluate_safe(a_uni)
    if np.isfinite(sc_uni) and sc_uni < best_score:
        best_score = sc_uni
        best_seq = a_uni.copy()
        print(f"Uniform n={n_uni}: score={sc_uni:.6f}")

    if best_seq is None:
        best_seq = np.ones(5000) / 5000
        best_score = 2.0
        print("Fallback to uniform.")

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = evaluate_safe(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")

    n0 = len(best_seq)

    # Phase 1: Mirror descent
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        t1 = min(remaining * 0.35, 500.0)
        print(f"\n=== Phase 1: Mirror descent (n={n0}, {t1:.0f}s) ===")
        a_out, _ = mirror_descent_v2(best_seq, t1, lr_init=0.05, beta_init=2.0,
                                     top_k_init=30, restart_every=2500, rng_seed=42)
        check_update(a_out, f"md_n{n0}")

    # Phase 2: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        t2 = min(remaining * 0.25, 350.0)
        print(f"\n=== Phase 2: Coord descent (n={n0}, {t2:.0f}s) ===")
        a_out, _ = coord_descent_v2(best_seq, t2, rng_seed=123)
        check_update(a_out, f"coord_n{n0}")

    # Phase 3: Second mirror descent (finer params)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        t3 = min(remaining * 0.20, 250.0)
        print(f"\n=== Phase 3: Fine MD (n={n0}, {t3:.0f}s) ===")
        a_out, _ = mirror_descent_v2(best_seq, t3, lr_init=0.02, beta_init=4.0,
                                     top_k_init=60, restart_every=2000, rng_seed=99)
        check_update(a_out, f"mdfine_n{n0}")

    # Phase 4: Length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 80:
        print(f"\n=== Phase 4: Length exploration (budget={remaining:.0f}s) ===")
        length_tries = []
        for delta in [-2000, -1000, -500, 500, 1000, 2000, 3000]:
            nn = n_best + delta
            if 1000 <= nn <= 10000 and nn != n_best:
                length_tries.append(nn)
        for rn in [2000, 3000, 4000, 5000, 6000, 7000, 8000]:
            if rn != n_best and 1000 <= rn <= 10000:
                length_tries.append(rn)
        length_tries = sorted(set(length_tries))

        for new_n in length_tries[:4]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            print(f"  Trying n={new_n} (rem={remaining:.0f}s)")
            a_interp = interpolate_seq(best_seq, new_n)
            t5a = min(remaining * 0.4, 100.0)
            if t5a > 15:
                a_out, _ = mirror_descent_v2(a_interp, t5a, lr_init=0.04,
                                             beta_init=3.0, top_k_init=40,
                                             restart_every=2000, rng_seed=new_n)
                check_update(a_out, f"md_n{new_n}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40:
                t5b = min(remaining * 0.3, 80.0)
                if t5b > 10:
                    a_out, _ = coord_descent_v2(a_interp, t5b, rng_seed=new_n % 1000)
                    check_update(a_out, f"coord_n{new_n}")

    # Phase 5: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t6 = min(remaining * 0.5, 120.0)
        print(f"\n=== Phase 5: Final polish (n={n_best}, {t6:.0f}s) ===")
        a_out, _ = mirror_descent_v2(best_seq, t6, lr_init=0.01, beta_init=5.0,
                                     top_k_init=80, restart_every=1500, rng_seed=200)
        check_update(a_out, f"final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        best_seq = np.ones(5000) / 5000
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
