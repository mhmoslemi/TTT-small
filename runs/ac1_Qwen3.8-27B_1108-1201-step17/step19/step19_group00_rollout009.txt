```python
def mwu_optimize(a0, time_limit, top_k=50, eta_base=1.0, verbose=True):
    n = len(a0)
    w = np.ones(n)  # log-weights
    a = a0 / a0.sum()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
        
        # Get top-k lags
        k_lags = np.argsort(ac)[::-1][:top_k]
        top_vals = ac[k_lags]
        weights = np.exp(50 * (top_vals - top_vals[0]))  # softmax
        weights /= weights.sum()
        
        # Compute gradient (subgradient of weighted max)
        g = np.zeros(n)
        for w_k, k in zip(weights, k_lags):
            ki = int(k)
            for i in range(max(0, ki-n+1), min(n, ki+1)):
                g[i] += 2.0 * w_k * a[ki - i]
        
        # MWU update
        eta = eta_base / np.sqrt(t + 1)
        g_mean = g.mean()
        log_w = np.log(a) - eta * (g - g_mean)
        log_w -= log_w.max()
        w_new = np.exp(log_w)
        w_new = np.maximum(w_new, 1e-30)
        a_new = w_new / w_new.sum()
        
        # Line search
        best_t = 0.5
        best_trial = fast_autocorr(a_new).max()
        for t_candidate in [0.1, 0.3, 0.5, 0.7, 0.9]:
            trial = np.maximum((1-t_candidate)*a + t_candidate*a_new, 0)
            trial = trial / trial.sum()
            Mt = fast_autocorr(trial).max()
            if Mt < best_trial:
                best_trial = Mt
                best_t = t_candidate
        a = np.maximum((1-best_t)*a + best_t*a_new, 0)
        a = a / a.sum()
    
    return best_a, best_m
```
