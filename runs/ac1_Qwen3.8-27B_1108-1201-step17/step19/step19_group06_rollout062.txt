"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer in log-space with soft-max gradient over top autocorrelation peaks
2. Periodic spectral flattening (flattening power spectrum via FFT)
3. Multi-population search with diverse restart strategies
4. Adaptive length exploration
5. Final coordinate descent polish
"""

import numpy as np
import time

def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    ac = np.fft.irfft(A * A, fft_size)[:size]
    return ac

def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    """Safely call the evaluation function."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def score_normalized(a):
    """Score for normalized sequence (sum=1): 2*n*max(ac)."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

def compute_gradient(a, ac, top_k, beta, gamma=1.0):
    """
    Compute soft-max weighted gradient of max autocorrelation.
    Uses top-K lags with exponential weighting.
    """
    n = len(a)
    k_alarm = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_alarm]
    top_vals = ac[top_idx]
    
    max_v = top_vals.max()
    expo = beta * (top_vals - max_v)
    expo = np.clip(expo, -60, 60)
    weights = np.exp(expo)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_alarm) / k_alarm
    else:
        weights = weights / w_sum
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n, k_int + 1)
        if i_lo >= i_hi:
            continue
        idx = np.arange(i_lo, i_hi)
        j_idx = k_int - idx
        valid = (j_idx >= 0) & (j_idx < n)
        grad[idx[valid]] += 2.0 * w * gamma * a[j_idx[valid]]
    
    return grad
```

def adam_log_descent(a0, time_limit, lr_init=0.01, beta1=0.9, beta2=0.999,
                     soft_beta=3.0, top_k_init=50, verbose=True, rng_seed=42):
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    # Adam state in log-space
    m_hat = np.zeros(n)  # first moment
    v_hat = np.zeros(n)  # second moment
    t_adam = 0
    
    lr = lr_init
    soft_beta = soft_beta
    top_k = top_k_init
    stall = 0
    eps = 1e-8
    
    for t in range(1, 500000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        grad = compute_gradient(a, ac, top_k, soft_beta)
        
        t_adam += 1
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad**2
        m_hat_hat = m_hat / (1 - beta1**t_adam)
        v_hat_hat = v_hat / (1 - beta2**t_adam)
        
        # Update in log-space
        log_a = np.log(np.maximum(a, 1e-30))
        log_a = log_a - lr * m_hat_hat / (np.sqrt(v_hat_hat) + eps)
        log_a -= log_a.mean()  # center for stability
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0

    """Compute soft-max weights on autocorrelation lags."""
    n_lags = len(ac)

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based Adam optimizer in log-space with soft-max gradient
2. FFT cross-correlation for O(n log n) gradient computation
3. Periodic spectral flattening via power spectrum compression
4. Multi-population search with diverse restart strategies
5. Adaptive length exploration
6. Final coordinate descent polish
"""

import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    ac = np.fft.irfft(A * A, fft_size)[:size]
    return ac


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / max(len(a), 1)
    return a / s


def evaluate_safe(a):
    """Safely call the evaluation function."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def score_normalized(a):
    """Score for normalized sequence (sum=1): 2*n*max(ac)."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * float(ac.max())


def compute_gradient_fft(a, lag_weights):
    """
    Compute gradient = 2 * cross_corr(lag_weights, a)
    where cross_corr[j] = sum_m lag_weights[m+j] * a[m]
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    size = 2 * n - 1
    needed = size + n - 1
    fft_size = 1 << (needed - 1).bit_length()
    if fft_size < needed:
        fft_size <<= 1

    f = np.zeros(fft_size)
    f[:size] = lag_weights[:size]
    g = np.zeros(fft_size)
    g[:n] = a

    F = np.fft.rfft(f)
    G = np.fft.rfft(g)
    cross = np.fft.irfft(F * np.conj(G), fft_size)

    return 2.0 * cross[:n]


def soft_max_weights(ac, top_k, beta):
    """Compute soft-max weights on autocorrelation lags."""
    n_lags = len(ac)
    k_use = min(top_k, n_lags)
    top_idx = np.argsort(ac)[::-1][:k_use]
    top_vals = ac[top_idx]

    max_v = top_vals.max()
    if max_v < 1e-30:
        return np.ones(n_lags) / n_lags

    expo = beta * (top_vals - max_v)
    expo = np.clip(expo, -60, 60)
    weights = np.exp(expo)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_use) / k_use
    else:
        weights = weights / w_sum

    lag_w = np.zeros(n_lags)
    lag_w[top_idx] = weights
    return lag_w


def adam_log_descent(a0, time_limit, lr_init=0.008, beta1=0.9, beta2=0.999,
                     soft_beta_init=2.0, top_k_init=50,
                     restart_every=4000, verbose=True, rng_seed=42):
    """
    Adam optimizer in log-space with soft-max gradient.
    Uses FFT-based gradient computation for efficiency.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t_adam = 0

    lr = lr_init
    soft_beta = soft_beta_init
    top_k = top_k_init
    stall = 0
    eps = 1e-8
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        lag_w = soft_max_weights(ac, top_k, soft_beta)
        grad = compute_gradient_fft(a, lag_w)

        # Gradient clipping for stability
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e6:
            grad = grad / g_norm * 1e6

        t_adam += 1
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * (grad * grad)
        m_unbias = m_hat / (1.0 - beta1 ** t_adam)
        v_unbias = v_hat / (1.0 - beta2 ** t_adam)

        # Adam update in log-space
        log_a = np.log(np.maximum(a, 1e-30))
        step = lr * m_unbias / (np.sqrt(v_unbias) + eps)
        log_a -= step
        log_a -= log_a.mean()

        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new
            stall += 1
            if stall > restart_every:
                pert = _generate_perturbation(n, rng, best_a)
                w_mix = float(rng.uniform(0.2, 0.5))
                a = normalize_nonneg((1 - w_mix) * best_a + w_mix * pert)
                m_hat = np.zeros(n)
                v_hat = np.zeros(n)
                t_adam = 0
                lr = max(lr * 0.6, 0.001)
                soft_beta = max(soft_beta - 1.0, 0.5)
                stall = 0

        iter_count += 1

        # Adaptive schedule every 3000 iterations
        if iter_count % 3000 == 0:
            top_k = min(top_k + 5, 200)
            soft_beta = min(soft_beta + 0.3, 15.0)
            lr = max(lr * 0.96, 0.0005)

    J = 2.0 * n * float(best_m)
    if verbose:
        print(f"  Adam descent ({time.monotonic()-t0:.0f}s): J={J:.6f}, n={n}, iters={iter_count}")
    return best_a, best_m


def generate_perturbation(n, rng, best_a):
    """Generate a diverse perturbation for restarts."""
    pert_type = int(rng.integers(6))
    x = np.linspace(0, 1, n)

    if pert_type == 0:
        c = float(rng.uniform(0.2, 0.8))
        sig = float(rng.uniform(0.05, 0.4))
        pert = np.exp(-((x - c) ** 2) / (2 * sig ** 2))
    elif pert_type == 1:
        i_arr = np.arange(n, dtype=np.float64)
        p = float(rng.uniform(0.3, 3.0))
        pert = (i_arr + 1.0) ** p
    elif pert_type == 2:
        freq = int(rng.integers(1, 10))
        phase = float(rng.uniform(0, 2 * np.pi))
        pert = np.sin(np.pi * freq * x + phase) ** 2
    elif pert_type == 3:
        pert = best_a * (1.0 + rng.normal(0, 0.15, n))
    elif pert_type == 4:
        c1 = float(rng.uniform(0.1, 0.35))
        c2 = float(rng.uniform(0.65, 0.9))
        sig = float(rng.uniform(0.05, 0.2))
        pert = (np.exp(-((x - c1) ** 2) / (2 * sig ** 2)) +
                np.exp(-((x - c2) ** 2) / (2 * sig ** 2)))
    else:
        # Cosine-modulated uniform
        freq = int(rng.integers(1, 5))
        amp = float(rng.uniform(0.1, 0.5))
        pert = 1.0 + amp * np.cos(2 * np.pi * freq * x + float(rng.uniform(0, np.pi)))

    return normalize_nonneg(pert)


def spectral_flatten(a, time_limit, strength=0.05, verbose=False):
    """
    Iterate: flatten power spectrum, project to non-negative simplex.
    Reduces spectral peaks which correspond to autocorrelation structure.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    a = normalize_nonneg(a)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()

        if P_mean < 1e-30:
            break

        # Compress dynamic range of power spectrum
        P_norm = P / P_mean
        P_flat = P_norm ** (1.0 - strength)

        # Reconstruct with new magnitude, keep phase
        mag_new = np.sqrt(P_flat)
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = normalize_nonneg(a_new)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break

    if verbose:
        J = 2.0 * n * float(best_m)
        print(f"  Spectral flatten ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a


def coord_descent(a0, time_limit, rng_seed=77, verbose=True):
    """
    Coordinate descent: try moving mass between coordinates to reduce peak.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                # Random perturbation to escape
                n_moves = int(rng.integers(3, 15))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = float(rng.uniform(0.01, 0.05))
                        amt = min(a[i] * frac, a[i] - 1e-12)
                        a[i] -= amt
                        a[j] += amt
                a = normalize_nonneg(a)
                stall = 0
                continue

        # Get top lags
        n_top = min(20, size)
        top_lags = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k_pos in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break

            k_int = int(k_pos)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo >= i_hi:
                continue

            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]

            n_pairs = min(8, len(idx))
            pair_order = np.argsort(contrib)[::-1][:n_pairs]

            for pi in pair_order:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[pi])
                src_val = a[src]
                if src_val < 1e-14:
                    continue

                # Find destination with lowest impact
                scores = np.zeros(n)
                for k2 in top_lags[:6]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n, k2i + 1)
                    if s2 < e2:
                        ii = np.arange(s2, e2)
                        jj = k2i - ii
                        scores[ii] += a[jj] * ac[k2i]
                        scores[jj] += a[ii] * ac[k2i]

                scores[src] = np.inf
                dst = int(np.argmin(scores))

                if a[dst] > 0.7 * src_val:
                    continue

                for frac in [0.03, 0.08, 0.15, 0.25]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] = max(0, a_trial[src] - amt)
                    a_trial[dst] += amt
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and stall > 100:
            n_moves = int(rng.integers(5, 20))
            for _ in range(n_moves):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = float(rng.uniform(0.005, 0.03))
                    amt = min(a[i] * frac, a[i] - 1e-12)
                    a[i] -= amt
                    a[j] += amt
            a = normalize_nonneg(a)

    if verbose:
        J = 2.0 * n * float(best_m)
        print(f"  Coord descent ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a


def parametric_init(n, time_limit=10.0):
    """Quick parametric search for good initializations."""
    t0 = time.monotonic()
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)

    candidates = []

    # Uniform
    candidates.append(np.ones(n))

    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)

    # Product of power laws (bimodal at ends)
    for p in [0.5, 1.0, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p * (n - i_arr + 1) ** p)

    # Gaussians
    for c in [0.3, 0.5, 0.7]:
        for sig in [0.1, 0.2, 0.3, 0.5]:
            candidates.append(np.exp(-((x - c) ** 2) / (2 * sig ** 2)))

    # Bimodal
    for sig in [0.08, 0.12, 0.2]:
        candidates.append(
            np.exp(-((x - 0.2) ** 2) / (2 * sig ** 2)) +
            np.exp(-((x - 0.8) ** 2) / (2 * sig ** 2)))

    # Cosine-modulated
    for f in [1, 2, 3]:
        for amp in [0.2, 0.4, 0.6]:
            candidates.append(1.0 + amp * np.cos(2 * np.pi * f * x))
            candidates.append(1.0 + amp * np.sin(2 * np.pi * f * x) ** 2)

    # Sine windows
    for p in [0.5, 1.0, 2.0]:
        candidates.append(np.sin(np.pi * x) ** p)

    # Evaluate
    best_cand = None
    best_val = float('inf')
    for c in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        c_norm = normalize_nonneg(c)
        v = score_normalized(c_norm)
        if v < best_val:
            best_val = v
            best_cand = c_norm.copy()

    return best_cand if best_cand is not None else np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Multi-phase: init -> Adam descent -> spectral -> length exploration -> polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize_nonneg(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 5000
        best_seq = parametric_init(n0, time_limit=8.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = evaluate_safe(a_n)
        if np.isfinite(sc) and sc < best_score:
            prev_score = best_score
            best_score = sc
            best_seq = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_n)}, t={elapsed:.0f}s, delta={prev_score-sc:.6f})")

    # === Phase 0: Try parametric inits at current length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 0: Parametric init (n={n0}, budget={remaining:.0f}s) ===")
    a_param = parametric_init(n0, time_limit=8.0)
    check_update(a_param, "param_init")

    # === Phase 1: Main Adam log-descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Adam descent (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 30:
        a_out, _ = adam_log_descent(best_seq, t1, lr_init=0.008,

    # === Phase 2: Spectral flattening ===

    print(f"\n=== Phase 2: Spectral flatten (n={len(best_seq)}, budget={remaining:.0f}s) ===")

    if t2 > 15:
        a_out = spectral_flatten(best_seq, t2, strength=0.04, verbose=True)
        check_update(a_out, f"spec_n{len(best_seq)}")

    # === Phase 3: Second Adam pass (finer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Fine Adam (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)

        a_out, _ = adam_log_descent(best_seq, t3, lr_init=0.004,
                                    soft_beta_init=5.0, top_k_init=80,
                                    restart_every=3000, verbose=True, rng_seed=123)
        check_update(a_out, f"adam_fine_n{len(best_seq)}")

    # === Phase 4: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

            length_tries.append(rn)

        t4a = min(remaining * 0.4, 150.0)
        if t4a > 15:
            a_out, _ = adam_log_descent(a_interp, t4a, lr_init=0.006,
                                        soft_beta_init=3.0, top_k_init=60,
                                        restart_every=3000, verbose=False,

            check_update(a_out, f"adam_n{new_n}")

        if remaining > 40:
            t4b = min(remaining * 0.3, 80.0)

                a_out = spectral_flatten(best_seq, t4b, strength=0.05, verbose=False)
                check_update(a
