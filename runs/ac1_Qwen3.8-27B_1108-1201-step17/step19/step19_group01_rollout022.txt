def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def coordinate_descent(a0, time_limit, verbose=True, label="cd"):
    """Peak-targeted coordinate descent."""
    n = len(a0)

        ac = fast_conv(a)
        top_idx = np.argsort(ac)[::-1][:min(30, len(ac))]
        improved = False
        
        for k in top_idx[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = int(idx[np.argmax(contrib)])
            
            # Find position with lowest "pressure" to move mass to
            pressure = np.zeros(n)
            for k2 in top_idx[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]
            
            # Exclude positions involved in top peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_idx[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2+1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)
            
            cand_idx = candidates[np.argmin(pressure[candidates])]
            if cand_idx == src:
                # Find second best
                sorted_cand = candidates[np.argsort(pressure[candidates])]
                cand_idx = sorted_cand[1] if len(sorted_cand) > 1 else sorted_cand[0]
            if cand_idx == src:
                continue
            
            # Try moving mass
            for frac in [0.02, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_tri = a.copy()
                a_tri[src] -= amt
                a_tri[cand_idx] += amt
                a_tri = np.maximum(a_tri, 0.0)
                s_t = a_tri.sum()
                if s_t < 1e-15:
                    continue
                a_tri = a_tri / s_t
                M_tri = fast_conv(a_tri).max()
                if M_tri < best_m - 1e-14:
                    a = a_tri
                    best_m = M_tri
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation when stalled
            if stall > 50:
                for _ in range(3):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:

                        amt = a[i] * frac
                        a[i] -= amt

                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_conv(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

def parametric_search(n, time_limit, best_init, best_score):

        for w in [0.1, 0.2, 0.3, 0.5]:

    # Two bumps
    for c1 in [0.25, 0.3, 0.4]:
        for c2 in [0.6, 0.7, 0.75]:

    # Product of sine and cosine
    for f1 in [1, 2]:
        for f2 in [1, 2, 3]:

    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(a_raw)
        m = fast_conv(a).max()

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.1f}s)")
    return best_seq, best_score

def propose_candidate(budget_s=1800.0):
    """Main search function."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        parent = None
    
    if best_seq is None:
        n_base = 6000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    n0 = len(best_seq)
    
    # Phase 1: Multi-scale progressive refinement
    print(f"\n=== Phase 1: Multi-scale L-BFGS ===")
    
    # Determine scales
    scales = []
    i = 0
    while 2**(i+1) <= n0:
        scales.append(n0 // (2**(i+1)))
        i += 1
    scales.append(n0 // 2)
    scales.append(n0)
    scales = sorted(set(s for s in scales if s >= 500))
    
    current_a = best_seq.copy()
    
    for scale_idx, target_n in enumerate(scales):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        
        # Interpolate to target scale
        if len(current_a) != target_n:
            current_a = interpolate_sequence(current_a, target_n)
        
        # Convert to x parameterization
        current_a = np.maximum(current_a, 1e-15)
        x_init = np.log(current_a)
        
        # Allocate time based on scale (more time for larger scales)
        scale_fraction = (scale_idx + 1) / len(scales)
        t_alloc = min(remaining * 0.35 * scale_fraction, remaining * 0.4)
        t_alloc = max(t_alloc, 10.0)
        
        beta_start = 20.0 if target_n > 3000 else 30.0
        beta_end = 300.0 if target_n > 3000 else 500.0
        freq_reg = 0.001 if target_n > 3000 else 0.005
        
        x_out, a_out = lbfgs_with_restarts(
            x_init, beta_start, beta_end, t_alloc,
            freq_reg=freq_reg, verbose=True, label=f"scale_n{target_n}")
        
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"scale_n{target_n}", t_global)
        current_a = a_out.copy()
    
    # Phase 2: Focused L-BFGS at full resolution with high beta
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: High-beta L-BFGS (n={len(current_a)}) ===")
    if remaining > 60:
        n_best = len(current_a)
        x_init = np.log(np.maximum(current_a, 1e-15))
        t2 = min(remaining * 0.25, 300.0)
        if t2 > 30:
            x_out, a_out = lbfgs_with_restarts(
                x_init, 100.0, 2000.0, t2,
                freq_reg=0.001, verbose=True, label="high_beta")
            best_seq, best_score = try_update(best_seq, best_score, a_out, "high_beta", t_global)
            current_a = a_out.copy()
    
    # Phase 3: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coordinate descent (n={len(current_a)}) ===")
    if remaining > 30:
        t3 = min(remaining * 0.20, 200.0)
        if t3 > 20:
            a_out, _ = coordinate_descent(current_a, t3, verbose=True, label="cd1")
            best_seq, best_score = try_update(best_seq, best_score, a_out, "cd1", t_global)
            current_a = a_out.copy()
    
    # Phase 4: More L-BFGS rounds
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: More L-BFGS (n={len(current_a)}) ===")
    if remaining > 40:
        n_best = len(current_a)
        x_init = np.log(np.maximum(current_a, 1e-15))
        t4 = min(remaining * 0.25, 250.0)
        if t4 > 20:
            x_out, a_out = lbfgs_with_restarts(
                x_init, 50.0, 1000.0, t4,
                freq_reg=0.002, verbose=True, label="more_lbfgs")
            best_seq, best_score = try_update(best_seq, best_score, a_out, "more_lbfgs", t_global)
            current_a = a_out.copy()
    
    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n0
    print(f"\n=== Phase 5: Multi-length exploration (base n={n_best}) ===")
    
    longer_lengths = [n_best + 200, n_best + 500, n_best + 1000, n_best + 2000]
    longer_lengths = [l for l in longer_lengths if l <= 12000]
    
    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(current_a, new_n)
        
        t5a = min(remaining * 0.30, 120.0)
        if t5a > 20:
            x_init = np.log(np.maximum(a_long, 1e-15))
            x_out, a_out = lbfgs_with_restarts(
                x_init, 30.0, 500.0, t5a,
                freq_reg=0.001, verbose=True, label=f"len{new_n}")
            best_seq, best_score = try_update(best_seq, best_score, a_out, f"len{new_n}", t_global)
            if a_out is not None:
                current_a = a_out.copy()
    
    # Phase 6: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final CD (n={len(current_a)}) ===")
    if remaining > 20:
        t6 = min(remaining * 0.30, 150.0)
        if t6 > 15:
            a_out, _ = coordinate_descent(current_a, t6, verbose=True, label="cd_final")
            best_seq, best_score = try_update(best_seq, best_score, a_out, "cd_final", t_global)
            current_a = a_out.copy()
    
    # Phase 7: Final L-BFGS
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 7: Final L-BFGS (n={len(current_a)}) ===")
        t7 = min(remaining * 0.50, 200.0)
        if t7 > 20:
            x_init = np.log(np.maximum(current_a, 1e-15))
            x_out, a_out = lbfgs_with_restarts(
                x_init, 80.0, 2000.0, t7,
                freq_reg=0.001, verbose=True, label="final")
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_lbfgs", t_global)
            current_a = a_out.copy()
    
    # Phase 8: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15 and current_a is not None:
        print(f"\n=== Phase 8: Final polish ===")
        t8 = min(remaining * 0.40, 100.0)
        if t8 > 10:
            a_out, _ = coordinate_descent(current_a, t8, verbose=True, label="polish")
            best_seq, best_score = try_update(best_seq, best_score, a_out, "polish", t_global)
    
    # Final validation
    if best_seq is not None:
        final_score = evaluate_safe(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={elapsed:.1f}s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
    
    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
