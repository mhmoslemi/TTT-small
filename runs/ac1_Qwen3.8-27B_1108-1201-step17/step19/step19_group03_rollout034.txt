    conv_max = conv.max()
    # Soft weighting over top peaks
    k_idx = np.argsort(conv)[::-1][:top_k]
    k_vals = np.maximum(conv[k_idx] - conv_max + 1e-30, 1e-30)
    w = k_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(top_k) / top_k
    else:
        w = w / w_sum
    g = np.zeros(n)
    for wi, k in zip(w, k_idx):
        ki = int(k)
        for i in range(n):
            j = ki - i
            if 0 <= j < n:
                g[i] += 2.0 * wi * a[j]
    return g


def compute_subgradient_fast(a, conv, top_k=20, beta=3.0):
    """Faster sub-gradient using vectorized operations."""
    n = len(a)
    conv_max = conv.max()
    k_idx = np.argsort(conv)[::-1][:top_k]
    k_vals = np.maximum(conv[k_idx] - conv_max + 1e-30, 1e-30)
    w = k_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(top_k) / top_k
    else:
        w = w / w_sum
    g = np.zeros(n)
    for wi, k in zip(w, k_idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        g[idx] += 2.0 * wi * a[j_idx]
    return g


def project_nonneg_sum1(v):
    """Project v onto {a >= 0, sum(a) = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones_like(v) / len(v)
    return v / s


def subgradient_optimize(a0, time_limit, top_k=30, beta=4.0, eta_init=0.1,
                         backtracking=True, perturb_every=2000, verbose=False,
                         rng_seed=42):
    """Sub-gradient descent with backtracking line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    conv = fast_conv(a)
    best_m = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    eta = eta_init
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_conv(a)
        M = conv.max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Compute sub-gradient
        g = compute_subgradient_fast(a, conv, top_k=top_k, beta=beta)

        # Project gradient onto tangent space of {sum=1}
        g_tangent = g - g.mean()

        # Step
        if backtracking:
            # Adaptive eta based on progress
            eta = eta_init / (1.0 + 0.01 * stall)
            if stall > 500:
                eta *= 0.5
            if eta < 1e-6:
                eta = eta_init * 0.1

        a_trial = a - eta * g_tangent
        a_new = project_nonneg_sum1(a_trial)

        M_new = fast_conv(a_new).max()
        if M_new <= M + 1e-12:
            a = a_new
        else:
            # Backtrack
            a = a_new  # Still move, but reduce eta
            eta *= 0.7

        # Periodic perturbation
        if stall > perturb_every:
            stall = 0
            eta = eta_init
            # Structured perturbation: mass transfer
            n_pert = max(3, n // 500)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = project_nonneg_sum1(a)
            M_pert = fast_conv(a).max()
            if M_pert < best_m:
                best_m = M_pert
                best_a = a.copy()

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SG it={it} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.6f}, stall={stall}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, top_k=30, verbose=False, rng_seed=0):
    """Coordinate descent: move mass from high-impact to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    conv = fast_conv(a)
    best_m = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        conv = fast_conv(a)
        M = conv.max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0

        # Find top peaks
        peak_idx = np.argsort(conv)[::-1][:top_k]

        # Compute sensitivity: for each position i, how much does it contribute to peaks?
        sens = np.zeros(n)
        for k in peak_idx[:15]:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            sens[idx] += a[ki - idx]

        # Find high-sensitivity positions and low-sensitivity positions
        high_sens = np.argsort(sens)[::-1][:max(5, n // 200)]
        low_sens = np.argsort(sens)[:max(5, n // 100)]

        improved = False
        for src in high_sens[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            for dst in low_sens[:10]:
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-14:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = project_nonneg_sum1(a)
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall > 0 and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def block_perturb_search(a0, time_limit, n_blocks=5, verbose=False, rng_seed=0):
    """Perturb contiguous blocks of the sequence and optimize locally."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    conv = fast_conv(a)
    best_m = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    block_size = max(5, n // 200)

    while time.monotonic() - t0 < time_limit:
        # Create block perturbation
        a_pert = a.copy()
        for _ in range(n_blocks):
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + block_size)
            # Random perturbation within block
            pert = rng.normal(0, 0.02 * a.mean(), end - start)
            a_pert[start:end] += pert
        a_pert = project_nonneg_sum1(a_pert)

        M_pert = fast_conv(a_pert).max()

        if M_pert < best_m:
            best_m = M_pert
            best_a = a_pert.copy()
            a = a_pert.copy()
        else:
            # Try a shorter sub-gradient run from perturbed point
            a_sub, _ = subgradient_optimize(a_pert, min(5.0, (time_limit - (time.monotonic() - t0)) * 0.3),
                                            top_k=20, beta=3.0, eta_init=0.05,
                                            perturb_every=500, verbose=False,
                                            rng_seed=int(rng.integers(100000)))
            M_sub = fast_conv(a_sub).max()
            if M_sub < best_m:
                best_m = M_sub
                best_a = a_sub.copy()
                a = a_sub.copy()

    return best_a, best_m


def multi_peak_balance(a0, time_limit, verbose=False, rng_seed=0):
    """Try to balance the top peaks of the convolution to be equal."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    conv = fast_conv(a)
    best_m = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_conv(a)
        M = conv.max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()

        # Find the top 3 peaks
        top3_idx = np.argsort(conv)[::-1][:3]
        top3_vals = conv[top3_idx]
        target = top3_vals.mean()

        # Compute "excess" for each peak
        g = np.zeros(n)
        for k, val in zip(top3_idx, top3_vals):
            if val > target:
                ki = int(k)
                i_start = max(0, ki - n + 1)
                i_end = min(n, ki + 1)
                idx = np.arange(i_start, i_end)
                j_idx = ki - idx
                weight = (val - target) / max(top3_vals[0], 1e-30)
                g[idx] += weight * a[j_idx]

        # Step to reduce excess peaks
        g_tangent = g - g.mean()
        eta = 0.05 / (1 + 0.001 * it)
        a_trial = a - eta * g_tangent
        a_new = project_nonneg_sum1(a_trial)

        M_new = fast_conv(a_new).max()
        if M_new <= M + 1e-12:
            a = a_new
        else:
            # Try smaller step
            a_trial = a - eta * 0.3 * g_tangent
            a_new = project_nonneg_sum1(a_trial)
            a = a_new

        # Periodic random restart from best
        if it % 5000 == 0:
            pert = rng.normal(0, 0.01, n) * best_a
            a_p = np.maximum(best_a + pert, 1e-10)
            a = a_p / a_p.sum()
            M_p = fast_conv(a).max()
            if M_p < best_m:
                best_m = M_p
                best_a = a.copy()

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Balance it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def mirror_descent_optimize(a0, time_limit, top_k=40, beta=5.0, c0=3.0,
                            verbose=False, rng_seed=0):
    """Away-step mirror descent on log-scale."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    conv = fast_conv(a)
    best_m = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        conv_y = fast_conv(y)
        g = compute_subgradient_fast(y, conv_y, top_k=top_k, beta=beta)

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = np.maximum(best_a + pert, 1e-10)
            x = x_r / x_r.sum()
            x_prev = x.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_moment_match_init(n):
    """Initialize based on moment matching: edges ~ 1/sqrt(2n), interior ~ 1/n."""
    a = np.ones(n) / n
    edge_val = 1.0 / np.sqrt(2.0 * n)
    a[0] = edge_val
    a[-1] = edge_val
    # Adjust interior to maintain sum = 1
    interior_sum = 1.0 - 2.0 * edge_val
    if interior_sum > 0 and n > 2:
        a[1:-1] = interior_sum / (n - 2)
    return a


def make_skewed_init(n, alpha=2.0):
    """Create a skewed (non-symmetric) initialization."""
    x = np.linspace(0, 1, n)
    a = (1.0 + alpha * x) ** 2
    return norm_sum1(a)


def make_bimodal_skewed(n):
    """Bimodal but asymmetric."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.2) / 0.15) ** 2) + 0.8 * np.exp(-((x - 0.8) / 0.2) ** 2) + 0.01
    return norm_sum1(a)


def make_random_smooth(n, seed=0):
    """Random smooth non-negative sequence."""
    rng = np.random.default_rng(seed)
    # Generate in frequency domain for smoothness
    fft_size = 1 << (n - 1).bit_length()
    spec = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
    spec[0] = 1.0
    for k in range(1, fft_size // 2 + 1):
        decay = np.exp(-k / (n * 0.1))
        spec[k] = decay * (rng.normal(0, 1) + 1j * rng.normal(0, 1))
    a_full = np.fft.irfft(spec, fft_size)
    a = a_full[:n]
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        best_seq = make_skewed_init(n_base, alpha=2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Sub-gradient descent from parent ===
    print(f"\n=== Phase 1: Sub-gradient (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_optimize(a_in, t1, top_k=40, beta=4.0,
                                         eta_init=0.08, perturb_every=3000,
                                         verbose=True, rng_seed=42)
        try_update(a_out, "sg_parent")

    # === Phase 2: Mirror descent ===
    print(f"\n=== Phase 2: Mirror Descent (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_optimize(a_in, t2, top_k=40, beta=5.0, c0=4.0,
                                            verbose=True, rng_seed=7)
        try_update(a_out, "md")

    # === Phase 3: Multi-peak balance ===
    print(f"\n=== Phase 3: Multi-peak Balance (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = multi_peak_balance(a_in, t3, verbose=True, rng_seed=12)
        try_update(a_out, "balance")

    # === Phase 4: Coordinate descent polish ===
    print(f"\n=== Phase 4: Coordinate Descent (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, top_k=30, verbose=True, rng_seed=99)
        try_update(a_out, "coord1")

    # === Phase 5: Block perturbation search ===
    print(f"\n=== Phase 5: Block Perturbation (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = block_perturb_search(a_in, t5, n_blocks=3, verbose=False, rng_seed=55)
        try_update(a_out, "block")

    # === Phase 6: Multi-init sub-gradient ===
    print(f"\n=== Phase 6: Multi-init Sub-gradient (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 250.0)
    if t6 > 60:
        inits = [
            (norm_sum1(best_seq), "best"),
            (make_moment_match_init(n0), "moment"),
            (make_skewed_init(n0, alpha=3.0), "skew3"),
            (make_skewed_init(n0, alpha=1.0), "skew1"),
            (make_bimodal_skewed(n0), "bimodal"),
            (make_random_smooth(n0, seed=77), "rand"),
        ]
        # Add parent blend
        try:
            p = np.asarray(height_sequence_1, dtype=np.float64)
            if p.ndim == 1 and p.size == n0 and p.sum() > 1e-10:
                inits.append((norm_sum1(0.5 * norm_sum1(p) + 0.5 * make_skewed_init(n0, 2.0)), "blend"))
        except Exception:
            pass

        t_per = max(10.0, t6 / len(inits))
        for idx, (init, lbl) in enumerate(inits):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 8:
                break
            print(f"  SG init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = subgradient_optimize(init, t_avail, top_k=30, beta=4.0,
                                             eta_init=0.06, perturb_every=2000,
                                             verbose=False, rng_seed=100 + idx)
            try_update(a_out, f"sg_{lbl}")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length Exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 2000, -2000, 3000, -3000]:
        new_n = n_best + delta
        if 500 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.25, 120.0)
        if t7a > 20:
            a_out, _ = subgradient_optimize(a_long, t7a, top_k=30, beta=4.0,
                                             eta_init=0.05, perturb_every=2000,
                                             verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"sg_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.3, 80.0)
            if t7b > 15:
                a_out, _ = coordinate_polish(a_long, t7b, top_k=25, verbose=False,
                                              rng_seed=new_n % 500)
                try_update(a_out, f"coord_n{new_n}")

    # === Phase 8: Intensive sub-gradient on best ===
    print(f"\n=== Phase 8: Intensive SG (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.20, 200.0)
    if t8 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_optimize(a_in, t8, top_k=50, beta=6.0,
                                         eta_init=0.04, perturb_every=4000,
                                         verbose=True, rng_seed=777)
        try_update(a_out, "sg_intensive")

    # === Phase 9: Final coordinate polish ===
    print(f"\n=== Phase 9: Final Polish (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t9 = min(remaining * 0.25, 150.0)
    if t9 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t9, top_k=40, verbose=True, rng_seed=314)
        try_update(a_out, "coord_final")

    # === Phase 10: Final micro-optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 10: Micro-opt (n={len(best_seq)}) ===")
        t10 = min(remaining * 0.7, 100.0)
        if t10 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = subgradient_optimize(a_in, t10, top_k=20, beta=8.0,
                                             eta_init=0.01, perturb_every=5000,
                                             verbose=False, rng_seed=271)
            try_update(a_out, "micro1")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_polish(a_in, min(remaining * 0.8, 60.0),
                                              top_k=50, verbose=False, rng_seed=161)
                try_update(a_out, "coord_micro")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = make_skewed_init(n_fb, alpha=2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
