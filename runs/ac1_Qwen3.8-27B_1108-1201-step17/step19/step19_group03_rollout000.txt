    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_soft_gradient(a, tau):
    """Compute gradient of log-sum-exp smooth max of autocorrelation."""
    n = len(a)
    ac = fast_autocorr(a)
    tau_shifted = tau * (ac - ac.max())
    max_exp = tau_shifted.max()
    exp_ac = np.exp(tau_shifted - max_exp)
    p = exp_ac / exp_ac.sum()  # softmax weights over autocorrelation peaks
    # g[j] = 2 * sum_m p[j+m] * a[m]
    # This is 2 * correlate(p, a, 'valid')
    g = 2.0 * sp_correlate(p, a, mode='valid')
    return g, ac


def compute_topk_gradient(a, top_k=50, beta=5.0):
    """Compute gradient using top-k autocorrelation peaks with power weighting."""
    n = len(a)
    ac = fast_autocorr(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        idx = np.arange(j_start, j_end + 1)
        g[idx] += 2.0 * wi * a[ki - idx]
    return g

def lp_frank_wolfe(a0, time_limit, solver, top_K_init=200, top_K_max=None, verbose=False):
    """LP-based Frank-Wolfe with line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0
    stall = 0
    max_stall = 100
    
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]
        
        # Build constraint matrix (sparse would be better but let's use dense for now)
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            idx = np.arange(j_start, j_end + 1)
            T_sel[ii, idx] = a[ki - idx]
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            iter_count += 1
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 25, top_K_max)
            iter_count += 1
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            # LP found nearly zero solution, try harder
            K = min(K + 100, top_K_max)
            iter_count += 1
            continue
        
        g_dir = b_val / b_sum
        
        # Line search
        base_M = fast_autocorr(a).max()
        best_alpha = 2.0 / (iter_count + 2.0)
        best_m_trial = base_M
        
        alpha_candidates = [
            2.0 / (iter_count + 2.0),
            min(1.0, 1.5 * 2.0 / (iter_count + 2.0)),
            min(1.0, 2.0 * 2.0 / (iter_count + 2.0)),
            1.0 / (iter_count + 2.0),
            0.5, 0.3, 0.2, 0.1, 0.05
        ]
        alpha_candidates = sorted(set([x for x in alpha_candidates if 0 < x <= 1.0]))
        
        for alpha in alpha_candidates:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_m_trial:
                best_m_trial = m_trial
                best_alpha = alpha
        
        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a
        
        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive K growth
        if iter_count % 5 == 0:
            K = min(K + 30, top_K_max)
        
        # Perturbation on stall
        if stall > 30:
            pert = rng.normal(0, 0.01, n) * best_a
            x_r = np.maximum(best_a + pert, 1e-15)
            a = x_r / x_r.sum()
            a = norm_sum1(a)
            stall = 0
            K = min(K - 50, top_K_max)  # Reduce K to escape
        
        iter_count += 1
        
        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, stall={stall}")
    
    return best_a, best_m
