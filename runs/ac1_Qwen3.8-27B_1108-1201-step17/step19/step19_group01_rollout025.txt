    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _MODULE_SOLVER


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad_v2(y, beta, top_k=None, pair_aware=True):
    """Compute gradient of smoothed max autocorrelation with pair awareness."""
    n = len(y)
    b = fast_autocorr(y)
    size = len(b)
    
    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    exp_sum = exp_vals.sum()
    if exp_sum < 1e-30:
        p = np.zeros(size)
        p[np.argmax(b_use)] = 1.0
    else:
        p = exp_vals / exp_sum
    
    # Gradient via FFT convolution
    fft_size = 1 << (2 * n - 1).bit_length()
    Ar = np.fft.rfft(y, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv_raw = np.fft.irfft(Ar * Pr, fft_size)
    
    # The gradient of (a*a)[k] w.r.t. a[i] is: a[k-i] (for valid k-i)
    # conv_raw[j] = sum_k p[k] * a[k] * a[j-k] ... no
    # Actually conv_raw = a * p (convolution)
    # We need sum_k p[k] * (a*a)_k gradient w.r.t a[i]
    # d(a*a)_k / da[i] = a[k-i] + a[i-k] (if both in range) = contribution from pair (i, k-i)
    # sum_k p[k] * (a[k-i] + a[i-k]) = 2 * (a * p_full)[i] where p_full is symmetric extension
    
    # Simpler: grad[i] = sum_k p[k] * (a[k-i] if 0<=k-i<n) + sum_k p[k] * (a[i-k] if 0<=i-k<n)
    # = (a * p)[i] + (a_reverse * p)[i] ... let me just use the direct FFT approach
    
    # grad[i] = sum_k p[k] * a[k-i] + sum_k p[k] * a[i-k]
    # First term: (p * a)[i] = sum_k p[k] * a[i-k] ... hmm indexing is tricky
    # Let's use: grad = (a * p_symmetric) where p_symmetric handles both directions
    # Actually the cleanest way:
    # (a*a)[k] = sum_i a[i]*a[k-i]
    # d/dx_i (a*a)[k] = a[k-i] + (a[i-k] if i-k valid)
    # For i in [0,n-1], k-i is valid when k-i in [0,n-1], i.k. i in [k-n+1, k]
    # i-k is valid when i-k in [0,n-1], i.k. i in [k, k+n-1]
    
    # So grad[i] = sum_{k: 0<=k-i<n} p[k] * a[k-i] + sum_{k: 0<=i-k<n} p[k] * a[i-k]
    # First sum: let j = k-i, so k = i+j, j in [0,n-1], k in [i, i+n-1]
    #   = sum_{j=0}^{n-1} p[i+j] * a[j] (if i+j < size)
    # Second sum: let j = i-k, so k = i-j, j in [0,n-1], k in [i-n+1, i]
    #   = sum_{j=0}^{n-1} p[i-j] * a[j] (if i-j >= 0)
    
    # This is essentially: grad[i] = sum_j a[j] * (p[i+j] + p[i-j])
    # = (a * p)[i] where * is a specific correlation-like operation
    
    # Let me just compute it directly using the FFT of p:
    # grad = ifft(fft(p) * fft(a)) doesn't quite work because of the symmetric structure
    
    # Simplest correct approach:
    # grad[i] = sum_{j=0}^{n-1} a[j] * p[i+j] + sum_{j=0}^{n-1} a[j] * p[i-j] (with p[x]=0 if x not in [0,size-1])
    # This is: (a * p_fwd)[i] + (a * p_bwd)[i] where p_fwd[i+j] and p_bwd[i-j]
    
    # Using FFT: sum_j a[j]*p[i+j] = (a * p)[i] in the "correlation" sense
    # Let's just use: conv_ap = np.fft.irfft(np.fft.rfft(a, fft_size) * np.fft.rfft(p, fft_size), fft_size)
    # conv_ap[k] = sum_i a[i]*p[k-i]
    # We want sum_j a[j]*p[i+j] = conv_ap[i+2*(n-1)] ... no this is getting confusing
    
    # Let me use the most direct method:
    # Create a "kernel" K where K[m] = sum of p contributions
    # grad[i] = sum_j a[j] * (p[i+j] + p[i-j])
    # Define q[m] = p[m] for m in [0, size-1], else 0
    # Then grad[i] = sum_j a[j]*q[i+j] + sum_j a[j]*q[i-j]
    # First term: (a * q)[i] in the "anti-convolution" sense = correlation
    
    # I'll use a clean FFT-based computation:
    # Let's define: for the first sum, S1[i] = sum_j a[j] * p[i+j]
    # This is the cross-correlation of a and p, evaluated at i
    # S1 = ifft(fft(a) * conj(fft(p))) ... in circular sense
    
    # Actually, let me just use the approach from the original code which works:
    # Original: a_rev = y[::-1]; conv = ifft(fft(a_rev)*fft(p)); grad = conv[n-1:2n-1]
    # Let me verify: (a_rev * p)[k] = sum_i a_rev[i]*p[k-i] = sum_i a[n-1-i]*p[k-i]
    # = sum_j a[j]*p[k-(n-1-j)] = sum_j a[j]*p[k-n+1+j]
    # So (a_rev*p)[n-1+i] = sum_j a[j]*p[i+j] -- this is S1[i]! Good.
    # And (a_rev*p)[n-1-i] = sum_j a[j]*p[-i+j] = sum_j a[j]*p[j-i] -- this is S2[i]! Good.
    # So grad[i] = (a_rev*p)[n-1+i] + (a_rev*p)[n-1-i]
    
    a_rev = y[::-1]
    fft_size2 = 1 << (2 * n - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size2)
    Pr = np.fft.rfft(p, fft_size2)
    conv_rp = np.fft.irfft(Ar * Pr, fft_size2)
    
    # grad[i] = conv_rp[n-1+i] + conv_rp[n-1-i] for i in [0, n-1]
    # But conv_rp has length fft_size2, and we need indices 0 to 2n-2
    valid_len = 2 * n - 1
    conv_valid = conv_rp[:valid_len]
    
    idx_fwd = np.arange(n - 1, 2 * n - 1)  # n-1+i for i in [0,n-1]
    idx_bwd = np.arange(n - 1, -1, -1)      # n-1-i for i in [0,n-1]
    
    grad = conv_valid[idx_fwd] + conv_valid[idx_bwd]
    grad = grad / beta
    
    return grad, p, b


def nesterov_v2(a0, time_limit, beta_start=10.0, beta_end=800.0,
                eta_base=1.0, top_k_ratio=0.05, restart_every=2000,
                verbose=True):
    """Nesterov with adaptive beta, cosine step-size, and pair-aware gradient."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(123)
    top_k = max(10, int(n * top_k_ratio))
    total_iters = 300000
    
    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        # Compute gradient with pair-awareness
        grad, p, b = compute_smoothed_grad_v2(y, beta, top_k, pair_aware=True)
        
        # Cosine-annealed step size with adaptive component
        progress = min(t / total_iters, 1.0)
        cos_factor = 0.5 * (1 + np.cos(np.pi * progress))
        eta = eta_base * cos_factor / max(1.0, beta * p.max())
        eta = min(eta, 0.5)
        eta = max(eta, 0.001)
        
        # Gradient step + projection
        z = y - eta * grad
        x_new = simplex_project(z)
        
        # Track best
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        # Adaptive beta: increase slowly, reset on restart
        if t % 50 == 0:
            beta = min(beta * 1.005, beta_end)
        
        # Restart with simulated annealing style perturbation
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            # Annealing: perturbation decreases over time
            anneal_temp = max(0.001, 0.03 * np.exp(-0.5 * stall / restart_every)) * (1 + 0.5 * (1 - progress))
            pert = rng.normal(0, anneal_temp, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r < 1e-15:
                x_r = best_a.copy()
            else:
                x_r = x_r / s_r
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, beta_start * 0.5)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest v2 restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " temp=" + repr(round(anneal_temp, 5)))
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest v2 t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " beta=" + repr(round(beta, 1)))
    
    return best_a, best_max_b


def lp_fw_v2(a0, time_limit, solver, initial_K=300, max_K=4000,
             barrier_decay=0.9995, min_barrier_frac=0.85,
             add_per_iter=20, verbose=True):
    """LP-Frank-Wolfe with decreasing barrier for tighter constraints."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K])
    
    # Barrier: start at current max, decrease over time
    barrier = best_m
    no_improve = 0
    max_no_improve = 800
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW v2: converged at iter " + str(t))
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            barrier = min(barrier, M_curr)
        else:
            no_improve += 1
        
        # Decrease barrier over time (tighten constraint)
        barrier = max(barrier * barrier_decay, best_m * min_barrier_frac)
        
        # Manage active set: add high-violation, remove low-margin
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([barrier - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)
        
        # Add new active constraints (top violations of barrier)
        viol = np.where(ac > barrier * 0.99)[0]
        if len(viol) > len(constraint_list):
            # Add top ones not already in set
            for k in viol[:add_per_iter]:
                if int(k) not in active_set:
                    active_set.add(int(k))
            constraint_list = sorted(active_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Trim if too large
        if K > max_K:
            constraint_list = constraint_list[:max_K]
            active_set = set(constraint_list)
            K = max_K
        
        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list[:K]):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        # LP: min M subject to sum(g)=1, g>=0, coeffs@g <= M, M <= barrier
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var, M_var <= barrier]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K:
                active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            # If infeasible, relax barrier
            barrier = min(barrier * 1.01, M_curr * 1.01)
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.01, default_alpha * 0.5),
                  max(0.01, default_alpha * 2.0), 1.0 / max(1, t + 1)]
        alphas = [max(0.01, min(1.0, a)) for a in alphas]
        alphas = list(set(alphas))
        
        best_alpha = default_alpha
        best_trial_M = fast_autocorr((1 - default_alpha) * y + default_alpha * g_dir).max()
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y
        
        y = y_new
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW v2 iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " |S|=" + str(len(active_set)) + " barrier=" + repr(round(barrier / best_m, 4)))
    
    return best_a, best_m


def pair_coordinate_descent(a0, time_limit, verbose=True):
    """Targeted pair-based coordinate descent focusing on peak contributors."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        peak_k = int(np.argmax(ac))
        n_top = min(30, len(ac))
        top_peaks = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for pk in top_peaks[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            pk_int = int(pk)
            # Find the pair (i, pk-i) with maximum contribution
            i_start = max(0, pk_int - n + 1)
            i_end = min(n - 1, pk_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[pk_int - idx]
            max_idx_local = int(np.argmax(contrib))
            src = int(idx[max_idx_local])
            partner = pk_int - src
            
            # Try to reduce the peak by moving mass from the larger of src/partner
            if src_val_check(a, src, partner):
                # Move from src to a "safe" location
                safe_dst = find_safe_destination(a, top_peaks, src, n)
                if safe_dst is not None:
                    for frac in [0.01, 0.03, 0.05, 0.1, 0.15]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            break
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[safe_dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            break
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
        
        if not improved:
            stall += 1
            # Random exploration when stuck
            if stall > 50:
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                break
        
        if verbose and stall % 200 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Pair CD (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " stall=" + str(stall))
    
    return best_a, best_m


def src_val_check(a, src, partner):
    """Check if src is a valid position to reduce from."""
    n = len(a)
    if src < 0 or src >= n:
        return False
    if a[src] < 1e-12:
        return False
    if 0 <= partner < n and a[partner] < a[src] * 0.3:
        return False  # Don't reduce if partner is much smaller
    return True


def find_safe_destination(a, top_peaks, src, n):
    """Find a position with low 'pressure' to receive mass."""
    pressure = np.zeros(n)
    for pk in top_peaks[:10]:
        pk_int = int(pk)
        s = max(0, pk_int - n + 1)
        e = min(n - 1, pk_int)
        if s <= e:
            idx = np.arange(s, e + 1)
            pressure[idx] += a[pk_int - idx]
            pressure[pk_int - idx] += a[idx]
    
    # Exclude src and positions near peaks
    in_peak = np.zeros(n, dtype=bool)
    for pk in top_peaks[:5]:
        pk_int = int(pk)
        s = max(0, pk_int - n + 1)
        e = min(n - 1, pk_int)
        in_peak[s:e + 1] = True
    in_peak[src] = True
    
    candidates = np.where(~in_peak)[0]
    if len(candidates) < 10:
        candidates = np.where(~in_peak[src] | (np.arange(n) == src))[0]
        candidates = np.setdiff1d(np.arange(n), [src])
    
    if len(candidates) == 0:
        return None
    
    cand_pressure = pressure[candidates]
    # Prefer low pressure and low current value
    score = cand_pressure + 0.1 * a[candidates]
    best_idx = int(candidates[np.argmin(score)])
    return best_idx


def parametric_init_v2(n, time_limit, best_init, best_score):
    """Diverse parametric initialization."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score
    
    candidates = []
    # Power laws (both directions)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
        candidates.append(((i_arr + 1.0) ** p) * (n - i_arr + 1.0) ** p)
    # Symmetric shapes
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((x ** p) + ((1.0 - x) ** p))
    # Gaussian mixtures
    for c1 in [0.2, 0.3, 0.4, 0.5]:
        for sigma in [0.05, 0.1, 0.15, 0.2]:
            candidates.append(np.exp(-((x - c1) ** 2) / (2.0 * sigma * sigma)))
            candidates.append(np.exp(-((x - c1) ** 2) / (2.0 * sigma * sigma)) +
                            np.exp(-((x - (1 - c1)) ** 2) / (2.0 * sigma * sigma)))
    # Cosine/sine
    for freq in [1, 2, 3, 4]:
        candidates.append(np.maximum(0.1 + 0.9 * np.cos(2.0 * np.pi * freq * x), 0.0))
        candidates.append(np.sin(np.pi * x) ** freq)
    # Bimodal
    for gap in [0.2, 0.3, 0.4]:
        mid = n // 2
        left = np.exp(-((i_arr[:mid]) / (n * gap)) ** 2)
        right = np.exp(-((i_arr[mid:] - n) / (n * gap)) ** 2)
        candidates.append(np.concatenate([left, right]))
    
    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(np.asarray(a_raw, dtype=np.float64))
        if len(a) != n:
            a = interpolate_sequence(a, n)
        b = fast_autocorr(a)
        m = b.max()
        if m < best_val:
            best_val = m
            best_a = a.copy()
    
    # Random initialization with various structures
    rng = np.random.default_rng(2024)
    for _ in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        # Random smooth curve
        freqs = rng.choice(np.arange(1, 20), size=rng.integers(2, 8), replace=False)
        amplitudes = rng.random(len(freqs)) ** rng.uniform(0.5, 2.0)
        phases = rng.uniform(0, 2 * np.pi, len(freqs))
        a_raw = np.ones(n)
        for f, amp, ph in zip(freqs, amplitudes, phases):
            a_raw += amp * np.cos(2 * np.pi * f * x + ph)
        a_raw = np.maximum(a_raw, 0.01)
        a = norm_sum1(a_raw)
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()
    
    return best_a, best_val


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

    Main search combining Nesterov v2, LP-FW v2 with barrier descent,
    pair-targeted coordinate descent, and multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None
    
    if best_seq is None:
        n_base = 6000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))
    
    n0 = len(best_seq)
    
    # Phase 1: Parametric diversity
    print("\n=== Phase 1: Parametric (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 40.0)
    if t1 > 10:
        a_param, _ = parametric_init_v2(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "param", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov v2 (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.25, 420.0)
    if t2 > 60:
        a_out, _ = nesterov_v2(norm_sum1(best_seq), t2,
                               beta_start=10.0, beta_end=600.0,
                               eta_base=1.0, top_k_ratio=0.04,
                               restart_every=2000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2_n" + str(n_best), t_global)
    
    # Phase 3: LP-FW with barrier descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP-FW v2 (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.25, 450.0)
    if t3 > 60 and n_best <= 8000:
        a_out, _ = lp_fw_v2(norm_sum1(best_seq), t3, solver,
                            initial_K=400, max_K=4000,
                            barrier_decay=0.9997, min_barrier_frac=0.88,
                            add_per_iter=25, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp2_n" + str(n_best), t_global)
    
    # Phase 4: Pair coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
