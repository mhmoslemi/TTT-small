"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe with adaptive active set, backtracking line search
2. Aggressive coordinate peeling for local refinement
3. Multi-length exploration
4. Smooth penalty-based gradient refinement

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None

def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  Solver: ECOS (fallback)")
    return cp.ECOS

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_lp_matrix(a, active_set, n):
    """Build the constraint matrix for the LP: rows are active lags, cols are n."""
    K = len(active_set)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(active_set):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[ki - idx]
    return coeffs

def lp_fw(a0, time_limit, solver, init_K=300, max_K=4000, add_per=30, verbose=True, reoptimize_every=50):
    """
    LP-based Frank-Wolfe algorithm with:
    - Adaptive active set growth
    - Backtracking line search
    - Periodic re-optimization of active set
    """
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:init_K])
    no_improve = 0
    stall_limit = 2000
    
    lp_prob = None
    g_var = None
    M_var = None
    prev_M = None
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update active set: add top violators, prune by margin
        if t % reoptimize_every == 0 or t == 1:
            top_lags = np.argsort(ac)[::-1]
            add_count = min(add_per, max(0, len(top_lags) - len(active_set)))
            for k in top_lags[:len(active_set) + add_count]:
                active_set.add(int(k))
            
            if len(active_set) > max_K:
                margins = np.array([M_curr - ac[k] for k in active_set])
                keep_idx = np.argsort(margins)[:max_K]
                clist = sorted(active_set)
                active_set = set(int(clist[i]) for i in keep_idx)
        
        # Also always include the current top lag
        active_set.add(int(np.argmax(ac)))
        
        clist = sorted(active_set)
        K = len(clist)
        if K == 0:
            break
        
        # Build LP
        try:
            coeffs = build_lp_matrix(y, clist, n)
            
            if lp_prob is None or g_var is None or len(g_var.shape) != 1 or g_var.shape[0] != n:
                g_var = cp.Variable(n, nonneg=True)
                M_var = cp.Variable()
                constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
                lp_prob = cp.Problem(cp.Minimize(M_var), constraints)
            
            if prev_M is not None and M_curr > 0:
                # Scale the constraint: use relative bound
                scale = max(M_curr, 1e-15)
                lp_prob = cp.Problem(
                    cp.Minimize(M_var),
                    [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
                )
            
            lp_prob.solve(solver=solver, verbose=False, warm_start=(lp_prob.status in ('optimal', 'optimal_inaccurate') if lp_prob is not None else False))
            
            if lp_prob.status not in ('optimal', 'optimal_inaccurate'):
                raise Exception("LP failed: " + str(lp_prob.status))
            
        except Exception:
            # Fallback: collapse active set
            if len(active_set) > init_K:
                clist = sorted(active_set)
                active_set = set(int(clist[i]) for i in np.argsort([M_curr - ac[k] for k in clist])[:init_K])
                clist = sorted(active_set)
                try:
                    coeffs = build_lp_matrix(y, clist, n)
                    g_var = cp.Variable(n, nonneg=True)
                    M_var = cp.Variable()
                    lp_prob = cp.Problem(cp.Minimize(M_var), [cp.sum(g_var) == 1, coeffs @ g_var <= M_var])
                    lp_prob.solve(solver=solver, verbose=False)
                    if lp_prob.status not in ('optimal', 'optimal_inaccurate'):
                        continue
                except Exception:
                    continue
            else:
                continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Check for violation to grow active set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per]]
            for k in top_v:
                active_set.add(int(k))
        
        # Backtracking line search
        best_alpha = 1.0 / (t + 1)
        best_trial = fast_autocorr(normalize(y + best_alpha * (g_dir - y))).max()
        
        # Try a range of alphas
        alphas_to_try = [1.0/(t+1), 2.0/(t+2), 0.3, 0.5, 0.7, 1.0]
        alphas_to_try = sorted(set(max(0.005, min(1.0, a)) for a in alphas_to_try))
        
        for alpha in alphas_to_try:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            Mt = fast_autocorr(trial).max()
            if Mt < best_trial - 1e-15:
                best_trial = Mt
                best_alpha = alpha
        
        # Bisection refinement
        lo = max(0.0, best_alpha - 0.1)
        hi = min(1.0, best_alpha + 0.1)
        for _ in range(6):
            mid1 = lo + (hi - lo) / 3.0
            mid2 = hi - (hi - lo) / 3.0
            t1 = np.maximum((1 - mid1) * y + mid1 * g_dir, 0.0)
            t2 = np.maximum((1 - mid2) * y + mid2 * g_dir, 0.0)
            s1, s2 = t1.sum(), t2.sum()
            m1 = fast_autocorr(t1 / s1).max() if s1 > 1e-15 else 1e10
            m2 = fast_autocorr(t2 / s2).max() if s2 > 1e-15 else 1e10
            if m1 < m2:
                hi = mid2
            else:
                lo = mid1
        best_alpha = (lo + hi) / 2.0
        
        # Update
        if t == 1:
            y = g_dir.copy()
        else:
            y = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t={:5d} ({:.1f}s) J={:.6f} |S|={}".format(t, elapsed, 2*n*best_m, len(active_set)))
    
    elapsed = time.monotonic() - t0
    if verbose:
        print("    LP-FW done ({:.1f}s) J={:.6f} |S|={}".format(elapsed, 2*n*best_m, len(active_set)))
    return best_a, 2.0 * n * best_m

def coord_peel(a0, time_limit, n_targets=15, max_frac=0.02, seed=42, verbose=True):
    """
    Aggressive coordinate peeling:
    - Identify top lags in autocorrelation
    - Find the pair contributing most to each peak
    - Move mass from the contributing position to the "quietest" position
    - Accept only if objective improves
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 1500
    
    # Precompute the "score" array: for each position, how much it contributes to top peaks
    def compute_pressure(a, ac, top_lags):
        pressure = np.zeros(n, dtype=np.float64)
        for k in top_lags:
            ki = int(k)
            ws = max(0, ki - n + 1)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            pressure[idx] += a[ki - idx]
            pressure[ki - idx] += a[idx]
        return pressure
    
    fracs = [0.002, 0.005, 0.008, 0.012, 0.018, max_frac]
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(n_targets, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False
        
        # Get all contributing pairs for top lags
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            ws = max(0, ki - n + 1)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            contrib = a[idx] * a[ki - idx]
            
            if contrib.max() < 1e-25:
                continue
            
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-15:
                continue
            
            # Compute pressure to find good destinations
            pressure = compute_pressure(a, ac, top_k[:8])
            pressure[src] = 1e10  # Never move to self
            
            # Try several candidate destinations
            quiet_positions = np.argsort(pressure)[:20]
            for dst in quiet_positions:
                if dst == src:
                    continue
                dst = int(dst)
                found = False
                for frac in fracs:
                    amt = src_val * frac
                    if amt < 1e-18:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                   Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-16:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break
            
            if not improved and stall > 200:
                # Random perturbation
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.001, 0.008)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if not improved:
            stall += 1
            if stall > 300:
                # Larger random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
        
        if verbose and stall % 500 == 0:
            print("    Peel ({:.1f}s) J={:.6f} stall={}".format(time.monotonic()-t0, 2*n*best_m, stall))
    
    return best_a, 2.0 * n * best_m

def smooth_refine(a0, time_limit, n_top=50, lr0=0.01, verbose=True):
    """
    Smooth penalty-based gradient refinement.
    Minimizes: max(b) + mu * sum_k relu(b[k] - target)^2
    using mirror descent with exponential updates.
    """
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    Yf = np.fft.rfft(x, fft_size)
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 5000:
            break
        
        ac = fast_autocorr(x)
        M_curr = ac.max()
        
        if 2.0 * n * M_curr < best_val - 1e-12:
            best_val = 2.0 * n * M_curr
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Compute gradient of top-K weighted autocorrelation
        k_max = min(n_top, len(ac))
        idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[idx]
        # Weight by excess over average
        avg = ac.mean()
        excess = np.maximum(top_vals - avg, 0.0)
        w = excess / (excess.sum() + 1e-30)
        
        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            i_s = max(0, ki)
            i_e = min(n, ki + 1)
            idx_arr = np.arange(i_s, i_e)
            # d b[ki] / d a[j] = a[ki-j] (or 2*a[ki-j] if ki != 2j)
            for j in idx_arr:
                partner = ki - j
                if 0 <= partner < n:
                    if ki == 2 * j:
                        grad[j] += wi * a_val  # Hmm this is getting complicated
                    else:
                        grad[j] += 2.0 * wi * x[partner]
        
        # Actually let me just use the simpler formulation
        # grad[j] = sum_{k in top} w_k * x[k-j] (for valid k-j)
        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            for j in range(n):
                p = ki - j
                if 0 <= p < n:
                    grad[j] += wi * x[p]
        
        # Mirror descent update
        lr = lr0 / (1.0 + 0.01 * t)
        log_x = np.log(x + 1e-30) - lr * grad
        log_x -= log_x.max()
        x_new = np.exp(log_x)
        x_new = np.maximum(x_new, 1e-25)
        x_new = x_new / x_new.sum()
        x = x_new
        
        if stall > 2000 and stall % 500 == 0:
            # Restart with noise
            noise = 0.01 * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x = x_r / x_r.sum()
            stall = 0
        
        if verbose and t % 500 == 0:
            print("    Smooth t={} ({:.1f}s) J={:.6f}".format(t, time.monotonic()-t0, best_val))
    
    return best_a, best_val

For the weighted sum over k: grad[j] = sum_k w_k * a[k-j] * (1 + 1[j != k-j]) ≈ sum_k w_k * a[k-j]

The factor of 2 (for j != k-j) is approximately 2 for most j, so we can just use:
grad[j] ≈ 2 * sum_k w_k * a[k-j] (for valid k-j)

This is a "cross-correlation" of a with the weight sequence w. We can compute it via FFT!

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe with adaptive active set and backtracking line search
2. FFT-accelerated mirror descent gradient refinement
3. Aggressive coordinate peeling for local refinement

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None

def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  Solver: ECOS (fallback)")
    return cp.ECOS

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_lp_matrix(a, active_list, n):
    K = len(active_list)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj in range(K):
        ki = int(active_list[jj])
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[ki - idx]
    return coeffs

def fft_gradient(a, top_lags, top_weights):
    """Compute gradient of sum_k w_k * b[k] with respect to a, using FFT."""
    n = len(a)
    w_padded = np.zeros(2 * n - 1)
    for i in range(len(top_lags)):
        ki = int(top_lags[i])
        if 0 <= ki < 2 * n - 1:
            w_padded[ki] = top_weights[i]
    flipped_w = w_padded[::-1]
    req_size = 3 * n - 2
    fft_size = 1 << (req_size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(flipped_w, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    grad = conv_result[2 * n - 2 : n - 2 : -1]
    return grad

def lp_fw(a0, time_limit, solver, init_K=200, max_K=4000, add_per=25, verbose=True):
    """LP-based Frank-Wolfe with adaptive active set and backtracking line search."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:init_K])
    no_improve = 0
    stall_limit = 2500
    lp_vars = None

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodically refresh active set
        if t % 30 == 0 or t == 1:
            top_lags = np.argsort(ac)[::-1]
            current_size = len(active_set)
            target_size = min(max_K, init_K + t * 2)
            if current_size < target_size:
                for k in top_lags[:target_size]:
                    active_set.add(int(k))
            if len(active_set) > max_K:
                clist = sorted(active_set)
                margins = np.array([M_curr - ac[k] for k in clist])
                keep = np.argsort(margins)[:max_K]
                active_set = set(int(clist[i]) for i in keep)
        active_set.add(int(np.argmax(ac)))

        clist = sorted(active_set)
        K = len(clist)
        if K == 0:
            break

        try:
            coeffs = build_lp_matrix(y, clist, n)
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            prob.solve(solver=solver, verbose=False)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                raise Exception("LP status: " + str(prob.status))
        except Exception:
            if len(active_set) > init_K // 2:
                clist = sorted(active_set)
                active_set = set(int(clist[i]) for i in np.argsort([M_curr - ac[k] for k in clist])[:init_K])
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations to grow active set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per]]
            for k in top_v:
                active_set.add(int(k))

        # Backtracking + bisection line search
        best_alpha = 2.0 / (t + 2.0)
        trial = np.maximum((1 - best_alpha) * y + best_alpha * g_dir, 0.0)
        ts = trial.sum()
        best_trial = fast_autocorr(trial / ts).max() if ts > 1e-15 else 1e10

        for alpha in [1.0/(t+1), 0.3, 0.5, 0.7, 0.9, 1.0]:
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            Mt = fast_autocorr(trial / ts).max()
            if Mt < best_trial - 1e-15:
                best_trial = Mt
                best_alpha = alpha

        # Bisection refinement
        lo = max(0.0, best_alpha - 0.15)
        hi = min(1.0, best_alpha + 0.15)
        for _ in range(6):
            mid1 = lo + (hi - lo) / 3.0
            mid2 = hi - (hi - lo) / 3.0
            t1 = np.maximum((1 - mid1) * y + mid1 * g_dir, 0.0)
            t2 = np.maximum((1 - mid2) * y + mid2 * g_dir, 0.0)
            s1, s2 = t1.sum(), t2.sum()
            m1 = fast_autocorr(t1 / s1).max() if s1 > 1e-15 else 1e10
            m2 = fast_autocorr(t2 / s2).max() if s2 > 1e-15 else 1e10
            if m1 < m2:
                hi = mid2
            else:
                lo = mid1
        best_alpha = (lo + hi) / 2.0

        if t <= 2:
            y = g_dir.copy()
        else:
            y = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t={:5d} ({:.1f}s) J={:.7f} |S|={}".format(t, elapsed, 2 * n * best_m, len(active_set)))

    elapsed = time.monotonic() - t0
    if verbose:
        print("    LP-FW done ({:.1f}s) J={:.7f} |S|={}".format(elapsed, 2 * n * best_m, len(active_set)))
    return best_a, 2.0 * n * best_m

def grad_refine(a0, time_limit, n_top=100, lr0=0.05, verbose=True):
    """FFT-accelerated mirror descent with soft top-k weighting."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    size = 2 * n - 1
    fft_size_grad = 1 << (3 * n - 3).bit_length()

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 8000:
            break

        ac = fast_autocorr(x)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Select top-K lags with temperature weighting
        k_sel = min(n_top, len(ac))
        idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = ac[idx]
        temp = 30.0 + 20.0 * (t / 1000.0)
        w = np.exp(temp * (top_vals - top_vals[0]))
        w = w / w.sum()

        # FFT-based gradient
        grad = fft_gradient(x, idx, w)

        # Mirror descent update with adaptive LR
        lr = lr0 / (1.0 + 0.005 * t)
        log_x = np.log(x + 1e-30) - lr * grad
        log_x -= log_x.max()
        x_new = np.exp(log_x)
        x_new = np.maximum(x_new, 1e-25)
        x_new = x_new / x_new.sum()
        x = x_new

        # Periodic restart with noise
        if stall > 3000 and stall % 1000 == 0:
            noise = 0.02 * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x = x_r / x_r.sum()
            stall = 0

        if verbose and t % 500 == 0:
            print("    GradRef t={} ({:.1f}s) J={:.7f} stall={}".format(t, time.monotonic() - t0, best_val, stall))

    return best_a, best_val

def coord_peel(a0, time_limit, n_targets=12, seed=42, verbose=True):
    """Aggressive coordinate peeling: move mass from peak contributors to quiet positions."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 2000
    fracs = [0.001, 0.003, 0.005, 0.008, 0.012, 0.02, 0.035]

    def pressure_(a_, ac_, top_lags_):
        pr = np.zeros(n, dtype=np.float64)
        for k in top_lags_:
            ki = int(k)
            ws = max(0, ki)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            pr[idx] += a_[ki - idx]
        return pr

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(n_targets, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            ws = max(0, ki)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-25:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-15:
                continue

            pr = pressure_(a, ac, top_k[:8])
            pr[src] = 1e10
            quiet_positions = np.argsort(pr)[:30]

            for dst in quiet_positions:
                dst = int(dst)
                if dst == src:
                    continue
                found = False
                for frac in fracs:
                    amt = src_val * frac
                    if amt < 1e-18:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-16:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break

            if not improved and stall > 300:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
                            improved = True
                break

        if not improved:
            stall += 1
            if stall > 500:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.003, 0.025)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 500 == 0 and stall > 0:
            print("    Peel ({:.1f}s) J={:.7f} stall={}".format(time.monotonic() - t0, 2 * n * best_m, stall))

    return best_a, 2.0 * n * best_m

        t5 = min(remaining * 0.15, 300.0)
        if t5 > 30:
            a_long = interpolate_seq(best_seq, n_long)
            a_out, v_out = grad_refine(a_long, t5 * 0.6, n_top=200, lr0=0.03, verbose=True)
            a_out2, v_out2 = coord_peel(a_out, t5 * 0.4, n_targets=10, seed=99, verbose=False)
            sc1 = eval_safe(a_out)
            sc2 = eval_safe(a_out2)
            if np.isfinite(min(sc1, sc2)) and min(sc1, sc2) < best_score:
                best_score = min(sc1, sc2)
                best_seq = a_out if sc1 < sc2 else a_out2
                print("  P5 improved: {}".format(repr(best_score)))

    # Phase 6: Final LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if n_cur <= 12000:
        print("\n=== P6: Final LP-FW (n={}) ===".format(n_cur))
        t6 = min(remaining * 0.15, 300.0)
        if t6 > 30:
            a_out, v_out = lp_fw(normalize(best_seq), t6, solver, init_K=300, max_K=4500, add_per=30, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  P6 improved: {}".format(repr(best_score)))
    else:
        print("\n=== P6: Final GradRef (n={}) ===".format(n_cur))
        t6 = min(remaining * 0.15, 300.0)
        if t6 > 30:
            a_out, v_out = grad_refine(normalize(best_seq), t6, n_top=200, lr0=0.02, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  P6 improved: {}".format(repr(best_score)))

    # Phase 7: Final coord polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P7: Final Peel (n={}) ===".format(n_cur))
    t7 = min(remaining * 0.20, 400.0)
    if t7 > 20:
        a_out, v_out = coord_peel(normalize(best_seq), t7, n_targets=15, seed=123, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P7 improved: {}".format(repr(best_score)))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={} n={} t={:.1f}s".format(repr(best_score), len(best_seq), elapsed))
    else:
        print("\nERROR fallback")
        n_fb = 10000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)

    return [float(x) for x in best_seq]
