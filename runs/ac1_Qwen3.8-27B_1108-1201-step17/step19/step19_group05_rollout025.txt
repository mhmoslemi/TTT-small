def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    
    if v.sum() <= 0:
        return np.ones(n) / n
    
    # Algorithm from the paper by Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    result = np.maximum(v - theta, 0)
    # Handle numerical issues
    if result.sum() < 1e-15:
        return np.ones(n) / n
    return result / result.sum()
