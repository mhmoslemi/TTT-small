def fast_autocorr(a):

    return np.fft.irfft(A * A, fft_size)[:size]

def multiplicative_updates(a0, time_limit, M_target=None, verbose=True, rng_seed=55):
    """
    Multiplicative weight updates targeting a specific peak level.
    If M_target is None, uses current peak. Otherwise tries to achieve that peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    if M_target is None:
        M_target = best_m * 0.99
    
    lr = 0.1
    stall = 0
    
    for t in range(1, 500000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Compute violated lags
        violated = ac > M_target
        n_violated = violated.sum()
        
        if n_violated == 0:
            # All constraints satisfied, try to reduce M_target
            M_target = M_curr * 0.995
            stall += 1
            if stall > 500:
                break
            continue
        else:
            stall = 0
        
        # Multiplicative update: reduce a_i where it contributes to violated lags
        # For each violated lag k, a[i] contributes via a[i]*a[k-i]
        penalty = np.zeros(n)
        top_v = np.argsort(ac)[::-1][:50]
        for k_val in top_v:
            if ac[k_val] <= M_target:
                break
            k_int = int(k_val)
            excess = (ac[k_int] - M_target) / max(M_target, 1e-10)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            valid = (j_idx >= 0) & (j_idx < n)
            penalty[idx[valid]] += excess * a[j_idx[valid]]
        
        # Apply multiplicative update
        lr_t = lr / (1.0 + 0.05 * np.sqrt(t))
        a = a * np.exp(-lr_t * penalty)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        ac_new = fast_autocorr(a)
        m_new = ac_new.max()
        if m_new < best_m - 1e-13:
            best_m = m_new
            best_a = a.copy()
        
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Mult updates done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss

def peak_clipping(a0, P_target, max_iter=10000):
    """
    Iteratively clip the peak of the autocorrelation to P_target.
    Returns (a, succeeded) where succeeded is True if max(a*a) <= P_target.
    """
    n = len(a0)
    a = norm_sum1(a0)
    
    for _ in range(max_iter):
        ac = fast_autocorr(a)
        M = ac.max()
        if M <= P_target * (1 + 1e-10):
            return a, True
        
        # Find violating lags
        viol = ac > P_target
        # For each violating lag, compute the "excess" and reduce contributions
        # Multiplicative: a[i] *= exp(-eta * sum_k excess[k] * a[k-i])
        penalty = np.zeros(n)
        n_viol = min(100, viol.sum())
        viol_idx = np.where(viol)[0][:n_viol]
        
        for k in viol_idx:
            excess = (ac[k] - P_target) / P_target
            i_lo = max(0, k - n + 1)
            i_hi = min(n, k + 1)
            idx = np.arange(i_lo, i_hi)
            j = k - idx
            penalty[idx] += excess * a[j]
        
        eta = 0.1
        a = a * np.exp(-eta * penalty)
        a = np.maximum(a, 0)
        a = a / a.sum()
    
    ac = fast_autocorr(a)
    return a, ac.max() <= P_target

def fast_autocorr(a):

    # Softmax weighting over all lags
    ac_clipped = np.maximum(ac, 1e-30)
    expo = beta * (ac_clipped - ac_clipped.max())
    expo = np.clip(expo, -50, 50)
    weights = np.exp(expo)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(size) / size
    else:
        weights = weights / w_sum
    
    # Gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] (for valid k-i)
    grad = np.zeros(n)
    
    # Use top weighted lags for efficiency
    top_idx = np.argsort(weights)[::-1][:top_k]
    top_w = weights[top_idx]
    
    for w_val, k_val in zip(top_w, top_idx):
        if w_val < 1e-15:
            continue
        k_int = int(k_val)
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n, k_int + 1)
        if i_lo >= i_hi:
            continue
        idx = np.arange(i_lo, i_hi)
        j_idx = k_int - idx
        valid = (j_idx >= 0) & (j_idx < n)
        grad[idx[valid]] += 2.0 * w_val * a[j_idx[valid]]
    
    return grad


def run_lse_gd(a0, time_limit, lr_init=0.05, beta_init=5.0,
               top_k=200, verbose=True, rng_seed=42):
    """
    Gradient descent with log-sum-exp smoothing + Adam optimizer.
    Works in log-space for non-negativity.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    # Adam state
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0
    
    lr = lr_init
    beta_lse = beta_init
    stall = 0
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Compute LSE-soft gradient
        grad = lse_soft_gradient(a, ac, beta_lse, top_k)
        
        # Adam update
        t_adam += 1
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad ** 2
        m_hat = m_v / (1 - beta1 ** t_adam)
        v_hat = v_v / (1 - beta2 ** t_adam)
        
        # Learning rate with decay
        lr_t = lr / (1.0 + 0.01 * np.sqrt(t_adam))
        
        # Update in log-space
        log_a = np.log(np.maximum(a, 1e-30)) - lr_t * m_hat / np.sqrt(v_hat + eps)
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        # Check improvement
        ac_new = fast_autocorr(a_new)
        m_new = ac_new.max()
        
        if m_new < best_m - 1e-13:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Don't accept, increment stall
            stall += 1
            if stall > 1500:
                # Restart with perturbation
                stall = 0
                pert = best_a * (1.0 + rng.normal(0, 0.02, n))
                a = norm_sum1(pert)
                m_v = np.zeros(n)
                v_v = np.zeros(n)
                t_adam = 0
                lr = max(lr * 0.85, 0.003)
                beta_lse = min(beta_lse * 1.1, 25.0)
                if beta_lse >= 25.0:
                    # Try decreasing beta (sharper)
                    beta_lse = 3.0
        
        # Slowly increase beta for convergence
        if t % 10000 == 0:
            beta_lse = min(beta_lse * 1.01, 25.0)
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    LSE-GD done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def peak_clipping(a0, P_target, time_limit, eta_init=0.1, verbose=False):
    """
    Iteratively clip the autocorrelation peak to P_target using multiplicative updates.
    Water-filling inspired: reduce contributions to over-constrained lags.
    Returns the best sequence achieved.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = eta_init
    stall = 0
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_m - 1e-13:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break
        
        if M <= P_target * (1 + 1e-8):
            # Achieved target, tighten it
            P_target = M * 0.999
            stall = 0
            continue
        
        # Find violating lags and compute penalty
        viol_mask = ac > P_target
        n_viol = min(80, int(viol_mask.sum()))
        if n_viol == 0:
            break
        
        viol_idx = np.where(viol_mask)[0]
        # Sort by excess to focus on worst
        viol_excess = ac[viol_idx] - P_target
        order = np.argsort(viol_excess)[::-1][:n_viol]
        viol_idx = viol_idx[order]
        
        penalty = np.zeros(n)
        for k in viol_idx:
            k_int = int(k)
            excess = (ac[k_int] - P_target) / max(P_target, 1e-10)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo >= i_hi:
                continue
            idx = np.arange(i_lo, i_hi)
            j = k_int - idx
            penalty[idx] += excess * a[j]
        
        # Adaptive step size
        eta_t = eta / (1.0 + 0.02 * np.sqrt(t))
        
        a = a * np.exp(-eta_t * penalty / max(penalty.max(), 1e-15))
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Peak clipping done ({time.monotonic()-t0:.0f}s): J={J:.6f}, target={P_target:.6f}")
    return best_a, best_m


def coord_descent_v2(a0, time_limit, rng_seed=77, verbose=True):
    """
    Improved coordinate descent with better destination scoring.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15000:
                break
        
        # Get top lags
        n_top = min(15, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k_pos in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            
            k_int = int(k_pos)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo >= i_hi:
                continue
            
            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            valid = (j_idx >= 0) & (j_idx < n)
            idx_v = idx[valid]
            j_v = j_idx[valid]
            contrib = a[idx_v] * a[j_v]
            
            # Find top contributing source positions
            n_pairs = min(15, len(idx_v))
            pair_order = np.argsort(contrib)[::-1][:n_pairs]
            
            for pi in pair_order[:6]:
                src = int(idx_v[pi])
                src_val = a[src]
                if src_val < 1e-12:
                    continue
                
                # Score destinations
                dst_scores = np.zeros(n)
                for k2 in top_lags[:6]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n, k2i + 1)
                    if s2 < e2:
                        ii = np.arange(s2, e2)
                        jj = k2i - ii
                        v2 = (jj >= 0) & (jj < n)
                        dst_scores[ii[v2]] += ac[k2i] * a[jj[v2]]
                
                dst_scores[src] = np.inf
                dst_scores += 0.05 * a  # penalize already-high positions
                
                dst = int(np.argmin(dst_scores))
                
                # Try several transfer amounts (binary search style)
                best_frac = None
                best_trial_m = best_m
                for frac in [0.02, 0.05, 0.08, 0.15, 0.25]:
                    amt = src_val * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-14:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_trial_m:
                        best_trial_m = m_trial
                        best_frac = frac
                
                if best_frac is not None and best_trial_m < best_m - 1e-14:
                    amt = src_val * best_frac
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    best_m = best_trial_m
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            if stall % 300 == 0:
                # Random perturbation
                n_moves = int(rng.integers(5, 20))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = float(rng.uniform(0.003, 0.02))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Coord descent v2 done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def generate_diverse_inits(n):
    """Generate diverse initial candidates emphasizing asymmetric shapes."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []
    
    # Uniform
    candidates.append(np.ones(n))
    
    # Power laws (asymmetric)
    for p in [-3, -2, -1.5, -1, -0.7, -0.5, -0.3, 0, 0.3, 0.5, 1, 2, 3, 4]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    
    # Exponentials (very asymmetric)
    for r in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0]:
        candidates.append(np.exp(-r * x))
        candidates.append(np.exp(r * x))
    
    # Skewed Gaussians
    for c in [0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.8, 0.85, 0.9]:
        for sig in [0.05, 0.08, 0.1, 0.15, 0.2, 0.3]:
            candidates.append(np.exp(-((x - c) ** 2) / (2 * sig ** 2)))
    
    # Bimodal asymmetric
    candidates.append(
        np.exp(-((x - 0.15) ** 2) / 0.02) + 0.5 * np.exp(-((x - 0.7) ** 2) / 0.05))
    candidates.append(
        0.5 * np.exp(-((x - 0.3) ** 2) / 0.03) + np.exp(-((x - 0.85) ** 2) / 0.02))
    
    # Sine-based
    for f in [1, 2, 3]:
        candidates.append(np.sin(np.pi * f * x) ** 2)
        candidates.append(np.abs(np.sin(np.pi * f * x)))
        candidates.append((1 + np.sin(np.pi * f * x)) ** 2)
    
    # Window functions
    candidates.append(0.54 + 0.46 * np.cos(2 * np.pi * x))  # Hanning
    candidates.append(0.42 + 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))
    
    # Beta-like distributions (approximated)
    for alpha, beta_b in [(2, 5), (3, 8), (1.5, 4), (4, 2), (5, 3), (2, 3), (3, 2)]:
        # Approximate beta pdf using gamma function sampling
        t = np.linspace(1e-6, 1 - 1e-6, n)
        f_beta = t ** (alpha - 1) * (1 - t) ** (beta_b - 1)
        candidates.append(f_beta)
    
    # Random smooth
    rng = np.random.default_rng(42)
    for _ in range(10):
        n_terms = int(rng.integers(3, 8))
        f_rand = np.ones(n) * 0.5
        for _ in range(n_terms):
            amp = float(rng.uniform(0.1, 0.5))
            freq = float(rng.uniform(0.5, 4.0))
            phase = float(rng.uniform(0, 2 * np.pi))
            f_rand += amp * np.cos(2 * np.pi * freq * x + phase)
        f_rand = np.maximum(f_rand, 0.01)
        candidates.append(f_rand)
    
    # Normalize
    normed = []
    for c in candidates:
        c = np.maximum(np.asarray(c, dtype=np.float64), 0.0)
        s = c.sum()
        if s > 1e-10:
            normed.append(c / s)
    
    return normed


def diverse_init_search(n, time_limit=20.0):
    """Find best initial candidate from diverse set."""
    t0 = time.monotonic()
    candidates = generate_diverse_inits(n)
    
    best = None
    best_score = float('inf')
    
    for c in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        s = score_from_ac(c)
        if s < best_score:
            best_score = s
            best = c.copy()
    
    return best if best is not None else np.ones(n) / n


def micro_polish(a0, time_limit, rng_seed=999):
    """Very fine-tuned local search with tiny perturbations."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    while time.monotonic() - t0 < time_limit:
        i = int(rng.integers(n))
        if a[i] < 1e-14:
            continue
        j = int(rng.integers(n))
        if j == i:
            continue
        
        best_frac = None
        best_trial_m = best_m
        for frac in [0.001, 0.003, 0.005, 0.01]:
            amt = a[i] * frac
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_frac = frac
        
        if best_frac is not None:
            amt = a[i] * best_frac
            a[i] -= amt
            a[j] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            best_m = best_trial_m
            best_a = a.copy()
    
    J = 2.0 * n * best_m
    print(f"    Micro polish done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining multiple optimization strategies:
    LSE-GD, peak clipping, coordinate descent, length exploration, micro polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n0 = 5000
        best_seq = diverse_init_search(n0, time_limit=10.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = evaluate_safe(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")
    
    n0 = len(best_seq)
    curr_peak = fast_autocorr(best_seq).max()
    
    # === Phase 1: LSE Gradient Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LSE-GD (n={n0}, budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.22, 330.0)
    if t1 > 30:
        a_out, _ = run_lse_gd(best_seq, t1, lr_init=0.06, beta_init=4.0,
                              top_k=200, verbose=True, rng_seed=42)
        check_update(a_out, f"lse1_n{n0}")
    
    # === Phase 2: Peak Clipping ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Peak clipping (n={n0}, budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.18, 270.0)
    if t2 > 20:
        # Set target slightly below current peak
        target = curr_peak * 0.98
        a_out, _ = peak_clipping(best_seq, target, t2, eta_init=0.15, verbose=True)
        check_update(a_out, f"clip_n{n0}")
    
    # === Phase 3: Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coord descent (n={n0}, budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 20:
        a_out, _ = coord_descent_v2(best_seq, t3, rng_seed=123, verbose=True)
        check_update(a_out, f"coord_n{n0}")
    
    # === Phase 4: Second LSE-GD (finer, different params) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LSE-GD fine (n={n0}, budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out, _ = run_lse_gd(best_seq, t4, lr_init=0.02, beta_init=8.0,
                              top_k=400, verbose=False, rng_seed=77)
        check_update(a_out, f"lse2_n{n0}")
    
    # === Phase 5: More Peak Clipping (aggressive) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Aggressive clipping (n={n0}, budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        curr_peak2 = fast_autocorr(best_seq).max()
        target2 = curr_peak2 * 0.97
        a_out, _ = peak_clipping(best_seq, target2, t5, eta_init=0.2, verbose=False)
        check_update(a_out, f"clip2_n{n0}")
    
    # === Phase 6: Length Exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration (n_best={n_best}, budget={remaining:.0f}s) ===")
    
    length_tries = []
    for delta in [-500, -200, 200, 500, 800, 1000, 1500, 2000]:
        nn = n_best + delta
        if 1500 <= nn <= 9000 and nn != n_best:
            length_tries.append(nn)
    for rn in [2000, 3000, 4000, 5000, 6000, 7000, 8000]:
        if rn != n_best and 1500 <= rn <= 9000:
            length_tries.append(rn)
    length_tries = sorted(set(length_tries))
    
    for new_n in length_tries[:5]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")
        
        a_interp = interpolate_sequence(best_seq, new_n)
        
        # LSE-GD on new length
        t6a = min(remaining * 0.35, 120.0)
        if t6a > 15:
            a_out, _ = run_lse_gd(a_interp, t6a, lr_init=0.04, beta_init=5.0,
                                  top_k=150, verbose=False, rng_seed=new_n % 1000)
            check_update(a_out, f"lse_n{new_n}")
        
        # Peak clipping on new length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.25, 60.0)
            if t6b > 10:
                pk = fast_autocorr(a_interp).max()
                a_out, _ = peak_clipping(a_interp, pk * 0.98, t6b, eta_init=0.12, verbose=False)
                check_update(a_out, f"clip_n{new_n}")
    
    # === Phase 7: Final Polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}, budget={remaining:.0f}s) ===")
    
    # Micro polish
    t7a = min(remaining * 0.25, 80.0)
    if t7a > 10:
        a_out, _ = micro_polish(best_seq, t7a, rng_seed=42)
        check_update(a_out, f"micro_n{n_best}")
    
    # Final LSE-GD
    remaining = time_budget - (time.monotonic() - t_global)
    t7b = min(remaining * 0.35, 120.0)
    if t7b > 15:
        a_out, _ = run_lse_gd(best_seq, t7b, lr_init=0.008, beta_init=10.0,
                              top_k=500, verbose=False, rng_seed=314)
        check_update(a_out, f"lse_fin_n{n_best}")
    
    # Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    t7c = min(remaining * 0.30, 60.0)
    if t7c > 10:
        a_out, _ = coord_descent_v2(best_seq, t7c, rng_seed=271, verbose=False)
        check_update(a_out, f"coord_fin_n{len(best_seq)}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
