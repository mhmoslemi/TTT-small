Actually, the LP is: `min_g max_{k in S} (y*g)[k]` subject to g in simplex. This is finding the point in the simplex that minimizes the maximum of a subset of bilinear forms. This is a valid Frank-Wolfe direction for the problem `min_g max_k (g*g)[k]` (which is our problem) because:

At the current point y, a valid subgradient of f(g) = max_k (g*g)[k] is `2 * sum_{k in argmax} w_k * y[k-.]` (the `. ` means the appropriate indexing). The FW direction is `min_{g in simplex} <subgrad, g> = min_{g in simplex} 2 * sum_k w_k * sum_i y[k-i]*g[i] = min_{g in simplex} 2 * sum_k w_k * (y*g)[k]`.

But the LP is solving `min_g max_{k in S} (y*g)[k]`, which is different from `min_g sum_k w_k * (y*g)[k]`. The LP gives a direction that's good for the max over S, while the FW direction is a weighted average.

def smooth_mirror_descent(a0, time_limit, beta_init=100.0, beta_final=10.0,

    """Mirror descent on the simplex with log-sum-exp smoothing."""
    n = len(a0)

    best_m = fast_convolve(a).max()

        beta = beta_init * (beta_final / beta_init) ** (progress ** 0.5)

        y = np.maximum(y, 1e-15)

            y = x.copy()

            y = y / s_y

        grad = get_smoothed_gradient(y, ac, beta)

        eta = gamma / (iters ** 0.5)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x = a_new
        
        # Check improvement
        M_new = fast_convolve(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

            stall = 0
            noise = 0.03 / (1.0 + 0.2 * (iters // max_stall))
            pert = rng.normal(0, noise, n) * best_a
            x = np.maximum(best_a + pert, 1e-10)
            s_x = x.sum()
            if s_x > 1e-15:

            vel = np.zeros(n)
            gamma = max(gamma * 0.85, 0.5)
            momentum = max(momentum * 0.95, 0.1)
            if verbose:

def lp_fw(a0, time_limit, solver, initial_K=300, max_K=3000, verbose=True):

    iters = 0
    
    while time.monotonic() - t0 < time_limit:

            # Keep only the most binding

            else:
                active_set = new_set
        
        K = len(active_set)
        if K == 0:
            break
        
        # Build constraint matrix
        active_list = sorted(active_set)
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(active_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_sum = g_val.sum()

        Yf = np.fft.rfft(y, fft_size)

        if len(violations) > 0:

            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                active_set.add(int(k))

        default_alpha = 2.0 / (iters + 2.0)
        alphas = [default_alpha, default_alpha * 0.5, default_alpha * 2.0, 0.05, 0.15, 0.3]
        alphas = [max(0.01, min(1.0, a)) for a in alphas]
        
        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:

                best_alpha = alpha

            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = y.copy()
            else:
                y = y / s_y

def coordinate_descent(a0, time_limit, verbose=True):

    n = len(a0)

    best_a = a.copy()

    max_stall = 2000
    iters = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iters += 1
        elapsed = time.monotonic() - t0
        
        ac = fast_convolve(a)
        # Find indices contributing to the max
        top_k = np.argsort(ac)[::-1][:30]

        improved = False
        
        for k in top_k[:15]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Source: index with max contribution to this peak
            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])
            src_val = a[src]
            
            if src_val < 1e-12:
                continue
            
            # Find destination: index that contributes least to all top peaks
            # and is not symmetric to src (to avoid trivial moves)
            src_sym = n - 1 - src

                s2 = max(0, k2i - n + 1)

                if s2 <= e2:

                    scores[idx2] += a[k2i - idx2]

            candidates = np.argsort(scores)[:min(n//3, 200)]

                    amt = src_val * frac
                    if amt < 1e-15:
                        continue

                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_convolve(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 100 == 0:
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True
                                break
        
        if verbose and stall % 500 == 0:
            print(f"    CD it={iters} ({elapsed:.0f}s): J={2.0*n*best_m:.6f}, stall={stall}")
    
    return norm_sum1(best_a), best_m

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    solver = init_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    # Get initial sequence
    best_seq = None
    best_score = float('inf')
    
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = norm_sum1(p)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    # Symmetrize
    best_seq = symmetrize(norm_sum1(best_seq))
    best_score = evaluate_safe(best_seq)
    print(f"After symmetrize: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"*** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
    
    # Phase 1: Quick parametric search
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Parametric (n={n0}) ===")
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 5:
        x = np.linspace(0.0, 1.0, n0)
        i_arr = np.arange(n0, dtype=np.float64)
        candidates = []
        for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            candidates.append(symmetrize(norm_sum1(i_arr ** p * (n0 - 1 - i_arr) ** p)))
        for c in [0.1, 0.2, 0.3, 0.4]:
            sigma = 0.15 + 0.1 * c
            candidates.append(symmetrize(norm_sum1(np.exp(-((x - 0.5)**2) / (2*sigma*sigma)) + c * x*(1-x))))
        for a in candidates:
            if time.monotonic() - t_global > time_budget - remaining * 0.97:
                break
            v = compute_obj(a)
            if v < best_score:
                best_score = v
                best_seq = a.copy()
                print(f"  Param: J={v:.6f}")
    
    # Phase 2: Smoothed Mirror Descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: SMD (n={n_best}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30:
        a_out, _ = smooth_mirror_descent(norm_sum1(best_seq), t2,
                                          beta_init=200.0, beta_final=20.0,
                                          c0=3.5, momentum=0.5, verbose=True)
        try_update(a_out, "smd1")
    
    # Phase 3: LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30 and n_best <= 10000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t3, solver,
                         initial_K=400, max_K=3500, verbose=True)
        try_update(a_out, "lpfw1")
    
    # Phase 4: More SMD with refined parameters
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: SMD refine (n={n_best}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out, _ = smooth_mirror_descent(norm_sum1(best_seq), t4,
                                          beta_init=50.0, beta_final=10.0,
                                          c0=2.5, momentum=0.4, verbose=True)
        try_update(a_out, "smd2")
    
    # Phase 5: Try different n
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    try_ns = [n_best - 500, n_best + 500, n_best + 1000, n_best - 1000]
    try_ns = [l for l in try_ns if 3000 <= l <= 12000]
    print(f"\n=== Phase 5: Multi-n exploration ===")
    for new_n in try_ns[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_seq(best_seq, new_n)
        a_long = symmetrize(a_long)
        
        t5a = min(remaining * 0.2, 150.0)
        if t5a > 30:
            a_out, _ = smooth_mirror_descent(a_long, min(t5a, 100)
