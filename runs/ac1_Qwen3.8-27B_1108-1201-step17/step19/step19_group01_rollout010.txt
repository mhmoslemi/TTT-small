def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 200:
                # Random exploration
                j = int(rng.integers(n))
                if j != reduce_idx and a[reduce_idx] > 1e-12:
                    frac = rng.uniform(0.002, 0.015)
                    amt = a[reduce_idx] * frac
                    a[reduce_idx] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        m_new = fast_autocorr(a).max()
                        if m_new < best_m:
                            best_m = m_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            # Periodic random perturbation
            if stall % 50 == 0:
                noise_scale = min(0.03, 0.005 * stall / 50)
                pert = rng.normal(0, noise_scale, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-12)
                a_r = a_r / a_r.sum()
                m_r = fast_autocorr(a_r).max()
                if m_r < best_m:
                    best_m = m_r
                    best_a = a_r.copy()
                a = a_r if m_r < best_m + 0.01 else best_a.copy()

            if verbose and stall % 200 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    PeakShave (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " stall=" + str(stall))

    return best_a, best_m


def subgradient_optimum(a0, time_limit, beta_start=20.0, beta_end=800.0,
                        beta_growth=1.005, momentum=0.9, verbose=True):
    """Subgradient method with heavy-ball momentum on log-sum-exp smoothed objective."""
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    v = np.zeros(n)  # velocity for heavy-ball
    stall = 0
    rng = np.random.default_rng(42)
    top_k_ratio = 0.04
    top_k = max(30, int(n * top_k_ratio))
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(x)
        # Use top-k peaks for smoothing
        idx_top = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros_like(b)
        b_use[idx_top] = b[idx_top]
        b_max = b[idx_top].max()

        shifted = np.clip(beta * (b_use - b_max), -300, 300)
        exp_vals = np.exp(shifted)
        p = exp_vals / exp_vals.sum()

        # Gradient: sum_k p_k * (a * reversed partial)
        # grad_i = (1/beta) * sum_k p_k * sum_j a[j]*a[k-j] * (delta_{i,j} + delta_{i,k-j})
        # Simplified: grad = (1/beta) * (p-fold convolution with a, restricted)
        conv_len = n + len(p) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        a_rev = x[::-1]
        Ar = np.fft.rfft(a_rev, fft_size)
        Pr = np.fft.rfft(p, fft_size)
        conv = np.fft.irfft(Ar * Pr, fft_size)
        grad = conv[n - 1:2 * n - 1]
        # For self-convolution, gradient of (a*a)[k] w.r.t a[i] is a[k-i] + a[i] (if both valid)
        # Use the FFT-based approximation
        grad = 2.0 * grad / beta

        # Adaptive step size
        eta = 1.0 / (np.linalg.norm(grad) + 1e-10)
        eta = min(eta * 0.5, 0.05)
        eta = max(eta, 1e-6)

        # Heavy-ball update
        v = momentum * v - eta * grad
        x_new = x + v
        x_new = simplex_project(x_new)

        # Check improvement
        b_new = fast_autocorr(x_new)
        m_new = b_new.max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
            x_new = x  # Revert if not improving

        x = x_new

        # Adaptive beta
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)
            momentum = 0.9 * (1.0 - 30.0 / (t + 10.0))
            momentum = min(0.95, max(0.5, momentum))

        # Restart
        if stall > 1500:
            stall = 0
            noise = 0.02 / (1.0 + 0.2 * (t // 1500))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            v = np.zeros(n)
            beta = max(beta * 0.6, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    SubGrad restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SubGrad t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

    return best_a, best_m


def block_coordinate_descent(a0, time_limit, block_size=100, verbose=True):
    """Optimize blocks of consecutive elements while keeping others fixed."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(333)
    n_blocks = (n + block_size - 1) // block_size

    for iteration in range(2000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        m_curr = b.max()

        # Find the block with the most "pressure"
        # Compute pressure per position, then aggregate per block
        top_peaks = np.argsort(b)[::-1][:20]
        pressure = np.zeros(n)
        for k_i in top_peaks[:10]:
            k_int = int(k_i)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                pressure[idx] += a[k_int - idx]
                pressure[k_int - idx] += a[idx]

        # Pick the block with highest total pressure
        block_pressures = np.zeros(n_blocks)
        for bi in range(n_blocks):
            s = bi * block_size
            e = min((bi + 1) * block_size, n)
            block_pressures[bi] = pressure[s:e].sum()

        bi = int(np.argmax(block_pressures))
        s = bi * block_size
        e = min((bi + 1) * block_size, n)

        # Optimize this block: try to redistribute mass within the block
        # to reduce the top peaks
        a_block = a[s:e].copy()
        block_sum = a_block.sum()

        # Compute which positions in this block contribute to top peaks
        for k_i in top_peaks[:8]:
            k_int = int(k_i)
            i_start = max(s, k_int - n + 1)
            i_end = min(e - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            src_local = int(idx[np.argmax(contrib)] - s)
            reduce_global = int(idx[np.argmax(contrib)])

            # Find the least-contributing position in the same block to boost
            all_local = np.arange(s, e)
            other_contrib = np.zeros(e - s)
            for pos in all_local:
                total_c = 0.0
                for k2 in top_peaks[:5]:
                    k2i = int(k2)
                    partner = k2i - int(pos)
                    if 0 <= partner < n:
                        total_c += a[int(pos)] * a[partner]
                other_contrib[pos - s] = total_c

            # Boost the position with lowest contribution (and not the reduce position)
            other_contrib[src_local] = np.inf
            dst_local = int(np.argmin(other_contrib))
            dst_global = s + dst_local

            if dst_global == reduce_global:
                continue

            # Move mass
            if a[reduce_global] < 1e-14:
                continue
            for frac in [0.02, 0.05, 0.1, 0.2]:
                amt = a[reduce_global] * frac
                a_trial = a.copy()
                a_trial[reduce_global] -= amt
                a_trial[dst_global] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    break

        # Occasionally shuffle block order
        if iteration % 10 == 0:
            pass  # Already picking max-pressure block

        if verbose and iteration % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    BlockCD iter=" + str(iteration) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

    return best_a, best_m


def lp_refinement(a0, time_limit, solver, verbose=True):
    """LP-Frank-Wolfe for the active set of top constraints."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    b = fast_autocorr(a)
    active_set = set(int(k) for k in np.argsort(b)[::-1][:800])
    no_improve = 0
    t = 0

    while time.monotonic() - t0 < time_limit and no_improve < 800:
        t += 1
        b = fast_autocorr(a)
        M_curr = b.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update active set: keep top constraints + violations
        constraint_list = sorted(active_set)
        if len(constraint_list) > 2500:
            margins = np.array([M_curr - b[k] for k in constraint_list])
            keep = np.argsort(margins)[:2500]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = a[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Af = np.fft.rfft(a, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ag = np.fft.irrift(Af * Gf, fft_size)[:size] if False else np.fft.irfft(Af * Gf, fft_size)[:size]

        violations = np.where(conv_ag > M_curr + 1e-6)[0]
        if len(violations) > 0:
            viol_amt = conv_ag[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:40]]
            for k in top_v:
                active_set.add(int(k))

        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5), 1.0 / (t + 1)]
        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * a + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        a_new = (1.0 - best_alpha) * a + best_alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        st = a_new.sum()
        if st > 1e-15:
            a_new = a_new / st
        a = a_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def parametric_sweep(n, time_limit, best_init, best_score):
    """Sweep through parametric families to find good initialization."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score

    # Asymmetric power laws
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
        for beta_p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
            if time.monotonic() - t0 > time_limit:
                break
            a_raw = (x + 0.01) ** alpha * (1.0 - x + 0.01) ** beta_p
            a = norm_sum1(a_raw)
            m = fast_autocorr(a).max()
            if m < best_val:
                best_val = m
                best_a = a.copy()

    # Gaussian mixtures
    for c1 in [0.2, 0.3, 0.4, 0.5]:
        for c2 in [0.5, 0.6, 0.7, 0.8]:
            for s1 in [0.05, 0.1, 0.15]:
                for s2 in [0.05, 0.1, 0.15]:
                    for w1 in [0.5, 1.0]:
                        if time.monotonic() - t0 > time_limit:
                            break
                        a_raw = np.exp(-((x - c1) ** 2) / (2 * s1 ** 2)) + w1 * np.exp(-((x - c2) ** 2) / (2 * s2 ** 2))
                        a = norm_sum1(a_raw)
                        m = fast_autocorr(a).max()
                        if m < best_val:
                            best_val = m
                            best_a = a.copy()

    # Polynomials
    for p in [2, 3, 4, 5]:
        a_raw = 1.0 + x ** p
        a = norm_sum1(a_raw)
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    # Random starts
    rng = np.random.default_rng(123)
    for _ in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(rng.random(n) ** rng.uniform(0.3, 5.0))
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    return best_a, best_val


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search: multi-phase optimization with peak-shaving, subgradient, block CD, and LP."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 7000
        x = np.linspace(0, 1, n_base)
        a_raw = (x + 0.01) ** 1.2 * (1.0 - x + 0.01) ** 0.8
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: Parametric sweep at parent's n and nearby
    print("\n=== Phase 1: Parametric sweep ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 50.0)
    if t1 > 10:
        a_param, _ = parametric_sweep(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "sweep_n" + str(n0), t_global)

    # Phase 2: Peak shaving (aggressive)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Peak shaving (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.12, 200.0)
    if t2 > 30:
        a_out, _ = peak_shaving(best_seq, t2, top_k=25, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peaksav_n" + str(n_best), t_global)

    # Phase 3: Subgradient with momentum
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Subgradient (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 60:
        a_out, _ = subgradient_optimum(best_seq, t3, beta_start=15.0, beta_end=600.0,
                                       beta_growth=1.005, momentum=0.9, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "subgrad_n" + str(n_best), t_global)

    # Phase 4: Peak shaving round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Peak shaving 2 (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.10, 160.0)
    if t4 > 20:
        a_out, _ = peak_shaving(best_seq, t4, top_k=20, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peaksav2_n" + str(n_best), t_global)

    # Phase 5: Block coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Block CD (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 30:
        a_out, _ = block_coordinate_descent(best_seq, t5, block_size=80, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "blockcd_n" + str(n_best), t_global)

    # Phase 6: LP refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: LP (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.15, 250.0)
    if t6 > 40 and n_best <= 8000:
        a_out, _ = lp_refinement(best_seq, t6, solver, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(n_best), t_global)

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    nearby_lengths = []
    for dn in [-500, -200, 200, 500, 800, 1200]:
        nl = n_best + dn
        if 4000 < nl <= 9000 and nl not in nearby_lengths:
            nearby_lengths.append(nl)

    for new_n in nearby_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print("\n=== Phase 7: n=" + str(new_n) + " ===")
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick peak shaving + subgradient
        t7a = min(remaining * 0.25, 150.0)
        if t7a > 20:
            a_out, _ = peak_shaving(a_long, min(t7a * 0.4, 60.0), top_k=20, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ps_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        t7b = min(remaining * 0.4, 200.0)
        if t7b > 40:
            a_out, _ = subgradient_optimum(best_seq, t7b, beta_start=20.0, beta_end=500.0,
                                           beta_growth=1.005, momentum=0.9, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "sg_n" + str(new_n), t_global)

    # Phase 8: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final polish (n=" + str(n_best) + ") ===")
    if remaining > 60:
        t8a = min(remaining * 0.4, 150.0)
        if t8a > 20:
            a_out, _ = peak_shaving(best_seq, t8a, top_k=15, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_ps", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        t8b = min(remaining * 0.4, 100.0)
        if t8b > 20:
            a_out, _ = subgradient_optimum(best_seq, t8b, beta_start=50.0, beta_end=1000.0,
                                           beta_growth=1.006, momentum=0.92, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_sg", t_global)

    # Phase 9: Final LP burst
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 50 and n_best <= 8000:
        print("\n=== Phase 9: Final LP burst (n=" + str(n_best) + ") ===")
        t9 = min(remaining * 0.7, 150.0)
        if t9 > 30:
            a_out, _ = lp_refinement(best_seq, t9, solver, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_lp", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 7000
        x = np.linspace(0, 1, n_fb)
        a_raw = (x + 0.01) ** 1.0 * (1.0 - x + 0.01) ** 1.0
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
