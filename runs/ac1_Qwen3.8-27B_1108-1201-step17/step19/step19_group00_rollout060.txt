def softmax_grad_descent(a0, time_limit, verbose=True):
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    
    # Adaptive parameters
    top_k = 50
    start_temp = 0.01
    decay_rate = 0.99999
    base_lr = 0.5
    
    x_prev = x.copy()
    stall = 0
    max_stall = 5000
    
    temp = start_temp
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        if stall > max_stall:
            break
        
        t += 1
        # Nesterov-like momentum
        beta = 0.9
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute autocorrelation
        ac = fast_autocorr(y)
        # Only consider lags 0 to n-1
        ac = ac[:n]
        
        # Soft-max weights over top lags
        k_max = min(top_k, n)
        idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[idx]
        
        # Adaptive temperature: decrease over time
        temp = max(1e-6, start_temp * (decay_rate ** t))
        
        w = np.exp(top_vals / temp - top_vals[0] / temp)
        w = w / w.sum()
        
        # Compute gradient
        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            if ki == 0:
                grad += 2.0 * wi * y
            else:
                # a_{j+k} for j+k < n
                grad[:n-ki] += wi * y[ki:]
                # a_{j-k} for j >= k
                grad[ki:] += wi * y[:n-ki]
        
        # Multiplicative update (mirror descent)
        lr = base_lr / (np.sqrt(t) + 1.0)
        log_x = np.log(np.maximum(y, 1e-30)) - lr * grad
        max_log = log_x.max()
        x_new = np.exp(log_x - max_log)
        x_new = np.maximum(x_new, 1e-30)
        x_new = x_new / x_new.sum()
        
        x_prev = x
        x = x_new
        
        # Evaluate
        val = obj_val(x)
        if val < best_val - 1e-12:
            best_val = val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > 2000 and stall % 2000 == 0:
            # Increase top_k and adjust temperature
            top_k = min(top_k + 20, 200)
            temp = max(start_temp * 0.5, temp * 0.3)
            base_lr = max(0.1, base_lr * 0.8)
            
            # Add small noise
            noise = 0.01 * np.random.default_rng(t).standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            stall = 0
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    SM-GD t=" + str(t) + " (" + str(round(elapsed,1)) + "s) J=" + repr(best_val) + " temp=" + repr(temp))
    
    return best_a, best_val

def fast_autocorr(a):

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n

    old_x = np.linspace(0.0, 1.0, old_n)

    ss = f_interp.sum()

        x = x_new

        val = obj_val(x)
        if val < best_val - 1e-13:
            best_val = val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive schedule
        if stall > 1000:
            # Cool down
            temp = max(min_temp, temp * 0.7)
            base_lr = max(0.05, base_lr * 0.7)
            top_k = min(top_k + 10, 200)
            if stall > 5000:
                # Restart with noise
                noise = 0.02 * rng.standard_normal(n) * best_a
                x_r = best_a + noise
                x_r = np.maximum(x_r, 1e-12)
                x_r = x_r / x_r.sum()
                x = x_r
                x_prev = x_r.copy()
                temp = max(temp * 2.0, min_temp * 5)
                base_lr = max(base_lr * 1.5, 0.1)
                stall = 0

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    SM-MD t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(best_val) + " temp=" + repr(temp) + " topk=" + str(top_k))

    return best_a, best_val

def build_lp_matrix(a, clist, n):
    K = len(clist)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(clist):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]
    return coeffs

def lp_fw_improved(a0, time_limit, solver, init_K=300, max_K=2500, verbose=True):
    """LP-Frank-Wolfe with adaptive constraint growth and better line search."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:init_K])
    no_improve = 0
    stall_limit = 2000
    add_per = 25

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        if no_improve > stall_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        clist = sorted(active_set)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = sorted(active_set)

        K = len(clist)
        if K == 0:
            break

        # Solve LP
        coeffs = build_lp_matrix(y, clist, n)
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        lp_ok = False
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                lp_ok = True
        except Exception:
            if len(active_set) > init_K // 2:
                active_set = set(int(k) for k in clist[:init_K // 2])
                clist = sorted(active_set)

        if not lp_ok:
            if no_improve > stall_limit // 3:
                break
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations and add constraints
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per]]
            for k in top_v:
                active_set.add(int(k))

        # Golden section line search
        if t <= 3:
            y = g_dir.copy()
            continue

        lo, hi = 0.0, 1.0
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)

        def trial_M(alpha):
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                return 1e10
            trial = trial / ts
            return fast_autocorr(trial).max()

        fc = trial_M(c)
        fd = trial_M(d)
        for _ in range(12):
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - gr * (hi - lo)
                fc = trial_M(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = trial_M(d)

        best_alpha = (lo + hi) / 2.0
        y_new = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy
        else:
            y = y.copy()

        if verbose and t % 30 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(2 * n * best_m) + " |S|=" + str(len(active_set)))

    elapsed = time.monotonic() - t0
    if verbose:
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s) J=" + repr(2 * n * best_m))
    return best_a, 2.0 * n * best_m

def coordinate_descent_intensive(a0, time_limit, seed=123, verbose=True):
    """Intensive coordinate descent with multiple strategies."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200

    fracs = [0.002, 0.004, 0.006, 0.01, 0.015, 0.02, 0.03, 0.05, 0.08]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        # Strategy 1: Move mass from peak contributor to low-score position
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score: how much would adding mass at j increase top autocorrs?
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)
            dst = int(candidates[np.argmin(scores[candidates])])
            if dst == src:
                continue

            found = False
            for frac in fracs:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t / st
                Mt = fast_autocorr(a_t).max()
                if Mt < best_m - 1e-16:
                    a = a_t
                    best_m = Mt
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if found:
                continue

        # Strategy 2: Pairwise swap (preserve total mass more naturally)
        if not improved and stall > 30:
            for _ in range(20):
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or abs(a[i] - a[j]) < 1e-14:
                    continue
                swap_amt = min(a[i], a[j]) * rng.uniform(0.1, 0.5)
                a_t = a.copy()
                a_t[i] -= swap_amt
                a_t[j] -= (a_t[j] - a[j])  # keep track
                a_t[i] = a[i] - swap_amt
                a_t[j] = a[j] + swap_amt
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st > 1e-15:
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-16:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

        # Strategy 3: Small random perturbation
        if not improved:
            stall += 1
            if stall % 50 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 500 == 0:
            print("    Coord (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(2 * n * best_m) + " stall=" + str(stall))

    return best_a, 2.0 * n * best_m

def deflation_refined(a0, time_limit, n_targets=8, verbose=True):
    """Refined deflation targeting top peaks with adaptive step size."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for round_i in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        targets = np.argsort(ac)[::-1][:n_targets]
        improved = False

        for k in targets[:5]:
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]

            for _ in range(5):
                if contrib.max() < 1e-18:
                    break
                src = int(idx[np.argmax(contrib)])
                src_val = a[src]
                if src_val < 1e-12:
                    break

                # Find best destination
                scores = np.zeros(n)
                for k2 in targets[:6]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                        scores[k2i - idx2] += a[idx2]
                scores[src] = 1e10

                # Try multiple destinations
                top_candidates = np.argsort(scores)[:20]
                found_move = False
                for dst in top_candidates:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.003, 0.005, 0.008, 0.012, 0.02]:
                        amt = src_val * frac
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < best_m - 1e-16:
                            a = a_t
                            best_m = Mt
                            best_a = a.copy()
                            improved = True
                            found_move = True
                            contrib = a[idx] * a[ki - idx]
                            break
                    if found_move:
                        break
                if not found_move:
                    break

        if not improved:
            # Random kick
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.001, 0.008)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    Mn = fast_autocorr(a).max()
                    if Mn < best_m:
                        best_m = Mn
                        best_a = a.copy()

        if verbose and round_i % 100 == 0:
            print("    Deflate r=" + str(round_i) + " (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(2 * n * best_m))

    return best_a, 2.0 * n * best_m

def grad_descent_fft(a0, time_limit, verbose=True, seed=99):
    """Full gradient descent using FFT for gradient computation."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Use FFT-based gradient: grad = 2n * (W * A + W_rev * A) in frequency domain
    # where W = FFT of soft-max weights
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    top_k = 80
    temp = 0.02 / n
    lr = 0.3
    momentum_buf = np.zeros(n, dtype=np.float64)
    beta1, beta2 = 0.9, 0.999
    t = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if stall > 8000:
            break
        t += 1

        ac = fast_autocorr(x)[:n]
        k_max = min(top_k, n)
        idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[idx]
        max_top = top_vals[0]

        # Soft-max weights
        w = np.exp((top_vals - max_top) / temp)
        w = w / w.sum()

        # Build weight vector of length n
        w_full = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            w_full[int(k)] = wi

        # FFT-based gradient
        Wf = np.fft.rfft(w_full, fft_size)
        Xf = np.fft.rfft(x, fft_size)

        # grad_j = 2n * sum_k w_k * (x_{j+k} + x_{j-k})
        # First term: sum_k w_k * x_{j+k} = (w_full * x)_j (cross-corr)
        # In FFT: ifft(Wf * conj(Xf)) gives cross-correlation
        # Actually: conv(w, x) in time = ifft(Wf * Xf)
        # We want: h_j = sum_k w_k * x_{j+k} which is cross-correlation
        # cross_corr = ifft(conj(Wf) * Xf) ... let me just use the direct approach

        # Direct approach using the top-k lags (faster for small k_max)
        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            if ki == 0:
                grad += 2.0 * wi * x
            else:
                grad[:n - ki] += wi * x[ki:]
                grad[ki:] += wi * x[:n - ki]

        # Adam-like update
        momentum_buf = beta1 * momentum_buf + (1 - beta1) * grad
        # Skip second moment for simplicity (multiplicative update handles scale)

        lr_t = lr / (1.0 + 0.05 * np.sqrt(t))
        log_x = np.log(np.maximum(x, 1e-30)) - lr_t * momentum_buf
        max_log = log_x.max()
        x_new = np.exp(log_x - max_log)
        x_new = np.maximum(x_new, 1e-30)
        x_new = x_new / x_new.sum()

        x = x_new
        val = obj_val(x)
        if val < best_val - 1e-13:
            best_val = val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adapt
        if stall > 3000:
            top_k = min(top_k + 20, 250)
            temp = max(temp * 0.5, 1e-5 / n)
            lr = max(lr * 0.7, 0.02)
            if stall > 6000:
                noise = 0.01 * rng.standard_normal(n) * best_a
                x_r = best_a + noise
                x_r = np.maximum(x_r, 1e-12)
                x_r = x_r / x_r.sum()
                x = x_r
                stall = 0

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print("    GradFFT t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(best_val))

    return best_a, best_val

def propose_candidate(budget_s=1800.0):
