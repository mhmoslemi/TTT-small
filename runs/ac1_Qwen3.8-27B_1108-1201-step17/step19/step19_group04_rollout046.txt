            # Multiplicative update
            move = -step * g
            move = np.clip(move, -2.0, 2.0)
            a_new = a * np.exp(move)
            a_new = normalize_vec(a_new)
            
            # Check improvement
            q_new = self_conv(a_new)
            score_new = 2.0 * n * float(q_new.max())
            
            if score_new < score - 1e-15:
                a = a_new
                step_base = min(step_base * 1.05, 2.0)
            else:
                # Try smaller step
                step_base = max(step_base * 0.6, 0.001)
            
            prev_grad = g
            prev_a = a
            total_iters += 1
        
        if verbose:
            print("  Continuation beta=%.0f: best_score=%.9f iters=%d t=%.1fs"
                  % (beta, best_score, total_iters, time.monotonic() - t0))
    
    # Final exact evaluation
    q = self_conv(best)
    final_score = 2.0 * n * float(q.max())
    if final_score < best_score:
        best_score = final_score
    
    return best

# ---------- LP-based direction finding ----------

def solve_lp_direction(y, active_lags, q_active):
    """
    Solve LP to find direction g:
    minimize T
    s.t. sum(g) = 1, g >= 0
         2*(y*g)[k] - T <= q_active[k] for k in active_lags
    """
    if not HAVE_LP:
        return None
    y = np.asarray(y, dtype=np.float64)
    n = y.size
    K = active_lags.size
    if K < 2 or n > 15000:
        return None
    if K * n > 8_000_000:
        return None
    
    try:
        # Build constraint matrix A (K x (n+1))
        A = np.zeros((K, n + 1), dtype=np.float64)
        for r, k in enumerate(active_lags):
            k = int(k)
            i0 = max(0, k - n + 1)
            i1 = min(n, k + 1)
            if i0 < i1:
                A[r, i0:i1] = 2.0 * y[i1-1-k+i0+1 : i1+1 : -1][:i1-i0]
                # Actually: A[r, j] for j in [i0, i1) should be 2*y[k-j]
                A[r, i0:i1] = 0.0
                for j in range(i0, i1):
                    A[r, j] = 2.0 * y[k - j]
        A[:, n] = -1.0
    except Exception:
        return None
    
    b = np.maximum(q_active, 0.0).copy()
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    Aeq = np.ones((1, n))
    bounds = [(0.0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=[1.0],
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    
    g = res.x[:n].copy()
    g = np.maximum(g, 0.0)
    s = g.sum()
    if s < 1e-10:
        return None
    g /= s
    return g
