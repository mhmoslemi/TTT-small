    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_matvec(w, v):
    """Compute H(w) @ v where H(w)_{ij} = w_{i+j}, w has length 2n-1, v has length n."""
    n = len(v)
    v_rev = v[::-1]
    size = (2*n - 1) + n - 1  # 3n - 2
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    V = np.fft.rfft(v_rev, fft_size)
    c = np.fft.irfft(W * V, fft_size)
    return c[n-1 : n-1+n]

    """Compute gradient of smooth max objective F_beta(a) = (2/beta)*log(sum exp(beta*C_k/n))."""
    n = len(a)
    C = fast_convolve(a)
    # Softmax weights
    scaled = beta * C / n
    scaled -= scaled.max()

    w = exp_scaled / exp_scaled.sum()
    # Gradient: (4/n) * H(w) @ a
    grad = (4.0 / n) * hankel_matvec(w, a)
    return grad, C, w, C.max()
