```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Power-law initialization (key insight: a_i ~ (i+1)^p gives score ~ 2(p+1)^2(p!)^2/(2p+1)! for large n)
2. FFT-based mirror descent with adaptive soft-max gradients on top autocorrelation lags
3. Spectral flattening in frequency domain with non-negativity enforcement
4. Coordinate descent with impact-aware mass transfer
5. Length exploration and multi-restart strategy
"""

import numpy as np
import time


def fast_conv(a):
    """Compute full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely call the evaluation function."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def score_from_conv(a):
    """Compute score from a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_conv(a)
    return float(2.0 * n * conv.max())


def power_law_init(n, p, reverse=False):
    """Create a power-law sequence."""
    i = np.arange(n, dtype=np.float64) + 1.0
    if reverse:
        i = n + 1.0 - i
    a = i ** p
    return norm_sum1(a)


def parametric_search(n, time_limit=10.0):
    """Search over parametric families for best initialization."""
    t0 = time.monotonic()
    best_a = None
    best_score = float('inf')
    
    # Power laws (key family)
    for p in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0, 4.5, 5.0, 6.0, 8.0, 10.0]:
        if time.monotonic() - t0 > time_limit:
            break
        a = power_law_init(n, p)
        s = score_from_conv(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()
    
    # Reversed power laws
    for p in [1.0, 2.0, 3.0, 4.0]:
        if time.monotonic() - t0 > time_limit:
            break
        a = power_law_init(n, p, reverse=True)
        s = score_from_conv(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()
    
    # Bimodal power laws
    x = np.linspace(0, 1, n)
    for p in [1.0, 2.0, 3.0]:
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(x**p + (1-x)**p)
        s = score_from_conv(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()
    
    # Uniform
    a = np.ones(n) / n
    s = score_from_conv(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()
    
    return best_a if best_a is not None else np.ones(n) / n, best_score


def compute_gradient(a, conv, top_k, beta):
    """

    grad[i] = 2 * sum_k w_k * sum_{j: i+j=k} a[j]
    """
    n = len(a)
    k_alarm = min(top_k, len(conv))
    top_idx = np.argsort(conv)[::-1][:k_alarm]
    top_vals = conv[top_idx]
    
    max_v = top_vals.max()
    if max_v < 1e-30:
        return np.zeros(n)
    
    expo = beta * (top_vals - max_v) / max_v
    expo = np.clip(expo, -50, 0)
    weights = np.exp(expo)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_alarm) / k_alarm
    else:
        weights = weights / w_sum
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        # conv[k] = sum_i a[i]*a[k-i], valid i: max(0,k-n+1) <= i <= min(n-1,k)
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n - 1, k_int)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        j_idx = k_int - idx
        # d(conv[k])/d(a[i]) = a[k-i] + a[i-k] (but for linear conv, it's a[k-i])
        # Actually for conv(a,a)[k] = sum_i a[i]*a[k-i]:
        # d/d(a[m]) = a[k-m] if 0<=k-m<n, plus a[m-k] if 0<=m-k<n (but these are the same term)
        # For linear convolution: d(conv[k])/d(a[m]) = a[k-m] (when valid)
        # But since conv is symmetric, d(conv[k])/d(a[m]) = a[k-m] + a[m-k]... 
        # No, for linear conv of a with itself:
        # conv[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i]*a[k-i]
        # d(conv[k])/d(a[m]) = a[k-m] (if 0<=k-m<n) + a[m-k] (if 0<=m-k<n)
        # But a[k-m] and a[m-k] are the same only if k-m = m-k, i.e., k=2m.
        # Generally: d(conv[k])/d(a[m]) = a[k-m] (if 0<=k-m<n)
        # Wait, conv[k] = sum_i a[i]*a[k-i]. If m is one of the i's, then term is a[m]*a[k-m].
        # If m is one of the (k-i)'s, then i=k-m, term is a[k-m]*a[m].
        # These are the same term! So d(conv[k])/d(a[m]) = a[k-m] (if valid) + a[m-k] (if valid).
        # But a[k-m] is valid when 0<=k-m<n, and a[m-k] is valid when 0<=m-k<n.
        # These are different conditions (unless k=2m).
        # So d(conv[k])/d(a[m]) = a[k-m]*1_{0<=k-m<n} + a[m-k]*1_{0<=m-k<n}
        
        # For our purposes, since we're just computing the gradient for descent:
        # The contribution to grad[m] from lag k is: a[k-m] (if 0<=k-m<n)
        # (we can ignore the second term as it's a mirror image and will be captured when
        #  we process the symmetric lag)
        
        valid = (j_idx >= 0) & (j_idx < n)
        grad[idx[valid]] += 2.0 * w * a[j_idx[valid]]
        # Also add the "other side" contribution
        valid2 = (idx >= 0) & (idx < n)
        # a[m-k] for m in idx: this is a[idx - k]
        alt_j = idx - k_int
        valid_alt = (alt_j >= 0) & (alt_j < n)
        if np.any(valid_alt):
            grad[idx[valid_alt]] += 2.0 * w * a[alt_j[valid_alt]]
    
    return grad


def compute_gradient_v2(a, conv, top_k, beta):
    """
    More efficient gradient computation using the structure of convolution.
    conv[k] = (a * a)[k] where * denotes linear convolution.
    d(conv[k])/d(a[m]) = a[k-m] + a[m-k] (with appropriate bounds)
    """
    n = len(a)
    k_alarm = min(top_k, len(conv))
    top_idx = np.argsort(conv)[::-1][:k_alarm]
    top_vals = conv[top_idx]
    
    max_v = top_vals.max()
    if max_v < 1e-30:
        return np.zeros(n)
    
    expo = beta * (top_vals - max_v) / max_v
    expo = np.clip(expo, -50, 0)
    weights = np.exp(expo)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_alarm) / k_alarm
    else:
        weights = weights / w_sum
    
    grad = np.zeros(n)
    
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        # For conv[k] = sum_i a[i] a[k-i]:
        # d/d(a[m]) = a[k-m] if 0 <= k-m < n (m appears as first index)
        #           + a[m-k] if 0 <= m-k < n (m appears as second index, i.e., k-i=m, i=k-m)
        # First term: m = i, need 0<=i<n and 0<=k-i<n => k-n+1<=i<=k and 0<=i<=n-1
        #   So i in [max(0,k-n+1), min(n-1,k)]
        #   Contribution to grad[i] += a[k-i]
        # Second term: m = k-i, need 0<=m<n and 0<=i<n => 0<=k-i<n => k-n+1<=i<=k
        #   And 0<=i<=n-1
        #   So i in [max(0,k-n+1), min(n-1,k)]
        #   m = k-i, contribution to grad[k-i] += a[i]
        # But these give the same set of (i, k-i) pairs! The first gives grad[i] += a[k-i]
        # and the second gives grad[k-i] += a[i]. These are the same: for pair (i, k-i),
        # we add a[k-i] to grad[i] and a[i] to grad[k-i].
        # So effectively: for each valid i, grad[i] += a[k-i] and grad[k-i] += a[i].
        # Which means grad[i] += a[k-i] (from first) and grad[i] += a[i-k] (from second, when
        # k-j = i, j = k-i, so grad[i] += a[k-i]... wait I'm confusing myself.
        
        # Let me just be direct: conv[k] = sum_{i} a[i]*a[k-i] (implied bounds)
        # d(conv[k])/d(a[m]) = sum_i [delta_{i,m}*a[k-i] + delta_{k-i,m}*a[i]]
        # = a[k-m] (if m in valid range for i) + a[k-m] (if k-m in valid range for i)
        # Wait: delta_{k-i,m} means i=k-m, so the term is a[k-m]*a[m] (when i=k-m is valid)
        # So d(conv[k])/d(a[m]) = a[k-m]*1_{valid} + a[m]*1_{k-m valid as i}
        # Hmm, let me just use: d(conv[k])/d(a[m]) = a[k-m] (if 0<=k-m<n, i.e., m is a valid first index)
        #                                         + a[m-k] (if 0<=m-k<n, i.e., m is a valid second index)
        # But a[k-m] with 0<=k-m<n means k-n+1 <= m <= k
        # And a[m-k] with 0<=m-k<n means k <= m <= k+n-1
        # So the two ranges are [max(0,k-n+1), min(n-1,k)] and [max(0,k), min(n-1,k+n-1)]
        # They overlap at m=k (if k < n).
        
        # Simpler approach: just compute it directly
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n - 1, k_int)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        j = k_int - idx
        # grad[idx] += a[j] (first term: m=i, contribution is a[k-i]=a[j])
        grad[idx] += w * a[j]
        # grad[j] += a[idx] (second term: m=k-i=j, contribution is a[i]=a[idx])
        # But j might have duplicates or be out of range... actually j = k - idx, and since
        # idx is in [i_lo, i_hi], j is in [k-i_hi, k-i_lo] which is within [0, n-1].
        # And the mapping i -> j=k-i is bijective on this range.
        grad[j] += w * a[idx]
    
    return 2.0 * grad


def mirror_descent(a0, time_limit, lr_init=0.1, beta_init=3.0, top_k_init=30,
                   restart_every=3000, verbose=True, rng_seed=42, min_top_k=10):
    """
    Multiplicative update (mirror descent) with soft-max gradient.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    a_prev = a.copy()
    momentum = 0.8
    lr = lr_init
    beta = beta_init
    top_k = top_k_init
    stall = 0
    best_stall_action = 0
    
    for t in range(1, 500000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov step in multiplicative space
        ratio = a / np.maximum(a_prev, 1e-30)
        ratio = np.clip(ratio, 0.01, 100.0)
        a_nest = a * np.power(ratio, momentum)
        a_nest = norm_sum1(a_nest)
        
        # Compute convolution and gradient
        conv = fast_conv(a_nest)
        grad = compute_gradient_v2(a_nest, conv, top_k, beta)
        
        # Adaptive learning rate
        eta = lr / (1.0 + 0.03 * np.sqrt(t))
        
        # Multiplicative update (mirror descent for simplex)
        log_a = np.log(np.maximum(a_nest, 1e-30)) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        if s < 1e-15:
            a_new = norm_sum1(a_nest)
        else:
            a_new = a_new / s
        
        a_prev = a
        a = a_new
        
        # Check improvement
        conv_new = fast_conv(a)
        m_new = conv_new.max()
        if m_new < best_m - 1e-13:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive top_k
        if t % 500 == 0:
            top_k = min(top_k + 3, 200)
            # Also adapt beta: increase to focus more on top peaks
            if stall > 500:
                beta = min(beta + 0.3, 20.0)
            elif stall < 100:
                beta = max(beta - 0.1, 1.0)
        
        # Restart when stuck
        if stall > restart_every:
            stall = 0
            best_stall_action = (best_stall_action + 1) % 5
            pert_type = best_stall_action
            
            if pert_type == 0:
                # Power law perturbation
                i_arr = np.arange(n, dtype=np.float64) + 1.0
                p = float(rng.uniform(1.0, 5.0))
                pert = i_arr ** p
            elif pert_type == 1:
                # Reversed power law
                i_arr = np.arange(n, dtype=np.float64) + 1.0
                p = float(rng.uniform(1.0, 5.0))
                pert = (n + 1.0 - i_arr) ** p
            elif pert_type == 2:
                # Mixed
                i_arr = np.arange(n, dtype=np.float64)
                p1 = float(rng.uniform(1.0, 3.0))
                p2 = float(rng.uniform(1.0, 3.0))
                pert = np.power(i_arr + 1.0, p1) + np.power(n - i_arr, p2)
            elif pert_type == 3:
                # Gaussian
                x = np.linspace(0, 1, n)
                c = float(rng.uniform(0.2, 0.8))
                sig = float(rng.uniform(0.1, 0.4))
                pert = np.exp(-((x - c)**2) / (2 * sig**2))
            else:
                # Small random perturbation of best
                pert = best_a * (1.0 + rng.normal(0, 0.1, n))
            
            pert = norm_sum1(pert)
            w_mix = float(rng.uniform(0.1, 0.5))
            a = norm_sum1((1 - w_mix) * best_a + w_mix * pert)
            a_prev = a.copy()
            lr = max(lr * 0.9, 0.01)
            
            if verbose and (time.monotonic() - t0) % 30 < 1:
                J = 2.0 * n * best_m
                print(f"    MD restart t={t} ({time.monotonic()-t0:.0f}s): J={J:.6f}, top_k={top_k}, beta={beta:.1f}, type={pert_type}")
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Mirror descent done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def spectral_flatten(a0, time_limit, lr_init=0.5, verbose=False):
    """
    Flatten the power spectrum to reduce convolution peaks.
    Iteratively reduces spectral peaks and enforces non-negativity.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    lr = lr_init
    stall = 0
    beta_spec = 2.0
    floor_ratio = 0.1  # don't reduce below this fraction of mean
    
    for t in range(1, 500000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A)**2
        P_max = P.max()
        P_mean = P.mean()
        
        if P_max < 1e-30 or P_mean < 1e-30:
            break
        
        # Compute a "flatness" target: geometric mean
        log_P = np.log(np.maximum(P, 1e-30))
        log_mean = log_P.mean()
        
        # Gradient direction: reduce peaks, boost valleys
        # Target: P_target = P_mean (flat spectrum)
        # In multiplicative space: A_new = A * exp(-lr * (log(P) - log(P_mean)) / log(P_max - log(P_mean)))
        dev = log_P - log_mean
        dev_range = dev.max() - dev.min()
        if dev_range < 1e-10:
            break
        dev_norm = dev / dev_range
        
        eta = lr / (1.0 + 0.05 * np.sqrt(t))
        scale = np.exp(-eta * dev_norm * beta_spec)
        scale = np.clip(scale, 0.5, 2.0)  # limit per-step change
        
        A_new = A * scale
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = norm_sum1(a_new)
        
        m_new = fast_conv(a_new).max()
        if m_new < best_m - 1e-13:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 2000:
                # Perturb and continue
                a = norm_sum1(best_a * (1.0 + rng.normal(0, 0.05, n)))
                stall = 0
                lr *= 0.85
                beta_spec = max(beta_spec - 0.3, 0.5)
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Spectral done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def coord_descent(a0, time_limit, rng_seed=77, verbose=True):
    """
    Coordinate descent: move mass to reduce top convolution peaks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv(a)
        M_curr = conv.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10000:
                break
        
        # Get top lags
        n_top = min(20, 2 * n - 1)
        top_lags = np.argsort(conv)[::-1][:n_top]
        
        improved = False
        
        for k_pos in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            
            k_int = int(k_pos)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n - 1, k_int)
            if i_lo > i_hi:
                continue
            
            idx = np.arange(i_lo, i_hi + 1)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            
            # Find top contributing pairs
            n_pairs = min(8, len(idx))
            pair_order = np.argsort(contrib)[::-1][:n_pairs]
            
            for pi in pair_order:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[pi])
                src_val = a[src]
                if src_val < 1e-14:
                    continue
                
                # Compute impact of reducing a[src]
                # For each top lag, how much does a[src] contribute?
                impact = np.zeros(n)
                for k2 in top_lags[:6]:
                    k2i = int(k2)
                    # a[src] contributes to conv[k2] via a[src]*a[k2-src] and a[k2-src]*a[src]
                    j1 = k2i - src
                    j2 = src - k2i
                    w2 = conv[k2i]
                    if 0 <= j1 < n:
                        impact[src] += a[j1] * w2
                    if 0 <= j2 < n:
                        impact[src] += a[j2] * w2
                    # Destination position d: how much would it add?
                    # For lag k2: a[d]*a[k2-d] and a[k2-d]*a[d]
                    # We want d to have low impact
                    if 0 <= j1 < n and j1 != src:
                        impact[j1] -= 0  # this is a source too
                    if 0 <= j2 < n and j2 != src:
                        impact[j2] -= 0
                
                # Find best destination: minimize total impact across top lags
                dst_scores = np.zeros(n)
                for k2 in top_lags[:6]:
                    k2i = int(k2)
                    for d in range(n):
                        j_a = k2i - d
                        j_b = d - k2i
                        if 0 <= j_a < n:
                            dst_scores[d] += a[j_a] * conv[k2i]
                        if 0 <= j_b < n:
                            dst_scores[d] += a[j_b] * conv[k2i]
                
                dst_scores[src] = np.inf
                dst = int(np.argmin(dst_scores))
                
                if dst == src:
                    continue
                
                # Try transfer
                for frac in [0.01, 0.03, 0.05, 0.1, 0.15]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] = max(0, a_trial[src] - amt)
                    a_trial[dst] += amt
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_conv(a_trial).max()
                    if m_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved and stall > 200:
            # Random perturbation
            n_moves = int(rng.integers(3, 15))
            for _ in range(n_moves):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = float(rng.uniform(0.005, 0.05))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            ss = a.sum()
            if ss > 1e-15:
                a = a / ss
            m_new = fast_conv(a).max()
            if m_new < best_m:
                best_m = m_new
                best_a = a.copy()
                stall = 0
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Coord descent done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Key insight: power laws a_i ~ (i+1)^p give score ~ 2(p+1)^2(p!)^2/(2p+1)! < 1 for p>=2.
    Multiple optimization phases with length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = evaluate_safe(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")
    
    # === Phase 1: Comprehensive parametric search at multiple lengths ===
    print("\n=== Phase 1: Parametric search ===")
    param_time = min(time_budget * 0.15, 150.0)
    
    # Try different lengths
    lengths_to_try = [3000, 5000, 5200, 7000, 8000, 10000, 12000]
    if parent_loaded:
        n_parent = len(best_seq)
        if n_parent not in lengths_to_try:
            lengths_to_try.insert(0, n_parent)
    
    for n_try in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t_param = min(remaining * 0.3, 30.0)
        
        a_init, s_init = parametric_search(n_try, time_limit=t_param)
        if a_init is not None:
            sc = evaluate_safe(a_init)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_init.copy()
                print(f"  Parametric n={n_try}: score={sc:.6f} (t={time.monotonic()-t_global:.0f}s)")
    
    # Also try the parent if loaded
    if parent_loaded:
        n_p = len(best_seq)
        a_parent = norm_sum1(np.asarray(height_sequence_1, dtype=np.float64))
        s_p = evaluate_safe(a_parent)
        if np.isfinite(s_p) and s_p < best_score:
            best_score = s_p
            best_seq = a_parent.copy()
    
    if best_seq is None:
        n0 = 5000
        best_seq, _ = parametric_search(n0, time_limit=10.0)
        best_score = evaluate_safe(best_seq)
        print(f"  Fallback parametric: n={n0}, score={best_score:.6f}")
    
    n_best = len(best_seq)
    print(f"  After parametric: best score={best_score:.6f}, n={n_best}")
    
    # === Phase 2: Mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Mirror descent (n={n_best}, budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.30, 400.0)
    if t2 > 30:
        a_out, _ = mirror_descent(best_seq, t2, lr_init=0.05, beta_init=3.0,
                                  top_k_init=30, restart_every=2000, verbose=True, rng_seed=42)
        check_update(a_out, f"md_n{n_best}")
    
    # === Phase 3: Spectral flattening ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Spectral (n={n_best}, budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 150.0)
    if t3 > 20:
        a_out, _ = spectral_flatten(best_seq, t3, lr_init=0.5, verbose=True)
        check_update(a_out, f"spectral_n{n_best}")
    
    # === Phase 4: Second mirror descent (finer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Fine MD (n={n_best}, budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 20:
        a_out, _ = mirror_descent(best_seq, t4, lr_init=0.02, beta_init=5.0,
                                  top_k_init=50, restart_every=3000, verbose=True, rng_seed=99)
        check_update(a_out, f"md_fine_n{n_best}")
    
    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord descent (n={n_best}, budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 120.0)
    if t5 > 15:
        a_out, _ = coord_descent(best_seq, t5, rng_seed=123, verbose=True)
        check_update(a_out, f"coord_n{n_best}")
    
    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration (budget={remaining:.0f}s) ===")
    
    length_tries = []
    for delta in [-2000, -1000, 500, 1000, 2000, 3000, 5000]:
        nn = n_best + delta
        if 2000 <= nn <= 15000 and nn != n_best:
            length_tries.append(nn)
    for rn in [4000, 6000, 8000, 10000, 12000]:
        if rn != n_best and 2000 <= rn <= 15000:
            length_tries.append(rn)
    length_tries = sorted(set(length_tries))
    
    for new_n in length_tries[:6]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")
        
        a_interp = interpolate_sequence(best_seq, new_n)
        
        # Quick optimization on new length
        t6a = min(remaining * 0.3, 100.0)
        if t6a > 20:
            a_out, _ = mirror_descent(a_interp, t6a, lr_init=0.04, beta_init=3.0,
                                      top_k_init=40, restart_every=2000, verbose=False, rng_seed=new_n % 1000)
            check_update(a_out, f"md_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t6b = min(remaining * 0.2, 60.0)
            if t6b > 15:
                a_out, _ = spectral_flatten(a_interp, t6b, lr_init=0.4, verbose=False)
                check_update(a_out, f"spec_n{new_n}")
    
    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}, budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.4, 200.0)
    if t7 > 20:
        a_out, _ = mirror_descent(best_seq, t7, lr_init=0.01, beta_init=8.0,
                                  top_k_init=80, restart_every=1500, verbose=True, rng_seed=200)
        check_update(a_out, f"final_md_n{n_best}")
    
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t8 = min(remaining * 0.5, 60.0)
        if t8 > 10:
            a_out, _ = coord_descent(best_seq, t8, rng_seed=42, verbose=False)
            check_update(a_out, f"final_coord_n{len(best_seq)}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = power_law_init(n_fb, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback power law: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
```
