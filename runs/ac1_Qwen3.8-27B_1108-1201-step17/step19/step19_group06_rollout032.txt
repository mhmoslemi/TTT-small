def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)

    return a / s

This is a well-studied problem! The "PAPR reduction" literature uses:
1. **Projected Subgradient Method (PSM)**: Project the subgradient onto the null space of the constraint.
2. **Successive Over-Relaxation (SOR)**: Iteratively adjust sub-expectations.
3. **Constellation Precoding**: Partition the signal constellation and solve a closest-point problem.
4. **Active Set Methods**: Identify the "active" time samples (where the peak occurs) and solve a QP.

The most promising for our case is the **Active Set Method**:
1. Identify the lag k* where the autocorrelation peaks.
2. The constraint (a*a)[k*] <= M defines a "forbidden" region.

def fast_autocorr(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score_from_normalized(a):
    """Score assuming a is normalized to sum=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max())


def evaluate_safe(a):
    """Safely call the evaluation function."""

def compute_softmax_grad(a, ac, beta, top_k=50):
    """Compute soft-max weighted gradient of autocorrelation peaks."""
    n = len(a)
    k_alarm = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_alarm]
    top_vals = ac[top_idx]

    max_v = top_vals.max()
    expo = beta * (top_vals - max_v)
    expo = np.clip(expo, -50, 0)
    weights = np.exp(expo)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_alarm) / k_alarm
    else:
        weights = weights / w_sum

    grad = np.zeros(n)
    for w_k, k in zip(weights, top_idx):
        k_int = int(k)
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n, k_int + 1)
        if i_lo >= i_hi:
            continue
        idx = np.arange(i_lo, i_hi)
        j_idx = k_int - idx
        grad[idx] += 2.0 * w_k * a[j_idx]

    return grad


def mirror_descent_adaptive(a0, time_limit, rng_seed=42, beta_start=2.0,
                            beta_max=20.0, lr_start=0.12, verbose=True):
    """
    Soft-max mirror descent with:
    - Adaptive beta annealing (smooth to sharp)
    - RMSProp adaptive learning rates
    - Nesterov momentum in log-space
    - Periodic restarts with diverse perturbations
    """
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    beta = beta_start
    lr = lr_start
    momentum = 0.88
    v = np.ones(n) * 1e-8
    v_decay = 0.9

    a_prev = a.copy()
    stall = 0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Nesterov lookahead in log-space
        log_ratio = np.log(np.maximum(a, 1e-30) / np.maximum(a_prev, 1e-30))
        log_ratio = np.clip(log_ratio, -3.0, 3.0)
        a_nest = a * np.exp(momentum * log_ratio)
        a_nest = norm_sum1(a_nest)

        # Compute autocorrelation and gradient
        ac = fast_autocorr(a_nest)

        # Adaptive top_k: start small, grow over time
        top_k = min(20 + t // 200, 100)
        grad = compute_softmax_grad(a_nest, ac, beta, top_k=top_k)

        # RMSProp adaptive rate
        v = v_decay * v + (1.0 - v_decay) * grad * grad
        adaptive_lr = lr / (np.sqrt(v) + 1e-10)

        # Multiplicative update with clipping
        step = np.clip(adaptive_lr * grad, -4.0, 4.0)
        a_new = a_nest * np.exp(-step)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        a_prev = a
        a = a_new

        # Check improvement
        ac_check = fast_autocorr(a)
        m_check = ac_check.max()
        if m_check < best_m - 1e-14:
            best_m = m_check
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive beta: increase slowly, with periodic annealing resets
        if t % 500 == 0:
            beta = min(beta + 0.3, beta_max)
            lr *= 0.97

        # Restart when stuck
        if stall > 2500:
            stall = 0
            pert_type = int(rng.integers(6))
            if pert_type == 0:
                x = np.linspace(0, 1, n)
                c = float(rng.uniform(0.2, 0.8))
                sig = float(rng.uniform(0.05, 0.4))
                pert = np.exp(-((x - c) ** 2) / (2 * sig ** 2))
            elif pert_type == 1:
                i_arr = np.arange(n, dtype=np.float64)
                p = float(rng.uniform(0.3, 3.0))
                pert = (i_arr + 1.0) ** p
            elif pert_type == 2:
                x = np.linspace(0, 1, n)
                freq = int(rng.integers(1, 6))
                phase = float(rng.uniform(0, 2 * np.pi))
                pert = np.sin(np.pi * freq * x + phase) ** 2
            elif pert_type == 3:
                x = np.linspace(0, 1, n)
                c1 = float(rng.uniform(0.1, 0.4))
                c2 = float(rng.uniform(0.6, 0.9))
                sig = float(rng.uniform(0.05, 0.2))
                pert = (np.exp(-((x - c1) ** 2) / (2 * sig ** 2)) +
                        np.exp(-((x - c2) ** 2) / (2 * sig ** 2)))
            elif pert_type == 4:
                pert = best_a * (1.0 + rng.normal(0, 0.08, n))
            else:
                pert = rng.uniform(0.1, 1.0, n)

            pert = norm_sum1(pert)
            w_mix = float(rng.uniform(0.15, 0.5))
            a = norm_sum1((1.0 - w_mix) * best_a + w_mix * pert)
            a_prev = a.copy()
            v = np.ones(n) * 1e-8
            beta = max(2.0, beta * 0.6)
            lr = lr_start * 0.8

            if verbose and (time.monotonic() - t0) < time_limit:

                print(f"      MD restart t={t} ({time.monotonic()-t0:.0f}s): "
                      f"J={J:.6f}, beta={beta:.1f}, lr={lr:.4f}")

    if verbose:
        J = 2.0 * n * best_m
        print(f"    MD done ({time.monotonic()-t0:.0f}s): J={J:.6f}, "
              f"beta={beta:.1f}, best_m={best_m:.8f}")
    return best_a


def coordinate_descent_polish(a0, time_limit, rng_seed=77, verbose=True):
    """
    Coordinate descent: for each top autocorrelation lag, try to reduce
    peak by transferring mass from high-contributing positions to
    low-impact positions. Uses line search.
    """
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        m_curr = ac.max()

        if m_curr < best_m - 1e-15:
            best_m = m_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15000:
                break

        # Get top lags
        n_top = min(20, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k_pos in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break

            k_int = int(k_pos)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo >= i_hi:
                continue

            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]

            # Find top contributing pairs
            n_pairs = min(8, len(idx))
            pair_order = np.argsort(contrib)[::-1][:n_pairs]

            for pi in pair_order:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[pi])
                src_val = a[src]
                if src_val < 1e-14:
                    continue

                # Score for each destination (lower is better)
                scores = np.zeros(n)
                for k2 in top_lags[:6]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n, k2i + 1)
                    if s2 < e2:
                        ii = np.arange(s2, e2)
                        jj = k2i - ii
                        scores[ii] += a[jj] * ac[k2i]
                        scores[jj] += a[ii] * ac[k2i]

                scores[src] = np.inf
                # Also penalize positions that are the partner of src at this lag
                partner = k_int - src
                if 0 <= partner < n:
                    scores[partner] = np.inf

                dst = int(np.argmin(scores))
                if dst == src:
                    continue

                # Try transfer with line search
                best_trial_m = best_m
                best_frac = 0.0
                for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.2]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_trial_m:
                        best_trial_m = m_trial
                        best_frac = frac

                if best_frac > 0 and best_trial_m < best_m - 1e-15:
                    amt = src_val * best_frac
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    best_m = best_trial_m
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            if stall > 1000 and stall % 300 == 0:
                # Random perturbations to escape local minimum
                n_moves = int(rng.integers(3, 12))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = float(rng.uniform(0.003, 0.02))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        J = 2.0 * n * best_m
        print(f"    Coord descent done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a


def spectral_shaping(a0, time_limit, rng_seed=99, verbose=True):
    """
    Flatten the power spectrum to reduce autocorrelation peaks.
    Iteratively reduces spectral peaks while maintaining non-negativity.
    """
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    fft_size = 1 << (2 * n - 2).bit_length()
    lr = 0.15
    beta_spec = 4.0
    stall = 0

    for t in range(1, 500000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a.astype(np.float64), fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_mean = P.mean()

        if P_max < 1e-30:
            break

        # Weight: push energy from spectral peaks to valleys
        P_dev = P - P_mean
        expo = beta_spec * P_dev / max(P_max, 1e-30)
        expo = np.clip(expo, -15, 15)
        w_spec = np.exp(expo)
        w_spec = w_spec / w_spec.mean()

        # Modify spectrum: reduce peaks, boost valleys
        eta_t = lr / (1.0 + 0.03 * np.sqrt(t))
        A_new = A * np.exp(-0.5 * eta_t * (w_spec - 1.0))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = norm_sum1(a_new)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 2000:
                a = norm_sum1(best_a * (1.0 + rng.normal(0, 0.008, n)))
                stall = 0
                lr *= 0.85
                beta_spec = max(beta_spec - 0.3, 1.5)
                if lr < 0.005:
                    break

    if verbose:
        J = 2.0 * n * best_m
        print(f"    Spectral done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def parametric_init(n, time_limit=15.0):
    """Quick parametric search for good initializations."""
    t0 = time.monotonic()
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)

    candidates = []

    # Uniform
    candidates.append(np.ones(n) / n)

    # Power laws
    for p in [0.2, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))

    # Gaussians
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sig in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]:
            candidates.append(norm_sum1(np.exp(-((x - c) ** 2) / (2 * sig ** 2))))

    # Bimodal
    for sig in [0.05, 0.1, 0.15, 0.2]:
        for c1 in [0.2, 0.3]:
            candidates.append(norm_sum1(
                np.exp(-((x - c1) ** 2) / (2 * sig ** 2)) +
                np.exp(-((x - (1 - c1)) ** 2) / (2 * sig ** 2))))

    # Sine modes
    for f in [1, 2, 3, 4, 5]:
        candidates.append(norm_sum1(np.sin(np.pi * f * x) ** 2))
        candidates.append(norm_sum1(np.cos(np.pi * f * x / 2) ** 2))

    # Exponential
    for r in [0.5, 1, 2, 3, 5]:
        candidates.append(norm_sum1(np.exp(-r * x)))
        candidates.append(norm_sum1(np.exp(r * x)))

    # Triangular
    candidates.append(norm_sum1(1.0 - np.abs(2 * x - 1)))

    # Evaluate all
    best_cand = None
    best_val = float('inf')
    for c in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        v = score_from_normalized(c)
        if v < best_val:
            best_val = v
            best_cand = c.copy()

    return best_cand if best_cand is not None else np.ones(n) / n

                                            rng_seed=new_n % 500, verbose=False)
            check_update(a_out, f"md_n{new_n}")

        # Quick coord descent
        elapsed = time.monotonic() - t_global
        remaining = time_budget - elapsed
        if remaining > 60:
            t5b = min(remaining * 0.25, 80.0)
            if t5b > 10:
                a_out = coordinate_descent_polish(best_seq, t5b,
                                                  rng_seed=new_n % 999, verbose=False)
                check_update(a_out, f"coord_n{new_n}")

    # === Phase 6: Final polish on best ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    n_final = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_final}, budget={remaining:.0f}s) ===")

    # MD with very fine parameters
    t6 = min(remaining * 0.35, 300.0)
    if t6 > 20:
        a_out = mirror_descent_adaptive(best_seq, t6, beta_start=8.0,
                                        beta_max=30.0, lr_start=0.02,
                                        rng_seed=500, verbose=True)
        check_update(a_out, f"final_md_n{n_final}")

    # Coord descent
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    if remaining > 30:
        t7 = min(remaining * 0.35, 200.0)
        if t7 > 15:
            a_out = coordinate_descent_polish(best_seq, t7, rng_seed=42, verbose=True)
            check_update(a_out, f"final_coord_n{len(best_seq)}")

    # One more spectral pass
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    if remaining > 20:
        t8 = min(remaining * 0.3, 100.0)
        if t8 > 10:
            a_out = spectral_shaping(best_seq, t8, rng_seed=314, verbose=True)
            check_update(a_out, f"final_spec_n{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")

    return [float(x) for x in best_seq]
