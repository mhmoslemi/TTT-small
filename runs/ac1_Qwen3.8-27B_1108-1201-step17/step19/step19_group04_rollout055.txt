def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]


def conv_ab(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]


def corr_shift(w, a):
    w = np.asarray(w, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = w.size
    size = next_pow2(L + n - 1)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fw * fa.conj(), size)[:n]
    return np.maximum(c, 0.0)

def weights_from_q(q, focus=4.0):
    q = np.asarray(q, dtype=np.float64)
    m = float(q.max())
    if m <= 1e-15:

    f1 = quad_max_val(c1, q0, c, qd)
    f2 = quad_max_val(d1, q0, c, qd)
    for _ in range(iters):
        if f1 < f2:
            b = d1
            d1 = c1
            f2 = f1
            c1 = b - phi * (b - a)
            f1 = quad_max_val(c1, q0, c, qd)
        else:
            a = c1
            c1 = d1
            f1 = f2
            d1 = a + phi * (b - a)
            f2 = quad_max_val(d1, q0, c, qd)
    return 0.5 * (a + b)


def line_search_direction(a, d, q0, T=None):
    a = np.asarray(a, dtype=np.float64)
    d = np.asarray(d, dtype=np.float64)
    q0 = np.asarray(q0, dtype=np.float64)
    M0 = float(np.max(q0))

    dot = float(np.dot(a, d))
    if dot > 1.0 - 1e-7:
        return a.copy(), M0

    c = conv_ab(a, d)
    qd = self_conv(d)

    best_alpha = 0.0
    best_M = M0

    alphas = [1e-4, 5e-4, 1e-3, 2e-3, 5e-3, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 0.9, 1.0]

    if T is not None and np.isfinite(T) and T < 0.0 and M0 > 1e-15:
        t1 = min(0.30, max(5e-4, 0.5 * (-float(T)) / M0))
        alphas.append(max(1e-5, 0.5 * t1))
        alphas.append(min(0.95, 1.5 * t1))

    uniq = []
    seen = set()
    for al in alphas:
        al = float(min(1.0, max(0.0, al)))
        key = round(al, 12)
        if key not in seen:
            seen.add(key)
            uniq.append(al)

    for al in uniq:
        val = quad_max_val(al, q0, c, qd)
        if val < best_M:
            best_M = val
            best_alpha = al

    if best_M < M0 - 1e-13 and 0.0 < best_alpha < 1.0:
        al2 = golden_alpha_max(q0, c, qd, max(0.0, best_alpha - 0.1), min(1.0, best_alpha + 0.1), 30)
        val2 = quad_max_val(al2, q0, c, qd)
        if val2 < best_M:
            best_M = val2
            best_alpha = al2

    if best_alpha <= 0.0 or best_M >= M0 - 1e-14:
        return a.copy(), M0

    new = (1.0 - best_alpha) * a + best_alpha * d
    new = np.maximum(new, 0.0)
    s = float(new.sum())
    if s <= 1e-15:
        return a.copy(), M0
    new = new / s
    return new, best_M


def get_lp_artifacts(n):
    art = _LP_CACHE.get(n)
    if art is not None:
        return art
    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0
    Aeq = np.zeros((1, n + 1), dtype=np.float64)
    Aeq[0, :n] = 1.0
    bounds = [(0.0, np.inf)] * n + [(None, None)]
    art = (c, Aeq, bounds)
    if len(_LP_CACHE) > 8:
        _LP_CACHE.clear()
    _LP_CACHE[n] = art
    return art


def solve_lp_offset(y, active, q_active):
    if not HAVE_LP:
        return None, None
    y = np.asarray(y, dtype=np.float64)
    active = np.asarray(active, dtype=int)
    q_active = np.asarray(q_active, dtype=np.float64)
    n = y.size
    K = active.size
    if K <= 0:
        return None, None
    if K * (n + 1) > 12_000_000:
        return None, None

    try:
        A = np.zeros((K, n + 1), dtype=np.float64)
    except Exception:
        return None, None

    yy = 2.0 * y
    for r, k in enumerate(active):
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            A[r, i0:i1] = yy[start:stop][::-1]

    A[:, n] = -1.0
    b = np.maximum(q_active, 0.0)

    c, Aeq, bounds = get_lp_artifacts(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq,
                      b_eq=np.array([1.0], dtype=np.float64),
                      bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception:
        return None, None

    status = getattr(res, 'status', None)
    success = getattr(res, 'success', False)
    x = getattr(res, 'x', None)
    if (status != 0 or not success) or x is None:
        return None, None

    g = np.asarray(x[:n], dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-10:
        return None, None
    g = g / s
    T = float(x[n])
    if not np.isfinite(T):
        T = 0.0
    return g, T


def soft_direction_best(a, q, current_score, max_dirs=6):
    a = np.asarray(a, dtype=np.float64)
    q = np.asarray(q, dtype=np.float64)
    n = a.size
    if n <= 1:
        return None, current_score / (2.0 * float(n))

    M_cur = current_score / (2.0 * float(n))

    focus_opts = (3.0, 8.0, 20.0, 50.0)
    trust_opts = (0.1, 0.4, 1.2, 3.0)

    best_new = None
    best_M = M_cur
    used = 0

    for focus in focus_opts:
        if used >= max_dirs:
            break
        w = weights_from_q(q, focus)
        c = corr_shift(w, a)
        cmax = float(np.max(c))
        cmin = float(np.min(c))
        scale = cmax - cmin
        if scale <= 1e-15:
            continue
        for trust in trust_opts:
            if used >= max_dirs:
                break
            move = trust * (cmax - c) / scale
            move = np.clip(move, -50.0, 50.0)
            d = a * np.exp(move)
            d = normalize_vec(d)
            new, Mnew = line_search_direction(a, d, q, None)
            used += 1
            if Mnew < best_M - 1e-13:
                best_M = Mnew
                best_new = new

    if best_new is None:
        return None, M_cur
    return best_new, best_M


def adaptive_lp_fw(a0, time_limit, verbose=True):
    """
    Adaptive Frank-Wolfe with multi-scale active set and simulated annealing escape.
    """
    if time_limit <= 0.0:
        return normalize_vec(a0)

    t0 = time.monotonic()
    y = normalize_vec(a0)
    n = y.size
    if n <= 5:
        return y

    best = y.copy()
    best_score = objective_norm(y)

    # Adaptive K schedule
    if n > 7000:
        K_schedule = [50, 80, 120, 200, 350, 500, 700, 900]
    elif n > 4000:
        K_schedule = [50, 100, 150, 250, 400, 600, 800, 1200]
    else:
        K_schedule = [50, 100, 200, 350, 550, 800, 1100, 1500]

    K_idx = 0
    K = K_schedule[0]
    use_lp = HAVE_LP
    lp_fails = 0
    no_improve = 0
    total_improve = 0
    last_print = t0
    iter_no = 0
    best_at_iter = 0
    rng = np.random.default_rng(42)

    # Temperature for simulated annealing
    temp = 0.0005
    temp_decay = 0.995

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining <= 0.0:
            break

        qy = self_conv(y)
        M_y = float(np.max(qy))
        score_y = 2.0 * n * M_y

        if score_y < best_score - 1e-12:
            best = y.copy()
            best_score = score_y
            best_at_iter = iter_no
            no_improve = 0
        else:
            no_improve += 1

        # Progressively widen active set
        if K_idx < len(K_schedule) - 1 and no_improve > 60:
            K_idx += 1
            K = K_schedule[K_idx]

        # Very long stall: try escape
        if no_improve > 200:
            if no_improve > 400 and time.monotonic() - t0 > 0.85 * time_limit:
                break

            # Simulated annealing: accept a random perturbation
            if no_improve % 100 == 50:
                scale = 0.005 * (1.0 + no_improve / 1000.0)
                cand = y + scale * rng.normal(size=n) * y
                cand = np.maximum(cand, 1e-15)
                cand = normalize_vec(cand)
                sc = objective_norm(cand)
                # Accept if not too bad (annealing)
                if sc < best_score + temp * best_score:
                    y = cand
                    no_improve = 50
                temp *= temp_decay

            # Reset to best and try soft direction
            if no_improve % 80 == 70:
                y = best.copy()
                new, M2 = soft_direction_best(y, self_conv(y), 2.0 * n * M2 if False else 2.0 * n * float(np.max(self_conv(y))), max_dirs=8)
                if M2 < float(np.max(self_conv(y))) - 1e-13:
                    y = new
                    no_improve = 40
                else:
                    y = normalize_vec(best + 0.002 * rng.normal(size=n) * best)
                    y = np.maximum(y, 1e-15)
                    y = normalize_vec(y)

        # Determine active set using adaptive threshold
        order = np.argsort(qy)[::-1]
        # Use lags within a relative threshold of the max
        threshold_rel = max(0.001, 0.02 - 0.001 * K_idx)
        threshold_abs = M_y * (1.0 - threshold_rel)
        act_all = order[qy >= threshold_abs]
        # Also ensure we have at least K lags
        if len(act_all) < K:
            act_all = order[:min(K, len(order))]
        Kcur = int(min(len(act_all), 1500))
        act = act_all[:Kcur]
        q_active = qy[act]

        g_dir = None
        T = None

        if use_lp and Kcur >= 10 and Kcur * n < 10_000_000:
            tstep = time.monotonic()
            g_dir, T = solve_lp_offset(y, act, q_active)
            spend = time.monotonic() - tstep

            if g_dir is None:
                lp_fails += 1
                if lp_fails >= 4:
                    use_lp = False
            else:
                lp_fails = 0
                if spend > 120.0:
                    use_lp = False
                elif spend > max(30.0, 0.15 * remaining):
                    K = max(30, int(K * 0.7))

        if g_dir is not None:
            new, Mnew = line_search_direction(y, g_dir, qy, T)
            if Mnew < M_y - 1e-13:
                y = new
                total_improve += 1
            else:
                # Try soft direction as fallback
                new2, M2 = soft_direction_best(y, qy, score_y, max_dirs=4)
                if M2 < M_y - 1e-13:
                    y = new2
                    total_improve += 1
                else:
                    K = min(int(K * 1.2), 1500)
        else:
            new, Mnew = soft_direction_best(y, qy, score_y, max_dirs=8)
            if Mnew < M_y - 1e-13:
                y = new
                total_improve += 1
            else:
                K = min(int(K * 1.15), 1500)
                if no_improve > 250:
                    break

        iter_no += 1
        now = time.monotonic()
        if verbose and (iter_no % 30 == 0 or (now - last_print) > 25.0):
            print("  AFW: it=%d t=%.0fs best=%.9f K=%d Kidx=%d noimp=%d imp=%d lp=%s"
                  % (iter_no, now - t0, best_score, Kcur, K_idx, no_improve, total_improve, use_lp))
            last_print = now

    return best


def smooth_polish(a0, time_limit, rng_seed=11, verbose=False):
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 1:
        return a

    best = a.copy()
    best_score = objective_norm(a)

    trust = 0.4
    focus = 4.0
    fails = 0
    restarts = 0
    rng = np.random.default_rng(rng_seed)

    while time.monotonic() - t0 < time_limit:
        q = self_conv(a)
        M = float(np.max(q))
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            fails = 0
        else:
            fails += 1

        if fails > 500:
            break

        if fails > 50 and fails % 25 == 0:
            focus = min(focus * 1.15, 80.0)
            trust = max(trust * 0.8, 0.02)

        if fails > 150 and fails % 150 == 0:
            restarts += 1
            if restarts > 40:
                break
            scale_noise = 0.015 / (1.0 + 0.05 * restarts)
            cand = best + scale_noise * rng.normal(size=n) * best
            cand = np.maximum(cand, 1e-12)
            cand = normalize_vec(cand)
            sc = objective_norm(cand)
            if np.isfinite(sc) and sc <= best_score + 1e-8:
                a = cand
                fails = 0
                trust = 0.3
                focus = min(focus * 1.1, 80.0)
            else:
                a = best.copy()

        w = weights_from_q(q, focus)
        c = corr_shift(w, a)
        cmax = float(np.max(c))
        cmin = float(np.min(c))
        scale = cmax - cmin
        if scale <= 1e-15:
            break

        move = trust * (cmax - c) / scale
        move = np.clip(move, -50.0, 50.0)
        b = a * np.exp(move)
        b = normalize_vec(b)

        Mb = float(np.max(self_conv(b)))
        score_b = 2.0 * n * Mb

        if score_b < best_score - 1e-13:
            best = b.copy()
            best_score = score_b
            a = b
            fails = 0
            trust = min(trust * 1.08, 8.0)
        elif score_b < score - 1e-14:
            a = b
            fails = 0
            trust = min(trust * 1.12, 8.0)
            focus = max(focus * 0.97, 1.0)
        else:
            fails += 1
            trust = max(trust * 0.7, 0.02)
            focus = min(focus * 1.12, 80.0)

        if verbose and (time.monotonic() - t0) % 30.0 < 0.02:
            print("  smooth: t=%.0f score=%.9f focus=%.1f trust=%.3f fails=%d"
                  % (time.monotonic() - t0, best_score, focus, trust, fails))

    return best


def peak_equalize(a, time_limit, verbose=False):
    """
    Targeted peak equalization: iteratively reduce the top peaks by
    solving small subproblems to balance them.
    """
    t0 = time.monotonic()
    a = normalize_vec(a)
    n = a.size
    best = a.copy()
    best_score = objective_norm(a)
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        c = self_conv(a)
        M = float(np.max(c))
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best = a.copy()
            best_score = score

        # Find top peaks
        top_order = np.argsort(c)[::-1]
        top_peaks = top_order[:min(20, len(top_order))]
        peak_vals = c[top_peaks]

        # If peaks are already well-equalized, switch to mirror descent
        peak_spread = float(peak_vals[0] - peak_vals[-1])
        if peak_spread < M * 0.0005:
            # Peaks are well balanced, use mirror descent for further reduction
            w = weights_from_q(c, 3.0)
            cc = corr_shift(w, a)
            cmax = float(np.max(cc))
            cmin = float(np.min(cc))
            sc = cmax - cmin
            if sc <= 1e-15:
                break
            for trust in [0.1, 0.3, 1.0, 3.0]:
                move = trust * (cmax - cc) / sc
                move = np.clip(move, -30.0, 30.0)
                b = a * np.exp(move)
                b = normalize_vec(b)
                sc_b = objective_norm(b)
                if sc_b < best_score - 1e-13:
                    best = b.copy()
                    best_score = sc_b
                    a = b
                elif sc_b < score - 1e-14:
                    a = b
            continue

        # Try to equalize: move mass from positions contributing to top peak
        # to positions that are under-contributing
        top_k = int(top_peaks[0])

        # Find which positions contribute most to the top peak
        m_min = max(0, top_k - n + 1)
        m_max = min(n - 1, top_k)
        contrib = np.zeros(n, dtype=np.float64)
        m = np.arange(m_min, m_max + 1)
        pairs = top_k - m
        contrib[m] = a[m] * a[pairs]

        # Find positions with low contribution (good targets for adding mass)
        low_pos = np.where(contrib < 0.1 * float(contrib.max()))[0]
        high_pos = np.where(contrib > 3.0 * float(contrib.mean()))[0]

        if len(low_pos) > 0 and len(high_pos) > 0:
            # Try moving a small amount from high to low
            for frac in [0.01, 0.03, 0.05, 0.1]:
                b = a.copy()
                total_move = 0.0
                for hp in high_pos[:5]:
                    amt = b[hp] * frac * 0.1
                    b[hp] -= amt
                    # Distribute to low positions
                    for lp in low_pos[:5]:
                        b[lp] += amt / max(1, len(low_pos[:5]))
                    total_move += amt
                b = np.maximum(b, 0.0)
                s = float(b.sum())
                if s > 1e-15:
                    b = b / s
                    sc = objective_norm(b)
                    if sc < best_score - 1e-13:
                        best = b.copy()
                        best_score = sc
                        a = b

        # Also trydirect coordinate descent on the peak
        # Reduce the element at the position that most contributes to the peak
        if m_min <= m_max:
            idx_max = m[np.argmax(contrib[m_min:m_max+1] if m_max >= m_min else np.array([0]))]
            # Move mass from idx_max to a random low position
            for target in low_pos[:3]:
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    b = a.copy()
                    amt = b[int(idx_max)] * frac
                    b[int(idx_max)] -= amt
                    b[int(target)] += amt
                    b = np.maximum(b, 0.0)
                    s = float(b.sum())
                    if s > 1e-15:
                        b = b / s
                        sc = objective_norm(b)
                        if sc < best_score - 1e-13:
                            best = b.copy()
                            best_score = sc
                            a = b

        # Periodic noise injection
        if (time.monotonic() - t0) % 60.0 < 0.5:
            cand = a + 0.003 * rng.normal(size=n) * a
            cand = np.maximum(cand, 1e-12)
            cand = normalize_vec(cand)
            sc = objective_norm(cand)
            if sc < best_score:
                best = cand.copy()
                best_score = sc
                a = cand

    return best


def frequency_session(a, time_limit):
    """
    Frequency-domain diversification: modify the magnitude spectrum to flatten
    the time-domain convolution peak.
    """
    t0 = time.monotonic()
    a = normalize_vec(a)
    n = a.size
    best = a.copy()
    best_score = objective_norm(a)
    m = 2 * n - 1
    size = next_pow2(m)

    while time.monotonic() - t0 < time_limit:
        A = np.fft.rfft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)
        power = mag ** 2

        p_max = float(np.max(power))
        p_min_pos = power[power > 1e-30]
        if len(p_min_pos) == 0:
            break
        p_min = float(np.min(p_min_pos))
        if p_max <= 1e-15:
            break

        # Try various power corrections
        improved = False
        for gamma in [0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2]:
            new_power = power ** gamma
            new_power = new_power / float(new_power.sum()) * float(power.sum())
            new_mag = np.sqrt(np.maximum(new_power, 0.0))
            new_A = new_mag * np.exp(1j * phase)
            new_a = np.fft.irfft(new_A, size)[:n]
            new_a = np.maximum(new_a, 0.0)
            s = float(new_a.sum())
            if s <=
