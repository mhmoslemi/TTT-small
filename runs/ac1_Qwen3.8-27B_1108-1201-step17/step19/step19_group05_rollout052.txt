    """L-BFGS-B optimization with smooth max and analytic gradients."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    
    x0 = np.log(a)
    best_x = x0.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    
    t0 = time.monotonic()
    iter_count = [0]
    
    def obj_and_grad(x):
        iter_count[0] += 1
        if time.monotonic() - t0 > time_limit:
            return 1e10, np.zeros(n)
        
        # Softmax parameterization: a = exp(x) / sum(exp(x))
        x_shifted = x - x.max()
        exp_x = np.exp(x_shifted)
        s = exp_x.sum()
        if s < 1e-30:
            return 1e10, np.zeros(n)
        a_norm = exp_x / s
        
        conv = fast_autocorr(a_norm)
        max_c = conv.max()
        if max_c < 1e-30:
            return 0.0, np.zeros(n)
        
        # Adaptive temperature: start wide, narrow down
        progress = min(1.0, iter_count[0] / 200.0)
        temp = max_c * (0.05 * (1 - progress) + 0.001 * progress)
        temp = max(temp, 1e-10)
        
        scaled = conv / temp
        shifted = scaled - scaled.max()
        exp_shifted = np.exp(shifted)
        log_sum_exp = scaled.max() + np.log(exp_shifted.sum())
        smooth_max = temp * log_sum_exp
        
        obj = 2.0 * n * smooth_max
        
        # Gradient of smooth_max w.r.t. conv
        weights = exp_shifted / exp_shifted.sum()
        
        # Gradient of obj w.r.t. a_norm
        # d(obj)/d(a_norm[i]) = 2*n * sum_k weights[k] * d(conv[k])/d(a_norm[i])
        # conv[k] = sum_j a_norm[j] * a_norm[k-j]
        # d(conv[k])/d(a_norm[i]) = a_norm[k-i] + a_norm[i-k] (valid indices only)
        grad_an = np.zeros(n)
        
        # Vectorized computation
        for k in range(2 * n - 1):
            if weights[k] < 1e-14:
                continue
            wk = 2.0 * n * weights[k]
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo > i_hi:
                continue
            i_range = np.arange(i_lo, i_hi + 1)
            # d(conv[k])/d(a[i]) = a[k-i] + a[i-k]
            ki = k - i_range  # k-i for each i in range
            ik = i_range - k  # i-k for each i in range
            g = np.zeros(len(i_range))
            valid_ki = (ki >= 0) & (ki < n)
            valid_ik = (ik >= 0) & (ik < n)
            g[valid_ki] += a_norm[ki[valid_ki]]
            g[valid_ik] += a_norm[ik[valid_ik]]
            grad_an[i_range] += wk * g
        
        # Chain rule: d(a_norm[i])/d(x[j]) = a_norm[j] * (delta_ij - a_norm[j])
        dot = np.dot(grad_an, a_norm)
        grad_x = a_norm * (grad_an - dot)
        
        return obj, grad_x
    
    def callback(xk):
        # Track best
        x_shifted = xk - xk.max()
        exp_x = np.exp(x_shifted)
        a_curr = exp_x / exp_x.sum()
        obj_curr = 2.0 * n * fast_autocorr(a_curr).max()
        if obj_curr < best_obj - 1e-12:
            best_obj = obj_curr
            best_x[0:n] = xk.copy()
        return time.monotonic() - t0 > time_limit * 0.95
    
    try:
        result = minimize(obj_and_grad, x0, method='L-BFGS-B',
                         options={'maxiter': 20000, 'maxfun': 10000,
                                  'ftol': 1e-12, 'gtol': 1e-10,
                                  'maxcor': 10})
        
        # Evaluate final
        x_final = result.x
        x_shifted = x_final - x_final.max()
        exp_x = np.exp(x_shifted)
        a_final = exp_x / exp_x.sum()
        obj_final = 2.0 * n * fast_autocorr(a_final).max()
        if obj_final < best_obj:
            best_obj = obj_final
            best_x = x_final.copy()
    except Exception as e:
        if verbose:
            print(f"    L-BFGS-B error: {e}")
    
    # Recover best
    x_best = best_x - best_x.max()
    a_best = np.exp(x_best)
    a_best = a_best / a_best.sum()
    
    if verbose:
        print(f"    L-BFGS-B: {iter_count[0]} iters, obj={best_obj:.8f}, time={time.monotonic()-t0:.1f}s")
    
    return a_best, best_obj


def mirror_descent(a0, time_limit, n_peaks=50, lr0=5.0, restart_every=2000,
                   rng_seed=42, verbose=False):
    """Nesterov-accelerated mirror descent on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    restart_count = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-20)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        conv = fast_autocorr(y)
        
        # Subgradient from top peaks with softmax weights
        top_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
        vals = conv[top_idx]
        max_v = vals.max()
        if max_v < 1e-30:
            break
        exp_v = np.exp((vals - max_v) / max(1e-10, max_v * 0.01))
        w = exp_v / exp_v.sum()
        
        grad = np.zeros(n)
        for idx_i, k in enumerate(top_idx):
            ki = int(k)
            wi = w[idx_i]
            if wi < 1e-12:
                continue
            j_lo = max(0, ki - n + 1)
            j_hi = min(n - 1, ki)
            if j_lo > j_hi:
                continue
            j_range = np.arange(j_lo, j_hi + 1)
            pair = ki - j_range
            grad[j_range] += wi * 2.0 * a[pair]
        
        eta = lr0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = conv.max()
        M_x = fast_autocorr(x).max()
        M_eff = min(M_new, M_x)
        if M_eff < best_m - 1e-16:
            best_m = M_eff
            best_a = (x if M_x <= M_new else y).copy()
            stall = 0
        else:
            stall += 1
        
        if stall > restart_every:
            stall = 0
            restart_count += 1
            strength = min(0.1, 0.02 + 0.01 * min(restart_count, 10))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-15)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
        
        if verbose and t % 5000 == 0:
            print(f"      MD t={t} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")
    
    return best_a, best_m


def spectral_flatten(a0, time_limit, verbose=False):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_mean = P.mean()
        if P_max < 1e-30:
            break
        
        alpha = 0.5 + 0.5 * min(1.0, it / 300.0)
        P_new = P + alpha * (P_mean - P)
        P_new = np.maximum(P_new, 1e-30)
        
        mag_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        mag_ratio = np.clip(mag_ratio, 0.3, 3.0)
        A_new = A * mag_ratio
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            blend = 0.1 + 0.1 * min(1.0, it / 1000.0)
            a = (1 - blend) * a + blend * a_new
            a = np.maximum(a, 0.0)
            a = a / max(a.sum(), 1e-15)
            M_curr = fast_autocorr(a).max()
            if M_curr < best_m:
                best_m = M_curr
                best_a = a.copy()
    
    return best_a, best_m


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss


def propose_candidate(budget_s=1800):
    """Main search function."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = safe_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Phase 0: Initial candidates ===
    print("=== Phase 0: Initial Candidates ===")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p.copy()
            try_update(parent, "parent")
    except Exception:
        pass
    
    # Diverse initializations
    inits = []
    for n in [3000, 5000, 7000, 10000]:
        inits.append((np.ones(n), f"uniform_n{n}"))
        inits.append((make_ramp(n, 1.0), f"ramp_n{n}"))
        inits.append((make_ramp(n, 0.5), f"sqrt_ramp_n{n}"))
        inits.append((make_cosine_window(n, 0.5, 1.0), f"coswin_n{n}"))
        inits.append((make_mixed(n, 2, 0.3), f"mix_n{n}_f2"))
    
    # Evaluate all
    for a, label in inits:
        try_update(a, label)
    
    # Sort by score for prioritization
    print(f"  After Phase 0: best={best_score:.6f}")
    
    # === Phase 1: L-BFGS-B on parent and best init ===
    print("\n=== Phase 1: L-BFGS-B Optimization ===")
    remaining = time_budget - (time.monotonic() - t_global)
    
    # Build list of starting points to optimize
    opt_starts = []
    if parent is not None:
        opt_starts.append((parent / parent.sum(), "parent", 250))
    if best_seq is not None:
        opt_starts.append((best_seq / best_seq.sum(), "best_init", 200))
    
    # Add a few diverse starts
    if remaining > 100:
        opt_starts.append((make_ramp(5000, 1.0), "ramp5k", 150))
        opt_starts.append((np.ones(5000), "uniform5k", 150))
    
    seen_labels = set()
    for a_norm, label, t_alloc in opt_starts:
        if label in seen_labels:
            continue
        seen_labels.add(label)
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t_alloc = min(t_alloc, remaining * 0.25)
        print(f"  L-BFGS-B [{label}] n={len(a_norm)} (t={t_alloc:.0f}s)...")
        try:
            a_out, _ = lgbfgs_optimize(a_norm, t_alloc, rng_seed=hash(label) % 10000, verbose=True)
            try_update(a_out, f"lgbfgs_{label}")
        except Exception as e:
            print(f"    Error: {e}")
    
    # === Phase 2: Mirror Descent ===
    print("\n=== Phase 2: Mirror Descent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    
    if best_seq is not None and remaining > 60:
        # Run MD on current best
        t2 = min(remaining * 0.2, 200.0)
        a_in = best_seq / best_seq.sum()
        print(f"  MD on best (n={len(a_in)}, t={t2:.0f}s)...")
        try:
            a_out, _ = mirror_descent(a_in, t2, n_peaks=60, lr0=5.0,
                                      restart_every=2000, rng_seed=42, verbose=False)
            try_update(a_out, "md_best")
        except Exception as e:
            print(f"    Error: {e}")
        
        # MD on parent if different
        if parent is not None and len(parent) != len(best_seq):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40:
                t2b = min(remaining * 0.15, 120.0)
                a_in_p = parent / parent.sum()
                print(f"  MD on parent (n={len(a_in_p)}, t={t2b:.0f}s)...")
                try:
                    a_out, _ = mirror_descent(a_in_p, t2b, n_peaks=50, lr0=4.0,
                                              restart_every=3000, rng_seed=7, verbose=False)
                    try_update(a_out, "md_parent")
                except Exception as e:
                    print(f"    Error: {e}")
    
    # === Phase 3: Spectral Flattening ===
    print("\n=== Phase 3: Spectral Flattening ===")
    remaining = time_budget - (time.monotonic() - t_global)
    
    if best_seq is not None and remaining > 30:
        t3 = min(remaining * 0.1, 100.0)
        a_in = best_seq / best_seq.sum()
        print(f"  Spectral (n={len(a_in)}, t={t3:.0f}s)...")
        try:
            a_out, _ = spectral_flatten(a_in, t3, verbose=False)
            try_update(a_out, "spectral")
        except Exception as e:
            print(f"    Error: {e}")
    
    # === Phase 4: L-BFGS-B refinement with tighter temperature ===
    print("\n=== Phase 4: Fine L-BFGS-B ===")
    remaining = time_budget - (time.monotonic() - t_global)
    
    if best_seq is not None and remaining > 30:
        t4 = min(remaining * 0.2, 180.0)
        a_in = best_seq / best_seq.sum()
        print(f"  Fine L-BFGS-B (n={len(a_in)}, t={t4:.0f}s)...")
        try:
            a_out, _ = lgbfgs_optimize(a_in, t4, rng_seed=99, verbose=True)
            try_update(a_out, "fine_lgbfgs")
        except Exception as e:
            print(f"    Error: {e}")
    
    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-length ===")
    remaining = time_budget - (time.monotonic() - t_global)
    
    if best_seq is not None:
        n_best = len(best_seq)
        for new_n in [n_best + 1000, n_best - 1000, n_best + 2000, max(2000, n_best - 2000)]:
            if new_n <= 100 or new_n > 15000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            a_interp = interpolate_to_length(best_seq, new_n)
            try_update(a_interp, f"interp_n{new_n}")
            
            # Quick MD on interpolated
            t5 = min(remaining * 0.15, 100.0)
            if t5 > 15:
                try:
                    a_out, _ = mirror_descent(a_interp, t5, n_peaks=40, lr0=3.0,
                                              restart_every=3000, rng_seed=new_n % 100, verbose=False)
                    try_update(a_out, f"md_n{new_n}")
                except Exception:
                    pass
    
    # === Phase 6: Final combined polish ===
    print("\n=== Phase 6: Final Polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    
    if best_seq is not None and remaining > 30:
        a_in = best_seq / best_seq.sum()
        
        # Spectral + L-BFGS-B
        t6a = min(remaining * 0.2, 60.0)
        t6b = min(remaining * 0.3, 120.0)
        
        if t6a > 10:
            try:
                a_out, _ = spectral_flatten(a_in, t6a, verbose=False)
                try_update(a_out, "spec_final")
                if safe_eval(a_out) < best_score:
                    a_in = a_out / a_out.sum()
            except Exception:
                pass
        
        remaining = time_budget - (time.monotonic() - t_global)
        if t6b > 10 and remaining > 20:
            try:
                a_out, _ = lgbfgs_optimize(a_in, t6b, rng_seed=555, verbose=True)
                try_update(a_out, "lgbfgs_final")
            except Exception:
                pass
        
        # One more MD
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20 and best_seq is not None:
            t6c = min(remaining * 0.3, 80.0)
            try:
                a_out, _ = mirror_descent(best_seq / best_seq.sum(), t6c,
                                          n_peaks=80, lr0=3.0,
                                          restart_every=5000, rng_seed=777, verbose=False)
                try_update(a_out, "md_final")
            except Exception:
                pass
    
    # === Final validation ===
    if best_seq is not None:
        fs = safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\n{'='*50}")
        print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
        print(f"{'='*50}")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        best_seq = np.ones(5000)
        best_score = safe_eval(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
