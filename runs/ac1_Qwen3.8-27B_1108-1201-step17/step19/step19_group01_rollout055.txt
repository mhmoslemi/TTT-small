def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def compute_smoothed_grad(y, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*R(k))) w.r.t. a.
    Returns (grad, max_b, p_weights)."""
    n = len(y)
    b = fast_autocorr(y)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros(len(b))
        p[np.argmax(b)] = 1.0
    else:
        p = exp_vals / total

    # Gradient: (1/beta) * sum_k p[k] * grad_R(k)
    # grad_R(k)[i] = 2*a[k-i] if 0<=k-i<n else 0
    # Sum over k: sum_k p[k]*2*a[k-i] = 2*sum_k p[k]*a[k-i]
    # = 2 * conv(a_flip, p)[n-1 : 2n-1] (verified analytically)
    a_flip = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Af = np.fft.rfft(a_flip, fft_size)
    Pf = np.fft.rfft(p, fft_size)
    conv_full = np.fft.irfft(Af * Pf, fft_size)[:conv_len]
    grad = 2.0 * conv_full[n - 1:2 * n - 1] / beta

    return grad, b_max, p

def nesterov_backtrack(a0, time_limit, beta_start=20.0, beta_end=1200.0,
                       beta_growth=1.008, max_iter=None,
                       restart_every=2500, verbose=True):
    """Nesterov accelerated projected gradient with backtracking line search."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(12345)
    max_iter = max_iter or 100000000

    for t in range(1, max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Compute gradient
        grad, b_max, p = compute_smoothed_grad(y, beta)

        # Backtracking line search
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Initial step size estimate
        eta = min(1.0 / (beta * max(p) * 4.0), 1.0 / (2.0 * gnorm + 1.0))
        eta = min(eta, 0.5)

        # Simple descent check (not full backtracking for speed, but adaptive)
        z = y - eta * grad
        x_new = simplex_project(z)

        # Check for improvement; if not, halve step
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        if max_new > b_max + 0.05:
            # Step was too large, try smaller
            for _ in range(3):
                eta *= 0.5
                z = y - eta * grad
                x_new = simplex_project(z)
                b_new = fast_autocorr(x_new)
                max_new = b_new.max()
                if max_new <= b_max + 0.05:
                    break

        if max_new < best_max_b - 1e-14:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        # Adaptive beta increase
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        # Restart mechanism
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            # Structured restart: target the peak lags
            noise_scale = 0.015 / (1.0 + 0.05 * (t // restart_every))
            pert = rng.normal(0, noise_scale, n) * best_a
            # Add a small "anti-peak" component
            ac_best = fast_autocorr(best_a)
            peak_lags = np.argsort(ac_best)[::-1][:5]
            for k in peak_lags:
                k_i = int(k)
                # Create small perturbation that reduces correlation at lag k_i
                for j in range(n):
                    partner = k_i - j
                    if 0 <= partner < n:
                        pert[j] -= 0.001 * best_a[partner]
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 5.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest restart t=%d (%.1fs): J=%.6f" % (t, elapsed, J))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=%d (%.1fs): J=%.6f, beta=%.1f, eta=%.4f" % (t, elapsed, J, beta, eta))

    return best_a, best_max_b

def lp_fw(a0, time_limit, solver, initial_K=500, max_K=4000,
          add_per_iter=40, verbose=True):
    """LP-Frank-Wolfe with adaptive active set."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 800

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: converged at iter %d" % t)
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set: keep lags within tolerance of max
        tol = M_curr * 0.01
        all_lags = np.where(ac >= M_curr - tol)[0]
        new_active = set(int(k) for k in all_lags)
        # Also keep some of the previously active
        for k in list(active_set):
            if k not in new_active and len(new_active) < max_K:
                new_active.add(k)
        # Add top violators
        sorted_lags = np.argsort(ac)[::-1]
        for k in sorted_lags[:initial_K]:
            if len(new_active) >= max_K:
                break
            new_active.add(int(k))
        active_set = new_active

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix (sparse)
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # Solve LP
        try:
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:max(100, initial_K // 2)])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            # Shrink active set and retry
            if len(active_set) > 200:
                current_ac = fast_autocorr(y)
                active_set = set(int(k) for k in np.argsort(current_ac)[::-1][:200])
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations to add
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr * 1.001 + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Line search: geometric
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = fast_autocorr((1 - best_alpha) * y + best_alpha * g_dir).max()
        for _ in range(5):
            trial_alpha = best_alpha * 0.7
            trial = (1.0 - trial_alpha) * y + trial_alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                break
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = trial_alpha
            else:
                break

        if t <= 2:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y
        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=%d (%.1fs): J=%.6f |S|=%d" % (t, elapsed, J, len(active_set)))

    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=True):
    """Peak-targeted coordinate descent with pair moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(4242)
    stall = 0
    max_stall = 600

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:min(30, len(ac))]
        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_idx_in = np.argmax(contrib)
            src = int(idx[max_idx_in])
            partner = k_int - src

            # Find best destination (low autocorrelation contribution)
            # Compute local "cost" of adding mass at each position
            cand_positions = np.random.choice(n, size=min(200, n), replace=False)
            best_dst = -1
            best_trial_M = best_m

            for dst in cand_positions:
                if dst == src and dst == partner:
                    continue
                amt = a[src] * 0.03
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_trial_M:
                    best_trial_M = M_trial
                    best_dst = int(dst)

            if best_dst >= 0 and best_trial_M < best_m - 1e-15:
                amt = a[src] * 0.03
                a[src] -= amt
                a[best_dst] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                a = a / s_t
                best_m = best_trial_M
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1
            if stall % 50 == 0:
                # Broader random exploration
                for _ in range(3):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (%.1fs): J=%.6f stall=%d" % (elapsed, J, stall))

    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    if old_n == 1:
        return np.ones(new_n) / new_n
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def parametric_init(n, time_limit, best_init, best_score):
    """Try various parametric families for good initialization."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score
    rng = np.random.default_rng(999)

    candidates = []
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 4.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    # Symmetric
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p * (n - i_arr) ** p)
        candidates.append(x ** p + (1.0 - x) ** p)
    # Gaussian
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.08, 0.1, 0.15, 0.2, 0.3]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    # Cosine modulated
    for freq in [1, 2, 3, 4]:
        for amp in [0.3, 0.5, 0.7]:
            candidates.append(np.maximum((1.0 + amp * np.cos(2.0 * np.pi * freq * x)) / (1.0 + amp), 0.001))
    # Sine powers
    for p in [1, 2, 3, 4, 5, 6]:
        candidates.append(np.sin(np.pi * x) ** p)
    # Product of sines
    candidates.append(np.sin(np.pi * x) * np.sin(2 * np.pi * x))
    candidates.append(np.maximum(np.sin(np.pi * x) * (1 + 0.3 * np.cos(4 * np.pi * x)), 0.001))

    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(a_raw)
        b = fast_autocorr(a)
        m = b.max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    # Random initializations
    for _ in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        alpha = rng.uniform(0.3, 3.0)
        a = norm_sum1(rng.random(n) ** alpha)
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    return best_a, best_val

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [%s]: %.8f (n=%d, t=%.1fs)" % (label, sc, len(a), elapsed))
    return best_seq, best_score

def propose_candidate(budget_s=1800.0):

    print("Time budget: %.1fs" % time_budget)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=%d, score=%.8f" % (len(parent), p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=%d, score=%.8f" % (len(best_seq), best_score))

    n0 = len(best_seq)

    # Phase 1: Parametric search (quick)
    print("\n=== Phase 1: Parametric (n=%d) ===" % n0)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 40.0)
    if t1 > 10:
        a_param, _ = parametric_init(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "param", t_global)

    # Phase 2: Main Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov main (n=%d) ===" % n_best)
    t2 = min(remaining * 0.28, 450.0)
    if t2 > 60:
        a_out, _ = nesterov_backtrack(norm_sum1(best_seq), t2,
                                      beta_start=20.0, beta_end=800.0,
                                      beta_growth=1.008, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_main_n%d" % n_best, t_global)

    # Phase 3: LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP-FW (n=%d) ===" % n_best)
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 40 and n_best <= 8000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t3, solver,
                         initial_K=600, max_K=3500,
                         add_per_iter=40, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n%d" % n_best, t_global)

    # Phase 4: Nesterov round 2 (different params)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Nesterov round 2 (n=%d) ===" % n_best)
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 50:
        # Start from a perturbed version to escape local minima
        rng = np.random.default_rng(555)
        perturbed = norm_sum1(best_seq + rng.normal(0, 0.01, n_best) * best_seq)
        a_out, _ = nesterov_backtrack(perturbed, t4,
                                      beta_start=40.0, beta_end=1000.0,
                                      beta_growth=1.01, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2_n%d" % n_best, t_global)

    # Phase 5: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Coordinate (n=%d) ===" % n_best)
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 20:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t5, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n%d" % n_best, t_global)

    # Phase 6: Multi-length homotopy
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = [n_best + 500, n_best + 1000]
    longer_lengths = [l for l in longer_lengths if l <= 9000]

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print("\n=== Phase 6: Scale n=%d ===" % new_n)
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.35, 200.0)
        if t6a > 40:
            a_out, _ = nesterov_backtrack(a_long, t6a,
                                          beta_start=25.0, beta_end=600.0,
                                          beta_growth=1.009, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n%d" % new_n, t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 7500:
            t6b = min(remaining * 0.40, 200.0)
            if t6b > 30:
                a_long2 = interpolate_sequence(best_seq, new_n)
                a_out, _ = lp_fw(norm_sum1(a_long2), t6b, solver,
                                 initial_K=500, max_K=3000,
                                 add_per_iter=30, verbose=True)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n%d" % new_n, t_global)

    # Phase 7: Sharp phase (high beta)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Sharp (n=%d) ===" % n_best)
    t7 = min(remaining * 0.25, 200.0)
    if t7 > 30:
        a_out, _ = nesterov_sharp(norm_sum1(best_seq), t7,
                                  beta_start=400.0, beta_end=3000.0,
                                  beta_growth=1.02, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sharp_n%d" % n_best, t_global)

    # Phase 8: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final coord (n=%d) ===" % n_best)
    t8 = min(remaining * 0.30, 150.0)
    if t8 > 15:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t8, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Phase 9: Final LP burst if time remains
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 50 and n_best <= 8000:
        print("\n=== Phase 9: Final LP (n=%d) ===" % n_best)
        t9 = min(remaining * 0.60, 180.0)
        if t9 > 25:
            a_out, _ = lp_fw(norm_sum1(best_seq), t9, solver,
                             initial_K=700, max_K=4000,
                             add_per_iter=50, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Final sharp polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print("\n=== Phase 10: Final sharp polish (n=%d) ===" % n_best)
        t10 = min(remaining * 0.50, 100.0)
        if t10 > 10:
            a_out, _ = nesterov_sharp(norm_sum1(best_seq), t10,
                                      beta_start=800.0, beta_end=5000.0,
                                      beta_growth=1.03, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "sharp_final", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=%.8f, n=%d, time=%.1fs" % (best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=%.8f" % best_score)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
