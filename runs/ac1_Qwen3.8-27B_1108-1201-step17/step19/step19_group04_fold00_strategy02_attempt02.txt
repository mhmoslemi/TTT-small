class KendraPriorityHelper:
    """Helper class to set and get priority values with clear display."""
    
    def __init__(self, items):
        """
        Initialize with a list of items including their priority.

        :param items: List of dictionaries with 'title' and 'priority' keys.
        """
        self.items = items
        self.next_item = 0

    def set_priority(self, title, priority):
        """Set a priority for an item and display it."""
        self.items[self.next_item]['priority'] = priority
        print(f"Set priority for '{title}' to {priority}")
        self.next_item += 1
        return priority

    def get_priority(self, title):
        """Get the priority for an item if it exists."""
        for item in self.items:
            if item['title'] == title:
                return item['priority']
        return None


# Example usage
