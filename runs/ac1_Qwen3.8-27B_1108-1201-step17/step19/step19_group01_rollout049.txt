def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def norm_sum1(a):

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def nesterov_opt(a0, time_limit, beta_start=10.0, beta_end=800.0,
                 beta_growth=1.005, eta_base=1.0, top_k=30,
                 restart_every=2000, verbose=True, seed=42):
    """Nesterov accelerated projected gradient with top-k log-sum-exp smoothing."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, b, idx = compute_topk_grad(y, beta, top_k)

        # Adaptive step size
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            eta = eta_base / max(1.0, beta)
        else:
            eta = eta_base / max(1.0, beta * p.max() * 1.5 + g_norm * 0.5)
        eta = min(eta, 0.2)
        eta = max(eta, 0.01 / beta)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            # Diverse restart strategies
            strategy = rng.integers(4)
            if strategy == 0:
                # Gaussian noise
                noise = rng.normal(0, 0.03 / (1.0 + 0.05 * (t // restart_every)), n) * best_a
                x_r = best_a + noise
            elif strategy == 1:
                # Multiplicative perturbation
                log_pert = rng.normal(0, 0.05, n)
                x_r = best_a * np.exp(log_pert)
            elif strategy == 2:
                # Smooth kernel convolution
                kernel_size = rng.integers(5, 20)
                kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size / 4.0))
                kernel /= kernel.sum()
                x_r = np.convolve(best_a, kernel, mode='same')
            else:
                # Mix with parametric
                xx = np.linspace(0, 1, n)
                shape = rng.choice([
                    np.sin(np.pi * xx) ** 2,
                    np.exp(-2 * (xx - 0.5) ** 2),
                    (1 + np.cos(2 * np.pi * xx)) / 2,
                    xx * (1 - xx) * 4
                ])
                mix_t = rng.uniform(0.1, 0.3)
                x_r = (1 - mix_t) * best_a + mix_t * norm_sum1(shape)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 5.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("      Nest restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("      Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " beta=" + str(round(beta, 1)))

    return best_a, best_max_b


def lp_fw_opt(a0, time_limit, solver, initial_K=100, max_K=2000,
              add_per_iter=20, verbose=True):
    """LP-Frank-Wolfe with adaptive active set management."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    ac = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 2000

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("      LP-FW: converged iter " + str(t))
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Prune active set: keep top-K by margin
        constraint_list = list(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            # Keep constraints that are most binding (smallest margin = closest to M)
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = list(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix efficiently
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Reset active set on failure
            active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find new violations using FFT
        conv_ys = fast_conv_cross(y, g_dir)
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Also add any new top peaks not already in active set
        top_peaks = np.argsort(ac)[::-1][:initial_K]
        for k in top_peaks:
            if int(k) not in active_set:
                active_set.add(int(k))

        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = sorted(set([
            default_alpha,
            max(0.01, default_alpha * 0.5),
            max(0.01, default_alpha * 1.5),
            1.0 / (t + 1.0),
            min(1.0, default_alpha * 3.0)
        ]))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("      LP-FW iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def coord_polish(a0, time_limit, verbose=True, seed=123):
    """Aggressive peak-shaving coordinate descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:30]
        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_c_idx = np.argmax(contrib)
            src = int(idx[max_c_idx])
            partner = k_int - src

            # Find best destination: minimize increase in autocorrelation
            # Compute "pressure" from top peaks
            n_pressure = min(10, len(top_k))
            pressure = np.zeros(n)
            for k2 in top_k[:n_pressure]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2] * 0.5
                    pressure[k2i - idx2] += a[idx2] * 0.5

            # Exclude high-pressure positions
            threshold = np.percentile(pressure, 70)
            candidates = np.where(pressure <= threshold)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)
            candidates = np.setdiff1d(candidates, [src, partner])
            if len(candidates) == 0:
                continue

            cand_pressure = pressure[candidates]
            # Try several destinations
            top_candidates = candidates[np.argsort(cand_pressure)[:5]]

            found = False
            for dst in top_candidates:
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[int(dst)] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                if found:
                    break

            if not found and stall > 100:
                # Random exploration
                j = int(rng.integers(n))
                if j != src and j != partner and a[src] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 50 == 0:
                # Broader random moves
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("      Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " stall=" + str(stall))

    return best_a, best_m


def parametric_search(n, time_limit, best_init, best_score, seed=7):
    """Search over parametric families for good initialization."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score
    rng = np.random.default_rng(seed)

    candidates = []
    # Sine powers
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        candidates.append(np.sin(np.pi * x) ** p)
    # Gaussian bumps
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.25, 0.3]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    # Two bumps
    for c1 in [0.2, 0.3]:
        for c2 in [0.7, 0.8]:
            for sigma in [0.1, 0.15]:
                candidates.append(np.exp(-((x - c1) ** 2) / (2 * sigma ** 2)) +
                                   np.exp(-((x - c2) ** 2) / (2 * sigma ** 2)))
    # Cosine modulated
    for freq in [1, 2, 3, 4]:
        for amp in [0.3, 0.5, 0.7]:
            candidates.append(np.maximum(np.sin(np.pi * x) ** 2 * (1 + amp * np.cos(2 * np.pi * freq * x)), 0.01))
    # Power laws
    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p * (n - i_arr + 1.0) ** p)
    # Triangle variants
    for p in [1.0, 1.5, 2.0]:
        candidates.append(np.maximum(1.0 - np.abs(2 * x - 1) ** p, 0.0))

    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(a_raw)
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    # Random smooth functions
    for _ in range(100):
        if time.monotonic() - t0 > time_limit:
            break
        freq = rng.integers(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 0.8)
        base = np.sin(np.pi * x) ** 2
        mod = 1.0 + amp * np.cos(2 * np.pi * freq * x + phase)
        a_raw = np.maximum(base * mod, 0.01)
        a = norm_sum1(a_raw)
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    return best_a, best_val


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search: LP-FW + Nesterov + multi-n homotopy + coordinate polish."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(round(p_score, 6)))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(round(best_score, 6)))

    n0 = len(best_seq)

    # Phase 1: Parametric search (quick)
    print("\n=== Phase 1: Parametric (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 10:
        a_param, _ = parametric_search(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "param", t_global)

    # Phase 2: Nesterov warmup (find good basin)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov warmup (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t2,
                                beta_start=10.0, beta_end=200.0,
                                beta_growth=1.006, eta_base=1.0,
                                top_k=30, restart_every=1500,
                                verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_warm", t_global)

    # Phase 3: LP-FW (main optimizer)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP-FW (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.35, 500.0)
    if t3 > 60 and n_best <= 9000:
        a_out, _ = lp_fw_opt(norm_sum1(best_seq), t3, solver,
                             initial_K=150, max_K=2500,
                             add_per_iter=20, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_main_n" + str(n_best), t_global)

    # Phase 4: Nesterov polish (high beta)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Nesterov polish (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t4,
                                beta_start=100.0, beta_end=1500.0,
                                beta_growth=1.008, eta_base=0.8,
                                top_k=20, restart_every=1000,
                                verbose=True, seed=99)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_polish", t_global)

    # Phase 5: Multi-n exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Multi-n exploration ===")
    # Try a few nearby lengths
    n_candidates = []
    for delta in [-500, +500, +1000, -1000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 10000:
            n_candidates.append(new_n)
    # Ensure unique and sorted
    n_candidates = sorted(set(n_candidates))

    for new_n in n_candidates:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print("  Trying n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.35, 150.0)
        if t5a > 30:
            a_out, _ = nesterov_opt(a_long, t5a,
                                    beta_start=50.0, beta_end=500.0,
                                    beta_growth=1.006, eta_base=0.8,
                                    top_k=25, restart_every=1000,
                                    verbose=False, seed=new_n % 1000)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 100 and new_n <= 8000:
            t5b = min(remaining * 0.40, 180.0)
            if t5b > 40:
                a_long2 = interpolate_sequence(best_seq, new_n)
                a_out, _ = lp_fw_opt(a_long2, t5b, solver,
                                     initial_K=100, max_K=1500,
                                     add_per_iter=15, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(new_n), t_global)

    # Phase 6: Second LP-FW pass on best found so far
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: LP-FW pass 2 (n=" + str(n_best) + ") ===")
    if remaining > 80 and n_best <= 9000:
        t6 = min(remaining * 0.30, 250.0)
        if t6 > 40:
            a_out, _ = lp_fw_opt(norm_sum1(best_seq), t6, solver,
                                 initial_K=200, max_K=3000,
                                 add_per_iter=25, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_pass2", t_global)

    # Phase 7: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Coordinate polish (n=" + str(n_best) + ") ===")
    if remaining > 30:
        t7 = min(remaining * 0.25, 150.0)
        if t7 > 15:
            a_out, _ = coord_polish(norm_sum1(best_seq), t7, verbose=True, seed=777)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord", t_global)

    # Phase 8: Final Nesterov with very high beta
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final Nesterov (n=" + str(n_best) + ") ===")
    if remaining > 40:
        t8 = min(remaining * 0.40, 200.0)
        if t8 > 20:
            a_out, _ = nesterov_opt(norm_sum1(best_seq), t8,
                                    beta_start=200.0, beta_end=3000.0,
                                    beta_growth=1.01, eta_base=0.6,
                                    top_k=15, restart_every=800,
                                    verbose=True, seed=2024)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

    # Phase 9: Final coordinate
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t9 = min(remaining * 0.50, 100.0)
        if t9 > 10:
            a_out, _ = coord_polish(norm_sum1(best_seq), t9, verbose=False, seed=31415)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
