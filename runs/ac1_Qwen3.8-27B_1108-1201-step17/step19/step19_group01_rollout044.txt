def simplex_project(z):
    """Project onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

    # The sum_k p_k * y[k-j] is a convolution: (p * y_flip)[j] where y_flip[i] = y[n-1-i]
    # Actually: sum_k p_k * y[k-j] = sum_k p_k * y[k-j]
    # Let l = k-j, then k = l+j: sum_l p[l+j] * y[l]
    # This is cross-correlation of p and y, or equivalently (p_rev * y)[j + n - 1] or similar.
    
    # Easiest: just compute it via FFT
    # sum_k p_k * y[k-j] for j = 0,...,n-1
    # This is the cross-correlation <p, shift_j(y)> which equals (p * y_rev)[j]
    # where y_rev[i] = y[n-1-i]
    
    # Actually the simplest: define c = p (length 2n-1, with zeros padded)
    # Then sum_k p[k] * y[k-j] is the cross-correlation.
    # Let me just compute it as: for each j, sum_k p[k]*y[k-j] where 0<=k-j<n.
    # This is (p * y)[2n-2-j] if we pad appropriately... 
    
    # You know what, let me just use the formula from the previous code which worked:
    # grad = conv(n-1 : 2n-1) where conv = irfft(FFT(y_rev) * FFT(p))
    
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta * 2.0  # factor of 2 for symmetry
    
    return grad, p, b, idx

def nesterov_v2(a0, time_limit, beta_start=20.0, beta_end=2000.0,
                 beta_growth=1.005, eta_base=1.0, top_k_ratio=0.03,
                 restart_every=2000, verbose=True):
    """
    Nesterov accelerated projected gradient with adaptive top-k smoothing.
    Key improvements over v1:
    - Adaptive top_k based on peak spread
    - Better step size adaptation
    - Spectral awareness in restarts
    """
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(1234)
    
    # Adaptive top_k: start with a larger fraction, shrink over time
    top_k = max(50, int(n * top_k_ratio))
    min_top_k = max(20, int(n * 0.005))
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        if t % 500 == 0 and time.monotonic() - t0 > time_limit * 0.95:
            break  # Safety margin
        
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        # Compute gradient with adaptive top_k
        grad, p, b, idx = compute_grad_topk(y, beta, top_k)
        
        # Adaptive top_k: shrink if the top values are well-separated
        b_sorted = np.sort(b)[::-1]
        b_max_val = b_sorted[0]
        # Effective width: how many values are within 50% of max
        eff_width = np.sum(b_sorted > 0.5 * b_max_val)
        target_top_k = max(min_top_k, min(int(n * 0.1), int(eff_width * 1.5)))
        top_k = int(0.8 * top_k + 0.2 * target_top_k)  # Smooth adaptation
        
        # Step size: adaptive based on gradient norm
        grad_norm = np.linalg.norm(grad)
        eta = eta_base / max(1e-10, beta * p.max() * 2.0)
        eta = min(eta, 0.1)
        if grad_norm > 0:
            eta = min(eta, 0.5 * (1/best_max_b - 1/((2*len(b)-1))) / grad_norm)
        
        # Gradient step + projection
        z = y - eta * grad
        x_new = simplex_project(z)
        
        # Track best on true objective
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        # Adaptive beta increase
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        # Restart mechanism with spectral awareness
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            # Spectral restart: perturb in frequency domain
            noise_level = 0.015 / (1.0 + 0.05 * (t // restart_every))
            A = np.fft.rfft(best_a)
            # Add phase-preserving noise (modify amplitude slightly)
            amp = np.abs(A)
            amp = amp * (1 + rng.normal(0, noise_level, len(amp)))
            amp = np.maximum(amp, 0)
            A_new = amp * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, n)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                x = a_new
            else:
                # Time-domain noise fallback
                pert = rng.normal(0, noise_level, n) * best_a
                x = best_a + pert
                x = np.maximum(x, 1e-10)
                x = x / x.sum()
            x_prev = x.copy()
            beta = max(beta * 0.6, 10.0)
            top_k = max(min_top_k, int(n * top_k_ratio))
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest2 restart t=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J))
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest2 t=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J) + " beta=" + repr(beta) + " topk=" + str(top_k))
    
    return best_a, best_max_b


def lp_fw_v2(a0, time_limit, solver, initial_K=500, max_K=4000,
             add_per_iter=20, verbose=True):
    """
    LP-Frank-Wolfe with improved active set management and line search.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K])
    
    no_improve = 0
    max_no_improve = 1500
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if t % 200 == 0 and time.monotonic() - t0 > time_limit * 0.92:
            break
        if no_improve > max_no_improve:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Manage active set: keep tightest constraints
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K // 2])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violations
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))
        
        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5),
                  max(0.02, default_alpha * 1.5), 1.0 / (t + 1)]
        alphas = [max(0.01, min(1.0, a)) for a in set(alphas)]
        
        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y > 1e-15:
                y_new = y_new / s_y
            else:
                y_new = y.copy()
        
        y = y_new
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP2 iter=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J) + " |S|=" + str(len(active_set)))
    
    return best_a, best_m
