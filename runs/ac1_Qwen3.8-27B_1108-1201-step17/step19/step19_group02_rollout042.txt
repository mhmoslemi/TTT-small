"""Docstring"""
import numpy as np
import time
import math
from scipy import optimize

def next_pow2(m):
    if m <=1: return 1
    return 1 << (int(m)-1).bit_length()

def norm_sum1(a):
    a=np.asarray(a, dtype=np.float64)
    a=np.maximum(a,0.0)
    s=a.sum()
    if s < 1e-15:
       return np.ones_like(a)/a.size
    return a/s

def fft_square(a):
    a=np.asarray(a, dtype=np.float64)
    n=a.size
    size=next_pow2(2*n-1)
    A=np.fft.rfft(a, size)
    c=np.fft.irfft(A*A, size)
    return c[:2*n-1]

def score_norm(a):
    a=norm_sum1(a)
    c=fft_square(a)
    return float(2*a.size*c.max())

def soft_from_q(q, temp_frac=0.05):
    n = (len(q)+1)//2
    M=float(np.max(q))
    if M <= 0: return 0.0, M, q
    temp=M*temp_frac
    z=q/temp
    z-=z.max()
    # optionally avoid underflow
    ssum=float(np.exp(z).sum())
    soft=2*n*(M+temp*math.log(ssum))
    return soft, M, q

def evaluate_pair(a, temp_frac=0.05):
    a=norm_sum1(a)
    q=fft_square(a)
    soft,M, _=soft_from_q(q,temp_frac)
    exact=2*a.size*M
    return exact, soft, q, M

def evaluate_safe(a):
    try:
       vals=[float(x) for x in np.asarray(a)]
       v=float(evaluate_sequence(vals))
       return v
    except Exception:
       return float('inf')

"""
Search for nonnegative coefficients minimizing
2*n*max(conv(a,a))/(sum(a))**2.

The implementation uses:
- a fixed working length (usually 8192) for efficient FFT evaluation;
- parametric seeds and interpolation of the provided previous state of the art;
- low-dimensional global refinements using positive spline and Legendre bases;
- a custom high-dimensional soft Frank-Wolfe / water-filling search using
  exact convolution via FFT;
- an exact-constraint linear-programming Frank-Wolfe polish;
- short final scaling/length checks and validation.
"""

import numpy as np
import time
import math
from scipy import optimize


def next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def norm_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15 or not math.isfinite(s):
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def fft_square(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(A * A, size)
    return c[:2 * n - 1]


def score_norm(a):
    a = norm_sum1(a)
    if a.size == 0:
        return float("inf")
    c = fft_square(a)
    if c.size == 0:
        return float("inf")
    m = float(np.max(c))
    if not math.isfinite(m):
        return float("inf")
    return float(2.0 * a.size * m)


def soft_from_q(q, temp_frac=0.05):
    q = np.asarray(q, dtype=np.float64)
    if q.size == 0:
        return 0.0, 0.0
    m = float(np.max(q))
    if m <= 0.0 or not math.isfinite(m):
        return 0.0, m
    n = (q.size + 1) // 2
    temp = max(m * float(temp_frac), 1e-300)
    z = q / temp
    z = z - float(np.max(z))
    ssum = float(np.exp(z).sum())
    if ssum <= 0.0 or not math.isfinite(ssum):
        return float(2.0 * n * m), m
    return float(2.0 * n * (m + temp * math.log(ssum))), m


def evaluate_pair(a, temp_frac=0.05):
    a = norm_sum1(a)
    q = fft_square(a)
    soft, m = soft_from_q(q, temp_frac)
    exact = float(2.0 * a.size * m)
    return exact, soft, q, m


def evaluate_safe(a):

def interpolate_seq(f, new_n):
    f = norm_sum1(f)
    new_n = int(new_n)
    if f.size == 0:
        return np.ones(new_n, dtype=np.float64) / max(1, new_n)
    if f.size == new_n:
        return f.copy()
    x_old = np.linspace(0.0, 1.0, f.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f)
    return norm_sum1(g)


def parametric_best(n, start_a, start_score, max_time):

    best_a = start_a.copy()
    best_score = float(start_score)

    i = np.arange(n, dtype=np.float64)
    x = i / max(1.0, float(n - 1))

    powers = [0.0, 0.25, 0.5, 0.75, 1.0, 1.5, 2.0, 3.0]

def spline_sequence(v, n, knots, xg):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return np.ones(n, dtype=np.float64) / n
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    f = np.interp(xg, knots, v)
    return norm_sum1(f)


def spline_soft_fun(v, n, knots, xg, temp_frac=0.07):
    v = np.asarray(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return 1e8
    v = np.maximum(v, 0.0)
    if float(v.sum()) < 1e-12:
        return 1e8
    f = np.interp(np.asarray(xg, dtype=np.float64),
                  np.asarray(knots, dtype=np.float64),
                  v)
    p = norm_sum1(f)
    q = fft_square(p)
    soft, _ = soft_from_q(q, temp_frac)
    if not math.isfinite(soft):
        return 1e8
    return soft


def spline_fit_values(a, knots, xg):
    a = norm_sum1(a)
    n = int(a.size)
    f = float(n) * a
    return np.interp(np.asarray(knots, dtype=np.float64),
                     np.asarray(xg, dtype=np.float64),
                     f)

def make_legendre_modes(n, m):
    z = 2.0 * np.linspace(0.0, 1.0, int(n)) - 1.0
    modes = [np.ones_like(z), z.copy()]
    for k in range(2, int(m) + 1):
        term = ((2.0 * k - 1.0) * z * modes[-1] - (k - 1.0) * modes[-2]) / float(k)
        modes.append(term)
    return modes


def legendre_soft_fun(theta, n, modes, temp_frac=0.07):
    theta = np.asarray(theta, dtype=np.float64)
    if theta.size == 0 or not np.all(np.isfinite(theta)):
        return 1e8
    f = np.ones(int(n), dtype=np.float64)
    for j in range(theta.size):
        f = f + theta[j] * modes[j + 1]
    f = np.maximum(f, 1e-12)
    p = f / float(f.sum())
    q = fft_square(p)
    soft, _ = soft_from_q(q, temp_frac)
    if not math.isfinite(soft):
        return 1e8
    return soft

def sensitivity(a, w):
    n = int(a.size)
    L = 2 * n - 1
    size = next_pow2(3 * n - 2)
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    if a.size == 0 or w.size == 0:
        return np.zeros(n, dtype=np.float64)

    ar = np.fft.rfft(a, size)
    wr = np.zeros(size, dtype=np.float64)
    wr[:min(L, w.size)] = w[:min(L, w.size)][::-1]
    wrf = np.fft.rfft(wr, size)
    conv = np.fft.irfft(ar * wrf, size)
    c = conv[n - 1:L][::-1]
    c = np.where(np.isfinite(c), c, 0.0)
    return np.maximum(c, 0.0)


def make_direction(s, dtype, param, n):
    s = np.asarray(s, dtype=np.float64)
    n = int(n)
    if s.size != n or not np.all(np.isfinite(s)):
        return np.ones(n, dtype=np.float64) / n

    s = np.maximum(s, 0.0)
    smax = float(np.max(s))
    smin = float(np.min(s))
    span = smax - smin
    if span <= 1e-30:
        return np.ones(n, dtype=np.float64) / n

    if param is None:
        paramv = 0.7
    else:
        paramv = float(param)

    if dtype == 0:
        if paramv >= 1.0:
            tau = smax + 1e-8 * span
        else:
            tau = float(np.quantile(s, min(0.999, max(1e-6, paramv)))) + 1e-7 * span
        g = np.maximum(tau - s, 0.0)
    elif dtype == 1:
        reg = max(1e-8 * smax, 1e-18)
        if paramv > 0.0:
            reg *= max(1.0, paramv)
        g = 1.0 / (s + reg)
    elif dtype == 2:
        g = np.exp(-max(0.5, paramv) * (s - smin) / span)
    elif dtype == 3:
        tau = smax + 1e-6 * span
        g = np.maximum(tau - s, 0.0)
    else:
        g = np.ones(n, dtype=np.float64)

    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    g = g + 1e-12
    gs = float(g.sum())
    if gs <= 0.0 or not math.isfinite(gs):
        return np.ones(n, dtype=np.float64) / n
    return g / gs


def wf_search(a0, max_time, seed=101, verbose=False):
    a = norm_sum1(a0).copy()
    if a.size == 0:
        a = np.ones(4096, dtype=np.float64) / 4096.0
    n = int(a.size)

    best_a = a.copy()
    best_score, _, _, _ = evaluate_pair(a)

    configs = [
        (0.020, 0, 0.45),
        (0.035, 0, 0.60),
        (0.060, 0, 0.75),
        (0.100, 0, 0.88),
        (0.035, 3, None),
        (0.070, 1, 1.0),
        (0.100, 2, 2.0),
        (0.020, 3, None)
    ]
    alphas = [0.02, 0.05, 0.09, 0.16, 0.27, 0.45, 0.75, 0.95]

    rng = np.random.default_rng(seed)
    cidx = 0
    no_improve = 0
    restarts = 0
    total_evals = 0
    last_print = 0.0

    while time.monotonic() < max_time:
        if restarts > 16 and no_improve > 160:
            break

        cfg = configs[cidx % len(configs)]
        cidx += 1
        temp_frac, dtype, dparam = cfg

        cur_exact, cur_soft, q, M = evaluate_pair(a, 0.05)
        if cur_exact < best_score - 1e-13:
            best_score = cur_exact
            best_a = a.copy()

        if M <= 1e-30:
            break

        temp = max(M * float(temp_frac), 1e-300)
        z = q / temp
        z = z - float(np.max(z))
        w = np.exp(z)
        ws = float(w.sum())
        if ws <= 0.0 or not math.isfinite(ws):
            top = np.argsort(q)[-max(50, n // 100):]
            w = np.zeros_like(q)
            w[top] = 1.0
            ws = float(w.sum())
        if ws > 0.0:
            w = w / ws

        s = sensitivity(a, w)
        d = make_direction(s, dtype, dparam, n)

        best_trial = a
        best_trial_soft = cur_soft
        best_trial_exact = cur_exact
        best_alpha = 0.0

        for al in alphas:
            al = float(al)
            if al <= 0.0:
                continue
            t = a + al * (d - a)
            t = np.maximum(t, 0.0)
            ts = float(t.sum())
            if ts <= 1e-15:
                continue
            t = t / ts
            te, ts_soft, _, _ = evaluate_pair(t, 0.05)
            total_evals += 1
            if ts_soft < best_trial_soft - 1e-14:
                best_trial_soft = ts_soft
                best_trial_exact = te
                best_trial = t.copy()
                best_alpha = al

        if best_alpha > 0.0:
            cands = []
            if best_alpha < 0.98:
                cands.append(min(1.0, best_alpha * 1.45))
            if best_alpha > 0.03:
                cands.append(max(0.01, best_alpha * 0.62))
            if best_alpha < 0.9:
                cands.append(0.5 * (best_alpha + 1.0))
            seen = set([round(best_alpha, 6)])
            for al2 in cands:
                key = round(float(al2), 6)
                if key in seen or al2 <= 0.0 or al2 > 1.0:
                    continue
                seen.add(key)
                t = a + al2 * (d - a)
                t = np.maximum(t, 0.0)
                ts = float(t.sum())
                if ts <= 1e-15:
                    continue
                t = t / ts
                te, ts_soft, _, _ = evaluate_pair(t, 0.05)
                total_evals += 1
                if ts_soft < best_trial_soft - 1e-14:
                    best_trial_soft = ts_soft
                    best_trial_exact = te
                    best_trial = t.copy()
                    best_alpha = al2

        accepted = False
        if best_trial_soft < cur_soft - 1e-14:
            if best_trial_exact < cur_exact or best_trial_exact <= cur_exact * 1.015 + 1e-12:
                a = best_trial.copy()
                accepted = True
        elif best_trial_exact < cur_exact - 1e-14:
            a = best_trial.copy()
            accepted = True

        if accepted:
            if best_trial_exact < best_score:
                best_score = best_trial_exact
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 35 == 0:
                scale = 0.035 / (1.0 + 0.35 * float(restarts))
                noise = rng.normal(0.0, 1.0, n) * best_a
                a = norm_sum1(best_a + scale * noise)
                no_improve = 0
                restarts += 1
                if restarts > 20:
                    break

        if verbose and total_evals > 0 and (total_evals % 120 == 0 or time.monotonic() - last_print > 30.0):
            last_print = time.monotonic()
            print("  wf_search: evals=" + str(total_evals) +
                  " best=" + repr(best_score) +
                  " cur=" + repr(cur_exact))

    final_exact, _, _, _ = evaluate_pair(best_a)
    return best_a, float(final_exact), a


def lpfw_search(a0, max_time, Ks=(30, 60, 100), verbose=False):
    a = norm_sum1(a0).copy()
    if a.size == 0:
        a = np.ones(4096, dtype=np.float64) / 4096.0
    n = int(a.size)

    best_a = a.copy()
    best_score = score_norm(a)

    alphas = [0.02, 0.05, 0.10, 0.18, 0.32, 0.55, 0.80, 1.00]
    cidx = 0
    no_improve = 0
    rng = np.random.default_rng(2024)

    while time.monotonic() < max_time:
        if no_improve > 220:
            break
        if time.monotonic() > max_time - 2.0:
            break

        K = int(Ks[cidx % len(Ks)])
        cidx += 1
        if K <= 0 or K > n:
            K = max(10, min(100, n))

        q = fft_square(a)
        M = float(np.max(q))
        cur_exact = float(2.0 * n * M)
        if cur_exact < best_score - 1e-13:
            best_score = cur_exact
            best_a = a.copy()

        top = np.argsort(q)[-K:]
        A = np.zeros((K, n + 1), dtype=np.float64)
        idx = np.arange(n, dtype=np.int64)
        for row, kk in enumerate(top):
            kk = int(kk)
            st = max(0, kk - n + 1)
            en = min(n, kk + 1)
            if st < en:
                A[row, st:en] = a[kk - idx[st:en]]
            A[row, -1] = -1.0

        c = np.zeros(n + 1, dtype=np.float64)
        c[-1] = 1.0
        Aeq = np.zeros((1, n + 1), dtype=np.float64)
        Aeq[0, -1] = 1.0
        bounds = [(0.0, None)] * n + [(None, None)]

        try:
            res = optimize.linprog(
                c,
                A_ub=A,
                b_ub=np.zeros(K, dtype=np.float64),
                A_eq=Aeq,
                b_eq=np.array([1.0], dtype=np.float64),
                bounds=bounds,
                method="highs",
                options={"presolve": True, "duality_gap_tolerance": 1e-8}
            )
        except Exception:
            continue

        if res.status not in (0, 3):
            continue

        g = np.asarray(res.x[:n], dtype=np.float64)
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)
        gs = float(g.sum())
        if gs <= 1e-15:
            continue
        d = g / gs

        best_trial = a
        best_trial_score = cur_exact
        for al in alphas:
            al = float(al)
            if al <= 0.0:
                continue
            t = a + al * (d - a)
            t = np.maximum(t, 0.0)
            ts = float(t.sum())
            if ts <= 1e-15:
                continue
            t = t / ts
            sc = score_norm(t)
            if sc < best_trial_score:
                best_trial_score = sc
                best_trial = t.copy()

        if best_trial_score < cur_exact - 1e-14:
            a = best_trial.copy()
            no_improve = 0
            if best_trial_score < best_score:
                best_score = best_trial_score
                best_a = a.copy()
        else:
            no_improve += 1
            if no_improve % 60 == 0:
                scale = 0.025 / (1.0 + 0.3 * float(no_improve // 60))
                noise = rng.normal(0.0, 1.0, n) * best_a
                a = norm_sum1(best_a + scale * noise)
                no_improve = 0

        if verbose and cidx % 8 == 0:
            print("  lpfw: step=" + str(cidx) + " K=" + str(K) +
                  " best=" + repr(best_score))

    final_exact = score_norm(best_a)
    return best_a, float(final_exact), a


def try_length_variants(best_a, best_score, max_time):
    best_a = norm_sum1(best_a)
    best_score = float(best_score)
    base_n = int(best_a.size)
    variants = [base_n, base_n - 1024, base_n + 1024, 6144, 8192, 10000]
    seen = set()
    for nn in variants:
        nn = int(nn)
        if nn in seen:
            continue
        seen.add(nn)
        if nn < 2048 or nn > 11000:
            continue
        if time.monotonic() > max_time:
            break
        cand = interpolate_seq(best_a, nn)
        sc = score_norm(cand)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = cand.copy()
    return best_a, best_score


def propose_candidate(budget_s=1800.0):
    t0 = time.monotonic()
    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 1800.0
    total_budget = min(max(total_budget, 5.0), 1800.0)
    time_budget = max(3.0, total_budget - 75.0)

    n_main = 8192

    parent = None
    try:
        parent = list(height_sequence_1)
    except Exception:
        parent = None

    if parent is None or len(parent) == 0:
        best_a = np.ones(n_main, dtype=np.float64) / float(n_main)
        best_score = score_norm(best_a)
        print("Fallback uniform: score=" + repr(best_score))
    else:
        try:
            p = np.asarray(parent, dtype=np.float64)
            if p.size < 50:
                raise Exception("parent too short")
            p = norm_sum1(p)
            best_a = interpolate_seq(p, n_main)
            best_score = score_norm(best_a)
            print("Parent interpolated to n=" + str(n_main) + ": score=" + repr(best_score))
        except Exception:
            best_a = np.ones(n_main, dtype=np.float64) / float(n_main)
            best_score = score_norm(best_a)

    fs = evaluate_safe(best_a)
    if math.isfinite(fs) and fs < best_score:
        best_score = fs

    def elapsed():
        return time.monotonic() - t0

    def remaining():
        return max(0.0, time_budget - elapsed())

    # Phase 1: cheap parametric seeds at working length.
    if remaining() > 45.0:
        alloc = min(remaining() * 0.10, 90.0)
        print("\n=== Phase 1: parametric seeds (alloc=" + str(round(alloc, 1)) + "s) ===")
        best_a, best_score = parametric_best(n_main, best_a, best_score, t0 + alloc)
        print("After parametric: best=" + repr(best_score))

    # Phase 2: low-dimensional spline search.
    if remaining() > 90.0:
        alloc = min(remaining() * 0.25, 260.0)
        print("\n=== Phase 2: spline search r=10 (alloc=" + str(round(alloc, 1)) + "s) ===")
        best_a, best_score = spline_search(n_main, best_a, best_score, t0 + alloc, r=10, max_iter=90)
        print("After spline r=10: best=" + repr(best_score))

    if remaining() > 70.0:
        alloc = min(remaining() * 0.10, 100.0)
        print("\n=== Phase 3: spline search r=16 (alloc=" + str(round(alloc, 1)) + "s) ===")
        best_a, best_score = spline_search(n_main, best_a, best_score, t0 + alloc, r=16, max_iter=55)
        print("After spline r=16: best=" + repr(best_score))

    # Phase 4: optional Legendre-basis search.
    if remaining() > 70.0:
        alloc = min(remaining() * 0.07, 80.0)
        print("\n=== Phase 4: Legendre basis search (alloc=" + str(round(alloc, 1)) + "s) ===")
        best_a, best_score = legendre_search(n_main, best_a, best_score, t0 + alloc, m=8)
        print("After Legendre: best=" + repr(best_score))

    # Phase 5: main high-dimensional soft water-filling search.
    if remaining() > 80.0 and len(best_a) <= 11000:
        alloc = min(remaining() * 0.42, 540.0)
        print("\n=== Phase 5: wf_search (alloc=" + str(round(alloc, 1)) + "s) ===")
        best_a, _, _ = wf_search(best_a, t0 + alloc, seed=101)
        print("After wf_search: best=" + repr(best_score))

    # Phase 6: LP Frank-Wolfe polish.
    if remaining() > 60.0 and len(best_a) <= 11000:
        alloc = min(remaining() * 0.25, 260.0)
        print("\n=== Phase 6: LP-FW polish (alloc=" + str(round(alloc, 1)) + "s) ===")
        best_a, _, _ = lpfw_search(best_a, t0 + alloc, Ks=(30, 60, 100))
        print("After LP-FW polish: best=" + repr(best_score))

    # Phase 7: a second high-dimensional refinement from the best point.
    if remaining() > 50.0 and len(best_a) <= 11000:
        alloc = min(remaining() * 0.14, 120.0)
        print("\n=== Phase 7: wf_search again (alloc=" + str(round(alloc, 1)) + "s) ===")
        best_a, _, _ = wf_search(best_a, t0 + alloc, seed=202)
        print("After second wf_search: best=" + repr(best_score))

    # Phase 8: length variants.
    if remaining() > 20.0:
        print("\n=== Phase 8: length variants ===")
        best_a, best_score = try_length_variants(best_a, best_score, t0 + max(5.0, remaining() * 0.5))
        print("After length variants: best=" + repr(best_score))

    # Final validation and safe return.
    best_a = np.asarray(best_a, dtype=np.float64)
    best_a = np.where(np.isfinite(best_a), best_a, 0.0)
    best_a = np.maximum(best_a, 0.0)
    best_score = score_norm(best_a)

    fs = evaluate_safe(best_a)
    if math.isfinite(fs) and fs < best_score:
        best_score = fs

    if not math.isfinite(best_score):
        best_a = np.ones(n_main, dtype=np.float64) / float(n_main)
        best_score = score_norm(best_a)
        fs = evaluate_safe(best_a)
        if math.isfinite(fs) and fs < best_score:
            best_score = fs

    final_result = [float(x) for x in best_a]
    print("\nFINAL: score=" + repr(best_score) +
          ", n=" + str(len(final_result)) +
          ", elapsed=" + str(round(time.monotonic() - t0, 1)) + "s")
    return final_result
