Let me implement a version of this:
1. Start with a non-negative sequence a
2. Compute A(ω) = DFT(a)
3. Compute C(ω) = A(ω)^2 (the spectrum of the convolution)
4. Compute c = IDFT(C(ω)) (should equal a*a)
5. Modify C(ω) to reduce peaks: C'(ω) = C(ω) * (1 - λ * (c_max - c) / c_max) or something
6. Recover A'(ω) = sqrt(C'(ω)) (taking the principal square root)
7. Recover a' = IDFT(A'(ω))
8. Project a' onto non-negative orthant: a'' = max(a', 0)
9. Normalize: a'' = a'' / sum(a'')
10. Repeat

            w = p[i] if top_k < len(b) else p[k_int]
            js = max(0, k_int - n + 1)
            je = min(n, k_int + 1)
            if js < je:
                idx_j = np.arange(js, je)
                grad[idx_j] += w * y[k_int - idx_j]

        grad /= beta

        eta = eta_base / max(1.0, beta * 0.1)
        eta = min(eta, 0.1)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_convolve(x_new)
        max_new = b_new.max()
        if max_new < best_m - 1e-14:
            best_m = max_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 100 == 0:
            beta = min(beta * growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.015
            pert = rng.normal(0, noise, n) * best_x
            x_r = best_x + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Nest restart t={t} ({elapsed:.1f}s): J={2*n*best_m:.6f}")

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nest t={t} ({elapsed:.1f}s): J={2*n*best_m:.6f}, beta={beta:.1f}")

    return best_x, best_m

def spectral_correct(a0, time_limit, n_iters=500, verbose=True):
    """
    Spectral correction: iteratively modify the frequency-domain representation
    to flatten the convolution, then project back to non-negative time domain.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute spectrum
        fft_size = 1 << (2 * n - 2).bit_length()
        A = np.fft.rfft(a, fft_size)
        C_spec = A * A  # Spectrum of convolution

        # Compute convolution and identify peaks
        c = np.fft.irfft(C_spec, fft_size)[:2*n-1]
        c_max = c.max()

        if c_max < best_m - 1e-14:
            best_m = c_max
            best_a = a.copy()

        # Create correction: reduce components that contribute to peaks
        # Inverse: find time-domain "penalty" from peaks
        # Modify C_spec to reduce peak contribution
        alpha = 0.05  # correction strength

        # Identify peak lags
        n_peaks = min(20, 2 * n - 1)
        peak_lags = np.argsort(c)[::-1][:n_peaks]

        # For each peak lag k, the contribution to C_spec is uniform
        # (since c_k = IDFT(C_spec)[k]), so we reduce C_spec uniformly
        # But we want to reduce it more where the phase aligns with peaks
        # Simple approach: multiply C_spec by (1 - alpha * peak_indicator_in_freq)
        # The peak indicator in frequency is just a constant (since peaks are in time domain)
        # Better: reduce C_spec where |A| is large (since those contribute most to c)

        # Simple spectral flattening: reduce high-magnitude spectral components
        mag_A = np.abs(A)
        mag_max = mag_A.max()
        if mag_max > 1e-15:
            # Soft threshold: reduce the top spectral components
            threshold = mag_max * 0.7
            reduction = np.ones_like(mag_A)
            mask = mag_A > threshold
            reduction[mask] = 1.0 - alpha * (mag_A[mask] - threshold) / mag_max
            # Apply correction (keeping phase)
            phase_A = np.angle(A)
            new_mag = mag_A * reduction
            A_corrected = new_mag * np.exp(1j * phase_A)

            # Recover time domain
            a_trial = np.fft.irfft(A_corrected, fft_size)
            a_trial = a_trial[:n]  # Take first n points (the original sequence)
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-15:
                a_trial = a_trial / s
                m_trial = fast_convolve(a_trial).max()
                if m_trial < best_m:
                    a = a_trial
                    # Don't update best here, let the main loop handle it
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral it={it} ({elapsed:.1f}s): J={2*n*best_m:.6f}")

    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=True):
    """Coordinate descent with pair-swap and targeted moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(314)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 1500:
        c = fast_convolve(a)
        c_max = c.max()
        # Top contributing lags
        n_top = min(30, 2 * n - 1)
        top_lags = np.argsort(c)[::-1][:n_top]

        improved = False

        # Strategy 1: Targeted reduction at peak-contributing positions
        for lag in top_lags[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(lag)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-20:
                continue

            # Find the position with highest contribution
            src = int(idx[np.argmax(contrib)])
            partner = k - src

            # Reduce src, add to a low-pressure position
            # Compute pressure at each position
            pressure = np.zeros(n)
            for lag2 in top_lags[:8]:
                k2 = int(lag2)
                s2 = max(0, k2 - n + 1)
                e2 = min(n - 1, k2)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2 - idx2]
                    pressure[k2 - idx2] += a[idx2]

            # Find low-pressure destination
            excl = set([src, partner]) if 0 <= partner < n else set([src])
            available = [i for i in range(n) if i not in excl]
            if not available:
                continue
            avail_arr = np.array(available)
            dst = int(avail_arr[np.argmin(pressure[avail_arr])])

            # Try different fractions
            for frac in [0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_convolve(a_trial).max()
                if m_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if improved:
                break  # Restart the outer loop with updated a

        if not improved:
            # Strategy 2: Pair swap
            stall += 1
            if stall % 10 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j:
                        continue
                    # Swap a small amount
                    amt = min(a[i], a[j]) * 0.1
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_convolve(a_trial).max()
                    if m_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

            if stall % 100 == 0 and verbose:
                elapsed = time.monotonic() - t0
                print(f"    Coord ({elapsed:.1f}s): J={2*n*best_m:.6f}, stall={stall}")

    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def parametric_search(n, time_limit, best_init, best_m):
    """Quick parametric search for good initialization."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_m

    candidates = []
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - 1.0 - i_arr) ** p)
    for p in [0.5, 1.0, 2.0]:
        candidates.append(((i_arr + 1.0) ** p) * ((n - i_arr) ** p))
    for c in [0.25, 0.5, 0.75]:
        for sigma in [0.1, 0.15, 0.25]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    for freq in [1, 2, 3, 5]:
        candidates.append(np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x)) / 2.0, 0.01))
    for p in [1, 2, 3, 4, 6]:
        candidates.append(np.sin(np.pi * x) ** p)
    # Asymmetric
    candidates.append(x ** 0.5 + (1-x) ** 0.3)
    candidates.append(x ** 2 + 0.5 * (1-x))
    candidates.append(1.0 / (x + 0.1))
    candidates.append(1.0 / (1.1 - x))

    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(a_raw)
        m = fast_convolve(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    rng = np.random.default_rng(7)
    for _ in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        # Random smooth function
        coef = rng.normal(0, 1, min(n, 50))
        t = np.linspace(0, 1, n)
        a_raw = np.zeros(n)
        for k, c in enumerate(coef):
            a_raw += c * np.cos(k * np.pi * t)
        a_raw = np.maximum(a_raw, 0)
        a = norm_sum1(a_raw)
        m = fast_convolve(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    return best_a, best_val

def propose_candidate(budget_s=1800.0):
    """
    Main search: multi-phase optimization combining LP-FW, Nesterov,
    spectral correction, and coordinate descent with multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.1f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # Phase 1: Parametric search
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 5:
        print(f"\n=== Phase 1: Parametric (n={n0}) ===")
        a_param, m_param = parametric_search(n0, t1, best_seq, best_seq.sum()**2 / (2*n0*fast_convolve(best_seq).max()))
        best_seq, best_score = try_update(best_seq, best_score, a_param, "param", t_global)

    # Phase 2: LP-FW (main workhorse, long run)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP-FW main (n={n_cur}) ===")
    t2 = min(remaining * 0.35, 550.0)
    if t2 > 60:
        a_out, _ = lp_fw_core(norm_sum1(best_seq), t2, solver,
                              initial_K=1000, max_K=3000,
                              margin_frac=0.02, restart_every=1200, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"lp1_n{n_cur}", t_global)

    # Phase 3: Nesterov (smooth exploration)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Nesterov (n={n_cur}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t3,
                                beta_start=30.0, beta_end=600.0,
                                growth=1.006, eta_base=0.6,
                                top_k_ratio=0.03, restart_every=2000,
                                verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"nest_n{n_cur}", t_global)

    # Phase 4: LP-FW (second pass, refined)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LP-FW refined (n={n_cur}) ===")
    t4 = min(remaining * 0.20, 350.0)
    if t4 > 40:
        a_out, _ = lp_fw_core(norm_sum1(best_seq), t4, solver,
                              initial_K=1500, max_K=4000,
                              margin_frac=0.01, restart_every=1000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"lp2_n{n_cur}", t_global)

    # Phase 5: Turn over to the mentor process
    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
