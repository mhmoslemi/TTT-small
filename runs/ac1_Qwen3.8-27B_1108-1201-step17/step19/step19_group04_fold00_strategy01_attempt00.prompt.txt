import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]


def conv_ab(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]


def corr_shift(w, a):
    """
    Returns c[j] = sum_m w[m] * a[m-j] for j = 0..n-1.
    This is the first n entries of the cross-correlation of w and a.
    """
    w = np.asarray(w, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = w.size
    size = next_pow2(L + n - 1)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fw * fa.conj(), size)[:n]
    return np.maximum(c, 0.0)


def objective_norm(a):

    q = self_conv(a)
    return 2.0 * float(a.size) * float(np.max(q))

def weights_from_q(q, focus=4.0):
    q = np.asarray(q, dtype=np.float64)
    m = float(q.max())
    if m <= 1e-15:
        return np.ones_like(q, dtype=np.float64) / float(q.size)
    scale = 0.5 * m
    gamma = focus / scale
    vals = q - m
    vals = gamma * vals
    vals = np.clip(vals, -50.0, 0.0)
    e = np.exp(vals)
    s = float(e.sum())
    if s <= 1e-300:
        return np.ones_like(q, dtype=np.float64) / float(q.size)
    return e / s


def quad_max(alpha, q0, c, qd):
    a1 = 1.0 - alpha
    val = (a1 * a1) * q0 + (2.0 * alpha * a1) * c + (alpha * alpha) * qd
    return float(np.max(val))


def golden_alpha_max(q0, c, qd, a, b, iters=35):
    phi = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = b - phi * (b - a)
    d1 = a + phi * (b - a)
    f1 = quad_max(c1, q0, c, qd)
    f2 = quad_max(d1, q0, c, qd)
    for _ in range(iters):
        if f1 < f2:
            b = d1
            d1 = c1
            f2 = f1
            c1 = b - phi * (b - a)
            f1 = quad_max(c1, q0, c, qd)
        else:
            a = c1
            c1 = d1
            f1 = f2
            d1 = a + phi * (b - a)
            f2 = quad_max(d1, q0, c, qd)
    return 0.5 * (a + b)


def line_search_direction(a, d, q0, T=None):
    """
    Minimize max q((1-alpha)a + alpha d) over alpha in [0,1].
    Uses exact quadratic form in alpha after FFT cross convolution.
    """
    a = np.asarray(a, dtype=np.float64)
    d = np.asarray(d, dtype=np.float64)
    q0 = np.asarray(q0, dtype=np.float64)
    M0 = float(np.max(q0))

    dot = float(np.dot(a, d))
    if dot > 1.0 - 1e-7:
        return a.copy(), M0

    c = conv_ab(a, d)
    qd = self_conv(d)

    best_alpha = 0.0
    best_M = M0

    alphas = [1e-4, 5e-4, 1e-3, 2e-3, 5e-3, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 0.9, 1.0]

    if T is not None and np.isfinite(T) and T < 0.0 and M0 > 1e-15:
        t1 = min(0.30, max(5e-4, 0.5 * (-float(T)) / M0))
        alphas.append(max(1e-5, 0.5 * t1))
        alphas.append(min(0.95, 1.5 * t1))

    uniq = []
    seen = set()
    for al in alphas:
        al = float(min(1.0, max(0.0, al)))
        key = round(al, 12)
        if key not in seen:
            seen.add(key)
            uniq.append(al)

    for al in uniq:
        val = quad_max(al, q0, c, qd)
        if val < best_M:
            best_M = val
            best_alpha = al

    if best_M < M0 - 1e-13 and 0.0 < best_alpha < 1.0:
        al2 = golden_alpha_max(q0, c, qd, 0.0, 1.0, 32)
        val2 = quad_max(al2, q0, c, qd)
        if val2 < best_M:
            best_M = val2
            best_alpha = al2

    if best_alpha <= 0.0 or best_M >= M0 - 1e-14:
        return a.copy(), M0

    new = (1.0 - best_alpha) * a + best_alpha * d
    new = np.maximum(new, 0.0)
    s = float(new.sum())
    if s <= 1e-15:
        return a.copy(), M0
    new = new / s
    return new, best_M


def get_lp_artifacts(n):
    art = _LP_CACHE.get(n)
    if art is not None:
        return art
    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0
    Aeq = np.zeros((1, n + 1), dtype=np.float64)
    Aeq[0, :n] = 1.0
    bounds = [(0.0, np.inf)] * n + [(None, None)]
    art = (c, Aeq, bounds)
    if len(_LP_CACHE) > 6:
        _LP_CACHE.clear()
    _LP_CACHE[n] = art
    return art


def solve_lp_offset(y, active, q_active):
    """
    Solve:
       min T
       s.t. sum g = 1,
            g >= 0,
            2*(y*g)[k] - T <= q_y[k], for active lags k.
    """
    if not HAVE_LP:
        return None, None
    y = np.asarray(y, dtype=np.float64)
    active = np.asarray(active, dtype=int)
    q_active = np.asarray(q_active, dtype=np.float64)
    n = y.size
    K = active.size
    if K <= 0:
        return None, None
    if K * (n + 1) > 9_000_000:
        return None, None

    try:
        A = np.zeros((K, n + 1), dtype=np.float64)
    except Exception:
        return None, None

    yy = 2.0 * y
    for r, k in enumerate(active):
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            A[r, i0:i1] = yy[start:stop][::-1]

    A[:, n] = -1.0
    b = np.maximum(q_active, 0.0)

    c, Aeq, bounds = get_lp_artifacts(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq,
                      b_eq=np.array([1.0], dtype=np.float64),
                      bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception:
        return None, None

    status = getattr(res, 'status', None)
    success = getattr(res, 'success', False)
    x = getattr(res, 'x', None)
    if (status != 0 or not success) or x is None:
        return None, None

    g = np.asarray(x[:n], dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-10:
        return None, None
    g = g / s
    T = float(x[n])
    if not np.isfinite(T):
        T = 0.0
    return g, T


def soft_direction_best(a, q, current_score, max_dirs=4):
    """
    Try a few softmax-gradient mirror directions and return best line-search result.
    """
    a = np.asarray(a, dtype=np.float64)
    q = np.asarray(q, dtype=np.float64)
    n = a.size
    if n <= 1:
        return None, current_score

    M_cur = current_score / (2.0 * float(n))

    focus_opts = (3.0, 10.0, 25.0)
    trust_opts = (0.15, 0.60, 1.80)

    best_new = None
    best_M = M_cur
    used = 0

    for focus in focus_opts:
        if used >= max_dirs:
            break
        w = weights_from_q(q, focus)
        c = corr_shift(w, a)
        cmax = float(np.max(c))
        cmin = float(np.min(c))
        scale = cmax - cmin
        if scale <= 1e-15:
            continue
        for trust in trust_opts:
            if used >= max_dirs:
                break
            move = trust * (cmax - c) / scale
            move = np.clip(move, -50.0, 50.0)
            d = a * np.exp(move)
            d = normalize_vec(d)
            new, Mnew = line_search_direction(a, d, q, None)
            used += 1
            if Mnew < best_M - 1e-13:
                best_M = Mnew
                best_new = new

    if best_new is None:
        return None, M_cur
    return best_new, best_M


def smooth_polish(a0, time_limit, rng_seed=11, verbose=False):
    """
    Mirrored multiplicative-gradient polish for min max self-convolution.
    """
    if time_limit <= 0.0:
        return normalize_vec(a0)

    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 1:
        return a

    best = a.copy()
    best_score = objective_norm(a)

    trust = 0.45
    focus = 4.0
    fails = 0
    restarts = 0
    rng = np.random.default_rng(rng_seed)

    while time.monotonic() - t0 < time_limit:
        q = self_conv(a)
        M = float(np.max(q))
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            fails = 0
        else:
            fails += 1

        if fails > 450:
            break

        if fails > 40 and fails % 20 == 0:
            focus = min(focus * 1.12, 60.0)
            trust = max(trust * 0.8, 0.03)

        if fails > 120 and fails % 120 == 0:
            restarts += 1
            if restarts > 30:
                break
            scale_noise = 0.018 / (1.0 + 0.06 * restarts)
            cand = best + scale_noise * rng.normal(size=best.size) * best
            cand = np.maximum(cand, 1e-12)
            cand = normalize_vec(cand)
            sc = objective_norm(cand)
            if np.isfinite(sc) and sc <= best_score + 1e-8:
                a = cand
                fails = 0
                trust = 0.35
                focus = min(focus * 1.1, 60.0)
            else:
                a = best.copy()

        w = weights_from_q(q, focus)
        c = corr_shift(w, a)
        cmax = float(np.max(c))
        cmin = float(np.min(c))
        scale = cmax - cmin
        if scale <= 1e-15:
            break

        move = trust * (cmax - c) / scale
        move = np.clip(move, -50.0, 50.0)
        b = a * np.exp(move)
        b = normalize_vec(b)

        Mb = float(np.max(self_conv(b)))
        score_b = 2.0 * n * Mb

        if score_b < best_score - 1e-13:
            best = b.copy()
            best_score = score_b
            a = b
            fails = 0
            trust = min(trust * 1.06, 5.0)
        elif score_b < score - 1e-14:
            a = b
            fails = 0
            trust = min(trust * 1.10, 5.0)
            focus = max(focus * 0.98, 1.0)
        else:
            fails += 1
            trust = max(trust * 0.72, 0.03)
            focus = min(focus * 1.10, 60.0)

        if verbose and (time.monotonic() - t0) % 30.0 < 0.02:
            print("    smooth_polish: elapsed=%.1fs score=%.9f focus=%.2f trust=%.3f fails=%d"
                  % (time.monotonic() - t0, best_score, focus, trust, fails))

    return best

def interpolate_sequence(f, new_n):
    f = normalize_vec(f)
    old_n = f.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return f.copy()
    if old_n <= 1:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f)
    return normalize_vec(g)


def make_starts(n, parent):
    out = []
    if parent is None:
        p = np.ones(n, dtype=np.float64) / float(n)
    else:
        p = normalize_vec(parent)
        if p.size != n:
            p = interpolate_sequence(p, n)

    out.append(p)
    rev = p[::-1].copy()
    out.append(rev)

    for w in (0.3, 0.5, 0.7):
        out.append(normalize_vec((1.0 - w) * p + w * rev))

    base = np.maximum(p, 1e-30)
    for g in (0.3, 0.5, 0.7, 1.5, 2.0, 4.0):
        out.append(normalize_vec(base ** g))

    out.append(np.ones(n, dtype=np.float64) / float(n))

    x = np.arange(n, dtype=np.float64) / float(max(n - 1, 1))

    for aexp in (0.5, 1.0, 2.0, 3.0):
        out.append(normalize_vec((x + 1e-6) ** aexp))
        out.append(normalize_vec(((1.0 - x) + 1e-6) ** aexp))

    for aexp in (0.5, 1.0, 2.0):
        for bexp in (0.5, 1.0, 2.0):
            seq = ((x + 1e-6) ** aexp) * (((1.0 - x) + 1e-6) ** bexp)
            out.append(normalize_vec(seq))

    for r in (0.5, 1.0, 2.0, 5.0):
        seq = np.exp(-r * x) + np.exp(-r * (1.0 - x))
        out.append(normalize_vec(seq))

    for ctr in (0.3, 0.5, 0.7):
        for wd in (0.05, 0.1, 0.2, 0.3):
            seq = np.exp(-((x - ctr) ** 2) / (2.0 * wd * wd))
            out.append(normalize_vec(seq))

    for c1, c2 in ((0.25, 0.75), (0.3, 0.7), (0.2, 0.8)):
        for wd in (0.08, 0.15, 0.25):
            seq = np.exp(-((x - c1) ** 2) / (2.0 * wd * wd)) + \
                  np.exp(-((x - c2) ** 2) / (2.0 * wd * wd))
            out.append(normalize_vec(seq))

    rng = np.random.default_rng(777)
    for _ in range(25):
        seq = np.zeros(n, dtype=np.float64)
        for __ in range(int(rng.integers(1, 4))):
            ctr = rng.uniform(0.05, 0.95)
            wd = rng.uniform(0.05, 0.35)
            ht = rng.uniform(0.3, 1.0)
            seq += ht * np.exp(-((x - ctr) ** 2) / (2.0 * wd * wd))
        seq += 1e-6
        out.append(normalize_vec(seq))

    return out
