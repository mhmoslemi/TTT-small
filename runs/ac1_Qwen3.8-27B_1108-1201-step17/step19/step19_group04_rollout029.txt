def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]


def conv_ab(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]


def corr_shift(w, a):
    """Returns c[j] = sum_m w[m] * a[m-j] for j = 0..n-1."""
    w = np.asarray(w, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = w.size
    size = next_pow2(L + n - 1)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fw * fa.conj(), size)[:n]
    return np.maximum(c, 0.0)


def objective_norm(a):

    q = self_conv(a)
    return 2.0 * float(a.size) * float(np.max(q))

def quad_max(alpha, q0, c, qd):
    a1 = 1.0 - alpha
    val = (a1 * a1) * q0 + (2.0 * alpha * a1) * c + (alpha * alpha) * qd
    return float(np.max(val))


def golden_alpha_max(q0, c, qd, a, b, iters=40):
    phi = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = b - phi * (b - a)
    d1 = a + phi * (b - a)
    f1 = quad_max(c1, q0, c, qd)
    f2 = quad_max(d1, q0, c, qd)
    for _ in range(iters):
        if f1 < f2:
            b = d1
            d1 = c1
            f2 = f1
            c1 = b - phi * (b - a)
            f1 = quad_max(c1, q0, c, qd)
        else:
            a = c1
            c1 = d1
            f1 = f2
            d1 = a + phi * (b - a)
            f2 = quad_max(d1, q0, c, qd)
    return 0.5 * (a + b)


def line_search_direction(a, d, q0, T=None):
    """Minimize max q((1-alpha)a + alpha d) over alpha in [0,1]."""
    a = np.asarray(a, dtype=np.float64)
    d = np.asarray(d, dtype=np.float64)
    q0 = np.asarray(q0, dtype=np.float64)
    M0 = float(np.max(q0))

    dot = float(np.dot(a, d))
    if dot > 1.0 - 1e-7:
        return a.copy(), M0

    c = conv_ab(a, d)
    qd = self_conv(d)

    best_alpha = 0.0
    best_M = M0

    alphas = [1e-5, 5e-5, 1e-4, 5e-4, 1e-3, 2e-3, 5e-3, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 0.9, 1.0]

    if T is not None and np.isfinite(T) and T < 0.0 and M0 > 1e-15:
        t1 = min(0.30, max(5e-4, 0.5 * (-float(T)) / M0))
        alphas.append(max(1e-5, 0.5 * t1))
        alphas.append(min(0.95, 1.5 * t1))

    uniq = []
    seen = set()
    for al in alphas:
        al = float(min(1.0, max(0.0, al)))
        key = round(al, 14)
        if key not in seen:
            seen.add(key)
            uniq.append(al)

    for al in uniq:
        val = quad_max(al, q0, c, qd)
        if val < best_M - 1e-15:
            best_M = val
            best_alpha = al

    if best_M < M0 - 1e-13 and 0.0 < best_alpha < 1.0:
        al_refine_lo = max(0.0, best_alpha - 0.1)
        al_refine_hi = min(1.0, best_alpha + 0.1)
        al2 = golden_alpha_max(q0, c, qd, al_refine_lo, al_refine_hi, 40)
        val2 = quad_max(al2, q0, c, qd)
        if val2 < best_M:
            best_M = val2
            best_alpha = al2

    if best_alpha <= 0.0 or best_M >= M0 - 1e-14:
        return a.copy(), M0

    new = (1.0 - best_alpha) * a + best_alpha * d
    new = np.maximum(new, 0.0)
    s = float(new.sum())
    if s <= 1e-15:
        return a.copy(), M0
    new = new / s
    return new, best_M

def solve_lp_offset(y, active, q_active, weights=None):
    """
    Solve:
       min T
       s.t. sum g = 1, g >= 0,
            2*(y*g)[k] - T <= q_y[k] * w_k  for active lags k with weights w_k.
    """
    if not HAVE_LP:
        return None, None
    y = np.asarray(y, dtype=np.float64)
    active = np.asarray(active, dtype=int)
    q_active = np.asarray(q_active, dtype=np.float64)
    n = y.size
    K = active.size
    if K <= 0:
        return None, None
    if K * (n + 1) > 12_000_000:
        return None, None

    if weights is None:
        weights = np.ones(K, dtype=np.float64)
    else:
        weights = np.asarray(weights, dtype=np.float64)

    try:
        A = np.zeros((K, n + 1), dtype=np.float64)
    except Exception:
        return None, None

    yy = 2.0 * y
    for r, k in enumerate(active):
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            A[r, i0:i1] = yy[start:stop][::-1]

    A[:, n] = -1.0
    b = np.maximum(q_active * weights, 0.0)

    c, Aeq, bounds = get_lp_artifacts(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq,
                      b_eq=np.array([1.0], dtype=np.float64),
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 120.0})
    except Exception:
        return None, None

    status = getattr(res, 'status', None)
    success = getattr(res, 'success', False)
    x = getattr(res, 'x', None)
    if (status != 0 or not success) or x is None:
        return None, None

    g = np.asarray(x[:n], dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-10:
        return None, None
    g = g / s
    T = float(x[n])
    if not np.isfinite(T):
        T = 0.0
    return g, T


def kkt_gradient(q, a, top_lags):
    """
    Compute the KKT-style gradient: sum over top lags of the subgradient.
    For each active lag k, the subgradient of q_k w.r.t. a_i is:
        a_{i+k} (if i+k < n) + a_{i-k} (if i-k >= 0)
    but with the convolution indexing: q[k] = sum_i a[i]*a[k-i]
    so dq[k]/da[i] = a[k-i] + (a[i-k] if i >= k else 0) -- need care.

    Actually for q = conv(a,a), q[k] = sum_i a[i] a[k-i].
    dq[k]/da[j] = a[k-j] + a[j-k] (where terms are 0 if index out of [0,n-1]).
    
    For the KKT residual, we want: at optimum, sum_l lambda_l * dq_l/da_j = const for all j in support.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    for k in top_lags:
        k = int(k)
        for j in range(n):
            val = 0.0
            idx1 = k - j
            if 0 <= idx1 < n:
                val += a[idx1]
            idx2 = j - k
            if 0 <= idx2 < n:
                val += a[idx2]
            grad[j] += val
    return grad


def soft_direction_best(a, q, current_score, max_dirs=6):
    """Try several softmax-gradient mirror directions."""
    a = np.asarray(a, dtype=np.float64)
    q = np.asarray(q, dtype=np.float64)
    n = a.size
    if n <= 1:
        return None, current_score

    M_cur = current_score / (2.0 * float(n))

    focus_opts = (2.0, 5.0, 12.0, 30.0)
    trust_opts = (0.1, 0.4, 1.0, 2.5)

    best_new = None
    best_M = M_cur
    used = 0

    for focus in focus_opts:
        if used >= max_dirs:
            break
        m = float(q.max())
        if m <= 1e-15:
            break
        scale = 0.5 * m
        gamma = focus / scale
        vals = gamma * (q - m)
        vals = np.clip(vals, -60.0, 0.0)
        e = np.exp(vals)
        s_e = float(e.sum())
        if s_e <= 1e-300:
            w = np.ones_like(q, dtype=np.float64) / float(q.size)
        else:
            w = e / s_e

        c = corr_shift(w, a)
        cmax = float(np.max(c))
        cmin = float(np.min(c))
        sc = cmax - cmin
        if sc <= 1e-15:
            continue
        for trust in trust_opts:
            if used >= max_dirs:
                break
            move = trust * (cmax - c) / sc
            move = np.clip(move, -60.0, 60.0)
            d = a * np.exp(move)
            d = normalize_vec(d)
            new, Mnew = line_search_direction(a, d, q, None)
            used += 1
            if Mnew < best_M - 1e-13:
                best_M = Mnew
                best_new = new.copy()

    if best_new is None:
        return None, M_cur
    return best_new, best_M


def peak_shave_polish(a0, time_limit, rng_seed=42):
    """

    """

    a = normalize_vec(a0)
    n = a.size
    if n <= 10 or time_limit <= 0.0:
        return a

    best = a.copy()
    best_score = objective_norm(a)
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        q = self_conv(a)
        M = float(np.max(q))
        score = 2.0 * n * M
        peak_lag = int(np.argmax(q))

        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            stall = 0
        else:
            stall += 1

        # Compute contribution of each a[i] to the peak lag
        # q[peak_lag] = sum_i a[i] * a[peak_lag - i]
        contrib = np.zeros(n, dtype=np.float64)
        for i in range(n):
            j = peak_lag - i
            if 0 <= j < n:
                contrib[i] += a[j]
            j2 = i - peak_lag
            if 0 <= j2 < n:
                contrib[i] += a[j2]

        # Also compute a "danger" score: how much would increasing a[i] hurt?
        # Look at top 15 lags
        top_idx = np.argsort(q)[::-1][:min(20, len(q))]
        danger = np.zeros(n, dtype=np.float64)
        for k in top_idx[:10]:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    danger[i] += a[j]
                j2 = i - k
                if 0 <= j2 < n:
                    danger[i] += a[j2]

        # Source: high contribution to peak, low danger elsewhere
        # Destination: low contribution to peak, low danger
        src_score = contrib - 0.3 * danger
        dst_score = -danger

        # Pick top sources and destinations
        src_candidates = np.argsort(src_score)[::-1][:20]
        dst_candidates = np.argsort(dst_score)[:20]

        improved = False
        for s_idx in src_candidates:
            if time.monotonic() - t0 > time_limit:
                break
            if improved:
                break
            s_int = int(s_idx)
            if a[s_int] < 1e-12:
                continue
            for d_idx in dst_candidates:
                d_int = int(d_idx)
                if s_int == d_int:
                    continue
                for frac in (0.01, 0.03, 0.08, 0.15, 0.25):
                    if time.monotonic() - t0 > time_limit:
                        break
                    amt = a[s_int] * frac
                    if amt < 1e-15:
                        continue
                    b = a.copy()
                    b[s_int] -= amt
                    b[d_int] += amt
                    b = np.maximum(b, 0.0)
                    ss = float(b.sum())
                    if ss <= 1e-15:
                        continue
                    b = b / ss
                    sc = objective_norm(b)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best = b.copy()
                        a = b
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try random perturbation to escape
            if stall % 30 == 0 and stall > 0:
                noise_scale = 0.005 / (1.0 + 0.05 * stall)
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    b = a.copy()
                    amt = a[i] * noise_scale
                    b[i] -= amt
                    b[j] += amt
                    b = np.maximum(b, 0.0)
                    ss = b.sum()
                    if ss > 1e-15:
                        b = b / ss
                        sc = objective_norm(b)
                        if sc < best_score + 0.001:
                            a = b
                            stall = max(0, stall - 10)

    return best


def kkt_guided_polish(a0, time_limit):
    """
    Use KKT conditions: at optimum, the weighted combination of
    subgradients at active lags should be flat on the support.
    Iteratively adjust the sequence to flatten this combination.
    """
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 10 or time_limit <= 0.0:
        return a

    best = a.copy()
    best_score = objective_norm(a)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        q = self_conv(a)
        M = float(np.max(q))
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            stall = 0
        else:
            stall += 1

        # Find active lags (within 1% of max)
        threshold = M * 0.99
        active_mask = q >= threshold
        active_lags = np.where(active_mask)[0]
        if len(active_lags) < 2:
            # Use top 5 lags
            active_lags = np.argsort(q)[::-1][:5]

        # Compute the combined gradient
        grad = np.zeros(n, dtype=np.float64)
        for k in active_lags:
            k = int(k)
            for i in range(n):
                val = 0.0
                j = k - i
                if 0 <= j < n:
                    val += a[j]
                j2 = i - k
                if 0 <= j2 < n:
                    val += a[j2]
                grad[i] += val

        # At KKT optimum: grad[i] should be constant on support
        support = a > 1e-10
        if support.sum() < 5:
            break

        grad_sup = grad[support]
        g_mean = float(np.mean(grad_sup))
        g_dev = grad[support] - g_mean
        g_range = float(np.max(grad_sup) - np.min(grad_sup))

        if g_range < 1e-8 * M:
            break  # KKT nearly satisfied

        # Move mass from high-grad to low-grad positions
        # Normalize the deviation
        g_dev_norm = g_dev / (g_range + 1e-15)

        # The direction: reduce a where grad is high, increase where grad is low
        # Use a multiplicative update for non-negativity
        step = 0.1
        move = -step * g_dev_norm
        move = np.clip(move, -3.0, 3.0)
        b = a * np.exp(move)
        b = normalize_vec(b)

        sc = objective_norm(b)
        if sc < best_score - 1e-13:
            best_score = sc
            best = b.copy()
            a = b
            stall = 0
        else:
            # Try smaller step
            step = 0.02
            move = -step * g_dev_norm
            move = np.clip(move, -2.0, 2.0)
            b = a * np.exp(move)
            b = normalize_vec(b)
            sc = objective_norm(b)
            if sc < score - 1e-13:
                a = b
                stall = 0
            else:
                stall += 1

    return best


def smooth_polish(a0, time_limit, rng_seed=11, verbose=False):
    """Mirrored multiplicative-gradient polish."""
    if time_limit <= 0.0:
        return normalize_vec(a0)

    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 1:
        return a

    best = a.copy()
    best_score = objective_norm(a)

    trust = 0.3
    focus = 4.0
    fails = 0
    rng = np.random.default_rng(rng_seed)

    while time.monotonic() - t0 < time_limit:
        q = self_conv(a)
        M = float(np.max(q))
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            fails = 0
        else:
            fails += 1
            if fails > 300:
                break

        if fails > 30 and fails % 20 == 0:
            focus = min(focus * 1.15, 80.0)
            trust = max(trust * 0.85, 0.02)

        if fails > 100 and fails % 100 == 0:
            # Restart with noise
            scale_noise = 0.015 / (1.0 + 0.04 * fails)
            cand = best + scale_noise * rng.normal(size=best.size) * (best + 1e-8)
            cand = np.maximum(cand, 1e-12)
            cand = normalize_vec(cand)
            sc = objective_norm(cand)
            if np.isfinite(sc) and sc <= best_score + 0.0005:
                a = cand
                fails = 0
                trust = 0.25
            else:
                a = best.copy()

        m = float(q.max())
        if m <= 1e-15:
            break
        scale = 0.5 * m
        gamma = focus / scale
        vals = gamma * (q - m)
        vals = np.clip(vals, -60.0, 0.0)
        e = np.exp(vals)
        s_e = float(e.sum())
        if s_e <= 1e-300:
            w = np.ones_like(q, dtype=np.float64) / float(q.size)
        else:
            w = e / s_e

        c = corr_shift(w, a)
        cmax = float(np.max(c))
        cmin = float(np.min(c))
        sc2 = cmax - cmin
        if sc2 <= 1e-15:
            break

        move = trust * (cmax - c) / sc2
        move = np.clip(move, -60.0, 60.0)
        b = a * np.exp(move)
        b = normalize_vec(b)

        Mb = float(np.max(self_conv(b)))
        score_b = 2.0 * n * Mb

        if score_b < best_score - 1e-13:
            best = b.copy()
            best_score = score_b
            a = b
            fails = 0
            trust = min(trust * 1.05, 4.0)
        elif score_b < score - 1e-14:
            a = b
            fails = 0
            trust = min(trust * 1.08, 4.0)
        else:
            fails += 1
            trust = max(trust * 0.75, 0.02)
            focus = min(focus * 1.08, 80.0)

        if verbose and (time.monotonic() - t0) % 30.0 < 0.02:
            print("    smooth: elapsed=%.1fs score=%.9f focus=%.2f trust=%.3f"
                  % (time.monotonic() - t0, best_score, focus, trust))

    return best


def lp_fw_v2(a0, time_limit, verbose=True):
    """
    Improved Frank-Wolfe with:
    - Soft weighted active set
    - Adaptive K
    - Periodic KKT check
    """
    if time_limit <= 0.0:
        return normalize_vec(a0)

    t0 = time.monotonic()
    y = normalize_vec(a0)
    n = y.size
    if n <= 5:
        return y

    best = y.copy()
    best_score = objective_norm(y)

    if n > 7000:
        initK = 150
        maxK = 400
    elif n > 4000:
        initK = 200
        maxK = 600
    else:
        initK = 250
        maxK = 800

    K = initK
    use_lp = HAVE_LP
    lp_fails = 0
    no_improve = 0
    last_print = t0
    iter_no = 0
    best_K = K

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining <= 2.0:
            break

        qy = self_conv(y)
        M_y = float(np.max(qy))
        score_y = 2.0 * n * M_y

        if score_y < best_score - 1e-12:
            best = y.copy()
            best_score = score_y

        # Adaptive K: if stuck, increase; if improving, maybe decrease
        if no_improve > 30 and no_improve % 15 == 0:
            K = min(K + 30, maxK)
        if no_improve == 0 and K > initK + 30:
            K = max(K - 10, initK)

        # Soft active set: weight by proximity to max
        order = np.argsort(qy)[::-1]
        Kcur = int(min(K, len(qy), 1000))
        act = order[:Kcur]
        q_active = qy[act]

        # Soft weights: exponential decay from max
        if len(act) > 1 and q_active[0] > 1e-15:
            rel = q_active / q_active[0]
            weights = np.exp(-2.0 * (1.0 - rel))
            weights = weights / weights.sum()
        else:
            weights = np.ones(Kcur) / Kcur

        g_dir = None
        T = None

        if use_lp and Kcur >= 15 and remaining > 5.0:
            tstep = time.monotonic()
            g_dir, T = solve_lp_offset(y, act, q_active, weights)
            spend = time.monotonic() - tstep

            if g_dir is None:
                lp_fails += 1
                if lp_fails >= 5:
                    use_lp = False
            else:
                lp_fails = 0
                if spend > 60.0:
                    use_lp = False
                elif spend > max(20.0, 0.15 * remaining):
                    K = max(100, K - 30)

        took = True
        if g_dir is not None:
            new, Mnew = line_search_direction(y, g_dir, qy, T)
            if Mnew < M_y - 1e-13:
                y = new
                no_improve = 0
                took = True
            else:
                no_improve += 1
                # Try soft direction as backup
                if no_improve in (1, 3, 8, 20):
                    new2, M2 = soft_direction_best(y, qy, score_y, max_dirs=5)
                    if M2 < M_y - 1e-13:
                        y = new2
                        no_improve = 0
                        took = True
        else:
            new, Mnew = soft_direction_best(y, qy, score_y, max_dirs=8)
            if Mnew < M_y - 1e-13:
                y = new
                no_improve = 0
                took = True
            else:
                no_improve += 1
                K = min(K + 20, maxK)

        if not took:
            if no_improve > 120:
                break

        iter_no += 1
        now = time.monotonic()
        if verbose and (iter_no % 30 == 0 or (now - last_print) > 25.0):
            print("    LP-FW2: it=%d elapsed=%.1fs best=%.9f cur=%.9f K=%d noimp=%d"
                  % (iter_no, now - t0, best_score, score_y, K, no_improve))
            last_print = now

    return best


def coarsen_optimize_refine(a0, time_limit, coarse_factor=2, verbose=False):
    """
    Multi-resolution: coarsen, optimize, refine, optimize again.
    Helps escape local minima by changing the landscape.
    """
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n < 100 or time_limit <= 30:
        return a

    best = a.copy()
    best_score = objective_norm(a)

    # Coarsen: average adjacent pairs
    n_coarse = n // coarse_factor
    if n_coarse < 50:
        return a

    a_coarse = a[:n_coarse * coarse_factor].reshape(n_coarse, coarse_factor).mean(axis=1)
    a_coarse = normalize_vec(a_coarse)

    # Optimize the coarse version
    t_coarse = min(60.0, 0.2 * time_limit)
    if t_coarse > 15:
        a_coarse_opt = smooth_polish(a_coarse, t_coarse, rng_seed=777, verbose=False)
        # Interpolate back
        x_old = np.linspace(0.0, 1.0, a_coarse_opt.size)
        x_new = np.linspace(0.0, 1.0, n)
        a_refined = np.interp(x_new, x_old, a_coarse_opt)
        a_refined = normalize_vec(a_refined)

        sc = objective_norm(a_refined)
        if sc < best_score:
            best = a_refined.copy()
            best_score = sc

            # Polish the refined version
            remaining = time_limit - (time.monotonic() - t0)
            if remaining > 30:
                a_refined = smooth_polish(a_refined, min(40.0, 0.3 * remaining), rng_seed=888, verbose=False)
                sc = objective_norm(a_refined)
                if sc < best_score:
                    best = a_refined.copy()
                    best_score = sc

    # Also try: blend original with coarse-optimized
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 20 and len(a_coarse_opt) > 0:
        for w in (0.1, 0.3, 0.5):
            x_old = np.linspace(0.0, 1.0, a_coarse_opt.size)
            x_new = np.linspace(0.0, 1.0, n)
            a_interp = np.interp(x_new, x_old, a_coarse_opt)
            a_interp = normalize_vec(a_interp)
            blend = normalize_vec((1.0 - w) * best + w * a_interp)
            sc = objective_norm(blend)
            if sc < best_score:
                best = blend.copy()
                best_score = sc

    return best


def transform_search(a0, time_limit):
    """Try power transforms, blends with uniform, reversals."""
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    best = a.copy()
    best_score = objective_norm(a)

    # Power transforms
    gammas = (0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0)
    for g in gammas:
        if time.monotonic() - t0 > time_limit:
            break
        b = np.maximum(a, 1e-30) ** g
        b = normalize_vec(b)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    # Blend with uniform
    u = np.ones(n, dtype=np.float64) / float(n)
    for w in (0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5):
        if time.monotonic() - t0 > time_limit:
            break
        b = normalize_vec((1.0 - w) * best + w * u)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    # Reversal blends
    rev = best[::-1].copy()
    for w in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7):
        if time.monotonic() - t0 > time_limit:
            break
        b = normalize_vec((1.0 - w) * best + w * rev)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    return best


def trim_search(a0, time_limit):
    """Try trimming edges to find better length."""
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n < 50 or time_limit <= 0.0:
        return a

    best = a.copy()
    best_score = objective_norm(a)

    samples = [1, 2, 5, 10, 20, 50, 100, 200, 500, 1000, 2000]
    max_side = min(n // 4, 3000)
    samples = [s for s in samples if s <= max_side]

    for l in samples:
        if time.monotonic() - t0 > time_limit:
            break
        if n - l < 20:
            continue
        b = normalize_vec(a[l:])
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    for r in samples:
        if time.monotonic() - t0 > time_limit:
            break
        if n - r < 20:
            continue
        b = normalize_vec(a[:-r])
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    # Fine-tune around best trim
    if best is not a:
        cur_n = best.size
        base = a if cur_n == n else best
        for dl in range(-5, 6):
            if time.monotonic() - t0 > time_limit:
                break
            l_new = max(0, l + dl) if l is not None else dl
            if 0 <= l_new < cur_n - 10:
                b = normalize_vec(base[l_new:])
                sc = objective_norm(b)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best = b.copy()

    return best


def make_starts(n, parent):
    """Generate diverse starting sequences."""
    out = []
    if parent is None:
        p = np.ones(n, dtype=np.float64) / float(n)
    else:
        p = normalize_vec(parent)
        if p.size != n:
            x_old = np.linspace(0.0, 1.0, p.size)
            x_new = np.linspace(0.0, 1.0, n)
            p = normalize_vec(np.interp(x_new, x_old, p))

    out.append(p.copy())
    rev = p[::-1].copy()
    out.append(rev)

    for w in (0.2, 0.4, 0.6, 0.8):
        out.append(normalize_vec((1.0 - w) * p + w * rev))

    base = np.maximum(p, 1e-30)
    for g in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 4.0):
        out.append(normalize_vec(base ** g))

    out.append(np.ones(n, dtype=np.float64) / float(n))

    x = np.arange(n, dtype=np.float64) / float(max(n - 1, 1))

    # Monotone ramps (these work well for self-convolution!)
    for aexp in (0.5, 1.0, 1.5, 2.0, 3.0):
        out.append(normalize_vec((x + 0.01) ** aexp))
        out.append(normalize_vec(((1.0 - x) + 0.01) ** aexp))
        out.append(normalize_vec((x + 0.01) ** aexp + 0.1))

    # Product of ramps (bell-shaped but asymmetric)
    for aexp in (0.5, 1.0, 1.5):
        for bexp in (0.5, 1.0, 1.5):
            seq = ((x + 0.01) ** aexp) * (((1.0 - x) + 0.01) ** bexp)
            out.append(normalize_vec(seq))

    # Exponential decays
    for r in (0.3, 1.0, 2.0, 4.0, 8.0):
        out.append(normalize_vec(np.exp(-r * x)))
        out.append(normalize_vec(np.exp(-r * (1.0 - x))))
        out.append(normalize_vec(np.exp(-r * x) + np.exp(-r * (1.0 - x))))

    # Sine-based
    for freq in (0.5, 1.0, 1.5, 2.0):
        out.append(normalize_vec(1.0 + 0.5 * np.sin(2 * math.pi * freq * x)))
        out.append(normalize_vec(1.0 + 0.3 * np.cos(2 * math.pi * freq * x)))

    # Gaussian bumps (asymmetric)
    for ctr in (0.3, 0.4, 0.5, 0.6, 0.7):
        for wd in (0.08, 0.15, 0.25):
            out.append(normalize_vec(np.exp(-((x - ctr) ** 2) / (2.0 * wd * wd))))

    # Double bump (asymmetric)
    for c1, c2 in ((0.25, 0.6), (0.3, 0.7), (0.35, 0.65), (0.2, 0.5)):
        for r1, r2 in ((1.0, 1.0), (1.0, 2.0), (2.0, 1.0)):
            seq = r1 * np.exp(-((x - c1) ** 2) / 0.02) + r2 * np.exp(-((x - c2) ** 2) / 0.02)
            out.append(normalize_vec(seq))

    # Random mixtures of Gaussians
    rng = np.random.default_rng(12345)
    for _ in range(20):
        seq = np.zeros(n, dtype=np.float64)
        for __ in range(int(rng.integers(1, 5))):
            ctr = rng.uniform(0.05, 0.95)
            wd = rng.uniform(0.04, 0.3)
            ht = rng.uniform(0.2, 1.0)
            seq += ht * np.exp(-((x - ctr) ** 2) / (2.0 * wd * wd))
        seq += 0.01
        out.append(normalize_vec(seq))

    return out


def reduce_starts(cand_scores, max_kept=6):
    """Keep diverse, good starts."""
    remaining = list(cand_scores)
    kept = []
    for _ in range(max_kept):
        bi = -1
        bs = float('inf')
        for i, item in enumerate(remaining):
            sc = float(item[0])
            if np.isfinite(sc) and sc < bs:
                bs = sc
                bi = i
        if bi < 0:
            break
        seq = remaining[bi][1]
        ok = True
        for kseq in kept:
            if seq.size == kseq.size:
                # Orthogonality check
                dp = float(np.dot(seq, kseq))
                norm_p = float(np.linalg.norm(seq))
                norm_k = float(np.linalg.norm(kseq))
                if norm_p > 0 and norm_k > 0:
                    cos_sim = dp / (norm_p * norm_k)
                    if cos_sim > 0.995:
                        ok = False
                        break
        if ok:
            kept.append(seq.copy())
        remaining.pop(bi)
    return kept


def interpolate_sequence(f, new_n):
    f = normalize_vec(f)
    old_n = f.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return f.copy()
    if old_n <= 1:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f)
    return normalize_vec(g)


def update_best(best_seq, best_score, a, label, t0):
    if a is None:
        return best_seq, best_score
    an = normalize_vec(a)
    if an.size == 0:
        return best_seq, best_score
    sc = objective_norm(an)
    if np.isfinite(sc) and sc < best_score - 1e-13:
        best_score = sc
        best_seq = an.copy()
        try:
            print("*** NEW BEST [%s]: %.10f n=%d t=%.1fs"
                  % (label, sc, an.size, time.monotonic() - t0))
        except Exception:
            pass
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main search function. Combines:
    1. Diverse starts from parent + parametric families
    2. LP-FW with soft active set
    3. KKT-guided polish
    4. Peak shaving
    5. Multi-resolution
    6. Transform/trim search
    7. Length exploration
    """
    t_global = time.monotonic()
    total_budget = min(1800.0, float(budget_s)) - 60.0
    total_budget = max(120.0, total_budget)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if p.ndim != 1:
            p = p.ravel()
        if p.size > 0 and p.sum() > 1e-12:
            parent = normalize_vec(p)
            if parent.size > 9000:
                parent = interpolate_sequence(parent, 8000)
    except Exception:
        parent = None

    if parent is not None:
        best_seq = parent.copy()
        best_score = objective_norm(best_seq)
        print("Parent: n=%d objective=%.10f" % (best_seq.size, best_score))
    else:
        n_default = 6500
        x = np.linspace(0.0, 1.0, n_default)
        # Start with a monotone ramp (works well for self-conv)
        best_seq = normalize_vec((x + 0.01) ** 1.5)
