def fast_autocorr(a):
    """Compute a*a (full convolution) using FFT."""
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A*A, fft_size)[:size]

def norm_nonneg(a):
    a = np.maximum(np.asarray(a, float), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a)/len(a)
    return a/s

def lp_frank_wolfe(a, time_limit, solver, top_K=500, K_increment=20, verbose=False):

    n = len(a)

    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Select top-K peaks
        if K >= 2*n-1:
            idx = np.arange(2*n-1)
        else:
            idx = np.argsort(ac)[::-1][:K]
        
        # Build constraint matrix T: T[i,j] = a[idx[i]-j] if valid
        # For efficiency, use the structure
        T = np.zeros((len(idx), n))
        for ii, k in enumerate(idx):
            k = int(k)

            j_end = min(n, k+1)
            for j in range(j_start, j_end):
                T[ii, j] = a[k-j]
        
        # Solve LP: max sum(b) s.t. T@b <= M, b >= 0
        b = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(b)), [T @ b <= M])
        try:
            prob.solve(solver=solver, verbose=False)

            continue
