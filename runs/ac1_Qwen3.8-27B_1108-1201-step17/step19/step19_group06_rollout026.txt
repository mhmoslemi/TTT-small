```python
def lp_direction(f, M, max_iter=50):
    """Solve LP to find direction that doesn't increase peak."""
    n = len(f)
    constraint_set = set()
    
    # Start with top constraints
    fb = fast_conv(f, f)  # f*f
    top_k = np.argsort(fb)[::-1][:20]
    constraint_set = set(int(k) for k in top_k)
    
    for iteration in range(max_iter):
        # Build constraint matrix
        k_list = sorted(constraint_set)
        A = np.zeros((len(k_list), n))
        for row, k in enumerate(k_list):
            for i in range(max(0, k-n+1), min(n, k+1)):
                A[row, i] = f[k-i]
        
        # Solve: max sum(b) s.t. A@b <= M, b >= 0
        c = -np.ones(n)  # minimize -sum(b)
        res = linprog(c, A_ub=A, b_ub=np.full(len(k_list), M), 
                      bounds=[(0, None)]*n, method='highs')
        
        if res.status != 0:
            break
        
        b = res.x
        
        # Check for violated constraints
        fb_new = np.zeros(2*n-1)
        # Compute (f*b)(k) for all k
        # This is the convolution of f and b
        size = 2*n - 1
        fft_size = 1 << (size-1).bit_length()
        F = np.fft.rfft(f, fft_size)
        B = np.fft.rfft(b, fft_size)
        fb_new = np.fft.irfft(F * B, fft_size)[:size]
        
        violated = np.where(fb_new > M + 1e-10)[0]
        
        if len(violated) == 0:
            return b
        
        # Add violated constraints (up to 10 per iteration)
        for k in violated[:10]:
            constraint_set.add(int(k))
    
    # If we didn't converge, return the best we have
    return b
```

```python
def coord_descent_step(a, i, ac):
    """Find optimal delta for coordinate i."""
    n = len(a)
    a_i = a[i]
    
    # For each lag k, the change is 2*delta*a(k-i) + delta^2*(k==2i)
    # We need: 0 <= a_i + delta (non-negativity)
    # And: (a*a)(k) + 2*delta*a(k-i) + delta^2*(k==2i) <= ac.max() (ideally)
    
    # Use ternary search on delta in [-a_i, some_max]
    # The objective is M(delta) = max_k [ac(k) + 2*delta*a(k-i) + delta^2*(k==2i)]
    
    def M_of_delta(delta):
        if a_i + delta < 0:
            return float('inf')
        # Compute new max
        max_val = 0
        for k in range(2*n-1):
            ki = k - i
            if 0 <= ki < n:
                val = ac[k] + 2*delta*a[ki] + (delta*delta if k == 2*i else 0)
            else:
                val = ac[k]
            if val > max_val:
                max_val = val
        return max_val
    
    # Ternary search
    lo = -a_i
    hi = a_i * 2  # reasonable upper bound
    for _ in range(50):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if M_of_delta(m1) < M_of_delta(m2):
            hi = m2
        else:
            lo = m1
    
    delta = (lo + hi) / 2
    return max(0, a_i + delta)
```

```python
    """Find LP direction that keeps (f*b)(k) <= M."""
    n = len(f)
    
    # Start with top-K constraints
    fb = np.convolve(f, f)  # or use FFT
    top_k = np.argsort(fb)[::-1][:K_init]
    active = set(int(k) for k in top_k)
    
    b = np.ones(n) / n  # initial guess
    for it in range(max_iters):
        # Build constraint matrix for active constraints
        k_list = sorted(active)
        A = np.zeros((len(k_list), n))
        for row, k in enumerate(k_list):
            i_lo = max(0, k - n + 1)
            i_hi = min(n, k + 1)
            for i in range(i_lo, i_hi):
                A[row, i] = f[k - i]
        
        # Solve LP: max sum(b) s.t. A@b <= M, b >= 0
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(len(k_list), M),
                      bounds=[(0, None)] * n, method='highs')
        
        if res.status != 0:
            return None
        
        b = res.x
        
        # Check for violated constraints
        # Compute (f*b)(k) for all k using FFT
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        F = np.fft.rfft(f, fft_size)
        B = np.fft.rfft(b, fft_size)
        fb_new = np.fft.irfft(F * B, fft_size)[:size]
        
        violated = np.where(fb_new > M * (1 + 1e-8) + 1e-12)[0]
        
        if len(violated) == 0:
            return b
        
        # Add violated constraints
        for k in violated[:20]:
            active.add(int(k))
        
        # Cap the number of active constraints
        if len(active) > 200:

            viol_vals = fb_new[list(active)]
            # Actually, let's keep all active ones but cap at 200
            # Re-sort by violation
            break
    
    return b
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Mirror descent with Nesterov momentum and adaptive soft-max gradients

import numpy as np
```

from scipy.optimize import linprog


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):

def score_from_ac(a):

    n = len(a)
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max())


def compute_soft_gradient(a, ac, top_k, beta):
    """Compute soft-max weighted gradient over top autocorrelation lags."""
    n = len(a)
    k_alarm = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_alarm]
    top_vals = ac[top_idx]
    max_v = top_vals.max()
    expo = beta * (top_vals - max_v)
    expo = np.clip(expo, -50, 50)
    weights = np.exp(expo)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_alarm) / k_alarm
    else:
        weights = weights / w_sum
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n, k_int + 1)
        if i_lo >= i_hi:
            continue
        idx = np.arange(i_lo, i_hi)
        j_idx = k_int - idx
        valid = (j_idx >= 0) & (j_idx < n)
        grad[idx[valid]] += 2.0 * w * a[j_idx[valid]]
    return grad

def lp_solve_direction(f, M, K_init=50, max_iters=30):
    """Solve LP with constraint generation to find a feasible direction."""
    n = len(f)

    fb = fast_autocorr(f)
    top_k = np.argsort(fb)[::-1][:K_init]
    active = set(int(k) for k in top_k)

    b = None
    for it in range(max_iters):
        k_list = sorted(active)
        n_con = len(k_list)
        if n_con > 300:
            k_list = k_list[:300]
            n_con = 300

        A = np.zeros((n_con, n))
        for row, k in enumerate(k_list):
            i_lo = max(0, k - n + 1)
            i_hi = min(n, k + 1)
            for i in range(i_lo, i_hi):
                A[row, i] = f[k - i]

        c = -np.ones(n)
        try:
            res = linprog(c, A_ub=A, b_ub=np.full(n_con, M),
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': 10})
        except Exception:
            return None

        if res.status != 0 or res.x is None:
            return None

        b = res.x

        # Check for violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        F = np.fft.rfft(f, fft_size)
        B = np.fft.rfft(b, fft_size)
        fb_new = np.fft.irfft(F * B, fft_size)[:size]

        violated = np.where(fb_new > M * (1 + 1e-8) + 1e-12)[0]

        if len(violated) == 0:
            return b

        for k in violated[:20]:
            active.add(int(k))

    return b


def lp_step(f, M, max_backtrack=15):
    """Take an LP-based optimization step with backtracking."""
    n = len(f)
    b = lp_solve_direction(f, M)
    if b is None:
        return None

    sum_b = b.sum()
    if sum_b < 1e-15:
        return None

    g = b * (f.sum() / sum_b)
    direction = g - f

    t = 1.0
    for _ in range(max_backtrack):
        trial = f + t * direction
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s < 1e-15:
            return None
        trial = trial / s
        trial_M = fast_autocorr(trial).max()
        if trial_M <= M * (1 + 1e-10):
            return trial
        t *= 0.5

    return None
