The multiplicative weights algorithm for zero-sum games:
- Initialize $\lambda_k = 1/|S|$ for all $k \in S$, and $p_j = 1/n$ for all $j$.
- For $t = 1, \ldots, T$:
  - Row player plays $\lambda^{(t)}$
  - Column player responds with $j^* = \arg\min_j \sum_k \lambda_k A_{k,j}$
  - Update $\lambda^{(t+1)}_k \propto \lambda^{(t)}_k \cdot \exp(-\eta \cdot A_{k,j^*})$
  - Update $p^{(t+1)}_j \propto p^{(t)}_j \cdot \exp(\eta \cdot \sum_k \lambda_k A_{k,j})$

After $T$ iterations, $\bar{\lambda} = \frac{1}{T}\sum_t \lambda^{(t)}$ and $\bar{p} = \frac{1}{T}\sum_t p^{(t)}$ approximate the optimal solution.

The FW direction is then $g^* = \bar{p}$ (the column player's average strategy).

This is MUCH faster than solving an LP! Each iteration is $O(|S| \cdot n)$ for the matrix-vector product, and we need $T = O(|S| \cdot n / \epsilon^2)$ iterations for $\epsilon$-accuracy.

def solve_game(A, T=200, eta=0.1):
    # A is |S| x n matrix
    S, n = A.shape
    lam = np.ones(S) / S
    p = np.ones(n) / n
    lam_sum = np.zeros(S)
    p_sum = np.zeros(n)
    
    for t in range(T):
        # Row player's expected payoff for each column
        col_payoffs = A.T @ lam  # n-dimensional vector
        j_star = np.argmin(col_payoffs)
        
        # Update row player (minimize over columns -> maximize lambda)
        row_payoff = A[:, j_star]  # S-dimensional
        lam = lam * np.exp(eta * row_payoff)
        lam /= lam.sum()
        
        # Update column player
        p = p * np.exp(-eta * col_payoffs)
        p /= p.sum()
        
        lam_sum += lam
        p_sum += p

- Row player chooses $k \in S$ (the constraint to focus on)
- Column player chooses $j \in \{0, \ldots, n-1\}$ (where to put mass)
- Payoff: $A_{k,j} = y_{k-j}$ (the current autocorrelation kernel)

The row player wants to MAXIMIZE the payoff (find the worst constraint).
The column player wants to MINIMIZE the payoff (find the best direction).

The value of the game is $M^* = \min_g \max_{k \in S} (y*g)_k = \max_\lambda \min_j (\lambda^T A_{:,j})$.

The FW direction is the column player's optimal strategy $p^*$.

The MWA updates:
- Row: $\lambda_k \propto \lambda_k \exp(\eta \cdot (A p)_k)$ (increase weight on high-payoff rows)
- Column: $p_j \propto p_j \exp(-\eta \cdot (\lambda^T A)_j)$ (decrease weight on high-payoff columns)

This is correct. The row player is "attacking" (finding the binding constraint), and the column player is "defending" (finding the best response).

"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with adaptive softmax gradient (FFT-accelerated, O(n log n) per step)
2. Game-based Frank-Wolfe (solving LP subproblem via multiplicative weights + FFT)
3. Hybrid polishing with alternating FISTA and Game-FW
4. Adaptive constraint set management
"""

import numpy as np
import time


def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_smoothed_grad(a, beta, top_k=None):
    """Compute smoothed gradient of max autocorrelation using softmax weighting."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]

    if top_k is not None and top_k < size:
        top_idx = np.argsort(c)[::-1][:top_k]
        c_top = c[top_idx]
        c_max = c_top.max()
        exp_c = np.zeros(size)
        exp_c[top_idx] = np.exp(beta * (c_top - c_max))
        w = exp_c / exp_c.sum()
    else:
        c_max = c.max()
        exp_c = np.exp(beta * (c - c_max))
        w = exp_c / exp_c.sum()

    # Gradient: grad[m] = 2 * sum_k w[k] * a[k-m]
    # This is 2 * cross_correlation(w, a)[:n]
    # Using FFT: corr = ifft(FFT(w) * conj(FFT(a)))
    W = np.fft.rfft(w, fft_size)
    A_f = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(W * np.conj(A_f), fft_size)[:n]
    grad = 2.0 * corr

    return grad, c, w


def fista_optimize(a0, time_limit, beta_start=3.0, beta_end=80.0,
                   eta0=0.8, verbose=True):
    """FISTA with adaptive softmax gradient in log-space (mirror descent)."""
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()
    best_x = x.copy()
    best_c = fast_autocorr(x)
    best_m = best_c.max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_threshold = 5000

    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        progress = min(iter_count / max(time_limit * 150, 1), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        eta_boost = 1.5 if (iter_count % 4000 == 3000) else 1.0
        eta = eta0 * eta_boost / (np.sqrt(t) + 1.0)

        grad, c_y, _ = compute_smoothed_grad(y, beta, top_k=min(150, 2 * n - 1))

        # Mirror descent in log-space
        log_y = np.log(np.maximum(y, 1e-30)) - eta * grad
        log_y -= log_y.max()
        z = np.exp(log_y)
        z = np.maximum(z, 1e-30)
        z = z / z.sum()

        # Nesterov momentum
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = z + momentum * (z - y)
            x_new = np.maximum(x_new, 1e-30)
            x_new = x_new / x_new.sum()
        else:
            x_new = z.copy()

        c_new = fast_autocorr(x_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > stall_threshold:
            rng = np.random.default_rng(iter_count + 42)
            pert = rng.normal(0, 0.03, n) * best_x
            x_restart = norm_sum1(best_x + pert)
            x_new = x_restart
            y = x_restart.copy()
            stall = 0
            stall_threshold = max(stall_threshold * 0.7, 1500)

        y = x_new
        x = x_new
        iter_count += 1

        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, stall={stall}")

    return best_x, best_m


def game_fw_step(y, S, fft_size, T=150, eta_game=0.3):
    """
    One FW step: solve the LP subproblem approximately using multiplicative weights.
    
    The LP is: min_M M s.t. sum(g)=1, g>=0, (y*g)_k <= M for k in S.
    Equivalent zero-sum game: row player picks k in S, column player picks j in {0..n-1}.
    Payoff: A[k,j] = y[k-j].
    
    Returns the FW direction g (column player's average strategy).
    """
    n = len(y)
    S_arr = np.array(S, dtype=np.int64)
    m = len(S_arr)

    lam = np.ones(m) / m
    p = np.ones(n) / n
    p_sum = np.zeros(n)

    # Precompute FFT of y
    Y_f = np.fft.rfft(y, fft_size)

    for t in range(T):
        # Row payoff: (A p)_k = (y * p)_k for k in S
        P_f = np.fft.rfft(p, fft_size)
        yp = np.fft.irfft(Y_f * P_f, fft_size)
        row_payoff = yp[S_arr]

        # Column payoff: (A^T lambda)_j = sum_k lam_k * y_{k-j}
        # Build vector v of length (2n-1) with v[k] = lam_idx(k) for k in S
        v = np.zeros(2 * n - 1)
        v[S_arr] = lam
        V_f = np.fft.rfft(v, fft_size)
        # cross-corr(v, y)[j] = sum_r v[r+j] * y[r] = ifft(FFT(v) * conj(FFT(y)))
        col_payoff = np.fft.irfft(V_f * np.conj(Y_f), fft_size)[:n]

        # Check for numerical issues
        if not np.all(np.isfinite(row_payoff)) or not np.all(np.isfinite(col_payoff)):
            break

        # MWA updates with clipping for numerical stability
        rp_clipped = np.clip(row_payoff - row_payoff.max(), -50, 0)
        cp_clipped = np.clip(col_payoff - col_payoff.min(), 0, 50)

        lam_new = lam * np.exp(eta_game * rp_clipped)
        lam_sum = lam_new.sum()
        if lam_sum < 1e-30:
            lam_new = np.ones(m) / m
        else:
            lam_new /= lam_sum

        p_new = p * np.exp(-eta_game * cp_clipped)
        p_sum_val = p_new.sum()
        if p_sum_val < 1e-30:
            p_new = np.ones(n) / n
        else:
            p_new /= p_sum_val

        p_sum += p_new
        lam = lam_new
        p = p_new

    g = p_sum / T
    g = np.maximum(g, 0.0)
    gs = g.sum()
    if gs < 1e-15:
        g = np.ones(n) / n
    else:
        g /= gs
    return g


def game_fw_optimize(a0, time_limit, initial_K=60, max_K=300, verbose=True):
    """Game-based Frank-Wolfe optimization with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_c = fast_autocorr(a)
    best_m = best_c.max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Initialize constraint set
    c_y = fast_autocorr(y)
    init_idx = np.argsort(c_y)[::-1][:initial_K]
    S_list = sorted(set(int(k) for k in init_idx))

    iter_count = 0
    nesterov_enabled = True

    while time.monotonic() - t0 < time_limit:
        if iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            if elapsed > time_limit:
                break

        # Compute current autocorrelation
        c_y = fast_autocorr(y)
        M_curr = c_y.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        c_x = fast_autocorr(x)
        if c_x.max() < best_m - 1e-15:
            best_m = c_x.max()
            best_a = x.copy()

        # Update constraint set: add violated lags
        if iter_count > 0 and iter_count % 5 == 0:
            # Find lags where autocorrelation is close to max
            threshold = M_curr * 0.95
            near_max = np.where(c_y >= threshold)[0]
            for k in near_max[:20]:
                k_int = int(k)
                if k_int not in S_list and len(S_list) < max_K:
                    S_list.append(k_int)
            S_list.sort()

            # Prune if too large
            if len(S_list) > max_K:
                keep = np.argsort([c_y[k] for k in S_list])[::-1][:max_K // 2]
                S_list = sorted(S_list[i] for i in keep)

        # Solve game for FW direction
        T_game = 100 + min(100, iter_count // 50)  # Increase game iters over time
        T_game = min(T_game, 300)
        g_dir = game_fw_step(y, S_list, fft_size, T=T_game, eta_game=0.3)

        # Nesterov-accelerated FW update
        alpha = 2.0 / (iter_count + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if iter_count == 0:
            x_new = y_new.copy()
        else:
            beta_t = (iter_count) / (iter_count + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Game-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, |S|={len(S_list)}")

    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
