def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_smc(a, p, fft_size):
    """Compute gradient of soft-max objective using FFT.
    gradient[i] = sum_k p[k]*(a[k-i] + a[i-k]) = term1[i] + term2[i]
    term1[i] = sum_j a[j]*p[j+i]
    term2[i] = sum_j p[j]*a[i-j]
    """
    A_full = np.fft.fft(a, fft_size)
    P_full = np.fft.fft(p, fft_size)
    term1 = np.fft.ifft(np.conj(A_full) * P_full, fft_size).real[:len(a)]
    term2 = np.fft.ifft(A_full * P_full, fft_size).real[:len(a)]
    return term1 + term2


def soft_max_mirror_descent(a0, time_limit, tau_init=10.0, tau_min=0.3,
                            verbose=False, rng_seed=42):
    """Soft-max gradient with mirror descent (multiplicative updates)."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    fft_size = 1 << (3 * n).bit_length()
    tau = tau_init
    eta = 2.0 / np.sqrt(n)
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        ac = fast_autocorr(a)
        ac_max = ac.max()
        tau_eff = max(tau, ac_max * 0.05)
        log_p = ac / tau_eff - ac_max / tau_eff
        p = np.exp(log_p)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.ones(len(ac)) / len(ac)
        else:
            p = p / p_sum

        grad = compute_gradient_smc(a, p, fft_size)
        grad_tangent = grad - grad.mean()

        # Mirror descent step (multiplicative)
        log_a = np.log(np.maximum(a, 1e-30)) - eta * grad_tangent
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = a_new / a_new.sum()

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.03, 5.0)
        else:
            stall += 1
            if stall % 20 == 0:
                eta *= 0.85
            if stall > 500:
                pert = rng.normal(0, 0.005, n) * best_a
                a_p = np.maximum(best_a + pert, 1e-15)
                a = a_p / a_p.sum()
                a_new = a.copy()
                stall = 0

def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K_init=200,
                   top_K_max=None, use_all=False):
    """LP-based Frank-Wolfe for the peak autocorrelation problem."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        if iter_count > 0 and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            if elapsed > time_limit:
                break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if use_all or K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 30, top_K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            K = min(K + 30, top_K_max)
            continue
        g_dir = b_val / b_sum

        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0),
                      0.5, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1
        if iter_count % 8 == 0:
            K = min(K + 40, top_K_max)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m

        it += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        # Compute sensitivity: how much each position contributes to peak lags
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:8]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved = False

        for k in top_k:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            top_cands = candidates[np.argsort(cand_sens)[:5]]

            found = False
            for dst in top_cands:
                if dst == src or found:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

            if not found and stall > 20:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord it={it} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def pocs_spectral(a0, time_limit, n_trials=3, rng_seed=7):
    """POCS: alternate between flat-spectrum and non-negativity projection."""
    n = len(a0)
    best_a = a0.copy()
    best_m = fast_autocorr(a0).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit - 5:
            break
        t_trial = (time_limit - (time.monotonic() - t0)) / (n_trials - trial)

        rng = np.random.default_rng(rng_seed + trial)
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        A_spec = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
        A_full = np.zeros(fft_size, dtype=np.complex128)
        A_full[:fft_size // 2 + 1] = A_spec
        for i in range(1, fft_size // 2):
            A_full[fft_size - i] = np.conj(A_spec[i])
        a = np.fft.irfft(A_full, fft_size)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s

        m_curr = fast_autocorr(a).max()
        if m_curr < best_m:
            best_m = m_curr
            best_a = a.copy()

        for it in range(50000):
            if time.monotonic() - t0 > time_limit - 5:
                break

            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            alpha_blend = 0.1 + 0.3 * min(1.0, it / 50.0)
            new_mag = (1 - alpha_blend) * mag + alpha_blend * mag_mean
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s

            m_new = fast_autocorr(a_new).max()
            if m_new < best_m - 1e-15:
                best_m = m_new
                best_a = a_new.copy()
            a = a_new

    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.1) / width) ** 2) + np.exp(-((x - 0.9) / width) ** 2) + 0.01
    return norm_sum1(a)


def make_cosine_mod(n, alpha=0.3):
    x = np.arange(n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * x / n)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_fourier_mix(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 15):
        coeff = rng.normal(0, 0.2 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def multi_tau_smc(a0, time_limit, verbose=False):
    """Run soft-max mirror descent at multiple tau values in sequence."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    tau_schedule = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5]
    num_phases = len(tau_schedule)
    time_per = time_limit / num_phases

    for tau_val in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        t_phase = min(time_per, time_limit - (time.monotonic() - t0))
        if t_phase < 5:
            break
        a_out, m_out = soft_max_mirror_descent(a, t_phase, tau_init=tau_val,
                                               tau_min=tau_val * 0.3,
                                               verbose=False, rng_seed=int(tau_val * 10))
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
        a = norm_sum1(best_a)
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MultiTau tau={tau_val:.1f} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator combining multiple optimization strategies."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 6000
        best_seq = make_u_shape(n_base, sharpness=2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: POCS Spectral (quick) ===
    print(f"\n=== Phase 1: POCS (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 120.0)
    if t1 > 20:
        a_out, _ = pocs_spectral(best_seq, t1, n_trials=3, rng_seed=7)
        try_update(a_out, "pocs")

    # === Phase 2: Multi-tau SMC (main workhorse) ===
    print(f"\n=== Phase 2: Multi-tau SMC (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = multi_tau_smc(a_in, t2, verbose=True)
        try_update(a_out, "smc_multi")

    # === Phase 3: SMC from diverse inits ===
    print(f"\n=== Phase 3: SMC diverse inits ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        inits = [
            (make_u_shape(n0, 1.5), "u1.5"),
            (make_u_shape(n0, 3.0), "u3.0"),
            (make_bimodal(n0, 0.12), "bimodal"),
            (make_cosine_mod(n0, 0.2), "cosmod"),
            (make_fourier_mix(n0, seed=42), "fourier"),
        ]
        if parent is not None:
            inits.append((norm_sum1(parent), "parent"))

        t_per = max(10.0, t3 / len(inits))
        for idx, (init, lbl) in enumerate(inits):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 120:
                break
            t_avail = min(t_per, time_budget - 120 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            a_out, _ = soft_max_mirror_descent(init, t_avail, tau_init=8.0,
                                               tau_min=0.5, verbose=False,
                                               rng_seed=42 + idx)
            try_update(a_out, f"smc_{lbl}")

    # === Phase 4: LP-Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30 and n_best <= 7000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 1500)
        top_K_init = min(300, 2 * n_best - 1)
        top_K_max = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True,
                                   top_K_init=top_K_init, top_K_max=top_K_max,
                                   use_all=use_all)
        try_update(a_out, "lpfw")

    # === Phase 5: SMC projected variant ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: SMC Projected (n={n_best}) ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = soft_max_projected(a_in, t5, tau_init=5.0, tau_min=0.3,
                                       verbose=True, rng_seed=100)
        try_update(a_out, "smc_proj")

    # === Phase 6: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coordinate polish (n={n_best}) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t6, verbose=True, rng_seed=456)
        try_update(a_out, "coord")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")

    trial_lengths = []
    for delta in [400, -400, 800, -800, 1200, -1200]:
        new_n = n_best + delta
        if 1000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.3, 100.0)
        if t7a > 15:
            a_out, _ = soft_max_mirror_descent(a_long, t7a, tau_init=8.0,
                                               tau_min=0.5, verbose=False,
                                               rng_seed=new_n % 1000)
            try_update(a_out, f"smc_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.3, 80.0)
            if t7b > 10:
                a_out, _ = coordinate_polish(a_long, t7b, verbose=False,
                                             rng_seed=new_n % 500)
                try_update(a_out, f"coord_n{new_n}")

    # === Phase 8: Final refinement on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final refinement (n={n_best}) ===")
    if remaining > 30:
        t8a = min(remaining * 0.35, 150.0)
        if t8a > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = soft_max_mirror_descent(a_in, t8a, tau_init=3.0,
                                               tau_min=0.2, verbose=False,
                                               rng_seed=999)
            try_update(a_out, "smc_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 25:
            t8b = min(remaining * 0.4, 100.0)
            if t8b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_polish(a_in, t8b, verbose=False, rng_seed=789)
                try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = make_u_shape(n_fb, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
