        cmin = float(np.min(c))
        scale_c = cmax - cmin
        if scale_c <= 1e-15:
            break
        move = trust * (cmax - c) / scale_c
        move = np.clip(move, -40.0, 40.0)
        b = a * np.exp(move)
        b = normalize_vec(b)
        Mb = float(np.max(self_conv(b)))
        score_b = 2.0 * n * Mb
        if score_b < best_score - 1e-14:
            best = b.copy()
            best_score = score_b
            a = b
            fails = 0
            trust = min(trust * 1.05, 3.0)
        elif score_b < score - 1e-15:
            a = b
            fails = max(fails - 2, 0)
            trust = min(trust * 1.08, 3.0)
        else:
            trust = max(trust * 0.7, 0.02)
            focus = min(focus * 1.05, 80.0)
    return best


def solve_lp_for_direction(y, active, q_active, max_size=5_000_000):
    """
    Solve: min T s.t. sum g = 1, g >= 0, 2*(y*g)[k] - T <= q_active[k]
    """
    if not HAVE_LP:
        return None, None
    y = np.asarray(y, dtype=np.float64)
    active = np.asarray(active, dtype=int)
    q_active = np.asarray(q_active, dtype=np.float64)
    n = y.size
    K = active.size
    if K <= 0 or K > 2000:
        return None, None
    if K * (n + 1) > max_size:
        return None, None
    try:
        A = np.zeros((K, n + 1), dtype=np.float64)
    except Exception:
        return None, None
    yy = 2.0 * y
    for r in range(K):
        k = int(active[r])
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            A[r, i0:i1] = yy[start:stop][::-1]
    A[:, n] = -1.0
    b = np.maximum(q_active, 0.0)
    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0
    Aeq = np.ones((1, n), dtype=np.float64)
    bounds = [(0.0, None)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=Aeq,
                      b_eq=np.array([1.0]), bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:
        return None, None
    if not getattr(res, 'success', False):
        return None, None
    x = res.x
    if x is None:
        return None, None
    g = np.maximum(np.asarray(x[:n], dtype=np.float64), 0.0)
    s = float(g.sum())
    if s <= 1e-10:
        return None, None
    g = g / s
    T = float(x[n])
    return g, T


def lp_fw_search(a0, time_limit, verbose=False):
    """LP-based Frank-Wolfe local search."""
    if time_limit <= 0.0:
        return normalize_vec(a0)
    t0 = time.monotonic()
    y = normalize_vec(a0)
    n = y.size
    if n <= 5:
        return y
    best = y.copy()
    best_score = objective_norm(y)
    if n > 7000:
        K = 200
        maxK = 500
    else:
        K = 300
        maxK = 900
    no_improve = 0
    use_lp = HAVE_LP
    lp_fails = 0
    iter_no = 0
    last_print = t0
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break
        qy = self_conv(y)
        M_y = float(np.max(qy))
        score_y = 2.0 * n * M_y
        if score_y < best_score - 1e-13:
            best = y.copy()
            best_score = score_y
        if no_improve > 200:
            y = best.copy()
            no_improve = 0
        # Select active lags: top K by value
        Kcur = min(K, len(qy) - 1, 1200)
        order = np.argsort(qy)[::-1]
        act = order[:Kcur]
        q_act = qy[act]
        g_dir = None
        T = None
        if use_lp and Kcur >= 20:
            ts = time.monotonic()
            g_dir, T = solve_lp_for_direction(y, act, q_act)
            spend = time.monotonic() - ts
            if g_dir is None:
                lp_fails += 1
                if lp_fails >= 4:
                    use_lp = False
            else:
                lp_fails = 0
                if spend > 60.0:
                    use_lp = False
                elif spend > 0.15 * remaining:
                    K = max(100, K - 30)
        # Line search
        improved = False
        if g_dir is not None:
            c_ab = conv_ab(y, g_dir)
            qd = self_conv(g_dir)
            dot = float(np.dot(y, g_dir))
            if dot < 1.0 - 1e-7:
                best_alpha = 0.0
                best_M = M_y
                alphas = [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0]
                if np.isfinite(T) and T < 0 and M_y > 1e-15:
                    t1 = min(0.3, max(0.005, -T / (2 * M_y)))
                    alphas = [max(0.001, 0.3*t1)] + alphas + [min(0.99, 1.3*t1)]
                seen_a = set()
                for al in alphas:
                    al = float(np.clip(al, 0.0, 1.0))
                    key = round(al, 10)
                    if key in seen_a:
                        continue
                    seen_a.add(key)
                    a1 = 1.0 - al
                    val = (a1*a1)*M_y + 2.0*al*a1*float(np.max(c_ab)) + al*al*float(np.max(qd))
                    # Actually need element-wise max
                    vec = a1*a1*qy + 2.0*al*a1*c_ab + al*al*qd
                    val = float(np.max(vec))
                    if val < best_M:
                        best_M = val
                        best_alpha = al
                if best_alpha > 0 and best_M < M_y - 1e-13:
                    # Golden section refine
                    a, b2 = 0.0, 1.0
                    phi = (math.sqrt(5.0) - 1.0) / 2.0
                    c1 = b2 - phi*(b2-a)
                    d1 = a + phi*(b2-a)
                    f1 = float(np.max(a1_sq(c1, qy, c_ab, qd)))
                    f2 = float(np.max(a1_sq(d1, qy, c_ab, qd)))
                    for _ in range(30):
                        if f1 < f2:
                            b2 = d1
                            d1 = c1
                            f2 = f1
                            c1 = b2 - phi*(b2-a)
                            f1 = float(np.max(a1_sq(c1, qy, c_ab, qd)))
                        else:
                            a = c1
                            c1 = d1
                            f1 = f2
                            d1 = a + phi*(b2-a)
                            f2 = float(np.max(a1_sq(d1, qy, c_ab, qd)))
                    best_alpha = 0.5*(a+b2)
                    best_M = float(np.max(a1_sq(best_alpha, qy, c_ab, qd)))
                    new = (1.0 - best_alpha)*y + best_alpha*g_dir
                    new = np.maximum(new, 0.0)
                    ns = float(new.sum())
                    if ns > 1e-15:
                        new = new / ns
                        y = new
                        no_improve = 0
                        improved = True
        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                K = min(K + 30, maxK)
            if no_improve > 50 and no_improve % 25 == 0:
                # Try a mirror-descent step
                a_tmp = smooth_polish(y, min(5.0, 0.05*remaining), rng_seed=iter_no)
                sc_tmp = objective_norm(a_tmp)
                if sc_tmp < best_score - 1e-13:
                    best = a_tmp.copy()
                    best_score = sc_tmp
                    y = a_tmp
                    no_improve = 0
                if no_improve > 180:
                    break
        iter_no += 1
        if verbose and (iter_no % 30 == 0 or time.monotonic() - last_print > 30):
            print("  LP-FW: it=%d t=%.1fs best=%.9f cur=%.9f K=%d noimp=%d"
                  % (iter_no, time.monotonic()-t0, best_score, score_y, K, no_improve))
            last_print = time.monotonic()
    return best


def a1_sq(alpha, q0, c, qd):
    """Element-wise quadratic in alpha for line search."""
    a1 = 1.0 - alpha
    return a1*a1*q0 + 2.0*alpha*a1*c + alpha*alpha*qd


def interpolate_to_n(f, new_n):
    f = normalize_vec(f)
    old_n = f.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return f.copy()
    if old_n == 1:
        return np.ones(new_n, dtype=np.float64) / new_n
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f)
    return normalize_vec(g)


def make_diverse_starts(parent, n, num_starts=6):
    """Generate diverse starting sequences."""
    starts = []
    if parent is not None:
        p = normalize_vec(parent)
        if p.size != n:
            p = interpolate_to_n(p, n)
    else:
        x = np.linspace(0, 1, n)
        p = normalize_vec(np.sin(np.pi * x)**2)
    starts.append(p.copy())
    # Power transforms
    for g in (0.4, 0.7, 1.5, 2.5):
        starts.append(normalize_vec(np.maximum(p, 1e-30)**g))
    # Asymmetric shifts
    x = np.arange(n, dtype=np.float64) / float(n)
    for shift in (-0.1, 0.1, -0.2, 0.2):
        shifted = np.roll(p, int(shift * n))
        starts.append(normalize_vec(np.maximum(shifted, 0.0)))
    # Blend with uniform
    u = np.ones(n) / n
    for w in (0.1, 0.3, 0.5):
        starts.append(normalize_vec((1-w)*p + w*u))
    # Gaussian bumps
    rng = np.random.default_rng(123)
    for _ in range(3):
        seq = np.zeros(n)
        for _ in range(rng.integers(2, 5)):
            ctr = rng.uniform(0.1, 0.9)
            wd = rng.uniform(0.05, 0.3)
            ht = rng.uniform(0.5, 1.0)
            seq += ht * np.exp(-((x - ctr)**2) / (2*wd*wd))
        seq += 1e-8
        starts.append(normalize_vec(seq))
    return starts[:num_starts]


def coordinate_polish(a0, time_limit):
    """Move mass from high-convtribution to low-contribution positions."""
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    best = a.copy()
    best_score = objective_norm(a)
    stall = 0
    while time.monotonic() - t0 < time_limit and stall < 100:
        q = self_conv(a)
        top_lags = np.argsort(q)[::-1][:10]
        # Compute "damage" of each position
        damage = np.zeros(n)
        for k in top_lags[:5]:
            k = int(k)
            i0 = max(0, k - n + 1)
            i1 = min(n, k + 1)
            s = k - (i1 - 1)
            e = k - i0 + 1
            damage[i0:i1] += a[s:e][::-1]
        # Move mass from high-damage to low-damage positions
        src_idx = np.argsort(damage)[::-1][:8]
        dst_idx = np.argsort(damage)[:8]
        moved = False
        for si in src_idx:
            si = int(si)
            if a[si] < 1e-14:
                continue
            for di in dst_idx:
                di = int(di)
                if di == si:
                    continue
                for frac in (0.05, 0.1, 0.2):
                    amt = a[si] * frac
                    b = a.copy()
                    b[si] -= amt
                    b[di] += amt
                    b = np.maximum(b, 0.0)
                    ss = b.sum()
                    if ss < 1e-15:
                        continue
                    b = b / ss
                    sc = objective_norm(b)
                    if sc < best_score - 1e-14:
                        best = b.copy()
                        best_score = sc
                        a = b
                        moved = True
                        break
                if moved:
                    break
            if moved:
                break
        if moved:
            stall = 0
        else:
            stall += 1
    return best
