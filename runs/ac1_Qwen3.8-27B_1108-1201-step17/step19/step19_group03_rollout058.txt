    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fft_cross_corr(p, a):
    """Compute correlate(p, a, mode='valid') using FFT.
    p has length 2n-1, a has length n. Output has length n."""
    n = len(a)
    N = 1 << (3 * n - 1).bit_length()
    P = np.fft.rfft(p, N)
    Ar = np.fft.rfft(a[::-1], N)
    full = np.fft.irfft(P * Ar, N)
    return full[n-1 : 2*n-1]

def lse_grad_descent(a0, time_limit, tau_init=2.0, tau_end=50.0,
                      verbose=False, rng_seed=0, eta_init=1.0):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    eta = eta_init
    min_eta = 1e-8
    stall = 0
    max_stall = 2000
    it = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        tau = tau_init + (tau_end - tau_init) * progress
        
        ac = fast_autocorr(a)
        ac_max = ac.max()
        
        # Softmax weights
        shifted = ac / tau - ac_max / tau
        exp_ac = np.exp(shifted)
        p = exp_ac / exp_ac.sum()
        
        # Gradient via FFT
        corr = fft_cross_corr(p, a)
        g = corr  # This is sum_l p[i+l]*a[l] for each i

        g_t = g - g.mean()
        
        # Backtracking line search
        current_val = ac_max
        found = False
        eta_try = eta
        for _ in range(30):
            a_trial = a - eta_try * g_t
            a_new = project_simplex(a_trial)
            ac_new = fast_autocorr(a_new)
            if ac_new.max() < current_val * (1 - 1e-10):
                found = True
                break
            eta_try *= 0.5
            if eta_try < min_eta:

            eta = min(eta * 1.05, 10.0)
            stall = 0
        else:
            if eta > min_eta:
                eta *= 0.7
            # Take a small step anyway
            a = a - eta * g_t
            a = project_simplex(a)
            stall += 1
        
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        it += 1
        if verbose and it % 500 == 0:
            print(f"    LSE-GD it={it} ({time.monotonic()-t0:.0f}s): tau={tau:.1f}, eta={eta:.4f}, score={best_score:.6f}")
    
    return best_a, best_score

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def full_cross_corr_fft(p, a):
    """Compute c[i] = sum_{l=0}^{n-1} p[i+l] * a[l] for i=0..n-1 using FFT.
    p has length 2n-1, a has length n."""
    n = len(a)
    N = 1 << (3 * n - 1).bit_length()
    P = np.fft.rfft(p, N)
    Ar = np.fft.rfft(a[::-1].copy(), N)
    full = np.fft.irfft(P * Ar, N)
    return full[n - 1 : 2 * n - 1]

def compute_lse_gradient(a, tau):
    """Compute gradient of (2n/tau)*logsumexp(conv(a,a)/tau) w.r.t. a.
    Returns gradient vector of length n."""
    n = len(a)
    ac = fast_autocorr(a)
    ac_max = ac.max()
    shifted = ac / tau - ac_max / tau
    exp_ac = np.exp(shifted)
    p = exp_ac / exp_ac.sum()
    corr = full_cross_corr_fft(p, a)
    return corr, ac

def lse_gradient_descent(a0, time_limit, tau_start=3.0, tau_end=40.0,
                         verbose=False, rng_seed=0, eta_init=0.3,
                         max_iters=200000):
    """LSE-smoothed projected gradient descent with backtracking."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    eta = eta_init
    min_eta = 1e-9
    stall = 0
    it = 0
    last_improve = 0
    
    while time.monotonic() - t0 < time_limit and it < max_iters:
        if stall > 3000:
            break
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        tau = tau_start + (tau_end - tau_start) * (progress ** 0.7)
        
        corr, ac = compute_lse_gradient(a, tau)
        g_t = corr - corr.mean()
        g_norm = np.linalg.norm(g_t)
        if g_norm < 1e-15:
            break
        
        # Adaptive step size
        eta_scaled = eta / (g_norm + 1e-10)
        
        # Try several step sizes
        current_max = ac.max()
        best_trial = None
        best_trial_max = current_max
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = eta_scaled * frac
            a_trial = a - step * g_t
            a_new = project_simplex(a_trial)
            ac_new = fast_autocorr(a_new)
            if ac_new.max() < best_trial_max:
                best_trial_max = ac_new.max()
                best_trial = a_new
        
        if best_trial is not None:
            a = best_trial
            stall = 0
            if best_trial_max < current_max * (1 - 1e-6):
                eta = min(eta * 1.02, 2.0)
        else:
            # No improvement; reduce step and try perturbation
            eta *= 0.7
            stall += 1
            if stall > 500:
                # Random perturbation to escape
                pert = rng.normal(0, 0.005, n) * a
                a_p = np.maximum(a + pert, 0.0)
                a_p = a_p / a_p.sum()
                ac_p = fast_autocorr(a_p)
                if ac_p.max() < current_max:
                    a = a_p
                    best_score = 2.0 * n * ac_p.max()
                    best_a = a.copy()
                    stall = 0
                    eta = eta_init
                
        score_now = 2.0 * n * fast_autocorr(a).max()
        if score_now < best_score:
            best_score = score_now
            best_a = a.copy()
            last_improve = it
        
        it += 1
        if verbose and it % 1000 == 0:
            print(f"    LSE-GD it={it} ({time.monotonic()-t0:.0f}s): tau={tau:.1f}, "
                  f"eta={eta:.4f}, score={best_score:.6f}, stall={stall}")
    
    return best_a, best_score

def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K_init=300,
                   top_K_max=None, use_all=False, min_step=0.1):
    """LP-based Frank-Wolfe with line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        K_sel = min(K, 2 * n - 1)
        if use_all or K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]
        
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 100, top_K_max)
            stall += 1
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            stall += 1
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            stall += 1
            continue
        g_dir = b_val / b_sum
        
        # Line search
        best_alpha = min_step
        best_trial_m = M_curr
        alphas = [2.0/(iter_count+2.0), 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
        alphas = [min(1.0, max(min_step, al)) for al in alphas]
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha
        
        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a
        
        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        iter_count += 1
        if iter_count % 5 == 0:
            K = min(K + 50, top_K_max)
        
        if verbose and iter_count % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, "
                  f"K={K_sel}, alpha={best_alpha:.4f}, stall={stall}")
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123, n_top=30):
    """Coordinate descent: move mass from high-sensitivity to low-sensitivity coords."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    total_rounds = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]
        
        # Compute sensitivity: how much each coordinate affects peak lags
        sens = np.zeros(n)
        for k in top_k[:15]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[ki - idx]
        
        improved = False
        
        # Try moving mass from high-sens to low-sens coordinates
        high_sens = np.argsort(sens)[::-1][:20]
        low_sens = np.argsort(sens)[:20]
        
        for src in high_sens:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue
            for dst in low_sens:
                if dst == src:
                    continue
                best_frac = 0
                best_trial_m = M_curr
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_trial_m:
                        best_trial_m = M_trial
                        best_frac = frac
                if best_frac > 0:
                    amt = a[src] * best_frac
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    improved = True
                    if best_trial_m < best_m:
                        best_m = best_trial_m
                        best_a = a.copy()
            
            if improved and time.monotonic() - t0 > t0 + time_limit * 0.8:
                break
        
        if not improved:
            stall += 1
            if stall > 50:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        
        total_rounds += 1
        if verbose and total_rounds % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord round={total_rounds} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def make_asymmetric_init(n, alpha=0.3, seed=0):
    """Create an asymmetric initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.3 * np.sin(np.pi * x) + alpha * np.cos(2 * np.pi * x)
    # Add random perturbation
    a += rng.normal(0, 0.05, n)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def make_tapered(n, skew=0.3):
    """Tapered (asymmetric) window."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x ** (1 + skew)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def make_bimodal_asym(n, pos1=0.2, pos2=0.7, w1=0.15, w2=0.25):
    a = np.exp(-((np.linspace(0, 1, n) - pos1) / w1) ** 2) + \
        np.exp(-((np.linspace(0, 1, n) - pos2) / w2) ** 2) + 0.001
    return norm_sum1(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: LSE gradient from parent (coarse tau) ===
    print(f"\n=== Phase 1: LSE-GD coarse (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.18, 280.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lse_gradient_descent(a_in, t1, tau_start=5.0, tau_end=50.0,
                                         verbose=True, rng_seed=0, eta_init=0.5)
        try_update(a_out, "lse_coarse")
    
    # === Phase 2: LSE gradient from diverse starts ===
    print(f"\n=== Phase 2: LSE-GD diverse starts (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 60:
        inits = [
            (norm_sum1(best_seq), "best"),
            (make_asymmetric_init(n0, alpha=0.4, seed=1), "asym1"),
            (make_tapered(n0, skew=0.4), "taper"),
            (make_bimodal_asym(n0, 0.15, 0.75, 0.12, 0.2), "bimodal"),
        ]
        t_per = max(15.0, t2 / len(inits))
        for idx, (init, lbl) in enumerate(inits):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  LSE-GD '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = lse_gradient_descent(init, t_avail, tau_start=3.0, tau_end=35.0,
                                             verbose=False, rng_seed=100 + idx, eta_init=0.4)
            try_update(a_out, f"lse_{lbl}")
    
    # === Phase 3: LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.22, 350.0)
    if t3 > 40 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 2000)
        top_K_init = min(400, 2 * n_best - 1)
        top_K_max = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True,
                                   top_K_init=top_K_init, top_K_max=top_K_max,
                                   use_all=use_all)
        try_update(a_out, "lp1")
    
    # === Phase 4: LSE fine-tune (small tau) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LSE-GD fine (n={n_best}) ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lse_refine(a_in, t4, tau_start=1.0, tau_end=10.0,
                              verbose=True, rng_seed=200, eta_init=0.15)
        try_update(a_out, "lse_fine")
    
    # === Phase 5: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW r2 (n={n_best}) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 2000)
        top_K_init = min(600, 2 * n_best - 1)
        top_K_max = min(5000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True,
                                   top_K_init=top_K_init, top_K_max=top_K_max,
                                   use_all=use_all)
        try_update(a_out, "lp2")
    
    # === Phase 6: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coordinate polish (n={n_best}) ===")
    t6 = min(remaining * 0.12, 200.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t6, verbose=True, rng_seed=456, n_top=40)
        try_update(a_out, "coord1")
    
    # === Phase 7: LSE ultra-fine ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: LSE ultra-fine ===")
    t7 = min(remaining * 0.12, 200.0)
    if t7 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = lse_refine(a_in, t7, tau_start=0.5, tau_end=3.0,
                              verbose=True, rng_seed=300, eta_init=0.05)
        try_update(a_out, "lse_ultra")
    
    # === Phase 8: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Length exploration (n0={n_best}) ===")
    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 1500, -1500]:
        new_n = n_best + delta
        if 1500 <= new_n <= 8000 and new_n != n_best:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    
    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        t8 = min(remaining * 0.3, 120.0)
        if t8 > 15:
            a_out, _ = lse_gradient_descent(a_long, t8, tau_start=3.0, tau_end=30.0,
                                             verbose=False, rng_seed=new_n % 500, eta_init=0.3)
            try_update(a_out, f"lse_n{new_n}")
    
    # === Phase 9: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final polish (n={n_best}) ===")
    if remaining > 60:
        t9a = min(remaining * 0.4, 150.0)
        if t9a > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lse_refine(a_in, t9a, tau_start=0.3, tau_end=2.0,
                                  verbose=False, rng_seed=500, eta_init=0.03)
            try_update(a_out, "lse_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t9b = min(remaining * 0.5, 100.0)
            if t9b > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_polish(a_in, t9b, verbose=False, rng_seed=789, n_top=50)
                try_update(a_out, "coord_final")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
