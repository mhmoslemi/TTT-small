def project_simplex(v):
    """O(n log n) Euclidean projection onto the probability simplex."""
    n = len(v)

    sw = np.cumsum(w)
    ks = np.arange(1, n + 1, dtype=np.float64)
    cond = w * ks > sw - 1.0
    if not cond.any():
        return np.ones(n) / n

    tau = (sw[rho - 1] - 1.0) / rho
    x = np.maximum(v - tau, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return x / s
