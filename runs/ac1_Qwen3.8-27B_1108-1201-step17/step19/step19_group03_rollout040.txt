```
def prepare_positive(a, uniform_frac=0.0):
    arr=np.asarray(a, dtype=np.float64)
    if arr.ndim != 1:
        arr=arr.reshape(-1)
    if arr.size == 0:
        return np.ones(1)
```

```
def autocorr(a):
    a=np.asarray(a,dtype=np.float64)
    n=a.size
    if n==0: return np.array([0.0])
    size=2*n-1
    fft_size=1 << (size-1).bit_length()
    R=np.fft.rfft(a, fft_size)
    c=np.fft.irfft(R*R, fft_size)[:size]
    c=np.where(c>0.0, c, 0.0)
```

```
def make_weights(vals, base, focus):
```

```
def resample_sequence(a, new_n):
    a=prepare_positive(a,0.0)
    old_n=a.size
    new_n=int(new_n)
    if new_n <=0: new_n=1
    if old_n == new_n: return a
    if old_n == 1: return np.full(new_n, a[0])
    x_old=np.linspace(0.0,1.0,old_n)
    x_new=np.linspace(0.0,1.0,new_n)
    b=np.interp(x_new,x_old,a)
    return prepare_positive(b,1e-9)
```

```python
"""
Search for non-negative step-function heights minimizing
    2*n*max(conv(a,a)) / (sum(a))^2.

Method overview:
- Work with normalized probability vectors (sum = 1), so the score is 2*n*max(a*a).
- Fast full convolution by zero-padded FFT.
- Main local search: multiplicative (Bregman/mirror) descent driven by a
  smooth maximum surrogate using top convolution lags, with adaptive focus,
  line search, restarts, and several diverse initializations.
- Secondary polish: random pair moves / coordinate descent.
- Optional LP-based safe-direction refinement using scipy.optimize.linprog.
- Multi-length exploration by interpolating the current best and re-optimizing.
- Official evaluate_sequence is used sparingly for final bookkeeping.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog as _linprog
    _HAVE_LINPROG = True
except Exception:
    _linprog = None
    _HAVE_LINPROG = False


def prepare_positive(a, uniform_frac=0.0):
    """Clip to nonnegative finite values, normalize to sum 1, add tiny uniform mass."""
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.reshape(-1)
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s < 1e-15:
        return np.full(arr.size, 1.0 / arr.size, dtype=np.float64)
    arr = arr / s
    if uniform_frac > 0.0:
        f = float(min(max(uniform_frac, 0.0), 1e-3))
        if f > 0.0:
            arr = (1.0 - f) * arr + f / arr.size
    return arr


def autocorr(a):
    """Full linear convolution a*a via FFT, clipped to nonnegative."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    R = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(R * R, fft_size)[:size]
    c = np.where(c > 0.0, c, 0.0)
    return c


def fast_score(a):
    """Fast proxy score for normalized vectors: 2*n*max(a*a)."""
    a = prepare_positive(a, 0.0)
    if a.size == 0:
        return float("inf")
    m = float(autocorr(a).max())
    return float(2.0 * a.size * m)


def official_score(a):
    """Call the supplied evaluate_sequence, with defensive sanitization."""
    try:
        arr = np.asarray(a, dtype=np.float64)
        if arr.ndim != 1:
            arr = arr.reshape(-1)
        if arr.size == 0:
            return float("inf")
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        s = float(arr.sum())
        if s < 0.01:
            return float("inf")
        arr = arr / s
        lst = [float(x) for x in arr.tolist()]
        sc = evaluate_sequence(lst)
        return float(sc)
    except Exception:
        return float("inf")


def select_top(c, K):
    """Return indices of K largest coefficients and their values (unsorted)."""
    c = np.asarray(c, dtype=np.float64)
    L = int(c.size)
    if L == 0:
        return np.empty(0, dtype=int), np.empty(0, dtype=np.float64)
    K = int(max(1, min(int(K), L)))
    if K >= L:
        idx = np.arange(L, dtype=int)
        vals = c.astype(np.float64, copy=False)
        return idx, vals
    idx = np.argpartition(c, -K)[-K:]
    vals = c[idx].astype(np.float64, copy=False)
    return idx, vals


def make_weights(vals, base, focus):
    """Soft weights over selected top peaks."""
    vals = np.asarray(vals, dtype=np.float64)
    if vals.size == 0:
        return np.empty(0, dtype=np.float64)
    base = float(max(base, np.max(vals), 1e-300))
    tau = float(focus) * base
    if tau <= 1e-300:
        tau = 1e-300
    diff = base - vals
    diff = np.where(diff > 0.0, diff, 0.0)
    w = np.exp(-np.minimum(diff / tau, 60.0))
    s = float(w.sum())
    if s <= 1e-300:
        w = np.zeros(vals.size, dtype=np.float64)
        w[int(np.argmax(vals))] = 1.0
    else:
        w = w / s
    return w


def compute_grad_top(a, top, weights):
    """
    Gradient of the weighted sum of top convolution coefficients.

    For c_k = sum_i a_i a_{k-i}, d(c_k)/d(a_i) = 2 a_{k-i}.
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    grad = np.zeros(n, dtype=np.float64)
    if top.size != weights.size:
        return grad
    for wi, k in zip(weights, top):
        if wi <= 1e-15:
            continue
        ki = int(k)
        s = ki - n + 1
        if s < 0:
            s = 0
        e = ki + 1
        if e > n:
            e = n
        if s >= e:
            continue
        high = ki - s
        low = ki - e + 1
        if low > 0:
            partner = a[high:low - 1:-1]
        else:
            partner = a[high::-1]
        grad[s:e] += (2.0 * float(wi)) * partner
    return grad


def resample_sequence(a, new_n):
    """Linearly resample a normalized sequence to another length."""
    a = prepare_positive(a, 0.0)
    old_n = int(a.size)
    new_n = int(max(1, new_n))
    if old_n == new_n:
        return a
    if old_n == 1:
        return np.full(new_n, float(a[0]), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    b = np.interp(x_new, x_old, a)
    return prepare_positive(b, 1e-9)


def param_init(n, mode, seed):
    """Diverse non-symmetric nonnegative initial densities."""
    n = int(max(1, n))
    rng = np.random.default_rng(seed)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    if mode == 0:
        a = np.exp(-3.0 * x) + 0.25 * np.exp(-30.0 * (x - 0.75) ** 2) + 1e-3
    elif mode == 1:
        a = 0.1 + np.power(x, 1.3) + 0.35 * np.power(np.maximum(1.0 - x, 1e-8), 0.4) + 1e-3
    elif mode == 2:
        a = (
            0.1
            + np.exp(-((x - 0.15) / 0.12) ** 2)
            + 0.6 * np.exp(-((x - 0.75) / 0.25) ** 2)
            + 1e-3
        )
    elif mode == 3:
        a = 1.0 / np.power(0.05 + x, 0.3) + 0.1 * np.exp(-((x - 0.5) / 0.2) ** 2) + 1e-3
    else:
        a = np.ones(n, dtype=np.float64) * (0.2 + 0.6 * x)
        a = a + 0.05 * rng.normal(0.0, 1.0, n)

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    a = a * (1.0 + 0.05 * rng.normal(0.0, 1.0, n))
    return prepare_positive(a, 1e-8)


def random_density_init(n, alpha, seed, uniform_frac=1e-7):
    """Random full-support density, optionally sparse via alpha > 1."""
    n = int(max(1, n))
    rng = np.random.default_rng(seed)
    a = rng.random(n, dtype=np.float64)
    if alpha != 1.0:
        a = np.power(np.maximum(a, 1e-12), float(alpha))
    a = a + 1e-6
    return prepare_positive(a, uniform_frac)


def spike_init(n, K, seed, uniform_frac=1e-7):
    """Random sparse spike initialization with optional tiny uniform floor."""
    n = int(max(1, n))
    K = int(max(1, min(int(K), n)))
    rng = np.random.default_rng(seed)
    if K < n:
        idx = rng.choice(n, K, replace=False)
    else:
        idx = np.arange(n, dtype=int)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / K
    return prepare_positive(a, uniform_frac)


def spectral_init(n, seed, iters=40):
    """
    Flat-spectrum projection initialization (Gerchberg-Saxton-like), then
    nonnegative normalization.
    """
    n = int(max(1, n))
    iters = int(max(0, iters))
    rng = np.random.default_rng(seed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    half = fft_size // 2 + 1

    phases = rng.uniform(0.0, 2.0 * np.pi, half)
    R = np.exp(1j * phases)
    a = np.fft.irfft(R, fft_size)[:n].astype(np.float64, copy=False)
    a = prepare_positive(a, 1e-6)

    best = a.copy()
    best_m = float(autocorr(a).max())

    for it in range(iters):
        R = np.fft.rfft(a, fft_size)
        mag = np.abs(R)
        if mag.size > 1:
            target = float(np.mean(mag[1:]))
        else:
            target = 1.0
        if target <= 1e-300:
            target = 1.0

        alpha_mix = 0.15 + 0.35 * float(it + 1) / float(iters + 1)
        new_mag = (1.0 - alpha_mix) * mag + alpha_mix * target
        if new_mag.size > 0:
            new_mag[0] = float(mag[0]) if mag[0] > 1e-300 else 1.0

        safe = np.maximum(mag, 1e-300)
        new_R = new_mag * (R / safe)
        a2 = np.fft.irfft(new_R, fft_size)[:n].astype(np.float64, copy=False)
        a2 = prepare_positive(a2, 1e-6)

        m2 = float(autocorr(a2).max())
        if m2 < best_m:
            best = a2.copy()
            best_m = m2
        a = a2

    return best


def top_candidates(cands, k):
    """Return the k best arrays from a list of (score, array) tuples."""
    if not cands:
        return []
    k = int(max(0, k))
    if k == 0:
        return []
    scores = np.array([float(sc) for sc, _ in cands], dtype=np.float64)
    order = np.argsort(scores)
    res = []
    lim = min(k, len(cands))
    for j in range(lim):
        res.append(cands[int(order[j])][1])
    return res


def smooth_mirror(a0, t_start, t_dur, seed=0, K=48, focus=0.15, eta0=0.12,
                  uniform_frac=1e-8):
    """
    Multiplicative mirror descent on nonnegative probability vectors.

    Uses a soft maximum over the current top convolution lags.  Line search
    is approximate: accept if hard max improves, or if the current weighted
    soft objective decreases while the hard max does not increase too much.
    """
    t_dur = float(max(0.0, t_dur))
    a = prepare_positive(a0, uniform_frac)
    n = int(a.size)
    if n == 0:
        return np.ones(1, dtype=np.float64), 2.0

    deadline = float(t_start) + t_dur
    if deadline - time.monotonic() <= 0.01:
        return a, fast_score(a)

    rng = np.random.default_rng(seed)
    best = a.copy()
    best_m = float(autocorr(a).max())
    best_sc = float(2.0 * n * best_m)

    eta = float(eta0)
    cur_K = int(K)
    cur_focus = float(focus)
    no_acc = 0
    no_hard = 0
    restarts = 0

    while time.monotonic() < deadline:
        c = autocorr(a)
        m = float(c.max())

        if m < best_m - 1e-18:
            best_m = m
            best = a.copy()
            best_sc = float(2.0 * n * m)
            no_hard = 0

        base = max(m, 1e-300)
        top, vals = select_top(c, cur_K)
        if top.size == 0:
            break

        weights = make_weights(vals, base, cur_focus)
        grad = compute_grad_top(a, top, weights)
        h = grad / base
        max_h = float(np.max(np.abs(h)))
        if max_h <= 1e-300 or not np.isfinite(max_h):
            break

        # Soft objective for line search: base + tau * logsumexp(top-base)
        tm = max(1e-300, 0.08 * base)
        cv = (c[top] - base) / tm
        lse_cur = np.log(np.sum(np.exp(np.clip(cv, -50.0, 0.0))))
        soft_cur = base + tm * lse_cur

        accepted = False
        trial = None
        trial_c = None
        mt = base

        for scale in (1.0, 0.5, 0.25):
            step = eta * scale
            if step <= 1e-12:
                break
            loga = np.log(a + 1e-300) - step * h
            loga -= float(np.max(loga))
            trial = np.exp(loga)
            trial = prepare_positive(trial, 0.0)

            trial_c = autocorr(trial)
            mt = float(trial_c.max())

            if mt < best_m - 1e-18:
                accepted = True
                break

            tcv = (trial_c[top] - mt) / tm
            lse_t = np.log(np.sum(np.exp(np.clip(tcv, -50.0, 0.0))))
            soft_t = mt + tm * lse_t

            if mt <= base + 0.10 * base and soft_t < soft_cur - 1e-18:
                accepted = True
                break

        if accepted:
            a = trial
            if mt < best_m - 1e-18:
                best_m = mt
                best = a.copy()
                best_sc = float(2.0 * n * mt)
                no_hard = 0
            else:
                no_hard += 1
            eta = min(eta * 1.03, 2.0)
            no_acc = 0
        else:
            eta = max(eta * 0.70, 1e-4)
            no_acc += 1
            no_hard += 1

        if no_acc > 80:
            if cur_focus < 0.25:
                cur_focus = min(0.50, cur_focus * 1.8)
            else:
                cur_focus = max(0.03, cur_focus * 0.55)
            cur_K = min(cur_K + 16, 256)
            no_acc = 0

        if no_hard > 1200:
            restarts += 1
            if restarts > 10:
                break
            noise = rng.normal(0.0, 0.05, n) * best
            a = prepare_positive(best + noise, 1e-9)
            eta = float(eta0)
            no_acc = 0
            no_hard = 0

    return best, best_sc


def coordinate_polish(a0, t_start, t_dur, seed=0):
    """
    Coordinate/pair-move polish.
    Move mass from high-sensitivity positions to low-sensitivity positions.
    """
    a = prepare_positive(a0, 1e-10)
    n = int(a.size)
    if n == 0:
        return np.ones(1, dtype=np.float64), 2.0
    deadline = float(t_start) + float(max(0.0, t_dur))
    if deadline - time.monotonic() <= 0.01:
        return a, fast_score(a)

    rng = np.random.default_rng(seed)
    best = a.copy()
    best_m = float(autocorr(a).max())
    best_sc = float(2.0 * n * best_m)
    stall = 0
    rounds = 0

    while time.monotonic() < deadline and rounds < 20000:
        rounds += 1
        c = autocorr(a)
        m = float(c.max())
        if m < best_m - 1e-18:
            best_m = m
            best = a.copy()
            best_sc = float(2.0 * n * m)

        top, vals = select_top(c, 6)
        if top.size == 0:
            break
        base = max(float(np.max(vals)), 1e-300)
        weights = make_weights(vals, base, 0.10)
        sens = compute_grad_top(a, top, weights)

        prod = a * sens
        src_order = np.argsort(prod)[::-1][:4]
        dst_order = np.argsort(sens)[:8]

        improved = False
        for src in src_order:
            if improved:
                break
            si = int(src)
            val = float(a[si])
            if val <= 1e-12:
                continue
            for dst in dst_order:
                if improved:
                    break
                di = int(dst)
                if di == si:
                    continue
                for frac in (0.03, 0.08):
                    amt = val * frac
                    trial = a.copy()
                    trial[si] -= amt
                    trial[di] += amt
                    trial = prepare_positive(trial, 0.0)
                    mt = float(autocorr(trial).max())

                    if (mt < min(m, best_m) - 1e-16) or \
                       (mt < m - 1e-14 and mt < best_m + 0.02 * best_m):
                        a = trial
                        if mt < best_m - 1e-18:
                            best_m = mt
                            best = a.copy()
                            best_sc = float(2.0 * n * mt)
                        improved = True
                        break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 100:
                noise = rng.normal(0.0, 0.02, n) * best
                a = prepare_positive(best + noise, 1e-9)
                stall = 0
            if stall > 1500:
                break

    return best, best_sc


def lp_refine(a0, t_start, t_dur, seed=0):
    """
    Optional LP safe-direction refinement.

    For fixed current a, solve:
        min t
        s.t. (a*b)_k <= t for selected active k,
             sum(b)=1, b>=0
    Then line-search blends toward b.
    """
    a = prepare_positive(a0, 1e-10)
    n = int(a.size)
    if n == 0:
        return np.ones(1, dtype=np.float64), 2.0
    deadline = float(t_start) + float(max(0.0, t_dur))
    if deadline - time.monotonic() <= 5.0:
        return a, fast_score(a)
    if not _HAVE_LINPROG or n > 12000:
        return a, fast_score(a)

    best = a.copy()
    best_m = float(autocorr(a).max())
    best_sc = float(2.0 * n * best_m)

    if n > 8000:
        K = 6
        Kmax = 10
    else:
        K = 8
        Kmax = 14

    it = 0
    no_improve = 0

    while time.monotonic() < deadline - 5.0 and it < 6:
        c = autocorr(a)
        m = float(c.max())
        if m < best_m - 1e-18:
            best_m = m
            best = a.copy()
            best_sc = float(2.0 * n * m)
        if m <= 1e-300:
            break

        top, vals = select_top(c, K)
        Kcur = int(top.size)
        if Kcur <= 0:
            break

        A = np.zeros((Kcur, n + 1), dtype=np.float64)
        for ii, ki in enumerate(top):
            k = int(ki)
            s = k - n + 1
            if s < 0:
                s = 0
            e = k + 1
            if e > n:
                e = n
            if s < e:
                high = k - s
                low = k - e + 1
                if low > 0:
                    A[ii, s:e] = a[high:low - 1:-1]
                else:
                    A[ii, s:e] = a[high::-1]
            A[ii, -1] = -1.0

        obj = np.zeros(n + 1, dtype=np.float64)
        obj[-1] = 1.0
        Aeq = np.zeros((1, n + 1), dtype=np.float64)
        Aeq[0, :n] = 1.0
        bounds = [(0.0, None)] * n + [(None, None)]

        try:
            res = _linprog(
                obj,
                A_ub=A,
                b_ub=np.zeros(Kcur, dtype=np.float64),
                A_eq=Aeq,
                b_eq=np.ones(1, dtype=np.float64),
                bounds=bounds,
                method="highs",
                options={"presolve": True, "time_limit": 5.0},
            )
        except Exception:
            try:
                res = _linprog(
                    obj,
                    A_ub=A,
                    b_ub=np.zeros(Kcur, dtype=np.float64),
                    A_eq=Aeq,
                    b_eq=np.ones(1, dtype=np.float64),
                    bounds=bounds,
                    method="highs",
                )
            except Exception:
                try:
                    res = _linprog(
                        obj,
                        A_ub=A,
                        b_ub=np.zeros(Kcur, dtype=np.float64),
                        A_eq=Aeq,
                        b_eq=np.ones(1, dtype=np.float64),
                        bounds=bounds,
                    )
                except Exception:
                    break

        if res is None or getattr(res, "status", None) not in (0, 3):
            K = min(K + 2, Kmax)
            it += 1
            no_improve += 1
            if no_improve >= 3:
                break
            continue

        b = np.asarray(res.x[:n], dtype=np.float64)
        b = np.where(np.isfinite(b), b, 0.0)
        b = np.maximum(b, 0.0)
        s = float(b.sum())
        if s <= 1e-12:
            K = min(K + 2, Kmax)
            it += 1
            no_improve += 1
            if no_improve >= 3:
                break
            continue
        b = b / s

        best_trial = None
        best_mt = float(m)
        for alpha in (0.5, 0.25, 1.0, 0.75, 0.1):
            trial = prepare_positive((1.0 - float(alpha)) * a + float(alpha) * b, 1e-10)
            mt = float(autocorr(trial).max())
            if mt < best_mt:
                best_mt = mt
                best_trial = trial

        if best_trial is not None:
            a = best_trial
            if best_mt < best_m - 1e-18:
                best_m = best_mt
                best = a.copy()
                best_sc = float(2.0 * n * best_m)
            no_improve = 0
        else:
            no_improve += 1

        K = min(K + 2, Kmax)
        it += 1
        if no_improve >= 3:
            break

    return best, best_sc


def optimize_length(n, parent, t_start, t_dur, seed):
    """
    Multi-start optimization for a fixed length n.
    Returns (best_array, best_fast_score).
    """
    n = int(max(100, n))
    deadline = float(t_start) + float(max(0.0, t_dur))

    if parent is not None and int(parent.size) == n:
        best = prepare_positive(parent, 0.0)
    else:
        best = param_init(n, 0, seed)

    best_sc = fast_score(best)
    cand = [(best_sc, best)]

    now = time.monotonic()
    rem = deadline - now
    if rem <= 20.0:
        return best, best_sc

    inits = []
    if parent is not None and int(parent.size) == n:
        inits.append(parent)

    if rem > 120:
        iters_spectral = int(max(20, min(80, rem * 0.08)))
        inits.append(spectral_init(n, seed + 11, iters_spectral))

    inits.append(spike_init(n, max(150, int(0.25 * n)), seed + 1))
    inits.append(spike_init(n, max(400, int(0.60 * n)), seed + 2))
    inits.append(random_density_init(n, 2.0, seed + 3))
    inits.append(param_init(n, 0, seed + 4))

    # Round 1: short runs from diverse inits.
    now = time.monotonic()
    rem = deadline - now
    if rem > 40.0:
        nini = max(1, len(inits))
        t_per = min(25.0, max(6.0, rem * 0.25 / float(nini)))
        left = nini
        for idx, init in enumerate(inits):
            rem = deadline - time.monotonic()
            left = max(1, nini - idx)
            t_use = min(t_per, max(0.0, rem / float(left) - 5.0))
            if t_use < 6.0:
                break
            out, sc = smooth_mirror(
                init,
                time.monotonic(),
                t_use,
                seed=seed * 100 + idx,
                K=48,
                focus=0.15,
                eta0=0.10,
                uniform_frac=1e-8,
            )
            cand.append((sc, out))

    # Round 2: medium runs from best current candidates.
    now = time.monotonic()
    rem = deadline - now
    if rem > 50.0:
        top = top_candidates(cand, 3)
        ntop = max(1, len(top))
        t_per = min(90.0, max(15.0, rem * 0.35 / float(ntop)))
        for idx, a_in in enumerate(top):
            rem = deadline - time.monotonic()
            left = max(1, ntop - idx)
            t_use = min(t_per, max(0.0, rem / float(left) - 5.0))
            if t_use < 8.0:
                break
            out, sc = smooth_mirror(
                a_in,
                time.monotonic(),
                t_use,
                seed=seed * 1000 + idx,
                K=64,
                focus=0.08,
                eta0=0.15,
                uniform_frac=1e-9,
            )
            cand.append((sc, out))

    # Round 3: long runs from top 2.
    now = time.monotonic()
    rem = deadline - now
    if rem > 60.0:
        top = top_candidates(cand, 2)
        ntop = max(1, len(top))
        t_per = min(180.0, max(20.0, rem * 0.50 / float(ntop)))
        for idx, a_in in enumerate(top):
            rem = deadline - time.monotonic()
            left = max(1, ntop - idx)
            t_use = min(t_per, max(0.0, rem / float(left) - 5.0))
            if t_use < 12.0:
                break
            out, sc = smooth_mirror(
                a_in,
                time.monotonic(),
                t_use,
                seed=seed * 10000 + idx,
                K=96,
                focus=0.05,
                eta0=0.18,
                uniform_frac=1e-10,
            )
            cand.append((sc, out))

    # Coordinate polish on current best for this length.
    now = time.monotonic()
    rem = deadline - now
    if rem > 35.0:
        best_now = top_candidates(cand, 1)
        if best_now:
            tc = min(45.0, rem * 0.18)
            if tc > 8.0:
                out, sc = coordinate_polish(
                    best_now[0],
                    time.monotonic(),
                    tc,
                    seed=seed + 70,
                )
                cand.append((sc, out))

    final_top = top_candidates(cand, 1)
    if not final_top:
        return best, best_sc
    best = final_top[0]
    best_sc = fast_score(best)
    return best, best_sc

    best_off_sc = float("inf")
    official_calls = 0

    if time.monotonic() - t0 < HARD - 45.0:
        try:
            os0 = official_score(best_off)
            official_calls = 1
            if np.isfinite(os0):
                best_off_sc = os0
            print(f"initial official score = {best_off_sc:.6f}", flush=True)
        except Exception:
            pass

    if not np.isfinite(best_off_sc):
        best_off_sc = best_sc

    # Helper-like update logic is inlined to avoid nested functions.
    def_update_msg = False

    # Main length optimization.
    n_main = int(best.size)
    if not (1000 <= n_main <= 10000):
        n_main = 6200
        if parent is not None:
            best = resample_sequence(best, n_main)
        else:
            best = param_init(n_main, 0, 777)
        best_sc = fast_score(best)

    rem = HARD - (time.monotonic() - t0)
    if rem > 320.0:
        if parent is not None:
            parent_main = resample_sequence(parent, n_main)
        else:
            parent_main = None
        t_main = min(750.0, rem * 0.45)
        print(f"[main] optimizing n={n_main} for up to {t_main:.0f}s", flush=True)
        out, sc = optimize_length(n_main, parent_main, time.monotonic(), t_main, seed=101)
        if np.isfinite(sc) and sc < best_sc:
            best = out
            best_sc = sc
            print(f"[main] improved fast score = {best_sc:.6f}", flush=True)
            if official_calls < 30 and time.monotonic() - t0 < HARD - 60.0:
                try:
                    os = official_score(best)
                    official_calls += 1
                    if np.isfinite(os) and os < best_off_sc:
                        best_off = best.copy()
                        best_off_sc = os
                        print(f"[main] improved official score = {best_off_sc:.6f}", flush=True)
                except Exception:
                    pass
        elif official_calls < 30 and np.isfinite(sc) and sc < best_off_sc + 0.002 \
                and time.monotonic() - t0 < HARD - 60.0:
            try:
                os = official_score(out)
                official_calls += 1
                if np.isfinite(os) and os < best_off_sc:
                    best_off = out
                    best_off_sc = os
                    print(f"[main] improved official score (no fast change) = {best_off_sc:.6f}", flush=True)
            except Exception:
                pass

    # Length exploration: usually a longer length first, then optionally shorter.
    trial_lengths = []
    if 1000 < n_main:
        n_long = int(min(10000, max(n_main + 800, int(n_main * 1.2))))
        if n_long > n_main + 300:
            trial_lengths.append(n_long)
        n_short = int(max(2000, min(n_main - 800, int(n_main * 0.78))))
        if 1500 <= n_short < n_main - 200:
            trial_lengths.append(n_short)
        if 5000 not in trial_lengths and n_main > 7000:
            trial_lengths.append(5000)
        if 8000 not in trial_lengths and n_main < 7000:
            trial_lengths.append(8000)

    seen = set()
    unique_lengths = []
    for ln in trial_lengths:
        if 1000 <= ln <= 10000 and ln != n_main and ln not in seen:
            seen.add(ln)
            unique_lengths.append(ln)

    seed_len = 202
    for new_n in unique_lengths[:2]:
        rem = HARD - (time.monotonic() - t0)
        if rem < 180.0:
            break
        t_exp = min(320.0, rem * 0.30)
        if t_exp < 80.0:
            break
        par_exp = resample_sequence(best, new_n)
        print(f"[length] optimizing n={new_n} for up to {t_exp:.0f}s", flush=True)
        out, sc = optimize_length(new_n, par_exp, time.monotonic(), t_exp, seed=seed_len)
        if np.isfinite(sc) and sc < best_sc:
            best = out
            best_sc = sc
            print(f"[length={new_n}] improved fast score = {best_sc:.6f}", flush=True)
            if official_calls < 30 and time.monotonic() - t0 < HARD - 60.0:
                try:
                    os = official_score(best)
                    official_calls += 1
                    if np.isfinite(os) and os < best_off_sc:
                        best_off = best.copy()
                        best_off_sc = os
                        print(f"[length={new_n}] improved official score = {best_off_sc:.6f}", flush=True)
                except Exception:
                    pass
        elif official_calls < 30 and np.isfinite(sc) and sc < best_off_sc + 0.002 \
                and time.monotonic() - t0 < HARD - 60.0:
            try:
                os = official_score(out)
                official_calls += 1
                if np.isfinite(os) and os < best_off_sc:
                    best_off = out
                    best_off_sc = os
                    print(f"[length={new_n}] improved official score (no fast change) = {best_off_sc:.6f}", flush=True)
            except Exception:
                pass
        seed_len += 7

    # Optional LP refinement on the current best.
    rem = HARD - (time.monotonic() - t0)
    if rem > 140.0 and best.size <= 9000:
        t_lp = min(180.0, rem * 0.22)
        if t_lp > 20.0:
            print(f"[LP] refinement n={best.size} for up to {t_lp:.0f}s", flush=True)
            out, sc = lp_refine(best, time.monotonic(), t_lp, seed=4242)
            if np.isfinite(sc) and sc < best_sc:
                best = out
                best_sc = sc
                print(f"[LP] improved fast score = {best_sc:.6f}", flush=True)
                if official_calls < 30 and time.monotonic() - t0 < HARD - 60.0:
                    try:
                        os = official_score(best)
                        official_calls += 1
                        if np.isfinite(os) and os < best_off_sc:
                            best_off = best.copy()
                            best_off_sc = os
                            print(f"[LP] improved official score = {best_off_sc:.6f}", flush=True)
                    except Exception:
                        pass
            elif official_calls < 30 and np.isfinite(sc) and sc < best_off_sc + 0.002 \
                    and time.monotonic() - t0 < HARD - 60.0:
                try:
                    os = official_score(out)
                    official_calls += 1
                    if np.isfinite(os) and os < best_off_sc:
                        best_off = out
                        best_off_sc = os
                        print(f"[LP] improved official score (no fast change) = {best_off_sc:.6f}", flush=True)
                except Exception:
                    pass

    # Final smooth polish.
    rem = HARD - (time.monotonic() - t0)
    if rem > 90.0:
        tf = min(110.0, rem * 0.33)
        print(f"[final smooth] n={best.size} for up to {tf:.0f}s", flush=True)
        out, sc = smooth_mirror(
            best,
            time.monotonic(),
            tf,
            seed=5050,
            K=96,
            focus=0.05,
            eta0=0.15,
            uniform_frac=1e-10,
        )
        if np.isfinite(sc) and sc < best_sc:
            best = out
            best_sc = sc
            print(f"[final smooth] improved fast score = {best_sc:.6f}", flush=True)
            if official_calls < 30 and time.monotonic() - t0 < HARD - 60.0:
                try:
                    os = official_score(best)
                    official_calls += 1
                    if np.isfinite(os) and os < best_off_sc:
                        best_off = best.copy()
                        best_off_sc = os
                        print(f"[final smooth] improved official score = {best_off_sc:.6f}", flush=True)
                except Exception:
                    pass

    # Final coordinate polish.
    rem = HARD - (time.monotonic() - t0)
    if rem > 70.0:
        tc = min(80.0, rem * 0.30)
        print(f"[final coord] n={best.size} for up to {tc:.0f}s", flush=True)
        out, sc = coordinate_polish(best, time.monotonic(), tc, seed=6060)
        if np.isfinite(sc) and sc < best_sc:
            best = out
            best_sc = sc
            print(f"[final coord] improved fast score = {best_sc:.6f}", flush=True)
            if official_calls < 30 and time.monotonic() - t0 < HARD - 60.0:
                try:
                    os = official_score(best)
                    official_calls += 1
                    if np.isfinite(os) and os < best_off_sc:
                        best_off = best.copy()
                        best_off_sc = os
                        print(f"[final coord] improved official score = {best_off_sc:.6f}", flush=True)
                except Exception:
                    pass

    # Tiny last-ditch smooth run.
    rem = HARD - (time.monotonic() - t0)
    if rem > 45.0:
        tt = min(45.0, rem * 0.45)
        out, sc = smooth_mirror(
            best,
            time.monotonic(),
            tt,
            seed=7777,
            K=64,
            focus=0.04,
            eta0=0.12,
            uniform_frac=1e-10,
        )
        if np.isfinite(sc) and sc < best_sc:
            best = out
            best_sc = sc

    # Resolve between fast-best and officially-evaluated best.
    final_arr = best
    final_sc = best_sc

    if time.monotonic() - t0 < HARD - 15.0:
        try:
            os_best = official_score(best)
            if np.isfinite(os_best) and os_best < best_off_sc:
                best_off = best.copy()
                best_off_sc = os_best
        except Exception:
            pass

    if np.isfinite(best_off_sc) and (not np.isfinite(best_sc) or best_off_sc < best_sc):
        final_arr = best_off
        final_sc = best_off_sc

    # Defensive final sanitization.
    try:
        final_arr = np.asarray(final_arr, dtype=np.float64).reshape(-1)
        if final_arr.size == 0:
            final_arr = np.ones(6000, dtype=np.float64) / 6000.0
        final_arr = np.where(np.isfinite(final_arr), final_arr, 0.0)
        final_arr = np.maximum(final_arr, 0.0)
        s = float(final_arr.sum())
        if s < 0.01:
            final_arr = np.ones(6000, dtype=np.float64) / 6000.0
            s = 1.0
        final_arr = final_arr / s
    except Exception:
        final_arr = np.ones(6000, dtype=np.float64) / 6000.0

    print(f"FINAL fast/selected score = {final_sc:.6f}, n = {final_arr.size}, "
          f"elapsed = {time.monotonic() - t0:.1f}s", flush=True)

    return [float(x) for x in final_arr.tolist()]
```
