def fast_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project z onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lse_objective_and_grad(a, beta, top_k=None):
    """Log-sum-exp smoothed objective and gradient (a lives on simplex)."""
    n = len(a)
    b = fast_conv(a)
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
    else:
        b_use = b
    b_max = b_use.max()
    if b_max < 1e-30:
        return 0.0, np.zeros(n)
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    # grad[j] = (1/beta) * sum_k p[k] * a[k-j] (valid indices)
    a_rev = a[::-1]
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    val = b_max + np.log(exp_vals.sum()) / beta
    return val, grad


def power_mean_obj_grad(a, p):
    """Power-mean smoothed objective and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_pos = np.maximum(b, 0.0)
    size = 2 * n - 1
    Mp = np.mean(b_pos ** p)
    if Mp < 1e-30:
        return 0.0, np.zeros(n)
    val = Mp ** (1.0 / p)
    # grad[j] = (1/(2n-1)) * Mp^{1/p - 1} * sum_k b[k]^{p-1} * a[k-j]
    w = b_pos ** (p - 1)
    a_rev = a[::-1]
    fft_size = 1 << (size - 1).bit_length()
    Wf = np.fft.rfft(w, fft_size)
    Arf = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(Wf * Arf, fft_size)
    grad = conv[n - 1:2 * n - 1] * (Mp ** (1.0 / p - 1.0)) / size
    return val, grad


def lbfgs_smoothed(a0, time_limit, method='lse', beta_start=30.0, beta_end=800.0,
                    beta_growth=1.008, p_start=10.0, p_end=200.0,
                    restart_every=5, verbose=True):

            else:
                v, _ = power_mean_obj_grad(x, p_curr)
            return v
        return obj

    def make_grad(beta_curr, p_curr):
        def grad(x):
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-15:
                return np.zeros(n)
            x = x / s
            if method == 'lse':
                _, g = lse_objective_and_grad(x, beta_curr, top_k=max(20, n // 20))
            else:
                _, g = power_mean_obj_grad(x, p_curr)
            return g
        return grad

    for restart in range(30):
        if time.monotonic() - t0 > time_limit:
            break

        if method == 'lse':
            beta = min(beta_start * (beta_growth ** (restart * 20)), beta_end)
            p_curr = p_start
        else:
            beta = beta_start
            p_curr = min(p_start * (1.5 ** restart), p_end)

        def obj_wrapped(x):
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-15:
                return 1e10
            x = x / s
            if method == 'lse':
                v, _ = lse_objective_and_grad(x, beta, top_k=max(20, n // 20))
            else:
                v, _ = power_mean_obj_grad(x, p_curr)

        def grad_wrapped(x):

    candidates = []

    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)

    # Symmetric bell shapes
    for p in [0.5, 1.0, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p * (n - i_arr) ** p)

    # Gaussian
    for c in [0.25, 0.5, 0.75]:
        for sigma in [0.08, 0.15, 0.25, 0.35]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))

    # Raised cosine
    candidates.append((1.0 + np.cos(2.0 * np.pi * x)) / 2.0)

    # Sine powers
    for p in [1, 2, 3, 4, 6]:
        candidates.append(np.maximum(np.sin(np.pi * x), 0) ** p)

    # Double peak
    for gap in [0.2, 0.3, 0.4]:
        candidates.append(np.exp(-((x - 0.5 + gap / 2) ** 2) / 0.02) +
                          np.exp(-((x - 0.5 - gap / 2) ** 2) / 0.02))

    # Multi-frequency cosine
    for f1 in [2, 3, 4]:
        for f2 in [5, 7, 11]:
            candidates.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f1 * x) +
                                        0.2 * np.cos(2 * np.pi * f2 * x), 0.01))

    # Chirp-like
    for f in [3, 5, 8]:
        phase = 2 * np.pi * f * x ** 2
        candidates.append(np.maximum(1.0 + 0.3 * np.cos(phase), 0.01))

    # Triangular
    candidates.append(1.0 - np.abs(2 * x - 1))
    candidates.append(1.0 - np.abs(2 * x - 1) ** 2)

    # Random structured
    rng = np.random.default_rng(42)
    for _ in range(30):
        freq = rng.uniform(1, 15)
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        candidates.append(np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * x + phase), 0.01))

    # Evaluate all
    best_a = best_init.copy()
    best_val = best_score

    for a_raw in candidates:
        if time.monotonic() - t0 > t_limit:
            break
        a = norm_sum1(a_raw)
        b = fast_conv(a)
        m = b.max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    return best_a, best_val


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()

    except NameError:
        pass
    except Exception:
