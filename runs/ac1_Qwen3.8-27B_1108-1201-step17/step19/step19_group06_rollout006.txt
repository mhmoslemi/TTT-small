"""

import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute score without calling evaluate_sequence (faster for iterations)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max() / (s * s))


def evaluate_safe_call(a):

    try:
        a_list = [float(max(0.0, x)) for x in a]
        return float(evaluate_sequence(a_list))

def norm1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_n)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def greedy_coord_descent(a0, time_limit, rng_seed=42, verbose=True):
    """
    Greedy coordinate descent: transfer mass between pairs to reduce
    the max autocorrelation peak. Uses FFT for fast evaluation.
    """
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        ac = fast_autocorr(a)
        cur_m = ac.max()

        if cur_m < best_m - 1e-14:
            best_m = cur_m
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50000:
                break

        # Find top autocorrelation lags
        n_top = min(20, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:n_top]

        improved = False

        # For each top lag, try targeted transfers
        for lag_idx in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(lag_idx)
            # Pairs contributing to ac[k]: a[i]*a[i+k] for valid i
            i_lo = max(0, k - n + 1)
            i_hi = min(n, k + 1)
            if i_lo >= i_hi:
                continue
            idx = np.arange(i_lo, i_hi)
            j_idx = k - idx
            valid = (j_idx >= 0) & (j_idx < n)
            idx_v = idx[valid]
            j_v = j_idx[valid]
            contrib = a[idx_v] * a[j_v]

            # Find highest contributing pair
            if len(contrib) == 0:
                continue
            pair_order = np.argsort(contrib)[::-1]

            for pi in pair_order[:8]:
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(idx_v[pi])
                j = int(j_v[pi])
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue

                # Score each potential destination by impact
                # Moving mass from i or j to position p affects ac at various lags
                # Use top lags for scoring
                n_score_lags = min(6, n_top)
                scores = np.zeros(n)
                for lk in top_lags[:n_score_lags]:
                    lk_int = int(lk)
                    sl = max(0, lk_int - n + 1)
                    el = min(n, lk_int + 1)
                    if sl >= el:
                        continue
                    ii = np.arange(sl, el)
                    jj = lk_int - ii
                    vv = (jj >= 0) & (jj < n)
                    ii_v = ii[vv]
                    jj_v = jj[vv]
                    # Impact of changing a[p] on ac[lk]:
                    # sum a[ii]*a[jj] where p is in ii or jj
                    impacts = np.zeros(n)
                    impacts[ii_v] += a[jj_v]
                    impacts[jj_v] += a[ii_v]
                    impacts[i] += abs(a[j])  # extra penalty for source
                    impacts[j] += abs(a[i])
                    scores += impacts * ac[lk_int]

                scores[i] = 1e18
                scores[j] = 1e18
                dst = int(np.argmin(scores))

                # Try moving small amount from higher-value source to dst
                src = i if a[i] > a[j] else j
                amt_frac = min(0.05, 0.5 * min(a[src] / max(a[dst], 1e-14), 1.0))
                amt = a[src] * amt_frac * 0.5

                if amt < 1e-15:
                    continue

                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial /= s_t
                m_trial = fast_autocorr(a_trial).max()

                if m_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True

        if not improved and stall > 0 and stall % 1000 == 0:
            # Perturb to escape
            n_pert = int(rng.integers(10, 50))
            for _ in range(n_pert):
                i = int(rng.integers(n))
                if a[i] > 1e-12:
                    j = int(rng.integers(n))
                    frac = float(rng.uniform(0.001, 0.01))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = norm1(a)
            m_new = fast_autocorr(a).max()
            if m_new < best_m:
                best_m = m_new
                best_a = a.copy()
                stall = 0

    if verbose:
        print(f"    Coord descent: n={n}, J={2.0*n*best_m:.6f}, iters={iteration}, time={time.monotonic()-t0:.0f}s")
    return best_a, best_m


def anneal_optimize(a0, time_limit, rng_seed=99, verbose=True):
    """
    Simulated annealing with multiplicative perturbations.
    Uses geometric random walk in log-space with acceptance criterion.
    """
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    T = 0.05  # initial temperature
    T_min = 0.0001
    cool_rate = 0.999
    iteration = 0
    accept_count = 0

    while time.monotonic() - t0 < time_limit and T > T_min:
        iteration += 1
        if iteration % 1000 == 0:
            T *= cool_rate ** 10

        # Geometric perturbation in log-space
        k_pert = int(rng.integers(5, min(50, n)))
        idx = rng.choice(n, size=k_pert, replace=False)
        log_factors = rng.normal(0, T, size=k_pert)
        log_factors = np.clip(log_factors, -2, 2)

        a_trial = a.copy()
        a_trial[idx] *= np.exp(log_factors)
        a_trial = np.maximum(a_trial, 1e-30)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            a_trial = best_a.copy()
            continue
        a_trial /= s_t

        m_trial = fast_autocorr(a_trial).max()
        delta = m_trial - best_m

        if delta < -1e-15:
            # Accept improvement
            a = a_trial
            accept_count += 1
        elif rng.random() < np.exp(-max(delta, 0) / max(T, 1e-10)):
            # Accept worse with probability
            a = a_trial
            accept_count += 1

        if best_m < fast_autocorr(a).max() + 1e-14:
            pass
        else:
            cur_m = fast_autocorr(a).max()
            if cur_m < best_m:
                best_m = cur_m
                best_a = a.copy()

    if verbose:
        print(f"    Anneal: n={n}, J={2.0*n*best_m:.6f}, iters={iteration}, acc={accept_count}, time={time.monotonic()-t0:.0f}s")
    return best_a, best_m


def spectral_optimize(a0, time_limit, rng_seed=7, verbose=True):
    """
    Work in frequency domain: flatten the power spectrum to reduce
    autocorrelation peaks. Uses iterative spectral shaping.
    """
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    fft_size = 1 << (2 * n - 2).bit_length()
    stall = 0
    beta = 2.0
    lr = 0.1

    for it in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        if P_max < 1e-30:
            break

        # Target: flatten spectrum
        P_target = np.full_like(P, P.max() / len(P))
        # Gradient in frequency domain: push toward flat
        P_dev = (P - P_target) / P_max
        expo = beta * P_dev
        expo = np.clip(expo, -15, 15)
        w = np.exp(expo)

        # Modify amplitude
        scale = 1.0 - lr * (w / w.mean() - 1.0)
        scale = np.clip(scale, 0.3, 1.7)
        A_new = A * scale

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = norm1(a_new)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-13:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 2000:
                # Restart with perturbation
                a = norm1(best_a * (1.0 + rng.normal(0, 0.02, n)))
                stall = 0
                beta = max(beta * 0.8, 0.5)
                lr = max(lr * 0.85, 0.01)

    if verbose:
        print(f"    Spectral: n={n}, J={2.0*n*best_m:.6f}, iters={it}, time={time.monotonic()-t0:.0f}s")

    print(f"\n=== Phase 4: Multi-start (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        a_out, _ = multi_start_optimize(best_seq, t4, rng_seed=4, verbose=True)
        check_update(a_out, f"multistart_n{len(best_seq)}")

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration (budget={remaining:.0f}s) ===")

    length_tries = []
    for delta in [-500, -200, 200, 500, 1000, -1000, 1500]:
        nn = n_best + delta
        if 500 <= nn <= 12000 and nn != n_best:
            length_tries.append(nn)
    for rn in [2000, 3000, 4000, 5000, 6000, 7000, 8000, 10000]:
        if rn != n_best and 500 <= rn <= 12000:
            length_tries.append(rn)
    length_tries = sorted(set(length_tries))

    for new_n in length_tries[:6]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")

        a_interp = interpolate_seq(best_seq, new_n)
        # Quick optimize
        t5a = min(remaining * 0.30, 100.0)
        if t5a > 15:
            a_out, _ = greedy_coord_descent(a_interp, t5a, rng_seed=new_n % 1000, verbose=False)
            check_update(a_out, f"coord_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t5b = min(remaining * 0.25, 80.0)
            if t5b > 10:
                a_out, _ = spectral_optimize(a_interp, t5b, rng_seed=new_n % 500, verbose=False)
                check_update(a_out, f"spectral_n{new_n}")

    # === Phase 6: Final polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_best}, budget={remaining:.0f}s) ===")

    t6 = min(remaining * 0.40, 200.0)
    if t6 > 20:
        a_out, _ = greedy_coord_descent(best_seq, t6, rng_seed=99, verbose=True)
        check_update(a_out, f"polish_coord_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t7 = min(remaining * 0.30, 120.0)
        if t7 > 15:
            a_out, _ = anneal_optimize(best_seq, t7, rng_seed=77, verbose=False)
            check_update(a_out, f"polish_anneal_n{len(best_seq)}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t8 = min(remaining * 0.40, 80.0)
        if t8 > 10:
            a_out, _ = spectral_optimize(best_seq, t8, rng_seed=55, verbose=False)
            check_update(a_out, f"polish_spec_n{len(best_seq)}")

    # Final: verify with actual evaluate_sequence
    if best_seq is not None:
        fs = evaluate_safe_call(best_seq)
        if np.isfinite(fs):
            if fs < best_score:
                best_score = fs
            print(f"\nFinal verify: evaluate_sequence={fs:.6f}, local_score={best_score:.6f}")
        elapsed = time.monotonic() - t_global
        print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = np.ones(n_fb) / n_fb
        best_score = eval_score(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    # Ensure all values are valid
    result = np.maximum(best_seq, 0.0)
    s = result.sum()
    if s > 0.01:
        result = result / s
    else:
        result = np.ones(1000) / 1000

    return [float(x) for x in result]
