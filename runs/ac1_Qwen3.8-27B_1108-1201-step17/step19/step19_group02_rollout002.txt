    """Project v onto the standard simplex {x >= 0, sum(x) = 1}"""
    n = len(v)

"""

import numpy as np
import time
import cvxpy as cp

# Global solver
_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(100, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                return s
        except:
            continue
    _solver = cp.ECOS
    return _solver

def fast_conv(a):
    """Fast linear convolution via FFT"""
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    return np.fft.irfft(A * A, fsize)[:size]

def obj_val(a):
    """Compute objective (assuming sum(a)=1)"""
    return 2.0 * len(a) * fast_conv(a).max()

def norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}"""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_gradient(a, beta, top_k=100):
    """Compute soft gradient of max(conv) using top-k terms"""
    n = len(a)
    conv = fast_conv(a)
    # Use top_k largest terms
    k_max = min(top_k, len(conv))
    top_idx = np.argsort(conv)[::-1][:k_max]
    top_vals = conv[top_idx]
    # Softmax weights
    max_v = top_vals.max()
    w = np.exp(beta * (top_vals - max_v))
    w /= w.sum()
    
    # Gradient: for each active k, contribution to grad[i] is w_k * a[k-i]
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] += wi * a[k - idx]
    return grad, conv.max()

def proj_gradient_opt(a0, time_limit, beta_init=10.0, top_k=80, c0=2.0,
                      momentum=0.5, restart_every=2000, verbose=True, label=""):
    """Projected gradient with softmax-smoothed max, momentum, and restarts"""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    x = a.copy()
    vel = np.zeros(n)
    gamma = c0
    stall = 0
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Momentum step
        y = x + momentum * vel
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute gradient
        grad, M_curr = softmax_gradient(y, beta_init, top_k)
        
        # Adaptive step size
        eta = gamma / (np.sqrt(t) + 1.0)
        
        # Gradient step
        x_new = project_simplex(y - eta * grad)
        x_new = np.maximum(x_new, 0.0)
        s = x_new.sum()
        if s < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s
        
        # Update velocity
        vel = momentum * vel + (1 - momentum) * (x_new - x)
        x = x_new
        
        # Check improvement
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Periodic restart
        if stall > restart_every:
            stall = 0
            noise = 0.03 / (1.0 + 0.2 * best_m)
            x = best_a + rng.normal(0, noise, n)
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            vel = np.zeros(n)
            gamma = max(gamma * 0.9, 0.5)
            if verbose:
                print(f"    {label}: restart at t={t}, J={2*n*best_m:.6f}")
        
        # Print progress
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label}: t={t} ({elapsed:.0f}s), J={2*n*best_m:.6f}, stall={stall}")
    
    return best_a

def lp_fw_v2(a0, time_limit, solver, K_init=300, K_max=3000, add_rate=50,
             verbose=True, label="lpfw"):
    """LP-based Frank-Wolfe with improved active set"""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    conv_y = fast_conv(y)
    active = sorted(np.argsort(conv_y)[::-1][:K_init].tolist())
    
    no_improve = 0
    max_stall = 800
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > max_stall:
            break
        it += 1
        
        M_curr = fast_conv(y).max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Manage active set size
        if len(active) > K_max:
            conv_a = fast_conv(y)
            margins = np.array([M_curr - conv_a[k] for k in active])
            keep_idx = np.argsort(margins)[:K_max]
            active = sorted([active[i] for i in keep_idx])
        
        if not active:
            break
        
        # Build constraint matrix
        K = len(active)
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(active):
            k = int(k)
            i_s = max(0, k - n + 1)
            i_e = min(n, k + 1)
            if i_s < i_e:
                idx = np.arange(i_s, i_e)
                coeffs[jj, idx] = y[k - idx]
        
        # Solve LP: min M s.t. A@x <= M, sum(x)=1, x>=0
        g = cp.Variable(n, nonneg=True)
        M_v = cp.Variable()
        cons = [cp.sum(g) == 1, coeffs @ g <= M_v]
        prob = cp.Problem(cp.Minimize(M_v), cons)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        if g_val.sum() < 1e-15:
            continue
        g_val /= g_val.sum()
        
        # Find violated constraints and add to active set
        conv_yg = fast_conv(y)
        # Actually we need conv of y with g_val to check violations
        # But for simplicity, use the current conv of y
        conv_ys = fast_conv(y)
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        # Add new violations
        new_ks = set(int(k) for k in violations[:add_rate] if int(k) not in set(active))
        active = sorted(set(active) | new_ks)
        
        # Frank-Wolfe step with best alpha
        alpha_opt = 2.0 / (it + 2.0)
        best_alpha = alpha_opt
        best_trial_M = M_curr
        
        # Line search
        for alpha in [alpha_opt, alpha_opt*0.5, alpha_opt*2, 0.1, 0.2, 0.3, 0.5, 1.0]:
            alpha = max(0.01, min(1.0, alpha))
            trial = (1 - alpha) * y + alpha * g_val
            trial = np.maximum(trial, 0.0)
            if trial.sum() < 1e-15:
                continue
            trial /= trial.sum()
            M_tr = fast_conv(trial).max()
            if M_tr < best_trial_M:
                best_trial_M = M_tr
                best_alpha = alpha
        
        y_new = (1 - best_alpha) * y + best_alpha * g_val
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new / y_new.sum()
        y = y_new
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label}: it={it} ({elapsed:.0f}s), J={2*n*best_m:.6f}, |S|={len(active)}")
    
    return best_a

def coord_polish_v2(a0, time_limit, verbose=True, label="coord"):

    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        top_k = np.argsort(conv)[::-1][:30]
        improved = False
        
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:

            k = int(k)

def get_solver():

            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    return np.fft.irfft(A * A, fsize)[:size]


def obj_val(a):
    return 2.0 * len(a) * fast_conv(a).max()


def norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

    if s < 1e-15:

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return norm1(f)
    f = norm1(f)
    old_x = np.linspace(0.0, 1.0, old_n)

    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    return norm1(f_new)

            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n, k_i + 1)
            if i_s < i_e:
                idx = np.arange(i_s, i_e)
                grad[idx] += wi * y[k_i - idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        x_new = project_simplex(y - eta * grad)
        x_new = np.maximum(x_new, 0.0)
        s_n = x_new.sum()
        if s_n < 1e-15:
            x_new = y.copy()
        else:
            x_new /= s_n

        vel = momentum * vel + (1 - momentum) * (x_new - x)
        x = x_new
        M_new = fast_conv(x).max()

        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            noise = 0.03
            x = best_a + rng.normal(0, noise, n)
            x = np.maximum(x, 1e-10)
            x /= x.sum()
            vel = np.zeros(n)
            gamma = max(gamma * 0.9, 0.5)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"  {label}: restart t={t} ({elapsed:.0f}s) J={2*n*best_m:.6f}")

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label}: t={t} ({elapsed:.0f}s) J={2*n*best_m:.6f} stall={stall}")

    return best_a
