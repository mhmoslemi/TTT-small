def next_pow2(m):
    m = int(max(1, m))

def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s

def self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]

def conv_ab(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]

def quad_max(alpha, q0, c, qd):
    a1 = 1.0 - alpha
    val = (a1 * a1) * q0 + (2.0 * alpha * a1) * c + (alpha * alpha) * qd
    return float(np.max(val))

def golden_alpha_max(q0, c, qd, a_lo, b_hi, iters=40):
    phi = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = b_hi - phi * (b_hi - a_lo)
    d1 = a_lo + phi * (b_hi - a_lo)
    f1 = quad_max(c1, q0, c, qd)
    f2 = quad_max(d1, q0, c, qd)
    for _ in range(iters):
        if f1 < f2:
            b_hi = d1
            d1 = c1
            f2 = f1
            c1 = b_hi - phi * (b_hi - a_lo)

        else:

            c1 = d1
            f1 = f2
            d1 = a_lo + phi * (b_hi - a_lo)
            f2 = quad_max(d1, q0, c, qd)

def peak_target_direction(a, q, K=500, eta=1.0):
    """Compute a multiplicative gradient direction targeting the peak lags."""
    n = a.size
    m = q.size
    M = float(np.max(q))
    if M <= 1e-15:
        return normalize_vec(a)
    threshold = M - 0.05 * M
    active = np.where(q >= threshold)[0]
    if active.size > K:
        active = active[:K]
    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            grad[i0:i1] += a[start:stop][::-1]
    grad = np.maximum(grad, 1e-30)
    base = np.maximum(a, 1e-30)
    d = base * np.exp(-eta * grad / base)
    d = np.where(np.isfinite(d), d, 0.0)
    d = np.maximum(d, 0.0)
    s = float(d.sum())
    if s <= 1e-15:
        return normalize_vec(a)
    return d / s

            best = cur_a.copy()
        active_mask = q >= (M - 0.02 * M)
        grad_blocks = np.zeros(n_blocks, dtype=np.float64)
        for k in np.where(active_mask)[0]:
            k = int(k)
            for b in range(n_blocks):
                i0 = b * B
                i1 = (b + 1) * B
                for i in range(i0, min(i1, n)):
                    j = k - i
                    if 0 <= j < n:
                        grad_blocks[b] += cur_a[j]
        grad_blocks = np.maximum(grad_blocks, 1e-30)
        eta_b = 0.5
        new_bs = block_sums * np.exp(-eta_b * grad_blocks / (block_sums + 1e-30))
        new_bs = np.maximum(new_bs, 0.0)
        s = float(new_bs.sum())
        if s > 1e-15:
            new_bs = new_bs / s
            test_a = np.zeros(n, dtype=np.float64)
            for b in range(n_blocks):
                test_a[b*B:(b+1)*B] = new_bs[b] / B
            if remain > 0:
                test_a[n_blocks*B:] = a[n_blocks*B:].copy()
            test_a = np.maximum(test_a, 1e-15)
            test_a = test_a / test_a.sum()
            sc = objective_norm(test_a)
            if sc < score - 1e-14:
                block_sums = new_bs
            else:
                eta_b *= 0.8
                if eta_b < 0.01:
                    block_sums = block_sums + 0.01 * rng.normal(size=n_blocks)
                    block_sums = np.maximum(block_sums, 0.0)
                    block_sums = block_sums / block_sums.sum()
    return best

def spectral_init(n, rng_seed=123):
    """Generate a sequence via spectral shaping."""
    rng = np.random.default_rng(rng_seed)
    size = next_pow2(2 * n - 1)
    A = np.zeros(size // 2 + 1, dtype=np.float64)
    for m in range(A.size):
        A[m] = 1.0 / (1.0 + 0.3 * (m / max(A.size - 1, 1)) ** 2)
    for _ in range(50):
        m_idx = int(rng.integers(0, A.size))
        A[m_idx] *= rng.uniform(0.5, 1.5)
    a = np.fft.irfft(A * A, size)[:n]
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(n, dtype=np.float64) / n
    return a / s

def make_diverse_starts(n, parent, num_starts=10):
    """Generate diverse starting points."""
    starts = []
    if parent is not None:
        p = normalize_vec(parent)
        if p.size != n:
            x_old = np.linspace(0.0, 1.0, p.size)
            x_new = np.linspace(0.0, 1.0, n)
            p = np.interp(x_new, x_old, p)
            p = normalize_vec(p)
        starts.append(p.copy())
        rev = p[::-1].copy()
        starts.append(rev)
        for w in (0.3, 0.5, 0.7):
            starts.append(normalize_vec((1.0 - w) * p + w * rev))
        base = np.maximum(p, 1e-30)
        for g in (0.5, 0.7, 1.3, 2.0):
            starts.append(normalize_vec(base ** g))
    x = np.arange(n, dtype=np.float64) / float(max(n - 1, 1))
    for aexp in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
        for bexp in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
            seq = (x + 1e-6) ** aexp * ((1.0 - x) + 1e-6) ** bexp
            starts.append(normalize_vec(seq))
    rng = np.random.default_rng(2024)
    for _ in range(8):
        seq = spectral_init(n, rng_seed=int(rng.integers(0, 100000)))
        starts.append(seq)
    for _ in range(5):
        num_peaks = int(rng.integers(1, 4))
        seq = np.zeros(n, dtype=np.float64)
        for _ in range(num_peaks):
            ctr = rng.uniform(0.1, 0.9)
            wd = rng.uniform(0.03, 0.3)
            ht = rng.uniform(0.3, 1.0)
            seq += ht * np.exp(-((x - ctr) ** 2) / (2.0 * wd * wd))
        seq += 1e-5
        starts.append(normalize_vec(seq))
    return starts

def select_diverse(cand_scores, max_kept=6):
    """Select diverse candidates by score and dissimilarity."""
    remaining = list(cand_scores)
    kept = []
    for _ in range(max_kept):
        bi = -1
        bs = float('inf')
        for i, item in enumerate(remaining):
            sc = float(item[0])
            if np.isfinite(sc) and sc < bs:
                bs = sc
                bi = i
        if bi < 0:
            break
        seq = remaining[bi][1]
        ok = True
        for kseq in kept:
            if seq.size == kseq.size:
                dp = float(np.dot(seq, kseq))
                if dp > 0.997:
                    ok = False
                    break
        if ok:
            kept.append(seq.copy())
        remaining.pop(bi)
    return kept

def interpolate_seq(f, new_n):
    f = normalize_vec(f)
    old_n = f.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return f.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f)
    return normalize_vec(g)

def power_blend_search(a0, time_limit):
    """Try power transforms and blends with uniform."""
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    best = a.copy()
    best_score = objective_norm(a)
    gammas = (0.3, 0.5, 0.7, 0.8, 1.2, 1.5, 2.0, 3.0)
    for g in gammas:
        if time.monotonic() - t0 > time_limit:
            break
        b = np.maximum(a, 1e-30) ** g
        b = normalize_vec(b)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()
    u = np.ones(n, dtype=np.float64) / n
    for w in (0.05, 0.1, 0.2, 0.3, 0.5):
        if time.monotonic() - t0 > time_limit:
            break
        b = normalize_vec((1.0 - w) * best + w * u)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()
    rev = best[::-1].copy()
    for w in (0.1, 0.2, 0.3, 0.5, 0.7):
        if time.monotonic() - t0 > time_limit:
            break
        b = normalize_vec((1.0 - w) * best + w * rev)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()
    return best

def trim_search(a0, time_limit):
    """Try trimming from edges."""
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n < 50 or time_limit <= 0.0:
        return a
    best = a.copy()
    best_score = objective_norm(a)
    max_side = min(n // 4, 3000)
    samples = list(range(0, max_side + 1, max(max_side // 20, 1)))
    if samples[-1] != max_side:
        samples.append(max_side)
    for l in samples:
        if time.monotonic() - t0 > time_limit:
            break
        if l >= n - 20:
            continue
        b = normalize_vec(a[l:])
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()
    a = best
    n = a.size
    max_side = min(n // 4, 3000)
    samples = list(range(0, max_side + 1, max(max_side // 20, 1)))
    for r in samples:
        if time.monotonic() - t0 > time_limit:
            break
        if r >= n - 20:
            continue
        b = normalize_vec(a[:-r])
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()
    return best
