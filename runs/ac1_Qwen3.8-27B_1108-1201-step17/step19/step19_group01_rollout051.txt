import scipy.sparse as sp

# ... helper functions ...

def lp_step(y, active_set, n, solver):
    """Solve one LP-FW step."""
    K = len(active_set)
    if K == 0:
        return np.ones(n) / n, 0.0
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for jj, k in enumerate(active_set):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        rows.extend([jj] * len(idx))
        cols.extend(idx.tolist())
        vals.extend(y[k - idx].tolist())
    
    A = sp.coo_matrix((vals, (rows, cols)), shape=(K, n)).tocsr()
    
    # LP: min M s.t. sum(g)=1, g>=0, A@g <= M
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [cp.sum(g) == 1, A @ g <= M]
    prob = cp.Problem(cp.Minimize(M), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    g_val = np.maximum(g.value, 0)
    g_sum = g_val.sum()
    if g_sum < 1e-15:
        return None, None
    return g_val / g_sum, M.value

"""
Minimizes 2*n*max(autocorr(a))/(sum(a))^2 using:
1. Nesterov accelerated projected gradient with log-sum-exp smoothing
2. LP-Frank-Wolfe with sparse constraint matrices and adaptive active set
3. Peak-directed coordinate descent for polishing
4. Multi-length exploration with interpolation
Key improvements: sparse LP constraints, better active set management,
faster solver usage, and adaptive step sizes.
"""

import numpy as np
import time
import scipy.sparse as sp
import cvxpy as cp

_SOLVER = None
_SOLVER_TRIED = False


def get_solver():
    global _SOLVER, _SOLVER_TRIED
    if _SOLVER TRIED:
        return _SOLVER
    _SOLVER_TRIED = True
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _SOLVER = cp.SCS
    print("  Solver (fallback): SCS")
    return s


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(y, beta, top_k=None):
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, p, b


def nesterov_accelerated(a0, time_limit, beta_start=20.0, beta_end=800.0,
                         beta_growth=1.003, eta_scale=1.0,
                         top_k_ratio=0.04, restart_every=2500, verbose=True):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(123 + n % 1000)
    top_k = max(20, int(n * top_k_ratio))

    for t in range(1, 80000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, b = compute_smoothed_grad(y, beta, top_k)

        eta = (eta_scale / max(1.0, beta * p.max() * 2.0))
        eta = min(eta, 0.12)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 60 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.015 / (1.0 + 0.05 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 5.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " beta=" + repr(round(beta, 1)))

    return best_a, best_max_b


def build_sparse_constraint(active_set, y, n):
    """Build sparse constraint matrix A where A[k,j] = y[k-j] for valid j."""
    rows = []
    cols = []
    vals = []
    K = len(active_set)
    for jj, k in enumerate(active_set):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        rows.extend([jj] * len(idx))
        cols.extend(idx.tolist())
        vals.extend(y[k_int - idx].tolist())
    if len(rows) == 0:
        return sp.csr_matrix((K, n))
    A = sp.coo_matrix((vals, (rows, cols)), shape=(K, n)).tocsr()
    return A


def solve_lp_fw_step(y, active_set, n, solver):
    """Solve one LP-FW step: min M s.t. sum(g)=1, g>=0, A@g<=M."""
    K = len(active_set)
    if K == 0:
        return None, None

    A = build_sparse_constraint(active_set, y, n)
    if A.nnz == 0:
        return None, None

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1, A @ g_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
    g_sum = g_val.sum()
    if g_sum < 1e-15:
        return None, None
    return g_val / g_sum, float(M_var.value)


def lp_fw_sparse(a0, time_limit, solver, initial_K=300, max_K=2500,
                 add_per_iter=20, remove_margin=0.002, verbose=True):
    """LP-Frank-Wolfe with sparse constraints and adaptive active set."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 800
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: converged at iter " + str(t))
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive active set management
        if t % 5 == 0:
            # Add violated constraints
            violations = np.where(ac > M_curr + 1e-8)[0]
            if len(violations) > 0:
                viol_amt = ac[violations] - M_curr
                top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
                for k in top_v:
                    active_set.add(int(k))

            # Remove non-binding constraints (if set is too large)
            if len(active_set) > max_K:
                margins = np.array([M_curr - ac[k] for k in active_set])
                keep = np.argsort(margins)[:max_K]
                active_set = set(int(list(active_set)[i]) for i in keep)
            elif len(active_set) > initial_K * 2:
                # Periodically prune: keep only near-binding constraints
                margins = np.array([M_curr - ac[k] for k in active_set])
                threshold = margins.min() + remove_margin
                active_set = set(int(k) for k, m in zip(active_set, margins) if m <= threshold)
                if len(active_set) < initial_K // 2:
                    active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])

        K = len(active_set)
        if K == 0:
            break

        # Solve LP
        g_dir, M_lp = solve_lp_fw_step(y, active_set, n, solver)
        if g_dir is None:
            # Fallback: reduce active set
            active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K // 2])
            continue

        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5),
                  max(0.02, default_alpha * 1.5), 1.0 / (t + 1)]
        alphas = list(set([max(0.01, min(1.0, a)) for a in alphas]))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True):
    """Peak-directed coordinate descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    max_stall = 600

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            src_val = a[src]
            partner_val = a[partner] if 0 <= partner < n else 0.0

            if src_val >= partner_val:
                reduce_idx = src
            else:
                reduce_idx = partner

            if a[reduce_idx] < 1e-12:
                continue

            # Find low-pressure destination
            pressure = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)

            cand_pressure = pressure[candidates]
            dst = int(candidates[np.argmin(cand_pressure)])
            if dst == reduce_idx:
                mask = candidates != reduce_idx
                if mask.sum() > 0:
                    dst = int(candidates[mask][np.argmin(cand_pressure[mask])])
                else:
                    continue

            found = False
            for frac in [0.02, 0.04, 0.08, 0.12]:
                amt = a[reduce_idx] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 100:
                j = int(rng.integers(n))
                if j != reduce_idx and a[reduce_idx] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[reduce_idx] * frac
                    a[reduce_idx] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0:
                for _ in range(3):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.003, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " stall=" + str(stall))

    return best_a, best_m


def parametric_init(n, time_limit, best_init, best_score):
    """Quick parametric search for good initialization."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score

    candidates = []
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    for p in [0.5, 1.0, 1.5]:
        candidates.append((i_arr + 1.0) ** p * (n - i_arr) ** p)
    candidates.append(np.sin(np.pi * x) ** 2)
    candidates.append(np.sin(np.pi * x) ** 4)
    candidates.append(np.maximum((1.0 + np.cos(2.0 * np.pi * x)) / 2.0, 0.01))
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.15, 0.2, 0.25, 0.3]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))

    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(np.maximum(a_raw, 0.0))
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    rng = np.random.default_rng(42)
    for _ in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(rng.random(n) ** rng.uniform(0.5, 3.0))
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    return best_a, best_val


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score
