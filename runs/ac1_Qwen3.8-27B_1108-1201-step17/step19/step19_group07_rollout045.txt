def fast_autocorr(a):

    A = np.fft.rfft(a, fft_size)

def norm_sum1(a):

def compute_subgradient(a, top_k=10):
    """Compute the subgradient of max(conv(a,a)) at a.
    
    Returns the subgradient vector g where g[j] = sum_k w_k * 2*a[k-j]
    with w_k proportional to exp(beta * (c[k] - c_max)) for top-k positions.
    """
    n = len(a)
    c = fast_autocorr(a)
    c_max = c.max()
    # Use top-k positions with softmax weights
    idx = np.argsort(c)[::-1][:top_k]
    weights = np.exp(50 * (c[idx] - c_max))
    weights /= weights.sum()
    
    g = np.zeros(n)
    for w_k, k in zip(weights, idx):
        k = int(k)
        # g[j] += 2 * w_k * a[k-j] for valid j
        # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        # a[k-j] for j in [j_start, j_end) is a[k-j_end+1 : k-j_start+1]
        # = a[k-(j_end-1) : k-j_start+1] = a[k-j_end+1 : k-j_start+1]
        # Let's just compute directly:
        for j in range(j_start, j_end):
            m = k - j
            if 0 <= m < n:
                g[j] += 2.0 * w_k * a[m]
    
    return g, c

        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_x = x.copy()
        
        # Top-k subgradient
        idx = np.argsort(c)[::-1][:top_k]
        weights = np.ones(top_k) / top_k
        # Actually use softmax for better weighting
        weights = np.exp(20 * (c[idx] - c[idx][0]))
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for ii in range(top_k):
            k = int(idx[ii])
            wk = weights[ii] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            if j_start < j_end:
                m_start = k - (j_end - 1)
                m_end = k - j_start + 1
                grad[j_start:j_end] += wk * x[m_start:m_end][::-1]
        
        # Mirror descent step
        log_x = np.log(np.maximum(x, 1e-30)) - step_size * grad
        log_x -= log_x.max()
        z = np.exp(log_x)
        z = np.maximum(z, 1e-30)
        x = z / z.sum()
    
    return best_x


def lp_frank_wolfe(a0, time_limit, solver, initial_K=500, max_K=3000,
                    line_search=True, verbose=True):
    """LP-based Frank-Wolfe with exact line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    
    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()
    
    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)
    
    iter_count = 0
    failed_solves = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
        
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx_arr = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx_arr] = y[k_int - idx_arr]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 8:
                constraint_list = sorted(constraint_set)[:min(init_K, len(constraint_set))]
                constraint_set = set(constraint_list)
                failed_solves = 0
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violated constraints and add them
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # Line search
        if line_search:
            best_t = 0.0
            best_val = M_curr
            for t_candidate in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
                x_try = (1.0 - t_candidate) * y + t_candidate * g_dir
                x_try = np.maximum(x_try, 0.0)
                s_try = x_try.sum()
                if s_try < 1e-15:
                    continue
                x_try = x_try / s_try
                m_try = fast_autocorr(x_try).max()
                if m_try < best_val:
                    best_val = m_try
                    best_t = t_candidate
            
            y_new = (1.0 - best_t) * y + best_t * g_dir
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / sy
        else:
            # Standard FW step
            t_candidate = 2.0 / (iter_count + 2.0)
            y_new = (1.0 - t_candidate) * y + t_candidate * g_dir
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / sy
        
        y = y_new
        iter_count += 1
        
        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")
    
    return best_a, best_m
