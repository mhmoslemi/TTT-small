"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.

Approach:
1. Log-sum-exp smoothed gradient descent with FFT-based gradient computation.
2. Accelerated mirror descent (multiplicative weights) with adaptive temperature.
3. Periodic exact subgradient Frank-Wolfe steps for fine-tuning.
4. Multi-start with diverse initializations and length exploration.
5. Final coordinate-descent polishing phase.

The key insight: the problem is convex (min max of quadratics over simplex),

import numpy as np
import time
import math


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a) / max(len(a), 1)
    return a / s


def _self_conv(a):
    n = a.size
    m = 2 * n - 1
    size = _next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]


def _score(a):
    b = _self_conv(a)
    return 2.0 * a.size * float(np.max(b))


def _grad_smoothed(a, b, T):
    """Compute gradient of (1/T)log(sum exp(T*b_k)) via FFT cross-correlation."""
    n = a.size
    m = b.size  # 2n-1
    # softmax(T * b)
    Tb = T * b
    Tb_max = float(np.max(Tb))
    w = np.exp(Tb - Tb_max)
    w_sum = float(w.sum())
    if w_sum <= 1e-300:
        w = np.ones(m) / m
    else:
        w = w / w_sum
    # grad[m] = 2 * sum_j a[j] * w[m+j] for m=0..n-1
    # This is cross-correlation: IFFT(FFT(w) * conj(FFT(a)))
    size = _next_pow2(m + n - 1)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    corr = np.fft.irfft(fw * fa.conj(), size)
    return 2.0 * corr[:n]


def _grad_exact(a, b):
    """Compute subgradient: 2*v_{k*} where k* = argmax b."""
    n = a.size
    k_star = int(np.argmax(b))
    # v_k[m] = a[k-m] for valid m
    grad = np.zeros(n, dtype=np.float64)
    i0 = max(0, k_star - n + 1)
    i1 = min(n, k_star + 1)
    # valid m range: m = k_star - j for j in [i0, i1), so m in [k_star-i1+1, k_star-i0]
    m_lo = max(0, k_star - i1 + 1)
    m_hi = min(n - 1, k_star - i0)
    if m_lo <= m_hi:
        j_vals = k_star - np.arange(m_lo, m_hi + 1)
        grad[m_lo:m_hi + 1] = a[j_vals]
    return 2.0 * grad


def _mirror_step(a, grad, eta):
    """Multiplicative weights update: a * exp(-eta*grad) normalized."""
    log_r = -eta * grad
    log_r = np.clip(log_r, -60.0, 60.0)
    a_new = a * np.exp(log_r)
    s = float(a_new.sum())
    if s <= 1e-15:
        return a.copy()
    return a_new / s


def _simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            importance[m_lo:m_hi + 1] = a[j_vals]

        # Also consider a few other top lags
        order = np.argsort(b)[::-1]
        for kk in order[1:min(5, len(order))]:
            k = int(kk)
            i0 = max(0, k - n + 1)
            i1 = min(n, k + 1)
            m_lo = max(0, k - i1 + 1)
            m_hi = min(n - 1, k - i0)
            if m_lo <= m_hi:
                j_vals = k - np.arange(m_lo, m_hi + 1)
                importance[m_lo:m_hi + 1] += 0.5 * a[j_vals]

        # Move mass from high-importance to low-importance
        src_idx = np.argsort(importance)[::-1]
        dst_idx = np.argsort(importance)[:min(20, n)]

        improved = False
        for s_i in src_idx[:10]:
            if improved:
                break
            s_i = int(s_i)
            if a[s_i] <= 1e-14:
                continue
            for d_i in dst_idx:
                d_i = int(d_i)
                if d_i == s_i:
                    continue
                for frac in (0.01, 0.03, 0.05, 0.1):
                    amt = a[s_i] * frac
                    if amt < 1e-16:
                        continue
                    a_new = a.copy()
                    a_new[s_i] -= amt
                    a_new[d_i] += amt
                    if a_new[s_i] < -1e-14:
                        continue
                    a_new = np.maximum(a_new, 0.0)
                    sc = _score(a_new)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best = a_new.copy()
                        a = a_new
                        improved = True
                        break
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1

        if time.monotonic() - t0 % 30 < 0.1:
            pass  # silent

    return best, best_score


def _try_length(a, new_n, time_limit, phase_fn, **kwargs):
    """Interpolate to new length and run optimization phase."""
    a_old = _normalize(a)
    n_old = a_old.size
    if n_old == new_n:
        a_int = a_old.copy()
    else:
        x_old = np.linspace(0, 1, n_old)
        x_new = np.linspace(0, 1, new_n)
        a_int = np.interp(x_new, x_old, a_old)
        a_int = _normalize(a_int)
    return phase_fn(a_int, time_limit, **kwargs)
