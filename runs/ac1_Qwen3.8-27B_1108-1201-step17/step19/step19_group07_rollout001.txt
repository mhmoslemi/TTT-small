    y = x.copy()
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_threshold = 3000
    best_eval_score = evaluate_safe(best_x)
    
    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        progress = min(iter_count / max(time_limit * 50, 1), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        top_k = int(top_k_start + (top_k_end - top_k_start) * progress)
        top_k = min(top_k, 2 * n - 1)

        z = np.exp(log_y)
        z = np.maximum(z, 1e-30)
        z = z / z.sum()
        
        # Nesterov momentum
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)

            best_x = x_new.copy()
            stall = 0

            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Reduce constraints on failure
            constraint_list = sorted(constraint_set)[:min(initial_K, len(constraint_set))]
            constraint_set = set(constraint_list)
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:60]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # Also add top autocorrelation values of y that aren't in the set
        top_ac = np.argsort(ac)[::-1][:20]
        for k in top_ac:
            if k not in constraint_set:
                constraint_set.add(int(k))
        
        # Nesterov acceleration
        alpha = 2.0 / (iter_count + 3.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if iter_count == 0:
            x_new = y_new.copy()
        else:
            beta_t = (iter_count - 1.0) / (iter_count + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")
    
    return best_a, best_m

def multi_scale_init(target_n, time_limit, solver, verbose=True):
    """Multi-scale: optimize at small n, interpolate up."""
    scales = [50, 200, min(1000, target_n), target_n]
    scales = [s for s in scales if s <= target_n]
    scales = sorted(set(scales))
    
    if scales[-1] != target_n:
        scales.append(target_n)
    
    a = np.ones(50) / 50.0  # Start uniform
    
    t0 = time.monotonic()
    for scale in scales:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        if len(a) != scale:
            # Interpolate
            old_n = len(a)
            if old_n == 1:
                a = np.ones(scale) / scale
            else:
                old_x = np.linspace(0, 1, old_n)
                new_x = np.linspace(0, 1, scale)
                a = np.interp(new_x, old_x, a)
                a = np.maximum(a, 0)
                a = a / a.sum()
        
        # FISTA at this scale
        t_fista = min(remaining * 0.7, 30.0 if scale < 1000 else 60.0)
        if t_fista > 5:
            a, _, _ = fista_optimize(a, t_fista, beta_start=5.0, beta_end=50.0,
                                     eta0=0.8, top_k_start=20, top_k_end=min(100, 2*scale-1),
                                     verbose=verbose and scale >= 1000)
        
        # LP at this scale (only for small scales where it's fast)
        remaining = time_limit - (time.monotonic() - t0)
        if scale <= 500 and remaining > 10:
            t_lp = min(remaining * 0.3, 20.0)
            if t_lp > 5:
                a, _ = lp_frank_wolfe(a, t_lp, solver, initial_K=50, max_K=200,
                                       verbose=verbose and scale >= 200)
        
        print(f"  Multi-scale: n={scale}, J={2*scale*fast_autocorr(a).max():.6f}, t={time.monotonic()-t0:.0f}s")
    
    # Final interpolation to target_n
    if len(a) != target_n:
        old_x = np.linspace(0, 1, len(a))
        new_x = np.linspace(0, 1, target_n)
        a = np.interp(new_x, old_x, a)
        a = np.maximum(a, 0)
        a = a / a.sum()
    
    return a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(30, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS
