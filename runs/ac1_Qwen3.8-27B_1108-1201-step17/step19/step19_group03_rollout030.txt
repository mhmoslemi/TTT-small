def fast_autocorr(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

Now let me think about the spectral refinement:

The idea is that at the optimum, the "energy" of the autocorrelation should be as flat as possible (excluding the DC component). We can use an iterative spectral method:

1. Compute FFT of a
2. The autocorrelation's FFT is |A(f)|^2
3. We want |A(f)|^2 to be as flat as possible (to minimize the peak of the inverse FFT)
4. But we also need a >= 0

def spectral_refine(a0, time_limit, verbose=False):
    n = len(a0)

    best_a = a.copy()

        # Compute current spectrum
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat magnitude (geometric mean of current)
        # Use log-domain correction
        log_mag = np.log(np.maximum(mag, 1e-30))
        log_target = np.mean(log_mag)
        
        # Blend: move toward flat
        blend = 0.1  # gentle blending
        log_mag_new = (1 - blend) * log_mag + blend * log_target
        mag_new = np.exp(log_mag_new)
        
        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

                A_new = mag_new * np.exp(1j * phase)

The constraint (a*a)_k <= M for each k is:

However, the FW approach works because:
- We're at a point a in the simplex
- We compute the "descent direction" g by solving an LP that approximates the best direction
- The LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0
  Here (a*b)_k = sum_j a[j] * b[k-j] is LINEAR in b (since a is fixed)
  So this LP is valid and convex.

The insight is that by fixing a and optimizing over b, we get a direction that, if followed, reduces the max convolution. This works because:
- At the current point a, the "gradient" of max_k (a*a)_k is dominated by the active constraints
- The LP direction g is the point in the simplex that best "balances" the constraints
- Moving towards g reduces the active constraint violations

This is a form of "block coordinate descent" where we alternate between the "a-block" (fixed) and the "b-block" (optimized via LP).

    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.12) / width) ** 2) + np.exp(-((x - 0.88) / width) ** 2) + 0.003
    return norm_sum1(a)


def make_cosine(n, alpha=0.4):
    x = np.arange(n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * x / n)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_poweredge(n, p=3.0):
    x = np.linspace(0, 1, n)
    a = x**p + (1-x)**p + 0.01
    return norm_sum1(a)


def spectral_init(n, n_modes=30, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lp_fw_adaptive(a0, time_limit, solver, verbose=False, K_init=300, K_max=None,
                   spectral_mix_every=50, spectral_mix_strength=0.05):
    """LP-Frank-Wolfe with adaptive binding constraints and golden-section line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if K_max is None:
        K_max = min(3000, 2 * n - 1)

    K = K_init
    it = 0
    stall = 0
    max_stall = 80
    rng = np.random.default_rng(12345)
    fft_size = 1 << (2 * n - 2).bit_length()

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < 1e-18:
            break

        # Adaptive constraint selection
        K_sel = min(K, 2 * n - 1)
        sorted_idx = np.argsort(ac)[::-1]
        threshold = 0.98 * M_curr
        binding = np.where(ac >= threshold)[0]
        top_k = sorted_idx[:K_sel]
        all_c = np.unique(np.concatenate([binding, top_k]))
        all_c = all_c[:min(K_sel + len(binding), 2 * n - 1)]
        K_act = len(all_c)

        # Build constraint matrix T where T[i,j] = a[k_i - j] if valid
        T = np.zeros((K_act, n))
        for ii, k in enumerate(all_c):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T[ii, j] = a[ki - j]

        # Solve LP: max sum(b) s.t. T @ b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        lp_start = time.monotonic()
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, K_max)
            continue
        lp_time = time.monotonic() - lp_start

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 30, K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            K = min(K + 30, K_max)
            continue

        g = b_val / b_sum

        # Check if g is meaningfully different from a
        if np.allclose(g, a, atol=1e-8):
            K = min(K + 50, K_max)
            stall += 1
            if stall > max_stall:
                break
            continue

        # Golden-section line search
        def obj_alpha(alpha):
            a_t = (1.0 - alpha) * a + alpha * g
            a_t = np.maximum(a_t, 0.0)
            s_t = a_t.sum()
            if s_t < 1e-15:
                return 1e10
            a_t = a_t / s_t
            return fast_autocorr(a_t).max()

        # Golden section in [0, 1]
        gratio = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        x1 = hi - gratio * (hi - lo)
        x2 = lo + gratio * (hi - lo)
        f1 = obj_alpha(x1)
        f2 = obj_alpha(x2)

        n_gs = 18
        for _ in range(n_gs):
            if time.monotonic() - t0 > time_limit:
                break
            if f1 < f2:
                hi = x2
                x2 = x1
                f2 = f1
                x1 = hi - gratio * (hi - lo)
                f1 = obj_alpha(x1)
            else:
                lo = x1
                x1 = x2
                f1 = f2
                x2 = lo + gratio * (hi - lo)
                f2 = obj_alpha(x2)

        best_alpha = (lo + hi) / 2.0
        m_alpha = obj_alpha(best_alpha)

        # Also evaluate a few fixed alphas for safety
        best_val = m_alpha
        for fa in [2.0 / (it + 2.0), 0.1, 0.2]:
            if 0 < fa < 1:
                v = obj_alpha(fa)
                if v < best_val:
                    best_val = v
                    best_alpha = fa

        if best_val >= M_curr * (1 - 1e-10):
            # No improvement
            K = min(K + 40, K_max)
            stall += 1
            if stall > max_stall:
                break
            # Try spectral perturbation to escape
            if stall > max_stall // 2:
                A = np.fft.rfft(a, fft_size)
                mag = np.abs(A)
                mag_mean = np.mean(mag)
                new_mag = 0.8 * mag + 0.2 * mag_mean
                A_new = new_mag * np.exp(1j * np.angle(A))
                a_p = np.fft.irfft(A_new, fft_size)[:n]
                a_p = np.maximum(a_p, 0.0)
                sp = a_p.sum()
                if sp > 1e-15:
                    a_p = a_p / sp
                    mp = fast_autocorr(a_p).max()
                    if mp < best_m:
                        a = a_p
                        best_m = mp
                        best_a = a.copy()
                        stall = 0
                        best_alpha = 0.5
                    else:
                        a = (1.0 - 0.3) * a + 0.3 * a_p
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                continue

        # Update
        a = (1.0 - best_alpha) * a + best_alpha * g
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        it += 1

        # Progressively increase K
        if it % 8 == 0:
            K = min(K + 25, K_max)

        # Periodic spectral mixing to avoid stagnation
        if spectral_mix_every > 0 and it % spectral_mix_every == 0:
            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            new_mag = (1 - spectral_mix_strength) * mag + spectral_mix_strength * mag_mean
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_mix = np.fft.irfft(A_new, fft_size)[:n]
            a_mix = np.maximum(a_mix, 0.0)
            sm = a_mix.sum()
            if sm > 1e-15:
                a_mix = a_mix / sm
                mm = fast_autocorr(a_mix).max()
                if mm < best_m:
                    best_m = mm
                    best_a = a_mix.copy()
                    a = a_mix

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW it={it} ({elapsed:.0f}s): J={J:.6f}, K={K_act}, alpha={best_alpha:.4f}, lp_t={lp_time:.2f}s")

    return best_a, best_m


def peak_shaving(a0, time_limit, verbose=False, rng_seed=42):
    """Reduce peak by moving mass from high-contributing pairs to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 300
    it = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_lags = np.argsort(ac)[::-1][:15]

        improved = False
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            idx = np.arange(j_start, j_end)
            contrib = a[idx] * a[ki - idx]

            if contrib.max() < 1e-15:
                continue

            # Get top contributing pairs
            pair_order = np.argsort(contrib)[::-1][:10]
            for pi in pair_order:
                j1 = int(idx[pi])
                j2 = int(ki - j1)
                if j2 < 0 or j2 >= n:
                    continue
                pair_val = a[j1] * a[j2]
                if pair_val < 1e-15:
                    continue

                # Move mass from the larger element
                if a[j1] >= a[j2]:
                    src = j1
                else:
                    src = j2
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Destination candidates: low-sensitivity positions
                # Sensitivity approx: for position d, it affects ac[k] for k in [d, d+n-1]
                # We want low sensitivity at peak lags
                n_cand = 8
                # Try positions far from peak lag contributions
                cand_set = set()
                for _ in range(n_cand * 2):
                    c = int(rng.integers(n))
                    cand_set.add(c)
                # Also add low-value positions
                low_positions = np.where(a < 0.05 * a.max())[0]
                if len(low_positions) > 0:
                    for _ in range(5):
                        c = int(low_positions[rng.integers(len(low_positions))])
                        cand_set.add(c)
                cand_set.discard(src)

                for dst in list(cand_set)[:n_cand]:
                    for frac in [0.005, 0.01, 0.02, 0.03]:
                        amt = src_val * frac
                        if amt < 1e-15:
                            continue
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        s_t = a_t.sum()
                        if s_t < 1e-15:
                            continue
                        a_t = a_t / s_t
                        m_t = fast_autocorr(a_t).max()
                        if m_t < best_m - 1e-16:
                            a = a_t
                            best_m = m_t
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation when stalled
            if stall % 50 == 0:
                n_pert = rng.integers(5, 20)
                srcs = rng.choice(n, size=n_pert, replace=False)
                dsts = rng.choice(n, size=n_pert, replace=False)
                a_p = a.copy()
                for s, d in zip(srcs, dsts):
                    if a_p[s] > 1e-10:
                        amt = a_p[s] * rng.uniform(0.01, 0.05)
                        a_p[s] -= amt
                        a_p[d] += amt
                a_p = np.maximum(a_p, 0.0)
                sp = a_p.sum()
                if sp > 1e-15:
                    a_p = a_p / sp
                    mp = fast_autocorr(a_p).max()
                    if mp < best_m:
                        a = a_p
                        best_m = mp
                        best_a = a.copy()
                        stall = 0
        else:
            stall = 0

        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave it={it} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m
