def normalize(a):
    a=np.asarray(a,dtype=np.float64).ravel()

def interpolate_to(a,new_n):

```python
"""
Search for low-max self-convolution non-negative step heights.

Algorithm summary:
- Maintain the sequence normalized to unit sum.
- Internal objective is fast FFT-based: 2*n*max(a*a)/(sum(a))^2.
- Main phase: adaptive-weight subgradient/mirror updates.
  * active lags are the largest convolution coefficients;
  * dual weights over active lags are updated to make the weighted
    gradient as flat as possible;
  * trial moves include multiplicative mirror steps and direct mass
    transfers from high-gradient coordinates to low-gradient ones.
- Polishing phase: local mass-transfer moves with exact quadratic
  estimates for the current top lags, followed by full FFT validation.
- Length sweeps and stochastic restarts are used to avoid local traps.
"""

import numpy as np
import time


def normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15 or a.size == 0:
        if a.size == 0:
            return np.ones(2, dtype=np.float64) * 0.5
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    a = a / s
    a = np.minimum(a, 1000.0)
    return a


def fast_conv_pair(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = len(a) + len(b) - 1
    if n <= 0:
        return np.array([], dtype=np.float64)
    size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    out = np.fft.irfft(A * B, size)[:n]
    return out


def fast_conv(a):
    return fast_conv_pair(a, a)


def conv_max(a):
    c = fast_conv(a)
    return float(np.max(c)) if c.size else 0.0


def fast_obj(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if s < 1e-15:
        return float("inf")
    m = conv_max(a)
    if m < 0.0:
        m = 0.0
    return float(2.0 * n * m / (s * s))


def eval_safe(a):
    try:
        func = globals().get("evaluate_sequence", None)
        if func is None:
            return float("inf")
        arr = np.asarray(a, dtype=np.float64).ravel()
        if arr.size == 0 or not np.isfinite(arr).all():
            return float("inf")
        lst = [float(x) for x in np.maximum(arr, 0.0)]
        return float(func(lst))
    except Exception:
        return float("inf")


def interpolate_to(a, new_n):
    old = np.asarray(a, dtype=np.float64).ravel()
    old_n = len(old)
    new_n = int(new_n)
    if old_n == new_n:
        return old.copy()
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) * 0.5
    f = normalize(old)
    if old_n == 1:
        return np.full(new_n, f[0], dtype=np.float64)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, f)
    return normalize(out)


def make_template(profile, n, seed=0):
    n = int(n)
    if n <= 0:
        return np.ones(2, dtype=np.float64) * 0.5
    x = np.linspace(0.0, 1.0, n)
    if profile == "sine":
        f = np.sin(np.pi * x / 2.0)
    elif profile == "sine2":
        f = np.sin(np.pi * x / 2.0) ** 2
    elif profile == "rootsine":
        f = np.sin(np.pi * x / 2.0) ** 0.5
    elif profile == "lin_up":
        f = 1.0 + 0.4 * x
    elif profile == "lin_down":
        f = 1.0 + 0.4 * (1.0 - x)
    elif profile == "u":
        f = np.ones(n, dtype=np.float64)
        m = max(2, n // 6)
        f[:m] = 1.6
        f[-m:] = 1.6
    elif profile == "random":
        return random_smooth(n, int(seed))
    else:
        f = np.sin(np.pi * x / 2.0)
    return normalize(f)


def random_smooth(n, seed):
    rng = np.random.default_rng(int(seed))
    x = np.linspace(0.0, 1.0, n)
    f = np.ones(n, dtype=np.float64)
    for k in range(1, 8):
        amp = np.exp(-0.30 * k) * rng.uniform(0.05, 0.55)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        f = f + amp * np.cos(2.0 * np.pi * k * x + ph)
    f = f - f.min() + 0.10
    return normalize(f)


def choose_active(c, M, K):
    c = np.asarray(c, dtype=np.float64).ravel()
    K = int(max(1, min(K, len(c))))
    idx = np.argsort(c)[::-1][:K].astype(int)
    if M > 0.0:
        close = np.where(c >= 0.985 * float(M))[0]
        if close.size > 0:
            if len(idx) == 0:
                new_add = close[:12]
            else:
                mask = ~np.isin(find_unique_idx(idx), close)
                new_add = close[mask][:12]
            idx = np.concatenate([idx, new_add])
    idx = find_unique_idx(idx)
    if idx.size > K + 20:
        order = np.argsort(c[idx])[::-1][:K]
        idx = idx[order]
    return idx


def find_unique_idx(idx):
    idx = np.asarray(idx, dtype=int).ravel()
    if idx.size == 0:
        return idx
    return np.unique(idx)


def build_gradient(p, active, weights):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = len(p)
    g = np.zeros(n, dtype=np.float64)
    active = np.asarray(active, dtype=int).ravel()
    weights = np.asarray(weights, dtype=np.float64).ravel()
    if active.size == 0 or n == 0:
        return g
    for k, w in zip(active, weights):
        if w == 0.0:
            continue
        ki = int(k)
        if ki < 0 or ki >= 2 * n - 1:
            continue
        i0 = 0 if ki <= n - 1 else ki - n + 1
        i1 = ki if ki < n else n - 1
        if i0 > i1:
            continue
        p0 = ki - i0
        p1 = ki - i1
        if p1 > 0:
            seg = p[p0:p1 - 1:-1]
        else:
            seg = p[p0::-1]
        if seg.size == i1 - i0 + 1:
            g[i0:i1 + 1] += float(w) * seg
    return g


def optimize_weights(p, active, prev_w, prev_active):
    active = np.asarray(active, dtype=int).ravel()
    K = len(active)
    if K == 0:
        return np.array([], dtype=np.float64)

    w = np.ones(K, dtype=np.float64) / float(K)
    if prev_w is not None and prev_active is not None and len(prev_active) > 0:
        prev_active = np.asarray(prev_active, dtype=int).ravel()
        prev_w = np.asarray(prev_w, dtype=np.float64).ravel()
        mapping = {}
        for i, k in enumerate(prev_active):
            mapping[int(k)] = i
        vals = []
        for k in active:
            j = mapping.get(int(k), None)
            vals.append(float(prev_w[j]) if j is not None else 0.0)
        s = sum(vals)
        if s > 1e-12:
            w = np.asarray(vals, dtype=np.float64) / s

    p = np.asarray(p, dtype=np.float64).ravel()
    g = build_gradient(p, active, w)
    mu = float(np.dot(p, g))
    r = p * (g - mu)
    if np.max(np.abs(r)) > 1e-15:
        pr = fast_conv_pair(r, p)
        grad = pr[active]
        kmin = int(np.argmin(grad))
        if kmin >= 0 and w[kmin] < 0.985:
            alpha = 0.35
            e = np.zeros(K, dtype=np.float64)
            e[kmin] = 1.0
            w = (1.0 - alpha) * w + alpha * e
            s = float(w.sum())
            if s > 0.0:
                w = w / s
            else:
                w = np.ones(K, dtype=np.float64) / float(K)
        elif w[kmin] >= 0.985:
            w = np.ones(K, dtype=np.float64) / float(K)
    return w


def awg_optimize(a0, time_limit=300.0, seed=1, K=None, verbose=True):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = len(a0)
    if n <= 1:
        p = normalize(a0)
        M = conv_max(p)
        return p, float(2.0 * n * M)

    if K is None:
        K = int(max(64, min(320, n // 50)))

    p = normalize(a0)
    c = fast_conv(p)
    M = float(np.max(c)) if c.size else 0.0
    best_p = p.copy()
    best_M = M

    t0 = time.monotonic()
    rng = np.random.default_rng(int(seed))
    prev_w = None
    prev_active = None
    stall = 0

    for it in range(1, 1000000):
        if time.monotonic() - t0 > float(time_limit):
            break

        c = fast_conv(p)
        M = float(np.max(c)) if c.size else 0.0
        if M < best_M - 1e-15:
            best_M = M
            best_p = p.copy()
            stall = 0
        else:
            stall += 1

        if stall > 180:
            break

        active = choose_active(c, M, K)
        w = optimize_weights(p, active, prev_w, prev_active)
        prev_w = w
        prev_active = active.copy()

        g = build_gradient(p, active, w)
        mu = float(np.dot(p, g))
        h = g - mu
        rg = float(np.max(h) - np.min(h))

        best_trial_M = M
        best_trial_p = None

        if rg > 1e-15:
            for beta in (0.25, 0.50, 0.90, 1.60, 2.80):
                eta = beta / rg
                trial = p * np.exp(-eta * h)
                trial = np.maximum(trial, 0.0)
                s = float(trial.sum())
                if s < 1e-14:
                    continue
                trial = trial / s
                Mt = conv_max(trial)
                if Mt < best_trial_M - 1e-15:
                    best_trial_M = Mt
                    best_trial_p = trial

        pg = p * g
        if pg.size:
            src = int(np.argmax(pg))
            if p[src] > 1e-14:
                dst_order = np.argsort(g)[:3]
                done_dst = 0
                for di in dst_order:
                    di = int(di)
                    if di == src:
                        continue
                    for frac in (0.10, 0.30):
                        trial = p.copy()
                        amt = p[src] * frac
                        trial[src] -= amt
                        trial[di] += amt
                        trial = np.maximum(trial, 0.0)
                        s = float(trial.sum())
                        if s < 1e-14:
                            continue
                        trial = trial / s
                        Mt = conv_max(trial)
                        if Mt < best_trial_M - 1e-15:
                            best_trial_M = Mt
                            best_trial_p = trial
                    done_dst += 1
                    if done_dst >= 2:
                        break

        if best_trial_p is not None and best_trial_M < M - 1e-15:
            p = best_trial_p
            stall = 0
        else:
            if stall > 18:
                temp = min(0.22, 0.012 + 0.0025 * float(stall))
                noise = rng.standard_normal(n) * temp * p
                p = normalize(p + noise)
                if stall > 80:
                    p = normalize(best_p + 0.08 * rng.standard_normal(n) * best_p)
                    stall = 35

        if verbose and it % 50 == 0:
            el = time.monotonic() - t0
            print("    AWG n={} it={} t={:.1f}s J={:.8f} stall={}".format(
                n, it, el, 2.0 * n * M, stall))

    c = fast_conv(p)
    M = float(np.max(c)) if c.size else 0.0
    if M < best_M - 1e-15:
        best_M = M
        best_p = p.copy()
    return best_p, float(2.0 * n * best_M)


def estimate_top_after_move(c, top_lags, p, src, dst, delta):
    c = np.asarray(c, dtype=np.float64).ravel()
    top_lags = np.asarray(top_lags, dtype=int).ravel()
    p = np.asarray(p, dtype=np.float64).ravel()
    n = len(p)
    if top_lags.size == 0:
        return -1.0
    vals = c[top_lags].astype(np.float64).copy()
    s2 = 2 * int(src)
    d2 = 2 * int(dst)
    sd = int(src) + int(dst)
    for j, lag in enumerate(top_lags):
        lag = int(lag)
        idst = lag - int(dst)
        isrc = lag - int(src)
        a_dst = float(p[idst]) if 0 <= idst < n else 0.0
        a_src = float(p[isrc]) if 0 <= isrc < n else 0.0
        val = vals[j] + 2.0 * delta * (a_dst - a_src)
        if lag == d2:
            val += delta * delta
        if lag == sd:
            val -= 2.0 * delta * delta
        if lag == s2:
            val += delta * delta
        vals[j] = val
    return float(np.max(vals))


def coord_polish(a0, time_limit=300.0, seed=1, verbose=True):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = len(a0)
    if n <= 1:
        p = normalize(a0)
        M = conv_max(p)
        return p, float(2.0 * n * M)

    p = normalize(a0)
    c = fast_conv(p)
    M = float(np.max(c)) if c.size else 0.0
    best_p = p.copy()
    best_M = M

    t0 = time.monotonic()
    rng = np.random.default_rng(int(seed))
    stall = 0
    it = 0
    fracs = (0.02, 0.05, 0.10, 0.18, 0.32, 0.50)
    top_L = 32

    while time.monotonic() - t0 < float(time_limit):
        it += 1
        if stall > 320:
            break

        c = fast_conv(p)
        M = float(np.max(c)) if c.size else 0.0
        if M < best_M - 1e-15:
            best_M = M
            best_p = p.copy()
            stall = 0
        else:
            stall += 1

        if c.size <= top_L:
            top = np.arange(len(c), dtype=int)
        else:
            top = np.argsort(c)[::-1][:top_L].astype(int)

        score = np.zeros(n, dtype=np.float64)
        score_lags = top[:8]
        Mscale = max(float(M), 1e-300)
        for k in score_lags:
            k = int(k)
            i0 = 0 if k <= n - 1 else k - n + 1
            i1 = k if k < n else n - 1
            if i0 > i1:
                continue
            p0 = k - i0
            p1 = k - i1
            if p1 > 0:
                seg = p[p0:p1 - 1:-1]
            else:
                seg = p[p0::-1]
            if seg.size == i1 - i0 + 1:
                score[i0:i1 + 1] += (float(c[k]) / Mscale) * seg

        low_dst = np.argsort(score)[:14]
        src_list = []
        seen_src = set()
        for k in top[:5]:
            k = int(k)
            i0 = 0 if k <= n - 1 else k - n + 1
            i1 = k if k < n else n - 1
            if i0 > i1:
                continue
            idx = np.arange(i0, i1 + 1, dtype=int)
            p_rev = p[k - idx]
            contrib = p[idx] * p_rev
            order = np.argsort(contrib)[::-1][:14]
            for oi in order:
                sidx = int(idx[oi])
                if sidx in seen_src:
                    continue
                if p[sidx] < 1e-14:
                    continue
                seen_src.add(sidx)
                src_list.append(sidx)
                if len(src_list) >= 14:
                    break
            if len(src_list) >= 14:
                break

        candidates = []
        for sidx in src_list[:12]:
            for didx in low_dst:
                didx = int(didx)
                if didx == sidx:
                    continue
                best_est = float("inf")
                best_frac = None
                for frac in fracs:
                    delta = p[sidx] * frac
                    est = estimate_top_after_move(c, top, p, sidx, didx, delta)
                    if est < best_est:
                        best_est = est
                        best_frac = float(frac)
                if best_frac is not None:
                    candidates.append((best_est, int(sidx), int(didx), best_frac))

        if len(candidates) < 4:
            for _ in range(4):
                sidx = int(rng.integers(n))
                didx = int(rng.integers(n))
                if didx == sidx or p[sidx] < 1e-14:
                    continue
                best_est = float("inf")
                best_frac = None
                for frac in fracs:
                    delta = p[sidx] * frac
                    est = estimate_top_after_move(c, top, p, sidx, didx, delta)
                    if est < best_est:
                        best_est = est
                        best_frac = float(frac)
                if best_frac is not None:
                    candidates.append((best_est, sidx, didx, best_frac))

        candidates.sort(key=lambda x: x[0])
        good = [z for z in candidates if z[0] < float(M) - 1e-15]
        if len(good) < 2 and candidates:
            good = candidates[:max(1, min(4, len(candidates)))]

        improved = False
        for _, sidx, didx, frac in good[:6]:
            trial = p.copy()
            amt = p[sidx] * float(frac)
            trial[sidx] -= amt
            trial[didx] += amt
            trial = np.maximum(trial, 0.0)
            s = float(trial.sum())
            if s < 1e-14:
                continue
            trial = trial / s
            Mt = conv_max(trial)
            if Mt < M - 1e-15:
                p = trial
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 25:
                temp = min(0.18, 0.008 + 0.0022 * float(stall))
                p = normalize(p + temp * rng.standard_normal(n) * p)
                if stall > 130:
                    p = normalize(best_p + 0.10 * rng.standard_normal(n) * best_p)
                    stall = 60

        if verbose and it % 100 == 0:
            el = time.monotonic() - t0
            print("    Coord n={} it={} t={:.1f}s J={:.8f} stall={}".format(
                n, it, el, 2.0 * n * M, stall))

    c = fast_conv(p)
    M = float(np.max(c)) if c.size else 0.0
    if M < best_M - 1e-15:
        best_M = M
        best_p = p.copy()
    return best_p, float(2.0 * n * best_M)


def candidate_lengths(base_n):
    base_n = int(base_n)
    out = []
    for add in (-1600, -800, -400, 400, 800, 1600):
        out.append(base_n + add)
    for f in (0.88, 0.94, 1.0, 1.06, 1.12, 1.20, 1.30):
        out.append(int(base_n * f))
    seen = set()
    res = []
    for x in out:
        x = int(x)
        if x < 4000 or x > 16000:
            continue
        if x not in seen:
            seen.add(x)
            res.append(x)
    res = [x for x in res if x != base_n]
    res.sort(key=lambda z: abs(z - base_n))
    return res


def propose_candidate(budget_s=1800.0):
    t_global0 = time.monotonic()
    budget = max(15.0, min(float(budget_s) - 75.0, 1740.0))

    best_seq = None
    best_score = float("inf")

    try:
        par = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if par.size > 20 and np.isfinite(par).all() and float(np.sum(np.maximum(par, 0.0))) > 1e-10:
            p0 = normalize(par)
            s0 = fast_obj(p0)
            best_seq = p0
            best_score = s0
            print("Parent: n={} fast_J={:.8f}".format(len(p0), best_score))
    except Exception:
        pass

    if best_seq is None:
        n0 = 12000
        best_seq = make_template("sine", n0)
        best_score = fast_obj(best_seq)
        print("No parent: use sine n={} fast_J={:.8f}".format(n0, best_score))

    n0 = len(best_seq)

    profiles = ("sine", "sine2", "rootsine", "lin_up", "lin_down", "u", "random")
    for i, prof in enumerate(profiles):
        cand = make_template(prof, n0, seed=17 + i)
        sc = fast_obj(cand)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cand
            print("  Template {} J={:.8f}".format(prof, best_score))

    remaining = budget - (time.monotonic() - t_global0)
    if remaining > 45.0:
        t1 = min(remaining * 0.34, 560.0)
        if t1 > 25.0:
            print("\nPhase 1: AWG main n={} t={:.1f}s".format(len(best_seq), t1))
            out, sc = awg_optimize(best_seq, t1, seed=12345, verbose=True)
            if sc < best_score:
                best_score = sc
                best_seq = out
                print("  Phase1 improved J={:.8f}".format(best_score))

    remaining = budget - (time.monotonic() - t_global0)
    if remaining > 45.0:
        t2 = min(remaining * 0.20, 330.0)
        if t2 > 25.0:
            print("\nPhase 2: Coord n={} t={:.1f}s".format(len(best_seq), t2))
            out, sc = coord_polish(best_seq, t2, seed=777, verbose=True)
            if sc < best_score:
                best_score = sc
                best_seq = out
                print("  Phase2 improved J={:.8f}".format(best_score))

    tried = {len(best_seq)}
    for nl in candidate_lengths(len(best_seq))[:3]:
        if nl in tried:
            continue
        tried.add(nl)
        remaining = budget - (time.monotonic() - t_global0)
        if remaining < 260.0:
            break
        print("\nPhase 3: Length n={} (from {})".format(nl, len(best_seq)))
        cand = interpolate_to(best_seq, nl)
        sc = fast_obj(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
            print("  Interpolated better J={:.8f}".format(best_score))

        ta = min(remaining * 0.35, 210.0)
        if ta > 30.0:
            out, sc = awg_optimize(best_seq, ta, seed=910 + nl, K=None, verbose=False)
            if sc < best_score:
                best_score = sc
                best_seq = out
                print("  Length AWG improved J={:.8f}".format(best_score))

        remaining = budget - (time.monotonic() - t_global0)
        tc = min(remaining * 0.20, 70.0)
        if tc > 15.0:
            out, sc = coord_polish(best_seq, tc, seed=911 + nl, verbose=False)
            if sc < best_score:
                best_score = sc
                best_seq = out
                print("  Length Coord improved J={:.8f}".format(best_score))

    remaining = budget - (time.monotonic() - t_global0)
    rng_global = np.random.default_rng(20240101)
    for restart_i in range(2):
        if remaining < 170.0:
            break
        print("\nPhase 4: Restart {} t~{:.1f}s".format(restart_i + 1, remaining * 0.10))
        base = normalize(best_seq)
        noise_scale = 0.02 + 0.022 * float(restart_i)
        pert = base + noise_scale * rng_global.standard_normal(len(base)) * base
        tmpl = random_smooth(len(base), 555 + 7 * restart_i)
        start = normalize(0.78 * pert + 0.22 * tmpl)
        sc = fast_obj(start)
        if sc < best_score:
            best_score = sc
            best_seq = start

        tr = min(remaining * 0.085, 130.0)
        if tr > 20.0:
            out, sc = awg_optimize(start, tr, seed=9000 + 23 * (restart_i + 1), verbose=False)
            if sc < best_score:
                best_score = sc
                best_seq = out
                print("  Restart {} improved J={:.8f}".format(restart_i + 1, best_score))

        remaining = budget - (time.monotonic() - t_global0)
        tc = min(remaining * 0.06, 45.0)
        if tc > 10.0:
            out, sc = coord_polish(best_seq, tc, seed=9100 + 31 * (restart_i + 1), verbose=False)
            if sc < best_score:
                best_score = sc
                best_seq = out
                print("  Restart {} coord improved J={:.8f}".format(restart_i + 1, best_score))

        remaining = budget - (time.monotonic() - t_global0)

    remaining = budget - (time.monotonic() - t_global0)
    if remaining > 40.0:
        tfin = min(remaining * 0.30, 360.0)
        print("\nPhase 5: Final coord n={} t={:.1f}s".format(len(best_seq), tfin))
        out, sc = coord_polish(best_seq, tfin, seed=60210, verbose=True)
        if sc < best_score:
            best_score = sc
            best_seq = out
            print("  Final coord improved J={:.8f}".format(best_score))

    remaining = budget - (time.monotonic() - t_global0)
    if remaining > 35.0:
        tf2 = min(remaining * 0.35, 240.0)
        print("\nPhase 6: Final AWG n={} t={:.1f}s".format(len(best_seq), tf2))
        out, sc = awg_optimize(best_seq, tf2, seed=41729, K=int(max(80, min(260, len(best_seq) // 70))), verbose=True)
        if sc < best_score:
            best_score = sc
            best_seq = out
            print("  Final AWG improved J={:.8f}".format(best_score))

        remaining = budget - (time.monotonic() - t_global0)
        if remaining > 25.0:
            tfc = min(remaining * 0.35, 160.0)
            print("\nPhase 7: Final coord n={} t={:.1f}s".format(len(best_seq), tfc))
            out, sc = coord_polish(best_seq, tfc, seed=42830, verbose=False)
            if sc < best_score:
                best_score = sc
                best_seq = out
                print("  Final coord2 improved J={:.8f}".format(best_score))

    best_seq = np.asarray(best_seq, dtype=np.float64).ravel()
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_template("sine", 12000)
        s = float(best_seq.sum())
    best_seq = best_seq / s
    best_seq = np.minimum(best_seq, 1000.0)

    actual = eval_safe(best_seq)
    if np.isfinite(actual):
        best_score = actual

    if not np.isfinite(best_score):
        try:
            par = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if par.size > 20 and np.isfinite(par).all():
                best_seq = normalize(par)
                best_score = eval_safe(best_seq)
                if not np.isfinite(best_score):
                    best_score = fast_obj(best_seq)
        except Exception:
            pass

    if not np.isfinite(best_score):
        best_seq = make_template("sine", 12000)
        best_score = fast_obj(best_seq)

    elapsed = time.monotonic() - t_global0
    print("\nFINAL: n={} J={:.8f} elapsed={:.1f}s".format(len(best_seq), best_score, elapsed))
    return [float(x) for x in best_seq]
```
