    2*n*max(conv(a,a))/(sum(a))^2.

Novel approaches in this version:

def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]


def objective_norm(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    q = self_conv(a)
    return 2.0 * float(a.size) * float(np.max(q))


def evaluate_safe(a):

def conv_ab(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]


def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return w / s


def coordinate_descent(a0, time_limit, verbose=False, rng_seed=42):
    """
    Normalized coordinate descent for min max_k self_conv(a)[k] on the simplex.
    For each coordinate i, find t minimizing M(t) = max_k q_new[k] where
    q_new is the convolution after setting a[i]=t and renormalizing.
    """
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 2:
        return a

    best = a.copy()
    best_M = float(np.max(self_conv(a)))
    rng = np.random.default_rng(rng_seed)
    no_improve = 0
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        q = self_conv(a)
        M_cur = float(np.max(q))
        if M_cur < best_M - 1e-14:
            best_M = M_cur
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                break

        # Precompute prefix and suffix max of q for fast non-affected max
        q_full = q  # length 2n-1
        prefix_max = np.zeros(2 * n - 1, dtype=np.float64)
        prefix_max[0] = q_full[0]
        for j in range(1, 2 * n - 1):
            prefix_max[j] = max(prefix_max[j - 1], q_full[j])
        suffix_max = np.zeros(2 * n - 1, dtype=np.float64)
        suffix_max[2 * n - 2] = q_full[2 * n - 2]
        for j in range(2 * n - 3, -1, -1):
            suffix_max[j] = max(suffix_max[j + 1], q_full[j])

        # Random coordinate order
        order = rng.permutation(n)

        for idx in order:
            if time.monotonic() - t0 > time_limit:
                break

            s = 1.0 - a[idx]  # sum of all except a[idx]
            if s <= 1e-15:
                continue

            # Affected lags: k in [idx, min(idx+n-1, 2n-2)]
            k_start = idx
            k_end = min(idx + n, 2 * n - 1)  # exclusive
            if k_start >= 2 * n - 1:
                continue

            # Non-affected max
            if k_start > 0:
                max_left = prefix_max[k_start - 1]
            else:
                max_left = 0.0
            if k_end < 2 * n - 1:
                max_right = suffix_max[k_end]
            else:
                max_right = 0.0
            max_non_aff = max(max_left, max_right)

            # Get affected q values and partner values
            aff_q = q_full[k_start:k_end].copy()
            # a[k-idx] for k in [k_start, k_end)
            partner_start = k_start - idx  # = 0
            partner_end = k_end - idx  # = n or less
            partner_end = min(partner_end, n)
            partners = a[partner_start:partner_end]
            # The [k==idx] indicator
            idx_offset = idx - k_start  # position of k=idx in the affected range
            # idx_offset should be 0 since k_start = idx
            # So [k==idx] is 1 at offset 0

            a_i = a[idx]

            def M_of_t(tv):
                """Compute max_k q_new[k] for a[idx] = tv, renormalized."""
                denom = (s + tv)
                c2 = 1.0 / (denom * denom)
                d = tv - a_i
                # Affected: aff_q[k] + 2*d*partners[k] + d^2 * [k==0] (offset 0 is k=idx)
                aff_new = aff_q + 2.0 * d * partners
                if idx_offset >= 0 and idx_offset < len(aff_new):
                    aff_new[idx_offset] += d * d
                # Non-affected
                non_aff_new = max_non_aff * c2
                aff_new = aff_new * c2
                return max(float(np.max(aff_new)), non_aff_new)

            # Grid search for rough minimum
            t_vals = np.array([0.0, 0.1 * a_i, 0.3 * a_i, 0.5 * a_i,
                                0.7 * a_i, a_i, 1.5 * a_i, 2.0 * a_i,
                                3.0 * a_i, 5.0 * a_i, 10.0 * a_i,
                                0.05, 0.1, 0.5, 1.0, 5.0, 10.0])
            t_vals = t_vals[t_vals >= 0.0]
            t_vals = t_vals[t_vals < 50.0 * s + 1.0]  # reasonable upper bound
            t_vals = np.unique(t_vals)

            best_t = a_i
            best_val = M_of_t(a_i)

            for tv in t_vals:
                if tv == a_i:
                    continue
                val = M_of_t(tv)
                if val < best_val:
                    best_val = val
                    best_t = tv

            # Golden section refinement around best_t
            if best_t > 1e-10:
                lo = max(0.0, best_t * 0.5)
                hi = best_t * 2.0 + 0.1
                phi = (math.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - phi * (hi - lo)
                d1 = lo + phi * (hi - lo)
                f1 = M_of_t(c1)
                f2 = M_of_t(d1)
                for _ in range(20):
                    if f1 < f2:
                        hi = d1
                        d1 = c1
                        f2 = f1
                        c1 = hi - phi * (hi - lo)
                        f1 = M_of_t(c1)
                    else:
                        lo = c1
                        c1 = d1
                        f1 = f2
                        d1 = lo + phi * (hi - lo)
                        f2 = M_of_t(d1)
                mid = 0.5 * (lo + hi)
                fmid = M_of_t(mid)
                if fmid < best_val:
                    best_val = fmid
                    best_t = mid

            # Apply update
            if best_t >= 0.0 and best_val < M_cur - 1e-15:
                a[idx] = best_t
                tot = a.sum()
                if tot > 1e-15:
                    a = a / tot
                else:
                    a[idx] = a_i
                    break

        sweep += 1
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - t0
            print("    coord_desc: sweep=%d elapsed=%.1fs best_M=%.10f noimp=%d"
                  % (sweep, elapsed, best_M, no_improve))

    return best


def minmax_dual_alternation(a0, time_limit, verbose=False, rng_seed=7):
    """
    Min-max dual alternation:
    1. Update dual weights p (softmax on lags).
    2. Solve inner QP: min a^T P a s.t. simplex via projected gradient.
    """
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 5:
        return a

    best = a.copy()
    best_M = float(np.max(self_conv(a)))
    temperature = 1.0
    pg_iters = 50
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        q = self_conv(a)
        M_cur = float(np.max(q))
        if M_cur < best_M - 1e-14:
            best_M = M_cur
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

        # Update dual weights: softmax on q
        scale = 0.5 * M_cur if M_cur > 1e-15 else 1.0
        gamma = 1.0 / (temperature * scale)
        shifted = q - M_cur
        shifted = np.clip(gamma * shifted, -50.0, 0.0)
        e = np.exp(shifted)
        p = e / e.sum()

        # Build Hankel matrix P[i,j] = p[i+j] (implicitly, via mat-vec product)
        # P*a: (P*a)[i] = sum_j p[i+j] * a[j]
        # Can compute via convolution: (P*a)[i] = sum_j p[i+j]*a[j]
        # Let's define b[j] = a[j], then (P*a)[i] = sum_j b[j]*p[i+j]
        # This is a cross-correlation-like operation.
        # p has length 2n-1 (indices 0 to 2n-2).
        # (P*a)[i] = sum_{j=0}^{n-1} p[i+j] * a[j]
        # = cross-correlation of p and a, evaluated at position i (with appropriate alignment)
        # Specifically: if we define p_extended and correlate:
        # result[i] = sum_j p[i+j]*a[j] for j=0..n-1
        # This is like: flip a, then convolve with p, take appropriate slice.
        # Let a_flip = a[::-1], then conv(p, a_flip)[i + n - 1] = sum_j p[i+j]*a_flip[n-1-j]
        # = sum_j p[i+j]*a[n-1-j+... ] hmm this is getting confusing.
        # Let me just do it directly with a loop or numpy.

        # Direct computation: (P*a)[i] = sum_{j=0}^{n-1} p[i+j] * a[j]
        # Using numpy: for each i, this is dot(p[i:i+n], a)
        # Vectorized: create a Toeplitz-like product.
        # Actually, I can compute this as:
        # (P*a) = the first n entries of the convolution of p with a (with appropriate alignment)
        # conv(p, a)[k] = sum_j p[k-j] * a[j]
        # I want sum_j p[i+j] * a[j] = sum_j p[(i+n-1) - (n-1-j)] * a[j]
        # Let j' = n-1-j, then j = n-1-j':
        # = sum_{j'} p[i + n - 1 - j'] * a[n-1-j']
        # = sum_{j'} p[i+n-1-j'] * a_flip[j']
        # = conv(p, a_flip)[i+n-1]

        a_flip = a[::-1]
        size = next_pow2(2 * n - 1 + n - 1)
        fp = np.fft.rfft(p, size)
        fa = np.fft.rfft(a_flip, size)
        corr = np.fft.irfft(fp * fa, size)
        # (P*a)[i] = corr[i + n - 1] for i = 0..n-1
        Pa = corr[n - 1:2 * n - 2]
        Pa = np.maximum(Pa, 0.0)

        # Projected gradient: minimize a^T P a ~ (1/2) a^T (2P) a
        # Gradient of (1/2) a^T (2P) a = 2P*a
        # But we're minimizing a^T P a, whose gradient is 2Pa.
        # Projected gradient step: a_new = proj_simplex(a - step * 2*Pa)

        # Adaptive step size
        step = 1.0 / (float(np.max(2 * Pa)) + 1e-15)

        # Line search on the QP objective
        def qp_obj(av):
            # Compute a^T P a = sum_{i,j} a[i]*a[j]*p[i+j]
            # = sum_k p[k] * sum_{i+j=k} a[i]*a[j] = sum_k p[k] * (a*a)[k]
            # = dot(p, self_conv(a))
            return float(np.dot(p, self_conv(av)))

        obj_cur = qp_obj(a)
        best_step = step
        best_obj = obj_cur
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 2.0]:
            st = step * mult
            a_trial = project_simplex(a - st * 2.0 * Pa)
            obj_trial = qp_obj(a_trial)
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_step = st
        a = project_simplex(a - best_step * 2.0 * Pa)

        # Temperature adaptation
        if no_improve > 20:
            temperature *= 1.1
        elif no_improve == 0:
            temperature = max(0.3, temperature * 0.95)

        it += 1
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print("    minmax_dual: it=%d elapsed=%.1fs best_M=%.10f temp=%.2f noimp=%d"

    return best


def lp_fw_search(a0, time_limit, verbose=False):
    """LP-based Frank-Wolfe local search."""
    if not HAVE_LP:
        return a0
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 10 or n > 9000:
        return a

    best = a.copy()
    best_M = float(np.max(self_conv(a)))
    K = min(300, n // 2)
    maxK = min(800, n)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining <= 0:
            break

        q = self_conv(a)
        M_cur = float(np.max(q))
        if M_cur < best_M - 1e-14:
            best_M = M_cur
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 150:
                break

        # Active set: top K lags
        order = np.argsort(q)[::-1]
        Kcur = min(K, len(q))
        active = order[:Kcur]
        q_active = q[active]

        # Solve LP: min T s.t. 2*(a*g)[k] - T <= q[k] for active k, sum(g)=1, g>=0
        K2 = Kcur
        if K2 * (n + 1) > 5_000_000:
            K = max(100, K - 30)
            continue

        try:
            A = np.zeros((K2, n + 1), dtype=np.float64)
            for r, k in enumerate(active):
                k = int(k)
                i0 = max(0, k - n + 1)
                i1 = min(n, k + 1)
                if i0 < i1:
                    start = k - (i1 - 1)
                    stop = k - i0 + 1
                    A[r, i0:i1] = 2.0 * a[start:stop][::-1]
            A[:, n] = -1.0
            b_ub = np.maximum(q_active, 0.0)

            c_obj = np.zeros(n + 1, dtype=np.float64)
            c_obj[n] = 1.0
            Aeq = np.zeros((1, n + 1), dtype=np.float64)
            Aeq[0, :n] = 1.0
            bounds = [(0.0, np.inf)] * n + [(None, None)]

            res = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=Aeq,
                          b_eq=np.array([1.0]),
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(1.0, remaining * 0.1)})
        except Exception:
            continue

        if res.status != 0 or res.x is None:
            no_improve += 1
            continue

        g = np.asarray(res.x[:n], dtype=np.float64)
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs <= 1e-10:
            continue
        g = g / gs

        # Line search: a_new = (1-t)*a + t*g
        c_cross = conv_ab(a, g)
        q_g = self_conv(g)
        M0 = M_cur

        best_t = 0.0
        best_Mv = M0
        for t in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            q_mix = (1 - t) ** 2 * q + 2 * t * (1 - t) * c_cross + t ** 2 * q_g
            Mv = float(np.max(q_mix))
            if Mv < best_Mv:
                best_Mv = Mv
                best_t = t

        if best_t > 0 and best_Mv < M_cur - 1e-14:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        else:
            # Try softmax direction
            gamma = 5.0 / max(M0, 1e-15)
            w = np.exp(np.clip(gamma * (q - M0), -50, 0))
            w = w / w.sum()
            c_cross2 = conv_ab(a, w)
            q_w = self_conv(w)
            for t in [0.05, 0.1, 0.2, 0.3]:
                q_mix = (1 - t) ** 2 * q + 2 * t * (1 - t) * c_cross2 + t ** 2 * q_w
                Mv = float(np.max(q_mix))
                if Mv < best_Mv:
                    best_Mv = Mv
                    best_t = t
            if best_t > 0 and best_Mv < M_cur - 1e-14:
                a = (1 - best_t) * a + best_t * w
                a = np.maximum(a, 0.0)
                a = a / a.sum()

        K = min(K + 10, maxK)

    return best


def smooth_mirror_polish(a0, time_limit, rng_seed=101):
    """Mirror-style multiplicative gradient polish."""
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 5:
        return a

    best = a.copy()
    best_M = float(np.max(self_conv(a)))
    trust = 0.3
    focus = 5.0
    fails = 0
    rng = np.random.default_rng(rng_seed)

    while time.monotonic() - t0 < time_limit:
        q = self_conv(a)
        M_cur = float(np.max(q))
        if M_cur < best_M - 1e-14:
            best_M = M_cur
            best = a.copy()
            fails = 0
        else:
            fails += 1
            if fails > 300:
                break

        if fails > 50:
            trust = max(0.05, trust * 0.85)
            focus = min(focus * 1.1, 50.0)
            if fails > 150:
                # Restart with noise
                noise = 0.01 * rng.normal(size=n) * a
                a_try = a + noise
                a_try = np.maximum(a_try, 1e-12)
                a_try = normalize_vec(a_try)
                M_try = float(np.max(self_conv(a_try)))
                if M_try < best_M + 0.01:
                    a = a_try
                    fails = 0
                    trust = 0.3

        # Compute gradient direction
        scale = 0.5 * M_cur if M_cur > 1e-15 else 1.0
        gamma = focus / scale
        w = np.exp(np.clip(gamma * (q - M_cur), -50, 0))
        w = w / w.sum()

        # Cross-correlation: c[i] = sum_k w[k] * a[k-i] (first n entries)
        size = next_pow2(2 * n - 1 + n - 1)
        fw = np.fft.rfft(w, size)
        fa = np.fft.rfft(a, size)
        corr = np.fft.irfft(fw * fa.conj(), size)[:n]
        corr = np.maximum(corr, 0.0)

        cmax = float(np.max(corr))
        cmin = float(np.min(corr))
        rngc = cmax - cmin
        if rngc <= 1e-15:
            break

        move = trust * (cmax - corr) / rngc
        move = np.clip(move, -40, 40)
        b = a * np.exp(move)
        b = normalize_vec(b)

        M_b = float(np.max(self_conv(b)))
        if M_b < best_M - 1e-14:
            best_M = M_b
            best = b.copy()
            a = b
            trust = min(trust * 1.05, 3.0)
        elif M_b < M_cur - 1e-14:
            a = b
            trust = min(trust * 1.08, 3.0)
        else:
            trust = max(trust * 0.75, 0.05)

    return best


def interpolate_seq(f, new_n):
    f = normalize_vec(f)
    old_n = f.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return f.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
