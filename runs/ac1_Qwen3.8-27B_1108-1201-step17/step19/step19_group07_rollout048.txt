    # = 2 * (w * y_rev)[n-1+j]
    W = np.fft.rfft(w, fft_size)
    Y_rev = np.fft.rfft(y[::-1], fft_size)
    conv = np.fft.irfft(W * Y_rev, fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    
    return grad, c, w


def fista_step_schedule(a0, time_limit, schedule, eta0=0.8, verbose=False):
    """
    FISTA with a discrete step schedule for beta and K.
    schedule: list of (beta, K, fraction_of_time) tuples
    """
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_threshold = 5000
    
    # Pre-compute iteration counts for each phase
    total_iters_est = int(time_limit * 300)  # rough estimate
    phases = []
    for beta, K, frac in schedule:
        phases.append((beta, K, int(frac * total_iters_est)))
    
    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        
        # Determine current phase
        cum_iter = 0
        beta_cur = schedule[-1][0]
        K_cur = schedule[-1][1]
        for beta_p, K_p, n_iters_p in phases:
            if iter_count < cum_iter + n_iters_p:
                beta_cur = beta_p
                K_cur = K_p
                break
            cum_iter += n_iters_p
        
        # Compute gradient
        grad, c_y, _ = compute_gradient(y, beta_cur, K_cur)
        
        # Step size
        eta = eta0 / (np.sqrt(t) + 1.0)
        
        # Mirror descent (multiplicative update)
        log_y = np.log(np.maximum(y, 1e-30)) - eta * grad
        log_y -= log_y.max()
        z = np.exp(log_y)
        z = np.maximum(z, 1e-30)
        z = z / z.sum()
        
        # Nesterov momentum
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = z + momentum * (z - y)
            x_new = np.maximum(x_new, 1e-30)
            x_new = x_new / x_new.sum()
        else:
            x_new = z.copy()
        
        # Evaluate
        c_new = fast_autocorr(x_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > stall_threshold:
            # Perturb and restart
            rng = np.random.default_rng(iter_count * 7 + 13)
            sigma = 0.03 + 0.03 * np.random.default_rng(iter_count).random()
            pert = rng.normal(0, sigma, n) * best_x
            x_restart = norm_sum1(best_x + pert)
            x_new = x_restart
            y = x_restart.copy()
            stall = 0
            stall_threshold = max(int(stall_threshold * 0.85), 2000)
        
        y = x_new
        x = x_new
        iter_count += 1
        
        if verbose and iter_count % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA-S iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={beta_cur:.1f}, K={K_cur}")
    
    return best_x, best_m


def fw_optimize(a0, time_limit, beta_start=2.0, beta_end=100.0, K_start=1, K_end=100, verbose=False):
    """Frank-Wolfe optimization with adaptive parameters."""
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        progress = min(iter_count / max(int(time_limit * 200), 1), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        K = max(1, int(K_start + (K_end - K_start) * progress))
        
        # Compute gradient
        grad, c_y, _ = compute_gradient(y, beta, K)
        
        # FW vertex: argmin of gradient
        j_star = np.argmin(grad)
        
        # FW step
        alpha = 2.0 / (t + 2.0)
        v = np.zeros(n)
        v[j_star] = 1.0
        x_new = (1.0 - alpha) * x + alpha * v
        
        # Nesterov
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            y_new = x_new + momentum * (x_new - x)
        else:
            y_new = x_new.copy()
        
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = x.copy()
        else:
            y_new = y_new / sy
        
        c_new = fast_autocorr(x_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 3000:
            rng = np.random.default_rng(iter_count * 3 + 7)
            pert = rng.normal(0, 0.05, n) * best_x
            x_restart = norm_sum1(best_x + pert)
            x_new = x_restart
            y = x_restart.copy()
            stall = 0
        
        x = x_new
        y = y_new
        iter_count += 1
        
        if verbose and iter_count % 50000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, K={K}")
    
    return best_x, best_m


def lp_frank_wolfe(a0, time_limit, solver, initial_K=300, max_K=3000, verbose=False):
    """LP-based Frank-Wolfe for fine polishing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x_fw = a.copy()
    
    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()
    
    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)
    
    failed_solves = 0
    
    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
        
        ac_x = fast_autocorr(x_fw)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x_fw.copy()
        
        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 10:
                constraint_list = sorted(list(constraint_set))[:min(initial_K, len(constraint_set))]
                constraint_set = set(constraint_list)
                failed_solves = 0
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Check for violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # Nesterov-accelerated FW step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x_fw = x_new
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={K}")
    
    return best_a, best_m


def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(30, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): SCS")
    return cp.SCS


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
