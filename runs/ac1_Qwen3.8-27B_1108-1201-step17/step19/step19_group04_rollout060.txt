def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]


def conv_ab(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]


def objective_norm(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    q = self_conv(a)
    return 2.0 * float(a.size) * float(np.max(q))

def make_pw_constant_seq(values, n):
    """Build a piecewise-constant sequence of length n from K values."""
    K = len(values)
    # Each segment gets approximately n/K elements
    seg_lens = np.ones(K, dtype=int) * (n // K)
    remainder = n - seg_lens.sum()
    # Distribute remainder to first 'remainder' segments
    for i in range(remainder):
        seg_lens[i] += 1
    a = np.empty(n, dtype=np.float64)
    pos = 0
    for i in range(K):
        L = seg_lens[i]
        a[pos:pos+L] = values[i]
        pos += L
    return a


def make_pw_linear_seq(control, n):
    """Build a piecewise-linear sequence of length n from K+1 control points."""
    K = len(control) - 1
    x = np.linspace(0.0, 1.0, len(control))
    a = np.interp(np.linspace(0.0, 1.0, n), x, control)
    return a


def pw_constant_obj(v, n, K):
    """Objective for piecewise-constant representation."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-10:
        return 1e10
    v = v / s
    a = make_pw_constant_seq(v, n)
    return objective_norm(a)


def pw_linear_obj(v, n):

    v = np.maximum(v, 1e-10)

    a = np.maximum(a, 0.0)

    if s2 < 1e-10:

    a = a / s2
    return objective_norm(a)

def interpolate_sequence(f, new_n):
    f = normalize_vec(f)
    old_n = f.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return f.copy()
    if old_n <= 1:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f)
    return normalize_vec(g)


def line_search_direction(a, d, q0, T=None):
    """Minimize max q((1-alpha)a + alpha d) over alpha in [0,1]."""
    a = np.asarray(a, dtype=np.float64)
    d = np.asarray(d, dtype=np.float64)
    q0 = np.asarray(q0, dtype=np.float64)
    M0 = float(np.max(q0))
    
    dot = float(np.dot(a, d))
    if dot > 1.0 - 1e-7:
        return a.copy(), M0
    
    c = conv_ab(a, d)
    qd = self_conv(d)
    
    best_alpha = 0.0
    best_M = M0
    
    alphas = [1e-4, 5e-4, 1e-3, 2e-3, 5e-3, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
    
    # Use T from LP to guess a good step
    if T is not None and np.isfinite(T) and T < 0.0 and M0 > 1e-15:
        t1 = min(0.5, max(1e-4, 0.8 * (-float(T)) / M0))
        alphas.append(max(1e-5, 0.5 * t1))
        alphas.append(min(0.95, 1.5 * t1))
        alphas.append(t1)
    
    for al in alphas:
        al = float(min(1.0, max(0.0, al)))
        a1 = 1.0 - al
        val = np.maximum(
            (a1 * a1) * q0 + (2.0 * al * a1) * c + (al * al) * qd,
            0.0
        )
        M = float(np.max(val))
        if M < best_M:
            best_M = M
            best_alpha = al
    
    # Golden section refinement
    if 0.0 < best_alpha < 1.0:
        best_M, best_alpha = golden_refine(q0, c, qd, best_alpha)
    
    if best_alpha <= 0.0 or best_M >= M0 - 1e-14:
        return a.copy(), M0
    
    new = (1.0 - best_alpha) * a + best_alpha * d
    new = np.maximum(new, 0.0)
    s = float(new.sum())
    if s <= 1e-15:
        return a.copy(), M0
    new = new / s
    return new, best_M


def quad_max_val(alpha, q0, c, qd):
    a1 = 1.0 - alpha
    val = (a1 * a1) * q0 + (2.0 * alpha * a1) * c + (alpha * alpha) * qd
    return float(np.max(val))


def golden_refine(q0, c, qd, best_alpha, iters=30):
    """Golden section search around best_alpha."""
    phi = (math.sqrt(5.0) - 1.0) / 2.0
    a = max(0.0, best_alpha - 0.1)
    b = min(1.0, best_alpha + 0.1)
    c1 = b - phi * (b - a)
    d1 = a + phi * (b - a)
    f1 = quad_max_val(c1, q0, c, qd)
    f2 = quad_max_val(d1, q0, c, qd)
    best_m = float(np.max(( (1-best_alpha)**2 * q0 + 2*best_alpha*(1-best_alpha)*c + best_alpha**2*qd )))
    best_a = best_alpha
    for _ in range(iters):
        if f1 < f2:
            b = d1
            d1 = c1
            f2 = f1
            c1 = b - phi * (b - a)
            f1 = quad_max_val(c1, q0, c, qd)
        else:
            a = c1
            c1 = d1
            f1 = f2
            d1 = a + phi * (b - a)
            f2 = quad_max_val(d1, q0, c, qd)
    result_a = 0.5 * (a + b)
    result_m = quad_max_val(result_a, q0, c, qd)
    if result_m < best_m:
        return result_m, result_a
    return best_m, best_a

            2*(y*g)[k] - T <= q_y[k], for active lags k.
    """
    if not HAVE_SCIPY:
        return None, None
    y = np.asarray(y, dtype=np.float64)
    active = np.asarray(active, dtype=int)
    q_active = np.asarray(q_active, dtype=np.float64)
    n = y.size
    K = active.size
    if K <= 0:
        return None, None
    if K * (n + 1) > 12_000_000:
        return None, None

    try:
        A = np.zeros((K, n + 1), dtype=np.float64)
    except Exception:
        return None, None

    yy = 2.0 * y
    for r, k in enumerate(active):
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            A[r, i0:i1] = yy[start:stop][::-1]

    A[:, n] = -1.0
    b = np.maximum(q_active, 0.0)

    c, Aeq, bounds = get_lp_artifacts(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq,
                      b_eq=np.array([1.0], dtype=np.float64),
                      bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception:
        return None, None

    status = getattr(res, 'status', None)
    success = getattr(res, 'success', False)
    x = getattr(res, 'x', None)
    if (status != 0 or not success) or x is None:
        return None, None

    g = np.asarray(x[:n], dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-10:
        return None, None
    g = g / s
    T = float(x[n])
    if not np.isfinite(T):
        T = 0.0
    return g, T


def select_active(q, K_max, strategy='top_neighbors'):
    """
    Select active lags for the LP.
    strategy='top': just top K lags
    strategy='top_neighbors': top lags + their neighbors
    strategy='spread': spread out across the correlation
    """
    n_lags = len(q)
    M = float(np.max(q))
    
    if strategy == 'top':
        order = np.argsort(q)[::-1]
        return order[:min(K_max, n_lags)]
    
    elif strategy == 'top_neighbors':
        # Take top K_max//2 lags, plus their neighbors
        order = np.argsort(q)[::-1]
        top = order[:min(K_max // 2, n_lags)]
        active_set = set(top.tolist())
        for k in top:
            k = int(k)
            for dk in (-1, -2, 1, 2):
                nk = k + dk
                if 0 <= nk < n_lags:
                    active_set.add(nk)
        active = np.array(sorted(active_set), dtype=int)
        if active.size > K_max:
            # Keep the highest ones
            vals = q[active]
            order2 = np.argsort(vals)[::-1]
            active = active[order2[:K_max]]
        return active
    
    elif strategy == 'spread':
        # Take top K_max//3, then spread the rest evenly
        order = np.argsort(q)[::-1]
        top = order[:min(K_max // 3, n_lags)]
        active_set = set(top.tolist())
        # Add evenly spaced lags
        spacing = max(1, n_lags // (K_max // 3))
        for k in range(0, n_lags, spacing):
            if k not in active_set:
                active_set.add(k)
                if len(active_set) >= K_max:
                    break
        active = np.array(sorted(active_set), dtype=int)
        if active.size > K_max:
            vals = q[active]
            order2 = np.argsort(vals)[::-1]
            active = active[order2[:K_max]]
        return active
    
    else:
        order = np.argsort(q)[::-1]
        return order[:min(K_max, n_lags)]

def try_soft_direction(y, qy):
    """Try a multiplicative gradient direction."""
    n = y.size
    M_cur = float(np.max(qy))
    
    # Compute correlation-based direction
    # Focus on the peak lags
    top_k = np.argsort(qy)[::-1][:min(20, len(qy))]
    scores = np.zeros(n, dtype=np.float64)
    for kk in top_k[:10]:
        k = int(kk)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            scores[i0:i1] += y[start:stop][::-1]
    
    if scores.max() <= scores.min() + 1e-15:
        return y.copy(), M_cur
    
    scale = scores.max() - scores.min()
    for trust in (0.3, 0.8, 2.0):
        move = trust * (scores.max() - scores) / scale
        d = y * np.exp(move)
        d = d / d.sum()
        new, Mnew = line_search_direction(y, d, qy)
        if Mnew < M_cur - 1e-13:
            return new, Mnew
    
    return y.copy(), M_cur


def smooth_polish(a0, time_limit, rng_seed=11):
    """Multiplicative gradient polish."""
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 1:
        return a

    best = a.copy()
    best_score = objective_norm(a)
    trust = 0.4
    focus = 5.0
    fails = 0
    rng = np.random.default_rng(rng_seed)

    while time.monotonic() - t0 < time_limit:
        q = self_conv(a)
        M = float(np.max(q))
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            fails = 0
        else:
            fails += 1
            if fails > 300:
                break

        if fails > 30:
            trust = max(trust * 0.85, 0.05)
            focus = min(focus * 1.08, 50.0)

        if fails > 80 and fails % 80 == 0:
            # Random perturbation to escape
            cand = best * (1.0 + 0.02 * rng.normal(size=best.size))
            cand = np.maximum(cand, 1e-12)
            cand = cand / cand.sum()
            a = cand
            fails = 0
            trust = 0.3

        # Compute direction
        top_k = np.argsort(q)[::-1][:min(30, len(q))]
        w = np.zeros(len(q), dtype=np.float64)
        for idx in top_k:
            w[idx] = np.exp(-focus * (M - q[idx]) / max(M, 1e-15))
        
        # Compute correlation
        size = next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, size)
        fa = np.fft.rfft(a, size)
        c = np.fft.irfft(fw * fa.conj(), size)[:n]
        c = np.maximum(c, 0.0)
        
        cmax = float(np.max(c))
        cmin = float(np.min(c))
        scale = cmax - cmin
        if scale <= 1e-15:
            break

        move = trust * (cmax - c) / scale
        move = np.clip(move, -40.0, 40.0)
        b = a * np.exp(move)
        b = b / b.sum()

        Mb = float(np.max(self_conv(b)))
        score_b = 2.0 * n * Mb

        if score_b < best_score - 1e-13:
            best = b.copy()
            best_score = score_b
            a = b
            fails = 0
            trust = min(trust * 1.05, 3.0)
        elif score_b < score - 1e-14:
            a = b
            fails = 0
        else:
            trust = max(trust * 0.75, 0.05)

    return best


def transform_search(a0, time_limit):
    """Try power transforms and blending."""
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    best = a.copy()
    best_score = objective_norm(a)

    gammas = (0.3, 0.5, 0.7, 1.2, 1.5, 2.0, 3.0)
    for g in gammas:
        if time.monotonic() - t0 > time_limit:
            break
        b = np.maximum(a, 1e-30) ** g
        b = b / b.sum()
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    # Blend with uniform
    u = np.ones(n, dtype=np.float64) / float(n)
    for w in (0.05, 0.1, 0.2, 0.3, 0.5):
        if time.monotonic() - t0 > time_limit:
            break
        b = (1.0 - w) * a + w * u
        b = b / b.sum()
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    # Reverse blend
    rev = a[::-1].copy()
    for w in (0.1, 0.2, 0.3, 0.5):
        if time.monotonic() - t0 > time_limit:
            break
        b = (1.0 - w) * a + w * rev
        b = b / b.sum()
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    return best

        sc = objective_norm(b)

            sc = objective_norm(b)

1. Piecewise-constant and piecewise-linear compressed representation optimization
   to find better global structures via L-BFGS-B.
2. Improved active set selection: "peak + epsilon" strategy captures all near-tight
   constraints, not just top-K.
3. Enhanced line search with golden section refinement.
4. Multiple diverse starts and adaptive time allocation.
5. Smooth multiplicative gradient polish with restarts.
"""

import math
import time
import numpy as np

try:
    from scipy.optimize import linprog, minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False

_LP_CACHE = {}


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]


def conv_ab(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]


def objective_norm(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    q = self_conv(a)
    return 2.0 * float(a.size) * float(np.max(q))


def evaluate_safe(a):
    try:
        x = [float(v) for v in np.asarray(a, dtype=np.float64)]
        val = evaluate_sequence(x)
        if np.isfinite(val):
            return float(val)
    except Exception:
        pass
    return float('inf')


def interpolate_sequence(f, new_n):
    f = normalize_vec(f)
    old_n = f.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return f.copy()
    if old_n <= 1:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f)
    return normalize_vec(g)


def make_pw_constant_seq(values, n):
    K = len(values)
    seg_lens = np.ones(K, dtype=int) * (n // K)
    remainder = n - int(seg_lens.sum())
    for i in range(min(remainder, K)):
        seg_lens[i] += 1
    a = np.empty(n, dtype=np.float64)
    pos = 0
    for i in range(K):
        L = int(seg_lens[i])
        a[pos:pos + L] = values[i]
        pos += L
    return a


def make_pw_linear_seq(control, n):
    x_ctrl = np.linspace(0.0, 1.0, len(control))
    a = np.interp(np.linspace(0.0, 1.0, n), x_ctrl, control)
    return a


def pw_constant_obj(v, n, K):
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if s < 1e-10:
        return 1e10
    v = v / s
    a = make_pw_constant_seq(v, n)
    return objective_norm(a)


def pw_linear_obj(v, n):
    v = np.maximum(v, 1e-10)
    s = float(v.sum())
    if s < 1e-10:
        return 1e10
    v = v / s
    a = make_pw_linear_seq(v, n)
    a = np.maximum(a, 0.0)
    s2 = float(a.sum())
    if s2 < 1e-10:
        return 1e10
    a = a / s2
    return objective_norm(a)


def compressed_search(parent, n, time_limit):
    t0 = time.monotonic()
    parent = normalize_vec(parent)
    if parent.size != n:
        parent = interpolate_sequence(parent, n)

    results = []

    K_list = [30, 50, 80, 120, 200]
    for K in K_list:
        if time.monotonic() - t0 > time_limit:
            break
        seg_lens = np.ones(K, dtype=int) * (n // K)
        remainder = n - int(seg_lens.sum())
        for i in range(min(remainder, K)):
            seg_lens[i] += 1
        init_vals = np.zeros(K)
        pos = 0
        for i in range(K):
            L = int(seg_lens[i])
            init_vals[i] = parent[pos:pos + L].mean()
            pos += L
        init_vals = np.maximum(init_vals, 1e-10)
        init_vals = init_vals / init_vals.sum()

        if HAVE_SCIPY:
            try:
                res = minimize(
                    pw_constant_obj, init_vals,
                    args=(n, K),
                    method='L-BFGS-B',
                    bounds=[(0.0, 1000.0)] * K,
                    options={'maxiter': 150, 'ftol': 1e-14, 'gtol': 1e-10}
                )
                v_opt = np.maximum(res.x, 0.0)
                s = float(v_opt.sum())
                if s > 1e-10:
                    v_opt = v_opt / s
                    a_opt = make_pw_constant_seq(v_opt, n)
                    sc = objective_norm(a_opt)
                    if np.isfinite(sc) and sc < 1e6:
                        results.append((sc, a_opt))
            except Exception:
                pass

    K_list_lin = [30, 60, 100, 150, 200]
    for K in K_list_lin:
        if time.monotonic() - t0 > time_limit:
            break
        x_ctrl = np.linspace(0.0, 1.0, K + 1)
        init_ctrl = np.interp(x_ctrl, np.linspace(0.0, 1.0, n), parent)
        init_ctrl = np.maximum(init_ctrl, 1e-10)
        init_ctrl = init_ctrl / init_ctrl.sum()

        if HAVE_SCIPY:
            try:
                res = minimize(
                    pw_linear_obj, init_ctrl,
                    args=(n,),
                    method='L-BFGS-B',
                    bounds=[(1e-10, 1000.0)] * (K + 1),
                    options={'maxiter': 120, 'ftol': 1e-14, 'gtol': 1e-10}
                )
                if res.fun < 1e6:
                    v_opt = np.maximum(res.x, 1e-10)
                    v_opt = v_opt / v_opt.sum()
                    a_opt = make_pw_linear_seq(v_opt, n)
                    a_opt = np.maximum(a_opt, 0.0)
                    s2 = float(a_opt.sum())
                    if s2 > 1e-10:
                        a_opt = a_opt / s2
                        sc = objective_norm(a_opt)
                        if np.isfinite(sc) and sc < 1e6:
                            results.append((sc, a_opt))
            except Exception:
                pass

    return results


def quad_max_val(alpha, q0, c, qd):
    a1 = 1.0 - alpha
    val = (a1 * a1) * q0 + (2.0 * alpha * a1) * c + (alpha * alpha) * qd
    return float(np.max(np.maximum(val, 0.0)))


def golden_refine(q0, c, qd, best_alpha, iters=35):
    phi = (math.sqrt(5.0) - 1.0) / 2.0
    a = max(0.0, best_alpha - 0.15)
    b = min(1.0, best_alpha + 0.15)
    if b - a < 1e-6:
        return quad_max_val(best_alpha, q0, c, qd), best_alpha
    c1 = b - phi * (b - a)
    d1 = a + phi * (b - a)
    f1 = quad_max_val(c1, q0, c, qd)
    f2 = quad_max_val(d1, q0, c, qd)
    for _ in range(iters):
        if f1 < f2:
            b = d1
            d1 = c1
            f2 = f1
            c1 = b - phi * (b - a)
            f1 = quad_max_val(c1, q0, c, qd)
        else:
            a = c1
            c1 = d1
            f1 = f2
            d1 = a + phi * (b - a)
            f2 = quad_max_val(d1, q0, c, qd)
    result_a = 0.5 * (a + b)
    result_m = quad_max_val(result_a, q0, c, qd)
    return result_m, result_a


def line_search_direction(a, d, q0, T=None):
    a = np.asarray(a, dtype=np.float64)
    d = np.asarray(d, dtype=np.float64)
    q0 = np.asarray(q0, dtype=np.float64)
    M0 = float(np.max(q0))

    dot = float(np.dot(a, d))
    if dot > 1.0 - 1e-7:
        return a.copy(), M0

    c = conv_ab(a, d)
    qd = self_conv(d)

    best_alpha = 0.0
    best_M = M0

    alphas = [1e-5, 5e-5, 1e-4, 5e-4, 1e-3, 2e-3, 5e-3, 0.01,
              0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]

    if T is not None and np.isfinite(T) and T < 0.0 and M0 > 1e-15:
        t1 = min(0.5, max(1e-4, (-float(T)) / M0))
        alphas.append(max(1e-5, 0.5 * t1))
        alphas.append(t1)
        alphas.append(min(0.95, 1.5 * t1))
        alphas.append(min(0.99, 2.0 * t1))

    for al in alphas:
        al = float(min(1.0, max(0.0, al)))
        M = quad_max_val(al, q0, c, qd)
        if M < best_M:
            best_M = M
            best_alpha = al

    if 0.0 < best_alpha < 1.0 and best_M < M0 - 1e-13:
        M2, a2 = golden_refine(q0, c, qd, best_alpha, 35)
        if M2 < best_M:
            best_M = M2
            best_alpha = a2

    if best_alpha <= 0.0 or best_M >= M0 - 1e-14:
        return a.copy(), M0

    new = (1.0 - best_alpha) * a + best_alpha * d
    new = np.maximum(new, 0.0)
    s = float(new.sum())
    if s <= 1e-15:
        return a.copy(), M0
    new = new / s
    return new, best_M


def get_lp_artifacts(n):
    art = _LP_CACHE.get(n)
    if art is not None:
        return art
    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0
    Aeq = np.zeros((1, n + 1), dtype=np.float64)
    Aeq[0, :n] = 1.0
    bounds = [(0.0, np.inf)] * n + [(None, None)]
    art = (c, Aeq, bounds)
    if len(_LP_CACHE) > 8:
        _LP_CACHE.clear()
    _LP_CACHE[n] = art
    return art


def solve_lp_offset(y, active, q_active):
    if not HAVE_SCIPY:
        return None, None
    y = np.asarray(y, dtype=np.float64)
    active = np.asarray(active, dtype=int)
    q_active = np.asarray(q_active, dtype=np.float64)
    n = y.size
    K = active.size
    if K <= 0:
        return None, None
    if K * (n + 1) > 12_000_000:
        return None, None

    try:
        A = np.zeros((K, n + 1), dtype=np.float64)
    except Exception:
        return None, None

    yy = 2.0 * y
    for r in range(K):
        k = int(active[r])
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            A[r, i0:i1] = yy[start:stop][::-1]

    A[:, n] = -1.0
    b_vec = np.maximum(q_active, 0.0)

    c, Aeq, bounds = get_lp_artifacts(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b_vec, A_eq=Aeq,
                      b_eq=np.array([1.0], dtype=np.float64),
                      bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception:
        return None, None

    status = getattr(res, 'status', None)
    success = getattr(res, 'success', False)
    x = getattr(res, 'x', None)
    if (status != 0 or not success) or x is None:
        return None, None

    g = np.asarray(x[:n], dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-10:
        return None, None
    g = g / s
    T = float(x[n])
    if not np.isfinite(T):
        T = 0.0
    return g, T


def select_active_peak_epsilon(q, K_top, epsilon):
    """Select active lags: top K_top plus all lags within epsilon of peak."""
    n_lags = len(q)
    M = float(np.max(q))
    threshold = M * (1.0 - epsilon)

    top_order = np.argsort(q)[::-1]
    top = top_order[:min(K_top, n_lags)]

    # All lags above threshold
    above_thresh = np.where(q >= threshold)[0]

    # Combine
    active_set = set(top.tolist())
    for k in above_thresh:
        active_set.add(int(k))

    active = np.array(sorted(active_set), dtype=int)

    # Cap at reasonable size
    max_size = max(K_top * 3, 500)
    if active.size > max_size:
        vals = q[active]
        order2 = np.argsort(vals)[::-1]
        active = active[order2[:max_size]]

    return active


def select_active_top(q, K):
    n_lags = len(q)
    order = np.argsort(q)[::-1]
    return order[:min(K, n_lags)]


def try_soft_direction(y, qy):
    n = y.size
    M_cur = float(np.max(qy))
    if M_cur <= 1e-15:
        return y.copy(), M_cur

    top_k = np.argsort(qy)[::-1][:min(20, len(qy))]
    scores = np.zeros(n, dtype=np.float64)
    for kk in top_k[:12]:
        k = int(kk)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            scores[i0:i1] += y[start:stop][::-1]

    cmax = float(scores.max())
    cmin = float(scores.min())
    scale = cmax - cmin
    if scale <= 1e-15:
        return y.copy(), M_cur

    for trust in (0.2, 0.5, 1.0, 2.0, 4.0):
        move = trust * (cmax - scores) / scale
        move = np.clip(move, -40.0, 40.0)
        d = y * np.exp(move)
        ds = float(d.sum())
        if ds <= 1e-15:
            continue
        d = d / ds
        new, Mnew = line_search_direction(y, d, qy)
        if Mnew < M_cur - 1e-13:
            return new, Mnew

    return y.copy(), M_cur


def lp_fw_improved(a0, time_limit, verbose=True):
    t0 = time.monotonic()
    y = normalize_vec(a0)
    n = y.size
    if n <= 5:
        return y

    best = y.copy()
    best_score = objective_norm(y)

    if n > 8000:
        K_base = 120
        K_max = 350
    elif n > 5000:
        K_base = 180
        K_max = 500
    else:
        K_base = 250
        K_max = 700

    K = K_base
    use_lp = HAVE_SCIPY
    lp_fails = 0
    no_improve = 0
    iter_no = 0
    eps = 0.02

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining <= 1.0:
            break

        qy = self_conv(y)
        M_y = float(np.max(qy))
        score_y = 2.0 * n * M_y

        if score_y < best_score - 1e-12:
            best = y.copy()
            best_score = score_y

        # Adaptive strategy
        if no_improve > 100:
            eps = max(0.005, eps * 0.8)
            K = min(K + 20, K_max)
        elif no_improve == 0 and iter_no > 10:
            eps = min(0.10, eps * 1.05)

        # Select active set
        if iter_no % 4 == 0:
            act = select_active_peak_epsilon(qy, K, eps)
        else:
            act = select_active_top(qy, K)
        q_active = qy[act]

        g_dir = None
        T = None

        if use_lp and act.size >= 15:
            tstep = time.monotonic()
            g_dir, T = solve_lp_offset(y, act, q_active)
            spend = time.monotonic() - tstep

            if g_dir is None:
                lp_fails += 1
                if lp_fails >= 5:
                    use_lp = False
            else:
                lp_fails = 0
                if spend > max(90.0, 0.35 * remaining):
                    K = max(80, K - 40)

        if g_dir is not None:
            new, Mnew = line_search_direction(y, g_dir, qy, T)
            if Mnew < M_y - 1e-13:
                y = new
                no_improve = 0
            else:
                no_improve += 1
                if no_improve in (1, 2, 5, 12, 30):
                    new2, M2 = try_soft_direction(y, qy)
                    if M2 < M_y - 1e-13:
                        y = new2
                        no_improve = 0
        else:
            new, Mnew = try_soft_direction(y, qy)
            if Mnew < M_y - 1e-13:
                y = new
                no_improve = 0
            else:
                no_improve += 1
                K = min(K + 15, K_max)
                if no_improve > 250:
                    break

        iter_no += 1
        if verbose and (iter_no % 30 == 0):
            print("    LP-FW: it=%d t=%.1fs best=%.9f K=%d eps=%.4f use_lp=%s noimp=%d"
                  % (iter_no, time.monotonic() - t0, best_score, K, eps, use_lp, no_improve))

    return best


def smooth_polish(a0, time_limit, rng_seed=11):
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 1:
        return a

    best = a.copy()
    best_score = objective_norm(a)
    trust = 0.4
    focus = 5.0
    fails = 0
    rng = np.random.default_rng(rng_seed)

    while time.monotonic() - t0 < time_limit:
        q = self_conv(a)
        M = float(np.max(q))
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            fails = 0
        else:
            fails += 1
            if fails > 400:
                break

        if fails > 30:
            trust = max(trust * 0.88, 0.05)
            focus = min(focus * 1.06, 60.0)

        if fails > 60 and fails % 60 == 0:
            cand = best * (1.0 + 0.015 * rng.normal(size=best.size))
            cand = np.maximum(cand, 1e-12)
            cand = cand / cand.sum()
            a = cand
            fails = 0
            trust = 0.35
            focus = min(focus * 1.1, 60.0)

        # Compute direction from peak lags
        top_k = np.argsort(q)[::-1][:min(30, len(q))]
        w = np.zeros(len(q), dtype=np.float64)
        for idx in top_k:
            w[idx] = np.exp(-focus * (M - q[idx]) / max(M, 1e-15))

        size = next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, size)
        fa = np.fft.rfft(a, size)
        c = np.fft.irfft(fw * fa.conj(), size)[:n]
        c = np.maximum(c, 0.0)

        cmax = float(np.max(c))
        cmin = float(np.min(c))
        scale = cmax - cmin
        if scale <= 1e-15:
            break

        move = trust * (cmax - c) / scale
        move = np.clip(move, -40.0, 40.0)
        b = a * np.exp(move)
        b = b / b.sum()

        Mb = float(np.max(self_conv(b)))
        score_b = 2.0 * n * Mb

        if score_b < best_score - 1e-13:
            best = b.copy()
            best_score = score_b
            a = b
            fails = 0
            trust = min(trust * 1.04, 3.0)
        elif score_b < score - 1e-14:
            a = b
            fails = 0
        else:
            trust = max(trust * 0.78, 0.05)

    return best


def transform_search(a0, time_limit):
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    best = a.copy()
    best_score = objective_norm(a)

    gammas = (0.3, 0.5, 0.7, 0.8, 1.2, 1.5, 2.0, 3.0)
    for g in gammas:
        if time.monotonic() - t0 > time_limit:
            break
        b = np.maximum(a, 1e-30) ** g
        b = b / b.sum()
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    u = np.ones(n, dtype=np.float64) / float(n)
    for w in (0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5):
        if time.monotonic() - t0 > time_limit:
            break
        b = (1.0 - w) * a + w * u
        b = b / b.sum()
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    rev = a[::-1].copy()
    for w in (0.1, 0.2, 0.3, 0.5):
        if time.monotonic() - t0 > time_limit:
            break
        b = (1.0 - w) * a + w * rev
        b = b / b.sum()
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    return best


def trim_search(a0, time_limit):
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n < 30:
        return a

    best = a.copy()
    best_score = objective_norm(a)
    max_side = min(n // 4, 3000)

    samples = [0, 1, 2, 3, 5, 8, 10, 15, 20, 30, 50, 75, 100, 150, 200, 300, 500, 800, 1200, 1800, 2584]
    samples = [v for v in samples if 0 <= v <= max_side]

    for l in samples:
        if time.monotonic() - t0 > time_limit:
            break
        if l >= n - 10:
            continue
        b = a[l:]
        b = b / b.sum()
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    for r in samples:
        if time.monotonic() - t0 > time_limit:
            break
        if r >= n - 10:
            continue
        b = a[:-r]
        b = b / b.sum()
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    for l in samples[:12]:
        if time.monotonic() - t0 > time_limit:
            break
        for r in samples[:12]:
            if l + r >= n - 10:
                continue
            b = a[l:n - r]
            b = b / b.sum()
            sc = objective_norm(b)
            if sc < best_score - 1e-12:
                best_score = sc
                best = b.copy()

    return best


def coordinate_polish(a0, time_limit, rng_seed=991):
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 10:
        return a

    best = a.copy()
    best_score = objective_norm(a)
    stall = 0
    rng = np.random.default_rng(rng_seed)

    while time.monotonic() - t0 < time_limit and stall < 150:
        q = self_conv(a)
        top = np.argsort(q)[::-1][:min(15, len(q))]

        scores = np.zeros(n, dtype=np.float64)
        for kk in top[:8]:
            k = int(kk)
            i0 = max(0, k - n + 1)
            i1 = min(n, k + 1)
            if i0 < i1:
                start = k - (i1 - 1)
                stop = k - i0 + 1
                scores[i0:i1] += a[start:stop][::-1]

        src = np.argsort(scores)[::-1][:8]
        dst = np.argsort(scores)[:8]

        improved = False
        for s in src:
            if time.monotonic() - t0 > time_limit or improved:
                break
            sv = float(a[s])
            if sv <= 1e-14:
                continue
            for j in dst:
                if improved or time.monotonic() - t0 > time_limit:
                    break
                if int(j) == int(s):
                    continue
                for frac in (0.05, 0.1, 0.2):
                    if improved or time.monotonic() - t0 > time_limit:
                        break
                    amt = sv * frac
                    b = a.copy()
                    b[s] -= amt
                    b[j] += amt
                    b = np.maximum(b, 0.0)
                    ss = float(b.sum())
                    if ss <= 1e-14:
                        continue
                    b = b / ss
                    sc = objective_norm(b)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best = b.copy()
                        a = b
                        improved = True

        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 30 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    b = a.copy()
                    amt = a[i] * 0.03
                    b[i] -= amt
                    b[j] += amt
                    b = np.maximum(b, 0.0)
                    ss = float(b.sum())
                    if ss > 1e-14:
                        b = b / ss
                        sc = objective_norm(b)
                        if sc < best_score - 1e-13:
                            best_score = sc
                            best = b.copy()
                            a = b
                            stall = 0

    return best
