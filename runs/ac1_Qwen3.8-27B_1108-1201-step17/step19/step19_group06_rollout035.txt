import numpy as np
import time

def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    power = A * A
    return np.fft.irfft(power, fft_size)[:size]

def norm_nonneg(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    """Safely call the evaluation function."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def soft_max_weights(vals, beta):
    """Compute soft-max weights for a set of values."""
    max_v = vals.max()
    expo = beta * (vals - max_v)
    expo = np.clip(expo, -100, 0)
    w = np.exp(expo)
    ws = w.sum()
    if ws < 1e-30:
        return np.ones(len(vals)) / len(vals)
    return w / ws

def compute_gradient(a, ac, top_k, beta):
    """
    Compute the gradient of max_b (soft version) with respect to a.
    Returns gradient vector of shape (n,).
    """
    n = len(a)
    k_alarm = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_alarm]
    top_vals = ac[top_idx]
    
    weights = soft_max_weights(top_vals, beta)
    
    # d(b_k)/d(a_i) = a_{i-k} + a_{i+k} (with boundary handling)
    # For linear autocorrelation: b_k = sum_j a_j * a_{j+k} for j in valid range
    # d(b_k)/d(a_i) = a_{i-k} (if 0 <= i-k < n and i+k >= 0 and i+k < n... )
    # Actually for linear (full) autocorrelation:
    # b_k = sum_j a_j * a_{j+k} where both indices must be in [0, n-1]
    # d(b_k)/d(a_i): a_i appears as a_j when j=i (giving a_{i+k}) and as a_{j+k} when j+k=i (giving a_{i-k})
    # So d(b_k)/d(a_i) = a_{i+k} (if i+k < n) + a_{i-k} (if i-k >= 0)
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        # a_i appears with contribution a_{i+k} if 0 <= i+k < n, i.e., i < n-k
        # and a_{i-k} if 0 <= i-k < n, i.e., i >= k
        if k_int == 0:
            # b_0 = sum a_i^2, db_0/da_i = 2*a_i
            grad += 2.0 * w * a
        else:
            # a_{i+k} for i in [0, n-k-1]
            if n - k_int > 0:
                grad[:n-k_int] += w * a[k_int:]
            # a_{i-k} for i in [k, n-1]
            if k_int < n:
                grad[k_int:] += w * a[:n-k_int]
    
    return grad

def log_adam_optimize(a0, time_limit, lr_init=0.05, beta_init=3.0, top_k_init=50,
                      beta_max=8.0, top_k_max=200, verbose=True, rng_seed=42):

    Adam optimizer in log-space with adaptive soft-max gradient.
    """
    n = len(a0)
    x = np.log(np.maximum(a0, 1e-30).astype(np.float64))
    x -= x.max()  # numerical stability
    
    best_a = norm_nonneg(a0)
    best_m = fast_autocorr(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    # Adam state
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_init
    
    beta = beta_init
    top_k = top_k_init
    stall = 0
    restart_every = 3000
    
    for it in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t += 1
        # Compute current a from x

        a = np.exp(x_shifted)

        ac = fast_autocorr(a)
        m_curr = ac.max()
        
        if m_curr < best_m - 1e-13:
            best_m = m_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        grad = compute_gradient(a, ac, top_k, beta)
        
        # Adam update
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad**2
        m_hat_hat = m_hat / (1 - beta1**t)
        v_hat_hat = v_hat / (1 - beta2**t)
        
        x -= lr * m_hat_hat / (np.sqrt(v_hat_hat) + eps)
        
        # Adaptive parameters
        if t % 500 == 0:
            top_k = min(top_k + 5, top_k_max)
            beta = min(beta + 0.2, beta_max)

            a_blend = norm_nonneg((1-w) * best_a + w * a_pert)
            x = np.log(np.maximum(a_blend, 1e-30))
            x -= x.max()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            t = 0
            lr = max(lr * 0.8, 0.001)
            
            if verbose:
                J = 2.0 * n * best_m

    if verbose:
        J = 2.0 * n * best_m
        print(f"    Log-Adam done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m

def pairwise_coord_descent(a0, time_limit, n_top_lags=20, n_trials=5, verbose=True, rng_seed=77):

    Pairwise coordinate descent: move mass from j to i to reduce the max autocorrelation.
    For lag k, db_k/d(move j->i) = a_{i+k} + a_{i-k} - a_{j+k} - a_{j-k}
    """

    a = norm_nonneg(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    stall = 0
    
    # Precompute the "influence" score for each position at each relevant lag
    def influence(i, k):
        """Return a_{i+k} + a_{i-k} with boundary handling."""
        val = 0.0
        if 0 <= i + k < n:
            val += a[i + k]
        if 0 <= i - k < n:
            val += a[i - k]
        return val
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break
        
        # Get top lags
        n_top = min(n_top_lags, 2*n-1)
        top_lags = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k_pos in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k_pos)
            
            # Compute influence scores for all positions at this lag
            infl = np.zeros(n)
            if k_int == 0:
                infl = 2.0 * a
            else:
                # a_{i+k} for i in [0, n-k-1]
                if n - k_int > 0:
                    infl[:n-k_int] += a[k_int:]
                # a_{i-k} for i in [k, n-1]
                if k_int < n:
                    infl[k_int:] += a[:n-k_int]
            
            # We want to move mass from high-influence to low-influence positions
            # Try the best few moves
            # Sort by influence: sources = highest, destinations = lowest
            order = np.argsort(infl)
            
            n_moves = min(n_trials, n // 4)
            for trial in range(n_moves):
                # Source: highest influence
                src_candidates = order[::-1][:max(1, n//20)]
                # Destination: lowest influence
                dst_candidates = order[:max(1, n//20)]
                
                best_move = None
                best_delta = 0
                
                for s_idx in src_candidates[:5]:

                        s_i = int(s_idx)
                        d_i = int(d_idx)
                        if s_i == d_i:
                            continue

                            continue

                        if a[s_i] < 1e-15:
                            continue
                        best_move = (s_i, d_i, delta_k)
                        best_delta = delta_k
                        break
                    if best_move is not None:
                        break
                
                if best_move is None:
                    continue
                
                s_i, d_i, _ = best_move
                
                # Binary search for optimal fraction
                lo, hi = 0.0, a[s_i]
                for _ in range(8):
                    mid = (lo + hi) / 2
                    a_trial = a.copy()
                    a_trial[s_i] -= mid
                    a_trial[d_i] += mid
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        break
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < M_curr:
                        lo = mid
                    else:
                        hi = mid
                
                if lo > 1e-15:
                    a[s_i] -= lo
                    a[d_i] += lo
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < M_curr:

def interpolated_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-space Adam optimizer with adaptive soft-max gradient on autocorrelation peaks
2. Fast pairwise coordinate descent using influence scores
3. Multi-length exploration via interpolation
4. Multiple restarts with diverse asymmetric initializations
5. Final fine-grained polish phase

import numpy as np
import time


def fast_autocorr(a):
    """Compute full autocorrelation (linear convolution of a with itself) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    power = A * A
    return np.fft.irfft(power, fft_size)[:size]


def norm_nonneg(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely call the evaluation function."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def soft_max_weights(vals, beta):
    """Compute normalized soft-max weights for a set of values."""
    max_v = vals.max()
    expo = beta * (vals - max_v)
    expo = np.clip(expo, -100.0, 0.0)
    w = np.exp(expo)
    ws = w.sum()
    if ws < 1e-30:
        return np.ones(len(vals)) / len(vals)
    return w / ws


def compute_gradient(a, ac, top_k, beta):
    """
    Compute soft-max gradient of max autocorrelation w.r.t. a.
    d(c_k)/d(a_i) = 2*a_{k-i} if 0 <= k-i < n, else 0.
    """
    n = len(a)
    k_alarm = min(top_k, 2 * n - 1)
    top_idx = np.argsort(ac)[::-1][:k_alarm]
    top_vals = ac[top_idx]
    weights = soft_max_weights(top_vals, beta)

    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        # grad[lo:hi+1] += 2*w*a[k-hi : k-lo+1][::-1]
        seg = a[k_int - hi : k_int - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * w * seg

    return grad


def log_adam_optimize(a0, time_limit, lr_init=0.03, beta_init=3.0, top_k_init=50,
                      beta_max=8.0, top_k_max=250, verbose=True, rng_seed=42):
    """
    Adam optimizer in log-space with adaptive soft-max gradient.
    Works on x = log(a), with a = softmax(x) to maintain sum=1 and a>0.
    """
    n = len(a0)
    a_start = norm_nonneg(a0)
    x = np.log(np.maximum(a_start, 1e-30))
    x -= x.max()

    best_a = a_start.copy()
    best_m = fast_autocorr(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    # Adam hyperparameters
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    step = 0
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    lr = lr_init

    beta = beta_init
    top_k = top_k_init
    stall = 0
    restart_every = 2500

    for it in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break

        step += 1

        # Compute a from x via softmax
        x_shifted = x - x.max()
        a = np.exp(x_shifted)
        a = a / a.sum()

        # Compute autocorrelation
        ac = fast_autocorr(a)
        m_curr = ac.max()

        if m_curr < best_m - 1e-13:
            best_m = m_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Compute gradient
        grad = compute_gradient(a, ac, top_k, beta)

        # Adam update in log-space
        m_hat = beta1 * m_hat + (1.0 - beta1) * grad
        v_hat = beta2 * v_hat + (1.0 - beta2) * grad * grad
        m_corr = m_hat / (1.0 - beta1 ** step)
        v_corr = v_hat / (1.0 - beta2 ** step)

        x -= lr * m_corr / (np.sqrt(v_corr) + eps)

        # Adaptive parameter tuning
        if step % 400 == 0:
            top_k = min(top_k + 8, top_k_max)
            beta = min(beta + 0.3, beta_max)
            lr = lr_init / (1.0 + 0.02 * np.sqrt(step))

        # Stagnation restart
        if stall > restart_every:
            stall = 0
            # Perturb around best solution
            noise_scale = rng.uniform(0.01, 0.05)
            a_pert = best_a * (1.0 + rng.normal(0, noise_scale, n))
            a_pert = norm_nonneg(a_pert)
            w_blend = rng.uniform(0.1, 0.3)
            a_blend = norm_nonneg((1.0 - w_blend) * best_a + w_blend * a_pert)
            x = np.log(np.maximum(a_blend, 1e-30))
            x -= x.max()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            step = 0
            lr = max(lr * 0.7, 0.0005)
            beta = max(beta - 0.5, 1.5)

            if verbose:
                J = 2.0 * n * best_m
                print(f"    LogAdam restart it={it} ({time.monotonic()-t0:.0f}s): "
                      f"J={J:.6f}, beta={beta:.1f}, top_k={top_k}, lr={lr:.4f}")

    if verbose:
        J = 2.0 * n * best_m
        print(f"    LogAdam done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def pairwise_coord_descent(a0, time_limit, n_top_lags=25, n_best_src=8, n_best_dst=8,
                           verbose=True, rng_seed=77):
    """
    Fast pairwise coordinate descent: move mass from high-influence to low-influence
    positions to reduce the maximum autocorrelation.
    """
    n = len(a0)
    a = norm_nonneg(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    stall_threshold = 3000

    def compute_influence(a_curr, k_int):
        """Compute influence vector: infl[i] = a[i+k] (if valid) + a[i-k] (if valid)."""
        infl = np.zeros(n)
        if k_int == 0:
            infl[:] = 2.0 * a_curr
        else:
            if n - k_int > 0:
                infl[:n - k_int] += a_curr[k_int:]
            if k_int < n:
                infl[k_int:] += a_curr[:n - k_int]
        return infl

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > stall_threshold:
                break

        # Get top lags
        n_top = min(n_top_lags, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:n_top]

        improved_this_round = False

        for k_pos in top_lags:
            if time.monotonic() - t0 > time_limit:
                break

            k_int = int(k_pos)
            infl = compute_influence(a, k_int)

            # Find best source-destination pairs
            order = np.argsort(infl)
            src_candidates = order[::-1][:n_best_src]
            dst_candidates = order[:n_best_dst]

            best_move_score = 0.0
            best_move = None

            for s_idx in src_candidates:
                s_i = int(s_idx)
                if a[s_i] < 1e-15:
                    continue
                for d_idx in dst_candidates:
                    d_i = int(d_idx)
                    if s_i == d_i:
                        continue
                    # We want infl[d_i] - infl[s_i] < 0 (reduces b_k)
                    delta = infl[d_i] - infl[s_i]
                    if delta < best_move_score:
                        best_move_score = delta
                        best_move = (s_i, d_i, delta)

            if best_move is None or best_move[2] >= -1e-15:
                continue

            s_i, d_i, _ = best_move

            # Try several fractions to find the best improvement
            max_frac = a[s_i] * 0.5
            best_frac = 0.0
            best_trial_m = M_curr

            fracs = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]
            for frac in fracs:
                amt = max_frac * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[s_i] -= amt
                a_trial[d_i] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_trial_m:
                    best_trial_m = m_trial
                    best_frac = amt

            if best_frac > 1e-18:
                a[s_i] -= best_frac
                a[d_i] += best_frac
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                improved_this_round = True

        if not improved_this_round:
            if stall > 0 and stall % 300 == 0:
                # Random perturbation to escape local minimum
                n_moves = int(rng.integers(3, 12))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = float(rng.uniform(0.002, 0.015))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        J = 2.0 * n * best_m
        print(f"    PairwiseCD done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def fine_pairwise_cd(a0, time_limit, verbose=True, rng_seed=111):
    """
    Even finer pairwise CD: processes lags in priority order and tries
    more destinations per source.
    """
    n = len(a0)
    a = norm_nonneg(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    def compute_influence(a_curr, k_int):

        if k_int == 0:
            infl[:] = 2.0 * a_curr
        else:
            if n - k_int > 0:
                infl[:n - k_int] += a_curr[k_int:]
            if k_int < n:
                infl[k_int:] += a_curr[:n - k_int]
        return infl

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break

        n_top = min(30, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k_pos in top_lags:
            if time.monotonic() - t0 > time_limit:
                break

            k_int = int(k_pos)
            infl = compute_influence(a, k_int)
            order = np.argsort(infl)

            # Try top sources with many destinations
            n_src = min(15, n // 10)
            n_dst = min(20, n // 10)

            for s_pos in range(n_src - 1, -1, -1):
                s_i = int(order[s_pos])
                if a[s_i] < 1e-15:
                    continue

                for d_pos in range(min(n_dst, n // 5)):
                    d_i = int(order[d_pos])
                    if s_i == d_i:
                        continue
                    delta = infl[d_i] - infl[s_i]
                    if delta >= -1e-15:
                        continue

                    # Quick trial with small step
                    amt = a[s_i] * 0.1
                    a_trial = a.copy()
                    a_trial[s_i] -= amt
                    a_trial[d_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    m_trial = fast_autocorr(a_trial).max()

                    if m_trial < M_curr - 1e-14:
                        # Accept and continue
                        a = a_trial
                        M_curr = m_trial
                        improved = True
                        break

                if improved:
                    break

        if not improved and stall > 200:
            # Perturb
            a_pert = best_a * (1.0 + rng.normal(0, 0.01, n))
            a = norm_nonneg(a_pert)
            m_new = fast_autocorr(a).max()
            if m_new < best_m:

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_int / ss


def make_asymmetric_init(n, rng):
    """Create a diverse asymmetric initial sequence."""
    x = np.linspace(0, 1, n)
    choice = int(rng.integers(8))

    if choice == 0:
        # Exponential decay
        r = float(rng.uniform(1.0, 5.0))
        a = np.exp(-r * x)
    elif choice == 1:
        # Power law decay
        p = float(rng.uniform(0.5, 3.0))
        a = (1.0 - x + 0.01) ** p
    elif choice == 2:
        # Asymmetric Gaussian
        c = float(rng.uniform(0.2, 0.4))
        sig = float(rng.uniform(0.1, 0.3))
        a = np.exp(-((x - c) ** 2) / (2 * sig ** 2))
    elif choice == 3:
        # Bimodal asymmetric
        c1 = float(rng.uniform(0.1, 0.3))
        c2 = float(rng.uniform(0.6, 0.9))
        sig = float(rng.uniform(0.05, 0.2))
        w1 = float(rng.uniform(0.3, 0.7))
        a = w1 * np.exp(-((x - c1) ** 2) / (2 * sig ** 2)) + \
            (1 - w1) * np.exp(-((x - c2) ** 2) / (2 * sig ** 2))
    elif choice == 4:
        # Sine half-wave
        f = float(rng.uniform(0.5, 2.0))
        shift = float(rng.uniform(0, 0.5))
        a = np.sin(np.pi * f * (x + shift)) ** 2
        a = np.maximum(a, 0.0)
    elif choice == 5:
        # Linear ramp with noise
        a = x * float(rng.uniform(0.5, 2.0)) + (1 - x) * float(rng.uniform(0.1, 0.5))
    elif choice == 6:
        # Two-exponential
        r1 = float(rng.uniform(1, 4))
        r2 = float(rng.uniform(1, 4))
        w = float(rng.uniform(0.3, 0.7))
        a = w * np.exp(-r1 * x) + (1 - w) * np.exp(-r2 * (1 - x))
    else:
        # Random smooth
        a = np.ones(n)
        for _ in range(int(rng.integers(3, 8))):
            c = float(rng.uniform(0, 1))
            sig = float(rng.uniform(0.05, 0.3))
            amp = float(rng.uniform(0.5, 2.0))
            a += amp * np.exp(-((x - c) ** 2) / (2 * sig ** 2))
        a = np.maximum(a, 0.0)

    return norm_nonneg(a)

        check_update(a_out, "logadam_fine")

    # === Phase 4: Fine pairwise CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Fine Pairwise CD (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_out, _ = fine_pairwise_cd(best_seq, t4, verbose=False, rng_seed=200)
        check_update(a_out, "fine_paircd")

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration (n_best={n_best}, budget={remaining:.0f}s) ===")

    length_tries = []
    for delta in [-1500, -1000, -500, 500, 1000, 1500, 2000, 3000]:
        nn = n_best + delta
        if 2000 <= nn <= 12000:
            length_tries.append(nn)
    for rn in [3000, 4000, 5000, 6000, 7000, 8000, 9000]:
        if 2000 <= rn <= 12000:
            length_tries.append(rn)
    length_tries = sorted(set(length_tries))

    for new_n in length_tries[:5]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        if new_n == n_best:
            continue
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")

        a_interp = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.40, 120.0)
        if t5a > 15:
            a_out, _ = log_adam_optimize(a_interp, t5a, lr_init=0.02, beta_init=3.0,
                                         top_k_init=50, beta_max=7.0, top_k_max=200,
                                         verbose=False, rng_seed=new_n % 10000)
            check_update(a_out, f"logadam_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t5b = min(remaining * 0.35, 80.0)
            if t5b > 10:
                a_out, _ = pairwise_coord_descent(best_seq, t5b, n_top_lags=20,
                                                  n_best_src=5, n_best_dst=5,
                                                  verbose=False, rng_seed=new_n % 5000)
                check_update(a_out, f"paircd_n{new_n}")

    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_final}, budget={remaining:.0f}s) ===")

    # Final Log-Adam with very fine settings
    t6a = min(remaining * 0.40, 150.0)
    if t6a > 10:
        a_out, _ = log_adam_optimize(best_seq, t6a, lr_init=0.003, beta_init=7.0,
                                     top_k_init=200, beta_max=12.0, top_k_max=500,
                                     verbose=False, rng_seed=555)
        check_update(a_out, "final_logadam")

    # Final fine pairwise CD
    remaining = time_budget - (time.monotonic() - t_global)
    t6b = min(remaining * 0.45, 100.0)
    if t6b > 10:
        a_out, _ = fine_pairwise_cd(best_seq, t6b, verbose=False, rng_seed=888)
        check_update(a_out, "final_fine_cd")

    # Absolute final: one more pairwise pass
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        a_out, _ = pairwise_coord_descent(best_seq, remaining * 0.8, n_top_lags=40,
                                          n_best_src=12, n_best_dst=12,
                                          verbose=False, rng_seed=999)
        check_update(a_out, "absolute_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        best_seq = np.ones(n_fb, dtype=np.float64) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")

    return [float(x) for x in best_seq]
