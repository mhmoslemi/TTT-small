"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Mirror descent on simplex with FFT-accelerated exact subgradient
2. Adaptive active set, backtracking step, path averaging
3. Multi-restart with diverse perturbations (including asymmetry tilt)
4. Coordinate-wise local search
5. LP-based Frank-Wolfe polishing

import numpy as np
import time
import cvxpy as cp


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def eval_obj(a):
    """Compute J = 2*n*max(conv) / sum^2 for the given sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    r = fast_autocorr(a)
    return float(2.0 * n * r.max() / (s * s))


def compute_subgradient(a, active_indices):
    """
    Compute the subgradient of max_{k in active} r_k(a) at point a.
    dr_k/da_m = 2*a_{k-m} if 0 <= k-m <= n-1, else 0.
    Gradient = (1/|S|) * sum_{k in S} dr_k/da
    Using: g_m = (2/|S|) * sum_j a_j * w[m+j] where w[k]=1/|S| if k in S
    """
    n = len(a)
    size = 2 * n - 1
    w = np.zeros(size)
    w[active_indices] = 1.0 / len(active_indices)
    # g = 2 * correlate(w, a, 'valid') gives length n
    g = 2.0 * np.correlate(w, a, 'valid')
    return g

def local_search(a0, time_limit, verbose=False):
    """Coordinate-wise local search: try moving mass between pairs of positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    acc = 0
    rej = 0
    
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        # Pick two random positions
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        
        # Try moving small amount from i to j (or vice versa)
        delta = rng.uniform(0.001, 0.01) * max(a[i], a[j], 1.0/n)

        a_trial = a.copy()
        a_trial[i] -= delta
        a_trial[j] += delta
        if a_trial[i] < 0:
            a_trial[i], a_trial[j] = a_trial[j], a_trial[i]
            delta = -delta  # flip
            a_trial = a.copy()
            a_trial[j] -= abs(delta)
            a_trial[i] += abs(delta)
            if a_trial[j] < 0:
                rej += 1
                continue

        m_trial = r_trial.max()
        
        if m_trial < best_m - 1e-14:
            a = a_trial
            best_a = a.copy()
            best_m = m_trial
            acc += 1
        else:
            rej += 1
    
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, initial_K=200, max_K=1000, verbose=False):
    """LP-based Frank-Wolfe for polishing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    r_y = fast_autocorr(y)
    M_curr = r_y.max()
    
    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(r_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)
    
    iter_count = 0
    failed_solves = 0
    
    while time.monotonic() - t0 < time_limit:
        r_y = fast_autocorr(y)
        M_curr = r_y.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
        
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - r_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 5:
                constraint_list = sorted(constraint_set)[:init_K]
                constraint_set = set(constraint_list)
                failed_solves = 0
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
