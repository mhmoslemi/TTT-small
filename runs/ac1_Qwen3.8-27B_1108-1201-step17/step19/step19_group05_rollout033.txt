import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft

def fast_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return ifft(A * A, fft_size).real[:size]

def objective(a):
    """Compute J(a) = 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_conv(a)
    return 2 * n * conv.max() / (s * s)

def gs_iterate(a, n_iters=100, beta=0.5, fft_size=None):
    """Gerchberg-Saxton type iteration for flat spectrum."""
    n = len(a)
    if fft_size is None:
        fft_size = 1 << (2 * n - 1).bit_length()
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = objective(a)
    
    for it in range(n_iters):
        A = fft(a, fft_size)
        mag = np.abs(A)
        mag_mean = mag.mean()
        
        # Flatten magnitude
        mag_new = mag + beta * (mag_mean - mag)
        mag_new = np.maximum(mag_new, 1e-10)
        
        # Preserve phase
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        
        a_new = ifft(A_new, fft_size).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        obj_new = objective(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend
            a = (1 - 0.3) * a + 0.3 * a_new
            a = np.maximum(a, 0)
            a = a / max(a.sum(), 1e-15)
    
    return best_a

    # Actually: obj = 2n * M / (sum)^2 = 2n * M (when sum=1)

max $\mathbf{1}^T b$ s.t. $C b \leq M \mathbf{1}$, $b \geq 0$

is:
min $M \mathbf{1}^T y$ s.t. $C^T y \geq \mathbf{1}$, $y \geq 0$

The dual variables $y_k$ indicate which lags are most "active." At the optimum, the complementary slackness condition says: if $(Cb)_k < M$, then $y_k = 0$, and if $b_j > 0$, then $(C^T y)_j = 1$.

We can use the dual to identify the "support" of the active constraints and the "active" variables.

For the gradient of the objective with respect to $a$:
$\nabla_a \max_k (a*a)_k = 2 \sum_{k \in K^*} y_k Q_k a$

where $K^*$ is the set of active lags. This gives a "natural gradient" direction.

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return ifft(A * A, fft_size).real[:size]

def obj_value(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * fast_conv(a).max() / (s * s)

def gs_flat(a, n_iters=200, beta=0.5, time_limit=60.0):

        mag = np.abs(A)
        mag_mean = mag.mean()
        mag_new = mag + beta * (mag_mean - mag)
        mag_new = np.maximum(mag_new, 1e-10)
        phase = np.angle(A)
        a_new = ifft(mag_new * np.exp(1j * phase), fft_size).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:

            best_a = a_new.copy()
