def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_softmax_gradient(a, p):
    """
    Compute gradient of soft-max objective.
    p is the softmax weights over the autocorrelation values.
    Returns grad[i] = 2 * sum_j p[i+j] * a[j]
    """
    n = len(a)
    fft_size = 1 << (3 * n - 3).bit_length()
    A = np.fft.rfft(a, fft_size)
    P_flip = np.fft.rfft(p[::-1], fft_size)
    conv_ap = np.fft.irfft(A * P_flip, fft_size)[:3 * n - 2]
    g = conv_ap[n - 1:2 * n - 1][::-1]
    return 2.0 * g


def fista_optimize(a0, time_limit, tau_init=5.0, tau_end=50.0, n_max_iters=20000,
                   verbose=False, rng_seed=77):
    """FISTA with soft-max smooth approximation of peak autocorrelation."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = np.maximum(a, 0.0) / s

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    y = a.copy()
    k = 1.0
    eta = 1.0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive temperature
        progress = min(1.0, (time.monotonic() - t0) / min(time_limit, 900.0))
        tau = tau_init + (tau_end - tau_init) * progress

        ac = fast_autocorr(y)
        ac_max = ac.max()
        shifted = (ac - ac_max) / tau
        exp_s = np.exp(shifted)
        p = exp_s / exp_s.sum()

        g = compute_softmax_gradient(y, p)
        g_tangent = g - g.mean()

        y = x - eta * g_tangent
        y = project_simplex(y)

        ac_y = fast_autocorr(y)
        m_y = ac_y.max()

        if m_y < best_m - 1e-15:
            best_m = m_y
            best_a = y.copy()
            eta = min(eta * 1.01, 5.0)
        else:
            eta = eta * 0.8

        k_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * k * k))
        y = x + ((k - 1.0) / k_new) * (x - y)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = x.copy()
        else:
            y = y / ys
        k = k_new
        x = y

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA it={it} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}")

    return best_a, best_m


def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=False, top_K_init=300,
                      top_K_max=None, adaptive_K=True, use_backtrack=True):
    """Improved LP-Frank-Wolfe with backtracking line search."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = np.maximum(a, 0.0) / s

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0
    stall = 0
    max_stall = 30

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if adaptive_K:
            # Use constraints that are within 5% of the peak
            threshold = M_curr * 0.95
            binding_mask = ac >= threshold
            binding_idx = np.where(binding_mask)[0]
            if len(binding_idx) > top_K_max:
                binding_idx = np.argsort(ac)[::-1][:top_K_max]
            elif len(binding_idx) < max(50, top_K_init):
                binding_idx = np.argsort(ac)[::-1][:min(top_K_init, 2 * n - 1)]
        else:
            binding_idx = np.argsort(ac)[::-1][:min(K, 2 * n - 1)]

        K_sel = len(binding_idx)

        # Build constraint matrix
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue

        # Backtracking line search
        if use_backtrack:
            alpha = 1.0
            base_m = fast_autocorr(a).max()
            best_alpha = 2.0 / (iter_count + 2.0)

            for trial_alpha in [1.0, 0.7, 0.5, 0.33, 0.25, 0.15, 0.1, 0.05,

                if m_trial < best_trial_m:

                    best_alpha = trial_alpha

        a = np.maximum(a, 0.0)

        if s_a < 1e-15:

        a = a / s_a

        m_new = fast_autocorr(a).max()

            m_p = fast_autocorr(a_p).max()
            if m_p < best_m:

                best_a = a_p.copy()
            a = a_p

        if adaptive_K and iter_count % 20 == 0:

        if verbose and iter_count % 10 == 0:

            J = 2.0 * n * best_m
            print(f"    LP-FW v2 iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={alpha:.4f}")

def peak_shaving(a0, time_limit, verbose=False, n_peaks=5, verbose_every=50):
    """Directly shave peaks by redistributing mass from contributing positions."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = np.maximum(a, 0.0) / s

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:n_peaks]

        improved = False
        for k in top_peaks:
            ki = int(k)
            # Find pairs (j, ki-j) that contribute to ac[ki]
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            js = np.arange(j_start, j_end + 1)
            pairs = [(j, ki - j) for j in js if 0 <= ki - j < n]

            if not pairs:
                continue

            # Find the pair with maximum contribution
            max_contrib = 0
            max_pair = None
            for j1, j2 in pairs:
                contrib = a[j1] * a[j2]
                if contrib > max_contrib:
                    max_contrib = contrib
                    max_pair = (j1, j2)

            if max_pair is None or max_contrib < 1e-20:
                continue

            j1, j2 = max_pair

            # Try to move mass from the higher-value position to somewhere that doesn't contribute to the peak
            src = j1 if a[j1] >= a[j2] else j2
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find good destinations (positions with low autocorrelation contribution)
            # Compute sensitivity: how much does each position contribute to top peaks?
            sens = np.zeros(n)
            for kp in top_peaks[:3]:
                kpi = int(kp)
                s2 = max(0, kpi - n + 1)
                e2 = min(n - 1, kpi)
                for j2_i in range(s2, e2 + 1):
                    pj = kpi - j2_i
                    if 0 <= pj < n:
                        sens[j2_i] += a[pj]
                        sens[pj] += a[j2_i]

            # Candidates: positions not in the peak's contributing set, with low sensitivity
            in_peak = set()
            for kp in top_peaks[:3]:
                kpi = int(kp)
                s2 = max(0, kpi - n + 1)
                e2 = min(n - 1, kpi)
                for j2_i in range(s2, e2 + 1):
                    pj = kpi - j2_i
                    if 0 <= pj < n:
                        in_peak.add(j2_i)
                        in_peak.add(pj)

            candidates = np.array([i for i in range(n) if i not in in_peak], dtype=int)
            if len(candidates) == 0:
                candidates = np.arange(n)

            cand_sens = sens[candidates]
            # Sort by sensitivity (lowest first)
            order = np.argsort(cand_sens)
            top_dsts = candidates[order[:5]]

            found = False
            for dst in top_dsts:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.003, 0.015, 0.03, 0.008]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break

        iter_count += 1
        if not improved and iter_count % verbose_every == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave it={iter_count} ({elapsed:.0f}s): J={J:.6f}")

        if not improved and iter_count > 500:
            break

    return best_a, best_m


def pocs_spectral(a0, time_limit, n_trials=3, rng_seed=11):
    """POCS with spectral flatness projection."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = np.maximum(a, 0.0) / s

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit - 5:
            break

        rng = np.random.default_rng(rng_seed + trial * 100)

        # Random phase init
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        A_spec = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
        A_full = np.zeros(fft_size, dtype=np.complex128)
        A_full[:fft_size // 2 + 1] = A_spec
        for i in range(1, fft_size // 2):
            A_full[fft_size - i] = np.conj(A_spec[i])
        a_cur = np.fft.irfft(A_full, fft_size)[:n]
        a_cur = np.maximum(a_cur, 0.0)
        ss = a_cur.sum()
        if ss < 1e-15:
            continue
        a_cur = a_cur / ss

        t_trial_start = time.monotonic()
        for it in range(100000):
            if time.monotonic() - t_trial_start > t_trial:
                break
            # Spectral projection
            A = np.fft.rfft(a_cur, fft_size)
            mag = np.abs(A)
            mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            alpha = 0.05 + 0.3 * (it / min(it + 1, 200))
            new_mag = (1 - alpha) * mag + alpha * mag_mean
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            # Time domain projection
            a_new = np.maximum(a_new, 0.0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = a_cur.copy()
            else:
                a_new = a_new / ss
            a_cur = a_new
            m = fast_autocorr(a_cur).max()
            if m < best_m - 1e-15:
                best_m = m
                best_a = a_cur.copy()

    return best_a, best_m


def coordinate_polish_v2(a0, time_limit, verbose=False, rng_seed=321):
    """Enhanced coordinate descent with multi-scale moves."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = np.maximum(a, 0.0) / s

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 200
    fracs = [0.002, 0.005, 0.01, 0.02, 0.003, 0.007, 0.015, 0.03]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:15]

        # Compute sensitivity for destinations
        sens = np.zeros(n)
        for k in top_k[:8]:
            ki = int(k)
            s2 = max(0, ki - n + 1)
            e2 = min(n - 1, ki)
            idx = np.arange(s2, e2 + 1)
            sens[idx] += a[ki - idx]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit - 2:
                break
            ki = int(k)
            s2 = max(0, ki - n + 1)
            e2 = min(n - 1, ki)
            idx = np.arange(s2, e2 + 1)
            contrib = a[idx] * a[ki - idx]

            if contrib.max() < 1e-20:
                continue

            src_local = idx[np.argmax(contrib)]
            src_val = a[src_local]
            if src_val < 1e-12:
                continue

            in_peak = np.zeros(n, dtype=bool)
            in_peak[idx] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            order = np.argsort(cand_sens)
            top_cands = candidates[order[:6]]

            found = False
            for dst in top_cands:
                if dst == src_local or found:
                    continue
                for frac in fracs:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src_local] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordV2 ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolated_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator combining multiple optimization strategies."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        try:
            sc = evaluate_sequence([float(x) for x in a_norm])
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = try_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    def try_eval(a):
        try:
            return float(evaluate_sequence([float(x) for x in a]))
        except Exception:
            return float('inf')

    if best_seq is None:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = try_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: POCS Spectral Init ===
    print(f"\n=== Phase 1: POCS (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 120.0)
    if t1 > 20:
        a_pocs, _ = pocs_spectral(best_seq, t1, n_trials=3, rng_seed=11)
        try_update(a_pocs, "pocs")

    # === Phase 2: LP-FW v2 (main workhorse) ===
    print(f"\n=== Phase 2: LP-FW v2 (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 40:
        a_in = best_seq.copy()
        a_out, _ = lp_frank_wolfe_v2(a_in, t2, solver, verbose=True,
                                      top_K_init=300, top_K_max=min(3000, 2 * n0 - 1),
                                      adaptive_K=True, use_backtrack=True)
        try_update(a_out, "lpfw1")

    # === Phase 3: FISTA ===
    print(f"\n=== Phase 3: FISTA (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_in = best_seq.copy()
        a_out, _ = fista_optimize(a_in, t3, tau_init=5.0, tau_end=50.0,
                                   verbose=True, rng_seed=77)
        try_update(a_out, "fista1")

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 20:
        a_in = best_seq.copy()
        a_out, _ = peak_shaving(a_in, t4, verbose=True, n_peaks=5)
        try_update(a_out, "peakshave1")

    # === Phase 5: LP-FW v2 round 2 ===
    print(f"\n=== Phase 5: LP-FW v2 r2 (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.2, 300.0)
    if t5 > 40:
        a_in = best_seq.copy()
        a_out, _ = lp_frank_wolfe_v2(a_in, t5, solver, verbose=True,
                                      top_K_init=500, top_K_max=min(5000, 2 * n0 - 1),
                                      adaptive_K=True, use_backtrack=True)
        try_update(a_out, "lpfw2")

    # === Phase 6: FISTA round 2 with different params ===
    print(f"\n=== Phase 6: FISTA r2 (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.1, 150.0)
    if t6 > 20:
        a_in = best_seq.copy()
        a_out, _ = fista_optimize(a_in, t6, tau_init=10.0, tau_end=80.0,
                                   verbose=True, rng_seed=88)
        try_update(a_out, "fista2")

    # === Phase 7: Coordinate polish ===
    print(f"\n=== Phase 7: Coordinate (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.1, 120.0)
    if t7 > 15:
        a_in = best_seq.copy()
        a_out, _ = coordinate_polish_v2(a_in, t7, verbose=True, rng_seed=321)
        try_update(a_out, "coord1")

    # === Phase 8: Length exploration ===
    print(f"\n=== Phase 8: Length exploration ===")
    n_best = len(best_seq)
    trial_lengths = []
    for delta in [300, -300, 600, -600, 1000, -1000]:
        new_n = n_best + delta
        if 1000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Exploring n={new_n}...")
        a_long = interpolated_sequence(best_seq, new_n)

        t8a = min(remaining * 0.3, 100.0)
        if t8a > 20:
            a_out, _ = lp_frank_wolfe_v2(a_long, t8a, solver, verbose=False,
                                          top_K_init=200, top_K_max=min(2000, 2 * new_n - 1),
                                          adaptive_K=True, use_backtrack=True)
            try_update(a_out, f"lpfw_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t8b = min(remaining * 0.3, 60.0)
            if t8b > 15:
                a_in = interpolated_sequence(best_seq, new_n)
                a_out, _ = fista_optimize(a_in, t8b, tau_init=5.0, tau_end=40.0,
                                          verbose=False, rng_seed=new_n % 500)
                try_update(a_out, f"fista_n{new_n}")

    # === Phase 9: Final intensive LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 9: Final LP-FW (n={n_final}) ===")
    if remaining > 40:
        t9 = min(remaining * 0.4, 200.0)
        if t9 > 30:
            a_in = best_seq.copy()
            a_out, _ = lp_frank_wolfe_v2(a_in, t9, solver, verbose=True,
                                          top_K_init=600, top_K_max=min(6000, 2 * n_final - 1),
                                          adaptive_K=True, use_backtrack=True)
            try_update(a_out, "lpfw_final")

    # === Phase 10: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 10: Final polish (n={n_final}) ===")
    if remaining > 15:
        t10 = min(remaining * 0.6, 60.0)
        if t10 > 10:
            a_in = best_seq.copy()
            a_out, _ = coordinate_polish_v2(a_in, t10, verbose=False, rng_seed=789)
            try_update(a_out, "coord_final")

    # Final validation
    if best_seq is not None:
        fs = try_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = try_eval(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
