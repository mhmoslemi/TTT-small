def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)

    sorted_v = np.sort(v)[::-1]

def spectral_flatten(a0, time_limit, verbose=False):
    """Improved Gerchberg-Saxton spectral flattening."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_mean = P.mean()
        if P_max < 1e-30:
            break
        
        # Aggressive flattening: replace values above threshold with threshold
        # Use a soft approach
        alpha = 0.5 + 0.5 * min(1.0, it / 100.0)
        threshold = P_mean * (1.0 + 1.0 / (1.0 + it * 0.01))
        P_capped = np.minimum(P, threshold + alpha * 0.1 * (threshold - P_mean))
        P_new = P_capped
        # Redistribute energy
        energy_deficit = (P - P_new).sum()
        P_new = P_new + energy_deficit / len(P_new)
        P_new = np.maximum(P_new, 1e-30)
        
        # Update amplitude
        mag_old = np.sqrt(np.maximum(P, 1e-30))
        mag_new = np.sqrt(P_new)
        ratio = mag_new / mag_old
        ratio = np.clip(ratio, 0.3, 3.0)
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            blend = 0.3
            a = (1 - blend) * a + blend * a_new
            a = np.maximum(a, 0.0)
            a = a / max(a.sum(), 1e-15)
            M_curr = fast_autocorr(a).max()
            if M_curr < best_m:
                best_m = M_curr
                best_a = a.copy()
        
        if verbose and it % 500 == 0:
            print(f"      Spec it={it} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")
    
    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=False, rng_seed=42):
    """Greedy pairwise moves targeting peak reduction."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_max = int(np.argmax(b))
        
        # Find the pair (i, k_max-i) with the largest product
        j_lo = max(0, k_max - n + 1)
        j_hi = min(n - 1, k_max)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        contrib = a[idx] * a[k_max - idx]
        src = idx[np.argmax(contrib)]
        
        if a[src] < 1e-14:
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * 0.05
                a[i] -= amt
                a[j] += amt
                a = normalize(a)
                if fast_autocorr(a).max() < best_m:
                    best_m = fast_autocorr(a).max()
                    best_a = a.copy()
            continue
        
        # Find destination: minimize impact on top peaks
        top_ks = np.argpartition(b, -15)[-15:]
        costs = np.zeros(n)
        for k2 in top_ks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            if lo2 <= hi2:
                valid = np.arange(lo2, hi2 + 1)
                costs[valid] += a[k2i - valid]
        
        # Find best destination (low cost, not the source)
        costs[src] = np.inf
        sorted_idx = np.argsort(costs)
        
        moved = False
        for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
            dst = int(sorted_idx[0])
            if a[src] * frac < 1e-15:
                continue
            a_trial = a.copy()
            amt = a[src] * frac
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = normalize(a_trial)
            M_t = fast_autocorr(a_trial).max()
            if M_t < best_m - 1e-16:
                a = a_trial
                best_m = M_t
                best_a = a.copy()
                moved = True
                break
        
        if not moved:
            # Small random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.01, 0.03)
                a[i] -= amt
                a[j] += amt
                a = normalize(a)
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
        
        if verbose:
            print(f"      CD ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")
    
    return best_a, best_m


def make_ramp(n, alpha=1.0):
    """Linear ramp initialization."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** alpha
    return a / a.sum()


def make_smooth(n, rng):
    """Random smooth initialization."""
    n_few = max(3, n // 50)
    a_few = rng.random(n_few)
    a_few = a_few + 0.5  # ensure positive
    a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_few), a_few)
    return a / a.sum()


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss
