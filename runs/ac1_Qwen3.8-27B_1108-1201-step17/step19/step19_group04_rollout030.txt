def solve_lp(y, active, q_active, n):
    """Solve the LP for the Frank-Wolfe direction."""
    K = len(active)
    if K < 5:
        return None
    if K * (n + 1) > 12_000_000:
        return None
    
    # Build sparse-ish constraint matrix
    # For each active lag k:
    #   sum_i 2*y[k-i] * g[i] - T <= q_y[k]
    #   where the sum is over valid i (0 <= i < n, 0 <= k-i < n)
    
    A = np.zeros((K, n + 1), dtype=np.float64)
    b = np.zeros(K, dtype=np.float64)
    
    yy = 2.0 * y
    
    for r, k in enumerate(active):
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            # A[r, i0:i1] = yy[k-(i1-1) : k-i0+1][::-1]
            # For i in [i0, i1): coefficient of g[i] is 2*y[k-i]
            # k-i ranges from k-i0 down to k-(i1-1)
            A[r, i0:i1] = yy[k - np.arange(i0, i1)][::-1]
        b[r] = q_active[r]
    
    A[:, n] = -1.0
    
    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0
    
    Aeq = np.tile(1.0, (1, n))  # sum g = 1
    bounds = [(0, np.inf)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=np.array([1.0]),
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    
    g = np.maximum(res.x[:n], 0.0)
    s = g.sum()
    if s < 1e-10:
        return None
    g /= s
    return g
