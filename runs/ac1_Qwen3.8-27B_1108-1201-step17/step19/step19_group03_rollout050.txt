    """Project onto probability simplex."""

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)

    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def nesterov_mirror_descent(a0, time_limit, top_k=80, beta_pow=5.0, c0=5.0,

    x_prev = a.copy()

    cur_beta = beta_pow
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max

            w = w / w_sum

                continue

            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        log_a = np.log(np.maximum(y, 1e-15)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)

        if M_new < best_m - 1e-13:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive parameters
        if stall > restart_every // 3:
            cur_beta = max(cur_beta * 0.92, 2.0)
            cur_top_k = min(cur_top_k + 10, 200)
        
        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            gamma *= 1.1
            cur_top_k = min(cur_top_k + 20, 300)
            cur_beta = max(cur_beta - 1.5, 2.0)
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def smooth_gradient_opt(a0, time_limit, verbose=False, tau_init=10.0, tau_max=500.0,
                         eta_init=0.5, rng_seed=100):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    eta = eta_init
    stall = 0
    max_stall = 800
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1.0))
        tau = tau_init * (tau_max / tau_init) ** progress
        
        ac = fast_autocorr(a)
        ac_max = ac.max()
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()
        
        # Gradient of smoothed objective
        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        
        # Cross term: for each i, sum_k p[k] * a[k-i] where k-i is valid
        # This is a cross-correlation of p and a
        cross = np.convolve(p, a, mode='full')
        # We need (cross)[n-1+i] - (cross)[n-1-1-i]... let me think
        # Actually, grad_i = a[i] * S[i] + sum_k p[k] * a[k-i] (where k-i valid and k-i != i)
        # The cross-correlation term: for fixed i, sum over k of p[k]*a[k-i] where 0<=k-i<n
        # This is like a convolution
        
        # Simpler: use the fact that d/d a[i] of c[k] = 2*a[k-i] (approximately)
        # So grad[i] = sum_k p[k] * 2 * a[k-i] (with valid indices)
        # = 2 * (p_convolve_a)[i] where appropriately defined
        
        # Let's compute it directly for correctness
        grad = 2.0 * np.zeros(n)
        for k in range(len(p)):
            if p[k] < 1e-15:
                continue
            ki = k
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            grad[idx] += 2.0 * p[k] * a[j_idx]
        
        g_tangent = grad - grad.mean()
        
        a_trial = a - eta * g_tangent
        a_new = project_simplex(a_trial)
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.03, 3.0)
        else:
            stall += 1
            if stall > 50:
                eta *= 0.8
            if stall > 150:
                pert = rng.normal(0, 0.003, n) * a
                a_p = a + pert
                a_p = np.maximum(a_p, 1e-10)
                a_p = a_p / a_p.sum()
                m_p = fast_autocorr(a_p).max()
                if m_p < best_m:
                    a = a_p
                    best_m = m_p
                    best_a = a_p.copy()
                    stall = 0
                    eta = eta_init * 0.5
                else:
                    a = a_new
            else:
                a = a_new
        
        total_iters += 1
        
        if verbose and total_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}, stall={stall}")
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    n_top = 15
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_top]
        
        # Sensitivity: how much each coordinate affects peaks
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:8]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]
        
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination (low sensitivity, not in peak)
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            top_cands = candidates[np.argsort(cand_sens)[:5]]
            
            found = False
            for dst in top_cands:
                if dst == src or found:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def make_cosine(n):
    x = np.linspace(0, 1, n)
    a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 2
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 6200
        x = np.linspace(0, 1, n_base)
        best_seq = make_u_shape(n_base, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Nesterov Mirror Descent (multiple inits) ===
    print(f"\n=== Phase 1: Nesterov MD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 250.0)
    
    if t1 > 60:
        inits = [norm_sum1(best_seq)]
        labels = ["parent"]
        if parent is not None:
            inits.append(make_u_shape(n0, 2.5))
            labels.append("u2.5")
            inits.append(make_bimodal(n0, 0.2))
            labels.append("bimodal")
        
        t_per = max(20.0, t1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 120:
                break
            t_avail = min(t_per, time_budget - 120 - (time.monotonic() - t_global))
            if t_avail < 15:
                break
            print(f"  NMD init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = nesterov_mirror_descent(init, t_avail, top_k=80, beta_pow=5.0,
                                                c0=5.0, restart_every=4000,
                                                verbose=False, rng_seed=42 + idx)
            try_update(a_out, f"nmd_{lbl}")
    
    # === Phase 2: Main LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW main (n={n_best}) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 60:
        a_in = norm_sum1(best_seq)
        max_K = min(6000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, max_K=max_K, grow_rate=80)
        try_update(a_out, "lp_main")
    
    # === Phase 3: Smooth gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smooth gradient (n={n_best}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_gradient_opt(a_in, t3, verbose=True, tau_init=8.0, tau_max=400.0,
                                        eta_init=0.6, rng_seed=100)
        try_update(a_out, "smooth")
    
    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinate (n={n_best}) ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, verbose=True, rng_seed=456)
        try_update(a_out, "coord")
    
    # === Phase 5: LP-FW round 2 ===

        new_n = n_best + delta
        if 3000 <= new_n <= 9000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]
    
    for new_n in trial_lengths[:3]:
