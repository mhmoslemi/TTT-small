        t_beta = min(time_limit - (t_beta_start - t0), 200.0)

        # Estimate Lipschitz constant

            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Grow K when stalling
        if stall > 5:
            K = min(K + 300, K_max)
            stall = 0

        iter_count += 1

        if verbose and iter_count % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW-K iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

    return best_a, best_m


def peak_local_search(a0, time_limit, verbose=False, rng_seed=0):
    """Target the top autocorrelation peaks and reduce them via mass transfer."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:20]

        improved = False
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start >= j_end:
                continue

            idx = np.arange(j_start, j_end)
            partner_idx = ki - idx
            contrib = a[idx] * a[partner_idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-20:
                continue

            # Find the index contributing most to this peak
            max_local = np.argmax(contrib)
            src = int(idx[max_local])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find candidate destinations: indices that don't participate in top peaks
            in_peak = np.zeros(n, dtype=bool)
            for kp in top_peaks[:10]:
                kpi = int(kp)
                ps = max(0, kpi - n + 1)
                pe = min(n - 1, kpi)
                in_peak[ps:pe + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            # Try to move mass from src to best candidate
            # Score candidates by how much they'd increase max autocorrelation
            best_dst = None
            best_new_m = M_curr

            # Pre-compute sensitivity: for each candidate dst, estimate cost
            # Moving delta from src to dst changes:
            # c[k] changes by: delta*(a[k-src] - a[k-dst]) + a[src]*(delta at k-src positions)
            # Approximate: the change in c[ki] is -a[ki-src]*(src_val old) + ... 
            # Simpler: just try a few candidates

            sorted_cands = candidates[np.argsort(a[candidates])[:20]]

            for dst in sorted_cands[:10]:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_dst = (dst, amt)

            if best_dst is not None and best_new_m < best_m - 1e-10:
                a[best_dst[0]] += best_dst[1]
                a[src] -= best_dst[1]
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_new_m
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall > 50:
                n_pert = max(10, n // 100)
                pert_idx = rng.choice(n, size=n_pert, replace=False)
                for pi in pert_idx:
                    j = int(rng.integers(n))
                    if j != pi and a[pi] > 1e-10:
                        frac = rng.uniform(0.001, 0.02)
                        amt = a[pi] * frac
                        a[pi] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                stall = 30

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakLocal ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def away_step_fw(a0, time_limit, verbose=False, rng_seed=0):
    """Away-step Frank-Wolfe for escaping vertices."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    # Maintain support
    support = set(np.where(a > 1e-6)[0].tolist())
    if len(support) < n // 10:
        support = set(np.argsort(a)[::-1][:n // 5].tolist())

    t_iter = 1
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        k_star = int(np.argmax(ac))

        # Compute subgradient (direction to minimize)
        # grad[j] = 2*a[k_star - j] for valid j
        grad = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n, k_star + 1)
        for j in range(j_start, j_end):
            grad[j] = 2.0 * a[k_star - j]

        # FW linearization: min grad^T b s.t. b in simplex
        # Optimal b is at vertex: b = e_{argmin grad}
        # But for away-step, consider support
        if len(support) > 2:
            sup_list = list(support)
            g_sup = grad[sup_list]
            # Find index with min gradient in support
            idx_min = np.argmin(g_sup)
            b_index = sup_list[idx_min]
            # Away direction
            a_idx = np.argmax(g_sup)
            away_index = sup_list[a_idx]

            # Away-step FW
            gamma_t = 2.0 / (t_iter + 2.0)
            a_t = 1.0 / (t_iter + 1.0)

            # Try both FW step and away-step
            # FW step
            b_fw = np.zeros(n)
            b_fw[b_index] = 1.0
            a_fw = (1 - gamma_t) * a + gamma_t * b_fw
            a_fw = np.maximum(a_fw, 0.0)
            s_fw = a_fw.sum()
            if s_fw > 1e-15:
                a_fw = a_fw / s_fw

            # Away step (remove mass from away_index)
            m_away = max(a[away_index] - 1e-10, 0.0)
            a_away = a.copy()
            a_away[away_index] = 0.0
            a_away[b_index] += m_away
            s_away = a_away.sum()
            if s_away > 1e-15:
                a_away = a_away / s_away

            # Choose better
            m_fw = fast_autocorr(a_fw).max()
            m_away = fast_autocorr(a_away).max()

            if m_away < m_fw:
                a_new = a_away
                m_new = m_away
            else:
                a_new = a_fw
                m_new = m_fw

            # Update support
            support.discard(away_index)
            support.add(b_index)
            # Remove very small elements from support
            if len(support) > n // 3:
                for si in list(support):
                    if a_new[si] < 1e-8:
                        support.discard(si)
        else:
            b_index = int(np.argmin(grad))
            gamma_t = 2.0 / (t_iter + 2.0)
            b_fw = np.zeros(n)
            b_fw[b_index] = 1.0
            a_new = (1 - gamma_t) * a + gamma_t * b_fw
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
            m_new = fast_autocorr(a_new).max()
            support.add(b_index)

        a = a_new
        t_iter += 1

        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        if verbose and t_iter % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AwayFW t={t_iter} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def momentum_gradient_optimize(a0, time_limit, beta=50, verbose=False, rng_seed=0):
    """Gradient descent with momentum on smoothed objective, projected to simplex."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    v = np.zeros(n)
    mu = 0.9
    eta = 1.0
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit and stall < 5000:
        it += 1
        grad, _ = smoothed_gradient(a, beta)
        v = mu * v + grad
        a_trial = a - eta * v
        a_new = project_simplex(a_trial)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 5.0)
        else:
            a = a_new
            stall += 1
            if stall > 100:
                eta *= 0.8
            if stall > 500:
                # Perturb
                pert = rng.normal(0, 0.01, n) * best_a
                a_p = np.maximum(best_a + pert, 1e-10)
                a = a_p / a_p.sum()
                v = np.zeros(n)
                eta = 0.5
                stall = 200

        # Adapt beta: increase for finer peak resolution
        if it % 200 == 0:
            beta = min(beta * 1.1, 200.0)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MomGrad it={it} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.0f}, eta={eta:.4f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_modulated(n, alpha=0.3, beta_mod=0.1):
    x = np.arange(n)
    a = 1.0 + alpha * (2.0 * x / (n - 1) - 1.0) ** 2 + beta_mod * np.cos(4.0 * np.pi * x / n)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """Main search: FISTA + LP-FW + peak local search + multi-length."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        try:
            a_list = [float(x) for x in a_norm]
            sc = evaluate_sequence(a_list)
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_norm = norm_sum1(parent)
            p_list = [float(x) for x in p_norm]
            p_score = evaluate_sequence(p_list)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = p_norm
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 6200
        best_seq = make_u_shape(n_base, 2.5)
        best_score = float('inf')
        try_update(best_seq, "default_u")
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: FISTA quick pass (multiple betas) ===
    print(f"\n=== Phase 1: FISTA (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 180.0)
    if t1 > 30:
        a_out, _ = fista_optimize(best_seq, t1, betas=[20, 50, 100, 150], verbose=True)
        try_update(a_out, "fista")

    # === Phase 2: Momentum gradient ===
    print(f"\n=== Phase 2: Momentum Gradient (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 20:
        a_out, _ = momentum_gradient_optimize(best_seq, t2, beta=50, verbose=True)
        try_update(a_out, "momgrad")

    # === Phase 3: LP-FW with large K ===
    print(f"\n=== Phase 3: LP-FW Large K (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30 and n0 <= 8000:
        K_init = min(2000, 2 * n0 - 1)
        a_out, _ = lp_fw_adaptive_k(best_seq, t3, solver, verbose=True, K_init=K_init)
        try_update(a_out, "lp_fw_large")

    # === Phase 4: Peak local search ===
    print(f"\n=== Phase 4: Peak Local Search (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out, _ = peak_local_search(best_seq, t4, verbose=True, rng_seed=42)
        try_update(a_out, "peaklocal")

    # === Phase 5: LP-FW all constraints on best ===
    print(f"\n=== Phase 5: LP-FW All (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 30 and n_best <= 7000:
        a_out, _ = lp_fw_all_constraints(best_seq, t5, solver, verbose=True)
        try_update(a_out, "lp_all")

    # === Phase 6: Second FISTA pass on improved ===
    print(f"\n=== Phase 6: FISTA r2 (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 20:
        a_out, _ = fista_optimize(best_seq, t6, betas=[80, 120, 160, 200], verbose=False, rng_seed=99)
        try_update(a_out, "fista_r2")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-length (base n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_base = len(best_seq)
    trial_lengths = [n_base - 400, n_base + 400, n_base - 800, n_base + 800]
    trial_lengths = [l for l in trial_lengths if 3000 <= l <= 9000]

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.35, 120.0)
        if t7a > 20:
            a_out, _ = fista_optimize(a_interp, t7a, betas=[30, 80, 150], verbose=False, rng_seed=new_n)
            try_update(a_out, f"fista_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 7000:
            t7b = min(remaining * 0.45, 150.0)
            if t7b > 20:
                K_init = min(1500, 2 * new_n - 1)
                a_out, _ = lp_fw_adaptive_k(a_interp, t7b, solver, verbose=False, K_init=K_init)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 8: Final combined polish ===
    print(f"\n=== Phase 8: Final Polish (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_final = len(best_seq)
        # FISTA
        t8a = min(remaining * 0.3, 100.0)
        if t8a > 15:
            a_out, _ = fista_optimize(best_seq, t8a, betas=[100, 150, 200], verbose=False, rng_seed=777)
            try_update(a_out, "fista_final")

        remaining = time_budget - (time.monotonic() - t_global)
        # Peak local
        if remaining > 20:
            t8b = min(remaining * 0.4, 100.0)
            if t8b > 10:
                a_out, _ = peak_local_search(best_seq, t8b, verbose=False, rng_seed=555)
                try_update(a_out, "peak_final")

        remaining = time_budget - (time.monotonic() - t_global)
        # LP-FW one more
        if remaining > 30 and n_final <= 7000:
            t8c = min(remaining * 0.5, 120.0)
            if t8c > 20:
                K_init = min(3000, 2 * n_final - 1)
                a_out, _ = lp_fw_adaptive_k(best_seq, t8c, solver, verbose=False, K_init=K_init)
                try_update(a_out, "lp_final")

    # === Final validation ===
    if best_seq is not None:
        try:
            fs = evaluate_sequence([float(x) for x in best_seq])
            if np.isfinite(fs):
                best_score = fs
        except Exception:
            pass
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 6200
        best_seq = make_u_shape(n_fb, 2.5)
        best_score = float('inf')
        try_update(best_seq, "fallback")
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
