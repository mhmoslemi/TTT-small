"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based full-softmax gradient (computed as single convolution)
2. Adam optimizer with adaptive learning rate
3. Temperature annealing for soft-max → hard-max transition
4. Multiple diverse initializations
5. Length exploration via interpolation
6. L-BFGS refinement
7. Coordinate descent polish
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def score_from_ac(a, ac=None):
    n = len(a)
    if ac is None:
        ac = fast_autocorr(a)
    return float(2.0 * n * ac.max())

def smooth_objective_and_grad(a, tau):
    """
    Compute the smoothed objective and its gradient.
    Objective: 2n * logsumexp(conv(a,a) / tau)
    Gradient: (4n/tau) * cross_corr(a, softmax(conv(a,a)/tau))
    Returns (objective_value, gradient_vector)
    """
    n = len(a)
    C = fast_autocorr(a)
    
    # Softmax weights
    C_scaled = C / tau
    C_scaled -= C_scaled.max()
    logits = np.exp(C_scaled)
    Z = logits.sum()
    p = logits / Z
    
    # Objective: 2n * (log(Z) + tau * C.max()... no)
    # Actually: logsumexp(C/tau) = log(sum exp(C[k]/tau))
    # = log(Z) + C.max()/tau... no
    # = (C.max()/tau) + log(sum exp((C[k]-C.max())/tau))
    # Let's just compute it:
    obj = 2.0 * n * (C_scaled.max() + np.log(Z))  # since C_scaled = C/tau - C.max()/tau
    # Wait: C/tau = C_scaled + C.max()/tau
    # logsumexp(C/tau) = logsumexp(C_scaled + C.max()/tau) = C.max()/tau + logsumexp(C_scaled)
    # = C.max()/tau + C_scaled.max() + log(sum exp(C_scaled - C_scaled.max()))
    # But C_scaled = C/tau - C.max()/tau, so C_scaled.max() = (C.max() - C.max())/tau = 0
    # So logsumexp(C/tau) = C.max()/tau + 0 + log(sum exp(C/tau - C.max()/tau))
    # = C.max()/tau + log(sum exp((C - C.max())/tau))
    
    obj = 2.0 * n * (C.max() / tau + np.log(np.sum(np.exp((C - C.max()) / tau))))
    
    # Gradient: (4n/tau) * cross_corr(a, p)
    # cross_corr(a, p)[i] = sum_m a[m] * p[m+i]
    # = conv(a[::-1], p)[n-1 + i]
    
    a_flip = a[::-1]
    size = n + len(p) - 1  # n + (2n-1) - 1 = 3n-2
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a_flip, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_full = np.fft.irfft(A * P, fft_size)
    
    grad = (4.0 * n / tau) * conv_full[n-1 : 2*n-1]
    
    return obj, grad

def adam_optimize(a0, time_limit, tau_init=0.5, tau_min=0.02, 
                  lr_init=0.005, beta1=0.9, beta2=0.999, eps=1e-8,
                  anneal_steps=20000, verbose=True, rng_seed=42):
    """Adam optimization with temperature annealing."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_score = score_from_ac(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    
    tau = tau_init
    lr = lr_init
    t = 0
    stall = 0
    stall_patience = 5000
    check_interval = 5
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal temperature
        tau = max(tau_min, tau_init * (tau_min / tau_init) ** (min(t, anneal_steps) / float(anneal_steps)))
        
        # Compute gradient
        _, grad = smooth_objective_and_grad(a, tau)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        a = norm_sum1(a)
        
        # Check improvement periodically
        if t % check_interval == 0:
            score = score_from_ac(a)
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        
        # Restart if stalled
        if stall > stall_patience:
            stall = 0
            if verbose:
                print(f"    Adam restart t={t}: score={best_score:.6f}, tau={tau:.4f}, lr={lr:.6f}")
            
            # Diverse perturbation
            pert_type = int(rng.integers(5))
            if pert_type == 0:
                # Gaussian bump
                x = np.linspace(0, 1, n)
                c = float(rng.uniform(0.2, 0.8))
                sig = float(rng.uniform(0.05, 0.3))
                pert = np.exp(-((x - c)**2) / (2 * sig**2))
            elif pert_type == 1:
                # Power law
                i_arr = np.arange(n, dtype=np.float64)
                p_exp = float(rng.uniform(-2.0, 2.0))
                pert = (i_arr + 1.0) ** p_exp
            elif pert_type == 2:
                # Sine
                x = np.linspace(0, 1, n)
                f = int(rng.integers(1, 4))
                pert = np.sin(np.pi * f * x + float(rng.uniform(0, np.pi)))**2
            elif pert_type == 3:
                # Random flat spectrum
                fft_size = 1 << (2*n-2).bit_length()
                phases = rng.uniform(0, 2*np.pi, fft_size//2+1)
                mags = rng.uniform(0.5, 1.5, fft_size//2+1)
                spec = mags * np.exp(1j * phases)
                spec[0] = mags[0]  # keep DC real and positive
                a_rand = np.fft.irfft(spec, fft_size)[:n]
                pert = np.abs(a_rand)
            else:
                # Small random
                pert = best_a * (1.0 + rng.normal(0, 0.1, n))
            
            pert = norm_sum1(pert)
            w = float(rng.uniform(0.1, 0.5))
            a = norm_sum1((1 - w) * best_a + w * pert)
            
            # Reset Adam state
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0  # Reset step count for bias correction
            tau = max(tau * 2, 0.3)
            lr = max(lr * 0.7, 1e-5)
            stall_patience = max(int(stall_patience * 0.8), 1000)
        
        # Slowly decrease learning rate
        if t % 10000 == 0:
            lr *= 0.9
    
    if verbose:
        print(f"    Adam done: score={best_score:.6f} (n={n}, time={time.monotonic()-t0:.0f}s)")
    return best_a, best_score

def lbfgs_refine(a0, time_limit, tau=0.1, maxiter=200, verbose=True):
    """Refine using L-BFGS-B on the smoothed objective."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_score = score_from_ac(a)
    t0 = time.monotonic()
    
    def fun_grad(x):
        if time.monotonic() - t0 > time_limit:
            return (float('inf'), np.zeros(n))
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            return (float('inf'), np.zeros(n))
        x = x / s
        obj, grad = smooth_objective_and_grad(x, tau)
        return obj, grad
    
    try:
        result = scipy_minimize(
            fun_grad, a, method='L-BFGS-B',
            options={'maxiter': maxiter, 'maxfun': 500, 'iprint': -1}
        )
        a_new = norm_sum1(result.x)
        score = score_from_ac(a_new)
        if score < best_score:
            best_a = a_new.copy()
            best_score = score
    except Exception as e:
        if verbose:
            print(f"    L-BFGS error: {e}")
    
    if verbose:
        print(f"    L-BFGS done: score={best_score:.6f} (n={n}, time={time.monotonic()-t0:.0f}s)")
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        max_k = int(np.argmax(ac))
        
        improved = False
        # Try adjusting a few random coordinates
        n_trials = min(50, n)
        idx = rng.choice(n, size=n_trials, replace=False)
        
        for i in idx:
            if time.monotonic() - t0 > time_limit:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based full-softmax gradient (single convolution for gradient computation)
2. Adam optimizer with adaptive per-coordinate learning rates
3. Temperature annealing (soft-max -> hard-max transition)
4. Diverse initializations (parametric, flat-spectrum, power-law)
5. L-BFGS-B second-order refinement
6. Length exploration via interpolation
7. Coordinate descent polish
Key insight: gradient of logsumexp(conv(a,a)/tau) w.r.t. a[i] is proportional to
cross-correlation of a with the softmax weights of autocorrelation lags, computable
via a single FFT convolution.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_autocorr(a):
    """Compute full autocorrelation (convolution of a with itself) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely call the evaluation function."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def score_from_ac(a, ac=None):
    """Compute score 2n*max(ac) from a normalized sequence."""
    n = len(a)
    if ac is None:
        ac = fast_autocorr(a)
    return float(2.0 * n * ac.max())


def smooth_obj_grad(a, tau):
    """
    Compute smoothed objective and gradient.
    Objective: 2n * logsumexp(conv(a,a) / tau)
    Gradient: (4n/tau) * cross_corr(a, softmax(conv(a,a)/tau))
    """
    n = len(a)
    C = fast_autocorr(a)

    # Softmax weights
    C_scaled = C / tau
    C_scaled -= C_scaled.max()
    logits = np.exp(C_scaled)
    Z = logits.sum()
    p = logits / Z

    # Objective value
    obj = 2.0 * n * (C.max() / tau + np.log(Z))

    # Gradient: (4n/tau) * cross_corr(a, p)
    # cross_corr(a,p)[i] = sum_m a[m] * p[m+i] = conv(a[::-1], p)[n-1+i]
    a_flip = a[::-1]
    size = n + len(p) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a_flip, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_full = np.fft.irfft(A * P, fft_size)

    grad = (4.0 * n / tau) * conv_full[n - 1:2 * n - 1]

    return obj, grad


def adam_optimize(a0, time_limit, tau_init=0.5, tau_min=0.02,
                  lr_init=0.003, beta1=0.9, beta2=0.999, eps=1e-8,
                  anneal_steps=20000, verbose=True, rng_seed=42):
    """Adam optimization with temperature annealing for the smoothed max autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_score = score_from_ac(a)

    m = np.zeros(n)
    v = np.zeros(n)

    tau = tau_init
    lr = lr_init
    t = 0
    stall = 0
    stall_patience = 5000
    check_interval = 5
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Anneal temperature
        progress = min(t, anneal_steps) / float(anneal_steps)
        tau = max(tau_min, tau_init * (tau_min / tau_init) ** progress)

        # Compute gradient
        _, grad = smooth_obj_grad(a, tau)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad ** 2
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        a = norm_sum1(a)

        # Check improvement periodically
        if t % check_interval == 0:
            score = score_from_ac(a)
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        # Restart if stalled
        if stall > stall_patience:
            stall = 0
            if verbose:
                print(f"    Adam restart t={t}: score={best_score:.6f}, tau={tau:.4f}, lr={lr:.6f}")

            pert_type = int(rng.integers(6))
            if pert_type == 0:
                x = np.linspace(0, 1, n)
                c = float(rng.uniform(0.2, 0.8))
                sig = float(rng.uniform(0.05, 0.3))
                pert = np.exp(-((x - c) ** 2) / (2 * sig ** 2))
            elif pert_type == 1:
                i_arr = np.arange(n, dtype=np.float64)
                p_exp = float(rng.uniform(-2.0, 2.0))
                pert = (i_arr + 1.0) ** p_exp
            elif pert_type == 2:
                x = np.linspace(0, 1, n)
                f = int(rng.integers(1, 5))
                pert = np.sin(np.pi * f * x + float(rng.uniform(0, np.pi))) ** 2
            elif pert_type == 3:
                fft_sz = 1 << (2 * n - 2).bit_length()
                phases = rng.uniform(0, 2 * np.pi, fft_sz // 2 + 1)
                mags = rng.uniform(0.5, 1.5, fft_sz // 2 + 1)
                spec = mags * np.exp(1j * phases)
                spec[0] = mags[0]
                a_rand = np.fft.irfft(spec, fft_sz)[:n]
                pert = np.abs(a_rand)
            elif pert_type == 4:
                # Bimodal with random centers
                x = np.linspace(0, 1, n)
                c1 = float(rng.uniform(0.1, 0.4))
                c2 = float(rng.uniform(0.6, 0.9))
                sig = float(rng.uniform(0.05, 0.2))
                pert = np.exp(-((x - c1) ** 2) / (2 * sig ** 2)) + np.exp(-((x - c2) ** 2) / (2 * sig ** 2))
            else:
                pert = best_a * (1.0 + rng.normal(0, 0.1, n))

            pert = norm_sum1(pert)
            w = float(rng.uniform(0.1, 0.5))
            a = norm_sum1((1.0 - w) * best_a + w * pert)

            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            tau = min(tau * 2.0, 1.0)
            lr = max(lr * 0.7, 1e-5)
            stall_patience = max(int(stall_patience * 0.8), 1000)

        # Slowly decrease learning rate
        if t % 10000 == 0:
            lr *= 0.9

    if verbose:
        print(f"    Adam done: score={best_score:.6f} (n={n}, time={time.monotonic()-t0:.0f}s)")
    return best_a, best_score


def lbfgs_wrap(x, tau, n, time_limit, t0):
    """Wrapper for L-BFGS objective and gradient."""
    if time.monotonic() - t0 > time_limit:
        return (float('inf'), np.zeros(n))
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        return (float('inf'), np.zeros(n))
    x = x / s
    obj, grad = smooth_obj_grad(x, tau)
    return obj, grad


def lbfgs_refine(a0, time_limit, tau=0.1, maxiter=150, verbose=True):
    """Refine using L-BFGS-B on the smoothed objective."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_score = score_from_ac(a)
    t0 = time.monotonic()

    def fun_grad(x):
        res = lbfgs_wrap(x, tau, n, time_limit, t0)
        return res[0], res[1]

    try:
        result = scipy_minimize(
            fun_grad, a, method='L-BFGS-B', jac=True,
            options={'maxiter': maxiter, 'maxfun': 500, 'iprint': -1}
        )
        a_new = norm_sum1(result.x)
        score = score_from_ac(a_new)
        if score < best_score:
            best_a = a_new.copy()
            best_score = score
    except Exception as e:
        if verbose:
            print(f"    L-BFGS error: {e}")

    if verbose:
        print(f"    L-BFGS done: score={best_score:.6f} (n={n}, time={time.monotonic()-t0:.0f}s)")
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def parametric_init(n, rng_seed=0):
    """Generate diverse parametric initializations and return the best."""
    rng = np.random.default_rng(rng_seed)
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)

    candidates = []

    # Uniform
    candidates.append(np.ones(n) / n)

    # Power laws
    for p in [-1.5, -1.0, -0.5, 0.0, 0.5, 1.0, 1.5, 2.0]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))

    # Gaussian bumps
    for c in [0.15, 0.3, 0.5, 0.7, 0.85]:
        for sig in [0.08, 0.15, 0.25, 0.35]:
            candidates.append(norm_sum1(np.exp(-((x - c) ** 2) / (2 * sig ** 2))))

    # Bimodal
    for sig in [0.08, 0.12, 0.18, 0.25]:
        for c1, c2 in [(0.2, 0.8), (0.25, 0.75), (0.3, 0.7)]:
            candidates.append(norm_sum1(
                np.exp(-((x - c1) ** 2) / (2 * sig ** 2)) + np.exp(-((x - c2) ** 2) / (2 * sig ** 2))))

    # Sine waves
    for f in [1, 2, 3, 4]:
        for phase in [0.0, np.pi / 2, np.pi]:
            candidates.append(norm_sum1(np.maximum(np.sin(np.pi * f * x + phase), 0.0) ** 2))

    # Exponential
    for r in [1, 2, 5]:
        candidates.append(norm_sum1(np.exp(-r * x)))
        candidates.append(norm_sum1(np.exp(r * (x - 1.0))))

    # Flat spectrum
    for trial in range(8):
        fft_sz = 1 << (2 * n - 2).bit_length()
        phases = rng.uniform(0, 2 * np.pi, fft_sz // 2 + 1)
        spec = np.exp(1j * phases)
        spec[0] = 1.0
        sig = np.fft.irfft(spec, fft_sz)[:n]
        candidates.append(norm_sum1(np.abs(sig)))

    best_cand = None
    best_val = float('inf')
    for c in candidates:
        v = score_from_ac(c)
        if v < best_val:
            best_val = v
            best_cand = c.copy()

    return best_cand if best_cand is not None else np.ones(n) / n


def coordinate_polish(a0, time_limit, verbose=True):
    """Coordinate descent: adjust individual coordinates to reduce max autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        improved = False
        n_trials = min(100, n)
        idx = rng.choice(n, size=n_trials, replace=False)

        for i in idx:
            if time.monotonic() - t0 > time_limit:
                break

            for delta_frac in [0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5]:
                a_trial = a.copy()
                a_trial[i] *= delta_frac
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < M_curr - 1e-14:
                    a = a_trial
                    M_curr = m_trial
                    improved = True
                    break

        if M_curr < best_m - 1e-13:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 3000:
                # Gradient step escape
                _, grad = smooth_obj_grad(a, 0.1)
                step = 0.01 / (np.abs(grad).max() + 1e-10)
                a = norm_sum1(a - step * grad)
                stall = 0

    if verbose:
        print(f"    Coord polish done: score={2.0 * n * best_m:.6f} (n={n})")
    return best_a, best_m


def try_update(a, best_score_ref, label, t_global):
    """Try to update best. Returns (new_best_seq, new_best_score, improved)."""
    if a is None:
        return best_score_ref[0], best_score_ref[1], False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_score_ref[0], best_score_ref[1], False
    a_n = a / s
    sc = evaluate_safe(a_n)
    if np.isfinite(sc) and sc < best_score_ref[1]:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")
        return a_n.copy(), sc, True
    return best_score_ref[0], best_score_ref[1], False


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator using Adam with FFT-based full-softmax gradient,
    L-BFGS refinement, diverse initializations, and length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            best_seq = parent.copy()
            best_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={best_score:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        n0 = 5000
        best_seq = np.ones(n0) / n0
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n0}, score={best_score:.6f}")

    n0 = len(best_seq)

    # Uniform baseline
    a_uni = np.ones(n0) / n0
    sc_uni = evaluate_safe(a_uni)
    if np.isfinite(sc_uni) and sc_uni < best_score:
        best_score = sc_uni
        best_seq = a_uni.copy()
        print(f"  Uniform improves: {sc_uni:.6f}")

    # === Phase 1: Adam on current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Adam (n={n0}, budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.28, 400.0)
    if t1 > 30:
        a_out, _ = adam_optimize(best_seq, t1, tau_init=0.5, tau_min=0.02,
                                 lr_init=0.003, anneal_steps=30000, verbose=True, rng_seed=42)
        best_seq, best_score, _ = try_update(a_out, (best_seq, best_score), f"adam_n{n0}", t_global)

    # === Phase 2: Diverse initializations ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Diverse inits (n={n0}, budget={remaining:.0f}s) ===")

    inits_to_try = [
        ("param", parametric_init(n0, rng_seed=42)),
        ("power15", norm_sum1((np.arange(n0, dtype=np.float64) + 1.0) ** 1.5)),
        ("exp_decay", norm_sum1(np.exp(-1.0 * np.linspace(0, 1, n0)))),
        ("bimodal", None),
    ]

    # Build bimodal
    x_tmp = np.linspace(0, 1, n0)
    inits_to_try[3] = ("bimodal", norm_sum1(
        np.exp(-((x_tmp - 0.3) ** 2) / 0.02) + np.exp(-((x_tmp - 0.7) ** 2) / 0.02)))

    for name, init in inits_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        s_init = score_from_ac(init)
        if s_init > best_score * 1.05:
            continue
        print(f"  Trying '{name}' (init score={s_init:.6f})")
        t2 = min(remaining * 0.18, 150.0)
        if t2 > 20:
            a_out, _ = adam_optimize(init, t2, tau_init=0.5, tau_min=0.02,
                                     lr_init=0.003, anneal_steps=20000, verbose=False,
                                     rng_seed=hash(name) % 10000)
            best_seq, best_score, _ = try_update(a_out, (best_seq, best_score), f"adam_{name}", t_global)

    # === Phase 3: L-BFGS on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: L-BFGS (n={n_cur}, budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 180.0)
    if t3 > 20:
        a_out, _ = lbfgs_refine(best_seq, t3, tau=0.1, maxiter=120, verbose=True)
        best_seq, best_score, _ = try_update(a_out, (best_seq, best_score), f"lbfgs_n{n_cur}", t_global)

    # === Phase 4: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Length exploration (n_best={n_best}, budget={remaining:.0f}s) ===")

    length_tries = []
    for delta in [-800, -400, -200, 200, 400, 800, 1200, 1600]:
        nn = n_best + delta
        if 2000 <= nn <= 9000 and nn != n_best:
            length_tries.append(nn)
    for rn in [3000, 4000, 5000, 6000, 7000, 8000]:
        if rn != n_best and 2000 <= rn <= 9000:
            length_tries.append(rn)
    length_tries = sorted(set(length_tries))

    for new_n in length_tries[:5]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")

        a_interp = interpolate_sequence(best_seq, new_n)

        t4 = min(remaining * 0.30, 120.0)
        if t4 > 15:
            a_out, _ = adam_optimize(a_interp, t4, tau_init=0.5, tau_min=0.02,
                                     lr_init=0.003, anneal_steps=20000, verbose=False,
                                     rng_seed=new_n % 10000)
            best_seq, best_score, _ = try_update(a_out, (best_seq, best_score), f"adam_n{new_n}", t_global)

    # === Phase 5: Fine-tuning on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Fine-tuning (n={n_best}, budget={remaining:.0f}s) ===")

    t5 = min(remaining * 0.25, 150.0)
    if t5 > 15:
        a_out, _ = adam_optimize(best_seq, t5, tau_init=0.2, tau_min=0.01,
                                 lr_init=0.001, anneal_steps=15000, verbose=True, rng_seed=999)
        best_seq, best_score, _ = try_update(a_out, (best_seq, best_score), f"adam_fine_n{n_best}", t_global)

    # L-BFGS fine
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t6 = min(remaining * 0.25, 100.0)
        if t6 > 15:
            a_out, _ = lbfgs_refine(best_seq, t6, tau=0.05, maxiter=100, verbose=False)
            best_seq, best_score, _ = try_update(a_out, (best_seq, best_score), f"lbfgs2_n{len(best_seq)}", t_global)

    # Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t7 = min(remaining * 0.50, 80.0)
        if t7 > 10:
            a_out, _ = coordinate_polish(best_seq, t7, verbose=False)
            best_seq, best_score, _ = try_update(a_out, (best_seq, best_score), f"coord_n{len(best_seq)}", t_global)

    # Final: one more Adam pass with very low temperature
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t8 = min(remaining * 0.60, 80.0)
        if t8 > 10:
            a_out, _ = adam_optimize(best_seq, t8, tau_init=0.1, tau_min=0.005,
                                     lr_init=0.0005, anneal_steps=10000, verbose=False, rng_seed=7777)
            best_seq, best_score, _ = try_update(a_out, (best_seq, best_score), f"adam_final_n{len(best_seq)}", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")

    return [float(x) for x in best_seq]
