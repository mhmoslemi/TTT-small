def line_search_conv(y, g, max_evals=20, min_improve=1e-15):
    y=normalize_pos(y); g=normalize_pos(g)
    m0 = conv_max_seq(y)
    if not np.any(np.abs(y-g)>1e-15): return y,m0
    # determine initial alphas
    alphas = [0.0, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.55, 0.7, 0.85, 1.0]
    best_a=0; best_m=m0
    for alpha in alphas:
       if alpha==0: continue
       z=y+alpha*(g-y)
       z=np.maximum(z,0)
       s=z.sum()
       if s<=0: continue
       z/=s
       m=conv_max_seq(z)
       if m<best_m:
           best_m=m; best_a=alpha
    if best_a == 0 or best_m >= m0 - 1e-16:

"""
Search for low-convolution-peak nonnegative sequences.

Objective (after sum-normalization):
    J(a) = 2*n*max_k (a*a)_k,  sum(a)=1, a>=0.

The implementation mixes:
  - exact local moves using the full update formula for one-coefficient transfers,
  - cutting-plane-like LP directions when SciPy is available,
  - mirrored Frank-Wolfe/mirror-descent directions on active peak constraints,
  - global transform and length-resampling polish.
"""

import numpy as np
import time

try:
    import scipy.sparse as sp
    from scipy.optimize import linprog
    SCIPY_OK = True
except Exception:
    sp = None
    linprog = None
    SCIPY_OK = False


def normalize_pos(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.size == 0:
        return arr
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = arr.sum()
    if s <= 1e-300:
        return np.ones_like(arr, dtype=np.float64) / float(arr.size)
    arr /= s
    return arr


def interpolate_seq(a, new_n):
    a = normalize_pos(a)
    old_n = int(a.size)
    new_n = int(new_n)
    if old_n == new_n:
        return a.copy()
    if new_n <= 0:
        return a.copy()
    xold = np.linspace(0.0, 1.0, old_n)
    xnew = np.linspace(0.0, 1.0, new_n)
    b = np.interp(xnew, xold, a)
    return normalize_pos(b)


def conv_full_seq(a):
    a = np.ascontiguousarray(np.asarray(a, dtype=np.float64))
    n = int(a.size)
    if n <= 0:
        return np.array([], dtype=np.float64)
    size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(A * A, size)
    return c[:2 * n - 1]


def conv_max_seq(a):
    a = np.ascontiguousarray(np.asarray(a, dtype=np.float64))
    n = int(a.size)
    if n <= 0:
        return 0.0
    size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(A * A, size)
    return float(c[:2 * n - 1].max())


def conv_two_full(u, v):
    u = np.ascontiguousarray(np.asarray(u, dtype=np.float64))
    v = np.ascontiguousarray(np.asarray(v, dtype=np.float64))
    n = int(u.size)
    if n <= 0:
        return np.array([], dtype=np.float64)
    size = 1 << (2 * n - 2).bit_length()
    U = np.fft.rfft(u, size)
    V = np.fft.rfft(v, size)
    return np.fft.irfft(U * V, size)[:2 * n - 1]


def score_fast(a):
    a = normalize_pos(a)
    c = conv_full_seq(a)
    m = float(c.max())
    return 2.0 * float(a.size) * m, c


def combo_vec(y, g, alpha):
    z = y + alpha * (g - y)
    z = np.maximum(z, 0.0)
    s = z.sum()
    if s <= 1e-300:
        return y.copy()
    return z / s


def combo_score(y, g, alpha):
    return conv_max_seq(combo_vec(y, g, alpha))


def line_search_conv(y, g, max_evals=18):
    y = normalize_pos(y)
    g = normalize_pos(g)
    m0 = conv_max_seq(y)
    if max_evals < 5:
        return y, m0
    alphas = [0.0, 0.05, 0.10, 0.15, 0.20, 0.30, 0.40, 0.55, 0.70, 0.90, 1.0]
    if max_evals < len(alphas):
        alphas = alphas[:max_evals]
    best_alpha = 0.0
    best_m = m0
    for alpha in alphas[1:]:
        m = combo_score(y, g, alpha)
        if m < best_m:
            best_m = m
            best_alpha = alpha
    if best_alpha == 0.0 or best_m >= m0 - 1e-15:
        return y, m0
    lo = max(0.0, best_alpha - 0.20)
    hi = min(1.0, best_alpha + 0.20)
    used = len(alphas)
    for _ in range(max(0, max_evals - used)):
        if hi - lo < 1e-7:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        v1 = combo_score(y, g, m1)
        v2 = combo_score(y, g, m2)
        if v1 < v2:
            hi = m2
            if v1 < best_m:
                best_m = v1
                best_alpha = m1
        else:
            lo = m1
            if v2 < best_m:
                best_m = v2
                best_alpha = m2
    z = combo_vec(y, g, best_alpha)
    return z, best_m


def top_indices(arr, K):
    K = int(K)
    n = int(arr.size)
    if K <= 0:
        return np.empty(0, dtype=int)
    if K >= n:
        return np.arange(n, dtype=int)
    return np.argpartition(arr, -K)[-K:].astype(int, copy=False)


def combine_active_indices(arr1, idx1, arr2, idx2, total):
    total = int(total)
    idx1 = np.asarray(idx1, dtype=int).ravel()
    idx2 = np.asarray(idx2, dtype=int).ravel()
    if idx1.size == 0 and idx2.size == 0:
        return np.empty(0, dtype=int)
    parts = []
    if idx1.size > 0:
        parts.append(idx1)
    if idx2.size > 0:
        parts.append(idx2)
    combined = np.unique(np.concatenate(parts))
    if combined.size <= total:
        return combined
    scores = np.zeros(combined.size, dtype=np.float64)
    if idx1.size > 0:
        scores += np.asarray(arr1, dtype=np.float64)[combined]
    if idx2.size > 0:
        scores += np.asarray(arr2, dtype=np.float64)[combined]
    order = np.argsort(scores)[::-1][:total]
    return combined[order]


def eval_safe(a):
    try:
        arr = np.asarray(a, dtype=np.float64)
        if arr.ndim != 1 or arr.size == 0:
            return float('inf')
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = arr.sum()
        if s < 1e-15:
            return float('inf')
        arr = arr / s
        lst = [float(x) for x in arr]
        val = float(evaluate_sequence(lst))
        if np.isfinite(val):
            return val
        return float('inf')
    except Exception:
        return float('inf')


def multiplicative_update(x, G, eta):
    x = np.maximum(x, 1e-18)
    gmax = float(G.max())
    gmin = float(G.min())
    rng = gmax - gmin
    if rng <= 1e-300:
        return x.copy()
    Gn = (G - gmin) / rng
    logp = np.log(x) - eta * Gn
    logp -= logp.max()
    y = np.exp(logp)
    y = np.maximum(y, 1e-18)
    s = y.sum()
    if s <= 1e-300:
        return x.copy()
    return y / s


def smooth_uniform_small(a, r):
    a = normalize_pos(a)
    n = int(a.size)
    r = int(r)
    if n <= 1 or r <= 0:
        return a.copy()
    r = min(r, max(1, n // 3))
    k = np.ones(2 * r + 1, dtype=np.float64) / float(2 * r + 1)
    b = np.convolve(a, k, mode='same')
    return normalize_pos(b)


def global_transform_phase(a0, time_limit=60.0):
    a = normalize_pos(a0)
    n = int(a.size)
    best_m = conv_max_seq(a)
    t0 = time.monotonic()
    count = 0
    x = np.linspace(0.0, 1.0, n)

    while time.monotonic() - t0 < time_limit and count < 200:
        improved = False

        for theta in [-3.0, -1.5, -0.75, 0.75, 1.5, 3.0]:
            if time.monotonic() - t0 > time_limit or count >= 200:
                break
            b = normalize_pos(a * np.exp(theta * (x - 0.5)))
            m = conv_max_seq(b)
            count += 1
            if m < best_m - 1e-17:
                a = b.copy()
                best_m = m
                improved = True
                break
        if improved:
            continue

        for p in [0.5, 1.5, 2.0]:
            if time.monotonic() - t0 > time_limit or count >= 200:
                break
            b = normalize_pos(np.power(a, p))
            m = conv_max_seq(b)
            count += 1
            if m < best_m - 1e-17:
                a = b.copy()
                best_m = m
                improved = True
                break
        if improved:
            continue

        b = normalize_pos(0.5 * (a + a[::-1]))
        m = conv_max_seq(b)
        count += 1
        if m < best_m - 1e-17:
            a = b.copy()
            best_m = m
            improved = True
            continue

        for r in [2, 3, 5]:
            if time.monotonic() - t0 > time_limit or count >= 200:
                break
            b = smooth_uniform_small(a, r)
            m = conv_max_seq(b)
            count += 1
            if m < best_m - 1e-17:
                a = b.copy()
                best_m = m
                improved = True
                break

        if not improved:
            break

    return a, 2.0 * n * best_m


def coord_polish(a0, time_limit=120.0, seed=7, verbose=False):
    a = normalize_pos(a0)
    a = np.maximum(a, 1e-18)
    n = int(a.size)
    L = 2 * n - 1
    c = conv_full_seq(a)
    best_m = float(c.max())
    best_a = a.copy()
    best_score = 2.0 * n * best_m

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    rnd = 0
    topT = int(min(60, max(16, 0.0025 * n)))
    fracs = (0.001, 0.003, 0.01, 0.03, 0.08, 0.20, 0.50)
    tmp = np.empty(L, dtype=np.float64)

    while time.monotonic() - t0 < time_limit:
        rnd += 1
        if rnd % 100 == 0:
            c = conv_full_seq(a)
            m = float(c.max())
        else:
            m = float(c.max())

        if m < best_m - 1e-18:
            best_m = m
            best_a = a.copy()
            best_score = 2.0 * n * m
            stall = 0
        else:
            stall += 1
            if stall > 8000:
                break

        K = min(topT, L)
        if K <= 0:
            break
        part = np.argpartition(c, -K)[-K:]
        cpart = c[part]
        maxp = float(cpart.max())
        meanp = float(cpart.mean())
        spread = max(maxp - meanp, maxp * 1e-7, 1e-300)
        wt = np.exp(6.0 * (cpart - maxp) / spread)
        ws = wt.sum()
        if ws <= 0:
            wt = np.ones_like(wt) / float(wt.size)
        else:
            wt /= ws

        src_score = np.zeros(n, dtype=np.float64)
        for wi, k in zip(wt, part):
            k = int(k)
            i0 = max(0, k - n + 1)
            i1 = min(n - 1, k)
            if i0 > i1:
                continue
            seg = a[k - i0 : k - i1 - 1 : -1]
            src_score[i0:i1 + 1] += wi * seg * a[i0:i1 + 1]

        sources = []
        if n > 4:
            topsrc = np.argpartition(src_score, -4)[-4:]
            for z in topsrc:
                sources.append(int(z))
        else:
            sources.extend(range(n))

        k0 = int(part[int(np.argmax(cpart))])
        i0 = max(0, k0 - n + 1)
        i1 = min(n - 1, k0)
        idx = np.arange(i0, i1 + 1)
        if idx.size > 0:
            contrib = a[idx] * a[k0 - idx]
            s0 = int(idx[int(np.argmax(contrib))])
            sources.append(s0)

        improved = False
        used_sources = []

        for s in sources:
            s = int(s)
            if s in used_sources:
                continue
            used_sources.append(s)
            base = float(a[s])
            if base <= 1e-16:
                continue

            dest_score = np.zeros(n, dtype=np.float64)
            for wi, k in zip(wt, part):
                k = int(k)
                d0 = max(0, k - n + 1)
                d1 = min(n - 1, k)
                if d0 > d1:
                    continue
                seg = a[k - d0 : k - d1 - 1 : -1]
                dest_score[d0:d1 + 1] += wi * seg
            dest_score += 0.05 * a

            if n > 8:
                cand = np.argpartition(dest_score, 8)[:8]
            else:
                cand = np.arange(n)

            if stall > 300:
                half = max(1, n // 2)
                low = np.argpartition(dest_score, half)[:half]
                for _ in range(3):
                    if low.size > 0:
                        cand = np.concatenate([cand, [low[rng.integers(low.size)]]])

            cand = np.unique(cand.astype(int))

            for d in cand:
                d = int(d)
                if d == s:
                    continue
                for frac in fracs:
                    amt = base * frac
                    if amt < 1e-18:
                        continue
                    tmp[:] = c
                    tmp[d:d + n] += 2.0 * amt * a
                    tmp[s:s + n] -= 2.0 * amt * a
                    dd = 2 * d
                    if dd < L:
                        tmp[dd] += amt * amt
                    ss = 2 * s
                    if ss < L:
                        tmp[ss] -= amt * amt
                    sd = s + d
                    if sd < L:
                        tmp[sd] -= 2.0 * amt * amt
                    nm = float(tmp.max())
                    if nm < best_m - 1e-18:
                        a[s] -= amt
                        a[d] += amt
                        a = np.maximum(a, 0.0)
                        c = tmp.copy()
                        best_m = nm
                        best_a = a.copy()
                        best_score = 2.0 * n * nm
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            tries = 8 if stall > 100 else 3
            for _ in range(tries):
                s = int(rng.integers(n))
                if a[s] <= 1e-14:
                    continue
                d = int(rng.integers(n))
                if d == s:
                    continue
                frac = float(rng.uniform(0.001, 0.03))
                amt = float(a[s]) * frac
                tmp[:] = c
                tmp[d:d + n] += 2.0 * amt * a
                tmp[s:s + n] -= 2.0 * amt * a
                dd = 2 * d
                if dd < L:
                    tmp[dd] += amt * amt
                ss = 2 * s
                if ss < L:
                    tmp[ss] -= amt * amt
                sd = s + d
                if sd < L:
                    tmp[sd] -= 2.0 * amt * amt
                nm = float(tmp.max())
                if nm < best_m - 1e-18:
                    a[s] -= amt
                    a[d] += amt
                    a = np.maximum(a, 0.0)
                    c = tmp.copy()
                    best_m = nm
                    best_a = a.copy()
                    best_score = 2.0 * n * nm
                    improved = True
                    break

            if improved:
                continue

            if stall > 700:
                scale = 0.02 + 0.01 * float(stall % 3)
                noise = best_a * (scale * rng.standard_normal(n))
                a2 = best_a + noise
                a2 = normalize_pos(np.maximum(a2, 0.0))
                c2 = conv_full_seq(a2)
                m2 = float(c2.max())
                if m2 < best_m * 1.5:
                    a = a2
                    c = c2
                    stall = 0

        if verbose and rnd % 300 == 0:
            print("    coord round=" + str(rnd) +
                  " elapsed=" + str(round(time.monotonic() - t0, 1)) + "s" +
                  " J=" + repr(best_score) + " stall=" + str(stall))

    best_a = normalize_pos(best_a)
    sf, _ = score_fast(best_a)
    return best_a, sf


def build_lp_constraint_matrix(y, active_idx):
    y = np.asarray(y, dtype=np.float64)
    active_idx = np.asarray(active_idx, dtype=int).ravel()
    n = int(y.size)
    K = int(active_idx.size)
    if K == 0:
        return sp.csr_matrix((0, n))
    rowptr = np.empty(K + 1, dtype=np.int64)
    rowptr[0] = 0
    cols_list = []
    vals_list = []
    pos = 0
    for r in range(K):
        k = int(active_idx[r])
        i0 = max(0, k - n + 1)
        i1 = min(n - 1, k)
        cnt = 0
        if i0 <= i1:
            idx = np.arange(i0, i1 + 1, dtype=np.int32)
            seg = y[k - idx]
            cols_list.append(idx)
            vals_list.append(seg.astype(np.float64, copy=False))
            cnt = int(idx.size)
        rowptr[r + 1] = pos + cnt
        pos += cnt
    if pos == 0:
        return sp.csr_matrix((0, n))
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    return sp.csr_matrix((vals, cols, rowptr), shape=(K, n))


def lp_direction(y, active_idx):
    if not SCIPY_OK or sp is None or linprog is None:
        return None
    y = normalize_pos(y)
    active_idx = np.asarray(active_idx, dtype=int).ravel()
    n = int(y.size)
    K = int(active_idx.size)
    if K <= 0 or n <= 0:
        return None

    try:
        A = build_lp_constraint_matrix(y, active_idx)
        A_aug = sp.hstack([A, -sp.eye(K, format='csr')], format='csr')
        cvec = np.zeros(n + 1, dtype=np.float64)
        cvec[-1] = 1.0
        Aeq = sp.hstack(
            [sp.csr_matrix(np.ones((1, n), dtype=np.float64)),
             sp.csr_matrix(np.zeros((1, 1), dtype=np.float64))],
            format='csr'
        )
        b_eq = np.array([1.0], dtype=np.float64)
        b_ub = np.zeros(K, dtype=np.float64)
        bounds = [(0.0, None)] * n + [(None, None)]
        res = linprog(
            cvec,
            A_ub=A_aug,
            b_ub=b_ub,
            A_eq=Aeq,
            b_eq=b_eq,
            bounds=bounds,
            method='highs',
            options={'presolve': True}
        )
        if int(getattr(res, 'status', -1)) == 0:
            g = np.asarray(res.x[:n], dtype=np.float64)
            g = np.maximum(g, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s, float(res.fun)
    except Exception:
        pass
    return None

def conv_two_cached(Fu, v, size, L):
    V = np.fft.rfft(v, size)
    return np.fft.irfft(Fu * V, size)[:L]


def correlate_cached(Fy, w, size, L, n):
    wr = np.ascontiguousarray(w[::-1])
    Fw = np.fft.rfft(wr, size)
    c = np.fft.irfft(Fy * Fw, size)
    return c[L - n:L][::-1]


def inner_cross_direction(Fu_cross, Fu_grad, active_idx, g, size_cross, size_grad, n,

            spread = max(maxf - float(v.mean()), maxf * 1e-7, 1e-300)
            xf = beta_f * (v - maxf) / spread
            xf = np.clip(xf, -700.0, 0.0)
            wf = np.exp(xf)
            sf_ = wf.sum()
            if sf_ > 0:
                w += rho * wf / sf_

        ws = w.sum()
        if ws <= 0:
            w[int(np.argmax(v))] = 1.0
        else:
            w /= ws

        G = correlate_cached(Fu_grad, w, size_grad, L, n)
        gmax = float(G.max())
        gmin = float(G.min())
        grng = gmax - gmin
        if grng <= 1e-18:
            break
        Gn = (G - gmin) / grng

        base_g = cur_g.copy()
        cand_g = None
        cand_score = score
        cand_mf = mf
        cand_ma = ma

        for eta in eta_opts:
            gn = multiplicative_update(base_g, Gn, eta)
            vn = conv_two_cached(Fu_cross, gn, size_cross, L)
            nmf = float(vn.max())
            nma = float(vn[act].max()) if act.size > 0 else nmf
            nsc = nma + 0.75 * max(0.0, nmf - current_m)
            if (nsc < cand_score - 1e-18 or
                (abs(nsc - cand_score) <= 1e-18 and
                 (nmf < cand_mf - 1e-18 or
                  (abs(nmf - cand_mf) <= 1e-18 and nma < cand_ma - 1e-18)))):
                cand_g = gn
                cand_score = nsc
                cand_mf = nmf
                cand_ma = nma

        if cand_g is None:
            no_improve += 1
            if no_improve > 8:
                break
        else:
            cur_g = cand_g
            cur_mf = cand_mf
            cur_ma = cand_ma
            no_improve = 0

    return best_g, best_score


def mirror_phase(a0, time_limit=250.0, K=180, rho=0.15, beta_a=8.0, beta_f=3.0, verbose=True):
    a = normalize_pos(a0)
    n = int(a.size)
    best_score, c = score_fast(a)
    best_m = best_score / (2.0 * n)
    best_a = a.copy()
    g = a.copy()

    size_cross = 1 << (2 * n - 2).bit_length()
    size_grad = 1 << (3 * n - 2).bit_length()
    t0 = time.monotonic()
    stall = 0
    outer = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 35:
            break
        m = float(c.max())
        if m < best_m - 1e-18:
            best_m = m
            best_score = 2.0 * n * m
            best_a = a.copy()

        Fu_cross = np.fft.rfft(a, size_cross)
        Fu_grad = np.fft.rfft(a, size_grad)
        active_idx = top_indices(c, K)
        remaining = time_limit - (time.monotonic() - t0)
        inner_limit = min(45.0, max(5.0, remaining * 0.45))

        g_best, _ = inner_cross_direction(
            Fu_cross, Fu_grad, active_idx, g,
            size_cross, size_grad, n, inner_limit,
            rho=rho, beta_a=beta_a, beta_f=beta_f,
            eta_opts=(1.0, 0.4, 2.2), current_m=m
        )
        line_a, line_m = line_search_conv(a, g_best, max_evals=16)
        if line_m < best_m - 1e-15:
            a = line_a
            c = conv_full_seq(a)
            best_m = line_m
            best_score = 2.0 * n * best_m
            best_a = a.copy()
            g = normalize_pos(0.5 * g_best + 0.5 * a)
            stall = 0
            outer += 1
        else:
            g = normalize_pos(g_best)
            stall += 1
            if stall > 4:
                break

        if verbose and (outer > 0 and outer % 4 == 0):
            print("    mirror outer=" + str(outer) +
                  " elapsed=" + str(round(time.monotonic() - t0, 1)) + "s" +
                  " J=" + repr(best_score))

    best_a = normalize_pos(best_a)
    sf, _ = score_fast(best_a)
    return best_a, sf


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    total = min(float(budget_s) - 180.0, 1650.0)
    if total < 120.0:
        total = max(120.0, float(budget_s) - 60.0)

    MAX_N = 18000
    best = None

    try:
        p = height_sequence_1
        arr = np.asarray(p, dtype=np.float64)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.ndim == 1 and arr.size > 10 and arr.sum() > 0:
            best = normalize_pos(arr)
    except Exception:
        best = None

    if best is None:
        n = 8000
        x = np.linspace(0.0, 1.0, n)
        best = normalize_pos(np.sin(np.pi * x) ** 2 + 0.05)

    best_score, best_c = score_fast(best)
    best_direct = eval_safe(best)
    best_direct_a = best.copy()
    initial_best = best.copy()
    print("INIT n=" + str(len(best)) + " fast=" + repr(best_score) + " direct=" + repr(best_direct))

    n0 = int(best.size)
    lengths = []
    for L in [n0,
              max(6000, n0 // 2),
              min(MAX_N, int(n0 * 1.25)),
              min(MAX_N, int(n0 * 1.5)),
              min(MAX_N, 14000),
              min(MAX_N, 16000)]:
        L = int(L)
        if L >= 2000 and L not in lengths:
            lengths.append(L)

    for L in lengths:
        if time.monotonic() - t_global > total * 0.12:
            break
        a = interpolate_seq(best, L)
        s, _ = score_fast(a)
        if s < best_score:
            best_score = s
            best = a
            best_c = conv_full_seq(best)
    if best_score < best_direct:
        best_direct = best_score
        best_direct_a = best.copy()

    rem = total - (time.monotonic() - t_global)
    if rem > 40:
        a, s = global_transform_phase(best, min(45.0, rem * 0.08))
        if s < best_score:
            best_score = s
            best = a

    rem = total - (time.monotonic() - t_global)
    if SCIPY_OK and rem > 120:
        t_lp = min(450.0, rem * 0.35)
        if t_lp > 80:
            a, s = improve_with_lp(best, t_lp, K=220, verbose=True)
            if s < best_score:
                best_score = s
                best = a
            if time.monotonic() - t_global < total - 120:
                d = eval_safe(a)
                if np.isfinite(d) and d < best_direct:
                    best_direct = d
                    best_direct_a = a.copy()

    rem = total - (time.monotonic() - t_global)
    if rem > 100:
        t_c = min(360.0, rem * 0.25)
        a, s = coord_polish(best, t_c, seed=111, verbose=True)
        if s < best_score:
            best_score = s
            best = a
        if time.monotonic() - t_global < total - 120:
            d = eval_safe(a)
            if np.isfinite(d) and d < best_direct:
                best_direct = d
                best_direct_a = a.copy()

    rem = total - (time.monotonic() - t_global)
    if rem > 120:
        t_m = min(330.0, rem * 0.25)
        a, s = mirror_phase(best, t_m, K=180, rho=0.15, verbose=True)
        if s < best_score:
            best_score = s
            best = a
        if time.monotonic() - t_global < total - 120:
            d = eval_safe(a)
            if np.isfinite(d) and d < best_direct:
                best_direct = d
                best_direct_a = a.copy()

    rem = total - (time.monotonic() - t_global)
    if rem > 90:
        t_c2 = min(260.0, rem * 0.22)
        a, s = coord_polish(best, t_c2, seed=222, verbose=True)
        if s < best_score:
            best_score = s
            best = a

    rem = total - (time.monotonic() - t_global)
    if rem > 50:
        a, s = global_transform_phase(best, min(60.0, rem * 0.18))
        if s < best_score:
            best_score = s
            best = a

    rem = total - (time.monotonic() - t_global)
    if rem > 180 and int(best.size) < MAX_N:
        target = min(MAX_N, int(best.size) * 5 // 4 + 800)
        if target > int(best.size) + 400:
            a2 = interpolate_seq(best, target)
            a2, s2 = coord_polish(a2, min(180.0, rem * 0.30), seed=555, verbose=True)
            if s2 < best_score:
                best_score = s2
                best = a2

    rem = total - (time.monotonic() - t_global)
    if rem > 70:
        a, s = coord_polish(best, min(100.0, rem * 0.35), seed=777, verbose=False)
        if s < best_score:
            best_score = s
            best = a

    final_candidates = [best, best_direct_a.copy(), initial_best.copy()]
    b = normalize_pos(0.5 * (best + best[::-1]))
    final_candidates.append(b)
    x = np.linspace(0.0, 1.0, int(best.size))
    final_candidates.append(normalize_pos(best * np.exp(-1.0 * (x - 0.5))))
    final_candidates.append(normalize_pos(best * np.exp(1.0 * (x - 0.5))))

    best_final = best.copy()
    best_final_direct = best_direct
    seen = 0
    for a in final_candidates:
        if time.monotonic() - t_global > total - 45:
            break
        if a is None:
            continue
        a = normalize_pos(a)
        if a.size <= 1:
            continue
        seen += 1
        if seen > 5:
            break
        d = eval_safe(a)
        if np.isfinite(d) and d < best_final_direct:
            best_final_direct = d
            best_final = a.copy()

    if time.monotonic() - t_global > total - 25:
        final = best_final
    else:
        final = best_final

    final = np.asarray(final, dtype=np.float64)
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.clip(final, 0.0, 1000.0)
    final = normalize_pos(final)
    if final.size == 0:
        final = np.ones(1000) / 1000.0

    print("FINAL fast=" + repr(best_score) +
          " direct=" + repr(best_final_direct) +
          " n=" + str(final.size) +
          " elapsed=" + repr(time.monotonic() - t_global))
    return [float(v) for v in final]
