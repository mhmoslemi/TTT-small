- $\int_0^1 f(x)^2 dx$ (middle, $k=n/2$) to be small
- $f(0)^2$ and $f(1)^2$ (edges, $k=0, 2n-2$) to be small
- All intermediate lags to be similar

The "flatness" condition suggests $f$ should be chosen so that its autocorrelation is as flat as possible. This is the "minimum variance" or "low sidelobe" problem.

In the continuous setting, the function that minimizes the maximum of the autocorrelation (subject to $\int f = 1, f \geq 0$) is related to the solution of a variational problem. The uniform function $f = 1$ gives a constant autocorrelation of $\int_0^{1-s} 1 \cdot 1 dx = 1-s$, which decreases linearly. The max is 1 (at $s=0$).

But in the discrete setting, the edges matter. For $a_i = 1/n$:
$(a*a)_0 = 1/n^2$
$(a*a)_1 = 2/n^2$
...
$(a*a)_{n-1} = n/n^2 = 1/n$
...
$(a*a)_{2n-2} = 1/n^2$

So the max is $1/n$ at the middle. Objective = $2n \cdot 1/n = 2$.

import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix

# Constants
TIME_BUDGET = 1800.0

def fast_convolve(a):
    """FFT-based linear convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_sparse_C(a, active_set, n):
    """Build sparse constraint matrix for LP/QP."""
    K = len(active_set)
    rows = []
    cols = []
    vals = []
    for ii, k in enumerate(active_set):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        for j in range(j_start, j_end):
            rows.append(ii)
            cols.append(j)
            vals.append(a[k_int - j])
    return csr_matrix((vals, (rows, cols)), shape=(K, n))

def norm1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def eval_score(a):

    ac = fast_convolve(a)
    s = a.sum()
    if s < 0.01:

    return float(2.0 * len(a) * ac.max() / (s * s))

def mirror_descent(a0, time_limit, top_k=100, beta=5.0, c0=5.0, verbose=False):

    best_m = fast_convolve(a).max()
    t0 = time.monotonic()

    stall = 0
    rng = np.random.default_rng(42)

        if time.monotonic() - t0 > time_limit:
            break
        
        y = np.maximum(x + 0.5 * vel, 1e-15)
        y = y / y.sum()
        
        ac = fast_convolve(y)
        M = ac.max()
        
        # Soft gradient: weighted average of gradients over top-k lags
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w = w / w.sum()
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx[:30]):  # Limit to 30 for efficiency

            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            if j_start >= j_end:
                continue
            idx = np.arange(j_start, j_end)
            grad[idx] += 2.0 * wi * y[k_int - idx]

        log_x = np.log(y) - eta * grad
        max_log = log_x.max()
        a_new = np.exp(log_x - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        vel = 0.5 * vel + 0.5 * (a_new - x)
        x = a_new

            best_m = M_new
            best_a = x.copy()
            stall = 0

            stall += 1

            stall = 0
            noise = 0.03 / (1 + 0.05 * (t // 3000))
            x_r = best_a + rng.normal(0, noise, n) * best_a
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            vel = np.zeros(n)
            gamma = max(gamma * 0.9, 0.5)
