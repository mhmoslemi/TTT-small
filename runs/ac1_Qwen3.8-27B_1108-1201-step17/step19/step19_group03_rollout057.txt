def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def pocs_optimize(a0, time_limit, n_iters_max=300000, rng_seed=0):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    fft_size = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    A_spec = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
    A_full = np.zeros(fft_size, dtype=np.complex128)
    A_full[:fft_size // 2 + 1] = A_spec
    for i in range(1, fft_size // 2):
        A_full[fft_size - i] = np.conj(A_spec[i])
    a = np.fft.irfft(A_full, fft_size)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = a0.copy()
    else:
        a = a / s

    ac = fast_autocorr(a)
    M = ac.max()
    if M < best_m:
        best_m = M
        best_a = a.copy()

    for it in range(n_iters_max):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        target_mag = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
        alpha_blend = 0.05 + 0.6 * (it / min(it + 1, 200))
        alpha_blend = min(alpha_blend, 0.75)
        new_mag = (1 - alpha_blend) * mag + alpha_blend * target_mag
        A_new = new_mag * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:

        if M_new < best_m - 1e-15:

            best_a = a_new.copy()

def smooth_gradient_nesterov(a0, time_limit, verbose=False, tau_init=5.0, tau_final=200.0):
    """Log-sum-exp smoothed gradient with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    v = np.zeros(n)
    mu = 1.0  # momentum
    stall = 0
    max_stall = 5000
    it = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / min(time_limit, 400.0))
        tau = tau_init * (tau_final / tau_init) ** progress

        # Nesterov look-ahead
        y = x + mu * v

        ac = fast_autocorr(y)
        ac_max = ac.max()
        # Softmax weights
        shifted = tau * (ac - ac_max)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(ac))
            w[ac_max == ac].nonzero()[0][0] = 1.0
        else:
            w = w / w_sum

        # Gradient: sum_k w_k * dc_k/dx
        g = np.zeros(n)
        # Only use top weights for efficiency
        threshold = w.max() * 1e-4
        active = np.where(w > threshold)[0]
        for k in active:
            ki = int(k)
            # dc_k/dx[j] = 2*a[ki-j] for valid j
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            idx = np.arange(j_start, j_end)
            vals = 2.0 * y[ki - idx]
            g[idx] += w[k] * vals

        # Normalized gradient (tangent to simplex)
        g_tangent = g - g.mean()
        g_norm = np.linalg.norm(g_tangent)
        if g_norm < 1e-15:
            stall += 1
            continue

        # Step size with Nesterov
        eta = 1.0 / np.sqrt(it + 1)
        new_mu = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
        v = -eta * g_tangent / g_norm
        x_new = x + new_mu * v
        x_new = np.maximum(x_new, 1e-15)
        x_new = x_new / x_new.sum()

        M_new = fast_autocorr(x_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                # Restart with perturbation
                pert = np.random.default_rng(it).normal(0, 0.02, n) * best_a
                x_r = np.maximum(best_a + pert, 1e-10)
                x_r = x_r / x_r.sum()
                x = x_r
                v = np.zeros(n)
                mu = 1.0
                stall = 0

        x = x_new

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothNest it={it} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, stall={stall}")

    return best_a, best_m


def lp_frank_wolfe_line_search(a0, time_limit, solver, verbose=False, top_K_init=200):
    """LP Frank-Wolfe with proper line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    K = top_K_init
    K_max = min(3 * n, 2 * n - 1)
    iter_count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr, cp.sum(b_var) >= 0.01]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Line search
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = M_curr
        for alpha in [best_alpha, min(1.0, best_alpha * 2), min(1.0, best_alpha * 1.5),
                      min(1.0, best_alpha * 0.5), 0.3, 0.2, 0.1, 0.7, 0.5]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        iter_count += 1
        if iter_count % 8 == 0:
            K = min(K + 40, K_max)

        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def pair_coordinate_descent(a0, time_limit, verbose=False, rng_seed=100):
    """Coordinate descent optimizing pairs of coordinates."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200
    n_peaks = 15

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_peaks]

        # Compute sensitivity: how much each coordinate affects peak peaks
        sens = np.zeros(n)
        for k in top_k[:8]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[ki - idx]

        # Find high-sensitivity coordinates in peaks
        peak_coords = set()
        for k in top_k[:n_peaks]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            peak_coords.update(range(s, e + 1))
        peak_list = list(peak_coords)

        # Low-sensitivity coordinates (safe to increase)
        low_sens = np.argsort(sens)[:n // 4]

        improved = False

        # Try moving mass from high-sens peak coords to low-sens coords
        for src in peak_list[:min(30, len(peak_list))]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue

            for dst in low_sens[:5]:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        sens = np.zeros(n)
                        for k2 in top_k[:8]:
                            ki2 = int(k2)
                            s2 = max(0, ki2 - n + 1)
                            e2 = min(n - 1, ki2)
                            idx2 = np.arange(s2, e2 + 1)
                            sens[idx2] += a[ki2 - idx2]
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:

            J = 2.0 * n * best_m

    return best_a, best_m


def spectral_init_shape(n, n_modes=15, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_u_shape(n, sharpness=2.0):

def make_cosine_mod(n, alpha=0.5, freq=1):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        a_out, _ = lp_frank_wolfe_line_search(a_in, t3, solver, verbose=True, top_K_init=min(300, 2*n_best-1))
        try_update(a_out, "lp_main")

    # === Phase 4: Smooth gradient on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Smooth Gradient on best (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_gradient_nesterov(a_in, t4, tau_init=20.0, tau_final=500.0)
        try_update(a_out, "sn_refine")

    # === Phase 5: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW r2 (n={n_best}) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_line_search(a_in, t5, solver, verbose=True, top_K_init=min(500, 2*n_best-1))
        try_update(a_out, "lp_r2")

    # === Phase 6: Pair coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Pair Coord Descent (n={n_best}) ===")
    t6 = min(remaining * 0.12, 200.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = pair_coordinate_descent(a_in, t6, verbose=True, rng_seed=456)
        try_update(a_out, "pair_coord")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length Exploration ===")

    trial_lengths = []
    for delta in [300, -300, 600, -600, 1000, -1000, 1500, -1500, 2000]:
        new_n = n_best + delta
        if 1000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.35, 120.0)
        if t7a > 20:
            a_out, _ = smooth_gradient_nesterov(a_long, t7a, tau_init=5.0, tau_final=200.0)
            try_update(a_out, f"sn_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 6000:
            t7b = min(remaining * 0.4, 100.0)
            if t7b > 20:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                a_out, _ = lp_frank_wolfe_line_search(a_in, t7b, solver, verbose=False,
                                                       top_K_init=min(300, 2*new_n-1))
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 8: Final intensive optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Optimization (n={n_best}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.3, 100.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = smooth_gradient_nesterov(a_in, t8, tau_init=50.0, tau_final=800.0)
            try_update(a_out, "sn_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and n_best <= 6000:
            t8b = min(remaining * 0.4, 120.0)
            if t8b > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe_line_search(a_in, t8b, solver, verbose=False,
                                                       top_K_init=min(800, 2*n_best-1))
                try_update(a_out, "lp_final")

    # === Phase 9: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 9: Final Polish (n={n_best}) ===")
        t9 = min(remaining * 0.8, 80.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = pair_coordinate_descent(a_in, t9, verbose=False, rng_seed=789)
            try_update(a_out, "coord_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
