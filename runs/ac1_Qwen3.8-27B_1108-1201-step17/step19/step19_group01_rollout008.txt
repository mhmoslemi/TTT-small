```python
def spectral_equalize(a, alpha, iters=100):
    """
    Alternate between:
    1. In freq domain: flatten |A(f)|^2 by raising to power (1-alpha)
    2. In time domain: project onto a>=0, sum=1
    
    alpha is the flattening strength (0=no change, 1=fully flat)
    """
    n = len(a)
    fft_size = 1 << (2*n - 1).bit_length()
    
    for _ in range(iters):
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A)**2
        max_power = power.max()
        if max_power < 1e-30:
            break
        
        # Flatten: new_power = power^(1-alpha) * max_power^(alpha)
        # This reduces the dynamic range
        new_power = np.power(np.maximum(power, 1e-30), 1-alpha) * max_power**alpha
        new_amp = np.sqrt(new_power)
        
        # Keep the phase
        phase = np.angle(A)
        A_new = new_amp * np.exp(1j * phase)
        
        # Back to time domain
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new /= s
    
    return a_new
```

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-30:
        return np.ones_like(x) / len(x)
    return e / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad_and_val(a, beta, top_k=None):
    n = len(a)
    size = 2 * n - 1
    b = fast_autocorr(a)
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
    else:
        b_use = b.copy()
    b_max = b_use.max()
    if b_max < 1e-30:
        b_max = 0.0
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    s_exp = exp_vals.sum()
    if s_exp < 1e-30:
        p = np.ones(len(b_use)) / len(b_use)
    else:
        p = exp_vals / s_exp
    val = b_max + np.log(s_exp) / beta
    # Gradient: (1/beta) * conv(a, p) restricted to valid indices
    fft_size = 1 << (n + len(p) - 2).bit_length()
    Ar = np.fft.rfft(a[::-1], fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, p, val, b

        f0, g0 = obj_and_grad(x)

def spectral_shave(a0, time_limit, n_steps=2000, shave_frac=0.02, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    rng = np.random.default_rng(123)

    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        mean_power = power.mean()
        peak_power = power.max()

        if peak_power < mean_power * 1.01:
            if it > 100:
                break

        # Find top frequency bins and shave them
        n_shave = max(1, len(power) // 50)
        top_idx = np.argsort(power)[::-1][:n_shave]

        # Reduce peak bins towards mean
        for idx in top_idx:
            if power[idx] > mean_power:
                power[idx] = power[idx] * (1.0 - shave_frac) + mean_power * shave_frac

        # Keep DC (index 0) fixed to maintain sum
        power[0] = np.abs(A[0]) ** 2

        # Reconstruct
        new_amp = np.sqrt(np.maximum(power, 0))
        phase = np.angle(A)
        A_new = new_amp * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            shave_frac = max(0.005, shave_frac * 0.98)
        else:
            a = a + 0.005 * rng.normal(0, 1, n) * a
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral iter=" + str(it) + " (" + str(elapsed) + "s): J=" + str(J))

    return best_a, best_m


def lp_fw_improved(a0, time_limit, solver, initial_K=400, max_K=3000,
                   add_per_iter=30, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 1500

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: converged at iter " + str(t))
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        fft_size = 1 << (2 * n - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5),
                  max(0.02, default_alpha * 1.5), 1.0 / (t + 1)]
        alphas = [max(0.01, min(1.0, a)) for a in alphas]

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(elapsed) + "s): J=" + str(J) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    max_stall = 1200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(50, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:30]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            src_val = a[src]
            partner_val = a[partner] if 0 <= partner < n else 0.0

            if src_val >= partner_val:
                reduce_idx = src
            else:
                reduce_idx = partner

            if a[reduce_idx] < 1e-12:
                continue

            # Find low-pressure destination
            pressure = np.zeros(n)
            for k2 in top_k[:15]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:15]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)

            cand_pressure = pressure[candidates]
            dst = int(candidates[np.argmin(cand_pressure)])
            if dst == reduce_idx:
                mask = candidates != reduce_idx
                if mask.sum() > 0:
                    dst = int(candidates[mask][np.argmin(cand_pressure[mask])])
                else:
                    continue

            found = False
            for frac in [0.01, 0.03, 0.06, 0.12, 0.25]:
                amt = a[reduce_idx] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 300:
                j = int(rng.integers(n))
                if j != reduce_idx and a[reduce_idx] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[reduce_idx] * frac
                    a[reduce_idx] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 200 == 0:
                for _ in range(8):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and (stall % 600 == 0):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (" + str(elapsed) + "s): J=" + str(J) + " stall=" + str(stall))

    return best_a, best_m


def nesterov_projected(a0, time_limit, beta_start=30.0, beta_end=1000.0,
                       beta_growth=1.005, eta_base=0.7, top_k_ratio=0.05,
                       restart_every=2500, verbose=True):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(99)
    top_k = max(20, int(n * top_k_ratio))

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, val, b = compute_smoothed_grad_and_val(y, beta, top_k)
        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.12)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 60 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.015 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 5.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest restart t=" + str(t) + " (" + str(elapsed) + "s): J=" + str(J))

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(elapsed) + "s): J=" + str(J) + ", beta=" + str(beta))

    return best_a, best_max_b


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(elapsed) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Time budget: " + str(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + str(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + str(best_score))

    n0 = len(best_seq)

    # Phase 1: L-BFGS continuation (primary optimizer)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 1: L-BFGS (n=" + str(n_cur) + ") ===")
    t1 = min(remaining * 0.22, 380.0)
    if t1 > 60:
        try:
            a_out, _ = lbfgs_continuation(norm_sum1(best_seq), t1,
                                          beta_schedule=[30, 60, 120, 250, 500, 1000, 2000],
                                          top_k_ratio=0.06, ms=20, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs_n" + str(n_cur), t_global)
        except Exception as e:
            print("  L-BFGS phase error: " + str(e))

    # Phase 2: Spectral shaving (new technique)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 2: Spectral (n=" + str(n_cur) + ") ===")
    t2 = min(remaining * 0.10, 150.0)
    if t2 > 30:
        try:
            a_out, _ = spectral_shave(norm_sum1(best_seq), t2,
                                      n_steps=3000, shave_frac=0.03, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "spectral_n" + str(n_cur), t_global)
        except Exception as e:
            print("  Spectral phase error: " + str(e))

    # Phase 3: L-BFGS again (benefiting from spectral step)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 3: L-BFGS again (n=" + str(n_cur) + ") ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 40:
        try:
            a_out, _ = lbfgs_continuation(norm_sum1(best_seq), t3,
                                          beta_schedule=[100, 300, 600, 1200, 2500, 5000],
                                          top_k_ratio=0.05, ms=20, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs2_n" + str(n_cur), t_global)
        except Exception as e:
            print("  L-BFGS2 phase error: " + str(e))

    # Phase 4: LP-Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 4: LP-FW (n=" + str(n_cur) + ") ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 40 and n_cur <= 8000:
        try:
            a_out, _ = lp_fw_improved(norm_sum1(best_seq), t4, solver,
                                      initial_K=500, max_K=3500,
                                      add_per_iter=40, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(n_cur), t_global)
        except Exception as e:
            print("  LP-FW phase error: " + str(e))

    # Phase 5: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 5: Coord (n=" + str(n_cur) + ") ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        try:
            a_out, _ = coordinate_polish(norm_sum1(best_seq), t5, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n" + str(n_cur), t_global)
        except Exception as e:
            print("  Coord phase error: " + str(e))

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    longer_lengths = [n_cur + 500, n_cur + 1000, n_cur + 2000]
    longer_lengths = [l for l in longer_lengths if l >= 4000 and l <= 10000]

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print("\n=== Phase 6: Scale n=" + str(new_n) + " ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.35, 200.0)
        if t6a > 40:
            try:
                a_out, _ = lbfgs_continuation(a_long, t6a,
                                              beta_schedule=[50, 150, 400, 1000, 2000],
                                              top_k_ratio=0.05, verbose=True)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs_n" + str(new_n), t_global)
            except Exception as e:
                print("  Scale L-BFGS error: " + str(e))

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 8000:
            t6b = min(remaining * 0.40, 200.0)
            if t6b > 30:
                try:
                    a_long2 = interpolate_sequence(best_seq, new_n)
                    a_out, _ = lp_fw_improved(norm_sum1(a_long2), t6b, solver,
                                              initial_K=400, max_K=3000,
                                              add_per_iter=35, verbose=True)
                    best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(new_n), t_global)
                except Exception as e:
                    print("  Scale LP error: " + str(e))

    # Phase 7: Nesterov (different from L-BFGS, may find different basin)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 7: Nesterov (n=" + str(n_cur) + ") ===")
    t7 = min(remaining * 0.18, 250.0)
    if t7 > 40:
        try:
            a_out, _ = nesterov_projected(norm_sum1(best_seq), t7,
                                          beta_start=50.0, beta_end=2000.0,
                                          beta_growth=1.006, eta_base=0.6,
                                          top_k_ratio=0.04, restart_every=2000,
                                          verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n" + str(n_cur), t_global)
        except Exception as e:
            print("  Nesterov phase error: " + str(e))

    # Phase 8: Final L-BFGS polish with very high beta
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 8: Final L-BFGS (n=" + str(n_cur) + ") ===")
    t8 = min(remaining * 0.25, 300.0)
    if t8 > 30:
        try:
            a_out, _ = lbfgs_continuation(norm_sum1(best_seq), t8,
                                          beta_schedule=[500, 1500, 4000, 8000, 15000],
                                          top_k_ratio=0.03, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs_final_n" + str(n_cur), t_global)
        except Exception as e:
            print("  Final L-BFGS error: " + str(e))

    # Phase 9: Final coordinate
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 9: Final coord (n=" + str(n_cur) + ") ===")
    t9 = min(remaining * 0.30, 200.0)
    if t9 > 20:
        try:
            a_out, _ = coordinate_polish(norm_sum1(best_seq), t9, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)
        except Exception as e:
            print("  Final coord error: " + str(e))

    # Final LP burst if time
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 50 and n_cur <= 8000:
        print("\n=== Phase 10: Final LP burst ===")
        t10 = min(remaining * 0.70, 200.0)
        if t10 > 20:
            try:
                a_out, _ = lp_fw_improved(norm_sum1(best_seq), t10, solver,
                                          initial_K=600, max_K=4000,
                                          add_per_iter=50, verbose=True)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_burst", t_global)
            except Exception as e:
                print("  LP burst error: " + str(e))

    # Validation
    if best_seq is not None:
        best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
        s = best_seq.sum()
        if s > 0.01:
            best_seq = best_seq / s
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + str(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(elapsed) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    # Ensure valid output
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    s_final = best_seq.sum()
    if s_final < 0.01:
        best_seq = np.ones(5000) / 5000.0
    else:
        best_seq = best_seq / best_seq.sum()

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
