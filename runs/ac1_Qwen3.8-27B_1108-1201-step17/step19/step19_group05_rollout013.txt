    A = np.fft.rfft(a, fft_size)

- a[j] -> a[j] + delta

The change in conv[m] is:
For lag = m - (n-1):
delta_conv[m] = delta * (a_old[i-lag] - a_old[j-lag]) + delta * (a_old[i+lag] - a_old[j+lag])
              = delta * (a[i-lag] + a[i+lag] - a[j-lag] - a[j+lag])

(where out-of-bounds = 0, and we use the old values)

So the new conv at the peak lag k_peak changes by:
delta * (a[i-k] + a[i+k] - a[j-k] - a[j+k])

If this is negative, the peak decreases. So we want to find (i, j) such that:
a[i-k] + a[i+k] < a[j-k] + a[j+k]

This means position j has more "neighborhood mass" at distance k than position i does. So we should move mass from i to j.

But we also need to check that no OTHER lag increases above the current maximum. This is the tricky part.

For a practical approach, I'll:
1. Compute the gradient at the current point
2. Use the gradient to identify good (i, j) pairs
3. Try the move and check if it actually improves

Let me now write the full algorithm. I'll aim for something that's different from the parent but builds on the same fundamental ideas:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def make_ramp(n, alpha=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** alpha
    return a / a.sum()

def make_bell(n, sigma_frac=0.3):
    x = np.arange(n, dtype=np.float64) - n/2
    a = np.exp(-x**2 / (2 * (sigma_frac * n)**2))
    return a / a.sum()

def make_two_ramp(n):
    half = n // 2
    a = np.zeros(n)
    a[:half] = np.arange(1, half+1)
    a[half:] = np.arange(half, 0, -1)
    if n % 2 == 0:
        a[half:] = np.arange(half-1, 0, -1)
    s = a.sum()
    if s > 0:
        a /= s
    return a

def compute_subgradient(a, C, n_lags=None, temperature=0.01):
    """Subgradient of max(C) with softmax over top lags."""
    n = len(a)
    if n_lags is None:
        n_lags = max(20, len(C) // 10)
    top_idx = np.argpartition(C, -n_lags)[-n_lags:]
    vals = C[top_idx]
    max_v = vals.max()
    if max_v < 1e-15:
        return np.zeros(n)
    w = np.exp((vals - max_v) / (max_v * temperature))
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx_i, m in enumerate(top_idx):
        mi = int(m)
        wi = w[idx_i]
        if wi < 1e-15:
            continue
        lag = mi - (n - 1)
        if lag == 0:
            grad += 2.0 * wi * a
        else:
            lag_abs = abs(lag)
            if lag_abs < n:
                grad[lag_abs:] += wi * a[:n - lag_abs]
                grad[:n - lag_abs] += wi * a[lag_abs:]
    
    return grad

def nesterov_subgradient(a0, time_limit, temp0=0.1, temp_min=0.005,
                          n_lags_frac=0.1, lr0=1.0, verbose=False):
    """Nesterov-accelerated subgradient descent on simplex."""
    n = len(a0)
    a = np.maximum(a0, 1e-30).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    C = fast_autocorr(a)
    best_m = C.max()
    t0 = time.time()
    
    x = a.copy()
    x_prev = a.copy()
    t = 0
    stall = 0
    
    while time.time() - t0 < time_limit:
        t += 1
        # Nesterov extrapolation
        beta = (t - 1.0) / (t + 2.0) if t > 1 else 0.0
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-20)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        C_y = fast_autocorr(y)
        
        # Adaptive temperature
        temp = max(temp_min, temp0 * (temp_min / temp0) ** (t / 5000.0))
        
        g = compute_subgradient(y, C_y, n_lags=int(len(C_y) * n_lags_frac), temperature=temp)
        
        # Step size with decay
        eta = lr0 / (np.sqrt(t) + 1.0)
        
        # Mirror step
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Check improvement
        C_x = fast_autocorr(x)
        M = min(C_y.max(), C_x.max())
        if M < best_m - 1e-18:
            best_m = M
            best_a = (x if C_x.max() <= C_y.max() else y).copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > 2000:
            stall = 0
            rng = np.random.default_rng(t)
            pert = rng.normal(0, 0.05, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-15)
            x = x / x.sum()
            x_prev = x.copy()
        
        if verbose and t % 5000 == 0:
            print(f"    NS t={t}: J={2*n*best_m:.6f}, stall={stall}")
    
    return best_a, best_m

def spectral_equalize(a0, time_limit, iters_per_sec=200, verbose=False):
    """Spectral domain flattening with soft non-negativity."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_autocorr(a).max()
    a = a / s
    best_a = a.copy()
    C = fast_autocorr(a)
    best_m = C.max()
    t0 = time.time()
    fft_size = 1 << (2*n - 1).bit_length()
    
    it = 0
    while time.time() - t0 < time_limit:
        it += 1
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_min = P.min()
        P_mean = P.mean()
        
        if P_max < 1e-30:
            break
        
        # Flatten spectrum: reduce peaks, boost valleys
        alpha = 0.1 + 0.3 * min(1.0, it / 1000.0)
        P_new = P.copy()
        # Reduce peaks
        mask_high = P > P_mean
        P_new[mask_high] = P[mask_high] - alpha * (P[mask_high] - P_mean)
        # Boost valleys
        mask_low = ~mask_high
        P_new[mask_low] = P[mask_low] + alpha * (P_mean - P[mask_low]) * 0.5
        P_new = np.maximum(P_new, 1e-30)
        
        mag_scale = np.sqrt(P_new / np.maximum(P, 1e-30))
        mag_scale = np.clip(mag_scale, 0.7, 1.5)
        A_new = A * mag_scale
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Soft non-negativity: use softplus-like projection
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        C_new = fast_autocorr(a_new)
        M_new = C_new.max()
        
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend
            blend = 0.1
            a = (1 - blend) * a + blend * a_new
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            C_curr = fast_autocorr(a)
            if C_curr.max() < best_m:
                best_m = C_curr.max()
                best_a = a.copy()
        
        if verbose and it % 500 == 0:
            print(f"    SE it={it}: J={2*n*best_m:.6f}")
    
    return best_a, best_m

def pairwise_local_search(a0, time_limit, verbose=False, rng_seed=42):
    """Exact pairwise swap local search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    C = fast_autocorr(a)
    best_m = C.max()
    t0 = time.time()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.time() - t0 < time_limit and stall < 50:
        C = fast_autocorr(a)
        m_max = C.max()
        peak_idx = np.argmax(C)
        peak_lag = peak_idx - (n - 1)
        
        # Compute "contribution score" for each position
        # How much each position contributes to the peak
        contrib = np.zeros(n)
        if peak_lag == 0:
            contrib = 2.0 * a
        else:
            lag_abs = abs(peak_lag)
            if lag_abs < n:
                contrib[lag_abs:] = a[:n - lag_abs]
                contrib[:n - lag_abs] = a[lag_abs:]
        
        # Also consider nearby peaks
        top_ks = np.argpartition(C, -10)[-10:]
        for k2 in top_ks:
            k2i = int(k2)
            if k2i == peak_idx:
                continue
            lag2 = abs(k2i - (n - 1))
            if lag2 < n and lag2 > 0:
                contrib[lag2:] += 0.5 * a[:n - lag2]
                contrib[:n - lag2] += 0.5 * a[lag2:]
        
        # Try to move mass from high-contrib to low-contrib position
        # Sort by contribution
        sorted_idx = np.argsort(contrib)
        
        improved = False
        for attempt in range(20):
            # Pick source (high contrib) and destination (low contrib)
            src = int(sorted_idx[-1 - rng.integers(0, min(10, n//10))])
            dst = int(sorted_idx[rng.integers(0, min(10, n//10))])
            
            if src == dst or a[src] < 1e-15:
                continue
            
            # Try different fractions
            for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
                delta = a[src] * frac
                if delta < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= delta
                a_trial[dst] += delta
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial /= s_t
                
                C_t = fast_autocorr(a_trial)
                M_t = C_t.max()
                
                if M_t < best_m - 1e-18:
                    a = a_trial
                    best_m = M_t
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                stall = 0
                break
        
        if not improved:
            # Random perturbation
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                delta = a[i] * rng.uniform(0.01, 0.05)
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                C_new = fast_autocorr(a)
                if C_new.max() < best_m:
                    best_m = C_new.max()
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 10 == 0:
            print(f"    LS: J={2*n*best_m:.6f}, stall={stall}")
    
    return best_a, best_m
