The sum distribution of two discrete variables is a convolution. We want to make this as "flat" as possible (minimize the peak).

For continuous variables, the sum of two iid variables has a smoother distribution. The analog of "flat" for continuous is uniform, which is achieved by triangular + triangular or similar constructions.

            active_set = sorted([active_set[i] for i in keep])
            if peak_pos not in active_set:
                active_set.append(peak_pos)
                active_set.sort()

        K = len(active_set)
        if K == 0:
            break

        coeffs = build_constraint_matrix_vec(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail_count += 1
            if lp_fail_count > 5:
                # Reset active set
                active_set = sorted(np.argsort(ac)[::-1][:initial_K].tolist())
                lp_fail_count = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail_count += 1
            if lp_fail_count > 10:
                active_set = sorted(np.argsort(ac)[::-1][:initial_K].tolist())
                lp_fail_count = 0
            continue
        lp_fail_count = 0

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find newly violated constraints
        conv_ys = fast_cross_conv(y, g_dir)
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)

        # Adaptive line search
        t_iter = time.monotonic() - t0
        remaining_frac = max(0.01, (time_limit - t_iter) / time_limit)
        base_alpha = 2.0 / (iter_count + 2.0)
        alphas = [base_alpha, base_alpha * 0.5, base_alpha * 1.5,
                  1.0 / (iter_count + 1.0), 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
        alphas = [max(min_alpha, min(1.0, a)) for a in alphas]
        alphas = list(dict.fromkeys(alphas))  # deduplicate preserving order

        best_alpha = base_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 3:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y / s_y

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s):"
                  + " J=" + repr(J) + ", |S|=" + str(len(active_set))
                  + ", alpha=" + repr(round(best_alpha, 4)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", iters=" + str(iter_count))
    return best_a, best_m


def peak_local_search(a0, time_limit, verbose=True, t_start=None):
    """
    Peak-aware coordinate descent: directly reduces the maximum convolution value
    by transferring mass from peak-contributing elements to low-impact positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = t_start if t_start is not None else time.monotonic()
    rng = np.random.default_rng(42)

    stall = 0
    max_stall = 500
    total_moves = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        M_curr = ac.max()
        peak_pos = int(np.argmax(ac))

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
            total_moves += 1
        else:
            stall += 1

        # Identify peak contributors
        k = peak_pos
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        peak_idx = np.arange(i_start, i_end)
        if len(peak_idx) == 0:
            break
        peak_contrib = a[peak_idx] * a[k - peak_idx]

        # Top contributors to the peak
        n_top = min(15, len(peak_idx))
        top_srcs = peak_idx[np.argsort(peak_contrib)[::-1][:n_top]]

        # Compute impact scores for all positions based on top convolution peaks
        top_peaks = np.argsort(ac)[::-1][:20]
        impact = np.zeros(n)
        for k2 in top_peaks:
            k2i = int(k2)
            s2 = max(0, k2i - n + 1)
            e2 = min(n - 1, k2i)
            if s2 <= e2:
                idx2 = np.arange(s2, e2 + 1)
                impact[idx2] += a[k2i - idx2]
                impact[k2i - idx2] += a[idx2]

        found = False
        for src in top_srcs:
            if time.monotonic() - t0 > time_limit:
                break
            src_val = a[src]
            if src_val < 1e-13:
                continue

            # Find good destinations: low impact, not in peak window
            dest_scores = impact.copy()
            dest_scores[peak_idx] += M_curr * 0.05  # penalize peak window
            dest_scores[src] = np.inf

            # Try a few best destinations with various transfer amounts
            n_candidates = min(10, n)
            best_dsts = np.argsort(dest_scores)[:n_candidates]

            for dst in best_dsts:
                if dst == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = src_val * frac
                    if amt < 1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_convolve(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        found = True
                        break
                if found:
                    break
            if found:
                break

        if not found:
            stall += 1
            if stall > 50:
                # Random perturbation for diversity
                n_pert = rng.integers(2, 15)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.008)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t

        if verbose and stall % 300 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSearch (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", stall=" + str(stall))

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print("    PeakSearch done (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", moves=" + str(total_moves))
    return best_a, best_m


def mirror_descent_refine(a0, time_limit, top_k=100, beta=4.0, c0=5.0,
                          restart_every=3000, t_start=None, verbose=False):
    """Soft mirror descent with adaptive parameters for fine refinement."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = t_start if t_start is not None else time.monotonic()
    rng = np.random.default_rng(7)

    x = a.copy()
    gamma = c0
    stall = 0
    restart_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_convolve(x)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * x[k_int - idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(x, 1e-30)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = x.copy()
        else:
            a_new = a_new / s_new
        x = a_new

        M_new = fast_convolve(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            restart_count += 1
            noise_level = 0.03 / (1.0 + 0.15 * restart_count)
            pert = rng.normal(0, noise_level, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r > 1e-15:
                x = x_r / s_r
            else:
                x = best_a.copy()
            cur_top_k = min(cur_top_k + 10, 180)
            cur_beta = max(cur_beta - 0.3, 2.0)
            gamma = max(gamma * 0.92, 0.5)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    MD restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print("    MirrorDescent done (" + str(round(elapsed, 1)) + "s): J=" + repr(J))
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search: multi-phase LP-FW + peak search + mirror descent refinement."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print("Total time budget: " + repr(time_budget) + "s")

    best_seq, best_score = get_best(height_sequence_1, t_global)
    n0 = len(best_seq)

    # Phase 1: Main LP-FW optimization
    elapsed = time.monotonic() - t_global
    t1 = min((time_budget - elapsed) * 0.35, 550.0)
    print("\n=== Phase 1: LP-FW (n=" + str(n0) + ", t=" + str(round(t1)) + "s) ===")
    if t1 > 30 and n0 <= 9000:
        a_out, _ = lp_fw_improved(best_seq, t1, solver,
                                  initial_K=500, max_K=4000, add_per_iter=50,
                                  verbose=True, min_alpha=0.02)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1", t_global)

    # Phase 2: Peak local search
    elapsed = time.monotonic() - t_global
    t2 = min((time_budget - elapsed) * 0.12, 180.0)
    print("\n=== Phase 2: PeakSearch (n=" + str(len(best_seq)) + ", t=" + str(round(t2)) + "s) ===")
    if t2 > 20:
        a_out, _ = peak_local_search(best_seq, t2, verbose=True, t_start=t_global)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak1", t_global)

    # Phase 3: Second LP-FW with fresh active set
    elapsed = time.monotonic() - t_global
    t3 = min((time_budget - elapsed) * 0.25, 400.0)
    print("\n=== Phase 3: LP-FW (n=" + str(len(best_seq)) + ", t=" + str(round(t3)) + "s) ===")
    if t3 > 30 and len(best_seq) <= 9000:
        a_out, _ = lp_fw_improved(best_seq, t3, solver,
                                  initial_K=600, max_K=4500, add_per_iter=60,
                                  verbose=True, min_alpha=0.015)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2", t_global)

    # Phase 4: Mirror descent refinement
    elapsed = time.monotonic() - t_global
    t4 = min((time_budget - elapsed) * 0.10, 150.0)
    print("\n=== Phase 4: MirrorDescent (n=" + str(len(best_seq)) + ", t=" + str(round(t4)) + "s) ===")
    if t4 > 20:
        a_out, _ = mirror_descent_refine(best_seq, t4, top_k=100, beta=4.0,
                                         c0=5.0, restart_every=2500,
                                         t_start=t_global, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "md1", t_global)

    # Phase 5: Try nearby n values
    n_best = len(best_seq)
    n_candidates = [n_best - 300, n_best - 100, n_best + 200, n_best + 500]
    n_candidates = [x for x in n_candidates if 3000 <= x <= 10000]
    # Remove duplicates while preserving order
    seen = set()
    n_unique = []
    for x in n_candidates:
        if x not in seen:
            seen.add(x)
            n_unique.append(x)
    n_candidates = n_unique

    for new_n in n_candidates:
        elapsed = time.monotonic() - t_global
        remaining = time_budget - elapsed
        if remaining < 120:
            break
        print("\n=== Phase 5: n=" + str(new_n) + " (remaining=" + str(round(remaining)) + "s) ===")
        a_long = interpolate_seq(best_seq, new_n)

        t5a = min(remaining * 0.40, 250.0)
        if t5a > 30 and new_n <= 9000:
            a_out, _ = lp_fw_improved(a_long, t5a, solver,
                                      initial_K=500, max_K=4000, add_per_iter=50,
                                      verbose=False, min_alpha=0.02)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

        elapsed = time.monotonic() - t_global
        remaining = time_budget - elapsed
        if remaining > 60:
            t5b = min(remaining * 0.35, 120.0)
            if t5b > 15:
                a_out, _ = peak_local_search(best_seq, t5b, verbose=False, t_start=t_global)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_n" + str(new_n), t_global)

    # Phase 6: Final LP-FW push
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    n_best = len(best_seq)
    print("\n=== Phase 6: Final LP-FW (n=" + str(n_best) + ", remaining=" + str(round(remaining)) + "s) ===")
    if remaining > 60 and n_best <= 9000:
        t6 = min(remaining * 0.45, 350.0)
        if t6 > 30:
            a_out, _ = lp_fw_improved(best_seq, t6, solver,
                                      initial_K=700, max_K=5000, add_per_iter=70,
                                      verbose=True, min_alpha=0.01)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # Phase 7: Final peak search polish
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    print("\n=== Phase 7: Final PeakSearch (remaining=" + str(round(remaining)) + "s) ===")
    if remaining > 30:
        t7 = min(remaining * 0.70, 200.0)
        if t7 > 15:
            a_out, _ = peak_local_search(best_seq, t7, verbose=True, t_start=t_global)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_final", t_global)

    # Phase 8: Final mirror descent
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    if remaining > 30:
        print("\n=== Phase 8: Final MD (remaining=" + str(round(remaining)) + "s) ===")
        t8 = min(remaining * 0.60, 100.0)
        if t8 > 10:
            a_out, _ = mirror_descent_refine(best_seq, t8, top_k=120, beta=3.5,
                                             c0=4.0, restart_every=2000,
                                             t_start=t_global, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "md_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(x ** 1.5)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
