    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Project onto probability simplex (non-negative, sum=1)."""
    a = np.maximum(a, 0.0)
    s = a.sum()

        return np.ones_like(a) / len(a)
    return a / s

def compute_agraph(f, beta, top_k):
    """
    Compute LSE-smoothed autocorrelation gradient.
    Returns (lse_value, gradient).
    """
    n = len(f)
    c = fast_convolve(f)
    total_len = len(c)
    
    if top_k < total_len:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    else:
        top_idx = np.arange(total_len)
    
    c_sub = c[top_idx]
    max_c = c_sub.max()
    
    # Compute softmax weights
    shifted = beta * (c_sub - max_c)
    shifted -= shifted.max()  # for numerical stability (should be <= 0)
    exp_vals = np.exp(shifted)
    exp_sum = exp_vals.sum()
    w = exp_vals / exp_sum
    
    # LSE value
    lse_val = (max_c + np.log(exp_sum)) / beta
    
    # Gradient: 2 * sum_k w_k * f_{k-j}
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        wi = w[i]
        if wi < 1e-12:
            continue
        k = int(top_idx[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        # f[k-j] for j in [js, je] = f[k-je], ..., f[k-js]
        grad[js:je+1] += 2.0 * wi * f[k-je:k-js+1]
    
    return lse_val, grad

def build_constraint_matrix_vec(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw(a0, time_limit, solver, initial_K=400, max_K=3000, add_per_iter=40, verbose=True):

        ac = fast_convolve(y)
        M_curr = ac.max()

        else:
            no_improve += 1
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep = np.argsort(margins)[:max_K]
            active_set = sorted([active_set[i] for i in keep])
        K = len(active_set)
        if K == 0:
            break
        coeffs = build_constraint_matrix_vec(y, active_set, n)
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()

        prob = cp.Problem(cp.Minimize(M_var), constraints)

            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = sorted(active_set[:initial_K])
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        default_alpha = 2.0 / (iter_count + 2.0)

            max(0.01, min(1.0, default_alpha)),

        if not improved:
            stall += 1
            if stall % 50 == 0:
                for _ in range(3):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.01)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (%.1fs): J=%.6f, stall=%d" % (elapsed, J, stall))
    if verbose:
        J = 2.0 * n * best_m
        print("    Coord done (%.1fs): J=%.6f" % (time.monotonic()-t0, J))
    return best_a, best_m


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def get_best(parent, t_global):
    best_seq = None
    best_score = float('inf')
    try:
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                best_seq = p_norm
                best_score = sc
                print("Parent: n=%d, score=%.6f" % (len(p), sc))
    except Exception:
        pass
    if best_seq is None:
        n = 5000
        x = np.linspace(0, 1, n)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=%d, score=%.6f" % (len(best_seq), best_score))
    return best_seq, best_score


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [%s]: %.6f (n=%d, t=%.1fs)" % (label, sc, len(a), elapsed))
    return best_seq, best_score


def parametric_quick(n, time_limit, best_init, best_score):
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score
    candidates = []
    for p in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(normalize((i_arr + 1.0) ** p))
        candidates.append(normalize((n - i_arr) ** p))
    for c in [0.2, 0.5, 0.8]:
        for sigma in [0.1, 0.2, 0.3, 0.5]:
            candidates.append(normalize(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))))
    for p in [0.5, 1.0, 2.0, 4.0]:
        candidates.append(normalize(x ** p + (1.0 - x) ** p))
    for freq in [1, 2, 3]:
        for ph in [0.0, np.pi]:
            a = np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x + ph)) / 2.0, 0.01)
            candidates.append(normalize(a))
    # Sine-based shapes
    for a_coeff in [0.1, 0.2, 0.3, 0.5]:
        candidates.append(normalize(np.sin(np.pi * x) ** 2 + a_coeff * np.sin(2 * np.pi * x) ** 2))
    for a_coeff in [0.1, 0.2, 0.3]:
        candidates.append(normalize(np.sin(np.pi * x) + a_coeff * np.cos(2 * np.pi * x)))
    for a in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        v = fast_convolve(a).max()
        J = 2.0 * n * v
        if J < best_val:
            best_val = J
            best_a = a.copy()
            print("    Param: J=%.6f" % best_val)
    return best_a, best_val


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Total time budget: %.1fs" % time_budget)

    best_seq, best_score = get_best(height_sequence_1, t_global)
    n0 = len(best_seq)

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # Phase 1: Quick parametric search
    print("\n=== Phase 1: Parametric ===")
    t1 = min(remaining_time() * 0.03, 30.0)
    if t1 > 5:
        a_param, _ = parametric_quick(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "param", t_global)

    # Phase 2: Main LSE gradient descent (long)
    remaining = remaining_time()
    print("\n=== Phase 2: LSE Descent (n=%d, %.0fs) ===" % (n0, remaining * 0.45))
    t2 = min(remaining * 0.45, 800.0)
    if t2 > 30:
        a_out, _ = lse_descent(normalize(best_seq), t2, top_k=80,
                               beta_start=3.0, beta_end=60.0,
                               c0=3.0, momentum=0.75,
                               restart_every=5000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lse1_n%d" % n0, t_global)

    # Phase 3: LP-FW polish
    remaining = remaining_time()
    n_best = len(best_seq)
    print("\n=== Phase 3: LP-FW (n=%d, %.0fs) ===" % (n_best, min(remaining * 0.2, 300.0)))
    t3 = min(remaining * 0.2, 300.0)
    if t3 > 30 and n_best <= 10000:
        a_out, _ = lp_fw(normalize(best_seq), t3, solver,
                         initial_K=500, max_K=4000, add_per_iter=50, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1_n%d" % n_best, t_global)

    # Phase 4: Second LSE pass with different params
    remaining = remaining_time()
    n_best = len(best_seq)
    print("\n=== Phase 4: LSE Descent v2 (n=%d, %.0fs) ===" % (n_best, min(remaining * 0.2, 350.0)))
    t4 = min(remaining * 0.2, 350.0)
    if t4 > 30:
        a_out, _ = lse_descent(normalize(best_seq), t4, top_k=100,
                               beta_start=5.0, beta_end=100.0,
                               c0=2.0, momentum=0.8,
                               restart_every=6000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lse2_n%d" % n_best, t_global)

    # Phase 5: Try different length
    remaining = remaining_time()
    n_best = len(best_seq)
    try_ns = [n_best + 500, n_best - 300, n_best + 1000]
    try_ns = [l for l in try_ns if 3000 <= l <= 10000]
    for new_n in try_ns[:2]:
        remaining = remaining_time()
        if remaining < 200:
            break
        print("\n=== Phase 5: Scale n=%d ===" % new_n)
        a_long = interpolate_seq(best_seq, new_n)
        t5a = min(remaining * 0.3, 200.0)
        if t5a > 30 and new_n <= 10000:
            a_out, _ = lse_descent(a_long, t5a, top_k=80,
                                   beta_start=3.0, beta_end=50.0,
                                   c0=3.0, momentum=0.75,
                                   restart_every=4000, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lse_n%d" % new_n, t_global)
        remaining = remaining_time()
        if remaining > 60 and new_n <= 8000:
            t5b = min(remaining * 0.3, 120.0)
            if t5b > 30:
                a_out, _ = lp_fw(normalize(best_seq), t5b, solver,
                                 initial_K=400, max_K=3000, add_per_iter=40, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n%d" % new_n, t_global)

    # Phase 6: Final LP-FW
    remaining = remaining_time()
    n_best = len(best_seq)
    print("\n=== Phase 6: Final LP-FW (n=%d, %.0fs) ===" % (n_best, min(remaining * 0.3, 300.0)))
    if remaining > 60 and n_best <= 10000:
        t6 = min(remaining * 0.3, 300.0)
        if t6 > 30:
            a_out, _ = lp_fw(normalize(best_seq), t6, solver,
                             initial_K=600, max_K=4500, add_per_iter=60, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # Phase 7: Coordinate polish
    remaining = remaining_time()
    n_best = len(best_seq)
    print("\n=== Phase 7: Coord Polish (n=%d, %.0fs) ===" % (n_best, min(remaining * 0.5, 200.0)))
    if remaining > 30:
        t7 = min(remaining * 0.5, 200.0)
        if t7 > 15:
            a_out, _ = coordinate_polish(normalize(best_seq), t7, rng_seed=42, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Phase 8: Final LSE micro-refinement
    remaining = remaining_time()
    n_best = len(best_seq)
    if remaining > 30:
        print("\n=== Phase 8: Final LSE micro (n=%d, %.0fs) ===" % (n_best, min(remaining * 0.6, 150.0)))
        t8 = min(remaining * 0.6, 150.0)
        if t8 > 20:
            a_out, _ = lse_descent(normalize(best_seq), t8, top_k=60,
                                   beta_start=20.0, beta_end=200.0,
                                   c0=1.5, momentum=0.85,
                                   restart_every=3000, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lse_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=%.6f, n=%d, time=%.1fs" % (best_score, len(best_seq), elapsed))
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=%.6f" % best_score)

    return [float(x) for x in best_seq]
