```python
def subgrad_descent(a, iterations, eta=0.1):
    n = len(a)
    a = a / a.sum()  # normalize to sum 1
    
    for it in range(iterations):
        b = convolve(a, a)
        k_star = argmax(b)
        
        # Compute gradient of b[k_star] w.r.t. a
        g = zeros(n)
        for i in range(n):
            j1 = k_star - i
            j2 = i - k_star
            if 0 <= j1 < n:
                g[i] += a[j1]
            if 0 <= j2 < n:
                g[i] += a[j2]
        
        # Update
        a = a - eta * g
        a = maximum(a, 0)
        a = a / a.sum()
    
    return a
```

```python
    # First term: a[k-i] for i where 0 <= k-i < n, i.e., k-n+1 <= i <= k
    i_lo = max(0, k - n + 1)

        g[idx] += a[k - idx]
    # Second term: a[i-k] for i where 0 <= i-k < n, i.e., k <= i <= k+n-1
    i_lo2 = max(0, k)
    i_hi2 = min(n - 1, k + n - 1)
    if i_lo2 <= i_hi2:
        idx2 = np.arange(i_lo2, i_hi2 + 1)
        g[idx2] += a[idx2 - k]
    return g
```

```python
def softmax_grad_topk(a, beta=10, topk=20):
    n = len(a)
    b = np.convolve(a, a)
    top_idx = np.argsort(b)[::-1][:topk]
    top_vals = b[top_idx]
    w = np.exp(beta * (top_vals - top_vals.max()))
    w /= w.sum()
    
    g = np.zeros(n)
    for k_val, w_val in zip(top_idx, w):
        if w_val > 1e-8:
            g += w_val * grad_bk(a, int(k_val))
    return g
```

This is O(topk * n) per iteration, which is fine.
