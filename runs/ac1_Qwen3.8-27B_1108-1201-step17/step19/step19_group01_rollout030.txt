def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_lse_objective_and_grad(a, beta, top_k=None):
    """Compute log-sum-exp smoothed objective and gradient."""
    n = len(a)
    b = fast_autocorr(a)
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    obj = (b_max + np.log(exp_vals.sum())) / beta
    # Gradient via FFT
    a_rev = a[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return obj, grad, b

def peak_shaving(a0, time_limit, verbose=True):

        # Find the top contributing pairs to the peak

        if total_contrib < 1e-20:
            stall += 100
            continue

                    pressure[idx2] += a[kp_int - idx2]

            # Find low-pressure destination (gateway)
            exclude = set()

            if len(candidates) < 10:
                candidates = np.arange(n)
                candidates = candidates[candidates != reduce_idx]

                a_trial = np.maximum(a_trial, 0.0)

                if s_t < 1e-15:

                a_trial = a_trial / s_t

                    a = a_trial
                    best_m = M_new
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if found:

        if not improved:

            if stall > 200:

                    i = int(rng.integers(n))
                    j = int(rng.integers(n))

                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)

            margins = np.array([M_curr - ac[k] for k in active_set if 0 <= k < len(ac)])
            active_list = sorted(active_set)
            margins_full = np.array([M_curr - ac[k] if 0 <= k < len(ac) else 1.0 for k in active_list])
            keep_thresh = np.percentile(margins_full, 70)
            active_set = set(k for k, m in zip(active_list, margins_full) if m <= max(keep_thresh, 1e-6))

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])
            constraint_list = sorted(active_set)
            K = len(constraint_list)
            if K == 0:
                break

        # Build constraint matrix (sparse-friendly)
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K:
                active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:40]]
            for k in top_v:
                active_set.add(int(k))

        # Line search with more aggressive alpha
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, 0.5, 0.3, 0.2, 1.0 / (t + 1), 2.0 / (t + 1)]
        alphas = sorted(set([max(0.01, min(1.0, a)) for a in alphas]), reverse=True)

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 150 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW2 iter=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def spectral_init(n, rng_seed=42):
    """Generate spectrally-motivated initial sequence."""
    rng = np.random.default_rng(rng_seed)
    x = np.linspace(0.0, 1.0, n)
    candidates = []
    # Monotone increasing (exploits asymmetry)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        candidates.append((x ** p + 0.1).astype(np.float64))
        candidates.append(((1.0 - x) ** p + 0.1).astype(np.float64))
    # quadratic
    for a in [0.5, 1.0, 2.0]:
        candidates.append(x ** 2 + a * x + 0.1)
    # Exponential
    for rate in [1.0, 2.0, 3.0, 5.0]:
        candidates.append(np.exp(rate * x))
        candidates.append(np.exp(-rate * x))
    # mixed
    candidates.append(np.exp(2.0 * x) * (1.0 - x) + 0.01)
    candidates.append(np.exp(3.0 * x) * (1.0 - 0.5 * x) + 0.01)
    candidates.append(x * np.exp(x) + 0.01)

    best = None
    best_m = float('inf')
    for a_raw in candidates:
        a = norm_sum1(a_raw)
        m = fast_autocorr(a).max()
        if m < best_m:
            best_m = m
            best = a.copy()

    # Random
    for _ in range(30):
        a_raw = rng.random(n) ** rng.uniform(0.5, 4.0)
        if rng.random() < 0.5:
            a_raw = a_raw[::-1]
        a = norm_sum1(a_raw)
        m = fast_autocorr(a).max()
        if m < best_m:
            best_m = m
            best = a.copy()

    return best, best_m


def multi_scale_optimize(base_seq, time_limit, n_base, verbose=True):
    """Try multiple scales around the base length."""
    t0 = time.monotonic()
    best_a = base_seq.copy()
    best_m = fast_autocorr(base_seq).max()
    n = len(base_seq)

    scales = [1.0, 0.9, 1.1, 0.8, 1.2]
    for scale in scales:
        if time.monotonic() - t0 > time_limit - 60:
            break
        new_n = int(n * scale)
        if new_n < 1000 or new_n > 12000 or new_n == n:
            continue
        a_init = interpolate_sequence(base_seq, new_n)
        # Quick L-BFGS
        remaining = time_limit - (time.monotonic() - t0)
        phase_t = min(remaining * 0.3, 120.0)
        if phase_t < 30:
            continue
        a_out, m_out = lbfgs_optimize(a_init, phase_t, beta_start=20.0, beta_end=400.0,
                                       beta_growth=1.005, top_k_ratio=0.05, verbose=False)
        J = 2.0 * new_n * m_out
        J_best = 2.0 * n * best_m
        if J < J_best:
            best_a = a_out.copy()
            best_m = m_out
            n = new_n
            if verbose:
                print("    MultiScale: n=" + str(new_n) + " J=" + repr(J))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(elapsed) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search: L-BFGS + peak shaving + LP-FW with multi-scale exploration."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6000
        a_init, _ = spectral_init(n_base)
        best_seq = a_init
        best_score = evaluate_safe(best_seq)
        print("Init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: Quick spectral/parametric init (only if parent is bad)
    if best_score > 1.52:
        remaining = time_budget - (time.monotonic() - t_global)
        t1 = min(remaining * 0.05, 40.0)
        if t1 > 10:
            a_param, _ = spectral_init(n0, rng_seed=123)
            best_seq, best_score = try_update(best_seq, best_score, a_param, "spectral", t_global)

    # Phase 2: L-BFGS (main optimizer, fast convergence)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: L-BFGS (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 60:
        a_out, _ = lbfgs_optimize(norm_sum1(best_seq), t2,
                                  beta_start=20.0, beta_end=600.0,
                                  beta_growth=1.005, top_k_ratio=0.05,
                                  verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs_n" + str(n_best), t_global)

    # Phase 3: Peak shaving (targeted local improvement)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Peak Shaving (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_out, _ = peak_shaving(norm_sum1(best_seq), t3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_n" + str(n_best), t_global)

    # Phase 4: LP-FW (structured improvement)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.20, 280.0)
    if t4 > 40 and n_best <= 8000:
        a_out, _ = lp_fw_v2(norm_sum1(best_seq), t4, solver,
                            initial_K=500, max_K=2500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(n_best), t_global)

    # Phase 5: Second L-BFGS pass with higher smoothing
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: L-BFGS pass 2 (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30:
        a_out, _ = lbfgs_optimize(norm_sum1(best_seq), t5,
                                  beta_start=50.0, beta_end=1200.0,
                                  beta_growth=1.006, top_k_ratio=0.03,
                                  verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs2_n" + str(n_best), t_global)

    # Phase 6: Multi-scale exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-Scale (base n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 60:
        a_out, _ = multi_scale_optimize(norm_sum1(best_seq), t6, n_best, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "multiscale", t_global)

    # Phase 7: Peak shaving on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Peak Shaving final (n=" + str(n_best) + ") ===")
    t7 = min(remaining * 0.12, 180.0)
    if t7 > 20:
        a_out, _ = peak_shaving(norm_sum1(best_seq), t7, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_final", t_global)

    # Phase 8: Final LP burst
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40 and n_best <= 8000:
        print("\n=== Phase 8: Final LP burst ===")
        t8 = min(remaining * 0.50, 200.0)
        if t8 > 20:
            a_out, _ = lp_fw_v2(norm_sum1(best_seq), t8, solver,
                                initial_K=600, max_K=3000, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_burst", t_global)

    # Phase 9: Final L-BFGS polish with very high smoothing
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print("\n=== Phase 9: Final L-BFGS polish ===")
        t9 = min(remaining * 0.60, 150.0)
        if t9 > 15:
            a_out, _ = lbfgs_optimize(norm_sum1(best_seq), t9,
                                      beta_start=100.0, beta_end=2000.0,
                                      beta_growth=1.008, top_k_ratio=0.02,
                                      verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs_polish", t_global)

    # Phase 10: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print("\n=== Phase 10: Final peak shave ===")
        t10 = min(remaining * 0.70, 100.0)
        if t10 > 10:
            a_out, _ = peak_shaving(norm_sum1(best_seq), t10, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_last", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(elapsed) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.exp(2.0 * x))
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
