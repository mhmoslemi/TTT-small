def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def safe_eval(a):
    """Safely compute the objective value."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]

def compute_all_pressure(a, conv, threshold_ratio=1.05):
    """Compute pressure vector using all peaks above threshold."""
    n = len(a)
    avg = (a.sum() ** 2) / (2 * n - 1)
    threshold = avg * threshold_ratio
    significant = np.where(conv > threshold)[0]
    
    pressure = np.zeros(n)
    for k in significant:
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        partner = ki - idx
        pressure[idx] += a[partner]
    return pressure, len(significant)


def find_best_pair_move(a, pressure, n_top_src=50, n_top_dst=50):
    """Find the best (source, dest) pair for a direct move."""
    n = len(a)
    # Source: high pressure AND has mass
    src_score = pressure * np.maximum(a, 0)
    # Dest: low pressure
    dst_score = pressure
    
    src_candidates = np.argsort(-src_score)[:n_top_src]
    dst_candidates = np.argsort(dst_score)[:n_top_dst]
    
    best_pair = None
    best_gap = 0
    for i in src_candidates:
        if a[i] < 1e-12:
            continue
        for j in dst_candidates:
            if i == j:
                continue
            gap = pressure[i] - pressure[j]
            if gap > best_gap:
                best_gap = gap
                best_pair = (i, j)
    return best_pair


def mirror_descent_step(a, conv, lr, threshold_ratio=1.05):
    """One mirror descent step using all-peak pressure."""
    pressure, n_peaks = compute_all_pressure(a, conv, threshold_ratio)
    if n_peaks == 0:
        return a.copy(), 0
    
    log_a = np.log(np.maximum(a, 1e-30)) - lr * pressure
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), n_peaks
    a_new = a_new / s
    return a_new, n_peaks


def direct_move_step(a, conv, frac_options=None, threshold_ratio=1.05):
    """Try direct pairwise moves."""
    if frac_options is None:
        frac_options = [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]
    
    n = len(a)
    pressure, _ = compute_all_pressure(a, conv, threshold_ratio)
    best_pair = find_best_pair_move(a, pressure)
    
    if best_pair is None:
        return a.copy()
    
    i, j = best_pair
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    for frac in frac_options:
        amt = a[i] * frac
        if amt < 1e-15:

        a_trial = a.copy()
        a_trial[i] -= amt
        a_trial[j] += amt
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:

        a_trial = a_trial / s
        M = fast_autocorr(a_trial).max()
        if M < best_m:
            best_m = M
            best_a = a_trial.copy()
    
    return best_a


def spectral_flatten_step(a, alpha=0.5):
    """One spectral flattening step."""
    n = len(a)
    fft_size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.abs(A) ** 2
    P_mean = P.mean()
    P_new = P + alpha * (P_mean - P)
    P_new = np.maximum(P_new, 1e-30)
    mag_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
    mag_ratio = np.clip(mag_ratio, 0.3, 3.0)
    A_new = A * mag_ratio
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def make_tapered(n, center=0.5, width=0.3, edge_decay=2.0):
    """Create a tapered sequence centered at 'center' with given width."""
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((x - center) / width) ** 2)
    # Edge decay
    edge = np.minimum(x, 1 - x)
    a *= edge ** (edge_decay * 0.3)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_twomax(n, pos1=0.25, pos2=0.75, width=0.15):
    """Two-Gaussian sequence."""
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((x - pos1) / width) ** 2) + np.exp(-((x - pos2) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_antisymmetric_tilt(n, strength=0.3):
    """Sequence tilted to be larger on one side (breaks symmetry to reduce center peak)."""
    x = np.arange(n, dtype=np.float64) / n
    a = 1.0 + strength * (2 * x - 1)  # linear tilt
    a = np.maximum(a, 0.01)
    s = a.sum()
    return a / s


def make_cosine_modulated(n, freq=1.0, depth=0.3):
    """Cosine-modulated sequence."""
    x = np.arange(n, dtype=np.float64) / n
    a = 1.0 - depth * np.cos(2 * np.pi * freq * x)
    a = np.maximum(a, 0.01)
    s = a.sum()
    return a / s


def main_optimize(a0, budget_s, verbose=False):
    """Main optimization loop combining multiple strategies."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    # Adaptive parameters
    lr = 0.5
    stall = 0
    strategy_cycle = 0
    n_strategies = 4  # md, direct, spectral, anneal
    
    iteration = 0
    while time.monotonic() - t0 < budget_s:
        iteration += 1
        elapsed = time.monotonic() - t0
        
        # Compute current convolution
        conv = fast_autocorr(a)
        M_curr = conv.max()
        
        # Update best
        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
            if verbose and iteration % 100 == 0:
                print(f"    it={iteration} ({elapsed:.0f}s): J={2*n*best_m:.6f}")
        else:
            stall += 1
        
        # Choose strategy based on stall count
        if stall > 500:
            strategy = 'anneal'
        elif stall > 200:
            strategy = 'spectral'
        elif stall > 50:
            strategy = strategy_cycle % n_strategies
            strategy_cycle += 1
        else:
            strategy = strategy_cycle % n_strategies
            strategy_cycle += 1
        
        if strategy == 0:  # Mirror descent
            lr_adaptive = lr / (1.0 + stall * 0.01)
            a_new, n_peaks = mirror_descent_step(a, conv, lr_adaptive)
            M_new = fast_autocorr(a_new).max()
            if M_new < best_m:
                a = a_new
            else:
                # Blend
                a = 0.7 * a + 0.3 * a_new
                a = np.maximum(a, 0)
                a = a / a.sum()
            lr = max(0.05, lr * 0.999)
        
        elif strategy == 1:  # Direct move
            a_new = direct_move_step(a, conv)
            M_new = fast_autocorr(a_new).max()
            if M_new < best_m:
                a = a_new
        
        elif strategy == 2:  # Spectral
            alpha = 0.3 + 0.4 * min(1.0, iteration / 500.0)
            a_new = spectral_flatten_step(a, alpha)
            M_new = fast_autocorr(a_new).max()
            if M_new < best_m:
                a = a_new
            else:
                blend = 0.2
                a = (1 - blend) * a + blend * a_new
                a = np.maximum(a, 0)
                a = a / a.sum()
        
        elif strategy == 3:  # Random perturbation (annealing)
            temperature = max(0.001, 0.05 / (1.0 + iteration * 0.001))
            a_pert = a + rng.normal(0, temperature, n) * a
            a_pert = np.maximum(a_pert, 0)
            s = a_pert.sum()
            if s > 1e-15:
                a_pert = a_pert / s
                M_pert = fast_autocorr(a_pert).max()
                # Accept if better or with probability based on temperature
                if M_pert < best_m or rng.random() < math.exp(-(M_pert - best_m) / (temperature * best_m + 1e-10)):
                    a = a_pert
        
        elif strategy == 'anneal':  # Strong annealing / restart
            strength = min(0.3, 0.05 + 0.02 * (stall // 500))
            a_pert = best_a + rng.normal(0, strength, n) * best_a
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / a_pert.sum()
            a = a_pert
            stall = 0
        
        # Periodic: try multiple direct moves in sequence
        if iteration % 200 == 0 and time.monotonic() - t0 < budget_s - 5:
            for _ in range(5):
                if time.monotonic() - t0 >= budget_s - 5:
                    break
                conv_c = fast_autocorr(a)
                a_try = direct_move_step(a, conv_c, [0.03, 0.07, 0.15])
                M_try = fast_autocorr(a_try).max()
                if M_try < fast_autocorr(a).max():
                    a = a_try
    
    return best_a, best_m


def main_optimize_v2(a0, budget_s, verbose=False):
    """Enhanced optimization with Nesterov acceleration and multi-scale approach."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    # Nesterov momentum
    x = a.copy()
    x_prev = a.copy()
    lr = 1.0
    stall = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > budget_s:
            break
        
        beta = (t - 1.0) / (t + 2.0)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-20)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        conv = fast_autocorr(y)
        pressure, n_sig = compute_all_pressure(y, conv, threshold_ratio=1.02)
        
        # Adaptive learning rate
        eta = lr / (math.sqrt(t) + 1.0)
        
        log_y = np.log(y) - eta * pressure
        max_log = log_y.max()
        a_new = np.exp(log_y - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall = 0
            if verbose and (t + 1) % 500 == 0:
                print(f"    Nesterov t={t} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")
        else:
            stall += 1
        
        # Adaptive restart
        if stall > 2000:
            stall = 0
            strength = min(0.15, 0.03 + 0.01 * (stall // 2000))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-15)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
        
        # Slowly decay learning rate
        lr = max(0.1, lr * 0.9999)
    
    return best_a, best_m


def multi_length_search(a_best, budget_s, verbose=False):
    """Try interpolating to different lengths and optimizing."""
    n_best = len(a_best)
    results = []
    
    candidate_ns = [n_best, n_best + 500, n_best + 1000, n_best - 500, n_best + 2000]
    candidate_ns = [n for n in candidate_ns if 1000 < n < 12000]
    
    for new_n in candidate_ns:
        if time.monotonic() - budget_s[0] > budget_s[1]:
            break
        # Interpolate
        old_x = np.linspace(0, 1, n_best)
        new_x = np.linspace(0, 1, new_n)
        a_interp = np.interp(new_x, old_x, a_best / a_best.sum())
        a_interp = np.maximum(a_interp, 0)
        s = a_interp.sum()
        if s < 1e-15:
            continue
        a_interp = a_interp / s
        
        sc = safe_eval(a_interp)
        if np.isfinite(sc):
            results.append((sc, a_interp, new_n))
            if verbose:
                print(f"    Interp n={new_n}: J={sc:.6f}")
    
    return results


def propose_candidate(budget_s=1800.0):
    """Main search function combining multiple optimization strategies."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = safe_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = safe_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Initialize with parent
    if parent is not None:
        a_init = parent / parent.sum()
        best_seq = a_init.copy()
        best_score = safe_eval(a_init)
    else:
        n = 6200
        a_init = np.ones(n) / n
        best_score = safe_eval(a_init)
        best_seq = a_init.copy()
    
    # Phase 1: Nesterov-accelerated mirror descent (main workhorse)
    print(f"\n=== Phase 1: Nesterov MD (budget={time_budget * 0.45:.0f}s) ===")
    t1_budget = time_budget * 0.45
    a_out, m_out = main_optimize_v2(a_init, t1_budget, verbose=True)
    try_update(a_out, "nesterov_md")
    
    # Phase 2: Multi-strategy local search on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Multi-strategy (budget={remaining * 0.30:.0f}s) ===")
    if remaining > 30 and best_seq is not None:
        t2_budget = remaining * 0.30
        a_in = best_seq / best_seq.sum()
        a_out, m_out = main_optimize(a_in, t2_budget, verbose=True)
        try_update(a_out, "multi_strategy")
    
    # Phase 3: Try alternative initializations and optimize
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Alternative initializations ===")
    if remaining > 60 and best_seq is not None:
        n_best = len(best_seq)
        
        alternatives = []
        # Tilted version of best
        x = np.arange(n_best, dtype=np.float64) / n_best
        tilt = 1.0 + 0.2 * (2 * x - 1)
        a_tilted = best_seq * tilt
        a_tilted = np.maximum(a_tilted, 0)
        a_tilted = a_tilted / a_tilted.sum()
        alternatives.append((a_tilted, "tilted"))
        
        # Cosine modulated
        a_cos = best_seq * (1.0 - 0.15 * np.cos(2 * np.pi * x))
        a_cos = np.maximum(a_cos, 0)
        a_cos = a_cos / a_cos.sum()
        alternatives.append((a_cos, "cosmod"))
        
        # Slightly different length
        for delta_n in [-500, 500, 1000]:
            new_n = n_best + delta_n
            if new_n < 2000 or new_n > 12000:
                continue
            old_x = np.linspace(0, 1, n_best)
            new_x = np.linspace(0, 1, new_n)
            a_interp = np.interp(new_x, old_x, best_seq / best_seq.sum())
            a_interp = np.maximum(a_interp, 0)
            a_interp = a_interp / a_interp.sum()
            alternatives.append((a_interp, f"n{new_n}"))
        
        for a_alt, label in alternatives:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            sc = safe_eval(a_alt)
            print(f"  Alt [{label}]: init J={sc:.6f}")
            try_update(a_alt, f"alt_{label}_init")
            
            # Optimize the alternative
            t_alt = min(remaining * 0.4, 120.0)
            if t_alt > 20:
                a_out, _ = main_optimize_v2(a_alt, t_alt, verbose=False)
                try_update(a_out, f"alt_{label}_optim")
    
    # Phase 4: Intensive direct moves on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Intensive direct moves (budget={remaining * 0.15:.0f}s) ===")
    if remaining > 20 and best_seq is not None:
        t4_budget = remaining * 0.15
        a_in = best_seq / best_seq.sum()
        a_out, m_out = main_optimize(a_in, t4_budget, verbose=True)
        try_update(a_out, "direct_intensive")
    
    # Phase 5: Final Nesterov polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Final polish (budget={remaining * 0.10:.0f}s) ===")
    if remaining > 15 and best_seq is not None:
        t5_budget = remaining * 0.10
        a_in = best_seq / best_seq.sum()
        a_out, _ = main_optimize_v2(a_in, t5_budget, verbose=True)
        try_update(a_out, "final_polish")
    
    # Final validation
    if best_seq is not None:
        fs = safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        n = 6200
        best_seq = np.ones(n) / n
        best_score = safe_eval(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
