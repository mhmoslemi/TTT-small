def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_convolve_xy(x, y):
    n = len(x)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    X = np.fft.rfft(x, fft_size)
    Y = np.fft.rfft(y, fft_size)
    return np.fft.irfft(X * Y, fft_size)[:size]

def subgradient_refine(a0, time_limit, verbose=True):
    """
    Subgradient method with Nesterov acceleration for the non-smooth objective max(conv(a,a)).
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    x = a.copy()
    y = a.copy()  # Nesterov momentum point
    vel = np.zeros(n)
    
    # Adaptive parameters
    top_k_start = 50
    top_k_max = 200
    beta = 3.0  # Sharpening parameter for soft max
    c0 = 3.0
    momentum = 0.8
    restart_every = 3000
    stall = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        y = x + momentum * vel
        y = np.maximum(y, 1e-10)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        ac = fast_convolve(y)
        M_curr = ac.max()
        
        # Compute soft gradient: weighted combination of subgradients at top peaks
        cur_top_k = min(top_k_start + t // 1000, top_k_max)
        top_idx = np.argsort(ac)[::-1][:cur_top_k]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(w)) / len(w)
        else:
            w = w / w_sum
        
        # Subgradient of conv[k] w.r.t. a is: 2*a[k-i] for valid i
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += 2.0 * wi * a[k_int - idx]
        
        eta = c0 / (np.sqrt(t) + 1.0)
        
        # Euclidean projected gradient step
        a_trial = x - eta * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            a_trial = x.copy()
        else:
            a_trial = a_trial / s_t
        
        # Line search: try a few step sizes
        best_step = eta
        best_M = fast_convolve(a_trial).max()
        for mult in [0.5, 2.0, 4.0]:
            eta_try = eta * mult
            a_try = x - eta_try * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            M_try = fast_convolve(a_try).max()
            if M_try < best_M:
                best_M = M_try
                best_step = eta_try
                a_trial = a_try
        
        vel = 0.9 * vel + 0.1 * (a_trial - x)
        x = a_trial
        
        M_new = fast_convolve(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > restart_every:
            stall = 0
            noise = 0.03 / (1.0 + 0.2 * (stall // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            s_x = x.sum()
            if s_x > 1e-15:
                x = x / s_x
            else:
                x = best_a.copy()
            vel = np.zeros(n)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    SGD restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))
    
    return best_a, best_m
