def fast_autocorr(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv(a, b):
    """Fast convolution of two sequences via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

def build_conv_matrix_row(a, k):
    """Build row k of the convolution matrix: T[k,i] = a[k-i] if 0<=k-i<n else 0."""
    n = len(a)
    row = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    for i in range(i_start, i_end):
        row[i] = a[k - i]
    return row


def build_conv_matrix(a, k_indices):
    """Build rows k_indices of the convolution matrix."""
    n = len(a)
    rows = []
    for k in k_indices:
        rows.append(build_conv_matrix_row(a, int(k)))
    return np.array(rows) if rows else None


def lp_solve_fw(a, M, k_indices, solver, b_init=None):
    """Solve: max sum(b) s.t. conv(a,b)[k] <= M for k in k_indices, b >= 0.
    Returns (b_opt, dual_values) or (None, None) on failure."""
    n = len(a)
    K = len(k_indices)
    
    if K == 0:
        return None, None
    
    T = build_conv_matrix(a, k_indices)
    if T is None:
        return None, None
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    
    # Add a sum constraint to prevent unboundedness
    # Actually, the problem is bounded because T has positive entries
    # and b >= 0, T@b <= M implies sum(b) <= M / min(T)
    # But let's add a upper bound for numerical safety
    constraints.append(cp.sum(b_var) <= M / (T.min() + 1e-30) * 10)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    
    # Extract dual values
    dual_vals = None
    try:
        if prob.dual_values is not None:
            dual_vals = np.asarray(prob.dual_values, dtype=np.float64)
            # First constraint is T@b <= M, dual should have K entries
            if dual_vals.shape[0] >= K:
                dual_vals = dual_vals[:K]
    except Exception:
        pass
    
    return b_val, dual_vals

        block_end = min(block_start + block_size, max_k)
        block = np.arange(block_start, block_end)
        if block_start + block_size > max_k:
            block = np.concatenate([block, np.arange(0, block_start + block_size - max_k)])
        
        k_indices = list(set(list(top_k) + list(block)))
        k_indices = k_indices[:K]
        
        b_opt, _ = lp_solve_fw(a, M_curr, k_indices, solver)
        if b_opt is None or b_opt.sum() < 1e-15:
            block_offset += block_size
            continue
        b_dir = b_opt / b_opt.sum()
        
        # Line search
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = M_curr
        for alpha in [best_alpha, 0.1, 0.05, 0.02, 0.5, 0.3, 0.2]:
            a_trial = (1.0 - alpha) * a + alpha * b_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha
        
        a = (1.0 - best_alpha) * a + best_alpha * b_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a
        
        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a.copy()
        
        iter_count += 1
        block_offset += block_size
        K = min(K + 20, K_max)
        
        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SubLP iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K}")
    
    return best_a, best_m


def gradient_descent_optimize(a0, time_limit, verbose=False, rng_seed=100,
                               top_k_grad=50, beta=5.0):
    """Mirror descent with soft subgradient (log barrier for non-negativity)."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    eta0 = 3.0
    stall = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov step
        beta_t = (t - 1.0) / (t + 1.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute soft subgradient
        ac = fast_autocorr(y)
        k_max = min(top_k_grad, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w = w / w.sum()
        
        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            g[idx] += 2.0 * wi * y[ki - idx]
        
        # Mirror descent update
        eta = eta0 / np.sqrt(t)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 3000:
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = np.maximum(best_a + pert, 1e-10)
            x = x_r / x_r.sum()
            x_prev = x.copy()
            stall = 0
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    GradDesc t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123, n_top=30):
    """Coordinate descent: move mass from high-contribution to low-contribution positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]
        
        # Compute sensitivity: how much does increasing a[i] increase the max conv?
        sens = np.zeros(n)
        for k in top_k[:15]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[ki - idx]
        
        improved = False
        
        # Try moving mass from high-sens to low-sens positions
        src_candidates = np.argsort(sens)[::-1][:10]
        dst_candidates = np.argsort(sens)[:10]
        
        for src in src_candidates:
            if a[src] < 1e-10:
                continue
            for dst in dst_candidates:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation
            stall += 1
            if stall % 10 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def pocs_spectral(a0, time_limit, verbose=False, n_trials=3, rng_seed=7):
    """POCS: alternate between flat-spectrum projection and non-negativity."""
    n = len(a0)
    best_a = norm_sum1(a0)
    best_m = fast_autocorr(best_a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit - 5:
            break
        t_trial = min((time_limit - (time.monotonic() - t0)) / (n_trials - trial), 60.0)
        
        rng = np.random.default_rng(rng_seed + trial * 100)
        a = best_a.copy()
        trial_rng = np.random.default_rng(rng_seed + trial)
        
        for it in range(100000):
            if time.monotonic() - t0 > time_limit - 5:
                break
            
            # Frequency: push toward flat magnitude
            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            mag_mask = mag > 1e-10
            if np.any(mag_mask):
                target = np.mean(mag[mag_mask])
                blend = 0.05 + 0.2 * min(1.0, it / 500)
                new_mag = (1 - blend) * mag + blend * target
                A = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A, fft_size)[:n]
            
            # Time: non-negativity + normalize
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s
            
            a = a_new
            M = fast_autocorr(a).max()
            if M < best_m:
                best_m = M
                best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    POCS trial {trial} ({elapsed:.0f}s): J={2.0*n*best_m:.6f}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """
    Multi-phase search:
    1. LP-FW with dual-guided active set (main optimizer)
    2. Multi-length exploration
    3. Refined LP-FW on best
    4. Coordinate polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            n = len(a)
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={n}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = make_u_shape(n_base, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: LP-FW on parent ===
    print(f"\n=== Phase 1: LP-FW (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    
    if n0 <= 3000:
        t1 = min(remaining * 0.25, 300.0)
        if t1 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_fw_full_constraints(a_in, t1, solver, verbose=True, rng_seed=42)
            try_update(a_out, "lp_full_p1")
    else:
        t1 = min(remaining * 0.25, 300.0)
        if t1 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = subsampled_lp_fw(a_in, t1, solver, verbose=True,
                                         K_init=400, K_max=3000, rng_seed=42)
            try_update(a_out, "lp_sub_p1")
    
    # === Phase 2: Multi-length exploration ===
    print(f"\n=== Phase 2: Multi-length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase2 = remaining * 0.30
    
    candidate_lengths = []
    # Smaller lengths (full LP feasible)
    for nl in [1000, 1500, 2000, 2500, 3000, 4000, 5000, 6200, 7000, 8000]:
        if nl != n0:
            candidate_lengths.append(nl)
    
    # Sort by proximity to current best
    candidate_lengths.sort(key=lambda x: abs(x - n0))
    candidate_lengths = candidate_lengths[:6]
    
    t_per = t_phase2 / max(1, len(candidate_lengths))
    for new_n in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t_avail = min(t_per, remaining - 30)
        if t_avail < 20:
            continue
        
        print(f"  Trying n={new_n} (t_avail={t_avail:.0f}s)...")
        a_interp = interpolate_sequence(best_seq, new_n)
        
        if new_n <= 2500:
            # Full constraint LP
            a_out, _ = lp_fw_full_constraints(a_interp, t_avail, solver,
                                              verbose=False, rng_seed=new_n % 1000)
        else:
            # Subsampled LP
            a_out, _ = subsampled_lp_fw(a_interp, t_avail, solver,
                                         K_init=300, K_max=
                                         min(2500, 2 * new_n - 1),
                                         rng_seed=new_n % 1000)
        try_update(a_out, f"lp_n{new_n}")
    
    # === Phase 3: Refined LP-FW on best ===
    print(f"\n=== Phase 3: Refined LP-FW (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    
    if n_best <= 2500:
        t3 = min(remaining * 0.20, 250.0)
        if t3 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_fw_full_constraints(a_in, t3, solver, verbose=True, rng_seed=777)
            try_update(a_out, "lp_refined")
    else:
        t3 = min(remaining * 0.20, 250.0)
        if t3 > 30:
            a_in = norm_sum1(best_seq)
            # Use larger K for refinement
            a_out, _ = subsampled_lp_fw(a_in, t3, solver, verbose=True,
                                         K_init=500, K_max=4000, rng_seed=777)
            try_update(a_out, "lp_refined")
    
    # === Phase 4: Gradient descent polish ===
    print(f"\n=== Phase 4: Gradient descent (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = gradient_descent_optimize(a_in, t4, verbose=False,
                                              top_k_grad=60, beta=5.0, rng_seed=200)
        try_update(a_out, "grad")
    
    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate descent (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True,
                                      rng_seed=456, n_top=30)
        try_update(a_out, "coord")
    
    # === Phase 6: Final LP-FW push ===
    print(f"\n=== Phase 6: Final LP-FW (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    
    if remaining > 40:
        t6 = min(remaining * 0.25, 200.0)
        if t6 > 30:
            a_in = norm_sum1(best_seq)
            if n_best <= 2500:
                a_out, _ = lp_fw_full_constraints(a_in, t6, solver, verbose=False, rng_seed=999)
            else:
                a_out, _ = subsampled_lp_fw(a_in, t6, solver, verbose=False,
                                             K_init=600, K_max=
                                             min(5000, 2 * n_best - 1), rng_seed=999)
            try_update(a_out, "lp_final")
    
    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 7: Final polish ===")
        t7 = min(remaining * 0.6, 80.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t7, verbose=False, rng_seed=789, n_top=40)
            try_update(a_out, "coord_final")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = make_u_shape(n_fb, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
