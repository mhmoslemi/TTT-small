def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]


def conv_ab(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]


def corr_shift(w, a):

        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            A[r, i0:i1] = yy[start:stop][::-1]
    A[:, n] = -1.0
    b = np.maximum(q_active, 0.0)

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0
    Aeq = np.ones((1, n + 1), dtype=np.float64)
    Aeq[0, n] = 0.0
    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=np.array([1.0]),
                      bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None, None

    status = getattr(res, 'status', None)
    success = getattr(res, 'success', False)
    x = getattr(res, 'x', None)
    if (status != 0 or not success) or x is None:
        return None, None

    g = np.asarray(x[:n], dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-10:
        return None, None
    g = g / s
    T = float(x[n])
    if not np.isfinite(T):
        T = 0.0
    return g, T


def smooth_polish_improved(a0, time_limit, rng_seed=11, verbose=False):
    """Adaptive mirror-descent polish with multiple focus levels and restarts."""
    if time_limit <= 0.0:
        return normalize_vec(a0)

    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 1:
        return a

    best = a.copy()
    best_score = objective_norm(a)

    trust = 0.5
    focus = 5.0
    fails = 0
    rng = np.random.default_rng(rng_seed)
    last_print = t0

    while time.monotonic() - t0 < time_limit:
        q = self_conv(a)
        M = float(np.max(q))
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            fails = 0
        else:
            fails += 1

        if fails > 500:
            break

        # Adaptive parameters
        if fails > 50 and fails % 25 == 0:
            focus = min(focus * 1.15, 80.0)
            trust = max(trust * 0.75, 0.02)

        if fails > 150 and fails % 150 == 0:
            # Perturbation restart
            scale_noise = 0.015 / (1.0 + 0.05 * (fails // 150))
            cand = best + scale_noise * rng.normal(size=best.size) * best
            cand = np.maximum(cand, 1e-12)
            cand = normalize_vec(cand)
            sc = objective_norm(cand)
            if np.isfinite(sc) and sc <= best_score + 0.001:
                a = cand
                fails = 0
                trust = 0.4
                focus = 5.0
            else:
                a = best.copy()

        w = weights_from_q(q, focus)
        c = corr_shift(w, a)
        cmax = float(np.max(c))
        cmin = float(np.min(c))
        scale = cmax - cmin
        if scale <= 1e-15:
            break

        move = trust * (cmax - c) / scale
        move = np.clip(move, -40.0, 40.0)
        b = a * np.exp(move)
        b = normalize_vec(b)

        Mb = float(np.max(self_conv(b)))
        score_b = 2.0 * n * Mb

        if score_b < best_score - 1e-13:
            best = b.copy()
            best_score = score_b
            a = b
            fails = 0
            trust = min(trust * 1.08, 5.0)
        elif score_b < score - 1e-14:
            a = b
            fails = 0
            trust = min(trust * 1.05, 5.0)
            focus = max(focus * 0.97, 1.0)
        else:
            fails += 1
            trust = max(trust * 0.75, 0.02)
            focus = min(focus * 1.08, 80.0)

        if verbose and (time.monotonic() - last_print) > 30.0:
            print("    smooth: t=%.1fs score=%.9f focus=%.1f trust=%.3f fails=%d"
                  % (time.monotonic() - t0, best_score, focus, trust, fails))
            last_print = time.monotonic()

    return best


def lp_fw_main(a0, time_limit, verbose=True):
    """Adaptive active-set LP-Frank-Wolfe with analytical line search."""
    if time_limit <= 0.0:
        return normalize_vec(a0)

    t0 = time.monotonic()
    y = normalize_vec(a0)
    n = y.size
    if n <= 5:
        return y

    best = y.copy()
    best_score = objective_norm(y)

    # Adaptive K
    if n > 7000:
        K_cur = 150
        K_max = 600
    elif n > 4000:
        K_cur = 200
        K_max = 800
    else:
        K_cur = 300
        K_max = 1200

    no_improve = 0
    use_lp = HAVE_LP
    iter_no = 0
    last_print = t0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining <= 5.0:
            break

        qy = self_conv(y)
        M_y = float(np.max(qy))
        score_y = 2.0 * n * M_y

        if score_y < best_score - 1e-12:
            best = y.copy()
            best_score = score_y

        # Select active set: top K plus some "near-active" lags
        K_use = int(min(K_cur, len(qy), 1500))
        order = np.argsort(qy)[::-1]
        act_main = order[:K_use]

        # Add some lags that are close to the peak (within 5% of max)
        threshold = M_y * 0.97
        near_active = np.where(qy >= threshold)[0]
        if near_active.size < K_use * 2:
            act = np.unique(np.concatenate([act_main, near_active[:K_use]]))
        else:
            act = act_main

        q_act = qy[act]

        g_dir = None
        T = None

        if use_lp and act.size >= 20:
            tstep = time.monotonic()
            g_dir, T = solve_lp_fw(y, act, q_act, n)
            spend = time.monotonic() - tstep

            if g_dir is None:
                use_lp = False
            else:
                if spend > max(30.0, 0.25 * remaining):
                    K_cur = max(80, K_cur - 60)

        if g_dir is not None:
            new, Mnew = line_search_direction(y, g_dir, qy, T)
            if Mnew < M_y - 1e-14:
                y = new
                no_improve = 0
                K_cur = min(K_cur + 15, K_max)
            else:
                no_improve += 1
        else:
            # Fallback: mirror descent direction
            w = weights_from_q(qy, focus=5.0)
            c = corr_shift(w, y)
            cmax = float(np.max(c))
            cmin = float(np.min(c))
            sc = cmax - cmin
            if sc > 1e-15:
                move = 0.5 * (cmax - c) / sc
                move = np.clip(move, -30.0, 30.0)
                d = y * np.exp(move)
                d = normalize_vec(d)
                new, Mnew = line_search_direction(y, d, qy, None)
                if Mnew < M_y - 1e-14:
                    y = new
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        # Adaptive K adjustment
        if no_improve > 0 and no_improve % 20 == 0:
            K_cur = min(K_cur + 30, K_max)
        if no_improve > 120:
            # Perturbation restart
            rng = np.random.default_rng(iter_no)
            cand = best + 0.01 * rng.normal(size=best.size) * best
            cand = np.maximum(cand, 1e-12)
            cand = normalize_vec(cand)
            sc_c = objective_norm(cand)
            if sc_c < best_score + 0.0005:
                y = cand
                K_cur = int(K_cur * 0.6)
            no_improve = 0
            if time.monotonic() - t0 > 0.9 * time_limit:
                break

        iter_no += 1
        now = time.monotonic()
        if verbose and (iter_no % 20 == 0 or (now - last_print) > 30.0):
            print("    LP-FW: it=%d t=%.1fs best=%.9f cur=%.9f K=%d noimp=%d"
                  % (iter_no, now - t0, best_score, score_y, K_cur, no_improve))
            last_print = now

    return best


def symmetric_optimize(a0, time_limit, verbose=False):
    """
    Optimize under palindromic constraint: a[i] = a[n-1-i].
    This halves the effective dimension.
    """
    if time_limit <= 0.0:
        return normalize_vec(a0)

    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 4:
        return a

    # Enforce symmetry (average with reverse)
    a = 0.5 * (a + a[::-1])
    a = normalize_vec(a)

    best = a.copy()
    best_score = objective_norm(a)

    trust = 0.4
    focus = 5.0
    fails = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        q = self_conv(a)
        M = float(np.max(q))
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            fails = 0
        else:
            fails += 1

        if fails > 400:
            break

        if fails > 60 and fails % 30 == 0:
            focus = min(focus * 1.12, 60.0)
            trust = max(trust * 0.78, 0.03)

        if fails > 100 and fails % 100 == 0:
            cand = best + 0.012 * rng.normal(size=best.size) * best
            cand = 0.5 * (cand + cand[::-1])
            cand = np.maximum(cand, 1e-12)
            cand = normalize_vec(cand)
            sc = objective_norm(cand)
            if sc < best_score + 0.001:
                a = cand
                fails = 0
                trust = 0.35

        w = weights_from_q(q, focus)
        c = corr_shift(w, a)
        # Enforce symmetry on gradient
        c = 0.5 * (c + c[::-1])
        cmax = float(np.max(c))
        cmin = float(np.min(c))
        sc = cmax - cmin
        if sc <= 1e-15:
            break

        move = trust * (cmax - c) / sc
        move = np.clip(move, -30.0, 30.0)
        # Symmetrize the move
        move = 0.5 * (move + move[::-1])
        b = a * np.exp(move)
        b = 0.5 * (b + b[::-1])
        b = normalize_vec(b)

        Mb = float(np.max(self_conv(b)))
        score_b = 2.0 * n * Mb

        if score_b < best_score - 1e-13:
            best = b.copy()
            best_score = score_b
            a = b
            fails = 0
            trust = min(trust * 1.06, 4.0)
        elif score_b < score - 1e-14:
            a = b
            fails = 0
            trust = min(trust * 1.04, 4.0)
        else:
            fails += 1
            trust = max(trust * 0.78, 0.03)
            focus = min(focus * 1.08, 60.0)

        if verbose and (time.monotonic() - t0) % 40.0 < 0.01:
            print("    sym: t=%.1fs score=%.9f fails=%d" % (time.monotonic() - t0, best_score, fails))

    return best


def spectral_equalize(a0, time_limit, verbose=False):
    """
    Frequency-domain approach: iteratively flatten the power spectrum.
    a -> FFT -> modify amplitude to make |A|^2 flatter -> IFFT -> project.
    """
    if time_limit <= 0.0:
        return normalize_vec(a0)

    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 4:
        return a

    best = a.copy()
    best_score = objective_norm(a)
    fails = 0
    alpha_blend = 0.3  # Blend factor between current and spectral target

    while time.monotonic() - t0 < time_limit:
        q = self_conv(a)
        M = float(np.max(q))
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            fails = 0
        else:
            fails += 1

        if fails > 200:
            break

        # FFT
        size = next_pow2(2 * n - 1)
        A = np.fft.rfft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)
        power = mag * mag

        # Target: make power more flat (geometric mean normalized)
        p_max = float(np.max(power))
        p_min = float(np.min(power[np.arange(len(power)) >= 1]))  # Skip DC
        if p_max <= 1e-30:
            break

        # Soft flatten: reduce peaks, boost valleys
        target_power = power ** 0.7  # Compression
        target_mag = np.sqrt(target_power)
        target_mag = target_mag / float(np.sum(target_mag / target_mag * mag))  # crude normalization

        # Actually, let's do it more carefully:
        # Scale target so that IFFT gives something with sum = 1
        target_mag_scaled = target_mag * (float(mag.sum()) / float(target_mag.sum()) if float(target_mag.sum()) > 0 else 1.0)

        A_new = target_mag_scaled * np.exp(1j * phase)
        a_spec = np.fft.irfft(A_new, size)[:n]

        # Blend with current
        a_new = (1.0 - alpha_blend) * a + alpha_blend * a_spec
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_vec(a_new)

        sc_new = objective_norm(a_new)
        if sc_new < best_score - 1e-13:
            best = a_new.copy()
            best_score = sc_new
            a = a_new
            fails = 0
            # Gradually increase blend
            alpha_blend = min(alpha_blend * 1.05, 0.7)
        elif sc_new < score - 1e-14:
            a = a_new
            fails = 0
        else:
            fails += 1
            alpha_blend = max(alpha_blend * 0.85, 0.05)

        if verbose and (time.monotonic() - t0) % 40.0 < 0.01:
            print("    spectral: t=%.1fs score=%.9f alpha=%.2f fails=%d"
                  % (time.monotonic() - t0, best_score, alpha_blend, fails))

    return best


def coordinate_polish(a0, time_limit, rng_seed=991):
    """Move mass from high-contribution elements to low-contribution ones."""
    if time_limit <= 0.0:
        return normalize_vec(a0)

    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 10:
        return a

    best = a.copy()
    best_score = objective_norm(a)
    stall = 0
    rng = np.random.default_rng(rng_seed)

    while time.monotonic() - t0 < time_limit and stall < 200:
        q = self_conv(a)
        top_lags = np.argsort(q)[::-1][:min(12, len(q))]

        # Compute contribution score for each element
        scores = np.zeros(n, dtype=np.float64)
        for kk in top_lags[:6]:
            k = int(kk)
            i0 = max(0, k - n + 1)
            i1 = min(n, k + 1)
            if i0 < i1:
                start = k - (i1 - 1)
                stop = k - i0 + 1
                scores[i0:i1] += a[start:stop][::-1]

        # Move from high-score to low-score elements
        src = np.argsort(scores)[::-1][:10]
        dst = np.argsort(scores)[:10]

        improved = False
        for s_idx in src:
            if time.monotonic() - t0 > time_limit:
                break
            if improved:
                break
            d = int(s_idx)
            v = float(a[d])
            if v <= 1e-15:
                continue
            for j in dst:
                j = int(j)
                if d == j:
                    continue
                for frac in (0.01, 0.03, 0.05, 0.1):
                    if time.monotonic() - t0 > time_limit:
                        break
                    amt = v * frac
                    if amt <= 1e-18:
                        continue
                    b = a.copy()
                    b[d] -= amt
                    b[j] += amt
                    b = np.maximum(b, 0.0)
                    ss = float(b.sum())
                    if ss <= 1e-15:
                        continue
                    b = b / ss
                    sc = objective_norm(b)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best = b.copy()
                        a = b
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 30 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    b = a.copy()
                    amt = a[i] * 0.02
                    b[i] -= amt
                    b[j] += amt
                    b = np.maximum(b, 0.0)
                    ss = b.sum()
                    if ss > 1e-15:
                        b = b / ss
                        sc = objective_norm(b)
                        if sc < best_score - 1e-13:
                            best_score = sc
                            best = b.copy()
                            a = b
                            stall = 0

    return best


def interpolate_sequence(f, new_n):
    f = normalize_vec(f)
    old_n = f.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return f.copy()
    if old_n <= 1:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f)
    return normalize_vec(g)


def power_transform(a, gamma):
    a = np.asarray(a, dtype=np.float64)
    b = np.maximum(a, 1e-30) ** gamma
    return normalize_vec(b)


def update_best_global(best_seq, best_score, a, label, t0):
    if a is None:
        return best_seq, best_score
    an = normalize_vec(a)
    if an.size == 0:
        return best_seq, best_score
    sc = objective_norm(an)
    if np.isfinite(sc) and sc < best_score - 1e-12:
        best_score = sc
        best_seq = an.copy()
        try:
            print("*** NEW BEST [%s]: %.9f n=%d t=%.1fs"
                  % (label, sc, an.size, time.monotonic() - t0))
        except Exception:
            pass
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    """
    t_global = time.monotonic()
    total_budget = min(1800.0, float(budget_s)) - 120.0
    total_budget = max(60.0, total_budget)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if p.ndim != 1:
            p = p.ravel()
        if p.size > 0 and p.sum() > 1e-12:
            parent = normalize_vec(p)
    except Exception:
        parent = None

    if parent is not None:
        best_seq = parent.copy()
        best_score = objective_norm(best_seq)
        print("Parent: n=%d objective=%.9f" % (best_seq.size, best_score))
    else:
        n_default = 6500
        x = np.linspace(0.0, 1.0, n_default)
        best_seq = normalize_vec(np.sin(np.pi * x) ** 2)
        best_score = objective_norm(best_seq)
        print("Default: n=%d objective=%.9f" % (best_seq.size, best_score))

    n0 = int(best_seq.size)

    # Phase 1: Quick smooth polish (60s)
    t1 = min(60.0, 0.08 * total_budget)
    if t1 > 10:
        out = smooth_polish_improved(best_seq, t1, rng_seed=101)
        best_seq, best_score = update_best_global(best_seq, best_score, out, "smooth_init", t_global)

    # Phase 2: Main LP-FW (largest chunk)
    remaining = total_budget - (time.monotonic() - t_global)
    t2 = min(900.0, 0.50 * remaining)
    if t2 > 60:
        out = lp_fw_main(best_seq, t2, verbose=True)
        best_seq, best_score = update_best_global(best_seq, best_score, out, "LP_main", t_global)

    # Phase 3: Symmetric optimization
    remaining = total_budget - (time.monotonic() - t_global)
    t3 = min(180.0, 0.20 * remaining)
    if t3 > 30:
        out = symmetric_optimize(best_seq, t3, verbose=True)
        best_seq, best_score = update_best_global(best_seq, best_score, out, "symmetric", t_global)

    # Phase 4: Spectral equalization
    remaining = total_budget - (time.monotonic() - t_global)
    t4 = min(150.0, 0.18 * remaining)
    if t4 > 20:
        out = spectral_equalize(best_seq, t4, verbose=True)
        best_seq, best_score = update_best_global(best_seq, best_score, out, "spectral", t_global)

    # Phase 5: Another LP-FW round on the improved point
    remaining = total_budget - (time.monotonic() - t_global)
    t5 = min(250.0, 0.25 * remaining)
    if t5 > 40:
        out = lp_fw_main(best_seq, t5, verbose=True)
        best_seq, best_score = update_best_global(best_seq, best_score, out, "LP_round2", t_global)

    # Phase 6: Power transforms
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 30:
        for g in (0.3, 0.5, 0.7, 1.5, 2.0):
            if time.monotonic() - t_global > total_budget + t_global:
                break
            out = power_transform(best_seq, g)
            sc = objective_norm(out)
            if sc < best_score - 1e-12:
                best_score = sc
                best_seq = out
                print("*** Power transform g=%.1f: %.9f" % (g, sc))
                # Quick polish after good transform
                t_pt = min(40.0, 0.05 * remaining)
                if t_pt > 10:
                    out2 = smooth_polish_improved(best_seq, t_pt, rng_seed=200 + int(g * 10))
                    best_seq, best_score = update_best_global(best_seq, best_score, out2, "post_power", t_global)

    # Phase 7: Try slightly different lengths
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 120:
        cur_n = int(best_seq.size)
        for dn in (-800, -400, 400, 800, 1500):
            remaining = total_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            nn = cur_n + dn
            if 3000 <= nn <= 10000 and nn != cur_n:
                a_ld = interpolate_sequence(best_seq, nn)
                t_ld = min(80.0, 0.30 * remaining)
                if t_ld > 20:
                    out = smooth_polish_improved(a_ld, t_ld, rng_seed=900 + nn % 500)
                    best_seq, best_score = update_best_global(best_seq, best_score, out, "len_%d" % nn, t_global)

    # Phase 8: Coordinate polish
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 50:
        tc = min(120.0, 0.40 * remaining)
        if tc > 20:
            out = coordinate_polish(best_seq, tc, rng_seed=314159)
            best_seq, best_score = update_best_global(best_seq, best_score, out, "coord", t_global)

    # Phase 9: Final smooth polish
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 30:
        tf = min(60.0, 0.80 * remaining)
        if tf > 10:
            out = smooth_polish_improved(best_seq, tf, rng_seed=9999)
            best_seq, best_score = update_best_global(best_seq, best_score, out, "final_smooth", t_global)

    # Final validation
    if best_seq is None:
        x = np.linspace(0.0, 1.0, 6500)
        best_seq = normalize_vec(np.sin(np.pi * x) ** 2)

    out_arr = np.asarray(best_seq, dtype=np.float64)
    out_arr = np.where(np.isfinite(out_arr), out_arr, 0.0)
    out_arr = np.maximum(out_arr, 0.0)
    out_arr = np.minimum(out_arr, 1000.0)
    s = float(out_arr.sum())
    if s <= 0.01:
        x = np.linspace(0.0, 1.0, 6500)
        out_arr = np.sin(np.pi * x) ** 2
        s = float(out_arr.sum())
    out_arr = out_arr / s

    final_score = objective_norm(out_arr)
    try:
        official = evaluate_safe(out_arr)
        if np.isfinite(official):
            final_score = min(final_score, official)
    except Exception:
        pass

    elapsed = time.monotonic() - t_global
    print("FINAL: internal_score=%.10f n=%d time=%.1fs" % (final_score, out_arr.size, elapsed))

    return [float(v) for v in out_arr]
