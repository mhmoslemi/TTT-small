def fast_autocorr(a):
    """Compute full autocorrelation (linear convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    """Project onto probability simplex."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_smoothed_grad(y, beta, top_k=None):
    """Compute gradient of smoothed max using log-sum-exp over top-k peaks."""
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    # Gradient of (1/beta)*log(sum exp(beta*b_k)) w.r.t. a
    # db_k/da_j = a[k-j] (for valid indices)
    # So grad_j = (1/beta) * sum_k p_k * a[k-j]
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, p, b

This is essentially the conditional gradient method with a vertex of the feasible set (which for the non-negative simplex is a unit vector e_j). The vertex that minimizes the linearization is the one with minimum c[j].

Wait, that's exactly what Frank-Wolfe does when the constraint set is the simplex! The LP min <c, g> s.t. g in simplex gives g = e_argmin(c). So the "greedy" approach IS Frank-Wolfe for the simplex constraint set.

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(y, beta, top_k=None):
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, p, b


def fw_line_search(y, g_dir, n_evals=25):
    """Bisection-style line search for Frank-Wolfe direction."""
    n = len(y)
    best_alpha = 2.0 / 2.0
    best_M = float('inf')
    # Grid search first
    alphas = np.linspace(0.0, 1.0, n_evals)
    for alpha in alphas:
        trial = (1.0 - alpha) * y + alpha * g_dir
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-15:
            continue
        trial = trial / s_t
        M = fast_autocorr(trial).max()
        if M < best_M:
            best_M = M
            best_alpha = alpha
    # Refine with bisection
    lo = max(0.0, best_alpha - 1.0 / n_evals)
    hi = min(1.0, best_alpha + 1.0 / n_evals)
    for _ in range(8):
        mid = 0.5 * (lo + hi)
        trial_mid = (1.0 - mid) * y + mid * g_dir
        trial_mid = np.maximum(trial_mid, 0.0)
        s_m = trial_mid.sum()
        if s_m < 1e-15:
            hi = mid
            continue
        trial_mid = trial_mid / s_m
        M_mid = fast_autocorr(trial_mid).max()

        q1 = 0.5 * (lo + mid)
        trial_q1 = (1.0 - q1) * y + q1 * g_dir
        trial_q1 = np.maximum(trial_q1, 0.0)
        s_q1 = trial_q1.sum()
        if s_q1 < 1e-15:
            lo = mid
            continue
        trial_q1 = trial_q1 / s_q1
        M_q1 = fast_autocorr(trial_q1).max()

        if M_q1 < M_mid:
            hi = mid
        else:
            lo = mid
    best_alpha = 0.5 * (lo + hi)
    trial = (1.0 - best_alpha) * y + best_alpha * g_dir
    trial = np.maximum(trial, 0.0)
    s_t = trial.sum()
    if s_t > 1e-15:
        trial = trial / s_t
    return trial, best_alpha


def lp_fw_optimize(a0, time_limit, solver, initial_K=400, max_K=2000,
                   add_per_iter=20, verbose=True):
    """LP-Frank-Wolfe with bisection line search and adaptive active set."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 800

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K // 2])
                constraint_list = sorted(active_set)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations to add to active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr * 1.001 + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Line search
        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new, _ = fw_line_search(y, g_dir, n_evals=20)

        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  LP-FW iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def fast_fw_optimize(a0, time_limit, top_k=300, verbose=True):
    """Fast conditional gradient (FW) without LP - uses subgradient direction."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    t_fw = 1.0
    no_improve = 0
    max_no_improve = 5000

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        b = fast_autocorr(y)
        M_curr = b.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute subgradient direction: minimize c[j] = sum_k p[k] * y[k-j]
        idx = np.argsort(b)[::-1][:top_k]
        p = np.zeros(len(b))
        p[idx] = 1.0 / len(idx)

        y_rev = y[::-1]
        conv_len = n + len(p) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        Yr = np.fft.rfft(y_rev, fft_size)
        Pr = np.fft.rfft(p, fft_size)
        conv = np.fft.irfft(Yr * Pr, fft_size)
        c = conv[n - 1:2 * n - 1]

        # The FW direction is the vertex of the simplex minimizing <c, g>
        # which is e_{argmin(c)}
        # But we can also use a "soft" direction for better convergence
        j_star = int(np.argmin(c))

        # Use a soft direction: concentrate mass where c is low
        c_shifted = c - c.min()
        c_exp = np.exp(-c_shifted * 50.0)
        c_exp = np.maximum(c_exp, 0)
        g_dir = c_exp / c_exp.sum()

        # Line search
        if t == 1:
            y_new = g_dir.copy()
        else:
            alpha = 2.0 / (t_fw + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  FastFW iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

    return best_a, best_m


def nesterov_optimize(a0, time_limit, beta_start=20.0, beta_end=800.0,
                      beta_growth=1.004, eta_base=0.8, top_k_ratio=0.04,
                      restart_every=2500, verbose=True):
    """Nesterov accelerated projected gradient with adaptive beta."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(123)
    top_k = max(20, int(n * top_k_ratio))

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, b = compute_smoothed_grad(y, beta, top_k)

        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.12)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 60 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.015 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 5.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("  Nest restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("  Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " beta=" + repr(round(beta, 1)))

    return best_a, best_max_b


def coordinate_polish(a0, time_limit, verbose=True):
    """Peak-targeted coordinate descent with systematic mass redistribution."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    max_stall = 1200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(50, len(ac))
        top_peaks = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_peaks[:25]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_idx = int(np.argmax(contrib))
            src = int(idx[max_idx])
            partner = k_int - src
            if partner < 0 or partner >= n:
                continue

            src_val = a[src]
            partner_val = a[partner] if 0 <= partner < n else 0.0

            if src_val >= partner_val:
                reduce_idx = src
            else:
                reduce_idx = partner

            if a[reduce_idx] < 1e-12:
                continue

            # Compute pressure at each position
            pressure = np.zeros(n)
            for k2 in top_peaks[:15]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]

            # Find low-pressure destinations
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:15]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)

            cand_pressure = pressure[candidates]
            # Try several candidates
            order = np.argsort(cand_pressure)[:20]
            found = False
            for ci in order:
                dst = int(candidates[ci])
                if dst == reduce_idx:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = a[reduce_idx] * frac
                    if amt < 1e-15:
                        break
                    a_trial = a.copy()
                    a_trial[reduce_idx] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                if found:
                    break

            if not found and stall > 300:
                # Random exploration
                for _ in range(3):
                    j = int(rng.integers(n))
                    if j != reduce_idx and a[reduce_idx] > 1e-10:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a[reduce_idx] * frac
                        a[reduce_idx] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True

        if not improved:
            stall += 1
            if stall % 200 == 0 and stall < max_stall:
                # Broader random moves
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " stall=" + str(stall))

    return best_a, best_m


def parametric_init(n, time_limit, best_init, best_score):
    """Quick parametric search for good initialization."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score

    candidates = []
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    for p in [0.5, 1.0, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p * (n - i_arr) ** p)
        candidates.append(x ** p + (1.0 - x) ** p + 0.1)
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    for freq in [1, 2, 3]:
        candidates.append(np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x)) / 2.0, 0.01))
    for p in [1, 2, 3, 4]:
        candidates.append(np.sin(np.pi * x) ** p)
    # Modulated sine
    for freq in [5, 10, 20]:
        candidates.append(np.sin(np.pi * x) ** 2 * (0.5 + 0.5 * np.cos(2 * np.pi * freq * x)) + 0.01)

    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(a_raw)
        b = fast_autocorr(a)
        m = b.max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    rng = np.random.default_rng(42)
    for _ in range(80):
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(rng.random(n) ** rng.uniform(0.3, 3.0))
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    return best_a, best_val


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search: LP-FW + Fast FW + Nesterov + coordinate descent with multi-length."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Time budget: " + repr(round(time_budget, 1)) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: Parametric search
    print("\n=== Phase 1: Parametric (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 40.0)
    if t1 > 10:
        a_param, _ = parametric_init(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "param", t_global)

    # Phase 2: LP-FW main optimizer (most important)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: LP-FW main (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.35, 600.0)
    if t2 > 60:
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t2, solver,
                                  initial_K=400, max_K=1500,
                                  add_per_iter=20, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_main", t_global)

    # Phase 3: Fast FW (fast iterations, good for refinement)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Fast FW (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_out, _ = fast_fw_optimize(norm_sum1(best_seq), t3, top_k=400, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fastfw", t_global)

    # Phase 4: Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Nesterov (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        a_out, _ = nesterov_optimize(norm_sum1(best_seq), t4,
                                     beta_start=20.0, beta_end=600.0,
                                     beta_growth=1.004, eta_base=0.7,
                                     top_k_ratio=0.04, restart_every=2000,
                                     verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest", t_global)

    # Phase 5: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Coordinate (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 20:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t5, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord", t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-length ===")
    longer_lengths = [n_best + 200, n_best + 500, n_best + 800]
    shorter_lengths = [max(1000, n_best - 500), max(1000, n_best - 1000)]
    all_lengths = longer_lengths + shorter_lengths
    all_lengths = [l for l in all_lengths if 2000 <= l <= 10000]

    for new_n in all_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print("  Trying n=" + str(new_n))
        a_new = interpolate_sequence(best_seq, new_n)

        # Quick LP-FW
        t6a = min(remaining * 0.35, 200.0)
        if t6a > 30:
            a_out, _ = lp_fw_optimize(a_new, t6a, solver,
                                      initial_K=300, max_K=1200,
                                      add_per_iter=15, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t6b = min(remaining * 0.40, 150.0)
            if t6b > 20:
                a_out, _ = coordinate_polish(norm_sum1(best_seq), t6b, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n" + str(new_n), t_global)

    # Phase 7: Final LP-FW burst on best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final LP-FW (n=" + str(n_best) + ") ===")
    if remaining > 50:
        t7 = min(remaining * 0.50, 300.0)
        if t7 > 30:
            a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t7, solver,
                                      initial_K=500, max_K=2000,
                                      add_per_iter=25, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Phase 8: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final coord (n=" + str(n_best) + ") ===")
    if remaining > 30:
        t8 = min(remaining * 0.60, 200.0)
        if t8 > 20:
            a_out, _ = coordinate_polish(norm_sum1(best_seq), t8, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Phase 9: Final fast FW
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t9 = min(remaining * 0.70, 150.0)
        if t9 > 20:
            a_out, _ = fast_fw_optimize(norm_sum1(best_seq), t9, top_k=500, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fastfw_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        # Ensure all values are valid
        best_seq = np.asarray(best_seq, dtype=np.float64)
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        elapsed = time.monotonic() - t_global
        final_score = evaluate_safe(best_seq)
        print("\nFINAL: score=" + repr(final_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
