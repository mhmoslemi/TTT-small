"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fienup/Gerchberg-Saxton iterations for flat autocorrelation (fast O(n log n))
2. Subgradient descent with Nesterov acceleration
3. LP-based Frank-Wolfe for final polishing
4. Multi-length exploration
"""

import numpy as np
import time
import cvxpy as cp

def fast_autocorr(a):
    """Compute full autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to non-negative, sum to 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Use the standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_max = np.max(np.where(cond)[0]) + 1
    theta = (cssv[rho_max - 1] - 1) / rho_max
    return np.maximum(v - theta, 0)

def fienup_flat(a, n_iter=500, alpha=0.7, verbose=False):
    """
    Fienup error reduction for flat autocorrelation.
    Alternates between:
    - Time domain: project onto non-negative simplex
    - Frequency domain: flatten the power spectrum
    """
    n = len(a)
    a = norm_sum1(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    for it in range(n_iter):
        # Current autocorrelation
        A = np.fft.rfft(a, N)
        power = np.abs(A)**2
        c = np.fft.irfft(power, N)  # circular autocorrelation
        # Linear autocorrelation is c[0:n] and c[N-n:N] (reversed)
        # For now, look at c[0:size]
        
        # Flatten the autocorrelation
        c_lin = c[:size]
        c_sym = np.concatenate([c[:n], c[N-n:N][::-1]])  # first half and mirrored second half
        # Actually, let's just work with the full circular autocorrelation
        # and "flatten" the relevant lags
        
        # Strategy: target = c^(alpha) which compresses peaks
        # Or: target = min(c, threshold)
        # Let's use a soft version: target = c * (1 + alpha) - alpha * max(c)
        # No, let's use the power compression
        c_flat = c_lin ** alpha  # compresses dynamic range
        # Ensure non-negative
        c_flat = np.maximum(c_flat, 0)
        
        # Compute target power spectrum
        P_target = np.fft.rfft(c_flat, N)
        P_target = np.maximum(P_target.real, 0)  # clip negative parts
        
        # Update magnitude, keep phase
        mag_new = np.sqrt(P_target / N)
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        
        # Back to time domain
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project onto non-negative simplex
        a_new = project_simplex(a_new)
        
        # Check improvement
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 100 == 0:
            print(f"    Fienup iter={it}: max_AC={best_m:.8f}")
    
    return best_a, best_m
