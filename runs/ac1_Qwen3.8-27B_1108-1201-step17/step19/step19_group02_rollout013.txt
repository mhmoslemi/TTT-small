def fast_conv(a):
    """Linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_fast(a):
    """Fast evaluation: 2*n*max(conv)/sum^2, assuming a>=0, sum>0."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    c = fast_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def eval_safe(a):
    """Safe evaluation using the actual evaluate_sequence."""
    try:
        a_list = [max(0.0, min(1000.0, float(x))) for x in np.asarray(a, dtype=np.float64).flatten()]
        if not a_list or sum(a_list) < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interp_seq(f, new_n):
    """Interpolate sequence f to new length new_n, preserving integral (sum)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / ss


def build_lp_matrix(y, k_list, n):
    """Build constraint matrix for LP: row k has y[k-j] at position j for valid j."""
    K = len(k_list)
    A = np.zeros((K, n))
    for jj, k in enumerate(k_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            A[jj, i_start:i_end] = y[i_end - 1 - np.arange(i_end - i_start)][::-1]
    return A


def lp_solve(y, active_set, solver, n):
    """Solve the LP: min M s.t. sum(g)=1, g>=0, (y*g)[k]<=M for k in active_set."""
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    A = build_lp_matrix(y, active_set, n)
    constraints = [cp.sum(g) == 1, A @ g <= M]
    prob = cp.Problem(cp.Minimize(M), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
    M_val = float(M.value) if M.value is not None else float('inf')
    gs = g_val.sum()
    if gs < 1e-15:
        return None, M_val
    return g_val / gs, M_val


def lp_fw_enhanced(a0, time_limit, solver, initial_K=400, max_K=3000,
                   add_per_iter=50, verbose=True):
    """Enhanced LP-Frank-Wolfe with direct jump and backtracking."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    ac = fast_conv(y)
    best_m = ac.max()
    t0 = time.monotonic()

    active_set = sorted(np.argsort(ac)[::-1][:min(initial_K, 2 * n - 1)].tolist())
    no_improve = 0
    max_no_improve = 2000
    it = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: stagnation at iter " + str(it))
            break
        it += 1

        ac_y = fast_conv(y)
        M_curr = ac_y.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update active set: keep top constraints by margin
        if len(active_set) > max_K:
            ac_full = fast_conv(y)
            margins = np.array([M_curr - ac_full[k] for k in active_set])
            keep_idx = np.argsort(margins)[:max_K]
            active_set = sorted([active_set[i] for i in keep_idx])

        K = len(active_set)
        if K == 0:
            break

        # Solve LP
        g_dir, M_lp = lp_solve(y, active_set, solver, n)
        if g_dir is None:
            # Try with fewer constraints
            active_set = sorted(active_set[:max(100, K // 2)])
            continue

        # Check if LP solution directly improves
        ac_g = fast_conv(g_dir)
        M_g = ac_g.max()

        # Add violated constraints
        ac_yg = np.fft.irfft(
            np.fft.rfft(y, 1 << (2 * n - 1).bit_length()) *
            np.fft.rfft(g_dir, 1 << (2 * n - 1).bit_length()),
            1 << (2 * n - 1).bit_length()

        violations = np.where(ac_yg > M_curr + 1e-9)[0]

                new_set.add(int(k))
            active_set = sorted(new_set)

            min(1.0, best_alpha * 0.5),
            min(1.0, best_alpha * 1.5),

            min(1.0, 0.2),
            min(1.0, 0.5),

        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir

def coord_peak_shave(a0, time_limit, verbose=False):
    """Coordinate descent focused on reducing the peak convolution value."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    ac = fast_conv(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    max_stall = 1500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M = ac.max()
        # Find top peak indices
        top_k = np.argsort(ac)[::-1][:min(50, len(ac))]

        improved = False
        for k in top_k[:25]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            # Contribution of each pair (idx[j], k_int - idx[j])
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            # Source: index with highest contribution to this peak
            best_j = int(idx[np.argmax(contrib)])
            src_val = a[best_j]
            if src_val < 1e-12:
                continue

            # Find destination: index that minimizes its contribution to all top peaks
            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            # Prefer indices far from peak-contributing regions
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < n // 5:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])
            if dst == best_j:
                continue

            # Try several transfer fractions
            found = False
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = src_val * frac
                if amt < 1e-15 or amt >= src_val:
                    continue
                a_trial = a.copy()
                a_trial[best_j] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 300:
                # Random perturbation
                j = int(rng.integers(n))
                if j != best_j and a[best_j] > 1e-10:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a[best_j] * frac
                    a[best_j] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            # Periodic random restart
            if stall % 200 == 0:
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_conv(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                break

        if verbose and stall % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 8)) + ", stall=" + str(stall))

    return best_a, best_m


def mirror_descent_smooth(a0, time_limit, top_k=60, beta=8.0, c0=3.0,
                          momentum=0.8, verbose=False):
    """Smooth mirror descent using log-sum-exp approximation of max."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    vel = np.zeros(n)
    stall = 0
    restart_every = 6000

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Momentum
        y = x + momentum * vel
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        ac = fast_conv(y)
        M = ac.max()

        # Soft gradient: weighted sum of gradients at top peaks
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx] - M + ac[top_idx].max(), 1e-30)
        # Use softmax weights
        exponents = beta * (ac[top_idx] - M)
        exponents -= exponents.max()
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += 2.0 * wi * y[k_int - idx]

        # Mirror descent update
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        vel = 0.9 * vel + 0.1 * (a_new - x)
        x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            noise = 0.03 / (1.0 + 0.05 * stall)
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r > 1e-15:
                x = x_r / s_r
            vel = np.zeros(n)

    return best_a, best_m


def spectral_init(n, rng):
    """Initialize using flat-spectrum-inspired approach."""
    # Try to construct a sequence with flatter power spectrum
    # Use inverse DFT of a random phase sequence with constant magnitude
    fft_size = 1 << (2 * n - 1).bit_length()
    # Random phases
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    # Constant magnitude (flat spectrum)
    spectrum = np.exp(1j * phases)
    # Build full spectrum (conjugate symmetric for real signal)
    full_spec = np.zeros(fft_size, dtype=complex)
    full_spec[:fft_size // 2 + 1] = spectrum
    for k in range(1, fft_size // 2 + 1):
        full_spec[fft_size - k] = np.conj(spectrum[k])
    # Inverse FFT
    signal = np.fft.ifft(full_spec).real
    # Take first n elements, make non-negative
    a = np.maximum(signal[:n], 0)
    s = a.sum()
    if s < 1e-10:
        # Fallback: try with different phase pattern
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        spectrum = np.exp(1j * phases)
        full_spec = np.zeros(fft_size, dtype=complex)
        full_spec[:fft_size // 2 + 1] = spectrum
        for k in range(1, fft_size // 2 + 1):
            full_spec[fft_size - k] = np.conj(spectrum[k])
        signal = np.fft.ifft(full_spec).real
        a = np.maximum(signal[:n], 0)
        s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def parametric_candidates(n, rng):
    """Generate diverse parametric initializations."""
    x = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)
    candidates = []

    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
        candidates.append(normalize((i + 1.0) ** p))
        candidates.append(normalize((n - i) ** p))

    # Product power laws
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        for beta in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            a = np.maximum(i + 1.0, 0.5) ** alpha * np.maximum(n - i, 0.5) ** beta
            candidates.append(normalize(a))

    # Sinusoidal
    for freq in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(normalize(np.sin(np.pi * freq * x) ** 2 + 0.01))

    # Gaussian at edges
    for sigma in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]:
        a = np.exp(-((x - 0) ** 2) / (2 * sigma ** 2)) + np.exp(-((x - 1) ** 2) / (2 * sigma ** 2))
        candidates.append(normalize(a))

    # Anti-Gaussian (U-shaped)
    for p in [1.0, 1.5, 2.0, 3.0, 5.0]:
        candidates.append(normalize(np.abs(2 * x - 1) ** p + 0.01))

    # Cosine windows
    for p in [0.5, 1.0, 2.0, 4.0]:
        candidates.append(normalize(np.abs(np.cos(np.pi * x)) ** p + 0.01))

    # Bimodal
    for c1 in [0.1, 0.15, 0.2]:
        for c2 in [0.8, 0.85, 0.9]:
            for sigma in [0.03, 0.05, 0.1, 0.15]:
                a = np.exp(-((x - c1) ** 2) / (2 * sigma ** 2)) + np.exp(-((x - c2) ** 2) / (2 * sigma ** 2))
                candidates.append(normalize(a))

    # Chebyshev-like
    for p in [1, 2, 3, 5]:
        t = 2 * x - 1
        candidates.append(normalize(np.abs(np.cos(p * np.arccos(np.clip(t, -1, 1)))) ** 2 + 0.01))

    # Random
    for _ in range(100):
        a = rng.random(n)
        a = a ** rng.uniform(0.3, 5.0)
        candidates.append(normalize(a))

    # Spectral
    for _ in range(30):
        candidates.append(spectral_init(n, rng))

    return candidates


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = eval_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) +
              ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    rng = np.random.default_rng(2024)

    print("Total time budget: " + repr(time_budget) + "s")

    # Get initial sequence
    best_seq = None
    best_score = float('inf')

    if 'height_sequence_1' in dir() or True:
        try:
            if height_sequence_1 is not None and len(height_sequence_1) > 0:
                p = np.asarray(height_sequence_1, dtype=np.float64)
                p = np.clip(p, 0.0, 1000.0)
                if p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
                    best_seq = p / p.sum()
                    best_score = eval_safe(best_seq)
                    print("Parent: n=" + str(len(p)) + ", score=" + repr(best_score))
        except Exception:
            pass

    if best_seq is None:
        n0 = 5000
        x = np.linspace(0, 1, n0)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # === Phase 1: Parametric search (multiple n values) ===
    print("\n=== Phase 1: Parametric search ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1_total = min(remaining * 0.06, 90.0)
    t1_start = time.monotonic()

    n_values = [n0, max(3000, n0 - 1000), min(8000, n0 + 1000), 4000, 6000]
    n_values = sorted(set([n for n in n_values if 2000 <= n <= 9000]))

    per_n_time = t1_total / max(1, len(n_values))
    for n_try in n_values:
        if time.monotonic() - t1_start > t1_total:
            break
        cands = parametric_candidates(n_try, rng)
        best_local = float('inf')
        best_local_a = None
        for c in cands:
            if time.monotonic() - t1_start > t1_total:
                break
            v = eval_fast(c)
            if v < best_local:
                best_local = v
                best_local_a = c
        if best_local_a is not None and best_local < best_score:
            best_score = best_local
            best_seq = best_local_a.copy()
            print("  Parametric n=" + str(n_try) + ": J=" + repr(round(best_score, 8)))

    # === Phase 2: LP-FW on best sequence ===
    print("\n=== Phase 2: LP-FW (n=" + str(len(best_seq)) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 600.0)
    if t2 > 30 and len(best_seq) <= 9000:
        a_out, _ = lp_fw_enhanced(best_seq, t2, solver,
                                  initial_K=500, max_K=4000,
                                  add_per_iter=50, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1", t_global)

    # === Phase 3: Peak-shaving coordinate descent ===
    print("\n=== Phase 3: Coord peak-shave (n=" + str(len(best_seq)) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 180.0)
    if t3 > 15:
        a_out, _ = coord_peak_shave(best_seq, t3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord1", t_global)

    # === Phase 4: Mirror descent ===
    print("\n=== Phase 4: Mirror descent (n=" + str(len(best_seq)) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 180.0)
    if t4 > 15:
        a_out, _ = mirror_descent_smooth(best_seq, t4, top_k=60, beta=8.0,
                                         c0=3.5, momentum=0.8, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "mirror1", t_global)

    # === Phase 5: LP-FW again with fresh active set ===
    print("\n=== Phase 5: LP-FW refine (n=" + str(len(best_seq)) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.20, 350.0)
    if t5 > 30 and len(best_seq) <= 9000:
        a_out, _ = lp_fw_enhanced(best_seq, t5, solver,
                                  initial_K=600, max_K=5000,
                                  add_per_iter=60, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2", t_global)

    # === Phase 6: Try different n values ===
    print("\n=== Phase 6: Multi-n exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    try_ns = sorted(set([
        n_best,
        max(3000, n_best - 500),
        min(9000, n_best + 500),
        max(3000, n_best - 1000),
        min(9000, n_best + 1000),
    ]))

    for new_n in try_ns:
        if new_n == n_best:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print("  Trying n=" + str(new_n) + " (remaining=" + str(round(remaining, 1)) + "s)")
        a_long = interp_seq(best_seq, new_n)

        t6a = min(remaining * 0.4, 180.0)
        if t6a > 20 and new_n <= 9000:
            a_out, _ = lp_fw_enhanced(a_long, t6a, solver,
                                      initial_K=400, max_K=3000,
                                      add_per_iter=40, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t6b = min(remaining * 0.3, 80.0)
            if t6b > 10:
                a_out, _ = coord_peak_shave(best_seq, t6b, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n" + str(new_n), t_global)

    # === Phase 7: Multi-start perturbation ===
    print("\n=== Phase 7: Multi-start perturbation ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100 and n_best <= 8000:
        for trial in range(min(5, int(remaining / 60))):
            if time.monotonic() - t_global > time_budget - 30:
                break
            # Perturb current best
            noise_scale = 0.05 * (1.0 + 0.2 * trial)
            pert = rng.normal(0, noise_scale, n_best) * best_seq
            a_pert = best_seq + pert
            a_pert = np.maximum(a_pert, 0.0)
            s_p = a_pert.sum()
            if s_p < 0.01:
                continue
            a_pert = a_pert / s_p

            t_ps = min(60.0, (time_budget - (time.monotonic() - t_global)) * 0.15)
            if t_ps > 10:
                a_out, _ = lp_fw_enhanced(a_pert, t_ps, solver,
                                          initial_K=300, max_K=2500,
                                          add_per_iter=30, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out,
                                                  "perturb" + str(trial), t_global)

    # === Phase 8: Final LP-FW + Coord polish ===
    print("\n=== Phase 8: Final polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    if remaining > 60 and n_best <= 9000:
        t8a = min(remaining * 0.4, 200.0)
        if t8a > 20:
            a_out, _ = lp_fw_enhanced(best_seq, t8a, solver,
                                      initial_K=700, max_K=5000,
                                      add_per_iter=70, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t8b = min(remaining * 0.6, 100.0)
        if t8b > 10:
            a_out, _ = coord_peak_shave(best_seq, t8b, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # === Final validation ===
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) +
              ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
