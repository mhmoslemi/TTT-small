def softmax_direction(a, beta, top_k=None):
    """Compute the softmax-weighted gradient direction."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    
    if top_k is not None and top_k < size:
        top_idx = np.argsort(c)[::-1][:top_k]
        c_top = c[top_idx]
        c_max = c_top.max()
        exp_vals = np.zeros(size)
        exp_vals[top_idx] = np.exp(beta * (c_top - c_max))
        w = exp_vals / exp_vals.sum()
    else:
        c_max = c.max()
        exp_vals = np.exp(beta * (c - c_max))
        w = exp_vals / exp_vals.sum()

    W = np.fft.rfft(w, fft_size)

    conv = np.fft.irfft(W * A_rev, fft_size)[:size]
    grad = 2.0 * conv[n - 1:2 * n - 1]
    
    return grad, c, w

"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with adaptive softmax gradient for fast convergence
2. Aggressive LP-based Frank-Wolfe with constraint management for polishing
3. Away-step FW variant for better vertex escaping
4. Multi-length exploration
5. Adaptive restarts

import numpy as np
import time
import cvxpy as cp

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize_nonneg(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_score(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def get_gradient(a, beta, top_k=200):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    
    if top_k is not None and top_k < size:
        top_idx = np.argsort(c)[::-1][:top_k]
        c_max = c[top_idx].max()
        w = np.zeros(size)
        w[top_idx] = np.exp(beta * (c[top_idx] - c_max))
        w /= w.sum()
    else:
        c_max = c.max()
        w = np.exp(beta * (c - c_max))
        w /= w.sum()
    
    a_rev = a[::-1].copy()
    W = np.fft.rfft(w, fft_size)
    A_rev = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(W * A_rev, fft_size)[:size]
    grad = 2.0 * conv[n-1:2*n-1]
    return grad, c, w

def fista_optimize(a0, time_limit, beta_start=5.0, beta_end=100.0, eta0=0.8, top_k=200, verbose=False):
    n = len(a0)
    x = normalize_nonneg(a0)
    y = x.copy()
    best_x = x.copy()
    best_m = fast_conv_self(x).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_limit = 5000
    
    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        progress = min(iter_count / max(time_limit * 80, 1), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        grad, c_y, _ = get_gradient(y, beta, top_k)
        eta = eta0 / (np.sqrt(t) + 1.0)
        
        log_y = np.log(np.maximum(y, 1e-300)) - eta * grad
        log_y -= log_y.max()
        z = np.exp(log_y)
        z = np.maximum(z, 1e-300)
        z /= z.sum()
        
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = z + momentum * (z - y)
            x_new = np.maximum(x_new, 1e-300)
            x_new /= x_new.sum()
        else:
            x_new = z.copy()
        
        c_new = fast_conv_self(x_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > stall_limit:
            seed = int(time.monotonic() * 1000000) % (2**31)
            rng = np.random.default_rng(seed)
            sigma = 0.01 + 0.02 * (stall_limit / 5000)
            pert = rng.normal(0, sigma, n) * best_x
            x_new = normalize_nonneg(best_x + pert)
            y = x_new.copy()
            stall = 0
            stall_limit = max(int(stall_limit * 0.75), 800)
        
        y = x_new
        x = x_new
        iter_count += 1
        
        if verbose and iter_count % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  FISTA iter={iter_count} ({elapsed:.0f}s): M={best_m:.10f}, beta={beta:.1f}")
    
    return best_x, best_m

def lp_fw_optimize(a0, time_limit, solver, init_K=200, max_K=3000, verbose=True):
    """LP-based Frank-Wolfe with away steps and dynamic constraint management."""
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    ac = fast_conv_self(a)
    best_m = ac.max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    size = 2 * n - 1
    
    # Initialize constraint set with top K lags
    ac_y = fast_conv_self(y)
    init_idx = np.argsort(ac_y)[::-1][:min(init_K, size)]
    constraint_set = set(int(k) for k in init_idx)
    
    iter_count = 0
    max_iters = 1000000
    
    while iter_count < max_iters:
        if time.monotonic() - t0 > time_limit:
            break
        
        ac_y = fast_conv_self(y)
        M_curr = ac_y.max()
        
        if M_curr < best_m - 1e-13:
            best_m = M_curr
            best_a = y.copy()
        
        # Check x as well
        ac_x = fast_conv_self(x)
        if ac_x.max() < best_m - 1e-13:
            best_m = ac_x.max()
            best_a = x.copy()
        
        # Manage constraint set
        constraint_list = sorted(constraint_set)
        # Add lags that are close to M
        close_lags = np.where(ac_y > M_curr * 0.95)[0]
        for k in close_lags:
            constraint_set.add(int(k))
        constraint_list = sorted(constraint_set)
        
        if len(constraint_list) > max_K:
            # Prune: keep the ones closest to M
            margins = np.array([M_curr - ac_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        # Solve LP: min M s.t. sum(g)=1, g>=0, coeffs@g <= M
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            constraint_set = set(np.argsort(ac_y)[::-1][:init_K].astype(int))
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violated constraints (not in current set)
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_yg > M_curr + 1e-7)[0]
        for k in violations[:50]:
            constraint_set.add(int(k))
        
        # Standard FW step with Nesterov acceleration
        alpha = 2.0 / (iter_count + 3.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if iter_count == 0:
            x_new = y_new.copy()
        else:
            beta_t = (iter_count) / (iter_count + 3.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP-FW iter={iter_count} ({elapsed:.0f}s): M={best_m:.10f}, |S|={len(constraint_set)}")
    
    return best_a, best_m

import numpy as np
import time
import cvxpy as cp


def fast_conv_self(a):
    """Compute full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_score(a):
    """Call evaluate_sequence and return the score."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def get_softmax_grad(a, beta, top_k=200):
    """Compute softmax-weighted gradient for the max-autocorrelation objective."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]

    if top_k is not None and top_k < size:
        top_idx = np.argsort(c)[::-1][:top_k]
        c_max = c[top_idx].max()
        w = np.zeros(size)
        w[top_idx] = np.exp(beta * (c[top_idx] - c_max))
        w /= w.sum()
    else:
        c_max = c.max()
        w = np.exp(beta * (c - c_max))
        w /= w.sum()

    a_rev = a[::-1].copy()
    W = np.fft.rfft(w, fft_size)
    A_rev = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(W * A_rev, fft_size)[:size]
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, c, w


def proj_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho = np.nonzero(u * idx > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fista_optimize(a0, time_limit, beta_start=5.0, beta_end=100.0, eta0=0.8, top_k=200, verbose=False):
    """FISTA with multiplicative updates (entropic mirror)."""
    n = len(a0)
    x = normalize_nonneg(a0)
    y = x.copy()
    best_x = x.copy()
    best_m = fast_conv_self(x).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_limit = 5000

    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        progress = min(iter_count / max(time_limit * 60, 1), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        grad, c_y, _ = get_softmax_grad(y, beta, top_k)
        eta = eta0 / (np.sqrt(t) + 1.0)

        log_y = np.log(np.maximum(y, 1e-300)) - eta * grad
        log_y -= log_y.max()
        z = np.exp(log_y)
        z = np.maximum(z, 1e-300)
        z /= z.sum()

        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = z + momentum * (z - y)
            x_new = np.maximum(x_new, 1e-300)
            x_new /= x_new.sum()
        else:
            x_new = z.copy()

        c_new = fast_conv_self(x_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > stall_limit:
            seed = int((time.monotonic() * 7919) % (2**31))
            rng = np.random.default_rng(seed)
            sigma = 0.005 + 0.015 * (stall_limit / 5000.0)
            pert = rng.normal(0, sigma, n) * best_x
            x_new = normalize_nonneg(best_x + pert)
            y = x_new.copy()
            stall = 0
            stall_limit = max(int(stall_limit * 0.7), 500)

        y = x_new
        x = x_new
        iter_count += 1

        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA iter={iter_count} ({elapsed:.0f}s): M={best_m:.10f}, beta={beta:.1f}, stall={stall}")

    return best_x, best_m


def lp_fw_optimize(a0, time_limit, solver, init_K=300, max_K=4000, use_away=True, verbose=True):
    """LP-based Frank-Wolfe with dynamic constraints and optional away-steps."""
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    ac = fast_conv_self(a)
    best_m = ac.max()
    t0 = time.monotonic()
    size = 2 * n - 1

    y = a.copy()
    x = a.copy()

    ac_y = fast_conv_self(y)
    M_init = ac_y.max()
    init_idx = np.argsort(ac_y)[::-1][:min(init_K, size)]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0

    while True:
        if time.monotonic() - t0 > time_limit:
            break
        if iter_count > 500000:
            break

        ac_y = fast_conv_self(y)
        M_curr = ac_y.max()

        if M_curr < best_m - 1e-13:
            best_m = M_curr
            best_a = y.copy()

        ac_x = fast_conv_self(x)
        if ac_x.max() < best_m - 1e-13:
            best_m = ac_x.max()
            best_a = x.copy()

        # Dynamically grow constraint set: add lags within 97% of M
        close_mask = ac_y > M_curr * 0.97
        close_lags = np.where(close_mask)[0]
        for k in close_lags:
            constraint_set.add(int(k))

        constraint_list = sorted(constraint_set)

        # Prune if too large
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            constraint_set = set(np.argsort(ac_y)[::-1][:init_K].astype(int))
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violated constraints
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_yg > M_curr + 1e-7)[0]
        for k in violations[:60]:
            constraint_set.add(int(k))

        # Frank-Wolfe step (with Nesterov)
        alpha = 2.0 / (iter_count + 3.0)

        if use_away and iter_count > 5:
            # Away-step: compute max away fraction
            support_y = np.where(y > 1e-12)[0]
            away_mask = (y[support_y] > 1e-12) & (g_dir[support_y] < 1e-10)
            if np.any(away_mask):
                away_indices = support_y[away_mask]
                max_away = alpha * 0.5
                y_new = (1.0 - alpha - max_away) * y + alpha * g_dir
                y_new[away_indices] -= max_away * y[away_indices]
                y_new = np.maximum(y_new, 0.0)
                sy = y_new.sum()
                if sy > 1e-15:
                    y_new = y_new / sy
                else:
                    y_new = (1.0 - alpha) * y + alpha * g_dir
                    y_new = np.maximum(y_new, 0.0)
                    y_new /= y_new.sum()
            else:
                y_new = (1.0 - alpha) * y + alpha * g_dir
                y_new = np.maximum(y_new, 0.0)
                y_new /= y_new.sum()
        else:
            y_new = (1.0 - alpha) * y + alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / sy

        if iter_count == 0:
            x_new = y_new.copy()
        else:
            beta_t = float(iter_count) / (iter_count + 3.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): M={best_m:.10f}, |S|={len(constraint_set)}")

    return best_a, best_m


def proj_grad_descend(a0, time_limit, beta=50.0, top_k=100, verbose=False):
    """Projected gradient descent with line search on the softmax-smoothed objective."""
    n = len(a0)
    x = normalize_nonneg(a0)
    best_x = x.copy()
    best_m = fast_conv_self(x).max()
    t0 = time.monotonic()
    lr = 0.1

    while time.monotonic() - t0 < time_limit:
        grad, c, w = get_softmax_grad(x, beta, top_k)
        # Gradient step
        x_trial = x - lr * grad
        x_trial = proj_simplex(x_trial)

        # Simple backtracking
        c_trial = fast_conv_self(x_trial)
        m_trial = c_trial.max()
        if m_trial > best_m * 1.01:
            lr *= 0.5
            x_trial = x - lr * grad
            x_trial = proj_simplex(x_trial)

        x = x_trial
        c_new = fast_conv_self(x)
        m_new = c_new.max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_x = x.copy()
        else:
            lr *= 0.8

    return best_x, best_m


def find_solver():
    """Find a working CVXPY solver."""
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    print("  Solver: ECOS (fallback)")
    return cp.ECOS


def interpolate_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize_nonneg(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    return normalize_nonneg(f_new)
