def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def nesterov_softmax_grad(a0, time_limit, top_k=50, temp=30.0, c0=3.0, verbose=True):
    """Nesterov-accelerated gradient descent on soft-max of top autocorrelation peaks."""
    n = len(a0)
    x = normalize(a0).copy()
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    x_prev = x.copy()
    stall = 0
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 5000:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute autocorrelation
        ac = fast_autocorr(y)
        n_ac = len(ac)
        
        # Soft-max weights on top peaks
        k_max = min(top_k, n_ac)
        idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[idx]
        w = np.exp(temp * (top_vals - top_vals[0]))
        w = w / w.sum()
        
        # Compute gradient
        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            if ki >= n:
                continue
            # grad[j] += wi * (a[j+ki] if j+ki < n) + wi * (a[j-ki] if j >= ki)
            # Vectorized:
            # For j in [0, n-ki): grad[j] += wi * y[j+ki]
            # For j in [ki, n): grad[j] += wi * y[j-ki]
            if ki == 0:
                grad += 2.0 * wi * y
            else:
                grad[:n-ki] += wi * y[ki:]
                grad[ki:] += wi * y[:n-ki]
        
        eta_t = c0 / (np.sqrt(max(t, 1)) + 1.0)
        log_x = np.log(y) - eta_t * grad
        max_log = log_x.max()
        x_new = np.exp(log_x - max_log)
        x_new = np.maximum(x_new, 1e-30)
        x_new = x_new / x_new.sum()
        
        x_prev = x
        x = x_new
        
        val = obj_val(x)
        if val < best_val - 1e-13:
            best_val = val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    Nesterov t=" + str(t) + " (" + str(round(elapsed,1)) + "s) J=" + repr(best_val) + " stall=" + str(stall))
    
    return best_a, best_val

def lp_fw(a0, time_limit, solver, init_K=500, max_K=4000, add_per=50, verbose=True):
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:init_K])
    no_improve = 0
    stall_limit = 1500
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        clist = sorted(active_set)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = sorted(active_set)
        
        K = len(clist)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs = np.zeros((K, n), dtype=np.float64)
        for jj, k in enumerate(clist):
            k_int = int(k)
            if k_int >= n:
                continue
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        lp_ok = False
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                lp_ok = True
        except Exception:
            if len(active_set) > init_K:
                active_set = set(int(clist[:init_K]))
                clist = sorted(active_set)
        
        if not lp_ok:
            if no_improve > stall_limit // 2:
                break
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Check violations and add to active set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per]]
            for k in top_v:
                int_k = int(k)
                if int_k < 2*n:
                    active_set.add(int_k)
        
        # Line search with ternary + golden section
        best_alpha = 2.0 / (t + 2.0)
        best_trial = fast_autocorr(np.maximum((1 - best_alpha) * y + best_alpha * g_dir, 0) / max(np.maximum((1 - best_alpha) * y + best_alpha * g_dir, 0).sum(), 1e-15)).max()
        
        # Golden section
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = 1.0
        
        def trial_obj(alpha):
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                return 1e10
            trial = trial / ts
            return fast_autocorr(trial).max()
        
        # Coarse search
        for alpha in [0.02, 0.05, 0.1, 0.2, 0.4, 0.6, 0.8, 0.95, 1.0]:
            Mt = trial_obj(alpha)
            if Mt < best_trial - 1e-15:
                best_trial = Mt
                best_alpha = alpha
        
        # Fine search around best_alpha
        center = best_alpha
        span = 0.2
        for _ in range(6):
            mid1 = center - span * gr
            mid2 = center + span * (1 - gr)
            m1 = trial_obj(max(0, mid1))
            m2 = trial_obj(min(1, mid2))
            if m1 < m2:
                center = center - span * gr * 0.5
                span *= gr
                if m1 < best_trial:
                    best_trial = m1
                    best_alpha = max(0, mid1)
            else:
                center = center + span * (1-gr) * 0.5
                span *= gr
                if m2 < best_trial:
                    best_trial = m2
                    best_alpha = min(1, mid2)
        
        y = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        if verbose and t % 30 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t=" + str(t) + " (" + str(round(elapsed,1)) + "s) J=" + repr(2*n*best_m) + " |S|=" + str(len(active_set)))
    
    return best_a, 2.0 * n * best_m

def aggressive_deflation(a0, time_limit, n_targets=8, verbose=True):
    """More aggressive deflation with backtracking and multiple lags."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    for round_i in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        targets = np.argsort(ac)[::-1][:n_targets]
        improved = False
        
        for k in targets:
            ki = int(k)
            if ki >= n:
                continue
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            
            for _ in range(5):
                if contrib.max() < 1e-20:
                    break
                src = int(idx[np.argmax(contrib)])
                src_val = a[src]
                if src_val < 1e-12:
                    break
                
                # Score destinations
                scores = np.zeros(n)
                for k2 in targets:
                    k2i = int(k2)
                    if k2i >= n:
                        continue
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                        scores[k2i - idx2] += a[idx2]
                scores[src] = 1e10
                
                # Try multiple destinations
                top_dsts = np.argsort(scores)[:20]
                found = False
                for dst in top_dsts:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.01, 0.005, 0.002, 0.001]:
                        amt = src_val * frac
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            break
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < best_m - 1e-15:
                            a = a_t
                            best_m = Mt
                            best_a = a.copy()
                            improved = True
                            found = True
                            break
                    if found:
                        break
                
                if not found:
                    break
            
            if improved:
                break
        
        if not improved:
            # Random perturbation
            if round_i % 50 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.001, 0.01)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
        
        if verbose and round_i % 100 == 0:
            print("    Deflate r=" + str(round_i) + " J=" + repr(2*n*best_m))
    
    return best_a, 2.0 * n * best_m

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated soft-max gradient descent (fast bulk optimization)
2. LP-Frank-Wolfe with adaptive constraints (precise convergence)
3. Aggressive deflation with backtracking (targeted peak reduction)

import numpy as np

import cvxpy as cp

_MODULE_SOLVER = None

def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def nesterov_gd_trial(y, g_dir, alpha):
    trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)

    trial = trial / ts
    return fast_autocorr(trial).max()

def nesterov_softmax_grad(a0, time_limit, top_k=50, temp_start=20.0, temp_end=80.0, c0=3.0, verbose=True):
    n = len(a0)
    x = normalize(a0).copy()
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    x_prev = x.copy()
    stall = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 6000:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        ac = fast_autocorr(y)
        n_ac = len(ac)

        k_max = min(top_k, n_ac)
        idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[idx]
        # Adaptive temperature: start focused, gradually widen
        frac = t / max(1, int(time_limit * 200))
        temp = temp_start + (temp_end - temp_start) * min(1.0, frac)
        w = np.exp(temp * (top_vals - top_vals[0]))
        w = w / w.sum()

        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            if ki >= n:
                continue
            if ki == 0:
                grad += 2.0 * wi * y
            else:
                grad[:n - ki] += wi * y[ki:]
                grad[ki:] += wi * y[:n - ki]

        eta_t = c0 / (np.sqrt(max(t, 1)) + 1.0)
        log_x = np.log(y) - eta_t * grad
        max_log = log_x.max()
        x_new = np.exp(log_x - max_log)
        x_new = np.maximum(x_new, 1e-30)
        x_new = x_new / x_new.sum()

        x_prev = x
        x = x_new

        val = obj_val(x)
        if val < best_val - 1e-13:
            best_val = val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

            elapsed = time.monotonic() - t0
            print("    Nesterov t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(best_val) + " temp=" + repr(round(temp, 1)) + " stall=" + str(stall))

def lp_fw_line_search(y, g_dir, t):
    """Golden section + coarse grid line search for LP-FW."""
    best_alpha = 2.0 / (t + 2.0)
    best_trial = nesterov_gd_trial(y, g_dir, best_alpha)

    for alpha in [0.01, 0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        Mt = nesterov_gd_trial(y, g_dir, alpha)
        if Mt < best_trial - 1e-16:
            best_trial = Mt
            best_alpha = alpha

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    center = best_alpha
    span = 0.15
    for _ in range(7):
        mid1 = center - span * gr
        mid2 = center + span * (1.0 - gr)
        mid1 = max(0.0, min(1.0, mid1))
        mid2 = max(0.0, min(1.0, mid2))
        m1 = nesterov_gd_trial(y, g_dir, mid1)
        m2 = nesterov_gd_trial(y, g_dir, mid2)
        if m1 < best_trial:
            best_trial = m1
            best_alpha = mid1
        if m2 < best_trial:
            best_trial = m2
            best_alpha = mid2
        if m1 < m2:
            center = mid1 + (mid2 - mid1) * (1.0 - gr)
        else:
            center = mid1 + (mid2 - mid1) * gr
        span *= gr

    return best_alpha, best_trial

            if src_val < 1e-12:
                continue

            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                if k2i >= n:
                    continue
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            candidates = np.argsort(scores)[:40]
            for dst in candidates:
                dst = int(dst)
                if dst == src:
                    continue
                for frac in fracs:
                    amt = src_val * frac
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        break
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-15:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 150:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.001, 0.005)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
        else:
            stall = 0

        if verbose and stall % 300 == 0:
            print("    Coord (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(2 * n * best_m) + " stall=" + str(stall))

    return best_a, 2.0 * n * best_m

def diverse_initialization(n, parent):
    """Generate and evaluate diverse initial candidates."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []

    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(normalize((i_arr + 1.0) ** p))
        candidates.append(normalize((n - i_arr).astype(np.float64) ** p + 1e-10))

    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sg in [0.08, 0.12, 0.2, 0.3]:
            candidates.append(normalize(np.exp(-((x - c) ** 2) / (2 * sg * sg))))

    for c1 in [0.2, 0.3, 0.4]:
        for c2 in [0.6, 0.7, 0.8]:
            for sg in [0.08, 0.15, 0.2]:
                cands = np.exp(-((x - c1) ** 2) / (2 * sg * sg)) + np.exp(-((x - c2) ** 2) / (2 * sg * sg))
                candidates.append(normalize(cands))

    for f in [1, 2, 3, 5, 7]:
        cands = np.maximum(1.0 + np.cos(2 * np.pi * f * x), 0.01)
        candidates.append(normalize(cands))

    for p in [1.0, 2.0, 3.0, 5.0]:
        candidates.append(normalize(x ** p + (1.0 - x) ** p + 1e-10))

    candidates.append(normalize(np.sin(np.pi * x) ** 2))
    candidates.append(normalize(np.sin(np.pi * x) ** 4))
    candidates.append(normalize(np.sin(np.pi * x) ** 8))

    if parent is not None:
        bi = normalize(parent)
        for w in [0.2, 0.4, 0.6, 0.8]:
            for cand in candidates[:8]:
                candidates.append(normalize(w * bi + (1.0 - w) * cand))

    return candidates

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1700.0)
    solver = get_solver()
    print("Budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent_arr = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent_arr = p
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            print("Parent: n=" + str(len(p)) + " score=" + repr(best_score))
    except Exception:
        try:
            p = np.asarray(height_sequence_1, dtype=np.float64)
            p = np.clip(p, 0.0, 1000.0)
            if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
                parent_arr = p
                best_seq = normalize(p)
                best_score = eval_safe(best_seq)
                print("Parent: n=" + str(len(p)) + " score=" + repr(best_score))
        except Exception:
            parent_arr = None

    if best_seq is None:
        n_init = 8000
        x = np.linspace(0, 1, n_init)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)
        print("Default init n=" + str(n_init) + " score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: Diverse initialization (30s)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Diverse Init (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.025, 35.0)
    if t1 > 5:
        cands = diverse_initialization(n0, parent_arr)
        bi = normalize(best_seq)
        for cand in cands:
            if time.monotonic() - t_global > time_budget - t1:
                break
            v = obj_val(cand)
            if v < 2 * n0 * best_m_val(cand):
                pass
            sc_local = 2.0 * n0 * fast_autocorr(cand).max()
            if sc_local < best_score:
                best_score = sc_local
                best_seq = cand.copy()
                print("  Diversified: J=" + repr(best_score))
        # Also try blends
        rng = np.random.default_rng(42)
        for _ in range(min(20, len(cands))):
            if time.monotonic() - t_global > time_budget - t1:
                break
            cand = cands[int(rng.integers(len(cands)))]
            for w in [0.3, 0.5, 0.7]:
                blend = normalize(w * bi + (1.0 - w) * cand)
                sc_local = 2.0 * n0 * fast_autocorr(blend).max()
                if sc_local < best_score:
                    best_score = sc_local
                    best_seq = blend.copy()

    # Phase 2: Nesterov GD on current length (300s)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P2: Nesterov GD (n=" + str(n_cur) + ") ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_out, v_out = nesterov_softmax_grad(normalize(best_seq), t2, top_k=60, temp_start=15.0, temp_end=60.0, c0=3.0, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  Nesterov improved: " + repr(best_score))

    # Phase 3: LP-FW (400s)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P3: LP-FW (n=" + str(n_cur) + ") ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 50:

        sc = eval_safe(a_out)

            best_seq = a_out
            print("  LP-FW improved: " + repr(best_score))
