def compute_subgradient(a, k_star):
    """Compute exact subgradient of C_k(a) = conv(a,a)[k] at point a."""
    n = len(a)
    # dC_k/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    grad = np.zeros(n)
    # First term: a[k-j] for valid j
    # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
    j_start1 = max(0, k_star - n + 1)
    j_end1 = min(n, k_star + 1)
    if j_start1 < j_end1:
        grad[j_start1:j_end1] += a[k_star - np.arange(j_start1, j_end1)]
    # Second term: a[j-k] for valid j
    # a[j-k] is valid when 0 <= j-k < n, i.e., k <= j <= k+n-1
    j_start2 = max(0, k_star)
    j_end2 = min(n, k_star + n)
    if j_start2 < j_end2:
        grad[j_start2:j_end2] += a[np.arange(j_start2, j_end2) - k_star]
    return grad

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def symmetrize(a):
    return (a + a[::-1]) / 2.0


def evaluate_safe(a):
    try:

def softmax_gradient_fft(a, beta, top_k=None):
    """Compute softmax-weighted subgradient of max_k conv(a,a)[k] using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]

    if top_k is not None and top_k < size:
        top_idx = np.argsort(c)[::-1][:top_k]
        c_top = c[top_idx]
        c_max = c_top.max()
        exp_c = np.zeros(size)
        exp_c[top_idx] = np.exp(beta * (c_top - c_max))
        w = exp_c / exp_c.sum()
    else:
        c_max = c.max()
        exp_c = np.exp(beta * (c - c_max))
        w = exp_c / exp_c.sum()

    # grad[j] = sum_k w[k] * (a[k-j] + a[j-k]) for valid indices
    # First term: sum_k w[k] * a[k-j] = cross-correlation of w and a
    # Second term: sum_k w[k] * a[j-k] = convolution of w and a (with appropriate indexing)
    
    # Both terms can be computed via FFT
    W = np.fft.rfft(w, fft_size)
    
    # First term: (w cross-corr a)[j] = sum_k w[k] * a[k-j]
    # This equals (w_rev conv a)[j] where w_rev is w reversed
    w_rev = w[::-1]
    Wr_rev = np.fft.rfft(w_rev, fft_size)
    term1 = np.fft.irfft(Wr_rev * A, fft_size)[:size]
    
    # Second term: (w conv a)[j] = sum_k w[k] * a[j-k]  
    term2 = np.fft.irfft(W * A, fft_size)[:size]
    
    # We only need grad for indices 0..n-1
    # The convolution results are indexed 0..2n-2
    grad = term1[:n] + term2[:n]
    
    return 2.0 * grad, c, w


def subgradient_mirror_descent(a0, time_limit, beta_start=5.0, beta_end=200.0,
                                verbose=True, symmetrize_flag=True):
    """
    Mirror descent (multiplicative weights) with softmax-smoothed subgradient.
    Uses AdaGrad-style adaptive learning rates.
    """

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_threshold = 2000
    
    # AdaGrad accumulator
    G2 = np.zeros(n)
    tau = 1e-6
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        # Adaptive beta
        progress = min(iter_count / max(int(time_limit * 50), 1), 1.0)
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.7)
        
        # Compute gradient
        grad, c, w = softmax_gradient_fft(a, beta, top_k=min(300, 2 * n - 1))
        
        # Track best
        m_curr = c.max()
        if m_curr < best_m - 1e-15:
            best_m = m_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # AdaGrad learning rate
        G2 += grad ** 2
        eta_vec = np.sqrt(tau) / (np.sqrt(G2) + 1e-8)
        
        # Global scaling to prevent too-large steps
        max_eta = 5.0 / (np.sqrt(iter_count + 1) + 1.0)
        eta_scale = min(1.0, max_eta / (np.max(eta_vec) + 1e-15))
        eta_vec *= eta_scale
        
        # Multiplicative weights update
        log_a = np.log(np.maximum(a, 1e-30)) - eta_vec * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        
        if symmetrize_flag:
            a_new = symmetrize(a_new)
        
        a_new = a_new / a_new.sum()
        a = a_new
        iter_count += 1
        
        if verbose and iter_count % 5000 == 0:
            J = 2.0 * n * best_m
            print(f"    MD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, stall={stall}")

                a = norm_sum1(symmetrize(a))
            G2 = np.zeros(n)
            stall = 0

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, symmetrize_flag=True):
    """
    LP-based Frank-Wolfe with Nesterov acceleration.

    a = norm_sum1(a0)

    y = a.copy()
    x = a.copy()

    ac = fast_autocorr(y)

    constraint_set = set(int(k) for k in init_idx)
    
    max_K = min(3000, 2 * n - 1)
    failed_solves = 0
    last_improve = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            last_improve = t
        
        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()
            last_improve = t
        
        # Constraint management
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]

        g_dir = g_val / g_sum

def parametric_init(n, best_ref=None):
    """Try diverse parametric families and return the best initial candidate."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')
    
    candidates = []
    
    # Power laws
    for p in [0.3, 0.5, 0.7, 0.8, 1.0, 1.2, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr + 1.0) ** p)
        # Symmetric
        candidates.append(((i_arr + 1.0) * (n - i_arr + 1.0)) ** (p / 2.0))
    
    # Beta-like distributions
    for alpha in [0.5, 0.8, 1.0, 1.5, 2.0]:
        for beta_p in [0.5, 0.8, 1.0, 1.5, 2.0]:
            candidates.append(np.maximum(i_arr + 0.5, 0.1) ** alpha *
                             np.maximum(n - i_arr - 0.5, 0.1) ** beta_p)
    
    # Gaussian bumps
    for c in [0.3, 0.4, 0.45, 0.5, 0.55, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    
    # Exponential
    for rate in [1, 2, 3, 5, 7]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1.0)))
        candidates.append(np.exp(-rate * np.abs(x - 0.5)))
    
    # Sine-based
    for freq in [1, 2, 3, 4]:
        candidates.append(np.maximum(np.sin(np.pi * freq * x), 0.0) ** 2)
    candidates.append(np.sin(np.pi * x) ** 2)
    candidates.append(np.sin(np.pi * x) ** 4)
    
    # Double bump
    for w in [0.3, 0.4, 0.5]:
        candidates.append(np.exp(-((x - 0.3) ** 2) / 0.02) + w * np.exp(-((x - 0.7) ** 2) / 0.02))
    
    # If we have a reference
    if best_ref is not None and len(best_ref) == n:
        ref_norm = norm_sum1(best_ref)
        for w in [0.3, 0.5, 0.7, 0.9]:
            candidates.append(w * ref_norm + (1.0 - w) * norm_sum1(np.sin(np.pi * x) ** 2))
    
    for a_cand in candidates:
        a_cand = np.maximum(a_cand, 0.0)
        if a_cand.sum() < 1e-10:
            continue
        a_norm = a_cand / a_cand.sum()
        ac = fast_autocorr(a_norm)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_a = a_norm
    
    return best_a, best_val

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
    
    # Determine starting length
    if parent is not None:
        n_start = len(parent)
    else:
        n_start = 5000
        x = np.linspace(0, 1, n_start)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_start}, score={best_score:.6f}")
    
    # === Phase 1: Parametric search ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 30.0)
    print(f"\n=== Phase 1: Parametric init (n={n_start}, t={t1:.0f}s) ===")
    if t1 > 5:
        a_param, val_param = parametric_init(n_start, best_seq)
        check_and_update(a_param, "parametric")
    
    # === Phase 2: Mirror descent (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: Mirror descent (n={n_best}, t={t2:.0f}s) ===")
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_mirror_descent(a_in, t2, beta_start=5.0, beta_end=150.0,
                                               verbose=True, symmetrize_flag=True)
        check_and_update(a_out, "md_main")
    
    # === Phase 3: LP-FW polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 3: LP-FW polish (n={n_best}, t={t3:.0f}s) ===")
    if t3 > 30 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True, symmetrize_flag=True)
        check_and_update(a_out, "lp_polish1")
    
    # === Phase 4: Mirror descent with higher beta ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 4: MD high-beta (n={n_best}, t={t4:.0f}s) ===")
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_mirror_descent(a_in, t4, beta_start=30.0, beta_end=400.0,
                                               verbose=True, symmetrize_flag=True)
        check_and_update(a_out, "md_high_beta")
    
    # === Phase 5: LP-FW second pass ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 5: LP-FW second (n={n_best}, t={t5:.0f}s) ===")
    if t5 > 30 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True, symmetrize_flag=True)
        check_and_update(a_out, "lp_polish2")
    
    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")
    
    length_candidates = []
    for delta in [-1000, -500, 500, 1000, 1500]:
        new_n = n_best + delta
        if 3000 <= new_n <= 9000:
            length_candidates.append(new_n)
    for rn in [4000, 5000, 6000, 7000, 8000]:
        if rn != n_best and 3000 <= rn <= 9000:
            length_candidates.append(rn)
    length_candidates = sorted(set(length_candidates))
    
    for new_n in length_candidates[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)
        
        # Quick MD
        t6a = min(remaining * 0.3, 120.0)
        if t6a > 15:
            a_out, _ = subgradient_mirror_descent(a_long, t6a, beta_start=10.0, beta_end=100.0,
                                                   verbose=False, symmetrize_flag=True)
            check_and_update(a_out, f"md_n{new_n}")
        
        # Quick LP if time permits
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 7000:
            t6b = min(remaining * 0.4, 150.0)
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t6b, solver, verbose=False, symmetrize_flag=True)
            check_and_update(a_out, f"lp_n{new_n}")
    
    # === Phase 7: Final LP polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP (n={n_best}, remaining={remaining:.0f}s) ===")
    if remaining > 40 and n_best <= 8000:
        t7 = min(remaining * 0.4, 200.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=True, symmetrize_flag=True)
            check_and_update(a_out, "lp_final")
    
    # === Phase 8: Final MD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t8 = min(remaining * 0.5, 100.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = subgradient_mirror_descent(a_in, t8, beta_start=50.0, beta_end=500.0,
                                                   verbose=False, symmetrize_flag=True)
            check_and_update(a_out, "md_final")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
