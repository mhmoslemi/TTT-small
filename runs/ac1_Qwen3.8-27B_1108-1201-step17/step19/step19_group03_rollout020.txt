def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_seq(a):

def norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_fw_step(a, M, K, top_k_idx):
    """One LP-FW step: solve LP to find best direction."""
    n = len(a)
    K_sel = min(K, len(top_k_idx))
    binding = top_k_idx[:K_sel]
    
    # Build constraint matrix T (K_sel × n)
    # T[k,j] = a[k_j - j] if 0 <= k_j - j < n else 0

    if K_sel * n < 50_000_000:  # Less than 50M elements
        T = np.zeros((K_sel, n))
        for ii, k in enumerate(binding):
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n, ki + 1)
            for j in range(js, je):
                T[ii, j] = a[ki - j]
    else:
        # Use sparse
        from scipy.sparse import csr_matrix
        rows, cols, vals = [], [], []
        for ii, k in enumerate(binding):
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n, ki + 1)
            for j in range(js, je):
                rows.append(ii)
                cols.append(j)
                vals.append(a[ki - j])
        T = csr_matrix((vals, (rows, cols)), shape=(K_sel, n))
    
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    cons = (T, np.full(K_sel, M))
    res = linprog(c, A_ub=T, b_ub=np.full(K_sel, M), bounds=[(0, None)]*n, method='highs')

    return np.maximum(res.x, 0)

    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    K = K_init
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()
        top_idx = np.argsort(ac)[::-1]
        
        b = lp_fw_step(a, M, K, top_idx)
        if b is None:
            K = min(K * 2, K_max)
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b / b_sum
        
        # Line search
        alpha_0 = 2.0 / (t + 2.0)
        best_alpha = alpha_0
        best_trial = fast_autocorr((1-alpha_0)*a + alpha_0*g).max()
        
        for alpha in [alpha_0 * 0.5, alpha_0, alpha_0 * 1.5, alpha_0 * 2.0, 0.5, 0.7, 1.0]:
            alpha = min(1.0, max(0.01, alpha))
            trial = (1 - alpha) * a + alpha * g
            trial = norm1(trial)
            m = fast_autocorr(trial).max()
            if m < best_trial:
                best_trial = m
                best_alpha = alpha
        
        a = (1 - best_alpha) * a + best_alpha * g
        a = norm1(a)
        
        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()
        
        t += 1
        if t % 10 == 0:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_seq(a):

def norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_sparse(a, binding_idx):
    n = len(a)
    K = len(binding_idx)
    rows = []
    cols = []
    vals = []
    for ii, k in enumerate(binding_idx):
        ki = int(k)
        js = max(0, ki - n + 1)
        je = min(n, ki + 1)
        for j in range(js, je):
            v = a[ki - j]
            if v > 1e-20:
                rows.append(ii)
                cols.append(j)
                vals.append(v)
    return csr_matrix((vals, (rows, cols)), shape=(K, n))


def lp_fw_direction(a, M, binding_idx):
    n = len(a)
    K = len(binding_idx)
    if K == 0:
        return None
    T = build_conv_matrix_sparse(a, binding_idx)
    c = -np.ones(n)
    b_ub = np.full(K, M)
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 30})
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_franks(a0, time_limit, K_init=200, K_max=8000, verbose=True):
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    K = K_init
    t = 0
    stall = 0
    max_stall = 100

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        top_idx = np.argsort(ac)[::-1]

        K_sel = min(K, 2 * n - 1)
        binding = top_idx[:K_sel]

        b = lp_fw_direction(a, M, binding)
        if b is None:
            K = min(K * 2, K_max)
            t += 1
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b / b_sum

        alpha_0 = 2.0 / (t + 2.0)
        candidates = [min(1.0, alpha_0 * f) for f in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]]
        candidates.append(0.5)
        candidates.append(0.8)
        candidates = sorted(set(min(1.0, max(0.01, x)) for x in candidates))

        best_alpha = candidates[0]
        best_trial = fast_autocorr(norm1((1 - best_alpha) * a + best_alpha * g)).max()
        for alpha in candidates:
            trial = norm1((1 - alpha) * a + best_alpha * 0 + alpha * g)
            m = fast_autocorr(trial).max()
            if m < best_trial:
                best_trial = m
                best_alpha = alpha

        a = norm1((1 - best_alpha) * a + best_alpha * g)
        m_new = fast_autocorr(a).max()

        if m_new < best_m - 1e-18:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        t += 1
        if t % 5 == 0:
            K = min(K + 100, K_max)

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, M={M:.2e}, stall={stall}")

    return best_a, best_m


def md_optimize(a0, time_limit, top_k=80, beta=6.0, c0=4.0, restart_every=3000,
                verbose=False, rng_seed=42):

    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        ac = fast_autocorr(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every // 2:
            cur_beta = max(cur_beta * 0.95, 2.0)
            cur_top_k = min(cur_top_k + 10, 300)

        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.05, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 20, 500)
            cur_beta = max(cur_beta - 0.5, 2.0)

            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

def pocs_optimize(a0, time_limit, n_trials=4, verbose=False, rng_seed=0):
    n = len(a0)
    best_a = a0.copy()
    best_m = fast_autocorr(a0).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit - 5:
            break
        t_trial = min((time_limit - (time.monotonic() - t0)) / (n_trials - trial), 60.0)
        if t_trial < 5:
            break

        rng = np.random.default_rng(rng_seed + trial)
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        A_spec = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
        A_full = np.zeros(fft_size, dtype=np.complex128)
        A_full[:fft_size // 2 + 1] = A_spec
        for i in range(1, fft_size // 2):
            A_full[fft_size - i] = np.conj(A_spec[i])
        a = np.fft.irfft(A_full, fft_size)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s

        t_pocs = time.monotonic()
        n_iters = max(100, int(t_trial * 200))
        for it in range(n_iters):
            if time.monotonic() - t_pocs > t_trial:
                break
            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            alpha = 0.05 + 0.3 * min(1.0, it / 200.0)
            new_mag = (1 - alpha) * mag + alpha * mag_mean
            new_mag = np.maximum(new_mag, 1e-10)
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s2 = a_new.sum()
            if s2 < 1e-15:
                continue
            a_new = a_new / s2
            a = a_new

        M = fast_autocorr(a).max()
        if M < best_m:
            best_m = M
            best_a = a.copy()
            if verbose:
                print(f"    POCS trial {trial}: J={2.0*n*best_m:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):

            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            for _ in range(5):
                dst = int(rng.integers(n))
                if dst == src:
                    continue

                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:

                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:

        if stall > 20:

                a = np.maximum(a, 0.0)

                M_new = fast_autocorr(a).max()
                if M_new < best_m:

                    best_a = a.copy()

            stall += 1

        if verbose and stall % 20 == 0:

def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm1(a)


def make_cosine(n, alpha=0.5):
    x = np.arange(n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * x / n)
    a = np.maximum(a, 0.0)
    return norm1(a)


def make_fourier(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.arange(n) / n
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.1 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.cos(2.0 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: MD R2 (n={n_best}) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 15:
        a_in = norm1(best_seq)
        a_out, _ = md_optimize(a_in, t6, top_k=120, beta=4.0, c0=6.0,
                               restart_every=2000, verbose=False, rng_seed=200)
        try_update(a_out, "md_r2")

    # Phase 7: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Coord (n={n_best}) ===")
    t7 = min(remaining * 0.10, 120.0)
    if t7 > 10:
        a_in = norm1(best_seq)
        a_out, _ = coordinate_polish(a_in, t7, verbose=True, rng_seed=456)
        try_update(a_out, "coord")

    # Phase 8: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Multi-length (base n={n_best}) ===")

    trial_lengths = []
    for delta in [200, -200, 500, -500, 1000, -1000, 1500, -1500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 12000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_seq(best_seq, new_n)

        t8a = min(remaining * 0.3, 100.0)
        if t8a > 15:
            a_out, _ = md_optimize(a_long, t8a, top_k=60, beta=5.0, c0=4.0,
                                   restart_every=3000, rng_seed=new_n % 1000)
            try_update(a_out, f"md_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 8000:
            t8b = min(remaining * 0.35, 100.0)
            if t8b > 15:
                a_in = interpolate_seq(best_seq, new_n)
                K_init = min(400, 2 * new_n - 1)
                K_max = min(6000, 2 * new_n - 1)
                a_out, _ = lp_franks(a_in, t8b, K_init=K_init, K_max=K_max, verbose=False)
                try_update(a_out, f"lp_n{new_n}")

    # Phase 9: Final LP-FW on current best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final LP-FW (n={n_best}) ===")
    if remaining > 30:
        t9 = min(remaining * 0.4, 200.0)
        if t9 > 15:
            a_in = norm1(best_seq)
            K_init = min(1000, 2 * n_best - 1)
            K_max = min(15000, 2 * n_best - 1)
            a_out, _ = lp_franks(a_in, t9, K_init=K_init, K_max=K_max, verbose=True)
            try_update(a_out, "lp_final")

    # Phase 10: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 10: Final polish (n={n_best}) ===")
        t10 = min(remaining * 0.8, 100.0)
        if t10 > 10:
            a_in = norm1(best_seq)
            a_out, _ = coordinate_polish(a_in, t10, verbose=False, rng_seed=789)
            try_update(a_out, "coord_final")

    # Final validation
    if best_seq is not None:
        fs = eval_seq(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 6200
        best_seq = make_u_shape(n_fb, 2.5)
        best_score = eval_seq(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
