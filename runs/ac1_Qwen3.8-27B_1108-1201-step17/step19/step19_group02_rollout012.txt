def norm_sum1(a):

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)

        # Line search over alphas
        default_alpha = 2.0 / (iter_count + 2.0)
        alphas = []
        for mult in [0.25, 0.5, 0.75, 1.0, 1.5, 2.0, 3.0]:
            alphas.append(max(0.01, min(1.0, default_alpha * mult)))
        alphas.append(0.05)
        alphas.append(0.1)
        alphas.append(0.3)
        alphas = list(set(alphas))
        alphas.sort(reverse=True)

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 5:
            y_new = g_dir.copy()
        else:
            y_new = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", |S|=" + str(len(active_set)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", iters=" + str(iter_count))
    return best_a, best_m


def peak_shaving_cd(a0, time_limit, rng_seed=42, verbose=False):
    """Targeted coordinate descent that specifically attacks the top convolution peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 1200
    step_scales = [0.001, 0.003, 0.005, 0.01, 0.02, 0.05, 0.1]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        # Focus on the top peaks
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        M_curr = ac[top_k[0]]

        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            # Find the most contributing pair
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score destination: where would adding mass cause least additional peak?
            dst_scores = np.zeros(n)
            for k2 in top_k[:12]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    # If we add mass at position j, it affects conv[k2] through:
                    # a[j] * a[k2-j] + a[k2-j] * a[j] = 2*a[j]*a[k2-j] (if j != k2-j)
                    # or a[j]^2 if j == k2-j
                    # The contribution to conv[k2] from position j is a[k2-j] if j != k2-j
                    # Actually, the derivative of conv[k2] w.r.t. a[j] is a[k2-j] (for the full conv)
                    # But since we're doing a*a, the "gradient" at position j for conv[k2] is a[k2-j]
                    # (because d/dx[sum a[i]*a[k-i]] = a[k-j] + a[k-j] = 2*a[k-j] if j != k-j, or 2*a[j] if j=k-j)
                    # For simplicity, use a[k2-j] as the "danger" score
                    valid_j = idx2  # positions that contribute to conv[k2]
                    # For position j, the "danger" is a[k2-j] (how much mass is at the partner position)
                    for j in valid_j:
                        partner = k2i - j
                        if 0 <= partner < n:
                            dst_scores[j] += a[partner]

            # Also check boundary peaks (k=0, k=1, ..., k=5)
            for k2 in range(min(6, n)):
                if a[k] > 0:  # boundary contribution
                    dst_scores[:min(n-k2, n)] += 0  # boundary peaks are a[i]^2 type

            # Exclude positions that are already in the top peaks
            hot = np.zeros(n, dtype=bool)
            for k2 in top_k[:12]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                hot[s2:e2 + 1] = True

            candidates = np.where(~hot)[0]
            if len(candidates) < n // 8:
                candidates = np.arange(n)

            cand_scores = dst_scores[candidates]
            # Add small random perturbation to break ties
            cand_scores += rng.random(len(candidates)) * 1e-6
            dst = int(candidates[np.argmin(cand_scores)])

            if dst == src:
                continue

            # Try transfer
            found = False
            for frac in step_scales:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 300:
                # Make a random small move
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 200 == 0:
                # Perturbation to escape
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakCD (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", stall=" + str(stall))

    return best_a, best_m


def soft_mirror_descent(a0, time_limit, top_k=100, beta=5.0, c0=4.0,
                        restart_every=4000, momentum=0.6, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(7)

    x = a.copy()
    vel = np.zeros(n)
    gamma = c0
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    restart_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        y = x + momentum * vel
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        ac = fast_convolve(y)
        # Compute soft gradient
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * y[k_int - idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        vel = 0.9 * vel + 0.1 * (a_new - x)
        x = a_new

        M_new = fast_convolve(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            restart_count += 1
            noise_level = 0.03 / (1.0 + 0.1 * restart_count)
            pert = rng.normal(0, noise_level, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r < 1e-15:
                x_r = best_a.copy()
            else:
                x_r = x_r / s_r
            x = x_r
            vel = np.zeros(n)
            cur_top_k = min(cur_top_k + 10, 150)
            cur_beta = max(cur_beta - 0.5, 2.0)
            gamma = max(gamma * 0.92, 0.5)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    SSD restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

    return best_a, best_m
