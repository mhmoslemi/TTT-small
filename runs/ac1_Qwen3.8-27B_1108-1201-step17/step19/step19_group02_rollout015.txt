def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        return np.zeros((0, n))
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k = int(constraint_list[jj])
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k - idx]
    return coeffs


def lp_fw(a0, time_limit, solver, initial_K=300, max_K=3000, add_per_iter=40,
          verbose=True, update_every=5):
    """
    LP-based Frank-Wolfe with improved active set management.
    Key improvements:
    - Update active set every `update_every` iterations
    - Use margin-based pruning
    - Backtracking line search for step size
    - Initial exploration with larger active set
    """
    n = len(a0)
    y = norm_sum1(a0)
    best_y = y.copy()
    best_m = fast_convolve(y).max()
    t0 = time.monotonic()

    ac_y = fast_convolve(y)
    # Symmetry: only need to check k in [0, n-1] since conv is symmetric
    active_set = sorted(np.argsort(ac_y[:n])[::-1][:initial_K].tolist())

    no_improve = 0
    max_no_improve = 800
    iter_count = 0
    prev_coeff_matrix = None
    prev_active_set = None

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: stagnation at iter " + str(iter_count))
            break
        iter_count += 1

        ac = fast_convolve(y)
        M_curr = ac[:n].max()  # Only need first half due to symmetry

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_y = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update active set periodically
        if iter_count % update_every == 1:
            # Prune: keep constraints with small margin
            if len(active_set) > initial_K * 2:
                margins = np.array([M_curr - ac[k] for k in active_set])
                keep_count = min(len(active_set), max_K)
                keep = np.argsort(margins)[:keep_count]
                active_set = sorted([active_set[i] for i in keep])

            # Add: top violators of the current point
            top_violations = np.argsort(ac[:n])[::-1][:initial_K]
            new_set = set(active_set)
            for k in top_violations[:add_per_iter]:
                new_set.add(int(k))
            # Also add some random constraints for exploration
            if iter_count % (update_every * 10) == 1:
                rand_ks = np.random.default_rng(iter_count).integers(0, n, size=initial_K // 5)
                for k in rand_ks:
                    new_set.add(int(k))

            if len(new_set) > max_K:
                active_set_full = sorted(new_set)
                margins = np.array([M_curr - ac[k] for k in active_set_full])
                keep = np.argsort(margins)[:max_K]
                active_set = sorted([active_set_full[i] for i in keep])
            else:
                active_set = sorted(new_set)

        K = len(active_set)
        if K == 0:
            break

        # Build constraint matrix (cache if active set didn't change)
        if prev_active_set != active_set or prev_coeff_matrix is None:
            coeffs = build_constraint_matrix(y, active_set, n)
            prev_coeff_matrix = coeffs
            prev_active_set = active_set[:]
        else:
            coeffs = prev_coeff_matrix
            # Need to update coeffs since y changed
            coeffs = build_constraint_matrix(y, active_set, n)
            prev_coeff_matrix = coeffs

        # Solve LP: min M s.t. sum(g)=1, g>=0, coeffs@g <= M
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Fallback: reduce active set
            active_set = active_set[:max(50, K // 2)]
            prev_coeff_matrix = None
            prev_active_set = None
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Backtracking line search
        best_alpha = 0.1
        best_trial_M = M_curr
        alphas_to_try = []
        if iter_count <= 5:
            alphas_to_try = [1.0, 0.8, 0.5, 0.3, 0.1]
        else:
            # Constrained step sizes
            t_base = 2.0 / (iter_count + 2.0)
            alphas_to_try = [
                min(1.0, max(0.01, t_base)),
                min(1.0, max(0.01, t_base * 2.0)),
                min(1.0, max(0.01, t_base * 0.5)),
                min(1.0, max(0.01, 1.0 / (iter_count + 1.0))),
                0.05, 0.1, 0.2, 0.3,
            ]
            # Deduplicate
            seen = set()
            unique_alphas = []
            for a in alphas_to_try:
                a_rounded = round(a, 4)
                if a_rounded not in seen:
                    seen.add(a_rounded)
                    unique_alphas.append(a)
            alphas_to_try = unique_alphas

        for alpha in alphas_to_try:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial)[:n].max()
            if M_trial < best_trial_M - 1e-18:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        # Invalidate cached matrix since y changed
        prev_coeff_matrix = None
        prev_active_set = None

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", |S|=" + str(len(active_set)) + ", alpha=" + str(round(best_alpha, 4)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", iters=" + str(iter_count))
    return best_y, best_m


def smooth_gradient_descent(a0, time_limit, top_k=60, beta_schedule=None,
                            c0=3.0, restart_every=3000, momentum=0.7,
                            verbose=False):
    """
    Mirror descent on the simplex with smoothed max (log-sum-exp) of top convolution values.
    Uses Nesterov momentum for acceleration.
    """
    n = len(a0)
    x = norm_sum1(a0)
    best_x = x.copy()
    best_m = fast_convolve(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    vel = np.zeros(n)
    gamma = c0
    stall = 0
    cur_top_k = top_k
    cur_beta = 8.0
    restart_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov lookahead
        y = x + momentum * vel
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        ac = fast_convolve(y)
        # Use log-sum-exp of top k values as smooth approximation to max
        top_vals = ac[:n]
        top_idx = np.argsort(top_vals)[::-1][:cur_top_k]
        top_k_vals = top_vals[top_idx]

        # Log-sum-exp for numerical stability
        max_vk = top_k_vals.max()
        log_sum_exp = max_vk + np.log(np.sum(np.exp(beta_ * (top_k_vals - max_vk))))
        
        # Gradient: softmax-weighted sum of convolution gradients
        # d/dy[i] of (y*y)[k] = y[k-i] (if valid)
        w = np.exp(cur_beta * (top_k_vals - max_vk))
        w_sum = w.sum()
        w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += 2.0 * wi * y[k_int - idx]

        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_x = np.log(y) - eta * grad
        max_log = log_x.max()
        x_new = np.exp(log_x - max_log)
        x_new = np.maximum(x_new, 1e-30)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new

        vel = 0.9 * vel + 0.1 * (x_new - x)
        x = x_new

        M_new = fast_convolve(x)[:n].max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_x = x.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive parameters
        if t % 1000 == 0:
            cur_beta = max(3.0, cur_beta * 0.95)
            cur_top_k = min(cur_top_k + 2, 120)

        # Restart
        if stall > restart_every:
            stall = 0
            restart_count += 1
            noise_level = 0.03 / (1.0 + 0.05 * restart_count)
            pert = rng.normal(0, noise_level, n) * best_x
            x_r = best_x + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r < 1e-15:
                x_r = best_x.copy()
            else:
                x_r = x_r / s_r
            x = x_r
            vel = np.zeros(n)
            cur_top_k = min(cur_top_k + 10, 150)
            cur_beta = max(cur_beta - 1.0, 3.0)
            gamma = max(gamma * 0.95, 0.5)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    SGD restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

    return best_x, best_m


def coordinate_polish(a0, time_limit, rng_seed=99, verbose=False):
    """
    Coordinate descent polish: identify indices contributing to convolution peaks
    and transfer mass to low-contribution indices.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a)[:n].max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(30, n)
        top_k = np.argsort(ac[:n])[::-1][:n_top]

        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            # Contribution of each pair (idx[j], k-idx[j])
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            # Find the most contributing source index
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-10:
                continue

            # Score all indices by their total contribution to top peaks
            in_peak = np.zeros(n, dtype=bool)
            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]
                    in_peak[s2:e2 + 1] = True

            # Find destination: low score, not in peak
            candidates = np.where(~in_peak)[0]
            if len(candidates) < n // 3:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])

            if dst == src:
                continue

            # Try different transfer fractions
            found = False
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = src_val * frac
                if amt < 1e-14:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial)[:n].max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 100:
                # Random perturbation
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a)[:n].max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 50 == 0:
                # Random mass transfer
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        frac = rng.uniform(0.001, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a)[:n].max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                break

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", stall=" + str(stall))

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print("Total time budget: " + repr(time_budget) + "s")

    # Get initial sequence
    best_seq, best_score = get_initial(5000, t_global)
    n0 = len(best_seq)

    # Phase 1: Parametric search
    print("\n=== Phase 1: Parametric (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 45.0)
    if t1 > 10:
        a_param, _ = parametric_init(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "parametric", t_global)

    # Phase 2: Main LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: LP-FW main (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.40, 600.0)
    if t2 > 60 and n_best <= 8000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t2, solver,
                         initial_K=400, max_K=3500, add_per_iter=50,
                         verbose=True, update_every=4)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_main", t_global)

    # Phase 3: Smooth gradient refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Smooth Gradient (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_out, _ = smooth_gradient_descent(norm_sum1(best_seq), t3, top_k=80,
                                           c0=3.5, restart_every=2500, momentum=0.7,
                                           verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sgd", t_global)

    # Phase 4: LP-FW with fresh active set
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW round 2 (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.25, 350.0)
    if t4 > 60 and n_best <= 8000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t4, solver,
                         initial_K=500, max_K=4000, add_per_iter=60,
                         verbose=True, update_every=3)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_r2", t_global)

    # Phase 5: Try different n values
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    try_ns = [n_best + 500, n_best - 300, n_best + 1000]
    try_ns = [l for l in try_ns if 2000 <= l <= 9000 and l != n_best]

    for new_n in try_ns[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("\n=== Phase 5: Scale n=" + str(new_n) + " ===")
        a_long = interpolate_seq(best_seq, new_n)

        t5a = min(remaining * 0.4, 180.0)
        if t5a > 30 and new_n <= 8000:
            a_out, _ = lp_fw(a_long, t5a, solver,
                             initial_K=300, max_K=2500, add_per_iter=40,
                             verbose=False, update_every=5)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t5b = min(remaining * 0.3, 80.0)
            if t5b > 15:
                a_out, _ = smooth_gradient_descent(norm_sum1(best_seq), t5b, top_k=70,
                                                   c0=3.0, restart_every=2000, momentum=0.6,
                                                   verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "sgd_n" + str(new_n), t_global)

    # Phase 6: Final LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Final LP-FW (n=" + str(n_best) + ") ===")
    if remaining > 60 and n_best <= 8000:
        t6 = min(remaining * 0.50, 300.0)
        if t6 > 30:
            a_out, _ = lp_fw(norm_sum1(best_seq), t6, solver,
                             initial_K=600, max_K=4500, add_per_iter=70,
                             verbose=True, update_every=3)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # Phase 7: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final Coord (n=" + str(n_best) + ") ===")
    if remaining > 20:
        t7 = min(remaining * 0.75, 120.0)
        if t7 > 15:
            a_out, _ = coordinate_polish(norm_sum1(best_seq), t7, rng_seed=42, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final smooth gradient pass
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t8 = min(remaining * 0.6, 100.0)
        if t8 > 10:
            print("\n=== Phase 8: Final SGD ===")
            a_out, _ = smooth_gradient_descent(norm_sum1(best_seq), t8, top_k=100,
                                               c0=2.5, restart_every=1500, momentum=0.7,
                                               verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "sgd_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
