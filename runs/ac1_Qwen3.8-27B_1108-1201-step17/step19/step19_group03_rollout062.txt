def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

                partner1 = np.arange(k_lo1, k_hi1 + 1) - int(i_hot)
                c[k_lo1:k_hi1 + 1] -= 2.0 * delta * a[partner1]

            # Part 2: +2*delta * a[k-j_cold] for k in [j_cold, j_cold+n-1]
            k_lo2 = max(0, int(j_cold))
            k_hi2 = min(2 * n - 2, int(j_cold) + n - 1)
            if k_lo2 <= k_hi2:
                partner2 = np.arange(k_lo2, k_hi2 + 1) - int(j_cold)
                c[k_lo2:k_hi2 + 1] += 2.0 * delta * a[partner2]

            # Check improvement
            updates_since_recalc += 1
            new_m = c.max()

            if new_m < best_m - 1e-13:
                best_m = new_m
                best_a = a.copy()
                stall = 0
                improved = True
                n_steps += 1
            else:
                # Undo
                a[i_hot] = old_ai
                a[j_cold] = old_aj
                a = a / a.sum()
                # Recompute c from scratch (simpler than undoing)
                # But that's O(n log n)... Let's just undo the incremental update:
                # Undo part 1
                if k_lo1 <= k_hi1:
                    partner1 = np.arange(k_lo1, k_hi1 + 1) - int(i_hot)
                    c[k_lo1:k_hi1 + 1] += 2.0 * delta * a[partner1]
                # Undo part 2
                if k_lo2 <= k_hi2:
                    partner2 = np.arange(k_lo2, k_hi2 + 1) - int(j_cold)
                    c[k_lo2:k_hi2 + 1] -= 2.0 * delta * a[partner2]

            # Periodic recalculation to fix accumulated errors
            if updates_since_recalc >= recalc_every:
                c = fast_conv(a)
                updates_since_recalc = 0

        if not improved:
            stall += 1
            # Perturbation to escape local minimum
            if stall % 100 == 0:
                for _ in range(5):
                    i_pert = rng.integers(n)
                    j_pert = rng.integers(n)
                    if i_pert == j_pert:
                        continue
                    amt = a[i_pert] * 0.02
                    a[i_pert] -= amt
                    a[j_pert] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    c = fast_conv(a)
                    if c.max() < best_m:
                        best_m = c.max()
                        best_a = a.copy()
                        stall = 0
                        break

        if verbose and n_steps % 10000 == 0 and n_steps > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    IncPeak it={n_steps} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def lp_frank_wolfe_line_search(a0, time_limit, solver, verbose=False,
                                top_K_init=200, top_K_max=None, use_all=False):
    """LP-FW with proper line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_conv(a)
        M_curr = c.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if use_all or K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(c)[::-1][:K_sel]

        # Build constraint matrix T: T[k,j] = a[k-j] if valid
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            iter_count += 1
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 20, top_K_max)
            iter_count += 1
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Line search: try multiple step sizes
        best_alpha = 0.5
        best_trial_m = fast_conv(a).max()
        alphas = [2.0 / (iter_count + 2.0), 0.5, 0.3, 0.2, 0.1, 0.4, 0.6, 0.15, 0.25]
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_conv(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1

        if iter_count % 5 == 0:
            K = min(K + 20, top_K_max)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

    return best_a, best_m


def multiplicative_peak_suppress(a_init, time_limit, K=30, verbose=False, rng_seed=0):
    """Multiplicative updates to suppress peak lags."""
    n = len(a_init)
    a = norm_sum1(a_init)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        it += 1
        c = fast_conv(a)
        c_max = c.max()

        if c_max < 1e-15:
            break

        top_lags = np.argsort(c)[::-1][:K]
        top_vals = np.maximum(c[top_lags], 1e-30)
        weights = top_vals ** 4.0
        weights = weights / weights.sum()

        # Compute suppression gradient
        g = np.zeros(n)
        for w, k in zip(weights, top_lags):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            g[idx] += 2.0 * w * a[ki - idx]

        # Multiplicative update
        eta = 1.0 / np.sqrt(it + 1.0)
        log_a = np.log(np.maximum(a, 1e-15)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        m_new = fast_conv(a_new).max()
        if m_new < best_m - 1e-13:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Perturb
                pert = rng.normal(0, 0.01, n) * a
                a_p = np.maximum(a + pert, 1e-10)
                a = a_p / a_p.sum()
                stall = 50

        a = a_new

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MultPeak it={it} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def coordinate_polish(a_init, time_limit, verbose=False, rng_seed=0):
    """Fast coordinate descent polish."""
    n = len(a_init)
    a = norm_sum1(a_init)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 50

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        top_k = np.argsort(c)[::-1][:15]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            i_src = idx[np.argmax(contrib)]
            src_val = a[i_src]
            if src_val < 1e-12:
                continue

            # Find destination: position with low sensitivity
            sens = np.zeros(n)
            for kk in top_k[:8]:
                kki = int(kk)
                hs = max(0, kki - n + 1)
                he = min(n, kki + 1)
                sens[hs:he] += a[kki - np.arange(hs, he)]

            cold = np.where(sens < np.median(sens))[0]
            if len(cold) == 0:
                cold = np.arange(n)

            # Try a few destinations
            candidates = cold[np.argsort(a[cold])[:5]]

            found = False
            for dst in candidates:
                if dst == i_src or found:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[i_src] -= amt
                    a_trial[dst] += amt
                    if a_trial[i_src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
            if found:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                amt = a[i] * 0.01
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 6200
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(1.0 + 0.3 * x)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Asymmetric POCS (150s) ===
    print(f"\n=== Phase 1: Asymmetric POCS (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.09, 150.0)
    if t1 > 30:
        a_pocs, _ = pocs_asymmetric(best_seq, t1, n_trials=5, rng_seed=7, verbose=True)
        try_update(a_pocs, "pocs_asym")

    # === Phase 2: Incremental Peak Chase (350s) ===
    print(f"\n=== Phase 2: Incremental Peak Chase (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 40:
        # Try from best and from asymmetric inits
        inits_to_try = [best_seq]
        asym_inits = make_asymmetric_init(n0, seed=100)
        inits_to_try.extend(asym_inits[:3])

        t_per = t2 / len(inits_to_try)
        for idx, init in enumerate(inits_to_try):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 15:
                break
            label = f"incpeak_{idx}"
            print(f"  IncPeak init {idx} (t={t_avail:.0f}s)...")
            a_out, _ = incremental_peak_chase(init, t_avail, K=50, verbose=False, rng_seed=42 + idx)
            try_update(a_out, label)

    # === Phase 3: LP-FW with line search (300s) ===
    print(f"\n=== Phase 3: LP-FW Line Search (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.17, 300.0)
    if t3 > 40 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 1500)
        top_K_init = min(300, 2 * n_best - 1)
        top_K_max = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe_line_search(a_in, t3, solver, verbose=True,
                                                top_K_init=top_K_init, top_K_max=top_K_max,
                                                use_all=use_all)
        try_update(a_out, "lp_ls")

    # === Phase 4: Incremental Peak Chase Round 2 (250s) ===
    print(f"\n=== Phase 4: IncPeak R2 (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = incremental_peak_chase(a_in, t4, K=80, verbose=True, rng_seed=777)
        try_update(a_out, "incpeak_r2")

    # === Phase 5: Multiplicative Peak Suppress (200s) ===
    print(f"\n=== Phase 5: MultPeak (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = multiplicative_peak_suppress(a_in, t5, K=40, verbose=True, rng_seed=55)
        try_update(a_out, "multpeak")

    # === Phase 6: LP-FW Round 2 (200s) ===
    print(f"\n=== Phase 6: LP-FW R2 (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.12, 200.0)
    if t6 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 1500)
        top_K_init = min(500, 2 * n_best - 1)
        top_K_max = min(5000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe_line_search(a_in, t6, solver, verbose=True,
                                                top_K_init=top_K_init, top_K_max=top_K_max,
                                                use_all=use_all)
        try_update(a_out, "lp_r2")

    # === Phase 7: Multi-length exploration (200s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Multi-length (base n={n_best}) ===")

    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 2000, -2000, 3000, -3000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.35, 150.0)
        if t7a > 20:
            a_out, _ = incremental_peak_chase(a_long, t7a, K=50, verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"inc_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 5000:
            t7b = min(remaining * 0.4, 120.0)
            if t7b > 20:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                use_all = (new_n <= 1500)
                a_out, _ = lp_frank_wolfe_line_search(a_in, t7b, solver, verbose=False,
                                                        top_K_init=min(300, 2 * new_n - 1),
                                                        top_K_max=min(2000, 2 * new_n - 1),
                                                        use_all=use_all)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 8: Final IncPeak on best (remaining * 0.3) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final IncPeak (n={n_best}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.3, 150.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = incremental_peak_chase(a_in, t8, K=100, verbose=True, rng_seed=999)
            try_update(a_out, "incpeak_final")

    # === Phase 9: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 9: Coord Polish (n={n_best}) ===")
        t9 = min(remaining * 0.6, 100.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t9, verbose=False, rng_seed=789)
            try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 6200
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(1.0 + 0.3 * x)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
