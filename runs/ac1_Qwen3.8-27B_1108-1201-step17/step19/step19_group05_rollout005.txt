def fast_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def make_ramp(n, alpha=1.0, window='none'):
    """Create normalized ramp (i+1)^alpha with optional window."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** alpha
    if window == 'cos2':
        t = np.arange(n, dtype=np.float64) / n
        a *= np.cos(np.pi * t / 2) ** 2
    elif window == 'taper':
        t = np.arange(n, dtype=np.float64) / n
        a *= (1 - 0.5 * t)
    elif window == 'half_sin':
        a = np.sin(np.pi * np.arange(1, n + 1) / (n + 1))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s

def spectral_flatten(a0, time_limit, verbose=False):
    """Iteratively flatten the power spectrum to reduce peak autocorrelation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        # Compute spectrum
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        
        # Flatten: push power from peaks toward mean
        P_max = P.max()
        P_mean = P.mean()
        if P_max < 1e-30 or P_mean < 1e-30:
            break
        
        # Progressive flattening
        alpha = min(1.0, 0.2 + 0.8 * it / 1000.0)
        # Use a soft-thresholding approach

        # P_new = P ** 0.9

        A_new = A * mag_ratio
        a_new = np.fft.irfft(A_new, fft_size)[:n]

            a = a_new

def subgradient_descent(a0, time_limit, n_peaks=20, lr0=2.0, verbose=False):
    """Subgradient descent on the simplex for minimizing max autocorrelation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_conv(a)
        # Find top peaks
        top_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
        vals = conv[top_idx]
        max_v = vals.max()
        # Softmax weights
        w = np.exp((vals - max_v) / max(1e-10, max_v * 0.05))
        w = w / w.sum()
        
        # Compute subgradient
        g = np.zeros(n)
        for idx_i, k in enumerate(top_idx):
            ki = int(k)
            wi = w[idx_i]
            if wi < 1e-12:
                continue
            # g_j contribution from lag ki: 2 * a[ki - j] for valid j
            j_lo = max(0, ki - n + 1)
            j_hi = min(n - 1, ki)
            if j_lo > j_hi:
                continue
            j_vals = np.arange(j_lo, j_hi + 1)
            pair_vals = ki - j_vals
            g[j_vals] += wi * 2.0 * a[pair_vals]
        
        # Step
        eta = lr0 / (np.sqrt(t) + 1.0)
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Small blend
            blend = min(0.05, 0.01 * np.sqrt(t) / 100)
            a = (1 - blend) * a + blend * a_new
            a = np.maximum(a, 0)
            a = a / max(a.sum(), 1e-15)
        
        if verbose and t % 100 == 0:
            print(f"      SubGrad t={t}: J={2*n*best_m:.6f}, time={time.monotonic()-t0:.0f}s")
    
    return best_a, best_m

def lp_update(a, M_target, time_limit):
    """LP-based update: maximize sum(b) s.t. b>=0, conv(a,b) <= M_target."""
    # This requires an LP solver. Try scipy or cvxpy.
    try:
        from scipy.optimize import linprog
        n = len(a)
        conv_size = 2 * n - 1
        
        # Variables: b_0, ..., b_{n-1}
        # Maximize sum(b) -> minimize -sum(b)
        c_obj = -np.ones(n)
        
        # Constraints: (conv(a,b))_k <= M_target for k=0,...,2n-2
        # conv(a,b)_k = sum_i a_i * b_{k-i} (with bounds)
        # This is a sparse linear constraint.
        # A_ub @ b <= M_target for all k
        
        # For each k, the constraint is: sum_{i} a_i * b[k-i] <= M_target
        # where b[j] = 0 if j < 0 or j >= n
        
        # Build sparse matrix
        from scipy.sparse import lil_matrix
        A_ub = lil_matrix((conv_size, n))
        for k in range(conv_size):
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    A_ub[k, j] = a[i]
        
        b_ub = np.full(conv_size, M_target)
        bounds = [(0, None)] * n
        
        res = linprog(c_obj, A_ub=A_ub.tocsr(), b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            b = np.maximum(res.x, 0)
            return b
        return None
    except Exception as e:
        print(f"      LP error: {e}")
        return None
