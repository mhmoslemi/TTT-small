def _fast_autocorr(a):

    n = len(a)

            continue

        grad[j_range] += wi * 2.0 * a[pair]
    return grad

def _lp_direction(a, conv, M, time_limit=30.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    # Build constraint matrix A_ub: each row k has a[k-j] at col j for valid j
    # (a*g)[k] = sum_j a[k-j]*g[j] <= M
    # A_ub[k, j] = a[k-j] if 0<=k-j<n else 0
    # Use sparse for efficiency
    from scipy import sparse
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A_ub = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(time_limit - 2, 1.0)})
    if res.status == 0:
        g = np.maximum(res.x, 0)
        sg = g.sum()
        if sg > 1e-10:
            return g / sg
    return None

def _interpolate_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss
