def next_pow2(m):

def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = float(a.sum())

def golden_alpha_max(q0, c, qd, lo, hi, iters=30):
    """Minimize max_k [q0[k] - 2*alpha*c[k] + alpha^2*qd[k]] over alpha in [lo, hi]."""
    phi = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - phi * (hi - lo)
    d1 = lo + phi * (hi - lo)
    f1 = quad_max(c1, q0, c, qd)
    f2 = quad_max(d1, q0, c, qd)
    for _ in range(iters):
        if f1 < f2:
            hi = d1
            d1 = c1
            f2 = f1
            c1 = hi - phi * (hi - lo)
            f1 = quad_max(c1, q0, c, qd)
        else:
            lo = c1
            c1 = d1
            f1 = f2
            d1 = lo + phi * (hi - lo)
            f2 = quad_max(d1, q0, c, qd)
    return 0.5 * (lo + hi)


def line_search_direction(a, d, q0, T=None):
    """
    Minimize max q((1-alpha)*a + alpha*d) over alpha in [0,1].
    Uses exact quadratic form in alpha.

    d = np.asarray(d, dtype=np.float64)
    q0 = np.asarray(q0, dtype=np.float64)
    M0 = float(np.max(q0))

    dot = float(np.dot(a, d))
    if dot > 1.0 - 1e-7:
        return a.copy(), M0

    c = conv_ab(a, d)
    qd = self_conv(d)

    best_alpha = 0.0
    best_M = M0

    alphas = [1e-4, 5e-4, 1e-3, 2e-3, 5e-3, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 0.9, 1.0]

    if T is not None and np.isfinite(T) and T < 0.0 and M0 > 1e-15:
        t1 = min(0.30, max(5e-4, 0.5 * (-float(T)) / M0))
        alphas.append(max(1e-5, 0.5 * t1))
        alphas.append(min(0.95, 1.5 * t1))

    uniq = []
    seen = set()
    for al in alphas:
        al = float(min(1.0, max(0.0, al)))
        key = round(al, 12)
        if key not in seen:
            seen.add(key)
            uniq.append(al)

    for al in uniq:
        val = quad_max(al, q0, c, qd)
        if val < best_M:
            best_M = val
            best_alpha = al

    if best_M < M0 - 1e-13 and 0.0 < best_alpha < 1.0:
        al2 = golden_alpha_max(q0, c, qd, 0.0, 1.0, 28)
        val2 = quad_max(al2, q0, c, qd)
        if val2 < best_M:
            best_M = val2
            best_alpha = al2

    if best_alpha <= 0.0 or best_M >= M0 - 1e-14:
        return a.copy(), M0

    new = (1.0 - best_alpha) * a + best_alpha * d
    new = np.maximum(new, 0.0)
    s = float(new.sum())
    if s <= 1e-15:
        return a.copy(), M0
    new = new / s
    return new, best_M


def subgradient_direction(a, q, top_K=10, mode='topk'):
    """
    Compute subgradient of max(conv(a,a)) using top-K peaks.
    g[j] = 2 * sum_k w_k * a[k-j] for valid indices.
    Uses FFT-based cross-correlation for efficiency.
    """
    n = a.size
    m = q.size  # 2n-1
    max_q = float(q.max())
    if max_q <= 1e-15:
        return np.zeros(n, dtype=np.float64)

    if mode == 'topk':
        order = np.argsort(q)[::-1][:top_K]
        weights = q[order].copy()
        total = weights.sum()
        if total <= 1e-15:
            return np.zeros(n, dtype=np.float64)
        weights /= total
    elif mode == 'threshold':
        thresh = max_q * 0.95
        active = np.where(q >= thresh)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(q))])
        weights = q[active] - thresh + 1e-10
        total = weights.sum()
        if total <= 1e-15:
            active = np.array([int(np.argmax(q))])
            weights = np.array([1.0])
            total = 1.0
        weights /= total
    else:
        active = np.array([int(np.argmax(q))])
        weights = np.array([1.0])

    # Build weight vector of length 2n-1
    w_vec = np.zeros(m, dtype=np.float64)
    for i, k in enumerate(active):
        w_vec[int(k)] = weights[i]

    # g[j] = 2 * sum_k w_vec[k] * a[k-j] = 2 * corr_shift(w_vec, a)[j]
    g = 2.0 * corr_shift(w_vec, a)
    return g


def multiplicative_subgradient_step(a, q, top_K=8, trust=0.1):
    """
    Multiplicative (mirror) subgradient step:
        a_new[i] = a[i] * exp(-trust * g_log[i])
    where g_log[i] = 2 * a[i] * h[i] and h[i] = sum_k w_k * a[k-i].
    This automatically maintains non-negativity.
    """
    n = a.size
    m = q.size
    max_q = float(q.max())
    if max_q <= 1e-15:
        return a.copy()

    order = np.argsort(q)[::-1][:top_K]
    weights = q[order].copy()
    total = weights.sum()
    if total <= 1e-15:
        return a.copy()
    weights /= total

    w_vec = np.zeros(m, dtype=np.float64)
    for i, k in enumerate(order):
        w_vec[int(k)] = weights[i]

    # h[i] = sum_k w_vec[k] * a[k-i]
    h = corr_shift(w_vec, a)

    # Log-subgradient: g_log[i] = 2 * a[i] * h[i]
    g_log = 2.0 * a * h

    # Clip to avoid numerical issues
    g_log = np.clip(-g_log * trust, -50.0, 50.0)

    a_new = a * np.exp(g_log)
    a_new = np.maximum(a_new, 0.0)
    s = float(a_new.sum())
    if s <= 1e-15:
        return a.copy()
    return a_new / s


def subgradient_descent(a0, time_limit, top_K=10, verbose=False):
    """
    Fast subgradient descent with exact line search.
    Alternates between additive and multiplicative updates.
    """
    if time_limit <= 0.0:
        return normalize_vec(a0)

    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 2:
        return a

    best = a.copy()
    best_score = objective_norm(a)
    fails = 0
    iters = 0

    # Adaptive parameters
    trust = 0.3
    use_mult = True
    K = top_K

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        q = self_conv(a)
        M = float(np.max(q))
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            fails = 0
        else:
            fails += 1

        if fails > 300:
            # Try a restart with perturbation
            if fails > 500:
                break
            rng = np.random.default_rng(fails * 7)
            noise = rng.normal(size=n) * 0.01 * best
            cand = np.maximum(best + noise, 0.0)
            cand = normalize_vec(cand)
            sc = objective_norm(cand)
            if sc < best_score + 0.01:
                a = cand
            else:
                a = best.copy()
            fails = 0
            K = min(K + 5, 30)

        # Compute subgradient direction
        g = subgradient_direction(a, q, K, mode='topk')
        if float(np.max(g)) <= 1e-15:
            break

        if use_mult:
            # Multiplicative update
            a_new = multiplicative_subgradient_step(a, q, K, trust)
            sc_new = objective_norm(a_new)
            if sc_new < score - 1e-14:
                a = a_new
                trust = min(trust * 1.05, 3.0)
            elif sc_new < best_score + 0.1:
                a = a_new
                trust = max(trust * 0.9, 0.05)
            else:
                trust = max(trust * 0.85, 0.05)
                if fails % 50 == 0:
                    use_mult = not use_mult
        else:
            # Additive update with line search
            d = normalize_vec(a - g * trust * M / max(float(np.max(g)), 1e-15))
            new, Mnew = line_search_direction(a, d, q)
            sc_new = 2.0 * n * Mnew
            if sc_new < score - 1e-14:
                a = new
                trust = min(trust * 1.05, 3.0)
            elif sc_new < best_score + 0.1:
                a = new
                trust = max(trust * 0.9, 0.05)
            else:
                trust = max(trust * 0.85, 0.05)
                if fails % 50 == 0:
                    use_mult = not use_mult

        iters += 1
        if verbose and iters % 1000 == 0:
            print("    SubGrad: it=%d elapsed=%.1fs best=%.9f cur=%.9f K=%d trust=%.3f fails=%d"
                  % (iters, time.monotonic() - t0, best_score, score, K, trust, fails))

    return best


def get_lp_artifacts(n):
    art = _LP_CACHE.get(n)
    if art is not None:
        return art
    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0
    Aeq = np.zeros((1, n + 1), dtype=np.float64)
    Aeq[0, :n] = 1.0
    bounds = [(0.0, np.inf)] * n + [(None, None)]
    art = (c, Aeq, bounds)
    if len(_LP_CACHE) > 8:
        _LP_CACHE.clear()
    _LP_CACHE[n] = art
    return art


def solve_lp_offset(y, active, q_active):
    """
    Solve:
       min T
       s.t. sum g = 1,
            g >= 0,
            2*(y*g)[k] - T <= q_y[k], for active lags k.
    """
    if not HAVE_LP:
        return None, None
    y = np.asarray(y, dtype=np.float64)
    active = np.asarray(active, dtype=int)
    q_active = np.asarray(q_active, dtype=np.float64)
    n = y.size
    K = active.size
    if K <= 0:
        return None, None
    if K * (n + 1) > 12_000_000:
        return None, None

    try:
        A = np.zeros((K, n + 1), dtype=np.float64)
    except Exception:
        return None, None

    yy = 2.0 * y
    for r, k in enumerate(active):
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            A[r, i0:i1] = yy[start:stop][::-1]

    A[:, n] = -1.0
    b = np.maximum(q_active, 0.0)

    c, Aeq, bounds = get_lp_artifacts(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq,
                      b_eq=np.array([1.0], dtype=np.float64),
                      bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception:
        return None, None

    status = getattr(res, 'status', None)
    success = getattr(res, 'success', False)
    x = getattr(res, 'x', None)
    if (status != 0 or not success) or x is None:
        return None, None

    g = np.asarray(x[:n], dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-10:
        return None, None
    g = g / s
    T = float(x[n])
    if not np.isfinite(T):
        T = 0.0
    return g, T


def soft_direction_best(a, q, current_score, max_dirs=4):
    """
    Try softmax-gradient mirror directions and return best line-search result.
    """
    a = np.asarray(a, dtype=np.float64)
    q = np.asarray(q, dtype=np.float64)
    n = a.size
    if n <= 1:
        return None, current_score

    M_cur = current_score / (2.0 * float(n))
    m = float(q.max())
    if m <= 1e-15:
        return None, current_score

    focus_opts = (3.0, 8.0, 20.0)
    trust_opts = (0.15, 0.5, 1.5)

    best_new = None
    best_M = M_cur
    used = 0

    for focus in focus_opts:
        if used >= max_dirs:
            break
        scale = 0.5 * m
        gamma = focus / max(scale, 1e-15)
        vals = np.clip(gamma * (q - m), -50.0, 0.0)
        e = np.exp(vals)
        s = float(e.sum())
        if s <= 1e-300:
            w = np.ones(n, dtype=np.float64) / n
        else:
            w = e / s

        c = corr_shift(w, a)
        cmax = float(np.max(c))
        cmin = float(np.min(c))
        scale_c = cmax - cmin
        if scale_c <= 1e-15:
            continue
        for trust in trust_opts:
            if used >= max_dirs:
                break
            move = trust * (cmax - c) / scale_c
            move = np.clip(move, -50.0, 50.0)
            d = a * np.exp(move)
            d = normalize_vec(d)
            new, Mnew = line_search_direction(a, d, q, None)
            used += 1
            if Mnew < best_M - 1e-13:
                best_M = Mnew
                best_new = new

    if best_new is None:
        return None, M_cur
    return best_new, best_M

    else:
        initK = 250
        maxK = 900

    lp_fails = 0
    no_improve = 0
    total_improve = 0

        remaining = time_limit - (time.monotonic() - t0)
        if remaining <= 0.0:
            break

        qy = self_conv(y)
        M_y = float(np.max(qy))
        score_y = 2.0 * n * M_y

        if score_y < best_score - 1e-12:
            best = y.copy()
            best_score = score_y

            break

        Kcur = int(min(K, 2 * n - 1, 1200))
        order = np.argsort(qy)[::-1]
        act = order[:Kcur]
        q_active = qy[act]

        T = None

            tstep = time.monotonic()
            g_dir, T = solve_lp_offset(y, act, q_active)
