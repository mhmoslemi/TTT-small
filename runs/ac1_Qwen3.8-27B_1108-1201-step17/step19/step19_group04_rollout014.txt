Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Strategy: Fast Frank-Wolfe with exact line search, supplemented by
multiplicative gradient updates and multi-start diversification.
"""

import numpy as np
import math
import time

def next_pow2(m):
    m = max(1, int(m))
    return 1 << (m - 1).bit_length()

def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(a.size) / a.size
    return a / s

def self_conv(a):
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]

def conv_cross(a, b):
    m = a.size + b.size - 1
    size = next_pow2(m)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]

def score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    q = self_conv(a)
    return 2.0 * a.size * float(np.max(q))

def fw_step(a, q, active_lags):

    Compute FW direction: vertex e_m where m minimizes the linearization.
    Returns direction d (which is e_m) and the predicted improvement.
    """
    n = a.size
    K = len(active_lags)
    if K == 0:
        return None

    # We want to minimize sum_j a_new[j] * grad_j = sum_j d[j] * grad_j (since d is a vertex)
    # = grad_m (the component at m)
    # So we want m = argmin_j grad_j = argmin_j sum_{k in active} a[k-j]
    
    grad = np.zeros(n)
    for k in active_lags:
        k = int(k)

        # j ranges from max(0, k-n+1) to min(n-1, k)

            # = a[k-j_max : k-j_min+1] reversed
            # k-j_min to k-j_max (reversed)
            idx_start = k - j_max
            idx_end = k - j_min + 1
            grad[j_min:j_max+1] += a[idx_start:idx_end][::-1]
    
    return int(np.argmin(grad))

def multiplicative_update(a, q, eta=0.1, focus=5.0):
    """
    Gradient-based multiplicative update.
    grad_j of max_k conv(a,a)[k] at active lags.
    a_new[j] = a[j] * exp(-eta * grad_j)
    """
    n = a.size
    M = float(np.max(q))
    # Weight lags by how close they are to the max
    # w_k = exp(-focus * (M - q[k]) / M)
    excess = M - q
    scale = M if M > 0 else 1.0
    w = np.exp(-focus * excess / scale)
    w = w / w.sum()
    
    # grad_j = sum_k w[k] * 2 * a[k-j] (approximately)
    grad = np.zeros(n)
    # Only use top few lags for efficiency
    top_k = min(20, len(q))
    order = np.argsort(q)[::-1][:top_k]
    top_w = w[order]
    top_w = top_w / top_w.sum()

        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:

    # Actually for K<=20, the direct approach is fine.
