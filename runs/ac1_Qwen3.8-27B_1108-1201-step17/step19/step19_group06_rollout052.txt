def fast_autocorr(a):

    n = len(a)

    result = np.fft.irfft(A * A, fft_size)[:size]
    return result

    n = len(a)

    if w_sum < 1e-30:

def lp_direction(a, top_k=50, verbose=False):
    """
    Solve LP to find a direction that reduces the convolution peak.
    LP: maximize sum(b) s.t. b >= 0, conv(a, b)[k] <= M for top-k lags k.
    Returns the LP solution (normalized to sum 1) or None if LP fails.
    """
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    
    # Get top-K lags
    top_idx = np.argsort(ac)[::-1][:top_k]
    
    # Build constraint matrix A_ub (top_k x n)
    # A_ub[k, j] = a[k-j] if max(0,k-n+1) <= j <= min(n-1,k)
    A_rows = []
    for k_int in top_idx:
        k_int = int(k_int)
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n, k_int + 1)
        row = np.zeros(n)
        idx = np.arange(i_lo, i_hi)
        j_idx = k_int - idx
        row[idx] = a[j_idx]
        A_rows.append(row)
    
    A_ub = np.array(A_rows)
    b_ub = np.full(top_k, M)
    
    # Objective: minimize -sum(b) = maximize sum(b)
    c_obj = -np.ones(n)
    
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success and result.x is not None:
            b = np.maximum(result.x, 0)
            s = b.sum()
            if s > 1e-10:
                return b / s
    except Exception:
        pass
    
    return None
