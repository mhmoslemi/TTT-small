def interpolate_sequence(a, new_n):
    a=np.asarray(a,float)
    if a.size == new_n: return a.copy()
    old_n=a.size
    if old_n <=0: return np.ones(new_n)/new_n
    a=norm_sum1_array(a)
    if new_n <=0: return a
    old_x=np.linspace(0.0,1.0,old_n)
    new_x=np.linspace(0.0,1.0,new_n)
    f=np.interp(new_x, old_x, a)
    f=np.maximum(f,0.0)
    s=f.sum()
    if s<=0: f=np.ones(new_n)/new_n
    else f/=s
    return f

```python
"""
Hybrid low-peak convolution optimizer.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over nonnegative sequences.
The main ingredients are:
  1. Fast FFT-based convolution and scoring.
  2. Multi-start geometric/mirror descent with softmax lag weights.
  3. Sparse Frank-Wolfe-style direction search using active top lags,
     solved with scipy.linprog when available.
  4. Targeted coordinate transfers for final polishing.
  5. Length exploration around the best found length.
"""

import numpy as np
import time
import math

try:
    import scipy.optimize as scopt
except Exception:
    scopt = None

_LP_BOUNDS_CACHE = {}


def next_pow2_int(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def clean_finite_array(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    return a


def norm_sum1_array(a):
    a = clean_finite_array(a)
    if a.size == 0:
        return a
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def clean_score(sc):
    if sc is None:
        return float('inf')
    sc = float(sc)
    if not np.isfinite(sc):
        return float('inf')
    return sc


def fft_convolve(a):
    a = clean_finite_array(a)
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    if a.sum() <= 0.0:
        return np.zeros(2 * n - 1, dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a, mode='full').astype(np.float64)
    size = 2 * n - 1
    N = next_pow2_int(size)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:size]
    return out.astype(np.float64)


def score_of_array(a):
    a = clean_finite_array(a)
    n = a.size
    if n == 0:
        return float('inf')
    s = float(a.sum())
    if s <= 1e-15:
        return float('inf')
    b = fft_convolve(a)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * n * m / (s * s))


def interpolated_score(a):
    a = norm_sum1_array(a)
    n = a.size
    if n == 0:
        return float('inf')
    b = fft_convolve(a)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * n * m)


def make_lag_weights(b, K, beta_scale=8.0, uniform_mix=0.05):
    b = np.asarray(b, dtype=np.float64)
    L = b.size
    if L == 0:
        return np.array([0.0], dtype=np.float64)
    if L == 1:
        return np.array([1.0], dtype=np.float64)
    bmax = float(np.max(b))
    if not np.isfinite(bmax) or bmax <= 0.0:
        return np.ones(L, dtype=np.float64) / float(L)
    K = int(max(1, min(K, L)))
    if K >= L:
        idx = np.arange(L, dtype=int)
    else:
        idx = np.argpartition(b, L - K)[L - K:].astype(int)
    vt = b[idx] - bmax
    beta = max(1.0, float(beta_scale) / max(bmax, 1e-300))
    w = np.exp(np.clip(beta * vt, -745.0, 0.0))
    wsum = float(np.sum(w))
    if not np.isfinite(wsum) or wsum <= 1e-300:
        w = np.ones(idx.size, dtype=np.float64)
        w /= float(idx.size)
    else:
        w /= wsum
    p = np.zeros(L, dtype=np.float64)
    p[idx] = w
    if uniform_mix and uniform_mix > 0.0:
        um = float(uniform_mix)
        p = (1.0 - um) * p
        p += um / float(L)
        ps = float(p.sum())
        if ps > 0.0:
            p /= ps
    return p


def fft_gradient_from_p(a, p):
    a = clean_finite_array(a)
    p = clean_finite_array(p)
    n = a.size
    Lc = 2 * n - 1
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    if p.size < Lc:
        p2 = np.zeros(Lc, dtype=np.float64)
        p2[:p.size] = p
        p = p2
    else:
        p = p[:Lc]
    ps = float(p.sum())
    if ps <= 1e-300:
        return np.zeros(n, dtype=np.float64)
    p = p / ps
    r = a[::-1].copy()
    conv_len = Lc + n - 1
    N = next_pow2_int(conv_len)
    P = np.fft.rfft(p, N)
    Ar = np.fft.rfft(r, N)
    c = np.fft.irfft(P * Ar, N)[:conv_len]
    start = n - 1
    return (2.0 * c[start:start + n]).astype(np.float64)


def smooth_mwd(a0, time_limit, seed=1, K_init=80, alpha0=0.45, beta_scale=8.0,
               verbose=False):
    a = norm_sum1_array(a0)
    n = a.size
    if n == 0 or time_limit <= 5.0:
        return a, float('inf')
    a = a + 1e-16
    a = norm_sum1_array(a)
    b = fft_convolve(a)
    cur_score = interpol_score_from_b(a, b)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    it = 0
    alpha = float(alpha0)
    cur_K = int(K_init)
    beta = float(beta_scale)
    stall = 0
    prev_p = None
    sqrt_n = max(1.0, math.sqrt(float(n)))

    while time.monotonic() - t0 < max(1.0, float(time_limit)):
        if time.monotonic() - t0 >= float(time_limit) - 2.0:
            break
        it += 1
        bmax = float(np.max(b))
        if bmax <= 1e-30:
            break

        p = make_lag_weights(b, cur_K, beta, 0.05)
        if prev_p is not None:
            p = 0.80 * p + 0.20 * prev_p
            ps = float(p.sum())
            if ps > 0.0:
                p /= ps
        prev_p = p

        g = fft_gradient_from_p(a, p)
        g = np.where(np.isfinite(g), g, 0.0)
        gmaxv = float(np.max(g))
        gminv = float(np.min(g))
        gcenter = g - 0.5 * (gmaxv + gminv)
        scale = float(np.max(np.abs(gcenter)))
        if scale <= 1e-300:
            noise = rng.normal(0.0, 0.03, n)
            ap = best_a * (1.0 + noise)
            ap = np.maximum(ap, 1e-16)
            ap = norm_sum1_array(ap)
            bp = fft_convolve(ap)
            sp = interpol_score_from_b(ap, bp)
            if sp < best_score:
                best_a = ap.copy()
                best_score = sp
            a = ap
            b = bp
            cur_score = sp
            stall += 1
            continue

        eta = alpha / scale
        cap = min(1.5, max(0.1, 1.2 * alpha))
        mult = np.exp(np.clip(-eta * gcenter, -cap, cap))
        a1 = norm_sum1_array(a * mult)
        b1 = fft_convolve(a1)
        s1 = interpol_score_from_b(a1, b1)
        if not np.isfinite(s1):
            s1 = cur_score * 10.0
        if s1 < best_score:
            best_a = a1.copy()
            best_score = s1

        if s1 <= cur_score + 1e-14:
            a = a1
            b = b1
            cur_score = s1
            stall = 0
            alpha = min(alpha * 1.05, 1.0)
        else:
            eta2 = 0.5 * eta
            mult2 = np.exp(np.clip(-eta2 * gcenter, -cap, cap))
            a2 = norm_sum1_array(a * mult2)
            b2 = fft_convolve(a2)
            s2 = interpol_score_from_b(a2, b2)
            if not np.isfinite(s2):
                s2 = cur_score * 10.0
            if s2 < best_score:
                best_a = a2.copy()
                best_score = s2
            if s2 <= cur_score + 1e-14:
                a = a2
                b = b2
                cur_score = s2
                stall = 0
                alpha = max(alpha * 0.85, 0.02)
            elif s2 <= cur_score * 1.02:
                a = a2
                b = b2
                cur_score = s2
                stall += 1
                alpha = max(alpha * 0.80, 0.02)
            else:
                stall += 1
                alpha = max(alpha * 0.75, 0.02)

        if stall > 250:
            cur_K = min(cur_K + 20, 500)
            beta = max(2.0, beta * 0.80)

        if stall > 900 or alpha <= 0.021:
            noise = rng.normal(0.0, 0.04, n)
            ap = best_a * (1.0 + noise)
            ap = np.maximum(ap, 1e-16)
            ap = norm_sum1_array(ap)
            bp = fft_convolve(ap)
            sp = interpol_score_from_b(ap, bp)
            if sp < best_score:
                best_a = ap.copy()
                best_score = sp
            if sp < cur_score * 1.25:
                a = ap
                b = bp
                cur_score = sp
            else:
                a = best_a.copy()
                b = fft_convolve(a)
                cur_score = best_score
            stall = 0
            alpha = max(0.25, 0.5 * float(alpha0))
            cur_K = min(cur_K + 40, 600)
            beta = max(2.0, beta)

        if verbose and (it % 250 == 0):
            elapsed = time.monotonic() - t0
            print(f"      smooth_mwd it={it} t={elapsed:.1f}s best={best_score:.6f} cur={cur_score:.6f} stall={stall}")

    return best_a, best_score


def interpol_score_from_b(a, b):
    n = a.size
    m = float(np.max(b))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * n * m)


def get_lp_bounds(n):
    key = int(n)
    if key not in _LP_BOUNDS_CACHE:
        _LP_BOUNDS_CACHE[key] = [(0.0, None)] * key
    return _LP_BOUNDS_CACHE[key]


def build_conv_matrix(a, idx):
    a = clean_finite_array(a)
    n = a.size
    K = int(len(idx))
    T = np.zeros((K, n), dtype=np.float64)
    for ii in range(K):
        k = int(idx[ii])
        s = max(0, k - n + 1)
        e = min(n - 1, k) + 1
        if s >= e:
            continue
        start = k - s
        stop = k - e
        length = e - s
        if stop <= -1:
            src = a[start::-1]
        else:
            src = a[start:stop:-1]
        if len(src) != length:
            cols = np.arange(s, e, dtype=int)
            T[ii, cols] = a[k - cols]
        else:
            T[ii, s:e] = src
    return T


def lp_direction(a, idx, M):
    if scopt is None:
        return None
    a = clean_finite_array(a)
    n = a.size
    if n == 0 or M <= 0.0 or len(idx) == 0:
        return None
    T = build_conv_matrix(a, idx)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(len(idx), float(M), dtype=np.float64)
    bounds = get_lp_bounds(n)
    try:
        options = {'presolve': True, 'time_limit': 30.0}
        res = scopt.linprog(c=c, A_ub=T, b_ub=b_ub, bounds=bounds,
                            method='highs', options=options)
    except Exception:
        try:
            options = {'presolve': True}
            res = scopt.linprog(c=c, A_ub=T, b_ub=b_ub, bounds=bounds,
                                method='highs', options=options)
        except Exception:
            return None
    if res is None or res.status != 0:
        return None
    x = np.asarray(res.x, dtype=np.float64)
    if x.size != n:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x[x < 0.0] = 0.0
    s = float(x.sum())
    if s <= 1e-14:
        return None
    return x / s


def lp_frank_wolfe(a0, time_limit, K_init=120, K_max=500, verbose=False):
    a = norm_sum1_array(a0)
    n = a.size
    if n == 0 or time_limit <= 20.0:
        return a, float('inf')
    a = a + 1e-16
    a = norm_sum1_array(a)
    b = fft_convolve(a)
    cur_score = interpol_score_from_b(a, b)
    best_a = a.copy()
    best_score = cur_score
    t0 = time.monotonic()
    it = 0
    K = int(K_init)
    K = int(max(20, min(K, min(2 * n - 1, int(K_max)))))
    stall = 0
    L = 2 * n - 1

    while time.monotonic() - t0 < max(1.0, float(time_limit)):
        if time.monotonic() - t0 >= float(time_limit) - 5.0:
            break
        it += 1
        bmax = float(np.max(b))
        if bmax <= 1e-30:
            break

        use_K = int(max(20, min(K, L)))
        if use_K >= L:
            idx = np.arange(L, dtype=int)
        else:
            idx = np.argpartition(b, L - use_K)[L - use_K:].astype(int)

        M = bmax * (1.005 + 0.01 * stall)
        M = min(M, bmax * 1.25)

        t_lp0 = time.monotonic()
        direction = lp_direction(a, idx, M)
        lp_time = time.monotonic() - t_lp0

        if direction is None:
            K = int(min(K + 40, K_max))
            stall += 1
            if lp_time > 80.0:
                break
            continue

        lams = [1.0 / (2.0 * it + 2.0), 0.15, 0.075, 0.03]
        used = []
        for x in lams:
            if 0.0 < x < 1.01 and all(abs(x - y) > 1e-9 for y in used):
                used.append(x)
        used = [min(1.0, x) for x in used]

        best_line = a.copy()
        best_line_score = cur_score
        best_line_b = b.copy()
        for lam in used:
            at = (1.0 - lam) * a + lam * direction
            at = norm_sum1_array(at)
            bt = fft_convolve(at)
            st = interpol_score_from_b(at, bt)
            if st < best_line_score:
                best_line = at
                best_line_score = st
                best_line_b = bt

        if best_line_score < cur_score:
            a = best_line
            b = best_line_b
            cur_score = best_line_score
            stall = 0
        else:
            stall += 1
            if stall > 4:
                K = int(min(K + 60, K_max))
            if stall > 14:
                a = best_a.copy()
                b = fft_convolve(a)
                cur_score = best_score
                stall = 0
                K = int(min(K + 80, K_max))

        if best_line_score < best_score:
            best_a = best_line
            best_score = best_line_score

        if verbose and (it % 4 == 0):
            elapsed = time.monotonic() - t0
            print(f"      LP-FW it={it} t={elapsed:.1f}s best={best_score:.6f} cur={cur_score:.6f} K={use_K} lp={lp_time:.1f}s")

    return best_a, best_score


def coordinate_polish(a0, time_limit, seed=0, topK=18, verbose=False):
    a = norm_sum1_array(a0)
    n = a.size
    if n == 0 or time_limit <= 10.0:
        return a, float('inf')
    a = a + 1e-16
    a = norm_sum1_array(a)
    b = fft_convolve(a)
    cur_score = interpol_score_from_b(a, b)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    stall = 0
    L = 2 * n - 1
    it = 0

    while time.monotonic() - t0 < max(1.0, float(time_limit)):
        if time.monotonic() - t0 >= float(time_limit) - 3.0:
            break
        it += 1
        bmax = float(np.max(b))
        if bmax <= 1e-30:
            break
        useK = int(max(5, min(topK, L)))
        if useK >= L:
            idx = np.arange(L, dtype=int)
        else:
            idx = np.argpartition(b, L - useK)[L - useK:].astype(int)

        load = np.zeros(n, dtype=np.float64)
        active = np.zeros(n, dtype=bool)
        best_contrib = -1.0
        best_src = -1

        for kk in idx:
            k = int(kk)
            s = max(0, k - n + 1)
            e = min(n - 1, k) + 1
            if s >= e:
                continue
            cols = np.arange(s, e, dtype=int)
            other = a[k - cols]
            load[cols] += other
            active[cols] = True
            prod = a[cols] * other
            if prod.size == 0:
                continue
            pos = int(np.argmax(prod))
            cmax = float(prod[pos])
            if cmax > best_contrib:
                best_contrib = cmax
                best_src = int(cols[pos])

        improved = False
        if best_src >= 0 and a[best_src] > 1e-15:
            if int(np.count_nonzero(active)) < n:
                candidates = np.where(~active)[0].astype(int)
            else:
                candidates = np.arange(n, dtype=int)
            if candidates.size > 80:
                cand_load = load[candidates]
                part = np.argpartition(cand_load, 80)[:80]
                destinations = candidates[part]
            else:
                destinations = candidates
            if destinations.size > 0:
                destinations = destinations[np.argsort(load[destinations])][:6]
            else:
                destinations = np.arange(n, dtype=int)

            for dst in destinations:
                dst = int(dst)
                if dst == best_src:
                    continue
                for frac in (0.02, 0.01, 0.05, 0.08, 0.15, 0.25):
                    amt = float(a[best_src]) * frac
                    if amt <= 1e-18 or amt >= a[best_src]:
                        continue
                    at = a.copy()
                    at[best_src] -= amt
                    at[dst] += amt
                    at = np.maximum(at, 0.0)
                    ssum = float(at.sum())
                    if ssum <= 1e-15:
                        continue
                    at /= ssum
                    bt = fft_convolve(at)
                    st = interpol_score_from_b(at, bt)
                    if st < cur_score:
                        a = at
                        b = bt
                        cur_score = st
                        if st < best_score:
                            best_a = at.copy()
                            best_score = st
                        improved = True
                        break
                if improved:
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall % 8 == 0:
                source_scores = a * load
                srcs = np.argsort(source_scores)[-4:].astype(int)
                if int(np.count_nonzero(active)) < n:
                    candidates = np.where(~active)[0].astype(int)
                else:
                    candidates = np.arange(n, dtype=int)
                if candidates.size > 40:
                    cand_load = load[candidates]
                    part = np.argpartition(cand_load, 40)[:40]
                    destinations = candidates[part]
                else:
                    destinations = candidates
                if destinations.size > 0:
                    destinations = destinations[np.argsort(load[destinations])][:5]
                else:
                    destinations = np.arange(n, dtype=int)

                for src in srcs:
                    src = int(src)
                    if a[src] <= 1e-15:
                        continue
                    for dst in destinations:
                        dst = int(dst)
                        if src == dst:
                            continue
                        frac = float(rng.uniform(0.005, 0.06))
                        amt = float(a[src]) * frac
                        if amt <= 1e-18 or amt >= a[src]:
                            continue
                        at = a.copy()
                        at[src] -= amt
                        at[dst] += amt
                        at = np.maximum(at, 0.0)
                        ssum = float(at.sum())
                        if ssum <= 1e-15:
                            continue
                        at /= ssum
                        bt = fft_convolve(at)
                        st = interpol_score_from_b(at, bt)
                        if st < cur_score:
                            a = at
                            b = bt
                            cur_score = st
                            if st < best_score:
                                best_a = at.copy()
                                best_score = st
                            improved = True
                            break
                    if improved:
                        stall = 0
                        break

            if stall > 180:
                noise = rng.normal(0.0, 0.025, n)
                ap = best_a * (1.0 + noise)
                ap = np.maximum(ap, 1e-16)
                ap = norm_sum1_array(ap)
                bp = fft_convolve(ap)
                sp = interpol_score_from_b(ap, bp)
                if sp < best_score:
                    best_a = ap.copy()
                    best_score = sp
                if sp < cur_score * 1.25:
                    a = ap
                    b = bp
                    cur_score = sp
                stall = 0

        if cur_score > best_score * 1.04:
            a = best_a.copy()
            b = fft_convolve(a)
            cur_score = best_score
            stall = 0

        if verbose and (it % 20 == 0):
            elapsed = time.monotonic() - t0
            print(f"      coord it={it} t={elapsed:.1f}s best={best_score:.6f} cur={cur_score:.6f} stall={stall}")

    return best_a, best_score


def cand_uniform(n):
    return np.ones(n, dtype=np.float64) / float(n)


def cand_shape(n, kind):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    kind = int(kind)
    if kind == 0:
        a = x.copy()
    elif kind == 1:
        a = 1.0 - x
    elif kind == 2:
        a = x * x
    elif kind == 3:
        t = 1.0 - x
        a = t * t
    elif kind == 4:
        a = x * (1.0 - x)
    elif kind == 5:
        a = 0.25 + x * (1.0 - x)
    elif kind == 6:
        a = np.exp(-((x - 0.35) / 0.22) ** 2)
    elif kind == 7:
        a = np.exp(-((x - 0.70) / 0.18) ** 2)
    elif kind == 8:
        a = a = 1.0 + 0.5 * np.sin(np.pi * x)
    elif kind == 9:
        a = 1.0 + 0.35 * np.cos(np.pi * x)
    elif kind == 10:
        a = 1.0 - 0.35 * np.cos(np.pi * x)
    elif kind == 11:
        a = np.sqrt(1.0 + np.cos(2.0 * np.pi * x))
    elif kind == 12:
        a = 0.1 + 0.9 * x ** 0.35
    elif kind == 13:
        t = 1.0 - x
        a = 0.1 + 0.9 * t ** 0.35
    else:
        a = np.ones(n, dtype=np.float64)
    a = np.maximum(a, 1e-12)
    return norm_sum1_array(a)


def cand_random(n, seed, kind=0):
    n = int(max(2, n))
    rng = np.random.default_rng(int(seed))
    kind = int(kind)
    if kind == 0:
        a = rng.gamma(1.0, n)
    elif kind == 1:
        a = rng.exponential(1.0, n)
    elif kind == 2:
        a = np.exp(rng.normal(0.0, 0.6, n))
    else:
        alpha = np.ones(n, dtype=np.float64)
        alpha *= 0.7
        try:
            a = rng.dirichlet(alpha)
        except Exception:
            a = rng.gamma(1.0, n)
    return norm_sum1_array(a)


def blend_sequences(a, b, w):
    a = norm_sum1_array(a)
    b = norm_sum1_array(b)
    n = a.size
    if b.size != n:
        b = interpolate_sequence(b, n)
        b = norm_sum1_array(b)
    wt = float(np.clip(w, 0.0, 1.0))
    out = (1.0 - wt) * a + wt * b
    return norm_sum1_array(out)


def interpolate_sequence(a, new_n):
    new_n = int(max(1, new_n))
    a = norm_sum1_array(a)
    if a.size == 0:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    old_n = int(a.size)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        out = np.ones(new_n, dtype=np.float64)
        if float(a.sum()) > 0:
            out[:] = float(a[0]) / float(new_n)
        return out
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, a)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 1e-15:
        out = np.ones(new_n, dtype=np.float64) / float(new_n)
    else:
        out /= s
    return out.astype(np.float64)


def sequence_signature(a, m=18):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return (0.0,)
    step = max(1, n // int(m))
    vals = a[::step]
    return tuple(np.round(vals, 5).tolist())


def get_top_candidates(n, parent, max_cand=6, seed=0):
    n = int(max(2, n))
    all_items = []
    seen = set()

    def add_item(sc, a, label):
        sc = clean_score(sc)
        if not np.isfinite(sc):
            return
        a = norm_sum1_array(a)
        if a.size != n:
            return
        sig = sequence_signature(a, 18)
        if sig in seen:
            return
        seen.add(sig)
        all_items.append((sc, a, label))

    if parent is not None:
        p = norm_sum1_array(parent)
        if p.size != n:
            p = interpolate_sequence(p, n)
            p = norm_sum1_array(p)
        s = score_of_array(p)
        add_item(s, p, "parent")
        for kind in (0, 2, 3, 6, 7):
            sh = cand_shape(n, kind)
            mix = blend_sequences(p, sh, 0.35)
            add_item(score_of_array(mix), mix, f"mix_parent_k{kind}")

    for kind in range(14):
        a = cand_shape(n, kind)
        add_item(score_of_array(a), a, f"shape_{kind}")

    for r in range(6):
        a = cand_random(n, seed + 100 + r * 17, r % 4)
        add_item(score_of_array(a), a, f"rand_{r}")

    top = []
    chosen_idx = []
    for _ in range(max_cand):
        best_i = -1
        best_v = float('inf')
        for i, item in enumerate(all_items):
            if i in chosen_idx:
                continue
            if item[0] < best_v:
                best_v = item[0]
                best_i = i
        if best_i < 0:
            break
        top.append(all_items[best_i])
        chosen_idx.append(best_i)
    return top

        cap = smooth_caps[i] if i < len(smooth_caps) else 25.0
        t_run = min(float(cap), remaining - 10.0)
        if t_run <= 5.0:

        print(f"Smooth candidate [{lbl}] for {t_run:.0f}s")
        a_out, s_out = smooth_mwd(a, t_run, seed=211 + i, K_init=90,
                                  alpha0=0.45, beta_scale=8.0, verbose=False)
        s_out = clean_score(s_out)
        if np.isfinite(s_out) and s_out < best_score:
            best_a = a_out
            best_score = s_out
            print(f"NEW BEST after smooth [{lbl}]: {best_score:.6f}, n={best_a.size}")
        remaining = time_budget - (time.monotonic() - t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = best_a.size
    if remaining > 120.0 and n_best <= max_search_n:
        if n_best <= 5000:
            lp_kmax = 520
        elif n_best <= 7000:
            lp_kmax = 420
        else:
            lp_kmax = 300
        t_run = min(330.0, remaining - 30.0)
        if t_run > 70.0:
            print(f"LP-FW on n={n_best} for {t_run:.0f}s")
            a_out, s_out = lp_frank_wolfe(best_a, t_run, K_init=110, K_max=lp_kmax, verbose=True)

            if np.isfinite(s_out) and s_out < best_score:

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 90.0:
        t_run = min(230.0, remaining - 30.0)
        if t_run > 30.0:
            print(f"Second smooth pass for {t_run:.0f}s")
            a_out, s_out = smooth_mwd(best_a, t_run, seed=555, K_init=140,
                                      alpha0=0.35, beta_scale=4.0, verbose=True)
            s_out = clean_score(s_out)
            if np.isfinite(s_out) and s_out < best_score:
                best_a = a_out
                best_score = s_out
                print(f"NEW BEST after smooth2: {best_score:.6f}, n={best_a.size}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80.0:
        t_run = min(110.0, remaining - 30.0)
        if t_run > 25.0:
            print(f"Coordinate polish for {t_run:.0f}s")
            a_out, s_out = coordinate_polish(best_a, t_run, seed=900, topK=18, verbose=False)
            s_out = clean_score(s_out)
            if np.isfinite(s_out) and s_out < best_score:

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = best_a.size
    trial_lengths = []
    deltas = [200, -200, 500, -500, 1000, -1000, 2000, -2000, 3000, -3000]
    for d in deltas:
        nn = n_best + int(d)
        if 2500 <= nn <= max_search_n and nn != n_best:
            trial_lengths.append(nn)
    uniq_lengths = []
    for x in trial_lengths:
        if x not in uniq_lengths:
            uniq_lengths.append(x)
    pairs = []
    for x in uniq_lengths:
        pairs.append((abs(x - n_best), x))
    pairs.sort()
    trial_lengths = []
    for _, x in pairs:
        trial_lengths.append(x)

    for nn in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 90.0:
            break
        a_len = interpolate_sequence(best_a, nn)

            best_score = s_len

        t_run = min(90.0, remaining - 40.0)
        if t_run > 25.0:
            print(f"Length n={nn}: smooth {t_run:.0f}s")
            a_out, s_out = smooth_mwd(a_len, t_run, seed=3210 + nn % 997,
                                      K_init=100, alpha0=0.40, beta_scale=6.0, verbose=False)

            if np.isfinite(s_out) and s_out < best_score:

                best_score = s_out

        remaining = time_budget - (time.monotonic() - t_global)

        if remaining > 90.0 and n_best <= 7000:

            if t_run > 30.0:

        t_run = min(80.0, remaining - 20.0)
        if t_run > 15.0:
            print(f"Final smooth {t_run:.0f}s")
            a_out, s_out = smooth_mwd(best_a, t_run, seed=7777, K_init=180,

    best_a = norm_sum1_array(best_a)

    try:

        for v in best_a:

        fsc = evaluate_sequence(final_list)
        fsc = clean_score(fsc)
        if np.isfinite(fsc):
            best_score = fsc
            best_a = np.asarray(final_list, dtype=np.float64)
    except Exception:
        pass

    print(f"FINAL: score={best_score:.6f}, n={best_a.size}, elapsed={time.monotonic()-t_global:.1f}s")

    final_list = []
    for v in best_a:
        v = float(v)
        if not np.isfinite(v) or v < 0.0:
            v = 0.0
        if v > 1000.0:
            v = 1000.0
        final_list.append(v)
    s = sum(final_list)
    if s < 0.01:
        final_list = []
        for i in range(4000):
            x = float(i) / 3999.0
            final_list.append(float(math.sin(math.pi * x) ** 2))
        s = sum(final_list)
        if s <= 0.0:
            final_list = [1.0]
        else:
            final_list = [v / s for v in final_list]
    else:
        final_list = [v / s for v in final_list]
    return final_list
```
