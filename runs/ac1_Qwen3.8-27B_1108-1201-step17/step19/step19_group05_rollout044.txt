def compute_subgradient(a, conv, n_active=30, softmax_temp=None):
    """
    Compute weighted subgradient of max_k conv(a,a)[k].
    Uses softmax weighting over top active peaks.
    """
    n = len(a)
    max_val = conv.max()
    
    # Select active peaks
    n_act = min(n_active, 2 * n - 1)
    top_idx = np.argpartition(conv, -n_act)[-n_act:]
    vals = conv[top_idx]
    max_v = vals.max()
    
    if softmax_temp is None:
        softmax_temp = max(max_v * 0.1, 1e-20)
    
    # Softmax weights
    exp_vals = np.exp((vals - max_v) / softmax_temp)
    w = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for idx_pos in range(len(top_idx)):
        k = int(top_idx[idx_pos])
        wi = w[idx_pos]
        if wi < 1e-14:

        # d(conv[k])/d(a[j]) = a[k-j]*I(0<=k-j<n) + a[j-k]*I(0<=j-k<n)
        # First term: k-j in [0, n-1] => j in [max(0,k-n+1), min(n-1,k)]
        j_lo1 = max(0, k - n + 1)
        j_hi1 = min(n - 1, k)
        if j_lo1 <= j_hi1:
            indices = k - np.arange(j_lo1, j_hi1 + 1)
            grad[j_lo1:j_hi1 + 1] += wi * a[indices]
        # Second term: j-k in [0, n-1] => j in [max(0,k), min(n-1,k+n-1)]
        j_lo2 = max(0, k)
        j_hi2 = min(n - 1, k + n - 1)
        if j_lo2 <= j_hi2:
            indices = np.arange(j_lo2, j_hi2 + 1) - k
            grad[j_lo2:j_hi2 + 1] += wi * a[indices]
    
    return grad


def projected_step(a, grad, eta):
    """Take a subgradient step with projection onto {a>=0, sum(a)=1}."""
    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def spectral_flatten_step(a, strength=0.3):
    """One step of spectral flattening (reduce peak in frequency domain)."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.abs(A) ** 2
    P_max = P.max()
    P_mean = P.mean()
    if P_max < 1e-30:
        return a.copy()
    # Compress peaks in power spectrum
    P_new = P + strength * (P_mean - P)
    P_new = np.maximum(P_new, 1e-30)
    ratio = np.sqrt(P_new / P)
    ratio = np.clip(ratio, 0.3, 3.0)
    A_new = A * ratio
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def peak_targeted_move(a, conv, n_candidates=20):
    """
    Targeted move: find the peak at lag k*, identify the pair (i, k*-i)
    with max product, and reduce it by transferring mass to a low-impact position.

    k_star = int(np.argmax(conv))

        return a.copy()

    products = a[idx] * a[k_star - idx]
    max_prod_idx = idx[np.argmax(products)]
    
    if a[max_prod_idx] < 1e-12:
        return a.copy()
    
    # Find a low-cost destination (position that doesn't contribute much to top peaks)
    top_ks = np.argpartition(conv, -min(10, 2*n-1))[-min(10, 2*n-1):]
    costs = np.zeros(n)
    for k2 in top_ks:

        if lo2 <= hi2:
            valid = np.arange(lo2, hi2 + 1)
            costs[valid] += a[k2 - valid]
    
    # Score: low cost + low current value = good destination
    dest_scores = costs + 0.5 * a
    dest_scores[max_prod_idx] = np.inf  # Can't move to self
    dest = int(np.argmin(dest_scores))

def mirrored_subgradient_descent(a0, time_limit, lr0=1.0, n_active=30,

                i = int(rng.integers(n))
                block_len = int(rng.integers(2, min(50, n - i)))
                if block_len >= 2:
                    a_trial[i:i+block_len] = a_trial[i+block_len-1:i-1:-1]
            elif move_type == 2:
                # Gaussian perturbation on random block
                i = int(rng.integers(n))
                block_len = int(rng.integers(5, min(100, n - i)))
                noise = rng.normal(0, 0.05, block_len) * a[i:i+block_len]
                a_trial[i:i+block_len] += noise
            else:
                # Peak-directed: reduce contribution to main peak
                k_star = int(np.argmax(conv))
                lo = max(0, k_star - n + 1)
                hi = min(n - 1, k_star)
                if lo <= hi:
                    idx = np.arange(lo, hi + 1)
                    products = a_trial[idx] * a_trial[k_star - idx]
                    worst = idx[np.argmax(products)]
                    # Transfer from worst to random low position
                    dest = int(rng.integers(n))
                    if dest != worst and a_trial[worst] > 1e-10:
                        amt = a_trial[worst] * rng.uniform(0.05, 0.3)
                        a_trial[worst] -= amt
                        a_trial[dest] += amt
            
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            
            sc = safe_eval(a_trial)
            delta = sc - best_score
            
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
                improved = True
            elif rng.uniform() < np.exp(-delta / T):
                a = a_trial.copy()
        
        if not improved and step % 500 == 0:
            # Occasionally reset to best
            a = best_a.copy()
        
        T *= cooling
        
        if step % 2000 == 0 and verbose:
            print(f"      SA step={step} ({time.monotonic()-t0:.0f}s): T={T:.6f}, J={best_score:.6f}")
    
    return best_a, best_score


def multi_scale_optimize(a0, time_limit, verbose=False):
    """
    Multi-scale approach: optimize at reduced resolution, then refine at full.
    """
    n = len(a0)
    best_a = a0.copy()
    best_score = safe_eval(a0)
    t0 = time.monotonic()
    
    # Try coarser resolutions
    for factor in [4, 2]:
        if time.monotonic() - t0 > time_limit * 0.7:
            break
        n_coarse = n // factor
        if n_coarse < 100:
            continue
        
        # Downsample
        a_coarse = np.zeros(n_coarse)
        for i in range(n_coarse):
            a_coarse[i] = a0[i*factor:(i+1)*factor].sum()
        a_coarse = a_coarse / a_coarse.sum()
        
        # Optimize at coarse level
        t_avail = min((time_limit - (time.monotonic() - t0)) * 0.3, 120.0)
        if t_avail < 10:
            continue
        a_opt, sc = mirrored_subgradient_descent(
            a_coarse, t_avail, lr0=1.5, n_active=20, verbose=False,
            rng_seed=42 + factor)
        
        # Upsample and use as starting point
        a_up = np.zeros(n)
        for i in range(n_coarse):
            a_up[i*factor:(i+1)*factor] = a_opt[i]
        a_up = a_up / a_up.sum()
        
        sc_up = safe_eval(a_up)
        if sc_up < best_score:
            best_score = sc_up
            best_a = a_up.copy()
            if verbose:
                print(f"    Multi-scale factor {factor}: J={sc_up:.6f}")
    
    # Final refinement at full resolution
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 30:
        a_opt, sc = mirrored_subgradient_descent(
            best_a, remaining * 0.8, lr0=0.5, n_active=50,
            softmax_frac=0.05, verbose=verbose, rng_seed=99)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
    
    return best_a, best_score


def coordinate_polish(a0, time_limit, rng_seed=42, verbose=False):
    """
    Coordinate descent: try individual element adjustments to reduce peaks.
    Uses line search for optimal step.
    """
    n = len(a0)
    a = a0 / a0.sum()
    best_a = a.copy()
    best_score = safe_eval(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        conv = fast_autocorr(a)
        k_max = int(np.argmax(conv))
        
        # Find the worst pair at the peak
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        if lo > hi:
            stall += 1
            break
        
        idx = np.arange(lo, hi + 1)
        products = a[idx] * a[k_max - idx]
        src = idx[np.argmax(products)]
        
        if a[src] < 1e-12:
            stall += 1
            continue
        
        # Compute cost map: how much each position contributes to top peaks
        top_ks = np.argpartition(conv, -min(15, 2*n-1))[-min(15, 2*n-1):]
        costs = np.zeros(n)
        for k2 in top_ks:
            k2 = int(k2)
            lo2 = max(0, k2 - n + 1)
            hi2 = min(n - 1, k2)
            if lo2 <= hi2:
                valid = np.arange(lo2, hi2 + 1)
                costs[valid] += a[k2 - valid]
        
        # Try multiple destinations
        candidates = np.argsort(costs + 0.3 * a)[:10]
        candidates = [c for c in candidates if c != src]
        
        moved = False
        for dest in candidates:
            if dest == src:
                continue
            # Bisect for best fraction
            best_frac = 0
            best_sc = best_score
            for frac in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
                amt = a[src] * frac
                if amt < 1e-14:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dest] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                sc = safe_eval(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_frac = frac
                    best_trial = a_trial.copy()
            
            if best_frac > 0 and best_sc < best_score:
                a = best_trial
                best_score = best_sc
                best_a = a.copy()
                moved = True
                stall = 0
                break
        
        if not moved:
            stall += 1
            # Random perturbation to escape
            if stall % 20 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.02, 0.1)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        sc = safe_eval(a)
                        if sc < best_score:
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
                            break
    
    if verbose:
        print(f"      CoordDescend done ({time.monotonic()-t0:.0f}s): J={best_score:.6f}")
    
    return best_a, best_score


def propose_candidate(budget_s=1800.0):
    """Main search function: multi-strategy optimization starting from parent."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a = np.minimum(a, 1000.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = safe_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = safe_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        # Fallback: create a reasonable initial sequence
        n_fallback = 4000
        parent = np.ones(n_fallback)
        parent_score = safe_eval(parent)
        print(f"Fallback constant: n={n_fallback}, score={parent_score:.6f}")
        best_seq = parent.copy()
        best_score = parent_score
    
    # Normalize for optimization (J is scale-invariant)
    a_work = parent / parent.sum()
    
    # === Phase 1: Subgradient Descent (main workhorse) ===
    print(f"\n=== Phase 1: Subgradient Descent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 30:
        print(f"  Running SGD for {t1:.0f}s...")
        a_out, sc = mirrored_subgradient_descent(
            a_work, t1, lr0=1.5, n_active=40, softmax_frac=0.08,
            verbose=True, rng_seed=42)
        try_update(a_out, "sgd_main")
        a_work = best_seq / best_seq.sum()
    
    # === Phase 2: Multi-scale optimization ===
    print(f"\n=== Phase 2: Multi-Scale ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        print(f"  Running multi-scale for {t2:.0f}s...")
        a_out, sc = multi_scale_optimize(a_work, t2, verbose=True)
        try_update(a_out, "multiscale")
        a_work = best_seq / best_seq.sum()
    
    # === Phase 3: Simulated Annealing ===
    print(f"\n=== Phase 3: Simulated Annealing ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 30:
        print(f"  Running SA for {t3:.0f}s...")
        a_out, sc = simulated_annealing(a_work, t3, n_trials_per_step=3,
                                         rng_seed=123)
        try_update(a_out, "sa")
        a_work = best_seq / best_seq.sum()
    
    # === Phase 4: Coordinate Descent Polish ===
    print(f"\n=== Phase 4: Coordinate Descent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 180.0)
    if t4 > 20:
        print(f"  Running CoordDescend for {t4:.0f}s...")
        a_out, sc = coordinate_polish(a_work, t4, rng_seed=42, verbose=True)
        try_update(a_out, "coord1")
        
        remaining = time_budget - (time.monotonic() - t_global)
        t4b = min(remaining * 0.10, 100.0)
        if t4b > 15:
            a_out, sc = coordinate_polish(a_work, t4b, rng_seed=777, verbose=True)
            try_update(a_out, "coord2")
    
    # === Phase 5: Spectral + SGD combo ===
    print(f"\n=== Phase 5: Spectral + SGD ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        # Spectral step
        a_spec = a_work.copy()
        for _ in range(50):
            a_spec = spectral_flatten_step(a_spec, strength=0.2)
        try_update(a_spec, "spectral")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t5 = min(remaining * 0.25, 200.0)
            a_in = best_seq / best_seq.sum()
            a_out, sc = mirrored_subgradient_descent(
                a_in, t5, lr0=0.8, n_active=60, softmax_frac=0.03,
                verbose=True, rng_seed=555)
            try_update(a_out, "sgd_fine")
    
    # === Phase 6: Final polish with different parameters ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        
        # Try peak-targeted moves directly
        conv = fast_autocorr(a_in)
        for _ in range(min(100, int(remaining / 2))):
            if time.monotonic() - t_global > time_budget:
                break
            a_pt = peak_targeted_move(a_in, conv)
            sc_pt = safe_eval(a_pt)
            if sc_pt < best_score:
                best_score = sc_pt
                best_seq = a_pt.copy()
                a_in = a_pt / a_pt.sum()
                conv = fast_autocorr(a_in)
                print(f"  Peak-targeted: J={best_score:.6f}")
            else:
                # Small random perturbation
                i = int(np.random.default_rng(_).integers(len(a_in)))
                j = int(np.random.default_rng(_ + 1).integers(len(a_in)))
                if i != j:
                    amt = a_in[i] * 0.05
                    a_in[i] -= amt
                    a_in[j] += amt
                    a_in = np.maximum(a_in, 0)
                    a_in = a_in / a_in.sum()
                    conv = fast_autocorr(a_in)
        
        # Final SGD with very fine params
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6 = min(remaining * 0.5, 100.0)
            a_out, sc = mirrored_subgradient_descent(
                a_in, t6, lr0=0.3, n_active=80, softmax_frac=0.02,
                verbose=False, rng_seed=999)
            try_update(a_out, "final_sgd")
    
    # Final validation
    if best_seq is not None:
        fs = safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = np.ones(1000)
        best_score = safe_eval(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    # Ensure output is a list of floats
    return [float(x) for x in best_seq]
