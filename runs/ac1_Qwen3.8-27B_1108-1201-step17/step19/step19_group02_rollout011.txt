def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}"""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    n = len(a)
    grad = np.zeros(n)

    # (a*a)[k] = sum_i a[i] * a[k-i]
    # d/da[j] = 2*a[k-j] (with a[m]=0 for m outside [0,n-1])

    #   grad[j] += 2 * weight * a[k-j]

        # For j = j_end: a[k - j_end]

1. Fast projected subgradient method with momentum and adaptive top-k smoothed max

_SOLVER = None

def _get_solver():

    if _SOLVER is not None:
        return _SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status == 'optimal':
                _SOLVER = s

        except Exception:

def _fft_conv(a):
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cv2 = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv2 - 1.0))[0][-1]
    theta = (cv2[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def _eval(a):
    try:
        return float(evaluate_sequence([float(x) for x in a]))
    except Exception:

def _subgrad(a, top_idx, weights):
    n = len(a)
    g = np.zeros(n)
    rev_a = a[::-1]
    for i in range(len(top_idx)):
        k = int(top_idx[i])
        w = 2.0 * weights[i]
        start = max(0, k - n + 1)
        end = min(n - 1, k)
        if start > end:
            continue
        idx_s = n - 1 - k + start
        idx_e = n - 1 - k + end
        g[start:end + 1] += w * rev_a[idx_s:idx_e + 1]
    return g

def _grad_opt(a0, time_limit, verbose=True):
    n = len(a0)
    a = _norm1(a0)
    best_a = a.copy()
    best_m = _fft_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    eta = 0.08
    mu = 0.55
    p = 15.0
    k_top = 25
    stall = 0
    max_stall = 4000
    restarts = 0
    v = np.zeros(n)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        ac = _fft_conv(a)
        M = ac.max()

        if M < best_m - 1e-16:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        a_new = _proj_simplex(a_new)

            v = np.zeros(n)

    return best_a, best_m

def _spectral_reweight(a, alpha=0.5, eps=1e-4):
    n = len(a)
    fs = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.rfft(a, fs)
    mag = np.abs(A)
    phase = A / np.maximum(mag, 1e-30)
    new_mag = (mag ** 2 + eps) ** alpha
    A_new = phase * new_mag
    a_new = np.fft.irfft(A_new, fs)[:n]
    return a_new

def _lp_fw(a0, time_limit, solver, init_K=300, max_K=2000, add_per=30, verbose=True):
    n = len(a0)
    a = _norm1(a0)
    best_a = a.copy()
    best_m = _fft_conv(a).max()
    t0 = time.monotonic()

    ac = _fft_conv(a)
    active = sorted(np.argsort(ac)[::-1][:init_K].tolist())
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        if stall > 800:
            break
        it += 1

        ac = _fft_conv(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if len(active) > max_K:
            margins = np.array([M_curr - ac[k] for k in active])
            keep = np.argsort(margins)[:max_K]
            active = sorted([active[i] for i in keep])

        K = len(active)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(active):
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            if s <= e:
                idx = np.arange(s, e + 1)
                coeffs[jj, idx] = a[ki - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        cons = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), cons)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active = sorted(active[:init_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        gv = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = gv.sum()
        if gs < 1e-15:
            continue
        g_dir = gv / gs

        fs = 1 << (2 * n - 2).bit_length()
        Af = np.fft.rfft(a, fs)
        Gf = np.fft.rfft(g_dir, fs)
        cross = np.fft.irfft(Af * Gf, fs)[:2 * n - 1]

        viol = np.where(cross > M_curr + 1e-10)[0]
        if len(viol) > 0:
            amt = cross[viol] - M_curr
            top_v = viol[np.argsort(amt)[::-1][:add_per]]
            new_set = set(active)
            for k in top_v:
                new_set.add(int(k))
            active = sorted(new_set)

        alpha_fw = 2.0 / (it + 2.0)
        if it <= 2:
            a_new = g_dir.copy()
        else:
            best_alpha = alpha_fw
            best_M = best_m
            for al in [alpha_fw, alpha_fw * 0.5, alpha_fw * 2.0, 0.1, 0.3, 0.5]:
                al = max(0.01, min(1.0, al))
                trial = (1.0 - al) * a + al * g_dir
                trial = _norm1(trial)
                Mt = _fft_conv(trial).max()
                if Mt < best_M:
                    best_M = Mt
                    best_alpha = al
            a_new = _norm1((1.0 - best_alpha) * a + best_alpha * g_dir)

        a = a_new

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print("      LPFW it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + " |S|=" + str(len(active)))

    return best_a, best_m

def _interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

    try_ns = [l for l in try_ns if 3000 <= l <= 9000 and l != n_cur]
    for new_n in try_ns[:1]:
        remaining = total_budget - (time.monotonic() - t0)
        if remaining < 120:
            break
        print("  n=" + str(new_n))
        a_long = _interpolate(best_seq, new_n)
        t5 = min(remaining * 0.4, 120.0)
        if t5 > 30:
            a_out, _ = _grad_opt(a_long, t5, verbose=False)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, "grad_n" + str(new_n), t0)
        remaining = total_budget - (time.monotonic() - t0)
        if remaining > 60 and new_n <= 8000:
            t5b = min(remaining * 0.3, 80.0)
            if t5b > 20:
                a_out, _ = _lp_fw(_norm1(best_seq), t5b, solver, init_K=300, max_K=1500, add_per=30, verbose=False)
                best_seq, best_score = _try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t0)

    # Phase 6: Coordinate polish (remaining)
    remaining = total_budget - (time.monotonic() - t0)
    n_cur = len(best_seq)
    print("\n=== Phase 6: Coord polish (n=" + str(n_cur) + ") ===")
    if remaining > 30:
        t6 = min(remaining * 0.8, 120.0)
        if t6 > 15:
            a_out, _ = _coord_polish(_norm1(best_seq), t6, rng_seed=42, verbose=True)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, "coord", t0)

    # Phase 7: Final LP-FW
    remaining = total_budget - (time.monotonic() - t0)
    n_cur = len(best_seq)
    print("\n=== Phase 7: Final LP-FW (n=" + str(n_cur) + ") ===")
    if remaining > 40 and n_cur <= 8000:
        t7 = min(remaining * 0.85, 200.0)
        if t7 > 30:
            a_out, _ = _lp_fw(_norm1(best_seq), t7, solver, init_K=500, max_K=3000, add_per=50, verbose=True)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, "lpfw_final", t0)

    # Final validation
    if best_seq is not None:
        fs = _eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        el = time.monotonic() - t0
        print("\nFINAL: score=" + repr(best_score) + " n=" + str(len(best_seq)) + " time=" + str(round(el, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = _norm1(np.sin(np.pi * x) ** 2)
        best_score = _eval(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
