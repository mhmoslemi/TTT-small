import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def fast_convolve(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_objective(a):
    """Compute J = 2*n*max_conv / sum^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def soft_max_objective(x, beta, n):
    """
    Soft-max smooth approximation of the objective.
    x is the parameter vector (length n), a = softmax(x) or just x (if already normalized).
    Returns (objective, gradient).
    """
    # Assume x is already the normalized sequence (sum = 1, non-negative)
    a = x  # For now, we'll handle the parameterization outside
    
    conv = fast_convolve(a)
    # Use only the first n values (by symmetry, these cover all unique values)
    conv_half = conv[:n]  # indices 0 to n-1
    
    beta_conv = beta * conv_half
    beta_conv -= beta_conv.max()  # numerical stability
    exp_vals = np.exp(beta_conv)
    log_sum = np.log(exp_vals.sum())
    M_soft = (log_sum + beta_conv.max()) / beta  # = (1/beta) * log(sum(exp(beta * conv_half)))
    
    w = exp_vals / exp_vals.sum()  # softmax weights
    
    # Gradient: dM_soft/da_j = 2 * sum_k w_k * a_{k-j} * 1[0 <= k-j < n]
    # This is the cross-correlation of w and a
    # Use FFT for efficiency
    fft_size = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    # Cross-correlation: IFFT(FFT(w) * conj(FFT(a)))
    corr = np.fft.irfft(W * np.conj(A), fft_size)
    # corr[j] = sum_k w_k * a_{k-j} (with zero padding)
    # We want sum over k where 0 <= k-j < n, which is automatically satisfied by zero-padding
    grad_M = 2.0 * corr[:n]
    
    # Full objective: 2*n*M_soft / S^2, but S = 1 (normalized)
    obj = 2.0 * n * M_soft  # since S = 1
    
    # Gradient of obj w.r.t. a: 2*n * grad_M (since S=1, the 1/S^2 factor is 1)
    grad_obj = 2.0 * n * grad_M
    
    return obj, grad_obj

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)

    a = np.maximum(a, 0.0)
    s = a.sum()

        return float('inf')

    n = len(a)
    conv = _fft_convolve(a)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def _interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = _norm(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, f)

def _fft_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]


def _score(a):
    """Compute J = 2n*max_conv/S^2 for normalized a."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    n = len(a)
    conv = _fft_conv(a)
    return 2.0 * n * conv.max()


def _norm(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _interp(f, new_n):
    """Normalize and interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return _norm(f)
    f = _norm(f)
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    r = np.interp(nx, ox, f)
    return _norm(r)

        conv_half = conv[:n]  # By symmetry, first n values suffice

        beta = 8.0

        bc -= bc.max()

        center = np.dot(a, grad_M)

def _build_lp_matrix(a_t, active_k, n):
    """Build the constraint matrix for the tangent LP.
    Row k: 2 * a_t[k-j] for valid j.
    Returns (K, n) matrix.
    """
    K = len(active_k)
    # Use sparse structure: each row has at most min(K_row_width, n) nonzeros
    # For efficiency, build as dense if K*n is manageable, else sparse
    if K * n <= 50_000_000:  # 50 million entries max for dense
        M = np.zeros((K, n))
        for ii, k in enumerate(active_k):
            k_int = int(k)
            # Row k: M[ii, j] = 2*a_t[k-j] if 0 <= k-j < n
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            if j_start < j_end:
                js = np.arange(j_start, j_end)
                M[ii, js] = 2.0 * a_t[k_int - js]
    else:
        # Sparse construction
        from scipy import sparse
        rows = []
        cols = []
        vals = []
        for ii, k in enumerate(active_k):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            if j_start < j_end:
                js = np.arange(j_start, j_end)
                rows.extend([ii] * len(js))
                cols.extend(js.tolist())
                vals.extend((2.0 * a_t[k_int - js]).tolist())
        M = sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))
    return M


def _cutting_plane_lp(a0, t0, time_limit, solver, initial_K=300, max_K=3000, verbose=True):
    """Cutting-plane LP method for minimizing max convolution."""
    n = len(a0)
    a = _norm(a0)
    best_a = a.copy()
    best_conv = _fft_conv(a)
    best_M = best_conv.max()
    t_start = time.monotonic()
    iter_count = 0
    no_improve = 0
    max_no_improve = 2000

    # Initial active set: top K convolution peaks
    conv = best_conv.copy()
    active_k = sorted(np.argsort(conv)[::-1][:initial_K].tolist())

    while time.monotonic() - t_start < time_limit:
        if no_improve > max_no_improve:
            if verbose:
                print("    CP-LP: stagnation at iter " + str(iter_count))
            break
        iter_count += 1

        # Compute current convolution and active peaks
        conv = _fft_conv(a)
        M_curr = conv.max()

        if M_curr < best_M - 1e-14:
            best_M = M_curr
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update active set: add newly violated peaks
        # Find peaks above current threshold
        threshold = M_curr * 0.95
        peak_candidates = np.where(conv >= threshold)[0]
        if len(peak_candidates) > 0:
            # Priority: highest peaks first
            peak_order = peak_candidates[np.argsort(conv[peak_candidates])[::-1]]
            new_set = set(active_k)
            added = 0
            for k in peak_order:
                if k not in new_set and added < 50:
                    new_set.add(int(k))
                    added += 1
            active_k = sorted(new_set)

        # Prune if too large: keep only the most "active" constraints
        K = len(active_k)
        if K > max_K:
            margins = np.array([M_curr - conv[k] for k in active_k])
            # Keep constraints with smallest margin (most binding)
            keep_idx = np.argsort(margins)[:max_K]
            active_k = sorted([active_k[i] for i in keep_idx])
            K = max_K

        if K == 0:
            break

        # Build and solve LP
        A_mat = _build_lp_matrix(a, active_k, n)

        if isinstance(A_mat, np.ndarray):
            A_flat = A_mat  # (K, n)
            b_vec = M_curr + conv[np.array(active_k)]
            # Constraint: A_mat @ g <= M_var + conv_k  =>  A_mat @ g - M_var <= conv_k
        else:
            A_flat = A_mat
            b_vec = M_curr + conv[np.array(active_k)]

        # LP formulation:
        # Variables: g (n, nonneg), M (free)
        # Minimize: M
        # Subject to: sum(g) = 1
        #             A_mat @ g - M <= b_vec  (i.e., A_mat@g <= M + b_vec)
        # Rewrite: A_mat@g - M <= b_vec

        # Total variables: n + 1
        # cvxpy formulation:
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1]
        if isinstance(A_flat, np.ndarray):
            constraints.append(A_flat @ g_var - M_var <= b_vec)
        else:
            constraints.append(A_flat @ g_var - M_var <= b_vec)
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.asarray(g_var.value, dtype=np.float64)
        g_val = np.maximum(g_val, 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Line search: find best alpha in (0, 1]
        alphas = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]
        best_alpha = 1.0
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * a + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = _fft_conv(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        a_new = (1.0 - best_alpha) * a + best_alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t_start
            J = 2.0 * n * best_M
            print("    CP-LP iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) +
                  "s): J=" + repr(J) + " |S|=" + str(len(active_k)))

    elapsed = time.monotonic() - t_start
    J = 2.0 * n * best_M
    if verbose:
        print("    CP-LP done (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " iters=" + str(iter_count))
    return best_a, best_M


def _peak_local_search(a0, t0, time_limit, seed=77):
    """Blame-based peak-targeted local search."""
    n = len(a0)
    a = _norm(a0)
    best_a = a.copy()
    best_M = _fft_conv(a).max()
    t_start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 500

    while time.monotonic() - t_start < time_limit and stall < max_stall:
        conv = _fft_conv(a)
        M_curr = conv.max()

        # Find top peaks (use all 2n-1, but by symmetry focus on 0..n-1)
        conv_half = conv[:n]
        n_top = min(30, n)
        top_peaks = np.argsort(conv_half)[::-1][:n_top]

        improved = False
        for k in top_peaks[:15]:
            if time.monotonic() - t_start > time_limit:
                break
            k_int = int(k)
            # Blame vector: which a_j contribute most to conv[k]
            # conv[k] = sum_j a_j * a_{k-j} (valid j)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            js = np.arange(j_start, j_end)
            blame = a[js] * a[k_int - js]
            total_blame = blame.sum()
            if total_blame < 1e-20:
                continue

            # Source: highest blame contributor
            src = int(js[np.argmax(blame)])

            # Find low-blame target (not in any top peak's support)
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                if s2 < e2:
                    in_peak[s2:e2] = True

            # Compute "score" for each position: how much it contributes to all peaks
            peak_contrib = np.zeros(n)
            for k2 in top_peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                if s2 < e2:
                    js2 = np.arange(s2, e2)
                    peak_contrib[js2] += a[k2i - js2]

            candidates = np.where(~in_peak)[0]
            if len(candidates) < n // 5:
                candidates = np.arange(n)
            dst = int(candidates[np.argmin(peak_contrib[candidates])])

            if dst == src:
                continue

            # Binary search for best transfer fraction
            src_val = a[src]
            if src_val < 1e-12:
                continue

            best_frac = 0
            best_trial_M = M_curr
            frac_range = np.logspace(-4, -1, 12)  # 0.0001 to 0.1
            for frac in frac_range:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                s_t = a_t.sum()
                if s_t < 1e-15:
                    continue
                a_t = a_t / s_t
                M_t = _fft_conv(a_t).max()
                if M_t < best_trial_M:
                    best_trial_M = M_t
                    best_frac = frac

            if best_frac > 0:
                amt = src_val * best_frac
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_M = best_trial_M
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 50 == 0:
                for _ in range(3):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = _fft_conv(a).max()
                        if M_new < best_M:
                            best_M = M_new
                            best_a = a.copy()
                            stall = 0

        if verbose_check(iter_count := iter_count, 200):
            elapsed = time.monotonic() - t_start
            J = 2.0 * n * best_M
            print("    Peak (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

    return best_a, best_M


def verbose_check(ic, period):
    return ic % period == 0


def _coord_polish(a0, t0, time_limit, seed=42):
    """Fine coordinate descent polish."""
    n = len(a0)
    a = _norm(a0)
    best_a = a.copy()
    conv = _fft_conv(a)
    best_M = conv.max()
    t_start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t_start < time_limit and stall < 300:
        conv = _fft_conv(a)
        M_curr = conv.max()
        conv_half = conv[:n]
        top_k = np.argsort(conv_half)[::-1][:min(20, n)]

        improved = False
        for k in top_k[:10]:
            if time.monotonic() - t_start > time_limit:
                break
            k_int = int(k)
            js = np.arange(max(0, k_int - n + 1), min(n, k_int + 1))
            blame = a[js] * a[k_int - js]

            src = int(js[np.argmax(blame)])
            src_val = a[src]
            if src_val < 1e-13:
                continue

            # Try small transfers to random targets
            for _ in range(5):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                    amt = src_val * frac
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    s_t = a_t.sum()
                    if s_t < 1e-15:
                        continue
                    a_t = a_t / s_t
                    M_t = _fft_conv(a_t).max()
                    if M_t < best_M - 1e-15:
                        a = a_t
                        best_M = M_t
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
        else:
            stall = 0

    return best_a, best_M


def _try_update(best_seq, best_score, a, t0, label=""):
    """Try to update best with new sequence a."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = _score(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t0
        print("*** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

    t_remaining = budget - (time.monotonic() - t0)
