"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full soft-max gradient descent with FFT-accelerated gradient computation
2. Nesterov acceleration in log-space with multiplicative updates
3. Adaptive beta schedule for soft-max approximation of max
4. Active-set coordinate descent for targeted peak reduction
5. Multi-length exploration via interpolation
6. Spectral shaping to flatten the self-convolution

import numpy as np
import time


def fast_autocorr(a):
    """Compute full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely call the evaluation function."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def score_from_ac(a):
    """Compute score from a normalized sequence (sum=1)."""
    n = len(a)
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max())


def compute_softmax_grad(a, c, beta):
    """
    Compute gradient of sum_k exp(beta * c_k) w.r.t. a_j.
    Uses FFT for efficiency.
    grad_j = 2 * sum_m a_m * exp(beta * c_{j+m}) for valid indices.
    """
    n = len(a)
    w = np.exp(beta * (c - c.max()))  # Numerical stability
    # We need: grad_j = 2 * sum_{m=0}^{n-1} a[m] * w[j+m] for j=0..n-1
    # Use FFT: convolve reversed a with w, extract middle
    a_rev = a[::-1]
    # Full convolution of a_rev (len n) and w (len 2n-1) has length 3n-3
    conv_size = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    A_rev = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_full = np.fft.irfft(A_rev * W, fft_size)[:conv_size]
    # Extract: grad_j / 2 = conv_full[j + n - 1]
    grad = 2.0 * conv_full[n - 1: 2 * n - 1]
    return grad


def compute_topk_grad(a, c, top_k, beta):
    """
    Compute gradient using only top-K lags (faster for large n).
    grad_j = sum_{k in top_k} w_k * 2 * a_{k-j} (valid indices only)
    """
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    expo = beta * (top_vals - top_vals.max())
    expo = np.clip(expo, -50, 50)
    weights = np.exp(expo)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(top_k) / top_k
    else:
        weights = weights / w_sum

    grad = np.zeros(n)
    for w_val, k in zip(weights, top_idx):
        k_int = int(k)
        # For c_k: grad_j = 2 * a_{k-j} for j in range
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start >= j_end:
            continue
        j_vals = np.arange(j_start, j_end)
        idx2 = k_int - j_vals
        valid = (idx2 >= 0) & (idx2 < n)
        grad[j_vals[valid]] += 2.0 * w_val * a[idx2[valid]]
    return grad


def nesterov_mirror_descent(a0, time_limit, lr_init=0.1, beta_init=2.0,
                            top_k_init=50, use_full_grad=False,
                            restart_every=3000, verbose=True, rng_seed=42):
    """
    Multiplicative Nesterov gradient descent minimizing soft-max of self-convolution.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    a_prev = a.copy()
    momentum = 0.8
    lr = lr_init
    beta = beta_init
    top_k = top_k_init
    stall = 0
    beta_target = 20.0  # Target beta for near-exact max

    for t in range(1, 1000000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation in multiplicative space
        ratio = a / np.maximum(a_prev, 1e-30)
        ratio = np.clip(ratio, 0.01, 100.0)
        a_nest = a * np.power(ratio, momentum)
        a_nest = norm_sum1(a_nest)

        # Compute convolution and gradient
        c = fast_autocorr(a_nest)
        if use_full_grad:
            grad = compute_softmax_grad(a_nest, c, beta)
        else:
            grad = compute_topk_grad(a_nest, c, top_k, beta)

        # Adaptive learning rate (decreasing)
        eta = lr / (1.0 + 0.02 * np.sqrt(t))

        # Multiplicative update in log-space
        log_a = np.log(np.maximum(a_nest, 1e-30)) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        a_prev = a
        a = a_new

        # Check improvement
        c_new = fast_autocorr(a)
        m_new = c_new.max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive beta: increase to approach exact max
        if t % 500 == 0:
            beta = min(beta + 0.1, beta_target)
        # Adaptive top_k
        if t % 1000 == 0 and not use_full_grad:
            top_k = min(top_k + 3, 200)

        # Restart when stuck
        if stall > restart_every:
            stall = 0
            pert_type = int(rng.integers(6))
            if pert_type == 0:
                x = np.linspace(0, 1, n)
                c_pos = float(rng.uniform(0.2, 0.8))
                sig = float(rng.uniform(0.05, 0.4))
                pert = np.exp(-((x - c_pos)**2) / (2 * sig**2))
            elif pert_type == 1:
                i_arr = np.arange(n, dtype=np.float64)
                p = float(rng.uniform(0.3, 3.0))
                pert = (i_arr + 1.0) ** p
            elif pert_type == 2:
                x = np.linspace(0, 1, n)
                freq1 = int(rng.integers(1, 4))
                freq2 = int(rng.integers(2, 7))
                pert = np.sin(np.pi * freq1 * x)**2 + 0.5 * np.sin(np.pi * freq2 * x)**2
            elif pert_type == 3:
                # Bimodal
                x = np.linspace(0, 1, n)
                sig = float(rng.uniform(0.08, 0.25))
                pert = np.exp(-((x - 0.25)**2) / (2*sig**2)) + np.exp(-((x - 0.75)**2) / (2*sig**2))
            elif pert_type == 4:
                # Inverse-U (bathtub)
                x = np.linspace(0, 1, n)
                pert = 1.0 + float(rng.uniform(0.5, 2.0)) * x**2 * (1-x)**2
            else:
                # Small perturbation of best
                pert = best_a * (1.0 + rng.normal(0, 0.08, n))
                pert = np.maximum(pert, 1e-10)

            pert = norm_sum1(pert)
            w_mix = float(rng.uniform(0.05, 0.35))
            a = norm_sum1((1 - w_mix) * best_a + w_mix * pert)
            a_prev = a.copy()
            lr = max(lr * 0.85, 0.005)
            beta = max(beta - 1.0, 1.5)

            if verbose:
                J = 2.0 * n * best_m
                print(f"    MD restart t={t} ({time.monotonic()-t0:.0f}s): J={J:.6f}, beta={beta:.1f}, top_k={top_k}")

    if verbose:
        J = 2.0 * n * best_m
        print(f"    MD done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def active_set_refinement(a0, time_limit, n_top=20, verbose=True, rng_seed=55):
    """
    Targeted refinement: for each top convolution lag, try to reduce it
    by redistributing mass from high-contributing pairs.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        M_curr = c.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50000:
                # Perturb to escape
                n_moves = int(rng.integers(3, 15))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = float(rng.uniform(0.003, 0.02))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                stall = 0
                continue

        # Get top lags
        n_top_use = min(n_top, 2 * n - 1)
        top_lags = np.argsort(c)[::-1][:n_top_use]

        improved = False
        for k_pos in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break

            k_int = int(k_pos)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            if j_start >= j_end:
                continue

            j_idx = np.arange(j_start, j_end)
            i2_idx = k_int - j_idx
            contrib = a[j_idx] * a[i2_idx]

            # Find the pair with maximum contribution to this lag
            pi = int(np.argmax(contrib))
            src1 = int(j_idx[pi])
            src2 = int(i2_idx[pi])

            # Try moving a small amount from src1 to a less impactful position
            for dst in [int(rng.integers(n)) for _ in range(5)]:
                if dst == src1 or dst == src2:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amt = a[src1] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src1] -= amt
                    a_trial[dst] += amt
                    if a_trial[src1] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

            # Also try symmetric move from src2
            if not improved:
                for dst in [int(rng.integers(n)) for _ in range(3)]:
                    if dst == src1 or dst == src2:
                        continue
                    for frac in [0.01, 0.03, 0.05]:
                        amt = a[src2] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src2] -= amt
                        a_trial[dst] += amt
                        if a_trial[src2] < 0:
                            continue
                        a_trial = a_trial / a_trial.sum()
                        m_trial = fast_autocorr(a_trial).max()
                        if m_trial < best_m - 1e-14:
                            a = a_trial
                            best_m = m_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break

    if verbose:
        J = 2.0 * n * best_m
        print(f"    Active-set done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def spectral_shape_optimize(a0, time_limit, lr_init=0.1, verbose=False):
    """
    Optimize in the frequency domain: flatten the power spectrum |A(f)|^2
    to reduce convolution peaks. Uses iterative spectral reweighting.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)

    fft_size = 1 << (2 * n - 2).bit_length()
    lr = lr_init
    stall = 0
    gamma = 2.0  # Flattening strength

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A)**2
        P_max = P.max()
        P_mean = P.mean() + 1e-30

        if P_max < 1e-30:
            break

        # Flattening: reduce peaks, boost valleys
        # A_new = A * exp(-alpha * (P - P_mean) / P_max)
        alpha = lr / (1.0 + 0.1 * np.sqrt(t))
        flatten = P / P_max
        flatten = np.clip(flatten * gamma, 0, 10)
        A_new = A * np.exp(-0.5 * alpha * (flatten - 1.0))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = norm_sum1(a_new)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 2000:
                # Reset with perturbation
                a = norm_sum1(best_a * (1.0 + rng.normal(0, 0.02, n)))
                stall = 0
                lr *= 0.8
                gamma = max(gamma - 0.3, 0.5)

    if verbose:
        J = 2.0 * n * best_m
        print(f"    Spectral done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length using cubic spline-like method."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    # Use linear interpolation then smooth
    f_int = np.interp(x_new, x_old, f_norm)
    # Light smoothing
    if new_n > 10:
        kernel = np.array([1, 2, 3, 2, 1], dtype=np.float64)
        kernel /= kernel.sum()
        f_smooth = np.convolve(f_int, kernel, mode='same')
        f_int = 0.7 * f_int + 0.3 * f_smooth
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def parametric_init(n, time_limit=15.0):
    """Quick parametric search for good initializations across families."""
    t0 = time.monotonic()
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)

    candidates = []

    # Uniform
    candidates.append(np.ones(n) / n)

    # Power laws (both directions)
    for p in [0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))

    # Gaussians
    for c_pos in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sig in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]:
            candidates.append(norm_sum1(np.exp(-((x - c_pos)**2) / (2 * sig**2))))

    # Bimodal
    for sig in [0.05, 0.1, 0.15, 0.2]:
        candidates.append(norm_sum1(
            np.exp(-((x - 0.25)**2) / (2*sig**2)) + np.exp(-((x - 0.75)**2) / (2*sig**2))))
        candidates.append(norm_sum1(
            np.exp(-((x - 0.3)**2) / (2*sig**2)) + np.exp(-((x - 0.7)**2) / (2*sig**2))))

    # Cosine-squared
    for f in [1, 2, 3, 4]:
        candidates.append(norm_sum1(np.sin(np.pi * f * x)**2))

    # Cosine modulation on uniform
    for alpha in [0.3, 0.5, 0.7, 0.9]:
        for f in [1, 2, 3]:
            candidates.append(norm_sum1(1.0 + alpha * np.cos(2 * np.pi * f * x / n * n)))

    # Exponential
    for r in [0.5, 1, 2, 5, 10]:
        candidates.append(norm_sum1(np.exp(-r * x)))
        candidates.append(norm_sum1(np.exp(r * x)))
        candidates.append(norm_sum1(np.exp(-r * (x - 0.5)**2)))

    # Bathtub (U-shaped)
    for r in [1, 2, 3, 5]:
        candidates.append(norm_sum1(1.0 + r * (x - 0.5)**2))

    # W-shaped
    for f in [2, 3]:
        candidates.append(norm_sum1(np.cos(np.pi * f * x)**4))

    # Evaluate
    best_cand = None
    best_val = float('inf')
    for c in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        v = score_from_ac(c)
        if v < best_val:
            best_val = v
            best_cand = c.copy()

    return best_cand if best_cand is not None else np.ones(n) / n


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with multi-phase optimization.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = evaluate_safe(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 5000
        best_seq = parametric_init(n0, time_limit=10.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_current = len(best_seq)

    # === Phase 0: Try alternative lengths with parametric init (fast) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 0: Alternative lengths (budget={remaining:.0f}s) ===")
    alt_lengths = [2000, 3000, 4000, 5000, 6000, 7000, 8000, 10000, 12000, 15000]
    alt_lengths = [nl for nl in alt_lengths if nl != n_current]

    for nl in alt_lengths[:6]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t_lim = min(remaining * 0.05, 8.0)
        if t_lim < 2:
            continue
        cand = parametric_init(nl, time_limit=t_lim)
        sc = evaluate_safe(cand)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            n_current = nl
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [param_n{nl}]: {sc:.6f} (t={elapsed:.0f}s)")

    # === Phase 1: Main mirror descent on best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Mirror descent (n={n_current}, budget={remaining:.0f}s) ===")
    t1 = remaining * 0.35
    if t1 > 30:
        a_out, _ = nesterov_mirror_descent(
            best_seq, t1, lr_init=0.08, beta_init=2.0,
            top_k_init=40, use_full_grad=False,
            restart_every=2500, verbose=True, rng_seed=1)
        check_update(a_out, f"md1_n{n_current}")

    # === Phase 2: Full-gradient mirror descent (more accurate) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Full-grad MD (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t2 = remaining * 0.20
    if t2 > 20:
        a_out, _ = nesterov_mirror_descent(
            best_seq, t2, lr_init=0.05, beta_init=5.0,
            top_k_init=50, use_full_grad=True,
            restart_every=2000, verbose=True, rng_seed=2)
        check_update(a_out, f"md2_full_n{len(best_seq)}")

    # === Phase 3: Spectral shaping ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Spectral (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t3 = remaining * 0.10
    if t3 > 15:
        a_out, _ = spectral_shape_optimize(best_seq, t3, lr_init=0.15, verbose=True)
        check_update(a_out, f"spectral_n{len(best_seq)}")

    # === Phase 4: Active-set refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Active-set (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t4 = remaining * 0.15
    if t4 > 15:
        a_out, _ = active_set_refinement(best_seq, t4, n_top=20, verbose=True, rng_seed=33)
        check_update(a_out, f"active_n{len(best_seq)}")

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration (budget={remaining:.0f}s) ===")

    length_tries = []
    for delta in [-1500, -1000, -500, 500, 1000, 1500, 2000, 3000, -2000, 2500]:
        nn = n_best + delta
        if 1500 <= nn <= 20000 and nn != n_best:
            length_tries.append(nn)
    for rn in [2000, 3000, 4000, 5000, 6000, 7000, 8000, 10000, 12000, 15000]:
        if rn != n_best and 1500 <= rn <= 20000:
            length_tries.append(rn)
    length_tries = sorted(set(length_tries))

    tried = 0
    for new_n in length_tries:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        if tried >= 5:
            break
        tried += 1
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")

        a_interp = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.30, 100.0)
        if t5a > 20:
            a_out, _ = nesterov_mirror_descent(
                a_interp, t5a, lr_init=0.06, beta_init=3.0,
                top_k_init=50, use_full_grad=False,
                restart_every=2000, verbose=False, rng_seed=new_n % 200)
            check_update(a_out, f"md_len{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t5b = min(remaining * 0.20, 60.0)
            if t5b > 15:
                a_out, _ = spectral_shape_optimize(best_seq, t5b, lr_init=0.1, verbose=False)
                check_update(a_out, f"spec_len{new_n}")

    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_best}, budget={remaining:.0f}s) ===")

    # Very fine gradient descent with high beta
    t6a = remaining * 0.40
    if t6a > 15:
        a_out, _ = nesterov_mirror_descent(
            best_seq, t6a, lr_init=0.02, beta_init=15.0,
            top_k_init=100, use_full_grad=True,
            restart_every=1500, verbose=False, rng_seed=42)
        check_update(a_out, f"polish_md_n{n_best}")

    # Active set refinement
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t6b = min(remaining * 0.30, 60.0)
        if t6b > 10:
            a_out, _ = active_set_refinement(best_seq, t6b, n_top=30, verbose=False, rng_seed=99)
            check_update(a_out, f"polish_active_n{n_best}")

    # Spectral
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t6c = min(remaining * 0.30, 40.0)
        if t6c > 8:
            a_out, _ = spectral_shape_optimize(best_seq, t6c, lr_init=0.05, verbose=False)
            check_update(a_out, f"polish_spec_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")

    return [float(x) for x in best_seq]
