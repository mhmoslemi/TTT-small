```python
def lse_gradient_fft(a, beta, top_k=None):
    n = len(a)
    size = 2 * n - 1  # length of autocorrelation
    
    # Compute autocorrelation
    fft1 = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft1)
    c = np.fft.irfft(A * A, fft1)[:size]
    
    # Compute softmax weights
    if top_k is not None and top_k < size:
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_top = c[idx]
        c_max = c_top.max()
        exp_c = np.zeros(size)
        exp_c[idx] = np.exp(beta * (c_top - c_max))
        w = exp_c / exp_c.sum()
    else:
        c_max = c.max()
        exp_c = np.exp(beta * (c - c_max))
        w = exp_c / exp_c.sum()
    
    # Compute gradient
    # S2[j] = (a * w)[j] for j = 0..n-1
    # Need FFT size >= n + size - 1 = 3n - 2
    fft2 = 1 << (3 * n - 2 - 1).bit_length()
    
    A2 = np.fft.rfft(a, fft2)
    W2 = np.fft.rfft(w, fft2)
    S2 = np.fft.irfft(A2 * W2, fft2)[:n]
    
    # S1[j] = (flip(a) * w)[j + n - 1]
    flip_a = a[::-1]
    FA = np.fft.rfft(flip_a, fft2)
    S1_full = np.fft.irfft(FA * W2, fft2)
    S1 = S1_full[n-1 : 2*n-1]
    
    grad = 2.0 * (S1 + S2)
    return grad, c, w
```

"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with FFT-accelerated log-sum-exp gradient (exact, O(n log n))
2. LP-based Frank-Wolfe with adaptive constraint management
3. Subgradient descent with simplex projection for polishing
4. Multi-length exploration with interpolation
5. Adaptive hyperparameter scheduling and restarts

import numpy as np
import time
import cvxpy as cp


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def lse_gradient_fft(a, beta, top_k=None):
    n = len(a)
    size = 2 * n - 1
    fft1 = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft1)
    c = np.fft.irfft(A * A, fft1)[:size]

    if top_k is not None and top_k < size:
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_top = c[idx]
        c_max = c_top.max()
        exp_c = np.zeros(size)
        exp_c[idx] = np.exp(beta * (c_top - c_max))
        w = exp_c / exp_c.sum()
    else:
        c_max = c.max()
        exp_c = np.exp(beta * (c - c_max))
        w = exp_c / exp_c.sum()

    fft2 = 1 << (3 * n - 2).bit_length()
    A2 = np.fft.rfft(a, fft2)
    W2 = np.fft.rfft(w, fft2)
    S2 = np.fft.irfft(A2 * W2, fft2)[:n]

    flip_a = a[::-1]
    FA = np.fft.rfft(flip_a, fft2)
    S1_full = np.fft.irfft(FA * W2, fft2)
    S1 = S1_full[n - 1:2 * n - 1]

    grad = 2.0 * (S1 + S2)
    return grad, c, w


def subgradient_single_peak(a, k_peak):
    n = len(a)
    s = np.zeros(n)
    j = np.arange(n)
    valid_left = (k_peak - j >= 0) & (k_peak - j < n)
    s[valid_left] += a[k_peak - j[valid_left]]
    valid_right = (j - k_peak >= 0) & (j - k_peak < n)
    s[valid_right] += a[j[valid_right] - k_peak]
    return 2.0 * s


def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def fista_optimize(a0, time_limit, beta_start=5.0, beta_end=100.0,
                   eta0=1.0, top_k_start=50, top_k_end=200,
                   verbose=True, restart_stall=6000):
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / max(time_limit, 1), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        top_k = int(top_k_start + (top_k_end - top_k_start) * progress)
        top_k = min(top_k, 2 * n - 1)

        grad, c_y, _ = lse_gradient_fft(y, beta, top_k=top_k)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-20:
            eta = eta0 / (np.sqrt(t) + 1.0)
        else:
            eta = eta0 / (np.sqrt(t) + 1.0)

        log_y = np.log(np.maximum(y, 1e-300)) - eta * grad
        log_y -= log_y.max()
        z = np.exp(log_y)
        z = np.maximum(z, 1e-300)
        z = z / z.sum()

        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = z + momentum * (z - y)
            x_new = np.maximum(x_new, 1e-300)
            x_new = x_new / x_new.sum()
        else:
            x_new = z.copy()

        c_new = fast_autocorr(x_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_stall:
            rng = np.random.default_rng(iter_count * 7 + 42)
            pert = rng.normal(0, 0.03, n) * best_x
            x_restart = norm_sum1(best_x + pert)
            x_new = x_restart
            y = x_restart.copy()
            stall = 0
            restart_stall = max(int(restart_stall * 0.7), 2000)

        y = x_new
        x = x_new
        iter_count += 1

        if verbose and iter_count % 5000 == 0:
            J = 2.0 * n * best_m
            print(f"    FISTA iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, top_k={top_k}, stall={stall}")

    return best_x, best_m


def subgradient_polish(a0, time_limit, eta0=0.05, verbose=True):
    n = len(a0)
    x = norm_sum1(a0)
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        c = fast_autocorr(x)
        k_peak = int(np.argmax(c))
        g = subgradient_single_peak(x, k_peak)
        g = g - g.mean()  # center gradient (projection onto sum=0 subspace)

        eta = eta0 / (np.sqrt(t) + 1.0)
        x_new = x - eta * g
        x_new = project_simplex(x_new)

        c_new = fast_autocorr(x_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_x = x_new.copy()
        x = x_new

        if verbose and t % 1000 == 0:
            J = 2.0 * n * best_m
            print(f"    SubGrad iter={t} ({time.monotonic()-t0:.0f}s): J={J:.6f}, k_peak={k_peak}")

    return best_x, best_m


def lp_frank_wolfe(a0, time_limit, solver, initial_K=200, max_K=800,
                   verbose=True, lp_interval=1):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()

    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0
    failed_solves = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        if iter_count % lp_interval != 0 and iter_count > 50:
            # Skip some LP solves, just do a gradient step
            c = fast_autocorr(y)
            k_peak = int(np.argmax(c))
            g = subgradient_single_peak(y, k_peak)
            g = g - g.mean()
            eta = 0.02 / (np.sqrt(iter_count) + 1.0)
            y_new = project_simplex(y - eta * g)
            if t == 1:
                x_new = y_new.copy()
            else:
                beta_t = (iter_count - 1.0) / (iter_count + 2.0)
                x_new = y_new + beta_t * (y_new - y)
                x_new = np.maximum(x_new, 0.0)
                sx = x_new.sum()
                if sx < 1e-15:
                    x_new = y_new.copy()
                else:
                    x_new = x_new / sx
            y = y_new
            x = x_new
            c_check = fast_autocorr(y)
            if c_check.max() < best_m - 1e-15:
                best_m = c_check.max()
                best_a = y.copy()
            continue

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 8:
                constraint_list = sorted(constraint_set)[:min(init_K, len(constraint_set))]
                constraint_set = set(constraint_list)
                failed_solves = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-6)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            n_add = min(30, len(violations))
            top_v = violations[np.argsort(viol_amt)[::-1][:n_add]]
            for k in top_v:
                constraint_set.add(int(k))

        # Nesterov acceleration
        alpha = 2.0 / (iter_count + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if iter_count == 1:
            x_new = y_new.copy()
        else:
            beta_t = (iter_count - 1.0) / (iter_count + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    return best_a, best_m


def parametric_init(n, ref=None):
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')

    candidates = []

    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 2.5, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr + 1.0) ** p)

    # Beta-like
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        for beta_p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            candidates.append(np.maximum(i_arr + 1.0, 0.5) ** alpha *
                              np.maximum(n - i_arr, 0.5) ** beta_p)

    # Gaussian
    for c in [0.2, 0.3, 0.35, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.08, 0.12, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))

    # Exponential
    for rate in [1, 2, 3, 5, 8]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1.0)))

    # Sine-based
    for freq in [1, 2, 3]:
        candidates.append(np.maximum(np.sin(np.pi * freq * x) ** 2, 0.01))

    # Bell-shaped
    for c in [0.4, 0.5, 0.6]:
        for w in [0.1, 0.15, 0.2, 0.25, 0.3]:
            candidates.append(np.exp(-((x - c) / w) ** 2))

    # If reference available
    if ref is not None and len(ref) == n:
        ref_norm = norm_sum1(ref)
        for w in [0.2, 0.5, 0.8]:
            candidates.append(w * ref_norm + (1.0 - w) * norm_sum1((i_arr + 1.0) ** 0.8))
            candidates.append(w * ref_norm + (1.0 - w) * norm_sum1(np.sin(np.pi * x) ** 2))

    for a_cand in candidates:
        a_norm = norm_sum1(a_cand)
        ac = fast_autocorr(a_norm)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_a = a_norm

    return best_a, best_val


def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            t6b = min(remaining * 0.3, 100.0)
            if t6b > 20:
                a_in2 = norm_sum1(best_seq if len(best_seq) == new_n else interpolate_sequence(best_seq, new_n))
                a_out2, _ = lp_frank_wolfe(a_in2, t6b, solver, initial_K=200, max_K=500,
                                           verbose=False, lp_interval=2)
                check_and_update(a_out2, f"lp_n{new_n}")

    # === Phase 7: Final LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW (n={n_best}) ===")
    if remaining > 50 and n_best <= 10000:
        t7 = min(remaining * 0.30, 200.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, initial_K=300, max_K=800,
                                      verbose=True, lp_interval=1)
            check_and_update(a_out, "lp_final")

    # === Phase 8: Final FISTA ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t8 = min(remaining * 0.5, 100.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = fista_optimize(a_in, t8, beta_start=30.0, beta_end=300.0,
                                      eta0=0.3, top_k_start=50, top_k_end=250,
                                      verbose=False, restart_stall=3000)
            check_and_update(a_out, "fista_final")

    # === Phase 9: Final subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t9 = min(remaining * 0.5, 60.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = subgradient_polish(a_in, t9, eta0=0.01, verbose=False)
            check_and_update(a_out, "subgrad_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
