def fast_autocorr(a):

    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def upsample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum() if a.sum() > 1e-15 else a
    old_x = np.linspace(0.0, 1.0 - 1.0/old_n, old_n)
    new_x = np.linspace(0.0, 1.0 - 1.0/new_n, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def pocs_refine(a0, time_limit, n_trials=3, rng_seed=7):
    """POCS: alternate flat-spectrum projection with non-negativity."""
    n = len(a0)
    best_a = norm_sum1(a0)
    best_m = fast_autocorr(best_a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit - 5:
            break
        t_trial = max(5.0, (time_limit - (time.monotonic() - t0)) / (n_trials - trial))

        rng = np.random.default_rng(rng_seed + trial)
        a = norm_sum1(a0)

        for it in range(500000):
            if time.monotonic() - t0 > time_limit - 5:
                break

            # Frequency domain: push toward flat magnitude
            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            nonzero = mag > 1e-12
            if not np.any(nonzero):
                break
            target_mag = np.mean(mag[nonzero])
            blend = min(0.3, 0.05 + 0.25 * it / min(it + 1, 200))
            new_mag = (1 - blend) * mag + blend * target_mag
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]

            # Time domain: non-negativity + normalization
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new / s

            M_new = fast_autocorr(a_new).max()
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = a_new.copy()
            a = a_new

    return best_a, best_m


def lp_fw_away(a0, time_limit, solver, K_init=200, K_max=None, verbose=False,

    iter_count = 0

            binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build constraint matrix T_sel (K_sel x n)
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        # Solve LP: max sum(b) s.t. T_sel @ b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K * 2, K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Store dual for next iteration's constraint selection
        try:
            dual_vec = np.abs(np.asarray(prob.dual_values, dtype=np.float64))
            dual_full = np.zeros(2 * n - 1)
            dual_full[binding_idx] = dual_vec[:len(dual_vec)]
            # EMA update
            if dual_weights is None:
                dual_weights = dual_full
            else:
                dual_weights = 0.7 * dual_weights + 0.3 * dual_full
        except Exception:
            pass

        # Line search with away-step
        base_obj = M_curr

        # Standard FW step
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        best_step_a = a.copy()

        # Try away-step: (1-lambda)*a + lambda*((1-mu)*away_a... no
        # Away-step: new_a = (1 - alpha*(1-mu))*a + alpha*mu*away_a + alpha*(1-mu... 
        # Actually: x_new = (1 - alpha*(1 - mu)) * x + alpha * mu * v_away + alpha*(1-alpha)...
        # Standard away-step FW:
        # x_new = (1 - t_a) * x + t_a * p where p = b/sum(b) (standard)
        # Away: x_new = (1 - t) * x + t * p, but also allow x_new = (1 + mu*t)*x - mu*t*away_v + t*p
        # Let's do: direction = g_dir - away_lambda * (a - away_a)
        if away_a is not None and rng.random() < away_prob:
            # Away direction
            away_dir = g_dir - away_lambda * (a - away_a)
            away_dir = np.maximum(away_dir, 0.0)
            away_dir_sum = away_dir.sum()
            if away_dir_sum > 1e-15:
                away_dir = away_dir / away_dir_sum
                for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    a_trial = (1.0 - alpha) * a + alpha * away_dir
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_trial_m:
                        best_trial_m = m_trial
                        best_alpha = alpha
                        best_step_a = a_trial.copy()
                        # Update away variables
                        if m_trial < best_m:
                            away_a = best_step_a.copy()
                            away_lambda = 0.5

        # Standard FW step
        for alpha in [2.0/(iter_count+2), 0.3, 0.2, 0.15, 0.1, 0.5, 0.7]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha
                best_step_a = a_trial.copy()

        a = best_step_a
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a.copy()
            # Reset away info on improvement
            if rng.random() < 0.3:
                away_a = a.copy()
                away_lambda = 0.3

        iter_count += 1

        # Adaptive K
        if iter_count % 5 == 0:
            K = min(K + 50, K_max)

        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LPAFW it={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0, restart_every=5000,
                 rng_seed=42):
    """Away-Step Mirror Descent with adaptive parameters."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

    return best_a, best_m


def spectral_anneal(a0, time_limit, rng_seed=100):
    """Spectral annealing: gradually flatten spectrum with simulated annealing acceptance."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    fft_size = 1 << (2 * n - 2).bit_length()
    T = 0.5  # initial temperature
    T_min = 0.01
    cool_rate = 0.9995
    stall = 0

    it = 0
    while time.monotonic() - t0 < time_limit and stall < 500:
        it += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Spectral perturbation
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        # Add noise to magnitude
        noise = rng.normal(0, T, len(mag))
        new_mag = mag + noise
        new_mag = np.maximum(new_mag, 1e-10)
        # Push toward flat
        mean_mag = np.mean(new_mag)
        new_mag = 0.8 * new_mag + 0.2 * mean_mag
        A_new = new_mag * np.exp(1j * phase)
        a_trial = np.fft.irfft(A_new, fft_size)[:n]
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / s_t

        M_trial = fast_autocorr(a_trial).max()
        delta = M_trial - M_curr

        # Metropolis acceptance
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_trial
            T *= cool_rate
            if M_trial < best_m - 1e-16:
                best_m = M_trial
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > 200:
            T = max(T * 2, T_min)
            stall = 100

    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=123):
    """Targeted coordinate descent on high-impact parameters."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:15]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            max_idx_local = np.argmax(contrib)
            src = idx[max_idx_local]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best sink
            for frac in [0.02, 0.01, 0.05, 0.005, 0.03]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                # Try moving to a random low-impact index
                for _ in range(3):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m - 1e-16:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def multi_scale_optimize(a0, time_limit, solver, rng_seed=0):
    """Solve at half resolution, upsample, then refine at full resolution."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Half-resolution solve
    n_half = max(200, n // 2)
    a_half = upsample(a, n_half)
    t_half = min(time_limit * 0.3, 120)

    a_h_out, _ = lp_fw_away(a_half, t_half, solver, K_init=100,
                            K_max=min(2*n_half-1, 2000), verbose=False)
    # Upsample back
    a_up = upsample(a_h_out, n)
    m_up = fast_autocorr(a_up).max()
    if m_up < best_m:
        best_m = m_up
        best_a = a_up.copy()

    # Refine at full resolution
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 30:
        a_ref, _ = lp_fw_away(best_a, remaining, solver, K_init=200,
                              K_max=min(2*n-1, 4000), verbose=False,
                              rng_seed=rng_seed+1)
        m_ref = fast_autocorr(a_ref).max()
        if m_ref < best_m:
            best_m = m_ref
            best_a = a_ref.copy()

    return best_a, best_m


def make_structures(n, seeds=None):
    """Generate diverse initial structures."""
    if seeds is None:
        seeds = [0, 1, 2, 3, 4]
    x = np.linspace(0, 1, n)
    structs = []

    # Sine squared
    a = np.sin(np.pi * x) ** 2
    structs.append(norm_sum1(a))

    # U-shape
    a = 1.0 + 2.0 * (2*x - 1)**2
    structs.append(norm_sum1(a))

    # Bimodal
    a = np.exp(-((x-0.2)/0.1)**2) + np.exp(-((x-0.8)/0.1)**2) + 0.01
    structs.append(norm_sum1(a))

    # Cosine modulated
    a = (1.0 + 0.4*np.cos(2*np.pi*x))**2
    a = np.maximum(a, 0)
    structs.append(norm_sum1(a))

    # Random
    rng = np.random.default_rng(99)
    a = rng.random(n)
    structs.append(norm_sum1(a))

    return structs


def propose_candidate(budget_s=1800.0):
    """Main search: AW-PW + POCS + multi-scale + coordinate polish."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x)**2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Quick POCS (45s) ===
    print(f"\n=== Phase 1: POCS (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.05, 45.0)
    if t1 > 10:
        a_pocs, _ = pocs_refine(best_seq, t1, n_trials=3, rng_seed=7)
        try_update(a_pocs, "pocs")

    # === Phase 2: Main LP-FW with away-step (600s) ===
    print(f"\n=== Phase 2: LP-FW Away (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.38, 600.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        K_init = min(300, 2*n0-1)
        K_max = min(2*n0-1, 5000)
        a_out, _ = lp_fw_away(a_in, t2, solver, K_init=K_init, K_max=K_max,
                              verbose=True, away_prob=0.4, rng_seed=42)
        try_update(a_out, "lpfw_main")

    # === Phase 3: Spectral Annealing (150s) ===
    print(f"\n=== Phase 3: Spectral Anneal (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_anneal(a_in, t3, rng_seed=100)
        try_update(a_out, "spec_anneal")

    # === Phase 4: AMD refinement (200s) ===
    print(f"\n=== Phase 4: AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.14, 200.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t4, top_k=60, beta=6.0, c0=5.0,
                                restart_every=4000, rng_seed=77)
        try_update(a_out, "amd")

    # === Phase 5: Second LP-FW round (300s) ===
    print(f"\n=== Phase 5: LP-FW R2 (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.22, 300.0)
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        K_init = min(500, 2*n0-1)
        K_max = min(2*n0-1, 6000)
        a_out, _ = lp_fw_away(a_in, t5, solver, K_init=K_init, K_max=K_max,
                              verbose=True, away_prob=0.5, rng_seed=200)
        try_update(a_out, "lpfw_r2")

    # === Phase 6: Multi-length exploration (150s) ===
    print(f"\n=== Phase 6: Multi-length ===")
    remaining = time_budget - (time.monotonic() - t_global)
    trial_lengths = []
    for delta in [1000, -1000, 2000, -2000, 500]:
        new_n = n0 + delta
        if 1000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        t_try = min(remaining * 0.4, 100)
        if t_try < 20:
            continue
        print(f"  Trying n={new_n}...")
        a_interp = upsample(best_seq, new_n)
        a_out, _ = lp_fw_away(a_interp, t_try, solver,
                              K_init=min(200, 2*new_n-1),
                              K_max=min(2*new_n-1, 3000),
                              verbose=False, rng_seed=new_n%100)
        try_update(a_out, f"len_{new_n}")

    # === Phase 7: Final LP-FW polish (remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final LP-FW (n={len(best_seq)}, t={remaining:.0f}s) ===")
    if remaining > 30:
        a_in = norm_sum1(best_seq)
        n_best = len(a_in)
        K_init = min(400, 2*n_best-1)
        K_max = min(2*n_best-1, 6000)
        a_out, _ = lp_fw_away(a_in, remaining * 0.7, solver,
                              K_init=K_init, K_max=K_max,
                              verbose=True, away_prob=0.3, rng_seed=500)
        try_update(a_out, "lpfw_final")

    # === Phase 8: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 8: Coord Polish (t={remaining:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, min(remaining * 0.8, 60), rng_seed=789)
        try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x)**2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
