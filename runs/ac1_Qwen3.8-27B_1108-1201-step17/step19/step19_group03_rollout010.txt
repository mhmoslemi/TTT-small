```python
def lp_framinge_wolfe_improved(a0, time_limit, solver, K_init=200, K_max=5000, K_increase=50, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    K = K_init
    it = 0
    total_lag = 2*n - 1
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        K_sel = min(K, total_lag)
        # Select active constraints: top K autocorrelation values
        binding_idx = np.argsort(ac)[::-1][:K_sel]
        
        # Build constraint matrix T where T[i,j] is the coefficient of a[j] in (a*b)[binding_idx[i]]
        # (a*b)[k] = sum_j a[j] * b[k-j]
        # So (T @ b)[i] = sum_j T[i,j] * b[j] = sum_j a[j] * b[binding_idx[i] - j]
        # T[i,j] = a[binding_idx[i] - j] if binding_idx[i] - j is in [0, n-1], else 0
        
        T = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n-1, ki)
            for j in range(j_start, j_end + 1):
                T[ii, j] = a[ki - j]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + K_increase, K_max)
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + K_increase, K_max)
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Backtracking line search
        base_m = M_curr
        best_alpha = 2.0 / (it + 2.0)  # default FW step size
        best_trial_m = base_m
        
        # Try several alpha values
        alphas = [best_alpha, 2*best_alpha, 0.5*best_alpha, 0.3, 0.2, 0.1, 0.5, 0.7]
        for alpha in alphas:
            alpha = min(alpha, 1.0)
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha
        
        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        
        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()
        
        it += 1
        if it % 20 == 0:
            K = min(K + K_increase, K_max)
        
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW2 it={it} ({elapsed:.0f}s): M={m_new:.6f}, best={best_m:.6f}, K={K_sel}, a={best_alpha:.4f}")
    
    return best_a, best_m
```

```python
        if M < best_m:
            best_m = M
            best_a = a.copy()

        t_trial_start = time.monotonic()
        for it in range(200000):
            if time.monotonic() - t_trial_start > t_trial:
                break
            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            alpha_blend = 0.08 + 0.2 * min(1.0, it / 200.0)
            new_mag = (1 - alpha_blend) * mag + alpha_blend * mag_mean
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s
            M_new = fast_autocorr(a_new).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a_new.copy()
            a = a_new

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, K_init=300, K_max=None,
                   K_increase=50, use_geometric=True, top_K_max_abs=None):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    total_lag = 2 * n - 1

    if K_max is None:
        K_max = min(3 * n, total_lag)
    if top_K_max_abs is not None:
        K_max = min(K_max, top_K_max_abs)

    K = K_init
    it = 0
    fw_gap_history = []

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-18:
            break

        K_sel = min(K, total_lag)
        binding_idx = np.argsort(ac)[::-1][:K_sel]

        T = np.zeros((K_sel, n))
        for ii in range(K_sel):
            ki = int(binding_idx[ii])
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            if j_start <= j_end:
                T[ii, j_start:j_end + 1] = a[ki - j_start:ki - j_end - 1]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + K_increase, K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + K_increase // 2, K_max)
            continue

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            K = min(K + K_increase, K_max)
            continue

        # FW gap: if b_sum/M_curr is close to 1, we're at a local optimum
        fw_gap = 1.0 - b_sum / (M_curr * n) if M_curr > 0 else 0
        fw_gap_history.append(fw_gap)

        g_dir = b_val / b_sum

        # Line search
        base_m = M_curr
        best_alpha = 2.0 / (it + 2.0)
        best_trial_m = base_m

        # Try multiple alpha values
        alphas = [best_alpha, min(1.0, best_alpha * 1.5), 0.5, 0.3, 0.2, 0.1, 0.7, 0.15]
        for alpha in alphas:
            if alpha <= 0 or alpha > 1:
                continue
            if use_geometric:
                log_a = np.log(np.maximum(a, 1e-30))
                log_d = np.log(np.maximum(g_dir, 1e-30))
                log_new = (1.0 - alpha) * log_a + alpha * log_d
                a_trial = np.exp(log_new - log_new.max())
            else:
                a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        # Apply update
        if use_geometric:
            log_a = np.log(np.maximum(a, 1e-30))
            log_d = np.log(np.maximum(g_dir, 1e-30))
            log_new = (1.0 - best_alpha) * log_a + best_alpha * log_d
            a = np.exp(log_new - log_new.max())
        else:
            a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            a = best_a.copy()
        else:
            a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a.copy()

        it += 1
        if it % 25 == 0:
            K = min(K + K_increase, K_max)

        # Adaptive: if gap is very small, increase K more aggressively
        if len(fw_gap_history) > 5:
            recent_gap = np.mean(fw_gap_history[-5:])
            if recent_gap < 0.01:
                K = min(K + K_increase * 2, K_max)

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW it={it} ({elapsed:.0f}s): M={m_new:.7f}, best={best_m:.7f}, K={K_sel}, gap={fw_gap:.4f}")

    return best_a, best_m


def lse_gradient_descent(a0, time_limit, verbose=False, tau_init=50.0, tau_final=500.0,
                         eta_init=0.3, rng_seed=200):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 1000:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(1.0, time_limit * 0.8))
        tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(a)
        ac_shifted = ac / tau
        ac_shifted -= ac_shifted.max()
        log_weights = np.exp(ac_shifted)
        log_weights /= log_weights.sum()

        # Gradient of logsumexp sum
        g = np.zeros(n)
        top_k = min(100, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:top_k]
        w = log_weights[top_idx]
        w_sum = w.sum()
        if w_sum < 1e-15:
            continue
        w = w / w_sum
        for wi, k in zip(w, top_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * wi * a[ki - idx]

        eta = eta_init / np.sqrt(1.0 + it * 0.01)
        a_trial = a - eta * g
        a_new = project_simplex(a_trial)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.01, 1.5)
        else:
            stall += 1
            if stall > 200:
                eta *= 0.7
            if stall > 500:
                pert = rng.normal(0, 0.01, n) * a
                a_p = np.maximum(a + pert, 0.0)
                a = a_p / a_p.sum()
                stall = 100
        a = a_new

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD it={it} ({elapsed:.0f}s): M={m_new:.7f}, best={best_m:.7f}, tau={tau:.1f}, stall={stall}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=300):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    n_top = 30

    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        # Compute sensitivity
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:15]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved = False

        # Try moving mass from high-sensitivity to low-sensitivity positions
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            cand_sens = np.abs(sens[candidates])
            top_cands = candidates[np.argsort(cand_sens)[:5]]

            found = False
            for dst in top_cands:
                if dst == src or found:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord ({elapsed:.0f}s): M={best_m:.7f}, stall={stall}")

    return best_a, best_m


def slsqp_polish(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Only polish if n is manageable
    if n > 8000:
        return best_a, best_m

    K = min(200, 2 * n - 1)

    def objective_and_grad(x):
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            return 1e10, np.zeros(n)
        x = x / s
        ac = fast_autocorr(x)
        top_idx = np.argsort(ac)[::-1][:K]
        M = ac.max()
        # Subgradient of max
        g_x = np.zeros(n)
        top_vals = ac[top_idx]
        w = top_vals ** 10.0
        w /= w.sum()
        for wi, k in zip(w, top_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            idx = np.arange(j_start, j_end + 1)
            g_x[idx] += wi * a[ki - idx]
        return M, g_x

    # Use scipy with a few iterations
    x0 = a.copy()

    def fun(x):
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            return 1e10
        x = x / s
        ac = fast_autocorr(x)
        return ac.max()

    def cons(x):
        return [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0, 'jac': lambda x: np.ones(n)}]

    try:
        res = sp_minimize(fun, x0, method='SLSQP', options={'maxiter': 50, 'ftol': 1e-10},
                          constraints=cons(x0))
        if res.x is not None:
            a_new = np.maximum(res.x, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
                m_new = fast_autocorr(a_new).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a_new.copy()
    except Exception:
        pass

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    SLSQP ({elapsed:.0f}s): M={best_m:.7f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 6200
        x = np.linspace(0, 1, n_base)
        best_seq = 1.0 + 2.0 * x ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Diverse initialization + POCS ===
    print(f"\n=== Phase 1: Initialization (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.06, 100.0)

    if t_phase1 > 20:
        inits = [norm_sum1(best_seq)]
        labels = ["current"]
        inits.append(make_u_shape(n0, 2.0))
        labels.append("u2")
        inits.append(make_u_shape(n0, 4.0))
        labels.append("u4")
        inits.append(make_bimodal(n0, 0.2))
        labels.append("bimodal")
        inits.append(make_cosine_mod(n0, 0.4, 1))
        labels.append("cos1")
        inits.append(make_cosine_mod(n0, 0.3, 3))
        labels.append("cos3")
        inits.append(make_w_shape(n0, 0.4))
        labels.append("w")
        inits.append(spectral_init(n0, 25, 42))
        labels.append("spectral")

        for init, lbl in zip(inits, labels):
            sc = evaluate_safe(init)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = init.copy()
                print(f"    Init '{lbl}': score={sc:.6f}")

        # Quick POCS on best
        t_pocs = min(t_phase1 * 0.5, 50.0)
        if t_pocs > 10:
            a_pocs, _ = pocs_optimize(best_seq, t_pocs, n_trials=3, rng_seed=7)
            try_update(a_pocs, "pocs")

    # === Phase 2: Main LP-FW ===
    print(f"\n=== Phase 2: LP-FW Main (n={n0}, current best={best_score:.6f}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase2 = min(remaining * 0.40, 600.0)

    if t_phase2 > 30:
        a_in = norm_sum1(best_seq)
        K_init = min(400, 2 * n0 - 1)
        K_max = min(2000, 2 * n0 - 1)
        a_out, _ = lp_frank_wolfe(a_in, t_phase2, solver, verbose=True,
                                    K_init=K_init, K_max=K_max, K_increase=40,
                                    use_geometric=True)
        try_update(a_out, "lfw_main")

        # Second pass with larger K
        remaining = time_budget - (time.monotonic() - t_global)
        t_phase2b = min(remaining * 0.15, 200.0)
        if t_phase2b > 30:
            a_in = norm_sum1(best_seq)
            K_init = min(800, 2 * n0 - 1)
            K_max = min(3000, 2 * n0 - 1)
            a_out, _ = lp_frank_wolfe(a_in, t_phase2b, solver, verbose=False,
                                       K_init=K_init, K_max=K_max, K_increase=60,
                                       use_geometric=True)
            try_update(a_out, "lfw_main2")

    # === Phase 3: LSE Gradient Descent ===
    print(f"\n=== Phase 3: LSE-GD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase3 = min(remaining * 0.12, 180.0)
    if t_phase3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = lse_gradient_descent(a_in, t_phase3, verbose=True,
                                         tau_init=30.0, tau_final=300.0,
                                         eta_init=0.5, rng_seed=200)
        try_update(a_out, "lse_gd")

    # === Phase 4: LP-FW with linear (non-geometric) steps ===
    print(f"\n=== Phase 4: LP-FW Linear (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase4 = min(remaining * 0.12, 180.0)
    if t_phase4 > 20:
        a_in = norm_sum1(best_seq)
        K_init = min(500, 2 * n0 - 1)
        K_max = min(2500, 2 * n0 - 1)
        a_out, _ = lp_frank_wolfe(a_in, t_phase4, solver, verbose=False,
                                    K_init=K_init, K_max=K_max, K_increase=50,
                                    use_geometric=False)
        try_update(a_out, "lfw_lin")

    # === Phase 5: Coordinate polish ===
    print(f"\n=== Phase 5: Coordinate (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase5 = min(remaining * 0.08, 120.0)
    if t_phase5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t_phase5, verbose=False, rng_seed=456)
        try_update(a_out, "coord")

    # === Phase 6: SLSQP polish ===
    print(f"\n=== Phase 6: SLSQP (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase6 = min(remaining * 0.06, 90.0)
    if t_phase6 > 20 and n0 <= 7000:
        a_in = norm_sum1(best_seq)
        a_out, _ = slsqp_polish(a_in, t_phase6, verbose=True)
        try_update(a_out, "slsqp")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length exploration (base n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)

    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 1500, -1500]:
        new_n = n0 + delta
        if 2000 <= new_n <= 9000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n0][:3]

    for new_n in trial_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.25, 100.0)
        if t7a > 20:
            a_out, _ = lp_frank_wolfe(a_long, t7a, solver, verbose=False,
                                       K_init=300, K_max=min(2000, 2 * new_n - 1),
                                       K_increase=40, use_geometric=True)
            try_update(a_out, f"lfw_n{new_n}")

    # === Phase 8: Final combined optimization ===
    print(f"\n=== Phase 8: Final optimization (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    if remaining > 20:
        t8a = min(remaining * 0.35, 150.0)
        if t8a > 15:
            a_in = norm_sum1(best_seq)
            K_init = min(600, 2 * n_best - 1)
            K_max = min(3500, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe(a_in, t8a, solver, verbose=False,
                                       K_init=K_init, K_max=K_max, K_increase=50,
                                       use_geometric=True)
            try_update(a_out, "lfw_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8b = min(remaining * 0.3, 100.0)
            if t8b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_polish(a_in, t8b, verbose=False, rng_seed=789)
                try_update(a_out, "coord_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t8c = min(remaining * 0.3, 80.0)
            if t8c > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = lse_gradient_descent(a_in, t8c, verbose=False,
                                                 tau_init=20.0, tau_final=100.0,
                                                 eta_init=0.4, rng_seed=500)
                try_update(a_out, "lse_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 6200
        x = np.linspace(0, 1, n_fb)
        best_seq = 1.0 + 2.0 * x ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
```
