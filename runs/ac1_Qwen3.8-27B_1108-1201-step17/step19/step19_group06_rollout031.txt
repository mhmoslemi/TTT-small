import numpy as np
import time
from scipy.optimize import linprog

def fast_autocorr(a):
    """Compute full convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def score_from_ac(a):
    n = len(a)
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max())

def solve_lp(f, M, n):
    """
    Solve LP: maximize sum(g) subject to A @ g <= M, g >= 0
    where A[k,i] = f[i] if 0 <= k-i < n, else 0
    """
    # Build constraint matrix A (sparse would be better, but let's start dense)
    # A has shape (2n-1, n)
    # A[k, i] = f[i] if 0 <= k-i < n, else 0
    
    # For each row k, the valid columns i are: max(0, k-n+1) <= i <= min(k, n-1)
    # A is banded with bandwidth n
    
    # Use sparse matrix for efficiency
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(f[i_lo:i_hi+1].tolist())
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: maximize sum(g) = -minimize -sum(g)
    c = -np.ones(n)
    
    result = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), method='highs')
    
    if result.success:
        return result.x
    else:
        return np.zeros(n)
```

Now let me write the optimized code. The key computation per iteration is:
1. Compute ac = conv(a, a) via FFT (O(n log n))
2. Compute w = softmax(beta * (ac - max)) (O(n))
3. Compute grad = 2*beta * cross_corr(w, a) via FFT (O(n log n))
4. Mirror descent update (O(n))

Total per iteration: O(n log n), which for n=5200 is about 5200*13 ≈ 68K operations. Very fast!

We can do millions of iterations in the time budget. The bottleneck will be the FFT computation in numpy, which for n=5200 (fft_size = 16384) takes about 10-20 microseconds. So we can do ~50K-100K iterations per second.

In 300 seconds, that's 15M-30M iterations. More than enough to converge.

Now, the question is: does this gradient method actually converge to the global optimum? Probably not in general (the problem is non-convex), but it should get close with good initialization and multiple restarts.

Let me now write the complete solution:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth log-sum-exp mirror descent with Nesterov acceleration
2. LP-based alternating minimization for exact max optimization
3. Multi-restart with diverse initializations
4. Length exploration via interpolation
5. Final coordinate descent polish
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_autocorr(a):
    """Compute full convolution (autocorrelation) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    """Safely call the evaluation function."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def score_from_ac(a):
    """Compute score from a normalized sequence (sum=1)."""
    n = len(a)
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max())

def compute_gradient_smooth(a, weight_vec, beta):
    """
    Compute gradient of log-sum-exp smooth approximation.
    
    Parameters:
        a: sequence of length n (normalized, sum=1)
        weight_vec: softmax weights over convolution lags, shape (2n-1,)
        beta: temperature parameter
    
    Returns:
        grad: gradient vector of shape (n,)
    """
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(weight_vec, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(W * np.conj(A), fft_size)
    grad = 2.0 * beta * corr[:n]
    return grad

def smooth_mirror_descent(a0, time_limit, beta_init=100.0, lr_init=0.3,
                          verbose=True, rng_seed=42):
    """
    Mirror descent with log-sum-exp smooth approximation.
    Nesterov acceleration in multiplicative space.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    beta = float(beta_init)
    lr = float(lr_init)
    momentum = 0.9
    a_prev = a.copy()
    stall = 0
    it = 0
    
    fft_size_corr = 1 << (3 * n - 2).bit_length()
    fft_size_ac = 1 << (2 * n - 2).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov step in multiplicative space
        ratio = a / np.maximum(a_prev, 1e-30)
        ratio = np.clip(ratio, 0.01, 100.0)
        a_nest = a * np.power(ratio, momentum)
        a_nest = norm_sum1(a_nest)
        
        # Compute autocorrelation
        A_fft = np.fft.rfft(a_nest, fft_size_ac)
        ac = np.fft.irfft(A_fft * A_fft, fft_size_ac)[:2*n-1]
        
        # Compute softmax weights
        max_ac = ac.max()
        expo = beta * (ac - max_ac)
        expo = np.clip(expo, -100, 0)
        weights = np.exp(expo)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.zeros_like(ac)
            weights[np.argmax(ac)] = 1.0
        else:
            weights = weights / w_sum
        
        # Compute gradient via cross-correlation
        W = np.fft.rfft(weights, fft_size_corr)
        An = np.fft.rfft(a_nest, fft_size_corr)
        corr = np.fft.irfft(W * np.conj(An), fft_size_corr)
        grad = 2.0 * beta * corr[:n]
        
        # Mirror descent update (multiplicative)
        eta = lr / (1.0 + 0.05 * np.sqrt(it))
        log_a = np.log(np.maximum(a_nest, 1e-30)) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        a_prev = a
        a = a_new
        
        # Check improvement
        m_new = ac.max()  # We already computed this
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adapt beta: increase over time (less smooth, more accurate)
        if it % 1000 == 0:
            beta = min(beta * 1.05, 10000.0)
        
        # Restart if stuck
        if stall > 2000:
            stall = 0
            pert_type = int(rng.integers(3))
            if pert_type == 0:
                pert = best_a * (1.0 + rng.normal(0, 0.03, n))
            elif pert_type == 1:
                x = np.linspace(0, 1, n)
                c = float(rng.uniform(0.2, 0.8))
                sig = float(rng.uniform(0.1, 0.4))
                pert = np.exp(-((x - c)**2) / (2 * sig**2))
            else:
                I = np.arange(n, dtype=float)
                p = float(rng.uniform(0.5, 2.0))
                pert = (I + 1.0) ** p
            a = norm_sum1(0.7 * best_a + 0.3 * norm_sum1(pert))
            a_prev = a.copy()
            lr *= 0.8
            
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Smooth MD done ({time.monotonic()-t0:.0f}s, {it} iters): J={J:.6f}")
    return best_a, best_m

def build_conv_matrix(f, n):
    """Build sparse convolution matrix for LP."""
    all_k = np.arange(2 * n - 1)
    i_lo = np.maximum(0, all_k - n + 1)
    i_hi = np.minimum(n - 1, all_k)
    counts = i_hi - i_lo + 1
    total = int(counts.sum())
    
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    
    offset = 0
    for k in range(2 * n - 1):
        cnt = int(counts[k])
        lo = int(i_lo[k])
        hi = int(i_hi[k])
        rows[offset:offset+cnt] = k
        cols[offset:offset+cnt] = np.arange(lo, hi + 1)
        vals[offset:offset+cnt] = f[lo:hi+1]
        offset += cnt
    
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def lp_iteration(f, M, n, time_limit_lp=20.0):
    """
    Solve LP: maximize sum(g) s.t. A@f * g <= M, g >= 0
    where A[k,i] = f[i] if 0<=k-i<n else 0
    Returns optimal g (unnormalized) or None if LP fails.
    """
    A = build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs',
                     options={'time_limit': time_limit_lp, 'presolve': True})
    
    if result.x is not None and result.x.sum() > 0:
        return np.maximum(result.x, 0.0)
    return None

def lp_alternating(f0, time_limit, max_iters=200, verbose=True, rng_seed=55):
    """
    LP-based alternating minimization.
    At each step: fix f, solve LP for best g, update f.
    """
    n = len(f0)
    f = norm_sum1(f0)
    best_f = f.copy()
    best_m = fast_autocorr(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit and it < max_iters:
        ac = fast_autocorr(f)
        M = ac.max()
        
        # Solve LP
        g = lp_iteration(f, M, n, time_limit_lp=15.0)
        if g is None or g.sum() < 1e-10:
            stall += 1
            if stall > 5:
                break
            it += 1
            continue
        
        # Rescale g to sum=1
        g_norm = g / g.sum()
        
        # Try different step sizes
        improved = False
        for t_step in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            f_trial = norm_sum1((1.0 - t_step) * f + t_step * g_norm)
            m_trial = fast_autocorr(f_trial).max()
            if m_trial < M - 1e-12:
                f = f_trial
                improved = True
                break
        
        it += 1
        m_curr = fast_autocorr(f).max()
        if m_curr < best_m - 1e-14:
            best_m = m_curr
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and it % 5 == 0:
            J = 2.0 * n * best_m
            print(f"      LP iter {it} ({time.monotonic()-t0:.0f}s): M={M:.8f}, best_J={J:.6f}")
        
        if stall > 10:
            # Perturb and continue
            f = norm_sum1(best_f * (1.0 + rng.normal(0, 0.01, n)))
            stall = 0
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    LP alternating done ({time.monotonic()-t0:.0f}s, {it} iters): J={J:.6f}")
    return best_f, best_m

def coordinate_descent(a0, time_limit, rng_seed=77, verbose=True):
    """
    Coordinate descent: try moving mass between pairs to reduce max convolution.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break
        
        # Focus on the top lags
        n_top = min(20, 2*n-1)
        top_lags = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k_pos in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            
            k_int = int(k_pos)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n - 1, k_int)
            if i_lo > i_hi:
                continue
            
            # Find the pair with maximum contribution
            idx = np.arange(i_lo, i_hi + 1)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            
            pi = int(np.argmax(contrib))
            src = int(idx[pi])
            dst = int(j_idx[pi])
            
            if src == dst:
                continue
            
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Try moving mass from src to a random destination
            for frac in [0.05, 0.1, 0.2, 0.3]:
                amt = src_val * frac
                if amt < 1e-14:
                    continue
                
                # Try several destinations
                for dst_try in [dst, int(rng.integers(n))]:
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst_try] += amt
                    if a_trial[src] < 0 or a_trial[dst_try] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = float(rng.uniform(0.01, 0.05))
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Coord descent done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
