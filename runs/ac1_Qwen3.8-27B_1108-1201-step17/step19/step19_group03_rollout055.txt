def smooth_max_gradient(a, tau):
    """Compute gradient of log-sum-exp approximation of max(conv(a,a))."""
    n = len(a)
    b = fast_autocorr(a)  # full convolution
    # Numerical stability: shift by max
    b_max = b.max()
    shifted = tau * (b - b_max)
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        # Fallback: uniform over top values
        p = np.zeros_like(b)
        top_k = np.argsort(b)[-10:]
        p[top_k] = 1.0/10
    else:
        p = p / p_sum
    
    # Gradient: c[i] = 2 * sum_m p[i+m] * a[m] for m=0..n-1

"""

import numpy as np
import time
from scipy.signal import correlate as sp_correlate

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    import cvxpy as cp
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_fast(a):
    n = len(a)
    b = fast_conv(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * b.max() / (s * s)


def norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def smooth_max_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*b[k])) w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Numerical stability
    if shifted.max() > 500:
        shifted = shifted - shifted.max()
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        top_k = np.argsort(b)[-min(20, len(b)):]
        p = np.zeros_like(b)
        p[top_k] = 1.0 / len(top_k)
    else:
        p = p / p_sum
    # gradient[i] = 2 * sum_m p[i+m] * a[m] for m=0..n-1
    # This is cross-correlation of p and a, 'valid' mode gives n values
    c = 2.0 * sp_correlate(p, a, mode='valid')
    return c


def nesterov_optimize(a0, time_limit, tau_start=5.0, tau_end=80.0,
                      c0=1.0, verbose=False, rng_seed=0,
                      n_restarts=3, momentum_start=0.9):
    """Nesterov accelerated gradient descent on smooth max of convolution."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_val = eval_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    total_time = time_limit
    n_iter_total = 500000
    restart_size = n_iter_total // max(n_restarts, 1)

    for restart in range(n_restarts):
        if time.monotonic() - t0 > total_time - 5:
            break
        t_restart_start = time.monotonic() - t0
        if restart > 0:
            # Perturb from best
            pert = rng.normal(0, 0.02 / np.sqrt(n), n) * best_a
            a = np.maximum(best_a + pert, 1e-10)
            a = a / a.sum()

        a_prev = a.copy()
        momentum = momentum_start
        eta = c0 / np.sqrt(n)

        for t in range(1, restart_size + 1):
            if time.monotonic() - t0 > total_time - 5:
                break

            # Adaptive tau
            progress = t / restart_size
            tau = tau_start * (tau_end / tau_start) ** progress

            # Nesterov look-ahead
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy

            # Gradient at y
            g = smooth_max_grad(y, tau)
            # Tangent projection (remove mean for sum=1 constraint)
            g_tan = g - g.mean()

            # Step
            eta_t = c0 / (np.sqrt(t + 1.0) * np.sqrt(n))
            a_new = y - eta_t * g_tan
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new

            # Evaluate
            val_new = eval_fast(a_new)

            # Backtracking (limited)
            if val_new > best_val + 1e-14:
                # Try smaller step
                for backtrack in range(3):
                    eta_t *= 0.5
                    a_new = y - eta_t * g_tan
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new < 1e-15:
                        break
                    a_new = a_new / s_new
                    val_new = eval_fast(a_new)
                    if val_new <= best_val + 1e-14:
                        break

            if val_new < best_val:
                best_val = val_new
                best_a = a_new.copy()

            # Update momentum (Nesterov)
            momentum = (1 + np.sqrt(1 + 4 * momentum * momentum)) / 2.0
            momentum = min(momentum, 2.0)

            a_prev = a
            a = a_new

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov restart {restart} ({elapsed:.0f}s): val={best_val:.6f}")

    return best_a, best_val


def coord_polish(a0, time_limit, verbose=False, rng_seed=0):
    """Coordinate descent: move mass from peak-contributing coords to low-sensitivity coords."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_val = eval_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_conv(a)
        M_curr = b.max()
        top_lags = np.argsort(b)[::-1][:15]

        # Compute sensitivity: how much each coordinate contributes to top peaks
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_lags[:8]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        # Find the coordinate that most contributes to the max peak
        k_max = int(np.argmax(b))
        s_max = max(0, k_max - n + 1)
        e_max = min(n - 1, k_max)
        idx_max = np.arange(s_max, e_max + 1)
        contrib = a[idx_max] * a[k_max - idx_max]
        src = idx_max[np.argmax(contrib)]
        src_val = a[src]

        if src_val < 1e-12:
            stall += 1
            continue

        # Find destination: low sensitivity, not in peak
        candidates = np.where(~in_peak)[0]
        if len(candidates) == 0:
            candidates = np.arange(n)
        cand_sens = sens[candidates]
        # Pick a few best destinations
        top_dests = candidates[np.argsort(cand_sens)[:5]]

        improved = False
        for dst in top_dests:
            if dst == src or improved:
                continue
            for frac in [0.02, 0.01, 0.05, 0.005, 0.1]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                M_trial = fast_conv(a_trial).max()
                if M_trial < M_curr * (1 - 1e-15):
                    a = a_trial
                    best_a = a.copy()
                    best_val = eval_fast(a)
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random perturbation
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.005, 0.03)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    val = eval_fast(a)
                    if val < best_val:
                        best_val = val
                        best_a = a.copy()
                        stall = 0
        else:
            stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord ({elapsed:.0f}s): val={best_val:.6f}, stall={stall}")

    return best_a, best_val


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K_init=200, top_K_max=3000):
    """LP-based Frank-Wolfe step."""
    import cvxpy as cp
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_val = eval_fast(a)
    t0 = time.monotonic()
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        M_curr = b.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(b)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            iter_count += 1
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            iter_count += 1
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            iter_count += 1
            continue
        g_dir = b_val / b_sum

        # Line search for step size
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = M_curr
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), 0.5, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_conv(a).max()
        if m_new < best_val / (2.0 * n):  # comparing normalized
            pass
        val_new = eval_fast(a)
        if val_new < best_val:
            best_val = val_new
            best_a = a.copy()

        iter_count += 1
        if iter_count % 10 == 0:
            K = min(K + 30, top_K_max)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): val={best_val:.6f}, K={K_sel}")

    return best_a, best_val


def make_raised_cosine(n):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    return norm1(a)


def make_sine_mod(n, freq=1, alpha=0.3):
    x = np.arange(n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0.0)
    return norm1(a)


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.2) / width) ** 2) + np.exp(-((x - 0.8) / width) ** 2) + 0.005
    return norm1(a)


def make_triangle(n):
    x = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    a = a ** 2 + 0.01
    return norm1(a)


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def pocs_init(n, n_trains=3, time_per=10.0, rng_seed=0):
    """POCS-style flat spectrum initialization."""
    best_a = make_raised_cosine(n)
    best_m = fast_conv(best_a).max()
    rng = np.random.default_rng(rng_seed)
    fft_size = 1 << (2 * n - 2).bit_length()

    for trial in range(n_trains):
        t_start = time.monotonic()
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        A_spec = np.exp(1j * phases)
        A_full = np.zeros(fft_size, dtype=np.complex128)
        A_full[:fft_size // 2 + 1] = A_spec
        for i in range(1, fft_size // 2):
            A_full[fft_size - i] = np.conj(A_spec[i])
        a = np.fft.irfft(A_full, fft_size)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s

        # POCS iterations
        n_pocs = 500
        for it in range(n_pocs):
            if time.monotonic() - t_start > time_per:
                break
            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            alpha_blend = 0.1 + 0.4 * min(1.0, it / 100)
            new_mag = (1 - alpha_blend) * mag + alpha_blend * mag_mean
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            a = a_new

        m = fast_conv(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()

    return best_a

        try_update(a_out, "nest_fine")

    # === Phase 6: Coordinate polish round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Coord polish 2 ===")
    t6 = min(remaining * 0.1, 120.0)
    if t6 > 20:
        a_out, _ = coord_polish(best_seq, t6, verbose=False, rng_seed=789)
        try_update(a_out, "coord2")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")

    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 1500, 2000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 10000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]
    trial_lengths = trial_lengths[:3]

    for new_n in trial_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)...")
        a_long = interpolate_seq(best_seq, new_n)

        t7a = min(remaining * 0.4, 150.0)
        if t7a > 20:
            a_out, _ = nesterov_optimize(a_long, t7a, tau_start=5.0, tau_end=50.0,
                                          c0=1.2, verbose=False, rng_seed=new_n % 1000,
                                          n_restarts=2, momentum_start=0.85)
            try_update(a_out, f"nest_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.3, 60.0)
            if t7b > 10:
                a_out, _ = coord_polish(interpolate_seq(best_seq, new_n), t7b,
                                         verbose=False, rng_seed=new_n + 100)
                try_update(a_out, f"coord_n{new_n}")

    # === Phase 8: Final Nesterov + Coord on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final optimization (n={n_best}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.4, 120.0)
        if t8 > 20:
            a_out, _ = nesterov_optimize(best_seq, t8, tau_start=40.0, tau_end=150.0,
                                          c0=0.5, verbose=True, rng_seed=999,
                                          n_restarts=3, momentum_start=0.92)
            try_update(a_out, "nest_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8b = min(remaining * 0.5, 80.0)
            if t8b > 10:
                a_out, _ = coord_polish(best_seq, t8b, verbose=False, rng_seed=1234)
                try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = eval_fast(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = make_raised_cosine(n_fb)
        best_score = eval_fast(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
