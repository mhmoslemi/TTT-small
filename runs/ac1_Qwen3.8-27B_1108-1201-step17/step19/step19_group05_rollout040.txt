"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using soft-subgradient descent with
FFT-accelerated gradient, adaptive beta annealing, Nesterov momentum,
multi-phase optimization, and diverse smooth initializations.

import numpy as np
import time


def fast_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgrad(a, w):
    """
    Compute g[j] = 2 * sum_k a[k] * w[j+k] via FFT cross-correlation.
    a has length n, w has length 2n-1.
    Returns g of length n.
    """
    n = len(a)
    L = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    X = np.fft.irfft(W * np.conj(A), L)[:n]
    return 2.0 * X


def proj_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def softmax_weights(C, beta):
    """Compute soft-max weights over C using log-sum-exp for stability."""
    C_shifted = beta * (C - C.max())
    exp_C = np.exp(C_shifted)
    return exp_C / exp_C.sum()


def make_ramp(n, alpha=1.0):
    """Create (i+1)^alpha sequence, normalized."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** alpha
    return a / a.sum()


def make_raised_cosine(n):
    """Raised cosine window, normalized."""
    x = np.arange(n, dtype=np.float64)
    t = x / (n - 1) if n > 1 else 0.0
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_gaussian(n, sigma=0.3):
    """Gaussian window centered, normalized."""
    x = np.arange(n, dtype=np.float64)
    t = (x - n / 2.0) / n
    a = np.exp(-0.5 * (t / sigma) ** 2)
    return a / a.sum()


def make_super_gaussian(n, p=4.0):
    """Super-Gaussian: exp(-(2t-1)^p), normalized."""
    x = np.arange(n, dtype=np.float64)
    t = 2.0 * x / (n - 1) - 1.0 if n > 1 else 0.0
    a = np.exp(-np.abs(t) ** p)
    return a / a.sum()


def make_skew_ramp(n, alpha=1.0, skew=0.0):
    """Skewed ramp: (i+1)^alpha * exp(skew * i/n), normalized."""
    x = np.arange(1, n + 1, dtype=np.float64)
    t = np.arange(n, dtype=np.float64) / n
    a = x ** alpha * np.exp(skew * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return make_ramp(n, alpha)
    return a / s


def interpolate_to_length(a, new_n):
    """Linearly interpolate sequence to new length, preserving normalization."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss


def soft_subgrad_optimize(a0, time_limit, beta0=5.0, beta_end=50.0,
                          lr0=1.0, momentum_decay=0.9, rng_seed=42,
                          verbose=False):
    """
    Soft-subgradient descent with Nesterov-style momentum.
    a0: initial point on simplex (sum=1, non-negative)
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    stall_limit = 5000
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        # Anneal beta: from beta0 to beta_end
        beta = beta0 + (beta_end - beta0) * progress ** 0.5

        # Nesterov momentum
        beta_nm = 0.5 / (1.0 + np.sqrt(total_iters + 1.0) / 10.0)
        y = x + beta_nm * (x - x_prev)
        y = np.maximum(y, 1e-30)
        y = y / y.sum()

        # Compute convolution and soft-max weights
        C = fast_conv(y)
        w = softmax_weights(C, beta)

        # Compute subgradient via FFT
        g = compute_subgrad(y, w)

        # Normalize gradient for stable step size
        g_norm = np.linalg.norm(g, ord=np.inf)
        if g_norm < 1e-20:
            break
        g_normed = g / g_norm

        # Step size: diminish slowly
        lr = lr0 / (1.0 + 0.01 * np.sqrt(total_iters + 1))

        # Update
        x_new = proj_simplex(x - lr * g_normed)

        x_prev = x
        x = x_new
        total_iters += 1

        # Track best
        C_x = fast_conv(x)
        M_x = C_x.max()
        if M_x < best_m - 1e-14:
            best_m = M_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart with perturbation if stuck
        if stall > stall_limit:
            stall = 0
            strength = 0.05 + 0.03 * rng.random()
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-15)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

        if verbose and total_iters % 20000 == 0:
            print(f"      SSG it={total_iters} ({elapsed:.0f}s): J={2*n*best_m:.6f}, beta={beta:.1f}, stall={stall}")

    return best_a, best_m


def peak_shaving(a0, time_limit, verbose=False, rng_seed=42):
    """
    Targeted peak reduction: identify top peaks and redistribute mass
    from high-contribution positions to low-contribution positions.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        C = fast_conv(a)
        # Get top K peaks
        K = min(5, 2 * n - 1)
        peak_indices = np.argpartition(C, -K)[-K:]
        peak_indices = peak_indices[np.argsort(C[peak_indices])[::-1]]

        # Compute "cost" map: contribution of each a[j] to all top peaks
        cost = np.zeros(n)
        for k in peak_indices:
            ki = int(k)
            j_lo = max(0, ki - n + 1)
            j_hi = min(n - 1, ki)
            if j_lo <= j_hi:
                j_range = np.arange(j_lo, j_hi + 1)
                cost[j_range] += a[ki - j_range] * C[ki]

        # Find highest-cost source
        src = int(np.argmax(cost * a))  # weight by both cost and current value
        if a[src] < 1e-12:
            stall += 1
            break

        # Find lowest-cost destination (prefer low a[j] and low cost)
        dst_score = cost * 0.5 + a
        dst = int(np.argmin(dst_score))
        if dst == src:
            # Find second best
            dst_score[src] = np.inf
            dst = int(np.argmin(dst_score))

        # Try to move mass
        moved = False
        for frac in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            amt = a[src] * frac
            if amt < 1e-15:
                break
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                break
            a_trial = a_trial / s_t
            M_t = fast_conv(a_trial).max()
            if M_t < best_m - 1e-14:
                a = a_trial
                best_m = M_t
                best_a = a.copy()
                moved = True
                stall = 0
                break

        if not moved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.01, 0.08)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def freq_phase_optimize(a0, time_limit, verbose=False):
    """
    Frequency-domain phase adjustment: modify the phase of A(k) to reduce
    peak of IFFT(A*A) while trying to maintain a >= 0.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)

    lr = 0.1
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute C from current A
        C = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
        C_max = C.max()
        k_max = int(np.argmax(C))

        # Gradient of C[k_max] w.r.t. phase of A
        # C[k] = (1/N) sum_m A[m]^2 exp(2 pimk/N)
        # dC[k]/dphase[m] = 2 * |A[m]|^2 * Im(conj(exp(2pi (k)*m/N))))  ... complex
        # Simpler: just do a small perturbation in phase and check improvement

        # Perturb phase at a few random frequencies
        n_perturb = min(5, fft_size // 2)
        perts = np.random.randint(1, fft_size // 2, size=n_perturb)
        dphase = np.random.uniform(-lr, lr, size=n_perturb)

        A_trial = A.copy()
        for p, dp in zip(perts, dphase):
            A_trial[p] += 1j * dp * mag[p]
            if p < fft_size - p:
                A_trial[fft_size - p] = np.conj(A_trial[p])

        a_trial = np.fft.irfft(A_trial, fft_size)[:n]
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 0.01:
            lr *= 0.9
            continue
        a_trial = a_trial / s_t
        M_t = fast_conv(a_trial).max()

        if M_t < best_m - 1e-14:
            best_m = M_t
            best_a = a_trial.copy()
            a = a_trial
            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            phase = np.angle(A)
            lr = min(lr * 1.05, 0.5)
        else:
            lr = max(lr * 0.95, 0.01)

        if verbose and it % 1000 == 0:
            print(f"      Freq it={it} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}, lr={lr:.4f}")

    return best_a, best_m


def gradient_step_exact(a, C, n_top=10):
    """
    Compute a weighted subgradient using top n_top peaks with exponential weighting.
    More targeted than the soft-max over all peaks.
    """
    n = len(a)
    conv_len = 2 * n - 1
    top = np.argpartition(C, -n_top)[-n_top:]
    vals = C[top]
    max_v = vals.max()
    if max_v < 1e-30:
        return np.zeros(n)
    # Exponential weights
    w = np.exp((vals - max_v) / (max_v * 0.1 + 1e-30))
    w = w / w.sum()

    g = np.zeros(n)
    for idx_i, k in enumerate(top):
        ki = int(k)
        wi = w[idx_i] * C[ki]  # weight by peak value too
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo <= j_hi:
            j_range = np.arange(j_lo, j_hi + 1)
            g[j_range] += wi * 2.0 * a[ki - j_range]
    return g


def multi_peak_subgrad(a0, time_limit, verbose=False, rng_seed=42):
    """
    Subgradient method using weighted combination of top-peak subgradients.
    More targeted than full soft-max, faster per iteration.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    total = 0
    lr = 0.5

    while time.monotonic() - t0 < time_limit:
        # Nesterov
        beta_nm = 0.5 / (1.0 + np.sqrt(total + 1.0) / 20.0)
        y = x + beta_nm * (x - x_prev)
        y = np.maximum(y, 1e-30)
        y = y / y.sum()

        C = fast_conv(y)
        g = gradient_step_exact(y, C, n_top=10)
        g_norm = np.linalg.norm(g, ord=np.inf)
        if g_norm < 1e-20:
            break
        g_normed = g / g_norm

        step = lr / (1.0 + 0.005 * np.sqrt(total + 1))
        x_new = proj_simplex(x - step * g_normed)
        x_prev = x
        x = x_new
        total += 1

        M_x = fast_conv(x).max()
        if M_x < best_m - 1e-14:
            best_m = M_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            stall = 0
            strength = 0.03 + 0.02 * rng.random()
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-15)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

        if verbose and total % 10000 == 0:
            print(f"      MPG it={total} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m

        t2 = min(remaining * 0.15, 200.0)
        a_out, _ = peak_shaving(best_seq, t2, rng_seed=42)
        try_update(a_out, "peak_shave_1")

        a_out, _ = peak_shaving(best_seq, t2, rng_seed=777)
        try_update(a_out, "peak_shave_2")

    # Phase 3: Frequency domain
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Frequency Domain ===")
    if remaining > 40 and best_seq is not None:
        t3 = min(remaining * 0.10, 100.0)
        a_out, _ = freq_phase_optimize(best_seq, t3)
        try_update(a_out, "freq")

    # Phase 4: Coordinate / peak shaving with different seeds
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: More Peak Shaving ===")
    if remaining > 60 and best_seq is not None:
        t4 = min(remaining * 0.15, 150.0)
        for seed in [1234, 5678, 9012]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            a_out, _ = peak_shaving(best_seq, min(t4, remaining - 10), rng_seed=seed)
            try_update(a_out, f"peak_s_{seed}")

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for new_n in [n_best + 1000, n_best + 2000, max(3000, n_best - 1500)]:
            if new_n <= 0 or new_n > 12000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            a_interp = interpolate_to_length(best_seq, new_n)
            try_update(a_interp, f"interp_n{new_n}")

            t5 = min(remaining * 0.20, 120.0)
            if t5 > 20:
                a_out, _ = soft_subgrad_optimize(a_interp, t5, beta0=10.0, beta_end=60.0,
                                                 lr0=0.5, rng_seed=100 + new_n % 100)
                try_update(a_out, f"ssg_n{new_n}")

    # Phase 6: Final combined polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final Polish ===")
    if remaining > 40 and best_seq is not None:
        # SSG then peak shave
        t6a = min(remaining * 0.35, 120.0)
        t6b = min(remaining * 0.35, 80.0)
        if t6a > 15:
            a_out, _ = soft_subgrad_optimize(best_seq, t6a, beta0=20.0, beta_end=100.0,
                                             lr0=0.3, rng_seed=555)
            try_update(a_out, "ssg_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if t6b > 10 and remaining > 20:
            a_out, _ = peak_shaving(best_seq, min(t6b, remaining - 10), rng_seed=4242)
            try_update(a_out, "peak_final")

    # Phase 7: One more SSG with very different params
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        t7 = min(remaining * 0.30, 80.0)
        a_out, _ = soft_subgrad_optimize(best_seq, t7, beta0=50.0, beta_end=200.0,
                                         lr0=0.2, rng_seed=31337)
        try_update(a_out, "ssg_sharp")

    # Final validation
    if best_seq is not None:
        fs = safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        best_seq = make_ramp(6200, 1.0)
        best_score = safe_eval(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
