def fast_conv(a):
    """Full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def subgrad_multi_peak(a, conv, n_peaks=80, width_factor=0.05):
    """Weighted subgradient of max(conv) using softmax over top peaks."""
    n = len(a)
    if n_peaks > conv.size:

    top_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]

        w[-1] = 1.0

    grad = np.zeros(n)

        width = max(width_min, width_initial * np.exp(-t / 5000.0))
        g = subgrad_multi_peak(y, conv, n_peaks=n_peaks, width_factor=width)

        eta = lr0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = conv.max()
        M_x = fast_conv(x).max()
        M_eff = min(M_new, M_x)
        if M_eff < best_m - 1e-15:
            best_m = M_eff
            best_a = (x if M_x <= M_new else y).copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            strength = 0.05 + 0.02 * min(5, stall // restart_every)
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-15)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

        if verbose and t % 5000 == 0:
            J = 2 * n * best_m
            print(f"      MD t={t} ({time.monotonic()-t0:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def peak_shaving(a0, time_limit, n_peaks_target=5, rng_seed=42, verbose=False):
    """Targeted coordinate descent: move mass from top-peak contributors to low-cost positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        conv = fast_conv(a)
        top_ks = np.argsort(conv)[-n_peaks_target:]

        changed = False
        for k in top_ks:
            k = int(k)
            p_lo = max(0, k - n + 1)
            p_hi = min(n - 1, k)
            if p_lo > p_hi:
                continue
            ps = np.arange(p_lo, p_hi + 1)
            contrib = a[ps] * a[k - ps]
            src_local = ps[np.argmax(contrib)]

            if a[src_local] < 1e-12:
                continue

            # Compute cost of each position (contribution to top-5 peaks)
            costs = np.zeros(n)
            for k2 in top_ks:
                k2i = int(k2)
                lo2 = max(0, k2i - n + 1)
                hi2 = min(n - 1, k2i)
                if lo2 <= hi2:
                    v2 = np.arange(lo2, hi2 + 1)
                    costs[v2] += a[k2i - v2]

            # Find low-cost, low-value destinations
            combined_cost = costs * 0.5 + a * 1.0
            sorted_idx = np.argsort(combined_cost)
            n_tries = min(10, n)
            moved = False
            for dst_pos in range(n_tries):
                dst = sorted_idx[dst_pos]
                if dst == src_local:
                    continue
                for frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
                    amt = a[src_local] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src_local] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_t = fast_conv(a_trial).max()
                    if M_t < best_m - 1e-15:
                        a = a_trial
                        best_m = M_t
                        best_a = a.copy()
                        changed = True
                        moved = True
                        break
                if moved:
                    break

        iteration += 1
        if not changed:
            stall += 1
            # Perturb to escape
            if stall % 10 == 0:
                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))
                if i1 != i2 and a[i1] > 1e-10:
                    amt = a[i1] * rng.uniform(0.02, 0.1)
                    a[i1] -= amt
                    a[i2] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        else:
            stall = 0

        if verbose and iteration % 100 == 0:
            print(f"      Peak it={iteration} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m


def spectral_reshape(a0, time_limit, n_iterations=3000, verbose=False):
    """Flatten the DFT magnitude, project back to non-negative simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    rng = np.random.default_rng(123)

    for it in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_mean = P.mean()
        if P_max < 1e-30:
            break

        # Gradually increase flattening strength
        alpha = 0.2 + 0.6 * min(1.0, it / 300.0)
        # Geometric mean interpolation for smoothness
        P_log = np.log(np.maximum(P, 1e-30))
        P_log_new = P_log + alpha * (np.log(P_mean + 1e-30) - P_log)
        P_new = np.exp(P_log_new)
        # Also try power-law flattening
        blend = 0.3
        P_new = (1 - blend) * P + blend * np.full_like(P, P_mean) * (P / np.maximum(P, 1e-30)) ** 0.5
        P_new = np.maximum(P_new, 1e-30)

        # Reconstruct with modified magnitude, keeping phase
        mag_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        mag_ratio = np.clip(mag_ratio, 0.3, 3.0)
        A_new = A * mag_ratio
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blended update
            a = 0.7 * a + 0.3 * a_new
            a = np.maximum(a, 0.0)
            a = a / max(a.sum(), 1e-15)
            M_c = fast_conv(a).max()
            if M_c < best_m:
                best_m = M_c
                best_a = a.copy()

        if verbose and it % 200 == 0:
            print(f"      Spec it={it} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m


def interp_to_length(a, new_n):
    """Interpolate to new length preserving shape."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss


def safe_eval(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def make_chirp_nonneg(n, rng):
    """Create a non-negative chirp-like sequence."""
    x = np.arange(n, dtype=np.float64) / n
    # Cosine-based shape with varying frequency
    freq = rng.uniform(2, 8)
    a = np.abs(np.cos(np.pi * freq * x + rng.uniform(0, 2 * np.pi))) ** rng.uniform(0.5, 2.0)
    a += 0.1 * np.ones(n)
    s = a.sum()
    return a / s


def make_two_pulse(n, rng):
    """Two Gaussian pulses at random positions."""
    c1 = rng.uniform(0.1, 0.4)
    c2 = rng.uniform(0.6, 0.9)
    w1 = rng.uniform(0.02, 0.15)
    w2 = rng.uniform(0.02, 0.15)
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((x - c1) / w1) ** 2) + np.exp(-((x - c2) / w2) ** 2)
    s = a.sum()
    return a / s


def make_uniform(n):
    return np.ones(n) / n


def make_power_law(n, alpha):
    """a_i proportional to (i+1)^(-alpha), creating a decaying shape."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** (-alpha)
    s = a.sum()
    return a / s


def propose_candidate(budget_s=1800.0):
    """Main search function combining multiple optimization phases."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = safe_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = safe_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    rng = np.random.default_rng(2024)
    n_base = 6200 if parent is None else len(parent)

    # Generate diverse initializations
    candidates = []
    if parent is not None:
        p_norm = parent / parent.sum()
        candidates.append((p_norm, "parent"))

    # Structured initializations
    candidates.append((make_uniform(n_base), "uniform"))
    candidates.append((make_power_law(n_base, 0.5), "powlaw_0.5"))
    candidates.append((make_power_law(n_base, 1.0), "powlaw_1.0"))

    # Random structured
    for i in range(4):
        candidates.append((make_chirp_nonneg(n_base, np.random.default_rng(100 + i)), f"chirp_{i}"))
    for i in range(3):
        candidates.append((make_two_pulse(n_base, np.random.default_rng(200 + i)), f"2pulse_{i}"))

    print(f"Initial candidates: {len(candidates)}")
    for a, label in candidates:
        sc = safe_eval(a)
        if np.isfinite(sc):
            print(f"  Init [{label}]: {sc:.6f}")

    # Evaluate and sort
    scored = []
    for a, label in candidates:
        sc = safe_eval(a)
        scored.append((sc, a, label))
    scored.sort(key=lambda x: x[0])

    # Phase 1: Mirror descent on top candidates
    print(f"\n=== Phase 1: Mirror Descent ===")
    n_cands = min(6, len(scored))
    remaining = time_budget - (time.monotonic() - t_global)
    t_per = max(30.0, min(250.0, remaining * 0.35 / n_cands))

    for i in range(n_cands):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget:
            break
        t_avail = min(t_per, time_budget - elapsed)
        if t_avail < 15:
            break
        sc0, a0, label = scored[i]
        print(f"  MD [{label}] (t={t_avail:.0f}s, init={sc0:.6f})...")
        a_out, _ = mirror_descent_v2(a0, t_avail, n_peaks=80, lr0=4.0,
                                      width_initial=0.05, width_min=0.01,
                                      restart_every=2000, rng_seed=42 + i * 7, verbose=False)
        try_update(a_out, f"md_{label}")

    # Phase 2: Spectral reshaping on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Spectral Reshaping ===")
    if remaining > 60 and best_seq is not None:
        t2 = min(remaining * 0.12, 150.0)
        a_in = best_seq / max(best_seq.sum(), 1e-15)
        a_out, _ = spectral_reshape(a_in, t2, n_iterations=2000, verbose=True)
        try_update(a_out, "spectral")

    # Phase 3: Peak shaving (coordinate descent)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Peak Shaving ===")
    if remaining > 50 and best_seq is not None:
        t3 = min(remaining * 0.15, 200.0)
        a_in = best_seq / max(best_seq.sum(), 1e-15)
        a_out, _ = peak_shaving(a_in, t3, n_peaks_target=5, rng_seed=777, verbose=True)
        try_update(a_out, "peak_shave")

    # Phase 4: Second MD round with different params
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: MD Round 2 (aggressive) ===")
    if remaining > 60 and best_seq is not None:
        t4 = min(remaining * 0.18, 250.0)
        a_in = best_seq / max(best_seq.sum(), 1e-15)
        a_out, _ = mirror_descent_v2(a_in, t4, n_peaks=120, lr0=6.0,
                                      width_initial=0.03, width_min=0.005,
                                      restart_every=1500, rng_seed=999, verbose=True)
        try_update(a_out, "md_round2")

    # Phase 5: Interleaved spectral + MD
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Spectral + MD Interleaved ===")
    if remaining > 80 and best_seq is not None:
        a_in = best_seq / max(best_seq.sum(), 1e-15)
        t5_total = min(remaining * 0.20, 250.0)
        t5_spec = t5_total * 0.4
        t5_md = t5_total * 0.6
        a_out, _ = spectral_reshape(a_in, t5_spec, n_iterations=1500, verbose=False)
        try_update(a_out, "spec5")
        a_in2 = a_out / max(a_out.sum(), 1e-15)
        a_out2, _ = mirror_descent_v2(a_in2, t5_md, n_peaks=100, lr0=3.0,
                                       width_initial=0.04, width_min=0.008,
                                       restart_every=3000, rng_seed=555, verbose=False)
        try_update(a_out2, "md5")

    # Phase 6: Peak shaving again on improved solution
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Peak Shaving Round 2 ===")
    if remaining > 40 and best_seq is not None:
        t6 = min(remaining * 0.12, 120.0)
        a_in = best_seq / max(best_seq.sum(), 1e-15)
        a_out, _ = peak_shaving(a_in, t6, n_peaks_target=3, rng_seed=31415, verbose=False)
        try_update(a_out, "peak_shave2")

    # Phase 7: Try different lengths via interpolation
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Multi-length ===")
    if best_seq is not None and remaining > 80:
        n_best = len(best_seq)
        for new_n in [n_best + 1000, n_best + 2000, max(2000, n_best - 1000)]:
            if new_n <= 0 or new_n > 15000 or remaining < 60:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            a_interp = interp_to_length(best_seq, new_n)
            try_update(a_interp, f"interp_n{new_n}")
            t7 = min(remaining * 0.25, 120.0)
            if t7 > 20:
                a_out, _ = mirror_descent_v2(a_interp, t7, n_peaks=80, lr0=3.5,
                                              width_initial=0.04, width_min=0.01,
                                              restart_every=2500, rng_seed=700 + new_n % 100, verbose=False)
                try_update(a_out, f"md_n{new_n}")
            remaining = time_budget - (time.monotonic() - t_global)

    # Phase 8: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final Polish ===")
    if remaining > 30 and best_seq is not None:
        a_in = best_seq / max(best_seq.sum(), 1e-15)
        # Quick spectral
        t8a = min(remaining * 0.3, 60.0)
        if t8a > 10:
            a_out, _ = spectral_reshape(a_in, t8a, n_iterations=1000, verbose=False)
            try_update(a_out, "spec_final")
            a_in = a_out / max(a_out.sum(), 1e-15)
        # Quick MD
        remaining = time_budget - (time.monotonic() - t_global)
        t8b = min(remaining * 0.4, 80.0)
        if t8b > 10:
            a_out, _ = mirror_descent_v2(a_in, t8b, n_peaks=150, lr0=2.0,
                                          width_initial=0.02, width_min=0.005,
                                          restart_every=4000, rng_seed=12345, verbose=False)
            try_update(a_out, "md_final")
        # Quick peak shave
        remaining = time_budget - (time.monotonic() - t_global)
        t8c = min(remaining * 0.3, 50.0)
        if t8c > 10:
            a_out, _ = peak_shaving(a_in, t8c, n_peaks_target=2, rng_seed=99999, verbose=False)
            try_update(a_out, "peak_final")

    # Final validation
    if best_seq is not None:
        fs = safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        best_seq = make_uniform(6200)
        best_score = safe_eval(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
