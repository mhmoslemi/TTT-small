    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_constraint_matrix(y, lag_indices, n):
    K = len(lag_indices)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lag_indices):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[ki - idx]
    return coeffs

def lp_frank_wolfe(y0, time_limit, solver, init_K=600, max_K=4000, verbose=True):
    n = len(y0)
    y = normalize(y0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:init_K])
    no_improve = 0
    stall_limit = 1500

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        clist = sorted(active_set)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = sorted(active_set)

        K = len(clist)
        if K == 0:
            break

        coeffs = build_constraint_matrix(y, clist, n)
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        lp_ok = False
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                lp_ok = True
        except Exception:
            if len(active_set) > init_K:
                active_set = set(int(k) for k in clist[:init_K])
                clist = sorted(active_set)

        if not lp_ok:
            if no_improve > stall_limit // 2:
                break
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violating constraints
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_yg > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_yg[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                active_set.add(int(k))

        # Exact 1D line search: minimize max_k ((1-t)*y + t*g_dir) convolved
        # Use ternary/golden section search on [0, 1]
        def trial_obj(alpha):
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                return 1e10
            trial = trial / ts
            return fast_autocorr(trial).max()

        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c = lo + (1.0 - gr) * (hi - lo)
        dd = lo + gr * (hi - lo)
        fc = trial_obj(c)
        fd = trial_obj(dd)

        for _ in range(40):
            if fc < fd:
                hi = dd
                dd = c
                fd = fc
                c = lo + (1.0 - gr) * (hi - lo)
                fc = trial_obj(c)
            else:
                lo = c
                c = dd
                fc = fd
                dd = lo + gr * (hi - lo)
                fd = trial_obj(dd)

        best_alpha = (lo + hi) / 2.0
        trial = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        ts = trial.sum()
        if ts > 1e-15:
            trial = trial / ts
            Mt = fast_autocorr(trial).max()
            if Mt < best_m:
                best_m = Mt
                best_a = trial.copy()

        y = trial if ts > 1e-15 else y.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(2 * n * best_m) + " |S|=" + str(len(active_set)))

    elapsed = time.monotonic() - t0
    if verbose:
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s) J=" + repr(2 * n * best_m))
    return best_a, 2.0 * n * best_m

def mult_grad_descent(y0, time_limit, top_k=100, temp=40.0, c0=3.0, verbose=True):
    n = len(y0)
    y = normalize(y0)
    best_a = y.copy()
    best_val = obj_val(y)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    y_prev = y.copy()
    stall = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 10000:
            break

        beta_t = (t - 1.0) / (t + 1.0)
        y_ex = y + beta_t * (y - y_prev)
        y_ex = np.maximum(y_ex, 1e-15)
        y_ex = y_ex / y_ex.sum()

        ac = fast_autocorr(y_ex)
        kk = min(top_k, len(ac))
        idx = np.argsort(ac)[::-1][:kk]
        top_vals = ac[idx]
        w = np.exp(temp * (top_vals - top_vals[0]))
        w = w / w.sum()

        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            idx_arr = np.arange(i_s, i_e)
            grad[idx_arr] += 2.0 * wi * y_ex[ki - idx_arr]

        eta_t = c0 / (np.sqrt(max(t, 1)) + 1.0)
        log_a = np.log(y_ex) - eta_t * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        y_prev = y
        y = a_new

        val = obj_val(y)
        if val < best_val - 1e-13:
            best_val = val
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        if stall > 5000 and stall % 5000 == 0:
            noise = 0.02 * rng.standard_normal(n) * best_a
            y_r = best_a + noise
            y_r = np.maximum(y_r, 1e-12)
            y_r = y_r / y_r.sum()
            y = y_r
            y_prev = y_r.copy()
            stall = 0

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    MGDes t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(best_val))

    return best_a, best_val

def peak_shaving(y0, time_limit, n_targets=8, verbose=True):
    n = len(y0)
    a = normalize(y0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    fracs = [0.002, 0.004, 0.008, 0.015, 0.03, 0.05]

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        targets = np.argsort(ac)[::-1][:n_targets]
        M_curr = ac.max()
        improved = False

        for k in targets:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]

            # Find top contributors
            top_contrib = np.argsort(contrib)[::-1][:5]
            for ci in top_contrib:
                src = int(idx[ci])
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Score all positions to find best destination
                scores = np.zeros(n)
                for k2 in targets:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                        scores[k2i - idx2] += a[idx2]
                scores[src] = 1e10

                dsts = np.argsort(scores)[:3]
                for dst in dsts:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in fracs:
                        amt = src_val * frac
                        if amt < 1e-15:
                            break
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            break
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < best_m - 1e-15:
                            a = a_t
                            best_m = Mt
                            best_a = a.copy()
                            improved = True
                            break

        if not improved:
            # Random perturbation
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.001, 0.008)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st < 1e-15:
                    a = best_a.copy()
                    break
                a = a / st
                Mn = fast_autocorr(a).max()
                if Mn < best_m:
                    best_m = Mn
                    best_a = a.copy()
                    improved = True
                    break

    elapsed = time.monotonic() - t0
    if verbose:
        print("    PeakShave done (" + str(round(elapsed, 1)) + "s) J=" + repr(2 * n * best_m))
    return best_a, 2.0 * n * best_m

def coord_polish(y0, time_limit, seed=77, verbose=True):
    n = len(y0)
    a = normalize(y0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fracs = [0.003, 0.006, 0.01, 0.02, 0.04, 0.08]

    stall = 0
    max_stall = 1200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(15, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            scores = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2+1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)
            dst = int(candidates[np.argmin(scores[candidates])])
            if dst == src:
                continue

            for frac in fracs:
                amt = src_val * frac
                if amt < 1e-15:
                    break
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    break
                a_t = a_t / st
                Mt = fast_autocorr(a_t).max()
                if Mt < best_m - 1e-15:
                    a = a_t
                    best_m = Mt
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            if stall > 100:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = max(0, stall - 50)

        if verbose and stall % 500 == 0 and stall > 0:
            print("    Coord (" + str(round(time.monotonic()-t0,1)) + "s) J=" + repr(2*n*best_m) + " stall=" + str(stall))

    return best_a, 2.0 * n * best_m

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Total budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            print("Parent: n=" + str(len(p)) + " score=" + repr(best_score))
    except Exception:
        best_seq = None

    if best_seq is None:
        n_init = 8000
        x = np.linspace(0, 1, n_init)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)
        print("Default init n=" + str(n_init) + " score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: LP-FW on parent length
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P1: LP-FW (n=" + str(n_cur) + ") ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 60:
        a_out, v_out = lp_frank_wolfe(normalize(best_seq), t1, solver,
                                       init_K=800, max_K=5000, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P1 improved: " + repr(best_score))

    # Phase 2: Continue LP-FW with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P2: LP-FW cont (n=" + str(n_cur) + ") ===")
    t2 = min(remaining * 0.22, 360.0)
    if t2 > 40:
        # Perturb slightly for diversity
        rng = np.random.default_rng(99)
        perturbation = 0.001 * rng.standard_normal(n_cur) * best_seq
        y2 = best_seq + perturbation
        y2 = np.maximum(y2, 0.0)
        y2 = y2 / y2.sum()
        a_out, v_out = lp_frank_wolfe(y2, t2, solver,
                                       init_K=1000, max_K=6000, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P2 improved: " + repr(best_score))

    # Phase 3: Multiplicative gradient descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P3: MultGradDes (n=" + str(n_cur) + ") ===")
    t3 = min(remaining * 0.15, 240.0)
    if t3 > 30:
        a_out, v_out = mult_grad_descent(normalize(best_seq), t3,
                                          top_k=120, temp=50.0, c0=3.0, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P3 improved: " + repr(best_score))

    # Phase 4: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P4: PeakShave (n=" + str(n_cur) + ") ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 20:
        a_out, v_out = peak_shaving(normalize(best_seq), t4, n_targets=10, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P4 improved: " + repr(best_score))

    # Phase 5: Try slightly longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 1000
    if remaining > 200:
        print("\n=== P5: Interpolate to n=" + str(n_long) + " ===")
        t5 = min(remaining * 0.20, 300.0)
        if t5 > 30:
            a_long = interpolate_seq(best_seq, n_long)
            a_out, v_out = mult_grad_descent(a_long, t5 * 0.5,
                                              top_k=120, temp=60.0, c0=2.5, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  P5 improved (longer): " + repr(best_score))

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 60:
                a_out, v_out = lp_frank_wolfe(normalize(best_seq), min(remaining * 0.5, 150.0),
                                               solver, init_K=800, max_K=4000, verbose=True)
                sc = eval_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out
                    print("  P5 LP improved: " + repr(best_score))

    # Phase 6: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P6: CoordPolish (n=" + str(n_cur) + ") ===")
    t6 = min(remaining * 0.25, 300.0)
    if t6 > 20:
        a_out, v_out = coord_polish(normalize(best_seq), t6, seed=42, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P6 improved: " + repr(best_score))

    # Phase 7: Final LP-FW if short enough
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P7: Final optimize (n=" + str(n_cur) + ") ===")
    t7 = min(remaining * 0.35, 350.0)
    if t7 > 30:
        if n_cur <= 15000:
            a_out, v_out = lp_frank_wolfe(normalize(best_seq), t7 * 0.6, solver,
                                           init_K=1200, max_K=7000, verbose=True)
        else:
            a_out, v_out = mult_grad_descent(normalize(best_seq), t7 * 0.6,
                                              top_k=150, temp=80.0, c0=2.0, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P7 improved: " + repr(best_score))

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            a_out, v_out = peak_shaving(normalize(best_seq), min(remaining * 0.7, 150.0),
                                         n_targets=12, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  P7b improved: " + repr(best_score))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + " n=" + str(len(best_seq)) + " t=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR fallback")
        n_fb = 8000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)

    return [float(x) for x in best_seq]
