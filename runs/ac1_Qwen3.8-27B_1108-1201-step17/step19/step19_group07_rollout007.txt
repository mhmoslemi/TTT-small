def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:

def compute_gradient_active(a, w, fft_size_cache=None):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    a_rev = a[::-1]
    W = np.fft.rfft(w, fft_size)
    A_rev = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(W * A_rev, fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad


def softmax_weights(c, beta, active_mask=None):
    if active_mask is not None:
        c_work = np.where(active_mask, c, -np.inf)
    else:
        c_work = c.copy()
    c_max = np.max(c_work[np.isfinite(c_work)])
    exp_c = np.zeros_like(c_work)
    valid = np.isfinite(c_work)
    exp_c[valid] = np.exp(beta * (c_work[valid] - c_max))
    s = exp_c.sum()
    if s < 1e-300:
        exp_c = np.zeros_like(c_work)
        top_idx = np.argmax(c_work[valid])
        indices = np.where(valid)[0]
        exp_c[indices[top_idx]] = 1.0
    else:
        exp_c /= s
    return exp_c


def fista_optimize(a0, time_limit, beta_start=3.0, beta_end=80.0,
                   eta0=0.8, eps_frac=0.05, verbose=True):
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()
    best_x = x.copy()
    best_c = fast_autocorr(x)
    best_m = best_c.max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_threshold = 5000

    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / max(time_limit, 1.0), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        c_y = fast_autocorr(y)
        c_max = c_y.max()
        threshold = c_max - eps_frac * c_max
        active = c_y > threshold
        if not active.any():
            active = np.zeros_like(c_y, dtype=bool)
            active[np.argmax(c_y)] = True

        w = softmax_weights(c_y, beta, active)
        grad = compute_gradient_active(y, w)

        eta = eta0 / (np.sqrt(t) + 1.0)
        if iter_count % 4000 == 3000:
            eta *= 2.5

        log_y = np.log(np.maximum(y, 1e-30)) - eta * grad
        log_y -= log_y.max()
        z = np.exp(log_y)
        z = np.maximum(z, 1e-30)
        z = z / z.sum()

        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = z + momentum * (z - y)
            x_new = np.maximum(x_new, 1e-30)
            x_new = x_new / x_new.sum()
        else:
            x_new = z.copy()

        c_new = fast_autocorr(x_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 2000:
                x_new = y.copy()

        if stall > stall_threshold:
            rng = np.random.default_rng(iter_count * 7 + 13)
            pert = rng.normal(0, 0.03, n) * best_x
            x_restart = norm_sum1(best_x + pert)
            x_new = x_restart
            y = x_restart.copy()
            stall = 0
            stall_threshold = max(stall_threshold * 0.7, 1500)

        y = x_new
        x = x_new
        iter_count += 1

        if verbose and iter_count % 10000 == 0:
            J = 2.0 * n * best_m
            print(f"    FISTA iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, stall={stall}")

    return best_x, best_m


def adam_optimize(a0, time_limit, beta_start=5.0, beta_end=100.0,
                  lr0=0.1, eps_frac=0.1, verbose=True):
    n = len(a0)
    x = norm_sum1(a0)
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    m_acc = np.zeros(n)
    v_acc = np.zeros(n)
    t0 = time.monotonic()
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / max(time_limit, 1.0), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        c_y = fast_autocorr(x)
        c_max = c_y.max()
        threshold = c_max - eps_frac * c_max
        active = c_y > threshold
        if not active.any():
            active = np.zeros_like(c_y, dtype=bool)
            active[np.argmax(c_y)] = True

        w = softmax_weights(c_y, beta, active)
        grad = compute_gradient_active(x, w)
        grad -= grad.mean()

        m_acc = 0.9 * m_acc + 0.1 * grad
        v_acc = 0.99 * v_acc + 0.01 * grad ** 2
        lr = lr0 / (np.sqrt(t) + 1.0)
        x_raw = x - lr * m_acc / (np.sqrt(v_acc) + 1e-8)
        x_raw = np.maximum(x_raw, 1e-15)
        x = x_raw / x_raw.sum()

        c_new = fast_autocorr(x)
        m_new = c_new.max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_x = x.copy()

        iter_count += 1
        if verbose and iter_count % 5000 == 0:
            J = 2.0 * n * best_m
            print(f"    Adam iter={iter_count} ({elapsed:.0f}s): J={J:.6f}")

    return best_x, best_m


def lp_frank_wolfe_improved(a0, time_limit, solver, initial_K=300, max_K=3000, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()

    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()

        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        conv_ys = np.fft.irfft(Yf * np.fft.rfft(g_dir, fft_size), fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                constraint_set.add(int(k))

        # Line search for step size
        best_alpha = 2.0 / (t + 2.0)
        best_obj = M_curr
        for alpha_try in [0.1, 0.2, best_alpha, 0.4, 0.6, 0.8, 1.0]:
            trial = (1.0 - alpha_try) * y + alpha_try * g_dir
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            ct = fast_autocorr(trial)
            mt = ct.max()
            if mt < best_obj:
                best_obj = mt
                best_alpha = alpha_try

        alpha = best_alpha
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    return best_a, best_m


def parametric_init(n, best_ref=None):
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')

    candidates = []

    for p in [0.3, 0.5, 0.7, 0.8, 1.0, 1.2, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr + 1.0) ** p)

    for alpha in [0.5, 0.8, 1.0, 1.2]:
        for beta_p in [0.5, 0.8, 1.0, 1.2]:
            candidates.append(np.maximum(i_arr + 1.0, 0.5) ** alpha *
                              np.maximum(n - i_arr, 0.5) ** beta_p)

    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))

    for rate in [1, 2, 3, 5]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1.0)))

    for freq in [1, 2, 3, 5]:
        candidates.append(np.maximum(np.sin(np.pi * freq * x) ** 2, 0.01))

    # LCD-like window
    i_range = np.arange(1, n + 1)
    candidates.append(1.0 / np.sin(np.pi * i_range / (n + 1)))
    candidates.append(1.0 / (1.0 + (i_arr - n / 2.0) ** 2 / (n * n * 0.1)))

    if best_ref is not None and len(best_ref) == n:
        ref_norm = norm_sum1(best_ref)
        for w in [0.2, 0.4, 0.6, 0.8]:
            candidates.append(w * ref_norm + (1.0 - w) * norm_sum1((i_arr + 1.0) ** 0.9))

    for a_cand in candidates:
        a_norm = norm_sum1(a_cand)
        ac = fast_autocorr(a_norm)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_a = a_norm

    return best_a, best_val


def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(30, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def sparse_refine(a0, time_limit, solver, verbose=True):
    """Identify near-zero coordinates, fix them, re-optimize the rest."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Identify coordinates below threshold
    thresh = np.percentile(a, 5)
    active_idx = np.where(a > thresh)[0]
    if len(active_idx) < max(100, n // 3):
        return best_a, best_m

    print(f"    Sparse refine: {len(active_idx)}/{n} active coords")

    # Optimize on active subset using LP-FW
    a_sub = a[active_idx]
    a_sub = norm_sum1(a_sub)

    def eval_sub(afull):
        ac = fast_autocorr(afull)
        return ac.max()

    # Simple: just run a short LP-FW on the full vector but with the constraint that
    # inactive coords stay at 0
    t_rem = time_limit - (time.monotonic() - t0)
    if t_rem < 20:
        return best_a, best_m

    a_out, m_out = lp_frank_wolfe_improved(a, t_rem, solver, initial_K=200, max_K=1500, verbose=False)
    # Zero out the inactive coordinates
    a_out[~np.isin(np.arange(n), active_idx)] = 0.0
    a_out = norm_sum1(a_out)
    m_final = fast_autocorr(a_out).max()

    if m_final < best_m:
        best_a = a_out
        best_m = m_final

    return best_a, best_m

    for rn in [4000, 4500, 5000, 5500, 6000, 6500, 7000]:
        if rn != n_best and 3000 <= rn <= 8000:
            length_candidates.append(rn)
    length_candidates = sorted(set(length_candidates))

    for new_n in length_candidates[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.4, 200.0)
        if t7a > 20:
            a_out, _ = fista_optimize(a_long, t7a, beta_start=5.0, beta_end=80.0,
                                      eta0=0.8, eps_frac=0.05, verbose=False)
            check_and_update(a_out, f"fista_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 7000:
            t7b = min(remaining * 0.4, 200.0)
            if t7b > 30:
                a_out, _ = lp_frank_wolfe_improved(a_long, t7b, solver,
                                                    initial_K=200, max_K=1500, verbose=False)
                check_and_update(a_out, f"lp_n{new_n}")

    # === Phase 8: Final LP-FW polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP-FW (n={n_best}) ===")
    if remaining > 40 and n_best <= 8000:
        t8 = min(remaining * 0.40, 200.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_improved(a_in, t8, solver, initial_K=400, max_K=3000, verbose=True)
            check_and_update(a_out, "lp_final")

    # === Phase 9: Final FISTA squeeze ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t9 = min(remaining * 0.5, 80.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = fista_optimize(a_in, t9, beta_start=50.0, beta_end=300.0,
                                      eta0=0.3, eps_frac=0.02, verbose=False)
            check_and_update(a_out, "fista_squeeze")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
