    size = 2 * n - 1
    y = a.copy()
    x = a.copy()

    # Build constraint matrix
    A = build_conv_matrix(y, n)
    row_idx = A.indices
    col_idx = A.indptr
    data = A.data

    iter_count = 0
    last_improvement = 0

    while time.monotonic() - t0 < time_limit:
        t_iter = iter_count + 1

        # Build and solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()

        # Use sparse matrix directly
        A_dense = None
        try:
            constraints = [cp.sum(g_var) == 1, A @ g_var <= M_var]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            prob.solve(solver=solver, verbose=False)
        except Exception as e:
            if iter_count < 3:
                # Try with dense matrix as fallback
                A_dense = A.toarray()
                constraints = [cp.sum(g_var) == 1, A_dense @ g_var <= M_var]
                prob = cp.Problem(cp.Minimize(M_var), constraints)
                prob.solve(solver=solver, verbose=False)
            else:
                break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            break
        g_dir = g_val / g_sum

        # Check improvement
        ac_g = fast_autocorr(g_dir)
        if ac_g.max() < best_m - 1e-15:
            best_m = ac_g.max()
            best_a = g_dir.copy()
            last_improvement = t_iter

        # Nesterov acceleration
        alpha = 2.0 / (t_iter + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        y_new = y_new / sy if sy > 1e-15 else y.copy()

        if t_iter == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t_iter - 1.0) / (t_iter + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            x_new = x_new / sx if sx > 1e-15 else y_new.copy()

        # Check Nesterov point
        ac_y = fast_autocorr(y_new)
        if ac_y.max() < best_m - 1e-15:
            best_m = ac_y.max()
            best_a = y_new.copy()
            last_improvement = t_iter

        ac_x = fast_autocorr(x_new)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x_new.copy()
            last_improvement = t_iter

        y = y_new
        x = x_new

        # Rebuild constraint matrix with updated y
        A = build_conv_matrix(y, n)

        iter_count += 1
        elapsed = time.monotonic() - t0

        if verbose and (t_iter % 5 == 0 or t_iter <= 5):
            J = 2.0 * n * best_m
            print(f"    LP-FW[full] iter={t_iter} ({elapsed:.0f}s): J={J:.6f}, M_lp={M_var.value:.6f}")

        # Convergence check
        if t_iter > 10 and (t_iter - last_improvement) > 50:
            print(f"    LP-FW[full]: No improvement for 50 iters, stopping.")
            break

    return best_a, best_m


def lp_fw_active(a0, time_limit, solver, initial_K=500, max_K=6000, verbose=True):
    """LP-Frank-Wolfe with active set management."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    size = 2 * n - 1
    y = a.copy()
    x = a.copy()

    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()

    init_K = min(initial_K, size)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        t_iter = iter_count + 1

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)

        # Build sparse constraint matrix
        rows = []
        cols = []
        vals = []
        for jj, k in enumerate(constraint_list):
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for j in range(i_start, i_end):
                rows.append(jj)
                cols.append(j)
                vals.append(y[k - j])

        if len(rows) == 0:
            break

        A_c = sp.csr_matrix((vals, (rows, cols)), shape=(K, n))

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A_c @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violated constraints
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                constraint_set.add(int(k))

        # Check improvement
        ac_g = fast_autocorr(g_dir)
        if ac_g.max() < best_m - 1e-15:
            best_m = ac_g.max()
            best_a = g_dir.copy()

        # Nesterov
        alpha = 2.0 / (t_iter + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        y_new = y_new / sy if sy > 1e-15 else y.copy()

        if t_iter == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t_iter - 1.0) / (t_iter + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            x_new = x_new / sx if sx > 1e-15 else y_new.copy()

        ac_y = fast_autocorr(y_new)
        if ac_y.max() < best_m - 1e-15:
            best_m = ac_y.max()
            best_a = y_new.copy()

        ac_x = fast_autocorr(x_new)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x_new.copy()

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t_iter % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW[active] iter={t_iter} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    return best_a, best_m


def parametric_init(n, best_ref=None):
    """Try diverse parametric families and return the best."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')

    candidates = []

    # Power laws
    for p in [0.3, 0.5, 0.7, 0.8, 1.0, 1.3, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr + 1.0) ** p)

    # Beta-like (symmetric and asymmetric)
    for alpha in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0]:
        for beta_p in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0]:
            candidates.append(np.maximum(i_arr + 1.0, 0.5) ** alpha *
                              np.maximum(n - i_arr, 0.5) ** beta_p)

    # Gaussian
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))

    # Exponential
    for rate in [1, 2, 3, 5, 8]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1.0)))

    # Sine-based
    for freq in [1, 2, 3, 4]:
        candidates.append(np.maximum(np.sin(np.pi * freq * x) ** 2, 0.01))
        candidates.append(np.maximum(np.sin(np.pi * freq * x) ** 4, 0.01))

    # Hyperbolic secant
    for c in [0.3, 0.5, 0.7]:
        for width in [0.3, 0.5, 1.0]:
            candidates.append(1.0 / np.cosh((x - c) / width))

    if best_ref is not None and len(best_ref) == n:
        ref_norm = norm_sum1(best_ref)
        for w in [0.3, 0.5, 0.7]:
            for p in [0.5, 0.8, 1.0]:
                candidates.append(w * ref_norm + (1.0 - w) * norm_sum1((i_arr + 1.0) ** p))

    for a_cand in candidates:
        a_cand = np.maximum(np.asarray(a_cand, dtype=np.float64), 0.0)
        s = a_cand.sum()
        if s < 1e-15:
            continue
        a_norm = a_cand / s
        ac = fast_autocorr(a_norm)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_a = a_norm

    return best_a, best_val


def get_solver():
    """Find the best available LP solver."""
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                # Test with a slightly larger problem for timing
                x2 = cp.Variable(500, nonneg=True)
                A_test = np.random.rand(1000, 500)
                prob2 = cp.Problem(cp.Minimize(cp.sum(x2)), [cp.sum(x2) == 1, A_test @ x2 <= 1])
                t_start = time.monotonic()
                prob2.solve(solver=s, verbose=False)
                t_elapsed = time.monotonic() - t_start
                if prob2.status in ('optimal', 'optimal_inaccurate') and t_elapsed < 5:
                    print(f"  LP solver: {s} (test time: {t_elapsed:.2f}s)")
                    return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search function: finds the best sequence of nonnegative heights."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Determine starting length
    if parent is not None:
        n_start = len(parent)
    else:
        n_start = 5000
        best_seq = norm_sum1((np.arange(n_start, dtype=np.float64) + 1.0) ** 0.7)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_start}, score={best_score:.6f}")

    n_best = len(best_seq)

    # === Phase 1: Quick parametric search ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 30.0)
    print(f"\n=== Phase 1: Parametric init (n={n_best}, t={t1:.0f}s) ===")
    if t1 > 5:
        a_param, val_param = parametric_init(n_best, best_seq)
        check_and_update(a_param, "parametric")

    # === Phase 2: FISTA main optimization (full softmax) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.30, 400.0)
    print(f"\n=== Phase 2: FISTA full-softmax (n={n_best}, t={t2:.0f}s) ===")
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_optimize(a_in, t2, mode="full_softmax",
                                  beta_start=5.0, beta_end=200.0,
                                  eta0=1.0, verbose=True)
        check_and_update(a_out, "fista_full")

    # === Phase 3: FISTA with balanced gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: FISTA balanced (n={n_best}, t={t3:.0f}s) ===")
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_optimize(a_in, t3, mode="balanced",
                                  m_bal=500, eta0=0.8, verbose=True)
        check_and_update(a_out, "fista_balanced")

    # === Phase 4: LP-FW with ALL constraints (key improvement) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.30, 400.0)
    print(f"\n=== Phase 4: LP-FW full constraints (n={n_best}, t={t4:.0f}s) ===")
    if t4 > 30 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_fw_full(a_in, t4, solver, verbose=True)
        check_and_update(a_out, "lp_full")

    # === Phase 5: Active-set LP-FW (complementary) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: LP-FW active set (n={n_best}, t={t5:.0f}s) ===")
    if t5 > 20 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_fw_active(a_in, t5, solver, initial_K=500, max_K=5000, verbose=True)
        check_and_update(a_out, "lp_active")

    # === Phase 6: Final FISTA polish (high beta) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 6: Final FISTA polish (n={n_best}, t={t6:.0f}s) ===")
    if t6 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_optimize(a_in, t6, mode="full_softmax",
                                  beta_start=50.0, beta_end=500.0,
                                  eta0=0.3, verbose=False)
        check_and_update(a_out, "fista_polish")

    # === Phase 7: One more LP-FW pass if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP pass (n={n_best}, remaining={remaining:.0f}s) ===")
    if remaining > 60 and n_best <= 8000:
        t7 = min(remaining * 0.5, 200.0)
        if t7 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_fw_full(a_in, t7, solver, verbose=True)
            check_and_update(a_out, "lp_final2")

    # === Phase 8: Try a slightly different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 120:
        print(f"\n=== Phase 8: Length exploration ===")
        for delta_n in [200, -200, 400]:
            new_n = n_best + delta_n
            if new_n < 3000 or new_n > 8000:
                continue
            if remaining < 80:
                break
            print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")
            a_long = interpolate_sequence(best_seq, new_n)

            t_fista = min(remaining * 0.4, 100.0)
            if t_fista > 15:
                a_out, _ = fista_optimize(a_long, t_fista, mode="full_softmax",
                                          beta_start=10.0, beta_end=100.0,
                                          eta0=0.8, verbose=False)
                check_and_update(a_out, f"fista_n{new_n}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 60 and new_n <= 7000:
                t_lp = min(remaining * 0.4, 120.0)
                if t_lp > 30:
                    a_in = norm_sum1(best_seq)
                    a_out, _ = lp_fw_full(a_in, t_lp, solver, verbose=False)
                    check_and_update(a_out, f"lp_n{new_n}")

            remaining = time_budget - (time.monotonic() - t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
