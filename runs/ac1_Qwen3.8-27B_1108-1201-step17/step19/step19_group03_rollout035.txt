def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, P):
    """
    Compute gradient of log-sum-exp(tau * conv(a,a)) w.r.t. a.
    P is the softmax weights for each lag (length 2n-1).
    Returns gradient of length n.
    """
    n = len(a)
    fft_size = 1 << (4 * n - 2).bit_length()
    
    # Part 2: g2 = conv(a, P) [first n elements]
    A = np.fft.rfft(a, fft_size)
    Pft = np.fft.rfft(P, fft_size)
    g2_full = np.fft.irfft(A * Pft, fft_size)[:2*n-1]
    g2 = g2_full[:n]
    
    # Part 1: g1[j] = sum_i a[i] * P[j+i]
    # = conv(a_rev, P)[n-1 : 2n-1]
    a_rev = a[::-1]
    Arev = np.fft.rfft(a_rev, fft_size)
    g1_full = np.fft.irfft(Arev * Pft, fft_size)[:2*n-1]
    g1 = g1_full[n-1:2*n-1]
    
    return g1 + g2


def smooth_objective_and_grad(a, tau):
    """
    Compute smooth objective (log-sum-exp approximation) and its gradient.
    Returns (smooth_obj, gradient, P) where P are the softmax weights.
    """
    n = len(a)
    ac = fast_autocorr(a)
    ac_max = ac.max()
    # Shift to avoid overflow
    shifted = tau * (ac - ac_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    log_sum = np.log(exp_vals.sum()) + ac_max * tau
    P = exp_vals / exp_vals.sum()
    
    grad = smooth_gradient(a, P)
    return log_sum / tau, grad, P

            print(f"    SmoothGrad it={it} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, stall={stall}")

def fista_proper(a0, time_limit, tau_init=1.0, tau_final=200.0,
                 verbose=False, rng_seed=0):
    """
    Proper FISTA with smooth log-sum-exp approximation.
    """
    n = len(a0)
    x = norm_sum1(a0)
    z = x.copy()
    best_a = x.copy()
    best_M = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    tau = tau_init
    t_k = 1.0
    step = 0.5 / (tau * 2.0)  # initial step
    stall = 0
    max_stall = 800
    
    for it in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        tau = tau_init * (tau_final / tau_init) ** progress
        
        # Compute gradient at z
        ac_z = fast_autocorr(z)
        ac_max = ac_z.max()
        shifted = np.clip(tau * (ac_z - ac_max), -50, 50)
        exp_vals = np.exp(shifted)
        P = exp_vals / exp_vals.sum()
        grad = smooth_gradient(z, P)
        
        # FISTA step
        x_new = project_simplex(z - step * grad)
        
        # Update momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        z = np.maximum(z, 0.0)
        s = z.sum()
        if s > 1e-15:
            z = z / s
        t_k = t_new
        
        # Check improvement
        M_new = fast_autocorr(x_new).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x = x_new
        
        # Adaptive step size (backtracking)
        if it % 50 == 0:
            # Estimate Lipschitz constant
            ac_x = fast_autocorr(x)
            L_est = 2.0 * tau * ac_x.max() * 2.0 * n
            step = min(step * 1.02, 1.0 / max(L_est, 1e-10))
        
        if stall > max_stall:
            # Momentum restart
            z = x.copy()
            t_k = 1.0
            stall = 0
            if stall // max_stall > 5:
                pert = rng.normal(0, 0.05, n) * best_a
                a_p = np.maximum(best_a + pert, 1e-10)
                x = a_p / a_p.sum()
                z = x.copy()
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    FISTA it={it} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}")
    
    return best_a, best_M


def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=False, 
                       top_K_init=100, top_K_growth=20, top_K_max=3000):
    """
    LP-based Frank-Wolfe with better step size selection.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    K = top_K_init
    iter_count = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]
        
        # Build constraint matrix
        T = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T[ii, j] = a[ki - j]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + top_K_growth, top_K_max)
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + top_K_growth, top_K_max)
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        v = b_val / b_sum  # normalized direction (sum to 1)
        
        # Line search for best step size
        best_alpha = 2.0 / (iter_count + 2)
        best_M_trial = M_curr
        alphas = np.linspace(0.1, 1.0, 10)
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * v
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_M_trial:
                best_M_trial = M_trial
                best_alpha = alpha
        
        a = (1.0 - best_alpha) * a + best_alpha * v
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        M_new = fast_autocorr(a).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        iter_count += 1
        K = min(K + top_K_growth, top_K_max)
        
        if stall > 20:
            K = min(K + 50, top_K_max)
            stall = 0
        
        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    LP-FW it={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.3f}")
    
    return best_a, best_M


def pocs_optimize_v2(a0, time_limit, n_trials=5, rng_seed=0, verbose=False):
    """POCS with adaptive blending."""
    n = len(a0)
    best_a = a0.copy()
    best_M = fast_autocorr(a0).max()
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 2).bit_length()
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit - 5:
            break
        t_trial = (time_limit - (time.monotonic() - t0)) / (n_trials - trial)
        
        rng = np.random.default_rng(rng_seed + trial)
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        A_spec = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
        A_full = np.zeros(fft_size, dtype=np.complex128)
        A_full[:fft_size // 2 + 1] = A_spec
        for i in range(1, fft_size // 2):
            A_full[fft_size - i] = np.conj(A_spec[i])
        a = np.fft.irfft(A_full, fft_size)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s
        
        # POCS iterations
        n_iters = max(100, int(t_trial * 50))
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit - 2:
                break
            
            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            target = np.median(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            blend = 0.05 + 0.3 * min(1.0, it / 200)
            new_mag = (1 - blend) * mag + blend * target
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            
            M_new = fast_autocorr(a_new).max()
            if M_new < best_M:
                best_M = M_new
                best_a = a_new.copy()
            a = a_new
        
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    POCS trial {trial} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_M


def mixed_method(a0, time_limit, solver, verbose=False, rng_seed=0):
    """
    Mixed method: alternate between smooth gradient, LP-FW, and POCS.
    """
    n = len(a0)
    a = norm_sum1(a0)

    # Phase allocation: 40% smooth grad, 35% LP-FW, 25% POCS+polish
    t_smooth = time_limit * 0.40
    t_lp = time_limit * 0.35
    t_pocs = time_limit * 0.25
    
    # Phase 1: Smooth gradient
    elapsed = time.monotonic() - t0

    if t1 > 10:
        a_out, _ = fista_proper(a, t1, tau_init=5.0, tau_final=300.0,
                                 verbose=verbose, rng_seed=rng_seed)
        M_out = fast_autocorr(a_out).max()
        if M_out < best_M:
            best_M = M_out
            best_a = a_out.copy()
    
    # Phase 2: LP-FW
    elapsed = time.monotonic() - t0
    t2 = min(t_lp, time_limit - elapsed)
    if t2 > 10 and n <= 8000:
        a_out, _ = lp_frank_wolfe_v2(best_a, t2, solver, verbose=verbose,
                                       top_K_init=200, top_K_growth=30, top_K_max=4000)
        M_out = fast_autocorr(a_out).max()
        if M_out < best_M:
            best_M = M_out
            best_a = a_out.copy()
    
    # Phase 3: Another round of smooth gradient from LP result
    elapsed = time.monotonic() - t0
    t3 = min(t_pocs, time_limit - elapsed)
    if t3 > 10:
        a_out, _ = fista_proper(best_a, t3 * 0.6, tau_init=10.0, tau_final=500.0,
                                 verbose=verbose, rng_seed=rng_seed + 1)
        M_out = fast_autocorr(a_out).max()
        if M_out < best_M:
            best_M = M_out
            best_a = a_out.copy()

    return best_a, best_M


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_asymmetric(n, skew=0.3):
    """Asymmetric U-shape: more mass on one side."""
    x = np.linspace(0, 1, n)
    a = (1.0 - skew) * np.exp(-5.0 * (x - 0.1) ** 2 / 0.05) + \
        skew * np.exp(-5.0 * (x - 0.9) ** 2 / 0.05) + 0.01
    return norm_sum1(a)


def make_gaussian(n, center=0.5, width=0.2):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2) + 0.001
    return norm_sum1(a)


def make_chebyshev(n, order=5):
    """Chebyshev polynomial based shape."""
    x = np.linspace(-1, 1, n)
    a = np.abs(np.cos(order * np.arccos(np.clip(x, -1, 1)))) + 0.01
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, P):
    n = len(a)
    fft_size = 1 << (4 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    Pft = np.fft.rfft(P, fft_size)
    g2_full = np.fft.irfft(A * Pft, fft_size)[:2 * n - 1]
    g2 = g2_full[:n]
    a_rev = a[::-1]
    Arev = np.fft.rfft(a_rev, fft_size)
    g1_full = np.fft.irfft(Arev * Pft, fft_size)[:2 * n - 1]
    g1 = g1_full[n - 1:2 * n - 1]
    return g1 + g2


def compute_softmax_weights(ac, tau):
    ac_max = ac.max()
    shifted = np.clip(tau * (ac - ac_max), -50, 50)
    exp_vals = np.exp(shifted)
    return exp_vals / exp_vals.sum()


def fista_smooth(a0, time_limit, tau_init=5.0, tau_final=500.0,
                 rng_seed=0, verbose=False):
    n = len(a0)
    x = norm_sum1(a0)
    z = x.copy()
    best_a = x.copy()
    best_M = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    t_k = 1.0
    step = 0.3 / (tau_init * 2.0)
    stall = 0
    max_stall = 600
    use_accel = True

    for it in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        tau = tau_init * (tau_final / tau_init) ** progress

        ac_z = fast_autocorr(z)
        P = compute_softmax_weights(ac_z, tau)
        grad = smooth_gradient(z, P)

        # Backtracking line search
        step_trial = step
        x_trial = project_simplex(z - step_trial * grad)
        M_trial = fast_autocorr(x_trial).max()
        back_count = 0
        while M_trial > best_M + 0.01 and back_count < 10:
            step_trial *= 0.5
            x_trial = project_simplex(z - step_trial * grad)
            M_trial = fast_autocorr(x_trial).max()
            back_count += 1

        step = max(step * 1.05, step_trial * 1.5)

        if use_accel:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            z_new = x_trial + ((t_k - 1.0) / t_new) * (x_trial - x)
            z_new = np.maximum(z_new, 0.0)
            s = z_new.sum()
            if s > 1e-15:
                z_new = z_new / s
            t_k = t_new
        else:
            z_new = x_trial.copy()
            t_k = 1.0

        M_new = fast_autocorr(x_trial).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = x_trial.copy()
            stall = 0
        else:
            stall += 1

        x = x_trial
        z = z_new

        if stall > max_stall:
            use_accel = not use_accel
            z = x.copy()
            t_k = 1.0
            stall = 0
            if stall // max_stall > 3 or (it > 2000 and M_new > 1.001 * best_M):
                pert = rng.normal(0, 0.03, n) * best_a
                a_p = np.maximum(best_a + pert, 1e-10)
                x = a_p / a_p.sum()
                z = x.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    FISTA it={it} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, stall={stall}")

    return best_a, best_M


def lp_frank_wolfe(a0, time_limit, solver, verbose=False,
                   top_K_init=200, top_K_growth=25, top_K_max=5000):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    K = top_K_init
    iter_count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]

        T = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + top_K_growth * 2, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + top_K_growth * 2, top_K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        v = b_val / b_sum

        best_alpha = 2.0 / (iter_count + 2)
        best_M_trial = M_curr
        alphas = np.array([best_alpha, 0.5, 0.3, 0.2, 0.7, 0.1, 1.0])
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * v
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_M_trial:
                best_M_trial = M_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * v
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        M_new = fast_autocorr(a).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        iter_count += 1
        K = min(K + top_K_growth, top_K_max)

        if stall > 15:
            K = min(K + 100, top_K_max)
            stall = 0

        if verbose and iter_count % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    LP-FW it={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_M


def pocs_flat_spectrum(a0, time_limit, n_trials=6, rng_seed=0, verbose=False):
    n = len(a0)
    best_a = a0.copy()
    best_M = fast_autocorr(a0).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit - 5:
            break
        t_trial = (time_limit - (time.monotonic() - t0)) / (n_trials - trial)

        rng = np.random.default_rng(rng_seed + trial)
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        A_spec = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
        A_full = np.zeros(fft_size, dtype=np.complex128)
        A_full[:fft_size // 2 + 1] = A_spec
        for i in range(1, fft_size // 2):
            A_full[fft_size - i] = np.conj(A_spec[i])
        a = np.fft.irfft(A_full, fft_size)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s

        n_iters = max(200, int(t_trial * 80))
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit - 2:
                break

            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            nonzero = mag > 1e-10
            if not np.any(nonzero):
                break
            target = np.median(mag[nonzero])
            blend = 0.03 + 0.2 * min(1.0, it / 500)
            new_mag = (1 - blend) * mag + blend * target
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s

            M_new = fast_autocorr(a_new).max()
            if M_new < best_M:
                best_M = M_new
                best_a = a_new.copy()
            a = a_new

        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    POCS trial {trial} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_M


def coordinate_polish(a0, time_limit, rng_seed=0, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:15]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            src_indices = range(max(0, ki - n + 1), min(n, ki + 1))
            max_contrib = 0
            best_src = -1
            for j in src_indices:
                contrib = a[j] * a[ki - j]
                if contrib > max_contrib and a[j] > 1e-12:
                    max_contrib = contrib
                    best_src = j

            if best_src < 0 or max_contrib < 1e-15:
                continue

            in_peak = set()
            for kk in top_k[:8]:
                ki2 = int(kk)
                for j in range(max(0, ki2 - n + 1), min(n, ki2 + 1)):
                    in_peak.add(j)

            candidates = [j for j in range(n) if j not in in_peak and j != best_src]
            if not candidates:
                candidates = [j for j in range(n) if j != best_src]

            sens = np.zeros(n)
            for kk in top_k[:10]:
                ki2 = int(kk)
                for j in range(max(0, ki2 - n + 1), min(n, ki2 + 1)):
                    sens[j] += a[ki2 - j]

            candidates.sort(key=lambda j: sens[j])
            top_cands = candidates[:5]

            found = False
            for dst in top_cands:
                if found:
                    break
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.1]:
                    amt = a[best_src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[best_src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M - 1e-16:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

        if not improved:
            stall += 1
            if stall % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_M

    return norm_sum1(a)


def make_power_law(n, exponent=1.5):
    x = np.linspace(0.01, 1.0, n)
    a = x ** (-exponent) * np.sin(np.pi * x) ** 2 + 0.001
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def mixed_optimize(a0, time_limit, solver, rng_seed=0, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Phase 1: FISTA smooth (45%)
    t1 = min(time_limit * 0.45, time_limit - 30)
    if t1 > 15:
        a_out, _ = fista_smooth(a, t1, tau_init=5.0, tau_final=300.0,
                                 rng_seed=rng_seed, verbose=verbose)
        M_out = fast_autocorr(a_out).max()
        if M_out < best_M:
            best_M = M_out
            best_a = a_out.copy()

    # Phase 2: LP-FW (30%)
    elapsed = time.monotonic() - t0
    t2 = min(time_limit * 0.30, time_limit - elapsed - 20)
    if t2 > 15 and n <= 8000:
        a_out, _ = lp_frank_wolfe(best_a, t2, solver, verbose=verbose,
                                    top_K_init=300, top_K_growth=30, top_K_max=4000)
        M_out = fast_autocorr(a_out).max()
        if M_out < best_M:
            best_M = M_out
            best_a = a_out.copy()

    # Phase 3: FISTA again (15%)
    elapsed = time.monotonic() - t0
    t3 = min(time_limit * 0.15, time_limit - elapsed - 10)
    if t3 > 10:
        a_out, _ = fista_smooth(best_a, t3, tau_init=20.0, tau_final=800.0,
                                 rng_seed=rng_seed + 1, verbose=verbose)
        M_out = fast_autocorr(a_out).max()
        if M_out < best_M:
            best_M = M_out
            best_a = a_out.copy()

    # Phase 4: Coordinate polish (10%)
    elapsed = time.monotonic() - t0
    t4 = max(0, time_limit - elapsed)
    if t4 > 10:
        a_out, _ = coordinate_polish(best_a, t4, rng_seed=rng_seed + 2, verbose=verbose)
        M_out = fast_autocorr(a_out).max()
        if M_out < best_M:
            best_M = M_out
            best_a = a_out.copy()

    return best_a, best_M


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            n0 = len(parent)
            print(f"Parent: n={n0}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        n0 = 5500
        print("No parent found, using default")

    if best_seq is None:
        n0 = 5500
        best_seq = make_u_shape(n0, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n0}, score={best_score:.6f}")

    n0 = len(best_seq)

    # Quick length sweep
    print(f"\n=== Quick length sweep (base n={n0}) ===")
    t_sweep_end = t_global + min(45, time_budget * 0.03)
    candidate_ns = []
    for delta in range(-2000, 2001, 250):
        new_n = n0 + delta
        if 3500 <= new_n <= 7500:
            candidate_ns.append(new_n)
    if n0 not in candidate_ns:
        candidate_ns.insert(0, n0)
    candidate_ns = list(dict.fromkeys(candidate_ns))

    quick_scores = {}
    for new_n in candidate_ns[:10]:
        if time.monotonic() > t_sweep_end:
            break
        a_test = interpolate_sequence(best_seq, new_n)
        M_test = fast_autocorr(a_test).max()
        quick_scores[new_n] = 2.0 * new_n * M_test

    sorted_ns = sorted(quick_scores.keys(), key=lambda x: quick_scores[x])
    top_ns = sorted_ns[:5]
    print(f"  Top: {[(n, f'{quick_scores[n]:.4f}') for n in top_ns]}")

    # Main optimization on best length
    n_best = top_ns[0]
    print(f"\n=== Main optimization (n={n_best}) ===")

    inits = []
    labels = []

    inits.append(interpolate_sequence(best_seq, n_best))
    labels.append("parent")
    inits.append(make_u_shape(n_best, 1.5))
    labels.append("u1.5")
    inits.append(make_u_shape(n_best, 3.0))
    labels.append("u3.0")
    inits.append(make_bimodal(n_best, 0.1))
    labels.append("bimodal")
    inits.append(make_cosine(n_best, 0.3))
    labels.append("cos03")
    inits.append(make_smooth_random(n_best, seed=42))
    labels.append("rand")
    inits.append(make_power_law(n_best, 1.0))
    labels.append("power1")
    inits.append(make_power_law(n_best, 2.0))
    labels.append("power2")

    # Deduplicate
    unique_inits = []
    unique_labels = []
    seen = set()
    for init, lbl in zip(inits, labels):
        sig = tuple(np.round(init[::max(1, n_best // 20)], 3))
        if sig not in seen:
            seen.add(sig)
            unique_inits.append(init)
            unique_labels.append(lbl)
    inits = unique_inits
    labels = unique_labels

    t_phase1_end = t_global + min(time_budget * 0.50, 800)
    t_per = max(30.0, (t_phase1_end - time.monotonic()) / len(inits))

    for idx, (init, lbl) in enumerate(zip(inits, labels)):
        elapsed = time.monotonic() - t_global
        if elapsed > t_phase1_end - 60:
            break
        t_avail = min(t_per, t_phase1_end - elapsed)
        if t_avail < 25:
            break
        print(f"  [{lbl}] t={t_avail:.0f}s...")
        a_out, _ = mixed_optimize(init, t_avail, solver, rng_seed=100 + idx * 7, verbose=False)
        a_out = np.maximum(a_out, 0.0)
        s = a_out.sum()
        if s > 0.01:
            a_norm = a_out / s
            sc = evaluate_safe(a_norm)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_norm.copy()
                print(f"  *** NEW BEST [{lbl}]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # POCS
    remaining = t_global + time_budget - time.monotonic()
    if remaining > 80:
        print(f"\n=== POCS ({len(best_seq)} elements) ===")
        t_pocs = min(remaining * 0.25, 250)
        a_out, _ = pocs_flat_spectrum(best_seq, t_pocs, n_trials=6, rng_seed=300, verbose=False)
        a_out = np.maximum(a_out, 0.0)
        s = a_out.sum()
        if s > 0.01:
            a_norm = a_out / s
            sc = evaluate_safe(a_norm)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_norm.copy()
                print(f"  *** NEW BEST [pocs]: {sc:.6f}")

    # Second round mixed on best
    remaining = t_global + time_budget - time.monotonic()
    if remaining > 100:
        n_cur = len(best_seq)
        print(f"\n=== Round 2 mixed (n={n_cur}) ===")
        t_r2 = min(remaining * 0.45, 450)
        a_out, _ = mixed_optimize(best_seq, t_r2, solver, rng_seed=700, verbose=True)
        a_out = np.maximum(a_out, 0.0)
        s = a_out.sum()
        if s > 0.01:
            a_norm = a_out / s
            sc = evaluate_safe(a_norm)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_norm.copy()
                print(f"  *** NEW BEST [r2]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # Length exploration
    remaining = t_global + time_budget - time.monotonic()
    n_cur = len(best_seq)
    if remaining > 120:
        print(f"\n=== Length exploration ===")
        for delta in [300, -300, 600, -600, 900, -900, 1200, -1200]:
            new_n = n_cur + delta
            if not (3500 <= new_n <= 7500):
                continue
            remaining = t_global + time_budget - time.monotonic()
            if remaining < 80:
                break
            print(f"  n={new_n}...")
            a_try = interpolate_sequence(best_seq, new_n)
            t_lim = min(remaining * 0.35, 200)
            if t_lim > 30:
                a_out, _ = mixed_optimize(a_try, t_lim, solver, rng_seed=new_n % 1000, verbose=False)
                a_out = np.maximum(a_out, 0.0)
                s = a_out.sum()
                if s > 0.01:
                    a_norm = a_out / s
                    sc = evaluate_safe(a_norm)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = a_norm.copy()
                        print(f"  *** NEW BEST [n={new_n}]: {sc:.6f}")

    # Final polish
    remaining = t_global + time_budget - time.monotonic()
    n_final = len(best_seq)
    if remaining > 30:
        print(f"\n=== Final polish (n={n_final}) ===")
        t_final = min(remaining * 0.85, 300)
        a_out, _ = mixed_optimize(best_seq, t_final, solver, rng_seed=999, verbose=True)
        a_out = np.maximum(a_out, 0.0)
        s = a_out.sum()
        if s > 0.01:
            a_norm = a_out / s
            sc = evaluate_safe(a_norm)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_norm.copy()
                print(f"  *** NEW BEST [final]: {sc:.6f}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("ERROR: fallback")
        n_fb = 5500
        best_seq = make_u_shape(n_fb, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
