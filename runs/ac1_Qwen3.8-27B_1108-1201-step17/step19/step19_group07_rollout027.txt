def fast_autocorr(a):

def norm_sum1(a):

def compute_gradient_smoothed(a, beta, top_k=None):
    """Compute smoothed gradient of max_k c[k] using log-sum-exp."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    
    if top_k is not None and top_k < size:
        # Focus on top-k largest correlations
        thresholds = np.sort(c)[::-1]
        cutoff = thresholds[min(top_k, size) - 1]
        mask = c >= cutoff
        c_masked = np.where(mask, c, -np.inf)
    else:
        c_masked = c.copy()
    
    c_max = c_masked.max()
    exp_c = np.exp(beta * (c_masked - c_max))
    w = exp_c / exp_c.sum()
    
    # Gradient of sum_k w[k] * c[k] with respect to a
    # = sum_k w[k] * d(c[k])/da
    # d(c[k])/da[i] involves a convolved with w
    # Specifically, d(c[k])/da = 2 * (a * w_shifted) at appropriate positions
    
    # More precisely: c[k] = sum_i a[i]*a[k-i]
    # d(c[k])/da[j] = a[k-j] (if valid) + a[j]*δ (from the other term)
    # The gradient wrt a is: (w * a)[j] + (w * a_rev)[j] ... 
    # Actually, it's: grad[j] = 2 * sum_k w[k] * a[k-j] (with bounds)
    # Which is 2 * (w * a_rev)[j] where a_rev is reverse of a
    
    a_rev = a[::-1]
    W = np.fft.rfft(w, fft_size)
    A_rev = np.fft.rfft(a_rev, fft_size)
    conv_w_a = np.fft.irfft(W * A_rev, fft_size)
    grad = 2.0 * conv_w_a[n - 1:2 * n - 1]
    
    return grad, c, w

def mirror_descent(a0, time_limit, beta_start=5.0, beta_end=200.0,
                   eta0=1.0, use_backtrack=True, verbose=True):

    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()
    best_x = x.copy()
    best_c = fast_autocorr(x).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_threshold = 5000
    eta = eta0
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        elapsed = time.monotonic() - t0
        progress = min(iter_count / max(time_limit * 100, 1), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute gradient at y (Nesterov point)
        grad, c_y, _ = compute_gradient_smoothed(y, beta, top_k=min(100, size))
        
        # Mirror descent step
        log_y = np.log(np.maximum(y, 1e-30)) - eta * grad
        log_y -= log_y.max()
        z = np.exp(log_y)
        z = np.maximum(z, 1e-30)
        z = z / z.sum()
        
        # Nesterov momentum
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = z + momentum * (z - y)
            x_new = np.maximum(x_new, 1e-30)
            x_new = x_new / x_new.sum()
        else:
            x_new = z.copy()
        
        c_new = fast_autocorr(x_new).max()
        
        if use_backtrack and c_new > best_c:
            # Backtrack: try smaller step
            for shrunk in [0.5, 0.25, 0.1]:
                log_y_bt = np.log(np.maximum(y, 1e-30)) - eta * shrunk * grad
                log_y_bt -= log_y_bt.max()
                z_bt = np.exp(log_y_bt)
                z_bt = np.maximum(z_bt, 1e-30)
                z_bt = z_bt / z_bt.sum()
                if t > 1:
                    momentum = (t - 1.0) / (t + 2.0)
                    x_bt = z_bt + momentum * (z_bt - y)
                    x_bt = np.maximum(x_bt, 1e-30)
                    x_bt = x_bt / x_bt.sum()
                else:
                    x_bt = z_bt.copy()
                c_bt = fast_autocorr(x_bt).max()
                if c_bt < best_c:
                    x_new = x_bt
                    c_new = c_bt
                    break
            else:
                x_new = y.copy()  # No improvement, stay
                c_new = c_y.max()
            eta *= 0.7  # Reduce step size
        elif c_new < best_c - 1e-16:

        else:
            eta = eta * 0.9  # Slight reduction
        
        if c_new < best_c - 1e-16:

            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > stall_threshold:
            rng = np.random.default_rng(iter_count * 7 + 13)
            pert = rng.normal(0, 0.01, n) * best_x
            x_restart = norm_sum1(best_x + pert)
            x_new = x_restart
            y = x_restart.copy()
            stall = 0
            stall_threshold = max(stall_threshold * 0.8, 2000)

        x = x_new
        iter_count += 1
        
        if verbose and iter_count % 5000 == 0:

def lp_frank_wolfe(a0, time_limit, solver, initial_K=200, max_K=3000, verbose=True):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    init_K = min(initial_K, 2 * n - 1)
    ac_y = fast_autocorr(y)
    constraint_list = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in constraint_list)
    
    iter_count = 0
    failed_solves = 0
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
        
        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()
        
        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 10:
                constraint_list = sorted(constraint_set)[:init_K]
                constraint_set = set(constraint_list)
                failed_solves = 0
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            failed_solves += 1
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find newly violated constraints
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-6)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")
    
    return best_a, best_m

def parametric_init(n, best_ref=None):
    """Try diverse parametric families."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')
    
    candidates = []
    
    # Triangular (Fejér-like)
    candidates.append(np.arange(1, n + 1, dtype=np.float64) * np.arange(n, 0, -1, dtype=np.float64))
    # Raised cosine
    candidates.append(0.5 * (1 + np.cos(2 * np.pi * x)))
    candidates.append(0.5 * (1 - np.cos(2 * np.pi * x)))
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    # Symmetric power
    for p in [0.5, 1.0, 1.5, 2.0]:
        d = np.minimum(i_arr, n - 1 - i_arr) + 1.0
        candidates.append(d ** p)
    # Gaussian
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    # Sine-based
    for freq in [1, 2, 3]:
        candidates.append(np.abs(np.sin(np.pi * freq * x)) ** 2)
    
    for a_cand in candidates:
        a_norm = norm_sum1(a_cand)
        ac = fast_autocorr(a_norm)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_a = a_norm
    
    return best_a, best_val

def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(30, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def symmetrize(a):
    """Try to improve by symmetrizing."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = norm_sum1(a_sym)
    c_sym = fast_autocorr(a_sym).max()
    c_orig = fast_autocorr(a).max()
    if c_sym < c_orig:
        return a_sym
    return a
