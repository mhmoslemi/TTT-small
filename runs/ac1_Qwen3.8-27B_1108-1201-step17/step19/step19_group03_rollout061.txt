"""

import numpy as np
import time
import cvxpy as cp

_global_solver = None

def get_solver():

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, tau):
    """Compute gradient of logsumexp(conv(a,a)/tau) * tau w.r.t. a."""
    b = fast_autocorr(a)
    b_max = b.max()
    shifted = (b - b_max) / tau
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    # g[i] = 2 * sum_k p[k] * a[k-i] (where valid)
    # This is 2 * cross-correlation of p and a
    # np.correlate(p, a, mode='valid') gives sum_j p[i+j]*a[j] for i=0..n-1
    n = len(a)
    g = 2.0 * np.correlate(p, a, mode='valid')
    return g, b


def smooth_lse_optimize(a0, time_limit, tau_init=50.0, tau_final=2.0,
                        eta_init=0.3, verbose=False, rng_seed=0):
    """Optimize using smooth log-sum-exp with annealed tau and multiplicative updates."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    eta = eta_init
    stall = 0
    max_stall = 800
    it = 0
    total_time = max(time_limit, 1.0)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / (total_time * 0.8))
        # Anneal tau: start large (smooth), end small (sharp)
        tau = tau_init * (tau_final / tau_init) ** (progress ** 0.5)
        tau = max(tau, 0.5)
        
        g, b = lse_gradient(a, tau)
        M_curr = b.max()
        
        # Multiplicative update (mirror descent with simplex)
        # a_new[i] = a[i] * exp(-eta * g[i]) / sum(...)
        # Scale gradient to have appropriate magnitude
        g_scale = g / (max(np.abs(g).max(), 1e-10))
        log_a = np.log(np.maximum(a, 1e-15)) - eta * 0.5 * g_scale
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.05, 1.5)
        else:
            stall += 1
            if stall > 50:
                eta *= 0.85
            if stall > 200:
                # Perturb and restart
                pert = rng.normal(0, 0.01, n) * best_a
                a_p = np.maximum(best_a + pert, 1e-10)
                a = a_p / a_p.sum()
                eta = eta_init * 0.5
                stall = 100
            else:
                a = a_new
        
        it += 1
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LSE it={it} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.2f}, eta={eta:.4f}, stall={stall}")
    
    return best_a, best_m


def lp_frank_wolfe_v2(a0, time_limit, solver, K_start=None, K_max=None, verbose=False):
    """LP-based Frank-Wolfe with line search and adaptive K."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    total_lags = 2 * n - 1
    if K_start is None:
        K_start = min(500, total_lags)
    if K_max is None:
        K_max = total_lags
    
    K = K_start
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 5:
            break
            
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        K_sel = min(K, total_lags)
        if K_sel < total_lags:
            binding_idx = np.argsort(ac)[::-1][:K_sel]
        else:
            binding_idx = np.arange(total_lags)
        K_sel = len(binding_idx)
        
        # Build constraint matrix (sparse-friendly)
        T = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            for j in range(j_start, j_end + 1):
                T[ii, j] = a[ki - j]
        
        # Solve LP: max sum(g) s.t. T@g <= M_curr, g >= 0
        g_var = cp.Variable(n, nonneg=True)
        constraints = [T @ g_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
        
        t_lp_start = time.monotonic()
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 100, K_max)
            continue
        t_lp_end = time.monotonic()
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Line search
        ac_ag = np.convolve(a, g_dir)
        ac_gg = np.convolve(g_dir, g_dir)
        
        best_alpha = 2.0 / (it + 2.0)
        best_f = float('inf')
        
        # Coarse grid
        alphas = np.linspace(0.01, 0.4, 40)
        for alpha in alphas:
            f_ = ac * (1 - alpha) ** 2 + 2 * alpha * (1 - alpha) * ac_ag + alpha ** 2 * ac_gg
            f_max = f_.max()
            if f_max < best_f:
                best_f = f_max
                best_alpha = alpha
        
        # Fine grid around best
        a_lo = max(0.005, best_alpha - 0.02)
        a_hi = min(0.8, best_alpha + 0.02)
        alphas_fine = np.linspace(a_lo, a_hi, 30)
        for alpha in alphas_fine:
            f_ = ac * (1 - alpha) ** 2 + 2 * alpha * (1 - alpha) * ac_ag + alpha ** 2 * ac_gg
            f_max = f_.max()
            if f_max < best_f:
                best_f = f_max
                best_alpha = alpha
        
        # Update
        a_new = (1 - best_alpha) * a + best_alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = a_new.copy()
        
        a = a_new
        it += 1
        
        # Increase K gradually
        K = min(K + 30, K_max)
        
        if verbose and it % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={it} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}, lp_t={t_lp_end-t_lp_start:.1f}s")
    
    return best_a, best_m


def coord_descent_polish(a0, time_limit, verbose=False, rng_seed=0):
    """Pairwise coordinate descent: move mass from hot to cold positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    # Precompute sensitivity: for each position i, how much does it contribute to peaks?
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find top sensitive lags
        top_k = np.argsort(ac)[::-1][:15]
        
        # Compute per-position sensitivity
        sens = np.zeros(n)
        for k in top_k:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += ac[ki] * a[ki - idx]
        
        improved = False
        
        # Try moving mass from high-sensitivity to low-sensitivity positions
        high_sens = np.argsort(sens)[::-1][:5]
        low_sens = np.argsort(sens)[:5]
        
        for src in high_sens:
            for dst in low_sens:
                if src == dst or a[src] < 1e-10:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-12:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
        
        # Random perturbations when stuck
        if not improved:
            stall += 1
            if stall > 20:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    elif stall > 50:
                        a = best_a.copy()
        
        if improved:
            stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def pocs_spectral(a0, time_limit, n_trials=3, verbose=False, rng_seed=0):
    """POCS: alternate between flat-spectrum and non-negativity projections."""
    n = len(a0)
    best_a = norm_sum1(a0)
    best_m = fast_autocorr(best_a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit - 3:
            break
        t_trial = (time_limit - (time.monotonic() - t0)) / (n_trials - trial)
        
        rng = np.random.default_rng(rng_seed + trial)
        # Initialize with random phases
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        A_spec = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
        A_full = np.zeros(fft_size, dtype=np.complex128)
        A_full[:fft_size // 2 + 1] = A_spec
        for i in range(1, fft_size // 2):
            A_full[fft_size - i] = np.conj(A_spec[i])
        a = np.fft.irfft(A_full, fft_size)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s
        
        for it in range(100000):
            if time.monotonic() - t0 > time_limit - 3:
                break
            
            # Frequency: push toward flat magnitude
            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            alpha_blend = 0.05 + 0.3 * min(1.0, it / 500)
            new_mag = (1 - alpha_blend) * mag + alpha_blend * mag_mean
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            
            # Time: enforce non-negativity
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s
            
            M_new = fast_autocorr(a_new).max()
            if M_new < best_m - 1e-12:
                best_m = M_new
                best_a = a_new.copy()
            a = a_new
        
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      POCS trial {trial} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def make_init_bump(n, center, width, alpha=1.0):
    """Create a bump-shaped initialization."""
    x = np.arange(n)
    a = np.exp(-((x - center) / width) ** 2) + 0.01
    a = a ** alpha
    return norm_sum1(a)


def make_init_ramp(n, power=1.0):
    """Create a ramp-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = x ** power + 0.01
    return norm_sum1(a)


def make_init_anti_sym(n, sharpness=1.0):
    """Create an anti-symmetric (high-left, low-right) initialization."""
    x = np.linspace(0, 1, n)
    a = (1.0 - x) ** sharpness + 0.05
    return norm_sum1(a)


def make_init_dual_bump(n, c1=0.3, c2=0.7, w=0.1):
    """Two bumps at different positions."""
    x = np.arange(n)
    a = np.exp(-((x - c1 * n) / (w * n)) ** 2) + 0.6 * np.exp(-((x - c2 * n) / (w * n)) ** 2) + 0.02
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search: LSE gradient + LP-FW + coordinate descent + length exploration."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = (1 - x) ** 1.5 + 0.05
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Smooth LSE gradient on parent ===
    print(f"\n=== Phase 1: LSE Gradient (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 250.0)
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_lse_optimize(a_in, t1, tau_init=40.0, tau_final=3.0,
                                        eta_init=0.4, verbose=True, rng_seed=1)
        try_update(a_out, "lse1")
    
    # === Phase 2: LP-FW with growing K ===
    print(f"\n=== Phase 2: LP-FW (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.25, 450.0)
    if t2 > 60 and n_best <= 7000:
        a_in = norm_sum1(best_seq)
        K_start = min(400, 2 * n_best - 1)
        K_max = min(2 * n_best - 1, 12000)
        a_out, _ = lp_frank_wolfe_v2(a_in, t2, solver, K_start=K_start, K_max=K_max, verbose=True)
        try_update(a_out, "lp1")
    
    # === Phase 3: LSE gradient round 2 ===
    print(f"\n=== Phase 3: LSE Gradient r2 (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_lse_optimize(a_in, t3, tau_init=20.0, tau_final=2.0,
                                        eta_init=0.3, verbose=True, rng_seed=10)
        try_update(a_out, "lse2")
    
    # === Phase 4: LP-FW round 2 ===
    print(f"\n=== Phase 4: LP-FW r2 (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.18, 320.0)
    if t4 > 60 and n_best <= 7000:
        a_in = norm_sum1(best_seq)
        K_start = min(800, 2 * n_best - 1)
        K_max = 2 * n_best - 1
        a_out, _ = lp_frank_wolfe_v2(a_in, t4, solver, K_start=K_start, K_max=K_max, verbose=True)
        try_update(a_out, "lp2")
    
    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coord Descent (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.1, 160.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_descent_polish(a_in, t5, verbose=True, rng_seed=42)
        try_update(a_out, "coord1")
    
    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length Exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    
    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 1500, -1500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]
    
    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)
        
        # Quick LSE
        t6a = min(remaining * 0.25, 120.0)
        if t6a > 20:
            a_out, _ = smooth_lse_optimize(a_interp, t6a, tau_init=30.0, tau_final=3.0,
                                            eta_init=0.3, verbose=False, rng_seed=new_n % 100)
            try_update(a_out, f"lse_n{new_n}")
        
        # LP-FW
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 7000:
            t6b = min(remaining * 0.35, 150.0)
            if t6b > 30:
                a_in = norm_sum1(best_seq)
                a_interp2 = interpolate_sequence(a_in, new_n)
                K_max_n = min(2 * new_n - 1, 10000)
                a_out, _ = lp_frank_wolfe_v2(a_interp2, t6b, solver,
                                              K_start=min(400, 2 * new_n - 1),
                                              K_max=K_max_n, verbose=False)
                try_update(a_out, f"lp_n{new_n}")
    
    # === Phase 7: Final LP-FW on best ===
    print(f"\n=== Phase 7: Final LP-FW (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.2, 200.0)
    if t7 > 40 and n_best <= 7000:
        a_in = norm_sum1(best_seq)
        K_max = 2 * n_best - 1
        a_out, _ = lp_frank_wolfe_v2(a_in, t7, solver,
                                      K_start=min(1000, K_max), K_max=K_max, verbose=True)
        try_update(a_out, "lp_final")
    
    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 8: Final Polish (n={len(best_seq)}) ===")
        t8 = min(remaining * 0.5, 80.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_descent_polish(a_in, t8, verbose=False, rng_seed=789)
            try_update(a_out, "coord_final")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = (1 - x) ** 1.5 + 0.05
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
