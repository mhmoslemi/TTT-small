A simple way: find the pair (i, j) with j-i = k (or j-i = n-k for the "wrapped" version) that contributes most to c_k, and reduce one of them while increasing a "random" position that doesn't contribute to any peak.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-Frank-Wolfe with away-step and warm-start (primary optimizer)
2. Nesterov accelerated gradient on smoothed objective (warm-up)
3. Spectral-aware coordinate descent (polishing)
4. Subgradient with Polyak step size (alternative)
5. Multi-length exploration
"""

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None


def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _MODULE_SOLVER


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_xcorr(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(y, beta, top_k=None):
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, p, b


def nesterov_projected(a0, time_limit, beta_start=15.0, beta_end=800.0,
                       beta_growth=1.004, eta_base=0.8, top_k_ratio=0.05,
                       restart_every=3000, verbose=True):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(99)
    top_k = max(20, int(n * top_k_ratio))

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, b = compute_smoothed_grad(y, beta, top_k)
        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.15)
        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 80 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.02 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 5.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest restart t=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J) + " beta=" + repr(beta))

    return best_a, best_max_b


def lp_fw_away(a0, time_limit, solver, initial_K=500, max_K=4000,
               add_per_iter=40, verbose=True, use_away=True):
    """LP-Frank-Wolfe with away-step and warm-start."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    active_list = list(np.argsort(fast_autocorr(y))[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 2000
    lp_time_total = 0.0
    lp_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW-Away: converged at iter " + str(t))
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if len(active_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_list])
            keep = np.argsort(margins)[:max_K]
            active_list = [int(active_list[i]) for i in keep]

        K = len(active_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(active_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        t_lp_start = time.monotonic()
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_list) > initial_K // 2:
                active_list = list(np.argsort(ac)[::-1][:initial_K // 2])
            continue
        lp_time_total += time.monotonic() - t_lp_start
        lp_count += 1

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                if int(k) not in active_list:
                    active_list.append(int(k))

        # Line search with more points
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, default_alpha * 0.5, default_alpha * 1.5,
                  1.0 / (t + 1), 0.5 / (t + 1), 0.1]
        alphas = [max(0.005, min(1.0, a)) for a in alphas]

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

            # Away-step: move away from high-contributing coordinates
            if use_away and no_improve > 200:
                grad_ac = fast_autocorr(y_new)
                top_peaks = np.argsort(grad_ac)[::-1][:30]
                # Find coordinates that contribute to peaks
                contrib = np.zeros(n)
                for k2 in top_peaks[:15]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        contrib[idx2] += y_new[k2i - idx2]
                        contrib[k2i - idx2] += y_new[idx2]
                # Move away from top contributors
                top_contrib = np.argsort(contrib)[::-1][:max(5, n // 100)]
                away_amt = best_alpha * 0.3
                for j in top_contrib:
                    y_new[j] -= away_amt * y_new[j]
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y

        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW-A iter=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J) +
                  " K=" + str(len(active_list)) + " lp_avg=" + str(lp_time_total / max(1, lp_count)) + "s")

    return best_a, best_m


def subgradient_polyak(a0, time_limit, verbose=True):
    """Subgradient method with Polyak step size on the exact max objective."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    x_best = a.copy()
    best_val = 2.0 * n * best_m  # upper bound on objective (with sum=1)

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        max_val = ac.max()
        current_val = 2.0 * n * max_val

        if current_val < best_val - 1e-12:
            best_val = current_val
            best_m = max_val
            best_a = a.copy()
            x_best = a.copy()

        # Subgradient: average gradient over active peaks
        active = np.where(np.abs(ac - max_val) < 1e-10 * max_val)[0]
        # Add some randomness for exploration
        if t % 50 == 0:
            rand_k = rng.integers(0, len(ac))
            active = np.append(active, rand_k)

        n_active = len(active)
        grad = np.zeros(n)
        for k in active[:min(len(active), 200)]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += a[k_int - idx]
                grad[k_int - idx] += a[idx]
        grad /= n_active

        # Polyak step size
        polyak_gap = best_val - 2.0 * n * 1.0 / n  # lower bound is 2 (for sum=1)
        # Actually lower bound is 2, so gap = best_val - 2
        # But we use a more practical lower bound
        lb = 2.0
        gap = max(best_val - lb, 0.01)
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            eta_t = 0.01
        else:
            eta_t = 2.0 * gap / grad_norm_sq
            eta_t = min(eta_t, 0.1)

        z = a - eta_t * grad
        a_new = simplex_project(z)

        # Also try a small random perturbation
        if t % 100 == 0:
            pert = rng.normal(0, 0.005, n)
            a_pert = a_new + pert * a_new
            a_pert = np.maximum(a_pert, 0)
            s_p = a_pert.sum()
            if s_p > 1e-15:
                a_pert = a_pert / s_p
                m_p = fast_autocorr(a_pert).max()
                if 2.0 * n * m_p < 2.0 * n * fast_autocorr(a_new).max():
                    a_new = a_pert

        a = a_new

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print("    SubGrad t=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(2.0 * n * best_m))

    return best_a, best_m


def spectral_coord_descent(a0, time_limit, verbose=True):
    """Coordinate descent using spectral awareness for move selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(314)
    stall = 0
    max_stall = 5000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        max_ac = ac.max()
        n_peaks = min(50, len(ac))
        peak_lags = np.argsort(ac)[::-1][:n_peaks]
        improved = False

        for k in peak_lags[:25]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            src_val = a[src]
            partner_val = a[partner] if 0 <= partner < n else 0.0

            if src_val >= partner_val:
                reduce_idx = src
            else:
                reduce_idx = partner

            if a[reduce_idx] < 1e-13:
                continue

            # Find destination using spectral criterion
            # Prefer positions where the local "autocorrelation pressure" is low
            # Compute pressure from all top peaks
            pressure = np.zeros(n)
            for k2 in peak_lags[:15]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]

            # Also add frequency-domain awareness: prefer positions that
            # have low "spectral energy" contribution
            A = np.fft.rfft(a)
            phase = np.angle(A)
            mag = np.abs(A)

            # Try several candidate destinations
            in_peak = np.zeros(n, dtype=bool)
            for k2 in peak_lags[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)
            cand_pressure = pressure[candidates]
            # Sort by pressure (lowest first) and take top candidates
            top_candidates = candidates[np.argsort(cand_pressure)[:min(30, len(candidates))]]

            found = False
            for frac in [0.02, 0.05, 0.1, 0.2]:
                for dst in top_candidates[:10]:
                    dst_int = int(dst)
                    if dst_int == reduce_idx:
                        continue
                    amt = a[reduce_idx] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_idx] -= amt
                    a_trial[dst_int] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial.copy()
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                if found:
                    break

            if not found and stall > 500:
                # Random exploration
                j = int(rng.integers(n))
                if j != reduce_idx and a[reduce_idx] > 1e-10:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[reduce_idx] * frac
                    a[reduce_idx] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 500 == 0:
                # Broader random moves with spectral guidance
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.003, 0.04)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SpectralCD (" + str(elapsed) + "s): J=" + repr(J) + " stall=" + str(stall))

    return best_a, best_m


def parametric_init(n, time_limit, best_init, best_score):
    """Parametric search for good initialization."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score

    candidates = []
    # Power laws (asymmetric)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    # Smooth asymmetric
    for p in [0.5, 1.0, 1.5, 2.0]:
        candidates.append(np.exp(-p * x))
        candidates.append(np.exp(-p * (1 - x)))
        candidates.append(np.exp(-p * (x - 0.5) ** 2 / 0.1))
    # Sine-based
    for p in [1, 2, 3, 4]:
        candidates.append(np.sin(np.pi * x) ** p)
        candidates.append(np.sin(np.pi * x * 1.5) ** 2)
        candidates.append(np.sin(np.pi * x) ** p + 0.3 * np.sin(2 * np.pi * x))
    # Cosine with phase
    for freq in [1, 2, 3]:
        for phase in [0, 0.3, 0.7]:
            candidates.append(np.maximum((1.0 + np.cos(2 * np.pi * freq * x + phase)) / 2.0, 0.01))
    # Linear斜坡
    candidates.append(i_arr.astype(float) + 1.0)
    candidates.append(np.maximum(n - i_arr, 1.0))
    # Exponential ramp
    candidates.append(np.exp(0.5 * x))
    candidates.append(np.exp(-0.5 * x))
    # Combination
    candidates.append(np.exp(1.0 * x) * np.sin(np.pi * x) ** 0.5)
    candidates.append(np.exp(-1.0 * x) * np.sin(np.pi * x) ** 0.5)

    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(a_raw)
        b = fast_autocorr(a)
        m = b.max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    rng = np.random.default_rng(42)
    for _ in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        # Random smooth sequences
        n_knots = rng.integers(3, 10)
        knot_vals = rng.random(n_knots) ** rng.uniform(0.3, 3.0)
        knot_positions = np.linspace(0, 1, n_knots)
        a_raw = np.interp(x, knot_positions, knot_vals)
        a = norm_sum1(a_raw)
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    return best_a, best_val


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(elapsed) + "s)")
    return best_seq, best_score
