import numpy as np

def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def eval_score(a):
    """Compute 2*n*max(conv)/(sum)^2 for non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = fast_conv(a)
    return float(2.0 * n * ac.max() / (s * s))

def gs_spectral_equalize(a, n_iter, target='flat', alpha=1.0):
    """
    Gerchberg-Saxton: iterate between frequency (flat magnitude) and time (non-negative).
    """
    n = len(a)
    a = normalize(a)
    fft_size = 1 << (4 * n - 2).bit_length()  # enough for linear conv spectrum
    best_a = a.copy()
    best_m = fast_conv(a).max()
    
    for it in range(n_iter):
        # To frequency domain
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target magnitude
        if target == 'flat':
            target_mag = np.ones_like(mag)
        elif target == 'tapered':
            x = np.linspace(0, 1, len(mag))
            target_mag = 1.0 - alpha * x  # reduce high frequencies
        elif target == 'peak_taper':
            x = np.linspace(0, 1, len(mag))
            target_mag = 1.0 - alpha * np.sin(np.pi * x)**2
        else:
            target_mag = np.ones_like(mag)
        
        # Blend: new_mag = mag * (target_mag / mag.mean())^blending
        # Or: new_mag = mag * target_mag^alpha_blend
        blend = 0.5  # 0 = no change, 1 = full replacement
        norm_target = target_mag / target_mag.mean()
        new_mag = mag * (norm_target ** blend)
        # Or full replacement:
        # new_mag = norm_target
        new_mag = mag * (norm_target ** blend)
        
        A_new = new_mag * np.exp(1j * phase)
        
        # Back to time domain
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = normalize(a_new)
        
        m_new = fast_conv(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

def subgradient_optimize(a0, time_limit, beta_start=5.0, beta_end=50.0, 
                         lr_init=0.1, top_k=50, verbose=True, rng_seed=42):
    """
    Nesterov subgradient descent with soft-max approximation of max.
    Works in log-space (multiplicative updates) for non-negativity.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    a_prev = a.copy()
    mu = 0.85  # Nesterov momentum
    
    iter_count = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        # Adaptive beta (increase over time for sharper max approximation)
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = beta_start + (beta_end - beta_start) * progress**0.5
        lr = lr_init / (1.0 + 0.1 * np.sqrt(iter_count))
        
        # Nesterov extrapolation in log-space
        log_ratio = np.log(np.maximum(a, 1e-30)) - np.log(np.maximum(a_prev, 1e-30))
        log_ratio = np.clip(log_ratio, -5, 5)
        a_nest = a * np.exp(mu * log_ratio)
        a_nest = normalize(a_nest)
        
        # Compute autocorrelation
        ac = fast_conv(a_nest)
        
        # Soft-max weights
        max_ac = ac.max()
        if max_ac < 1e-30:
            break
        expo = beta * (ac - max_ac) / n  # scale by n for stability
        expo = np.clip(expo, -50, 0)
        weights = np.exp(expo)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            # Use top-k
            top_idx = np.argsort(ac)[::-1][:top_k]
            weights = np.zeros(len(ac))
            weights[top_idx] = 1.0 / top_k
        else:
            weights = weights / w_sum
        
        # Gradient: grad[i] = 2 * sum_j weights[j+i] * a_nest[j]
        # Efficient: for top contributing lags
        n_top = min(100, len(ac))
        top_w_idx = np.argsort(weights)[::-1][:n_top]
        
        grad = np.zeros(n)
        for k in top_w_idx:
            w = weights[k]
            if w < 1e-12:
                break
            k_int = int(k)
            # grad[i] += 2*w * a[k-i] for 0 <= k-i < n
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo < i_hi:
                idx = np.arange(i_lo, i_hi)
                grad[idx] += 2.0 * w * a_nest[k_int - idx]
        
        # Normalized gradient (scale to be O(1))
        g_norm = np.sqrt(np.mean(grad**2))
        if g_norm > 1e-15:
            grad_norm = grad / g_norm
        else:
            grad_norm = grad
        
        # Multiplicative update: a_new = a * exp(-lr * grad_norm)
        log_a = np.log(np.maximum(a_nest, 1e-30)) - lr * grad_norm
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = normalize(a_new)
        
        a_prev = a
        a = a_new
        iter_count += 1
        
        # Check improvement
        m_new = fast_conv(a).max()
        if m_new < best_m - 1e-13:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > 2000:
            stall = 0
            # Perturb
            pert = best_a * (1.0 + rng.normal(0, 0.1, n))
            pert = normalize(pert)
            w_mix = float(rng.uniform(0.1, 0.3))
            a = normalize((1 - w_mix) * best_a + w_mix * pert)
            a_prev = a.copy()
        
        if verbose and iter_count % 5000 == 0:
            score = 2.0 * n * best_m
            elapsed = time.monotonic() - t0
            print(f"      SG iter={iter_count}, beta={beta:.1f}, score={score:.6f}, t={elapsed:.0f}s")
    
    score = 2.0 * n * best_m
    if verbose:
        print(f"      Subgradient done: score={score:.6f}, time={time.monotonic()-t0:.0f}s")
    return best_a

def coordinate_refine(a0, time_limit, n_swaps_per_iter=50, verbose=True, rng_seed=99):

    Local refinement via pair swaps and small transfers.
    Focus on reducing the top autocorrelation peaks.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_conv(a)
        M = ac.max()
        
        if M < best_m - 1e-13:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                # Random perturbation
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        amt = a[i] * float(rng.uniform(0.001, 0.02))
                        a[i] -= amt
                        a[j] += amt
                a = normalize(a)
                stall = 0
        
        # Find top autocorrelation lags
        n_top = min(20, 2*n-1)
        top_lags = np.argsort(ac)[::-1][:n_top]
        
        # For each top lag, try to reduce it
        for lag in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(lag)
            # Contributors to ac[k]: pairs (i, k-i) for valid i
            i_lo = max(0, k - n + 1)
            i_hi = min(n, k + 1)
            if i_lo >= i_hi:
                continue
            idx = np.arange(i_lo, i_hi)
            partners = k - idx
            contributions = a[idx] * a[partners]
            
            # Find the largest contributing pair
            order = np.argsort(contributions)[::-1][:5]
            for oi in order:
                i_src = int(idx[oi])
                i_dst_cand = int(partners[oi])
                # Try moving mass from the larger of the pair to reduce product
                if a[i_src] >= a[i_dst_cand]:
                    donor = i_src
                    other = i_dst_cand
                else:
                    donor = i_dst_cand
                    other = i_src
                
                if a[donor] < 1e-10:
                    continue
                
                # Find a good destination: somewhere that doesn't contribute to top lags
                # Simple heuristic: pick a random position with small value
                # that's far from the top lag contributions
                candidates = np.where((np.arange(n) != donor) & (a < 0.5 * a[donor]))[0]
                if len(candidates) == 0:
                    continue
                dst = int(rng.choice(candidates))
                
                # Try small transfer
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amt = a[donor] * frac
                    a_trial = a.copy()
                    a_trial[donor] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial / s_t
                    m_t = fast_conv(a_trial).max()
                    if m_t < best_m - 1e-14:
                        a = a_trial
                        best_m = m_t
                        best_a = a.copy()
                        break
    
    score = 2.0 * n * best_m
    if verbose:
        print(f"      Coord refine done: score={score:.6f}, time={time.monotonic()-t0:.0f}s")
    return best_a

def gen_templates(n):
    """Generate diverse parametric templates for initialization."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    templates = []
    
    # Uniform
    templates.append(np.ones(n) / n)
    
    # Linear ramps
    templates.append(normalize(i_arr + 1))
    templates.append(normalize(n - i_arr))
    
    # Exponential
    for r in [0.5, 1, 2, 5]:
        templates.append(normalize(np.exp(-r * x)))
        templates.append(normalize(np.exp(r * x)))
    
    # Power laws
    for p in [0.3, 0.5, 1, 1.5, 2, 3]:
        templates.append(normalize((i_arr + 1) ** p))
        templates.append(normalize((n - i_arr) ** p))
    
    # U-shaped (inverted parabola)
    for w in [0.3, 0.5, 0.7]:
        templates.append(normalize(x**2 + w * (1 - x)**2))
        templates.append(normalize((x - 0.5)**2 * 4 + 0.1))
    
    # Bimodal
    for sig in [0.1, 0.15, 0.2, 0.3]:
        for pos in [(0.2, 0.8), (0.3, 0.7), (0.1, 0.9)]:
            templates.append(normalize(
                np.exp(-((x - pos[0])**2) / (2*sig**2)) + 
                np.exp(-((x - pos[1])**2) / (2*sig**2))))
    
    # Single bump
    for c in [0.25, 0.5, 0.75]:
        for sig in [0.1, 0.2, 0.3, 0.4]:
            templates.append(normalize(np.exp(-((x - c)**2) / (2*sig**2))))
    
    # Sine-based
    for f in [1, 2, 3, 5]:
        phase = 0
        templates.append(normalize(np.sin(np.pi * f * x + phase)**2 + 0.01))
        templates.append(normalize(np.maximum(np.sin(2*np.pi*f*x + np.pi/4), 0)**2 + 0.01))
    
    # Asymmetric combinations
    templates.append(normalize((i_arr + 1) * np.exp(-2 * x)))
    templates.append(normalize((n - i_arr) * np.exp(-2 * (1 - x))))
    templates.append(normalize(np.exp(-3*x) + 0.5 * np.exp(-3*(1-x))))
    templates.append(normalize(i_arr**2 * np.exp(-2*x)))
    
    # Binary-like (0/1 sequences)
    for period in [3, 5, 7, 11]:
        b = np.zeros(n)
        b[np.arange(0, n, period)] = 1.0
        templates.append(normalize(b))
    
    return templates

def parametric_search(n, time_limit=30.0):
    """Quick search over parametric templates."""
    t0 = time.monotonic()
    templates = gen_templates(n)
    
    best_a = None
    best_score = float('inf')
    
    for tmpl in templates:
        if time.monotonic() - t0 > time_limit:
            break
        sc = eval_score(tmpl)
        if sc < best_score:
            best_score = sc
            best_a = tmpl.copy()
    
    # Also try random mixtures
    rng = np.random.default_rng(123)
    n_mix = min(30, len(templates))
    for _ in range(n_mix):
        if time.monotonic() - t0 > time_limit:
            break
        i, j = int(rng.integers(len(templates))), int(rng.integers(len(templates)))
        w = float(rng.uniform(0.2, 0.8))
        mix = normalize(w * templates[i] + (1-w) * templates[j])
        sc = eval_score(mix)
        if sc < best_score:
            best_score = sc
            best_a = mix.copy()
    
    if best_a is None:
        best_a = np.ones(n) / n
        best_score = eval_score(best_a)
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    return normalize(f_int)
