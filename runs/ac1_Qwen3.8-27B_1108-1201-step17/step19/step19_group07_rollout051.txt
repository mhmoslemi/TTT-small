    """Mirror descent with exact subgradient and Nesterov momentum."""
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()
    best_x = x.copy()
    best_c = fast_autocorr(x)
    best_m = best_c.max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_limit = 5000
    momentum = 0.0

    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        elapsed = time.monotonic() - t0

        # Adaptive top_k: increase as we get closer to convergence
        progress = min(elapsed / time_limit, 1.0)
        top_k = top_k_base + int(10 * progress)

        # Compute autocorrelation of y (for Nesterov)
        c = fast_autocorr(y)
        c_max = c.max()

        # Find active lags (within 5% of max)
        threshold = c_max * (1.0 - 0.05 * (1.0 - progress))
        active_mask = c >= threshold
        active_lags = np.where(active_mask)[0]

        # Limit to top_k most active
        if len(active_lags) > top_k:
            active_vals = c[active_lags]
            top_indices = np.argsort(active_vals)[::-1][:top_k]
            active_lags = active_lags[top_indices]

        if len(active_lags) == 0:
            active_lags = np.array([np.argmax(c)])

        # Compute subgradient
        g = subgradient_at_lags(y, active_lags)

        # Adaptive step size (cyclical)
        cycle = 2000
        cycle_pos = (iter_count % cycle) / cycle
        eta_modifier = 1.0 + 0.5 * np.sin(2 * np.pi * cycle_pos)
        eta = eta0 * eta_modifier / (np.sqrt(t) + 1.0)

        # Mirror descent step (exponential family)
        log_y = np.log(np.maximum(y, 1e-300)) - eta * g
        log_y -= log_y.max()
        z = np.exp(log_y)
        z = np.maximum(z, 1e-300)
        z = z / z.sum()

        # Nesterov momentum
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = z + momentum * (z - y)
            x_new = np.maximum(x_new, 1e-300)
            x_new = x_new / x_new.sum()
        else:
            x_new = z.copy()

        # Evaluate
        c_new = fast_autocorr(x_new)
        m_new = c_new.max()

        if m_new < best_m - 1e-15:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1

        # Restart if stalled
        if stall > stall_limit:
            rng = np.random.default_rng(iter_count * 7 + 13)
            # Try different perturbation strategies
            strategy = iter_count % 3
            if strategy == 0:
                pert = rng.normal(0, 0.01, n) * best_x
            elif strategy == 1:
                # Shift-based perturbation
                shift = rng.integers(1, max(2, n // 20))
                pert = np.roll(best_x, shift) - best_x
            else:
                # Boundary boost
                pert = np.zeros(n)
                edge = max(1, n // 50)
                pert[:edge] = 0.02
                pert[-edge:] = 0.02
                pert[n//3:2*n//3] = -0.01

            x_restart = norm_sum1(best_x + pert)
            x_new = x_restart
            y = x_restart.copy()
            stall = 0
            stall_limit = max(stall_limit * 0.7, 1000)

        y = x_new
        x = x_new
        iter_count += 1

        if verbose and iter_count % 5000 == 0:
            J = 2.0 * n * best_m
            print(f"      iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, max_c={best_m:.6e}, |active|={len(active_lags)}")

    return best_x, best_m


def directed_perturbation(a0, time_limit, verbose=False):
    """Local search: try directed perturbations based on active lags."""
    n = len(a0)
    x = norm_sum1(a0)
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(x)
        c_max = c.max()

        if c_max < best_m - 1e-15:
            best_m = c_max
            best_x = x.copy()

        # Find the argmax lag
        k_star = int(np.argmax(c))

        # Find entries that contribute most to c[k_star]
        # (a*a)[k_star] = sum_i a[i] * a[k_star - i]
        j_start = max(0, k_star - n + 1)
        j_end = min(n, k_star + 1)
        idx = np.arange(j_start, j_end)
        contributions = x[idx] * x[k_star - idx]

        # Reduce high-contributing entries, boost low ones
        if len(idx) > 2:
            top_contrib = np.argsort(contributions)[::-1][:max(1, len(idx)//5)]
            low_contrib = np.argsort(contributions)[:max(1, len(idx)//5)]

            total_reduce = contributions[top_contrib].sum()
            epsilon = 0.001 * total_reduce / max(1, len(low_contrib))

            for ti in top_contrib:
                x[idx[ti]] -= epsilon
            for li in low_contrib:
                x[idx[li]] += epsilon

            x = np.maximum(x, 0.0)
            s = x.sum()
            if s > 1e-15:
                x = x / s

        # Also try random swaps
        if iter_count % 3 == 0:
            rng = np.random.default_rng(iter_count)
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = 0.001 * min(x[i], x[j])
                x[i] -= delta
                x[j] += delta
                x = np.maximum(x, 0.0)
                s = x.sum()
                if s > 1e-15:
                    x = x / s

        iter_count += 1

        if verbose and iter_count % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      perturb iter={iter_count} ({elapsed:.0f}s): J={J:.6f}")

    return best_x, best_m


def parametric_init(n, seed=42):
    """Generate diverse parametric initializations."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    rng = np.random.default_rng(seed)
    best_a = None
    best_val = float('inf')

    candidates = []

    # Uniform
    candidates.append(np.ones(n))

    # Linear ramps
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr + 1.0) ** p)

    # U-shape (high at edges)
    for w in [0.1, 0.2, 0.3, 0.4]:
        candidates.append(1.0 + 2.0 * np.exp(-((x - 0.5)**2) / (w*w)))
        candidates.append(np.abs(x - 0.5)**0.5 + 0.1)

    # Gaussian bumps
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sigma in [0.05, 0.1, 0.15, 0.2, 0.3]:
            candidates.append(np.exp(-((x - c)**2) / (2.0*sigma*sigma)) + 0.01)

    # Two bumps
    for c1, c2 in [(0.2, 0.8), (0.25, 0.75), (0.3, 0.7), (0.35, 0.65)]:
        for sigma in [0.05, 0.1, 0.15]:
            candidates.append(np.exp(-((x-c1)**2)/(2*sigma*sigma)) +
                             np.exp(-((x-c2)**2)/(2*sigma*sigma)) + 0.01)

    # Sine-based
    for freq in [1, 2, 3, 5]:
        candidates.append(np.abs(np.sin(np.pi * freq * x)) + 0.1)
        candidates.append(np.sin(np.pi * x)**2 + 0.1)

    # Cosine window (high in middle)
    for beta in [0.5, 1.0, 2.0]:
        candidates.append(np.cos(np.pi * (x - 0.5))**(2*beta) + 0.01)

    # Random smooth
    for _ in range(5):
        rough = rng.random(n)
        # Smooth with moving average
        k = max(1, n // 50)
        kernel = np.ones(k) / k
        smooth = np.convolve(rough, kernel, mode='same')
        candidates.append(smooth + 0.01)

    for a_cand in candidates:
        a_cand = np.maximum(a_cand, 0.0)
        a_norm = norm_sum1(a_cand)
        ac = fast_autocorr(a_norm)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_a = a_norm

    return best_a, best_val


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def quick_eval(a, n_list):
    """Quick evaluation of score for a sequence."""
    c = fast_autocorr(a)
    return 2.0 * len(a) * c.max()


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Exploration (30s) ===
    elapsed = time.monotonic() - t_global
    t1 = min(time_budget * 0.03, 45.0)
    print(f"\n=== Phase 1: Exploration (t={t1:.0f}s) ===")

    # Try different lengths from parent
    n_candidates = []
    if parent is not None:
        n_parent = len(parent)
        for delta in [-1000, -500, 0, 500, 1000, 2000]:
            nn = n_parent + delta
            if 2000 <= nn <= 10000:
                n_candidates.append(nn)
    else:
        n_start = 5000
        best_seq = norm_sum1(np.ones(n_start))
        best_score = evaluate_safe(best_seq)
        n_candidates = [3000, 4000, 5000, 6000, 7000, 8000]

    n_candidates = sorted(set(n_candidates))

    # Quick parametric init for each candidate length
    length_best = {}
    for nn in n_candidates:
        if time.monotonic() - t_global > t1 + elapsed:
            break
        if nn == (len(parent) if parent is not None else 0):
            a_init = best_seq
        else:
            if parent is not None:
                a_init = interpolate_sequence(parent, nn)
            else:
                a_init, _ = parametric_init(nn, seed=nn)

        a_param, _ = parametric_init(nn, seed=nn + 100)
        # Use the better of interpolated parent and parametric
        c1 = fast_autocorr(a_init).max() if a_init is not None else float('inf')
        c2 = fast_autocorr(a_param).max()
        if c2 < c1:
            a_init = a_param

        check_and_update(a_init, f"init_n{nn}")
        length_best[nn] = a_init

    # Pick the best length
    best_n = len(best_seq) if best_seq is not None else 5000
    print(f"  Best after exploration: n={best_n}, score={best_score:.6f}")

    # === Phase 2: Main optimization (50% of remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.40
    n_opt = len(best_seq)
    print(f"\n=== Phase 2: Exact subgradient MD (n={n_opt}, t={t2:.0f}s) ===")

    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = exact_subgradient_mirror_descent(a_in, t2, verbose=True,
                                                     eta0=1.5, top_k_base=3)
        check_and_update(a_out, "exact_md_main")

    # === Phase 3: Second optimization with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.25
    n_opt = len(best_seq)
    print(f"\n=== Phase 3: Second round (n={n_opt}, t={t3:.0f}s) ===")

    if t3 > 20:
        # Try with different initialization (blend with parametric)
        a_param3, _ = parametric_init(n_opt, seed=999)
        a_blend = 0.7 * norm_sum1(best_seq) + 0.3 * a_param3
        a_blend = norm_sum1(a_blend)
        a_out, _ = exact_subgradient_mirror_descent(a_blend, t3 * 0.6, verbose=True,
                                                     eta0=1.0, top_k_base=5)
        check_and_update(a_out, "exact_md_r2_blend")

        # Also try directly on best
        remaining3 = time_budget - (time.monotonic() - t_global)
        if remaining3 > 15:
            a_out, _ = exact_subgradient_mirror_descent(norm_sum1(best_seq), remaining3,
                                                         verbose=False, eta0=0.8, top_k_base=8)
            check_and_update(a_out, "exact_md_r2_direct")

    # === Phase 4: Try a different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: Length exploration (t={t4:.0f}s) ===")

    # Try lengths around the best
    n_current = len(best_seq)
    alt_lengths = []
    for delta in [-1000, 1000, -2000, 2000, -3000, 3000]:
        nn = n_current + delta
        if 2000 <= nn <= 10000:
            alt_lengths.append(nn)
    # Also try some round numbers
    for rn in [3000, 4000, 5000, 6000, 7000, 8000, 9000, 10000]:
        if rn != n_current and abs(rn - n_current) > 500:
            alt_lengths.append(rn)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        t_alt = min(remaining * 0.4, 120.0)
        if t_alt < 10:
            continue
        print(f"  Trying n={alt_n} (t={t_alt:.0f}s)")
        a_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = parametric_init(alt_n, seed=alt_n)
        # Use better init
        if fast_autocorr(interpolate_sequence(best_seq, alt_n)).max() < fast_autocorr(a_alt).max():
            a_alt = interpolate_sequence(best_seq, alt_n)

        a_out, _ = exact_subgradient_mirror_descent(a_alt, t_alt, verbose=False,
                                                     eta0=1.2, top_k_base=3)
        check_and_update(a_out, f"exact_md_n{alt_n}")

    # === Phase 5: Local perturbation polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.15
    n_opt = len(best_seq)
    print(f"\n=== Phase 5: Local perturbation (n={n_opt}, t={t5:.0f}s) ===")

    if t5 > 15:
        a_out, _ = directed_perturbation(best_seq, t5, verbose=True)
        check_and_update(a_out, "perturb")

    # === Phase 6: Final short optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = remaining * 0.5
    n_opt = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_opt}, t={t6:.0f}s) ===")

    if t6 > 10:
        a_out, _ = exact_subgradient_mirror_descent(best_seq, t6, verbose=False,
                                                     eta0=0.5, top_k_base=10)
        check_and_update(a_out, "final_md")

    # Final validation
    if best_seq is not None:
        final_score = evaluate_safe(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 5000
        best_seq = norm_sum1(np.ones(n_fb))
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
