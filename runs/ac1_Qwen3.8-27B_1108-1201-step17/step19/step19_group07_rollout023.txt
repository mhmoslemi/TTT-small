def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    return a / s

def compute_topk_gradient(a, beta, K):
    """
    Compute approximate gradient using top-K lags with softmax weighting.
    Returns (grad, c, top_m) where top_m is the max autocorrelation value.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]

    # Find top-K lags
    if K < size:
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
        c_max = c_top.max()
        exp_vals = np.exp(beta * (c_top - c_max))
        w_full = np.zeros(size)
        w_full[top_idx] = exp_vals
        w_full /= w_full.sum()
    else:
        c_max = c.max()
        w_full = np.exp(beta * (c - c_max))
        w_full /= w_full.sum()

    # Gradient: grad[j] = 2 * sum_k w_full[j+k] * a[k]
    # This is a cross-correlation of w_full with a
    # conv(w_full_padded, a) evaluated at appropriate indices
    # Actually: grad[j] = 2 * (w_full * a)[j] where * is cross-correlation
    # We can compute as: reverse a, convolve, take appropriate slice
    # cross_corr(w_full, a)[j] = sum_k w_full[j+k] * a[k]
    # = conv(w_full, reverse(a))[j + n - 1] ... let me think more carefully
    # conv(w, a_rev)[m] = sum_i w[i] * a_rev[m-i] = sum_i w[i] * a[n-1-(m-i)] = sum_i w[i] * a[n-1-m+i]
    # We want sum_k w_full[j+k] * a[k]. Let i = j+k, so k = i-j.
    # = sum_i w_full[i] * a[i-j]
    # = conv(w_full, a)[j] ... no.
    # conv(w_full, a)[m] = sum_i w_full[i] * a[m-i]
    # Set m-i = k, i = m-k: sum_k w_full[m-k] * a[k]
    # We want sum_k w_full[j+k] * a[k]. Let m-k = j+k => m = j+2k. That's not a simple shift.
    # 
    # Alternative: use the reversal trick.
    # sum_k w_full[j+k] * a[k] = sum_k w_full[j+k] * a[k]
    # Let a_rev[k] = a[n-1-k]. Then a[k] = a_rev[n-1-k].
    # = sum_k w_full[j+k] * a_rev[n-1-k]
    # Let l = n-1-k, k = n-1-l:
    # = sum_l w_full[j+n-1-l] * a_rev[l]
    # = conv(w_full, a_rev)[j+n-1]
    # 
    # So grad[j] = 2 * conv(w_full, a_rev)[j+n-1] for j=0,...,n-1

    a_rev = a[::-1]
    w_full_padded = np.zeros(fft_size)
    w_full_padded[:size] = w_full
    Wf = np.fft.rfft(w_full_padded)
    Arf = np.fft.rfft(a_rev, fft_size)
    conv_result = np.fft.irfft(Wf * Arf, fft_size)
    grad = 2.0 * conv_result[n - 1:2 * n - 1]

    return grad, c, c.max()


def spectral_shaping_step(a, n_harmonics=0):
    """
    IAA-inspired spectral shaping: flatten the power spectrum.
    Returns a new non-negative, sum=1 sequence.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

    # Target: flatten |A| while keeping DC fixed
    abs_A = np.abs(A)
    # Don't change DC
    abs_A[0] = a.sum()
    # Compute target magnitude: geometric mean of non-DC bins
    non_dc = abs_A[1:]
    if non_dc.sum() < 1e-15:
        return a.copy()
    target_mag = np.median(non_dc) * (abs_A.sum() / max(non_dc.sum(), 1e-15)) ** 0.5
    target_mag = max(target_mag, 1e-10)

    # Partially flatten: blend current and target
    blend = 0.3
    new_mag = (1 - blend) * abs_A + blend * target_mag
    new_mag[0] = abs_A[0]  # Keep DC

    # Preserve phase, new magnitude
    phase = np.angle(A)
    A_new = new_mag * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]

    # Project to non-negative, sum=1
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def mirror_descent_optimize(a0, time_limit, beta_start=2.0, beta_end=100.0,
                            eta_base=1.0, K_start=5, K_end=100,
                            verbose=True, do_spectral=False, spectral_every=500):
    """
    Mirror descent (exponential updates) with Nesterov momentum,
    adaptive top-K, backtracking line search, and optional spectral shaping.
    """
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()  # Nesterov lookahead
    best_x = x.copy()
    best_c = fast_autocorr(x)
    best_m = best_c.max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_threshold = 3000
    t = 1

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed >= time_limit - 2:
            break

        progress = min(iter_count / max(50000, 1), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        K = max(K_start, int(K_end * progress))

        # Compute gradient at y (Nesterov point)
        grad, c_y, m_y = compute_topk_gradient(y, beta, K)

        # Backtracking line search
        eta = eta_base / (np.sqrt(t) + 1.0)
        max_backtrack = 6
        found = False
        for bt in range(max_backtrack):
            # Mirror descent step
            log_y = np.log(np.maximum(y, 1e-30)) - eta * grad
            log_y -= log_y.max()
            z = np.exp(log_y)
            z = np.maximum(z, 1e-30)
            z = z / z.sum()

            # Nesterov momentum
            if t > 1:
                momentum = (t - 1.0) / (t + 2.0)
                x_new = z + momentum * (z - y)
                x_new = np.maximum(x_new, 1e-30)
                x_new = x_new / x_new.sum()
            else:
                x_new = z.copy()

            c_new = fast_autocorr(x_new)
            m_new = c_new.max()

            if m_new < best_m * (1 + 1e-10):
                found = True
                break
            else:
                eta *= 0.5

        if not found:
            # Use the last (smallest eta) attempt
            log_y = np.log(np.maximum(y, 1e-30)) - eta * grad
            log_y -= log_y.max()
            z = np.exp(log_y)
            z = np.maximum(z, 1e-30)
            z = z / z.sum()
            if t > 1:
                momentum = (t - 1.0) / (t + 2.0)
                x_new = z + momentum * (z - y)
                x_new = np.maximum(x_new, 1e-30)
                x_new = x_new / x_new.sum()
            else:
                x_new = z.copy()
            c_new = fast_autocorr(x_new)
            m_new = c_new.max()

        # Track best
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1

        # Spectral shaping if stuck
        if do_spectral and stall > stall_threshold // 2 and iter_count % spectral_every < 10:
            x_spec = spectral_shaping_step(x_new)
            c_spec = fast_autocorr(x_spec)
            if c_spec.max() < best_m:
                best_m = c_spec.max()
                best_x = x_spec.copy()
                x_new = x_spec.copy()
                y = x_spec.copy()
                stall = 0
        elif stall > stall_threshold:
            # Restart with perturbation
            rng = np.random.default_rng(iter_count * 7 + 13)
            if rng.random() < 0.3:
                # Spectral restart
                x_restart = spectral_shaping_step(best_x)
            else:
                # Random perturbation
                sigma = 0.01 + 0.05 * (stall / stall_threshold)
                pert = rng.normal(0, sigma, n) * best_x
                x_restart = norm_sum1(best_x + pert)
            x_new = x_restart
            y = x_restart.copy()
            stall = 0
            stall_threshold = max(stall_threshold * 0.85, 1000)
            beta = max(beta_start, beta * 0.5)  # Reset beta

        y = x_new
        t += 1
        iter_count += 1

        if verbose and iter_count % 5000 == 0:
            J = 2.0 * n * best_m
            print(f"    MD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, K={K}, eta={eta:.4f}, stall={stall}")

    return best_x, best_m

def parametric_init(n, best_ref=None):
    """Try diverse parametric families and return the best."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')

    candidates = []

    # Power laws
    for p in [0.3, 0.5, 0.7, 0.85, 1.0, 1.2, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr + 1.0) ** p)

    # Beta-like
    for alpha in [0.5, 0.8, 1.0, 1.3]:
        for beta_p in [0.5, 0.8, 1.0, 1.3]:
            candidates.append(np.maximum(i_arr + 1.0, 0.5) ** alpha *
                              np.maximum(n - i_arr, 0.5) ** beta_p)

    # Gaussian
    for c in [0.3, 0.35, 0.4, 0.45, 0.5, 0.55, 0.6, 0.65, 0.7]:
        for sigma in [0.08, 0.12, 0.15, 0.2, 0.25, 0.3, 0.35]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))

    # Exponential
    for rate in [0.5, 1, 2, 3, 5, 8]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1.0)))

    # Sine-based
    for freq in [1, 2, 3, 4]:
        candidates.append(np.maximum(np.sin(np.pi * freq * x) ** 2, 0.005))
        candidates.append(np.abs(np.sin(np.pi * freq * x)) + 0.01)

    # Cosine window
    for width in [0.3, 0.5, 0.7]:
        candidates.append(np.maximum(np.cos(np.pi * (x - 0.5) / width), 0.0) ** 2 + 0.001)

    # If we have a reference, try blends
    if best_ref is not None and len(best_ref) == n:
        ref_norm = norm_sum1(best_ref)
        for w in [0.3, 0.5, 0.7, 0.9]:
            candidates.append(w * ref_norm + (1.0 - w) * norm_sum1(np.sin(np.pi * x) ** 2 + 0.01))

    for a_cand in candidates:
        a_norm = norm_sum1(a_cand)
        ac = fast_autocorr(a_norm)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_a = a_norm.copy()

    return best_a, best_val

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine starting length
    if parent is not None:
        n_start = len(parent)
    else:
        n_start = 5000
        best_seq = norm_sum1(np.sin(np.pi * np.linspace(0, 1, n_start)) ** 2 + 0.01)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_start}, score={best_score:.6f}")

    # === Phase 1: Parametric search ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 30.0)
    print(f"\n=== Phase 1: Parametric init (n={n_start}, t={t1:.0f}s) ===")
    if t1 > 5:
        a_param, val_param = parametric_init(n_start, best_seq)
        check_and_update(a_param, "parametric")

    # === Phase 2: Main mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.30, 450.0)
    print(f"\n=== Phase 2: Mirror descent (n={n_best}, t={t2:.0f}s) ===")
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_optimize(a_in, t2, beta_start=2.0, beta_end=80.0,
                                           eta_base=0.8, K_start=5, K_end=80,
                                           verbose=True, do_spectral=True, spectral_every=300)
        check_and_update(a_out, "md_main")

    # === Phase 3: LP-FW polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.25, 350.0)
    print(f"\n=== Phase 3: LP-FW polish (n={n_best}, t={t3:.0f}s) ===")
    if t3 > 30 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, initial_K=300, max_K=2000, verbose=True)
        check_and_update(a_out, "lp_pollish")

    # === Phase 4: Mirror descent round 2 (high beta) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 4: MD round 2 high-beta (n={n_best}, t={t4:.0f}s) ===")
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_optimize(a_in, t4, beta_start=20.0, beta_end=300.0,
                                           eta_base=0.4, K_start=10, K_end=150,
                                           verbose=True, do_spectral=False)
        check_and_update(a_out, "md_r2")

    # === Phase 5: Spectral shaping + MD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Spectral + MD (n={n_best}) ===")
    for spec_round in range(3):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        a_in = norm_sum1(best_seq)
        a_spec = spectral_shaping_step(a_in)
        if check_and_update(a_spec, f"spectral_r{spec_round}"):
            pass
        t5 = min(remaining * 0.3, 120.0)
        if t5 > 15:
            a_out, _ = mirror_descent_optimize(a_spec, t5, beta_start=5.0, beta_end=100.0,
                                               eta_base=0.6, K_start=5, K_end=60,
                                               verbose=False, do_spectral=True, spectral_every=200)
            check_and_update(a_out, f"md_after_spec_r{spec_round}")

    # === Phase 6: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")

    length_candidates = []
    for delta in [500, -500, 1000, -1000, 1500]:
        new_n = n_best + delta
        if 3000 <= new_n <= 7000:
            length_candidates.append(new_n)
    for rn in [4000, 5000, 6000]:
        if rn != n_best and 3000 <= rn <= 7000:
            length_candidates.append(rn)
    length_candidates = sorted(set(length_candidates))

    for new_n in length_candidates[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)

        t6 = min(remaining * 0.5, 100.0)
        if t6 > 10:
            a_out, _ = mirror_descent_optimize(a_long, t6, beta_start=3.0, beta_end=80.0,
                                               eta_base=0.7, K_start=5, K_end=80,
                                               verbose=False, do_spectral=False)
            check_and_update(a_out, f"md_n{new_n}")

    # === Phase 7: Final LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW (n={n_best}) ===")
    if remaining > 50 and n_best <= 8000:
        t7 = min(remaining * 0.4, 150.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, initial_K=400, max_K=2500, verbose=True)
            check_and_update(a_out, "lp_final")

    # === Phase 8: Final MD squeeze ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t8 = min(remaining * 0.6, 60.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = mirror_descent_optimize(a_in, t8, beta_start=50.0, beta_end=500.0,
                                               eta_base=0.3, K_start=20, K_end=200,
                                               verbose=False, do_spectral=False)
            check_and_update(a_out, "md_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2 + 0.01)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
