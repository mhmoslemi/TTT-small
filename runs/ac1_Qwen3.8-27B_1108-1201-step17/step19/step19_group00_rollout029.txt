def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def convolve_at_lags(a, lags):
    """Compute convolution values at specific lags. Returns array of values."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    full_conv = np.fft.irfft(A * A, fft_size)[:size]
    return full_conv[lags]

def build_constraint_matrix(a, active_lags, n):
    """Build the constraint matrix for LP: for each lag k, the row is the
    gradient of conv(a,a)[k] w.r.t. a, which gives the linearized constraint."""
    K = len(active_lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(active_lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]
    return coeffs

def lp_fw_improved(a0, time_limit, solver, init_K=600, max_K=6000, add_per=50, verbose=True):
    """Improved LP-based Frank-Wolfe with better constraint management."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    no_improve = 0
    stall_limit = 2000
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    ac_y = fast_autocorr(y)
    active_list = list(np.argsort(ac_y)[::-1][:init_K])
    active_set = set(active_list)

    for t in range(1, 3000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set size
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep_idx = np.argsort(margins)[:max_K]
            active_set = set(int(list(active_set)[i]) for i in keep_idx)

        active_list = sorted(active_set)
        K = len(active_list)
        if K == 0:
            break

        # Build and solve LP
        coeffs = build_constraint_matrix(y, active_list, n)
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        lp_ok = False
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                lp_ok = True
        except Exception:
            active_set = set(int(k) for k in active_list[:init_K])
            active_list = sorted(active_set)
            continue

        if not lp_ok:
            if no_improve > stall_limit // 3:
                break
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find new violations
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr

            else:

def multi_peak_coord(a0, time_limit, n_peaks=8, verbose=True):

    n = len(a0)
    a = normalize(a0)

        ac = fast_autocorr(a)
        peaks = np.argsort(ac)[::-1][:n_peaks]
        improved = False

        for pk in peaks:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(pk)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            # Try to reduce each significant contribution
            top_contrib = np.argsort(contrib)[::-1][:5]
            for ci in top_contrib:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[ci])
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Find best destination: minimize impact on all peaks
                scores = np.zeros(n)
                for pk2 in peaks[:6]:
                    k2i = int(pk2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                        scores[k2i - idx2] += a[idx2]
                scores[src] = 1e10

                # Sample from low-score positions
                n_sample = min(50, n)
                low_scores = np.argsort(scores)[:n_sample]
                dst_candidates = low_scores[np.random.default_rng(src + stall).choice(n_sample, size=min(10, n_sample), replace=False)]

                for dst in dst_candidates:
                    dst = int(dst)
                    if dst == src:
                        continue
                    # Try a few transfer fractions
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                        amt = src_val * frac
                        if amt < 1e-15:
                            continue
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < best_m - 1e-14:
                            a = a_t
                            best_m = Mt
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break

        if not improved:
            stall += 1
            # Random perturbations to escape local minima
            if stall % 10 == 0:
                n_pert = min(5, n)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.008)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        st = a.sum()
                        if st > 1e-15:
                            a = a / st
                            Mn = fast_autocorr(a).max()
                            if Mn < best_m:
                                best_m = Mn
                                best_a = a.copy()
                                stall = 0
                                improved = True
                                break

        if verbose and stall % 500 == 0:
            print("    MultiPeak (" + str(round(time.monotonic()-t0,1)) + "s) J=" + str(2*n*best_m) + " stall=" + str(stall))

    return best_a, 2.0 * n * best_m

def spectral_flatten(a0, time_limit, verbose=True):
    """Gradient descent to flatten the power spectrum (reduce autocorrelation variance)."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    stall = 0
    lr = 0.05

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 3000:
            break
        if t % 500 == 0:
            lr = max(0.001, lr * 0.7)

        A = np.fft.rfft(a, n=fft_size)
        power = np.abs(A) ** 2
        # Gradient of sum(power^2) w.r.t. a (via chain rule through FFT)
        # d/d(a) sum_k |A_k|^4 = 4 * IFFT(|A|^2 * A_conj * A) ... 
        # Actually: d(Re(A_k))/da_i = cos(2*pi*k*i/fft_size)/fft_size
        # Simpler: use the fact that sum_k c[k]^2 where c = conv(a,a)
        # dc/d(a) is the "correlation" of a with shifted a
        # For the max: use soft-max approximation
        ac = fast_autocorr(a)
        
        # Use top-k weighting for the max
        k_top = min(30, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_top]
        vals = ac[top_idx]
        w = np.exp(vals - vals[0])
        w = w / w.sum()

        # Compute gradient: for each top lag, gradient is shifted/reversed a
        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            for i in range(n):
                k_minus_i = ki - i
                if 0 <= k_minus_i < n:
                    grad[i] += 2.0 * wi * a[k_minus_i]
                i_minus_k = i - ki
                if 0 <= i_minus_k < n:
                    grad[i] += 2.0 * wi * a[i_minus_k]

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        val = fast_autocorr(a_new).max()
        if val < best_m - 1e-13:
            best_m = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                # Add noise
                noise = 0.01 * np.random.default_rng(t).standard_normal(n)
                a_new = a + noise
                a_new = np.maximum(a_new, 1e-15)
                a_new = a_new / a_new.sum()
                stall = 0
        a = a_new

        if verbose and t % 2000 == 0:
            print("    Spectral t=" + str(t) + " (" + str(round(time.monotonic()-t0,1)) + "s) J=" + str(2*n*best_m))

    return best_a, 2.0 * n * best_m

def grad_descent_long(a0, time_limit, top_k=80, temp=50.0, c0=2.5, verbose=True):
    """Multiplicative gradient descent with soft top-k for longer sequences."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    x_prev = x.copy()
    stall = 0
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 10000:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[idx]
        w = np.exp(temp * (top_vals - top_vals[0]))
        w = w / w.sum()

        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            idx_arr = np.arange(i_s, i_e)
            grad[idx_arr] += 2.0 * wi * y[ki - idx_arr]

        eta_t = c0 / (np.sqrt(max(t, 1)) + 1.0)
        log_a = np.log(y) - eta_t * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        val = obj_val(x)
        if val < best_val - 1e-12:
            best_val = val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000 and stall % 3000 == 0:
            noise = 0.01 * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            stall = 0

        if verbose and t % 500 == 0:
            print("    GradDes t=" + str(t) + " (" + str(round(time.monotonic()-t0,1)) + "s) J=" + str(best_val))

    return best_a, best_val

def parametric_init(n, best_init, best_score):
    """Quick diverse initialization."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = normalize(best_init)
    best_val = obj_val(best_a)

    candidates = []
    for p in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(normalize((i_arr + 1.0) ** p))
        candidates.append(normalize((n - i_arr) ** p))
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sg in [0.05, 0.1, 0.15, 0.25]:
            candidates.append(normalize(np.exp(-((x - c)**2)/(2*sg*sg))))
    for c1 in [0.2, 0.3, 0.4]:
        for c2 in [0.6, 0.7, 0.8]:
            for sg in [0.05, 0.1, 0.15]:
                cands = np.exp(-((x-c1)**2)/(2*sg*sg)) + np.exp(-((x-c2)**2)/(2*sg*sg))
                candidates.append(normalize(cands))
    for f in [1, 2, 3, 5, 7]:
        for phase in [0, 0.5]:
            cands = np.maximum(1.0 + np.cos(2*np.pi*f*x + phase*np.pi), 0.01)
            candidates.append(normalize(cands))
    for p in [1.0, 2.0, 4.0, 8.0]:
        candidates.append(normalize(x**p + (1.0-x)**p))
    # Sparse patterns
    for period in [3, 5, 7, 11]:
        mask = (i_arr % period < period // 2).astype(float)
        candidates.append(normalize(mask + 0.1))
    # Blend with best init
    bi = normalize(best_init)
    for w in [0.3, 0.5, 0.7]:
        for cand in candidates[:6]:
            candidates.append(normalize(w*bi + (1.0-w)*cand))

    for c in candidates:
        v = obj_val(c)
        if v < best_val:
            best_val = v
            best_a = c.copy()
            if verbose := True:
                if v < 1.6:
                    print("    Param: J=" + str(round(v, 6)))

    return best_a, best_val

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)
    solver = get_solver()
    print("Budget: " + str(round(time_budget,1)) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            print("Parent: n=" + str(len(p)) + " score=" + str(round(best_score, 8)))
    except Exception:
        best_seq = None

    if best_seq is None:
        n_init = 10000
        x = np.linspace(0, 1, n_init)
        best_seq = normalize(np.sin(np.pi * x)**2)
        best_score = eval_safe(best_seq)
        print("Default init n=" + str(n_init) + " score=" + str(round(best_score, 8)))

    n0 = len(best_seq)

    # Phase 1: Parametric init (quick)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Parametric (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.03, 40.0)
    if t1 > 5:
        a_p, v_p = parametric_init(n0, best_seq, best_score)
        sc = eval_safe(a_p)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_p
            print("  New best param: " + str(round(best_score, 8)))

    # Phase 2: Main LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P2: LP-FW (n=" + str(n_cur) + ") ===")
    t2 = min(remaining * 0.28, 450.0)
    if t2 > 60:
        a_out, v_out = lp_fw_improved(normalize(best_seq), t2, solver,
                                      init_K=600, max_K=5000, add_per=50, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  LP-FW improved: " + str(round(best_score, 8)))

    # Phase 3: Multi-peak coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P3: MultiPeak Coord (n=" + str(n_cur) + ") ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 30:
        a_out, v_out = multi_peak_coord(normalize(best_seq), t3, n_peaks=10, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  MultiPeak improved: " + str(round(best_score, 8)))

    # Phase 4: Continue LP-FW with larger active set
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P4: LP-FW cont (n=" + str(n_cur) + ") ===")
    t4 = min(remaining * 0.20, 350.0)
    if t4 > 40:
        a_out, v_out = lp_fw_improved(normalize(best_seq), t4, solver,
                                      init_K=800, max_K=7000, add_per=60, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  LP-FW2 improved: " + str(round(best_score, 8)))

    # Phase 5: Try longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = min(len(best_seq) + 2000, 15000)
    if n_long != len(best_seq) and remaining > 200:
        print("\n=== P5: Longer seq n=" + str(n_long) + " ===")
        t5 = min(remaining * 0.12, 200.0)
        if t5 > 30:
            a_long = interpolate_seq(best_seq, n_long)
            a_out, v_out = grad_descent_long(a_long, t5, top_k=100, temp=60.0, c0=2.0, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  Longer seq improved: " + str(round(best_score, 8)))

    # Phase 6: Spectral flattening
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if n_cur <= 12000:
        print("\n=== P6: Spectral (n=" + str(n_cur) + ") ===")
        t6 = min(remaining * 0.08, 120.0)
        if t6 > 20:
            a_out, v_out = spectral_flatten(normalize(best_seq), t6, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  Spectral improved: " + str(round(best_score, 8)))

    # Phase 7: Final LP-FW or GradDes
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if n_cur <= 10000:
        print("\n=== P7: Final LP-FW (n=" + str(n_cur) + ") ===")
        t7 = min(remaining * 0.18, 250.0)
        if t7 > 30:
            a_out, v_out = lp_fw_improved(normalize(best_seq), t7, solver,
                                          init_K=1000, max_K=8000, add_per=70, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  Final LP improved: " + str(round(best_score, 8)))
    else:
        print("\n=== P7: Final GradDes (n=" + str(n_cur) + ") ===")
        t7 = min(remaining * 0.18, 250.0)
        if t7 > 30:
            a_out, v_out = grad_descent_long(normalize(best_seq), t7, top_k=120, temp=80.0, c0=1.5, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  Final Grad improved: " + str(round(best_score, 8)))

    # Phase 8: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P8: Final Coord (n=" + str(n_cur) + ") ===")
    t8 = min(remaining * 0.15, 180.0)
    if t8 > 20:
        a_out, v_out = multi_peak_coord(normalize(best_seq), t8, n_peaks=12, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  Final Coord improved: " + str(round(best_score, 8)))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + str(round(best_score, 10)) + " n=" + str(len(best_seq)) + " t=" + str(round(elapsed,1)) + "s")
    else:
        print("\nERROR fallback")
        n_fb = 8000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x)**2)
        best_score = eval_safe(best_seq)

    return [float(x) for x in best_seq]
