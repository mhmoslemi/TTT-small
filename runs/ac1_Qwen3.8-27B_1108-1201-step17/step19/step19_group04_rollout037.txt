- For each active k, the gradient: grad_k = 2 * (shift of a0) (i.e., the gradient of a^T Q_k a at a0)

A linearization (cut) at a0 for active index k is:

The bundle method maintains a collection of such cuts and solves:

Actually, since each g_i(a) is linear in a, the inner problem is:

subject to t >= g_i(a) for all maintained cuts i

This is an LP with n+1 variables and (num_cuts + 1 + n) constraints.

The bundle method algorithm:
1. Initialize with one cut at the starting point.
2. Solve the master LP to get trial point a_trial.
3. If a_trial is close to a0 (the center), declare convergence (or use a stopping rule).
4. Otherwise, evaluate f(a_trial) and add a new cut.
5. Optionally, discard old cuts that are no longer tight.
6. Repeat.

This should converge to the optimal solution (for the non-smooth convex function) under appropriate conditions.

The key advantage over Frank-Wolfe is that the bundle method uses a QP/LP master problem that can make larger progress per iteration, and it naturally handles the non-smoothness.

"""
Bundle method for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative 
sequences of fixed length. The problem is convex, so we use a cutting-plane 
(bundle) method with an LP master problem.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / a.size
    return a / s

def self_conv(a):
    """Compute full self-convolution using FFT."""
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = rfft(a, size)
    return irfft(f * f, size)[:m]

def compute_max_conv(a):
    """Return (max_value, argmax_index) of self-convolution."""
    q = self_conv(a)
    k = int(np.argmax(q))
    return float(q[k]), k

def compute_gradient(a, k):
    """Compute gradient of conv(a,a)[k] w.r.t. a.
    grad[j] = 2*a[k-j] if 0 <= k-j < n, else 0.
    """
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        a_idx = k - np.arange(j_start, j_end + 1)
        grad[j_start:j_end+1] = 2.0 * a[a_idx]
    return grad

def get_active_set(a, threshold_frac=0.9995):
    """Get indices where conv(a,a)[k] >= threshold_frac * max."""
    q = self_conv(a)
    M = float(np.max(q))
    threshold = threshold_frac * M
    active = np.where(q >= threshold)[0]
    # Also include the top K by value
    K = min(50, len(q))
    top_K = np.argsort(q)[::-1][:K]
    active = np.unique(np.concatenate([active, top_K]))
    return active, q, M

def solve_master_lp(cuts, n):
    """
    Solve: min t
    s.t. t >= grad_i . a + offset_i for all cuts i
         sum(a) = 1
         a >= 0
    
    Returns (best_a, best_t) or (None, None) if infeasible.
    """
    M = len(cuts)
    if M == 0:
        return np.ones(n) / n, 0.0
    
    # Variables: a[0..n-1], t
    # Minimize: t
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Constraints: t - grad_i . a >= offset_i  =>  -(t - grad_i . a) <= -offset_i
    # i.e., -t + grad_i . a <= -offset_i
    A_ub = np.zeros((M, n + 1))
    b_ub = np.zeros(M)
    for i in range(M):
        grad, offset = cuts[i]
        A_ub[i, :n] = grad
        A_ub[i, n] = -1.0
        b_ub[i] = -offset
    
    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: a >= 0, t free
    bounds = [(0.0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

    a = np.asarray(a, dtype=np.float64).ravel()

        return np.ones(1)
    a = np.where(np.isfinite(a), a, 0.0)
