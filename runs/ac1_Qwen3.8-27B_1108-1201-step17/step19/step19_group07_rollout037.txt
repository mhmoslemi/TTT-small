"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Aggressive LP-Frank-Wolfe polishing with adaptive constraints

import numpy as np
import time
import cvxpy as cp


def fast_autocorr(a):
    """Compute full linear convolution (autocorrelation) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to be non-negative with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def sparse_softmax_gradient(a, beta, top_k=30):
    """
    Compute gradient of softmax approximation to max(conv(a,a)).
    Uses only top_k peaks for efficiency.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (3 * n - 3).bit_length()

    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]

    # Use only top_k peaks
    top_idx = np.argsort(c)[::-1][:top_k]
    c_top = c[top_idx]
    c_max = c_top.max()
    exp_c = np.zeros(size)
    exp_c[top_idx] = np.exp(beta * (c_top - c_max))
    w = exp_c / exp_c.sum()

    # Gradient: d/dc[k] * dc[k]/da[i] = w[k] * a[k-i] (reversed)
    a_rev = a[::-1]
    W = np.fft.rfft(w, fft_size)
    A_rev = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(W * A_rev, fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]

    return grad, c


def fista_optimize(a0, time_limit, beta_start=5.0, beta_end=100.0,
                   eta0=0.7, top_k=30, verbose=True):
    """FISTA optimization with sparse softmax gradient."""
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_threshold = 5000

    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1

        # Adaptive beta: increase over time
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        grad, c_y = sparse_softmax_gradient(y, beta, top_k=top_k)

        # Mirror descent step with 1/sqrt(t) learning rate
        eta = eta0 / (np.sqrt(t) + 1.0)

        log_y = np.log(np.maximum(y, 1e-30)) - eta * grad
        log_y -= log_y.max()
        z = np.exp(log_y)
        z = np.maximum(z, 1e-30)
        z = z / z.sum()

        # Nesterov momentum
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = z + momentum * (z - y)
            x_new = np.maximum(x_new, 1e-30)
            x_new = x_new / x_new.sum()
        else:
            x_new = z.copy()

        c_new = fast_autocorr(x_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > stall_threshold:
            # Perturb and restart
            rng = np.random.default_rng(iter_count * 7 + 13)
            pert = rng.normal(0, 0.01, n) * best_x
            x_restart = norm_sum1(best_x + pert)
            x_new = x_restart
            y = x_restart.copy()
            stall = 0

        y = x_new
        x = x_new
        iter_count += 1

        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}")

    return best_x, best_m


def lp_frank_wolfe(a0, time_limit, solver, initial_K=500, max_K=3000, verbose=True):
    """Aggressive LP-Frank-Wolfe with adaptive constraint management."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    init_K = min(initial_K, 2 * n - 1)
    ac_y = fast_autocorr(y)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0
    failed_solves = 0

    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1

        # Evaluate current best
        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()

        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 10:
                ac_new = fast_autocorr(y)
                new_idx = np.argsort(ac_new)[::-1][:init_K]
                constraint_set = set(int(k) for k in new_idx)
                failed_solves = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            failed_solves += 1
            continue

        failed_solves = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                constraint_set.add(int(k))

        # Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    return best_a, best_m


def get_solver():
    """Find the best available LP solver."""
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:

    print("  LP solver (fallback): ECOS")
    return cp.ECOS


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with sparse softmax gradient (FFT-accelerated)
2. Aggressive LP-Frank-Wolfe polishing with adaptive constraints
3. Multi-length exploration with interpolation
4. Cyclic FISTA/LP-FW refinement
Key improvements over previous: more LP-FW time, better constraint management,
sparse softmax for focused gradients, and multiple polishing cycles.

import numpy as np
import time
import cvxpy as cp


def fast_autocorr(a):
    """Compute full linear convolution (autocorrelation) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to be non-negative with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def sparse_softmax_gradient(a, beta, top_k=30):
    """Compute gradient of sparse softmax approximation to max(conv(a,a))."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (3 * n - 3).bit_length()

    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]

    top_k_actual = min(top_k, size)
    top_idx = np.argsort(c)[::-1][:top_k_actual]
    c_top = c[top_idx]
    c_max = c_top.max()
    exp_c = np.zeros(size)
    exp_c[top_idx] = np.exp(beta * (c_top - c_max))
    w = exp_c / exp_c.sum()

    a_rev = a[::-1]
    W = np.fft.rfft(w, fft_size)
    A_rev = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(W * A_rev, fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]

    return grad, c


def fista_optimize(a0, time_limit, beta_start=5.0, beta_end=100.0,
                   eta0=0.7, top_k=30, verbose=True):
    """FISTA optimization with sparse softmax gradient."""
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_threshold = 4000

    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / max(time_limit, 1.0), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        grad, c_y = sparse_softmax_gradient(y, beta, top_k=top_k)

        eta = eta0 / (np.sqrt(t) + 1.0)

        log_y = np.log(np.maximum(y, 1e-30)) - eta * grad
        log_y -= log_y.max()
        z = np.exp(log_y)
        z = np.maximum(z, 1e-30)
        z = z / z.sum()

        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = z + momentum * (z - y)
            x_new = np.maximum(x_new, 1e-30)
            x_new = x_new / x_new.sum()
        else:
            x_new = z.copy()

        c_new = fast_autocorr(x_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > stall_threshold:
            rng = np.random.default_rng(iter_count * 31 + 7)
            pert = rng.normal(0, 0.008, n) * best_x
            x_restart = norm_sum1(best_x + pert)
            x_new = x_restart
            y = x_restart.copy()
            stall = 0
            stall_threshold = max(stall_threshold * 0.85, 1500)

        y = x_new
        x = x_new
        iter_count += 1

        if verbose and iter_count % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}")

    return best_x, best_m


def lp_frank_wolfe(a0, time_limit, solver, initial_K=500, max_K=3000, verbose=True):

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    init_K = min(initial_K, 2 * n - 1)
    ac_y = fast_autocorr(y)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0
    failed_solves = 0

    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1

        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 8:
                ac_new = fast_autocorr(y)
                new_idx = np.argsort(ac_new)[::-1][:min(init_K, 2 * n - 1)]

            failed_solves += 1

                    else:
                        continue
                except Exception:
                    continue
                failed_solves = 0
            else:
                continue

        failed_solves = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()

            continue

        alpha = 2.0 / (t + 2.0)

        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:

    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
