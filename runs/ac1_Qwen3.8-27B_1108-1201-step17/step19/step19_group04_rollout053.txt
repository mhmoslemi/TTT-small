            a = a_new
            y = a_new.copy()
            beta = min(beta * 1.01, beta_max)
            eta = min(eta * 1.03, 8.0)
        elif score_new < score_y - 1e-14:
            a = a_new
            y = y_new
            beta = min(beta * 1.005, beta_max)
            eta = min(eta * 1.01, 8.0)
        else:
            # No improvement: revert and adjust
            a = best.copy()
            y = a.copy()
            eta = max(eta * 0.75, 0.05)
            beta = max(beta * 0.98, beta_min)
            no_improve += 1
            
            # Periodic perturbation to escape local minima
            if no_improve > 80 and no_improve % 40 == 0:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                noise_scale = 0.005 + 0.002 * (no_improve // 40)
                noise = noise_scale * rng.normal(size=n)
                a = np.maximum(a * np.exp(noise), 1e-15)
                a = a / a.sum()
                y = a.copy()
                eta = 0.5
                beta = 0.8
        
        if verbose and (int(time.monotonic() - t0) % 30 < 1):
            print("    ND: t=%.0fs best=%.9f cur=%.9f eta=%.3f beta=%.3f noimp=%d"
                  % (time.monotonic() - t0, best_score, score_new, eta, beta, no_improve))
    
    return best


def simple_mirror_descent(a_init, time_budget, verbose=False):
    """
    Simple mirror descent without Nesterov, with peak tracking.
    Uses the reversal gradient for center peaks.
    """
    t0 = time.monotonic()
    a = norm_vec(a_init)
    n = a.size
    best = a.copy()
    best_score = obj_norm(a)
    
    eta = 0.5
    no_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        q = self_conv(a)
        M = float(np.max(q))
        score = 2.0 * n * M
        k_star = int(np.argmax(q))
        
        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 250:
            break
        
        # Gradient
        if abs(k_star - (n - 1)) < n // 3:
            grad = a[::-1].copy()
        else:
            grad = compute_gradient_peak(a, q, k_star)
        
        gmax = float(np.max(grad))
        if gmax <= 1e-15:
            break
        grad = grad / gmax
        
        # Mirror update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        q_new = self_conv(a_new)
        M_new = float(np.max(q_new))
        score_new = 2.0 * n * M_new
        
        if score_new < best_score - 1e-13:
            best = a_new.copy()
            best_score = score_new
            a = a_new
            no_improve = 0
            eta = min(eta * 1.04, 6.0)
        elif score_new < score - 1e-15:
            a = a_new
            no_improve = 0
            eta = min(eta * 1.02, 6.0)
        else:
            a = best.copy()
            eta = max(eta * 0.7, 0.02)
            if no_improve > 60 and no_improve % 30 == 0:
                rng = np.random.default_rng(hash(int(time.monotonic()*1000)) % (2**31))
                noise = 0.008 * rng.normal(size=n)
                a = np.maximum(a * np.exp(noise), 1e-15)
                a = a / a.sum()
                eta = 0.4
        
    return best


def lp_direction(a, k_star, n_constraints=200):
    """
    Solve LP subproblem to find direction that reduces the peak.
    min T s.t. sum g = 1, g >= 0, 2*(a*g)[k] - T <= q[k] for active k.
    """
    if not HAVE_LP:
        return None
    n = a.size
    q = self_conv(a)
    M = float(np.max(q))
    
    # Select active constraints: lags closest to peak
    order = np.argsort(q)[::-1]
    K = min(n_constraints, 2 * n - 1)
    active = order[:K]
    q_active = q[active]
    
    # Build constraint matrix
    # For each active lag k: sum_i 2*a[k-i_valid]*g[i] - T <= q[k]
    try:
        A = np.zeros((K, n + 1), dtype=np.float64)
    except Exception:
        return None
    
    for r, k in enumerate(active):
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            indices = np.arange(i0, i1)
            j_indices = k - indices
            A[r, i0:i1] = 2.0 * a[j_indices]
        A[r, n] = -1.0
    
    b = np.maximum(q_active, 0.0)
    c = np.zeros(n + 1)
    c[n] = 1.0
    Aeq = np.zeros((1, n + 1))
    Aeq[0, :n] = 1.0
    bounds = [(0.0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=[1.0],
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None
    
    if res.status != 0 or not res.success:
        return None
    
    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-10:
        return None
    return g / s


def lp_fw_search(a_init, time_budget, verbose=False):
    """
    Frank-Wolfe with LP subproblem.
    """
    t0 = time.monotonic()
    a = norm_vec(a_init)
    n = a.size
    best = a.copy()
    best_score = obj_norm(a)
    
    K = min(150, 2 * n - 1)
    no_improve = 0
    use_lp = HAVE_LP
    lp_fails = 0
    
    while time.monotonic() - t0 < time_budget:
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 3.0:
            break
        
        q = self_conv(a)
        M = float(np.max(q))
        score = 2.0 * n * M
        
        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            break
        
        # Try LP direction
        direction = None
        if use_lp and no_improve < 100:
            t_lp = time.monotonic()
            direction = lp_direction(a, int(np.argmax(q)), K)
            lp_time = time.monotonic() - t_lp
            if direction is None:
                lp_fails += 1
                if lp_fails > 5:
                    use_lp = False
            elif lp_time > max(20.0, 0.1 * remaining):
                use_lp = False
        
        # Fallback: mirror descent direction
        if direction is None:
            k_star = int(np.argmax(q))
            if abs(k_star - (n - 1)) < n // 3:
                grad = a[::-1].copy()
            else:
                grad = compute_gradient_peak(a, q, k_star)
            gmax = float(np.max(grad))
            if gmax > 1e-15:
                direction = a * np.exp(-1.0 * grad / gmax)
                direction = direction / direction.sum()
        
        # Line search
        if direction is not None:
            d = direction - a  # step direction
            # Try several step sizes
            best_alpha = 0.0
            best_M = M
            for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_try = (1 - alpha) * a + alpha * direction
                a_try = np.maximum(a_try, 0.0)
                s_try = float(a_try.sum())
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    M_try = float(np.max(self_conv(a_try)))
                    if M_try < best_M:
                        best_M = M_try
                        best_alpha = alpha
            
            if best_alpha > 0 and best_M < M - 1e-14:
                a = (1 - best_alpha) * a + best_alpha * direction
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
        
        if verbose and int(time.monotonic() - t0) % 30 < 1:
            print("    LP-FW: t=%.0fs best=%.9f score=%.9f K=%d noimp=%d"
                  % (time.monotonic() - t0, best_score, score, K, no_improve))
    
    return best


def lopsided_start(n, pos=0.3, width=0.4):
    """Create a lopsided Gaussian-like start."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos) ** 2) / (2 * width * width))
    return norm_vec(a + 1e-8)


def make_diverse_starts(n, parent):
    """Generate diverse starting sequences."""
    starts = []
    x = np.linspace(0, 1, n)
    
    if parent is not None:
        p = norm_vec(parent)
        if p.size != n:
            p = np.interp(x, np.linspace(0, 1, p.size), p)
            p = norm_vec(p)
        starts.append(p)
        starts.append(p[::-1].copy())
    
    # Lopsided Gaussians
    for pos in [0.2, 0.3, 0.35, 0.4, 0.6, 0.7, 0.8]:
        for width in [0.15, 0.25, 0.35, 0.5]:
            starts.append(lopsided_start(n, pos, width))
    
    # Power-law shapes
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        starts.append(norm_vec((x + 1e-6) ** alpha))
        starts.append(norm_vec(((1 - x) + 1e-6) ** alpha))
    
    # Two-bump
    for c1, c2 in [(0.2, 0.8), (0.3, 0.7), (0.25, 0.75)]:
        for w in [0.1, 0.2, 0.3]:
            s = np.exp(-((x - c1) ** 2) / (2 * w * w)) + np.exp(-((x - c2) ** 2) / (2 * w * w))
            starts.append(norm_vec(s))
    
    # Sine-based
    for k in [1, 2, 3]:
        s = np.abs(np.sin(k * np.pi * x)) ** 2
        starts.append(norm_vec(s))
    
    # Linear + noise
    rng = np.random.default_rng(42)
    for _ in range(10):
        s = np.ones(n) * 0.5
        s += rng.normal(0, 0.3, n)
        s += 0.5 * np.sin(2 * np.pi * rng.uniform(1, 4) * x)
        starts.append(norm_vec(np.maximum(s, 0.1)))
    
    return starts


def power_transform_search(a, time_budget):
    """Try power transforms and blends to improve."""
    t0 = time.monotonic()
    a = norm_vec(a)
    n = a.size
    best = a.copy()
    best_score = obj_norm(a)
    
    base = np.maximum(a, 1e-30)
    for g in [0.3, 0.5, 0.7, 1.5, 2.0, 3.0, 5.0]:
        if time.monotonic() - t0 > time_budget:
            break
        b = norm_vec(base ** g)
        sc = obj_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()
    
    # Blend with uniform
    u = np.ones(n) / n
    for w in [0.05, 0.1, 0.2, 0.3, 0.5]:
        if time.monotonic() - t0 > time_budget:
            break
        b = norm_vec((1 - w) * best + w * u)
        sc = obj_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()
    
    # Blend with reversal
    rev = best[::-1].copy()
    for w in [0.1, 0.2, 0.3, 0.5]:
        if time.monotonic() - t0 > time_budget:
            break
        b = norm_vec((1 - w) * best + w * rev)
        sc = obj_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()
    
    return best


def interpolate_seq(a, new_n):
    """Interpolate a sequence to a new length."""
    a = norm_vec(a)
    old_n = a.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    g = np.interp(x_new, x_old, a)
    return norm_vec(g)
