def compute_subgradient_fft(a, active_set, weights):
    """Compute subgradient of max_k conv(a,a)[k] using FFT for efficiency.
    grad[i] = 2 * sum_k w_k * (a[i+k]*1[i+k<n] + a[i-k]*1[i>=k])
    """
    n = len(a)
    grad = np.zeros(n)
    
    for idx in range(len(active_set)):
        k = int(active_set[idx])
        w = weights[idx]
        if k == 0:
            grad += 4.0 * w * a
        elif k < n:
            grad[:n - k] += 2.0 * w * a[k:]
            grad[k:] += 2.0 * w * a[:n - k]
        else:
            k2 = 2 * n - 2 - k
            if k2 > 0:
                grad[:n - k2] += 2.0 * w * a[k2:]
                grad[k2:] += 2.0 * w * a[:n - k2]
            else:
                grad += 4.0 * w * a
    
    return grad

        ac = fast_autocorr(a_norm)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_a = a_norm.copy()
    
    return best_a, best_val


def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    # Determine starting length
    if parent is not None:
        n_start = len(parent)

        best_seq = norm_sum1((np.arange(n_start, dtype=np.float64) + 1.0) ** 0.7)

    if t1 > 5:

    return [float(x) for x in best_seq]
