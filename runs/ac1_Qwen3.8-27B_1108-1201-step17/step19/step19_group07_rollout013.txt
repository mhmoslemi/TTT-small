```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a)^2) for nonnegative a.

This implementation uses:
  * FFT-based automatic differentiation of the smooth maximum of the
    scaled convolution coefficients.
  * A variable-length base parameterization of the simplex that removes
    the useless global-log-scale degree of freedom.
  * L-BFGS-B on log-ratio variables with a temperature-annealed
    log-sum-exp max surrogate.
  * Multi-stage temporary annealing, perturbation/rotation/resampling
    seeds, and final polishing to squeeze the bound.

The returned sequence is normalized, nonnegative, and finite.
"""

import time
import math
import numpy as np
import scipy.optimize

# ------------------------------------------------------------------
# Globals for FFT / context / callback state (all helper-level helpers
# are top-level as required by the challenge rules).
# ------------------------------------------------------------------

_V_BOUND = 25.0

_CB_STATE = {"t0": 0.0, "limit": 0.0}

_FFT_N = 0
_FFT_AUTO = 0
_FFT_WEIGHT = 0

_CUR_N = 0
_CUR_BASE = 0
_CUR_NON = None


def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _set_fft_cache(n):
    global _FFT_N, _FFT_AUTO, _FFT_WEIGHT
    n = int(n)
    if _FFT_N == n:
        return
    _FFT_N = n
    _FFT_AUTO = _next_pow2(2 * n - 1)
    _FFT_WEIGHT = _next_pow2(3 * n - 2)


def autocorr(a):
    """Linear convolution a * a."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    _set_fft_cache(n)
    A = np.fft.rfft(a, _FFT_AUTO)
    b = np.fft.irfft(A * A, _FFT_AUTO)[:2 * n - 1]
    return b


def weighted_grad(w, a):
    """
    Gradient of H(a)=sum_k w_k (a*a)_k.
    Returns 2*conv(w, reversed(a))[n-1:2n-1].
    """
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    w = np.asarray(w, dtype=np.float64).reshape(-1)
    n = a.size
    _set_fft_cache(n)
    r = a[::-1]
    W = np.fft.rfft(w, _FFT_WEIGHT)
    R = np.fft.rfft(r, _FFT_WEIGHT)
    c = np.fft.irfft(W * R, _FFT_WEIGHT)
    return 2.0 * c[n - 1:2 * n - 1]


def normalized_nonnegative(seq):
    a = np.asarray(seq, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-14:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def fast_score(a):
    a = normalized_nonnegative(a)
    n = a.size
    if n == 0:
        return float("inf")
    b = autocorr(a)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def clean_seq(a, threshold=1e-12):
    a = normalized_nonnegative(a)
    n = a.size
    if n == 0:
        return a
    mask = a > threshold
    if int(np.count_nonzero(mask)) == 0:
        return a
    out = np.zeros(n, dtype=np.float64)
    out[mask] = a[mask]
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-14:
        return a
    return out / s


def resample_seq(a, new_n):
    a = normalized_nonnegative(a)
    n = a.size
    new_n = int(new_n)
    if new_n <= 0:
        new_n = n
    if new_n == n:
        return a
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f = np.interp(x_new, x_old, a)
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if not np.isfinite(s) or s <= 1e-14:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    return f / s


def smooth_seq(a, sigma):
    a = normalized_nonnegative(a)
    n = a.size
    sigma = float(sigma)
    if sigma <= 0.0 or n <= 2:
        return a
    radius = int(max(1, min(3 * sigma, n // 4)))
    x = np.arange(-radius, radius + 1, dtype=np.float64)
    k = np.exp(-(x * x) / (2.0 * sigma * sigma))
    ks = float(k.sum())
    if not np.isfinite(ks) or ks <= 0.0:
        return a
    k /= ks
    out = np.convolve(a, k, mode="same")
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-14:
        return a
    return out / s


# ------------------------------------------------------------------
# Base-coordinate simplex parameterization.
#
# Given a base index r, every probability vector with a_r > 0 can be
# written as
#   a_r = 1 / (1 + sum_i exp(v_i))
#   a_i = exp(v_i) / (1 + sum_i exp(v_i)), i != r
#
# This removes the additive constant degree of freedom of full softmax.
# ------------------------------------------------------------------


def _ensure_v_context(v, base):
    global _CUR_N, _CUR_BASE, _CUR_NON
    n = int(len(v)) + 1
    base = int(base)
    if base < 0 or base >= n:
        base = 0
    if _CUR_N != n or _CUR_BASE != base or _CUR_NON is None:
        _CUR_N = n
        _CUR_BASE = base
        mask = np.ones(n, dtype=bool)
        mask[base] = False
        _CUR_NON = np.nonzero(mask)[0]


def _v_to_seq(v, base):
    v = np.asarray(v, dtype=np.float64).reshape(-1)
    _ensure_v_context(v, base)
    n = _CUR_N
    non = _CUR_NON
    base_idx = _CUR_BASE
    a = np.empty(n, dtype=np.float64)

    if v.size > 0:
        vc = np.clip(v, -_V_BOUND, _V_BOUND)
        vmax = float(np.max(vc))
        shift = vmax if vmax > 0.0 else 0.0
        with np.errstate(over="ignore", under="ignore", invalid="ignore"):
            ev = np.exp(vc - shift)
            base_ev = np.exp(-shift)
        denom = float(ev.sum()) + float(base_ev)
        if not np.isfinite(denom) or denom <= 0.0:
            a[:] = 0.0
            a[base_idx] = 1.0
            return a
        a[non] = ev / denom
        a[base_idx] = base_ev / denom
    else:
        a[0] = 1.0

    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    a /= s
    return a


def _prepare_base_y(a):
    a = normalized_nonnegative(a)
    n = a.size
    base = int(np.argmax(a))
    denom = max(float(a[base]), 1e-15)
    ratios = np.maximum(a, 1e-15) / denom
    with np.errstate(divide="ignore", invalid="ignore"):
        v_full = np.log(ratios)
    v_full[base] = 0.0
    v_full = np.clip(v_full, -_V_BOUND, _V_BOUND)
    mask = np.ones(n, dtype=bool)
    mask[base] = False
    non = np.nonzero(mask)[0]
    return v_full[non], base, a


# ------------------------------------------------------------------
# Smooth objective:
#   s = n * (a*a)
#   phi_T(s) = T log(sum exp(s / T))  ~= max_s
#   score_smooth = 2 * phi_T
#
# The returned objective is multiplied by n, and the corresponding
# gradient is multiplied by n, to improve L-BFGS numerical conditioning.
# ------------------------------------------------------------------


def _objective_logsum(v, T, base):
    v = np.asarray(v, dtype=np.float64).reshape(-1)
    if v.size == 0:
        return 0.0, np.zeros(0, dtype=np.float64)

    if not np.all(np.isfinite(v)):
        return 1e300, np.zeros_like(v, dtype=np.float64)

    T = float(T)
    if T <= 1e-6:
        T = 1e-6

    a = _v_to_seq(v, base)
    n = _CUR_N

    b = autocorr(a)
    s = float(n) * b
    m = float(np.max(s))
    if not np.isfinite(m):
        return 1e300, np.zeros_like(v, dtype=np.float64)

    u = (s - m) / T
    with np.errstate(over="ignore", under="ignore", invalid="ignore"):
        e = np.exp(u)

    e_sum = float(np.sum(e))
    if not np.isfinite(e_sum) or e_sum <= 0.0:
        return 1e300, np.zeros_like(v, dtype=np.float64)

    logsum = math.log(e_sum)
    obj = 2.0 * float(n) * (m + T * logsum)

    w = e / e_sum
    g = weighted_grad(w, a)

    # Unscaled smooth score F = 2*phi_T.
    # grad_a(F) = n * weighted_grad(w,a).
    grad_a = float(n) * g
    dot = float(np.dot(a, grad_a))

    non = _CUR_NON
    grad_v = a[non].copy() * (grad_a[non] - dot)
    # Return gradient of n*F.
    grad_v *= float(n)

    if not np.all(np.isfinite(grad_v)):
        grad_v = np.zeros_like(grad_v, dtype=np.float64)

    return float(obj), grad_v


def _stop_callback(*args):
    if time.monotonic() - _CB_STATE["t0"] > _CB_STATE["limit"]:
        return True
    return False


def run_one_logsum(a, T, alloc_s, maxiter):
    a0 = normalized_nonnegative(a)
    s0 = fast_score(a0)
    if not np.isfinite(s0):
        return a0, s0

    alloc_s = float(alloc_s)
    if alloc_s < 2.0:
        return a0, s0

    v, base, _ = _prepare_base_y(a0)
    m = v.size
    if m == 0:
        return a0, s0

    _CB_STATE["t0"] = time.monotonic()
    _CB_STATE["limit"] = max(0.0, alloc_s - 5.0)

    options = {
        "maxiter": int(max(10, int(maxiter))),
        "gtol": 1e-7,
        "ftol": 1e-11,
        "maxcor": 30,
        "maxls": 12,
    }
    bounds = [(-_V_BOUND, _V_BOUND) for _ in range(m)]

    try:
        res = scipy.optimize.minimize(
            _objective_logsum,
            v,
            method="L-BFGS-B",
            jac=True,
            args=(T, base),
            options=options,
            bounds=bounds,
            callback=_stop_callback,
        )
    except Exception as ex:
        print("    L-BFGS-B exception, using start:", ex)
        return a0, s0

    x = getattr(res, "x", None)
    candidates = [a0]

    if x is not None:
        xv = np.asarray(x, dtype=np.float64).reshape(-1)
        if xv.size == m and np.all(np.isfinite(xv)):
            a_fin = _v_to_seq(xv, base)
            candidates.append(a_fin)
            candidates.append(clean_seq(a_fin, 1e-12))
            for alpha in (0.9, 0.8, 0.6, 0.4):
                candidates.append(a0 + alpha * (a_fin - a0))

    best_a = a0
    best_s = s0
    for cand in candidates:
        sc = fast_score(cand)
        if np.isfinite(sc) and sc < best_s:
            best_a = normalized_nonnegative(cand)
            best_s = sc

    return best_a, best_s


def anneal_sequence(a, time_limit, schedule):
    a = normalized_nonnegative(a)
    best_a = a.copy()
    best_s = fast_score(a)
    t_deadline = time.monotonic() + float(time_limit)

    for T, frac in schedule:
        rem = t_deadline - time.monotonic()
        if rem < 5.0:
            break

        alloc = max(2.0, min(rem * float(frac), 350.0))
        maxiter = int(max(30, min(1200, alloc * 2.5)))

        print(f"    L-SX stage T={T:.4f}, alloc={alloc:.1f}s", flush=True)
        a2, s2 = run_one_logsum(best_a, T, alloc, maxiter)
        if np.isfinite(s2) and s2 < best_s:
            best_a = a2
            best_s = s2
            print(f"      improved to {best_s:.6f}", flush=True)

    return best_a, best_s


def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()

    try:
        budget = float(budget_s)
    except Exception:
        budget = 1800.0

    time_budget = min(budget - 35.0, 1770.0)
    if time_budget < 30.0:
        time_budget = max(20.0, budget - 5.0)

    print(f"Time budget: {time_budget:.1f}s", flush=True)

    best_a = None
    best_s = float("inf")

    # ---------- Load parent construction if available. ----------
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if p.size > 0:
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.maximum(p, 0.0)
            ssum = float(p.sum())
            if ssum > 1e-12:
                pn = p / ssum
                sp = fast_score(pn)
                if np.isfinite(sp):
                    best_a = pn
                    best_s = sp
                    print(f"Parent loaded: n={len(pn)}, score={sp:.6f}", flush=True)
    except Exception:
        pass

    if best_a is None:
        n_fb = 4096
        best_a = np.ones(n_fb, dtype=np.float64) / float(n_fb)
        best_s = fast_score(best_a)
        print(f"Fallback init: n={n_fb}, score={best_s:.6f}", flush=True)

    # ---------- Phase 1: main temperature-annealed L-BFGS polish. ----------
    rem = time_budget - (time.monotonic() - t_start)
    t1 = min(rem * 0.55, 900.0)
    if t1 > 25.0:
        parent_schedule = [
            (0.50, 0.07),
            (0.20, 0.16),
            (0.08, 0.22),
            (0.04, 0.22),
            (0.02, 0.20),
            (0.01, 0.13),
        ]
        best_a, best_s = anneal_sequence(best_a, t1, parent_schedule)
        print(f"After main polish: score={best_s:.6f}, n={len(best_a)}", flush=True)

    # ---------- Phase 2: perturb / rotate / resample seeds. ----------
    rem = time_budget - (time.monotonic() - t_start)
    t2 = min(rem * 0.22, 320.0)
    if t2 > 30.0:
        n = len(best_a)
        rng = np.random.default_rng(20240607)
        seeds = []

        for sigma in (0.02, 0.05, 0.12):
            noise = rng.normal(0.0, sigma, size=n)
            seeds.append(best_a * (1.0 + noise))

        for sf in (0.1, 0.25, 0.4, 0.6, 0.8, 1.1):
            shift = int(round((sf % 1.0) * n))
            if shift != 0:
                seeds.append(np.roll(best_a, shift))

        for mult in (0.9, 1.1):
            nn = int(round(n * mult))
            if 2048 <= nn <= 8192 and nn != n:
                seeds.append(resample_seq(best_a, nn))

        for rn in (4096, 6144):
            if 2048 <= rn <= 8192 and rn != n:
                seeds.append(resample_seq(best_a, rn))

        seeds.append(smooth_seq(best_a, 3.0))
        seeds.append(smooth_seq(best_a, 8.0))

        seed_schedule = [
            (0.20, 0.30),
            (0.06, 0.40),
            (0.025, 0.30),
        ]

        for seed in seeds:
            rem2 = time_budget - (time.monotonic() - t_start)
            if rem2 < 40.0:
                break

            a_seed = normalized_nonnegative(seed)
            sc_seed = fast_score(a_seed)
            if not np.isfinite(sc_seed):
                continue

            if sc_seed < best_s:
                best_a = a_seed
                best_s = sc_seed
                print(f"  seed improved before run: score={best_s:.6f}", flush=True)

            if sc_seed < best_s + 0.05:
                if sc_seed < best_s:
                    alloc = 70.0
                elif sc_seed < best_s + 0.01:
                    alloc = 45.0
                else:
                    alloc = 30.0

                rem3 = time_budget - (time.monotonic() - t_start)
                alloc = min(alloc, max(10.0, rem3 * 0.15))
                if alloc > 15.0:
                    a_try, s_try = anneal_sequence(a_seed, alloc, seed_schedule)
                    if np.isfinite(s_try) and s_try < best_s:
                        best_a = a_try
                        best_s = s_try
                        print(f"  seed run improved: score={best_s:.6f}", flush=True)

    # ---------- Phase 3: final high-precision polish. ----------
    rem = time_budget - (time.monotonic() - t_start)
    t3 = min(rem - 20.0, 700.0)
    if t3 > 25.0:
        final_schedule = [
            (0.05, 0.30),
            (0.02, 0.35),
            (0.01, 0.35),
        ]
        best_a, best_s = anneal_sequence(best_a, t3, final_schedule)
        print(f"After final polish: score={best_s:.6f}, n={len(best_a)}", flush=True)

    # ---------- Phase 4: tiny noisy micro-restarts. ----------
    rem = time_budget - (time.monotonic() - t_start)
    if rem > 45.0:
        rng2 = np.random.default_rng(987654321)
        n = len(best_a)
        micro_runs = [
            (0.02, 0.004),
            (0.01, 0.002),
            (0.005, 0.0),
        ]
        for T, sigma in micro_runs:
            rem = time_budget - (time.monotonic() - t_start)
            if rem < 30.0:
                break
            if sigma > 0.0:
                seed = best_a * (1.0 + sigma * rng2.normal(0.0, 1.0, size=n))
            else:
                seed = best_a

            alloc = min(rem * 0.35, 80.0)
            a_try, s_try = run_one_logsum(seed, T, alloc, 350)
            if np.isfinite(s_try) and s_try < best_s:
                best_a = a_try
                best_s = s_try
                print(f"Micro restart improved: score={best_s:.6f}", flush=True)

    # ---------- Final cleanup / validation. ----------
    rem = time_budget - (time.monotonic() - t_start)
    if rem > 5.0:
        cand_clean = clean_seq(best_a, 1e-12)
        sc_clean = fast_score(cand_clean)
        if np.isfinite(sc_clean) and sc_clean < best_s:
            best_a = cand_clean
            best_s = sc_clean

    best_a = normalized_nonnegative(best_a)
    best_s = fast_score(best_a)

    final_list = [float(x) for x in best_a]

    try:
        eval_score = evaluate_sequence(final_list)
        if np.isfinite(eval_score):
            # Use the official evaluator as the last word for reporting.
            if eval_score < best_s:
                best_s = eval_score
    except Exception:
        pass

    print(f"FINAL: score={best_s:.6f}, n={len(best_a)}", flush=True)
    return final_list
```
