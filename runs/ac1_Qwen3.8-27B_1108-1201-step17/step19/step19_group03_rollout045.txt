def fast_autocorr(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Project onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gs_flat_spectrum(a0, n_iters=500, fft_size=None, blend_rate=0.15):
    """Gerchberg-Saxton: find non-negative a with flat power spectrum."""
    n = len(a0)
    a = norm_sum1(a0)
    if fft_size is None:
        fft_size = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    for it in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        # Target: flat magnitude
        target_mag = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
        new_mag = (1.0 - blend_rate) * mag + blend_rate * target_mag
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_m


def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    """Compute soft subgradient focusing on top-K convolution peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def nesterov_md_optimize(a0, time_limit, top_k_init=50, beta_init=5.0, c0=5.0,
                         restart_every=5000, verbose=False, rng_seed=42):
    """Nesterov-accelerated mirror descent with adaptive parameters."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k_init
    cur_beta = beta_init

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive parameter adjustment
        if stall_count > restart_every // 3:
            cur_beta = max(cur_beta * 0.95, 2.0)
            cur_top_k = min(cur_top_k + 3, 300)

        if stall_count > restart_every:
            stall_count = 0
            # Multi-scale perturbation
            scale = rng.uniform(0.01, 0.05)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 400)
            cur_beta = max(cur_beta - 1.0, 2.5)
            gamma = c0 * rng.uniform(0.8, 1.3)

        # Periodic spectral projection
        if t % 1000 == 0 and stall_count > 500:
            fft_size = 1 << (2 * n - 2).bit_length()
            A = np.fft.rfft(x, fft_size)
            mag = np.abs(A)
            target_mag = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            new_mag = 0.7 * mag + 0.3 * target_mag
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_spec = np.fft.irfft(A_new, fft_size)[:n]
            a_spec = np.maximum(a_spec, 0.0)
            s_spec = a_spec.sum()
            if s_spec > 1e-15:
                a_spec = a_spec / s_spec
                m_spec = fast_autocorr(a_spec).max()
                if m_spec < best_m:
                    best_m = m_spec
                    best_a = a_spec.copy()
                    x = a_spec.copy()
                    x_prev = a_spec.copy()
                    stall_count = 0

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}, K={cur_top_k}, beta={cur_beta:.1f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K_init=300, top_K_max=None,
                   K_increase=50, alpha_candidates=None):
    """LP Frank-Wolfe with adaptive K."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = min(2 * n - 1, 8000)
    K = top_K_init
    iter_count = 0

    if alpha_candidates is None:
        alpha_candidates = [2.0 / (iter_count + 2.0), 0.5, 0.3, 0.2, 0.15, 0.1, 0.05]

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build constraint matrix
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        solve_start = time.monotonic()
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + K_increase, top_K_max)
            continue
        solve_time = time.monotonic() - solve_start

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + K_increase, top_K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Line search
        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0),
                      0.5, 0.3, 0.2, 0.1, 0.05]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1
        K = min(K + K_increase, top_K_max)

        if verbose and iter_count % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}, t_solve={solve_time:.1f}s")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent: move mass from high-peak contributors to low-sensitivity locations."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 80
    n_top = 15

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_top]

        # Compute sensitivity
        sens = np.zeros(n)
        for k in top_k[:8]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[ki - idx]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-sensitivity destinations
            low_sens = np.argsort(sens)[:20]
            for dst in low_sens:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True

        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_hann(n):
    """Hann window, normalized."""
    x = np.arange(n)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n - 1)))
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_cosine_taper(n, power=1.0):
    """Cosine taper with adjustable power."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x / 2.0) ** (2.0 * power)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_flat_taper(n, edge_frac=0.1, edge_power=2.0):
    """Flat in the middle, tapered at edges."""
    a = np.ones(n)
    edge = int(n * edge_frac)
    if edge > 0:
        for i in range(edge):
            t = i / edge
            a[i] = t ** edge_power
            a[n - 1 - i] = t ** edge_power
    return norm_sum1(a)

            return

        if s < 0.01:
            return

        if np.isfinite(sc) and sc < best_score:

            best_seq = a_norm.copy()

            p_score = evaluate_safe(parent)

            best_seq = norm_sum1(parent)
            best_score = p_score

            t_per = max(5.0, (remaining - 10) / (len(alt_inits) + 1))

                    break

    # === Phase 2: Accelerated Gradient Descent ===
    print(f"\n=== Phase 2: Nesterov MD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)

        labels.append("hann")

        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 6200
        best_seq = make_hann(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
