def norm_nn(a):
    """Clip non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgrad_exact(a, S, S_weights):
    """
    Compute exact subgradient of max_{k in S} b[k] weighted by S_weights.
    g[m] = sum_k S_weights[k] * 2*a[k-m] * 1[0<=k-m<n]
    Uses FFT cross-correlation for O(n log n) computation.
    """
    n = len(a)
    L = 2 * n - 1
    s = np.zeros(L)
    for i in range(len(S)):
        s[int(S[i])] = S_weights[i]
    N_fft = 1 << (n + L - 2).bit_length()
    A = np.fft.rfft(a, N_fft)
    Sf = np.fft.rfft(s, N_fft)
    xcorr = np.fft.irfft(A * np.conj(Sf), N_fft)
    g = 2.0 * xcorr[:n]
    return g


def kkt_smeared(a, S, S_weights):
    """Compute h[i] = sum_k w_k * a[k-i] (the 'smeared' vector for KKT)."""
    n = len(a)
    L = 2 * n - 1
    s = np.zeros(L)
    for i in range(len(S)):
        s[int(S[i])] = S_weights[i]
    N_fft = 1 << (n + L - 2).bit_length()
    A = np.fft.rfft(a, N_fft)
    Sf = np.fft.rfft(s, N_fft)
    xcorr = np.fft.irfft(A * np.conj(Sf), N_fft)
    return xcorr[:n]


def mirror_subgrad(a0, time_budget, eta_init=0.15, tol_start=0.005,
                   tol_end=0.05, restart_every=4000, verbose=True, rng_seed=42):
    """
    Mirror descent with exact subgradient and adaptive active set.
    Multiplicative updates preserve non-negativity naturally.
    """
    n = len(a0)
    a = norm_nn(a0)
    best_a = a.copy()
    ac = fft_autocorr(a)
    best_max = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = eta_init
    tol = tol_start
    stall = 0
    total_iters = 0

    while time.monotonic() - t0 < time_budget:
        elapsed = time.monotonic() - t0
        ac = fft_autocorr(a)
        max_b = ac.max()

        if max_b < best_max - 1e-14:
            best_max = max_b
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            pert = best_a * (1.0 + rng.normal(0, 0.08, n))
            a = norm_nn(pert)
            eta = max(eta * 0.7, 0.01)
            tol = min(tol + 0.01, tol_end)
            if verbose:
                print(f"    MD restart ({elapsed:.0f}s): score={2*n*best_max:.6f}, eta={eta:.4f}, tol={tol:.3f}")
            continue

        # Adaptive active set threshold
        progress = min(1.0, stall / restart_every)
        tol_cur = tol_start + (tol_end - tol_start) * progress
        threshold = max_b * (1.0 - tol_cur)
        S = np.where(ac >= threshold)[0]
        S_count = len(S)

        # Weight active lags by how much they exceed threshold
        excess = ac[S] - threshold
        w_sum = excess.sum()
        if w_sum < 1e-30:
            S_weights = np.ones(S_count) / S_count
        else:
            S_weights = excess / w_sum

        # Compute subgradient
        g = compute_subgrad_exact(a, S, S_weights)

        # Multiplicative update with line search
        g_norm = np.sqrt(np.mean(g**2))
        if g_norm < 1e-15:
            break

        # Backtracking: try several step sizes
        found = False
        for trial in range(15):
            eta_try = eta * (0.7 ** trial)
            log_a = np.log(np.maximum(a, 1e-30)) - eta_try * g / max(g_norm, 1e-15)
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            a_new = norm_nn(a_new)
            ac_new = fft_autocorr(a_new)
            max_new = ac_new.max()
            if max_new < max_b * (1.0 - 1e-12):
                a = a_new
                eta = max(eta_try * 1.1, eta * 0.8)  # gentle increase if small step worked
                found = True
                break
        if not found:
            # Accept a small step anyway to avoid stalling
            log_a = np.log(np.maximum(a, 1e-30)) - eta * 0.1 * g / max(g_norm, 1e-15)
            log_a -= log_a.max()
            a = norm_nn(np.exp(log_a))
            eta *= 0.85

        total_iters += 1
        if verbose and total_iters % 20000 == 0:
            print(f"    MD iter={total_iters} ({elapsed:.0f}s): score={2*n*best_max:.6f}, stall={stall}, |S|={S_count}, eta={eta:.4f}")

    if verbose:
        print(f"    Mirror subgrad done ({time.monotonic()-t0:.0f}s): score={2*n*best_max:.6f}, iters={total_iters}")
    return best_a, 2.0 * n * best_max


def kkt_waterfill_iteration(a0, time_budget, verbose=True, rng_seed=7):
    """
    KKT-condition-inspired water-filling:
    At optimum, smeared vector h[i] = sum_k w_k*a[k-i] is constant on support.
    Iterate: compute h, then a_new[i] proportional to min(a[i] * mu / h[i], cap).
    """
    n = len(a0)
    a = norm_nn(a0)
    best_a = a.copy()
    best_max = fft_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_budget:
            break
        ac = fft_autocorr(a)
        max_b = ac.max()
        if max_b < best_max - 1e-14:
            best_max = max_b
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 3000:
                a = norm_nn(best_a * (1 + rng.normal(0, 0.05, n)))
                stall = 0
                if stall > 6000:
                    break

        # Active set (top lags)
        tol = 0.02
        S = np.where(ac >= max_b * (1 - tol))[0]
        S_count = len(S)
        excess = ac[S] - max_b * (1 - tol)
        w_sum = excess.sum()
        if w_sum < 1e-30:
            S_weights = np.ones(S_count) / S_count
        else:
            S_weights = excess / w_sum

        # Compute smeared vector
        h = kkt_smeared(a, S, S_weights)
        h = np.maximum(h, 1e-15)

        # Water-filling: a_new[i] = a[i] * (mu / h[i]) for h[i] > mu
        # Choose mu so that sum(a_new) = 1
        # a_new[i] = min(a[i] * mu / h[i], a[i]) for h[i] > mu
        # a_new[i] = a[i] for h[i] <= mu
        # Simpler: log-space water filling
        log_h = np.log(h)
        log_a = np.log(np.maximum(a, 1e-30))
        # Blend
        blend = 0.3
        log_a_new = (1 - blend) * log_a - blend * (log_h - log_h.mean())
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a = norm_nn(a_new)

    score = 2.0 * n * best_max
    if verbose:
        print(f"    KKT waterfill done ({time.monotonic()-t0:.0f}s): score={score:.6f}")
    return best_a, score


def coord_descent_active(a0, time_budget, verbose=True, rng_seed=55):
    """
    Coordinate descent focused on reducing active lags.
    For each top lag, move mass from high-contributing positions to low-impact ones.
    """
    n = len(a0)
    a = norm_nn(a0)
    best_a = a.copy()
    best_max = fft_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_budget:
        ac = fft_autocorr(a)
        M = ac.max()
        if M < best_max - 1e-14:
            best_max = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                # Perturb
                a = norm_nn(best_a * (1 + rng.normal(0, 0.03, n)))
                stall = 0
                if stall > 10000:
                    break

        # Get top 20 lags
        top_idx = np.argsort(ac)[::-1][:20]
        improved = False

        for k_pos in top_idx[:10]:
            if time.monotonic() - t0 > time_budget:
                break
            k_int = int(k_pos)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo >= i_hi:
                continue
            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            valid = (j_idx >= 0) & (j_idx < n)
            idx = idx[valid]
            j_idx = j_idx[valid]
            contrib = a[idx] * a[j_idx]

            # Find top contributing position
            pi = int(np.argmax(contrib))
            src = int(idx[pi])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Compute impact of each destination on top lags
            impact = np.zeros(n)
            for k2 in top_idx[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                if s2 < e2:
                    ii = np.arange(s2, e2)
                    jj = k2i - ii
                    v2 = (jj >= 0) & (jj < n)
                    ii = ii[v2]
                    jj = jj[v2]
                    impact[ii] += a[jj] * ac[k2i]
                    impact[jj] += a[ii] * ac[k2i]
            impact[src] = np.inf
            dst = int(np.argmin(impact))

            if dst == src:
                continue

            # Try transfer
            for frac in [0.03, 0.07, 0.12, 0.2, 0.35]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_try = a.copy()
                a_try[src] -= amt
                a_try[dst] += amt
                a_try = norm_nn(a_try)
                m_try = fft_autocorr(a_try).max()
                if m_try < best_max - 1e-15:
                    a = a_try
                    best_max = m_try
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

    score = 2.0 * np.sum(a) * 0 + 2.0 * n * best_max  # just 2*n*best_max since sum=1
    score = 2.0 * n * best_max
    if verbose:
        print(f"    Coord descent done ({time.monotonic()-t0:.0f}s): score={score:.6f}")
    return best_a, score


def parametric_candidates(n):
    """Generate diverse parametric candidates for initialization."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    cands = []

    # Uniform
    cands.append(np.ones(n) / n)

    # Raised cosine
    cands.append(norm_nn(0.5 + 0.5 * np.cos(2 * np.pi * x)))
    cands.append(norm_nn(0.5 - 0.5 * np.cos(2 * np.pi * x)))

    # Sinusoidal squared
    for f in [1, 2, 3, 4]:
        cands.append(norm_nn(np.sin(np.pi * f * x) ** 2))
        cands.append(norm_nn(0.5 + 0.5 * np.cos(2 * np.pi * f * x)))

    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        cands.append(norm_nn((i_arr + 1.0) ** p))
        cands.append(norm_nn((n - i_arr) ** p))

    # Symmetric power
    for p in [0.5, 1.0, 2.0, 3.0]:
        c = np.exp(-((x - 0.5) ** 2) * p * 4)
        cands.append(norm_nn(c))

    # Bimodal
    for sig in [0.05, 0.1, 0.15, 0.2]:
        cands.append(norm_nn(
            np.exp(-((x - 0.3) ** 2) / (2 * sig ** 2)) +
            np.exp(-((x - 0.7) ** 2) / (2 * sig ** 2))))

    # Triangular
    cands.append(norm_nn(1.0 - 2.0 * np.abs(x - 0.5)))

    # Sinc-like (zero-padded center)
    for width in [0.3, 0.5, 0.7]:
        mask = (np.abs(x - 0.5) < width / 2).astype(float)
        cands.append(norm_nn(mask + 0.01))

    # Exponential combinations
    for r in [1, 3, 5, 10]:
        cands.append(norm_nn(np.exp(-r * x) + np.exp(-r * (1 - x))))

    # Chebyshev-like
    cands.append(norm_nn(np.cos(np.pi * x) ** 2 + 0.01))
    cands.append(norm_nn(np.cos(3 * np.pi * x) ** 2 + 0.01))

    return cands


def interpolate_seq(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f)
    return norm_nn(f_int)


def eval_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """
    Main search: multi-phase optimization combining exact subgradient mirror descent,
    KKT water-filling, coordinate descent, and length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = eval_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a / s
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent or initialize ===
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_nn(p)
            p_score = eval_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # Try parametric initializations if no good parent
    if best_seq is None or best_score > 1.55:
        for n_try in [3000, 5000, 7000]:
            cands = parametric_candidates(n_try)
            for c in cands:
                sc = eval_safe(c)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = c.copy()
                    print(f"  Parametric init n={n_try}: score={sc:.6f}")

    if best_seq is None:
        n_fb = 5000
        best_seq = np.ones(n_fb) / n_fb
        best_score = eval_safe(best_seq)
        print(f"  Fallback uniform n={n_fb}: score={best_score:.6f}")

    n_primary = len(best_seq)
    print(f"Starting optimization: n={n_primary}, score={best_score:.6f}")

    # === Phase 1: Mirror descent with exact subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.28, 500.0)
    if t1 > 30:
        print(f"\n=== Phase 1: Mirror subgrad (n={n_primary}, budget={t1:.0f}s) ===")
        a_out, sc1 = mirror_subgrad(best_seq, t1, eta_init=0.12, tol_start=0.003,
                                    tol_end=0.06, restart_every=5000, verbose=True, rng_seed=1)
        update_best(a_out, "md1")

    # === Phase 2: KKT water-filling ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.12, 200.0)
    if t2 > 20:
        print(f"\n=== Phase 2: KKT waterfill (n={n_primary}, budget={t2:.0f}s) ===")
        a_out, sc2 = kkt_waterfill_iteration(best_seq, t2, verbose=True, rng_seed=3)
        update_best(a_out, "kkt1")

    # === Phase 3: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Coord descent (n={n_primary}, budget={t3:.0f}s) ===")
        a_out, sc3 = coord_descent_active(best_seq, t3, verbose=True, rng_seed=11)
        update_best(a_out, "cd1")

    # === Phase 4: Second mirror descent (finer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Fine MD (n={n_primary}, budget={t4:.0f}s) ===")
        a_out, sc4 = mirror_subgrad(best_seq, t4, eta_init=0.05, tol_start=0.002,
                                    tol_end=0.04, restart_every=3000, verbose=True, rng_seed=7)
        update_best(a_out, "md2")

    # === Phase 5: Multi-restart with diverse perturbations ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-restart (budget={remaining:.0f}s) ===")
    n_restarts = 3
    t_per_restart = remaining / (n_restarts + 0.5)
    rng = np.random.default_rng(99)
    for ri in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t_r = min(t_per_restart, remaining - 30)
        if t_r < 30:
            break
        # Generate perturbed starting point
        pert_type = ri % 4
        if pert_type == 0:
            start_a = norm_nn(best_seq * (1 + rng.normal(0, 0.15, n_primary)))
        elif pert_type == 1:
            # Structural perturbation: add a bump
            x = np.linspace(0, 1, n_primary)
            c = float(rng.uniform(0.2, 0.8))
            sig = float(rng.uniform(0.05, 0.2))
            bump = np.exp(-((x - c) ** 2) / (2 * sig ** 2))
            start_a = norm_nn(best_seq * 0.7 + bump * 0.3)
        elif pert_type == 2:
            # Frequency mixing
            x = np.linspace(0, 1, n_primary)
            f = int(rng.integers(1, 6))
            mod = 0.5 + 0.5 * np.cos(2 * np.pi * f * x + float(rng.uniform(0, 2 * np.pi)))
            start_a = norm_nn(best_seq * mod)
        else:
            # Random reweighting
            w = rng.uniform(0.3, 2.0, n_primary)
            start_a = norm_nn(best_seq * w)

        print(f"  Restart {ri+1} (type={pert_type}, budget={t_r:.0f}s)")
        a_out, _ = mirror_subgrad(start_a, t_r, eta_init=0.08, tol_start=0.005,
                                  tol_end=0.05, restart_every=3000, verbose=False,
                                  rng_seed=100 + ri)
        update_best(a_out, f"restart{ri+1}")

        # Quick coord descent after restart
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_out, _ = coord_descent_active(a_out, min(remaining * 0.3, 80.0), verbose=False,
                                            rng_seed=200 + ri)
            update_best(a_out, f"restart{ri+1}_cd")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration (n_best={n_best}, budget={remaining:.0f}s) ===")

    length_tries = []
    for delta in [-1000, -500, 500, 1000, 1500, 2000, 3000]:
        nn = n_best + delta
        if 2000 <= nn <= 10000 and nn != n_best:
            length_tries.append(nn)
    for rn in [2500, 3500, 4500, 5500, 6500, 7500, 8500, 9500]:
        if rn != n_best and 2000 <= rn <= 10000:
            length_tries.append(rn)
    length_tries = sorted(set(length_tries))

    for new_n in length_tries[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")

        a_interp = interpolate_seq(best_seq, new_n)
        sc_interp = eval_safe(a_interp)
        if np.isfinite(sc_interp) and sc_interp < best_score:
            best_score = sc_interp
            best_seq = a_interp.copy()
            print(f"    Interpolated already better: {sc_interp:.6f}")

        t_md = min(remaining * 0.5, 120.0)
        if t_md > 20:
            a_out, _ = mirror_subgrad(a_interp, t_md, eta_init=0.06, tol_start=0.003,
                                      tol_end=0.05, restart_every=3000, verbose=False,
                                      rng_seed=new_n % 1000)
            update_best(a_out, f"md_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t_cd = min(remaining * 0.4, 80.0)
            if t_cd > 15:
                a_out, _ = coord_descent_active(a_interp, t_cd, verbose=False,
                                                rng_seed=(new_n + 50) % 1000)
                update_best(a_out, f"cd_n{new_n}")

    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_final}, budget={remaining:.0f}s) ===")

    if remaining > 60:
        t_pd = min(remaining * 0.4, 150.0)
        a_out, _ = mirror_subgrad(best_seq, t_pd, eta_init=0.03, tol_start=0.001,
                                  tol_end=0.03, restart_every=2000, verbose=False, rng_seed=999)
        update_best(a_out, "final_md")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t_pcd = min(remaining * 0.6, 100.0)
        a_out, _ = coord_descent_active(best_seq, t_pcd, verbose=False, rng_seed=777)
        update_best(a_out, "final_cd")

    # Final evaluation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = np.ones(n_fb) / n_fb
        best_score = eval_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")

    return [float(x) for x in best_seq]
