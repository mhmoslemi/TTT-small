def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize_seq(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_conv(a)
    return 2.0 * n * conv.max() / (s * s)

def fista_optimize(a0, time_budget, beta_start=5.0, beta_end=100.0,
                   eta0=0.8, top_k_frac=0.05, verbose=True, label=""):
    n = len(a0)
    x = normalize_seq(a0)
    y = x.copy()
    best_x = x.copy()
    conv_best = fast_conv(x)
    best_m = conv_best.max()
    best_score = 2.0 * n * best_m
    t0 = time.monotonic()
    it = 0
    stall = 0
    stall_patience = 4000

    while time.monotonic() - t0 < time_budget:
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 0.1:
            break
        it += 1
        progress = min(it / max(20000, int(time_budget * 25)), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        conv_y = fast_conv(y)
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()

        top_k = max(20, int(top_k_frac * size))
        if top_k < size:
            top_idx = np.argpartition(conv_y, -top_k)[-top_k:]
            c_top = conv_y[top_idx].copy()
            c_max = c_top.max()
            exp_c = np.zeros(size)
            exp_c[top_idx] = np.exp(beta * (c_top - c_max))
            w = exp_c / exp_c.sum()
        else:
            c_max = conv_y.max()
            exp_c = np.exp(beta * (conv_y - c_max))
            w = exp_c / exp_c.sum()

        y_rev = y[::-1]
        W = np.fft.rfft(w, fft_size)
        Yrev = np.fft.rfft(y_rev, fft_size)
        conv_wy = np.fft.irfft(W * Yrev, fft_size)
        grad = 2.0 * conv_wy[n-1:2*n-1]

        # Adaptive step: scale by local curvature estimate
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-20:
            y = x.copy()
            continue
        eta = eta0 / (np.sqrt(it) + 1.0)

        log_y = np.log(np.maximum(y, 1e-30)) - eta * grad
        log_y -= log_y.max()
        z = np.exp(log_y)
        z = np.maximum(z, 1e-30)
        z /= z.sum()

        if it > 1:
            momentum = (it - 1.0) / (it + 2.0)
            x_new = z + momentum * (z - y)
            x_new = np.maximum(x_new, 1e-30)
            x_new /= x_new.sum()
        else:
            x_new = z.copy()

        m_new = fast_conv(x_new).max()
        score_new = 2.0 * n * m_new
        if score_new < best_score - 1e-18:
            best_score = score_new
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > stall_patience:
            rng = np.random.default_rng(it)
            pert = rng.normal(0, 0.02, n) * best_x
            x_restart = normalize_seq(best_x + pert)
            y = x_restart.copy()
            stall = 0
            stall_patience = max(int(stall_patience * 0.75), 800)

        y = x_new

        if verbose and it % 5000 == 0:

            print(f"    FISTA{label} it={it} ({elapsed:.0f}s): J={best_score:.6f}, beta={beta:.1f}, stall={stall}")

def build_constraint_matrix(a, constraint_list):
    n = len(a)
    K = len(constraint_list)
    A_mat = np.zeros((K, n))
    a_rev = a[::-1]
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            # A_mat[jj, i] = a[k_int - i] for i in [i_start, i_end)
            A_mat[jj, i_start:i_end] = a_rev[k_int + 1 - i_end : k_int + 1 - i_start]
    return A_mat

def lp_frank_wolfe(a0, time_budget, solver, initial_K=300, max_K=2000,
                   verbose=True, label=""):
    n = len(a0)
    a = normalize_seq(a0)
    best_a = a.copy()
    conv_a = fast_conv(a)
    best_m = conv_a.max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    init_K = min(initial_K, 2 * n - 1)
    ac_y = fast_conv(y)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    failed = 0
    it = 0

    while time.monotonic() - t0 < time_budget:
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 2.0:
            break
        it += 1

        ac = fast_conv(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()

        ac_x = fast_conv(x)
        if ac_x.max() < best_m - 1e-16:
            best_m = ac_x.max()
            best_a = x.copy()

        # Adaptive constraint management
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        # Remove very loose constraints (margin > 3x current max violation)
        if len(constraint_list) > 100:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            tight = margins < 0.5 * M_curr
            if tight.sum() < len(constraint_list):
                keep_set = set(int(constraint_list[i]) for i in range(len(constraint_list)) if tight[i])
                # Keep at least initial_K tightest
                if len(keep_set) < initial_K // 2:
                    top_tight = np.argsort(margins)[:initial_K // 2]
                    keep_set = set(int(constraint_list[i]) for i in top_tight)
                constraint_set = keep_set
                constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        A_mat = build_constraint_matrix(y, constraint_list)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A_mat @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed += 1
            if failed > 8:
                constraint_list = sorted(constraint_set)[:min(initial_K, len(constraint_set))]
                constraint_set = set(constraint_list)
                failed = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            failed += 1
            if failed > 8:
                constraint_list = sorted(constraint_set)[:min(initial_K, len(constraint_set))]
                constraint_set = set(constraint_list)
                failed = 0
            continue

        failed = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violated constraints and add them
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            n_add = min(30, len(violations))
            top_v = violations[np.argsort(viol_amt)[::-1][:n_add]]
            for k in top_v:
                constraint_set.add(int(k))

        # Frank-Wolfe step
        alpha = 2.0 / (it + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if it == 1:
            x_new = y_new.copy()
        else:
            beta_t = (it - 1.0) / (it + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW{label} it={it} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    return best_a, best_m

def peak_targeting(a0, time_budget, solver, top_lags=80, verbose=True):

        conv_a = fast_conv(a)
        M_curr = conv_a.max()
        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = a.copy()

        # Identify active lags (top top_lags)
        active = np.argpartition(conv_a, -top_lags)[-top_lags:]
        active_sorted = sorted(int(k) for k in active)

        # Solve LP: minimize M s.t. conv(a, g)[k] <= M, k in active, sum(g)=1, g>=0
        K = len(active_sorted)
        A_mat = build_constraint_matrix(a, active_sorted)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A_mat @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = g_val.sum()
        if gs < 1e-15:

        g_dir = g_val / gs

def parametric_init(n, ref=None):
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')

    candidates = []

    # U-shaped (high at edges)
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        candidates.append((1.0 + np.cos(2.0 * np.pi * x)) ** alpha)

    # Power laws (asymmetric)
    for p in [0.3, 0.7, 1.0, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)

    # Two-lobe
    for c in [0.3, 0.4, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.25]:
            candidates.append(np.exp(-((x - c)**2) / (2*sigma*sigma)) +
                              np.exp(-((x - (1-c))**2) / (2*sigma*sigma)))

    # Sine-based
    for freq in [1, 2, 3]:
        candidates.append(np.abs(np.sin(np.pi * freq * x)))

    # Gaussian center
    for sigma in [0.15, 0.2, 0.3]:
        candidates.append(np.exp(-((x - 0.5)**2) / (2*sigma*sigma)))

    if ref is not None and len(ref) == n:
        ref_n = normalize_seq(ref)
        for w in [0.3, 0.5, 0.7]:
            candidates.append(w * ref_n + (1.0-w) * normalize_seq((1.0 + np.cos(2*np.pi*x))**1.5))

    for cand in candidates:
        cand = np.maximum(cand, 1e-10)
        a_norm = normalize_seq(cand)
        val = fast_conv(a_norm).max()
        if val < best_val:
            best_val = val
            best_a = a_norm

    return best_a, best_val

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            parent_score = compute_J(parent)
            best_seq = parent.copy()
            best_score = parent_score
            print(f"Parent loaded: n={len(parent)}, J={parent_score:.6f}")
    except Exception:
        pass

    if parent is None:
        n_start = 5000
        x = np.linspace(0, 1, n_start)
        best_seq = normalize_seq((1.0 + np.cos(2.0*np.pi*x))**1.2)
        best_score = compute_J(best_seq)
        print(f"Default init: n={n_start}, J={best_score:.6f}")

    solver = get_lp_solver()
    print(f"Total time budget: {time_budget:.0f}s, n={len(best_seq)}")

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = compute_J(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Parametric init (15s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.01, 15.0)
    print(f"\n=== Phase 1: Parametric init (n={len(best_seq)}, t={t1:.0f}s) ===")
    if t1 > 3:
        a_param, _ = parametric_init(len(best_seq), best_seq)
        check_update(a_param, "parametric")

    # === Phase 2: FISTA main (400s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.28, 420.0)
    print(f"\n=== Phase 2: FISTA main (n={len(best_seq)}, t={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = fista_optimize(best_seq, t2, beta_start=3.0, beta_end=80.0,
                                   eta0=0.8, top_k_frac=0.04, verbose=True, label="_main")
        check_update(a_out, "fista_main")

    # === Phase 3: LP-FW (300s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.22, 320.0)
    print(f"\n=== Phase 3: LP-FW (n={len(best_seq)}, t={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = lp_frank_wolfe(best_seq, t3, solver, initial_K=400, max_K=1800,
                                   verbose=True, label="_main")
        check_update(a_out, "lp_main")

    # === Phase 4: FISTA round 2 (200s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.14, 220.0)
    print(f"\n=== Phase 4: FISTA r2 (n={len(best_seq)}, t={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = fista_optimize(best_seq, t4, beta_start=20.0, beta_end=250.0,
                                   eta0=0.5, top_k_frac=0.03, verbose=True, label="_r2")
        check_update(a_out, "fista_r2")

    # === Phase 5: LP-FW round 2 (200s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.14, 220.0)
    print(f"\n=== Phase 5: LP-FW r2 (n={len(best_seq)}, t={t5:.0f}s) ===")
    if t5 > 25:
        a_out, _ = lp_frank_wolfe(best_seq, t5, solver, initial_K=500, max_K=2000,
                                   verbose=True, label="_r2")
        check_update(a_out, "lp_r2")

    # === Phase 6: Length exploration (200s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration (n_best={n_best}, remaining={remaining:.0f}s) ===")

    length_candidates = []
    for delta in [-500, 500, -1000, 1000]:
        new_n = n_best + delta
        if 3000 <= new_n <= 9000:
            length_candidates.append(new_n)
    for rn in [4000, 5000, 6000, 7000, 8000]:
        if rn != n_best and 3000 <= rn <= 9000:
            length_candidates.append(rn)
    length_candidates = sorted(set(length_candidates))

    for new_n in length_candidates[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_seq(best_seq, new_n)

        t6f = min(remaining * 0.45, 150.0)
        if t6f > 10:
            a_out, _ = fista_optimize(a_long, t6f, beta_start=5.0, beta_end=100.0,
                                       eta0=0.7, top_k_frac=0.04, verbose=False, label=f"_n{new_n}")
            check_update(a_out, f"fista_n{new_n}")

    # === Phase 7: Peak targeting (150s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Peak targeting (n={n_best}, remaining={remaining:.0f}s) ===")
    if remaining > 60 and n_best <= 10000:
        t7 = min(remaining * 0.2, 160.0)
        if t7 > 30:
            a_out, _ = peak_targeting(best_seq, t7, solver, top_lags=60, verbose=True)
            check_update(a_out, "peak_tgt")

    # === Phase 8: Final combined (remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish (n={n_best}, remaining={remaining:.0f}s) ===")

    if remaining > 40:
        t8a = min(remaining * 0.5, 120.0)
        if t8a > 10:
            a_out, _ = fista_optimize(best_seq, t8a, beta_start=50.0, beta_end=500.0,
                                       eta0=0.3, top_k_frac=0.02, verbose=False, label="_final")
            check_update(a_out, "fista_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and len(best_seq) <= 10000:
        t8b = min(remaining * 0.6, 100.0)
        if t8b > 20:
            a_out, _ = lp_frank_wolfe(best_seq, t8b, solver, initial_K=600, max_K=2500,
                                       verbose=True, label="_final")
            check_update(a_out, "lp_final")

    # Final validation
    if best_seq is not None:
        fs = validate_score(best_seq)
        if np.isfinite(fs):
            if fs != best_score:
                print(f"  Validation: internal={best_score:.6f}, actual={fs:.6f}")
            best_score = min(best_score, fs)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: J={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize_seq(np.sin(np.pi * x) ** 2)
        best_score = validate_score(best_seq)
        print(f"Fallback: J={best_score:.6f}")

    return [float(x) for x in best_seq]
