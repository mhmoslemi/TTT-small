"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Softmax-parameterized L-BFGS on log-sum-exp smoothed objective
2. Adaptive peak-chasing coordinate descent
3. LP-Frank-Wolfe with scipy solver
4. Multi-length exploration
5. Iterative refinement with increasing smoothing parameter

import numpy as np

import cvxpy as cp

def softmax(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    return e / s

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_obj_and_grad(x, beta, top_k=None):
    a = softmax(x)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:size]
    
    if top_k is not None and top_k < size:
        top_idx = np.argsort(conv)[::-1][:top_k]
        conv_sel = np.zeros(size)
        conv_sel[top_idx] = conv[top_idx]
    else:
        conv_sel = conv
    
    m = conv_sel.max()
    shifted = beta * (conv_sel - m)
    log_s = np.log(np.exp(shifted).sum())
    obj = (m + log_s / beta)
    
    p = np.exp(shifted)
    p /= p.sum()
    
    # grad_a[i] = 2 * sum_j p[i+j] * a[j]
    p_rev = p[::-1]
    conv_len = n + size - 1
    fft_s2 = 1 << (conv_len - 1).bit_length()
    A2 = np.fft.rfft(a, fft_s2)
    P2 = np.fft.rfft(p_rev, fft_s2)
    conv_ap = np.fft.irfft(A2 * P2, fft_s2)[:conv_len]
    grad_a = 2.0 * conv_ap[n - 1:2 * n - 1][::-1]
    
    grad_x = a * (grad_a - np.dot(grad_a, a))
    
    return obj, grad_x

def lbfgs_smooth(a0, time_limit, beta_start=20.0, beta_end=2000.0,
                 beta_multiplier=1.05, iterations_per_beta=50,
                 top_k_ratio=0.1, verbose=True):
    """L-BFGS optimization of smoothed objective with increasing beta."""
    n = len(a0)
    a = norm_sum1(a0)
    # Initialize x from a: x = log(a) (with small epsilon)
    x = np.log(np.maximum(a, 1e-15))
    x -= x.mean()  # center to avoid numerical issues
    
    best_a = a.copy()
    best_max_b = fast_autocorr(a).max()
    t0 = time.monotonic()
    beta = beta_start
    total_iters = 0
    top_k = max(50, int(n * top_k_ratio))
    
    while time.monotonic() - t0 < time_limit:
        # Run L-BFGS for this beta
        def fun(x_eval):
            obj, grad = smooth_obj_and_grad(x_eval, beta, top_k)
            return obj, grad
        
        try:
            res = minimize(fun, x, method='L-BFGS-B', 
                         options={'maxiter': iterations_per_beta, 
                                  'gtol': 1e-8,
                                  'ftol': 1e-10})
            total_iters += res.get('nit', 0)
            x = res.x
        except Exception as e:
            if verbose:
                print("  L-BFGS error: " + str(e))
            break
        
        a_new = softmax(x)
        b_new = fast_autocorr(a_new)
        max_new = b_new.max()
        
        if max_new < best_max_b - 1e-12:
            best_max_b = max_new
            best_a = a_new.copy()
        
        if verbose and beta % 100 < beta_multiplier:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("  L-BFGS beta=" + str(int(beta)) + " (" + str(elapsed) + "s): J=" + repr(J))
        
        # Increase beta
        beta *= beta_multiplier
        if beta > beta_end:
            break
        
        # Don't let top_k grow too large
        top_k = min(top_k + n // 10000, int(n * 0.3))
    
    return best_a, best_max_b

def peak_chase(a0, time_limit, verbose=True):
    """Aggressive peak-chasing: reduce the maximum by redistributing mass."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_autocorr(a)
        max_val = conv.max()
        
        if max_val < best_m - 1e-12:
            best_m = max_val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find the peak lag
        k_peak = int(np.argmax(conv))
        
        # Find contributing pairs
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        if js > je:
            stall += 10
            continue
        
        idx = np.arange(js, je + 1)
        products = a[idx] * a[k_peak - idx]
        
        # Focus on the top contributors
        top_contrib = np.argsort(products)[::-1][:min(10, len(idx))]
        
        improved = False
        for ci in top_contrib:
            i = int(idx[ci])
            j = k_peak - i
            
            # Try to reduce a[i] or a[j] (whichever is larger relative to its "cost")
            # Cost of reducing a[i]: it contributes to other convolution values too
            # Simple heuristic: reduce the larger one
            if a[i] >= a[j]:
                src = i
            else:
                src = j
            
            if a[src] < 1e-12:
                continue
            
            # Try different amounts and destinations
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-14:
                    continue
                
                # Destination: try positions that minimize the increase in other peaks
                # Simple: try a few random positions + the position that minimizes gradient
                a_trial = a.copy()
                a_trial[src] -= amt
                
                # Try several destinations
                candidates = []
                # Random candidates
                candidates.extend([int(rng.integers(n)) for _ in range(5)])
                # Low-value positions
                low_idx = np.argsort(a)[:10]
                candidates.extend([int(kk) for kk in low_idx])
                # Position far from peak contributing pairs
                candidates.extend([int((src + n // 2) % n), int((src + n // 3) % n)])
                
                for dst in candidates:
                    if dst == src:
                        continue
                    a_trial2 = a_trial.copy()
                    a_trial2[dst] += amt
                    a_trial2 = np.maximum(a_trial2, 0)
                    s = a_trial2.sum()
                    if s < 1e-15:
                        continue
                    a_trial2 = a_trial2 / s
                    m_trial = fast_autocorr(a_trial2).max()
                    if m_trial < best_m - 1e-12:
                        a = a_trial2
                        best_m = m_trial
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation when stuck
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                        m_new = fast_autocorr(a).max()
                        if m_new < best_m:
                            best_m = m_new
                            best_a = a.copy()
                            stall = 0
    
    return best_a, best_m

def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2000, verbose=True):
    """LP-Frank-Wolfe method."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 500
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-12:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx_arr = np.arange(i_start, i_end)
                coeffs[jj, idx_arr] = y[k_int - idx_arr]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K // 2])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violations
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:20]]
            for k in top_v:
                active_set.add(int(k))
        
        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.01, default_alpha * 0.5),
                  max(0.01, default_alpha * 1.5), 1.0 / (t + 1)]
        alphas = list(set([max(0.01, min(1.0, a)) for a in alphas]))
        
        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y
        y = y_new
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  LP-FW iter=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J))
    
    return best_a, best_m

def get_solver():

            prob.solve(solver=s, verbose=False)

    return cp.ECOS

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
