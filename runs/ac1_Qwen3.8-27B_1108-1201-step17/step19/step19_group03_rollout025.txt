def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def norm_sum1(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, beta):
    """Compute gradient of (1/beta)*log(sum(exp(beta*conv_k))) w.r.t. a."""
    n = len(a)
    conv = fast_autocorr(a)
    max_conv = conv.max()
    exp_bc = beta * (conv - max_conv)
    p = np.exp(exp_bc)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        p = p / p_sum
    # grad_i = (1/beta) * sum_j p[i+j] * a[j] for j=0..n-1
    # This is correlate(p, a, 'valid')
    size = 1 << (3 * n - 3).bit_length()
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(P * np.conj(A), size)
    grad = corr[n - 1:2 * n - 1] / beta
    return grad, conv, p


def nesterov_smooth_optimize(a0, time_limit, verbose=False, beta_init=5.0,
                              beta_final=300.0, rng_seed=42, max_stall=200):
    """Nesterov-accelerated mirror descent on log-sum-exp smoothed objective."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    eta = 0.3
    best_eta = eta
    consecutive_decrease = 0
    it = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        beta = beta_init * (beta_final / beta_init) ** (progress ** 0.7)

        # Nesterov momentum
        momentum = (it - 1.0) / (it + 1.5)
        y = x + momentum * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute gradient
        grad, conv, p = compute_smooth_gradient(y, beta)

        # Mirror descent step
        log_y = np.log(np.maximum(y, 1e-15))
        log_new = log_y - eta * grad
        max_log = log_new.max()
        a_new = np.exp(log_new - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
            consecutive_decrease += 1
            if consecutive_decrease % 50 == 0:
                eta = min(eta * 1.05, 2.0)
        else:
            stall += 1
            consecutive_decrease = 0
            if stall > 50:
                eta *= 0.8
            if stall > 100:

                x_p = np.maximum(best_a + pert, 1e-10)
                x = x_p / x_p.sum()
                x_prev = x.copy()
                stall = 50
                eta = min(eta * 1.3, 1.0)

def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=False,
                      top_K_init=200, top_K_max=None, use_all=False,
                      line_search=True):
    """Improved LP-Frank-Wolfe with line search and adaptive K."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0
    stall = 0
    max_stall = 30

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if use_all or K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build constraint matrix
        K_act = len(binding_idx)
        T_sel = np.zeros((K_act, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start < j_end:
                idx = np.arange(j_start, j_end)
                T_sel[ii, idx] = a[ki - idx]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 50, top_K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Line search
        if line_search:
            best_alpha = 0.1
            best_trial_m = fast_autocorr((1 - 0.1) * a + 0.1 * g_dir).max()
            alphas = [2.0 / (iter_count + 2.0), 0.5, 0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03]
            for alpha in alphas:
                alpha = min(1.0, alpha)
                a_trial = (1.0 - alpha) * a + alpha * g_dir
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_trial_m:
                    best_trial_m = m_trial
                    best_alpha = alpha
            alpha = best_alpha
        else:
            alpha = 2.0 / (iter_count + 2.0)

        a_new = (1.0 - alpha) * a + alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            break
        a_new = a_new / s_a

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            a = a_new
        else:
            stall += 1
            a = a_new

        iter_count += 1
        if iter_count % 5 == 0:
            K = min(K + 40, top_K_max)

        if verbose and iter_count % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={alpha:.4f}, stall={stall}")

    return best_a, best_m

        improved = False

            if time.monotonic() - t0 > time_limit:

            # Find quiet destination (not in any top peak)
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            # Prefer destinations with low sensitivity
            cand_sens = sens[candidates]
            top_cands = candidates[np.argsort(cand_sens)[:5]]

            found = False
            for dst in top_cands:
                if dst == src or found:
                    continue
                for frac in [0.005, 0.01, 0.003, 0.02, 0.015, 0.008]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

            if found:
                stall = 0
                if stall == 0 and it % 100 == 0:
                    pass  # just tracking

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakRed it={it} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def coordinate_polish_v2(a0, time_limit, verbose=False, rng_seed=123):
    """Enhanced coordinate descent with multiple move sizes."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200
    it = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        it += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(15, 2 * n - 1)
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]

            if contrib.max() < 1e-20:
                continue

            src_local = np.argmax(contrib)
            src = idx[src_local]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Try moving to several candidate destinations
            # Destinations that minimize impact on top peaks
            impact = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                idx2 = np.arange(s2, e2 + 1)
                impact[idx2] += a[k2i - idx2]

            # Sort by impact (low impact first), exclude src
            sorted_idx = np.argsort(impact)
            sorted_idx = sorted_idx[sorted_idx != src]

            found = False
            for dst in sorted_idx[:8]:
                if found:
                    break
                for frac in [0.003, 0.007, 0.015, 0.005, 0.01, 0.02]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

        if not improved:
            stall += 1
            if stall > 50:
                # Larger random moves
                n_moves = rng.integers(2, 5)
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            break

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord it={it} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def pocs_quick(a0, time_limit, rng_seed=0, n_iters=50000):
    """Quick POCS for initialization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    fft_size = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    A_spec = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
    A_full = np.zeros(fft_size, dtype=np.complex128)
    A_full[:fft_size // 2 + 1] = A_spec
    for i in range(1, fft_size // 2):
        A_full[fft_size - i] = np.conj(A_spec[i])
    a = np.fft.irfft(A_full, fft_size)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = norm_sum1(a0)
    else:
        a = a / s

    M = fast_autocorr(a).max()
    if M < best_m:
        best_m = M
        best_a = a.copy()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        target_mag = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
        alpha_blend = 0.15 + 0.4 * min(1.0, it / 200.0)
        new_mag = (1 - alpha_blend) * mag + alpha_blend * target_mag
        A_new = new_mag * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_v_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * (1 - x ** 2)
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.1) / width) ** 2) + np.exp(-((x - 0.9) / width) ** 2) + 0.003
    return norm_sum1(a)


def make_trapezoid(n, flat_frac=0.3):
    a = np.zeros(n)
    ramp = int(n * (1 - flat_frac) / 2)
    a[:ramp] = np.linspace(0.3, 1.0, ramp)
    a[ramp:n - ramp] = 1.0
    a[n - ramp:] = np.linspace(1.0, 0.3, ramp)
    return norm_sum1(a)


def spectral_init(n, n_modes=25, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = local_eval(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 6200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: POCS Initialization (60s) ===
    print(f"\n=== Phase 1: POCS Init (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 60.0)
    if t1 > 20:
        a_pocs, _ = pocs_quick(best_seq, t1, rng_seed=7)
        try_update(a_pocs, "pocs")

    # === Phase 2: Nesterov Smooth Optimization (400s) ===
    print(f"\n=== Phase 2: Nesterov Smooth (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.22, 400.0)
    if t2 > 50:
        # Run with different temperature schedules
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_smooth_optimize(a_in, t2, verbose=True,
                                            beta_init=5.0, beta_final=200.0,
                                            rng_seed=42)
        try_update(a_out, "nesterov1")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t2b = min(remaining * 0.1, 100.0)
            if t2b > 30:
                a_in = norm_sum1(best_seq)
                a_out, _ = nesterov_smooth_optimize(a_in, t2b, verbose=False,
                                                    beta_init=3.0, beta_final=100.0,
                                                    rng_seed=99)
                try_update(a_out, "nesterov2")

    # === Phase 3: LP-FW Main (700s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW Main (n={n_best}) ===")
    t3 = min(remaining * 0.38, 700.0)
    if t3 > 60 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(300, 2 * n_best - 1)
        top_K_max = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe_v2(a_in, t3, solver, verbose=True,
                                      top_K_init=top_K_init,
                                      top_K_max=top_K_max,
                                      line_search=True)
        try_update(a_out, "lp_main")

    # === Phase 4: Peak Reduction (250s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Peak Reduction (n={n_best}) ===")
    t4 = min(remaining * 0.14, 250.0)
    if t4 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_reduction_direct(a_in, t4, verbose=True, rng_seed=77)
        try_update(a_out, "peakred")

    # === Phase 5: LP-FW Round 2 (300s) ===
    remaining = time_budget - (time.monotonic() - t_global)
