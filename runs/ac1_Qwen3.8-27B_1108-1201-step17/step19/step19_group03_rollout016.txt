    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, ac, top_k=50, beta=4.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def mirror_descent(a0, time_limit, top_k=50, beta=4.0, c0=3.0,
                   restart_every=5000, verbose=False, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every // 3:
            cur_beta = max(cur_beta * 0.95, 2.0)
            cur_top_k = min(cur_top_k + 3, 150)

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 200)
            cur_beta = max(cur_beta - 0.5, 2.0)

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def pocs_spectral(a0, time_limit, n_trials=3, verbose=False, rng_seed=0):
    n = len(a0)
    best_a = norm_sum1(a0)
    best_m = fast_autocorr(best_a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit - 10:
            break
        t_trial = min((time_limit - (time.monotonic() - t0)) / max(1, n_trials - trial), 60.0)
        t_start = time.monotonic()

        rng = np.random.default_rng(rng_seed + trial)
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        A_spec = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
        A_full = np.zeros(fft_size, dtype=np.complex128)
        A_full[:fft_size // 2 + 1] = A_spec
        for i in range(1, fft_size // 2):
            A_full[fft_size - i] = np.conj(A_spec[i])
        a = np.fft.irfft(A_full, fft_size)[:n]
        a = np.maximum(a, 1e-15)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s

        for it in range(200000):
            if time.monotonic() - t_start > t_trial:
                break

            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            target_mag = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            alpha_blend = 0.05 + 0.3 * min(1.0, it / 500.0)
            new_mag = (1 - alpha_blend) * mag + alpha_blend * target_mag
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]

            a_new = np.maximum(a_new, 1e-15)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s

            M_new = fast_autocorr(a_new).max()
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = a_new.copy()
                if verbose:
                    J = 2.0 * n * best_m
                    print(f"    POCS trial={trial} it={it}: J={J:.6f}")
            a = a_new

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K_init=200, top_K_max=None):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = min(2 * n - 1, 3000)
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0), 0.5, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1

        if iter_count % 10 == 0:
            K = min(K + 30, top_K_max)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.01
    return norm_sum1(a)


def make_cosine(n, freq=1):
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x)
    a = a ** 2
    return norm_sum1(a)


def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.1 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200
    n_top = 15

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-contribution destination
            dest_sens = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                dest_sens[s2:e2+1] += 1.0

            candidates = np.where(dest_sens < dest_sens.max() * 0.5)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            for dst in candidates[:5]:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try uniform first to establish baseline
    n_uniform = 6200
    a_uni = np.ones(n_uniform)
    try_update(a_uni, "uniform")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            try_update(parent, "parent")
            print(f"Parent: n={len(parent)}, score={best_score:.6f}")
    except Exception as e:
        print(f"No parent: {e}")

    if best_seq is None:
        n_base = 6200
        best_seq = np.ones(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default uniform: score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting n={n0}, best={best_score:.6f}")

    # === Phase 1: Try different lengths with uniform ===
    print("\n=== Phase 1: Length exploration (uniform) ===")
    for n_try in [4000, 5000, 5500, 6200, 7000, 8000, 10000]:
        a_u = np.ones(n_try)
        try_update(a_u, f"uni_n{n_try}")

    # === Phase 2: POCS Spectral ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: POCS Spectral (n={n0}) ===")
    t2 = min(remaining * 0.1, 120.0)
    if t2 > 20:
        a_out, _ = pocs_spectral(best_seq, t2, n_trials=3, verbose=True, rng_seed=7)
        try_update(a_out, "pocs")

    # === Phase 3: Mirror Descent from multiple initializations ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Mirror Descent (n={n0}) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 60:
        inits = []
        labels = []

        inits.append(norm_sum1(best_seq))
        labels.append("best")
        inits.append(make_u_shape(n0, sharpness=1.0))
        labels.append("u1")
        inits.append(make_u_shape(n0, sharpness=3.0))
        labels.append("u3")
        inits.append(make_bimodal(n0, width=0.2))
        labels.append("bim")
        inits.append(make_cosine(n0, freq=1))
        labels.append("cos")
        inits.append(spectral_init(n0, n_modes=30, seed=42))
        labels.append("spec")

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        t_per = max(15.0, t3 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  MD init '{lbl}' (t_avail={t_avail:.0f}s)...")

            try_update(a_out, f"md_{lbl}")

    # === Phase 4: LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)

    print(f"\n=== Phase 4: LP-FW (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)

                                    top_K_init=top_K_init, top_K_max=top_K_max)

        try_update(a_out, "md_final")

    # === Phase 7: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Coordinate (n={n_best}) ===")
    t7 = min(remaining * 0.08, 100.0)
    if t7 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t7, verbose=False, rng_seed=456)
        try_update(a_out, "coord")

    # === Phase 8: Length search for LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Length search ===")
    for new_n in [2000, 2500, 4000, 5000]:
        remaining = time_budget - (time.monotonic() - t_global)

                                        top_K_init=top_K_init, top_K_max=top_K_max)
            try_update(a_out, f"lp_n{new_n}")

    # === Phase 9: Final LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40 and n_best <= 5000:
        print(f"\n=== Phase 9: Final LP-FW (n={n_best}) ===")
        t9 = min(remaining * 0.5, 200.0)
        a_in = norm_sum1(best_seq)

                                    top_K_init=top_K_init, top_K_max=top_K_max)

        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")

    return [float(x) for x in best_seq]
