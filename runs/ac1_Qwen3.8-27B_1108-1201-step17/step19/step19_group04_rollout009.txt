"""
Search for non-negative step-function heights minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based Frank-Wolfe with adaptive active sets, exponential-weight
mirror descent, parametric transforms, and coordinate polish.
Target: beat 1.5027.

import math
import time
import numpy as np

try:
    from scipy.optimize import linprog
    HAVE_LP = True
except ImportError:
    HAVE_LP = False


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a) / a.size
    return a / s


def self_conv(a):
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]


def conv_ab(a, b):
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:m]


def objective_norm(a):
    n = a.size
    if n <= 1:
        return 2.0
    q = self_conv(a)
    return float(2.0 * n * q.max())

def quad_at_alpha(alpha, q0, c, qd):
    a1 = 1.0 - alpha
    vals = (a1 * a1) * q0 + (2.0 * alpha * a1) * c + (alpha * alpha) * qd
    return float(np.max(vals))


def golden_min(q0, c, qd, lo, hi, iters=38):
    phi = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - phi * (hi - lo)
    d1 = lo + phi * (hi - lo)
    f1 = quad_at_alpha(c1, q0, c, qd)
    f2 = quad_at_alpha(d1, q0, c, qd)
    for _ in range(iters):
        if f1 < f2:
            hi = d1
            d1 = c1
            f2 = f1
            c1 = hi - phi * (hi - lo)
            f1 = quad_at_alpha(c1, q0, c, qd)
        else:
            lo = c1
            c1 = d1
            f1 = f2
            d1 = lo + phi * (hi - lo)
            f2 = quad_at_alpha(d1, q0, c, qd)
    return 0.5 * (lo + hi)


def line_search(a, d, q0, T_hint=None):
    M0 = float(np.max(q0))
    if float(np.dot(a, d)) > 1.0 - 1e-8:
        return a.copy(), M0
    c = conv_ab(a, d)
    qd = self_conv(d)
    best_alpha = 0.0
    best_M = M0
    candidates = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
    if T_hint is not None and np.isfinite(T_hint) and T_hint < 0 and M0 > 1e-15:
        t1 = min(0.35, max(0.005, 0.5 * (-T_hint) / M0))
        candidates.append(max(0.005, 0.5 * t1))
        candidates.append(min(0.95, 1.5 * t1))
    for al in sorted(set(candidates)):
        al = float(min(1.0, max(0.0, al)))
        v = quad_at_alpha(al, q0, c, qd)
        if v < best_M - 1e-14:
            best_M = v
            best_alpha = al
    if best_alpha <= 0 or best_M >= M0 - 1e-14:
        return a.copy(), M0
    al2 = golden_min(q0, c, qd, 0.0, 1.0, 35)
    v2 = quad_at_alpha(al2, q0, c, qd)
    if v2 < best_M:
        best_M = v2
        best_alpha = al2
    new = np.maximum((1.0 - best_alpha) * a + best_alpha * d, 0.0)
    s = float(new.sum())
    if s <= 1e-15:
        return a.copy(), M0
    return new / s, best_M

def solve_lp(y, active_idx, q_vals):
    if not HAVE_LP:
        return None, None
    n = y.size
    K = len(active_idx)
    if K == 0:
        return None, None
    if K * (n + 1) > 15_000_000:
        return None, None
    try:
        A = np.zeros((K, n + 1))
    except MemoryError:
        return None, None
    yy = 2.0 * y
    for r in range(K):
        k = int(active_idx[r])
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)

            seg_end = k - i0 + 1
            A[r, i0:i1] = yy[seg_start:seg_end][::-1]

                      options={'presolve': True, 'time_limit': 150})
    except Exception:
        return None, None

    x = getattr(res, 'x', None)

    s = float(g.sum())
    if s <= 1e-10:
        return None, None

def mirror_dir(a, q, focus):
    n = a.size
    m = 2 * n - 1
    M = float(q.max())
    if M <= 1e-15:
        return a.copy()
    thr = 0.80 * M
    w = np.zeros(m)
    mask = q >= thr
    w[mask] = (q[mask] / M) ** 2
    ws = float(w.sum())
    if ws <= 1e-15:
        kp = int(np.argmax(q))
        if 0 <= kp < n:
            s = np.zeros(n)
            for i in range(n):
                j = kp - i
                if 0 <= j < n:
                    s[i] = a[j]
        else:
            return a.copy()
    else:
        w /= ws
        a_rev = a[::-1].copy()
        size = next_pow2(m + n - 1)
        fw = np.fft.rfft(w, size)
        fra = np.fft.rfft(a_rev, size)
        conv = np.fft.irfft(fw * fra, size)
        s = conv[n - 1: n - 1 + n].copy()
        s = np.maximum(s, 0.0)
    s_max = float(s.max())
    if s_max <= 1e-15:
        return a.copy()
    s /= s_max
    move = -focus * s
    move -= move.max()
    move = np.clip(move, -50.0, 50.0)
    d = a * np.exp(move)
    ds = float(d.sum())
    if ds <= 1e-15:
        return a.copy()
    return d / ds
