def lp_direction(f, M):
    """
    Find the direction g that maximizes sum(g) subject to
    (f * g)[k] <= M for all k, g >= 0.
    
    f: current sequence (non-negative)
    M: max autocorrelation of f

    size = 2 * n - 1

    # A[k, i] = f[k-i] if 0 <= k-i < n, else 0

def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def softmax_gradient(a, ac, beta, top_k=None):
    """
    Compute gradient of log-sum-exp(β*c) with respect to a.
    Uses top-K lags for efficiency.
    """
    n = len(a)
    size = len(ac)
    
    if top_k is not None and top_k < size:
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = ac[top_idx]
    else:
        top_idx = np.arange(size)
        top_vals = ac.copy()
    
    # Soft-max weights
    max_v = top_vals.max()
    expo = beta * (top_vals - max_v)
    expo = np.clip(expo, -50, 50)
    weights = np.exp(expo)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        weights = weights / w_sum
    
    # Compute gradient
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        # ∂c[k]/∂a[j] = 2*a[k-j] if 0 <= k-j < n
        j_lo = max(0, k_int - n + 1)
        j_hi = min(n, k_int + 1)
        if j_lo >= j_hi:
            continue
        j = np.arange(j_lo, j_hi)
        Kj = k_int - j
        grad[j] += 2.0 * w * a[Kj]
    
    return grad

def lp_direction_finding(a, time_per_lp=5.0, top_k_lp=100, verbose=True):
    """
    Use LP to find a direction that reduces max autocorrelation.
    Solves: max sum(g) s.t. (a*g)[k] <= M for all top-k lags, g >= 0.
    """
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    size = 2 * n - 1
    
    # Select top-K lags for constraints
    k = min(top_k_lp, size)
    top_idx = np.argsort(ac)[::-1][:k]
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for row_k in top_idx:
        ki = int(row_k)
        for i in range(max(0, ki - n + 1), min(n, ki + 1)):
            rows.append(len(rows))  # row index in reduced matrix
            cols.append(i)
            vals.append(a[ki - i])
    
    # Actually, let me use proper row indices
    rows = []
    cols = []
    vals = []
    for idx, row_k in enumerate(top_idx):
        ki = int(row_k)
        for i in range(max(0, ki - n + 1), min(n, ki + 1)):
            rows.append(idx)
            cols.append(i)
            vals.append(a[ki - i])
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(k, n))
    
    # Solve LP: max sum(g) subject to A@g <= M, g >= 0
    c = -np.ones(n)
    b_ub = np.full(k, M)
    
    t_start = time.monotonic()
    try:
        result = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        t_elapsed = time.monotonic() - t_start
    except Exception as e:
        if verbose:
            print(f"      LP failed: {e}")
        return None
    
    if not result.success:
        if verbose:
            print(f"      LP not successful: {result.message}")
        return None
    
    g = result.x
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None
    
    if verbose:
        print(f"      LP solved in {t_elapsed:.2f}s, sum(g)={g_sum:.4f}")
    
    return g

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length using smooth interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    # Use cubic interpolation for smoothness
    f_int = np.interp(x_new, x_old, a_norm, kind='cubic')
    f_int = np.maximum(f_int, 0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss

        nonlocal best_seq, best_score
        if a is None:
            return

        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = score_from_ac(a_n)

            best_seq = a_n.copy()
