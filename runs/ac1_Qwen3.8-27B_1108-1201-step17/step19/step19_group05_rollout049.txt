"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Frank-Wolfe (conditional gradient)
method with LP-based direction finding, combined with spectral flattening and
coordinate descent for polishing.

Key insight: The ramp a_i = i+1 gives J -> 4/3 asymptotically. We use the

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import scipy.sparse as sp


def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    return result


def compute_J(a):
    """Compute J = 2*n*max_b/(sum_a)^2."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-15:
        return float('inf')
    return float(2 * n * max_b / (sum_a ** 2))


def make_ramp(n, alpha=1.0):
    """Create normalized ramp (i+1)^alpha."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** alpha
    s = a.sum()
    return a / s


def make_parent_ramp(n):
    """Create a ramp that matches the parent's structure."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x.astype(np.float64)
    return a / a.sum()


def solve_lp_direction(f, M, S, time_per_lp=5.0):
    """
    Solve the LP:
    maximize sum g
    subject to:
      (f*g)_k <= M for k = 0, ..., 2n-2
      sum g = S
      g >= 0
    
    Returns the optimal g, or None if the LP fails.
    """
    n = len(f)
    # Constraint: (f*g)_k <= M for each k
    # (f*g)_k = sum_{j: 0<=k-j<n} f_j * g_{k-j}
    # For each k, the constraint is: sum_j A[k,j] * g_j <= M
    # where A[k,j] = f_j if 0 <= k-j < n, else 0.
    
    # Build the constraint matrix A (sparse)
    # Row k, column j: A[k,j] = f_{k-j} if 0 <= k-j <= n-1, else 0
    # i.e., A[k,j] = f_{k-j} if max(0, k-n+1) <= j <= min(n-1, k), else 0
    
    # Actually, let me re-derive. (f*g)_k = sum_i f[i] * g[k-i]
    # For this to be non-zero, we need 0 <= k-i < n, i.e., k-n+1 <= i <= k.
    # And 0 <= i < n.
    # So the valid range of i is [max(0, k-n+1), min(n-1, k)].
    # And j = k-i, so j ranges over [k - min(n-1,k), k - max(0, k-n+1)] = [max(0, k-n+1), min(n-1, k)].
    # Wait, j = k - i, so when i = max(0, k-n+1), j = k - max(0, k-n+1) = min(k, n-1).
    # And when i = min(n-1, k), j = k - min(n-1, k) = max(0, k-n+1).
    # So j ranges from max(0, k-n+1) to min(n-1, k).
    # And A[k, j] = f[k-j] (since i = k-j).
    
    # Let me just build it row by row.
    rows = []
    cols = []
    data = []
    
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo > j_hi:
            continue
        for j in range(j_lo, j_hi + 1):
            i = k - j
            # A[k, j] = f[i] = f[k-j]
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    
    # Equality constraint: sum g = S
    A_eq = np.ones((1, n))
    b_eq = np.array([S])
    
    bounds = [(0, None)] * n
    
    # Objective: maximize sum g, so minimize -sum g
    c = -np.ones(n)
    
    try:
        t_start = time.monotonic()
        res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                      method='highs', options={'time_limit': time_per_lp})
        if res.success:
            return res.x
        else:
            return None
    except Exception:
        return None


def frank_wolfe(f0, time_limit=100.0, verbose=False, n_lp_calls=50):
    """
    Frank-Wolfe (conditional gradient) method.
    f0: initial point (will be normalized to sum=1)
    """
    n = len(f0)
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    S = f.sum()
    if S < 1e-15:
        return f0
    f = f / S  # normalize to sum = 1
    
    best_f = f.copy()
    best_M = fast_autocorr(f).max()
    t0 = time.monotonic()
    
    lp_count = 0
    grad_count = 0
    
    for t in range(1, 100000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        b = fast_autocorr(f)
        M = b.max()
        
        # Check improvement
        if M < best_M - 1e-15:
            best_M = M
            best_f = f.copy()
        
        # Every few iterations, use LP; otherwise use greedy (subgradient)
        use_lp = (t % 20 == 1) and lp_count < n_lp_calls
        
        if use_lp:
            t_lp = min(10.0, (time_limit - elapsed) / max(1, n_lp_calls - lp_count))
            if t_lp < 0.5:
                use_lp = False
        
        if use_lp:
            g = solve_lp_direction(f, M, 1.0, time_per_lp=t_lp)
            lp_count += 1
            if g is None or not np.all(np.isfinite(g)):
                g = None
        
        if g is None:
            # Greedy: pick the coordinate with minimum subgradient
            # Subgradient: g_j = f[k*-j] (if valid) + f[j-k*] (if valid)
            k_star = int(np.argmax(b))
            grad = np.zeros(n)
            j_lo = max(0, k_star - n + 1)
            j_hi = min(n - 1, k_star)
            for j in range(j_lo, j_hi + 1):
                i = k_star - j
                if 0 <= i < n:
                    grad[j] += f[i]
                # Also check the other direction
                i2 = j - k_star
                if 0 <= i2 < n:
                    grad[j] += f[i2]
            
            # Pick the coordinate with minimum gradient (and f_j > 0 or just minimum)
            g = np.zeros(n)
            g[np.argmin(grad)] = 1.0
            grad_count += 1
        
        # Frank-Wolfe step
        gamma = 2.0 / (t + 2.0)
        f_new = (1 - gamma) * f + gamma * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            continue
        f_new = f_new / s_new
        
        M_new = fast_autocorr(f_new).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_f = f_new.copy()
        
        f = f_new
        
        if verbose and t % 100 == 0:
            J = 2 * n * best_M
            print(f"    FW t={t} ({elapsed:.0f}s): J={J:.6f}, M={best_M:.8f}, lp={lp_count}")
    
    return best_f
