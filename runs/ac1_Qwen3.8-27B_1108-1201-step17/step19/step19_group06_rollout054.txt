    for c in [0.5]:
        for sig in [0.15, 0.25, 0.35]:
            candidates.append(np.exp(-((x - c) ** 2) / (2 * sig ** 2)))

    # Bimodal
    for sig in [0.15, 0.25]:
        candidates.append(np.exp(-((x - 0.25) ** 2) / (2 * sig ** 2)) +
                          np.exp(-((x - 0.75) ** 2) / (2 * sig ** 2)))

    # Sine power
    for f in [1, 2]:
        for p in [1, 2, 3]:
            candidates.append(np.sin(np.pi * f * x) ** p)

    # Power laws
    for p in [0.5, 1.0, 1.5]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)

    # Linear
    candidates.append(x)
    candidates.append(1.0 - x)

    # Truncated normal
    mu = 0.5
    for sig in [0.15, 0.2, 0.3]:
        candidates.append(np.exp(-((x - mu) ** 2) / (2 * sig ** 2)))

    # Evaluate and pick best
    best_cand = np.ones(n)
    best_val = float('inf')
    for c in candidates:
        c = _norm_nonneg(c)
        v = _score(c)
        if v < best_val:
            best_val = v
            best_cand = c.copy()

    return best_cand


def _mirror_descent(a0, time_limit, rng, lr_init=0.15, beta_init=4.0,
                    K_init=30, verbose=True, min_lr=0.005):
    """
    Mirror descent (multiplicative updates) with soft-max gradient.
    Adaptive beta, K, and learning rate.
    """
    n = len(a0)
    a = _norm_nonneg(a0.copy())
    best_a = a.copy()
    best_m = _autocorr(a).max()

    t0 = time.monotonic()
    lr = lr_init
    beta = beta_init
    K = K_init
    stall = 0
    stall_limit = 3000
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        # Compute autocorrelation
        ac = _autocorr(a)
        ac_max = ac.max()

        # Check improvement
        if ac_max < best_m - 1e-14:
            best_m = ac_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Get top K lags and soft-max weights
        if K >= 2 * n - 1:
            top_idx = np.arange(2 * n - 1)
            top_vals = ac
        else:
            part = np.argpartition(ac, -K)[-K:]
            top_idx = part
            top_vals = ac[part]

        # Soft-max weights
        shifted = beta * (top_vals - ac_max)
        shifted = np.clip(shifted, -60.0, 0.0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(w) / len(w)
        else:
            w = w / w_sum

        # Compute gradient
        grad = _grad_from_lags(a, top_idx, w)

        # Multiplicative (mirror) update
        log_a = np.log(np.maximum(a, 1e-30))
        log_a = log_a - lr * grad / max(grad.max(), 1e-15)
        log_a = log_a - log_a.max()  # numerical stability
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        # Validate
        ac_new = _autocorr(a_new)
        m_new = ac_new.max()

        if m_new < best_m:
            a = a_new
        else:
            # If no improvement, try smaller step
            a_half = _norm_nonneg(0.5 * a + 0.5 * a_new)
            ac_half = _autocorr(a_half)
            if ac_half.max() < ac_max:
                a = a_half
            # else keep a, reduce lr

        # Adaptive parameters
        if stall > stall_limit:
            # Restart with perturbation
            if verbose:
                J = 2.0 * n * best_m
                print(f"      MD restart iter={iter_count} ({time.monotonic()-t0:.0f}s): J={J:.6f}")
            pert_type = int(rng.integers(4))
            if pert_type == 0:
                sig = float(rng.uniform(0.05, 0.2))
                a = _norm_nonneg(best_a * (1.0 + sig * rng.normal(0, 1, n)))
            elif pert_type == 1:
                # Spectral perturbation
                A = np.fft.rfft(a, 1 << (2 * n - 1 - 1).bit_length())
                phase = rng.normal(0, 0.1, len(A))
                A_pert = A * np.exp(1j * phase)
                a = _norm_nonneg(np.fft.irfft(A_pert, 1 << (2 * n - 1 - 1).bit_length())[:n])
            elif pert_type == 2:
                # Shift
                shift = int(rng.integers(1, max(2, n // 10)))
                a = _norm_nonneg(np.roll(a, shift))
            else:
                # Blend with parametric
                x = np.linspace(0, 1, n)
                c = float(rng.uniform(0.2, 0.8))
                sig = float(rng.uniform(0.1, 0.4))
                pert = _norm_nonneg(np.exp(-((x - c) ** 2) / (2 * sig ** 2)))
                a = _norm_nonneg(0.7 * best_a + 0.3 * pert)

            stall = 0
            lr = max(lr * 0.7, min_lr)
            beta = max(beta - 1.0, 2.0)
            K = max(K - 10, 15)
        else:
            # Gradually sharpen and reduce lr
            if iter_count % 200 == 0:
                lr *= 0.995
                beta = min(beta + 0.02, 30.0)
                K = min(K + 1, 150)

    J = 2.0 * n * best_m
    if verbose:
        print(f"      MD done ({time.monotonic()-t0:.0f}s): J={J:.6f}, best_m={best_m:.8f}")
    return best_a


def _spectral_shape(a0, time_limit, rng, iterations=2000):
    """
    Shape the spectrum to reduce peak.
    In frequency domain, flatten |A|^2 while maintaining real, non-negative time signal.
    """
    n = len(a0)
    a = _norm_nonneg(a0.copy())
    best_a = a.copy()
    best_m = _autocorr(a).max()
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    t0 = time.monotonic()

    for it in range(iterations):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        magnitude = np.abs(A)
        phase = np.angle(A)

        # Target: flatten magnitude spectrum
        mag_mean = magnitude.mean()
        mag_std = magnitude.std()

        if mag_std < 1e-15:
            break

        # Push magnitude toward mean
        alpha = 0.1
        mag_new = magnitude * (1.0 - alpha) + mag_mean * alpha

        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = _norm_nonneg(a_new)

        m_new = _autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        # else don't update (keep a)

    J = 2.0 * n * best_m
    print(f"      Spectral done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a

    """

    a = _norm_nonneg(a0.copy())

    stall = 0

        M_curr = ac.max()

            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break

        # Get top lags
        n_top = min(20, 2 * n - 1)
        top_lags = np.arg
