def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(a, tau):
    """Compute gradient of smoothed log-sum-exp of convolution."""
    n = len(a)
    c = fast_autocorr(a)
    c_shift = c / tau
    c_shift -= c_shift.max()
    exp_c = np.exp(c_shift)
    p = exp_c / exp_c.sum()
    obj = tau * (np.log(exp_c.sum()) + c_shift.max())
    grad = 2.0 * sp_correlate(p, a, mode='valid')
    return obj, grad, c.max()

                         rng_seed=0):

    a = norm_sum1(a0)

    best_m = fast_autocorr(a).max()

    it = 0
    stall = 0
    max_stall = 500
    momentum = momentum_init
    
    while time.monotonic() - t0 < time_limit:

        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:

        a_new = y - step_frac * grad / grad_norm
        a_new = project_simplex(a_new)
        
        # Evaluate

        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
        else:

    return np.maximum(v - theta, 0.0)


def smooth_obj_grad(a, tau):
    """Compute smoothed log-sum-exp objective and gradient."""
    n = len(a)
    c = fast_autocorr(a)
    c_shift = c / tau
    c_shift -= c_shift.max()
    exp_c = np.exp(c_shift)
    p = exp_c / exp_c.sum()
    obj = tau * (np.log(exp_c.sum()) + c_shift.max())
    grad = 2.0 * sp_correlate(p, a, mode='valid')
    return obj, grad, c.max()


def smoothed_grad_descent(a0, time_limit, tau_init=2.0, tau_final=0.3,
                          step_frac=0.08, verbose=False, rng_seed=0):
    """Nesterov-accelerated projected gradient on smoothed log-sum-exp."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    a_prev = a.copy()
    it = 0
    stall = 0
    max_stall = 400
    momentum = 0.85
    cur_step = step_frac

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        tau = tau_init * (tau_final / tau_init) ** (progress ** 0.3)

        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y / s_y

        obj, grad, m_curr = smooth_obj_grad(y, tau)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        a_new = y - cur_step * grad
        a_new = project_simplex(a_new)
        m_new = fast_autocorr(a_new).max()

        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            cur_step = min(cur_step * 1.02, 0.3)
        else:
            stall += 1
            if stall > 50:
                cur_step = max(cur_step * 0.85, 0.01)
            if stall > max_stall:
                pert_std = 0.03
                pert = rng.normal(0, pert_std, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-10)
                a_r = a_r / a_r.sum()
                a = a_r
                a_prev = a_r.copy()
                stall = max_stall // 2
                cur_step = step_frac

        momentum = max(0.5, 0.85 - 0.05 * (it / 20000.0))
        a_prev = a
        a = a_new

        if verbose and it % 1000 == 0:
            J = 2.0 * n * best_m
            print(f"    SGD it={it} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.3f}, "
                  f"step={cur_step:.4f}, stall={stall}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K_init=500,
                   top_K_max=None):
    """LP-based Frank-Wolfe: maximize sum(b) s.t. (a*b)[k] <= M, b >= 0."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 100, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = fast_autocorr(a).max()
        for alpha in [best_alpha, min(1.0, best_alpha * 2.0), 0.5, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1
        if iter_count % 15 == 0:
            K = min(K + 50, top_K_max)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def spectral_equalize(a0, time_limit, verbose=False, rng_seed=10):
    """POCS-style: alternate flat-spectrum projection and non-negativity."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    stall = 0
    max_stall = 300

    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Frequency domain: push toward flat magnitude
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        mag_mean = np.mean(mag[mag > 1e-12]) if np.any(mag > 1e-12) else 1.0
        blend = 0.15 + 0.3 * min(1.0, it / 1000.0)
        new_mag = (1 - blend) * mag + blend * mag_mean
        A_new = new_mag * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        # Time domain: enforce non-negativity and normalization
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new

        if stall > max_stall:
            pert = rng.normal(0, 0.02, n) * best_a
            a = np.maximum(best_a + pert, 1e-10)
            a = a / a.sum()
            stall = max_stall // 2

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SpecEq it={it} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def targeted_coord_descent(a0, time_limit, verbose=False, rng_seed=0):
    """Coordinate descent targeting the worst autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    n_top = 15

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find destination: low-sensitivity position
            sens = np.zeros(n)
            for kk in top_k[:8]:
                kki = int(kk)
                ss = max(0, kki - n + 1)
                ee = min(n - 1, kki)
                ii = np.arange(ss, ee + 1)
                sens[ii] += a[kki - ii]

            low_sens = np.where(sens < np.median(sens))[0]
            if len(low_sens) == 0:
                low_sens = np.arange(n)
            dst_candidates = low_sens[np.argsort(sens[low_sens])[:5]]

            found = False
            for dst in dst_candidates:
                if dst == src or found:
                    continue
                for frac in [0.02, 0.01, 0.03, 0.05, 0.08]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    Mt = fast_autocorr(a_trial).max()
                    if Mt < best_m - 1e-16:
                        a = a_trial
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

        if not improved:
            stall += 1
            if stall > 20:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    Mn = fast_autocorr(a).max()
                    if Mn < best_m:
                        best_m = Mn
                        best_a = a.copy()
                        stall = 0
                    elif stall > 100:
                        break
        else:
            stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_w_shape(n, num_peaks=3, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.01
    for i in range(num_peaks):
        center = (i + 0.5) / num_peaks
        a += np.exp(-((x - center) / width) ** 2)
    a += np.exp(-(x / width) ** 2) * 0.5
    a += np.exp(-((x - 1) / width) ** 2) * 0.5
    return norm_sum1(a)


def make_periodic(n, freq=4, alpha=0.4):
    x = np.arange(n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_cosine_sq(n):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def pocs_init(n, seed=0, blend=0.3):
    """Generate initial sequence via flat-spectrum + non-negativity projection."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    A = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
    A_full = np.zeros(fft_size, dtype=np.complex128)
    A_full[:fft_size // 2 + 1] = A
    for i in range(1, fft_size // 2):
        A_full[fft_size - i] = np.conj(A[i])
    a = np.fft.irfft(A_full, fft_size)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    return a


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator: smoothed GD + LP-FW + spectral + coord + length exploration."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(arr, label):
        nonlocal best_seq, best_score
        if arr is None:
            return
        arr = np.maximum(np.asarray(arr, dtype=np.float64), 0.0)
        s = arr.sum()
        if s < 0.01:
            return
        arr_norm = arr / s
        sc = evaluate_safe(arr_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(arr_norm)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 6200
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Smoothed gradient descent on parent ===
    print(f"\n=== Phase 1: Smoothed GD on parent (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 60:
        a_out, _ = smoothed_grad_descent(best_seq, t1, tau_init=3.0, tau_final=0.2,
                                          step_frac=0.06, verbose=True, rng_seed=1)
        try_update(a_out, "sgd_parent")

    # === Phase 2: Smoothed GD with diverse inits ===
    print(f"\n=== Phase 2: Smoothed GD diverse inits (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 60:
        inits = []
        labels = []
        inits.append(make_u_shape(n0, 1.5))
        labels.append("u1.5")
        inits.append(make_w_shape(n0, 4, 0.08))
        labels.append("w4")
        inits.append(make_periodic(n0, 6, 0.3))
        labels.append("per6")
        inits.append(pocs_init(n0, seed=42))
        labels.append("pocs42")
        inits.append(pocs_init(n0, seed=123))
        labels.append("pocs123")

        if best_score < float('inf'):
            inits.append(norm_sum1(best_seq))
            labels.append("best_cur")

        t_per = max(20.0, t2 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 15:
                break
            print(f"  SGD init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = smoothed_grad_descent(init, t_avail, tau_init=2.0, tau_final=0.2,
                                              step_frac=0.06, verbose=False, rng_seed=100 + idx)
            try_update(a_out, f"sgd_{lbl}")

    # === Phase 3: LP-Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30 and n_best <= 7000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(400, 2 * n_best - 1)
        top_K_max = min(4000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True,
                                   top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp_main")

    # === Phase 4: Spectral equalization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Spectral Eq (n={n_best}) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_equalize(a_in, t4, verbose=True, rng_seed=20)
        try_update(a_out, "speceq")

    # === Phase 5: Smoothed GD round 2 on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Smoothed GD r2 (n={n_best}) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smoothed_grad_descent(a_in, t5, tau_init=1.5, tau_final=0.1,
                                          step_frac=0.05, verbose=True, rng_seed=777)
        try_update(a_out, "sgd_r2")

    # === Phase 6: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LP-FW r2 (n={n_best}) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 20 and n_best <= 7000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(800, 2 * n_best - 1)
        top_K_max = min(2 * n_best - 1, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=True,
                                   top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp_r2")

    # === Phase 7: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Coord (n={n_best}) ===")
    t7 = min(remaining * 0.08, 100.0)
    if t7 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = targeted_coord_descent(a_in, t7, verbose=True, rng_seed=555)
        try_update(a_out, "coord")

    # === Phase 8: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Length exploration ===")
    trial_lengths = []
    for delta in [200, -200, 500, -500, 1000, -1000, 1500, -1500]:
        new_n = n_best + delta
        if 1000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t8a = min(remaining * 0.25, 100.0)
        if t8a > 15:
            a_out, _ = smoothed_grad_descent(a_long, t8a, tau_init=2.0, tau_final=0.2,
                                              step_frac=0.06, verbose=False, rng_seed=new_n % 500)
            try_update(a_out, f"sgd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 6000:
            t8b = min(remaining * 0.3, 80.0)
            if t8b > 15:
                a_out, _ = lp_frank_wolfe(norm_sum1(interpolate_sequence(best_seq, new_n)),
                                           t8b, solver, verbose=False,
                                           top_K_init=min(300, 2 * new_n - 1),
                                           top_K_max=min(3000, 2 * new_n - 1))
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 9: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len.best_seq) if best_seq is not None else n0
    print(f"\n=== Phase 9: Final polish (n={n_best}) ===")
    if remaining > 30:
        t9 = min(remaining * 0.35, 120.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = smoothed_grad_descent(a_in, t9, tau_init=1.0, tau_final=0.05,
                                              step_frac=0.04, verbose=False, rng_seed=999)
            try_update(a_out, "sgd_final")

        remaining = time_budget - (time.monotonic() - t_global)
