def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize a to be non-negative with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_topk_subgradient(a, K=10):
    """
    Compute normalized subgradient using top-K lags.
    Returns (gradient, max_val, top_k_indices).
    """
    n = len(a)
    conv = fast_autocorr(a)
    size = 2 * n - 1
    
    # Find top K
    if K >= size:
        top_idx = np.arange(size)
    else:
        top_idx = np.argpartition(conv, -K)[-K:]
        top_idx = top_idx[np.argsort(conv[top_idx])[::-1]]
    
    conv_max = conv[top_idx[0]]
    
    # Compute subgradient: g[j] = sum_{k in top_K} a[k-j] / (K * conv[k])
    # where we only include terms where 0 <= k-j < n
    g = np.zeros(n)
    
    for k in top_idx:
        k_int = int(k)
        # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start < j_end:
            indices = np.arange(j_start, j_end)
            a_vals = a[k_int - indices]
            g[j_start:j_end] += a_vals / (K * conv[k_int])
    
    return g, conv_max, top_idx, conv

def subgrad_mirror_descent(a0, time_limit, K_start=1, K_end=50, eta0=1.0,
                           decay=0.5, verbose=True):
    """
    Subgradient mirror descent in log-space (multiplicative weights).
    """
    n = len(a0)
    theta = np.log(np.maximum(a0, 1e-30))
    theta -= theta.max()
    a = np.exp(theta)
    a = a / a.sum()
    
    theta_prev = theta.copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    t = 0
    stall = 0
    stall_limit = 3000
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        
        # Adaptive K
        progress = min(t / (time_limit * 50), 1.0)
        K = max(K_start, int(K_start + (K_end - K_start) * progress))
        
        # Compute subgradient at current point
        g, cmax, top_idx, conv = compute_topk_subgradient(a, K)
        
        # Step size
        eta = eta0 / (1.0 + decay * t)
        
        # Nesterov: evaluate gradient at extrapolated point
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            theta_y = theta + momentum * (theta - theta_prev)
            a_y = np.exp(theta_y - theta_y.max())
            a_y = a_y / a_y.sum()
            g_y, cmax_y, _, _ = compute_topk_subgradient(a_y, K)
            # Update theta using gradient at y
            theta_new = theta_y - eta * g_y
        else:
            theta_new = theta - eta * g
        
        # Renormalize (maintain sum=1 via soft normalization)
        theta_new -= theta_new.max()
        a_new = np.exp(theta_new)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        # Check improvement
        c_new = fast_autocorr(a_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > stall_limit:
            rng = np.random.default_rng(t)
            pert = rng.normal(0, 0.05, n)
            theta_restart = np.log(np.maximum(best_a + abs(pert) * best_a, 1e-30))
            theta_restart -= theta_restart.max()
            theta = theta_restart
            theta_prev = theta_restart.copy()
            a = np.exp(theta)
            a = a / a.sum()
            stall = 0
            stall_limit = max(stall_limit // 2, 500)
            K = K_start  # Reset K
        
        theta_prev = theta
        theta = theta_new
        a = a_new
        
        if verbose and t % 500 == 0:
            print(f"    SubGrad iter={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, K={K}, stall={stall}")
    
    return best_a, best_m

def projected_subgrad(a0, time_limit, K_start=1, K_end=100, eta0=0.1,
                      verbose=True):
    """
    Subgradient method with Euclidean projection onto simplex.
    """
    n = len(a0)
    a = norm_sum1(a0)
    a_prev = a.copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    t = 0
    stall = 0
    stall_limit = 2000
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        
        progress = min(t / (time_limit * 30), 1.0)
        K = max(K_start, int(K_start + (K_end - K_start) * progress))
        
        g, cmax, top_idx, conv = compute_topk_subgradient(a, K)
        
        eta = eta0 / (1.0 + 0.1 * t)
        
        # Nesterov extrapolation
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            a_y = a + momentum * (a - a_prev)
            a_y = np.maximum(a_y, 0)
            sy = a_y.sum()
            if sy < 1e-15:
                a_y = a.copy()
            else:
                a_y = a_y / sy
            g_y, _, _, _ = compute_topk_subgradient(a_y, K)
            a_new = a_y - eta * g_y
        else:
            a_new = a - eta * g
        
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = project_simplex(a_new)
        
        # Check improvement
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > stall_limit:
            rng = np.random.default_rng(t)
            a_restart = norm_sum1(best_a + rng.dirichlet(np.ones(n) * 10))
            a = a_restart
            a_prev = a_restart.copy()
            stall = 0
            stall_limit = max(stall_limit // 2, 500)
            K = K_start
        
        a_prev = a
        a = a_new
        
        if verbose and t % 500 == 0:
            print(f"    ProjSubGrad iter={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, K={K}, stall={stall}")
    
    return best_a, best_m

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def lp_frank_wolfe(a0, time_limit, solver, initial_K=200, max_K=2000, verbose=True):
    """LP-based Frank-Wolfe for polishing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    init_K = min(initial_K, 2 * n - 1)
    ac_y = fast_autocorr(y)
    constraint_set = set(int(k) for k in np.argsort(ac_y)[::-1][:init_K])
    
    iter_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
        
        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()
        
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # Frank-Wolfe step with Nesterov
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, |S|={len(constraint_set)}")
    
    return best_a, best_m

def parametric_init(n, ref=None):
    """Generate diverse parametric initializations."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')
    
    candidates = []
    
    # U-shaped (large at edges, small in middle) - key insight!
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        candidates.append(x**p + (1-x)**p)
    
    # Inverted (large in middle)
    for p in [0.5, 1.0, 2.0]:
        candidates.append(1 - (2*x - 1)**p)
    
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    
    # Bimodal
    for c in [0.2, 0.3, 0.4]:
        for sigma in [0.05, 0.1, 0.15, 0.2]:
            candidates.append(np.exp(-((x-c)**2)/(2*sigma*sigma)) + np.exp(-((x-1+c)**2)/(2*sigma*sigma)))
    
    # If reference available
    if ref is not None:
        ref_norm = norm_sum1(ref)
        candidates.append(ref_norm)
        for w in [0.3, 0.7]:
            candidates.append(w * ref_norm + (1-w) * norm_sum1(x**2 + (1-x)**2))
    
    for a_cand in candidates:
        a_norm = norm_sum1(a_cand)
        ac = fast_autocorr(a_norm)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_a = a_norm
    
    return best_a, best_val

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fista_softmax(a0, time_limit, beta_start=3.0, beta_end=80.0,
                   eta0=0.8, verbose=False):
    """FISTA with softmax gradient (from previous approach, kept as fallback)."""
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_threshold = 5000
    
    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        progress = min(iter_count / max(time_limit * 100, 1), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        eta = eta0 / (np.sqrt(t) + 1.0)
        
        # Compute softmax-weighted gradient
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        
        A = np.fft.rfft(y, fft_size)
        c = np.fft.irfft(A * A, fft_size)[:size]
        
        top_k = min(100, size)
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        c_top = c[top_idx]
        c_max = c_top.max()
        exp_c = np.zeros(size)
        exp_c[top_idx] = np.exp(beta * (c_top - c_max))
        w = exp_c / exp_c.sum()
        
        a_rev = y[::-1]
        W = np.fft.rfft(w, fft_size)
        A_rev = np.fft.rfft(a_rev, fft_size)
        conv_w = np.fft.irfft(W * A_rev, fft_size)
        grad = 2.0 * conv_w[n - 1:2 * n - 1]
        
        # Mirror descent step
        log_y = np.log(np.maximum(y, 1e-30)) - eta * grad
        log_y -= log_y.max()
        z = np.exp(log_y)
        z = np.maximum(z, 1e-30)
        z = z / z.sum()
        
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = z + momentum * (z - y)
            x_new = np.maximum(x_new, 1e-30)
            x_new = x_new / x_new.sum()
        else:
            x_new = z.copy()
        
        c_new = fast_autocorr(x_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > stall_threshold:
            rng = np.random.default_rng(iter_count)
            pert = rng.normal(0, 0.03, n) * best_x
            x_restart = norm_sum1(best_x + pert)
            x_new = x_restart
            y = x_restart.copy()
            stall = 0
            stall_threshold = max(stall_threshold * 0.7, 1000)
        
        y = x_new
        x = x_new
        iter_count += 1

    return best_x, best_m
