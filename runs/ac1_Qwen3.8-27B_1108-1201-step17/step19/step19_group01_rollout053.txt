    A = np.fft.rfft(a, fft_size)

def subgradient_at_lag(a, k):
    """Compute gradient of g_k(a) = (a*a)[k]."""
    n = len(a)
    sg = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        sg[j] = 2.0 * a[k - j]
    return sg

def subgradient_topk(a, c, top_k, beta=10.0):
    """Compute smoothed subgradient using top-K lags with softmax weights."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    max_c = c[top_idx[0]]
    weights = np.exp(beta * (c[top_idx] - max_c))
    weights /= weights.sum()
    sg = np.zeros(n)
    for w, k in zip(weights, top_idx):
        sg += w * subgradient_at_lag(a, int(k))
    return sg
