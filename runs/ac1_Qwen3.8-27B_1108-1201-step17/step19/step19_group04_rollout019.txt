"""
Improved search for nonnegative step-function heights minimizing
    2*n*max(conv(a,a))/(sum(a))^2.

Key innovations over previous approach:
1. Multi-scale mirror descent with broad-to-narrow focus annealing
2. LP-FW with "ring" active set (top K + spread lags) to prevent new peaks
3. Parabolic refinement after golden-section line search
4. Progressive constraint tightening (start relaxed, tighten over time)
5. Log-parametrization (softmax) for cleaner optimization
6. Diverse starts with different structural shapes
7. Adaptive length search around the previous best
8. Peak-shaving coordinate descent as final polish
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import linprog
    HAVE_LP = True
except Exception:
    HAVE_LP = False


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]


def conv_ab(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]


def objective_norm(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    q = self_conv(a)
    M = float(np.max(q))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * float(a.size) * M / (s * s)


def corr_shift(w, a):
    w = np.asarray(w, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = w.size
    size = next_pow2(L + n - 1)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fw * fa.conj(), size)[:n]
    return np.maximum(c, 0.0)


def weights_from_q_broad(q, focus_width):
    """
    Compute soft weights with a controllable spread.
    focus_width small = focus on peak, large = broad.
    """
    q = np.asarray(q, dtype=np.float64)
    m = float(q.max())
    if m <= 1e-15:
        return np.ones_like(q, dtype=np.float64) / float(q.size)
    scale = max(focus_width, 1e-10)
    vals = (q - m) / scale
    vals = np.clip(vals, -100.0, 0.0)
    e = np.exp(vals)
    s = float(e.sum())
    if s <= 1e-300:
        return np.ones_like(q, dtype=np.float64) / float(q.size)
    return e / s


def quad_max_alpha(alpha, q0, c, qd):
    a1 = 1.0 - alpha
    val = (a1 * a1) * q0 + (2.0 * alpha * a1) * c + (alpha * alpha) * qd
    return float(np.max(val))


def golden_section_max(q0, c, qd, a_lo, a_hi, iters=40):
    phi = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = a_hi - phi * (a_hi - a_lo)
    d1 = a_lo + phi * (a_hi - a_lo)
    f1 = quad_max_alpha(c1, q0, c, qd)
    f2 = quad_max_alpha(d1, q0, c, qd)
    for _ in range(iters):
        if f1 < f2:
            a_hi = d1
            d1 = c1
            f2 = f1
            c1 = a_hi - phi * (a_hi - a_lo)
            f1 = quad_max_alpha(c1, q0, c, qd)
        else:
            a_lo = c1
            c1 = d1
            f1 = f2
            d1 = a_lo + phi * (a_hi - a_lo)
            f2 = quad_max_alpha(d1, q0, c, qd)
    alpha = 0.5 * (a_lo + a_hi)
    return alpha, quad_max_alpha(alpha, q0, c, qd)


def parabolic_refine(q0, c, qd, a1, f1, a2, f2, a3, f3):
    """Fit parabola through 3 points and find its minimum."""
    if a1 == a2 or a2 == a3 or a1 == a3:
        mid = sorted([a1, a2, a3])[1]
        return mid
    A = (f1 - f3) / (a1 - a3) - (f1 - f2) / (a1 - a2)
    B = (f1 - f2) / (a1 - a2)
    if abs(A) < 1e-30:
        return 0.5 * (a1 + a3)
    apex = -B / A
    apex = max(0.0, min(1.0, apex))
    return apex


def line_search_direction(a, d, q0):
    a = np.asarray(a, dtype=np.float64)
    d = np.asarray(d, dtype=np.float64)
    q0 = np.asarray(q0, dtype=np.float64)
    M0 = float(np.max(q0))

    c = conv_ab(a, d)
    qd = self_conv(d)

    best_alpha = 0.0
    best_M = M0

    alphas = [1e-4, 5e-4, 1e-3, 2e-3, 5e-3, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4, 0.6, 0.8, 0.95, 1.0]
    vals = []
    for al in alphas:
        al = float(min(1.0, max(0.0, al)))
        v = quad_max_alpha(al, q0, c, qd)
        vals.append((al, v))
        if v < best_M:
            best_M = v
            best_alpha = al

    if best_M < M0 - 1e-14 and 0.0 < best_alpha < 1.0:
        # Golden section
        alpha_g, val_g = golden_section_max(q0, c, qd, 0.0, 1.0, 38)
        if val_g < best_M:
            best_M = val_g
            best_alpha = alpha_g

    new = (1.0 - best_alpha) * a + best_alpha * d
    new = np.maximum(new, 0.0)
    s = float(new.sum())
    if s <= 1e-15:
        return a.copy(), M0
    new = new / s
    return new, best_M


def softmax_parametrize(log_b, temperature=1.0):
    """Convert log-space vector to normalized positive vector."""
    log_b = np.asarray(log_b, dtype=np.float64)
    log_b = log_b - np.max(log_b)
    b = np.exp(log_b / max(temperature, 1e-6))
    s = float(b.sum())
    if s <= 1e-15:
        return np.ones_like(log_b, dtype=np.float64) / float(log_b.size)
    return b / s


def mirror_descent_multi_scale(a0, time_limit, rng_seed=42):
    """
    Mirror descent with broad-to-narrow focus annealing.
    Start with broad focus (all lags matter) and gradually narrow to peak.
    """
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 2:
        return a

    best = a.copy()
    best_score = objective_norm(a)

    rng = np.random.default_rng(rng_seed)
    trust = 0.3
    iter_count = 0
    fails = 0

    total_iters_estimate = int(time_limit / 0.05)
    if total_iters_estimate < 10:
        total_iters_estimate = 10

    while time.monotonic() - t0 < time_limit:
        remaining_frac = 1.0 - (time.monotonic() - t0) / time_limit
        # Progress: 0 at start, 1 at end
        progress = 1.0 - remaining_frac

        # Focus width: start broad (all lags), narrow to peak
        # At progress=0: width = M (broad), at progress=1: width = 0.01*M (narrow)
        q = self_conv(a)
        M = float(np.max(q))
        if M <= 1e-15:
            break

        # Anneal focus width
        broad_width = M * 0.5
        narrow_width = max(M * 0.005, 1e-12)
        focus_width = broad_width * (narrow_width / broad_width) ** progress
        focus_width = max(focus_width, narrow_width)

        # Also modulate trust: larger early, smaller late
        trust_base = 0.6 * (1.0 - 0.5 * progress)
        trust = max(trust * (0.85 + 0.15 * trust_base / trust), 0.02)
        trust = min(trust, trust_base * 1.5)

        w = weights_from_q_broad(q, focus_width)
        c = corr_shift(w, a)
        cmax = float(np.max(c))
        cmin = float(np.min(c))
        scale = cmax - cmin
        if scale <= 1e-15:
            fails += 5
            if fails > 50:
                break
            continue

        move = trust * (cmax - c) / scale
        move = np.clip(move, -60.0, 60.0)
        b = a * np.exp(move)
        b = normalize_vec(b)

        score_b = objective_norm(b)
        iter_count += 1

        if score_b < best_score - 1e-13:
            best = b.copy()
            best_score = score_b
            a = b
            fails = 0
        elif score_b < objective_norm(a) - 1e-14:
            a = b
            fails = 0
        else:
            fails += 1
            trust *= 0.8
            if fails > 200:
                # Perturb to escape
                noise = 0.02 * (1.0 + 0.1 * fails / 200.0)
                log_a = np.log(np.maximum(a, 1e-30))
                log_a += noise * rng.normal(size=n)
                a = softmax_parametrize(log_a)
                fails = 0
                trust = 0.3
                if fails > 400:
                    break

    return best


def solve_lp_ring(y, active, q_active, n):
    """
    Solve LP for FW direction with given active set.
    min T s.t. sum g = 1, g >= 0, 2*(y*g)[k] - T <= q_y[k] for active k.
    """
    if not HAVE_LP:
        return None
    active = np.asarray(active, dtype=int)
    q_active = np.asarray(q_active, dtype=np.float64)
    K = active.size
    if K <= 0 or K * (n + 1) > 12_000_000:
        return None

    try:
        A = np.zeros((K, n + 1), dtype=np.float64)
    except Exception:
        return None

    yy = 2.0 * y
    for r, k in enumerate(active):
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            A[r, i0:i1] = yy[start:stop][::-1]

    A[:, n] = -1.0
    b_ub = np.maximum(q_active, 0.0)

    c_vec = np.zeros(n + 1, dtype=np.float64)
    c_vec[n] = 1.0
    Aeq = np.zeros((1, n + 1), dtype=np.float64)
    Aeq[0, :n] = 1.0
    bounds = [(0.0, np.inf)] * n + [(None, None)]

    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, A_eq=Aeq,
                      b_eq=np.array([1.0]), bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:
        return None

    if getattr(res, 'status', 1) != 0:
        return None
    x = getattr(res, 'x', None)
    if x is None:
        return None

    g = np.asarray(x[:n], dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-10:
        return None
    g = g / s
    return g


def ring_active_set(q, K_top, spread_count=3, spread_ratio=2.0):
    """
    Build active set: top K_top lags + spread lags at regular intervals.
    This prevents new peaks from forming at unmonitored positions.
    """
    n_q = len(q)
    order = np.argsort(q)[::-1]
    top = order[:K_top]

    # Add spread lags
    spread = []
    for s in range(1, spread_count + 1):
        lag = int(K_top * spread_ratio * s)
        if lag < n_q:
            # Add a few lags near this position
            start = max(0, lag - 2)
            end = min(n_q, lag + 3)
            for idx in order[start:end]:
                if idx not in set(top.tolist()):
                    spread.append(int(idx))
                    if len(spread) >= K_top // 2:
                        break
        if len(spread) >= K_top // 2:
            break

    combined = list(top) + spread
    # Remove duplicates while preserving order
    seen = set()
    unique = []
    for v in combined:
        if v not in seen:
            seen.add(v)
            unique.append(v)
    return np.array(unique, dtype=int)


def lp_fw_ring(a0, time_limit, verbose=False):

    best = y.copy()

    # Adaptive K based on n and progress

        K_start = 100
        K_max = 600
    elif n > 4000:
        K_start = 150
        K_max = 800
    else:
        K_start = 200
        K_max = 1200

    K = K_start
    use_lp = HAVE_LP
    lp_fails = 0
    no_improve = 0
    iter_no = 0
    last_print = t0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining <= 0.0:
            break

        qy = self_conv(y)
        M_y = float(np.max(qy))
        score_y = 2.0 * n * M_y

        if score_y < best_score - 1e-12:
            best = y.copy()
            best_score = score_y
            no_improve = 0

        # Progressive K increase
        progress = (time.monotonic() - t0) / time_limit
        K_target = int(K_start + (K_max - K_start) * min(1.0, progress * 1.5))
        K = min(K_target, K_max)

        # Ring active set
        act = ring_active_set(qy, K, spread_count=2, spread_ratio=2.0)

        g_dir = None

                    no_improve = 0
        else:
            # Mirror descent step

                no_improve += 1

            K = min(K + 20, K_max)

def peak_shaving(a0, time_limit, rng_seed=777):
    """

    """

    while time.monotonic() - t0 < time_limit and stall < 300:
        q = self_conv(a)
        top_lags = np.argsort(q)[::-1][:min(10, len(q))]

        # Compute contribution scores for each a_i
        scores = np.zeros(n, dtype=np.float64)

            k = int(kk)
            i0 = max(0, k - n + 1)
            i1 = min(n, k + 1)
            if i0 < i1:
                start = k - (i1 - 1)
                stop = k - i0 + 1
                scores[i0:i1] += a[start:stop][::-1]

        # Find high-contributing and low-contributing elements
        src_order = np.argsort(scores)[::-1][:15]
        dst_order = np.argsort(scores)[:15]

        improved = False
        for s_idx in src_order:
            if time.monotonic() - t0 > time_limit:
                break
            if improved:
                break
            d_pos = int(s_idx)
            for j_idx in dst_order:
                j_pos = int(j_idx)
                if d_pos == j_pos:
                    continue
                v = float(a[d_pos])
                if v <= 1e-15:
                    continue
                for frac in (0.01, 0.03, 0.05, 0.1, 0.15, 0.25):
                    amt = v * frac
                    if amt <= 1e-18:
                        continue
                    b = a.copy()
                    b[d_pos] -= amt
                    b[j_pos] += amt
                    b = np.maximum(b, 0.0)
                    ss = float(b.sum())
                    if ss <= 1e-15:
                        continue
                    b = b / ss
                    sc = objective_norm(b)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best = b.copy()
                        a = b
                        improved = True
                        break
                if improved:
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                b = a.copy()
                amt = a[i] * 0.03
                b[i] -= amt
                b[j] += amt
                b = np.maximum(b, 0.0)
                ss = b.sum()
                if ss > 1e-15:
                    b = b / ss
                    sc = objective_norm(b)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best = b.copy()
                        a = b
                        stall = 0

    return best


def interpolate_sequence(f, new_n):
    f = normalize_vec(f)
    old_n = f.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return f.copy()
    if old_n <= 1:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f)
    return normalize_vec(g)


def make_diverse_starts(n, parent):
    """Generate diverse starting sequences."""
    out = []
    if parent is not None:
        p = normalize_vec(parent)
        if p.size != n:
            p = interpolate_sequence(p, n)
    else:
        x = np.linspace(0.0, 1.0, n)
        p = normalize_vec(np.sin(np.pi * x) ** 2)

    out.append(p.copy())
    out.append(p[::-1].copy())

    # Power transforms
    base = np.maximum(p, 1e-30)
    for g in (0.3, 0.5, 0.7, 1.5, 2.0, 3.0):
        out.append(normalize_vec(base ** g))

    # Parametric shapes
    x = np.arange(n, dtype=np.float64) / float(max(n - 1, 1))
    for aexp in (0.3, 0.5, 1.0, 2.0, 3.0):
        out.append(normalize_vec((x + 1e-8) ** aexp))
        out.append(normalize_vec(((1.0 - x) + 1e-8) ** aexp))

    # Product shapes
    for ae in (0.3, 0.5, 1.0):
        for be in (0.3, 0.5, 1.0):
            seq = ((x + 1e-8) ** ae) * (((1.0 - x) + 1e-8) ** be)
            out.append(normalize_vec(seq))

    # Gaussian mixtures
    for ctr in (0.3, 0.5, 0.7):
        for wd in (0.05, 0.1, 0.2, 0.3):
            seq = np.exp(-((x - ctr) ** 2) / (2.0 * wd * wd))
            out.append(normalize_vec(seq))

    # Bimodal
    for c1, c2 in ((0.25, 0.75), (0.3, 0.7), (0.2, 0.8)):
        for wd in (0.05, 0.1, 0.2):
            seq = np.exp(-((x - c1) ** 2) / (2.0 * wd * wd)) + \
                  np.exp(-((x - c2) ** 2) / (2.0 * wd * wd))
            out.append(normalize_vec(seq))

    # Cosine-based
    for freq in (1.0, 1.5, 2.0, 3.0):
        seq = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x)
        seq = np.maximum(seq, 1e-8)
        out.append(normalize_vec(seq))

    # Random smooth
    rng = np.random.default_rng(12345)
    for _ in range(20):
        seq = np.zeros(n)
        n_peaks = int(rng.integers(1, 5))
        for _ in range(n_peaks):
            ctr = rng.uniform(0.05, 0.95)
            wd = rng.uniform(0.03, 0.4)
            ht = rng.uniform(0.3, 1.0)
            seq += ht * np.exp(-((x - ctr) ** 2) / (2.0 * wd * wd))
        seq += 1e-6
        out.append(normalize_vec(seq))

    return out


def score_and_sort(starts):
    scored = []
    for s in starts:
        sc = objective_norm(s)
        if np.isfinite(sc):
            scored.append((sc, s))
    scored.sort(key=lambda x: x[0])
    # Keep top diverse
    kept = []
    for sc, s in scored:
        duplicate = False
        for _, ks in kept:
            if s.size == ks.size:
                dp = float(np.dot(s, ks))
                ss = float(np.sqrt(np.dot(s, s) * np.dot(ks, ks)))
                if ss > 1e-15 and dp / ss > 0.995:
                    duplicate = True
                    break
        if not duplicate:
            kept.append((sc, s))
        if len(kept) >= 5:
            break
    return kept


def transform_search(a0, time_limit):
    """Try power transforms, blending with uniform, reversal."""
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    best = a.copy()
    best_score = objective_norm(a)

    for g in (0.3, 0.5, 0.7, 1.2, 1.5, 2.0, 3.0):
        if time.monotonic() - t0 > time_limit:
            break
        b = np.maximum(a, 1e-30) ** g
        b = normalize_vec(b)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    u = np.ones(n, dtype=np.float64) / float(n)
    for w in (0.01, 0.03, 0.05, 0.1, 0.2, 0.3):
        if time.monotonic() - t0 > time_limit:
            break
        b = normalize_vec((1.0 - w) * a + w * u)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    rev = a[::-1].copy()
    for w in (0.05, 0.1, 0.2, 0.3, 0.5):
        if time.monotonic() - t0 > time_limit:
            break
        b = normalize_vec((1.0 - w) * a + w * rev)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    return best


def propose_candidate():
    t_global = time.monotonic()
    total_budget = 1800.0 - 120.0  # Reserve time for final processing
    total_budget = max(60.0, total_budget)

    # Get parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if p.ndim != 1:
            p = p.ravel()
        if p.size > 0 and p.sum() > 1e-12:
            parent = p
    except Exception:
        parent = None

    if parent is not None:
        n0 = int(parent.size)
        if n0 > 9000:
            n0 = 8000
        best_seq = interpolate_sequence(parent, n0)
    else:
        n0 = 6500
        x = np.linspace(0.0, 1.0, n0)
        best_seq = normalize_vec(np.sin(np.pi * x) ** 2)

    best_score = objective_norm(best_seq)
    print("Initial: n=%d score=%.9f" % (n0, best_score))

    # Phase 1: Generate diverse starts and score them
    starts = make_diverse_starts(n0, best_seq)
    # Ensure parent is included
    if parent is not None:
        starts.insert(0, interpolate_sequence(parent, n0))

    scored = score_and_sort(starts)
    print("Best start score: %.9f" % scored[0][0])

    # Update best if a start is better
    if scored[0][0] < best_score:
        best_score = scored[0][0]
        best_seq = scored[0][1].copy()

    # Phase 2: Run mirror descent on top diverse starts
    for idx in range(min(3, len(scored))):
        sc, seq = scored[idx]
        remaining = total_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t_lim = min(80.0, 0.12 * remaining)
        if t_lim < 20:
            continue
        out = mirror_descent_multi_scale(seq, t_lim, rng_seed=1000 + 13 * idx)
        out_sc = objective_norm(out)
        if out_sc < best_score - 1e-12:
            best_score = out_sc
            best_seq = out.copy()
            print("Mirror %d: NEW BEST %.9f (n=%d)" % (idx, best_score, best_seq.size))

    # Phase 3: Main LP-FW ring optimization
    remaining = total_budget - (time.monotonic() - t_global)
    lp_time = min(1000.0, 0.50 * remaining)
    if lp_time > 90.0:
        print("Starting LP-FW ring: time_budget=%.0fs" % lp_time)
        out = lp_fw_ring(best_seq, lp_time, verbose=True)
        out_sc = objective_norm(out)
        if out_sc < best_score - 1e-12:
            best_score = out_sc
            best_seq = out.copy()
            print("LP-FW: NEW BEST %.9f" % best_score)

    # Phase 4: Second diverse start with LP-FW if time allows
    remaining = total_budget - (time.monotonic() - t_global)
    if len(scored) >= 2 and remaining > 400:
        sc2, seq2 = scored[1]
        # Check diversity
        dp = float(np.dot(seq2, best_seq))
        ss = float(np.sqrt(np.dot(seq2, seq2) * np.dot(best_seq, best_seq)))
        if ss > 1e-15 and dp / ss < 0.99:
            t2 = min(300.0, 0.25 * remaining)
            if t2 > 60:
                out2 = lp_fw_ring(seq2, t2, verbose=False)
                out2_sc = objective_norm(out2)
                if out2_sc < best_score - 1e-12:
                    best_score = out2_sc
                    best_seq = out2.copy()
                    print("LP-FW second: NEW BEST %.9f" % best_score)

    # Phase 5: Mirror descent polish on current best
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 100:
        t3 = min(200.0, 0.20 * remaining)
        if t3 > 50:
            out3 = mirror_descent_multi_scale(best_seq, t3, rng_seed=9999)
            out3_sc = objective_norm(out3)
            if out3_sc < best_score - 1e-12:
                best_score = out3_sc
                best_seq = out3.copy()
                print("Mirror polish: NEW BEST %.9f" % best_score)

    # Phase 6: Transform search
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 60:
        t4 = min(60.0, 0.15 * remaining)
        if t4 > 10:
            out4 = transform_search(best_seq, t4)
            out4_sc = objective_norm(out4)
            if out4_sc < best_score - 1e-12:
                best_score = out4_sc
                best_seq = out4.copy()
                print("Transform: NEW BEST %.9f" % best_score)

            nn = cur_n + dn

    # Phase 8: Peak shaving coordinate descent
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 80:

        if tc > 20:

            if out_f_sc < best_score - 1e-12:
                best_score = out_f_sc
                best_seq = out_f.copy()

        remaining = total_budget - (time.monotonic() - t_global)
        if remaining > 30:
            tp = min(30.0, 0.7 * remaining)
            if tp > 5:

    out_arr = np.maximum(out_arr, 0.0)

        out_arr = np.sin(np.pi * x) ** 2

    elapsed = time.monotonic() - t_global
    print("FINAL: score=%.10f n=%d time=%.1fs" % (final_score, out_arr.size, elapsed))

    return [float(v) for v in out_arr]
