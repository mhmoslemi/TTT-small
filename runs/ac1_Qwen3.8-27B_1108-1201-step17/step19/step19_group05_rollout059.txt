"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-start mirror descent
with multi-peak subgradients, spectral deflation, and coordinate polishing.
Key insight: non-palindromic sequences allow the autoconvolution peak to be
lower than the sum of squares, enabling J < 2. Asymptotically, a decreasing
ramp gives J -> 4/3. We start from the parent (1.504) and polish aggressively.

import numpy as np
import time


def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_multi_peak_subgrad(a, conv, n_peaks=80, temperature=0.05):
    """Compute weighted subgradient using top peaks with softmax-like weights."""
    n = len(a)
    if n_peaks >= len(conv):
        n_peaks = len(conv) - 1 if len(conv) > 1 else 1
    # Use indices of top peaks
    top_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
    vals = conv[top_idx]
    max_v = vals.max()
    if max_v < 1e-30:
        return np.zeros(n)
    # Softmax weights (sharper = more focus on the true max)
    exp_v = np.exp((vals - max_v) / (max_v * temperature))
    w = exp_v / exp_v.sum()

    grad = np.zeros(n)
    for idx_i, k in enumerate(top_idx):
        ki = int(k)
        wi = w[idx_i]
        if wi < 1e-14:
            continue
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        j_range = np.arange(j_lo, j_hi + 1)
        pair = ki - j_range
        # Gradient of b_k w.r.t. a_j is 2*a_{k-j} (or a_{k-j} if k=2j)
        contrib = 2.0 * a[pair]
        # Fix diagonal: where k - j == j, i.e., j == k/2
        if ki % 2 == 0:
            diag = ki // 2
            if j_lo <= diag <= j_hi:
                pos = np.where(j_range == diag)[0]
                if len(pos) > 0:
                    contrib[pos[0]] = a[diag]
        grad[j_range] += wi * contrib
    return grad


def mirror_descent(a0, time_limit, n_peaks=80, lr0=3.0, restart_every=2000,
                   rng_seed=42, temperature=0.05, verbose=False):
    """Nesterov-accelerated mirror descent on the simplex with multi-peak subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    restart_count = 0
    best_ever_m = best_m
    best_ever_a = a.copy()

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        if t < 5:
            beta_t = 0.0
        else:
            beta_t = (t - 1.0) / (t + 2.0)

        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-20)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        conv = fast_autocorr(y)
        g = compute_multi_peak_subgrad(y, conv, n_peaks=n_peaks, temperature=temperature)

        # Adaptive learning rate
        eta = lr0 / (np.sqrt(t) + 1.0)
        # Cap the gradient to prevent explosion
        g_max = np.max(np.abs(g))
        if g_max > 1e10:
            g = g / g_max * 1e10

        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Evaluate
        M_x = fast_autocorr(x).max()
        if M_x < best_m - 1e-18:
            best_m = M_x
            best_a = x.copy()
            if M_x < best_ever_m:
                best_ever_m = M_x
                best_ever_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart mechanism
        if stall > restart_every:
            stall = 0
            restart_count += 1
            strength = min(0.15, 0.03 + 0.02 * min(restart_count, 5))
            if rng_seed is not None:
                pert = rng.normal(0, strength, n) * best_a
                x_r = best_a + pert
                x_r = np.maximum(x_r, 1e-15)
                x_r = x_r / x_r.sum()
                x = x_r

            if verbose:

            print(f"      MD t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, stall={stall}")

def spectral_deflation(a0, time_limit, verbose=False):
    """Flatten the autoconvolution peaks using spectral domain manipulation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    stall = 0

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        # Identify peaks that are significantly above the mean
        mean_b = conv.sum() / len(conv)
        threshold = mean_b * 1.05
        peak_mask = conv > threshold

        if not np.any(peak_mask):
            break

        # In frequency domain, reduce the peaks
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2

        # Compute target spectrum: we want the IFFT of P to have reduced peaks
        # Use a soft deflation: reduce P where it contributes to peaks
        # Simple approach: reduce the overall magnitude slightly and re-project
        alpha = 0.1 + 0.5 * min(1.0, it / 200.0)
        # Reduce P at frequencies that correlate with the peak structure
        P_deflated = P.copy()
        # Scale down to reduce peak height
        scale = 1.0 - alpha * 0.02
        P_deflated[1:] = P_deflated[1:] * scale
        # Boost the DC to maintain sum
        P_deflated[0] = P[0]

        A_new = A * np.sqrt(P_deflated / np.maximum(P, 1e-30))
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Blend
            blend = 0.1
            a = (1 - blend) * a + blend * a_new
            a = np.maximum(a, 0.0)
            a = a / max(a.sum(), 1e-15)
            M_curr = fast_autocorr(a).max()
            if M_curr < best_m:
                best_m = M_curr
                best_a = a.copy()
            stall += 1

        if stall > 100:
            break

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Spec it={it} ({elapsed:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=42):
    """Local coordinate descent to polish the solution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit and stall < 50:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1
        conv = fast_autocorr(a)
        k_max = int(np.argmax(conv))

        # Find the pair with max contribution at k_max
        j_lo = max(0, k_max - n + 1)
        j_hi = min(n - 1, k_max)
        if j_lo > j_hi:
            stall += 1
            continue
        idx = np.arange(j_lo, j_hi + 1)
        contrib = a[idx] * a[k_max - idx]
        src = idx[np.argmax(contrib)]

        if a[src] < 1e-14:
            stall += 1
            continue

        # Compute cost of each position (how much moving to it creates peaks)
        top_ks = np.argpartition(conv, -15)[-15:]
        costs = np.zeros(n)
        for k2 in top_ks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            if lo2 <= hi2:
                valid = np.arange(lo2, hi2 + 1)
                costs[valid] += a[k2i - valid]

        # Pick destination with low cost
        sort_key = costs + 0.1 * a
        sorted_idx = np.argsort(sort_key)
        dst = sorted_idx[0]
        if dst == src:
            dst = sorted_idx[1] if len(sorted_idx) > 1 else int(rng.integers(n))

        moved = False
        for frac in [0.02, 0.05, 0.1, 0.15, 0.25, 0.4]:
            amt = a[src] * frac
            if amt < 1e-16:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            M_t = fast_autocorr(a_trial).max()
            if M_t < best_m - 1e-18:
                a = a_trial
                best_m = M_t
                best_a = a.copy()
                moved = True
                stall = 0
                break

        if not moved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                amt = a[i] * rng.uniform(0.01, 0.05)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss


def make_decreasing_ramp(n, alpha=1.0):
    """Create a decreasing ramp (i+1)^alpha, normalized."""
    x = np.arange(n, 0, -1, dtype=np.float64)
    a = x ** alpha
    return a / a.sum()


def safe_eval(a):
    """Safely compute the objective value."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """
    Main search function. Strategy:
    1. Start from parent sequence (score ~1.504)
    2. Aggressive mirror descent with multi-peak subgradients
    3. Spectral deflation
    4. Coordinate polishing
    5. Multi-length exploration
    6. Final combined polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = safe_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = safe_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Build initial candidates
    candidates = []
    if parent is not None:
        p_norm = parent / parent.sum()
        candidates.append((p_norm, "parent"))

    # Also try decreasing ramps (asymptotic J -> 4/3)
    n_ramp = 6000
    for alpha in [1.0, 0.8, 1.2, 1.5]:
        a = make_decreasing_ramp(n_ramp, alpha)
        candidates.append((a, f"ramp_dec_a{alpha}"))

    # Try the parent at different effective normalizations (already scale-invariant, so just normalize)
    if parent is not None:
        # Slightly perturb the parent for diversity
        rng_init = np.random.default_rng(0)
        for seed_val in [1, 2, 3]:
            pert = rng_init.normal(0, 0.05, len(parent)) * (parent / parent.sum())
            a_pert = parent / parent.sum() + pert
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / a_pert.sum()
            candidates.append((a_pert, f"parent_pert{seed_val}"))

    # Evaluate initial candidates
    print(f"\nInitial candidates: {len(candidates)}")
    scored = []
    for a, label in candidates:
        sc = safe_eval(a)
        if np.isfinite(sc):
            print(f"  Init [{label}]: n={len(a)}, J={sc:.6f}")
            scored.append((sc, a, label))
    scored.sort(key=lambda x: x[0])

    # Phase 1: Mirror descent on top candidates (most time here)
    print(f"\n{'='*50}")
    print("Phase 1: Mirror Descent (primary optimization)")
    print(f"{'='*50}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_cands = min(5, len(scored))

    # Allocate time: 50% to best candidate, 20% to second, 15% to third, rest split
    time_fracs = [0.45, 0.20, 0.15, 0.10, 0.10][:n_cands]
    # Normalize
    tf_sum = sum(time_fracs)
    time_fracs = [f / tf_sum for f in time_fracs]

    for i in range(n_cands):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget:
            break
        t_avail = min(remaining * time_fracs[i], time_budget - elapsed)
        if t_avail < 10:
            break
        sc0, a0, label = scored[i]
        print(f"\n  MD [{label}] init_J={sc0:.6f}, time={t_avail:.0f}s")

        # Use different parameters for different candidates
        if i == 0:
            # Best candidate: long run, careful
            a_out, _ = mirror_descent(a0, t_avail, n_peaks=100, lr0=2.5,
                                      restart_every=3000, rng_seed=42,
                                      temperature=0.03, verbose=True)
        elif i == 1:
            a_out, _ = mirror_descent(a0, t_avail, n_peaks=60, lr0=4.0,
                                      restart_every=2000, rng_seed=77,
                                      temperature=0.08, verbose=False)
        else:
            a_out, _ = mirror_descent(a0, t_avail, n_peaks=50, lr0=5.0,
                                      restart_every=1500, rng_seed=100 + i,
                                      temperature=0.12, verbose=False)
        try_update(a_out, f"md_{label}")

    # Phase 2: Second round of mirror descent on best
    print(f"\n{'='*50}")
    print("Phase 2: Refined Mirror Descent on Best")
    print(f"{'='*50}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and best_seq is not None:
        t2 = min(remaining * 0.20, 300.0)
        a_in = best_seq / best_seq.sum()
        print(f"  MD round 2, time={t2:.0f}s, current best={best_score:.6f}")

        # Multiple seeds for diversity
        for seed in [200, 500, 999]:
            t_sub = t2 / 3.0
            a_out, _ = mirror_descent(a_in, t_sub, n_peaks=120, lr0=2.0,
                                      restart_every=2500, rng_seed=seed,
                                      temperature=0.02, verbose=False)
            try_update(a_out, f"md_r2_s{seed}")
            a_in = best_seq / best_seq.sum()  # Always start from best

    # Phase 3: Spectral deflation
    print(f"\n{'='*50}")
    print("Phase 3: Spectral Deflation")
    print(f"{'='*50}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40 and best_seq is not None:
        t3 = min(remaining * 0.10, 150.0)
        a_in = best_seq / best_seq.sum()
        print(f"  Spectral, time={t3:.0f}s")
        a_out, _ = spectral_deflation(a_in, t3, verbose=True)
        try_update(a_out, "spectral")

        # MD after spectral
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t3b = min(remaining * 0.08, 100.0)
            a_in = best_seq / best_seq.sum()
            a_out, _ = mirror_descent(a_in, t3b, n_peaks=80, lr0=2.0,
                                      restart_every=2000, rng_seed=777,
                                      temperature=0.03, verbose=False)
            try_update(a_out, "md_post_spec")

    # Phase 4: Coordinate descent polishing
    print(f"\n{'='*50}")
    print("Phase 4: Coordinate Descent Polishing")
    print(f"{'='*50}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        t4 = min(remaining * 0.12, 180.0)
        a_in = best_seq / best_seq.sum()
        print(f"  Coord descent, time={t4:.0f}s")
        a_out, _ = coordinate_polish(a_in, t4, verbose=True, rng_seed=42)
        try_update(a_out, "coord1")

        a_out, _ = coordinate_polish(a_in, t4, verbose=False, rng_seed=12345)
        try_update(a_out, "coord2")

    # Phase 5: Multi-length exploration
    print(f"\n{'='*50}")
    print("Phase 5: Multi-Length Exploration")
    print(f"{'='*50}")

    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is not None and remaining > 60:
        n_best = len(best_seq)
        # Try a few nearby lengths
        for new_n in [n_best + 500, n_best + 1000, n_best + 2000,
                      max(3000, n_best - 1000)]:
            if new_n <= 0 or new_n > 15000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break

            a_interp = interpolate_to_length(best_seq, new_n)
            sc_interp = safe_eval(a_interp)
            print(f"  Interp n={new_n}: J={sc_interp:.6f}")
            try_update(a_interp, f"interp_n{new_n}")

            t5 = min(remaining * 0.15, 120.0)
            if t5 > 15:
                a_out, _ = mirror_descent(a_interp, t5, n_peaks=60, lr0=3.0,
                                          restart_every=2000, rng_seed=100 + new_n % 100,
                                          temperature=0.05, verbose=False)
                try_update(a_out, f"md_n{new_n}")

    # Phase 6: Final aggressive polish
    print(f"\n{'='*50}")
    print("Phase 6: Final Polish")
    print(f"{'='*50}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20 and best_seq is not None:
        a_in = best_seq / best_seq.sum()

        # One more MD pass with very fine temperature
        t6 = min(remaining * 0.40, 200.0)
        if t6 > 10:
            print(f"  Final MD, time={t6:.0f}s")
            a_out, _ = mirror_descent(a_in, t6, n_peaks=150, lr0=1.5,
                                      restart_every=4000, rng_seed=4242,
                                      temperature=0.01, verbose=True)
            try_update(a_out, "md_final")

        # Quick coord descent
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15 and best_seq is not None:
            t7 = min(remaining * 0.30, 80.0)
            a_in = best_seq / best_seq.sum()
            a_out, _ = coordinate_polish(a_in, t7, verbose=False, rng_seed=99999)
            try_update(a_out, "coord_final")

            # One more spectral
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 10 and best_seq is not None:
                t8 = min(remaining * 0.30, 60.0)
                a_in = best_seq / best_seq.sum()
                a_out, _ = spectral_deflation(a_in, t8, verbose=False)
                try_update(a_out, "spec_final")

    # Final validation
    if best_seq is not None:
        fs = safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\n{'='*50}")
        print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
        print(f"{'='*50}")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        best_seq = make_decreasing_ramp(5000, 1.0)
        best_score = safe_eval(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
