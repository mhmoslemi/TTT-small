Now, for the LP itself, let me think about the formulation:

```
minimize M
subject to: sum(g) = 1
            g >= 0
            A @ g <= M * ones(K)
```

This is a standard LP. With cvxpy, this should be fast.

But actually, I realize I can reformulate this as:
```
minimize M
subject to: sum(g) = 1
            g >= 0
            A @ g - M <= 0
```

Where `M` is a free variable. This is what cvxpy would do anyway.

Let me now think about the step size selection. After getting the FW direction `g`, we want to find the best `alpha` in `[0, 1]` such that:
`J((1-alpha)*y + alpha*g)` is minimized.

Since `J` depends on `max(conv)`, and the convolution is a quadratic form, the function `alpha -> max(conv((1-alpha)*y + alpha*g))` is a convex function (max of convex functions, and each component is a quadratic in alpha).

So we can do a ternary search or golden section search to find the minimum. But each evaluation requires an FFT, which is O(n log n). For n=8000, that's about 0.01s. So 50 evaluations would take 0.5s. That's too much if we're doing 600+ iterations.

Alternative: use a cheap approximation. The maximum of the convolution is dominated by a few peaks. We can:
1. Find the top 20 peaks of the current convolution
2. For each peak, compute `conv((1-alpha)*y + alpha*g)[k]` as a quadratic in alpha
3. Find the alpha that minimizes the maximum of these 20 quadratics
4. This is a 1D problem that can be solved by bisection

Actually, let me just use the default FW step size `alpha = 2/(t+2)` and occasionally try a few alternatives. The key is to have many iterations rather than perfect step sizes.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Informed initialization from parent + spectral equalization
2. High-speed LP-based Frank-Wolfe with vectorized constraint building
3. Adaptive active set management
4. Multi-length exploration
5. Peak-aware coordinate descent polish

import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix

_S = None

def get_solver():
    global _S
    if _S is not None:
        return _S
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(200, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _S = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _S = cp.SCS
    print("  Solver (fallback): SCS")
    return _S


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_seq(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_cmat(f, active_set, n):
    """Build constraint matrix: row k has f[i] at column i for valid i."""
    K = len(active_set)
    if K == 0:
        return csr_matrix((0, n))
    ks = np.array([int(k) for k in active_set])
    i_starts = np.maximum(0, ks - n + 1)
    i_ends = np.minimum(n, ks + 1)
    lengths = i_ends - i_starts
    total = int(lengths.sum())
    if total == 0:
        return csr_matrix((K, n))
    offsets = np.concatenate([[0], np.cumsum(lengths)[:-1]])
    row_idx = np.repeat(np.arange(K, dtype=np.int32), lengths)
    col_idx = (np.repeat(i_starts, lengths) + np.arange(total, dtype=np.int64) 
               - np.repeat(offsets, lengths)).astype(np.int32)
    data = f[col_idx].astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(K, n))


def spectral_equalize(a0, n_iters=200, target_flatness=0.0):
    """
    Gerchberg-Saxton style spectral equalization.
    Tries to flatten the autocorrelation by adjusting the magnitude spectrum.
    """
    a = norm_seq(a0).copy()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    # Target: flat power spectrum (which gives flat autocorrelation)
    # But we need to maintain sum(a) = 1 and a >= 0
    
    for it in range(n_iters):
        # FFT
        A = np.fft.rfft(a, fft_size)
        # Compute autocorrelation
        ac = np.fft.irfft(np.abs(A)**2, fft_size)[:2*n-1]
        # Find max
        M = ac.max()
        # Compute "pressure" at each frequency
        # We want to reduce frequencies that contribute to the peak
        # Inverse: for peak at lag k, the contribution from freq w is |A(w)|^2 * exp(-i*w*k)
        # So we should reduce |A(w)| where it contributes positively to the peak
        
        # Simpler approach: clip the power spectrum
        power = np.abs(A)**2
        # Try to make power more uniform
        # But this might make a negative in time domain
        
        # Alternative: multiplicative update in frequency domain
        # Reduce magnitude where it leads to high autocorrelation peaks
        peak_lag = int(np.argmax(ac))
        phase_exp = np.exp(-1j * 2 * np.pi * peak_lag * np.arange(len(A)) / fft_size)
        # The contribution of each frequency to the peak is:
        # sum_w |A(w)|^2 * exp(-i*w*peak_lag) / fft_size
        # We want to minimize this, so reduce |A| where phase aligns
        # This is complex, let's use a simpler heuristic
        
        # Simpler: just try to equalize the power spectrum
        target = power.mean()
        alpha_eq = 0.1  # Equalization strength
        new_power = power + alpha_eq * (target - power)
        new_A = np.sqrt(new_power) * A / np.maximum(np.abs(A), 1e-30)
        
        # IFFT and clip
        a_new = np.fft.irfft(new_A, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check improvement
        ac_new = fast_conv(a_new)
        if ac_new.max() < M * 0.99:
            a = a_new
        else:
            # Try weaker equalization
            break
    
    return a


def lp_fw(a0, time_limit, solver, initial_K=200, max_K=4000, verbose=False):
    """LP-based Frank-Wolfe method."""
    n = len(a0)
    y = norm_seq(a0).copy()
    best_y = y.copy()
    best_M = fast_conv(y).max()
    t0 = time.monotonic()
    
    # Initialize active set
    ac = fast_conv(y)
    active_set = sorted(np.argsort(ac)[::-1][:initial_K].tolist())
    
    no_improve = 0
    max_no_improve = 500
    iter_count = 0
    initial_m = best_M
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        elapsed = time.monotonic() - t0
        if elapsed >= time_limit:
            break
        
        ac = fast_conv(y)
        M_curr = ac.max()
        
        if M_curr < best_M - 1e-16:
            best_M = M_curr
            best_y = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            if verbose:
                print("    FW stagnated at iter " + str(iter_count))
            break
        
        # Update active set: add worst violators, remove least tight
        if iter_count % 10 == 1:
            # Add top violators
            top_viol = np.argsort(ac)[::-1][:max_K]
            new_set = set(active_set)
            for k in top_viol:
                new_set.add(int(k))
            active_set = sorted(new_set)
        
        # Trim active set
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep = np.argsort(margins)[:max_K]
            active_set = sorted([active_set[i] for i in keep])
        
        K = len(active_set)
        if K == 0:
            break
        
        # Build constraint matrix
        A_mat = build_cmat(y, active_set, n)
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A_mat @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if verbose:
                print("    LP solve failed at iter " + str(iter_count))
            active_set = sorted(active_set[:initial_K])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Step size: try a few values
        default_alpha = 2.0 / (iter_count + 2.0)
        alphas = [default_alpha, min(1.0, default_alpha * 2), min(1.0, default_alpha * 0.5), 0.3, 0.5]
        if iter_count <= 5:
            alphas.append(1.0)
        
        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_conv(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        if iter_count <= 3:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_y.copy()
            else:
                y = y / s_y
        
        if verbose and iter_count % 100 == 0:
            J = 2.0 * n * best_M
            print("    FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", |S|=" + str(K))
    
    J = 2.0 * n * best_M
    if verbose:
        print("    FW done (" + str(round(time.monotonic() - t0, 1)) + "s): J=" + repr(J) + ", iters=" + str(iter_count))
    return best_y, best_M


def coord_polish(a0, time_limit, verbose=False):
    """Peak-aware coordinate descent."""
    n = len(a0)
    a = norm_seq(a0).copy()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = int(idx[np.argmax(contrib)])
            if a[src] < 1e-12:
                continue
            
            # Find best destination (lowest contribution to peaks)
            scores = np.zeros(n)
            for k2 in top_k[:15]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
            
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:15]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2+1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < n // 4:
                candidates = np.arange(n)
            
            dst = int(candidates[np.argmin(scores[candidates])])
            if dst == src:
                continue
            
            # Try to move mass
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M - 1e-16:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation when stuck
            if stall > 50:
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.01, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_conv(a).max()
                            if M_new < best_M:
                                best_M = M_new
                                best_a = a.copy()
                                stall = 0
    
    return best_a, best_M


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

Actually, for the spectral approach, the key insight is:
- `conv(a,a)` as a function of lag is the inverse DFT of `|FFT(a)|^2` (for circular correlation, approximately for linear with zero-padding)
- We want to minimize `max(conv(a,a))`
- By Parseval, `sum(conv(a,a)) = (sum(a))^2 = 1` (after normalization)
- So if we can make `conv(a,a)` perfectly flat, `max = 1/(2n-1)`
- In the frequency domain, this means making `|FFT(a)|^2` have a specific form

The connection: if `|A(k)|^2` is constant for all k, then the circular autocorrelation is a delta function (concentrated at lag 0). That's the opposite of what we want! We want the autocorrelation to be flat, which means `|A(k)|^2` should be a delta function (all energy at one frequency). But that would make `a` a pure sinusoid, which can be negative.

So the problem is more subtle. We want:
- `a >= 0`
- `sum(a) = 1`
- `max(conv(a,a))` minimized

The flat autocorrelation would require `|A(k)|^2` to be concentrated, but that conflicts with `a >= 0`.

Actually, let me reconsider. For the circular case:
- `c[k] = sum_i a[i] * a[(i+k) mod n]` (circular autocorrelation)
- `c[k] = IFFT(|FFT(a)|^2)[k]`
- If `|FFT(a)|^2` is flat, then `c = delta` (spike at 0)
- If `|FFT(a)|^2` is a delta (all at DC), then `a` is constant and `c` is flat

So for a constant `a`, the autocorrelation is flat! That's the uniform sequence, which gives `J = 2`.

The issue is that for linear convolution, the uniform sequence doesn't have a flat autocorrelation - it has a triangular shape. The maximum is at the center (lag n-1).

So the problem is inherently different for linear vs circular convolution.

For the linear case, the optimal solution is NOT the uniform sequence. It's something that has a more "spread out" profile, perhaps with higher values in the middle and lower at the edges (or vice versa), to equalize the convolution peaks.

Let me think about this more carefully. For a sequence of length n:
- `conv(a,a)[k]` for k near 0 or 2n-2 involves few terms (short overlap)
- `conv(a,a)[n-1]` involves all n terms (maximum overlap)

The optimal solution is a trade-off. This is exactly what the LP-FW finds.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Informed initialization from parent sequence
2. High-speed LP-based Frank-Wolfe with vectorized sparse constraint building
3. Adaptive active set management (coarse-to-fine)
4. Multi-length exploration with interpolation
5. Peak-aware coordinate descent for final polish

import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix

_S = None

def get_solver():
    global _S
    if _S is not None:
        return _S
    for s in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(200, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _S = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _S = cp.SCS
    print("  Solver (fallback): SCS")
    return _S


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_seq(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_cmat(f, active_set, n):
    """Build constraint matrix: row k has f[j] at column j for j in valid range."""
    K = len(active_set)
    if K == 0:
        return csr_matrix((0, n))
    ks = np.array([int(k) for k in active_set])
    i_starts = np.maximum(0, ks - n + 1)
    i_ends = np.minimum(n, ks + 1)
    lengths = i_ends - i_starts
    total = int(lengths.sum())
    if total == 0:
        return csr_matrix((K, n))
    offsets = np.concatenate(([0], np.cumsum(lengths)[:-1]))
    row_idx = np.repeat(np.arange(K, dtype=np.int32), lengths)
    col_idx = (np.repeat(i_starts.astype(np.int64), lengths) + np.arange(total, dtype=np.int64)
               - np.repeat(offsets.astype(np.int64), lengths)).astype(np.int32)
    data = f[col_idx].astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(K, n))


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lp_fw(a0, time_limit, solver, initial_K=200, max_K=3000, verbose=True):
    """LP-based Frank-Wolfe with adaptive active set."""
    n = len(a0)
    y = norm_seq(a0).copy()
    best_y = y.copy()
    best_M = fast_conv(y).max()
    t0 = time.monotonic()

    ac = fast_conv(y)
    active_set = sorted(np.argsort(ac)[::-1][:initial_K].tolist())

    no_improve = 0
    max_no_improve = 400
    iter_count = 0
    initial_m = best_M

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        if time.monotonic() - t0 >= time_limit:
            break

        ac = fast_conv(y)
        M_curr = ac.max()

        if M_curr < best_M - 1e-16:
            best_M = M_curr
            best_y = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            if verbose:
                print("    FW stagnated at iter " + str(iter_count) + ", J=" + repr(2.0 * n * best_M))
            break

        # Adaptively grow active set
        target_K = min(max_K, initial_K + iter_count * 5)

        # Periodically refresh active set
        if iter_count % 15 == 1:
            top_viol = np.argsort(ac)[::-1][:target_K]
            new_set = set(active_set)
            for k in top_viol:
                new_set.add(int(k))
            active_set = sorted(new_set)

        # Trim active set
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep = np.argsort(margins)[:max_K]
            active_set = sorted([active_set[i] for i in keep])

        K = len(active_set)
        if K == 0:
            break

        A_mat = build_cmat(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A_mat @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = sorted(active_set[:initial_K])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Step size selection
        default_alpha = 2.0 / (iter_count + 2.0)
        alphas = [default_alpha, min(1.0, default_alpha * 2.0), min(1.0, default_alpha * 0.5), 0.2, 0.4]
        if iter_count <= 5:
            alphas.append(1.0)

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_conv(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 3:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_y.copy()
            else:
                y = y / s_y

        if verbose and iter_count % 200 == 0:
            J = 2.0 * n * best_M
            elapsed = time.monotonic() - t0
            print("    FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", |S|=" + str(K))

    J = 2.0 * n * best_M
    if verbose:
        print("    FW done (" + str(round(time.monotonic() - t0, 1)) + "s): J=" + repr(J) + ", iters=" + str(iter_count))
    return best_y, best_M


def lp_fw_phased(a0, time_limit, solver, verbose=True):
    """
    Two-phase LP-FW: Phase A uses small active set for fast iterations,
    Phase B uses larger active set for refinement.
    """
    n = len(a0)
    t0 = time.monotonic()

    # Phase A: fast iterations with small K
    tA = time_limit * 0.5
    yA, MA = lp_fw(a0, tA, solver, initial_K=200, max_K=1500, verbose=verbose)
    best_y = yA
    best_M = MA

    # Phase B: refinement with larger K
    elapsed = time.monotonic() - t0
    tB = time_limit - elapsed
    if tB > 30:
        yB, MB = lp_fw(yA, tB, solver, initial_K=500, max_K=4000, verbose=verbose)
        if MB < best_M:
            best_y = yB
            best_M = MB

    return best_y, best_M


def coord_polish(a0, time_limit, verbose=False):
    """Peak-aware coordinate descent for fine-grained improvements."""
    n = len(a0)
    a = norm_seq(a0).copy()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)

    stall = 0
    max_stall = 250

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            if a[src] < 1e-12:
                continue

            # Score destinations by their contribution to peaks
            scores = np.zeros(n)
            for k2 in top_k[:12]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            # Avoid peak regions
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:12]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < n // 4:
                candidates = np.arange(n)

            dst = int(candidates[np.argmin(scores[candidates])])
            if dst == src:
                continue

            for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M - 1e-16:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 40:
                for _ in range(8):
                    if time.monotonic() - t0 > time_limit:
                        break
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.01, 0.06)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_conv(a).max()
                            if M_new < best_M:
                                best_M = M_new
                                best_a = a.copy()
                                stall = 0

    if verbose:
        print("    Coord done (" + str(round(time.monotonic() - t0, 1)) + "s): J=" + repr(2.0 * n * best_M))
    return best_a, best_M


def diversity_restart(a0, n_perturbs=5, noise_level=0.05):
    """Generate diverse perturbations of the current solution."""
    n = len(a0)
    rng = np.random.default_rng(42)
    best = a0.copy()
    best_M = fast_conv(a0).max()

    for _ in range(n_perturbs):
        # Multiplicative noise
        noise = rng.lognormal(0, noise_level, n)
        a_pert = a0 * noise
        a_pert = norm_seq(a_pert)
        M_pert = fast_conv(a_pert).max()
        if M_pert < best_M:
            best_M = M_pert
            best = a_pert.copy()

        # Additive noise with renormalization
        a_add = a0 + rng.normal(0, noise_level * 0.5, n)
        a_add = norm_seq(a_add)
        M_add = fast_conv(a_add).max()
        if M_add < best_M:
            best_M = M_add
            best = a_add.copy()

    return best, best_M

            print("  *** LP-FW improved: J=" + repr(sc_out))

    # Phase 3: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 3: Coord Polish ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 20:
        a_out, M_out = coord_polish(norm_seq(best_seq), t3, verbose=True)
        sc_out = 2.0 * len(a_out) * M_out
        if sc_out < best_score:
            best_score = sc_out
            best_seq = a_out.copy()
            print("  *** Coord improved: J=" + repr(sc_out))

    # Phase 4: LP-FW refinement (starting from polished solution)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 4: LP-FW Refine (n=" + str(n_cur) + ") ===")
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 60 and n_cur <= 12000:
        a_out, M_out = lp_fw_phased(norm_seq(best_seq), t4, solver, verbose=True)
        sc_out = 2.0 * len(a_out) * M_out
        if sc_out < best_score:
            best_score = sc_out
            best_seq = a_out.copy()
            print("  *** LP-FW Refine improved: J=" + repr(sc_out))

    # Phase 5: Try different lengths
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Length Exploration ===")
    try_ns = [n_best + 2000, n_best - 1000, n_best + 4000]
    try_ns = [l for l in try_ns if 4000 <= l <= 14000 and l != n_best]

    for new_n in try_ns[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("  Trying n=" + str(new_n))
        a_interp = interpolate_seq(best_seq, new_n)
        t5 = min(remaining * 0.45, 300.0)
        if t5 > 60:
            a_out, M_out = lp_fw_phased(a_interp, t5, solver, verbose=False)
            sc_out = 2.0 * len(a_out) * M_out
            if sc_out < best_score:
                best_score = sc_out
                best_seq = a_out.copy()
                print("  *** n=" + str(new_n) + " improved: J=" + repr(sc_out))

    # Phase 6: Final LP-FW on best so far
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 6: Final LP-FW (n=" + str(n_cur) + ") ===")
    if remaining > 60 and n_cur <= 12000:
        t6 = min(remaining * 0.50, 300.0)
        if t6 > 30:
            a_out, M_out = lp_fw_phased(norm_seq(best_seq), t6, solver, verbose=True)
            sc_out = 2.0 * len(a_out) * M_out
            if sc_out < best_score:
                best_score = sc_out
                best_seq = a_out.copy()
                print("  *** Final LP-FW improved: J=" + repr(sc_out))

    # Phase 7: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 7: Final Coord ===")
    if remaining > 20:
        t7 = min(remaining * 0.65, 120.0)
        if t7 > 10:
            a_out, M_out = coord_polish(norm_seq(best_seq), t7, verbose=True)
            sc_out = 2.0 * len(a_out) * M_out
            if sc_out < best_score:
                best_score = sc_out
                best_seq = a_out.copy()
                print("  *** Final Coord improved: J=" + repr(sc_out))

    # Final output
    if best_seq is not None:
        best_seq = norm_seq(best_seq)
        ac = fast_conv(best_seq)
        final_J = 2.0 * len(best_seq) * ac.max()
        elapsed = time.monotonic() - t_global
        print("\nFINAL: J=" + repr(final_J) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        best_seq = norm_seq(np.sin(np.pi * np.linspace(0, 1, 5000)) ** 2)
        print("Fallback used")

    return [float(x) for x in best_seq]
