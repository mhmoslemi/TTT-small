"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using linear ramp initialization
and subgradient optimization with strict acceptance criteria.
Key insight: pure linear ramp gives J→4/3 asymptotically, far below previous 1.504.

import numpy as np

def fast_autocorr(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A*A, fft_size)[:size]

def compute_J(a):
    n = len(a)
    conv = fast_autocorr(a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2*n*max_b/(s*s))

def make_ramp(n, alpha=1.0):
    x = np.arange(1, n+1, dtype=np.float64)
    a = x**alpha
    s = a.sum()
    return a/s

def make_sine(n, power=1.0):
    x = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi*(x+0.5)/n)**power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return make_ramp(n)
    return a/s

def make_triangle(n):
    if n == 1:
        return np.array([1.0])
    if n % 2 == 1:
        a = np.concatenate([np.arange(1, n//2+1), np.arange(n//2, 0, -1)])
    else:
        a = np.concatenate([np.arange(1, n//2+1), np.arange(n//2, 0, -1)])[:n]
    s = a.sum()
    return a/s

def subgradient(a, conv, n_peaks=50):
    n = len(a)
    top_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
    vals = conv[top_idx]
    max_v = vals.max()
    spread = max_v - vals.min()
    if spread < 1e-15:
        w = np.ones(n_peaks) / n_peaks
    else:
        w = np.exp((vals - max_v) / spread)
        w = w / w.sum()
    
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        ki = int(k)
        wi = w[i]
        if wi < 1e-12:
            continue
        j_lo = max(0, ki - n + 1)
        j_hi = min(n-1, ki)
        j_range = np.arange(j_lo, j_hi+1)
        pair = ki - j_range
        grad[j_range] += wi * a[pair]
    return grad

def subgrad_optimize(a0, time_limit, lr=0.05, n_peaks=50, seed=42):
    n = len(a0)
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 1000
    step = 0
    current_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        conv = fast_autocorr(a)
        g = subgradient(a, conv, n_peaks)
        
        # Scale gradient to unit "energy"
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        a_new = a - current_lr * g / a.sum()
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            current_lr *= 0.5
            stall += 1
            if stall > max_stall:
                # Restart
                pert = rng.normal(0, 0.005, n)
                a = best_a * (1 + pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
            continue
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Slightly increase lr if progress
            current_lr = min(lr, current_lr * 1.01)
        else:
            stall += 1
            if stall % 100 == 0:
                current_lr *= 0.7
            if stall > max_stall:
                # Perturb and restart
                strength = 0.01 + 0.005 * (stall // max_stall)
                pert = rng.normal(0, strength, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                current_lr = lr
        
        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    subgrad step={step}, t={elapsed:.0f}s, J={best_J:.6f}, stall={stall}")
    
    return best_a, best_J

def pair_swap_optimize(a0, time_limit, seed=42):
    """Try pair swaps to reduce the autocorrelation peak."""
    n = len(a0)

        conv = fast_autocorr(a)
        k_peak = int(np.argmax(conv))
        
        # Find the pair that contributes most to the peak
        j_lo = max(0, k_peak - n + 1)
        j_hi = min(n-1, k_peak)
        idx = np.arange(j_lo, j_hi+1)
        contrib = a[idx] * a[k_peak - idx]
        src = idx[np.argmax(contrib)]
        
        # Find a good destination: low contribution to all top peaks
        top_ks = np.argpartition(conv, -20)[-20:]
        costs = np.zeros(n)
        for k2 in top_ks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n-1, k2i)
            if lo2 <= hi2:
                v = np.arange(lo2, hi2+1)
                costs[v] += a[k2i - v]
        
        # Pick destination with lowest cost
        sorted_dst = np.argsort(costs)
        
        moved = False
        for frac in [0.05, 0.1, 0.2, 0.3]:
            for di in range(5):
                dst = sorted_dst[di]
                if dst == src:
                    continue
                amt = a[src] * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial / s_t
                J_t = compute_J(a_trial)
                if J_t < best_J - 1e-12:
                    a = a_trial
                    best_a = a.copy()
                    best_J = J_t
                    stall = 0
                    moved = True
                    break
            if moved:
                break
        
        if not moved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.02, 0.1)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                J_t = compute_J(a_trial)
                if J_t < best_J:
                    best_J = J_t
                    best_a = a_trial.copy()
                    a = a_trial
                    stall = 0
    
    return best_a, best_J
