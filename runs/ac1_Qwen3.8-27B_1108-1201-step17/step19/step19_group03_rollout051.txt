        M_new = c_new.max()

        if M_new < best_c - 1e-15:
            best_c = M_new
            best_a = a_new.copy()
            stall = 0
            # Increase tau slowly
            tau = min(tau * tau_growth, tau_max)
        else:
            stall += 1
            if stall > 50:
                eta = max(eta * 0.85, 0.01)
            if stall > max_stall:
                # Restart with perturbation
                pert_scale = 0.05 + 0.03 * rng.random()
                pert = rng.normal(0, pert_scale, n) * best_a
                x_r = best_a + pert
                x_r = np.maximum(x_r, 1e-10)
                x_r = x_r / x_r.sum()
                x = x_r
                x_prev = x_r.copy()
                eta = 0.5
                tau = max(tau * 0.5, 10.0)
                stall = 0
                c_restart = fast_convolve(x).max()
                if c_restart < best_c:
                    best_c = c_restart
                    best_a = x.copy()

        x_prev = x
        x = a_new

        # Periodic verbose
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_c
            print(f"    SGD it={it} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}, stall={stall}")

    return best_a, best_c


def softmax_gradient_multi(a0, time_limit, n_trials=3, verbose=True):
    """Run soft-max GD with different tau schedules."""
    n = len(a0)
    best_a = a0.copy()
    best_m = fast_convolve(a0).max()
    t0 = time.monotonic()

    tau_schedules = [
        (20.0, 300.0, 1.005),
        (50.0, 400.0, 1.007),
        (10.0, 200.0, 1.010),
    ]

    t_per = (time_limit - 5) / n_trials
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit - 5:
            break
        t_avail = min(t_per, time_limit - 5 - (time.monotonic() - t0))
        if t_avail < 10:
            break
        tau_init, tau_max, tau_growth = tau_schedules[trial % len(tau_schedules)]
        print(f"  SGD trial {trial}: tau_init={tau_init}, t_avail={t_avail:.0f}s")
        a_out, m_out = softmax_gradient_descent(a0, t_avail, rng_seed=trial * 100, verbose=(trial == n_trials - 1))
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False):
    """LP-based Frank-Wolfe with adaptive K and line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()

    K = min(200, 2 * n - 1)
    K_max = min(3000, 2 * n - 1)
    iter_count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        M_curr = c.max()

        if M_curr < 1e-15:
            break

        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(c)[::-1][:K_sel]

        # Build constraint matrix
        T = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            for j in range(i_start, i_end):
                T[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 30, K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Line search for optimal alpha
        base_M = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_M
        alphas = [best_alpha, best_alpha * 0.5, best_alpha * 1.5, 0.1, 0.2, 0.3, 0.5]
        alphas = [min(1.0, max(0.01, al)) for al in alphas]

        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_convolve(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_convolve(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        iter_count += 1
        if iter_count % 15 == 0:
            K = min(K + 40, K_max)

        if stall > 100:
            break

        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Aggressive coordinate descent: move mass from peak-contributing to low-sensitivity coords."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200
    n_top = 30

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_convolve(a)
        M_curr = c.max()
        top_k = np.argsort(c)[::-1][:n_top]

        # Compute sensitivity: which coordinates contribute most to peak lags
        sens = np.zeros(n)
        for k in top_k[:15]:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            sens[idx] += a[ki - idx]

        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-sensitivity targets
            low_sens = np.argsort(sens)[:20]
            candidates = [j for j in low_sens if j != src and a[j] < src_val]

            found = False
            for dst in candidates[:5]:
                if found:
                    break
                for frac in [0.02, 0.01, 0.03, 0.05, 0.005]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_convolve(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

        if not improved:
            stall += 1
            # Random perturbation
            if stall > 30:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_convolve(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_u_shape(n, sharpness=2.0, asymmetry=0.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2 + asymmetry * x
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """Main search: soft-max GD + LP-FW + coordinate descent + length exploration."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = safe_evaluate(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = safe_evaluate(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 6000
        best_seq = make_u_shape(n_base, sharpness=2.5, asymmetry=0.1)
        best_score = safe_evaluate(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    a_start = norm_sum1(best_seq)

    # === Phase 1: Soft-max gradient descent (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.45, 750.0)
    print(f"\n=== Phase 1: Soft-max GD (n={n0}, t={t_phase1:.0f}s) ===")
    if t_phase1 > 30:
        a_out, _ = softmax_gradient_multi(a_start, t_phase1, n_trials=3, verbose=True)
        try_update(a_out, "sgd")

    # === Phase 2: LP-FW refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}, t_avail={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 30 and n_best <= 7000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True)
        try_update(a_out, "lp1")

    # === Phase 3: Another round of soft-max GD (sharper tau) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Soft-max GD round 2 (t_avail={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = softmax_gradient_multi(a_in, t3, n_trials=2, verbose=True)
        try_update(a_out, "sgd2")

    # === Phase 4: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW round 2 (t_avail={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20 and n_best <= 7000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True)
        try_update(a_out, "lp2")

    # === Phase 5: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coordinate polish (t_avail={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 150.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True, rng_seed=456)
        try_update(a_out, "coord")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration (t_avail={remaining:.0f}s) ===")

    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 1500, -1500]:
        new_n = n_best + delta
        if 3000 <= new_n <= 9000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)...")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.4, 120.0)
        if t6a > 15:
            a_out, _ = softmax_gradient_descent(a_long, t6a, rng_seed=new_n % 1000, verbose=False)
            try_update(a_out, f"sgd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 6000:
            t6b = min(remaining * 0.4, 100.0)
            if t6b > 15:
                a_in = norm_sum1(best_seq)
                a_interp = interpolate_sequence(a_in, new_n)
                a_out, _ = lp_frank_wolfe(a_interp, t6b, solver, verbose=False)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (t_avail={remaining:.0f}s) ===")
    if remaining > 20:
        t7 = min(remaining * 0.5, 100.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = softmax_gradient_descent(a_in, t7, rng_seed=999, verbose=False)
            try_update(a_out, "sgd_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.6, 80.0)
            if t7b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_polish(a_in, t7b, verbose=False, rng_seed=789)
                try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = safe_evaluate(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 6000
        best_seq = make_u_shape(n_fb, sharpness=2.5)
        best_score = safe_evaluate(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
