def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_lp_constraint_matrix(a, active_list, n):
    K = len(active_list)
    if K == 0:
        return None
    coeffs = np.zeros((K, n), dtype=np.float64)

        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]
    return coeffs

def mirror_descent_main(a0, time_limit, top_k=60, temp_base=40.0, c0=3.0, verbose=True):
    """Main optimization loop using mirror descent with adaptive top-k subgradient."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    x_prev = x.copy()
    stall = 0
    max_stall = 6000

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        ac = fast_autocorr(y)
        size = len(ac)
        k_eff = min(top_k, size)
        idx = np.argsort(ac)[::-1][:k_eff]
        top_vals = ac[idx]

        # Adaptive temperature: scale so that top few peaks have significant weight
        if len(top_vals) > 1 and top_vals[0] > 1e-15:
            spread = max(top_vals[0] - top_vals[min(10, len(top_vals)-1)], 1e-15)
            temp = max(temp_base * spread / max(ac.max(), 1e-15), 1.0)
        else:
            temp = 40.0

        w = np.exp(temp * (top_vals - top_vals[0]))
        w = w / w.sum()

        # Compute subgradient
        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            idx_arr = np.arange(i_s, i_e)
            grad[idx_arr] += 2.0 * wi * y[ki - idx_arr]

        # Adaptive learning rate
        eta_t = c0 / (np.sqrt(max(t, 1)) + 1.0)

        # Mirror descent update (exponential family)
        log_a = np.log(y) - eta_t * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        val = obj_val(x)
        if val < best_val - 1e-10:
            best_val = val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Periodic restart from best with noise
        if stall > 2000 and stall % 2000 == 0:
            rng = np.random.default_rng(t)
            noise_scale = 0.01 + 0.02 * (stall // 2000)
            noise = noise_scale * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            stall = 0

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    MirrorDes t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(best_val) + " stall=" + str(stall))

    return best_a, best_val

def lp_fw_refine(a0, time_limit, solver, init_K=300, max_K=2500, add_per=20, verbose=True):
    """LP-Frank-Wolfe refinement with tighter active sets."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    ac_y = fast_autocorr(y)
    active_set = list(np.argsort(ac_y)[::-1][:init_K])
    no_improve = 0
    stall_limit = 800

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep = np.argsort(margins)[:max_K]
            active_set = list(active_set[i] for i in keep)

        K = len(active_set)
        if K == 0:
            break

        coeffs = build_lp_constraint_matrix(y, active_set, n)
        if coeffs is None:
            break

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        lp_ok = False
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                lp_ok = True
        except Exception:
            if len(active_set) > init_K:
                active_set = list(active_set[:init_K])

        if not lp_ok:
            if no_improve > stall_limit // 2:
                break
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations and add new constraints
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per]]
            for k in top_v:
                if int(k) not in active_set:
                    active_set.append(int(k))

        # Line search: combine of grid + bisection
        best_alpha = 2.0 / (t + 1.0)
        best_trial = fast_autocorr((1 - best_alpha) * y + best_alpha * g_dir).max()

        alphas = [1.0 / (t + 1.0), 2.0 / (t + 2.0), 0.5, 0.8, 1.0]
        for alpha in alphas:
            alpha = max(0.01, min(1.0, alpha))
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            Mt = fast_autocorr(trial).max()
            if Mt < best_trial - 1e-15:
                best_trial = Mt
                best_alpha = alpha

        # Bisection refinement
        lo = max(0.0, best_alpha - 0.2)
        hi = min(1.0, best_alpha + 0.2)
        for _ in range(6):
            mid1 = lo + (hi - lo) / 3.0
            mid2 = hi - (hi - lo) / 3.0
            t1 = np.maximum((1 - mid1) * y + mid1 * g_dir, 0.0)
            t2 = np.maximum((1 - mid2) * y + mid2 * g_dir, 0.0)
            s1, s2 = t1.sum(), t2.sum()
            if s1 > 1e-15:
                t1 = t1 / s1
            if s2 > 1e-15:
                t2 = t2 / s2
            m1 = fast_autocorr(t1).max() if s1 > 1e-15 else 1e10
            m2 = fast_autocorr(t2).max() if s2 > 1e-15 else 1e10
            if m1 < m2:
                hi = mid2
            else:
                lo = mid1
        best_alpha = (lo + hi) / 2.0

        if t == 1:
            y = g_dir.copy()
        else:
            y = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(2 * n * best_m) + " |S|=" + str(len(active_set)))

    return best_a, 2.0 * n * best_m

def coord_descent_polish(a0, time_limit, seed=42, verbose=True):
    """Aggressive coordinate descent with golden-section search on transfer amounts."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 1500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score destinations: how much would moving mass here increase other peaks?
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            # Find best destination
            mask = np.ones(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                mask[s2:e2 + 1] = False
            mask[src] = False
            candidates = np.where(mask)[0]
            if len(candidates) < 10:
                candidates = np.where(mask)[0] if len(mask) > 0 else np.arange(n)
            if len(candidates) == 0:
                candidates = np.arange(n)
                candidates = candidates[candidates != src]

            dst = int(candidates[np.argmin(scores[candidates])]) if len(candidates) > 0 else int(rng.integers(n))
            if dst == src:
                continue

            # Golden section search for optimal transfer amount
            lo_amt = 0.0
            hi_amt = src_val * 0.5
            if hi_amt < 1e-15:
                continue

            def trial_eval(frac):
                amt = src_val * frac
                if amt < 1e-15 or amt >= src_val:
                    return 1e10
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    return 1e10
                a_t = a_t / st
                return fast_autocorr(a_t).max()

            # Quick grid
            fracs = [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]
            best_frac = 0.0
            best_mt = trial_eval(0.01) if 0.01 < 0.5 else 1e10
            for f in fracs:
                if f > src_val * 0.5 / max(src_val, 1e-15):
                    continue
                mt = trial_eval(f)
                if mt < best_mt:
                    best_mt = mt
                    best_frac = f

            if best_frac > 0 and best_mt < best_m - 1e-15:
                amt = src_val * best_frac
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                best_m = best_mt
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1
            if stall % 100 == 0 and stall > 50:
                # Random perturbation to escape
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.005)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 500 == 0:
            print("    Coord (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(2 * n * best_m) + " stall=" + str(stall))

    return best_a, 2.0 * n * best_m

def peak_deflation(a0, time_limit, n_targets=8, verbose=True):
    """Targeted deflation of the highest peaks."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)

    for round_i in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        targets = np.argsort(ac)[::-1][:n_targets]
        improved = False

        for k in targets[:5]:
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            n_attempts = 4

            for _ in range(n_attempts):
                if contrib.max() < 1e-18:
                    break
                src = int(idx[np.argmax(contrib)])
                src_val = a[src]
                if src_val < 1e-12:
                    break

                # Score all positions for destination
                scores = np.zeros(n)
                for k2 in targets[:6]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                        scores[k2i - idx2] += a[idx2]

                scores[src] = 1e10
                dst = int(np.argmin(scores))
                if dst == src:
                    break

                # Try several transfer fractions
                found = False
                for frac in [0.003, 0.006, 0.01, 0.015, 0.02]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-15:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        contrib = a[idx] * a[ki - idx]
                        found = True
                        break
                if not found:
                    break

        if not improved:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.001, 0.004)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    Mn = fast_autocorr(a).max()
                    if Mn < best_m:
                        best_m = Mn
                        best_a = a.copy()
                        improved = True

        if verbose and round_i % 100 == 0:
            print("    Deflate round=" + str(round_i) + " J=" + repr(2 * n * best_m))

    return best_a, 2.0 * n * best_m

def parametric_sweep(n, best_init, best_score, verbose=True):
    """Diverse parametric initialization."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = normalize(best_init)
    best_val = obj_val(best_a)

    candidates = []

    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(normalize((i_arr + 1.0) ** p))
        candidates.append(normalize((n - i_arr) ** p))
        candidates.append(normalize(((i_arr + 1.0) / (n + 1.0)) ** p + 0.1 * ((n - i_arr) / (n + 1.0)) ** p))

    # Gaussians
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sg in [0.05, 0.1, 0.15, 0.2, 0.3]:
            candidates.append(normalize(np.exp(-((x - c) ** 2) / (2 * sg * sg))))

    # Double bump
    for c1 in [0.15, 0.25, 0.3]:
        for c2 in [0.7, 0.75, 0.85]:
            for sg in [0.05, 0.08, 0.12, 0.15]:
                cnd = np.exp(-((x - c1) ** 2) / (2 * sg * sg)) + np.exp(-((x - c2) ** 2) / (2 * sg * sg))
                candidates.append(normalize(cnd))

    # Triangular / V-shape
    for w in [0.1, 0.2, 0.3, 0.4]:
        candidates.append(normalize(np.ones(n) - w * np.sin(np.pi * x) ** 2))
        candidates.append(normalize(w + (1 - w) * (1 - np.sin(np.pi * x) ** 2)))

    # Sine waves
    for f in [1, 2, 3, 4, 5, 7]:
        for amp in [0.3, 0.5, 0.7]:
            cnd = np.maximum(1.0 + amp * np.cos(2 * np.pi * f * x), 0.01)
            candidates.append(normalize(cnd))

    # Symmetric power
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(normalize(x ** p + (1.0 - x) ** p))
        candidates.append(normalize(np.maximum(np.minimum(x, 1 - x) ** p, 0.001)))

    # Blend with best init
    bi = normalize(best_init)
    for w in [0.3, 0.5, 0.7]:
        for cand in candidates[:6]:
            candidates.append(normalize(w * bi + (1.0 - w) * cand))

    for c in candidates:
        v = obj_val(c)
        if v < best_val - 1e-12:
            best_val = v
            best_a = c.copy()
            if verbose:
                print("    Param: J=" + repr(v))

    return best_a, best_val

def propose_candidate(budget_s=1800.0):
    """Main search function with multi-phase optimization."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)
    solver = get_solver()
    print("Budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            print("Parent: n=" + str(len(p)) + " score=" + repr(best_score))
    except Exception:
        best_seq = None

    if best_seq is None:
        n_init = 8000
        x = np.linspace(0, 1, n_init)
        best_seq = normalize(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * x), 0.01))
        best_score = eval_safe(best_seq)
        print("Default init n=" + str(n_init) + " score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: Parametric sweep at current length
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Parametric (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.02, 30.0)
    if t1 > 5:
        a_p, v_p = parametric_sweep(n0, best_seq, best_score, verbose=True)
        sc = eval_safe(a_p)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_p
            print("  Param improved: " + repr(best_score))

    # Phase 2: Mirror descent main loop
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P2: Mirror Descent (n=" + str(n_cur) + ") ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 60:
        a_out, v_out = mirror_descent_main(normalize(best_seq), t2, top_k=60, temp_base=40.0, c0=3.0, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  MirrorDes improved: " + repr(best_score))

    # Phase 3: LP-FW refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P3: LP-FW (n=" + str(n_cur) + ") ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40:
        a_out, v_out = lp_fw_refine(normalize(best_seq), t3, solver,
                                     init_K=400, max_K=3000, add_per=30, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  LP-FW improved: " + repr(best_score))

    # Phase 4: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P4: Coord Descent (n=" + str(n_cur) + ") ===")
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 30:
        a_out, v_out = coord_descent_polish(normalize(best_seq), t4, seed=42, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  Coord improved: " + repr(best_score))

    # Phase 5: Peak deflation
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P5: Peak Deflation (n=" + str(n_cur) + ") ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_out, v_out = peak_deflation(normalize(best_seq), t5, n_targets=8, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  Deflate improved: " + repr(best_score))

    # Phase 6: Try nearby lengths
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P6: Length exploration (n=" + str(n_cur) + ") ===")
    length_offsets = [-300, -100, 100, 300, 500]
    for offset in length_offsets:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        n_new = n_cur + offset
        if n_new < 100 or n_new > 30000:
            continue
        t_test = min(remaining * 0.15, 100.0)
        if t_test < 20:
            continue
        print("  Trying n=" + str(n_new))
        a_interp = interpolate_seq(best_seq, n_new)
        # Quick mirror descent at new length
        a_out, v_out = mirror_descent_main(a_interp, t_test, top_k=50, temp_base=35.0, c0=2.5, verbose=False)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            n_cur = n_new
            print("  Length " + str(n_new) + " improved: " + repr(best_score))

    # Phase 7: Final LP-FW on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P7: Final LP-FW (n=" + str(n_cur) + ") ===")
    t7 = min(remaining * 0.15, 200.0)
    if t7 > 30:
        a_out, v_out = lp_fw_refine(normalize(best_seq), t7, solver,
                                     init_K=500, max_K=3500, add_per=40, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  Final LP improved: " + repr(best_score))

    # Phase 8: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P8: Final Coord (n=" + str(n_cur) + ") ===")
    t8 = min(remaining * 0.20, 200.0)
    if t8 > 20:
        a_out, v_out = coord_descent_polish(normalize(best_seq), t8, seed=99, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  Final Coord improved: " + repr(best_score))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + " n=" + str(len(best_seq)) + " t=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR fallback")
        n_fb = 8000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * x), 0.01))
        best_score = eval_safe(best_seq)

    # Ensure all values are valid finite non-negative floats
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0.0, 1000.0)
    return [float(x) for x in best_seq]
