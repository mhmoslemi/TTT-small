def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s

def self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]

def conv_ab(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]

def quad_max(alpha, q0, c, qd):
    a1 = 1.0 - alpha
    val = (a1 * a1) * q0 + (2.0 * alpha * a1) * c + (alpha * alpha) * qd

def golden_alpha_max(q0, c, qd, a, b, iters=40):
    phi = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = b - phi * (b - a)
    d1 = a + phi * (b - a)
    f1 = quad_max(c1, q0, c, qd)
    f2 = quad_max(d1, q0, c, qd)

            d1 = c1
            f2 = f1

            c1 = d1
            f1 = f2
            d1 = a + phi * (b - a)

def solve_lp_for_direction(y, active, q_active):
    """Solve the LP to find the FW direction."""
    y = np.asarray(y, dtype=np.float64)
    active = np.asarray(active, dtype=int)
    q_active = np.asarray(q_active, dtype=np.float64)
    n = y.size
    K = active.size
    if K <= 0 or n > 12000:
        return None
    
    try:
        A = np.zeros((K, n + 1), dtype=np.float64)
    except Exception:
        return None
    
    yy = 2.0 * y
    for r, k in enumerate(active):
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            A[r, i0:i1] = yy[start:stop][::-1]
    
    A[:, n] = -1.0
    b = np.maximum(q_active, 0.0)
    
    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0
    Aeq = np.ones((1, n + 1), dtype=np.float64)
    Aeq[0, n] = 0.0
    bounds = [(0.0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=np.array([1.0]),
                      bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30.0})
    except Exception:
        return None
    
    if not getattr(res, 'success', False):
        return None
    
    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-10:
        return None
    g = g / s
    return g

def mirror_descent_step(a, q, trust, focus):
    """One step of mirror descent."""
    n = a.size
    m = float(q.max())
    if m <= 1e-15:
        return a.copy()
    scale = 0.5 * m
    gamma = focus / scale
    vals = q - m
    vals = np.clip(gamma * vals, -50.0, 0.0)
    w = np.exp(vals)
    s = float(w.sum())
    if s <= 1e-300:
        return a.copy()
    w = w / s
    
    # Cross-correlation: c[j] = sum_m w[m] * a[m-j]
    L = w.size
    size = next_pow2(L + n - 1)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fw * fa.conj(), size)[:n]
    c = np.maximum(c, 0.0)
    
    cmax = float(np.max(c))
    cmin = float(np.min(c))
    scale2 = cmax - cmin
    if scale2 <= 1e-15:
        return a.copy()
    
    move = trust * (cmax - c) / scale2
    move = np.clip(move, -50.0, 50.0)
    b = a * np.exp(move)
    b = normalize_vec(b)
    return b

def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())

    return a / s

def self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]

def conv_ab(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = 2 * n - 1

    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]

def evaluate_safe(a):
    try:
        x = [float(v) for v in np.asarray(a, dtype=np.float64)]
        val = evaluate_sequence(x)
        if np.isfinite(val):
            return float(val)
    except Exception:
        pass
    return float('inf')

def quad_max(alpha, q0, c, qd):
    a1 = 1.0 - alpha
    val = (a1 * a1) * q0 + (2.0 * alpha * a1) * c + (alpha * alpha) * qd
    return float(np.max(val))

def golden_alpha_max(q0, c, qd, a, b, iters=45):
    phi = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = b - phi * (b - a)
    d1 = a + phi * (b - a)
    f1 = quad_max(c1, q0, c, qd)
    f2 = quad_max(d1, q0, c, qd)
    for _ in range(iters):
        if f1 < f2:
            b = d1
            d1 = c1
            f2 = f1
            c1 = b - phi * (b - a)
            f1 = quad_max(c1, q0, c, qd)
        else:
            a = c1
            c1 = d1
            f1 = f2
            d1 = a + phi * (b - a)
            f2 = quad_max(d1, q0, c, qd)
    return 0.5 * (a + b)

def line_search_direction(a, d, q0, T_hint=None):
    a = np.asarray(a, dtype=np.float64)
    d = np.asarray(d, dtype=np.float64)
    q0 = np.asarray(q0, dtype=np.float64)
    M0 = float(np.max(q0))

    dot = float(np.dot(a, d))
    if dot > 1.0 - 1e-7:
        return a.copy(), M0

    c = conv_ab(a, d)
    qd = self_conv(d)

    best_alpha = 0.0
    best_M = M0

    alphas = [5e-5, 1e-4, 2e-4, 5e-4, 1e-3, 2e-3, 5e-3, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]

    if T_hint is not None and np.isfinite(T_hint) and T_hint < 0.0 and M0 > 1e-15:
        t1 = min(0.35, max(5e-4, 0.6 * (-float(T_hint)) / M0))
        for mult in (0.3, 0.6, 0.9, 1.2, 1.5):
            al = min(0.98, max(1e-5, mult * t1))
            alphas.append(al)

    uniq = []
    seen = set()
    for al in alphas:
        al = float(min(1.0, max(0.0, al)))
        key = round(al, 12)
        if key not in seen:
            seen.add(key)
            uniq.append(al)

    for al in uniq:
        val = quad_max(al, q0, c, qd)
        if val < best_M:
            best_M = val
            best_alpha = al

    if best_M < M0 - 1e-13 and 0.0 < best_alpha < 1.0:
        al2 = golden_alpha_max(q0, c, qd, 0.0, 1.0, 45)
        val2 = quad_max(al2, q0, c, qd)
        if val2 < best_M:
            best_M = val2
            best_alpha = al2

        # Also try golden section in a narrower range around best
        lo = max(0.0, best_alpha - 0.05)
        hi = min(1.0, best_alpha + 0.05)
        if hi - lo > 1e-6:
            al3 = golden_alpha_max(q0, c, qd, lo, hi, 30)
            val3 = quad_max(al3, q0, c, qd)
            if val3 < best_M:
                best_M = val3
                best_alpha = al3

    if best_alpha <= 0.0 or best_M >= M0 - 1e-14:
        return a.copy(), M0

    new = (1.0 - best_alpha) * a + best_alpha * d
    new = np.maximum(new, 0.0)
    s = float(new.sum())
    if s <= 1e-15:
        return a.copy(), M0
    new = new / s
    return new, best_M

def solve_lp_direction(y, active, q_active, time_limit=20.0):
    y = np.asarray(y, dtype=np.float64)
    active = np.asarray(active, dtype=int)
    q_active = np.asarray(q_active, dtype=np.float64)
    n = y.size
    K = active.size
    if K <= 0 or n > 15000:
        return None, None

    try:
        A = np.zeros((K, n + 1), dtype=np.float64)
    except Exception:
        return None, None

    yy = 2.0 * y
    for r, k in enumerate(active):
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            A[r, i0:i1] = yy[start:stop][::-1]

    A[:, n] = -1.0
    b = q_active.copy()

    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0
    Aeq = np.ones((1, n + 1), dtype=np.float64)
    Aeq[0, n] = 0.0
    bounds = [(0.0, None)] * n + [(None, None)]

    t_start = time.monotonic()
    try:
        sl = min(time_limit, max(5.0, 0.3 * (time_limit - (time.monotonic() - t_start))))
        res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=np.array([1.0]),
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': sl})
    except Exception:
        return None, None

    if not getattr(res, 'success', False):
        return None, None

    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-10:
        return None, None
    g = g / s
    T = float(res.x[n])
    if not np.isfinite(T):
        T = 0.0
    return g, T

def mirror_direction(a, q, trust, focus):
    n = a.size
    m = float(q.max())
    if m <= 1e-15:
        return a.copy()
    scale = 0.5 * m
    gamma = focus / scale
    vals = q - m
    vals = np.clip(gamma * vals, -50.0, 0.0)
    w = np.exp(vals)
    s = float(w.sum())
    if s <= 1e-300:
        return a.copy()
    w = w / s

    L = w.size
    size = next_pow2(L + n - 1)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fw * fa.conj(), size)[:n]
    c = np.maximum(c, 0.0)

    cmax = float(np.max(c))
    cmin = float(np.min(c))
    scale2 = cmax - cmin
    if scale2 <= 1e-15:
        return a.copy()

    move = trust * (cmax - c) / scale2
    move = np.clip(move, -50.0, 50.0)
    b = a * np.exp(move)
    return normalize_vec(b)

def fw_optimize(a0, time_limit, verbose=False):
    t0 = time.monotonic()
    y = normalize_vec(a0)
    n = y.size
    if n <= 5:
        return y

    best = y.copy()
    best_score = objective_norm(y)

    if n > 9000:
        initK = 250
        maxK = 700
    elif n > 5000:
        initK = 400
        maxK = 1000
    else:
        initK = 500
        maxK = 1500

    K = initK
    no_improve = 0
    iter_no = 0
    last_print = t0
    lp_count = 0
    lp_fails = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining <= 8.0:
            break

        qy = self_conv(y)
        M_y = float(np.max(qy))
        score_y = 2.0 * n * M_y

        if score_y < best_score - 1e-12:
            best = y.copy()
            best_score = score_y

        if no_improve > 350 and time.monotonic() - t0 > 0.85 * time_limit:
            break

        order = np.argsort(qy)[::-1]
        total_lags = len(qy)

        # Try multiple active set sizes
        best_direction = None
        best_T = 0.0
        best_dir_score = float('inf')

        K_options = []
        K_cur = int(min(K, total_lags, 2000))
        K_options.append(K_cur)
        K_options.append(int(min(K_cur // 2, total_lags, 1000)))
        K_options.append(int(min(K_cur + 200, total_lags, 2000)))
        # Remove duplicates and sort descending
        K_options = sorted(set(max(20, k) for k in K_options), reverse=True)

        used_lp = False
        for Kopt in K_options[:3]:
            if remaining < 15:
                break
            act = order[:Kopt]
            q_act = qy[act]
            t_lp = min(15.0, max(5.0, 0.2 * remaining))
            g_dir, T = solve_lp_direction(y, act, q_act, time_limit=t_lp)
            if g_dir is not None:
                lp_count += 1
                used_lp = True
                # Evaluate quality: smaller T is better
                new_test, M_test = line_search_direction(y, g_dir, qy, T)
                if M_test < best_dir_score:
                    best_dir_score = M_test
                    best_direction = g_dir
                    best_T = T
                    best_new = new_test
                if M_test < M_y - 1e-13:
                    break  # Good enough, stop trying other active sets
            else:
                lp_fails += 1

        if used_lp and best_direction is not None and best_new is not None:
            if best_new is not None and best_dir_score < M_y - 1e-13:
                y = best_new
                no_improve = 0
            else:
                no_improve += 1
                # Fallback to mirror descent
                if no_improve in (1, 3, 7, 15, 30):
                    for trust in (0.3, 0.8, 2.0, 5.0):
                        for foc in (4.0, 10.0, 25.0):
                            d = mirror_direction(y, qy, trust, foc)
                            new2, M2 = line_search_direction(y, d, qy, None)
                            if M2 < M_y - 1e-13:
                                y = new2
                                no_improve = 0
                                break
                        if no_improve == 0:
                            break
        else:
            # Mirror descent as primary
            improved = False
            for trust in (0.3, 0.8, 2.0, 5.0, 10.0):
                for foc in (4.0, 10.0, 25.0, 50.0):
                    d = mirror_direction(y, qy, trust, foc)
                    new, Mnew = line_search_direction(y, d, qy, None)
                    if Mnew < M_y - 1e-13:
                        y = new
                        no_improve = 0
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                no_improve += 1
                K = min(K + 40, maxK)

        if no_improve > 0 and no_improve % 15 == 0:
            K = min(K + 50, maxK)

        iter_no += 1
        now = time.monotonic()
        if verbose and (iter_no % 30 == 0 or (now - last_print) > 30.0):
            print("  FW: it=%d elapsed=%.1fs best=%.9f cur=%.9f K=%d noimp=%d lp=%d"
                  % (iter_no, now - t0, best_score, score_y, K, no_improve, lp_count))
            last_print = now

    return best

def smooth_polish(a0, time_limit, rng_seed=11, verbose=False):
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 1:
        return a

    best = a.copy()
    best_score = objective_norm(a)
    trust = 0.6
    focus = 5.0
    fails = 0
    restarts = 0
    rng = np.random.default_rng(rng_seed)

    while time.monotonic() - t0 < time_limit:
        q = self_conv(a)
        M = float(np.max(q))
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            fails = 0
        else:
            fails += 1

        if fails > 600:
            break

        if fails > 50 and fails % 25 == 0:
            focus = min(focus * 1.15, 100.0)
            trust = max(trust * 0.82, 0.02)

        if fails > 150 and fails % 150 == 0:
            restarts += 1
            if restarts > 40:
                break
            scale_noise = 0.02 / (1.0 + 0.06 * restarts)
            cand = best + scale_noise * rng.normal(size=best.size) * best
            cand = np.maximum(cand, 1e-12)
            cand = normalize_vec(cand)
            sc = objective_norm(cand)
            if np.isfinite(sc) and sc <= best_score * 1.001:
                a = cand
                fails = 0
                trust = 0.5
                focus = min(focus * 1.1, 100.0)
            else:
                a = best.copy()

        d = mirror_direction(a, q, trust, focus)
        new, Mnew = line_search_direction(a, d, q, None)

        if Mnew < M - 1e-14:
            a = new
            score_new = 2.0 * n * Mnew
            if score_new < best_score - 1e-12:
                trust = min(trust * 1.08, 10.0)
            else:
                trust = min(trust * 1.04, 10.0)
        else:
            trust = max(trust * 0.75, 0.02)
            focus = min(focus * 1.1, 100.0)

        if verbose and (fails % 100 == 0 or time.monotonic() - t0 - t0 > 30.0):
            print("    polish: elapsed=%.1fs best=%.9f fails=%d" % (time.monotonic() - t0, best_score, fails))

    return best

def interpolate_sequence(f, new_n):
    f = normalize_vec(f)
    old_n = f.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return f.copy()
    if old_n <= 1:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f)
    return normalize_vec(g)

def transform_search(a0, time_limit):
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    best = a.copy()
    best_score = objective_norm(a)

    # Power transforms
    for g in (0.2, 0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0, 5.0):
        if time.monotonic() - t0 > time_limit:
            break
        b = np.maximum(a, 1e-30) ** g
        b = normalize_vec(b)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    # Blend with uniform
    u = np.ones(n) / n
    for w in (0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5):
        if time.monotonic() - t0 > time_limit:
            break
        b = normalize_vec((1.0 - w) * best + w * u)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    # Symmetrize
    rev = best[::-1].copy()
    b = normalize_vec((best + rev) / 2.0)
    sc = objective_norm(b)
    if sc < best_score - 1e-12:
        best_score = sc
        best = b.copy()

    # Blend with reverse at various weights
    rev = best[::-1].copy()
    for w in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7):
        if time.monotonic() - t0 > time_limit:
            break
        b = normalize_vec((1.0 - w) * best + w * rev)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    return best

def coordinate_polish(a0, time_limit, rng_seed=991):
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 10 or time_limit <= 0.0:
        return a

    best = a.copy()
    best_score = objective_norm(a)
    stall = 0
    rng = np.random.default_rng(rng_seed)

    while time.monotonic() - t0 < time_limit and stall < 150:
        q = self_conv(a)
        peak_idx = int(np.argmax(q))
        k = peak_idx
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)

        if i0 >= i1:
            break

        importance = np.zeros(n)
        j_arr = np.arange(i0, i1)
        paired = j_arr - k
        paired_clipped = np.clip(paired, 0, n - 1)
        importance[j_arr] = a[j_arr] * a[paired_clipped]

        src_candidates = np.argsort(importance)[::-1][:10]
        dst_candidates = np.argsort(importance)[:10]

        improved = False
        for s_idx in src_candidates:
            if time.monotonic() - t0 > time_limit or improved:
                break
            s_pos = int(s_idx)
            if importance[s_pos] <= 1e-15:
                continue
            for d_idx in dst_candidates:
                d_pos = int(d_idx)
                if d_pos == s_pos:
                    continue
                for frac in (0.03, 0.05, 0.1, 0.15, 0.25):
                    amt = a[s_pos] * frac
                    if amt <= 1e-15:
                        continue
                    b = a.copy()
                    b[s_pos] = a[s_pos] - amt
                    b[d_pos] = a[d_pos] + amt
                    b = np.maximum(b, 0.0)
                    ss = float(b.sum())
                    if ss <= 1e-15:
                        continue
                    b = b / ss
                    sc = objective_norm(b)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best = b.copy()
                        a = b
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 15 == 0:
                for _ in range(3):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        b = a.copy()
                        amt = a[i] * 0.05
                        b[i] -= amt
                        b[j] += amt
                        b = np.maximum(b, 0.0)
                        ss = b.sum()
                        if ss > 1e-15:
                            b = b / ss
                            sc = objective_norm(b)
                            if sc < best_score - 1e-13:
                                best_score = sc
                                best = b.copy()
                                a = b
                                stall = 0
                                break

    return best

def trim_search(a0, time_limit):
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n < 50 or time_limit <= 0.0:
        return a

    best = a.copy()
    best_score = objective_norm(a)
    max_side = min(n // 3, 3000)

    step = max(1, max_side // 30)
    for l in range(0, max_side + 1, step):
        if time.monotonic() - t0 > time_limit:
            break
        if n - l < 20:
            continue
        b = normalize_vec(a[l:])
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    for r in range(1, max_side + 1, step):
        if time.monotonic() - t0 > time_limit:
            break
        if n - r < 20:
            continue
        b = normalize_vec(a[:n - r])
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()

    return best

def spectral_init(n, rng):
    x = np.linspace(0, 1, n, endpoint=False)
    candidates = [
        np.sin(np.pi * x) ** 2,
        np.sin(np.pi * x) ** 3,
        np.exp(-4.0 * (x - 0.5) ** 2),
        np.exp(-8.0 * (x - 0.5) ** 2),
        np.exp(-12.0 * (x - 0.5) ** 2),
        1.0 / (1.0 + 30.0 * (x - 0.5) ** 2),
        (x * (1.0 - x)) ** 0.5,
        (x * (1.0 - x)) ** 0.7,
        np.cos(np.pi * (x - 0.5)) ** 2,
        np.cos(np.pi * (x - 0.5)) ** 4,
        np.cos(np.pi * (x - 0.5)) ** 6,
    ]

    best = None
    best_score = float('inf')
    for seq_raw in candidates:
        seq = np.maximum(seq_raw, 0.0) + 1e-10
        seq = normalize_vec(seq)
        sc = objective_norm(seq)
        if sc < best_score:
            best_score = sc
            best = seq.copy()

    for _ in range(15):
        freqs = rng.integers(1, 10, size=rng.integers(2, 6))
        phases = rng.uniform(0, 2 * np.pi, size=len(freqs))
        amps = rng.uniform(0.1, 1.0, size=len(freqs))
        seq = np.ones(n)
        for i in range(len(freqs)):
            seq += amps[i] * np.cos(2 * np.pi * freqs[i] * x + phases[i])
        seq = np.maximum(seq, 0.0) + 1e-10
        seq = normalize_vec(seq)
        sc = objective_norm(seq)
        if sc < best_score:
            best_score = sc
            best = seq.copy()

    if best is None:
        best = np.ones(n) / n
    return best

def propose_candidate(budget_s=1800.0):
    """Main search: multi-phase optimization with FW, mirror descent, and structural moves."""
    t_global = time.monotonic()
    total_budget = min(1800.0, float(budget_s)) - 80.0
    total_budget = max(120.0, total_budget)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if p.ndim != 1:
            p = p.ravel()
        if p.size > 0 and p.sum() > 1e-12:
            parent = p
    except Exception:
        parent = None

    if parent is not None:
        parent_n = parent.size
        print("Parent loaded: n=%d" % parent_n)
    else:
        parent_n = 6700
        x = np.linspace(0, 1, parent_n)
        parent = normalize_vec(np.sin(np.pi * x) ** 2)
        print("Using default parent: n=%d" % parent_n)

    best_seq = normalize_vec(parent)
    best_score = objective_norm(best_seq)
    print("Initial best: score=%.9f n=%d" % (best_score, best_seq.size))

    # Phase 1: Main FW on parent
    remaining = total_budget - (time.monotonic() - t_global)
    fw_time = min(900.0, 0.55 * remaining)
    if fw_time > 60:
        print("Phase 1: FW optimize (%.0f sec)" % fw_time)
        out = fw_optimize(best_seq, fw_time, verbose=True)
        sc = objective_norm(out)
        if sc < best_score:
            print("*** NEW BEST Phase1: %.9f n=%d" % (sc, out.size))
            best_score = sc
            best_seq = out

    # Phase 2: Mirror descent polish
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 100:
        pol_time = min(200.0, 0.25 * remaining)
        if pol_time > 40:
            print("Phase 2: Smooth polish (%.0f sec)" % pol_time)
            out = smooth_polish(best_seq, pol_time, rng_seed=42, verbose=False)
            sc = objective_norm(out)
            if sc < best_score:
                print("*** NEW BEST Phase2: %.9f n=%d" % (sc, out.size))
                best_score = sc
                best_seq = out

    # Phase 3: FW polish (second pass)
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 100:
        fw_time2 = min(400.0, 0.4 * remaining)
        if fw_time2 > 50:
            print("Phase 3: FW polish (%.0f sec)" % fw_time2)
            out = fw_optimize(best_seq, fw_time2, verbose=False)
            sc = objective_norm(out)
            if sc < best_score:
                print("*** NEW BEST Phase3: %.9f n=%d" % (sc, out.size))
                best_score = sc
                best_seq = out

    # Phase 4: Transform search
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 50:
        tf_time = min(50.0, 0.3 * remaining)
        if tf_time > 10:
            print("Phase 4: Transform search")
            out = transform_search(best_seq, tf_time)
            sc = objective_norm(out)
            if sc < best_score:
                print("*** NEW BEST Phase4: %.9f" % sc)
                best_score = sc
                best_seq = out

    # Phase 5: Coordinate polish
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 50:
        cp_time = min(120.0, 0.5 * remaining)
        if cp_time > 20:
            print("Phase 5: Coordinate polish (%.0f sec)" % cp_time)
            out = coordinate_polish(best_seq, cp_time, rng_seed=314159)
            sc = objective_norm(out)
            if sc < best_score:
                print("*** NEW BEST Phase5: %.9f" % sc)
                best_score = sc
                best_seq = out

    # Phase 6: Try alternate length
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 200:
        cur_n = best_seq.size
        for dn in (500, -500, 1000, -1000, 1500):
            if remaining < 80:
                break
            alt_n = cur_n + dn
            if 3000 <= alt_n <= 12000 and alt_n != cur_n:
                alt = interpolate_sequence(best_seq, alt_n)
                alt_time = min(150.0, 0.5 * remaining)
                if alt_time > 30:
                    out = fw_optimize(alt, alt_time, verbose=False)
                    sc = objective_norm(out)
                    if sc < best_score:
                        print("*** NEWBEST Phase6 (n=%d): %.9f" % (alt_n, sc))
                        best_score = sc
                        best_seq = out
                    remaining = total_budget - (time.monotonic() - t_global)
                    break

    # Phase 7: Final FW squeeze
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 60:
        fw_time3 = min(remaining - 10, 300.0)
        if fw_time3 > 30:
            print("Phase 7: Final FW (%.0f sec)" % fw_time3)
            out = fw_optimize(best_seq, fw_time3, verbose=False)
            sc = objective_norm(out)
            if sc < best_score:
                print("*** NEW BEST Phase7: %.9f" % sc)
                best_score = sc
                best_seq = out

    # Phase 8: Final transform + polish
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 30:
        out = transform_search(best_seq, min(20.0, remaining - 5))
        sc = objective_norm(out)
        if sc < best_score:
            print("*** NEW BEST Phase8: %.9f" % sc)
            best_score = sc
            best_seq = out

    # Final validation
    out_arr = np.asarray(best_seq, dtype=np.float64)
    out_arr = np.where(np.isfinite(out_arr), out_arr, 0.0)
    out_arr = np.maximum(out_arr, 0.0)
    out_arr = np.minimum(out_arr, 1000.0)
    s = float(out_arr.sum())
    if s <= 0.01:
        x = np.linspace(0, 1, 6700)
        out_arr = np.sin(np.pi * x) ** 2
        s = float(out_arr.sum())
    out_arr = out_arr / s

    final_score = objective_norm(out_arr)
    try:
        official = evaluate_safe(out_arr)
        if np.isfinite(official):
            final_score = min(final_score, official)
    except Exception:
        pass

    elapsed = time.monotonic() - t_global
    print("FINAL: score=%.10f n=%d time=%.1fs" % (final_score, out_arr.size, elapsed))

    return [float(v) for v in out_arr]
