def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)

    s = float(a.sum())

def self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]

def compute_soft_gradient(a, q, temperature):
    """Compute soft-max weighted gradient of max(q)."""
    n = a.size
    m = q.size
    qmax = float(np.max(q))
    if qmax <= 1e-15:
        return np.ones(n, dtype=np.float64) / n
    w = np.exp((q - qmax) / temperature)
    wsum = float(w.sum())
    if wsum <= 1e-300:
        w = np.zeros(m, dtype=np.float64)
        w[int(np.argmax(q))] = 1.0
    else:
        w = w / wsum
    # g_j = sum_k w[k] * (a[j-k] + a[j+k]) with boundary handling
    # Use FFT: g = conv(w_ext, a) + conv(w_ext_rev, a)
    size = next_pow2(m + n - 1)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    # First term: sum_k w[k] * a[j-k] = (w_rev * a)[j]
    w_rev = w[::-1].copy()
    fw_rev = np.fft.rfft(w_rev, size)
    t1 = np.fft.irfft(fw * fa, size)[:n]
    t2 = np.fft.irfft(fw_rev * fa, size)[:n]
    g = t1 + t2
    g = np.maximum(g, 0.0)
    return g

                            scores[jj] += a[ii] + 1e-10

                # Move a small fraction from the larger of a[i], a[j] to the lowest-score position
                if a[i] >= a[j]:
                    src = i
                else:
                    src = j
                amt = a[src] * 0.05
                if amt < 1e-15:
                    continue
                dst = int(np.argmin(scores))
                b = a.copy()
                b[src] -= amt
                b[dst] += amt
                b = np.maximum(b, 0.0)
                ss = float(b.sum())
                if ss <= 1e-15:
                    continue
                b = b / ss
                sc = objective_norm(b)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best = b.copy()
                    a = b
                    improved = True
                    break
            if improved:
                fails = 0
                break

        if not improved:
            fails += 1

    return best


def lp_constraint_gen(a, q, max_iter=30, tol=1e-8):
    """Solve LP with constraint generation for the FW direction."""
    n = a.size
    m = q.size
    if not HAVE_LP or n > 12000:
        return None

    # Start with top 3 lags
    top_k = np.argsort(q)[::-1][:3]
    active = set(int(k) for k in top_k)

    for _ in range(max_iter):
        K = len(active)
        if K > 50:
            # Too many constraints, just use top K by q value
            active_set = sorted(np.argsort(q)[::-1][:K])
        else:
            active_set = sorted(active)

        # Build LP: min T s.t. sum g = 1, g >= 0, 2*(a*g)[k] - T <= q(k) for k in active
        K = len(active_set)
        try:
            A = np.zeros((K, n + 1), dtype=np.float64)
        except Exception:
            return None

        for r, k in enumerate(active_set):
            k = int(k)
            i0 = max(0, k - n + 1)
            i1 = min(n, k + 1)
            if i0 < i1:
                # (a*g)[k] = sum_i a[i] * g[k-i] for valid i
                # Row r: sum_i a[i] * g[k-i] - T <= q(k)
                start = k - (i1 - 1)
                stop = k - i0 + 1
                A[r, i0:i1] = 2.0 * a[start:stop][::-1]
            A[r, n] = -1.0

        b_ub = np.array([float(q[k]) for k in active_set], dtype=np.float64)
        c = np.zeros(n + 1, dtype=np.float64)
        c[n] = 1.0
        A_eq = np.ones((1, n + 1), dtype=np.float64)
        A_eq[0, n] = 0.0
        bounds = [(0.0, np.inf)] * n + [(None, None)]

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=np.array([1.0]),
                          bounds=bounds, method='highs', options={'presolve': True})
        except Exception:
            return None

        if res.status != 0 or res.x is None:
            return None

        g = np.maximum(res.x[:n], 0.0)
        gs = float(g.sum())
        if gs <= 1e-10:
            return None
        g = g / gs
        T = float(res.x[n])

        # Check for violated constraints
        ag = conv_ab(a, g)
        violations = 2.0 * ag - T - q
        viol_idx = np.where(violations > tol)[0]
        if len(viol_idx) == 0:
            return g
        # Add most violated constraint
        worst = int(viol_idx[np.argmax(violations[viol_idx])])
        active.add(worst)

    # Fallback: return direction from top lags
    return None


def interleaved_optimize(a0, time_limit, verbose=True):
    """Main optimization loop alternating momentum GD, LP, and anti-peak."""
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    best = a.copy()
    best_score = objective_norm(a)

    phase = "mom"
    phase_timer = 0.0
    mom_time = 40.0
    lp_time = 15.0
    anti_time = 10.0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5.0:
            break

        if phase == "mom":
            t_lim = min(mom_time, remaining)
            out = momentum_gradient_descent(a, t_lim, verbose=False)
            sc = objective_norm(out)
            if sc < best_score - 1e-12:
                best = out.copy()
                best_score = sc
                no_improve = 0
            else:
                no_improve += 1
            a = out
            phase = "lp" if HAVE_LP else "anti"

        elif phase == "lp":
            t_lim = min(lp_time, remaining)
            q = self_conv(a)
            g_dir = lp_constraint_gen(a, q, max_iter=20)
            if g_dir is not None:
                # Line search
                c = conv_ab(a, g_dir)
                qd = self_conv(g_dir)
                q0 = q
                M0 = float(np.max(q0))
                best_alpha = 0.0
                best_M = M0
                for al in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    a1 = 1.0 - al
                    val = np.max(a1 * a1 * q0 + 2.0 * al * a1 * c + al * al * qd)
                    if val < best_M:
                        best_M = val
                        best_alpha = al
                if best_alpha > 0:
                    new = (1.0 - best_alpha) * a + best_alpha * g_dir
                    new = np.maximum(new, 0.0)
                    s = float(new.sum())
                    if s > 1e-15:
                        new = new / s
                        sc = objective_norm(new)
                        if sc < best_score - 1e-12:
                            best = new.copy()
                            best_score = sc
                            no_improve = 0
                        a = new
            else:
                no_improve += 1
            phase = "anti"

        elif phase == "anti":
            t_lim = min(anti_time, remaining)
            out = anti_peak_moves(a, t_lim)
            sc = objective_norm(out)
            if sc < best_score - 1e-12:
                best = out.copy()
                best_score = sc
                no_improve = 0
            else:
                no_improve += 1
            a = out
            phase = "mom"

        if verbose and (time.monotonic() - t0 - phase_timer) > 60:
            elapsed = time.monotonic() - t0
            print("    interleaved: elapsed=%.1fs best=%.9f phase=%s noimp=%d"
                  % (elapsed, best_score, phase, no_improve))
            phase_timer = time.monotonic() - t0

        if no_improve > 50 and (time.monotonic() - t0) > 0.7 * time_limit:
            break

    return best


def make_spectral_start(n, seed=42):
    """Create a start with approximately flat spectrum (Zadoff-Chu inspired)."""
    rng = np.random.default_rng(seed)
    # Generate random phases, take real part, make non-negative
    phases = rng.uniform(0, 2 * math.pi, size=n)
    # Use a window to make it smoother
    x = np.linspace(0, 1, n)
    window = np.sin(np.pi * x) ** 0.5
    # Combine: real part of exp(i*phases) weighted by window, then clamp
    a = window * (1.0 + np.cos(phases))
    a = np.maximum(a, 0.0)
    a = a + 1e-6
    return normalize_vec(a)


def make_parametric_starts(n, base):
    """Generate diverse parametric starts."""
    x = np.linspace(0, 1, n)
    out = []
    base = normalize_vec(base) if base is not None else np.ones(n) / n

    out.append(base.copy())
    out.append(base[::-1].copy())

    for w in (0.3, 0.5, 0.7):
        out.append(normalize_vec((1 - w) * base + w * base[::-1]))

    for p in (0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0):
        out.append(normalize_vec((base + 1e-10) ** p))

    for aexp in (0.3, 0.5, 1.0, 2.0, 3.0):
        out.append(normalize_vec((x + 1e-6) ** aexp))
        out.append(normalize_vec(((1 - x) + 1e-6) ** aexp))

    for r in (0.5, 1.0, 2.0, 4.0, 8.0):
        out.append(normalize_vec(np.exp(-r * x) + np.exp(-r * (1 - x)) + 1e-6))

    for ctr in (0.3, 0.4, 0.5, 0.6, 0.7):
        for wd in (0.05, 0.1, 0.15, 0.2, 0.3):
            out.append(normalize_vec(np.exp(-((x - ctr) ** 2) / (2 * wd * wd)) + 1e-6))

    for c1, c2 in ((0.25, 0.75), (0.3, 0.7), (0.2, 0.8), (0.35, 0.65)):
        for wd in (0.08, 0.12, 0.2):
            out.append(normalize_vec(
                np.exp(-((x - c1) ** 2) / (2 * wd * wd)) +
                np.exp(-((x - c2) ** 2) / (2 * wd * wd)) + 1e-6))

    # Spectral starts
    for s in (42, 137, 999):
        out.append(make_spectral_start(n, seed=s))

    # Random smooth
    rng = np.random.default_rng(12345)
    for _ in range(8):
        freqs = rng.uniform(1, 5, size=4)
        amps = rng.uniform(0.5, 1.0, size=4)
        phs = rng.uniform(0, 2 * math.pi, size=4)
        a = np.ones(n, dtype=np.float64)
        for i in range(4):
            a += amps[i] * np.cos(2 * math.pi * freqs[i] * x + phs[i])
        a = np.maximum(a, 0.0) + 1e-6
        out.append(normalize_vec(a))

    return out


def select_diverse(candidates, max_kept=6):
    """Select diverse candidates by score and dissimilarity."""
    scored = []
    for c in candidates:
        c = normalize_vec(c)
        sc = objective_norm(c)
        if np.isfinite(sc):
            scored.append((sc, c))
    scored.sort(key=lambda x: x[0])
    kept = []
    for sc, seq in scored:
        if len(kept) >= max_kept:
            break
        ok = True
        for kseq in kept:
            if seq.size == kseq.size:
                dp = float(np.dot(seq, kseq))
                norm_s = float(np.sqrt(np.dot(seq, seq) * np.dot(kseq, kseq)))
                if norm_s > 1e-15 and dp / norm_s > 0.99:
                    ok = False
                    break
        if ok:
            kept.append(seq.copy())
    return kept


def interpolate_sequence(f, new_n):
    f = normalize_vec(f)
    old_n = f.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return f.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f)
    return normalize_vec(g)
