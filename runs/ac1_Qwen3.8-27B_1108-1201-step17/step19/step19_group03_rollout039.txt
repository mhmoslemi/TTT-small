def spectral_flatten(a, n_iters=200, timestep=60.0, seed=0):
    """Iteratively flatten the magnitude spectrum while preserving non-negativity."""
    a = norm_sum1(a).copy()

    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    for it in range(n_iters):
        if time.monotonic() - t0 > timestep:
            break
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        # Soft blend toward flat magnitude
        mean_mag = np.mean(mag)
        # Use a non-linear blend to preserve the DC component
        alpha = 0.05 + 0.15 * (1.0 - np.exp(-it / 50.0))
        new_mag = (1.0 - alpha) * mag + alpha * mean_mag
        # Preserve DC exactly
        new_mag[0] = mag[0]
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()

        m = fast_autocorr(a_new).max()

                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

    return best_a


def lp_frank_wolfe_diverse(a0, timestep, solver, n_groups=5, K_init=300, K_max=None, seed=0):
    """LP-Frank-Wolfe with diverse (stratified) constraint selection."""
    a = norm_sum1(a0).copy()
    t0 = time.monotonic()
    n = len(a)
    n_lags = 2 * n - 1
    rng = np.random.default_rng(seed)

    if K_max is None:
        K_max = min(2 * n - 1, 3000)

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    K = K_init
    it = 0

    while time.monotonic() - t0 < timestep - 5:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < 1e-16:
            break

        # Stratified constraint selection
        K_sel = min(K, n_lags)
        group_size = n_lags // n_groups
        binding_idx = []
        per_group = max(10, K_sel // n_groups)

        for g in range(n_groups):
            lo = g * group_size
            hi = min((g + 1) * group_size, n_lags)
            if g == n_groups - 1:
                hi = n_lags
            group_ac = ac[lo:hi]
            top_in_group = np.argsort(group_ac)[::-1][:per_group]
            binding_idx.extend(top_in_group + lo)

        binding_idx = np.array(binding_idx[:K_sel])

        # Build constraint matrix T
        # T[i, j] = contribution of b[j] to constraint i (lag binding_idx[i])
        # conv(a, b)[k] = sum_j a[j] * b[k-j]... 
        # Actually: (a*b)[k] = sum_j a[j]*b[k-j]
        # We need (a*b)[ki] <= M for each selected ki
        # (a*b)[ki] = sum_j a[j] * b[ki-j] (valid j)
        # = sum_j b[j] * a[ki-j] (reindex: let i' = ki-j, j = ki-i')
        # So T[i, j] = a[ki - j] if 0 <= ki - j < n, else 0
        T = np.zeros((len(binding_idx), n))
        for ii, ki in enumerate(binding_idx):
            ki = int(ki)
            lo = max(0, ki - n + 1)
            hi = min(n, ki + 1)
            if lo < hi:
                # T[ii, j] = a[ki - j] for j in [lo, hi)
                T[ii, lo:hi] = a[ki - np.arange(lo, hi)][::-1]

        # Solve LP: maximize sum(b) subject to T@b <= M, b >= 0
        c = -np.ones(n)  # minimize -sum(b)
        A_ub = T
        b_ub = np.full(len(binding_idx), M_curr)

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        except Exception:
            try:
                b_var = cp.Variable(n, nonneg=True)
                prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M_curr])
                prob.solve(solver=solver, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    b_val = np.maximum(np.asarray(b_var.value), 0.0)
                else:
                    b_val = a.copy()
            except Exception:
                b_val = a.copy()
        else:
            if res.success and res.x is not None:
                b_val = np.maximum(res.x, 0.0)
            else:
                b_val = a.copy()

        b_sum = b_val.sum()
        if b_sum < 1e-15:
            K = min(K + 50, K_max)
            it += 1
            continue

        g_dir = b_val / b_sum

        # Line search
        best_alpha = 2.0 / (it + 2.0)
        best_trial_m = base_obj = M_curr
        for alpha in [best_alpha, min(1.0, best_alpha * 2), 0.3, 0.5, 0.1, 0.8]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-10:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        it += 1
        if it % 5 == 0:
            K = min(K + 40, K_max)

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW div it={it} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a


def soft_subgradient_md(a0, timestep, top_k=60, beta_init=5.0, c0=5.0, seed=100):
    """Away-step-less mirror descent with soft subgradient."""
    a = norm_sum1(a0).copy()
    t0 = time.monotonic()
    n = len(a)
    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    cur_beta = beta_init
    cur_top_k = top_k

    for t in range(1, 10000000):
        if time.monotonic() - t0 > timestep - 2:
            break

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-10:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_sel = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w = w / w.sum()

        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n, ki + 1)
            if lo < hi:
                idx = np.arange(lo, hi)
                g[lo:hi] += 2.0 * wi * y[ki - idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = np.maximum(best_a + pert, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_beta = max(cur_beta - 1.0, 2.0)
            cur_top_k = min(cur_top_k + 10, 200)

    return best_a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search: spectral init + stratified LP-FW + sensitivity CD + soft gradient MD.
    Time budget: 1800s total, with 30s reserved for cleanup.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Total time budget: {time_budget:.0f}s")

    solver = get_lp_solver()

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5500
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(make_u_shape(n_base, 2.0))
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Spectral Initialization ===
    print(f"\n=== Phase 1: Spectral Init (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 120.0)

    if t1 > 30:
        candidates = [norm_sum1(best_seq)]
        labels = ["parent"]

        candidates.append(spectral_flatten(best_seq, n_iters=300, timestep=t1 * 0.4, seed=1))
        labels.append("spec_flat")

        candidates.append(make_u_shape(n0, 1.5))
        labels.append("u1.5")
        candidates.append(make_u_shape(n0, 3.0))
        labels.append("u3.0")
        candidates.append(make_cosine(n0, 0.4))
        labels.append("cos")
        candidates.append(make_bimodal(n0, 0.12))
        labels.append("bimodal")

        for cand, lbl in zip(candidates, labels):
            try_update(cand, f"init_{lbl}")

    # === Phase 2: Sensitivity Coordinate Descent ===
    print(f"\n=== Phase 2: Sensitivity CD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)

    if t2 > 30:
        t2a = t2 * 0.5
        a_out = sensitivity_coordinate_descent(best_seq, t2a, top_k=50, seed=42)
        try_update(a_out, "cd1a")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t2b = min(remaining * 0.12, 200.0)
            a_out = sensitivity_coordinate_descent(best_seq, t2b, top_k=80, seed=99)
            try_update(a_out, "cd1b")

    # === Phase 3: Stratified LP-FW ===
    print(f"\n=== Phase 3: Stratified LP-FW (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.25, 450.0)

    if t3 > 60 and n0 <= 7500:
        K_init = min(400, 2 * n0 - 1)
        K_max = min(2500, 2 * n0 - 1)
        a_out = lp_frank_wolfe_diverse(best_seq, t3, solver, n_groups=5,
                                        K_init=K_init, K_max=K_max, seed=0)
        try_update(a_out, "lpfwd1")

    # === Phase 4: CD after LP-FW ===
    print(f"\n=== Phase 4: CD round 2 ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 180.0)
    if t4 > 20:
        a_out = sensitivity_coordinate_descent(best_seq, t4, top_k=60, seed=200)
        try_update(a_out, "cd2")

    # === Phase 5: Soft Subgradient MD ===
    print(f"\n=== Phase 5: Soft Gradient MD ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 280.0)
    if t5 > 40:
        a_out = soft_subgradient_md(best_seq, t5, top_k=80, beta_init=6.0, c0=6.0, seed=300)
        try_update(a_out, "sgmd1")

    # === Phase 6: LP-FW round 2 (larger K) ===
    print(f"\n=== Phase 6: LP-FW round 2 ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.18, 300.0)
    if t6 > 60 and n0 <= 7500:
        K_init = min(600, 2 * n0 - 1)
        K_max = min(4000, 2 * n0 - 1)
        a_out = lp_frank_wolfe_diverse(best_seq, t6, solver, n_groups=7,
                                        K_init=K_init, K_max=K_max, seed=1)
        try_update(a_out, "lpfwd2")

    # === Phase 7: CD round 3 ===
    print(f"\n=== Phase 7: CD round 3 ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.08, 120.0)
    if t7 > 15:
        a_out = sensitivity_coordinate_descent(best_seq, t7, top_k=100, seed=500)
        try_update(a_out, "cd3")

    # === Phase 8: Multi-length exploration ===
    print(f"\n=== Phase 8: Multi-length ===")
    remaining = time_budget - (time.monotonic() - t_global)

    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 1500, -1500]:
        new_n = n0 + delta
        if 4000 <= new_n <= 8500:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))

    for new_n in trial_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)
        try_update(a_interp, f"interp_n{new_n}")

        t8 = min(remaining * 0.25, 150.0)
        if t8 > 20:
            a_out = sensitivity_coordinate_descent(a_interp, t8, top_k=50, seed=new_n % 1000)
            try_update(a_out, f"cd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 7000:
            t8b = min(remaining * 0.3, 120.0)
            a_out = lp_frank_wolfe_diverse(a_interp, t8b, solver, n_groups=5,
                                            K_init=300, K_max=min(2000, 2 * new_n - 1), seed=new_n % 999)
            try_update(a_out, f"lp_n{new_n}")

    # === Phase 9: Final intensive optimization on best ===
    print(f"\n=== Phase 9: Final optimization ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        # CD
        t9a = min(remaining * 0.3, 80.0)
        if t9a > 10:
            a_out = sensitivity_coordinate_descent(best_seq, t9a, top_k=80, seed=700)
            try_update(a_out, "cd_final")

        # LP-FW
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and n_best <= 7500:
            t9b = min(remaining * 0.4, 150.0)
            a_out = lp_frank_wolfe_diverse(best_seq, t9b, solver, n_groups=5,
                                            K_init=500, K_max=min(3000, 2 * n_best - 1), seed=2)
            try_update(a_out, "lp_final")

        # Final CD
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t9c = min(remaining * 0.5, 60.0)
            a_out = sensitivity_coordinate_descent(best_seq, t9c, top_k=120, seed=800)
            try_update(a_out, "cd_ult")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5500
        best_seq = make_u_shape(n_fb, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
