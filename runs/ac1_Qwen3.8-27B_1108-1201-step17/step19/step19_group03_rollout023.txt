def fast_autocorr(a):
    """Compute full linear convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(a, indices):
    """Build the constraint matrix T where T[i,j] = a[indices[i]-j] for valid j."""
    n = len(a)
    K = len(indices)
    T = np.zeros((K, n))
    for ii, k in enumerate(indices):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        for j in range(j_start, j_end):
            T[ii, j] = a[ki - j]
    return T


def lp_direction(a, M, binding_idx, solver):
    """Solve LP to find direction b that maximizes sum(b) subject to T@b <= M, b >= 0."""
    n = len(a)
    K = len(binding_idx)
    b_var = cp.Variable(n, nonneg=True)
    T = build_constraint_matrix(a, binding_idx)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val / b_val.sum()

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_T(a, indices):

        je = min(n, ki + 1)
        for j in range(js, je):
            T[ii, j] = a[ki - j]
    return T


def solve_lp_direction(a, M, binding_idx, solver):

                continue

            # Block move: move mass from a block around src to a low-sensitivity region
            blk_s = max(0, src - block_size // 2)
            blk_e = min(n, src + block_size // 2 + 1)
            blk = np.arange(blk_s, blk_e)
            blk_mass = a[blk].sum()
            if blk_mass < 1e-12:
                continue

            # Find low-sensitivity destination
            cand_mask = (~np.zeros(n, dtype=bool))
            cand_mask[blk_s:blk_e] = False
            candidates = np.where(cand_mask)[0]
            if len(candidates) == 0:
                continue
            cand_sens = sens[candidates]
            dst_idx = candidates[np.argmin(cand_sens)]

            # Try moving a fraction
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                amt = blk_mass * frac
                # Distribute from source block proportionally
                source_weights = np.maximum(a[blk], 0.0)
                sw_sum = source_weights.sum()
                if sw_sum < 1e-15:
                    continue
                reduction = source_weights / sw_sum * amt
                a_trial = a.copy()
                a_trial[blk] -= reduction
                a_trial[dst_idx] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random block move
            i = int(rng.integers(0, max(1, n - block_size)))
            j = int(rng.integers(0, max(1, n - block_size)))
            if i != j:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                if amt > 1e-12:
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    BlockCoord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def spectral_gradient(a0, time_limit, verbose=False, rng_seed=50):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        it += 1
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        if it % 3 == 0:
            # Spectral step
            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            new_mag = 0.6 * mag + 0.4 * mag_mean
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
            m_new = fast_autocorr(a_new).max()
            if m_new < best_m:
                best_m = m_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            # Gradient step
            ac = fast_autocorr(a)
            top_idx = np.argsort(ac)[::-1][:40]
            top_vals = np.maximum(ac[top_idx], 1e-30)
            w = top_vals ** 5.0
            w = w / w.sum()
            g = np.zeros(n)
            for wi, k in zip(w, top_idx):
                ki = int(k)
                is_ = max(0, ki - n + 1)
                ie = min(n, ki + 1)
                if is_ >= ie:
                    continue
                idx = np.arange(is_, ie)
                j_idx = ki - idx
                g[idx] += 2.0 * wi * a[j_idx]

            eta = 0.8 / np.sqrt(it + 1.0)
            log_a = np.log(np.maximum(a, 1e-15)) - eta * g
            ml = log_a.max()
            a_new = np.exp(log_a - ml)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()
            m_new = fast_autocorr(a_new).max()
            if m_new < best_m:
                best_m = m_new
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
            a = a_new

        if stall > 150:
            pert = rng.normal(0, 0.015, n) * best_a
            a_p = np.maximum(best_a + pert, 1e-10)
            a = a_p / a_p.sum()
            stall = 50

        if verbose and it % 500 == 0:
            J = 2.0 * n * best_m
            print(f"    SpecGrad it={it} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_asymmetric(n, alpha=0.5):
    """Create an asymmetric shape that reduces the middle peak."""
    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * np.sin(np.pi * x)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_dual_peak(n, pos1=0.2, pos2=0.7, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) / width) ** 2) + np.exp(-((x - pos2) / width) ** 2) + 0.01
    return norm_sum1(a)


def make_fourier_init(n, n_freq=30, seed=0):
    rng = np.random.default_rng(seed)
    x = np.arange(n) / n
    a = np.ones(n) * 0.5
    for freq in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search: LP-FW + AMD + spectral + block coordinate with smart scheduling."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Initialization and POCS ===
    print(f"\n=== Phase 1: Init + POCS (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.07, 100.0)

    if t1 > 15:
        a_pocs, _ = pocs_init(best_seq, t1, n_trials=4, verbose=True, rng_seed=42)
        try_update(a_pocs, "pocs")

        # Try a few analytical inits
        inits = [
            (make_u_shape(n0, 1.5), "u1.5"),
            (make_asymmetric(n0, 0.4), "asym"),
            (make_dual_peak(n0, 0.25, 0.7, 0.18), "dual"),
        ]
        for init, lbl in inits:
            sc = evaluate_safe(init)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = init.copy()
                print(f"  *** NEW BEST [{lbl}]: {sc:.6f}")

    # === Phase 2: Main LP-FW ===
    print(f"\n=== Phase 2: LP-FW main (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.3, 500.0)

    if t2 > 30:
        K_init = min(200, 2 * n_cur - 1)
        K_max = min(3000, 2 * n_cur - 1)
        a_out, _ = lp_frank_wolfe_improved(best_seq, t2, solver, verbose=True,
                                           K_init=K_init, K_max=K_max,
                                           use_dual_selection=True)
        try_update(a_out, "lp_main")

    # === Phase 3: AMD ===
    print(f"\n=== Phase 3: AMD (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.2, 350.0)

    if t3 > 30:
        a_out, _ = amd_improved(best_seq, t3, top_k=60, beta=6.0, c0=5.0,
                                restart_every=4000, verbose=True, rng_seed=77)
        try_update(a_out, "amd")

    # === Phase 4: LP-FW round 2 (more constraints) ===
    print(f"\n=== Phase 4: LP-FW r2 (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.2, 350.0)

    if t4 > 30:
        K_init = min(400, 2 * n_cur - 1)
        K_max = min(5000, 2 * n_cur - 1)
        a_out, _ = lp_frank_wolfe_improved(best_seq, t4, solver, verbose=True,
                                           K_init=K_init, K_max=K_max,
                                           use_dual_selection=True)
        try_update(a_out, "lp_r2")

    # === Phase 5: Spectral-Gradient ===
    print(f"\n=== Phase 5: SpecGrad (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.1, 170.0)

    if t5 > 20:
        a_out, _ = spectral_gradient(best_seq, t5, verbose=True, rng_seed=55)
        try_update(a_out, "specgrad")

    # === Phase 6: Block coordinate polish ===
    print(f"\n=== Phase 6: Block Coord (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.08, 130.0)

    if t6 > 15:
        a_out, _ = block_coordinate_polish(best_seq, t6, verbose=True, rng_seed=321, block_size=3)
        try_update(a_out, "blockcoord")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 1500, -1500]:
        new_n = n_best + delta
        if 1000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.35, 120.0)
        if t7a > 15:
            a_out, _ = amd_improved(a_interp, t7a, top_k=50, beta=5.0, c0=4.0,
                                    restart_every=5000, verbose=False, rng_seed=new_n % 997)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 5000:
            t7b = min(remaining * 0.4, 100.0)
            if t7b > 15:
                K_i = min(200, 2 * new_n - 1)
                K_m = min(2000, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe_improved(a_interp, t7b, solver, verbose=False,
                                                   K_init=K_i, K_max=K_m)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 8: Final AMD on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final AMD (n={n_best}) ===")
    if remaining > 20:
        t8 = min(remaining * 0.3, 150.0)
        if t8 > 15:
            a_out, _ = amd_improved(best_seq, t8, top_k=80, beta=8.0, c0=6.0,
                                    restart_every=3000, verbose=False, rng_seed=999)
            try_update(a_out, "amd_final")

    # === Phase 9: Final LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 9: Final LP-FW (n={n_best}) ===")
        t9 = min(remaining * 0.4, 150.0)
        if t9 > 20:
            K_i = min(500, 2 * n_best - 1)
            K_m = min(6000, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe_improved(best_seq, t9, solver, verbose=False,
                                               K_init=K_i, K_max=K_m)
            try_update(a_out, "lp_final")

    # === Phase 10: Final block polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        print(f"\n=== Phase 10: Final polish (n={len(best_seq)}) ===")
        t10 = min(remaining * 0.6, 60.0)
        if t10 > 8:
            a_out, _ = block_coordinate_polish(best_seq, t10, verbose=False,
                                              rng_seed=789, block_size=2)
            try_update(a_out, "polish_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
