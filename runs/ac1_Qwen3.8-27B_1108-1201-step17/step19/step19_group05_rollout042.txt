Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using soft-max guided mirror descent

import numpy as np
import time


def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def make_ramp(n, alpha=1.0):
    """Create normalized (i+1)^alpha sequence."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** alpha
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_cosine_taper(n, p=1.0):
    """Create normalized cosine-tapered sequence: a_i = sin(pi*(i+0.5)/n)^p."""
    x = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (x + 0.5) / n) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_gaussian_taper(n, sigma=1.0):
    """Create normalized Gaussian-tapered sequence centered."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - center) / (sigma * center)) ** 2)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_two_ramp(n, split=0.5):
    """Two-sided ramp that goes up then down, but asymmetric."""
    a = np.zeros(n)
    ns = int(n * split)
    if ns < 1:
        ns = 1
    nr = n - ns
    a[:ns] = np.arange(1, ns + 1, dtype=np.float64)
    if nr > 0:
        a[ns:] = np.arange(nr, 0, -1, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def compute_soft_gradient(a, conv, T, n_top):
    """
    Compute subgradient of soft-max of conv with respect to a.
    Uses top-n_top peaks for efficiency.
    Returns gradient vector of same length as a.
    """
    n = len(a)
    # Get top peaks
    if n_top >= len(conv):
        top_idx = np.arange(len(conv))
        vals = conv
    else:
        top_idx = np.argpartition(conv, -n_top)[-n_top:]
        vals = conv[top_idx]

    # Softmax weights
    max_v = vals.max()
    if max_v < 1e-30:
        return np.zeros(n)
    exp_v = np.exp((vals - max_v) / T)
    w = exp_v / exp_v.sum()

    # Compute gradient: d(conv[k])/d(a_i) = a[k-i] + a[i-k+n-1] (valid indices)
    grad = np.zeros(n)
    for idx_i, k in enumerate(top_idx):
        wi = w[idx_i]
        if wi < 1e-12:
            continue
        ki = int(k)
        # d(conv[ki])/d(a_i) = a[ki-i] (when 0<=ki-i<n) + a[i-ki+n-1] (when 0<=i-ki+n-1<n)
        # First term: valid when 0 <= ki-i < n, i.e., ki-n+1 <= i <= ki
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo <= j_hi:
            i_range = np.arange(j_lo, j_hi + 1)
            grad[i_range] += wi * a[ki - i_range]

        # Second term: valid when 0 <= i-ki+n-1 < n, i.e., ki-n+1 <= i-ki+n-1... 
        # Actually: i - ki + n - 1 is in [0, n-1] means ki-n+1 <= i <= ki
        # Wait, that's the same range. Let me reconsider.
        # conv[k] = sum_j a[j]*a[k-j]
        # d(conv[k])/d(a_i) = a[k-i] (from first factor, j=i) + a[k-i] (from second factor, k-j=i => j=k-i... no)
        # Actually: d/d(a_i) of a[j]*a[k-j] is nonzero when j=i (gives a[k-i]) or k-j=i (i.e., j=k-i, gives a[k-i])
        # But k-j=i means j=k-i, and the term is a[j]*a[k-j] = a[k-i]*a[i], derivative w.r.t. a_i is a[k-i]
        # So total: d(conv[k])/d(a_i) = a[k-i] + a[k-i] = 2*a[k-i] when both i and k-i are in [0,n-1]
        # If only i is in range (k-i out of range): just a[k-i] = 0, so contribution is 0
        # Wait no. If k-i is out of [0,n-1], then a[k-i]=0, so the contribution is 0.
        # Similarly if i is out of range, a[i]=0 so the whole term is 0.
        # So: d(conv[k])/d(a_i) = a[k-i] + a[k-i] = 2*a[k-i] when 0<=i<n and 0<=k-i<n
        # But wait, we might double count when i = k-i (i.e., k = 2i).
        # Actually no, the two terms are distinct: one comes from varying the first a[j], the other from the second a[k-j].
        # When j=i: term is a[i]*a[k-i], derivative wrt a_i is a[k-i]
        # When k-j=i: term is a[k-i]*a[i], derivative wrt a_i is a[k-i]
        # These are the same term! So we should only count it once when k=2i.
        # But in the sum, each pair (j, k-j) appears once, so if j=i and k-j=i (i=k/2),
        # it's a single term a[i]*a[i], and derivative is 2*a[i].
        # So the formula d(conv[k])/d(a_i) = 2*a[k-i] is correct (gives 2*a[k-i] which for k=2i is 2*a[i]).
        
        # The range where 0<=k-i<n is: k-n+1 <= i <= k (and 0<=i<n)
        # This is exactly j_lo to j_hi computed above.
        # But I already added a[ki-i_range] above. I need to add it again (factor of 2).
        # Wait, let me re-examine. I think I made an error above.
        # Let me redo: the gradient should be 2*a[k-i] for valid i.
        # In my code above, I added wi * a[ki-i_range] once. I need to add it once more (or multiply by 2).
        # Let me just redo this properly below.
        pass

    # Redo properly: gradient is 2 * sum_k w_k * a[k-i] for valid (i,k)
    grad = np.zeros(n)
    for idx_i, k in enumerate(top_idx):
        wi = w[idx_i]
        if wi < 1e-12:
            continue
        ki = int(k)
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo <= j_hi:
            i_range = np.arange(j_lo, j_hi + 1)
            grad[i_range] += 2.0 * wi * a[ki - i_range]

    return grad


def soft_max_value(conv, T):
    """Compute log-sum-exp temperature approximation of max."""
    max_v = conv.max()
    if max_v < 1e-30:
        return 0.0
    return max_v + np.log(np.exp((conv - max_v) / T).sum()) * T


def mirror_descent_smooth(a0, time_limit, T_init=0.05, T_final=0.005, n_top=50,
                          lr0=5.0, rng_seed=42, verbose=True):
    """
    Adaptive-temperature mirror descent (multiplicative weights) to minimize
    the soft-max of the convolution.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-20)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    restart_count = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        # Adaptive temperature: anneal from T_init to T_final
        progress = min(1.0, step / 5000.0)
        T = T_init * (T_final / T_init) ** progress

        # Nesterov momentum
        beta = 0.9 + 0.05 * np.sin(step * 0.01)  # oscillating momentum
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-20)
        sy = y.sum()
        y = y / sy

        # Compute convolution and gradient
        conv = fast_autocorr(y)
        g = compute_soft_gradient(y, conv, T, n_top=n_top)

        # Learning rate
        eta = lr0 / (np.sqrt(step) + 1.0)

        # Mirror descent update (exponential family)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Track best
        M_new = conv.max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = y.copy()
            stall = 0
        else:
            # Also check x
            M_x = fast_autocorr(x).max()
            if M_x < best_m - 1e-15:
                best_m = M_x
                best_a = x.copy()
                stall = 0
            else:
                stall += 1

        # Restart if stalled
        if stall > 3000:
            stall = 0
            restart_count += 1
            strength = min(0.2, 0.05 * restart_count)
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-15)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                print(f"        MD restart {restart_count} step={step}: J={2*n*best_m:.6f}")

        if verbose and step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        MD step={step} ({elapsed:.0f}s) T={T:.5f}: J={2*n*best_m:.6f}")

    return best_a, best_m


def projected_gradient_smooth(a0, time_limit, T_init=0.05, T_final=0.005, n_top=50,
                              lr0=0.01, rng_seed=42, verbose=True):
    """
    Projected gradient descent (additive) with soft-max objective.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-20)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    step = 0
    stall = 0
    restart_count = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        progress = min(1.0, step / 5000.0)
        T = T_init * (T_final / T_init) ** progress

        conv = fast_autocorr(a)
        g = compute_soft_gradient(a, conv, T, n_top=n_top)

        # Gradient step with projection onto simplex-like constraint
        eta = lr0 / (np.sqrt(step) + 1.0)
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        M_new = conv.max()
        M_a = fast_autocorr(a_new).max()
        if M_a < best_m - 1e-15:
            best_m = M_a
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Try smaller step
            for frac in [0.5, 0.1, 0.01]:
                a_trial = a - eta * frac * g
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                Mt = fast_autocorr(a_trial).max()
                if Mt < best_m - 1e-15:
                    best_m = Mt
                    best_a = a_trial.copy()
                    a = a_trial
                    stall = 0
                    break
            else:
                stall += 1

        if stall > 2000:
            stall = 0
            restart_count += 1
            pert = rng.normal(0, 0.05 * restart_count, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-15)
            x_r = x_r / x_r.sum()
            a = x_r
            if verbose:
                print(f"        PGD restart {restart_count} step={step}: J={2*n*best_m:.6f}")

        if verbose and step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        PGD step={step} ({elapsed:.0f}s) T={T:.5f}: J={2*n*best_m:.6f}")

    return best_a, best_m


def spectral_flatten(a0, time_limit, verbose=True):
    """
    Gerchberg-Saxton style: iteratively flatten the power spectrum to make
    the time-domain convolution more uniform.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_mean = P.mean()
        if P_max < 1e-30:
            break

        # Flatten: push power from peaks toward mean
        alpha = 0.1 + 0.4 * min(1.0, it / 2000.0)
        P_new = P + alpha * (P_mean - P)
        P_new = np.maximum(P_new, 1e-30)

        # Update magnitude only, preserve phase
        mag_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        mag_ratio = np.clip(mag_ratio, 0.7, 1.5)
        A_new = A * mag_ratio
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend
            blend = 0.1
            a = (1 - blend) * a + blend * a_new
            a = np.maximum(a, 0.0)
            a = a / max(a.sum(), 1e-15)
            M_curr = fast_autocorr(a).max()
            if M_curr < best_m:
                best_m = M_curr
                best_a = a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"        Spec it={it} ({elapsed:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m


def peak_shaving(a0, time_limit, verbose=True):
    """
    Directly reduce peaks by moving mass from high-contributing positions
    to low-contributing ones.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit and stall < 50:
        it += 1
        conv = fast_autocorr(a)
        k_max = int(np.argmax(conv))

        # Find which a[i] contributes most to conv[k_max]
        j_lo = max(0, k_max - n + 1)
        j_hi = min(n - 1, k_max)
        if j_lo > j_hi:
            stall += 1
            continue
        idx = np.arange(j_lo, j_hi + 1)
        contrib = a[idx] * a[k_max - idx]
        src = idx[np.argmax(contrib)]

        if a[src] < 1e-12:
            stall += 1
            continue

        # Find best destination: minimize increase in top peaks
        top_ks = np.argpartition(conv, -20)[-20:]
        costs = np.zeros(n)
        for k2 in top_ks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            if lo2 <= hi2:
                valid = np.arange(lo2, hi2 + 1)
                costs[valid] += a[k2i - valid]

        # Also add the current value as a soft constraint (don't fill already-large positions)
        total_cost = costs + 0.5 * a
        sorted_idx = np.argsort(total_cost)

        moved = False
        for dst in sorted_idx[:5]:
            if dst == src:
                continue
            for frac in [0.1, 0.05, 0.02]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_t = fast_autocorr(a_trial).max()
                if M_t < best_m - 1e-16:
                    a = a_trial
                    best_m = M_t
                    best_a = a.copy()
                    moved = True
                    stall = 0
                    break
            if moved:
                break

        if not moved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.01, 0.03)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / max(a.sum(), 1e-15)
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"        Peak it={it} ({elapsed:.0f}s): J={2*n*best_m:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss


def safe_eval(a):
    """Safely compute the objective value using the external evaluator."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """
    Main search: multiple initializations + mirror descent + spectral + peak shaving.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = safe_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_norm = parent / parent.sum()
            p_score = safe_eval(p_norm)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score):
                best_seq = p_norm.copy()
                best_score = p_score
    except Exception:
        pass

    # Generate initial candidates at various lengths
    print("\n=== Generating Initial Candidates ===")
    candidates = []

    # Ramps at different lengths (key: gives ~4/3)
    for n_try in [4000, 6000, 8000, 10000]:
        a = make_ramp(n_try, 1.0)
        candidates.append((a, f"ramp_n{n_try}"))

    # Cosine tapers
    for n_try in [5000, 8000]:
        a = make_cosine_taper(n_try, 1.0)
        candidates.append((a, f"cos_n{n_try}"))
        a = make_cosine_taper(n_try, 2.0)
        candidates.append((a, f"cos2_n{n_try}"))

    # Two-sided ramps
    for n_try in [6000]:
        a = make_two_ramp(n_try, 0.5)
        candidates.append((a, f"two_ramp_n{n_try}"))

    # Slightly modified ramps (super/sub-linear)
    for alpha in [0.8, 1.2]:
        a = make_ramp(6000, alpha)
        candidates.append((a, f"ramp_a{alpha}_n6000"))

    # If parent exists, include it
    if parent is not None:
        p_norm = parent / parent.sum()
        candidates.insert(0, (p_norm, "parent_norm"))

    # Evaluate all initial candidates
    print(f"Total candidates: {len(candidates)}")
    scored = []
    for a, label in candidates:
        sc = safe_eval(a)
        if np.isfinite(sc):
            scored.append((sc, a, label))
            print(f"  Init [{label}]: {sc:.6f}")

    scored.sort(key=lambda x: x[0])

    # Phase 1: Mirror descent on top candidates
    print(f"\n=== Phase 1: Mirror Descent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_cands = min(4, len(scored))
    for i in range(n_cands):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget:
            break
        t_avail = min(remaining * 0.25, time_budget - elapsed)
        if t_avail < 30:
            break
        sc0, a0, label = scored[i]
        print(f"  MD [{label}] (t={t_avail:.0f}s, init={sc0:.6f})...")
        a_out, _ = mirror_descent_smooth(a0, t_avail, T_init=0.1, T_final=0.01,
                                         n_top=60, lr0=4.0, rng_seed=42 + i, verbose=False)
        try_update(a_out, f"md1_{label}")

    # Phase 2: Spectral flattening on current best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Spectral Flattening ===")
    if remaining > 60 and best_seq is not None:
        t2 = min(remaining * 0.15, 180.0)
        a_in = best_seq / best_seq.sum()
        a_out, _ = spectral_flatten(a_in, t2, verbose=False)
        try_update(a_out, "spectral")

        # MD after spectral
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t2b = min(remaining * 0.12, 120.0)
            a_in = best_seq / best_seq.sum()
            a_out, _ = mirror_descent_smooth(a_in, t2b, T_init=0.05, T_final=0.005,
                                             n_top=80, lr0=3.0, rng_seed=99, verbose=False)
            try_update(a_out, "md_post_spec")

    # Phase 3: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Peak Shaving ===")
    if remaining > 30 and best_seq is not None:
        t3 = min(remaining * 0.15, 150.0)
        a_in = best_seq / best_seq.sum()
        a_out, _ = peak_shaving(a_in, t3, verbose=False)
        try_update(a_out, "peak_shave")

    # Phase 4: Projected gradient on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Projected Gradient ===")
    if remaining > 30 and best_seq is not None:
        t4 = min(remaining * 0.15, 150.0)
        a_in = best_seq / best_seq.sum()
        a_out, _ = projected_gradient_smooth(a_in, t4, T_init=0.05, T_final=0.005,
                                             n_top=60, lr0=0.02, rng_seed=7, verbose=False)
        try_update(a_out, "pgd")

    # Phase 5: Try different lengths via interpolation
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length Search ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for new_n in [n_best + 2000, n_best + 4000, max(3000, n_best - 2000)]:
            if new_n <= 100 or new_n > 15000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            a_interp = interpolate_to_length(best_seq, new_n)
            try_update(a_interp, f"interp_n{new_n}")

            if remaining > 80:
                t5 = min(remaining * 0.25, 120.0)
                a_out, _ = mirror_descent_smooth(a_interp, t5, T_init=0.08, T_final=0.008,
                                                 n_top=60, lr0=3.0, rng_seed=200 + new_n % 100,
                                                 verbose=False)
                try_update(a_out, f"md_n{new_n}")

    # Phase 6: Final combined polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final Polish ===")
    if remaining > 30 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        t6a = min(remaining * 0.3, 100.0)
        if t6a > 20:
            a_out, _ = mirror_descent_smooth(a_in, t6a, T_init=0.03, T_final=0.003,
                                             n_top=100, lr0=2.0, rng_seed=555, verbose=False)
            try_update(a_out, "md_final")
            a_in = a_out / max(a_out.sum(), 1e-15)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6b = min(remaining * 0.3, 80.0)
            a_out, _ = peak_shaving(a_in, t6b, verbose=False)
            try_update(a_out, "peak_final")

    # Phase 7: One more spectral pass
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20 and best_seq is not None:
        t7 = min(remaining * 0.3, 60.0)
        a_in = best_seq / best_seq.sum()
        a_out, _ = spectral_flatten(a_in, t7, verbose=False)
        try_update(a_out, "spec_final")

    # Final validation
    if best_seq is not None:
        fs = safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback ramp.")
        best_seq = make_ramp(6000, 1.0)
        best_score = safe_eval(best_seq)
        print(f"Fallback ramp: score={best_score:.6f}")

    return [float(x) for x in best_seq]
