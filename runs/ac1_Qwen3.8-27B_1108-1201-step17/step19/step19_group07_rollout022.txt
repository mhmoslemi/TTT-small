def softmax_gradient(a, beta, top_k=None):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (3 * n - 3).bit_length()

    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]

    if top_k is not None and top_k < size:
        top_idx = np.argsort(c)[::-1][:top_k]
        c_top = c[top_idx]
        c_max = c_top.max()
        exp_c = np.zeros(size)
        exp_c[top_idx] = np.exp(beta * (c_top - c_max))
        w = exp_c / exp_c.sum()
    else:
        c_max = c.max()
        exp_c = np.exp(beta * (c - c_max))
        w = exp_c / exp_c.sum()

    a_rev = a[::-1]
    W = np.fft.rfft(w, fft_size)
    A_rev = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(W * A_rev, fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]

    return grad, c, w

def parametric_init(n, best_ref=None):
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')

    candidates = []

    # Power laws
    for p in [0.3, 0.5, 0.7, 0.8, 1.0, 1.3, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr + 1.0) ** p)

    # Beta-like shapes
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5]:
        for beta_p in [0.3, 0.5, 0.7, 1.0, 1.5]:
            candidates.append(np.maximum(i_arr + 1.0, 0.3) ** alpha *
                              np.maximum(n - i_arr, 0.3) ** beta_p)

    # Gaussian
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))

    # Exponential
    for rate in [1, 2, 3, 5, 8]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1.0)))

    # Sine-based
    for freq in [1, 2, 3]:
        candidates.append(np.maximum(np.sin(np.pi * freq * x) ** 2, 0.01))
        candidates.append(np.maximum(np.sin(np.pi * (x + 0.1)) ** 2, 0.01))

    # Triangle
    candidates.append(1.0 - 2.0 * np.abs(x - 0.5))

    # If we have a reference, try blends
    if best_ref is not None and len(best_ref) == n:
        ref_norm = norm_sum1(best_ref)
        for w in [0.3, 0.5, 0.7, 0.9]:
            candidates.append(w * ref_norm + (1.0 - w) * norm_sum1((i_arr + 1.0) ** 0.8))
            candidates.append(w * ref_norm + (1.0 - w) * norm_sum1(np.sin(np.pi * x) ** 2))

    for a_cand in candidates:
        a_norm = norm_sum1(a_cand)
        ac = fast_autocorr(a_norm)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_a = a_norm

    return best_a, best_val

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for i in range(n_perturb):
        if time.monotonic() - t0 > time_limit:
            break

        # Different perturbation scales
        scale = 0.005 * (1.0 + i / float(n_perturb))
        pert = rng.normal(0, scale, n)
        a_pert = norm_sum1(a + pert)
        m = fast_autocorr(a_pert).max()
        if m < best_m - 1e-15:
            best_m = m
            best_a = a_pert.copy()
            a = a_pert.copy()
            if verbose and i % 10 == 0:
                J = 2.0 * n * best_m
                print(f"    Perturb iter={i}: J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Determine starting length
    if parent is not None:
        n_start = len(parent)
    else:
        n_start = 5000
        best_seq = norm_sum1(np.sin(np.pi * np.linspace(0, 1, n_start)) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_start}, score={best_score:.6f}")

    # === Phase 1: Parametric search ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 30.0)
    print(f"\n=== Phase 1: Parametric init (n={n_start}, t={t1:.0f}s) ===")
    if t1 > 5:
        a_param, val_param = parametric_init(n_start, best_seq)
        check_and_update(a_param, "parametric")

    # === Phase 2: FISTA main optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.30, 450.0)
    print(f"\n=== Phase 2: FISTA main (n={n_best}, t={t2:.0f}s) ===")
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_optimize(a_in, t2, beta_start=3.0, beta_end=80.0,
                                  eta0=0.7, top_k_base=50, verbose=True)
        check_and_update(a_out, "fista_main")

    # === Phase 3: FISTA with higher beta ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: FISTA high-beta (n={n_best}, t={t3:.0f}s) ===")
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_optimize(a_in, t3, beta_start=20.0, beta_end=200.0,
                                  eta0=0.4, top_k_base=20, verbose=True)
        check_and_update(a_out, "fista_high_beta")

    # === Phase 4: LP-FW polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP-FW polish (n={n_best}, t={t4:.0f}s) ===")
    if t4 > 30 and n_best <= 10000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, initial_K=600, max_K=5000, verbose=True)
        check_and_update(a_out, "lp_polish")

    # === Phase 5: FISTA after LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 5: FISTA after LP (n={n_best}, t={t5:.0f}s) ===")
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_optimize(a_in, t5, beta_start=50.0, beta_end=400.0,
                                  eta0=0.3, top_k_base=10, verbose=False)
        check_and_update(a_out, "fista_after_lp")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration (current n={n_best}) ===")

    length_candidates = []
    for delta in [-500, -250, 250, 500, 750, 1000]:
        new_n = n_best + delta
        if 3000 <= new_n <= 10000:
            length_candidates.append(new_n)
    for rn in [4000, 4500, 5000, 5500, 6000, 6500, 7000]:
        if rn != n_best:
            length_candidates.append(rn)
    length_candidates = sorted(set(length_candidates))

    # Sort by proximity to current best (try closest first)
    length_candidates.sort(key=lambda x: abs(x - n_best))

    for new_n in length_candidates[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        if new_n == n_best:
            continue
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.4, 120.0)
        if t6a > 15:
            a_out, _ = fista_optimize(a_long, t6a, beta_start=5.0, beta_end=100.0,
                                      eta0=0.7, top_k_base=40, verbose=False)
            check_and_update(a_out, f"fista_n{new_n}")

    # === Phase 7: Local perturbation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.08, 80.0)
    print(f"\n=== Phase 7: Local perturbation (n={n_best}, t={t7:.0f}s) ===")
    if t7 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = local_perturbation_search(a_in, t7, n_perturb=200, verbose=True)
        check_and_update(a_out, "perturb")

    # === Phase 8: Final LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP-FW (n={n_best}) ===")
    if remaining > 50 and n_best <= 10000:
        t8 = min(remaining * 0.40, 250.0)
        if t8 > 25:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t8, solver, initial_K=800, max_K=6000, verbose=True)
            check_and_update(a_out, "lp_final")

    # === Phase 9: Final FISTA ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t9 = min(remaining * 0.5, 100.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = fista_optimize(a_in, t9, beta_start=100.0, beta_end=500.0,
                                      eta0=0.2, top_k_base=5, verbose=False)
            check_and_update(a_out, "fista_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
