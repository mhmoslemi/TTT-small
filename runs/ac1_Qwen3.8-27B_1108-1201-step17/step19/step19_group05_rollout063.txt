        max_v = vals.max()
        if max_v < 1e-30:
            break
        exp_v = np.exp((vals - max_v) / (max_v * 0.01))
        w = exp_v / exp_v.sum()

        g = np.zeros(n)
        for idx_i, k in enumerate(top_idx):
            ki = int(k)
            wi = w[idx_i]
            if wi < 1e-14:
                continue
            j_lo = max(0, ki - n + 1)
            j_hi = min(n - 1, ki)
            j_range = np.arange(j_lo, j_hi + 1)
            g[j_range] += wi * 2.0 * y[ki - j_range]

        eta = lr0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_x = fast_autocorr(x).max()
        M_eff = min(M_y, M_x)
        if M_eff < best_M - 1e-15:
            best_M = M_eff
            best_a = (x if M_x <= M_y else y).copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            stall = 0
            pert = rng.normal(0, 0.05, n) * best_a
            x_r = np.maximum(best_a + pert, 1e-15)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

        if verbose and t % 5000 == 0:
            print(f"    MD t={t}: J={2*n*best_M:.6f}, stall={stall}, time={time.monotonic()-t0:.0f}s")

    return best_a, best_M
