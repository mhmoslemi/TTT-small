def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v, fixed_zero_idx=None):
    """Project v onto {x >= 0, sum x = 1}, with optional fixed zero indices."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if fixed_zero_idx is not None:
        v[fixed_zero_idx] = 0.0
    # Standard simplex projection
    active = np.ones(n, dtype=bool)
    if fixed_zero_idx is not None:
        active[fixed_zero_idx] = False
    v[active] = np.sort(v[active])[::-1]
    n_active = active.sum()
    cssv = np.cumsum(v[active])
    rho = np.nonzero(v[active] * np.arange(1, n_active + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho_idx = rho[-1]
        theta = (cssv[rho_idx] - 1) / (rho_idx + 1.0)
    result = np.zeros(n)
    result[active] = np.maximum(v[active] - theta, 0.0)
    s = result.sum()
    if s < 1e-15:
        result[active] = 1.0 / n_active
        return result
    return result / s

def spectral_equalize(a, n_iters=100, blend=0.3, seed=0):
    """Iteratively equalize the magnitude spectrum while maintaining non-negativity."""
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    a = norm_sum1(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat magnitude
        mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
        
        # Blend toward flat
        new_mag = (1 - blend) * mag + blend * mag_mean
        A_new = new_mag * np.exp(1j * phase)
        
        # Back to time domain
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        
        a = a_new
    
    return a

def subgradient_descent(a0, time_limit, top_k=50, beta=5.0, momentum_decay=0.9,
                        eta0=1.0, rng_seed=0):
    """Nesterov-accelerated subgradient descent with adaptive parameters."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    vel = np.zeros(n)
    g_ema = np.zeros(n)
    ema_beta = 0.8
    cur_top_k = top_k
    cur_beta = beta
    stall = 0
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Nesterov lookahead
        a_look = a + momentum_decay * vel
        a_look = np.maximum(a_look, 1e-15)
        a_look = a_look / a_look.sum()
        
        ac = fast_autocorr(a_look)
        M_curr = ac.max()
        
        # Compute subgradient
        k_sel = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_sel) / k_sel
        else:
            w = w / w_sum
        
        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            g[idx] += 2.0 * wi * a_look[j_idx]
        
        # EMA of gradient
        g_ema = ema_beta * g_ema + (1 - ema_beta) * g
        
        # Step
        eta = eta0 / (np.sqrt(t) + 1.0)
        vel = momentum_decay * vel - eta * g_ema
        a_new = a + vel
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a = a_new
        M_new = fast_autocorr(a).max()
        
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive parameters
        if stall > 2000:

            a = np.maximum(best_a + pert, 1e-10)
            a = a / a.sum()
            vel = np.zeros(n)
            g_ema = np.zeros(n)
            stall = 0
    
    return best_a, best_m

def pairwise_descent(a0, time_limit, n_worst=10, n_moves=5000, rng_seed=0):
    """Pairwise coordinate descent: move mass from contributors to non-contributors."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        worst = np.argsort(ac)[::-1][:n_worst]
        
        # Find high contributors to worst lags
        contributors = set()
        for k in worst:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            for i in range(i_start, i_end):
                if a[i] > 1e-6 and a[ki - i] > 1e-6:
                    contributors.add(i)
                    contributors.add(ki - i)
        
        non_contrib = np.array([j for j in range(n) if j not in contributors and a[j] > 0])
        if len(non_contrib) == 0:
            non_contrib = np.arange(n)
        
        # Try moving mass
        improved = False
        for _ in range(n_moves):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick a contributor to reduce
            src_list = np.array([j for j in contributors if a[j] > 1e-8])
            if len(src_list) == 0:
                break
            src = src_list[rng.integers(len(src_list))]
            
            # Pick a destination
            dst = non_contrib[rng.integers(len(non_contrib))]
            
            # Try small moves
            for frac in [0.001, 0.003, 0.005, 0.01]:
                amt = a[src] * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.001, 0.005)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        else:
            stall = 0
    
    return best_a, best_m

def freq_phase_optimize(a0, time_limit, seed=0):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
