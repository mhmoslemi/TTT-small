    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    eta = 0.5
    stall = 0
    restarts = 0
    it = 0
    temp = temp_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        ac = fast_autocorr(x)
        M_cur = ac.max()
        
        # Compute subgradient
        grad = compute_peak_subgradient(x, ac, top_k=top_k, temp=temp)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Backtracking line search
        best_trial = x.copy()
        best_trial_m = M_cur
        found_improvement = False
        
        for attempt in range(8):
            step = eta / (attempt + 1)
            log_x = np.log(np.maximum(x, 1e-30)) - step * grad
            log_x -= log_x.max()
            x_new = np.exp(log_x)
            x_new = np.maximum(x_new, 1e-30)
            x_new = x_new / x_new.sum()
            
            M_new = fast_autocorr(x_new).max()
            if M_new < M_cur - 1e-16:
                best_trial = x_new
                best_trial_m = M_new
                found_improvement = True
                break
        else:
            # No improvement found, use smallest step
            step = eta / 9.0
            log_x = np.log(np.maximum(x, 1e-30)) - step * grad
            log_x -= log_x.max()
            x_new = np.exp(log_x)
            x_new = np.maximum(x_new, 1e-30)
            x_new = x_new / x_new.sum()
            best_trial = x_new
            best_trial_m = fast_autocorr(x_new).max()
        
        if found_improvement:
            eta = min(eta * 1.05, 5.0)
        else:
            eta *= 0.7
            stall += 1
        x = best_trial
        
        if best_trial_m < best_m - 1e-16:
            best_m = best_trial_m
            best_a = x.copy()
            stall = 0
            # Gradually decrease temperature (sharpen peak selection)
            temp = max(temp * 0.995, 0.003)
        elif stall > 200:
            # Restart with perturbation
            restarts += 1
            if restarts > max_restarts:
                break
            scale = 0.02 + 0.03 * (restarts % 5)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            eta = 0.3
            stall = 0
            temp = temp_init * 0.5
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD it={it} ({elapsed:.0f}s): J={J:.6f}, M={best_m:.8f}, eta={eta:.4f}, restarts={restarts}")
    
    return best_a, best_m


def lp_frank_wolfe_optimize(a0, time_limit, solver, verbose=False, top_K_init=150,
                            top_K_max=3000, multi_target=True, away_step=True,
                            rng_seed=0):
    """LP-Frank-Wolfe with adaptive K, multi-target M, and away-steps."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    K = top_K_init
    iter_count = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]
        
        # Build constraint matrix: (a*b)_k = sum_j a[j] * b[k-j]
        # For each constraint k: sum_j a[j] * b[k-j] <= M_target
        # = sum_j b[j] * a[k-j] <= M_target (reindexing)
        # So constraint k is: (A_k) @ b <= M_target where (A_k)[j] = a[k-j] (valid j)
        
        # Try multiple M targets
        targets = [M_curr]
        if multi_target and M_curr > 0:
            targets = [0.95 * M_curr, 0.98 * M_curr, M_curr]
        
        b_val = None
        for M_target in targets:
            T = np.zeros((K_sel, n))
            for ii, k in enumerate(binding_idx):
                ki = int(k)
                # (a*b)_ki = sum_j a[j] * b[ki-j] = sum_j b[j] * a[ki-j]
                # b[j] appears with coefficient a[ki-j] when ki-j is in [0,n-1]
                # i.e., j in [ki-n+1, ki] intersect [0, n-1]
                j_start = max(0, ki - n + 1)
                j_end = min(ki, n - 1)
                if j_start > j_end:
                    continue
                T[ii, j_start:j_end+1] = a[ki - np.arange(j_start, j_end+1)]
            
            b_var = cp.Variable(n, nonneg=True)
            constraints = [T @ b_var <= M_target, cp.sum(b_var) >= 0.01]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
            
            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                continue
            
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
                b_sum = b_val.sum()
                if b_sum > 1e-10:
                    break
        
        if b_val is None or b_val.sum() < 1e-10:
            K = min(K + 20, top_K_max)
            continue
        
        b_dir = b_val / b_val.sum()
        
        # Try multiple step sizes
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = fast_autocorr(a).max()
        best_trial_a = a.copy()
        
        alphas = [2.0/(iter_count+2), min(1.0, 2.5/(iter_count+2)), 0.5, 0.3, 0.2, 0.1]
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * b_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha
                best_trial_a = a_trial.copy()
        
        # Away-step: try pushing away from a "bad" coordinate
        if away_step and iter_count % 10 == 0:
            # Find the coordinate most contributing to the peak
            peak_k = int(np.argmax(fast_autocorr(a)))
            # The "bad" direction is the coordinate with max a[i] that contributes to peak
            s = max(0, peak_k - n + 1)
            e = min(peak_k, n - 1)
            if s <= e:
                contrib = a[s:e+1] * a[peak_k - np.arange(s, e+1)]
                bad_idx_local = np.argmax(contrib)
                bad_idx = s + bad_idx_local
                if a[bad_idx] > 0.01 / n:
                    # Try away-step: reduce bad_idx, compensate with b_dir
                    for away_frac in [0.01, 0.02, 0.05]:
                        amt = a[bad_idx] * away_frac
                        a_away = a.copy()
                        a_away[bad_idx] -= amt
                        a_away = np.maximum(a_away, 0.0)
                        a_away = a_away / a_away.sum()
                        # Then move toward b_dir
                        a_away2 = (1 - 0.3) * a_away + 0.3 * b_dir
                        a_away2 = np.maximum(a_away2, 0.0)
                        a_away2 = a_away2 / a_away2.sum()
                        m_away = fast_autocorr(a_away2).max()
                        if m_away < best_trial_m:
                            best_trial_m = m_away
                            best_trial_a = a_away2.copy()
        
        a = best_trial_a
        m_new = best_trial_m
        
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                # Perturb and reset
                pert = rng.normal(0, 0.01, n) * best_a
                a_p = np.maximum(best_a + pert, 1e-10)
                a = a_p / a_p.sum()
                stall = 0
                K = min(K + 50, top_K_max)
        
        iter_count += 1
        if iter_count % 10 == 0:
            K = min(K + 15, top_K_max)
        
        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, M={best_m:.8f}")
    
    return best_a, best_m


def coordinate_descent_optimize(a0, time_limit, verbose=False, rng_seed=123, 
                                 n_top_peaks=15, n_candidates=5):
    """
    Targeted coordinate descent: move mass from peak-contributing positions
    to low-impact positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    it = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        it += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top_peaks]
        
        # Compute sensitivity: which positions contribute most to peaks
        sens = np.zeros(n)
        peak_mask = np.zeros(n, dtype=bool)
        for k in top_k:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(ki, n - 1)
            if s > e:
                continue
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            sens[idx] += contrib
            peak_mask[idx] = True
        
        improved = False
        
        # Find top contributors to peaks and try to move their mass
        top_contributors = np.argsort(sens)[::-1][:30]
        
        for src in top_contributors:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-8:
                continue
            
            # Find good destinations: low sensitivity, not in peak
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            # Prefer low sensitivity destinations
            top_dsts = candidates[np.argsort(cand_sens)[:n_candidates]]
            
            for dst in top_dsts:
                if dst == src:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    amt = a[src] * frac
                    if amt < 1e-12:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_train, 0.0) if False else np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-17:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                stall = 0
                break
        
        if not improved:
            stall += 1
            # Random move to escape
            if a[src if len(top_contributors) > 0 else 0] > 1e-10:
                src_i = int(top_contributors[0]) if len(top_contributors) > 0 else int(rng.integers(n))
                dst_i = int(rng.integers(n))
                if src_i != dst_i and a[src_i] > 1e-10:
                    frac = rng.uniform(0.02, 0.1)
                    amt = a[src_i] * frac
                    a[src_i] -= amt
                    a[dst_i] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord it={it} ({elapsed:.0f}s): J={J:.6f}, M={best_m:.8f}, stall={stall}")
    
    return best_a, best_m


def pocs_spectral_init(n, n_trials=3, time_per=30.0, rng_seed=0):
    """POCS with flat-spectrum target for initialization."""
    best_a = np.ones(n) / n
    best_m = fast_autocorr(best_a).max()
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 2).bit_length()
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_per * n_trials:
            break
        rng = np.random.default_rng(rng_seed + trial)
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        A_spec = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
        A_full = np.zeros(fft_size, dtype=np.complex128)
        A_full[:fft_size // 2 + 1] = A_spec
        for i in range(1, fft_size // 2):
            A_full[fft_size - i] = np.conj(A_spec[i])
        a = np.fft.irfft(A_full, fft_size)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s
        
        # POCS iterations
        for it in range(20000):
            if time.monotonic() - t0 > time_per * (trial + 1):
                break
            # Frequency: push toward flat magnitude
            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            target = np.median(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            blend = min(0.3, 0.05 + 0.25 * it / 5000)
            new_mag = (1 - blend) * mag + blend * target
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            a = a_new
            
            M_new = fast_autocorr(a).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
    
    return best_a, best_m


def make_structured_init(n, kind='sine'):
    if kind == 'sine':
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif kind == 'u_shape':
        x = np.linspace(-1, 1, n)
        a = 1.0 + 2.0 * x ** 2
    elif kind == 'bimodal':
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.2) / 0.15) ** 2) + np.exp(-((x - 0.8) / 0.15) ** 2) + 0.01
    elif kind == 'cosine':
        x = np.arange(n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * x / n)
        a = a ** 2
    elif kind == 'triangular':
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * x - 1.0)
    else:
        a = np.ones(n)
    return norm_sum1(np.maximum(a, 0.0))


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    trial_lengths = [l for l in trial_lengths if l != n_best]
    
    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)...")
        a_long = interpolate_sequence(best_seq, new_n)
        
        # Quick MD
        t6a = min(remaining * 0.3, 120.0)
        if t6a > 20:
            a_out, _ = mirror_descent_optimize(a_long, t6a, verbose=False,
                                                top_k=5, temp_init=0.015,
                                                max_restarts=5, rng_seed=new_n % 1000)
            try_update(a_out, f"md_n{new_n}")
        
        # Quick LP-FW
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 5000:
            t6b = min(remaining * 0.4, 100.0)
            if t6b > 20:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                top_K = min(1500, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe_optimize(a_in, t6b, solver, verbose=False,
                                                    top_K_init=min(200, 2*new_n-1),
                                                    top_K_max=top_K,
                                                    multi_target=True,
                                                    away_step=True,
                                                    rng_seed=new_n % 500)
                try_update(a_out, f"lp_n{new_n}")
    
    # === Phase 7: Final combined refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Refinement (n={n_best}) ===")
    
    if remaining > 40:
        # MD
        t7a = min(remaining * 0.25, 120.0)
        if t7a > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = mirror_descent_optimize(a_in, t7a, verbose=False,
                                                top_k=3, temp_init=0.008,
                                                max_restarts=5, rng_seed=777)
            try_update(a_out, "md_final")
        
        # LP-FW
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and n_best <= 5000:
            t7b = min(remaining * 0.35, 150.0)
            if t7b > 20:
                a_in = norm_sum1(best_seq)
                top_K_init = min(500, 2 * n_best - 1)
                top_K_max = min(8000, 2 * n_best - 1)
                a_out, _ = lp_frank_wolfe_optimize(a_in, t7b, solver, verbose=False,
                                                    top_K_init=top_K_init,
                                                    top_K_max=top_K_max,
                                                    multi_target=True,
                                                    away_step=True,
                                                    rng_seed=888)
                try_update(a_out, "lp_final")
        
        # Coordinate
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7c = min(remaining * 0.4, 100.0)
            if t7c > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_descent_optimize(a_in, t7c, verbose=False,
                                                        rng_seed=555, n_top_peaks=20,
                                                        n_candidates=8)
                try_update(a_out, "coord_final")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        best_seq = make_structured_init(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
