def smooth_gradient_update(a, q, beta, lr):
    n = a.size
    m = 2*n - 1
    # Focus on top lags
    top_k = min(50, m)
    top_idx = np.argsort(q)[::-1][:top_k]
    top_q = q[top_idx]
    
    # Softmax weights
    vals = beta * (top_q - top_q[0])
    vals = np.clip(vals, -50, 0)
    w = np.exp(vals)
    w = w / w.sum()
    
    # Gradient: weighted sum of "shifted" versions of a
    g = np.zeros(n)
    for i, k in enumerate(top_idx):
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            g[i0:i1] += w[i] * a[start:stop][::-1]
    
    return g

def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]


def conv_ab(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]


def objective_norm(a):

def weights_from_q(q, beta):
    """Compute softmax weights on the convolution spectrum, focused on peaks."""
    q = np.asarray(q, dtype=np.float64)
    m = float(q.max())

def corr_shift(w, a):
    """Cross-correlation: c[j] = sum_k w[k] * a[k-j] for j=0..n-1."""
    w = np.asarray(w, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = w.size
    size = next_pow2(L + n - 1)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fw * fa.conj(), size)[:n]
    return np.maximum(c, 0.0)


def quad_max_alpha(alpha, q0, c, qd):
    """Compute max_k of (1-a)^2*q0[k] + 2*a*(1-a)*c[k] + a^2*qd[k]."""
    a1 = 1.0 - alpha
    val = (a1 * a1) * q0 + (2.0 * alpha * a1) * c + (alpha * alpha) * qd
    return float(np.max(val))


def exact_line_search(a, d, q0):
    """Exact 1D minimization of max conv along direction d from a."""
    a = np.asarray(a, dtype=np.float64)

        return a.copy(), M0

    qd = self_conv(d)

    best_alpha = 0.0
    best_M = M0

    lo = 0.0
    hi = 1.0
    c1 = hi - phi * (hi - lo)
    d1 = lo + phi * (hi - lo)
    f1 = quad_max_alpha(c1, q0, c, qd)

            d1 = c1
            f2 = f1
            c1 = hi - phi * (hi - lo)
            f1 = quad_max_alpha(c1, q0, c, qd)
        else:
            lo = c1
            c1 = d1
            f1 = f2
            d1 = lo + phi * (hi - lo)
            f2 = quad_max_alpha(d1, q0, c, qd)

    new = np.maximum(new, 0.0)
    s = float(new.sum())
    if s <= 1e-15:
        return a.copy(), M0
    new = new / s
    return new, best_M


def compute_direction(a, q, beta, top_k):
    """Compute the mirrored gradient direction using weighted top-k lags."""
    n = a.size
    m = q.size
    k_use = min(top_k, m)
    
    # Get top k lags
    if k_use < m:
        top_idx = np.argpartition(q, -k_use)[-k_use:]
        top_idx = top_idx[np.argsort(q[top_idx])[::-1]]
    else:
        top_idx = np.arange(m)
    
    top_q = q[top_idx]
    
    # Softmax weights on top lags
    tq_max = float(top_q.max())
    vals = beta * (top_q - tq_max)
    vals = np.clip(vals, -60.0, 0.0)
    w = np.exp(vals)
    s = float(w.sum())
    if s <= 1e-300:
        w = np.ones_like(top_q) / len(top_q)
    else:
        w = w / s
    
    # Build full weight vector (sparse)
    w_full = np.zeros(m, dtype=np.float64)
    w_full[top_idx] = w
    
    # Cross-correlation gives the gradient
    g = corr_shift(w_full, a)
    return g


def mirrored_update(a, g, lr):
    """Multiplicative weights update: a_new = a * exp(-lr * g) / Z."""
    move = -lr * g
    move = np.clip(move, -50.0, 50.0)
    new = a * np.exp(move)
    new = np.maximum(new, 0.0)
    s = float(new.sum())
    if s <= 1e-30:
        return a.copy()
    return new / s

def interpolate_sequence(f, new_n):
    f = normalize_vec(f)
    old_n = f.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return f.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f)
    return normalize_vec(g)

            out.append(normalize_vec(seq))

    # Cosine bumps
    for shift in (0.0, 0.1, 0.2):

        print("Parent: n=%d objective=%.10f" % (best_seq.size, best_score))
    else:
        n_default = 6500
        x = np.linspace(0.0, 1.0, n_default)
        best_seq = normalize_vec(np.sin(np.pi * x) ** 2)
        best_score = objective_norm(best_seq)
        print("Default: n=%d objective=%.10f" % (best_seq.size, best_score))

    # Phase 1: Diversified starts at the best length
    n0 = int(best_seq.size)
    starts = make_diverse_starts(n0, best_seq)
    cand_scores = []
    for s in starts:
        s = normalize_vec(s)
        sc = objective_norm(s)
        if np.isfinite(sc):
            cand_scores.append((sc, s))

    kept = select_diverse(cand_scores, 6)
    if len(kept) == 0:
        kept = [best_seq.copy()]

    # Ensure parent is in the list
    matched = False
    for k in kept:
        if k.size == best_seq.size and float(np.dot(k, best_seq)) > 0.999:
            matched = True
            break
    if not matched:
        kept.insert(0, best_seq.copy())

    # Phase 2: Multi-length exploration
    # Try a few different lengths with quick optimization
    remaining = total_budget - (time.monotonic() - t_global)
    lengths_to_try = []
    for dn in (-1000, -500, 500, 1000, 1500, 2000):
        nn = n0 + dn
        if 3500 <= nn <= 10000 and nn != n0:
            lengths_to_try.append(nn)
    # Deduplicate
    seen = set()
    unique_lengths = []
    for ln in lengths_to_try:
        if ln not in seen:
            seen.add(ln)
            unique_lengths.append(ln)

    for ln in unique_lengths[:4]:
        remaining = total_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        a_long = interpolate_sequence(best_seq, ln)
        t_lim = min(60.0, 0.15 * remaining)
        if t_lim < 20:
            continue
        out = gradient_optimize(a_long, t_lim, rng_seed=500 + ln % 997, verbose=False)
        best_seq, best_score = update_best(best_seq, best_score, out, "len_%d" % ln, t_global)

    # Phase 3: Main gradient optimization on best candidate
    remaining = total_budget - (time.monotonic() - t_global)
    main_time = min(1100.0, 0.55 * remaining)
    if main_time > 100:
        print("\nMain optimization: n=%d time=%.0fs" % (best_seq.size, main_time))
        out = gradient_optimize(best_seq, main_time, rng_seed=42, verbose=True)
        best_seq, best_score = update_best(best_seq, best_score, out, "main_grad", t_global)

    # Phase 4: Try second best diverse start
    remaining = total_budget - (time.monotonic() - t_global)
    if len(kept) >= 2 and remaining > 300:
        for i in range(1, min(3, len(kept))):
            remaining = total_budget - (time.monotonic() - t_global)
            if remaining < 150:
                break
            cand = kept[i]
            if cand.size == best_seq.size and float(np.dot(cand, best_seq)) > 0.995:
                continue
            t2 = min(200.0, 0.3 * remaining)
            if t2 < 60:
                continue
            out = gradient_optimize(cand, t2, rng_seed=200 + i * 37, verbose=False)
            best_seq, best_score = update_best(best_seq, best_score, out, "diverse_%d" % i, t_global)

    # Phase 5: More optimization on the best
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 200:
        t3 = min(300.0, 0.4 * remaining)
        if t3 > 80:
            print("\nSecond optimization pass: time=%.0fs" % t3)
            out = gradient_optimize(best_seq, t3, rng_seed=999, verbose=False)
            best_seq, best_score = update_best(best_seq, best_score, out, "pass2", t_global)

    # Phase 6: Transform search
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 60:
        t4 = min(60.0, 0.3 * remaining)
        if t4 > 10:
            out = transform_search(best_seq, t4)
            best_seq, best_score = update_best(best_seq, best_score, out, "transform", t_global)

    # Phase 7: Trim search
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 60:
        t5 = min(80.0, 0.3 * remaining)
        if t5 > 10:
            out = trim_search(best_seq, t5)
            best_seq, best_score = update_best(best_seq, best_score, out, "trim", t_global)

    # Phase 8: Coordinate polish
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 40:
        tc = min(100.0, 0.5 * remaining)
        if tc > 15:
            out = coordinate_polish(best_seq, tc, rng_seed=314159)
            best_seq, best_score = update_best(best_seq, best_score, out, "coord", t_global)

    # Phase 9: Final short gradient pass
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 40:
        tf = min(80.0, 0.6 * remaining)
        if tf > 15:
            out = gradient_optimize(best_seq, tf, rng_seed=7777, verbose=False)
            best_seq, best_score = update_best(best_seq, best_score, out, "final_grad", t_global)

    # Phase 10: Final transform
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 20:
        tt = min(30.0, 0.8 * remaining)
        if tt > 5:
            out = transform_search(best_seq, tt)
            best_seq, best_score = update_best(best_seq, best_score, out, "final_tf", t_global)

    # Final validation
    if best_seq is None:
        x = np.linspace(0.0, 1.0, 6500)
        best_seq = normalize_vec(np.sin(np.pi * x) ** 2)

    out_arr = np.asarray(best_seq, dtype=np.float64)
    out_arr = np.where(np.isfinite(out_arr), out_arr, 0.0)
    out_arr = np.maximum(out_arr, 0.0)
    out_arr = np.minimum(out_arr, 1000.0)
    s = float(out_arr.sum())
    if s <= 0.01:
        x = np.linspace(0.0, 1.0, 6500)
        out_arr = np.sin(np.pi * x) ** 2
        s = float(out_arr.sum())
    out_arr = out_arr / s

    final_score = objective_norm(out_arr)
    try:
        official = evaluate_safe(out_arr)
        if np.isfinite(official):
            final_score = min(final_score, official)
    except Exception:
        pass

    elapsed = time.monotonic() - t_global
    print("\nFINAL: score=%.10f n=%d time=%.1fs" % (final_score, out_arr.size, elapsed))

    return [float(v) for v in out_arr]
