def norm_sum1(a):

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad_full(y, beta, top_k=None):
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    a_rev = y[::-1]

    Ar = np.fft.rfft(a_rev, fft_size)

    return grad, p, b

    # Cosine
    for freq in [1, 2, 3]:
        for offset in [0, 0.3]:
            candidates.append(np.maximum((1.0 + offset * np.cos(2.0 * np.pi * freq * x)) / (1.0 + offset), 0.01))
    # Sine powers
    for p in [1, 2, 3, 4, 6]:
        candidates.append(np.sin(np.pi * x) ** p)
    # Exponential
    for rate in [0.5, 1.0, 2.0, 5.0]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(-rate * (1.0 - x)))
        candidates.append(np.exp(-rate * np.abs(x - 0.5)))

    rng = np.random.default_rng(42)
    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit * 0.6:
            break
        a = norm_sum1(a_raw)
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    # Random
    for _ in range(80):
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(rng.random(n) ** rng.uniform(0.3, 4.0))
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    return best_a, best_val


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(elapsed) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: Parametric search (quick)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Parametric (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.03, 40.0)
    if t1 > 10:
        a_param, _ = parametric_init(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "param", t_global)

    # Phase 2: Nesterov (main optimizer, longer than before)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.28, 450.0)
    if t2 > 60:
        a_out, _ = nesterov_projected(norm_sum1(best_seq), t2,
                                      beta_start=20.0, beta_end=700.0,
                                      beta_growth=1.005, eta_base=0.7,
                                      top_k_ratio=0.04, restart_every=2500,
                                      verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n" + str(n_best), t_global)

    # Phase 3: Fourier perturbation (explore)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Fourier Perturb (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.08, 100.0)
    if t3 > 20:
        a_out, _ = fourier_perturb(norm_sum1(best_seq), t3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fourier_n" + str(n_best), t_global)
        # Quick Nesterov from Fourier result
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 100:
            t3b = min(remaining * 0.15, 180.0)
            if t3b > 30:
                a_out, _ = nesterov_projected(norm_sum1(best_seq), t3b,
                                              beta_start=40.0, beta_end=600.0,
                                              beta_growth=1.005, eta_base=0.6,
                                              top_k_ratio=0.03, restart_every=2000,
                                              verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_fourier", t_global)

    # Phase 4: LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 40 and n_best <= 8000:
        a_out, _ = lp_fw_refined(norm_sum1(best_seq), t4, solver,
                                 initial_K=600, max_K=4000,
                                 add_per_iter=40, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(n_best), t_global)

    # Phase 5: Peak-safe coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: PeakSafe Coord (n=" + str(n_best) + ") ===")

        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakcoord_n" + str(n_best), t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-length ===")
    longer_lengths = [n_best + 400, n_best + 800]
    longer_lengths = [l for l in longer_lengths if l <= 9000]
    shorter_lengths = [n_best - 300]
    shorter_lengths = [l for l in shorter_lengths if l >= 3000]

        print("  Scaling to n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

                                          top_k_ratio=0.035, restart_every=2000,
                                          verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n" + str(new_n), t_global)

            if t6b > 30:
                a_long2 = interpolate_sequence(best_seq, new_n)

                                         initial_K=500, max_K=3000,
                                         add_per_iter=35, verbose=True)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(new_n), t_global)

    # Phase 7: Final Nesterov polish (high beta)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final Nesterov (n=" + str(n_best) + ") ===")
    if remaining > 80:
        t7 = min(remaining * 0.30, 200.0)
        if t7 > 30:
            a_out, _ = nesterov_projected(norm_sum1(best_seq), t7,
                                          beta_start=60.0, beta_end=1000.0,
                                          beta_growth=1.006, eta_base=0.5,
                                          top_k_ratio=0.025, restart_every=1500,
                                          verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

    # Phase 8: Final peak-safe coord
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final PeakCoord (n=" + str(n_best) + ") ===")
    if remaining > 30:
        t8 = min(remaining * 0.40, 120.0)
        if t8 > 15:
            a_out, _ = peak_safe_coord(norm_sum1(best_seq), t8, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peakcoord_final", t_global)

    # Phase 9: Final LP burst
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

        print("\n=== Phase 9: Final LP burst ===")
        t9 = min(remaining * 0.60, 180.0)
        if t9 > 25:
            a_out, _ = lp_fw_refined(norm_sum1(best_seq), t9, solver,

                                     add_per_iter=50, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_burst", t_global)

    # Phase 10: Very final coord polish

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):

    else:

        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
