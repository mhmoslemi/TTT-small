def simplex_project(z):
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

    no_improve = 0
    max_no_improve = 30000

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        t += 1

        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_M - 1e-16:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t % 50 == 1:
            threshold = M * 0.92
            n_keep = max(40, int(size * 0.04))
            sorted_idx = np.argsort(ac)[::-1][:n_keep]
            active = np.zeros(size, dtype=np.float64)
            active[sorted_idx] = 1.0
            above = np.where(ac > threshold)[0]
            active[above] = 1.0

        A_fft = np.fft.rfft(a, press_fft_size)
        B_fft = np.fft.rfft(active, press_fft_size)
        corr = np.fft.irfft(np.conj(B_fft) * A_fft, press_fft_size)
        pressure = np.maximum(corr[:n], 0.0)

        i_star = int(np.argmin(pressure))

        # Backtracking: try alpha = 1, 0.5, 0.25, 0.125, 0.0625
        alphas = [1.0, 0.5, 0.25, 0.125, 0.0625]
        best_alpha = 0.0625
        best_trial_M = M  # current value (fallback)

        for alpha in alphas:
            a_trial = a * (1.0 - alpha)
            a_trial[i_star] += alpha
            a_trial = symmetrize(a_trial)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            # Quick evaluation: check only active positions
            ac_trial = fast_autocorr(a_trial)
            M_trial = ac_trial.max()
            if M_trial < best_trial_M - 1e-18:
                best_trial_M = M_trial
                best_alpha = alpha
                if alpha >= 0.5:
                    break  # Good enough

        a_new = a * (1.0 - best_alpha)
        a_new[i_star] += best_alpha
        a_new = symmetrize(a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    FW-BT t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_M


def nesterov_polish(a0, time_limit, beta=100.0, verbose=True):
    """Nesterov accelerated gradient on log-sum-exp smoothed objective."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_M = fast_autocorr(x).max()
    t0 = time.monotonic()
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(777)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        y = y / s_y

        # Compute gradient of log-sum-exp smoothed max
        ac_y = fast_autocorr(y)
        b_max = ac_y.max()
        shifted = beta * (ac_y - b_max)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        p = exp_vals / exp_vals.sum()

        # Gradient: sum_k p[k] * (a[k-i] + a[i-k]) / 2  ... = sum_k p[k] * a[k-i]
        # Use cross-correlation
        p_fft_size = press_fft_size = 1 << (3 * n - 1).bit_length()
        P_fft = np.fft.rfft(p, p_fft_size)
        Y_fft = np.fft.rfft(y, p_fft_size)
        corr = np.fft.irfft(np.conj(P_fft) * Y_fft, p_fft_size)
        grad = corr[:n] / beta
        grad = np.maximum(grad, 0.0)

        # Step size
        L = beta * p.max() * 2.0 + 1e-8
        eta = 1.0 / L
        eta = min(eta, 0.3)

        z = y - eta * grad
        x_new = simplex_project(z)

        ac_new = fast_autocorr(x_new)
        M_new = ac_new.max()
        if M_new < best_M - 1e-16:
            best_M = M_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if stall > 2000:
            stall = 0
            t_nest = 1.0
            noise = 0.03 / (1.0 + 0.05 * (t // 2000))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r = symmetrize(x_r)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Nest t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_M


def coordinate_descent(a0, time_limit, verbose=True):
    """Peak-targeted coordinate descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_M = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    max_stall = 3000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        if M < best_M - 1e-16:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Find top peaks
        top_k = np.argsort(ac)[::-1][:min(30, len(ac))]
        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            src_val = a[src]
            partner_val = a[partner] if 0 <= partner < n else 0.0

            if src_val >= partner_val:
                reduce_idx = src
            else:
                reduce_idx = partner

            if a[reduce_idx] < 1e-12:
                continue

            # Find low-pressure destination
            # Use the current autocorrelation to identify "quiet" positions
            # A position j is "quiet" if it doesn't contribute much to any top peak
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)

            # Pick destination with minimum a value among candidates
            dst = int(candidates[np.argmin(a[candidates])])
            if dst == reduce_idx:
                # Pick next best
                mask = candidates != reduce_idx
                if mask.sum() > 0:
                    dst = int(candidates[mask][np.argmin(a[candidates[mask]])])
                else:
                    continue

            # Try moves
            found = False
            for frac in [0.05, 0.02, 0.01, 0.005]:
                amt = a[reduce_idx] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = symmetrize(a_trial)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_M - 1e-17:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 500:
                # Random exploration
                j = int(rng.integers(n))
                if j != reduce_idx and a[reduce_idx] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[reduce_idx] * frac
                    a[reduce_idx] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = symmetrize(a)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_M:
                            best_M = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1

        if verbose and stall % 2000 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f} stall={stall}")

    return best_a, best_M


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main search: Frank-Wolfe (exact max) + Nesterov polish + coordinate descent.
    Exploits convexity of the min-max autocorrelation problem.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # Phase 1: FW on parent length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: FW (n={n0}) ===")
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 60:
        a_out, _ = fw_algorithm(norm_sum1(best_seq), t1, use_away=True, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"fw1_n{n0}", t_global)

    # Phase 2: Nesterov polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Nesterov polish (n={n_best}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 40:
        a_out, _ = nesterov_polish(norm_sum1(best_seq), t2, beta=80.0, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"nest_n{n_best}", t_global)

    # Phase 3: FW with backtracking
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: FW-BT (n={n_best}) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40:
        a_out, _ = fw_with_backtrack(norm_sum1(best_seq), t3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"fwbt_n{n_best}", t_global)

    # Phase 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord (n={n_best}) ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 30:
        a_out, _ = coordinate_descent(norm_sum1(best_seq), t4, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"coord_n{n_best}", t_global)

    # Phase 5: Try longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    new_n = min(n_best + 800, 8000)
    if new_n > n_best and remaining > 200:
        print(f"\n=== Phase 5: Scale to n={new_n} ===")
        # Interpolate
        old_n = n_best
        f_norm = best_seq / best_seq.sum()
        old_x = np.linspace(0.0, 1.0, old_n)
        new_x = np.linspace(0.0, 1.0, new_n)
        f_interp = np.interp(new_x, old_x, f_norm)
        f_interp = np.maximum(f_interp, 0.0)
        f_interp = symmetrize(f_interp)
        f_interp = f_interp / f_interp.sum()

        t5a = min(remaining * 0.35, 350.0)
        if t5a > 50:
            a_out, _ = fw_algorithm(f_interp, t5a, use_away=True, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, f"fw_n{new_n}", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 150:
            n_best = len(best_seq)
            t5b = min(remaining * 0.35, 200.0)
            if t5b > 40:
                a_out, _ = nesterov_polish(norm_sum1(best_seq), t5b, beta=100.0, verbose=True)
                best_seq, best_score = try_update(best_seq, best_score, a_out, f"nest_n{new_n}", t_global)

    # Phase 6: Final Nesterov with higher beta
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final Nesterov (n={n_best}) ===")
    if remaining > 60:
        t6 = min(remaining * 0.40, 300.0)
        if t6 > 30:
            a_out, _ = nesterov_polish(norm_sum1(best_seq), t6, beta=200.0, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, f"nest_final_n{n_best}", t_global)

    # Phase 7: Final coordinate
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 7: Final Coord ===")
        t7 = min(remaining * 0.45, 150.0)
        if t7 > 15:
            a_out, _ = coordinate_descent(norm_sum1(best_seq), t7, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final FW burst
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 8: Final FW burst ===")
        t8 = min(remaining * 0.80, 300.0)
        if t8 > 30:
            a_out, _ = fw_algorithm(norm_sum1(best_seq), t8, use_away=True, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fw_final", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.1f}s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
