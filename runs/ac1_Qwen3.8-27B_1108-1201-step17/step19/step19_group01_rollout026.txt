def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

            best_m = m
            best_a = a_new.copy()
        a = a_new
        if verbose and it % 50 == 0:
            print("    SpecShape it={}: max_b={:.6f}".format(it, m))

    return best_a, best_m


def spectral_shape_multi(a, verbose=False):
    """Try multiple alpha values for spectral shaping."""
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    for alpha in [0.3, 0.5, 0.7, 0.85]:
        a_out, m = spectral_shape(a, alpha=alpha, n_iter=80, verbose=False)
        if m < best_m:
            best_m = m
            best_a = a_out.copy()
    return best_a, best_m


def lp_fw_v2(a0, time_limit, solver, initial_K=800, max_K=5000,
             verbose=True):
    """
    Improved LP-Frank-Wolfe with:
    - Aggressive violation detection
    - Margin-based constraint removal
    - Parabolic line search
    - Adaptive restart
    """
    n = len(a0)
    y = norm_sum1(a0).copy()
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Initial active set: top K autocorrelation values
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 2000
    iter_count = 0

    # Pre-compute FFT of y for reuse
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW v2: converged/stalled at iter {}".format(iter_count))
            # Restart with perturbation
            rng = np.random.default_rng(iter_count)
            pert = rng.normal(0, 0.02, n) * best_a
            y = best_a + pert
            y = np.maximum(y, 1e-10)
            y = y / y.sum()
            no_improve = 0
            active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K])
            continue

        # Prune active set: remove constraints with large margin
        K = len(active_set)
        if K > max_K // 2:
            margins = np.array([M_curr - ac[k] for k in active_set])
            KEEP_FRAC = 0.6
            keep_count = max(initial_K // 2, int(K * KEEP_FRAC))
            keep_idx = np.argsort(margins)[:keep_count]
            active_set = set(int(list(active_set)[i]) for i in keep_idx)

        constraint_list = sorted(active_set)
        K = len(constraint_list)
        if K == 0:
            active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])
            continue

        # Build constraint matrix: coeffs[k_idx, i] = y[k_i - i] if valid
        # c_k(g) = sum_i g_i * y_{k-i} (linearization of autocorrelation at y)
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k - idx]

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Fallback: reduce active set
            if K > initial_K:
                active_set = set(list(active_set)[:initial_K])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations: lags where c_k(g_dir) > M_curr
        conv_yc = convolve_pair(y, g_dir)
        violations = np.where(conv_yc > M_curr + 1e-8)[0]
        if len(violations) > 0:
            # Add top violations (limit to avoid too-large constraint set)
            viol_amt = conv_yc[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:200]]
            for k in top_v:
                active_set.add(int(k))

        # Line search with parabolic interpolation
        # Try several alpha values
        base_alpha = 2.0 / (iter_count + 2.0)
        alpha_candidates = [base_alpha, base_alpha * 0.5, base_alpha * 1.5,
                           0.1, 0.2, 0.3, 0.5, 1.0 / (iter_count + 1)]
        alpha_candidates = list(set([max(0.01, min(1.0, a)) for a in alpha_candidates]))

        best_alpha = base_alpha
        best_trial_M = float('inf')
        trial_values = []
        for alpha in alpha_candidates:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                trial_values.append((alpha, float('inf')))
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            trial_values.append((alpha, M_trial))
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        # Parabolic refinement if we have 3+ finite values
        finite_vals = [(a, m) for a, m in trial_values if np.isfinite(m)]
        if len(finite_vals) >= 3:
            finite_vals.sort()
            # Try midpoints between best neighbors
            best_idx = int(np.argmin([m for _, m in finite_vals]))
            neighbors = []
            if best_idx > 0:
                neighbors.append(finite_vals[best_idx - 1])
            if best_idx < len(finite_vals) - 1:
                neighbors.append(finite_vals[best_idx + 1])
            for na, nm in neighbors:
                mid_alpha = (na + finite_vals[best_idx][0]) / 2.0
                trial = (1.0 - mid_alpha) * y + mid_alpha * g_dir
                trial = np.maximum(trial, 0.0)
                s_t = trial.sum()
                if s_t > 1e-15:
                    trial = trial / s_t
                    M_trial = fast_autocorr(trial).max()
                    if M_trial < best_trial_M:
                        best_trial_M = M_trial
                        best_alpha = mid_alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW v2 iter={} ({:.0f}s): J={:.6f} |S|={}".format(
                iter_count, elapsed, J, len(active_set)))

    return best_a, best_m


def nesterov_v2(a0, time_limit, beta_start=20.0, beta_end=800.0,
                beta_growth=1.005, eta_base=0.7, top_k_ratio=0.05,
                restart_every=2000, verbose=True):
    """Nesterov accelerated projected gradient with adaptive smoothing."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(123)
    top_k = max(50, int(n * top_k_ratio))
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        b = fast_autocorr(y)
        if top_k < len(b):
            idx = np.argsort(b)[::-1][:top_k]
            b_use = np.zeros(len(b))
            b_use[idx] = b[idx]
            b_max = b[idx].max()
        else:
            b_use = b
            b_max = b.max()

        shifted = np.clip(beta * (b_use - b_max), -300, 300)
        exp_vals = np.exp(shifted)
        p = exp_vals / exp_vals.sum()

        # Gradient via convolution
        a_rev = y[::-1]
        conv_len = n + len(p) - 1
        cfft = 1 << (conv_len - 1).bit_length()
        Ar = np.fft.rfft(a_rev, cfft)
        Pr = np.fft.rfft(p, cfft)
        conv = np.fft.irfft(Ar * Pr, cfft)
        grad = conv[n - 1:2 * n - 1] / beta

        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.03 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest v2 restart t={} ({:.0f}s): J={:.6f}".format(
                    t, elapsed, J))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest v2 t={} ({:.0f}s): J={:.6f} beta={:.0f}".format(
                t, elapsed, J, beta))

    return best_a, best_max_b


def coord_descent_v3(a0, time_limit, verbose=True):
    """Peak-targeted coordinate descent with gradient-based destination selection."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    max_stall = 600

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:30]
        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_ci = np.argmax(contrib)
            src = int(idx[max_ci])
            partner = k_int - src

            src_val = a[src]
            partner_val = a[partner] if 0 <= partner < n else 0.0

            if src_val >= partner_val:
                reduce_idx = src
            else:
                reduce_idx = partner

            if a[reduce_idx] < 1e-12:
                continue

            # Find low-pressure destination
            pressure = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)
            candidates = candidates[candidates != reduce_idx]
            if len(candidates) == 0:
                continue

            cand_pressure = pressure[candidates]
            dst = int(candidates[np.argmin(cand_pressure)])

            found = False
            for frac in [0.02, 0.04, 0.08, 0.12, 0.2]:
                amt = a[reduce_idx] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 150:
                j = int(rng.integers(n))
                if j != reduce_idx and a[reduce_idx] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[reduce_idx] * frac
                    a[reduce_idx] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 200 == 0:
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.003, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord v3 ({:.0f}s): J={:.6f} stall={}".format(
                elapsed, J, stall))

    return best_a, best_m


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n, best_init, best_score, time_limit=None, t0=None):
    """Parametric initialization search."""
    if t0 is None:
        t0 = time.monotonic()
    if time_limit is None:
        time_limit = 30.0

    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score

    candidates = []
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    for p in [0.5, 1.0, 1.5]:
        candidates.append((i_arr + 1.0) ** p * (n - i_arr) ** p)
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.3]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    for p in [1, 2, 3, 4]:
        candidates.append(np.sin(np.pi * x) ** p)
    for freq in [1, 2, 3]:
        candidates.append(np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x)) / 2.0, 0.01))
    # Exponential
    for rate in [0.5, 1.0, 2.0, 5.0]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(-rate * (1.0 - x)))
    # Two-hump
    for c1, c2 in [(0.25, 0.75), (0.2, 0.8), (0.3, 0.7)]:
        candidates.append(np.exp(-((x - c1) ** 2) / 0.02) + np.exp(-((x - c2) ** 2) / 0.02))

    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit + 30:
            break
        a = norm_sum1(a_raw)
        b = fast_autocorr(a)
        m = b.max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    rng = np.random.default_rng(42)
    for _ in range(30):
        if time.monotonic() - t0 > time_limit + 30:
            break
        a = norm_sum1(rng.random(n) ** rng.uniform(0.5, 3.0))
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    return best_a, best_val


def propose_candidate(budget_s=1800.0):
    """
    Main search function combining multiple optimization strategies:
    - LP-Frank-Wolfe (primary optimizer)
    - Spectral shaping (diversity/shaking)
    - Nesterov gradient (polishing)
    - Coordinate descent (fine tuning)
    - Multi-length exploration
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Time budget: {:.0f}s".format(time_budget))

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n={}, score={:.6f}".format(len(parent), p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n={}, score={:.6f}".format(len(best_seq), best_score))

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            old_score = best_score
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print("  *** NEW BEST [{}]: {:.6f} (n={}, t={:.0f}s, delta={:.6f})".format(
                label, sc, len(a), elapsed, old_score - sc))

    n0 = len(best_seq)

    # Phase 1: Parametric + Spectral shaping initialization (30s)
    print("\n=== Phase 1: Init (n={}) ===".format(n0))
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        a_param, _ = parametric_init(n0, best_seq, 2.0, time_limit=15, t0=time.monotonic())
        try_update(a_param, "param")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_spec, _ = spectral_shape_multi(best_seq, verbose=True)
            try_update(a_spec, "spec_init")

    # Phase 2: Main LP-FW optimization (50% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: LP-FW v2 (n={}) ===".format(n_best))
    t2 = min(remaining * 0.45, 700.0)
    if t2 > 60:
        a_out, _ = lp_fw_v2(norm_sum1(best_seq), t2, solver,
                            initial_K=min(1000, n_best // 5),
                            max_K=min(5000, n_best // 2),
                            verbose=True)
        try_update(a_out, "lpv2_n{}".format(n_best))

    # Phase 3: Spectral shaping shake (30s)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 3: Spectral shake ===")
    if remaining > 30:
        a_spec, _ = spectral_shape_multi(best_seq, verbose=True)
        try_update(a_spec, "spec_shake")
        # Also try from the spectral-shaped version, run LP-FW
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 120:
            n_s = len(a_spec)
            t3b = min(remaining * 0.4, 200.0)
            a_out, _ = lp_fw_v2(norm_sum1(a_spec), t3b, solver,
                                initial_K=min(800, n_s // 5),
                                max_K=min(4000, n_s // 2),
                                verbose=True)
            try_update(a_out, "lpv2_from_spec")

    # Phase 4: Nesterov gradient polishing (15% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Nesterov v2 (n={}) ===".format(n_best))
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 40:
        a_out, _ = nesterov_v2(norm_sum1(best_seq), t4,
                               beta_start=30.0, beta_end=1000.0,
                               beta_growth=1.005, eta_base=0.6,
                               top_k_ratio=0.03, restart_every=2000,
                               verbose=True)
        try_update(a_out, "nestv2_n{}".format(n_best))

    # Phase 5: LP-FW again with refined active set
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: LP-FW v2 refined (n={}) ===".format(n_best))
    t5 = min(remaining * 0.25, 300.0)
    if t5 > 60:
        a_out, _ = lp_fw_v2(norm_sum1(best_seq), t5, solver,
                            initial_K=min(1200, n_best // 4),
                            max_K=min(5000, n_best // 2),
                            verbose=True)
        try_update(a_out, "lpv2_refined")

    # Phase 6: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Coord v3 (n={}) ===".format(n_best))
    t6 = min(remaining * 0.12, 120.0)
    if t6 > 20:
        a_out, _ = coord_descent_v3(norm_sum1(best_seq), t6, verbose=True)
        try_update(a_out, "coordv3")

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Multi-length ===")
    test_lengths = []
    for delta in [500, 1000, -500, 1500]:
        nl = n_best + delta
        if 3000 <= nl <= 10000 and nl != n_best:
            test_lengths.append(nl)

    for new_n in test_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("  Trying n={}".format(new_n))
        a_long = interpolate_seq(best_seq, new_n)

        # Quick spectral shape at new length
        t7a = min(remaining * 0.2, 60.0)
        if t7a > 10:
            a_s, _ = spectral_shape(a_long, alpha=0.6, n_iter=50)
            try_update(a_s, "spec_n{}".format(new_n))

        # LP-FW at new length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 100 and new_n <= 8000:
            t7b = min(remaining * 0.5, 200.0)
            a_out, _ = lp_fw_v2(norm_sum1(best_seq), t7b, solver,
                                initial_K=min(800, new_n // 5),
                                max_K=min(4000, new_n // 2),
                                verbose=False)
            try_update(a_out, "lpv2_n{}".format(new_n))

    # Phase 8: Final LP-FW burst
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final LP-FW burst ===")
    if remaining > 60:
        t8 = min(remaining * 0.6, 200.0)
        a_out, _ = lp_fw_v2(norm_sum1(best_seq), t8, solver,
                            initial_K=min(1500, n_best // 3),
                            max_K=min(6000, n_best // 2),
                            verbose=True)
        try_update(a_out, "lpv2_final")

    # Phase 9: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t9 = min(remaining * 0.7, 80.0)
        a_out, _ = coord_descent_v3(norm_sum1(best_seq), t9, verbose=False)
        try_update(a_out, "coord_final")

    # Validation and final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.6f}, n={}, time={:.0f}s".format(
            best_score, len(best_seq), elapsed))
    else:
        print("ERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    # Ensure output is valid
    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = best_seq[np.isfinite(best_seq)]
    if len(best_seq) == 0:
        best_seq = np.ones(100)

    return [float(x) for x in best_seq]
