def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_autocorr_and_gradient(a, top_k=50):
    """Compute autocorrelation and subgradient for top lags."""
    ac = fast_autocorr(a)
    n = len(a)
    top_idx = np.argsort(ac)[::-1][:top_k]
    
    # Weight by autocorrelation values
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** 6.0
    w = w / w.sum()
    
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        g[idx] += 2.0 * wi * a[j_idx]
    return ac, g, top_idx


def subgradient_nesterov(a0, time_limit, top_k=80, verbose=False, rng_seed=0):
    """Nesterov-accelerated subgradient with multiple active lags."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    y = a.copy()
    beta_prev = 0.0
    eta = 0.5
    stall = 0
    max_stall = 3000
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 50 == 0 and it % 5000 == 0:
            pass  # just for structure
        
        # Nesterov momentum
        beta = (1.0 + np.sqrt(1.0 + 4.0 * beta_prev ** 2)) / 2.0
        y = x + (beta_prev / beta) * (x - y)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute gradient
        ac, g, top_idx = compute_autocorr_and_gradient(y, top_k=top_k)
        
        # Adaptive step size
        eta = 1.0 / (np.sqrt(it + 1.0) + 1.0)
        
        # Gradient step
        a_trial = y - eta * g
        a_new = project_simplex(a_trial)
        
        # Check improvement
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        x = a_new
        beta_prev = beta
        
        # Restart if stuck
        if stall > max_stall:
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            y = x_r.copy()
            beta_prev = 0.0
            stall = 0
            top_k = min(top_k + 20, 200)
        
        if verbose and it % 2000 == 0:

            J = 2.0 * n * best_m
            print(f"    SG it={it} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.4f}, stall={stall}")

def lp_frank_wolfe_away(a0, time_limit, solver, verbose=False, top_K_init=100,
                         max_iterations=None, away_coeff=0.5):
    """LP Frank-Wolfe with away-step and adaptive active set."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    if max_iterations is None:
        max_iterations = 100000
    
    K = top_K_init
    iter_count = 0
    
    # Pre-build constraint matrix structure
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]
        
        # Build constraint matrix T: T[i] = a[binding_idx[i] - j] for valid j
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 20, min(500, 2 * n - 1))
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        s_dir = b_val / b_sum  # linear minimization oracle
        
        # Away-step: identify the "away" direction
        # The away vertex is e_j where j has the largest component in a
        away_idx = np.argmax(a)
        away_dir = np.zeros(n)
        away_dir[away_idx] = 1.0
        
        # Standard FW step
        alpha_fw = 2.0 / (iter_count + 2.0)
        
        # Try away-step
        best_alpha = alpha_fw
        best_trial_m = fast_autocorr(a).max()
        
        # Try standard FW
        a_trial = (1.0 - alpha_fw) * a + alpha_fw * s_dir
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t > 1e-15:
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha_fw
                best_is_away = False
            else:
                best_is_away = False
        
        # Try away-step: a_new = (1-alpha)*a + alpha*s - lambda*(1-alpha)*e_away / (1 - alpha*lambda)
        for lam in [away_coeff * 0.5, away_coeff, away_coeff * 1.5]:
            for alpha in [alpha_fw, alpha_fw * 0.5, 0.5]:
                denom = 1.0 - alpha * lam
                if denom < 0.1:
                    continue
                a_away = (1.0 - alpha) * a + alpha * s_dir - lam * (1.0 - alpha) / denom * away_dir
                a_away = np.maximum(a_away, 0.0)
                s_a = a_away.sum()
                if s_a < 1e-15:
                    continue
                a_away = a_away / s_a
                m_away = fast_autocorr(a_away).max()
                if m_away < best_trial_m:
                    best_trial_m = m_away
                    best_alpha = alpha
                    best_is_away = True
                    best_lam = lam
        
        # Apply update
        if best_is_away:
            denom = 1.0 - best_alpha * best_lam
            if denom > 0.1:
                a = (1.0 - best_alpha) * a + best_alpha * s_dir - best_lam * (1.0 - best_alpha) / denom * away_dir
        else:
            a = (1.0 - best_alpha) * a + best_alpha * s_dir
        
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a
        
        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()
        
        iter_count += 1
        
        # Adapt K
        if iter_count % 20 == 0:
            K = min(K + 10, min(500, 2 * n - 1))
        
        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LPFW-A it={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")
    
    return best_a, best_m

        # Compute autocorrelation to find peaks
        ac = fast_autocorr(a)
        ac_max = ac.max()
        
        # Identify which frequency bins contribute most to the peak
        # The autocorrelation at lag k is: sum_f |A(f)|^2 * exp(2*pi*i*f*k/N)
        # To reduce the peak, we want to reduce |A(f)|^2 where it correlates with the peak lag
        
        # Use a soft threshold: reduce magnitudes where |A|^2 is large
        # This approximates making the spectrum flatter
        mag_target = np.ones_like(mag)
        
        # Compute "flatness" score
        mag_nonzero = mag[mag > 1e-10]
        if len(mag_nonzero) > 0:
            mag_mean = np.mean(mag_nonzero)
            # Blend toward flat
            alpha = 0.05 + 0.1 * (it / min(it + 1, 100))
            mag_new = (1 - alpha) * mag + alpha * mag_mean
            # But don't increase magnitudes that are already small
            mag_new = np.where(mag < mag_mean * 0.5, mag, mag_new)
        else:
            mag_new = mag
        
        A_new = mag_new * phase
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            stall += 1
            continue
        a_new = a_new / s
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            # Try smaller step
            if stall > 100:
                # Revert and try with smaller alpha
                alpha = 0.02
                mag_new2 = (1 - alpha) * mag + alpha * mag_mean
                mag_new2 = np.where(mag < mag_mean * 0.5, mag, mag_new2)
                A_new2 = mag_new2 * phase
                a_new2 = np.fft.irfft(A_new2, fft_size)[:n]
                a_new2 = np.maximum(a_new2, 0.0)
                s2 = a_new2.sum()
                if s2 > 1e-15:
                    a_new2 = a_new2 / s2
                    m2 = fast_autocorr(a_new2).max()
                    if m2 < best_m:
                        best_m = m2
                        best_a = a_new2.copy()
                        a = a_new2
                        stall = 0
                    else:
                        # Add random perturbation
                        rng = np.random.default_rng(it)
                        pert = rng.normal(0, 0.01, n) * a
                        a_p = np.maximum(a + pert, 0.0)
                        a_p = a_p / a_p.sum()
                        mp = fast_autocorr(a_p).max()
                        if mp < best_m:
                            best_m = mp
                            best_a = a_p.copy()
                            a = a_p
                            stall = 0
                        else:
                            a = a_new  # accept the spectral step anyway for diversity
    
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FreqShave it={it} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def coordinate_polish_fast(a0, time_limit, verbose=False, rng_seed=0):
    """Fast coordinate descent: move mass from high-contribution coords to low-contribution ones."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 50
    n_top = 30
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_top]
        
        # Compute sensitivity: how much does each coordinate contribute to peak lags?
        sens = np.zeros(n)
        for k in top_k[:15]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[ki - idx]
        
        improved = False
        
        # Find coordinates with high sensitivity that are large
        high_sens = np.where(sens > np.median(sens) * 2)[0]
        high_sens = high_sens[a[high_sens] > np.median(a) * 1.5]
        
        if len(high_sens) == 0:
            high_sens = np.argsort(sens * a)[::-1][:10]
        
        low_sens = np.where(sens < np.median(sens))[0]
        if len(low_sens) == 0:
            low_sens = np.argsort(sens)[:n // 4]
        
        # Try moving mass from high-sens to low-sens coordinates
        for src_idx in high_sens[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src_idx)
            if a[src_i] < 1e-12:
                continue
            
            # Choose destination: low sensitivity, preferentially near edges
            candidates = low_sens[np.argsort(sens[low_sens])[:10]]
            if len(candidates) == 0:
                continue
            
            for dst_i in candidates[:3]:
                if dst_i == src_i:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    amt = a[src_i] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def pocs_flat_spectrum(a0, time_limit, verbose=False, n_trials=5, rng_seed=0):
    """POCS: alternate between flat-spectrum projection and non-negativity."""
    n = len(a0)
    best_a = a0.copy()
    best_m = fast_autocorr(a0).max()
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 2).bit_length()
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit - 5:
            break
        t_trial = min((time_limit - (time.monotonic() - t0)) / max(1, n_trials - trial), 60.0)
        
        rng = np.random.default_rng(rng_seed + trial * 17)
        
        # Initialize with random flat spectrum
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        A_full = np.zeros(fft_size, dtype=np.complex128)
        A_full[:fft_size // 2 + 1] = np.exp(1j * phases)
        for i in range(1, fft_size // 2):
            A_full[fft_size - i] = np.conj(A_full[i])
        a = np.fft.irfft(A_full, fft_size)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s
        
        M = fast_autocorr(a).max()
        if M < best_m:
            best_m = M
            best_a = a.copy()
        
        t_trial_start = time.monotonic()
        it = 0
        while time.monotonic() - t_trial_start < t_trial:
            it += 1
            
            # Step 1: Project onto flat spectrum in frequency domain
            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            phase = np.angle(A)
            
            # Smooth toward flat magnitude
            mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            alpha = 0.3 + 0.4 * min(1.0, it / 50.0)
            new_mag = (1 - alpha) * mag + alpha * mag_mean
            A_new = new_mag * phase
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            
            # Step 2: Project onto non-negative, sum=1
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            
            M_new = fast_autocorr(a_new).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a_new.copy()
            a = a_new
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    POCS ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.1) / width) ** 2) + np.exp(-((x - 0.9) / width) ** 2) + 0.001
    return norm_sum1(a)


def make_w_shape(n, width=0.12):
    x = np.linspace(0, 1, n)
    centers = [0.05, 0.4, 0.6, 0.95]
    a = np.ones(n) * 0.001
    for c in centers:
        a += np.exp(-((x - c) / width) ** 2)
    return norm_sum1(a)


def make_edge_concentrated(n, power=2.0):
    x = np.linspace(0, 1, n)
    a = x ** power + (1 - x) ** power + 0.001
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def mirror_descent_away(a0, time_limit, top_k=60, beta=6.0, c0=5.0,
                        restart_every=5000, verbose=False, rng_seed=42):
    """Away-step mirror descent with adaptive parameters."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        g, _, _ = compute_autocorr_and_gradient(y, top_k=cur_top_k)
        
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        # Away step: reduce the largest component slightly
        if t % 10 == 0:
            max_idx = np.argmax(a_new)
            a_new[max_idx] *= 0.95
            s_new2 = a_new.sum()
            if s_new2 > 1e-15:
                a_new = a_new / s_new2
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Adaptive parameters
        if stall_count > restart_every // 2:
            cur_beta = max(cur_beta * 0.9, 2.0)
            cur_top_k = min(cur_top_k + 10, 200)
        
        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 20, 250)
            cur_beta = max(cur_beta - 1.0, 2.0)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD-A t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")
    
    return best_a, best_m


def combined_polish(a0, time_limit, solver, verbose=False, rng_seed=0):
    """Combined polishing: alternate between LP-FW, subgradient, and coordinate descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    phase_time = min(60.0, time_limit / 4)
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break
        
        # Phase A: Quick LP-FW
        t_a = min(phase_time * 0.4, remaining * 0.3)
        if t_a > 10 and n <= 6000:
            a_out, _ = lp_frank_wolfe_away(a, t_a, solver, verbose=False, top_K_init=150)
            if fast_autocorr(a_out).max() < best_m:
                best_m = fast_autocorr(a_out).max()
                best_a = a_out.copy()
                a = a_out.copy()
                if verbose:
                    print(f"      Polish LPFW: J={2.0*n*best_m:.6f}")
        
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break
        
        # Phase B: Subgradient
        t_b = min(phase_time * 0.3, remaining * 0.3)
        if t_b > 10:
            a_out, m_out = subgradient_nesterov(a, t_b, top_k=80, rng_seed=int(time.monotonic()*1000) % 10000)
            if m_out < best_m:
                best_m = m_out
                best_a = a_out.copy()
                a = a_out.copy()
                if verbose:
                    print(f"      Polish SG: J={2.0*n*best_m:.6f}")
        
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        # Phase C: Coordinate descent
        t_c = min(phase_time * 0.3, remaining * 0.3)
        if t_c > 5:
            a_out, m_out = coordinate_polish_fast(a, t_c, verbose=False, rng_seed=int(time.monotonic()*1000) % 10000)
            if m_out < best_m:
                best_m = m_out
                best_a = a_out.copy()
                a = a_out.copy()
                if verbose:
                    print(f"      Polish Coord: J={2.0*n*best_m:.6f}")
    
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search: multi-phase optimization with LP-FW, subgradient, freq shaving, and coord descent."""
    t_global = time.monotonic()

    solver = get_solver()

    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global

    parent = None

        p = np.asarray(height_sequence_1, dtype=np.float64)

            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:

        t7a = min(remaining * 0.3, 120.0)
        if t7a > 15:
            a_out, _ = mirror_descent_away(a_long, t7a, top_k=60, beta=6.0, c0=5.0,
                                            restart_every=4000, verbose=False,
                                            rng_seed=new_n % 5000)
            try_update(a_out, f"md_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 7000:
            t7b = min(remaining * 0.4, 100.0)
            if t7b > 15:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                a_out, _ = lp_frank_wolfe_away(a_in, t7b, solver, verbose=False,
                                                top_K_init=100)
                try_update(a_out, f"lpfw_n{new_n}")
    
    # === Phase 8: Final intensive polish (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Polish (n={n_best}, remaining={remaining:.0f}s) ===")
    
    if remaining > 20:
        # LP-FW with more iterations
        t_8a = min(remaining * 0.4, 150.0)
        if t_8a > 15 and n_best <= 8000:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_away(a_in, t_8a, solver, verbose=False,
                                            top_K_init=200, away_coeff=0.6)
            try_update(a_out, "lpfw_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t_8b = min(remaining * 0.3, 80.0)
            if t_8b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = subgradient_nesterov(a_in, t_8b, top_k=120, rng_seed=1234)
                try_update(a_out, "sg_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t_8c = min(remaining * 0.3, 60.0)
            if t_8c > 5:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_polish_fast(a_in, t_8c, verbose=False, rng_seed=5678)
                try_update(a_out, "coord_final")
    
    # === Final validation ===
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0.0)
        s = best_seq.sum()
        if s > 0.01:
            best_seq = best_seq / s
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        best_seq = make_edge_concentrated(n_fb, power=3.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
