def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_mirror_grad(a, ac, top_k, beta):
    """
    Compute gradient for mirror descent step.
    Uses soft-max weighting over top autocorrelation lags.
    Returns gradient vector of shape (n,).
    """
    n = len(a)
    k_alarm = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_alarm]
    top_vals = ac[top_idx]
    
    # Exponential weighting (soft-max)
    max_v = top_vals.max()
    expo = beta * (top_vals - max_v)
    expo = np.clip(expo, -50, 50)  # numerical stability
    weights = np.exp(expo)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_alarm) / k_alarm
    else:
        weights = weights / w_sum
    
    # Compute gradient: d(ac[k])/da[i] = 2*a[k-i] if 0 <= k-i < n
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n, k_int + 1)
        if i_lo >= i_hi:
            continue
        idx = np.arange(i_lo, i_hi)
        j_idx = k_int - idx
        valid = (j_idx >= 0) & (j_idx < n)
        grad[idx[valid]] += 2.0 * w * a[j_idx[valid]]
    
    return grad

def batch_coord_descent(a0, time_limit, rng_seed=77, verbose=True):

    Batch coordinate descent: for each top autocorrelation lag,

    best_a = a.copy()
    best_m = fast_autocorr(a).max()

        else:
            stall += 1
            if stall > 20000:
                break
        
        # Get top lags
        n_top = min(30, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k_pos in top_lags[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            
            k_int = int(k_pos)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo >= i_hi:
                continue
            
            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            
            # Find top contributing pairs
            n_pairs = min(10, len(idx))
            pair_order = np.argsort(contrib)[::-1][:n_pairs]
            
            for pi in pair_order:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[pi])
                src_val = a[src]
                if src_val < 1e-14:
                    continue
                
                # Compute impact score for each destination
                scores = np.zeros(n)
                for k2 in top_lags[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n, k2i + 1)
                    if s2 < e2:
                        ii = np.arange(s2, e2)
                        jj = k2i - ii
                        scores[ii] += a[jj] * ac[k2i]
                        scores[jj] += a[ii] * ac[k2i]
                
                scores[src] = np.inf
                dst = int(np.argmin(scores))
                
                if dst == src or a[dst] > 0.8 * src_val:
                    continue
                
                # Try transfer fractions
                for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            if stall % 500 == 0 and stall > 0:
                # Random perturbations
                n_moves = int(rng.integers(5, 20))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = float(rng.uniform(0.005, 0.03))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Coord descent done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def spectral_flatten(a0, time_limit, lr_init=0.3, verbose=False):
    """
    Flatten the power spectrum to reduce autocorrelation peaks.
    Works by iteratively reducing spectral peaks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    
    fft_size = 1 << (2 * n - 2).bit_length()
    lr = lr_init
    stall = 0
    beta_spec = 3.0
    
    for t in range(1, 500000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A)**2
        P_max = P.max()
        P_mean = P.mean()
        
        if P_max < 1e-30:
            break
        
        # Gradient: push energy from spectral peaks to valleys
        # In frequency domain: reduce A at peaks, increase at valleys
        P_dev = P - P_mean
        expo = beta_spec * P_dev / max(P_max, 1e-30)
        expo = np.clip(expo, -20, 20)
        w_spec = np.exp(expo)
        
        # Modify spectrum
        A_new = A * np.exp(-0.5 * lr / (1 + np.sqrt(t)) * w_spec / max(w_spec.mean(), 1e-10))
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = norm_sum1(a_new)
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 3000:
                a = norm_sum1(best_a + rng.normal(0, 0.01, n) * best_a)
                stall = 0
                lr *= 0.8
                beta_spec = max(beta_spec - 0.5, 1.0)
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Spectral done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def parametric_init(n, time_limit=10.0):
    """Quick parametric search for good initializations."""
    t0 = time.monotonic()
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    
    candidates = []
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))
    
    # Gaussians
    for c in [0.3, 0.5, 0.7]:
        for sig in [0.1, 0.2, 0.3, 0.5]:
            candidates.append(norm_sum1(np.exp(-((x - c)**2) / (2 * sig**2))))
    
    # Bimodal
    for sig in [0.1, 0.2, 0.3]:
        candidates.append(norm_sum1(
            np.exp(-((x - 0.25)**2) / (2*sig**2)) + np.exp(-((x - 0.75)**2) / (2*sig**2))))
    
    # Sine
    for f in [1, 2, 3]:
        candidates.append(norm_sum1(np.sin(np.pi * f * x)**2))
    
    # Exponential
    for r in [1, 2, 5]:
        candidates.append(norm_sum1(np.exp(-r * x)))
        candidates.append(norm_sum1(np.exp(r * x)))
    
    # Evaluate all
    best_cand = None
    best_val = float('inf')
    for c in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        v = score_from_ac(c)
        if v < best_val:
            best_val = v
            best_cand = c.copy()
    
    return best_cand if best_cand is not None else np.ones(n) / n
