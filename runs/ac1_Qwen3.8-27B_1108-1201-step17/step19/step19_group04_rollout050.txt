def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def self_conv(a):
    n = a.size
    if n == 0:
        return np.zeros(1)
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]


def conv_ab(a, b):
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]


def objective(a):
    n = a.size
    if n == 0:
        return float('inf')
    q = self_conv(a)
    M = float(np.max(q))
    S = float(np.sum(a))
    if S < 1e-15:
        return float('inf')
    return 2.0 * n * M / (S * S)


def normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / a.size
    return a / s


def downsample(a, new_n):
    old_n = a.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return a.copy()
    if old_n < new_n:
        return np.interp(np.linspace(0, 1, new_n), np.linspace(0, 1, old_n), a)
    step = old_n / new_n
    idx = (np.arange(new_n) * step + step * 0.5).astype(int)
    idx = np.clip(idx, 0, old_n - 1)
    return normalize(a[idx])


def upsample(a, new_n):
    old_n = a.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return a.copy()
    x = np.linspace(0, 1, new_n)
    return normalize(np.interp(x, np.linspace(0, 1, old_n), a))


def weights_from_q(q, focus):
    q = np.asarray(q, dtype=np.float64)
    m = float(q.max())
    if m <= 1e-15:
        return np.ones_like(q, dtype=np.float64) / q.size
    scale = 0.5 * m
    if scale < 1e-15:

    gamma = focus / scale
    vals = np.clip(gamma * (q - m), -50.0, 0.0)
    e = np.exp(vals)
    s = float(e.sum())
    if s < 1e-300:
        return np.ones_like(q, dtype=np.float64) / q.size
    return e / s


def corr_shift(w, a):

    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fw * fa.conj(), size)[:n]

def quad_max(alpha, q0, c, qd):
    a1 = 1.0 - alpha
    val = (a1 * a1) * q0 + (2.0 * alpha * a1) * c + (alpha * alpha) * qd
    return float(np.max(val))

def smooth_gradient_step(a, focus, trust):
    q = self_conv(a)
    w = weights_from_q(q, focus)
    c = corr_shift(w, a)
    cmax = float(np.max(c))
    cmin = float(np.min(c))

def parametric_gaussian(params, n):
    """Asymmetric Gaussian: h * exp(-(x-mu)^2 / (2*sig^2)) with optional asymmetry"""
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    h = max(params[0], 0.01)
    mu = np.clip(params[1], 0.0, 1.0)
    sig = max(abs(params[2]), 0.02)
    asym = np.clip(params[3], -1.0, 1.0) if len(params) > 3 else 0.0
    sig_l = sig * (1.0 - 0.5 * asym)
    sig_r = sig * (1.0 + 0.5 * asym)
    sig_eff = np.where(x < mu, sig_l, sig_r)
    result = h * np.exp(-0.5 * ((x - mu) / sig_eff) ** 2)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    return result


def parametric_beta(params, n):
    """Beta-distribution-like: x^a * (1-x)^b"""
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    x = np.clip(x, 1e-8, 1 - 1e-8)
    a = max(params[0], 0.1)
    b = max(params[1], 0.1)
    result = x ** a * (1.0 - x) ** b
    s = result.sum()
    if s > 1e-15:
        result = result / s
    return result


def parametric_two_gauss(params, n):
    """Sum of two Gaussians"""
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    result = np.zeros(n, dtype=np.float64)
    for k in range(2):
        h = max(params[3 * k], 0.01)
        mu = np.clip(params[3 * k + 1], -0.5, 1.5)
        sig = max(abs(params[3 * k + 2]), 0.02)
        result += h * np.exp(-0.5 * ((x - mu) / sig) ** 2)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    return result


def parametric_poly(params, n):
    """Polynomial: sum_k params[k] * x^k, clipped to non-negative"""
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    result = np.zeros(n, dtype=np.float64)
    d = len(params) - 1
    for k in range(d + 1):
        result += params[k] * x ** k
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    return result

                               options={'maxiter': 200, 'xatol': 1e-6, 'fatol': 1e-7})
                if res.success and res.fun < best_score:
                    best_score = res.fun
                    best_seq = parametric_gaussian(np.array([max(res.x[0],0.01), np.clip(res.x[1],-0.1,1.1), max(abs(res.x[2]),0.01), np.clip(res.x[3],-1,1)]), n)
                    print("    Param: refined gauss score=%.9f" % best_score)

                p0_b = np.array([1.5, 2.0])
                p0_b += rng.normal(0, 0.3, size=2)

                res = minimize(obj_func_beta, p0_b, method='Nelder-Mead',
                               options={'maxiter': 200, 'xatol': 1e-6, 'fatol': 1e-7})
                if res.success and res.fun < best_score:
                    best_score = res.fun
                    best_seq = parametric_beta(np.array([max(res.x[0],0.1), max(res.x[1],0.1)]), n)
                    print("    Param: refined beta score=%.9f" % best_score)

    if best_seq is None:
        x = np.linspace(0, 1, n)
        best_seq = normalize((x * (1 - x)) ** 0.5)
        best_score = objective(best_seq)

    return best_seq, best_score


def multi_scale_refine(a0, time_limit):
    """Coarse-to-fine optimization."""
    t0 = time.monotonic()
    a = normalize(a0)
    n_full = a.size

    scales = [100, 200, 400, 800, 0]  # 0 means full
    for scale in scales:
        if time.monotonic() - t0 > time_limit * 0.8:
            break
        if scale == 0:
            n_cur = n_full
        else:
            n_cur = min(scale, n_full)
        if n_cur < 10:
            continue
        a_cur = downsample(a, n_cur)
        # Do a few smooth gradient steps at this scale
        focus = 4.0
        trust = 0.5
        for it in range(30):
            if time.monotonic() - t0 > time_limit:
                break
            q = self_conv(a_cur)
            M = float(np.max(q))
            score = 2.0 * n_cur * M
            w = weights_from_q(q, focus)
            c = corr_shift(w, a_cur)
            cmax = float(np.max(c))
            cmin = float(np.min(c))
            sc = cmax - cmin
            if sc <= 1e-15:
                break
            move = trust * (cmax - c) / sc
            move = np.clip(move, -50.0, 50.0)
            d = a_cur * np.exp(move)
            d = normalize(d)
            new_a, new_M = line_search_direction(a_cur, d, q)
            new_score = 2.0 * n_cur * new_M
            if new_score < score - 1e-13:
                a_cur = new_a
                trust = min(trust * 1.1, 3.0)
            else:
                trust = max(trust * 0.8, 0.1)
                focus = min(focus * 1.05, 30.0)
        # Upsample back
        if n_cur < n_full:
            a = upsample(a_cur, n_full)

    return a


def asymmetric_perturb(a, rng):
    """Apply a small asymmetric perturbation to escape local minima."""
    n = a.size
    mode = rng.integers(0, 4)
    if mode == 0:
        # Exponential tilt
        lam = rng.uniform(-0.3, 0.3)
        x = np.arange(n, dtype=np.float64) / (n - 1)
        a_new = a * np.exp(lam * x)
    elif mode == 1:
        # Shift peak
        shift = int(rng.integers(1, max(2, n // 40)))
        if rng.random() < 0.5:
            a_new = np.zeros(n)
            a_new[shift:] = a[:-shift]
            a_new[:shift] = a[:shift] * 0.5
        else:
            a_new = np.zeros(n)
            a_new[:-shift] = a[shift:]
            a_new[-shift:] = a[-shift:] * 0.5
    elif mode == 2:
        # Add small secondary bump
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.03, 0.1)
        height = rng.uniform(0.01, 0.05) * a.max()
        x = np.arange(n, dtype=np.float64) / (n - 1)
        bump = height * np.exp(-0.5 * ((x - center) / width) ** 2)
        a_new = a + bump
    else:
        # Multiplicative noise
        noise = rng.normal(1.0, 0.02, size=n)
        a_new = a * np.maximum(noise, 0.5)
    a_new = normalize(np.maximum(a_new, 0.0))
    return a_new


def lp_fw_search(a0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive parameters."""
    t0 = time.monotonic()
    y = normalize(a0)
    n = y.size
    if n <= 10:
        return y

    best = y.copy()
    best_score = objective(y)

    if n > 7000:
        K = 200
        maxK = 500
    elif n > 4000:
        K = 280
        maxK = 800
    else:
        K = 350
        maxK = 1200

    use_lp = HAVE_LP
    lp_fails = 0
    no_improve = 0
    iter_no = 0
    last_print = t0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining <= 1.0:
            break

        qy = self_conv(y)
        M_y = float(np.max(qy))
        score_y = 2.0 * n * M_y

        if score_y < best_score - 1e-12:
            best = y.copy()
            best_score = score_y
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50 and no_improve % 25 == 0:
            # Asymmetric perturbation
            rng = np.random.default_rng(iter_no * 7919)
            y_pert = asymmetric_perturb(best, rng)
            sc_pert = objective(y_pert)
            if sc_pert < best_score + 0.01:
                y = y_pert
                no_improve = 0

        if no_improve > 250:
            break

        order = np.argsort(qy)[::-1]
        Kcur = int(min(K, len(qy), 1500))
        act = order[:Kcur]
        q_active = qy[act]

        g_dir = None
        if use_lp and Kcur >= 20:
            tstep = time.monotonic()
            g_dir, T = solve_lp_offset(y, act, q_active)
            spend = time.monotonic() - tstep
            if g_dir is None:
                lp_fails += 1
                if lp_fails >= 5:
                    use_lp = False
            elif spend > 90.0:
                use_lp = False
                K = max(100, K - 50)

        if g_dir is not None:
            new, Mnew = line_search_direction(y, g_dir, qy)
            new_score = 2.0 * n * Mnew
            if new_score < score_y - 1e-13:
                y = new
                no_improve = 0
            else:
                no_improve += 1
                if no_improve in (2, 5, 12, 30):
                    # Try smooth gradient as fallback
                    for focus in (3.0, 8.0, 20.0):
                        new2, sc2 = smooth_gradient_step(y, focus, 0.4)
                        if sc2 < score_y - 1e-13:
                            y = new2
                            no_improve = 0
                            break
                K = min(K + 30, maxK)
        else:
            improved = False
            for focus in (3.0, 6.0, 12.0, 25.0):
                new, sc = smooth_gradient_step(y, focus, 0.4)
                if sc < score_y - 1e-13:
                    y = new
                    no_improve = 0
                    improved = True
                    break
            if not improved:
                no_improve += 1
                K = min(K + 20, maxK)
                if no_improve > 180:
                    break

        iter_no += 1
        if verbose and (iter_no % 30 == 0 or (time.monotonic() - last_print) > 30.0):
            print("    LP-FW: it=%d t=%.1fs best=%.9f K=%d noimp=%d" %
                  (iter_no, time.monotonic() - t0, best_score, K, no_improve))
            last_print = time.monotonic()

    return best


def smooth_polish(a0, time_limit, rng_seed=7, verbose=False):
    """Multiplicative gradient polish with adaptive parameters."""
    t0 = time.monotonic()
    a = normalize(a0)
    n = a.size
    if n <= 5:
        return a

    best_score = objective(a)

    while time.monotonic() - t0 < time_limit:

        M = float(np.max(q))
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            fails = 0
        else:
            fails += 1

        if fails > 500:
            break

        if fails > 60 and fails % 30 == 0:
            focus = min(focus * 1.15, 80.0)
            trust = max(trust * 0.85, 0.05)

            a = best.copy()
            scale_noise = 0.015 / (1.0 + 0.1 * (fails // 150))
            a = a * (1.0 + scale_noise * rng.normal(size=n))
            a = normalize(np.maximum(a, 0.0))

        d = a * np.exp(move)
        d = normalize(d)
