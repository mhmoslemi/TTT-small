import numpy as np

def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0, v[0])]) if v[0] > 0 else np.array([1.0])

        return np.ones(n) / n
    r = rho[-1]

    return np.maximum(v - theta, 0.0)

Let me also think about a key improvement: **the spectral flattening step can be made much more effective**.

In standard GS:
- Time domain: p = max(0, IFFT(P)) / sum
- Frequency domain: |P| = |P|^alpha * mean(|P|)^(1-alpha)

The alpha parameter controls how aggressive the flattening is. Starting with alpha close to 1 (very aggressive) and annealing to alpha = 0.5 (mild) might work well.

But there's a subtlety: we want to flatten |P|^2 (the power spectrum), not |P|. So:
- |P|_new = (|P|^2)^{(1-alpha)/2} * mean(|P|^2)^{alpha/2} = |P|^{1-alpha} * mean(|P|^2)^{alpha/2}

Or equivalently, work with the power spectrum directly:
- S = |P|^2
- S_new = S^{1-alpha} * mean(S)^{alpha}
- |P|_new = sqrt(S_new)

This makes more sense. Let me use this formulation.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Adaptive full-subgradient mirror descent
- Spectral flattening (power spectrum domain)
- Coordinate descent polish
- Multi-length exploration
"""

import numpy as np
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_autocorr_m(a):
    """Return max of autocorrelation."""
    return fast_autocorr(a).max()


def proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(p, c, rel_thresh=0.995, max_active=100):
    """Compute subgradient of max(c) using active lags."""
    n = len(p)
    max_c = c.max()
    if max_c < 1e-30:
        return np.zeros(n)
    
    active = np.where(c >= rel_thresh * max_c)[0]
    if len(active) > max_active:
        # Keep only top max_active
        active_vals = c[active]
        top_idx = np.argsort(active_vals)[::-1][:max_active]
        active = active[top_idx]
    
    grad = np.zeros(n)
    for k in active:
        lag = int(k) - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[:n - lag] += p[lag:]
                grad[lag:] += p[:n - lag]
        else:
            s = -lag
            if s < n:
                grad[s:] += p[:n - s]
                grad[:n - s] += p[s:]
    
    grad /= len(active)
    return grad


def mirror_descent_subgrad(p0, time_limit, rel_thresh=0.995, max_active=50,
                           lr=2.0, verbose=False):
    """Mirror descent with full subgradient."""
    n = len(p0)
    p = np.maximum(np.asarray(p0, dtype=np.float64), 1e-30)
    p = p / p.sum()
    best_p = p.copy()
    best_m = fast_autocorr(p).max()
    t0 = time.monotonic()
    
    log_p = np.log(p)
    stall = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_autocorr(p)
        g = compute_subgrad(p, c, rel_thresh, max_active)
        
        eta = lr / (np.sqrt(t) + 1.0)
        log_p = log_p - eta * g
        # Renormalize (this is the mirror descent step on the simplex)
        max_log = log_p.max()
        p_new = np.exp(log_p - max_log)
        p_new = p_new / p_new.sum()
        
        M_new = fast_autocorr(p_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_p = p_new.copy()
            p = p_new
            log_p = np.log(np.maximum(p, 1e-30))
            stall = 0
        else:
            stall += 1
            # Don't update p, keep old log_p
        
        if stall > 2000:
            # Restart with perturbation
            pert = np.random.normal(0, 0.05, n) * p
            p_r = p + pert
            p_r = np.maximum(p_r, 1e-15)
            p_r = p_r / p_r.sum()
            p = p_r
            log_p = np.log(p)
            stall = 0
        
        if verbose and t % 5000 == 0:
            print(f"    MD t={t} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")
    
    return best_p, best_m


def spectral_flatten(p0, time_limit, verbose=False):
    """Flatten power spectrum to reduce autocorrelation peaks."""
    n = len(p0)
    p = np.maximum(np.asarray(p0, dtype=np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s
    best_p = p.copy()
    best_m = fast_autocorr(p).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        P = np.fft.rfft(p, fft_size)
        S = np.abs(P) ** 2
        S_max = S.max()
        S_mean = S.mean()
        
        if S_max < 1e-30:
            break
        
        # Flatten: S_new = S^(1-alpha) * S_mean^alpha
        progress = min(1.0, it / 1000.0)
        alpha = 0.3 + 0.5 * progress  # 0.3 to 0.8
        
        S_new = S ** (1 - alpha) * S_mean ** alpha
        P_new_mag = np.sqrt(S_new)
        P_new = P_new_mag * np.exp(1j * np.angle(P))
        
        p_new = np.fft.irfft(P_new, fft_size)[:n]
        p_new = np.maximum(p_new, 0.0)
        s_new = p_new.sum()
        if s_new < 1e-15:
            continue
        p_new = p_new / s_new
        
        M_new = fast_autocorr(p_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_p = p_new.copy()
            p = p_new
        else:
            # Blend
            blend = 0.1
            p = (1 - blend) * p + blend * p_new
            p = np.maximum(p, 0.0)
            p = p / p.sum()
            M_curr = fast_autocorr(p).max()
            if M_curr < best_m:
                best_m = M_curr
                best_p = p.copy()
        
        if verbose and it % 500 == 0:
            print(f"    Spectral it={it} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")
    
    return best_p, best_m


def coordinate_descent(p0, time_limit, verbose=False, rng_seed=42):
    """Coordinate descent: move mass from high-contribution to low-contribution positions."""
    n = len(p0)
    p = np.maximum(np.asarray(p0, dtype=np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s
    best_p = p.copy()
    best_m = fast_autocorr(p).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        c = fast_autocorr(p)
        k_max = int(np.argmax(c))
        lag_max = k_max - (n - 1)
        
        # Find the pair (i, i+lag) that contributes most to c[k_max]
        if lag_max >= 0:
            valid_i = np.arange(0, n - lag_max)
        else:
            valid_i = np.arange(-lag_max, n)
        
        if len(valid_i) == 0:
            stall += 1
            continue
        
        contribs = p[valid_i] * p[valid_i + lag_max] if lag_max >= 0 else p[valid_i] * p[valid_i - lag_max]
        idx_max = valid_i[np.argmax(contribs)]
        
        if p[idx_max] < 1e-12:
            stall += 1
            continue
        
        # Find destination: position that minimizes impact on top peaks
        top_ks = np.argpartition(c, -20)[-20:]
        costs = np.zeros(n)
        for k2 in top_ks:
            lag2 = int(k2) - (n - 1)
            if lag2 >= 0:
                costs[:n - lag2] += p[lag2:]
                costs[lag2:] += p[:n - lag2]
            else:
                s2 = -lag2
                costs[s2:] += p[:n - s2]
                costs[:n - s2] += p[s2:]
        
        # Pick lowest cost destination (not the source)
        costs[idx_max] = np.inf
        dst = int(np.argmin(costs))
        
        moved = False
        for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
            amt = p[idx_max] * frac
            p_trial = p.copy()
            p_trial[idx_max] -= amt
            p_trial[dst] += amt
            p_trial = np.maximum(p_trial, 0.0)
            p_trial = p_trial / p_trial.sum()
            M_t = fast_autocorr(p_trial).max()
            if M_t < best_m - 1e-15:
                best_m = M_t
                best_p = p_trial.copy()
                p = p_trial
                moved = True
                stall = 0
                break
        
        if not moved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and p[i] > 1e-10:
                amt = p[i] * rng.uniform(0.005, 0.03)
                p[i] -= amt
                p[j] += amt
                p = np.maximum(p, 0.0)
                p = p / p.sum()
                M_new = fast_autocorr(p).max()
                if M_new < best_m:
                    best_m = M_new
                    best_p = p.copy()
                    stall = 0
    
    return best_p, best_m

            stall += 1

        if stall > 3000:
            strength = min(0.08, 0.02 + 0.01 * (stall // 3000))
            pert = rng.normal(0, strength, n) * p
            p_r = p + pert
            p_r = np.maximum(p_r, 1e-15)
            p_r = p_r / p_r.sum()
            p = p_r
            log_p = np.log(p)
            stall = 0

        if verbose and t % 5000 == 0:
            print(f"    MD t={t} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}, stall={stall}")

    return best_p, best_m


def spectral_flatten(p0, time_limit, verbose=False):
    n = len(p0)
    p = np.maximum(np.asarray(p0, dtype=np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s
    best_p = p.copy()
    best_m = fast_autocorr(p).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        P = np.fft.rfft(p, fft_size)
        S = np.abs(P) ** 2
        S_max = S.max()
        S_mean = S.mean()

        if S_max < 1e-30:
            break

        progress = min(1.0, it / 2000.0)
        alpha = 0.2 + 0.6 * progress

        S_new = S ** (1.0 - alpha) * S_mean ** alpha
        P_new_mag = np.sqrt(S_new)
        P_new = P_new_mag * np.exp(1j * np.angle(P))

        p_new = np.fft.irfft(P_new, fft_size)[:n]
        p_new = np.maximum(p_new, 0.0)
        s_new = p_new.sum()
        if s_new < 1e-15:
            continue
        p_new = p_new / s_new

        M_new = fast_autocorr(p_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_p = p_new.copy()
            p = p_new
        else:
            blend = 0.05 + 0.1 * progress
            p = (1.0 - blend) * p + blend * p_new
            p = np.maximum(p, 0.0)
            p = p / p.sum()
            M_curr = fast_autocorr(p).max()
            if M_curr < best_m:
                best_m = M_curr
                best_p = p.copy()

        if verbose and it % 1000 == 0:
            print(f"    Spectral it={it} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")

    return best_p, best_m


def projected_subgrad(p0, time_limit, verbose=False, rng_seed=0):
    """Projected subgradient method on the simplex."""
    n = len(p0)
    p = np.maximum(np.asarray(p0, dtype=np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s
    best_p = p.copy()
    best_m = fast_autocorr(p).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autocorr(p)
        g = compute_subgrad(p, c, rel_thresh=0.995, max_active=40)

        alpha = 2.0 / (np.sqrt(t) + 1.0)
        p_new = p - alpha * g
        p_new = proj_simplex(p_new)

        M_new = fast_autocorr(p_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_p = p_new.copy()
            p = p_new
            stall = 0
        else:
            stall += 1

        if stall > 2000:
            pert = rng.normal(0, 0.03, n) * p
            p_r = p + pert
            p_r = np.maximum(p_r, 0.0)
            p_r = p_r / p_r.sum()
            p = p_r
            stall = 0

        if verbose and t % 5000 == 0:
            print(f"    PS t={t} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")

    return best_p, best_m


def coordinate_descent(p0, time_limit, verbose=False, rng_seed=42):
    n = len(p0)
    p = np.maximum(np.asarray(p0, dtype=np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s
    best_p = p.copy()
    best_m = fast_autocorr(p).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        c = fast_autocorr(p)
        k_max = int(np.argmax(c))
        lag_max = k_max - (n - 1)

        if lag_max >= 0:
            valid_i = np.arange(0, n - lag_max)
        else:
            valid_i = np.arange(-lag_max, n)

        if len(valid_i) == 0:
            stall += 1
            continue

        if lag_max >= 0:
            contribs = p[valid_i] * p[valid_i + lag_max]
        else:
            contribs = p[valid_i] * p[valid_i - lag_max]
        src = int(valid_i[np.argmax(contribs)])

        if p[src] < 1e-12:
            stall += 1
            continue

        top_ks = np.argpartition(c, -15)[-15:]
        costs = np.zeros(n)
        for k2 in top_ks:
            lag2 = int(k2) - (n - 1)
            if lag2 >= 0:
                if lag2 < n:
                    costs[:n - lag2] += p[lag2:]
                    costs[lag2:] += p[:n - lag2]
            else:
                s2 = -lag2
                if s2 < n:
                    costs[s2:] += p[:n - s2]
                    costs[:n - s2] += p[s2:]

        costs[src] = np.inf
        dst = int(np.argmin(costs))

        moved = False
        for frac in [0.02, 0.05, 0.1, 0.2, 0.35]:
            amt = p[src] * frac
            p_trial = p.copy()
            p_trial[src] -= amt
            p_trial[dst] += amt
            p_trial = np.maximum(p_trial, 0.0)
            p_trial = p_trial / p_trial.sum()
            M_t = fast_autocorr(p_trial).max()
            if M_t < best_m - 1e-16:
                best_m = M_t
                best_p = p_trial.copy()
                p = p_trial
                moved = True
                stall = 0
                break

        if not moved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and p[i] > 1e-10:
                amt = p[i] * rng.uniform(0.005, 0.02)
                p[i] -= amt
                p[j] += amt
                p = np.maximum(p, 0.0)
                p = p / p.sum()
                M_new = fast_autocorr(p).max()
                if M_new < best_m:
                    best_m = M_new
                    best_p = p.copy()
                    stall = 0

    return best_p, best_m


def interpolate_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss


def safe_eval(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def simulated_annealing(p0, time_limit, verbose=False, rng_seed=0):
    n = len(p0)
    p = np.maximum(np.asarray(p0, dtype=np.float64), 1e-15)
    p = p / p.sum()
    best_p = p.copy()
    best_m = fast_autocorr(p).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    T = 0.01
    stall = 0

    while time.monotonic() - t0 < time_limit:
        T *= 0.9995
        if T < 1e-6:
            T = 1e-6

        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        amt = p[i] * rng.uniform(0.01, 0.1)
        p_trial = p.copy()
        p_trial[i] -= amt
        p_trial[j] += amt
        p_trial = np.maximum(p_trial, 0.0)
        p_trial = p_trial / p_trial.sum()
        M_new = fast_autocorr(p_trial).max()
        delta = M_new - best_m

        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            p = p_trial
            if M_new < best_m - 1e-16:
                best_m = M_new
                best_p = p_trial.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > 5000:
            p = best_p.copy()
            stall = 0

        if verbose and (int(time.monotonic() - t0)) % 50 == 0:
            print(f"    SA ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")

    return best_p, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1755.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = safe_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = safe_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Create diverse initializations
    candidates = []
    if parent is not None:
        candidates.append((parent.copy(), "parent"))

    n_base = 6000
    for n_try in [5000, 6000, 7000, 8000]:
        a = np.ones(n_try) / n_try
        candidates.append((a, f"uniform_n{n_try}"))

    # Tapered profiles
    for n_try in [5000, 7000]:
        x = np.arange(n_try, dtype=np.float64) / n_try
        a = np.cos(np.pi * x) ** 2
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            candidates.append((a, f"cos2_n{n_try}"))

    # Evaluate and sort
    scored = []
    for a, label in candidates:
        sc = safe_eval(a)
        if np.isfinite(sc):
            scored.append((sc, a, label))
            print(f"  Init [{label}]: {sc:.6f}")
    scored.sort(key=lambda x: x[0])

    # Phase 1: Spectral flattening on top candidates
    print(f"\n=== Phase 1: Spectral Flattening ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_cands = min(3, len(scored))
    t_per = max(30.0, min(300.0, remaining * 0.25 / max(1, n_cands)))

    for i in range(n_cands):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget:
            break
        t_avail = min(t_per, time_budget - elapsed)
        if t_avail < 15:
            break
        sc0, a0, label = scored[i]
        print(f"  Spectral [{label}] (t={t_avail:.0f}s, init={sc0:.6f})...")
        a_out, _ = spectral_flatten(a0, t_avail, verbose=False)
        try_update(a_out, f"spec_{label}")

    # Phase 2: Adaptive mirror descent
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Adaptive Mirror Descent ===")
    if remaining > 60 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        t2 = min(remaining * 0.30, 500.0)
        print(f"  MD (t={t2:.0f}s)...")
        a_out, _ = mirror_descent_adaptive(a_in, t2, rel_thresh=0.99, max_active=60,
                                            lr=3.0, verbose=True, rng_seed=42)
        try_update(a_out, "md_main")

    # Phase 3: Projected subgradient
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Projected Subgradient ===")
    if remaining > 40 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        t3 = min(remaining * 0.15, 250.0)
        print(f"  PS (t={t3:.0f}s)...")
        a_out, _ = projected_subgrad(a_in, t3, verbose=False, rng_seed=7)
        try_update(a_out, "ps_main")

    # Phase 4: Second round of spectral + MD
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Spectral + MD Round 2 ===")
    if remaining > 60 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        t4a = min(remaining * 0.15, 200.0)
        t4b = min(remaining * 0.20, 250.0)
        if t4a > 20:
            a_out, _ = spectral_flatten(a_in, t4a, verbose=False)
            try_update(a_out, "spec_r2")
            a_in = a_out / max(a_out.sum(), 1e-15)
        remaining = time_budget - (time.monotonic() - t_global)
        if t4b > 20 and remaining > 30:
            a_out, _ = mirror_descent_adaptive(a_in, t4b, rel_thresh=0.995, max_active=40,
                                                lr=2.0, verbose=False, rng_seed=99)
            try_update(a_out, "md_r2")

    # Phase 5: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Coordinate Descent ===")
    if remaining > 30 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        t5 = min(remaining * 0.10, 150.0)
        if t5 > 15:
            a_out, _ = coordinate_descent(a_in, t5, verbose=False, rng_seed=42)
            try_update(a_out, "cd_1")
            a_out, _ = coordinate_descent(a_in, t5, verbose=False, rng_seed=777)
            try_update(a_out, "cd_2")

    # Phase 6: Simulated annealing
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Simulated Annealing ===")
    if remaining > 30 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        t6 = min(remaining * 0.08, 100.0)
        if t6 > 10:
            a_out, _ = simulated_annealing(a_in, t6, verbose=False, rng_seed=123)
            try_update(a_out, "sa_1")

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Multi-length Exploration ===")
    if best_seq is not None and remaining > 100:
        n_best = len(best_seq)
        for delta_n in [1000, -1000, 2000, -2000]:
            new_n = n_best + delta_n
            if new_n < 3000 or new_n > 12000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            a_interp = interpolate_to_length(best_seq, new_n)
            sc_interp = safe_eval(a_interp)
            print(f"  Interp n={new_n}: {sc_interp:.6f}")
            if np.isfinite(sc_interp) and sc_interp < best_score:
                best_score = sc_interp
                best_seq = a_interp.copy()
                print(f"  *** NEW BEST [interp_n{new_n}]: {sc_interp:.6f}")

            t7 = min(remaining * 0.15, 120.0)
            if t7 > 20:
                a_out, _ = mirror_descent_adaptive(a_interp, t7, rel_thresh=0.99,
                                                    max_active=50, lr=3.0,
                                                    verbose=False, rng_seed=200 + new_n % 50)
                try_update(a_out, f"md_n{new_n}")

    # Phase 8: Final combined polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final Polish ===")
    if remaining > 30 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        t8a = min(remaining * 0.25, 100.0)
        t8b = min(remaining * 0.30, 120.0)
        t8c = min(remaining * 0.15, 60.0)
        if t8a > 10:
            a_out, _ = spectral_flatten(a_in, t8a, verbose=False)
            try_update(a_out, "spec_final")
            a_in = a_out / max(a_out.sum(), 1e-15)
        remaining = time_budget - (time.monotonic() - t_global)
        if t8b > 10 and remaining > 20:
            a_out, _ = mirror_descent_adaptive(a_in, t8b, rel_thresh=0.995, max_active=30,
                                                lr=2.5, verbose=False, rng_seed=555)
            try_update(a_out, "md_final")
            a_in = a_out / max(a_out.sum(), 1e-15)
        remaining = time_budget - (time.monotonic() - t_global)
        if t8c > 10 and remaining > 15:
            a_out, _ = coordinate_descent(a_in, t8c, verbose=False, rng_seed=9999)
            try_update(a_out, "cd_final")

    # Final validation
    if best_seq is not None:
        fs = safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        best_seq = np.ones(6000) / 6000
        best_score = safe_eval(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
