        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False,
                       **_get_solver_kwargs(solver))
        except Exception:
            failed_solves += 1
            if failed_solves > 8:
                constraint_list = sorted(constraint_set)[:min(initial_K, len(constraint_set))]
                constraint_set = set(constraint_list)
                failed_solves = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            n_add = min(50, len(violations))
            top_v = violations[np.argsort(viol_amt)[::-1][:n_add]]
            for k in top_v:
                constraint_set.add(int(k))

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 15 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    return best_a, best_m


def _get_solver_kwargs(solver):
    if solver in (cp.CLARABEL,):
        return {'tol_gap_abs': 1e-9, 'tol_gap_rel': 1e-9,
                'tol_feas': 1e-9, 'tol_infeas_abs': 1e-9, 'tol_infeas_rel': 1e-9}
    elif solver in (cp.PDLP,):
        return {}
    elif solver in (cp.ECOS,):
        return {'ABSTOL': 1e-8, 'RELTOL': 1e-8, 'MAXITAL': 10000}
    elif solver in (cp.SCS,):
        return {'eps': 1e-8, 'max_iters': 10000}
    return {}


def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(30, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_smooth(a, alpha=0.1):
    """Apply mild spectral smoothing to reduce high-frequency components."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    freqs = np.fft.rfftfreq(fft_size)
    cutoff = 0.8
    mask = np.ones(len(A))
    for i in range(1, len(A)):
        if freqs[i] > cutoff:
            mask[i] = alpha
    A_smooth = A * mask
    a_smooth = np.fft.irfft(A_smooth, fft_size)[:n]
    return np.maximum(a_smooth, 0.0)


def multi_start_fista(a0, time_limit, solver, n_starts=3, verbose=True):
    """Run FISTA from multiple perturbed starting points, keep best."""
    n = len(a0)
    best_a = norm_sum1(a0)
    best_m = fast_autocorr(best_a).max()
    t0 = time.monotonic()

    time_per_start = time_limit / n_starts

    for start_i in range(n_starts):
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 30:
            break

        if start_i == 0:
            a_init = norm_sum1(a0)
        else:
            rng = np.random.default_rng(1000 + start_i * 17)
            noise = rng.normal(0, 0.05, n) * a0
            a_init = norm_sum1(a0 + noise)

        t_start = min(remaining * 0.85, time_per_start * 1.2)
        a_out, m_out = fista_optimize(a_init, t_start,
                                      beta_start=5.0 + start_i * 5.0,
                                      beta_end=80.0 + start_i * 20.0,
                                      eta0=0.7 - start_i * 0.05,
                                      top_k_start=400, top_k_end=100,
                                      verbose=False)

        if m_out < best_m:
            best_m = m_out
            best_a = a_out
            if verbose:
                print(f"    Multi-start {start_i}: improved to J={2*n*best_m:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_start = 5000
        best_seq = norm_sum1((np.arange(n_start, dtype=np.float64) + 1.0) ** 0.7)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_start}, score={best_score:.6f}")

    # === Phase 1: FISTA main optimization (exploit phase) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t1 = min(remaining * 0.30, 450.0)
    print(f"\n=== Phase 1: FISTA main (n={n_best}, t={t1:.0f}s) ===")
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_optimize(a_in, t1, beta_start=5.0, beta_end=120.0,
                                  eta0=0.75, top_k_start=400, top_k_end=80,
                                  verbose=True)
        check_and_update(a_out, "fista_main")

    # === Phase 2: LP-FW polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: LP-FW polish (n={n_best}, t={t2:.0f}s) ===")
    if t2 > 30 and n_best <= 10000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, initial_K=500, max_K=4000,
                                  verbose=True)
        check_and_update(a_out, "lp_polish_1")

    # === Phase 3: FISTA after LP (sharper) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: FISTA sharp (n={n_best}, t={t3:.0f}s) ===")
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_optimize(a_in, t3, beta_start=30.0, beta_end=300.0,
                                  eta0=0.4, top_k_start=200, top_k_end=50,
                                  dual_temp=True, verbose=True)
        check_and_update(a_out, "fista_sharp")

    # === Phase 4: LP-FW second round ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: LP-FW round 2 (n={n_best}, t={t4:.0f}s) ===")
    if t4 > 30 and n_best <= 10000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, initial_K=600, max_K=5000,
                                  verbose=True)
        check_and_update(a_out, "lp_polish_2")

    # === Phase 5: Multi-start FISTA for diversity ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 5: Multi-start FISTA (n={n_best}, t={t5:.0f}s) ===")
    if t5 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = multi_start_fista(a_in, t5, solver, n_starts=2, verbose=True)
        check_and_update(a_out, "multi_start")

    # === Phase 6: Spectral smoothing + FISTA ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 6: Spectral smooth + FISTA (t={t6:.0f}s) ===")
    if t6 > 20:
        a_smooth = spectral_smooth(norm_sum1(best_seq), alpha=0.3)
        a_smooth = norm_sum1(a_smooth)
        a_out, _ = fista_optimize(a_smooth, t6, beta_start=10.0, beta_end=100.0,
                                  eta0=0.6, top_k_start=300, top_k_end=80,
                                  verbose=False)
        check_and_update(a_out, "spectral_smooth")

    # === Phase 7: Final LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW (n={n_best}) ===")
    if remaining > 40 and n_best <= 10000:
        t7 = min(remaining * 0.4, 150.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, initial_K=800, max_K=5000,
                                      verbose=True)
            check_and_update(a_out, "lp_final")

    # === Phase 8: Final FISTA polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t8 = min(remaining * 0.5, 60.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = fista_optimize(a_in, t8, beta_start=50.0, beta_end=500.0,
                                      eta0=0.3, top_k_start=100, top_k_end=30,
                                      verbose=False)
            check_and_update(a_out, "fista_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
