def fast_autocorr(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_init(n, n_modes=30, seed=0):
    rng = np.random.default_rng(seed)

        a += coeff * np.sin(2 * np.pi * freq * x + phase)

def pocs_optimize(a0, time_limit, rng_seed=0, n_iters_max=200000):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    fft_size = 1 << (2 * n - 2).bit_length()

    for trial in range(3):
        if time.monotonic() - t0 > time_limit:
            break
        t_trial = (time_limit - (time.monotonic() - t0)) / (3 - trial)

        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        A_spec = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
        A_full = np.zeros(fft_size, dtype=np.complex128)
        A_full[:fft_size // 2 + 1] = A_spec
        for i in range(1, fft_size // 2):
            A_full[fft_size - i] = np.conj(A_spec[i])
        a_init = np.fft.irfft(A_full, fft_size)[:n]
        a_init = np.maximum(a_init, 0.0)
        s = a_init.sum()
        if s < 1e-15:
            a_init = a.copy()
        else:
            a_init = a_init / s

        cur = a_init.copy()
        for it in range(n_iters_max):
            if time.monotonic() - t0 > time_limit:
                break
            A = np.fft.rfft(cur, fft_size)
            mag = np.abs(A)
            mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            alpha_b = 0.15 + 0.5 * min(1.0, it / 50.0)
            new_mag = (1 - alpha_b) * mag + alpha_b * mag_mean
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = cur.copy()
            else:
                a_new = a_new / sa
            M_new = fast_autocorr(a_new).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a_new.copy()
            cur = a_new

    return best_a, best_m


def compute_smooth_grad(a, ac, tau):
    n = len(a)
    ac_max = ac.max()
    shifted = tau * (ac - ac_max)
    exp_ac = np.exp(shifted)
    p = exp_ac / exp_ac.sum()

    p_cum = np.concatenate([[0.0], np.cumsum(p)])
    k_lo = np.arange(n)
    k_hi = np.minimum(k_lo + n, len(p))
    S = p_cum[k_hi] - p_cum[k_lo]
    cross = sp_correlate(p, a, mode='valid')
    g = 2.0 * (a * S + cross)
    g_tangent = g - g.mean()
    return g_tangent


def smooth_grad_optimize(a0, time_limit, rng_seed=100):

def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    K = 400
    K_max = min(5000, 2 * n - 1)
    iter_count = 0
    active_set = None

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-18:
            break

        top_idx = np.argsort(ac)[::-1][:K]
        if active_set is not None:
            combined = np.union1d(top_idx, active_set)
            combined = combined[:K_max]
        else:
            combined = top_idx
        K_sel = len(combined)
        binding_idx = combined

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 200, K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        base_obj = fast_autocorr(a).max()
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        for alpha in [best_alpha, 1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        active_set = binding_idx.copy()
        iter_count += 1

        if iter_count % 5 == 0:
            K = min(K + 100, K_max)

        if iter_count % 10 == 0 and verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

    return best_a, best_m


def batch_coord_descent(a0, time_limit, rng_seed=456):

    a = norm_sum1(a0)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_top]

        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:15]:
            ki = int(k)
            s_max = max(0, ki - n + 1)
            e_max = min(n - 1, ki)
            idx = np.arange(s_max, e_max + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved = False

        for k in top_k:

            s_max = max(0, ki - n + 1)

                continue

            src = idx[np.argmax(contrib)]

def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def amd_quick(a0, time_limit, rng_seed=42, top_k=50, c0=4.0):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** 5.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(top_k) / top_k
        else:
            w = w / w_sum
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = np.maximum(best_a + pert, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            stall = 0

    return best_a, best_m

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5500
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # Phase 1: POCS + quick AMD on diverse inits
    print(f"\n=== Phase 1: Multi-init warmup (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.10, 150.0)

    if t_phase1 > 30:
        a_pocs, _ = pocs_optimize(best_seq, min(t_phase1 * 0.4, 50.0), rng_seed=7)
        try_update(a_pocs, "pocs")

    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1b = min(remaining * 0.06, 90.0)
    if t_phase1b > 15:
        inits = []
        labels = []
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        inits.append(make_u_shape(n0, 1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, 3.0))
        labels.append("u3.0")
        inits.append(make_sin2(n0))
        labels.append("sin2")
        inits.append(make_cos_mod(n0, 0.4))
        labels.append("cosmod")
        inits.append(make_power_law(n0, 0.4))
        labels.append("powlaw")

        t_per = max(8.0, t_phase1b / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 5:
                break
            a_out, _ = amd_quick(init, t_avail, rng_seed=100 + idx, top_k=50, c0=4.0)
            try_update(a_out, f"amd_{lbl}")

    # Phase 2: LP-FW main
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW main (n={n_best}, rem={remaining:.0f}s) ===")
    t2 = min(remaining * 0.40, 600.0)
    if t2 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True)
        try_update(a_out, "lp_main")

    # Phase 3: Smooth gradient
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smooth gradient (n={n_best}, rem={remaining:.0f}s) ===")
    t3 = min(remaining * 0.18, 250.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_grad_optimize(a_in, t3, rng_seed=200)
        try_update(a_out, "smooth1")

    # Phase 4: LP-FW round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW r2 (n={n_best}, rem={remaining:.0f}s) ===")
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True)
        try_update(a_out, "lp_r2")

    # Phase 5: Batch coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord descent (n={n_best}, rem={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = batch_coord_descent(a_in, t5, rng_seed=789)
        try_update(a_out, "coord")

    # Phase 6: Length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration (rem={remaining:.0f}s) ===")

    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 2000, -2000]:
        new_n = n_best + delta
        if 3000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interp_seq(best_seq, new_n)

        t6a = min(remaining * 0.35, 120.0)
        if t6a > 15:
            a_out, _ = amd_quick(a_long, t6a, rng_seed=new_n % 1000, top_k=50, c0=4.0)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 7000:
            t6b = min(remaining * 0.5, 100.0)
            if t6b > 15:
                a_in = norm_sum1(interp_seq(best_seq, new_n))
                a_out, _ = lp_frank_wolfe(a_in, t6b, solver, verbose=False)
                try_update(a_out, f"lp_n{new_n}")

    # Phase 7: Final optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final optimization (n={n_best}, rem={remaining:.0f}s) ===")
    if remaining > 30:
        t7 = min(remaining * 0.30, 120.0)
        if t7 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=False)
            try_update(a_out, "lp_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.30, 80.0)
            if t7b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = smooth_grad_optimize(a_in, t7b, rng_seed=500)
                try_update(a_out, "smooth_final")

    # Phase 8: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 8: Final polish (n={n_best}) ===")
        t8 = min(remaining * 0.7, 80.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = batch_coord_descent(a_in, t8, rng_seed=1111)
            try_update(a_out, "coord_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5500
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
