```python
    """Compute gradient of smooth_max(conv(a,a)) with respect to a.
    
    Args:
        a: sequence of length n
        w: weight vector of length 2n-1 (softmax weights over convolution lags)
        fft_size: FFT size for computation
    
    Returns:
        grad: gradient vector of length n
    """
    n = len(a)
    W = np.zeros(fft_size)
    W[:2*n-1] = w
    A = np.zeros(fft_size)
    A[:n] = a
    # Cross-correlation: (W * A)[i] = sum_j W[i+j] * A[j]
    # = ifft(fft(W) * conj(fft(A)))
    Wf = np.fft.rfft(W)
    Af = np.fft.rfft(A)
    cross = np.fft.irfft(Wf * np.conj(Af), fft_size)
    grad = 2.0 * cross[:n]
    return grad
```

```python
import numpy as np
```

```python
def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def score_from_conv(a, conv):
    """Compute score given a and its autocorrelation (sum(a)=1 assumed)."""
    n = len(a)
    return float(2.0 * n * conv.max())


def smooth_max_weights(conv, beta):
    """Compute softmax weights for smooth max approximation."""
    m = conv.max()
    shifted = beta * (conv - m)
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones_like(conv) / len(conv)
    else:
        w = w / ws
    return w


def compute_grad_fft(a, w):
    """
    Compute gradient of smooth_max(conv(a,a)) w.r.t. a using FFT cross-correlation.
    grad[i] = 2 * sum_k w[k] * a[k-i] (where valid)
    This equals 2 * cross_correlation(w, a)[i]
    """
    n = len(a)
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.zeros(fft_size)
    W[:len(w)] = w
    A = np.zeros(fft_size)
    A[:n] = a
    Wf = np.fft.rfft(W)
    Af = np.fft.rfft(A)
    cross = np.fft.irfft(Wf * np.conj(Af), fft_size)
    return 2.0 * cross[:n]


def mirror_descent_smooth(a0, time_limit, beta_init=20.0, beta_max=200.0,

def spectral_shaping(a0, time_limit, lr_init=0.1, verbose=False):

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss

            best_seq = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")

    # Try uniform as baseline
    a_uni = np.ones(n0) / n0
    sc_uni = evaluate_safe(a_uni)
    if np.isfinite(sc_uni) and sc_uni < best_score:
        best_score = sc_uni
        best_seq = a_uni.copy()
        print(f"  Uniform improves: {sc_uni:.6f}")

    # === Phase 1: Smooth-max mirror descent (main optimization) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Smooth-max MD (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 30:
        a_out, _ = mirror_descent_smooth(best_seq, t1, beta_init=15.0, beta_max=150.0,
                                         lr_init=0.4, momentum=0.7, verbose=True, rng_seed=42)
        check_update(a_out, f"md_smooth_n{len(best_seq)}")

    # === Phase 2: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Coord descent (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 20:
        a_out, _ = coordinate_descent(best_seq, t2, rng_seed=123, verbose=True)
        check_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 3: Spectral shaping ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Spectral (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 15:
        a_out, _ = spectral_shaping(best_seq, t3, lr_init=0.15)
        check_update(a_out, f"spectral_n{len(best_seq)}")

    # === Phase 4: Second MD pass with higher beta ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Fine MD (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out, _ = mirror_descent_smooth(best_seq, t4, beta_init=50.0, beta_max=400.0,
                                         lr_init=0.15, momentum=0.8, verbose=False, rng_seed=99)
        check_update(a_out, f"md_fine_n{len(best_seq)}")

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration (budget={remaining:.0f}s) ===")

    length_tries = []
    for delta in [-2000, -1500, -1000, -500, 500, 1000, 1500, 2000, 2500, 3000]:
        nn = n_best + delta
        if 2000 <= nn <= 12000 and nn != n_best:
            length_tries.append(nn)
    for rn in [3000, 4000, 5000, 6000, 7000, 8000, 9000, 10000]:
        if rn != n_best and 2000 <= rn <= 12000:
            length_tries.append(rn)
    length_tries = sorted(set(length_tries))

    n_lengths_tried = 0
    for new_n in length_tries:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        if n_lengths_tried >= 4:
            break
        n_lengths_tried += 1
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")

        a_interp = interpolate_sequence(best_seq, new_n)

        # Quick mirror descent on new length
        t5a = min(remaining * 0.4, 150.0)
        if t5a > 20:
            a_out, _ = mirror_descent_smooth(a_interp, t5a, beta_init=20.0, beta_max=200.0,
                                             lr_init=0.3, momentum=0.7, verbose=False, rng_seed=new_n % 10000)
            check_update(a_out, f"md_n{new_n}")

        # Quick coord descent
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t5b = min(remaining * 0.3, 80.0)
            if t5b > 15:
                a_out, _ = coordinate_descent(a_interp, t5b, rng_seed=new_n % 9999, verbose=False)
                check_update(a_out, f"coord_n{new_n}")

    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_best}, budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.45, 180.0)
    if t6 > 20:
        a_out, _ = mirror_descent_smooth(best_seq, t6, beta_init=80.0, beta_max=600.0,
                                         lr_init=0.1, momentum=0.85, verbose=False, rng_seed=2024)
        check_update(a_out, f"final_md_n{n_best}")

    # Final coord descent
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t7 = min(remaining * 0.5, 80.0)
        if t7 > 10:
            a_out, _ = coordinate_descent(best_seq, t7, rng_seed=42, verbose=False)
            check_update(a_out, f"final_coord_n{len(best_seq)}")

    # Final spectral
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t8 = min(remaining * 0.4, 50.0)
        if t8 > 10:
            a_out, _ = spectral_shaping(best_seq, t8, lr_init=0.08)
            check_update(a_out, f"final_spec_n{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")

    return [float(x) for x in best_seq]
```
