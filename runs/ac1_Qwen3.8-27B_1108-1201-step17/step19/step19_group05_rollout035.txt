Function `fast_convolve(a)`:
```
def fast_convolve(a):
    a=np.asarray(a,float)
    n=len(a)
    if n==0: return np.array([])
    size=2*n-1
    # For small n maybe direct np.convolve for accuracy?
    if size <= 2048:
        return np.convolve(a,a)
    nfft=1 << (size-1).bit_length()
    A=np.fft.rfft(a,nfft)
    c=np.fft.irfft(A*A,nfft)
    return c[:size]
```
For small, np.convolve exact. For large, FFT may produce small negative; take max with 0.

s=a.sum()

return 2.0*len(a)*M/(s*s)
```

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = 2*n - 1
    if size <= 4096:
        c = np.convolve(a, a)
    else:
        nfft = 1 << (size - 1).bit_length()
        A = np.fft.rfft(a, nfft)
        c = np.fft.irfft(A * A, nfft)[:size]
    c = np.asarray(c, dtype=np.float64)
    c[c < 0.0] = 0.0
    return c

```python
"""
Search for a low-scoring non-negative step-function sequence.

Core idea:
- The objective is scale invariant. Work with probability vectors (sum = 1).
- A strong analytic seed family is a one-sided power law a_i ~ (i+1)^alpha.
  For alpha near 2 it gives an asymptotic score well below 1.5.
- Refine with:
  * smoothed mirror descent on the simplex using top-peak subgradients;
  * fast exact coordinate moves using the algebraic update of a * a after
    transferring mass between two coordinates;
  * occasional restarts / random perturbations.

All helpers are top-level, no lambdas, no closures.
"""

import numpy as np
import time


def normalize_sequence(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return None
    arr = arr.copy()
    bad = ~np.isfinite(arr)
    if bad.any():
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    arr /= s
    return arr


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = 2 * n - 1
    if size <= 4096:
        c = np.convolve(a, a)
    else:
        nfft = 1 << (size - 1).bit_length()
        A = np.fft.rfft(a, nfft)
        c = np.fft.irfft(A * A, nfft)[:size]
    c = np.asarray(c, dtype=np.float64)
    c[c < 0.0] = 0.0
    return c


def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return float('inf')
    c = fast_convolve(a)
    if c.size == 0:
        return float('inf')
    m = float(c.max())
    if m < 0.0:
        m = 0.0
    if not np.isfinite(m):
        return float('inf')
    v = 2.0 * float(a.size) * m / (s * s)
    return float(v) if np.isfinite(v) else float('inf')


def safe_eval(a):
    try:
        arr = np.asarray(a, dtype=np.float64)
        if arr.ndim != 1 or arr.size == 0:
            return float('inf')
        if not np.all(np.isfinite(arr)):
            return float('inf')
        arr = np.clip(arr, 0.0, 1000.0)
        if float(arr.sum()) < 0.01:
            return float('inf')
        return float(evaluate_sequence(arr.tolist()))
    except Exception:
        return float('inf')


def convergence_tol(m):
    m = float(m)
    if not np.isfinite(m) or m <= 0.0:
        return 1e-13
    return max(1e-13, 1e-10 * m)


def make_state():
    return {
        'fast_seq': None,
        'fast_score': float('inf'),
        'fast_label': '',
        'exact_seq': None,
        'exact_score': float('inf'),
        'exact_label': '',
    }


def set_fast_best(state, a, label):
    arr = normalize_sequence(a)
    if arr is None:
        return state
    sf = score_fast(arr)
    if not np.isfinite(sf):
        return state
    if sf < state['fast_score'] - 1e-12:
        state['fast_seq'] = arr.copy()
        state['fast_score'] = sf
        state['fast_label'] = str(label)
    return state


def validate_exact(state, a, label):
    arr = normalize_sequence(a)
    if arr is None:
        return state
    sf = score_fast(arr)
    if np.isfinite(sf) and sf < state['fast_score']:
        state['fast_seq'] = arr.copy()
        state['fast_score'] = sf
        state['fast_label'] = str(label)
    se = safe_eval(arr)
    if not np.isfinite(se):
        se = sf
    if state['exact_seq'] is None or (np.isfinite(se) and se < state['exact_score'] - 1e-12):
        state['exact_seq'] = arr.copy()
        state['exact_score'] = se
        state['exact_label'] = str(label)
    return state


def add_seed(seeds, a, label, counter, limit=20, state=None):
    arr = normalize_sequence(a)
    if arr is None:
        return counter
    sf = score_fast(arr)
    if not np.isfinite(sf):
        return counter

    if state is not None and sf < state['fast_score']:
        state['fast_seq'] = arr.copy()
        state['fast_score'] = sf
        state['fast_label'] = str(label)

    item = (float(sf), int(counter), arr, str(label))
    counter += 1

    if len(seeds) < limit:
        seeds.append(item)
    else:
        max_i = 0
        max_val = seeds[0][0]
        for i in range(1, len(seeds)):
            if seeds[i][0] > max_val:
                max_val = seeds[i][0]
                max_i = i
        if sf < max_val:
            seeds[max_i] = item
    return int(counter)


def make_power(n, alpha, beta=0.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(x, float(alpha))
    if beta != 0.0:
        r = float(n) - x + 1.0
        a = a * np.power(r, float(beta))
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    a /= s
    return a


def make_exp_ramp(n, t):
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(float(t) * x)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    a /= s
    return a


def make_gaussian(n, center, width):
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    w = max(float(width), 1e-6)
    a = np.exp(-((x - float(center)) / w) ** 2)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    a /= s
    return a


def make_endpoint_atom(n, base_alpha, frac, use_left, use_right):
    a = make_power(n, base_alpha)
    frac = float(max(0.0, min(0.999, frac)))
    if use_left:
        a[0] = (1.0 - frac) * a[0] + frac
    if use_right:
        a[-1] = (1.0 - frac) * a[-1] + frac
    return normalize_sequence(a)


def interpolate_sequence(a, new_n):
    arr = normalize_sequence(a)
    if arr is None:
        return None
    if arr.size == new_n:
        return arr.copy()
    old_x = np.linspace(0.0, 1.0, arr.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    b = np.interp(new_x, old_x, arr)
    return normalize_sequence(b)


def gradient_top_peaks(a, c, n_peaks=30, temp_frac=0.03):
    n = len(a)
    cmax = float(c.max())
    if cmax <= 0.0 or not np.isfinite(cmax):
        return np.zeros(n, dtype=np.float64)

    k = min(int(n_peaks), int(c.size))
    idx = np.argpartition(c, -k)[-k:]
    vals = c[idx]
    maxv = float(vals.max())
    scale = max(cmax * float(temp_frac), 1e-300)
    w = np.exp((vals - maxv) / scale)
    sw = float(w.sum())
    if not np.isfinite(sw) or sw <= 0.0:
        w = np.ones(k, dtype=np.float64) / float(k)
    else:
        w /= sw

    grad = np.zeros(n, dtype=np.float64)
    for ii in range(k):
        ki = int(idx[ii])
        wi = float(w[ii])
        if wi <= 0.0:
            continue
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo > hi:
            continue
        grad[lo:hi + 1] += (2.0 * wi) * a[ki - hi:ki - lo + 1][::-1]
    return grad


def mirror_descent(a0, time_limit, n_peaks=30, lr0=1.0, restart_every=300, rng_seed=42):
    a = normalize_sequence(a0)
    if a is None:
        return None, float('inf')
    if time_limit <= 0.0:
        c = fast_convolve(a)
        return a, float(c.max())

    t_start = time.monotonic()
    n = len(a)
    x = a.copy()
    x_prev = x.copy()
    M_cur = float(fast_convolve(x).max())
    best_a = x.copy()
    best_m = M_cur

    it = 0
    stall = 0
    restarts = 0
    lr = float(lr0)
    rng = np.random.default_rng(int(rng_seed))

    while time.monotonic() - t_start < time_limit:
        it += 1
        beta = float(it - 1) / float(it + 2)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-300)
        sy = float(y.sum())
        if not np.isfinite(sy) or sy <= 1e-150:
            y = x.copy()
        else:
            y = y / sy

        cy = fast_convolve(y)
        M_y = float(cy.max())
        tol = convergence_tol(M_cur)

        if M_y < M_cur - tol:
            old = x
            x = y.copy()
            x_prev = old
            M_cur = M_y
            if M_cur < best_m:
                best_m = M_cur
                best_a = x.copy()
            stall = 0
            lr = min(5.0, lr * 1.02)
            continue

        grad = gradient_top_peaks(y, cy, n_peaks=n_peaks, temp_frac=0.03)
        gp = grad[grad > 0.0]
        improved = False

        if gp.size > 0:
            gscale = float(np.percentile(gp, 60))
            if not np.isfinite(gscale) or gscale <= 1e-300:
                gscale = 1.0

            etas = [lr, lr * 0.7, lr * 1.3, lr * 0.4]
            for eta in etas:
                if improved:
                    break
                if not np.isfinite(eta) or eta <= 0.0:
                    continue

                loga = np.log(y + 1e-300)
                loga -= (float(eta) / gscale) * grad
                mx = float(loga.max())
                if not np.isfinite(mx):
                    continue
                loga -= mx
                np.clip(loga, -50.0, 50.0, out=loga)
                a_new = np.exp(loga)
                sn = float(a_new.sum())
                if not np.isfinite(sn) or sn <= 1e-150:
                    continue
                a_new /= sn

                for tb in (1.0, 0.7, 0.4, 0.2, 0.05):
                    if tb >= 1.0:
                        cand = a_new
                    else:
                        cand = (1.0 - tb) * x + tb * a_new
                        sc = float(cand.sum())
                        if not np.isfinite(sc) or sc <= 1e-150:
                            continue
                        cand = cand / sc

                    cand = np.maximum(cand, 0.0)
                    sc = float(cand.sum())
                    if not np.isfinite(sc) or sc <= 1e-150:
                        continue
                    cand = cand / sc

                    Mc = float(fast_convolve(cand).max())
                    if Mc < M_cur - tol:
                        old = x
                        x = cand.copy()
                        x_prev = old
                        M_cur = Mc
                        if M_cur < best_m:
                            best_m = M_cur
                            best_a = x.copy()
                        improved = True
                        break

        if not improved:
            x_prev = x
            stall += 1
            if stall % 20 == 0:
                lr = max(0.1, lr * 0.85)

            if stall > restart_every:
                strength = 0.05 + 0.02 * min(restarts, 5)
                x = best_a * np.exp(rng.normal(0.0, strength, n))
                x = np.maximum(x, 0.0)
                sx = float(x.sum())
                if not np.isfinite(sx) or sx <= 1e-150:
                    x = best_a.copy()
                else:
                    x = x / sx
                M_cur = float(fast_convolve(x).max())
                x_prev = x.copy()
                restarts += 1
                stall = 0
                lr = float(lr0)
                if M_cur < best_m:
                    best_m = M_cur
                    best_a = x.copy()

    return best_a, best_m


def active_weighted_cost(a, c, k_active=24):
    n = len(a)
    cmax = float(c.max())
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.zeros(n, dtype=np.float64), np.array([], dtype=np.int64)

    k = min(int(k_active), int(c.size))
    idx = np.argpartition(c, -k)[-k:]
    order = np.argsort(c[idx])[::-1]
    peaks = idx[order]
    vals = c[peaks]

    w = vals - 0.95 * float(vals[0])
    w = np.maximum(w, 0.0)
    sw = float(w.sum())
    if not np.isfinite(sw) or sw <= 0.0:
        w = np.ones(len(peaks), dtype=np.float64) / float(len(peaks))
    else:
        w /= sw

    cost = np.zeros(n, dtype=np.float64)
    for ii in range(len(peaks)):
        ki = int(peaks[ii])
        wi = float(w[ii])
        if wi <= 0.0:
            continue
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo > hi:
            continue
        cost[lo:hi + 1] += wi * a[ki - hi:ki - lo + 1][::-1]
    return cost, peaks


def apply_move(c, a, s, d, delta):
    n = len(a)
    trial = c.copy()
    two_delta = 2.0 * float(delta)
    trial[d:d + n] += two_delta * a
    trial[s:s + n] -= two_delta * a

    dd = float(delta) * float(delta)
    trial[2 * d] += dd
    trial[2 * s] += dd
    trial[s + d] -= 2.0 * dd
    return trial


def coordinate_polish(a0, time_limit, rng_seed=1, k_active=24,
                      max_peak_cases=1, max_src=3, max_dst=4):
    a = normalize_sequence(a0)
    if a is None:
        return None, float('inf')
    if time_limit <= 0.0:
        c = fast_convolve(a)
        return a, float(c.max())

    t_start = time.monotonic()
    n = len(a)
    c = fast_convolve(a)
    M_cur = float(c.max())

    best_a = a.copy()
    best_c = c.copy()
    best_m = M_cur

    rng = np.random.default_rng(int(rng_seed))
    stall = 0
    restarts = 0
    iterations = 0

    while time.monotonic() - t_start < time_limit:
        iterations += 1

        if iterations % 400 == 0:
            s = float(a.sum())
            if not np.isfinite(s) or abs(s - 1.0) > 1e-6 or s <= 1e-150:
                a = normalize_sequence(a)
                if a is None:
                    break
            c = fast_convolve(a)
            M_cur = float(c.max())
            if M_cur < best_m:
                best_m = M_cur
                best_a = a.copy()
                best_c = c.copy()
            if time.monotonic() - t_start >= time_limit:
                break

        if M_cur <= 0.0 or not np.isfinite(M_cur):
            break

        cost, peaks = active_weighted_cost(a, c, k_active=k_active)
        if peaks.size == 0:
            break

        tol = convergence_tol(M_cur)
        improved = False

        for ptr in range(min(int(max_peak_cases), int(peaks.size))):
            k0 = int(peaks[ptr])
            lo = max(0, k0 - n + 1)
            hi = min(n - 1, k0)
            if lo > hi:
                continue

            idx_arr = np.arange(lo, hi + 1, dtype=np.int64)
            prod = a[idx_arr] * a[k0 - idx_arr]

            if prod.size < max_src:
                base_order = np.argsort(prod)[::-1]
            else:
                base_order = np.argpartition(prod, -int(max_src))[::-1]

            sel_idx = idx_arr[base_order[:max_src]]
            sel_prod = prod[base_order[:max_src]]
            sel_order = np.argsort(sel_prod)[::-1]
            src_candidates = sel_idx[sel_order]

            dest_score = cost + 0.05 * a
            dest_order = np.argsort(dest_score)
            dest_candidates = dest_order[:max_dst]

            for ss in src_candidates:
                if improved:
                    break
                s = int(ss)
                if a[s] <= 1e-14:
                    continue

                ks0 = k0 - s
                partner_s = float(a[ks0]) if 0 <= ks0 < n else 0.0

                for dd in dest_candidates:
                    if improved:
                        break
                    d = int(dd)
                    if d == s:
                        continue

                    kd0 = k0 - d
                    partner_d = float(a[kd0]) if 0 <= kd0 < n else 0.0

                    if partner_d >= partner_s - 1e-15:
                        continue

                    best_delta = None
                    best_M = M_cur
                    best_trial = None

                    for frac in (0.1, 0.3, 0.6, 0.9):
                        delta = frac * float(a[s])
                        if delta <= 1e-14:
                            continue
                        if delta > float(a[s]):
                            delta = float(a[s])

                        trial = apply_move(c, a, s, d, delta)
                        Mt = float(trial.max())
                        if Mt < best_M:
                            best_M = Mt
                            best_delta = delta
                            best_trial = trial

                    if best_delta is not None and best_M < M_cur - tol:
                        a[s] -= best_delta
                        a[d] += best_delta
                        if a[s] < 0.0 and a[s] > -1e-12:
                            a[s] = 0.0
                        if a[d] < 0.0:
                            a[d] = 0.0
                        c = best_trial
                        M_cur = best_M
                        if M_cur < best_m - tol:
                            best_m = M_cur
                            best_a = a.copy()
                            best_c = c.copy()
                        stall = 0
                        improved = True
                        break

        if improved:
            continue

        stall += 1

        if stall >= 5 and (stall % 5) == 0:
            for _ in range(12):
                s = int(rng.integers(n))
                d = int(rng.integers(n))
                if s == d or a[s] <= 1e-14:
                    continue
                delta = float(a[s]) * float(rng.uniform(0.03, 0.25))
                trial = apply_move(c, a, s, d, delta)
                Mt = float(trial.max())
                if Mt < M_cur - tol:
                    a[s] -= delta
                    a[d] += delta
                    c = trial
                    M_cur = Mt
                    if M_cur < best_m:
                        best_m = M_cur
                        best_a = a.copy()
                        best_c = c.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                continue

        if stall > 120:
            strength = 0.05 + 0.02 * min(restarts, 5)
            x = best_a * np.exp(rng.normal(0.0, strength, n))
            x = np.maximum(x, 0.0)
            sx = float(x.sum())
            if not np.isfinite(sx) or sx <= 1e-150:
                x = best_a.copy()
            else:
                x = x / sx
            cx = fast_convolve(x)
            Mx = float(cx.max())
            if Mx < best_m * 1.05:
                a = x
                c = cx
                M_cur = Mx
            else:
                a = best_a.copy()
                c = best_c.copy()
                M_cur = best_m
            restarts += 1
            stall = 0

    # final correction
    if a is not None:
        c_final = fast_convolve(a)
        M_final = float(c_final.max())
        if M_final < best_m:
            best_m = M_final
            best_a = a.copy()
            best_c = c_final

    return best_a, best_m


def propose_candidate():
    t0 = time.monotonic()
    deadline = t0 + 1760.0
    N = 7000

    state = make_state()
    seeds = []
    counter = 0
    limit = 20

    # Try to use supplied parent, but do not depend on it.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            p = interpolate_sequence(parent, N)
            if p is not None:
                counter = add_seed(seeds, p, 'parent', counter, limit=limit, state=state)
    except Exception:
        pass

    # Strong analytic seed family: power-law ramps.
    for i in range(46):
        alpha = -0.5 + i * (4.5 / 45.0)
        a = make_power(N, alpha)
        counter = add_seed(seeds, a, 'power_{:.3f}'.format(alpha), counter, limit=limit, state=state)

    explicit_alphas = (0.8, 1.0, 1.5, 1.8, 2.0, 2.15, 2.3, 2.5, 3.0)
    for alpha in explicit_alphas:
        a = make_power(N, float(alpha))
        counter = add_seed(seeds, a, 'power_exp_{:.3f}'.format(alpha), counter, limit=limit, state=state)

    # Mild asymmetries for diversification.
    for alpha in (1.8, 2.0, 2.2):
        for beta in (-0.5, -0.2, 0.2, 0.5):
            a = make_power(N, alpha, beta)
            counter = add_seed(seeds, a, 'power_asym_{:.2f}_{:.2f}'.format(alpha, beta),
                               counter, limit=limit, state=state)

    # Exponential ramps.
    for t in (-2.0, -1.0, 0.0, 1.0, 2.0, 3.0, 4.0, 6.0, 8.0, 10.0):
        a = make_exp_ramp(N, t)
        counter = add_seed(seeds, a, 'exp_t_{:.1f}'.format(t), counter, limit=limit, state=state)

    # Skewed Gaussians.
    for center in (0.8, 0.9, 1.0):
        for width in (0.05, 0.1, 0.2):
            a = make_gaussian(N, center, width)
            counter = add_seed(seeds, a, 'gauss_{:.2f}_{:.2f}'.format(center, width),
                               counter, limit=limit, state=state)

    # Small endpoint atoms.
    for alpha in (1.5, 2.0, 2.5):
        for frac in (0.001, 0.005, 0.01):
            a = make_endpoint_atom(N, alpha, frac, True, False)
            counter = add_seed(seeds, a, 'atom_L_{:.2f}_{:.3f}'.format(alpha, frac),
                               counter, limit=limit, state=state)
            a = make_endpoint_atom(N, alpha, frac, False, True)
            counter = add_seed(seeds, a, 'atom_R_{:.2f}_{:.3f}'.format(alpha, frac),
                               counter, limit=limit, state=state)

    # Fallback if something went wrong.
    if not seeds:
        a = make_power(N, 2.0)
        counter = add_seed(seeds, a, 'fallback_power2', counter, limit=limit, state=state)

    seeds.sort()

    try:
        print('Seed scan done: n={}'.format(N))
        for item in seeds[:5]:
            print('  seed {score:.6f}  {label}'.format(score=item[0], label=item[3]))
    except Exception:
        pass

    # Exact evaluation of a few best seeds to anchor the final objective.
    for item in seeds[:3]:
        if deadline - time.monotonic() < 40.0:
            break
        validate_exact(state, item[2], item[3])

    try:
        if state['exact_seq'] is not None:
            print('Initial exact best: {score:.6f}  {label}'.format(
                score=state['exact_score'], label=state['exact_label']))
        else:
            print('Initial fast best: {score:.6f}  {label}'.format(
                score=state['fast_score'], label=state['fast_label']))
    except Exception:
        pass

    # Phase 1: mirror descent from a few good starts.
    starts = []
    if state['fast_seq'] is not None:
        starts.append((state['fast_seq'], state['fast_label']))
    for item in seeds[:6]:
        if len(starts) >= 3:
            break
        if state['fast_seq'] is not None and state['fast_label'] == item[3]:
            continue
        starts.append((item[2], item[3]))

    for si in range(len(starts)):
        if deadline - time.monotonic() < 120.0:
            break
        sa, slabel = starts[si]
        tlim = min(300.0, max(30.0, (deadline - time.monotonic()) * 0.25))
        try:
            a_out, _ = mirror_descent(sa, tlim, n_peaks=30, lr0=1.0,
                                      restart_every=300, rng_seed=42 + si)
            set_fast_best(state, a_out, 'mirror_{}_{}'.format(slabel, si))
            if deadline - time.monotonic() > 50.0:
                validate_exact(state, a_out, 'mirror_{}_{}'.format(slabel, si))
        except Exception:
            pass

    # Phase 2: coordinate polish of the best fast sequence.
    if state['fast_seq'] is not None and deadline - time.monotonic() > 120.0:
        tlim = min(700.0, max(60.0, (deadline - time.monotonic()) * 0.55))
        try:
            a_out, _ = coordinate_polish(state['fast_seq'], tlim, rng_seed=101,
                                         k_active=24, max_peak_cases=1,
                                         max_src=3, max_dst=4)
            set_fast_best(state, a_out, 'coord_best')
            if deadline - time.monotonic() > 50.0:
                validate_exact(state, a_out, 'coord_best')
        except Exception:
            pass

    # Phase 3: coordinate polish of a second independent seed if available.
    if len(seeds) > 1 and state['fast_seq'] is not None and deadline - time.monotonic() > 150.0:
        tlim = min(400.0, max(60.0, (deadline - time.monotonic()) * 0.40))
        try:
            a_out, _ = coordinate_polish(seeds[1][2], tlim, rng_seed=202,
                                         k_active=24, max_peak_cases=1,
                                         max_src=3, max_dst=4)
            set_fast_best(state, a_out, 'coord_seed2')
            if deadline - time.monotonic() > 50.0:
                validate_exact(state, a_out, 'coord_seed2')
        except Exception:
            pass

    # Phase 4: short mirror pass on current best fast.
    if state['fast_seq'] is not None and deadline - time.monotonic() > 120.0:
        tlim = min(250.0, max(30.0, (deadline - time.monotonic()) * 0.30))
        try:
            a_out, _ = mirror_descent(state['fast_seq'], tlim, n_peaks=40, lr0=2.0,
                                      restart_every=250, rng_seed=7)
            set_fast_best(state, a_out, 'mirror_late')
            if deadline - time.monotonic() > 50.0:
                validate_exact(state, a_out, 'mirror_late')
        except Exception:
            pass

    # Phase 5: final coordinate polish.
    if state['fast_seq'] is not None and deadline - time.monotonic() > 90.0:
        tlim = min(450.0, max(30.0, (deadline - time.monotonic()) * 0.55))
        try:
            a_out, _ = coordinate_polish(state['fast_seq'], tlim, rng_seed=303,
                                         k_active=28, max_peak_cases=2,
                                         max_src=4, max_dst=5)
            set_fast_best(state, a_out, 'coord_final')
            if deadline - time.monotonic() > 50.0:
                validate_exact(state, a_out, 'coord_final')
        except Exception:
            pass

    # If the fast best is noticeably below exact best, validate it before finishing.
    if (state['fast_seq'] is not None and state['exact_seq'] is not None and
            state['fast_score'] < state['exact_score'] - 1e-5 and
            deadline - time.monotonic() > 60.0):
        try:
            validate_exact(state, state['fast_seq'], 'final_check_fast')
        except Exception:
            pass

    elif state['exact_seq'] is None and state['fast_seq'] is not None and \
            deadline - time.monotonic() > 80.0:
        try:
            validate_exact(state, state['fast_seq'], 'final_check_fast')
        except Exception:
            pass

    # Choose final sequence. Prefer a sequence with a proven exact score when
    # available; otherwise rely on the best fast candidate.
    final_seq = None
    if state['fast_seq'] is not None and state['exact_seq'] is not None:
        if state['fast_score'] < state['exact_score'] - 1e-5:
            final_seq = state['fast_seq']
        else:
            final_seq = state['exact_seq']
    elif state['fast_seq'] is not None:
        final_seq = state['fast_seq']
    elif state['exact_seq'] is not None:
        final_seq = state['exact_seq']

    if final_seq is None:
        final_seq = make_power(N, 2.0)

    final_seq = normalize_sequence(final_seq)
    if final_seq is None:
        final_seq = make_power(N, 2.0)

    out = [float(max(0.0, x)) for x in final_seq.tolist()]

    try:
        print('FINAL returned: n={}'.format(len(out)))
        if state['exact_seq'] is not None:
            print('FINAL exact best known: {score:.6f}  {label}'.format(
                score=state['exact_score'], label=state['exact_label']))
        if state['fast_seq'] is not None:
            print('FINAL fast best known: {score:.6f}  {label}'.format(
                score=state['fast_score'], label=state['fast_label']))
        print('Elapsed: {:.1f}s'.format(time.monotonic() - t0))
    except Exception:
        pass

    return out
