```python
"""
Improved high-dimensional optimizer for
    J(a) = 2 * n * max conv(a, a) / (sum a)^2.

Strategy:
1. Normalize to the probability simplex (scale invariance of the objective).
2. Start from the provided parent when available.
3. Use entropy mirror descent (softmax/weighted-gradient) with Nesterov-style
   momentum, where the weights on convolution coefficients are periodically
   updated by a small LP dual.  The LP dual produces weights that try to
   flatten the first-order directional derivative over the simplex.
4. Use an LP-based Frank-Wolfe/cutting-plane polishing pass:
       min   M
       s.t. sum(q)=1, q>=0,
            (p*q)_k <= M  for generated constraints k.
   This gives strong local descent directions for the bilinear peak objective.
5. Alternate these passes, with restarts, and a small amount of length
   exploration near the end.
"""

import numpy as np
import time

try:
    import cvxpy as cp
except Exception:
    cp = None


def _next_power2(m: int) -> int:
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _norm_sum1(a) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _fast_autocorr(a) -> np.ndarray:
    """Full linear convolution a*a, length 2n-1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    size = 2 * n - 1
    N = _next_power2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    # Remove small negative numerical artifacts.
    return np.maximum(c, 0.0)


def _cross_conv(a, b) -> np.ndarray:
    """Full linear convolution a*b, length 2n-1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size
    if n == 0 or b.size != n:
        return np.zeros(2 * max(n, b.size) - 1, dtype=np.float64)
    size = 2 * n - 1
    N = _next_power2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    return np.maximum(c, 0.0)


def _sequence_score(a) -> float:
    """Internal score equivalent to evaluate_sequence for nonnegative a."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    a = a / s
    m = float(_fast_autocorr(a).max())
    return float(2.0 * a.size * m)


def _better_candidate(a, best_seq, best_score):
    if a is None:
        return best_seq, best_score
    sc = _sequence_score(a)
    if np.isfinite(sc) and sc < best_score:
        return _norm_sum1(a), sc
    return best_seq, best_score


def _softmax_weights(ac, beta) -> np.ndarray:
    ac = np.asarray(ac, dtype=np.float64)
    if ac.size == 0:
        return np.array([1.0])
    cmax = float(ac.max())
    if not np.isfinite(cmax):
        return np.ones(ac.size, dtype=np.float64) / ac.size
    arg = beta * (ac - cmax)
    arg = np.where(arg < -700.0, -700.0, arg)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(ac.size, dtype=np.float64) / ac.size
    return e / s


def _weighted_gradient(p, w) -> np.ndarray:
    """
    Gradient of sum_k w_k (p*p)_k, scaled by n, mean removed.

    The mean removal is harmless in the entropy simplex step because the
    resulting probability vector is renormalized.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return np.array([0.0])
    full_size = 2 * n - 1
    if w.size != full_size:
        tmp = np.zeros(full_size, dtype=np.float64)
        m = min(w.size, full_size)
        if m > 0:
            tmp[:m] = w[:m]
        s = float(tmp.sum())
        if s <= 0.0 or not np.isfinite(s):
            w = np.full(full_size, 1.0 / full_size)
        else:
            w = tmp / s
    else:
        w = np.where(np.isfinite(w), w, 0.0)
        s = float(w.sum())
        if s <= 0.0 or not np.isfinite(s):
            w = np.full(full_size, 1.0 / full_size)
        else:
            w = w / s

    N = _next_power2(full_size)
    P = np.fft.rfft(p[::-1], N)
    W = np.fft.rfft(w, N)
    conv = np.fft.irfft(W * P, N)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    # Scale to the actual objective magnitude (up to a constant factor).
    grad = grad * float(n)
    return grad - float(grad.mean())


def _mirror_step(y, grad, eta) -> np.ndarray:
    n = y.size
    if n == 0:
        return y
    log_y = np.log(np.maximum(y, 1e-300)) - eta * grad
    maxl = float(log_y.max())
    if not np.isfinite(maxl):
        return np.ones(n, dtype=np.float64) / float(n)
    z = np.exp(log_y - maxl)
    z = np.maximum(z, 1e-300)
    s = float(z.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return z / s


def _nesterov_combine(z, y, t) -> np.ndarray:
    if t == 1:
        return z.copy()
    mu = float(t - 1.0) / float(t + 2.0)
    x = z + mu * (z - y)
    x = np.maximum(x, 1e-300)
    s = float(x.sum())
    if s <= 0.0 or not np.isfinite(s):
        return z.copy()
    return x / s


def _perturb_sequence(best, seed, sigma) -> np.ndarray:
    n = best.size
    rng = np.random.default_rng(seed)
    best = np.asarray(best, dtype=np.float64).ravel()
    # Multiplicative positive perturbation in log-space.
    noise = rng.normal(0.0, float(sigma), n)
    x = np.maximum(best * np.exp(-noise), 1e-16)
    s = float(x.sum())
    if s <= 0.0 or not np.isfinite(s):
        return _norm_sum1(best)
    return x / s


def _build_row_matrix(p, constraints) -> np.ndarray:
    """Dense matrix A where A[k,j] = p_{j+k offset} for constraint k."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    constraints = [int(k) for k in constraints]
    K = len(constraints)
    A = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(constraints):
        if 0 <= k < n:
            A[jj, 0:k + 1] = p[k::-1]
        elif n <= k <= 2 * n - 2:
            start = k - n + 1
            if start <= 0:
                A[jj, 0:n] = p[n - 1::-1]
            else:
                A[jj, start:n] = p[n - 1:start - 1:-1]
    return A


def _dual_lp_weights(p, constraints, solver):
    """
    Solve the active-constraint dual:
        max nu
        s.t. sum(lambda)=1, lambda>=0,
             A.T lambda >= nu.

    Returns full normalized weights on convolution lags, or None.
    """
    if cp is None or solver is None:
        return None
    p = np.asarray(p, dtype=np.float64).ravel()
    N = p.size
    constraints = [int(k) for k in constraints if 0 <= k <= 2 * N - 2]
    if not constraints:
        return None
    try:
        A = _build_row_matrix(p, constraints)
        lam = cp.Variable(len(constraints), nonneg=True)
        nu = cp.Variable()
        cons = [
            cp.sum(lam) == 1.0,
            A.T @ lam >= nu,
            nu >= 0.0
        ]
        prob = cp.Problem(cp.Maximize(nu), cons)
        prob.solve(solver=solver, verbose=False)
        if prob.status not in ("optimal", "optimal_inaccurate"):
            return None
        vals = np.asarray(lam.value, dtype=np.float64).ravel()
        if vals is None or vals.size != len(constraints):
            return None
        vals = np.where(np.isfinite(vals), vals, 0.0)
        vals = np.maximum(vals, 0.0)
        s = float(vals.sum())
        if s <= 0.0:
            return None
        w = np.zeros(2 * N - 1, dtype=np.float64)
        for k, v in zip(constraints, vals):
            w[k] = v
        w /= w.sum()
        return w
    except Exception:
        return None


def _fista_lp_guided(a0, time_limit, solver,
                     beta_start=3.0, beta_end=120.0,
                     eta0=0.06,
                     lp_K=300, lp_every=400,
                     mix_start=0.25, mix_end=0.85,
                     restart_threshold=4500,
                     verbose=True):
    n = len(a0)
    y = _norm_sum1(a0)
    best = y.copy()
    best_m = float(_fast_autocorr(best).max())
    t0 = time.monotonic()
    it = 0
    stall = 0
    restart_count = 0
    lp_w = None
    last_lp = -10 ** 9

    K_cap = max(120, int(1.5 * lp_K))

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        ac_y = _fast_autocorr(y)
        m_y = float(ac_y.max())
        if m_y < best_m - 1e-15:
            best_m = m_y
            best = y.copy()

        # Occasionally update LP-guided weights.
        do_lp = (
            solver is not None
            and (lp_w is None or it - last_lp >= lp_every)
            and (time_limit - elapsed) > 25.0
        )
        if do_lp:
            top_idx = np.argsort(ac_y)[::-1][:lp_K]
            cons_set = set(int(i) for i in top_idx)
            cons_set.add(n - 1)
            cons_set.add(0)
            cons_set.add(2 * n - 2)
            if lp_w is not None:
                w_top = np.argsort(lp_w)[::-1][:max(40, lp_K // 2)]
                for i in w_top:
                    cons_set.add(int(i))
            if len(cons_set) > K_cap:
                arr = np.array(list(cons_set), dtype=int)
                keep = np.argsort(ac_y[arr])[::-1][:K_cap]
                cons_set = set(int(v) for v in arr[keep])
            new_w = _dual_lp_weights(y, sorted(cons_set), solver)
            if new_w is not None:
                lp_w = new_w
                last_lp = it

        progress = min(1.0, elapsed / max(0.75 * time_limit, 1.0))
        beta = beta_start * ((beta_end / max(beta_start, 1e-12)) ** progress)
        sw = _softmax_weights(ac_y, beta)

        if lp_w is not None:
            mix = mix_start + (mix_end - mix_start) * progress
            w = mix * lp_w + (1.0 - mix) * sw
            s = float(w.sum())
            if s <= 0.0 or not np.isfinite(s):
                w = sw.copy()
            else:
                w = w / s
        else:
            w = sw

        grad = _weighted_gradient(y, w)

        eta = eta0 / (np.sqrt(float(it)) + 1.0)
        # Mild cyclical boost to escape plateaus.
        if it % 3000 < 250:
            eta *= 1.5
        # Later iterations: slightly stronger exploration if stalled.
        if stall > restart_threshold // 2:
            eta *= 1.25

        z = _mirror_step(y, grad, eta)
        x_new = _nesterov_combine(z, y, it)
        m_new = float(_fast_autocorr(x_new).max())

        # Conservative damping if the step is clearly bad.
        if m_new > m_y * 1.10:
            x_new = 0.90 * y + 0.10 * x_new
            x_new = np.maximum(x_new, 1e-300)
            s = float(x_new.sum())
            if s > 0.0:
                x_new = x_new / s
            m_new = float(_fast_autocorr(x_new).max())

        if m_new > m_y * 1.30:
            x_new = y.copy()
            m_new = m_y

        if m_new < best_m - 1e-15:
            best_m = m_new
            best = x_new.copy()
            stall = 0
        else:
            stall += 1

        y = x_new

        if verbose and it % 20000 == 0:
            print(f"    FISH-LP iter={it} t={elapsed:.0f}s "
                  f"best_m={best_m:.10f} score={2.0*n*best_m:.6f}")

        if stall > restart_threshold:
            sigma = min(0.28, 0.025 + 0.006 * restart_count)
            y = _perturb_sequence(best, 123456 + restart_count, sigma)
            best_m = float(_fast_autocorr(best).max())
            stall = 0
            restart_count += 1
            lp_w = None
            last_lp = -10 ** 9

    return best, best_m


def _lp_minm_polish(a0, time_limit, solver,
                    initial_K=250, max_K=750, verbose=True):
    """LP cutting-plane Frank-Wolfe polish (min M with sum(q)=1)."""
    if cp is None or solver is None:
        p = _norm_sum1(a0)
        return p, float(_fast_autocorr(p).max())

    n = len(a0)
    p = _norm_sum1(a0)
    p = np.maximum(p, 1e-16)
    p = p / p.sum()

    best = p.copy()
    ac = _fast_autocorr(p)
    best_m = float(ac.max())
    t0 = time.monotonic()
    it = 0
    fails = 0
    stall = 0

    cons_set = set(int(i) for i in np.argsort(ac)[::-1][:initial_K])
    cons_set.add(n - 1)
    cons_set.add(0)
    cons_set.add(2 * n - 2)

    score = ac.copy()

    while time.monotonic() - t0 < time_limit:
        if len(cons_set) == 0:
            cons_set = set(int(i) for i in np.argsort(ac)[::-1][:initial_K])
            cons_set.add(n - 1)

        if it % 10 == 0:
            score = np.maximum(score, ac)
            top_self = np.argsort(ac)[::-1][:min(30, max(10, max_K // 30))]
            for i in top_self:
                cons_set.add(int(i))

        if len(cons_set) > max_K:
            arr = np.array(list(cons_set), dtype=int)
            vals = score[arr]
            keep = np.argsort(vals)[::-1][:max_K]
            cons_set = set(int(v) for v in arr[keep])
            cons_set.add(n - 1)

        cons_list = sorted(cons_set)
        K = len(cons_list)
        if K == 0:
            break

        try:
            A = _build_row_matrix(p, cons_list)
            q = cp.Variable(n, nonneg=True)
            Mv = cp.Variable()
            cons = [
                cp.sum(q) == 1.0,
                A @ q <= Mv,
                Mv >= 0.0
            ]
            prob = cp.Problem(cp.Minimize(Mv), cons)
            prob.solve(solver=solver, verbose=False)
            if prob.status not in ("optimal", "optimal_inaccurate"):
                raise RuntimeError("LP status")
            qv = np.asarray(q.value, dtype=np.float64).ravel()
            if qv is None or qv.size != n or not np.isfinite(qv).all():
                raise RuntimeError("bad LP q")
        except Exception:
            fails += 1
            if fails > 3:
                keep = max(40, K // 2)
                arr = np.array(cons_list, dtype=int)
                vals = score[arr]
                keep_idx = np.argsort(vals)[::-1][:keep]
                cons_set = set(int(v) for v in arr[keep_idx])
                cons_set.add(n - 1)
            continue

        fails = 0
        qv = np.maximum(qv, 0.0)
        S = float(qv.sum())
        if S <= 1e-14:
            qv = p.copy()
        else:
            qv = qv / S

        Mv_val = float(Mv.value) if Mv.value is not None else float(best_m)
        if not np.isfinite(Mv_val):
            Mv_val = float(best_m)

        ac_q = _fast_autocorr(qv)
        m_q = float(ac_q.max())
        if m_q < best_m - 1e-14:
            best_m = m_q
            best = qv.copy()

        b_cross = _cross_conv(p, qv)
        tol = max(1e-12, 1e-7 * max(abs(Mv_val), 1e-12))
        viol = np.where(b_cross > Mv_val + tol)[0]
        if viol.size > 0:
            order = np.argsort(b_cross[viol])[::-1]
            add = viol[order[:35]]
            for k in add:
                cons_set.add(int(k))

        arr = np.array(cons_set, dtype=int)
        if arr.size > 0:
            score[arr] = np.maximum(score[arr], b_cross[arr])

        if len(cons_set) > max_K:
            arr = np.array(cons_set, dtype=int)
            vals = score[arr]
            keep = np.argsort(vals)[::-1][:max_K]
            cons_set = set(int(v) for v in arr[keep])
            cons_set.add(n - 1)

        # Mix toward q; larger step when q directly improves the self peak.
        if m_q < best_m * (1.0 - 1e-5):
            gain = max(0.0, (best_m - m_q)) / max(best_m, 1e-15)
            alpha = min(0.88, 0.30 + 4.0 * gain + 2.0 / (float(it) + 2.0))
        else:
            alpha = min(0.22, 0.05 + 2.0 / (float(it) + 2.0))

        p_new = (1.0 - alpha) * p + alpha * qv
        p_new = np.maximum(p_new, 1e-16)
        p_new = p_new / p_new.sum()
        ac_new = _fast_autocorr(p_new)
        m_new = float(ac_new.max())

        if m_new > best_m * 1.05:
            alpha2 = 0.10
            p_new = (1.0 - alpha2) * p + alpha2 * qv
            p_new = np.maximum(p_new, 1e-16)
            p_new = p_new / p_new.sum()
            ac_new = _fast_autocorr(p_new)
            m_new = float(ac_new.max())

        if m_new < best_m - 1e-14:
            best_m = m_new
            best = p_new.copy()
            stall = 0
        else:
            stall += 1

        p = p_new
        ac = ac_new

        dist = float(np.sqrt(np.mean((qv - p) * (qv - p))))
        if stall > 18 or (it > 10 and dist < 1e-7):
            p = _perturb_sequence(best, 900000 + stall, 0.035)
            ac = _fast_autocorr(p)
            best_m = max(best_m, float(ac.max()))
            score = ac.copy()
            cons_set = set(int(i) for i in np.argsort(ac)[::-1][:initial_K])
            cons_set.add(n - 1)
            stall = 0
            fails = 0

        it += 1
        if verbose and it % 5 == 0:
            el = time.monotonic() - t0
            print(f"    LP-MinM iter={it} t={el:.0f}s |S|={len(cons_set)} "
                  f"best_m={best_m:.10f} score={2.0*n*best_m:.6f}")

    return best, best_m


def _interpolate_sequence(f, new_n: int) -> np.ndarray:
    f = np.asarray(f, dtype=np.float64).ravel()
    old_n = f.size
    if old_n == 0:
        return np.ones(max(1, new_n), dtype=np.float64) / max(1, new_n)
    new_n = max(1, int(new_n))
    if old_n == new_n:
        return _norm_sum1(f)
    f = _norm_sum1(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    g = np.interp(new_x, old_x, f)
    return _norm_sum1(g)


def _get_solver():
    if cp is None:
        return None
    names = ["CLARABEL", "PDLP", "ECOS", "SCS", "MOSEK"]
    for name in names:
        s = getattr(cp, name, None)
        if s is None:
            continue
        try:
            x = cp.Variable(25, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1.0, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                print(f"  LP solver: {name}")
                return s
        except Exception:
            continue
    print("  LP solver: none available")
    return None


def _default_initial(n: int) -> np.ndarray:
    n = max(1, int(n))
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.45 * np.sin(2.0 * np.pi * x) + 0.25 * np.cos(4.0 * np.pi * x + 0.7)
    a = np.maximum(a, 0.05)
    return _norm_sum1(a)


def propose_candidate(budget_s: float = 1800.0):
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    if time_budget < 20.0:
        best_seq = _default_initial(4000)
        return [float(x) for x in best_seq]

    solver = _get_solver()
    print(f"Total search budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float("inf")

    # Load parent if present.
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if p.size > 0 and p.sum() > 1e-10:
            best_seq = _norm_sum1(p)
            best_score = _sequence_score(best_seq)
            print(f"Parent loaded: n={best_seq.size}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = _default_initial(5200)
        best_score = _sequence_score(best_seq)
        print(f"Default initial: n={best_seq.size}, score={best_score:.6f}")

    # Keep length in a numerically/time-friendly range.
    if best_seq.size > 6500:
        best_seq = _interpolate_sequence(best_seq, 6500)
        best_score = _sequence_score(best_seq)
        print(f"Parent downsampled: n={best_seq.size}, score={best_score:.6f}")

    def _remaining():
        return max(0.0, time_budget - (time.monotonic() - t_start))

    # Phase 1: main guided entropy optimization.
    rem = _remaining()
    t1 = min(rem * 0.34, 570.0)
    if t1 > 30.0:
        print(f"\n=== Phase 1: guided entropy optimization (t={t1:.0f}s) ===")
        a1, _ = _fista_lp_guided(
            best_seq, t1, solver,
            beta_start=3.0, beta_end=120.0,
            eta0=0.06,
            lp_K=300, lp_every=400,
            mix_start=0.25, mix_end=0.85,
            restart_threshold=4500,
            verbose=True
        )
        old = best_score
        best_seq, best_score = _better_candidate(a1, best_seq, best_score)
        if best_score < old:
            print(f"  Phase 1 improved to {best_score:.6f}")

    # Phase 2: LP cutting-plane polish.
    rem = _remaining()
    t2 = min(rem * 0.25, 450.0)
    if t2 > 35.0:
        n_cur = best_seq.size
        max_K = int(max(200, min(800, 4_000_000 // max(1, n_cur))))
        init_K = int(max(100, max_K // 3))
        print(f"\n=== Phase 2: LP polish (t={t2:.0f}s, K<= {max_K}) ===")
        a2, _ = _lp_minm_polish(best_seq, t2, solver,
                                initial_K=init_K, max_K=max_K, verbose=True)
        old = best_score
        best_seq, best_score = _better_candidate(a2, best_seq, best_score)
        if best_score < old:
            print(f"  Phase 2 improved to {best_score:.6f}")

    # Phase 3: high-temperature / low-step guided entropy.
    rem = _remaining()
    t3 = min(rem * 0.20, 330.0)
    if t3 > 25.0:
        print(f"\n=== Phase 3: guided refinement (t={t3:.0f}s) ===")
        a3, _ = _fista_lp_guided(
            best_seq, t3, solver,
            beta_start=20.0, beta_end=500.0,
            eta0=0.03,
            lp_K=320, lp_every=500,
            mix_start=0.70, mix_end=0.95,
            restart_threshold=3500,
            verbose=True
        )
        old = best_score
        best_seq, best_score = _better_candidate(a3, best_seq, best_score)
        if best_score < old:
            print(f"  Phase 3 improved to {best_score:.6f}")

    # Phase 4: final LP polish.
    rem = _remaining()
    t4 = min(rem * 0.17, 300.0)
    if t4 > 30.0:
        n_cur = best_seq.size
        max_K = int(max(200, min(800, 4_000_000 // max(1, n_cur))))
        init_K = int(max(100, max_K // 3))
        print(f"\n=== Phase 4: final LP polish (t={t4:.0f}s, K<= {max_K}) ===")
        a4, _ = _lp_minm_polish(best_seq, t4, solver,
                                initial_K=init_K, max_K=max_K, verbose=True)
        old = best_score
        best_seq, best_score = _better_candidate(a4, best_seq, best_score)
        if best_score < old:
            print(f"  Phase 4 improved to {best_score:.6f}")

    # Phase 5: limited length exploration.
    rem = _remaining()
    if rem > 180.0:
        print(f"\n=== Phase 5: length exploration ===")
        n_cur = best_seq.size
        trial_ns = []
        for d in (500, -500, 1000, -1000, 1500, -1500):
            nn = n_cur + d
            if 3500 <= nn <= 6500:
                trial_ns.append(nn)
        trial_ns = sorted(set(trial_ns), reverse=True)

        count = 0
        for nn in trial_ns:
            rem = _remaining()
            if rem < 90.0 or count >= 2:
                break
            t6 = min(rem * 0.45, 140.0)
            print(f"  Trying length n={nn} (t={t6:.0f}s)")
            aa = _interpolate_sequence(best_seq, nn)
            aa, _ = _fista_lp_guided(
                aa, t6, solver,
                beta_start=5.0, beta_end=120.0,
                eta0=0.05,
                lp_K=260, lp_every=600,
                mix_start=0.30, mix_end=0.85,
                restart_threshold=3500,
                verbose=False
            )
            old = best_score
            best_seq, best_score = _better_candidate(aa, best_seq, best_score)
            if best_score < old:
                print(f"  Length n={nn} improved to {best_score:.6f}")
            count += 1

    # Final validation with the actual evaluator.
    try:
        candidate = [float(x) for x in best_seq]
        final_val = float(evaluate_sequence(candidate))
        if np.isfinite(final_val):
            best_score = final_val
        else:
            raise ValueError("nonfinite final value")
    except Exception:
        best_seq = _default_initial(5000)
        candidate = [float(x) for x in best_seq]
        try:
            best_score = float(evaluate_sequence(candidate))
        except Exception:
            best_score = float("inf")
        if not np.isfinite(best_score):
            # Extremely defensive fallback.
            x = np.linspace(0.0, 1.0, 4000)
            best_seq = _norm_sum1(np.sin(np.pi * x) ** 2)
            candidate = [float(v) for v in best_seq]
            try:
                best_score = float(evaluate_sequence(candidate))
            except Exception:
                best_score = float("inf")

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: score={best_score:.6f}, n={best_seq.size}, time={elapsed:.0f}s")
    return [float(x) for x in best_seq]
```
