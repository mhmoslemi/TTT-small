def fast_autocorr(a):

    a = np.asarray(a, dtype=np.float64)

    fft_size = 1 << (size - 1).bit_length()

def make_shape(n, shape_type='ramp', param=1.0):
    """Create various analytical shapes, all normalized to sum=1."""
    x = np.arange(n, dtype=np.float64)
    if shape_type == 'ramp':
        a = (x + 1) ** param
    elif shape_type == 'sine':
        a = np.sin(np.pi * (x + 1) / (n + 1))
    elif shape_type == 'cosine':
        a = 0.5 + 0.5 * np.cos(np.pi * x / max(1, n - 1))
    elif shape_type == 'exp':
        a = np.exp(-param * x / n)
    elif shape_type == 'quadratic':

    elif shape_type == 'sqrt_ramp':

def subgradient_step(a, conv, n_peaks=50, step_size=0.1):
    """
    One subgradient step: decrease a[i] where it contributes to peaks,
    increase a[j] where it contributes least.
    """
    n = len(a)
    # Find top peaks
    top_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
    
    # Compute gradient: how much each a[i] contributes to top peaks
    grad = np.zeros(n)
    for k in top_idx:
        ki = int(k)
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        for j in range(j_lo, j_hi + 1):
            pair = ki - j
            grad[j] += a[pair]
    
    # Total contribution
    total = grad.sum()
    if total < 1e-15:
        return a.copy()
    
    # Normalize gradient
    grad = grad / total
    
    # Step: move mass from high-grad to low-grad
    # New a: a_i * (1 - step * (grad_i - mean_grad))
    mean_grad = grad.mean()
    a_new = a * (1.0 - step_size * (grad - mean_grad))
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    return a_new

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using analytical shape initialization
(ramps achieve J~4/3) combined with careful peak-reduction via coordinate
descent, mirror descent, and spectral flattening. Only accepts strict improvements.

import numpy as np
import time


def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def make_shape(n, shape_type, param):
    """Create various analytical shapes, all normalized to sum=1."""
    x = np.arange(n, dtype=np.float64)
    if shape_type == 'ramp':
        a = (x + 1) ** param
    elif shape_type == 'sine':
        a = np.sin(np.pi * (x + 1) / (n + 1))
    elif shape_type == 'cosine':
        a = 0.5 + 0.5 * np.cos(np.pi * x / max(1, n - 1))
    elif shape_type == 'exp':
        a = np.exp(-param * x / n)
    elif shape_type == 'sqrt_ramp':
        a = np.sqrt(x + 1)
    elif shape_type == 'triangle':
        mid = (n - 1) / 2.0
        a = 1.0 - np.abs(x - mid) / max(1.0, mid)
        a = np.maximum(a, 0.0)
    elif shape_type == 'raised_cos':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    else:
        a = (x + 1)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def safe_eval(a):
    """Safely compute the objective value."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_peak_contrib(a, conv, n_peaks):
    """Compute per-element contribution to top peaks."""
    n = len(a)
    top_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
    grad = np.zeros(n)
    for k in top_idx:
        ki = int(k)
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        j_range = np.arange(j_lo, j_hi + 1)
        pair = ki - j_range
        grad[j_range] += a[pair]
    return grad


def coordinate_descent_peak(a0, time_limit, n_peaks, rng_seed, verbose):
    """
    Coordinate descent targeting the autocorrelation peak.
    Only accepts moves that strictly reduce the max convolution value.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), fast_autocorr(a0).max()
    a = a / s
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        it += 1
        conv = fast_autocorr(a)
        k_max = int(np.argmax(conv))

        j_lo = max(0, k_max - n + 1)
        j_hi = min(n - 1, k_max)
        if j_lo > j_hi:
            stall += 1
            continue

        idx = np.arange(j_lo, j_hi + 1)
        contrib = a[idx] * a[k_max - idx]

        if contrib.max() < 1e-18:
            stall += 1
            continue

        src = idx[np.argmax(contrib)]

        # Compute cost of each position (how much it adds to top peaks)
        top_ks = np.argpartition(conv, -n_peaks)[-n_peaks:]
        costs = np.zeros(n)
        for k2 in top_ks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            if lo2 <= hi2:
                valid = np.arange(lo2, hi2 + 1)
                costs[valid] += a[k2i - valid]

        # Score: prefer low cost and low current value
        scores = costs + 0.005 * a
        sorted_idx = np.argsort(scores)

        moved = False
        for dst in sorted_idx[:80]:
            if dst == src:
                continue
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if a[src] - amt < -1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] = a[src] - amt
                a_trial[dst] += amt
                M_t = fast_autocorr(a_trial).max()
                if M_t < best_m - 1e-18:
                    a = a_trial
                    best_m = M_t
                    best_a = a.copy()
                    stall = 0
                    moved = True
                    break
            if moved:
                break

        if not moved:
            stall += 1
            # Random perturbation to escape
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    amt = a[i] * 0.005
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    M_t = fast_autocorr(a_trial).max()
                    if M_t < best_m - 1e-18:
                        a = a_trial
                        best_m = M_t
                        best_a = a.copy()
                        stall = 0
                        break

        if verbose and it % 200 == 0:
            print(f"      CD it={it} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}, stall={stall}")

    return best_a, best_m


def mirror_descent_peak(a0, time_limit, n_peaks, lr0, rng_seed, verbose):
    """
    Mirror descent on the simplex using peak subgradient.
    Only tracks improvements (reverts if no progress).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-20)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        conv = fast_autocorr(y)
        grad = compute_peak_contrib(x, conv, n_peaks)

        eta = lr0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_x = fast_autocorr(x).max()
        if M_x < best_m - 1e-18:
            best_m = M_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            stall = 0
            x = best_a.copy()
            x_prev = x.copy()
            pert = rng.normal(0, 0.005, n) * best_a
            x = np.maximum(x + pert, 1e-20)
            x = x / x.sum()
            x_prev = x.copy()

        if verbose and t % 10000 == 0:
            print(f"      MD t={t} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}, stall={stall}")

    return best_a, best_m


def spectral_flatten_peak(a0, time_limit, verbose):
    """
    Gerchberg-Saxton style: flatten the power spectrum to reduce time-domain peaks.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_mean = P.mean()
        if P_max < 1e-30:
            break

        alpha = 0.2 + 0.5 * min(1.0, it / 200.0)
        P_new = P + alpha * (P_mean - P)
        P_new = np.maximum(P_new, 1e-30)

        mag_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        mag_ratio = np.clip(mag_ratio, 0.5, 2.0)
        A_new = A * mag_ratio
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend: keep some of the current to avoid divergence
            blend = 0.1
            a = (1 - blend) * a + blend * a_new
            a = np.maximum(a, 0.0)
            a = a / max(a.sum(), 1e-15)
            M_curr = fast_autocorr(a).max()
            if M_curr < best_m:
                best_m = M_curr
                best_a = a.copy()

        if verbose and it % 500 == 0:
            print(f"      Spec it={it} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss


def subgradient_polish(a0, time_limit, n_peaks, lr, rng_seed):
    """
    Simple subgradient method: reduce a[i] where it contributes to peaks,
    increase where it doesn't. Accept only if J improves.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        grad = compute_peak_contrib(a, conv, n_peaks)
        total_g = grad.sum()
        if total_g < 1e-15:
            break
        grad_norm = grad / total_g

        # Multiplicative update
        eta = lr / (np.sqrt(t) + 1.0)
        a_new = a * (1.0 + eta * (0.5 - grad_norm))
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                a = best_a.copy()
                stall = 0
                pert = rng.normal(0, 0.003, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 1e-20)
                a = a / a.sum()

    return best_a, best_m

        pass

    if best_seq is None:
        print("  No valid sequence found in Phase 0, using ramp fallback")
        best_seq = make_shape(4000, 'ramp', 1.0)
        best_score = safe_eval(best_seq)

    # Phase 1: Coordinate descent (most reliable peak reducer)
    print(f"\n=== Phase 1: Coordinate Descent (score={best_score:.6f}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        n_in = len(a_in)
        t1 = min(remaining * 0.30, 350.0)
        print(f"  CD: n={n_in}, t={t1:.0f}s")
        a_out, _ = coordinate_descent_peak(a_in, t1, n_peaks=20,
                                            rng_seed=42, verbose=True)
        try_update(a_out, "cd_phase1")

        # Second CD pass with different seed
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and best_seq is not None:
            a_in = best_seq / best_seq.sum()
            t1b = min(remaining * 0.20, 200.0)
            a_out, _ = coordinate_descent_peak(a_in, t1b, n_peaks=30,
                                                rng_seed=777, verbose=False)
            try_update(a_out, "cd_phase1b")

    # Phase 2: Mirror descent
    print(f"\n=== Phase 2: Mirror Descent (score={best_score:.6f}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        t2 = min(remaining * 0.20, 200.0)
        print(f"  MD: n={len(a_in)}, t={t2:.0f}s")
        a_out, _ = mirror_descent_peak(a_in, t2, n_peaks=50, lr0=2.0,
                                        rng_seed=99, verbose=True)
        try_update(a_out, "md_phase2")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and best_seq is not None:
            a_in = best_seq / best_seq.sum()
            t2b = min(remaining * 0.15, 120.0)
            a_out, _ = mirror_descent_peak(a_in, t2b, n_peaks=80, lr0=1.0,
                                            rng_seed=555, verbose=False)
            try_update(a_out, "md_phase2b")

    # Phase 3: Spectral flattening
    print(f"\n=== Phase 3: Spectral Flattening (score={best_score:.6f}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        t3 = min(remaining * 0.10, 100.0)
        a_out, _ = spectral_flatten_peak(a_in, t3, verbose=False)
        try_update(a_out, "spec_phase3")

        # CD after spectral
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and best_seq is not None:
            a_in = best_seq / best_seq.sum()
            t3b = min(remaining * 0.15, 120.0)
            a_out, _ = coordinate_descent_peak(a_in, t3b, n_peaks=20,
                                                rng_seed=333, verbose=False)
            try_update(a_out, "cd_post_spec")

    # Phase 4: Multi-length exploration
    print(f"\n=== Phase 4: Multi-length (score={best_score:.6f}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is not None and remaining > 60:
        n_best = len(best_seq)
        candidates_n = [n_best]
        for delta in [-2000, -1000, 1000, 2000, 3000, 5000]:
            nn = n_best + delta
            if 1000 <= nn <= 15000 and nn not in candidates_n:
                candidates_n.append(nn)

        for new_n in candidates_n:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            a_interp = interpolate_to_length(best_seq, new_n)
            try_update(a_interp, f"interp_n{new_n}")

            if remaining > 50:
                t4 = min(remaining * 0.15, 100.0)
                a_out, _ = coordinate_descent_peak(a_interp, t4, n_peaks=20,
                                                    rng_seed=100 + new_n % 1000,
                                                    verbose=False)
                try_update(a_out, f"cd_n{new_n}")

    # Phase 5: Subgradient polish
    print(f"\n=== Phase 5: Subgradient (score={best_score:.6f}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        t5 = min(remaining * 0.20, 150.0)
        a_out, _ = subgradient_polish(a_in, t5, n_peaks=40, lr=0.5,
                                       rng_seed=456)
        try_update(a_out, "sg_phase5")

    # Phase 6: Final combined polish
    print(f"\n=== Phase 6: Final Polish (score={best_score:.6f}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        # CD
        t6a = min(remaining * 0.30, 100.0)
        if t6a > 15:
            a_out, _ = coordinate_descent_peak(a_in, t6a, n_peaks=30,
                                                rng_seed=9999, verbose=False)
            try_update(a_out, "cd_final")
            a_in = best_seq / best_seq.sum()

        # MD
        remaining = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining * 0.30, 100.0)
        if t6b > 15:
            a_out, _ = mirror_descent_peak(a_in, t6b, n_peaks=60, lr0=1.5,
                                            rng_seed=7777, verbose=False)
            try_update(a_out, "md_final")

    # Phase 7: One last CD with high peak count
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        t7 = min(remaining * 0.50, 100.0)
        a_out, _ = coordinate_descent_peak(a_in, t7, n_peaks=50,
                                            rng_seed=2024, verbose=False)
        try_update(a_out, "cd_last")

    # Final validation
    if best_seq is not None:
        fs = safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        best_seq = make_shape(4000, 'ramp', 1.0)
        best_score = safe_eval(best_seq)
        print(f"\nFALLBACK: score={best_score:.6f}")

    return [float(x) for x in best_seq]
