def compute_grad_topk(a, c, k):

    n = len(a)

def parametric_init(n, ref=None):
    """Try diverse parametric families."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')

    candidates = []

    # Power laws (increasing and decreasing)
    for p in [0.3, 0.5, 0.7, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr + 1.0) ** p)

    # Exponential
    for rate in [0.5, 1, 1.5, 2, 3, 5, 8]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1.0)))

    # Gaussian centered
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.3, 0.4]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))

    # Tilted Gaussian
    for c in [0.4, 0.5, 0.6]:
        for sigma in [0.2, 0.3, 0.4]:
            for tilt in [0.5, 1.0, 2.0]:
                candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)) *
                                  np.exp(tilt * (x - 0.5)))

    # Sine-based
    for freq in [1, 2, 3]:
        candidates.append(np.maximum(np.sin(np.pi * freq * x) ** 2, 0.001))
        candidates.append(np.maximum(np.cos(np.pi * freq * x) ** 2, 0.001))

    # Beta-like (rise and fall)
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        for beta_p in [0.5, 1.0, 1.5, 2.0]:
            candidates.append(np.maximum(i_arr + 1.0, 0.5) ** alpha *
                              np.maximum(n - i_arr, 0.5) ** beta_p)

    # If reference available, try blends
    if ref is not None and len(ref) == n:
        ref_norm = norm_sum1(ref)
        for w in [0.3, 0.5, 0.7]:
            for p in [0.7, 1.0, 1.3]:
                candidates.append(w * ref_norm + (1.0 - w) * norm_sum1((i_arr + 1.0) ** p))

    for a_cand in candidates:
        a_norm = norm_sum1(a_cand)
        ac = fast_autocorr(a_norm)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_a = a_norm

    return best_a, best_val

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Determine starting length
    if parent is not None:
        n_start = len(parent)
    else:
        n_start = 5000
        best_seq = norm_sum1((np.arange(n_start, dtype=np.float64) + 1.0) ** 0.8)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_start}, score={best_score:.6f}")

    # === Phase 1: Parametric search ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 30.0)
    print(f"\n=== Phase 1: Parametric init (n={n_start}, t={t1:.0f}s) ===")
    if t1 > 5:
        a_param, _ = parametric_init(n_start, best_seq)
        check_and_update(a_param, "parametric")

    # === Phase 2: FISTA Top-K (primary) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.30, 450.0)
    print(f"\n=== Phase 2: FISTA Top-K (n={n_best}, t={t2:.0f}s) ===")
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_topk(a_in, t2, k_start=1, k_end=60, eta0=1.2,
                              restart_interval=4000, verbose=True)
        check_and_update(a_out, "fista_topk")

    # === Phase 3: FISTA Softmax (different dynamics) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 3: FISTA Softmax (n={n_best}, t={t3:.0f}s) ===")
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_softmax(a_in, t3, beta_start=8.0, beta_end=120.0,
                                 eta0=0.6, top_k_limit=150, verbose=True)
        check_and_update(a_out, "fista_softmax")

    # === Phase 4: LP-FW polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP-FW polish (n={n_best}, t={t4:.0f}s) ===")
    if t4 > 40 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, initial_K=300, max_K=2000, verbose=True)
        check_and_update(a_out, "lp_polish")

    # === Phase 5: FISTA Top-K after LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 5: FISTA after LP (n={n_best}, t={t5:.0f}s) ===")
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_topk(a_in, t5, k_start=5, k_end=100, eta0=0.5,
                              restart_interval=3000, verbose=False)
        check_and_update(a_out, "fista_after_lp")

    # === Phase 6: Multi-start ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-start (remaining={remaining:.0f}s) ===")
    if remaining > 60:
        n_starts = min(4, int(remaining // 60))
        for si in range(n_starts):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            t_ms = min(remaining * 0.25, 120.0)
            if t_ms < 20:
                break
            rng = np.random.default_rng(1000 + si * 7919)
            # Different perturbation structures
            freqs = np.fft.rfftfreq(n_best)
            noise = np.zeros(n_best)
            n_comp = 3 + si * 2
            for fi in range(1, min(n_comp, len(freqs))):
                amp = rng.uniform(0.005, 0.02)
                phase = rng.uniform(0, 2 * np.pi)
                noise += amp * np.cos(2 * np.pi * freqs[fi] * np.arange(n_best) + phase)
            a_pert = norm_sum1(best_seq + noise * best_seq)
            a_out, _ = fista_topk(a_pert, t_ms, k_start=1, k_end=50, eta0=1.0,
                                  restart_interval=2000, verbose=False)
            check_and_update(a_out, f"multistart_{si}")

    # === Phase 7: Final LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW (n={n_best}, remaining={remaining:.0f}s) ===")
    if remaining > 50 and n_best <= 8000:
        t7 = min(remaining * 0.4, 200.0)
        if t7 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, initial_K=400, max_K=2500, verbose=True)
            check_and_update(a_out, "lp_final")

    # === Phase 8: Final FISTA squeeze ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t8 = min(remaining * 0.5, 100.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = fista_topk(a_in, t8, k_start=10, k_end=150, eta0=0.3,
                                  restart_interval=2000, verbose=False)
            check_and_update(a_out, "fista_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.exp(-2.0 * x) + 0.1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
