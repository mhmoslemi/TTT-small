def fast_autocorr(a):
    """Compute full convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]

    except Exception:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def exact_subgradient(a, ac, top_k=20, beta=3.0):
    """Compute weighted subgradient using top-k lags with exact gradient formula."""
    n = len(a)
    n_ac = len(ac)
    k_max = min(top_k, n_ac)
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        # grad[i] += 2 * a[ki - i] for valid i where 0 <= ki-i < n
        # i ranges from max(0, ki-n+1) to min(ki, n-1)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        # Check for double counting when i = ki/2
        vals = 2.0 * a[j_idx]
        # If ki is even, the middle element should only be counted once
        if ki % 2 == 0:
            mid = ki // 2
            if i_start <= mid < i_end:
                vals[mid - i_start] = a[mid]
        grad[idx] += wi * vals
    return grad


def nesterov_subgradient(a0, time_limit, verbose=False, top_k=40, beta=4.0,
                          nesterov=True, rng_seed=0):
    """Nesterov-accelerated subgradient method with exact subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    y = a.copy()
    t_param = 1.0
    gamma = 3.0
    stall_count = 0
    max_stall = 5000
    restart_every = 4000
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_param += 1.0

        # Nesterov lookahead
        beta_t = (t_param - 1.0) / (t_param + 1.5)
        y = x + beta_t * (x - y) if it > 1 else x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute subgradient
        ac = fast_autocorr(y)
        M_curr = ac.max()
        g = exact_subgradient(y, ac, top_k=top_k, beta=beta)

        # Step size
        eta = gamma / (np.sqrt(t_param) + 1.0)

        # Mirror descent update (multiplicative weights)
        log_y = np.log(np.maximum(y, 1e-15)) - eta * g
        max_log = log_y.max()
        a_new = np.exp(log_y - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        # Check improvement
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        x = a_new

        # Adaptive parameters
        if stall_count > restart_every // 2:
            beta = max(beta * 0.95, 2.0)
            top_k = min(top_k + 3, 100)

        if stall_count > restart_every:
            stall_count = 0
            # Perturbation restart
            noise_scale = min(0.05, 0.01 * (it / restart_every))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            y = x_r.copy()
            top_k = min(top_k + 10, 120)
            beta = max(beta - 1.0, 2.0)

        if stall_count > max_stall:
            break

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NSG it={it} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}, top_k={top_k}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K_init=200, top_K_max=None):
    """LP-based Frank-Wolfe with adaptive K."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Line search
        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        for alpha in [2.0/(iter_count+2.0), min(1.0, 3.0/(iter_count+3.0)), 0.5, 0.3, 0.2, 0.15, 0.1]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1
        if iter_count % 8 == 0:
            K = min(K + 40, top_K_max)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def peeling_coordinate_descent(a0, time_limit, verbose=False, rng_seed=42):
    """Aggressive targeted coordinate descent: move mass from peak-contributing
    positions to low-contributing positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200
    round_count = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:15]

        # Compute sensitivity: how much each position contributes to peak lags
        sens = np.zeros(n)
        for k in top_k:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            # Contribution of a[i] to (conv)[k] is 2*a[j] (or a[i] if at center)
            contrib = 2.0 * a[j_idx]
            if ki % 2 == 0:
                mid = ki // 2
                if i_start <= mid < i_end:
                    contrib[mid - i_start] = a[mid]
            sens[idx] += contrib

        improved = False

        # For each peak lag, try to reduce the contributing pairs
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            contrib = a[idx] * a[j_idx]

            # Find the highest contributing pair
            pair_order = np.argsort(contrib)[::-1]
            for pi in pair_order[:5]:
                src_i = idx[pi]
                src_j = j_idx[pi]

                # Try moving mass from the larger of the pair
                if a[src_i] >= a[src_j]:
                    src = src_i
                else:
                    src = src_j

                if a[src] < 1e-12:
                    continue

                # Find destination: positions with low sensitivity
                low_sens = np.argsort(sens)[:max(5, n // 20)]
                candidates = [int(c) for c in low_sens if c != src and a[c] < np.median(a)]

                if not candidates:
                    candidates = [int(c) for c in np.argsort(sens)[:20] if c != src]

                found = False
                for dst in candidates[:8]:
                    for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            found = True
                            break
                    if found:
                        break
            if time.monotonic() - t0 > time_limit:
                break

        # Random exploration when stalled
        if not improved:
            stall += 1
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.002, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        # Periodic larger perturbation
        if stall > 50 and stall % 50 == 0:
            noise = rng.normal(0, 0.01, n) * best_a
            a_p = np.maximum(best_a + noise, 1e-10)
            a_p = a_p / a_p.sum()
            M_p = fast_autocorr(a_p).max()
            if M_p < best_m:
                a = a_p
                best_a = a.copy()
                stall = 0

        round_count += 1
        if verbose and round_count % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Peel round={round_count} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def block_coordinate_descent(a0, time_limit, verbose=False, rng_seed=99, block_size=3):
    """Block coordinate descent: move mass between blocks of positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 100

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:10]

        # Identify "hot" positions (high contribution to peak lags)
        hot = np.zeros(n)
        for k in top_lags:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            hot[idx] += a[idx] * a[j_idx]

        hot_threshold = np.percentile(hot, 70)
        hot_positions = set(np.where(hot > hot_threshold)[0].tolist())
        cold_positions = set(np.where(hot <= hot_threshold)[0].tolist())

        # Try block moves: move mass from a block of hot to a block of cold
        n_hot = list(hot_positions)
        n_cold = list(cold_positions)

        if not n_hot or not n_cold:
            stall += 1
            continue

        improved = False
        n_attempts = min(50, n // 10)

        for _ in range(n_attempts):
            if time.monotonic() - t0 > time_limit:
                break

            # Select a random hot position and a cold position
            src = int(rng.choice(n_hot))
            dst = int(rng.choice(n_cold))

            if a[src] < 1e-12:
                continue

            for frac in [0.005, 0.01, 0.02, 0.04, 0.06, 0.1]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            # Larger random perturbation
            if stall % 20 == 0:
                mask = rng.random(n) < 0.1
                noise = rng.normal(0, 0.02, n)
                a_p = a + noise * mask
                a_p = np.maximum(a_p, 1e-10)
                a_p = a_p / a_p.sum()
                M_p = fast_autocorr(a_p).max()
                if M_p < best_m:
                    a = a_p
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def smooth_iterative_shaping(a0, time_limit, verbose=False, rng_seed=100):
    """Iterative spectral shaping with non-negativity enforcement."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    fft_size = 1 << (2 * n - 2).bit_length()
    stall = 0
    max_stall = 300
    it = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        it += 1

        # Frequency domain: push toward flat spectrum
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: flat magnitude
        mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0

        # Adaptive blending: more aggressive early, less later
        progress = min(1.0, it / 5000)
        blend = 0.15 + 0.5 * progress

        new_mag = (1 - blend) * mag + blend * mag_mean
        # Slightly boost low frequencies to help with non-negativity
        freqs = np.arange(len(mag))
        low_boost = 1.0 + 0.1 * np.exp(-freqs / (n // 4))
        new_mag = new_mag * low_boost

        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1

        if stall > 100:
            # Perturb and restart
            noise = rng.normal(0, 0.005, n) * best_a
            a_p = np.maximum(best_a + noise, 1e-10)
            a = a_p / a_p.sum()
            stall = 50

    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.1) / width) ** 2) + np.exp(-((x - 0.9) / width) ** 2) + 0.005
    return norm_sum1(a)


def make_cosine(n, alpha=0.5):
    x = np.arange(n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * x / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_power_edge(n, p=2.0):
    """Power-law edge weighting: a[i] proportional to (dist_to_center)^p"""
    x = np.linspace(-1, 1, n)
    a = np.abs(x) ** p + 0.1
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator: Nesterov subgradient + LP-FW + targeted coordinate descent."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        best_seq = make_u_shape(n_base, sharpness=2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Nesterov Subgradient on parent ===
    print(f"\n=== Phase 1: Nesterov Subgradient (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 200.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_subgradient(a_in, t1, verbose=True, top_k=40, beta=4.0,
                                         nesterov=True, rng_seed=0)
        try_update(a_out, "nsg_main")

    # === Phase 2: LP-Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-Frank-Wolfe (n={n_best}) ===")
    t2 = min(remaining * 0.2, 300.0)
    if t2 > 50 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 1500)
        top_K_init = min(250, 2 * n_best - 1)
        top_K_max = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True,
                                   top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp_main")

    # === Phase 3: Peeling Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Peeling Coord Descent (n={n_best}) ===")
    t3 = min(remaining * 0.2, 280.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = peeling_coordinate_descent(a_in, t3, verbose=True, rng_seed=42)
        try_update(a_out, "peel_1")

    # === Phase 4: Multi-restart Nesterov with diverse inits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Multi-restart (n={n_best}) ===")
    t4 = min(remaining * 0.2, 250.0)
    if t4 > 60:
        inits = []
        labels = []

        inits.append(norm_sum1(best_seq))
        labels.append("best")

        # Perturbed versions of best
        rng_local = np.random.default_rng(777)
        for pi in range(3):
            noise = rng_local.normal(0, 0.02 * (pi + 1), n_best) * norm_sum1(best_seq)
            a_p = np.maximum(norm_sum1(best_seq) + noise, 1e-10)
            a_p = a_p / a_p.sum()
            inits.append(a_p)
            labels.append(f"pert{pi}")

        # Different shapes
        inits.append(make_u_shape(n_best, 1.8))
        labels.append("u18")
        inits.append(make_u_shape(n_best, 3.5))
        labels.append("u35")
        inits.append(make_power_edge(n_best, 1.5))
        labels.append("pow15")
        inits.append(make_power_edge(n_best, 3.0))
        labels.append("pow30")
        inits.append(make_bimodal(n_best, 0.12))
        labels.append("bimod")
        inits.append(make_cosine(n_best, 0.4))
        labels.append("cos")

        # Blend of best with U-shape
        blend = 0.6 * norm_sum1(best_seq) + 0.4 * make_u_shape(n_best, 2.0)
        inits.append(norm_sum1(blend))
        labels.append("blend")

        t_per = max(10.0, t4 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t_avail = min(t_per, remaining - 60)
            if t_avail < 8:
                break
            print(f"  Restart '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = nesterov_subgradient(init, t_avail, verbose=False,
                                             top_k=50, beta=4.0, rng_seed=100 + idx)
            try_update(a_out, f"restart_{lbl}")

    # === Phase 5: Block Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Block Coord Descent (n={n_best}) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = block_coordinate_descent(a_in, t5, verbose=True, rng_seed=99)
        try_update(a_out, "block")

    # === Phase 6: Spectral Shaping ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Spectral Shaping (n={n_best}) ===")
    t6 = min(remaining * 0.1, 120.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_iterative_shaping(a_in, t6, verbose=True, rng_seed=200)
        try_update(a_out, "spectral")

    # === Phase 7: Second LP-FW pass ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: LP-FW Round 2 (n={n_best}) ===")
    t7 = min(remaining * 0.1, 130.0)
    if t7 > 30 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(400, 2 * n_best - 1)
        top_K_max = min(5000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=True,
                                   top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp_r2")

    # === Phase 8: Final Peeling + Nesterov on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Polish (n={n_best}) ===")
    if remaining > 40:
        t8a = min(remaining * 0.35, 100.0)
        if t8a > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = peeling_coordinate_descent(a_in, t8a, verbose=False, rng_seed=555)
            try_update(a_out, "peel_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t8b = min(remaining * 0.5, 100.0)
            if t8b > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = nesterov_subgradient(a_in, t8b, verbose=False,
                                                 top_k=60, beta=5.0, rng_seed=888)
                try_update(a_out, "nsg_final")

    # === Phase 9: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Length Exploration (n={n_best}) ===")

    trial_lengths = []
    for delta in [300, -300, 600, -600, 1000, -1000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 9000 and new_n != n_best:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))

    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t9a = min(remaining * 0.35, 80.0)
        if t9a > 15:
            a_out, _ = nesterov_subgradient(a_long, t9a, verbose=False,
                                             top_k=50, beta=4.0, rng_seed=new_n % 500)
            try_update(a_out, f"nsg_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 6000:
            t9b = min(remaining * 0.4, 60.0)
            if t9b > 15:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                top_K = min(2000, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe(a_in, t9b, solver, verbose=False,
                                           top_K_init=min(300, 2 * new_n - 1),
                                           top_K_max=top_K)
                try_update(a_out, f"lp_n{new_n}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        best_seq = make_u_shape(n_fb, 2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
