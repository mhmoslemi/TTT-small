def lp_fw_step(a, M, active_idx, solver):
    """Single LP-FW step: find best direction g."""
    n = len(a)
    K = len(active_idx)
    
    # Build constraint matrix: T[k,i] = a[i + k] or a[k - i] or 0
    T = np.zeros((K, n))
    for row, k in enumerate(active_idx):
        ki = int(k)
        # (a*g)[k] = sum_i a[i] * g[k-i]
        # For g[j] to appear, we need j = k-i, so i = k-j
        # (a*g)[k] = sum_j a[k-j] * g[j] where k-j is in [0,n-1] and j is in [0,n-1]
        # So j ranges from max(0, k-(n-1)) to min(n-1, k)

        j_hi = min(n - 1, ki)

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        # d/d(a[i]) [(a*a)[k]] = a[k-i] + a[k-i] for the two terms... 
        # Actually (a*a)[k] = sum_j a[j]*a[k-j], so d/d(a[i]) = a[k-i] + a[k-i] if i != k-i, else 2*a[i]
        # More precisely: d/d(a[i]) sum_j a[j]*a[k-j] = a[k-i] (from j=i term) + a[k-i] (from k-j=i, j=k-i term)
        # = 2*a[k-i] if i != k-i, and 2*a[i] if i == k-i (which is same thing)
        # So gradient w.r.t. a[i] of (a*a)[k] = 2*a[k-i] if k-i is in [0,n-1], else 0
        # Wait, that's not right either. Let me think more carefully.
        # (a*a)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j] * a[k-j]
        # d/d(a[i]) (a*a)[k] = a[k-i] (if 0 <= k-i <= n-1) * [i is in valid range for j]
        #                       + a[i] * [k-i is in valid range for the other factor]
        # Actually since (a*a)[k] = sum_j a[j]*a[k-j], the derivative w.r.t. a[i] is:
        # = a[k-i] (when j=i, we need 0<=k-i<=n-1) + a[i] (when k-j=i, i.e., j=k-i, we need 0<=k-i<=n-1)
        # But these are the same term when i = k-i, i.e., i = k/2.
        # So in general: d/d(a[i]) (a*a)[k] = a[k-i] if 0<=k-i<n, and ALSO a[k-i] if 0<=k-i<n
        # Wait no. (a*a)[k] = sum_j a[j]*a[k-j]. The term involving a[i] appears when:
        # - j = i: contributes a[i]*a[k-i] (valid if 0<=k-i<n)
        # - k-j = i, i.e., j = k-i: contributes a[k-i]*a[i] (valid if 0<=k-i<n)
        # So total contribution: 2*a[k-i] if 0<=k-i<n and i != k-i
        # Or a[i] if i = k-i (i.e., k is even and i=k/2)
        # Actually in both cases the derivative is:
        # If 0 <= k-i < n: derivative = a[k-i] + a[k-i] = 2*a[k-i] (counting both j=i and j=k-i terms)
        # Unless i = k-i (k=2i), in which case it's just a[i] (one term)
        # Hmm, but in the sum, when j=i we get a[i]*a[k-i], and when j=k-i we get a[k-i]*a[i].
        # If i != k-i, these are two separate terms, each contributing a[k-i] to d/d(a[i]).
        # If i = k-i, it's a single term a[i]*a[i], contributing 2*a[i] to d/d(a[i]).
        # So in all cases: d/d(a[i]) (a*a)[k] = 2*a[k-i] if 0<=k-i<n (with the understanding that when i=k-i, a[k-i]=a[i]).

        # Efficient computation:
        # grad[i] = sum_k p[k] * 2 * a[k-i] (where 0 <= k-i < n)
        # = 2 * sum_k p[k] * a[k-i] for valid k
        # This is 2 * (p * a)[i] in the cross-correlation sense!
        # Where (p * a)[i] = sum_k p[k] * a[k-i]... hmm, let me be careful with indexing.
        # grad[i] = 2 * sum_{k: 0<=k-i<n} p[k] * a[k-i] = 2 * sum_{j: 0<=j<n} p[i+j] * a[j] (substituting j=k-i)
        # = 2 * sum_j a[j] * p[i+j]
        # This is the cross-correlation of a and p!

        from scipy.signal import correlate as sp_correlate
        # cross[i] = sum_j a[j] * p[i+j] for valid i, j
        # np.correlate(p, a, 'full') gives sum_j p[j] * a[j+i]... let me use the right function.
        # Actually: cross_corr[i] = sum_j a[j] * p[j+i]
        # This is correlate(p, a, mode='valid') if len(p) >= len(a), but p has length 2n-1 and a has length n.
        # correlate(p, a, 'valid')[i] = sum_j p[i+j] * a[j] for j=0..n-1
        # That's what we want!

        cross = sp_correlate(p, a, mode='valid')  # length n, cross[i] = sum_j p[i+j]*a[j]
        grad = 2.0 * cross

        # Project gradient to simplex tangent: remove mean
        grad_tan = grad - grad.mean()

        # Mirror descent step
        a_trial = a - eta * grad_tan
        a_new = project_simplex(a_trial)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.01, 1.0)
            a = a_new
        else:
            stall += 1
            if stall > 50:
                eta *= 0.8
            if stall > 100:
                # Perturb
                pert = rng.normal(0, 0.005, n) * best_a
                a_p = np.maximum(best_a + pert, 1e-10)
                a = a_p / a_p.sum()
                stall = 0
                eta = 0.3
            else:
                a = a_new

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Smooth it={it} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum (random phases)."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    A = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
    A_full = np.zeros(fft_size, dtype=np.complex128)
    A_full[:fft_size // 2 + 1] = A
    for i in range(1, fft_size // 2):
        A_full[fft_size - i] = np.conj(A[i])
    a = np.fft.irfft(A_full, fft_size)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_superposition_init(n, seed=0):
    """Initialize as superposition of sinusoids with random frequencies."""
    rng = np.random.default_rng(seed)
    x = np.arange(n) / n
    a = np.ones(n) * 0.5
    n_modes = 15
    for f in range(1, n_modes + 1):
        freq = rng.uniform(f * 0.5, f * 1.5)
        amp = rng.uniform(0.05, 0.15) / np.sqrt(f)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """Main orchestrator: LP-FW with adaptive K, restarts, multi-length, and polish."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        best_seq = make_flat_spectrum_init(n_base, seed=7)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: LP-FW on parent (main optimization) ===
    print(f"\n=== Phase 1: LP-FW on parent (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.35, 550.0)

    if t_phase1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_fw_optimize(a_in, t_phase1, solver, verbose=True,
                                   K_start=min(300, 2*n0-1), K_max=min(4000, 2*n0-1),
                                   K_growth=30, max_stall=35, seed=42)
        try_update(a_out, "lp_phase1")

    # === Phase 2: LP-FW with different K schedule ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW alt schedule (n={len(best_seq)}) ===")
    t_phase2 = min(remaining * 0.25, 350.0)

    if t_phase2 > 60:
        a_in = norm_sum1(best_seq)
        # Start with larger K, grow more slowly
        a_out, _ = lp_fw_optimize(a_in, t_phase2, solver, verbose=False,
                                   K_start=min(800, 2*len(best_seq)-1),
                                   K_max=min(6000, 2*len(best_seq)-1),
                                   K_growth=15, max_stall=50, seed=123)
        try_update(a_out, "lp_phase2")

    # === Phase 3: Smoothed gradient optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Smoothed gradient (n={len(best_seq)}) ===")
    t_phase3 = min(remaining * 0.15, 200.0)

    if t_phase3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_subgradient_optimize(a_in, t_phase3, verbose=True, seed=200)
        try_update(a_out, "smooth_grad")

    # === Phase 4: LP-FW round 3 on current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP-FW round 3 (n={len(best_seq)}) ===")
    t_phase4 = min(remaining * 0.12, 180.0)

    if t_phase4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_fw_optimize(a_in, t_phase4, solver, verbose=False,
                                   K_start=min(1500, 2*len(best_seq)-1),
                                   K_max=min(7000, 2*len(best_seq)-1),
                                   K_growth=20, max_stall=45, seed=456)
        try_update(a_out, "lp_phase4")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length exploration (base n={n_best}) ===")

    trial_lengths = []
    for delta in [-500, 500, -1000, 1000, -2000, 2000, -3000, 3000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)

        t_trial = min(remaining * 0.35, 200.0)
        if t_trial > 40:
            a_out, _ = lp_fw_optimize(a_interp, t_trial, solver, verbose=False,
                                       K_start=min(300, 2*new_n-1),
                                       K_max=min(3000, 2*new_n-1),
                                       K_growth=25, max_stall=35, seed=new_n % 997)
            try_update(a_out, f"lp_n{new_n}")

    # === Phase 6: Alternative initialization LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Alt init (n={n_best}) ===")
    if remaining > 100:
        # Try flat spectrum init
        a_flat = make_flat_spectrum_init(n_best, seed=777)
        t6a = min(remaining * 0.3, 150.0)
        if t6a > 30:
            a_out, _ = lp_fw_optimize(a_flat, t6a, solver, verbose=False,
                                       K_start=min(200, 2*n_best-1),
                                       K_max=min(3000, 2*n_best-1),
                                       K_growth=25, max_stall=30, seed=888)
            try_update(a_out, "lp_flat_init")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            a_super = make_superposition_init(n_best, seed=999)
            t6b = min(remaining * 0.4, 120.0)
            if t6b > 20:
                a_out, _ = lp_fw_optimize(a_super, t6b, solver, verbose=False,
                                           K_start=min(200, 2*n_best-1),
                                           K_max=min(3000, 2*n_best-1),
                                           K_growth=20, max_stall=30, seed=1000)
                try_update(a_out, "lp_super_init")

    # === Phase 7: Final LP-FW on absolute best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW (n={n_best}) ===")
    t_phase7 = min(remaining * 0.25, 250.0)

    if t_phase7 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_fw_optimize(a_in, t_phase7, solver, verbose=True,
                                   K_start=min(1000, 2*n_best-1),
                                   K_max=min(8000, 2*n_best-1),
                                   K_growth=20, max_stall=40, seed=2024)
        try_update(a_out, "lp_final")

    # === Phase 8: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Coordinate polish (n={n_best}) ===")
    t_phase8 = min(remaining * 0.7, 120.0)

    if t_phase8 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t_phase8, verbose=False, seed=31415)
        try_update(a_out, "coord_final")

    # === Phase 9: One more LP-FW pass if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 9: Final LP-FW pass (n={n_best}) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_fw_optimize(a_in, remaining * 0.8, solver, verbose=False,
                                   K_start=min(2000, 2*n_best-1),
                                   K_max=min(10000, 2*n_best-1),
                                   K_growth=30, max_stall=30, seed=99999)
        try_update(a_out, "lp_last")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        best_seq = make_flat_spectrum_init(n_fb, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
