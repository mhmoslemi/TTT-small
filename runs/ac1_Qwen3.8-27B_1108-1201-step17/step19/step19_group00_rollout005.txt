import numpy as np

def fast_convolution(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(a, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    a = np.asarray(a, dtype=np.float64)
    # If already satisfied
    if a.min() >= 0 and abs(a.sum() - target_sum) < 1e-12:
        return a.copy()
    
    # Standard simplex projection
    n = len(a)
    # We want to find μ such that sum(max(a - μ, 0)) = target_sum
    # Use binary search on μ
    lo = a.min() - 1e6
    hi = a.max() + 1e6
    
    # Actually, the standard algorithm:
    # Sort a in descending order, find the threshold
    sorted_a = np.sort(a)[::-1]
    cumulative = np.cumsum(sorted_a)
    # Find largest k such that (cumulative[k-1] - (k-1)*a_k) / k > a_k... 
    # Simpler: use the O(n) algorithm
    
    # Binary search approach (robust):
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.maximum(a - mid, 0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2
    result = np.maximum(a - mu, 0)
    # Fix rounding
    diff = target_sum - result.sum()
    if abs(diff) > 1e-14:
        # Distribute to non-zero elements
        nz = np.where(result > 0)[0]
        if len(nz) > 0:
            result[nz] += diff / len(nz)
            result = np.maximum(result, 0)
            s2 = result.sum()
            result = result * (target_sum / s2)
    return result

    n = len(a)

    s = a.sum()
    if s < 1e-15:

    c = fast_convolution(a)
    return 2.0 * n * c.max() / (s * s)

def subgradient_descent(a0, time_limit, top_k=50, temp=30.0, verbose=True, seed=42):

def fast_convolution(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / max(len(a), 1)
    return a / s

def project_simplex(a, target_sum=1.0):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return a
    lo = a.min() - 1.0
    hi = a.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.maximum(a - mid, 0.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    result = np.maximum(a - mu, 0.0)
    ss = result.sum()
    if ss > 1e-15:
        result = result * (target_sum / ss)
    else:
        result = np.full(n, target_sum / n)
    return result

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)

def lp_fw(a0, time_limit, solver, init_K=400, max_K=3000, add_per=40, verbose=True):
    """LP-based Frank-Wolfe with adaptive constraint set."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_convolution(y).max()
    t0 = time.monotonic()
    
    ac_y = fast_convolution(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:init_K])
    no_improve = 0
    stall_limit = 2000
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break
        
        ac = fast_convolution(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        clist = sorted(active_set)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = sorted(active_set)
        
        K = len(clist)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs = np.zeros((K, n), dtype=np.float64)
        for jj, k in enumerate(clist):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        lp_ok = False
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                lp_ok = True
        except Exception:
            if len(active_set) > init_K:
                active_set = set(int(clist[i] for i in range(init_K)))
                clist = sorted(active_set)
        
        if not lp_ok:
            if no_improve > stall_limit // 2:
                break
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violated constraints

        Gf = np.fft.rfft(g_dir, fft_size)

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per]]
            for k in top_v:
                active_set.add(int(k))
        
        # Line search (ternary + grid)
        alphas = [1.0 / (t + 1), 2.0 / (t + 2), 0.5, 0.8, 1.0]
        alphas = sorted(set(max(0.01, min(1.0, a)) for a in alphas))
        best_alpha = 2.0 / (t + 2.0)

            if ts < 1e-15:

            trial = trial / ts
            Mt = fast_convolution(trial).max()
            if Mt < best_trial - 1e-15:
                best_trial = Mt
                best_alpha = alpha
        
        # Ternary refinement
        lo_a = max(0.0, best_alpha - 0.2)

            m1 = lo_a + (hi_a - lo_a) / 3.0

            t1 = np.maximum((1 - m1) * y + m1 * g_dir, 0.0)

            if s1 > 1e-15:

            m1v = fast_convolution(t1).max() if s1 > 1e-15 else 1e10

            if m1v < m2v:

            else:
                lo_a = m1

                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t / st
                Mt = fast_convolution(a_t).max()
                if Mt < best_m - 1e-15:
                    a = a_t
                    best_m = Mt
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            
            if not found and stall > 100:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_convolution(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    Mn = fast_convolution(a).max()
                    if Mn < best_m:
                        best_m = Mn
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 200 == 0 and stall > 0:
            print("    Coord ({}s) J={} stall={}".format(round(time.monotonic() - t0, 1), repr(2 * n * best_m), stall))
    
    return best_a, 2.0 * n * best_m

def get_solver():
    global _SOLVER_CACHE
    try:
        _SOLVER_CACHE
        return _SOLVER_CACHE
    except NameError:
        pass
    
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(15, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _SOLVER_CACHE = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _SOLVER_CACHE = cp.ECOS
    print("  Solver fallback: ECOS")
    return _SOLVER_CACHE

def parametric_init(n, reference_seq):
    """Generate diverse parametric initializations."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []
    
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append(normalize((i_arr + 1.0) ** p))
        candidates.append(normalize((n - i_arr) ** p))
        candidates.append(normalize(x ** p + (1.0 - x) ** p + 1e-10))
    
    # Gaussian bumps
    for c in [0.3, 0.5, 0.7]:
        for sg in [0.08, 0.12, 0.2, 0.3]:
            candidates.append(normalize(np.exp(-((x - c) ** 2) / (2 * sg * sg))))
    
    # Double bumps
    for c1 in [0.25, 0.35, 0.4]:
        for c2 in [0.6, 0.65, 0.75]:
            for sg in [0.06, 0.1, 0.15]:
                cands = np.exp(-((x - c1) ** 2) / (2 * sg * sg)) + np.exp(-((x - c2) ** 2) / (2 * sg * sg))
                candidates.append(normalize(cands))
    
    # Cosine modulations
    for f in [1, 2, 3, 5, 7]:
        cands = np.maximum(1.0 + 0.8 * np.cos(2 * np.pi * f * x), 0.01)
        candidates.append(normalize(cands))
    
    # Blend with reference
    if reference_seq is not None:
        ref = normalize(interpolate_seq(reference_seq, n))
        for w in [0.2, 0.4, 0.6, 0.8]:
            for cand in candidates[:8]:
                candidates.append(normalize(w * ref + (1.0 - w) * cand))
    
    # Find best
    best_a = candidates[0]
    best_val = obj_val(best_a)
    for c in candidates[1:]:
        v = obj_val(c)
        if v < best_val:
            best_val = v
            best_a = c
    
    return best_a, best_val

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Budget: {}s".format(time_budget))
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            print("Parent: n={} score={}".format(len(p), repr(best_score)))
    except Exception:
        best_seq = None
    
    if best_seq is None:
        n_init = 8000
        x = np.linspace(0, 1, n_init)
        best_seq = normalize(1.0 + 0.5 * np.cos(2 * np.pi * x))
        best_score = eval_safe(best_seq)
        print("Default init n={} score={}".format(n_init, repr(best_score)))
    
    n0 = len(best_seq)
    
    # Phase 1: Parametric init at same length
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Parametric (n={}) ===".format(n0))
    t1 = min(remaining * 0.04, 40.0)
    if t1 > 5:
        t1s = time.monotonic()
        a_p, v_p = parametric_init(n0, best_seq)
        sc = eval_safe(a_p)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_p
            print("  Param improved: {}".format(repr(best_score)))
        print("  P1 done ({}s) best={}".format(round(time.monotonic() - t1s, 1), repr(best_score)))
    
    # Phase 2: SoftMax descent (fast, many iterations)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P2: SoftMax (n={}) ===".format(n_cur))
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 30:
        t2s = time.monotonic()
        a_out, v_out = softmax_descent(normalize(best_seq), t2, top_k=60, T_init=20.0, T_max=200.0, c0=1.5, verbose=True, seed=101)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  SoftMax improved: {}".format(repr(best_score)))
        print("  P2 done ({}s) best={}".format(round(time.monotonic() - t2s, 1), repr(best_score)))
    
    # Phase 3: LP-FW refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P3: LP-FW (n={}) ===".format(n_cur))
    t3 = min(remaining * 0.22, 300.0)
    if t3 > 30:
        t3s = time.monotonic()
        a_out, v_out = lp_fw(normalize(best_seq), t3, solver, init_K=500, max_K=3500, add_per=50, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  LP-FW improved: {}".format(repr(best_score)))
        print("  P3 done ({}s) best={}".format(round(time.monotonic() - t3s, 1), repr(best_score)))
    
    # Phase 4: SoftMax at longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = min(n_cur + 3000, 15000)
    if n_long != n_cur:
        print("\n=== P4: SoftMax n={} ===".format(n_long))
        t4 = min(remaining * 0.15, 200.0)
        if t4 > 30:
            t4s = time.monotonic()
            a_long = interpolate_seq(best_seq, n_long)
            a_out, v_out = softmax_descent(a_long, t4, top_k=80, T_init=30.0, T_max=250.0, c0=1.2, verbose=True, seed=202)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  Long SoftMax improved: {}".format(repr(best_score)))
            print("  P4 done ({}s) best={}".format(round(time.monotonic() - t4s, 1), repr(best_score)))
    
    # Phase 5: LP-FW continue (on best)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P5: LP-FW2 (n={}) ===".format(n_cur))
    t5 = min(remaining * 0.18, 250.0)
    if t5 > 30 and n_cur <= 15000:
        t5s = time.monotonic()
        a_out, v_out = lp_fw(normalize(best_seq), t5, solver, init_K=600, max_K=4000, add_per=60, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  LP-FW2 improved: {}".format(repr(best_score)))
        print("  P5 done ({}s) best={}".format(round(time.monotonic() - t5s, 1), repr(best_score)))
    
    # Phase 6: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P6: Coord (n={}) ===".format(n_cur))
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 20:
        t6s = time.monotonic()
        a_out, v_out = coord_polish(normalize(best_seq), t6, seed=42, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  Coord improved: {}".format(repr(best_score)))
        print("  P6 done ({}s) best={}".format(round(time.monotonic() - t6s, 1), repr(best_score)))
    
    # Phase 7: Final SoftMax + LP interleave
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P7: Final (n={}) ===".format(n_cur))
    t7 = min(remaining * 0.15, 200.0)
    if t7 > 20:
        # Short softmax burst
        t7a = t7 * 0.5
        a_out, v_out = softmax_descent(normalize(best_seq), t7a, top_k=80, T_init=50.0, T_max=400.0, c0=0.8, verbose=False, seed=303)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
        
        # Short LP burst
        t7b = t7 - (time.monotonic() - t_global + t7 - t7)  # remaining in phase
        t7b = min(remaining - (time.monotonic() - t_global), t7 * 0.5)
        if t7b > 15 and n_cur <= 15000:
            a_out, v_out = lp_fw(normalize(best_seq), t7b, solver, init_K=800, max_K=5000, add_per=80, verbose=False)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
        print("  P7 done best={}".format(repr(best_score)))
    
    # Phase 8: Final coord polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 15:
        print("\n=== P8: Final Coord (n={}) ===".format(n_cur))
        a_out, v_out = coord_polish(normalize(best_seq), remaining * 0.8, seed=99, verbose=False)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
        print("  P8 done best={}".format(repr(best_score)))
    
    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={} n={} t={}s".format(repr(best_score), len(best_seq), round(elapsed, 1)))
    else:
        print("\nERROR fallback")
        n_fb = 7000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(1.0 + 0.5 * np.cos(2 * np.pi * x))
        best_score = eval_safe(best_seq)
    
    return [float(x) for x in best_seq]
