def compute_gradient(a, ac, top_k, beta):
    """Compute subgradient of max(c) using top-K lags."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def nesterov_mirror_descent(a0, time_limit, top_k_init=20, top_k_final=300, 
                             beta=4.0, c0=3.0, restart_every=5000,
                             rng_seed=42, verbose=False):
    """Nesterov-accelerated mirror descent on the simplex."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_hist = a.copy()
    gamma = c0
    stall_count = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        momentum = (t - 1.0) / (t + 1.5)
        y = x + momentum * (x - x_hist)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute gradient
        progress = min(1.0, t / (restart_every * 3))
        cur_k = int(top_k_init + (top_k_final - top_k_init) * progress)
        ac = fast_autocorr(y)
        g = compute_gradient(y, ac, cur_k, beta)
        
        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_hist = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Adaptive restart
        if stall_count > restart_every:
            stall_count = 0
            pert_scale = 0.02 / (1 + t / 10000)
            pert = rng.normal(0, pert_scale, n) * best_a
            x_r = np.maximum(best_a + pert, 1e-10)
            x = x_r / x_r.sum()
            x_hist = x.copy()
            gamma = max(gamma * 0.9, 1.0)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.0f}s): J={J:.6f}, K={cur_k}, stall={stall_count}")
    
    return best_a, best_m

def lp_frank_wolfe_v2(a0, time_limit, solver, top_K_init=200, top_K_max=3000, 
                       verbose=False, rng_seed=0):
    """Improved LP Frank-Wolfe with progressive K and better line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    K = top_K_init
    iter_count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        K_sel = min(K, 2 * n - 1)
        if K_sel < 2 * n - 1:
            binding_idx = np.argsort(ac)[::-1][:K_sel]
        else:
            binding_idx = np.arange(2 * n - 1)
        
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 1
            if no_improve > 5:
                break
            K = min(K + 50, top_K_max)
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            no_improve += 1
            if no_improve > 5:
                break
            continue
        b_val = b_val / b_sum
        
        # Line search: try multiple alpha values
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = M_curr
        alphas = [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0), 
                  0.5, 0.3, 0.2, 0.1, 0.7, 0.8]
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * b_val
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha
        
        a = (1.0 - best_alpha) * a + best_alpha * b_val
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a
        
        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        iter_count += 1
        
        # Progressive K increase
        if iter_count % 5 == 0:
            K = min(K + 20, top_K_max)
        
        # Reset if stuck
        if no_improve > 15:
            no_improve = 0
            K = min(K + 100, top_K_max)
            # Small perturbation
            rng = np.random.default_rng(iter_count)
            pert = rng.normal(0, 0.005, n) * a
            a = np.maximum(a + pert, 0.0)
            a = a / a.sum()
        
        if verbose and iter_count % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW v2 iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")
    
    return best_a, best_m

def spectral_shaping(a0, time_limit, verbose=False, rng_seed=0):
    """Spectral domain optimization: shape |A(w)|^2 to reduce time-domain peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    fft_size = 1 << (2 * n - 2).bit_length()
    stall = 0
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: reduce concentration in frequency domain
        # This should make time-domain autocorrelation flatter
        # Strategy: reduce the variance of |A|^2
        mag_sq = mag ** 2
        mag_sq_mean = np.mean(mag_sq)
        
        # Push toward flat magnitude (reduces peak in time domain autocorrelation)
        # But be gentle to maintain non-negativity
        alpha = 0.05 + 0.1 * (it / min(it + 1, 100))
        new_mag_sq = (1 - alpha) * mag_sq + alpha * mag_sq_mean
        new_mag = np.sqrt(np.maximum(new_mag_sq, 0))
        
        A_new = new_mag * np.exp(1j * phase)
        
        # Ensure Hermitian symmetry
        A_full = np.zeros(fft_size, dtype=np.complex128)
        A_full[:fft_size // 2 + 1] = A_new
        for i in range(1, fft_size // 2):
            A_full[fft_size - i] = np.conj(A_new[i])
        
        a_new = np.fft.irfft(A_full, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            stall += 1
        else:
            a_new = a_new / s
            m_new = fast_autocorr(a_new).max()
            if m_new < best_m - 1e-15:
                best_m = m_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        
        if stall > 200:
            # Reset with perturbation
            pert = rng.normal(0, 0.01, n) * best_a
            a = np.maximum(best_a + pert, 1e-10)
            a = a / a.sum()
            stall = 0
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral it={it} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def coordinate_polish_v2(a0, time_limit, verbose=False, rng_seed=0):
    """Improved coordinate descent: move mass from high-contribution to low-contribution positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    n_top = 30
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]
        
        # Compute sensitivity: how much each coordinate contributes to top peaks
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:15]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]
        
        improved = False
        
        # Try moving mass from high-sensitivity to low-sensitivity positions
        src_candidates = np.argsort(sens)[::-1][:20]
        dst_candidates = np.argsort(sens)[:20]
        
        for src in src_candidates:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            
            for dst in dst_candidates:
                if dst == src:
                    continue
                
                # Try different fractions
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                a[i] -= a[i] * frac
                a[j] += a[i] * frac / (1 - frac)  # approximately
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord v2 ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)

def make_bimodal(n, width=0.15, pos1=0.15, pos2=0.85):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) / width) ** 2) + np.exp(-((x - pos2) / width) ** 2) + 0.001
    return norm_sum1(a)

def make_cosine_mod(n, alpha=0.5):
    x = np.arange(n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * x / n)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def make_power_law(n, p=2.0):
    x = np.linspace(-1, 1, n)
    a = np.abs(x) ** p + 0.01
    return norm_sum1(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
