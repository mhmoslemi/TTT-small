        j_range = np.arange(lo, hi + 1)
        a_shifted[j_range] = a[k_star - j_range]

    grad_x = 4.0 * n * a * (a_shifted - M)
    return J, grad_x


def obj_grad_multi_peak(x, n, n_peaks=10, temperature=0.0):
    """
    Objective and gradient in log-space, using multi-peak soft subgradient.
    If temperature > 0, uses softmax-tempered max for smoothness.
    """
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()

    b = fast_conv(a)

    if temperature > 0.0:
        # Soft max with temperature
        b_scaled = b / temperature
        b_max = b_scaled.max()
        exp_b = np.exp(b_scaled - b_max)
        weights = exp_b / exp_b.sum()
        M = np.dot(weights, b)
        # Use top peaks for gradient
        top_idx = np.where(weights > 1e-10)[0]
        top_idx = top_idx[np.argsort(weights[top_idx])[::-1]][:min(n_peaks, len(top_idx))]
        top_w = weights[top_idx]
        top_w = top_w / top_w.sum()
    else:
        top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        top_vals = b[top_idx]
        # Weight by proximity to max
        max_v = top_vals.max()
        min_v = top_vals.min()
        if max_v - min_v < 1e-15:
            top_w = np.ones(n_peaks) / n_peaks
        else:
            t = (top_vals - min_v) / (max_v - min_v)
            top_w = t ** 4
            top_w = top_w / top_w.sum()
        M = np.dot(top_w, top_vals)

    J = 2.0 * n * M

    # Compute gradient: grad_x[j] = 4n * a[j] * (sum_k w_k * a[k*j] - M)
    a_shifted = np.zeros(n)
    for wi, k in zip(top_w, top_idx):
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            j_range = np.arange(lo, hi + 1)
            a_shifted[j_range] += wi * a[ki - j_range]

    grad_x = 4.0 * n * a * (a_shifted - M)
    return J, grad_x


def run_lbfgs(x0, n, time_limit, n_peaks=1, temperature=0.0, tol=1e-10, maxiter=50000):
    """
    Run L-BFGS-B with time limit. Returns best x found.
    """
    t_start = time.monotonic()
    best_x = x0.copy()
    best_J = obj_grad_single_peak(x0, n)[0] if n_peaks == 1 else obj_grad_multi_peak(x0, n, n_peaks, temperature)[0]

    state = {'best_x': best_x, 'best_J': best_J}

    def callback(xk):
        nonlocal best_J, best_x
        if time.monotonic() - t_start > time_limit:
            raise TimeLimitExceeded()
        # Evaluate current point
        J_k = obj_grad_single_peak(xk, n)[0]
        if J_k < best_J:
            best_J = J_k
            best_x = xk.copy()
            state['best_x'] = best_x
            state['best_J'] = best_J
        return None

    if n_peaks == 1:
        fun = lambda x: obj_grad_single_peak(x, n)
    else:
        fun = lambda x: obj_grad_multi_peak(x, n, n_peaks, temperature)

    try:
        result = minimize(
            fun, x0, method='L-BFGS-B', jac=True,
            options={'maxiter': maxiter, 'gtol': tol, 'maxls': 40, 'maxcor': 20},
            callback=callback
        )
    except TimeLimitExceeded:
        pass
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    return best_x, best_J


def coordinate_polish(a0, time_limit, n_peaks=5, rng_seed=42):
    """
    Coordinate descent polish: move mass from positions that contribute
    most to the peak to positions that contribute least.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        M = b.max()
        J_curr = 2.0 * n * M

        top_ks = np.argpartition(b, -n_peaks)[-n_peaks:]

        # Find the position that most contributes to the peak
        # For each top k, find j that maximizes a[j]*a[k-j]
        best_move = None
        best_reduction = 0

        for k in top_ks:
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            if lo > hi:
                continue
            idx = np.arange(lo, hi + 1)
            contrib = a[idx] * a[ki - idx]
            src = idx[np.argmax(contrib)]
            red = contrib.max() * 0.1  # estimated reduction from moving 10% of src
            if red > best_reduction:
                best_reduction = red
                best_src = src
                best_k = ki

        if best_move is None and best_reduction > 0:
            src = best_src
            # Find destination: position that least contributes to any top peak
            costs = np.zeros(n)
            for k2 in top_ks:
                k2i = int(k2)
                lo2 = max(0, k2i - n + 1)
                hi2 = min(n - 1, k2i)
                if lo2 <= hi2:
                    valid = np.arange(lo2, hi2 + 1)
                    costs[valid] += a[k2i - valid]
            dst = int(np.argmin(costs))
            if dst == src:
                dst = (dst + 1) % n

            # Try different amounts to move
            for frac in [0.02, 0.05, 0.1, 0.15, 0.2]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_t = fast_conv(a_trial).max()
                if M_t < M - 1e-16:
                    a = a_trial
                    M = M_t
                    if M < best_m:
                        best_m = M
                        best_a = a.copy()
                    break
            else:
                # No improvement, try random perturbation
                moved = False
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-10:
                        continue
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    M_t = fast_conv(a_trial).max()
                    if M_t < M:
                        a = a_trial
                        M = M_t
                        if M < best_m:
                            best_m = M
                            best_a = a.copy()
                        moved = True
                        break
                if not moved:
                    # More aggressive perturbation
                    idx = rng.integers(n, size=10)
                    for i in idx:
                        j = (i + int(rng.integers(-5, 6)) + n) % n
                        if j != i and a[i] > 1e-10:
                            amt = a[i] * 0.01
                            a[i] -= amt
                            a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M = fast_conv(a).max()
                    if M < best_m:
                        best_m = M
                        best_a = a.copy()

    return best_a


def propose_candidate():
    """
    Main search: L-BFGS-B in log-space with exact gradients,
    multi-peak subgradients, temperature annealing, and coordinate polish.
    """
    t_global = time.monotonic()
    time_budget = 1740.0  # Leave margin

    best_score = float('inf')
    best_seq = None

    def try_update(a, label=""):
        nonlocal best_score, best_seq
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = safe_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            parent_norm = parent / parent.sum()
            p_score = safe_eval(parent_norm)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent_norm.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        # Fallback initialization
        best_seq = make_ramp(5000, 1.0)
        best_score = safe_eval(best_seq)
        print(f"Fallback init: n={len(best_seq)}, score={best_score:.6f}")

    n = len(best_seq)
    a0 = best_seq / best_seq.sum()
    a0_safe = np.maximum(a0, 1e-30)
    x0 = np.log(a0_safe)
    x0 -= x0.max()

    # ============================================================
    # Phase 1: L-BFGS-B with single peak (exact subgradient)
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: L-BFGS-B single peak (n={n}, t_avail={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 600.0)
    if t1 > 20:
        x_best, J_best = run_lbfgs(x0, n, t1, n_peaks=1, temperature=0.0, tol=1e-12, maxiter=100000)
        a_out = np.exp(x_best - x_best.max())
        a_out = a_out / a_out.sum()
        try_update(a_out, "lbfgs_p1")
        x0 = x_best.copy()
        a0 = a_out

    # ============================================================
    # Phase 2: L-BFGS-B with multi-peak (smoother)
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: L-BFGS-B multi-peak (t_avail={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 20:
        # Try with different numbers of peaks
        for npk in [5, 10]:
            t_avail = t2 / 2.0
            if t_avail < 15:
                break
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            t_use = min(t_avail, remaining)
            x_best, J_best = run_lbfgs(x0, n, t_use, n_peaks=npk, temperature=0.0, tol=1e-12, maxiter=50000)
            a_out = np.exp(x_best - x_best.max())
            a_out = a_out / a_out.sum()
            try_update(a_out, f"lbfgs_p2_k{npk}")
            x0 = x_best.copy()
            a0 = a_out

    # ============================================================
    # Phase 3: Temperature annealing L-BFGS-B
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Temperature annealing (t_avail={remaining:.0f}s) ===")
    if remaining > 60:
        # Start with high temperature, decrease
        b_curr = fast_conv(a0)
        M_curr = b_curr.max()
        temps = [M_curr * 0.5, M_curr * 0.2, M_curr * 0.1, M_curr * 0.05, M_curr * 0.02, M_curr * 0.01]
        t_per_temp = min(remaining * 0.15, 200.0)
        for temp in temps:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            t_use = min(t_per_temp, remaining)
            x_best, J_best = run_lbfgs(x0, n, t_use, n_peaks=20, temperature=temp, tol=1e-12, maxiter=30000)
            a_out = np.exp(x_best - x_best.max())
            a_out = a_out / a_out.sum()
            try_update(a_out, f"anneal_T{temp:.2e}")
            x0 = x_best.copy()
            a0 = a_out

    # ============================================================
    # Phase 4: Coordinate descent polish
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coordinate polish (t_avail={remaining:.0f}s) ===")
    if remaining > 30:
        t4 = min(remaining * 0.20, 200.0)
        a_in = best_seq / best_seq.sum()
        a_out = coordinate_polish(a_in, t4, n_peaks=5, rng_seed=42)
        try_update(a_out, "coord1")

        a_out = coordinate_polish(a_in, min(t4, time_budget - (time.monotonic() - t_global)), n_peaks=10, rng_seed=99)
        try_update(a_out, "coord2")

    # ============================================================
    # Phase 5: Final L-BFGS-B on best solution
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Final L-BFGS-B (t_avail={remaining:.0f}s) ===")
    if remaining > 30 and best_seq is not None:
        a_best = best_seq / best_seq.sum()
        a_best_safe = np.maximum(a_best, 1e-30)
        x_best = np.log(a_best_safe)
        x_best -= x_best.max()
        t5 = min(remaining * 0.40, 300.0)
        if t5 > 15:
            x_final, J_final = run_lbfgs(x_best, len(a_best), t5, n_peaks=1, temperature=0.0, tol=1e-14, maxiter=100000)
            a_out = np.exp(x_final - x_final.max())
            a_out = a_out / a_out.sum()
            try_update(a_out, "lbfgs_final")

    # ============================================================
    # Phase 6: Try slightly different n values
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-n exploration (t_avail={remaining:.0f}s) ===")
    if remaining > 60 and best_seq is not None:
        n_best = len(best_seq)
        n_trials = [n_best + 500, n_best + 1000, n_best - 500, n_best + 2000]
        n_trials = [nt for nt in n_trials if 3000 <= nt <= 12000]
        n_trials = list(set(n_trials))[:3]

        for new_n in n_trials:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            a_interp = interpolate_to_length(best_seq, new_n)
            try_update(a_interp, f"interp_n{new_n}")

            if remaining > 80:
                a_safe = np.maximum(a_interp, 1e-30)
                x_init = np.log(a_safe)
                x_init -= x_init.max()
                t_use = min(remaining * 0.30, 200.0)
                if t_use > 20:
                    x_out, J_out = run_lbfgs(x_init, new_n, t_use, n_peaks=1, temperature=0.0, tol=1e-12, maxiter=50000)
                    a_out = np.exp(x_out - x_out.max())
                    a_out = a_out / a_out.sum()
                    try_update(a_out, f"lbfgs_n{new_n}")

    # ============================================================
    # Phase 7: Very final polish
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20 and best_seq is not None:
        a_best = best_seq / best_seq.sum()
        a_best_safe = np.maximum(a_best, 1e-30)
        x_best = np.log(a_best_safe)
        x_best -= x_best.max()
        t7 = min(remaining * 0.5, 100.0)
        if t7 > 10:
            x_final, _ = run_lbfgs(x_best, len(a_best), t7, n_peaks=3, temperature=0.0, tol=1e-14, maxiter=50000)
            a_out = np.exp(x_final - x_final.max())
            a_out = a_out / a_out.sum()
            try_update(a_out, "final_polish")

    # Final validation
    if best_seq is not None:
        fs = safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        best_seq = make_ramp(5000, 1.0)
        best_score = safe_eval(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
