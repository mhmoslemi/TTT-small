def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""

    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

def bregman_grad_desc(a0, time_limit, top_k=100, beta_init=8.0, c0=6.0,

            break
        
        # NAG extrapolation
        beta_t = (t - 1.0) / (t + 1.5)

            y = x.copy()
        else:
            y = y / sy

        # Adaptive temperature

        w = np.exp(cur_beta * (top_vals - top_vals[0]) / T)

        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            g[idx] += 2.0 * wi * y[j_idx]
        
        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

                x = x_r
                x_prev = x_r.copy()

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='sine', seed=0):

        phase = chirp_rate * freqs ** 2 / n
        a = 1.0 + np.cos(phase)
        a = np.maximum(a, 0.0)
    elif kind == 'u_shape':
        x_center = x - 0.5
        a = 1.0 + 3.0 * (2 * x_center) ** 2

    elif kind == 'random':
        rng = np.random.default_rng(seed)
        a = rng.uniform(0.1, 1.0, n)
    else:
        a = np.ones(n)
    return norm_sum1(a)

2. Away-step Bregman gradient descent with adaptive temperature cooling

import numpy as np
import time


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def pocs_optimize(a0, time_limit, rng_seed=0):
    """Aggressive POCS: alternate flat-spectrum projection and non-negativity."""
    n = len(a0)
    best_a = norm_sum1(a0)
    best_m = fast_autocorr(best_a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Frequency: push toward flat magnitude spectrum
        A = np.fft.rfft(a0, fft_size)
        mag = np.abs(A)
        active = mag > 1e-10
        mag_mean = np.mean(mag[active]) if active.any() else 1.0

        # Adaptive blending: start gentle, become aggressive
        progress = min(1.0, it / 3000)
        alpha = 0.2 + 0.7 * progress
        new_mag = (1.0 - alpha) * mag + alpha * mag_mean

        # Extra: suppress peaks in power spectrum
        power = new_mag ** 2
        p_mean = np.mean(power[power > 1e-20])
        suppress = 0.98 * np.ones_like(new_mag)
        above = power > p_mean * 1.5
        suppress[above] = 0.92
        new_mag = new_mag * suppress

        A_new = new_mag * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        # Time: enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a0.copy()
        else:
            a_new = a_new / s

        a0 = a_new
        m = fast_autocorr(a0).max()
        if m < best_m:
            best_m = m
            best_a = a0.copy()

    return best_a, best_m


def pocs_multi_init(a0, time_limit, n_trials=8, rng_seed=0):
    """POCS with multiple structured phase initializations."""
    n = len(a0)

    t0 = time.monotonic()

            phases = base + mod

        A_full = np.zeros(fft_size, dtype=np.complex128)

        a_init = np.fft.irfft(A_full, fft_size)[:n]

        s = a_init.sum()

            a_init = best_a.copy()

        if m_out < best_m:

            best_a = a_out.copy()

def bregman_gd(a0, time_limit, top_k_init=80, beta_init=10.0, c0=7.0,
               restart_every=4000, rng_seed=0, k_max=250):
    """Away-step Bregman gradient descent with adaptive temperature."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    cur_top_k = top_k_init
    cur_beta = beta_init
    T = float(n)  # softmax temperature

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # NAG momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute weighted subgradient at top peaks
        ac = fast_autocorr(y)
        k_sel = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = np.maximum(ac[top_idx], 1e-30)

        # Softmax weighting with cooling temperature
        T = max(0.5, T * 0.999)
        w = np.exp(cur_beta * (top_vals - top_vals[0]) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_sel) / k_sel
        else:
            w = w / w_sum

        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            g[idx] += 2.0 * wi * y[j_idx]

        # Mirror descent step (KL divergence Bregman)
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
            if stall > restart_every // 3:
                cur_beta = max(cur_beta * 0.95, 5.0)
                cur_top_k = min(cur_top_k + 8, k_max)
            if stall > restart_every:
                stall = 0
                # Aggressive restart from best with perturbation
                scale = 0.03 + 0.05 * (t % 100) / 100.0
                pert = rng.normal(0, scale, n) * best_a
                x_r = best_a + pert
                x_r = np.maximum(x_r, 1e-10)
                x_r = x_r / x_r.sum()
                x = x_r
                x_prev = x_r.copy()
                cur_top_k = min(cur_top_k + 15, k_max + 50)
                cur_beta = max(cur_beta - 2.0, 5.0)
                T = float(n)

    return best_a, best_m


def spectral_shape(a0, time_limit, rng_seed=0):
    """Iterative spectral shaping: alternate peak suppression and non-negativity."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        power = mag ** 2

        active = power > 1e-20
        p_mean = np.mean(power[active]) if active.any() else 1.0

        # Peak suppression: reduce high-power bins
        ratio = power / p_mean
        damp = np.ones_like(mag)
        damp[ratio > 1.2] = 1.0 / np.sqrt(ratio[ratio > 1.2])
        damp[ratio > 2.0] = 0.85

        mag_new = mag * damp
        # Fill in low-power bins
        mag_new[ratio < 0.5] = mag_new[ratio < 0.5] * 1.05 + 0.02 * np.sqrt(p_mean)

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a = a_new / s

        m = fast_autocorr(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()

    return best_a, best_m


def coord_descent(a0, time_limit, rng_seed=0, n_trials_peak=30, n_random_dst=40):
    """Exact coordinate descent: move mass from peak contributors to cold spots."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_trials_peak]

        improved = False

        # Identify "hot" and "cold" coordinates
        hot = np.zeros(n, dtype=bool)
        for k in top_k[:15]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            hot[s:e+1] = True

        cold_indices = np.where(~hot)[0]

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]

            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Try moving to cold spots
            best_m_trial = best_m
            best_move = None

            candidates = cold_indices
            if len(candidates) < 10:
                candidates = np.arange(n)

            # Sample a subset for speed
            if len(candidates) > 200:
                sampled = rng.choice(candidates, 200, replace=False)
            else:
                sampled = candidates

            for dst in sampled:
                if dst == src:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m_trial:
                        best_m_trial = M_trial
                        best_move = (src, int(dst), frac)

            if best_move is not None:

                amt = a[src_i] * frac
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = fast_autocorr(a).max()
                best_a = a.copy()
                improved = True

        if not improved:
            stall += 1
            # Random kick
            n_kicks = min(10, n // 10)
            for _ in range(n_kicks):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = max(0, stall - 5)

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_init(n, kind='sine', seed=0):
    """Create various initial sequences."""
    x = np.linspace(0, 1, n)
    if kind == 'sine':
        a = np.sin(np.pi * x) ** 2
    elif kind == 'chirp':
        rng = np.random.default_rng(seed)
        freqs = np.arange(n, dtype=np.float64)
        rate = rng.uniform(1.0, 5.0)
        phase = 2 * np.pi * rate * (freqs / n) ** 2
        a = 1.0 + 0.8 * np.cos(phase)
        a = np.maximum(a, 0.0)
    elif kind == 'u_shape':
        xc = x - 0.5
        a = 1.0 + 4.0 * (2 * xc) ** 2
    elif kind == 'bimodal':
        a = (np.exp(-((x - 0.15) / 0.12) ** 2) +
             np.exp(-((x - 0.5) / 0.1) ** 2) +
             np.exp(-((x - 0.85) / 0.12) ** 2) + 0.01)
    elif kind == 'cosine':
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 2
    elif kind == 'random_smooth':
        rng = np.random.default_rng(seed)
        a = rng.uniform(0.3, 1.0, n)
        # Smooth
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')
    else:
        a = np.ones(n)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator:
    Phase 1: POCS multi-init (structured spectral initialization)
    Phase 2: Multi-init Bregman gradient descent
    Phase 3: Spectral peak shaping
    Phase 4: Bregman on best (longer run)
    Phase 5: Length exploration
    Phase 6: Final Bregman + spectral
    Phase 7: Coordinate descent polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 6200
        best_seq = make_init(n_base, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: POCS Multi-Init ===
    print(f"\n=== Phase 1: POCS Multi-Init (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.14, 220.0)
    if t1 > 30:
        a_pocs, _ = pocs_multi_init(best_seq, t1, n_trials=8, rng_seed=7)
        try_update(a_pocs, "pocs")

    # === Phase 2: Multi-init Bregman GD ===
    print(f"\n=== Phase 2: Bregman GD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.26, 420.0)
    if t2 > 60:
        inits = []
        labels = []
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        inits.append(norm_sum1(best_seq))
        labels.append("best")
        inits.append(make_init(n0, 'chirp', seed=42))
        labels.append("chirp")
        inits.append(make_init(n0, 'u_shape'))
        labels.append("u_shape")
        inits.append(make_init(n0, 'bimodal'))
        labels.append("bimodal")
        inits.append(make_init(n0, 'random_smooth', seed=99))
        labels.append("rand_smooth")

        t_per = max(20.0, t2 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            if time.monotonic() - t_global > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 15:
                break
            print(f"  Bregman '{lbl}' (t={t_avail:.0f}s) score_0={evaluate_safe(init):.6f}...")
            a_out, _ = bregman_gd(init, t_avail, top_k_init=80, beta_init=10.0,
                                   c0=7.0, restart_every=4000, rng_seed=42 + idx * 13)
            try_update(a_out, f"breg_{lbl}")

    # === Phase 3: Spectral peak shaping ===
    print(f"\n=== Phase 3: Spectral Shape (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 25:
        a_out, _ = spectral_shape(best_seq, t3, rng_seed=50)
        try_update(a_out, "spectral")

    # === Phase 4: Bregman on best (longer) ===
    print(f"\n=== Phase 4: Bregman r2 (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.2, 320.0)
    if t4 > 30:
        a_out, _ = bregman_gd(best_seq, t4, top_k_init=100, beta_init=12.0,
                               c0=9.0, restart_every=3500, rng_seed=200)
        try_update(a_out, "breg_r2")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length Exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    trial_lengths = []
    for delta in [1000, -1000, 2000, -2000, 1500, -1500, 2500, -500, 500, 3000]:
        new_n = n_best + delta
        if 1500 <= new_n <= 9000 and new_n != n_best:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))

    for new_n in trial_lengths[:5]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 70:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.22, 100.0)
        if t5a > 20:
            a_out, _ = bregman_gd(a_long, t5a, top_k_init=70, beta_init=8.0,
                                   c0=6.0, restart_every=3500, rng_seed=new_n % 500)
            try_update(a_out, f"breg_n{new_n}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 25:
                t5b = min(remaining * 0.25, 70.0)
                if t5b > 15:
                    a_out2, _ = spectral_shape(interpolate_sequence(best_seq, new_n), t5b)
                    try_update(a_out2, f"spec_n{new_n}")

    # === Phase 6: Final Bregman + Spectral ===
    print(f"\n=== Phase 6: Final Opt (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.2, 180.0)
    if t6 > 20:
        a_out, _ = bregman_gd(best_seq, t6, top_k_init=120, beta_init=15.0,
                               c0=10.0, restart_every=3000, rng_seed=999)
        try_update(a_out, "breg_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.5, 100.0)
            if t6b > 15:
                a_out2, _ = spectral_shape(best_seq, t6b, rng_seed=1234)
                try_update(a_out2, "spec_final")

    # === Phase 7: Coordinate descent polish ===
    print(f"\n=== Phase 7: Coord Descent (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.25, 150.0)
    if t7 > 15:
        a_out, _ = coord_descent(best_seq, t7, rng_seed=789)
        try_update(a_out, "coord1")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.6, 80.0)
            if t7b > 10:
                a_out2, _ = coord_descent(best_seq, t7b, rng_seed=456)
                try_update(a_out2, "coord2")

    # === Phase 8: One more Bregman round on final best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 8: Final Bregman (n={len(best_seq)}) ===")
        t8 = min(remaining * 0.5, 100.0)
        if t8 > 15:
            a_out, _ = bregman_gd(best_seq, t8, top_k_init=150, beta_init=20.0,
                                   c0=12.0, restart_every=2000, rng_seed=31415)
            try_update(a_out, "breg_v3")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        n_fb = 6200
        best_seq = make_init(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
