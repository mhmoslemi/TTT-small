def evaluate_sequence(seq):
    """Evaluates a sequence of coefficients with defensive checks. Returns np.inf if invalid."""
    if not isinstance(seq, list):
        return np.inf
    if not seq:
        return np.inf
    # Convert to float32 for speed and check types
    seq_float = []
    for val in seq:
        try:
            f_val = float(val)
            if isinstance(f_val, bool):
                raise TypeError
            seq_float.append(f_val)
        except TypeError:
            return np.inf
    # Check for NaN and Infinity
    if any(np.isnan(x) for x in seq_float):
        return np.inf
    if any(x < 0 for x in seq_float):
        return np.inf
    if any(np.isinf(x) for x in seq_float):
        return np.inf
    
    # Defensive clipping (already in the definition)
    n = len(seq_float)
    res_seq = np.convolve(seq_float, seq_float)
    sum_val = np.sum(seq_float)
    
    # Return config if needed, or directly the objective
    return float(2.0 * n * np.max(res_seq) / (sum_val ** 2))

def normalize(seq):
    """Convert to nonnegative and normalize sum to at least 0.01"""
    s = np.sum(np.asarray(seq, dtype=np.float64))
    if s < 1e-15:
        return np.ones_like(seq) * 0.01
    # Ensure nonnegative and clipped to bounds if needed
    res = np.asarray(seq, dtype=np.float64)
    res = np.clip(res * 0.99, 0.0, 1000.0)  # Keep coefficients centered and within bounds
    res_sum = np.sum(res)
    return res if res_sum >= 0.01 else res * 0.02
