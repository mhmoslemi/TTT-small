Minimize 2*n*max(conv(a,a))/(sum(a))^2 for nonnegative sequence a.
Combined LP-Frank-Wolfe + mirror descent + subgradient + restarts.
"""

import math
import time
import numpy as np

try:
    from scipy.optimize import linprog
    HAVE_LP = True
except Exception:
    HAVE_LP = False

try:
    from scipy import sparse
    HAVE_SPARSE = True
except Exception:
    HAVE_SPARSE = False


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]


def conv_ab(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]


def objective_norm(a):
    """Objective assuming sum(a) = 1."""
    n = a.size
    q = self_conv(a)
    return 2.0 * n * float(np.max(q))


def full_evaluate(a):
    """Full objective 2n*max(conv)/(sum)^2."""
    n = a.size
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    q = self_conv(a)
    return 2.0 * n * float(np.max(q)) / (S * S)

    try:
        x = [float(v) for v in np.asarray(a, dtype=np.float64)]
        val = evaluate_sequence(x)
        if np.isfinite(val):
            return float(val)
    except Exception:

    return float('inf')


def line_search(a, d, q0, qd, c_cross):
    """
    Minimize max_k [(1-α)^2 q0[k] + 2α(1-α) c[k] + α^2 qd[k]] over α ∈ [0,1].
    Returns (best_alpha, best_max).
    """
    M0 = float(np.max(q0))
    
    best_alpha = 0.0
    best_M = M0
    
    # Coarse grid
    alphas = np.array([0.0, 0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0])
    
    for alpha in alphas:
        a1 = 1.0 - alpha
        val = a1 * a1 * q0 + 2.0 * alpha * a1 * c_cross + alpha * alpha * qd
        M = float(np.max(val))
        if M < best_M:
            best_M = M
            best_alpha = alpha
    
    # Fine golden section around the best grid point
    if 0.0 < best_alpha < 1.0:
        # Determine interval
        grid_idx = np.argmin(alphas - best_alpha)
        a_lo = float(alphas[max(0, grid_idx - 1)])
        a_hi = float(alphas[min(len(alphas) - 1, grid_idx + 1)])
    else:
        a_lo, a_hi = 0.0, 1.0
    
    if a_hi - a_lo > 1e-10:
        # Golden section
        phi = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = a_hi - phi * (a_hi - a_lo)
        d1 = a_lo + phi * (a_hi - a_lo)
        a1_c1 = 1.0 - c1
        a1_d1 = 1.0 - d1
        f1 = float(np.max(a1_c1*a1_c1*q0 + 2*c1*a1_c1*c_cross + c1*c1*qd))
        f2 = float(np.max(a1_d1*a1_d1*q0 + 2*d1*a1_d1*c_cross + d1*d1*qd))
        
        for _ in range(50):
            if a_hi - a_lo < 1e-12:
                break
            if f1 < f2:
                a_hi = d1
                d1 = c1
                f2 = f1
                c1 = a_hi - phi * (a_hi - a_lo)
                a1_c1 = 1.0 - c1
                f1 = float(np.max(a1_c1*a1_c1*q0 + 2*c1*a1_c1*c_cross + c1*c1*qd))
            else:
                a_lo = c1
                c1 = d1
                f1 = f2
                d1 = a_lo + phi * (a_hi - a_lo)
                a1_d1 = 1.0 - d1
                f2 = float(np.max(a1_d1*a1_d1*q0 + 2*d1*a1_d1*c_cross + d1*d1*qd))
        
        mid = 0.5 * (a_lo + a_hi)
        a1 = 1.0 - mid
        fmid = float(np.max(a1*a1*q0 + 2*mid*a1*c_cross + mid*mid*qd))
        if fmid < best_M:
            best_M = fmid
            best_alpha = mid
    
    return best_alpha, best_M


def solve_lp_direction(y, active, q_active, n):
    """
    Solve: min T s.t. sum g = 1, g >= 0,
            2*(y*g)[k] - T <= q_active[r] for r-th active lag.
    Returns (g, T) or (None, None).
    """
    if not HAVE_LP:
        return None, None
    
    K = len(active)
    if K <= 0 or K > 3000:
        return None, None
    
    t_start = time.monotonic()
    
    # Build sparse A matrix
    if HAVE_SPARSE:
        try:
            active_arr = np.asarray(active, dtype=int)
            rows_list = []
            cols_list = []
            vals_list = []
            
            for r in range(K):
                k = int(active_arr[r])
                i0 = max(0, k - n + 1)
                i1 = min(n, k + 1)
                if i0 < i1:
                    i_vals = np.arange(i0, i1)
                    j_vals = k - i_vals
                    rows_list.extend([r] * len(i_vals))
                    cols_list.extend(i_vals.tolist())
                    vals_list.extend((2.0 * y[j_vals]).tolist())
                # T column
                rows_list.append(r)
                cols_list.append(n)
                vals_list.append(-1.0)
            
            A = sparse.csr_matrix((np.array(vals_list), (np.array(rows_list), np.array(cols_list))),
                                  shape=(K, n + 1))
            
            c = np.zeros(n + 1)
            c[n] = 1.0
            
            Aeq = np.ones((1, n + 1))
            Aeq[0, n] = 0.0  # T doesn't appear in equality
            
            bounds = [(0.0, None)] * n + [(None, None)]
            
            res = linprog(c, A_ub=A, b_ub=np.asarray(q_active, dtype=np.float64),
                         A_eq=Aeq, b_eq=np.array([1.0]),
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 60.0})
            
            if res.status != 0 or res.x is None:
                return None, None
            
            g = np.maximum(res.x[:n], 0.0)
            s = float(g.sum())
            if s < 1e-10:
                return None, None
            g = g / s
            T = float(res.x[n])
            return g, T
            
        except Exception:
            return None, None
    else:
        return None, None


def mirror_step(a, q, trust=0.5):
    """
    Multiplicative gradient step: a_new[i] ∝ a[i] * exp(-trust * grad[i]).
    grad[i] = sum of q contributions to row i.
    """
    n = a.size
    m = 2 * n - 1
    
    # Gradient of max_k q[k] w.r.t. a[i]
    # Use the top few lags as subgradient
    top_k = np.argsort(q)[::-1][:min(20, m)]
    avg_q = float(np.mean(q[top_k]))
    
    # Subgradient: average of ∂q[k*]/∂a[i] for active k*
    # ∂q[k]/∂a[i] = a[k-i] + a[i-k] (with valid index check)
    grad = np.zeros(n, dtype=np.float64)
    count = 0
    for k in top_k:
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        i_vals = np.arange(i0, i1)
        j_vals = k - i_vals
        # ∂q[k]/∂a[i] includes a[j] where j = k-i
        # But since q[k] = sum_i a[i]*a[k-i], ∂/∂a[i] = a[k-i] (if valid) + a[i-k] (if valid, and i≠k-i)
        # Actually for the full sum: ∂q[k]/∂a[i] = a[k-i] (if 0≤k-i<n)
        # Plus a[i-k] (if 0≤i-k<n and i-k != k-i)
        # Simplified: ∂q[k]/∂a[i] = a[k-i] + a[i-k] (counting each term)
        # But this double-counts when i = k-i, i.e., 2i = k
        # For subgradient purposes, we can just use a[k-i] (one side)
        grad[i_vals] += a[j_vals]
        count += 1
    
    if count > 0:
        grad /= count  # average subgradient
    
    # Mirror step
    move = -trust * (grad - grad.mean()) / (grad.std() + 1e-15)
    move = np.clip(move, -50.0, 50.0)
    b = a * np.exp(move)
    return normalize_vec(b)
