For the score to be small, we want all C[k] to be small. In particular:
- C[n-1] = sum_i a[i] * a[n-1-i] (the "center")
- C[0] = a[0]^2
- C[2n-2] = a[n-1]^2
- C[1] = a[0]*a[1] + a[1]*a[0] = 2*a[0]*a[1]

The sum of all C[k] = (sum a)^2 = 1 (when normalized). So if there are 2n-1 lags, the average is 1/(2n-1). For the max to be close to the average, we want all lags to be approximately equal, i.e., C[k] ≈ 1/(2n-1) for all k.

This is exactly the "flat autocorrelation" or "flat spectrum" condition! A sequence with perfectly flat autocorrelation would have score = 2n/(2n-1) ≈ 1 for large n. But with non-negative sequences, this is not achievable in general.

The best we can do is to make the autocorrelation as flat as possible while maintaining non-negativity.

This gives me a clearer optimization target: minimize the variance of the autocorrelation (or minimize the max while keeping the sum fixed).

def mirror_optimize(a0, time_limit, lr_init=0.1, beta_init=4.0, top_k_init=50,
                    restart_every=3000, rng_seed=42, verbose=False):
    """
    Multiplicative update optimization with adaptive parameters.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    C = fast_conv(a)
    best_m = C.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    a_prev = a.copy()
    momentum = 0.8
    lr = lr_init
    beta = beta_init
    top_k = top_k_init
    stall = 0
    best_beta = beta_init

    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov in multiplicative space
        ratio = a / np.maximum(a_prev, 1e-25)
        ratio = np.clip(ratio, 0.02, 50.0)
        a_nest = a * np.power(ratio, momentum)
        a_nest = normalize(a_nest)

        C = fast_conv(a_nest)
        grad = compute_grad(a_nest, C, top_k, beta)

        # Project: subtract mean to preserve sum
        grad -= grad.mean()

        # Learning rate decay
        eta = lr / (1.0 + 0.03 * np.sqrt(t))

        # Multiplicative update
        log_a = np.log(np.maximum(a_nest, 1e-25)) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-18)
        a_new = a_new / a_new.sum()

        a_prev = a
        a = a_new

        C_new = fast_conv(a)
        m_new = C_new.max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive parameters
        if t % 500 == 0:
            top_k = min(top_k + 3, 200)
            beta = max(beta - 0.3, 1.0)

        # Restart when stuck
        if stall > restart_every:
            stall = 0
            pert_type = int(rng.integers(6))
            if pert_type == 0:
                # Gaussian bump
                x = np.linspace(0, 1, n)
                c = float(rng.uniform(0.2, 0.8))
                sig = float(rng.uniform(0.05, 0.3))
                pert = np.exp(-((x - c)**2) / (2 * sig**2))
            elif pert_type == 1:
                # Power law
                i_arr = np.arange(n, dtype=np.float64)
                p = float(rng.uniform(0.3, 3.0))
                pert = (i_arr + 1.0) ** p
            elif pert_type == 2:
                # Inverse power law
                i_arr = np.arange(n, dtype=np.float64)
                p = float(rng.uniform(0.3, 3.0))
                pert = (n - i_arr) ** p
            elif pert_type == 3:
                # Sine
                x = np.linspace(0, 1, n)
                freq = int(rng.integers(1, 7))
                pert = np.abs(np.sin(np.pi * freq * x + float(rng.uniform(0, np.pi))))**2
            elif pert_type == 4:
                # Bimodal
                x = np.linspace(0, 1, n)
                sig = float(rng.uniform(0.08, 0.25))
                pert = np.exp(-((x - 0.25)**2) / (2*sig**2)) + np.exp(-((x - 0.75)**2) / (2*sig**2))
            else:
                # Small perturbation of best
                pert = best_a * (1.0 + rng.normal(0, 0.1, n))

            pert = normalize(pert)
            w_blend = float(rng.uniform(0.1, 0.5))
            a = normalize((1 - w_blend) * best_a + w_blend * pert)
            a_prev = a.copy()
            lr *= 0.8
            beta = max(beta - 0.5, 1.5)
            momentum = float(rng.uniform(0.5, 0.9))

            if verbose:
                print(f"    MD restart t={t} ({time.monotonic()-t0:.0f}s): m={best_m:.8f}, top_k={top_k}, beta={beta:.1f}")

        print(f"    MD done ({time.monotonic()-t0:.0f}s): m={best_m:.8f}")

def coord_polish(a0, time_limit, rng_seed=77, verbose=False):
    """
    Coordinate descent: move small amounts from high-impact to low-impact positions
    to reduce the peak autocorrelation.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    C = fast_conv(a)
    best_m = C.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        C = fast_conv(a)
        if C.max() < best_m - 1e-14:
            best_m = C.max()
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30000:
                break

        # Top lags
        n_top = min(20, 2 * n - 1)
        top_lags = np.argsort(C)[::-1][:n_top]
        improved = False

        for k_pos in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k_pos)
            s_lo = max(0, ki - n + 1)
            s_hi = min(n - 1, ki)
            if s_lo > s_hi:
                continue

            s_arr = np.arange(s_lo, s_hi + 1)
            other = ki - s_arr
            contrib = a[s_arr] * a[other]

            # Find highest-contribution element
            n_pairs = min(5, len(s_arr))
            pair_order = np.argsort(contrib)[::-1][:n_pairs]

            for pi in pair_order:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(s_arr[pi])
                src_val = a[src]
                if src_val < 1e-14:
                    continue

                # Score each destination by its impact on top lags
                scores = np.zeros(n)
                for k2 in top_lags[:6]:
                    k2i = int(k2)
                    s2_lo = max(0, k2i - n + 1)
                    s2_hi = min(n - 1, k2i)
                    if s2_lo > s2_hi:
                        continue
                    s2 = np.arange(s2_lo, s2_hi + 1)
                    o2 = k2i - s2
                    scores[s2] += a[o2] * max(C[k2i], 1e-15)

                scores[src] = np.inf
                # Prefer low-score destinations
                dst_candidates = np.argsort(scores)[:20]
                dst_candidates = dst_candidates[scores[dst_candidates] < scores[src] * 1.5]
                if len(dst_candidates) == 0:
                    dst_candidates = np.argsort(scores)[:10]

                for dst in dst_candidates[:3]:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.01, 0.03, 0.05, 0.1]:
                        amt = src_val * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[src] = src_val - amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial /= st
                        m_trial = fast_conv(a_trial).max()
                        if m_trial < best_m - 1e-15:
                            a = a_trial
                            best_m = m_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break

        if not improved:
            if stall > 1000 and stall % 200 == 0:
                # Random perturb
                n_moves = int(rng.integers(3, 15))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = float(rng.uniform(0.003, 0.02))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a /= a.sum()
                m_new = fast_conv(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        print(f"    Coord done ({time.monotonic()-t0:.0f}s): m={best_m:.8f}")
    return best_a, best_m

    fft_size = 1 << (2 * n - 2).bit_length()
    lr = lr_init
    stall = 0
    beta_spec = 3.0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A)**2
        P_max = P.max()
        P_mean = P.mean()

        if P_max < 1e-30:
            break

        # Push energy from spectral peaks to valleys
        P_dev = (P - P_mean) / max(P_max, 1e-30)
        expo = beta_spec * P_dev
        expo = np.clip(expo, -15, 15)
        w_spec = np.exp(expo)
        w_mean = w_spec.mean()

        step = lr / (1.0 + np.sqrt(t) * 0.1)
        A_new = A * np.exp(-0.5 * step * w_spec / max(w_mean, 1e-10))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = normalize(a_new)

        m_new = fast_conv(a_new).max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 2000:
                a = normalize(best_a * (1.0 + rng.normal(0, 0.02, n)))
                stall = 0
                lr *= 0.85
                beta_spec = max(beta_spec - 0.3, 0.8)

    if verbose:
        print(f"    Spectral done ({time.monotonic()-t0:.0f}s): m={best_m:.8f}")
    return best_a, best_m


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def parametric_init(n, time_limit=8.0):
    """Quick parametric search for good initializations."""
    t0 = time.monotonic()
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)

    candidates = []

    # Uniform
    candidates.append(np.ones(n) / n)

    # Power laws (ascending and descending)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(normalize((i_arr + 1.0)**p))
        candidates.append(normalize((n - i_arr)**p))

    # Gaussians at various positions
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sig in [0.08, 0.12, 0.2, 0.3, 0.4]:
            candidates.append(normalize(np.exp(-((x - c)**2) / (2 * sig**2))))

    # Bimodal
    for sig in [0.08, 0.15, 0.2]:
        candidates.append(normalize(
            np.exp(-((x - 0.2)**2) / (2*sig**2)) + np.exp(-((x - 0.8)**2) / (2*sig**2))))
        candidates.append(normalize(
            np.exp(-((x - 0.3)**2) / (2*sig**2)) + np.exp(-((x - 0.7)**2) / (2*sig**2))))

    # Sine waves
    for f in [1, 2, 3, 4, 5]:
        candidates.append(normalize(np.abs(np.sin(np.pi * f * x))**2))
        candidates.append(normalize(np.sin(np.pi * f * x)**2))

    # Exponential decay/growth
    for r in [0.5, 1, 2, 3, 5]:
        candidates.append(normalize(np.exp(-r * x)))
        candidates.append(normalize(np.exp(r * x)))

    # Trapezoidal
    for w1 in [0.1, 0.2, 0.3]:
        for w2 in [0.4, 0.5, 0.6]:
            trap = np.zeros(n)
            trap[:int(w1*n)] = np.linspace(0, 1, int(w1*n))
            trap[int(w1*n):int(w2*n)] = 1.0
            trap[int(w2*n):] = np.linspace(1, 0, n - int(w2*n))
            candidates.append(normalize(trap))

    # Random smooth
    for seed in [1, 7, 42]:
        rng_c = np.random.default_rng(seed)
        freqs = rng_c.integers(1, 5, 5)
        phases = rng_c.uniform(0, 2*np.pi, 5)
        amps = rng_c.uniform(0.5, 1.0, 5)
        sig = np.zeros(n)
        for fi, ph, am in zip(freqs, phases, amps):
            sig += am * np.sin(2*np.pi*fi*x + ph)
        candidates.append(normalize(np.abs(sig) + 0.1))

    # Evaluate all
    best_cand = None
    best_val = float('inf')
    for c in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        Cv = fast_conv(c)
        v = 2.0 * n * Cv.max()
        if v < best_val:
            best_val = v
            best_cand = c.copy()

    return best_cand if best_cand is not None else np.ones(n) / n


def evaluate_safe(a):
    """Safely call the evaluation function."""
    try:
        a_list = [float(x) for x in a]
        val = float(evaluate_sequence(a_list))
        return val
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with multi-phase optimization.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)

    best_seq = None
    best_score = float('inf')

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01 or a.size == 0:
            return
        a_n = a / s
        sc = evaluate_safe(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score) and p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # Also try uniform and parametric inits
    n_default = 5000
    a_uni = np.ones(n_default) / n_default
    sc_uni = evaluate_safe(a_uni)
    if np.isfinite(sc_uni) and sc_uni < best_score:
        best_score = sc_uni
        best_seq = a_uni.copy()
        print(f"  Uniform: {sc_uni:.6f}")

    if best_seq is None:
        best_seq = parametric_init(n_default, time_limit=5.0)
        best_score = evaluate_safe(best_seq)
        print(f"  Parametric init: n={len(best_seq)}, score={best_score:.6f}")

    # === Phase 1: Multiple starting points with coarse MD ===
    print(f"\n=== Phase 1: Multi-start MD (budget ~{time_budget*0.3:.0f}s) ===")
    n_best = len(best_seq)
    t_phase1 = min(time_budget * 0.25, 400.0)
    t1_start = time.monotonic()

    # Run MD on parent
    t1a = min(t_phase1 * 0.4, 150.0)
    if t1a > 20:
        a_out, _ = mirror_optimize(best_seq, t1a, lr_init=0.1, beta_init=4.0,
                                   top_k_init=60, restart_every=2500, rng_seed=42, verbose=True)
        check_update(a_out, f"md_parent_n{n_best}")

    # Try different parametric inits at the same length
    remaining_p1 = t_phase1 - (time.monotonic() - t1_start)
    if remaining_p1 > 30:
        a_param = parametric_init(n_best, time_limit=5.0)
        check_update(a_param, f"param_n{n_best}")
        if evaluate_safe(a_param) < best_score - 0.01:
            t1b = min(remaining_p1 * 0.4, 100.0)
            if t1b > 15:
                a_out, _ = mirror_optimize(a_param, t1b, lr_init=0.1, beta_init=3.0,
                                           top_k_init=50, restart_every=2000, rng_seed=100, verbose=False)
                check_update(a_out, f"md_param_n{n_best}")

    # Try a different length (shorter, for different dynamics)
    remaining_p1 = t_phase1 - (time.monotonic() - t1_start)
    if remaining_p1 > 40:
        n_alt = max(2000, n_best - 1000)
        a_alt = interpolate_seq(best_seq, n_alt)
        t1c = min(remaining_p1 * 0.35, 80.0)
        if t1c > 15:
            a_out, _ = mirror_optimize(a_alt, t1c, lr_init=0.08, beta_init=3.5,
                                       top_k_init=50, restart_every=2000, rng_seed=200, verbose=False)
            check_update(a_out, f"md_n{n_alt}")

    # === Phase 2: Fine MD on best ===
    print(f"\n=== Phase 2: Fine MD (budget ~{time_budget*0.2:.0f}s) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.18, 300.0)
    if t2 > 20:
        a_out, _ = mirror_optimize(best_seq, t2, lr_init=0.04, beta_init=6.0,
                                   top_k_init=100, restart_every=2000, rng_seed=300, verbose=True)
        check_update(a_out, f"md_fine_n{n_best}")

    # === Phase 3: Spectral flattening ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Spectral ({remaining*0.12:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 15:
        a_out, _ = spectral_flatten(best_seq, t3, lr_init=0.25, verbose=True)
        check_update(a_out, f"spectral_n{n_best}")

    # === Phase 4: Another MD pass after spectral ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Post-spectral MD ({remaining*0.15:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_out, _ = mirror_optimize(best_seq, t4, lr_init=0.03, beta_init=7.0,
                                   top_k_init=120, restart_every=1500, rng_seed=400, verbose=False)
        check_update(a_out, f"md_post_spec_n{n_best}")

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration ({remaining:.0f}s) ===")

    length_tries = set()
    for delta in [-500, -300, 300, 500, 800, 1200, 1500]:
        nn = n_best + delta
        if 1500 <= nn <= 10000 and nn != n_best:
            length_tries.add(nn)
    for rn in [3000, 4000, 5000, 6000, 7000, 8000]:
        if 1500 <= rn <= 10000 and rn != n_best:
            length_tries.add(rn)
    length_tries = sorted(length_tries)

    tried = 0
    for new_n in length_tries[:6]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        tried += 1
        print(f"  Length {new_n} (tried {tried}, remaining={remaining:.0f}s)")

        a_interp = interpolate_seq(best_seq, new_n)
        # Quick parametric init at this length too
        a_p = parametric_init(new_n, time_limit=3.0)
        sc_p = evaluate_safe(a_p)
        if np.isfinite(sc_p) and sc_p < 2.0:
            check_update(a_p, f"param_n{new_n}")
            if evaluate_safe(a_p) < evaluate_safe(a_interp):
                a_interp = a_p

        t5a = min(remaining * 0.25, 80.0)
        if t5a > 10:
            a_out, _ = mirror_optimize(a_interp, t5a, lr_init=0.06, beta_init=4.0,
                                       top_k_init=60, restart_every=2000, rng_seed=(new_n % 1000) + 500, verbose=False)
            check_update(a_out, f"md_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t5b = min(remaining * 0.15, 50.0)
            if t5b > 10:
                a_out, _ = mirror_optimize(best_seq if len(best_seq)==new_n else a_interp, t5b,
                                           lr_init=0.03, beta_init=5.5,
                                           top_k_init=80, restart_every=1500, rng_seed=new_n, verbose=False)
                check_update(a_out, f"md2_n{new_n}")

    # === Phase 6: Final coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coord polish (n={n_best}, {remaining:.0f}s) ===")
    t6 = min(remaining * 0.35, 200.0)
    if t6 > 10:
        a_out, _ = coord_polish(best_seq, t6, rng_seed=555, verbose=True)
        check_update(a_out, f"coord_n{n_best}")

    # === Phase 7: Final fine MD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final MD (n={n_best}, {remaining:.0f}s) ===")
    t7 = min(remaining * 0.35, 180.0)
    if t7 > 10:
        a_out, _ = mirror_optimize(best_seq, t7, lr_init=0.015, beta_init=8.0,
                                   top_k_init=150, restart_every=1000, rng_seed=777, verbose=False)
        check_update(a_out, f"final_md_n{n_best}")

    # === Phase 8: Final coord polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t8 = min(remaining * 0.5, 80.0)
        if t8 > 5:
            a_out, _ = coord_polish(best_seq, t8, rng_seed=888, verbose=False)
            check_update(a_out, f"final_coord_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = np.ones(5000) / 5000
        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")

    # Ensure we return with time to spare
    elapsed = time.monotonic() - t_global
    if elapsed > 1790:
        print(f"WARNING: Running out of time (t={elapsed:.0f}s)")

    return [float(x) for x in best_seq]
