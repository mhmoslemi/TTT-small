def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute the objective value (lower is better)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    C = fast_conv(a)
    return float(2 * len(a) * C.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_subgradient(a, C, K, beta):
    """Compute soft subgradient of max(C) using top-K lags."""
    n = len(a)
    k_max = min(K, len(C))
    top_idx = np.argsort(C)[::-1][:k_max]
    top_vals = np.maximum(C[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def lp_direction(a, C, K, M, solver):
    """Solve LP-FW direction: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    K_sel = min(K, 2 * n - 1)
    top_idx = np.argsort(C)[::-1][:K_sel]
    
    # Build constraint matrix T
    # T[k, i, j]... actually T is K_sel x n where
    # T[k, i] = sum over j of a[j] * a[k-j] partial... 
    
    # Actually, the constraint is: for each lag k, (f * g)[k] = sum_i a[i]*b[k-i] <= M
    # where f = a (fixed) and g = b (variable)
    # So T[k, i] = a[k-i] if 0 <= k-i < n, else 0
    
    T = np.zeros((K_sel, n))
    for ii, k in enumerate(top_idx):
        ki = int(k)
        for i in range(n):
            j = ki - i
            if 0 <= j < n:
                T[ii, i] = a[j]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

def make_init_geometric(n, r, rng=None):
    """Geometric decay initialization."""
    a = r ** np.arange(n)
    return normalize(a)


def make_init_shifted_gauss(n, center, width, rng=None):
    """Shifted Gaussian initialization."""
    x = np.arange(n) / n
    a = np.exp(-((x - center) / width) ** 2)
    a += 1e-4  # avoid exact zeros
    return normalize(a)


def make_init_two_bump(n, c1, c2, w1, w2, h2=0.5):
    """Two-bump asymmetric initialization."""
    x = np.arange(n) / n
    a = np.exp(-((x - c1) / w1) ** 2) + h2 * np.exp(-((x - c2) / w2) ** 2)
    a += 1e-4
    return normalize(a)


def make_init_powerlaw(n, p, side='left'):
    """Power-law decay."""
    if side == 'left':
        a = (n - np.arange(n)) ** p + 1e-4
    else:
        a = (np.arange(n) + 1) ** p + 1e-4
    return normalize(a)


def make_init_random_smooth(n, rng, smoothness=0.1):

    kernel /= kernel.sum()
    smoothed = np.convolve(noise, kernel, mode='same')
    a = np.exp(smoothed)

1. Diverse asymmetric initialization with short AMD screening
2. Embedded LP-FW + Away-Step Mirror Descent (AMDLP)
3. LP-dual informed soft subgradient for better non-LP iterations

5. Adaptive coordinate descent polish
"""

import numpy as np
import time
import cvxpy as cp

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS

def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute the objective value (lower is better)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    C = fast_conv(a)
    return float(2 * len(a) * C.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_subgradient(a, C, K, beta):
    """Compute soft subgradient of max(C) using top-K lags with temperature weighting."""
    n = len(a)
    k_max = min(K, len(C))
    top_idx = np.argsort(C)[::-1][:k_max]
    top_vals = np.maximum(C[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def lp_direction(a, C, K, M, solver):
    """Solve LP-FW direction: max sum(b) s.t. (a*b)[k] <= M for top-K lags, b >= 0."""
    n = len(a)
    K_sel = min(K, 2 * n - 1)
    top_idx = np.argsort(C)[::-1][:K_sel]

    T = np.zeros((K_sel, n))
    for ii, k in enumerate(top_idx):
        ki = int(k)
        for i in range(n):
            j = ki - i
            if 0 <= j < n:
                T[ii, i] = a[j]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None, None

    # Also extract dual variables for informed subgradient
    dual_vals = None
    try:
        dual_vals = np.maximum(np.asarray(constraints[0].dual.value, dtype=np.float64), 0.0)
    except Exception:
        pass

    return b_val / b_sum, dual_vals

        K_cur = min(K + int(150 * progress), 400)
        beta_cur = max(beta * (1.0 - 0.4 * progress), 2.5)
        lp_int_cur = max(40, int(lp_interval * (1.0 - 0.4 * progress)))

        C = fast_conv(x)
        M = C.max()

        if M < best_C - 1e-15:
            best_C = M
            best_a = x.copy()
            stall = 0
            gamma = min(gamma * 1.01, 12.0)
        else:
            stall += 1

        use_lp = (it % lp_int_cur == 0) and (stall > 15) and (n <= 7000)

        if use_lp:
            g_dir, dual_vals = lp_direction(x, C, K_cur, M, solver)
            if g_dir is not None:
                if dual_vals is not None and len(dual_vals) > 0:
                    dual_weights = dual_vals / max(dual_vals.sum(), 1e-15)
                best_alpha = 2.0 / (it + 2.0)
                best_trial = M
                for alpha_cand in [best_alpha, min(1.0, best_alpha * 2.0), 0.5, 0.3, 0.2, 0.1, 0.05]:
                    a_trial = (1.0 - alpha_cand) * x + alpha_cand * g_dir
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    mt = fast_conv(a_trial).max()
                    if mt < best_trial:
                        best_trial = mt
                        best_alpha = alpha_cand

                x_prev = x.copy()
                x = (1.0 - best_alpha) * x + best_alpha * g_dir
                x = np.maximum(x, 0.0)
                x = x / x.sum()
            else:
                grad = soft_subgradient(x, C, K_cur, beta_cur)
                eta = gamma / (np.sqrt(it) + 1.0)
                log_x = np.log(np.maximum(x, 1e-15)) - eta * grad
                log_x -= log_x.max()
                x_new = np.exp(log_x)
                x_new = np.maximum(x_new, 1e-30)
                x_new = x_new / x_new.sum()
                x_prev = x.copy()
                x = x_new
        else:
            beta_t = (it - 1.0) / (it + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            y = y / y.sum()

            C_y = fast_conv(y)
            if dual_weights is not None:
                grad = soft_subgradient_dual(y, C_y, K_cur, dual_weights, n)
            else:
                grad = soft_subgradient(y, C_y, K_cur, beta_cur)

            eta = gamma / (np.sqrt(it) + 1.0)
            log_y = np.log(y) - eta * grad
            log_y -= log_y.max()
            x_new = np.exp(log_y)
            x_new = np.maximum(x_new, 1e-30)
            x_new = x_new / x_new.sum()

            x_prev = x.copy()
            x = x_new

        if stall > 3000:
            stall = 0
            pert = rng.normal(0, 0.015, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x = x_r / x_r.sum()
            x_prev = x.copy()
            K = min(K + 30, 400)
            beta = max(beta - 0.5, 2.5)
            dual_weights = None

        if stall > 1500 and stall % 500 == 0:
            K = min(K + 15, 400)

        if verbose and it % 200 == 0:
            J = 2.0 * n * best_C
            print(f"    AMDLP it={it} ({elapsed:.0f}s): J={J:.6f}, K={K_cur}, stall={stall}")

    return best_a, best_C


def soft_subgradient_dual(a, C, K, dual_w, n):
    """Subgradient using LP-dual weights."""
    k_max = min(K, len(C))
    top_idx = np.argsort(C)[::-1][:k_max]
    grad = np.zeros(n)
    if len(dual_w) >= k_max:
        w = dual_w[:k_max]
        w_sum = w.sum()
        if w_sum > 1e-15:
            w = w / w_sum
        else:
            w = np.ones(k_max) / k_max
    else:
        top_vals = np.maximum(C[top_idx], 1e-30)
        w = top_vals ** 4.0
        w = w / w.sum()
    for wi, k in zip(w, top_idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def make_init_geometric(n, r):
    a = r ** np.arange(n)
    return normalize(a)


def make_init_shifted_gauss(n, center, width):
    x = np.arange(n) / n
    a = np.exp(-((x - center) / width) ** 2) + 1e-4
    return normalize(a)


def make_init_two_bump(n, c1, c2, w1, w2, h2):
    x = np.arange(n) / n
    a = np.exp(-((x - c1) / w1) ** 2) + h2 * np.exp(-((x - c2) / w2) ** 2) + 1e-4
    return normalize(a)


def make_init_powerlaw(n, p, side):
    if side == 'left':
        a = (n - np.arange(n)) ** p + 1e-4
    else:
        a = (np.arange(n) + 1) ** p + 1e-4
    return normalize(a)


def make_init_random_smooth(n, rng, smoothness=0.1):
    noise = rng.normal(0, smoothness, n)
    kernel = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)
    kernel /= kernel.sum()
    smoothed = np.convolve(noise, kernel, mode='same')
    a = np.exp(smoothed)
    return normalize(a)


def generate_candidates(parent, n, time_budget, solver, rng_seed=0):
    """Generate diverse candidates via short AMD runs."""
    t0 = time.monotonic()

    inits = []
    labels = []

    if parent is not None:
        inits.append(normalize(parent))
        labels.append("parent")
        inits.append(normalize(parent[::-1]))
        labels.append("parent_rev")

    for r in [0.92, 0.97]:
        inits.append(make_init_geometric(n, r))
        labels.append(f"geom_{r}")

    for c, w in [(0.1, 0.08), (0.2, 0.12), (0.8, 0.08), (0.3, 0.05)]:
        inits.append(make_init_shifted_gauss(n, c, w))
        labels.append(f"g_{c}_{w}")

    inits.append(make_init_two_bump(n, 0.12, 0.75, 0.06, 0.1, 0.4))
    labels.append("2b1")
    inits.append(make_init_two_bump(n, 0.25, 0.85, 0.08, 0.06, 0.6))
    labels.append("2b2")

    inits.append(make_init_powerlaw(n, 2.0, 'left'))
    labels.append("pw2")
    inits.append(make_init_powerlaw(n, 1.5, 'right'))
    labels.append("pw1.5r")

    rng = np.random.default_rng(rng_seed)
    for s in [0, 1]:
        inits.append(make_init_random_smooth(n, np.random.default_rng(rng_seed + s)))
        labels.append(f"rand_{s}")

    n_per = max(8.0, time_budget / len(inits))
    best_a = inits[0].copy()
    best_score = eval_score(best_a)

    for idx, (init, lbl) in enumerate(zip(inits, labels)):
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 5:
            break
        t_avail = min(n_per, remaining - 2)
        if t_avail < 5:
            break

        a_out, _ = amdlp_optimize(init, t_avail, solver, verbose=False, rng_seed=rng_seed + idx)
        sc = eval_score(a_out)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            print(f"    Cand '{lbl}': J={sc:.6f} (t={time.monotonic()-t0:.0f}s)")

    return best_a, best_score


def explore_lengths(best_a, best_score, time_budget, solver, verbose=False):
    """Try different sequence lengths."""
    t0 = time.monotonic()
    n_best = len(best_a)

    deltas = [300, -300, 600, -600, 1000, -1000, 1500]

    for delta in deltas:
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 25:
            break
        new_n = n_best + delta
        if new_n < 400 or new_n > 9000:
            continue

        old_x = np.linspace(0, 1, len(best_a))
        new_x = np.linspace(0, 1, new_n)
        a_interp = np.interp(new_x, old_x, best_a)
        a_interp = normalize(a_interp)

        sc = eval_score(a_interp)
        if sc < best_score:
            t_opt = min(remaining * 0.4, 50.0)
            a_out, _ = amdlp_optimize(a_interp, t_opt, solver, verbose=False, rng_seed=new_n % 1000)
            sc = eval_score(a_out)
            a_interp = a_out

        if sc < best_score:
            best_score = sc
            best_a = a_interp.copy()
            print(f"  Len {new_n}: J={sc:.6f} (improved!)")
        elif verbose:
            print(f"  Len {new_n}: J={sc:.6f}")

    return best_a, best_score


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Adaptive coordinate descent for final polishing."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_C = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 60

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        C = fast_conv(a)
        top_k = min(15, len(C))
        top_idx = np.argsort(C)[::-1][:top_k]

        improved = False

        for k in top_idx[:6]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[ki - idx]

            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            sens = np.zeros(n)
            for kk in top_idx[:4]:
                kki = int(kk)
                s = max(0, kki - n + 1)
                e = min(n, kki + 1)
                idx2 = np.arange(s, e)
                sens[idx2] += a[kki - idx2]

            low_sens = np.argsort(sens)[:30]
            candidates = [j for j in low_sens if j != src][:8]

            found = False
            for dst in candidates:
                if found:
                    break
                for frac in [0.003, 0.008, 0.015, 0.03, 0.06]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    a_t = a_t / a_t.sum()
                    Mt = fast_conv(a_t).max()
                    if Mt < best_C - 1e-16:
                        a = a_t
                        best_C = Mt
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

            if not found and stall > 20:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.008)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    Mt = fast_conv(a).max()
                    if Mt < best_C:
                        best_C = Mt
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.008)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                Mt = fast_conv(a).max()
                if Mt < best_C:
                    best_C = Mt
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 15 == 0:
            J = 2.0 * n * best_C
            print(f"    Coord ({time.monotonic()-t0:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_C


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with embedded LP-FW and diverse initialization."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def _update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = eval_score(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = eval_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5500
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = normalize(best_seq)
        best_score = eval_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Diverse candidate generation ===
    print(f"\n=== Phase 1: Diverse candidates (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)

    if t1 > 30:
        a_cand, sc_cand = generate_candidates(best_seq, n0, t1, solver, rng_seed=7)
        _update(a_cand, "candidates")

    # === Phase 2: Main AMDLP optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Main AMDLP (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.38, 680.0)

    if t2 > 30:
        a_in = normalize(best_seq)
        a_out, _ = amdlp_optimize(a_in, t2, solver, verbose=True, rng_seed=42)
        _update(a_out, "amdlp_main")

    # === Phase 3: Second AMDLP round ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: AMDLP r2 (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.20, 350.0)

    if t3 > 30:
        a_in = normalize(best_seq)
        a_out, _ = amdlp_optimize(a_in, t3, solver, verbose=True, rng_seed=99)
        _update(a_out, "amdlp_r2")

    # === Phase 4: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Length exploration ===")
    t4 = min(remaining * 0.12, 180.0)

    if t4 > 25:
        a_len, sc_len = explore_lengths(best_seq, best_score, t4, solver, verbose=True)
        _update(a_len, f"len_{len(a_len)}")

    # === Phase 5: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Coordinate polish (n={len(best_seq)}) ===")
    t5 = min(remaining * 0.10, 130.0)

    if t5 > 15:
        a_in = normalize(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True, rng_seed=456)
        _update(a_out, "coord")

    # === Phase 6: Final AMDLP ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final AMDLP (n={len(best_seq)}) ===")
    t6 = min(remaining * 0.40, 250.0)

    if t6 > 20:
        a_in = normalize(best_seq)
        a_out, _ = amdlp_optimize(a_in, t6, solver, verbose=True, rng_seed=777)
        _update(a_out, "amdlp_final")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 7: Final polish ===")
        t7 = min(remaining * 0.7, 60.0)
        if t7 > 10:
            a_in = normalize(best_seq)
            a_out, _ = coordinate_polish(a_in, t7, verbose=False, rng_seed=789)
            _update(a_out, "coord_final")

    # Final validation
    if best_seq is not None:
        fs = eval_score(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        validate_score = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(validate_score):
            best_score = min(best_score, validate_score)
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5500
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = normalize(best_seq)
        best_score = eval_score(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
