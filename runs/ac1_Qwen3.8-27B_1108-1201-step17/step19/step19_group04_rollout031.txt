"""
Search for non-negative step-function heights minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Uses a fast FFT-based pressure mirror descent:
1. Compute self-convolution and identify hot lags via soft weighting.
2. Compute per-position pressure via paired FFT cross-correlation + convolution.
3. Apply exponential reweighting to reduce mass at high-pressure positions.
4. Grid line search over reweighting strength.
5. Diverse structured starts, adaptive restarts, and length tuning.
"""

import math
import time
import numpy as np


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]


def conv_ab(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]


def objective_score(a):
    q = self_conv(a)
    return 2.0 * float(a.size) * float(np.max(q))


def evaluate_safe(a):
    try:
        x = [float(v) for v in np.asarray(a, dtype=np.float64)]
        val = evaluate_sequence(x)
        if np.isfinite(val):
            return float(val)
    except Exception:
        pass
    return float('inf')


def compute_pressure(a, q, alpha):
    n = a.size
    m = q.size
    M = float(np.max(q))
    if M <= 1e-15:
        return np.ones(n, dtype=np.float64)
    ratio = np.maximum(q, 0.0) / M
    w = ratio ** alpha
    size = next_pow2(m + n)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    cross = np.fft.irfft(fw * fa.conj(), size)[:n]
    convol = np.fft.irfft(fw * fa, size)[:n]
    P = cross + convol
    return np.maximum(P, 0.0)


def mirror_step(a, P, eta):
    Pmax = float(np.max(P))
    if Pmax <= 1e-15:
        return a.copy()
    scaled = eta * P / Pmax
    scaled = np.clip(scaled, -50.0, 50.0)
    new_a = a * np.exp(-scaled)
    new_a = np.maximum(new_a, 0.0)
    s = float(new_a.sum())
    if s <= 1e-15:
        return a.copy()
    return new_a / s


def local_search_pressure(a0, time_limit, alpha=15.0, verbose=False):
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 2:
        return a

    best = a.copy()
    best_score = objective_score(a)
    eta = 1.0
    stall = 0
    rng = np.random.default_rng(42)
    iters = 0

    eta_grid = [0.1, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0, 12.0]

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        q = self_conv(a)
        M_cur = float(np.max(q))
        score_cur = 2.0 * n * M_cur

        if score_cur < best_score - 1e-13:
            best = a.copy()
            best_score = score_cur
            stall = 0
        else:
            stall += 1

        if stall > 300:
            break

        if stall > 30 and stall % 30 == 0:
            noise_scale = 0.005 * (1.0 + 0.1 * (stall // 30))
            pert = a * (1.0 + noise_scale * rng.normal(size=n))
            pert = np.maximum(pert, 1e-12)
            pert = normalize_vec(pert)
            s_pert = objective_score(pert)
            if s_pert < best_score + 0.001:
                a = pert
                stall = 0
                eta = 1.0
            elif stall > 200:
                a = best.copy()
                stall = 0
                eta = 1.0

        P = compute_pressure(a, q, alpha)
        Pmax = float(np.max(P))
        if Pmax <= 1e-15:
            break

        best_eta = eta
        best_new_score = score_cur
        best_new_a = a

        for e in eta_grid:
            if time.monotonic() - t0 > time_limit:
                break
            candidate = mirror_step(a, P, e)
            sc = objective_score(candidate)
            if sc < best_new_score - 1e-14:
                best_new_score = sc
                best_eta = e
                best_new_a = candidate

        if best_new_score < score_cur - 1e-14:
            a = best_new_a
            eta = best_eta
            iters += 1
        else:
            stall += 1
            eta = max(0.1, eta * 0.8)

        if verbose and iters > 0 and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    pressure_ls: it=%d elapsed=%.1fs best=%.9f eta=%.2f stall=%d"
                  % (iters, elapsed, best_score, eta, stall))

    return best


def spectral_equalize(a, iters=50, verbose=False):
    """Try to flatten the autocorrelation by adjusting in frequency domain."""
    a = normalize_vec(a)
    n = a.size
    best = a.copy()
    best_score = objective_score(a)

    for it in range(iters):
        size = next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, size)
        S = fa * fa.conj()
        S = np.abs(S)
        Smax = float(np.max(S))
        if Smax <= 1e-15:
            break
        S_norm = S / Smax
        # Reduce peaks in the periodogram
        S_flat = np.maximum(S, 0.3 * Smax)
        new_fa = fa * np.sqrt(S_flat / np.maximum(S, 1e-300))
        new_a = np.fft.irfft(new_fa, size)[:n]
        new_a = np.maximum(new_a, 0.0)
        s = float(new_a.sum())
        if s <= 1e-15:
            break
        new_a = new_a / s
        sc = objective_score(new_a)
        if sc < best_score - 1e-13:
            best = new_a.copy()
            best_score = sc
            a = new_a
        else:
            # Try blending
            for w in (0.1, 0.3, 0.5):
                blend = (1 - w) * a + w * new_a
                blend = np.maximum(blend, 0.0)
                sb = float(blend.sum())
                if sb > 1e-15:
                    blend /= sb
                    scb = objective_score(blend)
                    if scb < best_score - 1e-13:
                        best = blend.copy()
                        best_score = scb
                        a = blend
                        break

    return best


def diverse_starts(n, parent):
    out = []
    x = np.linspace(0.0, 1.0, n)

    if parent is not None:
        p = parent
        if p.size != n:
            p = np.interp(x, np.linspace(0, 1, p.size), p)
        p = normalize_vec(p)
        out.append(p.copy())
        out.append(p[::-1].copy())
        for w in (0.2, 0.5, 0.8):
            out.append(normalize_vec((1 - w) * p + w * p[::-1]))
    else:
        out.append(np.ones(n, dtype=np.float64) / float(n))

    # Sinc-like
    for fc in (1.0, 1.5, 2.0, 3.0):
        seq = np.sinc(fc * (x - 0.5)) ** 2
        seq = np.maximum(seq, 0.0)
        out.append(normalize_vec(seq))

    # Raised cosine
    for beta in (0.5, 1.0, 2.0, 4.0):
        seq = np.cos(np.pi * (x - 0.5)) ** (2 * beta)
        seq = np.maximum(seq, 0.0)
        out.append(normalize_vec(seq))

    # Polynomial shapes
    for p_exp in (0.5, 1.0, 1.5, 2.0, 3.0, 5.0):
        out.append(normalize_vec((x + 1e-8) ** p_exp))
        out.append(normalize_vec(((1 - x) + 1e-8) ** p_exp))

    # Product of powers
    for a_e in (0.5, 1.0, 2.0):
        for b_e in (0.5, 1.0, 2.0):
            seq = (x + 1e-8) ** a_e * ((1 - x) + 1e-8) ** b_e
            out.append(normalize_vec(seq))

    # Gaussian bumps
    for ctr in (0.2, 0.3, 0.5, 0.7, 0.8):
        for wd in (0.05, 0.1, 0.15, 0.2, 0.3):
            seq = np.exp(-((x - ctr) ** 2) / (2 * wd * wd))
            out.append(normalize_vec(seq))

    # Double bumps
    for c1, c2 in ((0.3, 0.7), (0.25, 0.75), (0.2, 0.8)):
        for wd in (0.08, 0.12, 0.2):
            seq = (np.exp(-((x - c1) ** 2) / (2 * wd * wd)) +
                   np.exp(-((x - c2) ** 2) / (2 * wd * wd)))
            out.append(normalize_vec(seq))

    # Random smooth
    rng = np.random.default_rng(12345)
    for _ in range(10):
        nbumps = int(rng.integers(2, 6))
        seq = np.zeros(n)
        for _ in range(nbumps):
            ctr = rng.uniform(0.1, 0.9)
            wd = rng.uniform(0.03, 0.3)
            ht = rng.uniform(0.3, 1.0)
            seq += ht * np.exp(-((x - ctr) ** 2) / (2 * wd * wd))
        seq += 1e-8
        out.append(normalize_vec(seq))

    # Frequency-domain starts (flat-ish spectrum)
    rng2 = np.random.default_rng(777)
    for _ in range(5):
        size = next_pow2(2 * n - 1)
        fa = rng2.standard_normal(size // 2 + 1).astype(np.complex128)
        fa[0] = float(n)  # DC component for positive sum
        fa = fa + fa.conj()[::-1] * 0  # keep real-valued
        a_real = np.fft.irfft(fa, size)[:n]
        a_real = np.maximum(a_real, 0.0)
        s = float(a_real.sum())
        if s > 1e-10:
            out.append(a_real / s)

    return out


def reduce_diverse(cands, max_keep=6):
    scored = []
    for c in cands:
        c = normalize_vec(c)
        sc = objective_score(c)
        if np.isfinite(sc):
            scored.append((sc, c))
    scored.sort(key=lambda x: x[0])
    kept = []
    for sc, seq in scored:
        ok = True
        for kseq in kept:
            if seq.size == kseq.size:
                sim = float(np.dot(seq, kseq)) / (float(np.dot(seq, seq)) * float(np.dot(kseq, kseq))) ** 0.5
                if sim > 0.995:
                    ok = False
                    break
        if ok:
            kept.append(seq)
        if len(kept) >= max_keep:
            break
    return kept

def swap_polish(a0, time_limit, rng_seed=271):
    """Try small mass transfers between positions to reduce the peak."""
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    best = a.copy()
    best_score = objective_score(a)
    stall = 0
    rng = np.random.default_rng(rng_seed)

    while time.monotonic() - t0 < time_limit and stall < 100:
        q = self_conv(a)
        M = float(np.max(q))
        active = np.where(q > M * 0.95)[0]

        pressure = np.zeros(n)
        for k in active[:10]:
            ki = int(k)
            for j in range(n):
                jk = ki - j
                if 0 <= jk < n:
                    pressure[j] += a[j] * a[jk]
                jk2 = j - ki
                if 0 <= jk2 < n:
                    pressure[j] += a[j] * a[jk2]

        # Move from high pressure to low pressure
        src = np.argmin(pressure)
        dst = np.argmax(pressure)
        if a[int(src)] > 1e-10 and pressure[int(dst)] > pressure[int(src)] + 1e-10:
            for frac in (0.01, 0.05, 0.1):
                b = a.copy()
                amt = a[int(dst)] * frac
                b[int(dst)] -= amt
                b[int(src)] += amt
                b = np.maximum(b, 0)
                s = float(b.sum())
                if s > 1e-15:
                    b /= s
                    sc = objective_score(b)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best = b.copy()
                        a = b
                        stall = 0
                        break
                stall += 1
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                b = a.copy()
                amt = a[i] * 0.02
                b[i] -= amt
                b[j] += amt
                b = np.maximum(b, 0)
                s = b.sum()
                if s > 1e-15:
                    b /= s
                    sc = objective_score(b)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best = b.copy()
                        a = b
                        stall = 0

    return best

            out = transform_search(best_seq, t_tf)
            update_best_new(out, "transform1")

    # Try different starting structures at same n
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 200:
        print("Phase 3: diverse starts at n=%d, budget=%.0fs" % (n_cur, remaining))
        starts = diverse_starts(n_cur, parent)
        kept = reduce_diverse(starts, max_keep=5)
        # Ensure current best is included
        matched = False
        for k in kept:
            if k.size == best_seq.size and float(np.dot(k, best_seq)) / (float(np.dot(k, k)) * float(np.dot(best_seq, best_seq)))**0.5 > 0.999:
                matched = True
                break
        if not matched:
            kept.insert(0, best_seq.copy())

        for idx, start_seq in enumerate(kept[:4]):
            remaining = total_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t_d = min(180.0, 0.25 * remaining)
            if t_d < 30:
                continue
            print("  diverse start %d, budget=%.0fs" % (idx, t_d))
            out = local_search_pressure(start_seq, t_d, alpha=12.0 + 3 * idx, verbose=False)
            update_best_new(out, "diverse%d_n%d" % (idx, n_cur))

    # Try different lengths
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 200:
        cur_n = int(best_seq.size)
        trial_ns = []
        for dn in (-500, -1000, 500, 1000, 1500, 2000):
            nn = cur_n + dn
            if 4000 <= nn <= 10000 and nn != cur_n:
                trial_ns.append(nn)
        seen = set()
        unique_ns = []
        for nn in trial_ns:
            if nn not in seen:
                seen.add(nn)
                unique_ns.append(nn)

        print("Phase 4: try different lengths: %s" % unique_ns[:4])
        for nn in unique_ns[:3]:
            remaining = total_budget - (time.monotonic() - t_global)
            if remaining < 100:
                break
            x_old = np.linspace(0, 1, cur_n)
            x_new = np.linspace(0, 1, nn)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = normalize_vec(a_interp)
            t_l2 = min(200.0, 0.22 * remaining)
            if t_l2 < 40:
                continue
            print("  n=%d, budget=%.0fs" % (nn, t_l2))
            out = local_search_pressure(a_interp, t_l2, alpha=15.0, verbose=True)
            update_best_new(out, "len_n%d" % nn)

    # Trim
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 50:
        t_tr = min(80.0, 0.12 * remaining)
        if t_tr > 15:
            out = trim_search(best_seq, t_tr)
            update_best_new(out, "trim")

    # Final pressure polish
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 60:
        t_fp = min(200.0, 0.30 * remaining)
        if t_fp > 30:
            print("Phase 5: final pressure polish, budget=%.0fs" % t_fp)
            out = local_search_pressure(best_seq, t_fp, alpha=25.0, verbose=True)
            update_best_new(out, "final_pressure")

    # Transform again
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t_t2 = min(50.0, 0.5 * remaining)
        if t_t2 > 10:
            out = transform_search(best_seq, t_t2)
            update_best_new(out, "transform2")

    # Swap polish
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 50:
        t_sp2 = min(100.0, 0.4 * remaining)
        if t_sp2 > 20:
            out = swap_polish(best_seq, t_sp2)
            update_best_new(out, "swap")

    # Final validation
    if best_seq is None:
        x = np.linspace(0, 1, 6700)
        best_seq = normalize_vec(np.sinc(2 * (x - 0.5)) ** 2)

    out_arr = np.asarray(best_seq, dtype=np.float64)
    out_arr = np.where(np.isfinite(out_arr), out_arr, 0.0)
    out_arr = np.maximum(out_arr, 0.0)
    out_arr = np.minimum(out_arr, 1000.0)
    s = float(out_arr.sum())
    if s <= 0.01:
        x = np.linspace(0, 1, 6700)
        out_arr = np.sinc(2 * (x - 0.5)) ** 2
        s = float(out_arr.sum())
    out_arr = out_arr / s

    final_score = objective_score(out_arr)
    official = evaluate_safe(out_arr)
    if np.isfinite(official):
        final_score = official

    elapsed = time.monotonic() - t_global
    print("FINAL: score=%.10f n=%d time=%.1fs" % (final_score, out_arr.size, elapsed))
    return [float(v) for v in out_arr]
