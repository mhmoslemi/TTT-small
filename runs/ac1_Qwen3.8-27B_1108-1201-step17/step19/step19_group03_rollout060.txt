Let me write the LP-FW more carefully:

The LP-FW algorithm:
- Given current point a (on the simplex), compute autocorrelation b = conv(a,a).
- Select top K lags where b_k is largest.
- Solve LP: max sum(b) subject to T @ b ≤ M, b ≥ 0, where T is the constraint matrix for the selected lags and M = max(b).
- The LP solution gives a "vertex" direction g.
- Update: a ← (1-α)a + αg, where α is chosen by line search.

The key improvement: use a proper line search to find the best α.

For the line search, we want to minimize max_k ((1-α)a + αg) * ((1-α)a + αg) over k, subject to (1-α)a + αg ≥ 0.

This is a 1D optimization problem. We can use bisection or golden section search.

Actually, the max of a set of quadratic functions in α is a piecewise quadratic function. We can find the minimum by:
1. Identifying the "active" lags (those that achieve the max).
2. Finding the α where two adjacent active lags cross.
3. Evaluating at these candidate points.

But this is complex. A simpler approach: use a few trial values of α (e.g., 0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0) and pick the best.

Or even simpler: use the standard FW step α = 2/(t+2) and also try a few multiples.

Let me also think about the "peak shaving" approach:

Given the current sequence a and the peak lag k*:
- The peak is b_{k*} = sum_j a_j * a_{k*-j}.
- To reduce b_{k*}, we can:
  a) Reduce a_j for some j that contributes significantly to the peak.
  b) Move the "mass" to a position j' that doesn't contribute to the peak (or contributes less).

The sensitivity of b_k to a change in a_i is:
∂b_k/∂a_i = a_{k-i} (for valid k-i)
So the total sensitivity (across all peaks) is:
S_i = sum_{k in top_lags} w_k * a_{k-i}

We want to move mass from high-sensitivity positions to low-sensitivity positions.

def pocs_solve(a0, time_limit, verbose=False, n_iters=100000, rng_seed=0, omega=1.3):
    """POCS with over-relaxation toward flat spectrum."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    fft_size = 1 << (2 * n - 2).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        target = np.median(mag[mag > 1e-30]) if np.any(mag > 1e-30) else 1.0
        new_mag = mag + omega * (target - mag)
        new_mag = np.maximum(new_mag, 1e-30)
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        m = fast_autocorr(a_new).max()
        if m < best_m - 1e-15:
            best_m = m
            best_a = a_new.copy()
        a = a_new

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K_init=200, top_K_max=None):
    """LP-Frank-Wolfe with geometric line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 5:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_rows = []
        for k in binding_idx:
            ki = int(k)
            row = np.zeros(n)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                row[j] = a[ki - j]
            T_rows.append(row)
        T_sel = np.array(T_rows)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 20, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 20, top_K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Geometric line search
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = fast_autocorr(a).max()
        alphas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, best_alpha]
        alphas = [min(1.0, max(0.001, al)) for al in alphas]
        alphas = sorted(set(alphas))
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        iter_count += 1

        if iter_count % 5 == 0:
            K = min(K + 40, top_K_max)

        if no_improve > 30:
            no_improve = 0
            K = min(K + 50, top_K_max)

        if verbose and iter_count % 10 == 0:

            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

    return best_a, best_m


def peak_shave(a0, time_limit, verbose=False, rng_seed=42):
    """Peak-shaving: move mass from high-sensitivity to low-sensitivity positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(30, 2 * n - 1)
        top_k = np.argsort(ac)[::-1][:n_top]
        top_vals = np.maximum(ac[top_k], 1e-30)
        w = top_vals ** 4.0
        w = w / w.sum()

        # Compute sensitivity: how much each a_i contributes to the peaks
        sens = np.zeros(n)
        for wi, k in zip(w, top_k):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                sens[j] += wi * a[ki - j]

        improved = False
        # Try moving mass from high-sensitivity to low-sensitivity positions
        high_sens_idx = np.argsort(sens)[::-1][:10]
        low_sens_idx = np.argsort(sens)[:10]

        for src in high_sens_idx:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue
            for dst in low_sens_idx:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Random perturbation
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def mirror_descent_solve(a0, time_limit, verbose=False, rng_seed=100, c0=5.0, top_k_init=50, beta_init=5.0):

    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    cur_top_k = top_k_init
    cur_beta = beta_init
    restart_every = 5000

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)

            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            g[idx] += 2.0 * wi * a[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(y, 1e-15)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 200)
            cur_beta = max(cur_beta - 1.0, 3.0)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
