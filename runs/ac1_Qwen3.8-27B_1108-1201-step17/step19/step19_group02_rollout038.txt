def lp_fw(a0, time_limit, solver, initial_K=400, max_K=3000, add_per_iter=40, verbose=True):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    ac_y = fast_convolve(y)
    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())
    
    no_improve = 0
    max_no_improve = 2000
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            break
        iter_count += 1
        
        ac = fast_convolve(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Manage active set
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set if k < len(ac)])
            valid_idx = [i for i, k in enumerate(active_set) if k < len(ac)]
            keep = [valid_idx[i] for i in np.argsort(margins)[:max_K]]
            active_set = sorted([active_set[i] for i in keep])
        
        K = len(active_set)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(active_set):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = sorted(active_set[:initial_K])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Check for violations and add new constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            new_set = set(active_set)
            for k in violations[:add_per_iter]:
                new_set.add(int(k))
            active_set = sorted(new_set)
        
        # Step size selection
        default_alpha = 2.0 / (iter_count + 2.0)
        alphas = sorted(set([
            max(0.01, min(1.0, default_alpha)),
            max(0.01, min(1.0, default_alpha * 0.5)),
            max(0.01, min(1.0, default_alpha * 2.0)),
            max(0.01, min(1.0, default_alpha * 0.25)),
            max(0.01, min(1.0, default_alpha * 4.0)),

def scipy_polish(a0, time_limit, verbose=True):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    
    def objective(x):
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            return 1e10
        x = x / s
        return fast_convolve(x).max()
    
    def gradient(x):
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            return np.zeros_like(x)
        x = x / s
        ac = fast_convolve(x)
        max_val = ac.max()
        # Subgradient: average over all max positions
        max_mask = ac >= max_val - 1e-12
        max_indices = np.where(max_mask)[0]
        
        grad = np.zeros(n)
        for k in max_indices:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += 2.0 * x[k_int - idx]
        grad /= len(max_indices)
        return grad
    
    constraints = [
        {'type': 'eq', 'fun': lambda x: x.sum() - 1.0, 'jac': lambda x: np.ones(n)}
    ]
    bounds = [(0.0, None)] * n
    
    result = minimize(objective, a, method='SLSQP', jac=gradient,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 200, 'ftol': 1e-14})
    
    if result.success or (result.fun < best_m):
        a_new = normalize(result.x)
        m_new = fast_convolve(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new
    
    return best_a, best_m

def projected_gradient(a0, time_limit, verbose=True):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    
    lr = 0.01
    momentum = 0.9
    vel = np.zeros(n)
    stall = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_convolve(a)
        max_val = ac.max()
        max_mask = ac >= max_val - 1e-12
        max_indices = np.where(max_mask)[0]
        
        # Subgradient
        grad = np.zeros(n)
        for k in max_indices:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += 2.0 * a[k_int - idx]
        grad /= len(max_indices)
        
        # Projected gradient step with momentum
        eta = lr / (1.0 + 0.001 * t)
        y = a + momentum * vel - eta * grad
        y = np.maximum(y, 0.0)
        s = y.sum()
        if s < 1e-15:
            y = a.copy()
        else:
            y = y / s
        vel = momentum * vel + (y - a)
        a = y
        
        m_new = fast_convolve(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 1000:
            break
    
    return best_a, best_m

def parametric_blend(a0, time_limit):
    """Try blending the current sequence with various parametric shapes."""
    n = len(a0)

    # U-shape (high at edges)

        shapes.append(u / u.sum())
    # Inverted U (peaked in middle)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        inv = 1.0 - (np.abs(x - 0.5) * 2.0) ** p
        inv = np.maximum(inv, 0.01)

        wshp = np.maximum(wshp, 0.01)

        for ph in [0, np.pi]:

            c = np.maximum(c, 0.01)
            shapes.append(c / c.sum())
    # Power-law edges
    for p in [1.0, 2.0, 3.0, 5.0, 8.0]:
        edge = np.maximum(i_arr + 1, 0.5) ** (-p) + np.maximum(n - i_arr, 0.5) ** (-p)
        shapes.append(edge / edge.sum())

        if time.monotonic() - t0 > time_limit:

    ac_y = fast_convolve(y)
    active_set = sorted(np.argsort(ac_y)[::-1][:min(initial_K, 2 * n - 1)].tolist())

    no_improve = 0
    max_no_improve = 3000
    iter_count = 0
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: stagnation at iter " + str(iter_count))
            break
        iter_count += 1

        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic active set refresh to avoid stale constraints
        if iter_count % 400 == 0:
            active_set = sorted(np.argsort(ac)[::-1][:min(max_K, 2 * n - 1)].tolist())

        # Trim active set to most binding constraints
        if len(active_set) > max_K:
            valid = [k for k in active_set if 0 <= k < len(ac)]
            if len(valid) <= max_K:
                active_set = sorted(valid)
            else:
                margins = np.array([M_curr - ac[k] for k in valid])
                keep_idx = np.argsort(margins)[:max_K]
                active_set = sorted([valid[i] for i in keep_idx])

        K = len(active_set)
        if K == 0:
            break

        coeffs = build_coeff_matrix(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if K > initial_K:
                active_set = sorted(active_set[:initial_K])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations and add new constraints
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)

        # Adaptive step size: decrease over time for precision
        base_alpha = 2.0 / (iter_count + 2.0)
        if iter_count > 1000:
            base_alpha = min(base_alpha, 0.002)
        elif iter_count > 500:
            base_alpha = min(base_alpha, 0.005)
        elif iter_count > 200:
            base_alpha = min(base_alpha, 0.01)
        elif iter_count > 50:
            base_alpha = min(base_alpha, 0.03)

        alphas = [base_alpha, base_alpha * 0.5, base_alpha * 2.0, base_alpha * 0.25, base_alpha * 4.0]
        alphas = sorted(set([max(0.001, min(1.0, a)) for a in alphas]))

        best_alpha = base_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 3:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and iter_count % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", |S|=" + str(len(active_set)))

    elapsed = time.monotonic() - t0
    if verbose:
        J = 2.0 * n * best_m
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", iters=" + str(iter_count))
    return best_a, best_m


def laplacian_smooth(a0, time_limit, verbose=True):
    """Apply Laplacian smoothing to reduce convolution peaks."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 500:
            break

        # Compute Laplacian (with boundary handling)
        lap = np.zeros(n)
        lap[1:-1] = a[:-2] - 2.0 * a[1:-1] + a[2:]
        lap[0] = a[1] - a[0]
        lap[-1] = a[-2] - a[-1]

        # Try multiple step sizes
        improved = False
        for eps in [0.001, 0.002, 0.005, 0.01, 0.02]:
            trial = a + eps * lap
            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s < 1e-15:
                continue
            trial = trial / s
            m = fast_convolve(trial).max()
            if m < best_m - 1e-15:
                a = trial
                best_m = m
                best_a = a.copy()
                stall = 0
                improved = True
                break

        if not improved:
            stall += 1

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Smooth iter=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

    return best_a, best_m


def peak_squeeze(a0, time_limit, verbose=True):
    """Targeted mass transfer to reduce the top convolution peaks."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 1000:
        ac = fast_convolve(a)
        top_k = np.argsort(ac)[::-1][:15]
        max_val = ac[top_k[0]]
        improved = False

        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contribs = np.array([a[i] * a[k_int - i] for i in idx])
            if contribs.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contribs)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score destinations: low contribution to all top peaks
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            candidates = np.argsort(scores)[:min(n // 2, 2000)]

            for dst in candidates[:30]:
                dst = int(dst)
                if dst == src:
                    continue
                for frac in [0.003, 0.001, 0.005, 0.0005]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        break
                    trial = a.copy()
                    trial[src] -= amt
                    trial[dst] += amt
                    trial = np.maximum(trial, 0.0)
                    s_t = trial.sum()
                    if s_t < 1e-15:
                        break
                    trial = trial / s_t
                    M_trial = fast_convolve(trial).max()
                    if M_trial < best_m - 1e-16:
                        a = trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 200 == 0:
                # Random perturbation to escape
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.005)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 300 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Squeeze (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", stall=" + str(stall))

    return best_a, best_m

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Informed initialization from parent + shape blending

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):

    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

def parametric_blend(a0, time_limit):
    n = len(a0)
    t0 = time.monotonic()
    a = normalize(a0)
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)

    shapes = []
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        u = (np.abs(x - 0.5) * 2.0) ** p
        shapes.append(u / u.sum())
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        inv = 1.0 - (np.abs(x - 0.5) * 2.0) ** p
        inv = np.maximum(inv, 0.01)
        shapes.append(inv / inv.sum())
    for w in [0.25, 0.3, 0.4]:
        wshp = (np.abs(x - w) - 0.5) ** 2 + (np.abs(x - (1.0 - w)) - 0.5) ** 2 + (1.0 - np.abs(x - 0.5)) ** 2
        wshp = np.maximum(wshp, 0.01)
        shapes.append(wshp / wshp.sum())
    for freq in [1, 2, 3]:
        for ph in [0.0, np.pi]:
            c = 1.0 + 0.5 * np.cos(2.0 * np.pi * freq * x + ph)
            c = np.maximum(c, 0.01)
            shapes.append(c / c.sum())
    for p in [1.0, 2.0, 3.0, 5.0, 8.0]:
        edge = np.maximum(i_arr + 1.0, 0.5) ** (-p) + np.maximum(n - i_arr, 0.5) ** (-p)
        shapes.append(edge / edge.sum())

    for shape in shapes:
        if time.monotonic() - t0 > time_limit:
            break
        for blend in [0.05, 0.1, 0.2, 0.3, 0.5]:
            trial = (1.0 - blend) * a + blend * shape
            trial = normalize(trial)
            m = fast_convolve(trial).max()
            if m < best_val:
                best_val = m
                best_a = trial.copy()

    return best_a


def build_coeff_matrix(y, active_set, n):
    K = len(active_set)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(active_set):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs
