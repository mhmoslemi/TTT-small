import numpy as np
import time

# --- FFT-based fast convolution ---

def fast_convolve(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

# --- Utilities ---

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def interpolate(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = normalize(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f = np.interp(new_x, old_x, a)
    return normalize(f)

def evaluate_safe(a):
    """Evaluate the sequence safely."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

# --- Exact line search for Frank-Wolfe ---

def exact_line_search(a, d, ac_a, verbose=False):
    """
    Find t in [0,1] that minimizes max_k conv((1-t)*a + t*d, (1-t)*a + t*d).
    Uses binary search on the peak value M.
    
    a: current point (sum=1, non-negative)
    d: direction (sum=1, non-negative), step is a + t*(d-a)
    ac_a: convolution of a with a (precomputed)
    """
    n = len(a)
    # Precompute conv(d, d) and conv(a, d)
    ac_d = fast_convolve(d)
    ac_ad = fast_convolve(a)  # this is ac_a
    # Actually need cross term: conv(a, d)
    # conv(a+t*d, a+t*d) = conv(a,a) + 2t*conv(a,d) + t^2*conv(d,d)
    # But conv(a,d) != conv(d,a) in general? No, it is: conv(a,d)_k = sum a_i d_{k-i} = conv(d,a)_k
    fft_size = 1 << (2*n - 2).bit_length()
    Af = np.fft.rfft(a, fft_size)
    Df = np.fft.rfft(d, fft_size)
    ac_cross = np.fft.irfft(Af * Df, fft_size)[:2*n-1]
    
    # f_k(t) = ac_a[k] + 2*t*ac_cross[k] + t^2*ac_d[k]
    # We want min_t max_k f_k(t) for t in [0,1]
    
    def peak_val(t):
        vals = ac_a + 2*t*ac_cross + t*t*ac_d
        return vals.max()
    
    # Binary search on M: find smallest M such that there exists t in [0,1] with all f_k(t) <= M
    # For a given M, f_k(t) <= M means: ac_d[k]*t^2 + 2*ac_cross[k]*t + (ac_a[k]-M) <= 0
    # This is a quadratic in t. For it to be <= 0, t must be between the roots (if they exist).
    # The feasible set is the intersection of all such intervals, intersected with [0,1].
    
    def feasible_M(M):
        """Check if there exists t in [0,1] with all f_k(t) <= M."""
        t_low = 0.0
        t_high = 1.0
        for k in range(len(ac_a)):
            A = ac_d[k]
            B = 2.0 * ac_cross[k]
            C = ac_a[k] - M
            # A*t^2 + B*t + C <= 0
            if A < 1e-20:
                # Linear: B*t + C <= 0
                if abs(B) < 1e-20:
                    if C > 1e-20:
                        return False
                elif B > 0:
                    t_low = max(t_low, -C / B)
                else:
                    t_high = min(t_high, -C / B)
            else:
                disc = B*B - 4*A*C
                if disc < -1e-15:
                    # No real roots. If A > 0, the quadratic is always positive (if C > 0) 
                    # or always negative (if C < 0 and vertex is below 0)
                    if A > 0:
                        # Parabola opening up. Check if vertex value <= 0
                        t_v = -B / (2*A)
                        if t_v < 0 or t_v > 1:
                            # Vertex outside [0,1], check endpoints
                            if A*0 + B*0 + C > 1e-20 or A*1 + B*1 + C > 1e-20:
                                return False
                        else:
                            if A*t_v*t_v + B*t_v + C > 1e-20:
                                return False
                        # Actually for A > 0 and disc < 0, the quadratic is always positive
                        # if C > 0 (since the min value is C - B^2/(4A) and disc < 0 means B^2 < 4AC, so C - B^2/(4A) > 0)
                        # So if A > 0 and disc < 0, the quadratic is always > 0, meaning infeasible
                        return False
                    else:
                        # A < 0 (shouldn't happen since ac_d[k] >= 0)
                        pass
                elif disc < 1e-15:
                    # Double root
                    t_root = -B / (2*A)
                    if A > 0:
                        # Only t = t_root satisfies <= 0
                        # Check if t_root is in [t_low, t_high]
                        if t_root < t_low - 1e-10 or t_root > t_high + 1e-10:
                            return False
                        t_low = max(t_low, t_root)
                        t_high = min(t_high, t_root)
                    else:
                        pass
                else:
                    sqrt_disc = np.sqrt(disc)
                    t1 = (-B - sqrt_disc) / (2*A)
                    t2 = (-B + sqrt_disc) / (2*A)
                    if A > 0:
                        # Between roots
                        t_low = max(t_low, t1)
                        t_high = min(t_high, t2)
                    else:
                        # Outside roots (shouldn't happen)
                        t_low = max(t_low, t2)
                        t_high = min(t_high, t1)
            if t_low > t_high + 1e-12:
                return False
        return True
    
    # Binary search on M
    M_low = 0.0
    M_high = peak_val(0.0)  # = ac_a.max()
    
    # First check if t=0 is already optimal (i.e., moving in direction d increases the peak)
    if peak_val(1e-8) >= M_high - 1e-12:
        return 0.0
    
    # Binary search
    for _ in range(60):
        M_mid = (M_low + M_high) / 2
        if feasible_M(M_mid):
            M_high = M_mid
        else:
            M_low = M_mid
        if M_high - M_low < 1e-14:
            break
    
    # Now find the optimal t: it's the midpoint of the feasible interval at M_high
    t_low = 0.0
    t_high = 1.0
    M_target = M_high
    for k in range(len(ac_a)):
        A = ac_d[k]
        B = 2.0 * ac_cross[k]
        C = ac_a[k] - M_target
        if A < 1e-20:
            if abs(B) < 1e-20:
                if C > 1e-20:
                    pass  # infeasible, shouldn't happen
            elif B > 0:
                t_low = max(t_low, -C / B)
            else:
                t_high = min(t_high, -C / B)
        else:
            disc = B*B - 4*A*C
            if disc < 0:
                if A > 0:
                    pass  # infeasible, shouldn't happen
            else:
                sqrt_disc = np.sqrt(max(disc, 0))
                t1 = (-B - sqrt_disc) / (2*A)
                t2 = (-B + sqrt_disc) / (2*A)
                if A > 0:
                    t_low = max(t_low, t1)
                    t_high = min(t_high, t2)
    
    t_opt = (t_low + t_high) / 2
    t_opt = np.clip(t_opt, 0.0, 1.0)
    return t_opt

The key insight is: at the optimum, the active set S* is the set of k where conv_k = max. The LP-FW converges when the LP solution b = a (or the direction d = b - a = 0).

For the active set:
- Start with the top-K conv values
- After each iteration, add violated constraints (where the new point violates the bound)
- Periodically prune constraints that are "far" from active (their margin M - conv_k is large)

The pruning criterion: if M - conv_k > threshold, remove k from the active set. This keeps the LP small.

Let me also think about the **dual of the LP** for guidance. The LP is:

The dual variables for the constraints A_S b <= M 1 are y >= 0. The complementary slackness condition says y_k * (M - (A_S b)_k) = 0, so y_k > 0 only for active constraints.

    t0 = time.monotonic()

    ac = fast_convolve(a)
    active_set = np.argsort(ac)[::-1][:initial_K]
    active_set = set(active_set.tolist())

    iter_count = 0

            break
        iter_count += 1

            best_m = M_curr
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update active set: add violations, prune inactive
        # Add violated constraints
        violations = np.where(ac > M_curr + 1e-12)[0]  # shouldn't happen, M_curr IS the max
        # Actually, we need to add constraints that will be violated by the NEW point, not the current one
        # For now, let's just keep the active set and prune

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def exact_line_search(ac_a, ac_d, ac_cross):
    """Find t in [0,1] minimizing max_k (ac_a[k] + 2t*ac_cross[k] + t^2*ac_d[k])."""
    def peak(t):
        return (ac_a + 2.0 * t * ac_cross + t * t * ac_d).max()

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_g = 0.0
    b_g = 1.0
    c_g = b_g - gr * (b_g - a_g)
    d_g = a_g + gr * (b_g - a_g)
    fc = peak(c_g)
    fd = peak(d_g)

    for _ in range(60):
        if fc < fd:
            b_g = d_g
            d_g = c_g
            fd = fc
            c_g = b_g - gr * (b_g - a_g)
            fc = peak(c_g)
        else:
            a_g = c_g
            c_g = d_g
            fc = fd
            d_g = a_g + gr * (b_g - a_g)
            fd = peak(d_g)

    t_opt = (a_g + b_g) / 2.0
    return max(0.0, min(1.0, t_opt))

def soft_mirror_descent(a0, time_limit, top_k=80, beta=5.0, c0=4.0,
                        restart_every=4000, momentum=0.7, verbose=False):

    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(7)

    x = a.copy()
    vel = np.zeros(n)
    gamma = c0
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    restart_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        y = x + momentum * vel
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        ac = fast_convolve(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * y[k_int - idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        vel = 0.9 * vel + 0.1 * (a_new - x)
        x = a_new

        M_new = fast_convolve(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            restart_count += 1
            noise_level = 0.04 / (1.0 + 0.1 * restart_count)
            pert = rng.normal(0, noise_level, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r < 1e-15:
                x_r = best_a.copy()
            else:
                x_r = x_r / s_r
            x = x_r
            vel = np.zeros(n)
            cur_top_k = min(cur_top_k + 15, 150)
            cur_beta = max(cur_beta - 0.5, 2.0)
            gamma = max(gamma * 0.9, 0.5)

    return best_a, best_m

        ac = fast_convolve(a)
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find destination: minimize contribution to peaks
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < n // 4:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])

            if dst == src:
                continue

            found = False
            for frac in step_scales:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 150:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 80 == 0:
                for _ in range(8):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", stall=" + str(stall))

    return best_a, best_m


def get_best(parent, t_global):
    best_seq = None
    best_score = float('inf')
    try:
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                best_seq = p_norm
                best_score = sc
                print("Parent: n=" + str(len(p)) + ", score=" + repr(sc))
    except Exception:
        pass
    if best_seq is None:
        n = 5000
        x = np.linspace(0, 1, n)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))
    return best_seq, best_score


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Total time budget: " + repr(time_budget) + "s")

    best_seq, best_score = get_best(height_sequence_1, t_global)
    n0 = len(best_seq)

    # Phase 1: Parametric search
    print("\n=== Phase 1: Parametric (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 40.0)
    if t1 > 10:
        a_param, _ = parametric_init(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "parametric", t_global)

    # Phase 2: Main LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: LP-FW main (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30 and n_best <= 9000:
        a_out, _ = lp_frank_wolfe(normalize(best_seq), t2, solver,
                                  initial_K=400, max_K=3000, add_per_iter=40,
                                  verbose=True, stagnation_limit=600)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_main_n" + str(n_best), t_global)

    # Phase 3: SSD refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: SSD (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.08, 120.0)
    if t3 > 20:
        a_out, _ = soft_mirror_descent(normalize(best_seq), t3, top_k=80, beta=5.0,
                                       c0=4.5, restart_every=3000, momentum=0.7, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ssd_n" + str(n_best), t_global)

    # Phase 4: LP-FW second round
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW round 2 (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 30 and n_best <= 9000:
        a_out, _ = lp_frank_wolfe(normalize(best_seq), t4, solver,
                                  initial_K=500, max_K=3500, add_per_iter=50,
                                  verbose=True, stagnation_limit=500)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2_n" + str(n_best), t_global)

    # Phase 5: Try different n values
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    try_ns = [n_best - 500, n_best + 500, n_best + 1000, n_best - 1000]
    try_ns = sorted(set([l for l in try_ns if 2000 <= l <= 10000]))

    for new_n in try_ns[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("\n=== Phase 5: Scale n=" + str(new_n) + " ===")
        a_long = interpolate_seq(best_seq, new_n)

        t5a = min(remaining * 0.30, 180.0)
        if t5a > 30 and new_n <= 9000:
            a_out, _ = lp_frank_wolfe(a_long, t5a, solver,
                                      initial_K=350, max_K=2500, add_per_iter=35,
                                      verbose=False, stagnation_limit=400)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t5b = min(remaining * 0.20, 80.0)
            if t5b > 15:
                a_out, _ = soft_mirror_descent(normalize(best_seq), t5b, top_k=70, beta=4.5,
                                               c0=3.5, restart_every=2500, momentum=0.6, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "ssd_n" + str(new_n), t_global)

    # Phase 6: Final LP-FW on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Final LP-FW (n=" + str(n_best) + ") ===")
    if remaining > 60 and n_best <= 9000:
        t6 = min(remaining * 0.40, 280.0)
        if t6 > 30:
            a_out, _ = lp_frank_wolfe(normalize(best_seq), t6, solver,
                                      initial_K=600, max_K=4000, add_per_iter=60,
                                      verbose=True, stagnation_limit=400)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # Phase 7: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final Coord (n=" + str(n_best) + ") ===")
    if remaining > 30:
        t7 = min(remaining * 0.65, 150.0)
        if t7 > 15:
            a_out, _ = coordinate_polish(normalize(best_seq), t7, rng_seed=42, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
