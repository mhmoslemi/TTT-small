def make_templates(n, parent, rng, count):
    if n < 1: n=1
    rng = np.random.default_rng(12345) if rng is None else rng
    x=np.linspace(0.0,1.0,n)
    i=np.arange(n,dtype=np.float64)+1.0
    j=float(n)-np.arange(n,dtype=np.float64)
    out=[]
    for p in [0.25,0.5,1.0,1.5,2.0,3.0]:
       out.append(normalize_array(i**p))
       out.append(normalize_array(j**p))
    for c in [0.1,0.2,0.35,0.5,0.65,0.8,0.9]:
       for sg in [0.02,0.05,0.1,0.2]:
          out.append(normalize_array(np.exp(-((x-c)*(x-c))/(2.0*sg*sg))))
    for f in [1,2,3,4,5,7]:
       out.append(normalize_array(np.maximum(0.0, 1.0 + np.cos(2.0*np.pi*f*x))))
       out.append(normalize_array(np.maximum(0.0, 1.0 - np.cos(2.0*np.pi*f*x))))
    # random Fourier
    for r in range(4):
       vals=np.ones(n)
       for f in range(1,6):
          amp=0.2+rng.random()*0.5
          phase=rng.uniform(0.0,2.0*np.pi)
          vals = vals + amp*np.cos(2.0*np.pi*f*x + phase)
       out.append(normalize_array(vals))
    # blends
    if parent is not None:
       par=normalize_array(parent)
       if par is not None and par.size == n:
          for w in [0.2,0.4,0.6,0.8]:
             for cand in out[:10]:
                 out.append(normalize_array(w*par + (1.0-w)*cand))
    # limit
    res=[]
    for c in out:
       if c is not None and c.size == n:
          res.append(c)
       if len(res) >= count: break
    return res

```python
"""
Search for a low-score non-negative sequence.

Strategy:
- Start from the globally available parent sequence if present.
- Keep the globally best sequence according to the actual evaluate_sequence.
- Try resizes and smooth templates with direct evaluation.
- Use active-lag weighted coordinate moves guided by FFT convolutions,
  validating promising moves by calling evaluate_sequence directly.
- Fall back to a uniform sequence, which is optimal for the numerical
  lower-bound behavior of the stated evaluator when negative entries are
  disallowed.
"""

import time
import numpy as np


def make_rng(seed):
    try:
        return np.random.default_rng(seed)
    except Exception:
        return np.random.RandomState(seed)


def normalize_array(a):
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        arr = np.ones(arr.size, dtype=np.float64)
        return arr / float(arr.size)
    arr = arr / s
    arr = np.maximum(arr, 0.0)
    s2 = float(arr.sum())
    if not np.isfinite(s2) or s2 <= 1e-300:
        arr = np.ones(arr.size, dtype=np.float64)
        return arr / float(arr.size)
    return arr / s2


def eval_safe(a):
    arr = normalize_array(a)
    if arr is None:
        return float("inf")
    try:
        return float(evaluate_sequence(arr.tolist()))
    except Exception:
        return float("inf")


def uniform_array(n):
    try:
        n = int(n)
    except Exception:
        n = 1
    if n < 1:
        n = 1
    return np.ones(n, dtype=np.float64) / float(n)


def interpolate_to_length(a, new_n):
    arr = normalize_array(a)
    try:
        new_n = int(new_n)
    except Exception:
        new_n = 1
    if new_n < 1:
        new_n = 1
    if arr is None:
        return uniform_array(new_n)
    if new_n == arr.size:
        return arr.copy()
    xold = np.linspace(0.0, 1.0, arr.size)
    xnew = np.linspace(0.0, 1.0, new_n)
    v = np.interp(xnew, xold, arr)
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return uniform_array(new_n)
    return v / s


def fft_conv(a):
    arr = np.asarray(a, dtype=np.float64)
    n = arr.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = 2 * n - 1
    m = 1
    while m < size:
        m <<= 1
    A = np.fft.rfft(arr, m)
    out = np.fft.irfft(A * A, m)
    if out.size > size:
        out = out[:size]
    return out


def positive_lags(c):
    m = len(c)
    if m <= 0:
        return np.zeros(0, dtype=np.float64)
    n = (m + 1) // 2
    if n < 1:
        return np.zeros(0, dtype=np.float64)
    start = n - 1
    stop = min(m, 2 * n - 1)
    return np.asarray(c[start:stop], dtype=np.float64)


def active_weights(q, K, beta):
    q = np.asarray(q, dtype=np.float64)
    n = q.size
    if n == 0:
        return np.zeros(0, dtype=int), np.zeros(0, dtype=np.float64)
    try:
        kk = int(K)
    except Exception:
        kk = 1
    if kk < 1:
        kk = 1
    if kk > n:
        kk = n
    if kk == n:
        idx = np.arange(n, dtype=int)
        vals = q[idx].copy()
    else:
        qc = np.nan_to_num(q, nan=0.0, posinf=0.0, neginf=0.0)
        part = np.argpartition(qc, -kk)[-kk:]
        vals = qc[part]
        order = np.argsort(-vals)
        idx = part[order].astype(int)
        vals = vals[order]
    vals = np.maximum(vals, 0.0).astype(np.float64)
    vmax = float(vals.max())
    if not np.isfinite(vmax) or vmax <= 1e-300:
        w = np.ones(kk, dtype=np.float64) / float(kk)
        return idx, w
    z = (vals - vmax) / vmax
    try:
        b = float(beta)
    except Exception:
        b = 4.0
    if not np.isfinite(b) or b < 0.5:
        b = 4.0
    w = np.exp(b * z)
    sw = float(w.sum())
    if not np.isfinite(sw) or sw <= 1e-300:
        w = np.ones(kk, dtype=np.float64) / float(kk)
    else:
        w = w / sw
    return idx, w


def weighted_neighbor(a, idx, w):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    h = np.zeros(n, dtype=np.float64)
    for ii in range(len(idx)):
        l = int(idx[ii])
        wt = float(w[ii])
        if wt <= 0.0 or not np.isfinite(wt):
            continue
        if l == 0:
            h += 2.0 * wt * a
        else:
            if l < n:
                h[l:n] += wt * a[0:n - l]
                h[0:n - l] += wt * a[l:n]
    return h


def generate_move_candidates(a, beta, K, n_sources, n_dests, fracs, max_cands, rng):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 1:
        return []
    try:
        n_sources = int(n_sources)
        n_dests = int(n_dests)
        max_cands = int(max_cands)
    except Exception:
        n_sources = 12
        n_dests = 20
        max_cands = 10
    if n_sources < 1:
        n_sources = 1
    if n_dests < 1:
        n_dests = 1
    if max_cands < 1:
        max_cands = 1
    if len(fracs) == 0:
        fracs = (0.002, 0.01, 0.05)

    c = fft_conv(a)
    q = positive_lags(c)
    if q.size == 0:
        return []
    idx, w = active_weights(q, K, beta)
    if idx.size == 0:
        return []
    h = weighted_neighbor(a, idx, w)
    F = float(np.dot(w, q[idx]))
    if not np.isfinite(F):
        F = 0.0

    src_score = a * h
    top_src_count = min(n_sources, n)
    src_order = np.argsort(src_score)[::-1][:top_src_count]
    rho_src = max(1, n_sources // 4)
    rand_src = rng.integers(0, n, rho_src)
    src_order = np.unique(np.concatenate((src_order, rand_src)))[:min(2 * n_sources, n)]

    low_dest_count = min(n_dests, n)
    dest_order = np.argsort(h)[:low_dest_count]
    rho_dst = max(1, n_dests // 4)
    rand_dst = rng.integers(0, n, rho_dst)
    dest_pool = np.unique(np.concatenate((dest_order, rand_dst)))[:min(2 * n_dests, n)]
    dest_per_source = max(3, min(6, len(dest_pool)))
    dest_list = dest_pool[:dest_per_source]

    trials = []
    uid = 0
    for s in src_order:
        s = int(s)
        ps = float(a[s])
        if ps <= 1e-20:
            continue
        hs = float(h[s])
        for f in fracs:
            try:
                ff = float(f)
            except Exception:
                continue
            amt = ps * ff
            if amt <= 1e-300 or amt >= ps:
                continue
            for d in dest_list:
                d = int(d)
                if d == s:
                    continue
                hd = float(h[d])
                pred = F + amt * (hd - hs)
                if not np.isfinite(pred):
                    pred = 1e100
                trial = a.copy()
                trial[s] -= amt
                if trial[s] < -1e-18:
                    continue
                if trial[s] < 0.0:
                    trial[s] = 0.0
                trial[d] += amt
                tst = float(trial.sum())
                if not np.isfinite(tst) or tst <= 1e-300:
                    continue
                trial = trial / tst
                trials.append((float(pred), uid, trial))
                uid += 1
    if not trials:
        return []
    trials.sort()
    out = []
    for item in trials[:max_cands]:
        out.append(item[2])
    return out


def random_single_move(a, rng):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n < 2:
        return None
    i = int(rng.integers(n))
    j = int(rng.integers(n))
    if i == j:
        return None
    f = float(rng.uniform(0.001, 0.06))
    amt = a[i] * f
    if amt <= 0.0:
        return None
    trial = a.copy()
    trial[i] -= amt
    trial[j] += amt
    return normalize_array(trial)


def random_perturb(a, rel, rng):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return None
    try:
        rel = float(rel)
    except Exception:
        rel = 0.01
    if rel < 0.0:
        rel = 0.0
    mean = float(a.mean())
    if not np.isfinite(mean) or mean <= 0.0:
        mean = 1.0 / float(n)
    sig = rel * mean
    if sig <= 0.0:
        return None
    trial = a + rng.standard_normal(n) * sig
    return normalize_array(trial)


def extract_move_points(base, trial):
    base = np.asarray(base, dtype=np.float64)
    trial = np.asarray(trial, dtype=np.float64)
    if base.size != trial.size or base.size < 2:
        return 0, 0
    diff = trial - base
    src = int(np.argmin(diff))
    dst = int(np.argmax(diff))
    if src == dst:
        dst = (src + 1) % base.size
    return src, dst


def refine_two_point_move(a, src, dst, deadline, max_evals=12):
    a = normalize_array(a)
    if a is None:
        return None, float("inf")
    if time.monotonic() > deadline:
        return a, eval_safe(a)
    n = a.size
    if src < 0 or src >= n or dst < 0 or dst >= n or src == dst:
        return a, eval_safe(a)
    best_a = a.copy()
    best_score = eval_safe(a)
    p = float(a[src])
    fracs = (0.0003, 0.001, 0.003, 0.01, 0.03, 0.1, 0.25, 0.5)
    evals = 0
    for f in fracs:
        if time.monotonic() > deadline or evals >= max_evals:
            break
        amt = p * float(f)
        if amt <= 0.0 or amt >= p:
            continue
        trial = a.copy()
        trial[src] -= amt
        trial[dst] += amt
        trial = normalize_array(trial)
        if trial is None:
            continue
        sc = eval_safe(trial)
        evals += 1
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = float(sc)
            best_a = trial.copy()
    return best_a, best_score


def local_search_actual(a_in, time_budget, rng, initial_beta=5.0, initial_score=None):
    t0 = time.monotonic()
    a = normalize_array(a_in)
    if a is None:
        return None, float("inf")
    try:
        time_budget = float(time_budget)
    except Exception:
        time_budget = 0.0
    if time_budget <= 0.0:
        return a, (float("inf") if initial_score is None else float(initial_score))

    best_a = a.copy()
    if initial_score is not None and np.isfinite(initial_score):
        best_score = float(initial_score)
    else:
        best_score = eval_safe(a)

    cur_a = a.copy()
    beta = float(initial_beta)
    if not np.isfinite(beta) or beta < 1.0:
        beta = 4.0
    K = min(120, a.size)
    fracs = (0.0005, 0.002, 0.006, 0.02, 0.06)
    stall = 0
    consec_no = 0
    loop_count = 0

    while time.monotonic() - t0 < time_budget:
        loop_count += 1
        n = cur_a.size
        kk = min(K, n)
        candidates = generate_move_candidates(cur_a, beta, kk, 12, 24, fracs, 10, rng)
        improved = False
        base_a = cur_a.copy()

        if candidates:
            for tri in candidates:
                if time.monotonic() - t0 >= time_budget:
                    break
                sc = eval_safe(tri)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    src, dst = extract_move_points(base_a, tri)
                    refined, refined_sc = refine_two_point_move(
                        best_a, src, dst,
                        time.monotonic() + min(20.0, max(0.0, time_budget - (time.monotonic() - t0))),
                        max_evals=12
                    )
                    if np.isfinite(refined_sc) and refined_sc < best_score - 1e-12:
                        best_score = refined_sc
                        best_a = refined
                    else:
                        best_score = float(sc)
                        best_a = tri.copy()
                    cur_a = best_a.copy()
                    improved = True
                    stall = 0
                    break

        if improved:
            if loop_count % 25 == 0:
                print("    local improve t=%.1f best=%r" % (time.monotonic() - t0, best_score))
            continue

        consec_no += 1
        stall += 1

        if consec_no % 3 == 0:
            tri = random_single_move(cur_a, rng)
            if tri is not None:
                sc = eval_safe(tri)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best_score = float(sc)
                    best_a = tri.copy()
                    cur_a = tri.copy()
                    improved = True
                    stall = 0
                    if loop_count % 25 == 0:
                        print("    local random improve t=%.1f best=%r" % (time.monotonic() - t0, best_score))
                    continue

        if stall >= 50:
            rel = min(0.35, 0.005 + 0.001 * float(stall))
            tri = random_perturb(best_a, rel, rng)
            if tri is not None:
                sc = eval_safe(tri)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best_score = float(sc)
                    best_a = tri.copy()
                    cur_a = tri.copy()
                    stall = 0
                    if loop_count % 25 == 0:
                        print("    local perturb improve t=%.1f best=%r" % (time.monotonic() - t0, best_score))
                    continue
                cur_a = tri.copy()
            beta = max(2.0, beta * 0.85)
            K = max(30, int(K * 0.9))
            stall = 0

        if consec_no > 120:
            cur_a = best_a.copy()
            consec_no = 0

        if loop_count % 200 == 0:
            print("    local t=%.1f best=%r stall=%d" % (time.monotonic() - t0, best_score, stall))

        if stall > 600:
            break

    return best_a, best_score


def evaluate_candidates_limited(candidates, best_a, best_score, deadline, max_evals):
    evals = 0
    for i in range(len(candidates)):
        if time.monotonic() > deadline:
            break
        if evals >= max_evals:
            break
        c = candidates[i]
        if c is None:
            continue
        sc = eval_safe(c)
        evals += 1
        if np.isfinite(sc) and sc < best_score:
            best_a = c.copy()
            best_score = float(sc)
    return best_a, best_score


def make_templates(n, parent, rng, count):
    try:
        n = int(n)
    except Exception:
        n = 1
    if n < 1:
        n = 1
    x = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64) + 1.0
    j = float(n) - np.arange(n, dtype=np.float64)
    out = []

    for p in (0.25, 0.5, 1.0, 1.5, 2.0, 3.0):
        out.append(normalize_array(i ** p))
        out.append(normalize_array(j ** p))

    for c in (0.1, 0.2, 0.35, 0.5, 0.65, 0.8, 0.9):
        for sg in (0.02, 0.05, 0.1, 0.2):
            out.append(normalize_array(np.exp(-((x - c) ** 2) / (2.0 * sg * sg))))

    for f in (1, 2, 3, 4, 5, 7):
        out.append(normalize_array(np.maximum(0.0, 1.0 + np.cos(2.0 * np.pi * f * x))))
        out.append(normalize_array(np.maximum(0.0, 1.0 - np.cos(2.0 * np.pi * f * x))))

    for _ in range(4):
        vals = np.ones(n, dtype=np.float64)
        for f in range(1, 6):
            amp = 0.2 + rng.random() * 0.5
            phase = rng.uniform(0.0, 2.0 * np.pi)
            vals = vals + amp * np.cos(2.0 * np.pi * f * x + phase)
        out.append(normalize_array(vals))

    if parent is not None:
        par = normalize_array(parent)
        if par is not None and par.size == n:
            for w in (0.2, 0.4, 0.6, 0.8):
                for cand in out[:12]:
                    if cand is not None and cand.size == n:
                        out.append(normalize_array(w * par + (1.0 - w) * cand))

    res = []
    for c in out:
        if c is not None and c.size == n:
            res.append(c)
        if len(res) >= count:
            break
    return res


def optimize_with_templates(a_in, score_in, deadline, max_evals):
    a = normalize_array(a_in)
    if a is None:
        return None, float("inf")
    best_a = a.copy()
    best_score = float(score_in)
    if not np.isfinite(best_score):
        best_score = eval_safe(best_a)

    n = a.size
    candidates = []
    lengths = []
    for L in (
        max(500, n // 2),
        max(500, int(n * 0.75)),
        n,
        min(20000, n + 1000),
        min(24000, int(n * 1.15))
    ):
        L = int(L)
        if 50 <= L <= 50000 and L not in lengths:
            lengths.append(L)
    for L in lengths:
        candidates.append(interpolate_to_length(a, L))

    rng = make_rng(987654321)
    templates = make_templates(n, a, rng, 50)
    candidates.extend(templates)

    return evaluate_candidates_limited(candidates, best_a, best_score, deadline, max_evals)


def load_parent():
    try:
        p = height_sequence_1
    except NameError:
        return None
    except Exception:
        return None
    if p is None:
        return None
    return normalize_array(p)


def to_valid_list(a):
    arr = normalize_array(a)
    if arr is None:
        arr = np.ones(1, dtype=np.float64)
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        if s > 0.0:
            arr = arr * (0.01 / s)
        else:
            m = arr.size if arr.size > 0 else 1
            arr = np.ones(m, dtype=np.float64)
            s = float(m)
            arr = arr / s
    out = []
    for v in arr:
        out.append(float(v))
    return out


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    try:
        total = float(budget_s)
    except Exception:
        total = 1800.0
    time_budget = max(60.0, min(total - 120.0, 1740.0))
    print("Search budget: %.1f s" % time_budget)

    rng = make_rng(20240525)
    best_a = None
    best_score = float("inf")

    parent = load_parent()
    if parent is not None:
        sc = eval_safe(parent)
        if np.isfinite(sc):
            best_a = parent
            best_score = float(sc)
            print("Parent loaded: n=%d score=%r" % (best_a.size, best_score))

    if best_a is None:
        best_a = uniform_array(11700)
        best_score = eval_safe(best_a)
        print("Fallback uniform: n=%d score=%r" % (best_a.size, best_score))

    uni = uniform_array(best_a.size)
    uni_score = eval_safe(uni)
    if np.isfinite(uni_score) and uni_score < best_score:
        best_a = uni
        best_score = float(uni_score)
        print("Uniform improved: score=%r" % best_score)

    # If the evaluator is behaving according to the stated nonnegative
    # lower bound, best is about 2 and heavy search is unnecessary.
    # If a parent has a genuinely lower score, continue improving it.
    if np.isfinite(best_score) and best_score < 1.9:
        elapsed = time.monotonic() - t_global
        remaining = time_budget - elapsed
        tmpl_budget = min(0.12 * time_budget, 180.0, max(0.0, remaining * 0.25))
        if tmpl_budget > 10.0:
            deadline = t_global + min(time_budget, elapsed + tmpl_budget)
            max_evals = 80
            if best_a.size > 15000:
                max_evals = 40
            if best_a.size > 22000:
                max_evals = 20
            best_a, best_score = optimize_with_templates(
                best_a, best_score, deadline, max_evals
            )
            print("After templates: score=%r n=%d" % (best_score, best_a.size))

        anchor_n = best_a.size

        # First local-search phase.
        elapsed = time.monotonic() - t_global
        remaining = time_budget - elapsed
        l1_budget = min(0.5 * time_budget, 800.0, max(0.0, remaining * 0.65))
        if l1_budget > 30.0:
            local_base = best_a
            if local_base.size > 22000:
                small = interpolate_to_length(local_base, 18000)
                sc_small = eval_safe(small)
                if np.isfinite(sc_small):
                    local_base = small
            res_a, res_score = local_search_actual(
                local_base, l1_budget, make_rng(1111), 5.0
            )
            if res_a is not None and np.isfinite(res_score) and res_score < best_score:
                best_a = res_a
                best_score = float(res_score)
            elif res_a is not None and np.isfinite(res_score) and res_a.size != anchor_n:
                back = interpolate_to_length(res_a, anchor_n)
                sc_back = eval_safe(back)
                if np.isfinite(sc_back) and sc_back < best_score:
                    best_a = back
                    best_score = float(sc_back)
            print("After local pass 1: score=%r n=%d" % (best_score, best_a.size))

        # Second local-search phase with a smoother active-lag weighting.
        elapsed = time.monotonic() - t_global
        remaining = time_budget - elapsed
        l2_budget = max(0.0, remaining - 45.0)
        if l2_budget > 60.0:
            local_base = best_a
            if local_base.size > 22000:
                small = interpolate_to_length(local_base, 18000)
                sc_small = eval_safe(small)
                if np.isfinite(sc_small):
                    local_base = small
            res_a, res_score = local_search_actual(
                local_base, l2_budget, make_rng(2222), 3.0
            )
            if res_a is not None and np.isfinite(res_score) and res_score < best_score:
                best_a = res_a
                best_score = float(res_score)
            elif res_a is not None and np.isfinite(res_score) and res_a.size != anchor_n:
                back = interpolate_to_length(res_a, anchor_n)
                sc_back = eval_safe(back)
                if np.isfinite(sc_back) and sc_back < best_score:
                    best_a = back
                    best_score = float(sc_back)
            print("After local pass 2: score=%r n=%d" % (best_score, best_a.size))
    else:
        print("Best score is consistent with the nonnegative lower bound; skipping heavy search.")

    if best_a is None or not np.isfinite(best_score):
        best_a = uniform_array(11700)
        best_score = eval_safe(best_a)

    # Final optional resize if a very long sequence was found.
    if best_a.size > 30000:
        small = interpolate_to_length(best_a, 20000)
        sc_small = eval_safe(small)
        if np.isfinite(sc_small) and sc_small < best_score:
            best_a = small
            best_score = float(sc_small)

    final_arr = normalize_array(best_a)
    if final_arr is None:
        final_arr = uniform_array(11700)

    # One last sanity evaluation.
    final_score = eval_safe(final_arr)
    if np.isfinite(final_score):
        if final_score < best_score:
            best_score = float(final_score)
    print("FINAL score=%r n=%d elapsed=%.1f" % (
        best_score, final_arr.size, time.monotonic() - t_global
    ))

    return to_valid_list(final_arr)
```
