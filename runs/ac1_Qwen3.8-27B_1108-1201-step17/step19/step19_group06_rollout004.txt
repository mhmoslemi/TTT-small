```python
import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_nonneg(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_score(a):
    """Score = 2n * max_ac (assuming sum(a) = 1)"""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

def soft_max_grad(a, ac, top_k, beta):
    """Compute gradient of soft-max over top k autocorrelation lags."""
    n = len(a)
    k_alarm = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_alarm]
    top_vals = ac[top_idx]
    
    max_v = top_vals.max()
    expo = beta * (top_vals - max_v)
    expo = np.clip(expo, -50, 50)
    weights = np.exp(expo)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_alarm) / k_alarm
    else:
        weights = weights / w_sum
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n, k_int + 1)
        if i_lo >= i_hi:
            continue
        idx = np.arange(i_lo, i_hi)
        j_idx = k_int - idx
        grad[idx] += 2.0 * w * a[j_idx]
    
    return grad, top_idx
```

```python
def adam_optimize(a0, time_limit, lr=0.5, beta1=0.9, beta2=0.999, top_k=50, temp_beta=3.0):
    n = len(a0)
    a = norm_nonneg(a0)
    # Work in log-space for non-negativity
    log_a = np.log(np.maximum(a, 1e-30))
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        ac = fast_autocorr(a)
        grad, _ = soft_max_grad(a, ac, top_k, temp_beta)
        
        # Adam update in log-space
        # grad is for a, but we're updating log_a
        # dL/dlog_a = a * dL/da
        grad_log_a = a * grad
        
        m = beta1 * m + (1 - beta1) * grad_log_a
        v = beta2 * v + (1 - beta2) * grad_log_a**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        lr_t = lr * np.sqrt(1 - beta2**t) / (1 - beta1**t)
        log_a = log_a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        a_new = np.exp(log_a - log_a.max())  # stability
        a_new = a_new / a_new.sum()
        
        score_new = compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score
```

But that's not achievable with non-negative sequences (the autocorrelation at lag 0 is a[0]^2 >= 0, and at lag n-1 is the "overlap" which for non-negative sequences has structure).

Actually, the sum of all autocorrelation values is:
sum_k ac[k] = sum_k sum_i a[i]*a[k-i] = sum_i sum_k a[i]*a[k-i] = sum_i a[i] * sum_j a[j] = (sum_i a[i])^2 = 1

So the mean autocorrelation is 1/(2n-1). If all lags were equal, max = 1/(2n-1), score = 2n/(2n-1) → 1.

```python
"""

import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_nonneg(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_score_normalized(a):
    """Score assuming sum(a)=1: 2n * max_ac."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):
    """Safely call the evaluation function."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def soft_max_gradient(a, ac, top_k, beta):
    """
    Compute gradient of soft-max objective over top k autocorrelation lags.
    Returns (gradient_vector, top_lag_indices).
    """
    n = len(a)
    k_alarm = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_alarm]
    top_vals = ac[top_idx]

    max_v = top_vals.max()
    expo = beta * (top_vals - max_v)
    expo = np.clip(expo, -50, 50)
    weights = np.exp(expo)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_alarm) / k_alarm
    else:
        weights = weights / w_sum

    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n, k_int + 1)
        if i_lo >= i_hi:
            continue
        idx = np.arange(i_lo, i_hi)
        j_idx = k_int - idx
        grad[idx] += 2.0 * w * a[j_idx]

    return grad, top_idx


def adam_logspace(a0, time_budget, lr_init=0.3, top_k_init=30, beta_sm=3.0,
                  restart_every=3000, verbose=True, rng_seed=42):
    """
    Adam optimizer in log-space with soft-max gradient.
    """
    n = len(a0)
    a = norm_nonneg(a0)
    log_a = np.log(np.maximum(a, 1e-30))

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    beta_sm_curr = beta_sm
    top_k = top_k_init

    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_budget:
        t += 1
        ac = fast_autocorr(a)
        grad, top_idx = soft_max_gradient(a, ac, top_k, beta_sm_curr)

        # Gradient in log-space: dL/d(log_a) = a * dL/da
        grad_log = a * grad

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log * grad_log)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr_t = lr / (1.0 + 0.02 * np.sqrt(t))
        log_a = log_a - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        a_new = np.exp(log_a - log_a.max())
        a_new = norm_nonneg(a_new)

        score_new = compute_score_normalized(a_new)
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new

        # Adaptive parameters
        if t % 500 == 0:
            top_k = min(top_k + 3, 120)
            beta_sm_curr = max(beta_sm_curr - 0.2, 1.0)
            if stall > 500:
                lr *= 0.9

        # Restart if stuck
        if stall > restart_every:
            stall = 0
            # Structured perturbation
            pert_type = int(rng.integers(5))
            x = np.linspace(0, 1, n)
            if pert_type == 0:
                c = float(rng.uniform(0.2, 0.8))
                sig = float(rng.uniform(0.05, 0.3))
                pert = np.exp(-((x - c) ** 2) / (2 * sig ** 2))
            elif pert_type == 1:
                freq = int(rng.integers(1, 7))
                phase = float(rng.uniform(0, 2 * np.pi))
                pert = (1.0 + np.sin(np.pi * freq * x + phase))
            elif pert_type == 2:
                # Super-Gaussian
                p = float(rng.uniform(0.3, 3.0))
                center = 0.5
                pert = np.exp(-np.abs(x - center) ** (2.0 / max(p, 0.1)))
            elif pert_type == 3:
                # Bimodal
                pert = (np.exp(-((x - 0.3) ** 2) / 0.04) +
                        np.exp(-((x - 0.7) ** 2) / 0.04))
            else:
                # Smooth random
                base = best_a * (1.0 + rng.normal(0, 0.1, n))
                # Smooth it
                kernel = np.exp(-np.arange(-3, 4) ** 2 / 4.0)
                pert = np.convolve(base, kernel, mode='same')

            pert = norm_nonneg(pert)
            w = float(rng.uniform(0.05, 0.3))
            a = norm_nonneg((1 - w) * best_a + w * pert)
            log_a = np.log(np.maximum(a, 1e-30))
            m *= 0.5
            v *= 0.5
            lr = max(lr * 0.9, 0.05)
            beta_sm_curr = max(beta_sm_curr - 0.3, 1.0)

            if verbose:
                el = time.monotonic() - t0
                print(f"      Adam restart t={t} ({el:.0f}s): score={best_score:.8f}, lr={lr:.4f}, top_k={top_k}")

    if verbose:
        el = time.monotonic() - t0
        print(f"      Adam done ({el:.0f}s): score={best_score:.8f}")
    return best_a, best_score


def spectral_shaping(a0, time_budget, lr_init=0.5, beta_spec=4.0,
                     iterations_per_cycle=200, verbose=True, rng_seed=55):
    """
    Flatten the power spectrum to reduce autocorrelation peaks.
    """
    n = len(a0)
    a = norm_nonneg(a0)
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    lr = lr_init
    stall = 0

    while time.monotonic() - t0 < time_budget:
        # Compute spectrum
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_mean = P.mean()

        if P_max < 1e-30:
            break

        # Shape spectrum: push energy from peaks to valleys
        P_dev = (P - P_mean) / max(P_max, 1e-30)
        expo = beta_spec * P_dev
        expo = np.clip(expo, -15, 15)
        w_spec = np.exp(-expo)  # Suppress peaks, boost valleys

        # Modify amplitude
        lr_t = lr / (1.0 + 0.01 * stall)
        A_new = A * (1.0 - lr_t * 0.5 * (w_spec - 1.0) / max(w_spec.mean(), 1e-10))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = norm_nonneg(a_new)

        score_new = compute_score_normalized(a_new)
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Random kick
                a = norm_nonneg(best_a + rng.normal(0, 0.005, n))
                stall = 0
                lr *= 0.9
                beta_spec = max(beta_spec - 0.5, 1.5)

        if time.monotonic() - t0 > time_budget:
            break

    if verbose:
        el = time.monotonic() - t0
        print(f"      Spectral done ({el:.0f}s): score={best_score:.8f}")
    return best_a, best_score


def batch_coordinate_descent(a0, time_budget, rng_seed=77, verbose=True):
    """
    Batched coordinate descent: move mass from high-impact to low-impact positions.
    """
    n = len(a0)
    a = norm_nonneg(a0)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    n_score = 2.0 * n * best_score
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_budget:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr * 2.0 * n < n_score - 1e-12:
            n_score = 2.0 * n * M_curr
            best_score = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break

        # Get top lags
        n_top = min(20, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k_pos in top_lags[:8]:
            if time.monotonic() - t0 > time_budget:
                break

            k_int = int(k_pos)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo >= i_hi:
                continue

            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]

            # Find top contributing positions
            n_pairs = min(5, len(idx))
            pair_order = np.argsort(contrib)[::-1][:n_pairs]

            for pi in pair_order:
                if time.monotonic() - t0 > time_budget:
                    break
                src = int(idx[pi])
                src_val = a[src]
                if src_val < 1e-14:
                    continue

                # Compute impact of each position on top lags
                scores = np.zeros(n)
                for k2 in top_lags[:6]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n, k2i + 1)
                    if s2 < e2:
                        ii = np.arange(s2, e2)
                        jj = k2i - ii
                        scores[ii] += a[jj] * ac[k2i]
                        scores[jj] += a[ii] * ac[k2i]

                scores[src] = np.inf
                dst = int(np.argmin(scores))

                if dst == src or a[dst] > 0.7 * src_val:
                    continue

                # Try transfer
                for frac in [0.01, 0.03, 0.08, 0.15]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial * 2.0 * n < n_score - 1e-13:
                        a = a_trial
                        best_score = m_trial
                        n_score = 2.0 * n * m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            if stall % 300 == 0:
                # Random perturbations
                n_moves = int(rng.integers(10, 30))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = float(rng.uniform(0.003, 0.02))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new * 2.0 * n < n_score:
                    n_score = 2.0 * n * m_new
                    best_score = m_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        el = time.monotonic() - t0
        print(f"      Coord descent done ({el:.0f}s): score={n_score:.8f}")
    return best_a, n_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def parametric_init(n, rng_seed=0):
    """Generate diverse parametric initializations and return the best."""
    rng = np.random.default_rng(rng_seed)
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)

    candidates = []

    # Uniform
    candidates.append(np.ones(n))

    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)

    # Super-Gaussians
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        for c in [0.3, 0.5, 0.7]:
            sig = float(rng.uniform(0.1, 0.4))
            candidates.append(np.exp(-np.abs(x - c) ** (2.0 / p) / sig))

    # Raised cosine windows
    for freq in [1, 2, 3]:
        candidates.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * x))

    # Bimodal
    for sig in [0.1, 0.15, 0.2, 0.3]:
        candidates.append(
            np.exp(-((x - 0.25) ** 2) / (2 * sig ** 2)) +
            np.exp(-((x - 0.75) ** 2) / (2 * sig ** 2)))

    # Sine windows
    for f in [1, 2, 3, 4]:
        candidates.append(np.sin(np.pi * (x + 0.5) * f) ** 2)

    # Exponential
    for r in [1, 2, 5, 10]:
        candidates.append(np.exp(-r * x))
        candidates.append(np.exp(-r * (1 - x)))

    # Tukey windows
    for r in [0.1, 0.3, 0.5]:
        seg = np.ones_like(x)
        mask_low = x < r
        mask_high = x > 1 - r
        seg[mask_low] = 0.5 + 0.5 * np.cos(np.pi * (0.5 - x[mask_low] / r))
        seg[mask_high] = 0.5 + 0.5 * np.cos(np.pi * (x[mask_high] - (1 - 0.5 * r)) / r)
        candidates.append(seg)

    best_cand = None
    best_val = float('inf')
    for c in candidates:
        c_norm = norm_nonneg(c)
        v = compute_score_normalized(c_norm)
        if v < best_val:
            best_val = v
            best_cand = c_norm.copy()

    return best_cand

        check_update(a_out, f"coord2_n{len(best_seq)}")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration (n_best={n_best}, budget={remaining:.0f}s) ===")

    length_tries = set()
    for delta in [-2000, -1000, -500, 500, 1000, 2000, 3000]:
        nn = n_best + delta
        if 2000 <= nn <= 12000:
            length_tries.add(nn)
    for rn in [3000, 4000, 5000, 6000, 7000, 8000, 10000]:
        if rn != n_best and 2000 <= rn <= 12000:
            length_tries.add(rn)
    length_tries = sorted(length_tries)

    for new_n in length_tries[:6]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")

        a_interp = interpolate_sequence(best_seq, new_n)

        # Quick Adam on new length
        t6a = min(remaining * 0.25, 100.0)
        if t6a > 10:
            a_out, _ = adam_logspace(a_interp, t6a, lr_init=0.15, top_k_init=40,
                                     beta_sm=3.5, restart_every=2000, verbose=False,
                                     rng_seed=new_n % 10000)
            check_update(a_out, f"adam_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.2, 60.0)
            if t6b > 10:
                a_out, _ = spectral_shaping(best_seq, t6b, lr_init=0.3, beta_spec=3.0,
                                            verbose=False, rng_seed=new_n % 5000)
                check_update(a_out, f"spectral_n{new_n}")

    # === Phase 7: Final fine-tuning ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_final}, budget={remaining:.0f}s) ===")
    if remaining > 20:
        t7 = min(remaining * 0.5, 150.0)
        a_out, _ = adam_logspace(best_seq, t7, lr_init=0.03, top_k_init=80,
                                 beta_sm=6.0, restart_every=1500, verbose=False,
                                 rng_seed=314)
        check_update(a_out, f"final_adam_n{n_final}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t8 = min(remaining * 0.6, 80.0)
        a_out, _ = batch_coordinate_descent(best_seq, t8, rng_seed=42, verbose=False)
        check_update(a_out, f"final_coord_n{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        el = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={el:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.8f}")

    return [float(x) for x in best_seq]
```
