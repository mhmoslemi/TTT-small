"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Symmetrized initialization from parent or parametric families
2. LP-based Frank-Wolfe with scipy HiGHS (fast repeated solves)
3. Adaptive constraint management with warm K
4. Nesterov-accelerated FW with stall detection and restarts
5. Final FISTA polish with top-k softmax gradient (FFT-accelerated)
6. Multi-length exploration

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def symmetrize(a):
    """Symmetrize: a_sym[i] = (a[i] + a[n-1-i])/2."""
    return 0.5 * (a + a[::-1].copy())


def evaluate_safe(a):
    """Call the global evaluate_sequence safely."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def build_constraint_matrix_sparse(y, constraint_list, n):
    """Build sparse constraint matrix where row k has y[k-i] at column i (valid indices)."""
    rows = []
    cols = []
    vals = []
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            rows.extend([jj] * len(idx))
            cols.extend(idx.tolist())
            vals.extend((y[k_int - idx]).tolist())
    mat = sp.csc_matrix((vals, (rows, cols)), shape=(len(constraint_list), n))
    return mat


def solve_fw_lp(y, constraint_list, n, M_curr):
    """
    Solve the FW subproblem:
      min M
      s.t. (y*g)[k] <= M for k in constraint_list
           sum(g) = 1, g >= 0
    Returns the optimal g direction.
    """
    K = len(constraint_list)
    if K == 0:
        return np.ones(n) / n

    # Build A_ineq: K x (n+1) matrix [A | -1]
    # A[k, i] = y[k-i] for valid i
    rows = []
    cols = []
    vals = []
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            y_vals = y[k_int - idx]
            rows.extend([jj] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(y_vals.tolist())
            # Add the -1 for M variable (column n)
            rows.append(jj)
            cols.append(n)
            vals.append(-1.0)

    A_ineq = sp.csc_matrix((vals, (rows, cols)), shape=(K, n + 1))
    b_ineq = np.zeros(K)

    # A_eq: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize M

    bounds = [(0, None)] * n + [(0, None)]  # g >= 0, M >= 0

    res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': 30, 'presolve': True})

    if res.status not in (0, 3):  # 0=optimal, 3=optimal (infeasible obj)
        return None

    g_val = np.maximum(np.asarray(res.x[:n], dtype=np.float64), 0.0)
    g_sum = g_val.sum()
    if g_sum < 1e-15:
        return None
    return g_val / g_sum


def fista_step(y, beta, eta, n, top_k=200):
    """One FISTA step with top-k softmax gradient."""
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    A = np.fft.rfft(y, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]

    if top_k is not None and top_k < size:
        top_idx = np.argsort(c)[::-1][:top_k]
        c_top = c[top_idx]
        c_max = c_top.max()
        exp_c = np.zeros(size)
        exp_c[top_idx] = np.exp(beta * (c_top - c_max))
        w = exp_c / exp_c.sum()
    else:
        c_max = c.max()
        exp_c = np.exp(beta * (c - c_max))
        w = exp_c / exp_c.sum()

    a_rev = y[::-1]
    W = np.fft.rfft(w, fft_size)
    A_rev = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(W * A_rev, fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]

    log_y = np.log(np.maximum(y, 1e-30)) - eta * grad
    log_y -= log_y.max()
    z = np.exp(log_y)
    z = np.maximum(z, 1e-30)
    z = z / z.sum()
    return z, c


def fista_optimize(a0, time_limit, beta_start=5.0, beta_end=100.0,
                   eta0=0.8, top_k=200, verbose=True):
    """FISTA with adaptive softmax gradient."""
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_threshold = 5000

    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        progress = min(iter_count / max(time_limit * 50, 1), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        eta = eta0 / (np.sqrt(t) + 1.0)

        z, c = fista_step(y, beta, eta, n, top_k=top_k)

        momentum = (t - 1.0) / (t + 2.0)
        x_new = z + momentum * (z - y)
        x_new = np.maximum(x_new, 1e-30)
        x_new = x_new / x_new.sum()

        c_new = fast_autocorr(x_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > stall_threshold:
            rng = np.random.default_rng(iter_count)
            pert = rng.normal(0, 0.03, n) * best_x
            x_restart = norm_sum1(best_x + pert)
            x_new = x_restart
            y = x_restart.copy()
            stall = 0
            stall_threshold = max(stall_threshold * 0.7, 1500)

        y = x_new
        x = x_new
        iter_count += 1

        if verbose and iter_count % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}")

    return best_x, best_m


def lp_frank_wolfe(a0, time_limit, initial_K=800, max_K=2500, verbose=True):
    """LP-based Frank-Wolfe with Nesterov acceleration and adaptive constraints."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()

    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0
    stall = 0
    stall_threshold = 150

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Evaluate current best
        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            stall = 0

        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-14:
            best_m = ac_x.max()
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > stall_threshold:
            # Restart with perturbation
            rng = np.random.default_rng(iter_count)
            pert = rng.normal(0, 0.02, n) * best_a
            x_restart = norm_sum1(best_a + pert)
            y = x_restart.copy()
            x = x_restart.copy()
            stall = 0
            stall_threshold = max(stall_threshold * 0.8, 80)
            # Reset constraints
            ac_new = fast_autocorr(y)
            init_idx2 = np.argsort(ac_new)[::-1][:min(initial_K, 2 * n - 1)]
            constraint_set = set(int(k) for k in init_idx2)
            continue

        # Manage constraint set size
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        if len(constraint_list) == 0:
            break

        # Solve LP
        g_dir = solve_fw_lp(y, constraint_list, n, M_curr)
        if g_dir is None:
            # LP failed, try with fewer constraints
            constraint_list = sorted(constraint_set)[:min(500, len(constraint_set))]
            constraint_set = set(constraint_list)
            g_dir = solve_fw_lp(y, constraint_list, n, M_curr)
            if g_dir is None:
                continue

        # Find violated constraints and add them
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                constraint_set.add(int(k))

        # Nesterov FW step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    return best_a, best_m


def parametric_init(n, best_ref=None):
    """Try diverse parametric families, return best."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')

    candidates = []

    # Symmetric power laws (U-shaped and inverted)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        d = np.abs(i_arr - (n - 1) / 2.0) / ((n - 1) / 2.0)
        candidates.append(d ** p)  # U-shaped
        candidates.append((1.0 - d) ** p)  # Inverted U (hump)

    # Cosine-based two-hump
    for alpha in [0.3, 0.5, 0.7, 0.9]:
        candidates.append(1.0 - alpha * np.cos(2.0 * np.pi * (i_arr + 0.5) / n))

    # Gaussian mixtures
    for c1, c2, s in [(0.2, 0.8, 0.1), (0.25, 0.75, 0.15), (0.3, 0.7, 0.2),
                       (0.15, 0.85, 0.08), (0.2, 0.8, 0.15)]:
        candidates.append(np.exp(-((x - c1) ** 2) / (2 * s * s)) +
                          np.exp(-((x - c2) ** 2) / (2 * s * s)))

    # Beta-like
    for a_p, b_p in [(0.5, 0.5), (0.7, 0.7), (0.3, 0.3), (1.5, 1.5), (2.0, 2.0)]:
        candidates.append(np.maximum(i_arr + 1.0, 0.1) ** a_p *
                          np.maximum(n - i_arr, 0.1) ** b_p)

    # Reference blends
    if best_ref is not None and len(best_ref) == n:
        ref_norm = norm_sum1(best_ref)
        for w in [0.3, 0.5, 0.7]:
            d = np.abs(i_arr - (n - 1) / 2.0) / ((n - 1) / 2.0)
            candidates.append(w * ref_norm + (1.0 - w) * norm_sum1(d ** 0.5))

    for a_cand in candidates:
        a_norm = norm_sum1(a_cand)
        ac = fast_autocorr(a_norm)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_a = a_norm

    return best_a, best_val


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        check_and_update(a_out, "fista_mid")

    # Phase 4: Second LP-FW round
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 4: LP-FW round 2 (n={n_best}, t={t4:.0f}s) ===")
    if t4 > 30:
        a_in = norm_sum1(symmetrize(best_seq))
        a_out, _ = lp_frank_wolfe(a_in, t4, initial_K=1200, max_K=3500, verbose=True)
        check_and_update(a_out, "lp_r2")

    # Phase 5: FISTA high-beta refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: FISTA high-beta (n={n_best}, t={t5:.0f}s) ===")
    if t5 > 10:
        a_in = norm_sum1(symmetrize(best_seq))
        a_out, _ = fista_optimize(a_in, t5, beta_start=50.0, beta_end=300.0,
                                  eta0=0.3, top_k=100, verbose=True)
        check_and_update(a_out, "fista_high_beta")

    # Phase 6: Try different n
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")

    length_candidates = []
    for delta in [-1000, -500, 500, 1000]:
        new_n = n_best + delta
        if 2500 <= new_n <= 8000:
            length_candidates.append(new_n)
    for rn in [3000, 4000, 5000, 6000, 7000]:
        if rn != n_best and 2500 <= rn <= 8000:
            length_candidates.append(rn)
    length_candidates = sorted(set(length_candidates))

    for new_n in length_candidates[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)
        a_long = symmetrize(a_long)
        a_long = norm_sum1(a_long)

        # Quick FISTA on new length
        t6a = min(remaining * 0.4, 120.0)
        if t6a > 10:
            a_out, _ = fista_optimize(a_long, t6a, beta_start=5.0, beta_end=80.0,
                                      eta0=0.7, top_k=150, verbose=False)
            check_and_update(a_out, f"fista_n{new_n}")

        # Quick LP-FW on new length
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 50 and new_n <= 6000:
            t6b = min(remaining2 * 0.4, 150.0)
            a_in2 = norm_sum1(symmetrize(best_seq))
            a_in2 = interpolate_sequence(a_in2, new_n)
            a_out2, _ = lp_frank_wolfe(a_in2, t6b, initial_K=800, max_K=2000, verbose=False)
            check_and_update(a_out2, f"lp_n{new_n}")

    # Phase 7: Final LP-FW on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW (n={n_best}) ===")
    if remaining > 30:
        t7 = min(remaining * 0.4, 150.0)
        if t7 > 15:
            a_in = norm_sum1(symmetrize(best_seq))
            a_out, _ = lp_frank_wolfe(a_in, t7, initial_K=1500, max_K=4000, verbose=True)
            check_and_update(a_out, "lp_final")

    # Phase 8: Final FISTA
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t8 = min(remaining * 0.5, 100.0)
        if t8 > 10:
            a_in = norm_sum1(symmetrize(best_seq))
            a_out, _ = fista_optimize(a_in, t8, beta_start=80.0, beta_end=400.0,
                                      eta0=0.2, top_k=80, verbose=False)
            check_and_update(a_out, "fista_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
