import time
from scipy.signal import correlate as sp_correlate


def fast_conv(a):
    """Compute full convolution conv(a,a) via FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_weighted_gradient(a, weights):
    """Compute g[j] = 2 * sum_m weights[j+m] * a[m] via FFT cross-correlation."""
    n = len(a)
    wlen = len(weights)
    fft_size = 1 << (n + wlen - 2).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size)
    w_pad[:wlen] = weights
    A_f = np.fft.rfft(a_pad, fft_size)
    W_f = np.fft.rfft(w_pad, fft_size)
    cross = np.fft.irfft(A_f * np.conj(W_f), fft_size)
    return 2.0 * cross[:n]


def softmax_weights(c, tau):
    """Compute softmax weights over convolution values."""
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    s = exp_c.sum()
    if s < 1e-300:
        return np.ones_like(c) / len(c)
    return exp_c / s


def topk_weights(c, K):
    """Uniform weights on top-K largest convolution values."""
    K = min(K, len(c))
    top_idx = np.argsort(c)[::-1][:K]
    w = np.zeros_like(c)
    w[top_idx] = 1.0 / K
    return w


def pocs_init(n, n_trials, time_budget, rng_seed=0):
    """POCS-style spectral initialization: alternate flat-spectrum and non-negativity."""
    t0 = time.monotonic()
    best_a = np.sin(np.pi * np.linspace(0, 1, n)) ** 2
    best_a = norm_sum1(best_a)
    best_m = fast_conv(best_a).max()

    fft_size = 1 << (2 * n - 2).bit_length()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_budget - 1:
            break
        t_avail = (time_budget - (time.monotonic() - t0)) / (n_trials - trial)

        rng = np.random.default_rng(rng_seed + trial)
        freqs = np.linspace(0, 1, fft_size // 2 + 1)
        # Start with decaying magnitude to help non-negativity
        mag = np.ones(fft_size // 2 + 1)
        mag *= np.exp(-freqs * 0.5)
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        A_spec = mag * np.exp(1j * phases)
        A_full = np.zeros(fft_size, dtype=np.complex128)
        A_full[:fft_size // 2 + 1] = A_spec
        for i in range(1, fft_size // 2):
            A_full[fft_size - i] = np.conj(A_spec[i])
        a = np.fft.irfft(A_full, fft_size)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s

        ac = fast_conv(a)
        M = ac.max()
        if M < best_m:
            best_m = M
            best_a = a.copy()

        # POCS iterations
        n_iters = max(200, int(t_avail * 50))
        for it in range(n_iters):
            A = np.fft.rfft(a, fft_size)
            mags = np.abs(A)
            mags[mags < 1e-15] = 1e-15
            target = np.mean(mags)
            blend = 0.05 + 0.3 * min(1.0, it / 500.0)
            new_mags = (1 - blend) * mags + blend * target
            A_new = new_mags * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s2 = a_new.sum()
            if s2 < 1e-15:
                break
            a_new = a_new / s2
            M_new = fast_conv(a_new).max()
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = a_new.copy()
            a = a_new

    return best_a, best_m


def soft_max_mirror_descent(a0, time_limit, tau_init, tau_final, eta0,
                            nesterov_alpha, restart_interval, rng_seed=0):
    """Nesterov-accelerated mirror descent with adaptive soft-max weights."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    tau = tau_init
    stall = 0
    total_steps = 0

    while time.monotonic() - t0 < time_limit:
        total_steps += 1
        progress = min(1.0, total_steps / 50000.0)
        tau_cur = tau_init * (tau_final / tau_init) ** progress
        eta = eta0 / (np.sqrt(total_steps + 1.0) + 1.0)

        # Nesterov extrapolation
        beta = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * total_steps * total_steps))
        y = x + (total_steps / (total_steps + beta + 1.0)) * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = x.copy()

        # Compute convolution and weights
        c = fast_conv(y)
        M_cur = c.max()

        # Adaptive weights: mix of softmax and top-K
        K = min(50 + 100 * progress, 2 * n - 1)
        w_soft = softmax_weights(c, tau_cur)
        w_top = topk_weights(c, K)
        w_mix = 0.5 * w_soft + 0.5 * w_top

        # Compute gradient via FFT cross-correlation
        g = compute_weighted_gradient(y, w_mix)

        # Mirror descent step (exponential family)
        log_a = np.log(np.maximum(y, 1e-15)) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = y.copy()

        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = a_new

        # Periodic restart
        if stall > restart_interval:
            stall = 0
            sigma = 0.02 + 0.03 * (stall / restart_interval)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            eta *= 0.5

        if total_steps % 2000 == 0:
            j_val = 2.0 * n * best_m
            print(f"      SMMD step={total_steps} ({time.monotonic()-t0:.0f}s): "
                  f"J={j_val:.6f}, tau={tau_cur:.4f}, eta={eta:.6f}, stall={stall}")

    return best_a, best_m


def freq_domain_equalize(a0, time_limit, n_steps=5000, rng_seed=0):
    """Optimize frequency-domain phases to flatten |A(m)|^2."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    fft_size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    phases = np.angle(A)
    mags = np.abs(A).copy()
    mags[mags < 1e-10] = 1e-10

    eta = 0.1
    stall = 0

    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Current a
        A_cur = mags * np.exp(1j * phases)
        A_full = np.zeros(fft_size, dtype=np.complex128)
        A_full[:fft_size // 2 + 1] = A_cur
        A_full[1:fft_size // 2] = A_full[1:fft_size // 2]  # keep as is
        for i in range(1, fft_size // 2):
            A_full[fft_size - i] = np.conj(A_cur[i])
        a_cur = np.fft.irfft(A_full, fft_size)[:n]
        a_cur = np.maximum(a_cur, 0.0)
        sc = a_cur.sum()
        if sc < 1e-15:
            break
        a_cur = a_cur / sc

        # Gradient of variance of |A|^2 w.r.t. phases
        # |A(m)|^2 = A(m)*conj(A(m)), d/dphi of |A(m)|^2 = 2*Im(A(m)*conj(dA/dphi))
        # dA(m)/dphi(m) = i*A(m)
        # So d|A(m)|^2/dphi(m) = 2*Im(A(m)*conj(-i*A(m))) = 2*Im(-i*|A(m)|^2) = -2*|A(m)|^2*Im(1)/... 
        # Actually this is self-referential. Let me use a different approach.
        
        # Instead: compute c = conv(a,a), and compute gradient of max(c) w.r.t. phases
        c = fast_conv(a_cur)
        c_mean = c.mean()
        c_var = np.mean((c - c_mean) ** 2)
        
        if c_var < 1e-20:
            break

        # Gradient of c_var w.r.t. each phase: use finite differences for a few,
        # then extrapolate. Actually, let's use the analytical gradient.
        # c[k] = sum_i a[i]*a[k-i], and a = IFFT(mags*exp(i*phases))
        # This is complex. Let's just do a simple perturb-and-accept.
        
        # Perturb phases
        n_perturb = min(100, fft_size // 2)
        idx = rng.integers(0, fft_size // 2 + 1, n_perturb)
        dphi = rng.normal(0, 0.05, n_perturb)
        
        phases_trial = phases.copy()
        phases_trial[idx] += dphi
        A_trial = mags * np.exp(1j * phases_trial)
        A_full_t = np.zeros(fft_size, dtype=np.complex128)
        A_full_t[:fft_size // 2 + 1] = A_trial
        for i in range(1, fft_size // 2):
            A_full_t[fft_size - i] = np.conj(A_trial[i])
        a_trial = np.fft.irfft(A_full_t, fft_size)[:n]
        a_trial = np.maximum(a_trial, 0.0)
        st = a_trial.sum()
        if st > 1e-15:
            a_trial = a_trial / st
            c_trial = fast_conv(a_trial)
            c_var_trial = np.mean((c_trial - c_trial.mean()) ** 2)
            if c_var_trial < c_var * 0.999:
                phases = phases_trial
                a = a_trial
                best_m_new = c_trial.max()
                if best_m_new < best_m:
                    best_m = best_m_new
                    best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > 200:
            stall = 0
            phases += rng.normal(0, 0.1, fft_size // 2 + 1)

    return best_a, best_m


def topk_subgradient_polish(a0, time_limit, K_init=30, K_max=200, rng_seed=0):
    """Polish using exact subgradient on top-K convolution peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    K = K_init
    eta = 0.5
    stall = 0
    total = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        total += 1
        c = fast_conv(a)
        M_cur = c.max()
        if M_cur < 1e-20:
            break

        # Top-K weights
        K_cur = min(K, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:K_cur]
        top_vals = c[top_idx]
        # Soft weighting within top-K
        w_top = top_vals ** 3.0
        w_sum = w_top.sum()
        if w_sum < 1e-30:
            w_top = np.ones(K_cur) / K_cur
        else:
            w_top = w_top / w_sum

        w = np.zeros(2 * n - 1)
        w[top_idx] = w_top

        # Exact gradient
        g = compute_weighted_gradient(a, w)

        # Projected gradient step onto simplex
        a_trial = a - eta * g
        a_trial = np.maximum(a_trial, 0.0)
        st = a_trial.sum()
        if st < 1e-15:
            break
        a_trial = a_trial / st

        M_new = fast_conv(a_trial).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_trial.copy()
            a = a_trial
            stall = 0
            eta = min(eta * 1.03, 2.0)
            K = min(K + 2, K_max)
        else:
            stall += 1
            if stall > 50:
                eta *= 0.8
            if stall > 100:
                K = min(K + 10, K_max)
            if stall > 200:
                pert = rng.normal(0, 0.01, n) * best_a
                a_p = np.maximum(best_a + pert, 1e-10)
                a_p = a_p / a_p.sum()
                M_p = fast_conv(a_p).max()
                if M_p < best_m:
                    a = a_p
                    best_m = M_p
                    best_a = a_p.copy()
                    stall = 0
                    eta = 0.3
                else:
                    a = a_trial
            else:
                a = a_trial

        if total % 200 == 0:
            j_val = 2.0 * n * best_m
            print(f"      SG polish it={total} ({time.monotonic()-t0:.0f}s): "
                  f"J={j_val:.6f}, K={K_cur}, eta={eta:.4f}, stall={stall}")

    return best_a, best_m


def peak_shaving(a0, time_limit, rng_seed=0):
    """Directly reduce the maximum convolution peak by redistributing mass."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    total = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        total += 1
        c = fast_conv(a)
        M_cur = c.max()
        k_max = int(np.argmax(c))

        # For peak at lag k_max, the dominant pairs are (i, k_max-i)
        i_start = max(0, k_max - n + 1)
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        pairs = idx * a[k_max - idx]
        pair_sum = pairs.sum()
        if pair_sum < 1e-20:
            break

        # Find the dominant pair
        dom_i = idx[np.argmax(pairs)]
        dom_k = k_max - dom_i

        # Compute "loading" for all coordinates: how much they contribute to top peaks
        top_K = 10
        top_idx = np.argsort(c)[::-1][:top_K]
        loading = np.zeros(n)
        for ki in top_idx:
            ks = max(0, int(ki) - n + 1)
            ke = min(n, int(ki) + 1)
            ii = np.arange(ks, ke)
            loading[ii] += a[int(ki) - ii]
            loading[int(ki) - ii] += a[ii]

        # Find destination: low loading
        mask = np.ones(n, dtype=bool)
        mask[dom_i] = False
        mask[dom_k] = False
        low_idx = np.where(mask)[0]
        if len(low_idx) == 0:
            stall += 100
            continue
        dst = low_idx[np.argmin(loading[low_idx])]

        # Move a small fraction of mass
        src_val = a[dom_i]
        if src_val < 1e-12:
            stall += 1
            continue

        improved = False
        for frac in [0.005, 0.002, 0.01, 0.003, 0.02, 0.001]:
            amt = src_val * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[dom_i] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_m - 1e-16:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                stall = 0
                improved = True
                break
            elif M_trial < M_cur * 1.001:
                a = a_trial
                stall = min(stall, 50)
                break

        if not improved:
            stall += 1
            if stall > 50:
                # Random perturbation
                i_rand = int(rng.integers(n))
                j_rand = int(rng.integers(n))
                if i_rand != j_rand and a[i_rand] > 1e-10:
                    amt = a[i_rand] * rng.uniform(0.001, 0.01)
                    a[i_rand] -= amt
                    a[j_rand] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if total % 100 == 0:
            j_val = 2.0 * n * best_m
            print(f"      PeakShave it={total} ({time.monotonic()-t0:.0f}s): "
                  f"J={j_val:.6f}, k_max={k_max}, stall={stall}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, n_top=15, rng_seed=0):
    """Coordinate descent: move mass from high-contribution to low-contribution coords."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        c = fast_conv(a)
        top_k = np.argsort(c)[::-1][:n_top]

        # Compute sensitivity map
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:8]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            top_cands = candidates[np.argsort(sens[candidates])[:5]]

            found = False
            for dst in top_cands:
                if found:
                    break
                for frac in [0.005, 0.002, 0.01, 0.02, 0.003, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def make_sin_sq(n):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    return norm_sum1(a)


def make_cosine_sq(n):
    x = np.linspace(0, 1, n)
    a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 2
    return norm_sum1(a)


def make_gaussian(n, width=0.25):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5) / width) ** 2)
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.2) / width) ** 2) + np.exp(-((x - 0.8) / width) ** 2) + 0.01
    return norm_sum1(a)


def make_smooth_random(n, n_knots=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    knots = np.linspace(0, 1, n_knots)
    vals = rng.uniform(0.3, 1.0, n_knots)
    a = np.interp(x, knots, vals)
    a += 0.1 * rng.normal(0, 1, n)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_fourier_smooth(n, n_modes=10, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.7
    for f in range(1, n_modes + 1):
        c = rng.normal(0, 0.2 / np.sqrt(f))
        p = rng.uniform(0, 2 * np.pi)
        a += c * np.sin(2 * np.pi * f * x + p)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search: multi-stage optimization with FFT-based gradient methods."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(norm_sum1(parent))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5500
        best_seq = make_sin_sq(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-start initialization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Multi-start init (n={n0}) ===")
    t1 = min(remaining * 0.06, 80.0)

    if t1 > 15:
        inits = [
            (norm_sum1(best_seq), "current"),
            (make_sin_sq(n0), "sin2"),
            (make_cosine_sq(n0), "cos2"),
            (make_gaussian(n0, 0.25), "gauss"),
            (make_bimodal(n0, 0.15), "bimodal"),
            (make_fourier_smooth(n0, 15, 42), "fourier"),
            (make_smooth_random(n0, 30, 7), "rand_spline"),
        ]
        if parent is not None:
            inits.insert(0, (norm_sum1(parent), "parent"))

        for init_i, (init_a, lbl) in enumerate(inits):
            if time.monotonic() - t_global > time_budget - 60:
                break
            t_avail = min(t1 / (len(inits) - init_i), 15.0)
            if t_avail < 5:
                break
            a_out, _ = pocs_init(n0, 2, t_avail, rng_seed=init_i * 10)
            try_update(a_out, f"pocs_{lbl}")

    # === Phase 2: Main soft-max mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Soft-max Mirror Descent (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 60:
        a_in = norm_sum1(best_seq)
        # Run with different tau schedules
        for trial in range(3):
            r = time_budget - (time.monotonic() - t_global)
            if r < 60:
                break
            t_avail = min(t2 / (3 - trial), 250.0)
            print(f"  SMMD trial {trial} (t={t_avail:.0f}s)...")
            tau_f = [5.0, 20.0, 50.0][trial]
            a_out, _ = soft_max_mirror_descent(a_in, t_avail, tau_init=3.0,
                                                tau_final=tau_f, eta0=1.0,
                                                nesterov_alpha=0.9,
                                                restart_interval=3000,
                                                rng_seed=100 + trial)
            try_update(a_out, f"smmd_{trial}")
            a_in = norm_sum1(best_seq)

    # === Phase 3: Frequency-domain equalization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Freq-domain equalization (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = freq_domain_equalize(a_in, t3, n_steps=8000, rng_seed=55)
        try_update(a_out, "freq_eq")

    # === Phase 4: Top-K subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Top-K Subgradient (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = topk_subgradient_polish(a_in, t4, K_init=40, K_max=150,
                                           rng_seed=200)
        try_update(a_out, "sg_polish")

    # === Phase 5: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Peak Shaving (n={len(best_seq)}) ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving(a_in, t5, rng_seed=300)
        try_update(a_out, "peak_shave")

    # === Phase 6: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Coordinate Polish (n={len(best_seq)}) ===")
    t6 = min(remaining * 0.08, 120.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t6, n_top=20, rng_seed=400)
        try_update(a_out, "coord")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration (base n={n_best}) ===")

    trial_lengths = []
    for delta in [300, -300, 600, -600, 1000, -1000, 1500, -1500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 9000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick mirror descent
        t7a = min(remaining * 0.3, 120.0)
        if t7a > 20:
            a_out, _ = soft_max_mirror_descent(a_long, t7a, tau_init=3.0,
                                                tau_final=10.0, eta0=0.8,
                                                nesterov_alpha=0.9,
                                                restart_interval=2000,
                                                rng_seed=new_n % 500)
            try_update(a_out, f"smmd_n{new_n}")

        # Quick subgradient
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t7b = min(remaining * 0.3, 80.0)
            if t7b > 15:
                a_in2 = norm_sum1(interpolate_sequence(best_seq, new_n))
                a_out2, _ = topk_subgradient_polish(a_in2, t7b, K_init=30,
                                                     K_max=80, rng_seed=new_n % 300)
                try_update(a_out2, f"sg_n{new_n}")

    # === Phase 8: Final optimization rounds ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final optimization (n={n_best}) ===")

    if remaining > 60:
        t8a = min(remaining * 0.25, 200.0)
        if t8a > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = soft_max_mirror_descent(a_in, t8a, tau_init=2.0,
                                                tau_final=5.0, eta0=0.6,
                                                nesterov_alpha=0.9,
                                                restart_interval=2000,
                                                rng_seed=999)
            try_update(a_out, "smmd_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t8b = min(remaining * 0.25, 150.0)
            if t8b > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = topk_subgradient_polish(a_in, t8b, K_init=50,
                                                    K_max=200, rng_seed=888)
                try_update(a_out, "sg_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t8c = min(remaining * 0.25, 100.0)
            if t8c > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = peak_shaving(a_in, t8c, rng_seed=777)
                try_update(a_out, "peak_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8d = min(remaining * 0.3, 60.0)
            if t8d > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_polish(a_in, t8d, n_top=15, rng_seed=666)
                try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = make_sin_sq(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
