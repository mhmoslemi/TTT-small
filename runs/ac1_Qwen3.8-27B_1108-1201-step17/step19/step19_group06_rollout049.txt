def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def build_lp_matrix(a):
    """Build the sparse constraint matrix for the LP.
    Constraint k: sum_j A[k,j] * b[j] <= M
    where A[k,j] = a[k-j] if 0 <= k-j < n
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n, k + 1)
        for i in range(i_lo, i_hi):
            j = k - i
            # constraint k involves b[j] with coefficient a[i]
            if a[i] > 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_direction(a, M, max_iter=50, verbose=False):
    """Solve LP to find direction b that maximizes sum(b) subject to conv(a,b) <= M, b >= 0.
    Returns b_norm (normalized to sum 1) or None if infeasible.
    """
    n = len(a)
    A = build_lp_matrix(a)
    
    # Objective: maximize sum(b) = minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: A @ b <= M (all same M)
    b_vec = np.full(2 * n - 1, M)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 60})
    
    if res.status == 0:
        b = res.x
        b = np.maximum(b, 0)
        s = b.sum()
        if s > 1e-10:
            return b / s
    return None

def compute_mirror_grad(a, ac, top_k, beta):
    """Compute gradient for mirror descent step."""
    n = len(a)
    k_alarm = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_alarm]
    top_vals = ac[top_idx]
    
    max_v = top_vals.max()
    expo = beta * (top_vals - max_v)
    expo = np.clip(expo, -50, 50)
    weights = np.exp(expo)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_alarm) / k_alarm
    else:
        weights = weights / w_sum
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n, k_int + 1)
        if i_lo >= i_hi:
            continue
        idx = np.arange(i_lo, i_hi)
        j_idx = k_int - idx
        valid = (j_idx >= 0) & (j_idx < n)
        grad[idx[valid]] += 2.0 * w * a[j_idx[valid]]
    
    return grad

                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)

                    if s_t < 1e-15:

                    if m_trial < best_m - 1e-14:

                    break

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based iterative optimization with reduced constraint set (top-K lags)
2. FFT-based mirror descent with soft-max gradients and Nesterov momentum
3. Batch coordinate descent for fine-tuning
4. Spectral flattening in frequency domain
5. Multi-scale length exploration with interpolation
6. Adaptive restarts with structured perturbations
"""

import numpy as np
import time
from scipy import sparse
from scipy.optimize import linprog


def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely call the evaluation function."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def score_from_ac(a):
    """Compute score from a normalized sequence (sum=1)."""
    n = len(a)
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max())


def precompute_lp_structure(n):
    """Precompute sparsity structure for LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def lp_direction_reduced(a, M, top_lags, row_idx, col_idx, time_limit=30):
    """Solve reduced LP: max sum(b) s.t. conv(a,b)[k] <= M for k in top_lags, b >= 0."""
    n = len(a)
    # Select rows corresponding to top_lags
    lag_set = set(int(k) for k in top_lags)
    mask = np.isin(row_idx, list(lag_set))
    r_sub = row_idx[mask]
    c_sub = col_idx[mask]
    
    if len(r_sub) == 0:
        return None
    
    # Values: for constraint k, column j: a[k-j]
    a_idx = r_sub.astype(np.int64) - c_sub.astype(np.int64)
    vals = a[a_idx]
    
    # Filter out zero coefficients
    nonzero = vals > 1e-15
    r_sub = r_sub[nonzero]
    c_sub = c_sub[nonzero]
    vals = vals[nonzero]
    
    if len(vals) == 0:
        return None
    
    K = len(top_lags)
    A_sp = sparse.csr_matrix((vals, (r_sub, c_sub)), shape=(K, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    
    t_start = time.monotonic()
    res = linprog(c_obj, A_ub=A_sp, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    
    elapsed = time.monotonic() - t_start
    if res.status == 0 and elapsed < time_limit + 5:
        b = res.x
        b = np.maximum(b, 0.0)
        s = b.sum()
        if s > 1e-10:
            return b / s
    return None


def lp_iterate(a0, time_limit, verbose=True, top_k=80, rng_seed=42):
    """Iterative LP-based optimization with reduced constraints."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    # Precompute LP structure
    row_idx, col_idx = precompute_lp_structure(n)
    
    stall = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > 1650:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Get top K lags
        n_top = min(top_k, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:n_top]
        top_vals = ac[top_lags]
        
        # Use adaptive threshold: include lags that are > 80% of max
        threshold = 0.7 * M
        relevant_mask = ac >= threshold
        relevant_lags = np.where(relevant_mask)[0]
        if len(relevant_lags) > n_top:
            relevant_lags = np.argsort(ac)[::-1][:n_top]
        
        # Solve LP
        b = lp_direction_reduced(a, M, relevant_lags, row_idx, col_idx,
                                  time_limit=min(20, max(5, time_limit - (time.monotonic() - t0))))
        
        if b is None:
            stall += 100
            if stall > 500:
                break
            # Try mirror descent step instead
            a_nest = norm_sum1(a * (1 + rng.normal(0, 0.02, n)))
            m_test = fast_autocorr(a_nest).max()
            if m_test < best_m:
                a = a_nest
                stall = 0
            continue
        
        it += 1
        
        # Line search for optimal t
        best_t = 0
        best_m_trial = M
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_trial = norm_sum1((1 - t) * a + t * b)
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_m_trial:
                best_m_trial = m_trial
                best_t = t
        
        if best_t == 0:
            # Try smaller steps
            for t in [0.01, 0.02, 0.03]:
                a_trial = norm_sum1((1 - t) * a + t * b)
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m_trial:
                    best_m_trial = m_trial
                    best_t = t
        
        if best_t == 0:
            stall += 1
            if stall > 200:
                # Perturb and restart
                a = norm_sum1(best_a * (1 + rng.normal(0, 0.05, n)))
                stall = 0
            continue
        
        a = norm_sum1((1 - best_t) * a + best_t * b)
        m_new = fast_autocorr(a).max()
        
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a.copy()
            stall = 0
            if verbose and it % 10 == 0:
                print(f"    LP iter {it}: M={M:.8f}, score={2*n*M:.6f}, t={best_t:.3f}")
        else:
            stall += 1
    
    if verbose:
        print(f"    LP optimize done ({time.monotonic()-t0:.0f}s, {it} iters): score={2*n*best_m:.6f}")
    return best_a, best_m


def compute_mirror_grad(a, ac, top_k, beta):
    """Compute gradient for mirror descent step using soft-max weighting."""
    n = len(a)
    k_alarm = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_alarm]
    top_vals = ac[top_idx]
    
    max_v = top_vals.max()
    expo = beta * (top_vals - max_v)
    expo = np.clip(expo, -50, 50)
    weights = np.exp(expo)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_alarm) / k_alarm
    else:
        weights = weights / w_sum
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n, k_int + 1)
        if i_lo >= i_hi:
            continue
        idx = np.arange(i_lo, i_hi)
        j_idx = k_int - idx
        valid = (j_idx >= 0) & (j_idx < n)
        grad[idx[valid]] += 2.0 * w * a[j_idx[valid]]
    
    return grad


def mirror_descent(a0, time_limit, lr_init=0.08, beta_init=3.0, top_k_init=40,
                   restart_every=3000, verbose=True, rng_seed=42):
    """Multiplicative update (mirror descent) with soft-max gradient and Nesterov momentum."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    a_prev = a.copy()
    momentum = 0.85
    lr = lr_init
    beta = beta_init
    top_k = top_k_init
    stall = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > 1650:
            break
        it += 1
        
        ratio = a / np.maximum(a_prev, 1e-30)
        ratio = np.clip(ratio, 0.01, 100.0)
        a_nest = a * np.power(ratio, momentum)
        a_nest = norm_sum1(a_nest)
        
        ac = fast_autocorr(a_nest)
        grad = compute_mirror_grad(a_nest, ac, top_k, beta)
        
        eta = lr / (1.0 + 0.05 * np.sqrt(it))
        
        log_a = np.log(np.maximum(a_nest, 1e-30)) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        a_prev = a
        a = a_new
        
        ac_new = fast_autocorr(a)
        m_new = ac_new.max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if it % 500 == 0:
            top_k = min(top_k + 3, 200)
        
        if stall > restart_every:
            stall = 0
            pert_type = int(rng.integers(6))
            if pert_type == 0:
                x = np.linspace(0, 1, n)
                c = float(rng.uniform(0.3, 0.7))
                sig = float(rng.uniform(0.1, 0.4))
                pert = np.exp(-((x - c)**2) / (2 * sig**2))
            elif pert_type == 1:
                i_arr = np.arange(n, dtype=np.float64)
                p = float(rng.uniform(0.5, 2.5))
                pert = (i_arr + 1.0) ** p
            elif pert_type == 2:
                x = np.linspace(0, 1, n)
                freq = int(rng.integers(1, 5))
                pert = np.sin(np.pi * freq * x + float(rng.uniform(0, np.pi)))**2
            elif pert_type == 3:
                pert = best_a * (1.0 + rng.normal(0, 0.03, n))
            elif pert_type == 4:
                i_arr = np.arange(n, dtype=np.float64)
                pert = np.sqrt(np.maximum(i_arr * (n - 1 - i_arr), 1.0))
            else:
                x = np.linspace(0, 1, n)
                pert = np.sin(np.pi * x**0.7)
            
            pert = norm_sum1(pert)
            w = float(rng.uniform(0.05, 0.3))
            a = norm_sum1((1 - w) * best_a + w * pert)
            a_prev = a.copy()
            lr = max(lr * 0.9, 0.01)
            beta = max(beta - 0.3, 1.0)
    
    if verbose:
        print(f"    MD done ({time.monotonic()-t0:.0f}s, {it} iters): score={2*n*best_m:.6f}")
    return best_a, best_m

                stall = 0
                lr *= 0.85
                beta_spec = max(beta_spec - 0.3, 1.0)
    
    if verbose:
        print(f"    Spectral done ({time.monotonic()-t0:.0f}s): score={2*n*best_m:.6f}")
    return best_a, best_m


def batch_coord_descent(a0, time_limit, rng_seed=77, verbose=True):
    """Batch coordinate descent for fine-tuning."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > 1650:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-13:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15000:
                break
        
        n_top = min(25, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k_pos in top_lags[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            
            k_int = int(k_pos)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo >= i_hi:
                continue
            
            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            
            n_pairs = min(8, len(idx))
            pair_order = np.argsort(contrib)[::-1][:n_pairs]
            
            for pi in pair_order:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[pi])
                src_val = a[src]
                if src_val < 1e-14:
                    continue
                
                # Find best destination
                scores = np.zeros(n)
                for k2 in top_lags[:6]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n, k2i + 1)
                    if s2 < e2:
                        ii = np.arange(s2, e2)
                        jj = k2i - ii
                        scores[ii] += a[jj] * ac[k2i]
                        scores[jj] += a[ii] * ac[k2i]
                
                scores[src] = np.inf
                dst = int(np.argmin(scores))
                
                if dst == src or a[dst] > 0.9 * src_val:
                    continue
                
                for frac in [0.01, 0.03, 0.05, 0.1, 0.15]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            if stall % 300 == 0:
                n_moves = int(rng.integers(3, 15))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = float(rng.uniform(0.003, 0.02))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                    m_new = fast_autocorr(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                        stall = 0
    
    if verbose:
        print(f"    Coord descent done ({time.monotonic()-t0:.0f}s): score={2*n*best_m:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def parametric_init(n):
    """Quick parametric search for good initializations."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    
    candidates = []
    candidates.append(np.ones(n) / n)
    
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))
    
    for c in [0.3, 0.5, 0.7]:
        for sig in [0.1, 0.2, 0.3, 0.5]:
            candidates.append(norm_sum1(np.exp(-((x - c)**2) / (2 * sig**2))))
    
    for sig in [0.1, 0.2, 0.3]:
        candidates.append(norm_sum1(
            np.exp(-((x - 0.25)**2) / (2 * sig**2)) + np.exp(-((x - 0.75)**2) / (2 * sig**2))))
    
    for f in [1, 2, 3]:
        candidates.append(norm_sum1(np.sin(np.pi * f * x)**2))
    
    for r in [1, 2, 5]:
        candidates.append(norm_sum1(np.exp(-r * x)))
        candidates.append(norm_sum1(np.exp(r * x)))
    
    # Triangular sqrt
    candidates.append(norm_sum1(np.sqrt(np.maximum(i_arr * (n - 1 - i_arr), 1.0))))
    # Sine power
    candidates.append(norm_sum1(np.sin(np.pi * x**0.5)))
    candidates.append(norm_sum1(np.sin(np.pi * x**0.7)))
    
    best_cand = None
    best_val = float('inf')
    for c in candidates:
        v = score_from_ac(c)
        if v < best_val:
            best_val = v
            best_cand = c.copy()
    
    return best_cand if best_cand is not None else np.ones(n) / n


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining LP-based optimization, mirror descent,
    spectral methods, coordinate descent, and length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent sequence
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n0 = 5000
        best_seq = parametric_init(n0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = evaluate_safe(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")
    
    # Try uniform baseline
    a_uni = np.ones(n0) / n0
    sc_uni = evaluate_safe(a_uni)
    if np.isfinite(sc_uni) and sc_uni < best_score:
        best_score = sc_uni
        best_seq = a_uni.copy()
        print(f"  Uniform improves: {sc_uni:.6f}")
    
    remaining = time_budget - (time.monotonic() - t_global)
    
    # === Phase 1: Mirror descent (quick, to get a good starting point for LP) ===
    print(f"\n=== Phase 1: Mirror descent (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.15, 200.0)
    if t1 > 20:
        a_out, _ = mirror_descent(best_seq, t1, lr_init=0.08, beta_init=3.0,
                                  top_k_init=50, restart_every=2000, verbose=True, rng_seed=42)
        check_update(a_out, f"md1_n{len(best_seq)}")
    
    # === Phase 2: LP-based optimization (main method) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    # Use LP for moderate n (up to ~6000 for reasonable speed)
    print(f"\n=== Phase 2: LP optimization (n={n_cur}, budget={remaining:.0f}s) ===")
    if n_cur <= 6000:
        t2 = min(remaining * 0.35, 500.0)
        if t2 > 30:
            a_out, _ = lp_iterate(best_seq, t2, verbose=True, top_k=60, rng_seed=123)
            check_update(a_out, f"lp_n{n_cur}")
    
    # === Phase 3: Due to time, verify and try different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    
    # Try a couple of different lengths
    length_tries = []
    for delta in [-500, -1000, 500, 1000]:
        nn = n_cur + delta
        if 2000 <= nn <= 7000 and nn != n_cur:
            length_tries.append(nn)
    
    for new_n in length_tries[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        
        a_interp = interpolate_sequence(best_seq, new_n)
        
        # Quick MD on new length
        t3a = min(remaining * 0.3, 120.0)
        if t3a > 15:
            a_out, _ = mirror_descent(a_interp, t3a, lr_init=0.05, beta_init=3.0,
                                      top_k_init=50, restart_every=2000, verbose=False, rng_seed=new_n % 1000)
            check_update(a_out, f"md_n{new_n}")
        
        # LP on new length if feasible
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 6000:
            t3b = min(remaining * 0.4, 200.0)
            if t3b > 20:
                a_out, _ = lp_iterate(best_seq, t3b, verbose=False, top_k=50, rng_seed=new_n % 2000)
                check_update(a_out, f"lp_n{new_n}")
    
    # === Phase 4: Mirror descent fine-tuning on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Fine MD (n={n_best}, budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.25, 250.0)
    if t4 > 20:
        a_out, _ = mirror_descent(best_seq, t4, lr_init=0.03, beta_init=5.0,
                                  top_k_init=80, restart_every=2000, verbose=True, rng_seed=77)
        check_update(a_out, f"md_fine_n{n_best}")
    
    # === Phase 5: Spectral flattening ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Spectral (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 150.0)
    if t5 > 15:
        a_out, _ = spectral_flatten(best_seq, t5, lr_init=0.3, verbose=True)
        check_update(a_out, f"spectral_n{len(best_seq)}")
    
    # === Phase 6: LP again if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 6000 and remaining > 100:
        print(f"\n=== Phase 6: LP again (n={n_best}, budget={remaining:.0f}s) ===")
        t6 = min(remaining * 0.35, 300.0)
        if t6 > 30:
            a_out, _ = lp_iterate(best_seq, t6, verbose=True, top_k=80, rng_seed=2024)
            check_update(a_out, f"lp2_n{n_best}")
    
    # === Phase 7: Final coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Coord descent (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.4, 150.0)
    if t7 > 10:
        a_out, _ = batch_coord_descent(best_seq, t7, rng_seed=42, verbose=True)
        check_update(a_out, f"coord_n{len(best_seq)}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
