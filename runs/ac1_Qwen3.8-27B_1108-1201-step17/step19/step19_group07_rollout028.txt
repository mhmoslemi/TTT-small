def fista_topk(a0, time_limit, k_frac=0.05, beta_schedule='linear', verbose=True):
    """
    FISTA-style optimization with top-k subgradient.
    Only uses the top k_frac fraction of convolution lags in the gradient.
    """
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()
    best_x = x.copy()
    best_c = fast_conv(x)
    best_m = best_c.max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_threshold = 5000
    
    size = 2 * n - 1
    k_top = max(10, int(k_frac * size))
    
    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        
        # Compute convolution of y
        c_y = fast_conv(y)
        m_y = c_y.max()
        
        # Find top-k positions
        if k_top < size:
            top_idx = np.argsort(c_y)[::-1][:k_top]
        else:
            top_idx = np.arange(size)
        
        # Compute gradient at top-k positions
        grad = np.zeros(n)
        for k in top_idx:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[i_start:i_end] += 2.0 * y[k_int - idx]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm
        
        # Step size (diminishing)
        eta = 1.0 / np.sqrt(t + 1.0)
        
        # Mirror step: x_new = proj_simplex(x - eta * grad)
        # Project onto simplex {a >= 0, sum(a) = 1}
        z = x - eta * grad
        # Project z onto simplex
        z_proj = project_simplex(z)
        
        # Nesterov momentum
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            y_new = z_proj + momentum * (z_proj - y)
            y_new = project_simplex(y_new)
        else:
            y_new = z_proj.copy()
        
        # Evaluate
        c_new = fast_conv(y_new)
        m_new = c_new.max()
        
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_x = y_new.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > stall_threshold:
            # Restart with perturbation
            rng = np.random.default_rng(iter_count + 42)
            pert = rng.normal(0, 0.01, n)
            x_restart = project_simplex(best_x + pert)
            y_new = x_restart
            stall = 0
            stall_threshold = max(int(stall_threshold * 0.85), 1000)
            k_top = max(5, int(k_top * 0.7))
        
        y = y_new
        x = y_new
        iter_count += 1
        
        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA-topk iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, k_top={k_top}, stall={stall}")
    
    return best_x, best_m


def project_simplex(v, sum_target=1.0):
    """Project v onto {x >= 0, sum(x) = sum_target}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - sum_target))[0][-1]
    theta = (cssv[rho] - sum_target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_frank_wolfe_v2(a0, time_limit, solver, initial_K=200, max_K=2000, verbose=True):
    """
    Improved LP-based Frank-Wolfe for the convolution peak minimization.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    init_K = min(initial_K, size)
    
    y = a.copy()
    x = a.copy()
    
    # Initial constraint set: top-K lags
    ac_y = fast_conv(y)
    M_curr = ac_y.max()
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_list = sorted(int(k) for k in init_idx)
    
    iter_count = 0
    failed_solves = 0
    no_improve_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current M
        ac_y = fast_conv(y)
        M_curr = ac_y.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Also check x (Nesterov point)
        ac_x = fast_conv(x)
        if ac_x.max() < best_m - 1e-14:
            best_m = ac_x.max()
            best_a = x.copy()
            no_improve_count = 0
        
        # Manage constraint set
        if len(constraint_list) > max_K:
            # Remove constraints with largest margin
            margins = np.array([M_curr - ac_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_list = sorted(int(constraint_list[i]) for i in keep)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        # Constraint: (y * g)[k] <= M_curr for k in constraint_list
        # (y*g)[k] = sum_i y[k-i] * g[i]
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 10:
                constraint_list = sorted(constraint_list)[:min(init_K, len(constraint_list))]
                failed_solves = 0
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            failed_solves += 1
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violated constraints and add them
        conv_ys = fast_cross_conv(y, g_dir)
        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:min(50, len(violations))]]
            for k in top_v:
                if int(k) not in constraint_list:
                    constraint_list.append(int(k))
            constraint_list = sorted(constraint_list)
        
        # FW update with Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW2 iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_list)}, M_var={M_var.value:.6f}")
    
    return best_a, best_m


def coord_descent(a0, time_limit, block_size=5, verbose=True):
    """
    Block coordinate descent: optimize small blocks of coordinates at a time.
    For a block of coordinates, the objective is quadratic, so we can solve exactly.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    no_improve = 0
    no_improve_limit = 200
    
    while time.monotonic() - t0 < time_limit and no_improve < no_improve_limit:
        improved = False
        
        # Scan through blocks
        for start in range(0, n - block_size + 1, block_size):
            if time.monotonic() - t0 > time_limit:
                break
            
            end = start + block_size
            block = a[start:end].copy()
            fixed = a.copy()
            
            # Compute the "background" convolution contribution from fixed parts
            # (a*a)[k] = contribution_from_block + contribution_from_fixed
            # For the block coordinates, the objective is:
            # maximize_k (fixed_conv[k] + 2*cross[k] + block_conv[k]) <= M
            # where cross[k] = sum of cross terms between block and fixed
            # and block_conv[k] = self-conv of block (small)
            
            # Simplification: for small blocks, the block_conv is negligible
            # So we approximately minimize max_k (fixed_conv[k] + 2*cross[k])
            # subject to block >= 0, sum(block) = sum(a[start:end])
            
            # Compute fixed contribution
            # a = [fixed_0 | block | fixed_1]
            a_fixed_zero = a.copy()
            a_fixed_zero[start:end] = 0
            conv_fixed = fast_conv(a_fixed_zero)
            
            # Cross terms: (a_fixed_zero * block_padded)
            block_padded = np.zeros(n)
            block_padded[start:end] = block
            cross_terms = fast_cross_conv(a_fixed_zero, block_padded)
            
            # Total: conv_fixed + 2*cross_terms + conv_block
            # For optimality: we want to minimize max_k (conv_fixed[k] + 2*cross_terms[k])
            # (ignoring block self-conv for now)
            
            # This is: minimize M subject to conv_fixed[k] + 2*sum_i block[i]*a_fixed_zero[k-start-i] <= M
            # for all k, block >= 0, sum(block) = target_sum
            
            target_sum = a[start:end].sum()
            
            # For a small block (size 3-5), we can solve this with a small LP
            # or even analytically.
            
            # Simple approach: use a small LP
            # Variables: block[0], ..., block[block_size-1], M
            # Constraints: for each k (0 to 2n-2):
            #   conv_fixed[k] + 2*sum_i block[i]*a_fixed_zero[k-i-start] <= M
            #   (but many of these are automatically satisfied)
            #   block[i] >= 0, sum(block) = target_sum, M >= 0
            
            # Only include constraints that are "active" (close to current max)
            current_max = (conv_fixed + 2 * cross_terms).max()
            
            # Find relevant k values
            total_approx = conv_fixed + 2 * cross_terms
            active_k = np.where(total_approx > current_max * 0.95)[0]
            
            if len(active_k) == 0:
                continue
            
            # Build small LP
            n_active = min(len(active_k), 50)
            active_k = active_k[np.argsort(total_approx[active_k])[::-1][:n_active]]
            
            b_var = cp.Variable(block_size, nonneg=True)
            M_var = cp.Variable()
            b_constraints = [cp.sum(b_var) == target_sum]
            
            for k in active_k:
                k_int = int(k)
                # cross[k] = sum_i block[i] * a_fixed_zero[k - i - start]
                # (where k - i - start is the index in a_fixed_zero)
                cross_coeff = np.zeros(block_size)
                for i in range(block_size):
                    idx = k_int - i - start
                    if 0 <= idx < n:
                        cross_coeff[i] = a_fixed_zero[idx]
                b_constraints.append(conv_fixed[k_int] + 2.0 * (cross_coeff @ b_var) <= M_var)
            
            prob = cp.Problem(cp.Minimize(M_var), b_constraints)
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    new_block = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
                    bs = new_block.sum()
                    if bs > 1e-10:
                        new_block = new_block * (target_sum / bs)
                        a_new = a.copy()
                        a_new[start:end] = new_block
                        m_new = fast_conv(a_new).max()
                        if m_new < best_m - 1e-14:
                            best_m = m_new
                            best_a = a_new.copy()
                            a = a_new
                            improved = True
            except Exception:
                pass
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
    
    return best_a, best_m
