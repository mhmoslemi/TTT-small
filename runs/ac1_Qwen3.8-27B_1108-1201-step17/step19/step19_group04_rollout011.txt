    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    best = a.copy()
    best_score = objective_normalized(best)
    
    # Adam state
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    # Parameters
    lr = 0.1
    tau_param = 1.0  # τ = tau_param * n
    tau_anneal_rate = 1.015  # multiply tau by this each step
    fails = 0
    rng = np.random.default_rng(42)
    
    step_count = 0
    while time.monotonic() - t0 < time_limit:
        # Compute gradient
        q = self_conv(a)
        M = float(np.max(q))
        score = 2.0 * n * M  # since sum(a) = 1 after normalization
        
        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            fails = 0
        else:
            fails += 1
        
        # Softmax weights
        tau = tau_param * n
        q_scaled = tau * q - tau * np.max(q)
        w = np.exp(q_scaled)
        w = w / w.sum()
        
        # Gradient
        grad = compute_gradient(a, w)
        
        # Adam update
        t_adam += 1
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * grad**2
        m_hat = m_adam / (1 - beta1**t_adam)
        v_hat = v_adam / (1 - beta2**t_adam)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_vec(a_new)
        
        # Check improvement
        q_new = self_conv(a_new)
        M_new = float(np.max(q_new))
        score_new = 2.0 * n * M_new
        
        if score_new < score - 1e-14:
            a = a_new
            # Increase tau (make smoother → sharper)
            tau_param = min(tau_param * tau_anneal_rate, 50.0)
            lr = min(lr * 1.01, 1.0)
            fails = 0
        else:
            a = a_new  # still move, even if not improving (momentum)

                a = a + noise

                a = normalize_vec(a)
                fails = 0
                lr = 0.05
                tau_param = max(tau_param * 0.5, 0.5)
        
        step_count += 1
        if verbose and step_count % 500 == 0:

"""
Search for non-negative step function heights minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based smoothed gradient descent (log-sum-exp max approximation) with Adam optimizer,
multiple diverse parametric starts, tau-annealing, and final coordinate polish.
Key improvements over previous: cheap per-step cost enables millions of iterations,
tau-annealing schedule, population diversity, and symmetry-aware optimization.
"""

import math
import time
import numpy as np

# --- Helper Functions ---

def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]


def objective_raw(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    q = self_conv(a)
    return 2.0 * float(a.size) * float(np.max(q))


def compute_gradient(a, w):
    """
    Compute grad[i] = 2 * sum_j a[j] * w[i+j] (w out-of-range = 0).
    Uses FFT: grad = 2 * (a conv rev(w))[n-1 : 2n-1][::-1]
    """
    n = a.size
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fw_rev = np.fft.rfft(w[::-1], size)
    conv = np.fft.irfft(fa * fw_rev, size)[:m]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return grad


def softmax_weights(q, tau):
    """Compute softmax(tau*q) normalized to sum 1."""
    q = np.asarray(q, dtype=np.float64)
    tq = tau * q
    tq = tq - np.max(tq)
    tq = np.clip(tq, -700.0, 0.0)
    e = np.exp(tq)
    s = float(e.sum())
    if s < 1e-300:
        return np.ones_like(q) / float(q.size)
    return e / s


def smooth_descent(a0, time_limit, tau_start, tau_end, lr_start, lr_end,
                   rng_seed=0, verbose=False, noise_interval=3000, noise_scale=0.005):
    """
    Smoothed gradient descent with Adam and tau annealing.
    Minimize 2n*soft_max(conv(a,a)) subject to a>=0, sum(a)=1.
    """
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 2:
        return a

    best = a.copy()
    best_score = 2.0 * n * float(np.max(self_conv(best)))

    # Adam state
    m_adam = np.zeros(n, dtype=np.float64)
    v_adam = np.zeros(n, dtype=np.float64)
    t_adam = 0

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-10

    tau = tau_start
    lr = lr_start
    n_steps_target = max(1, int((tau_end / tau_start) * 5000)) if tau_end > tau_start else 10000
    log_tau_start = math.log(tau_start)
    log_tau_end = math.log(tau_end)

    rng = np.random.default_rng(rng_seed)
    fails = 0
    step = 0
    best_fail_reset = 0

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, float(step) / max(1, n_steps_target * 3))

        # Anneal tau and lr
        cur_tau = math.exp(log_tau_start + progress * (log_tau_end - log_tau_start))
        cur_lr = lr_start * (lr_end / lr_start) ** progress

        # Compute q and score
        q = self_conv(a)
        M = float(np.max(q))
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best = a.copy()
            best_score = score
            fails = 0
        else:
            fails += 1

        # Noise injection
        if fails > noise_interval or (step > best_fail_reset + noise_interval * 3 and (step - best_fail_reset) % noise_interval == 0):
            noise = noise_scale * rng.normal(0.0, 1.0, size=n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = normalize_vec(a)
            fails = 0
            best_fail_reset = step
            m_adam *= 0.5
            v_adam *= 0.5

        # Compute softmax weights and gradient
        w = softmax_weights(q, cur_tau)
        grad = compute_gradient(a, w)

        # Normalize gradient by the soft_max value to get relative gradient
        soft_val = float(np.dot(w, q))
        if soft_val > 1e-30:
            grad = grad / soft_val

        # Adam update
        t_adam += 1
        m_adam = beta1 * m_adam + (1.0 - beta1) * grad
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad * grad)
        m_hat = m_adam / (1.0 - beta1 ** t_adam)
        v_hat = v_adam / (1.0 - beta2 ** t_adam)

        step_size = cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        step_size = np.clip(step_size, -1.0, 1.0)
        a_new = a - step_size
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        a = a_new
        step += 1

        # Early termination if converged
        if fails > 20000 and progress > 0.9:
            break

        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print("      smooth: step=%d t=%.1fs tau=%.1f lr=%.5f best=%.9f fails=%d"
                  % (step, elapsed, cur_tau, cur_lr, best_score, fails))

    # Final evaluation
    final_score = 2.0 * n * float(np.max(self_conv(best)))
    if final_score < best_score:
        best_score = final_score

    if best_score > objective_raw(best) * 1.001:
        # Re-evaluate
        a = normalize_vec(best)
        sc = objective_raw(a)
        if sc < best_score:
            best = a
            best_score = sc

    return best


def coordinate_polish(a0, time_limit, iters_per_cycle=200):
    """Polish by adjusting individual coordinates to reduce max convolution."""
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 5:
        return a

    best = a.copy()
    best_score = 2.0 * n * float(np.max(self_conv(best)))
    fails = 0
    rng = np.random.default_rng(777)

    q = self_conv(a)
    peak_idx = int(np.argmax(q))

    while time.monotonic() - t0 < time_limit:
        if fails > 50 * iters_per_cycle:
            break

        q = self_conv(a)
        M = float(np.max(q))
        peak_idx = int(np.argmax(q))
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best = a.copy()
            best_score = score
            fails = 0

        # Find which elements contribute most to the peak lag
        k = peak_idx
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 >= i1:
            fails += 1
            continue

        # Contribution of a[j] to q[k]: 2*a[k-j] (for j in valid range)
        contrib = np.zeros(n, dtype=np.float64)
        for j in range(i0, i1):
            idx = k - j
            if 0 <= idx < n:
                contrib[j] = 2.0 * a[idx]

        # Also check neighboring lags
        for dk in (-1, 0, 1):
            kk = k + dk
            if 0 <= kk < 2 * n - 1 and q[kk] > 0.9 * M:
                j0 = max(0, kk - n + 1)
                j1 = min(n, kk + 1)
                for j in range(j0, j1):
                    idx = kk - j
                    if 0 <= idx < n:
                        contrib[j] += a[idx]

        src = np.argsort(contrib)[::-1][:6]
        dst = np.argsort(contrib)[:6]

        improved = False
        for s_idx in src[:4]:
            si = int(s_idx)
            if a[si] < 1e-15:
                continue
            for d_idx in dst[:4]:
                di = int(d_idx)
                if di == si:
                    continue
                for frac in (0.01, 0.03, 0.08):
                    amt = a[si] * frac
                    if amt < 1e-18:
                        break
                    b = a.copy()
                    b[si] -= amt
                    b[di] += amt
                    b = np.maximum(b, 0.0)
                    ss = float(b.sum())
                    if ss < 1e-15:
                        break
                    b = b / ss
                    sc = 2.0 * n * float(np.max(self_conv(b)))
                    if sc < best_score - 1e-13:
                        best = b.copy()
                        best_score = sc
                        a = b
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            fails += 1
            if fails % 10 == 0:
                # Random pair swap
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    b = a.copy()
                    amt = a[i] * 0.02
                    b[i] -= amt
                    b[j] += amt
                    b = np.maximum(b, 0.0)
                    ss = float(b.sum())
                    if ss > 1e-15:
                        b = b / ss
                        sc = 2.0 * n * float(np.max(self_conv(b)))
                        if sc < best_score - 1e-13:
                            best = b.copy()
                            best_score = sc
                            a = b
                            fails = 0

    return best


def make_diverse_starts(n, parent=None):
    """Generate diverse starting sequences of length n."""
    starts = []
    x = np.arange(n, dtype=np.float64) / float(n - 1)

    if parent is not None:
        p = normalize_vec(parent)
        if p.size != n:
            p = np.interp(x, np.linspace(0, 1, p.size), p)
            p = normalize_vec(p)
        starts.append(p.copy())
        starts.append(p[::-1].copy())

    # Parabola
    xx = np.linspace(0, 1, n)
    parab = (xx * (1 - xx) + 1e-10)
    starts.append(normalize_vec(parab))

    # Raised cosine
    cos_seq = 0.5 + 0.5 * np.cos(2.0 * np.pi * (xx - 0.5))
    starts.append(normalize_vec(cos_seq))

    # Bump (narrower)
    for width in (0.3, 0.5, 0.7):
        bump = np.exp(-((xx - 0.5) ** 2) / (2.0 * width * width * 0.1))
        starts.append(normalize_vec(bump))

    # Power of parabola
    for g in (0.5, 1.0, 2.0, 3.0):
        starts.append(normalize_vec(parab ** g))

    # Two-peak
    for pos in (0.3, 0.4):
        two = np.exp(-((xx - pos) ** 2) / 0.01) + np.exp(-((xx - (1 - pos)) ** 2) / 0.01)
        starts.append(normalize_vec(two))

    # Slight asymmetry
    for w in (0.1, 0.3, 0.5):
        asym = (1 - w) * xx * (1 - xx) + w * (1 - xx) * xx
        asym = xx ** (1 + w) * (1 - xx) ** (1 - w) + 1e-10
        starts.append(normalize_vec(asym))

    # Random smooth bumps
    rng = np.random.default_rng(12345)
    for _ in range(10):
        seq = np.ones(n, dtype=np.float64) * 0.1
        for _ in range(int(rng.integers(1, 4))):
            c = rng.uniform(0.1, 0.9)
            w = rng.uniform(0.05, 0.3)
            h = rng.uniform(0.5, 2.0)
            seq += h * np.exp(-((xx - c) ** 2) / (2.0 * w * w))
        starts.append(normalize_vec(seq))

    return starts


def interpolate_to_length(a, new_n):
    a = normalize_vec(a)
    old_n = a.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return a.copy()
    if old_n <= 1:
        return np.ones(new_n, dtype=np.float64) / new_n
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, a)
    return normalize_vec(g)


def try_transforms(a, time_limit):
    """Try power, shift, and blend transforms."""
    t0 = time.monotonic()
    a = normalize_vec(a)
    n = a.size
    best = a.copy()
    best_score = 2.0 * n * float(np.max(self_conv(best)))

    # Power transforms
    for g in (0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 3.0):
        if time.monotonic() - t0 > time_limit:
            break
        b = np.maximum(a, 1e-30) ** g
        b = normalize_vec(b)
        sc = 2.0 * n * float(np.max(self_conv(b)))
        if sc < best_score - 1e-13:
            best_score = sc
            best = b.copy()

    # Blend with uniform
    u = np.ones(n, dtype=np.float64) / n
    for w in (0.02, 0.05, 0.1, 0.2, 0.3):
        if time.monotonic() - t0 > time_limit:
            break
        b = (1 - w) * a + w * u
        b = normalize_vec(b)
        sc = 2.0 * n * float(np.max(self_conv(b)))
        if sc < best_score - 1e-13:
            best_score = sc
            best = b.copy()

    # Reverse blend
    for w in (0.1, 0.3, 0.5):
        if time.monotonic() - t0 > time_limit:
            break
        b = (1 - w) * a + w * a[::-1]
        b = normalize_vec(b)
        sc = 2.0 * n * float(np.max(self_conv(b)))
        if sc < best_score - 1e-13:
            best_score = sc
            best = b.copy()

    return best


def propose_candidate():
    """Main search function. Returns best sequence found within time budget."""
    t_global = time.monotonic()
    total_budget = 1800.0 - 120.0  # safety margin
    total_budget = max(100.0, total_budget)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = normalize_vec(a)
        if a.size < 2:
            return
        sc = 2.0 * float(a.size) * float(np.max(self_conv(a)))
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print("*** NEW BEST [%s]: %.9f n=%d t=%.1fs" % (label, sc, a.size, elapsed))

    # Try to use parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if p.ndim != 1:
            p = p.ravel()
        if p.size > 0 and float(p.sum()) > 1e-12:
            parent = p
            if parent.size > 10000:
                parent = interpolate_to_length(parent, 9000)
            elif parent.size < 3000:
                parent = interpolate_to_length(parent, 6000)
            print("Parent: n=%d score=%.9f" % (parent.size,
                  2.0 * parent.size * np.max(self_conv(normalize_vec(parent)))))
    except Exception:
        parent = None

    # Determine working length
    if parent is not None:
        n_main = int(parent.size)
    else:
        n_main = 6500

    # Generate diverse starts
    starts = make_diverse_starts(n_main, parent)

    # Score all starts
    scored = []
    for s in starts:
        s = normalize_vec(s)
        sc = 2.0 * n_main * float(np.max(self_conv(s)))
        scored.append((sc, s))
    scored.sort(key=lambda x: x[0])

    # Keep top diverse starts (similarity check)
    kept = []
    for sc, s in scored:
        too_similar = False
        for ks in kept:
            if float(np.dot(s, ks)) > 0.995:
                too_similar = True
                break
        if not too_similar:
            kept.append(s)
        if len(kept) >= 5:
            break

    if len(kept) == 0:
        kept = [normalize_vec(starts[0])]

    # Main optimization loop over diverse starts
    n_starts = len(kept)
    time_per_start = total_budget / max(1, n_starts)

    for idx, start in enumerate(kept[:3]):
        remaining = total_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break

        print("\n=== Start %d/%d: initial_score=%.9f, budget=%.0fs ==="
              % (idx + 1, n_starts, 2.0 * n_main * np.max(self_conv(start)), remaining))

        # Phase 1: Coarse optimization (smooth, large steps)
        t1 = min(remaining * 0.35, 400.0)
        if t1 > 30:
            out = smooth_descent(start, t1,
                                 tau_start=0.5, tau_end=10.0,
                                 lr_start=0.3, lr_end=0.05,
                                 rng_seed=100 + 7 * idx,
                                 verbose=True, noise_interval=2000,
                                 noise_scale=0.008)
            update_best(out, "coarse_%d" % idx)
            if best_seq is not None:
                start = best_seq.copy()

        # Phase 2: Medium optimization
        remaining = total_budget - (time.monotonic() - t_global)
        t2 = min(remaining * 0.30, 350.0)
        if t2 > 30:
            out = smooth_descent(start, t2,
                                 tau_start=5.0, tau_end=50.0,
                                 lr_start=0.1, lr_end=0.01,
                                 rng_seed=200 + 13 * idx,
                                 verbose=True, noise_interval=3000,
                                 noise_scale=0.004)
            update_best(out, "medium_%d" % idx)
            if best_seq is not None:
                start = best_seq.copy()

        # Phase 3: Fine optimization
        remaining = total_budget - (time.monotonic() - t_global)
        t3 = min(remaining * 0.25, 300.0)
        if t3 > 30:
            out = smooth_descent(start, t3,
                                 tau_start=20.0, tau_end=200.0,
                                 lr_start=0.03, lr_end=0.003,
                                 rng_seed=300 + 17 * idx,
                                 verbose=False, noise_interval=5000,
                                 noise_scale=0.002)
            update_best(out, "fine_%d" % idx)

    # Coordinate polish
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 60 and best_seq is not None:
        tp = min(remaining * 0.25, 200.0)
        if tp > 30:
            out = coordinate_polish(best_seq, tp)
            update_best(out, "coord_polish")

    # Transform search
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 40 and best_seq is not None:
        tt = min(remaining * 0.2, 100.0)
        if tt > 15:
            out = try_transforms(best_seq, tt)
            update_best(out, "transform_1")

    # Try different lengths
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 200 and best_seq is not None:
        cur_n = best_seq.size
        for dn in (-500, 500, -1000, 1000):
            remaining = total_budget - (time.monotonic() - t_global)
            if remaining < 150:
                break
            nn = cur_n + dn
            if nn < 3000 or nn > 10000:
                continue
            a_n = interpolate_to_length(best_seq, nn)
            tt = min(100.0, remaining * 0.35)
            if tt < 30:
                continue
            out = smooth_descent(a_n, tt,
                                 tau_start=5.0, tau_end=80.0,
                                 lr_start=0.08, lr_end=0.005,
                                 rng_seed=999 + dn,
                                 verbose=False, noise_interval=3000,
                                 noise_scale=0.003)
            update_best(out, "length_%d" % nn)

    # Final fine polish
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 40 and best_seq is not None:
        tf = min(remaining * 0.5, 200.0)
        if tf > 20:
            out = smooth_descent(best_seq, tf,
                                 tau_start=50.0, tau_end=500.0,
                                 lr_start=0.01, lr_end=0.001,
                                 rng_seed=5000,
                                 verbose=False, noise_interval=8000,
                                 noise_scale=0.001)
            update_best(out, "final_fine")

    # Final coordinate polish
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        tc = min(remaining * 0.6, 100.0)
        if tc > 15:
            out = coordinate_polish(best_seq, tc)
            update_best(out, "final_coord")

    # Final transforms
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 20 and best_seq is not None:
        out = try_transforms(best_seq, min(remaining * 0.7, 60.0))
        update_best(out, "final_transform")

    # Validate and return
    if best_seq is None:
        x = np.linspace(0.0, 1.0, 6500)
        best_seq = normalize_vec((x * (1 - x) + 1e-8))

    out_arr = np.asarray(best_seq, dtype=np.float64)
    out_arr = np.where(np.isfinite(out_arr), out_arr, 0.0)
    out_arr = np.maximum(out_arr, 0.0)
    out_arr = np.minimum(out_arr, 1000.0)
    s = float(out_arr.sum())
    if s <= 0.01:
        x = np.linspace(0.0, 1.0, 6500)
        out_arr = (x * (1 - x) + 1e-8)
        s = float(out_arr.sum())
    out_arr = out_arr / s

    # Verify with evaluator
    try:
        official = evaluate_sequence([float(v) for v in out_arr])
        if np.isfinite(official):
            print("Official evaluate_sequence: %.9f" % official)
        else:
            print("WARNING: evaluate_sequence returned non-finite!", official)
    except Exception as e:
        print("Warning: evaluate_sequence error:", e)

    internal = 2.0 * out_arr.size * float(np.max(self_conv(out_arr))) / (float(out_arr.sum()) ** 2)
    elapsed = time.monotonic() - t_global
    print("\nFINAL: internal=%.10f n=%d time=%.1fs" % (internal, out_arr.size, elapsed))

    # Ensure we return within time
    if time.monotonic() - t_global > 1750:
        print("Time warning, returning immediately")

    return [float(v) for v in out_arr]
