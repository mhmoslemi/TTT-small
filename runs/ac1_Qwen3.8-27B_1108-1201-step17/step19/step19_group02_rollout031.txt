Let me think about the LP-FW more carefully. The LP at each step is:

minimize M
subject to:
  sum_i g_i = 1
  g_i >= 0 for all i
  sum_i A[k,i] * g_i <= M for k in active_set

where A[k,i] = y[k-i] if 0 <= k-i < n, else 0.

The number of constraints is |active_set| + 1 (equality) + n (inequality). The number of variables is n + 1.

For n=6700 and |active_set|=2000, this is a LP with ~6700 variables and ~8701 constraints. This should be solvable in maybe 0.1-1 seconds with a good solver.

The key to speed is:
1. Keep the active set manageable (not too large)
2. Use a fast solver (Clarabel or PDLP)
3. Don't rebuild the constraint matrix from scratch every time (but with numpy, it's fast enough)

Let me estimate: if each LP takes 0.5 seconds, and we have 1000 seconds for LP-FW, we can do about 2000 iterations. That should be enough for good convergence.

Actually, let me reconsider the active set size. With n=6700, the convolution has 13399 values. The active set should contain the lags where the constraint is tight or nearly tight. At the optimum, maybe 100-500 lags are active. Starting with 500-1000 and growing to 2000-3000 should be fine.

Let me also think about the step size. The standard FW step size is alpha_t = 2/(t+2). But near the optimum, we want smaller steps. A better approach is to do a line search:

Given current y and new vertex g, find alpha that minimizes:
  f((1-alpha)*y + alpha*g) = max_k conv((1-alpha)*y + alpha*g, (1-alpha)*y + alpha*g)

This is a 1D optimization problem. We can solve it by bisection or golden section search.

Actually, since we're minimizing a convex function of alpha (the max of convex quadratic functions of alpha), we can use a simple line search:
- Try alpha in [0, 1]
- Evaluate the objective at a few points
- Pick the best

This is what the previous code does with its list of alphas. I'll keep this approach but make it more systematic.

One more thought: the previous code uses `fast_convolve` which is FFT-based. For n=6700, the FFT size is 16384 (next power of 2 for 2*6700-1=13399). This is fast.

But in the LP, we need to build the constraint matrix. For each active constraint k, we need the row A[k,:] which has at most n non-zero entries. Building this matrix for 2000 active constraints takes O(K*n) = O(2000*6700) = 13.4 million operations, which is fast in numpy.

One more important consideration: the `evaluate_sequence` function clamps values to [0, 1000]. So if any value in our sequence exceeds 1000, it will be clamped. For a normalized sequence (sum=1), this shouldn't be an issue since each value is at most 1.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-Frank-Wolfe with adaptive active set management and line search
2. Multi-phase strategy: initialization, coarse FW, fine FW, polish
3. Multi-length exploration around the best found length
4. Symmetry-aware coordinate descent polish
Starts from height_sequence_1 if available.
"""

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None

def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    return _MODULE_SOLVER

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_local(a):
    """Fast local evaluation assuming a is a numpy array, non-negative, sum > 0."""
    ac = fast_convolve(a)
    max_b = ac.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2 * len(a) * max_b / (sum_a ** 2))

def evaluate_via_api(a):
    """Evaluate using the actual evaluate_sequence function."""
    a_list = [float(x) for x in a]
    return evaluate_sequence(a_list)

def build_lp_matrix(y, active_set, n):
    """Build the constraint matrix for the LP step.
    For each active lag k, the constraint is sum_i y[k-i] * g_i <= M
    where the sum is over valid i (0 <= i < n and 0 <= k-i < n).
    """
    K = len(active_set)
    A = np.zeros((K, n))
    for jj, k in enumerate(active_set):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            A[jj, idx] = y[k_int - idx]
    return A

def lp_fw_step(y, active_set, n, solver):
    """Solve one LP-FW step: find g that minimizes M subject to constraints."""
    K = len(active_set)
    A = build_lp_matrix(y, active_set, n)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [cp.sum(g) == 1, A @ g <= M]
    prob = cp.Problem(cp.Minimize(M), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
    g_sum = g_val.sum()
    if g_sum < 1e-15:
        return None, None
    g_val = g_val / g_sum
    M_val = M.value
    return g_val, M_val

def line_search(y, g, n, max_evals=15):
    """Find best alpha in [0,1] to minimize max conv of (1-alpha)*y + alpha*g."""
    best_alpha = 0.5
    best_M = fast_convolve(y).max()
    
    # Try a set of alphas
    alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]
    
    for alpha in alphas:
        trial = (1.0 - alpha) * y + alpha * g
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s < 1e-15:
            continue
        trial = trial / s
        M_trial = fast_convolve(trial).max()
        if M_trial < best_M:
            best_M = M_trial
            best_alpha = alpha
    
    return best_alpha, best_M

def run_lp_fw(a0, time_limit, solver, initial_K=300, max_K=3000, 
              add_per_iter=50, verbose=True):
    """Run LP-Frank-Wolfe for the given time limit."""
    n = len(a0)
    y = norm_sum1(a0)
    best_y = y.copy()
    best_m = fast_convolve(y).max()
    t0 = time.monotonic()
    
    # Initialize active set with top-K convolution values
    ac = fast_convolve(y)
    active_set = sorted(np.argsort(ac)[::-1][:initial_K].tolist())
    
    no_improve = 0
    max_no_improve = 2000
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        elapsed = time.monotonic() - t0
        
        # Check if we should stop due to stagnation
        if no_improve > max_no_improve:
            break
        
        # Ensure active set doesn't exceed max_K
        if len(active_set) > max_K:
            ac_curr = fast_convolve(y)
            M_curr = ac_curr.max()
            margins = np.array([M_curr - ac_curr[k] for k in active_set])
            # Keep the ones with smallest margin (most binding)
            keep_idx = np.argsort(margins)[:max_K]
            active_set = sorted([active_set[i] for i in keep_idx])
        
        # Solve LP
        g_val, M_lp = lp_fw_step(y, active_set, n, solver)
        if g_val is None:
            # LP failed, reduce active set
            active_set = active_set[:initial_K]
            continue
        
        # Compute conv(y, g) to find new active constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_val, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        # Update active set: add violated constraints
        ac_curr = fast_convolve(y)
        M_curr = ac_curr.max()
        violations = np.where(conv_ys > M_curr + 1e-12)[0]
        if len(violations) > 0:
            # Sort by violation amount, add top ones
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1]][:add_per_iter]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)
        
        # Also periodically add top conv(y,y) values
        if iter_count % 50 == 0:
            top_conv = np.argsort(ac_curr)[::-1][:initial_K]
            for k in top_conv:
                if int(k) not in set(active_set):
                    active_set.append(int(k))
            active_set = sorted(active_set)
        
        # Line search for step size
        alpha, M_trial = line_search(y, g_val, n)
        
        # Update
        if iter_count <= 3:
            y_new = g_val.copy()
        else:
            y_new = (1.0 - alpha) * y + alpha * g_val
            y_new = np.maximum(y_new, 0.0)
            s = y_new.sum()
            if s < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s
        
        # Check improvement
        M_new = fast_convolve(y_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_y = y_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        y = y_new
        
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  FW iter={:d} ({:.1f}s): J={:.8f}, |S|={}".format(
                iter_count, elapsed, J, len(active_set)))
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("  FW done ({:.1f}s): J={:.8f}, iters={}".format(elapsed, J, iter_count))
    return best_y, best_m

def gradient_polish(a0, time_limit, verbose=True):
    """Gradient-based polish using softmax-weighted gradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    stall = 0
    max_stall = 5000
    learning_rate = 0.5
    decay = 0.999
    
    top_k_values = [30, 50, 80, 100, 150, 200]
    cur_top_k = 50
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break
        
        ac = fast_convolve(a)
        M_curr = ac.max()
        
        # Compute softmax-weighted gradient
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        # Use temperature-scaled softmax
        temp = M_curr / (1.0 + 0.01 * (t // 1000))
        w = np.exp(top_vals / temp - top_vals.max() / temp)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * a[k_int - idx]
        
        # Mirror descent step
        eta = learning_rate / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        M_new = fast_convolve(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            a = a_new  # Still take the step for exploration
        if M_new <= best_m + 1e-12:
            a = a_new
        
        # Adaptive parameters
        if t % 2000 == 0:
            learning_rate *= decay
            cur_top_k = min(cur_top_k + 10, len(ac))
        
        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  Grad t={:d} ({:.1f}s): J={:.8f}".format(t, elapsed, J))
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=True):
    """Coordinate descent polish: move mass from peak-contributing positions to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 1000
    n_top = 30
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-14:
                continue
            
            # Score each position by its total contribution to top peaks
            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]
            
            # Find position with minimum score (least impact on peaks)
            dst = int(np.argmin(scores))
            if dst == src:
                dst = int(np.argmin(np.where(np.arange(n) == src, 1e10, scores)))
            
            # Try to move mass
            found = False
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = src_val * frac
                if amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break
            
            if not found:
                # Try random destination when stalled
                if stall > 200 and src_val > 1e-10:
                    dst_rand = int(rng.integers(n))
                    if dst_rand != src:
                        frac = rng.uniform(0.002, 0.01)
                        amt = src_val * frac
                        a[src] -= amt
                        a[dst_rand] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True
        
        if not improved:
            stall += 1
            # Random perturbation when stalled
            if stall % 50 == 0:
                for _ in range(3):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.01)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
        
        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  Coord ({:.1f}s): J={:.8f}, stall={}".format(elapsed, J, stall))
    
    return best_a, best_m

def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()

        return np.ones_like(a) / len(a)

def build_lp_matrix(y, active_set, n):
    K = len(active_set)
    A = np.zeros((K, n))
    for jj in range(K):
        k = int(active_set[jj])
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            A[jj, idx] = y[k - idx]
    return A

def solve_lp-fw(y, active_set, n, solver):
    K = len(active_set)
    if K == 0:
        return None, None
    A = build_lp_matrix(y, active_set, n)
    g = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g) == 1, A @ g <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver, verbose=False)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
    g_sum = g_val.sum()
    if g_sum < 1e-15:
        return None, None
    g_val = g_val / g_sum
    return g_val, M_var.value

def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        candidates.append(norm_sum1((n - i_arr) ** p))
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        for beta in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            candidates.append(norm_sum1(a))
    for c in [0.15, 0.2, 0.25, 0.3, 0.35]:
        for sigma in [0.05, 0.08, 0.1, 0.12, 0.15, 0.2, 0.3]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)) + np.exp(-((x - (1.0 - c)) ** 2) / (2.0 * sigma * sigma))
            candidates.append(norm_sum1(a))
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
        candidates.append(norm_sum1(x ** p + (1.0 - x) ** p))
    for freq in [1, 2, 3, 5, 7]:
        for ph in [0.0, np.pi / 2.0, np.pi / 3.0]:
            a = np.maximum(1.0 + np.cos(2.0 * np.pi * freq * x + ph), 0.0)
            candidates.append(norm_sum1(a))
    for rate in [0.5, 1, 2, 3, 5]:
        a = np.exp(-rate * x) + np.exp(rate * (x - 1.0))
        candidates.append(norm_sum1(a))

    rng = np.random.default_rng(2024)
    for trial in range(100):
        if time.monotonic() - t0 > time_limit:
            break
        a = rng.random(n)
        a = a ** rng.uniform(0.3, 4.0)
        candidates.append(norm_sum1(a))

    for a in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        v = evaluate_local(a)
        if v < best_val:
            best_val = v
            best_a = a.copy()

    return best_a, best_val
