def fast_autocorr(a):

    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def multiplicative_grad_desc(a0, time_limit, top_k=100, temperature=30.0, c0=3.0, verbose=True):
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    x_prev = x.copy()
    stall = 0
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 10000:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute autocorrelation
        Yf = np.fft.rfft(y, fft_size)
        ac = np.fft.irfft(Yf * Yf, fft_size)[:size]
        M = ac.max()
        
        # Select top-k lags with softmax weights
        k_max = min(top_k, size)
        idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[idx]
        # Use temperature-scaled softmax
        w = np.exp(temperature * (top_vals - top_vals[0]))
        w = w / w.sum()
        
        # Compute gradient: d(max sum)/da ≈ sum_w w_k * (ac gradient at lag k)
        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            idx_arr = np.arange(i_s, i_e)
            contrib = y[ki - idx_arr]
            grad[idx_arr] += 2.0 * wi * contrib
        
        # Multiplicative update
        eta_t = c0 / (np.sqrt(max(t, 1)) + 1.0)
        log_a = np.log(y) - eta_t * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        val = obj_val(x)
        if val < best_val - 1e-12:
            best_val = val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Random restart on long stall
        if stall > 3000 and stall % 3000 == 0:
            noise_scale = min(0.1, 5.0 / np.sqrt(n))
            noise = noise_scale * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            stall = 0
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    MG t={} ({}s) J={}".format(t, round(elapsed,1), repr(best_val)))
    
    return best_a, best_val

def lp_fractional_wallflower(a0, time_limit, solver, init_K=400, max_K=2500, verbose=True):
    n = len(a0)
    if n > 12000:
        print("    LP-FW: n too large, skipping")
        return a0, obj_val(a0)
    
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    ac_y = fast_autocorr(y)
    active_set = list(np.argsort(ac_y)[::-1][:init_K].astype(int))
    no_improve = 0
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 800:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Manage active set
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep = np.argsort(margins)[:max_K]
            active_set = [int(active_set[i]) for i in keep]
        
        K = len(active_set)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs = np.zeros((K, n), dtype=np.float64)
        for jj, k in enumerate(active_set):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        lp_ok = False
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                lp_ok = True
        except Exception:
            if len(active_set) > init_K:
                active_set = list(np.argsort(ac)[::-1][:init_K].astype(int))
        
        if not lp_ok:
            if no_improve > 400:
                break
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Check violations and add to active set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                if int(k) not in active_set:
                    active_set.append(int(k))
        
        # Line search: golden section
        lo = 0.0
        hi = 1.0
        best_alpha = 2.0 / (t + 2.0)
        best_trial = obj_val(np.maximum((1-best_alpha)*y + best_alpha*g_dir, 0.0))
        
        for _ in range(8):
            alpha1 = lo + (hi - lo) * 0.382
            alpha2 = hi - (hi - lo) * 0.382
            t1 = np.maximum((1-alpha1)*y + alpha1*g_dir, 0.0)
            t2 = np.maximum((1-alpha2)*y + alpha2*g_dir, 0.0)
            s1, s2 = t1.sum(), t2.sum()
            if s1 > 1e-15:
                t1 = t1 / s1
            if s2 > 1e-15:
                t2 = t2 / s2
            m1 = obj_val(t1) if s1 > 1e-15 else 1e10
            m2 = obj_val(t2) if s2 > 1e-15 else 1e10
            if m1 < m2:
                hi = alpha2
            else:
                lo = alpha1
            if m1 < best_trial:
                best_trial = m1
                best_alpha = alpha1
            if m2 < best_trial:
                best_trial = m2
                best_alpha = alpha2
        
        best_alpha = (lo + hi) / 2.0
        y = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        if verbose and t % 30 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t={} ({}s) J={} |S|={}".format(t, round(elapsed,1), repr(2*n*best_m), len(active_set)))
    
    return best_a, 2.0 * n * best_m

def coord_descent_polish(a0, time_limit, seed=42, verbose=True):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    size = 2 * n - 1
    
    while time.monotonic() - t0 < time_limit and stall < 1500:
        ac = fast_autocorr(a)
        n_top = min(30, size)
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False
        
        # Try to reduce top peaks by moving mass
        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find destination: minimize score (contribution to other top peaks)
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]
            scores[src] = 1e10
            dst = int(np.argmin(scores))
            if dst == src:
                continue
            
            # Try different transfer fractions
            found = False
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t / st
                Mt = fast_autocorr(a_t).max()
                if Mt < best_m - 1e-15:
                    a = a_t
                    best_m = Mt
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            
            if not found and stall > 200:
                # Random move
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.001, 0.005)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if not improved:
            stall += 1
            if stall % 200 == 0 and stall > 100:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.005)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
        
        if verbose and stall % 500 == 0:
            print("    Coord ({}s) J={} stall={}".format(round(time.monotonic()-t0,1), repr(2*n*best_m), stall))
    
    return best_a, 2.0 * n * best_m

For an asymmetric sequence, the center lag `c[n-1] = sum a[i]*a[n-1-i]` can be less than `sum a[i]^2` (by Cauchy-Schwarz, it's ≤ `sum a[i]^2` with equality iff a is symmetric).
