def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:

    if a.size == 0:

def weighted_conv_gradient(a, lag_weights):
    """
    Compute the gradient of sum_k w[k] * (a*a)[k] w.r.t. a.
    This equals 2 * sum_k w[k] * (a shifted by -k) (truncated to [0,n)).
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    m = lag_weights.size
    # For each lag k, (a*a)[k] = sum_i a[i] * a[i+k-n+1] ... 
    # More precisely: for k in [0, 2n-2], (a*a)[k] = sum_{i} a[i] * a[i - k + n - 1] (with bounds)
    # The gradient w.r.t. a[j] is: sum_k w[k] * (partial (a*a)[k] / partial a[j])
    # = sum_k w[k] * a[j - k + n - 1] (with appropriate bounds checking)
    # This is a correlation-like operation.
    # Use FFT for efficiency:
    # sum_k w[k] * corr(a, k)[j] where corr(a,k)[j] = a[(j-k) mod ...]
    # Actually, let's use the FFT cross-correlation approach.
    
    # The gradient is: grad = 2 * (w * a) in a specific correlation sense
    # Let's compute it as: for each active lag k with weight w[k],
    # add w[k] * a_shifted where a_shifted[i] = a[i-k] (with zero padding)
    
    # Efficient FFT-based computation:
    # The "blurred" shift sum is: sum_k w[k] * a_{-k} * a  (cross-correlation)
    # Which is: real( FFT(a) * conj(FFT(w_padded)) )
    
    size = next_pow2(2 * n + m)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(lag_weights, size)
    # Cross-correlation: sum_k w[k] * a[i-k]
    c = np.fft.irfft(fa * fw.conj(), size)[:n]
    return 2.0 * c


def quad_max_val(alpha, q0, c, qd):
    """Compute max of (1-a)^2*q0 + 2*a*(1-a)*c + a^2*qd at given alpha."""
    a1 = 1.0 - alpha
    val = (a1 * a1) * q0 + (2.0 * alpha * a1) * c + (alpha * alpha) * qd
    return float(np.max(val))


def line_search_exact(a, d, q0):
    """
    Minimize max of (1-t)*a + t*d self-convolution over t in [0,1].
    Uses exact quadratic form and golden section + parabolic refinement.
    """
    a = np.asarray(a, dtype=np.float64)
    d = np.asarray(d, dtype=np.float64)
    q0 = np.asarray(q0, dtype=np.float64)
    M0 = float(np.max(q0))
    
    c = conv_ab(a, d)
    qd = self_conv(d)
    
    def f(t):
        return quad_max_val(t, q0, c, qd)
    
    # Golden section search
    phi = (math.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c1 = hi - phi * (hi - lo)
    d1 = lo + phi * (hi - lo)
    f1, f2 = f(c1), f(d1)
    
    for _ in range(40):
        if f1 < f2:
            hi = d1
            d1 = c1
            f2 = f1
            c1 = hi - phi * (hi - lo)
            f1 = f(c1)
        else:
            lo = c1
            c1 = d1
            f1 = f2
            d1 = lo + phi * (hi - lo)
            f2 = f(d1)
    
    t_best = 0.5 * (lo + hi)
    M_best = f(t_best)
    
    # Also evaluate some fixed points for safety
    for t_eval in [0.0, 0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        v = f(t_eval)
        if v < M_best:
            M_best = v
            t_best = t_eval
    
    if t_best <= 0.0 or M_best >= M0 - 1e-14:
        return a.copy(), M0
    
    new = (1.0 - t_best) * a + t_best * d
    new = np.maximum(new, 0.0)
    s = float(new.sum())
    if s <= 1e-15:
        return a.copy(), M0
    new = new / s
    return new, M_best


def solve_lp_direction(y, active_k, q_vals, n):
    """
    Frank-Wolfe LP: find g >= 0, sum(g)=1, minimizing T
    s.t. 2*(y*g)[k] - T <= q[k] for k in active set.
    
    Returns (g, T) or (None, None) on failure.
    """
    if not HAVE_LP:
        return None, None
    
    K = len(active_k)
    if K <= 0 or K > 800:
        return None, None
    if K * (n + 1) > 12_000_000:
        return None, None
    
    try:
        A = np.zeros((K, n + 1), dtype=np.float64)
    except Exception:
        return None, None
    
    yy = 2.0 * np.asarray(y, dtype=np.float64)
    for r, k in enumerate(active_k):
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            A[r, i0:i1] = yy[start:stop][::-1]
    A[:, n] = -1.0
    
    b = np.array(q_vals, dtype=np.float64)
    b = np.maximum(b, 0.0)
    
    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0
    Aeq = np.ones((1, n + 1), dtype=np.float64)
    Aeq[0, n] = 0.0
    bounds = [(0.0, np.inf)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=[1.0],
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None, None
    
    if res.status != 0 or res.x is None:
        return None, None
    
    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-10:
        return None, None
    g = g / s
    T = float(res.x[n])
    if not np.isfinite(T):
        T = 0.0
    return g, T


def mirror_direction(a, q, temperature=1.0, trust=0.5):
    """
    Compute a mirror descent direction using softmax-weighted gradient.
    Returns normalized direction d.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    w = softmax_lag_weights(q, temperature)
    grad = weighted_conv_gradient(a, w)
    # Mirror step: d = a * exp(-trust * grad / max_grad)
    gmax = float(np.max(np.abs(grad)))
    if gmax < 1e-15:
        return a.copy()
    move = trust * grad / gmax
    move = np.clip(move, -50.0, 50.0)
    d = a * np.exp(-move)
    d = np.maximum(d, 1e-30)
    s = float(d.sum())
    if s <= 1e-15:
        return a.copy()
    return d / s


def peak_peel_direction(a, q, top_k=5):
    """
    Create a direction that specifically targets the top-k peaks.
    For each peak lag, compute the correlation and move mass away from
    the positions that contribute most to those peaks.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = q.size
    
    order = np.argsort(q)[::-1]
    peaks = order[:min(top_k, m)]
    
    # Build a "peak indicator" vector
    peak_w = np.zeros(m, dtype=np.float64)
    for i, k in enumerate(peaks):
        d = float(q[k] - q[order[min(top_k + 1, m - 1)]])
        peak_w[k] = max(d, 0.0) + 1.0
    
    # Normalize
    ps = float(peak_w.sum())
    if ps > 0:
        peak_w /= ps
    
    grad = weighted_conv_gradient(a, peak_w)
    gmax = float(np.max(np.abs(grad)))
    if gmax < 1e-15:
        return a.copy()
    
    move = 0.8 * grad / gmax
    move = np.clip(move, -50.0, 50.0)
    d = a * np.exp(-move)
    d = np.maximum(d, 1e-30)
    s = float(d.sum())
    if s <= 1e-15:
        return a.copy()
    return d / s


def diverse_starts(n, parent=None):
    """Generate diverse initial sequences."""
    out = []
    x = np.linspace(0.0, 1.0, n)
    
    if parent is not None:
        p = normalize_vec(parent)
        if p.size != n:
            p = np.interp(x, np.linspace(0, 1, p.size), p)
            p = normalize_vec(p)
        out.append(p)
        out.append(p[::-1].copy())
        for w in (0.3, 0.5):
            out.append(normalize_vec((1-w)*p + w*p[::-1]))
    
    # Sine-based
    out.append(normalize_vec(np.sin(np.pi * x)**2))
    out.append(normalize_vec(np.sin(np.pi * x)**4))
    out.append(normalize_vec(np.sin(np.pi * x)**0.5))
    
    # Gaussian
    for mu in (0.5,):
        for sig in (0.15, 0.2, 0.25, 0.35):
            out.append(normalize_vec(np.exp(-0.5*((x-mu)/sig)**2)))
    
    # Bimodal
    for mu1, mu2 in ((0.25, 0.75), (0.2, 0.8), (0.3, 0.7)):
        for sig in (0.08, 0.12, 0.18):
            seq = np.exp(-0.5*((x-mu1)/sig)**2) + np.exp(-0.5*((x-mu2)/sig)**2)
            out.append(normalize_vec(seq))
    
    # Raised cosine
    out.append(normalize_vec(0.5 * (1 + np.cos(2 * np.pi * x))))
    out.append(normalize_vec(0.5 * (1 + np.cos(4 * np.pi * x))))
    
    # Polynomial bases
    for deg in (1, 2, 3):
        out.append(normalize_vec((x + 1e-8)**deg))
        out.append(normalize_vec(((1-x) + 1e-8)**deg))
    
    # Chebyshev-like
    theta = np.arccos(1 - 2*x)
    for k in (1, 2, 3, 5):
        out.append(normalize_vec(np.abs(np.cos(k * theta)) + 1e-8))
    
    # Power of parent
    if parent is not None:
        p = normalize_vec(parent)
        if p.size != n:
            p = np.interp(x, np.linspace(0, 1, p.size), p)
            p = normalize_vec(p)
        base = np.maximum(p, 1e-20)
        for g in (0.5, 0.7, 1.5, 2.0, 3.0):
            out.append(normalize_vec(base**g))
    
    # Random smooth
    rng = np.random.default_rng(42)
    for _ in range(15):
        seq = np.zeros(n, dtype=np.float64)
        n_peaks = int(rng.integers(1, 5))
        for _ in range(n_peaks):
            ctr = rng.uniform(0.1, 0.9)
            sig = rng.uniform(0.05, 0.3)
            ht = rng.uniform(0.3, 1.0)
            seq += ht * np.exp(-0.5*((x-ctr)/sig)**2)
        seq += 1e-5
        out.append(normalize_vec(seq))
    
    return out


def interpolate_seq(a, new_n):
    a = normalize_vec(a)
    old_n = a.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, a)
    return normalize_vec(g)


def local_search_improved(a0, time_limit, rng_seed=123, verbose=False):
    """
    Main local search combining LP-FW, mirror descent, and peak peeling.
    """
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    best = a.copy()
    best_score = objective_norm(a)
    
    rng = np.random.default_rng(rng_seed)
    no_improve = 0
    total_iters = 0
    last_print = t0
    
    # Adaptive parameters
    temperature = 0.5 * float(np.max(self_conv(a)))
    trust = 0.4
    lp_enabled = HAVE_LP
    lp_fails = 0
    
    # Active set K
    if n > 7000:
        K_init, K_max = 150, 400
    elif n > 4000:
        K_init, K_max = 200, 500
    else:
        K_init, K_max = 250, 600
    K = K_init
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        
        q = self_conv(a)
        M_cur = float(np.max(q))
        score_cur = 2.0 * n * M_cur
        
        if score_cur < best_score - 1e-13:
            best = a.copy()
            best_score = score_cur
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive temperature
        temperature = max(0.1 * M_cur, 1e-10)
        
        # Choose direction strategy based on no_improve
        improved = False
        
        # Strategy 1: LP-FW (primary)
        if lp_enabled and no_improve < 80 and remaining > 10:
            order = np.argsort(q)[::-1]
            # Threshold-based active set
            threshold = M_cur - 0.05 * M_cur  # within 5% of max
            act_idx = order[q[order] > threshold]
            act_idx = act_idx[:min(K, len(act_idx))]
            
            if len(act_idx) >= 10:
                t_lp = time.monotonic()
                g_dir, T = solve_lp_direction(a, act_idx, q[act_idx], n)
                lp_time = time.monotonic() - t_lp
                
                if g_dir is None:
                    lp_fails += 1
                    if lp_fails >= 5:
                        lp_enabled = False
                else:
                    lp_fails = 0
                    if lp_time > max(20, 0.1 * remaining):
                        lp_enabled = False
                    else:
                        new, Mnew = line_search_exact(a, g_dir, q)
                        if Mnew < M_cur - 1e-13:
                            a = new
                            improved = True
                        elif T < M_cur - 1e-10:
                            # LP says improvement possible but line search didn't find it
                            # Try a smaller step
                            small_t = 0.1
                            cand = (1 - small_t) * a + small_t * g_dir
                            cand = np.maximum(cand, 0)
                            cs = cand.sum()
                            if cs > 1e-15:
                                cand /= cs
                                Mc = float(np.max(self_conv(cand)))
                                if Mc < M_cur - 1e-13:
                                    a = cand
                                    improved = True
        elif lp_enabled and no_improve >= 80:
            # LP is slow to help, switch to gradient methods
            pass
        
        # Strategy 2: Mirror gradient (secondary)
        if not improved and remaining > 5:
            for trust_try in (trust, trust * 0.5, trust * 1.5):
                d = mirror_direction(a, q, temperature, trust_try)
                new, Mnew = line_search_exact(a, d, q)
                if Mnew < M_cur - 1e-13:
                    a = new
                    improved = True
                    trust = trust_try
                    break
                if Mnew < M_cur:
                    # Small improvement, accept
                    a = new
                    M_cur = Mnew
                    improved = True
                    break
        
        # Strategy 3: Peak peeling (tertiary)
        if not improved and remaining > 5:
            d = peak_peel_direction(a, q, top_k=min(5, n // 10))
            new, Mnew = line_search_exact(a, d, q)
            if Mnew < M_cur - 1e-13:
                a = new
                improved = True
                M_cur = Mnew
        
        # Strategy 4: Random perturbation (exploration)
        if not improved and no_improve > 30 and remaining > 5:
            for _ in range(3):
                noise_scale = 0.01 + 0.005 * (no_improve // 10)
                noise = rng.normal(size=n) * noise_scale
                cand = a * (1.0 + noise)
                cand = np.maximum(cand, 1e-30)
                cand = normalize_vec(cand)
                Mc = objective_norm(cand)
                if Mc < best_score + 0.001:
                    # Accept even if slightly worse (simulated annealing)
                    if no_improve > 50 and rng.random() < 0.3:
                        a = cand
                    elif Mc < best_score - 1e-13:
                        a = cand
                        improved = True
                    break
        
        # Adapt K
        if improved:
            K = max(K_init, K - 10)
        else:
            K = min(K + 20, K_max)
        
        # Adapt trust
        if improved:
            trust = min(trust * 1.05, 3.0)
        else:
            trust = max(trust * 0.9, 0.05)
        
        # Periodic larger perturbation
        if no_improve > 100 and no_improve % 50 == 0 and remaining > 20:
            scale = 0.03
            cand = best * (1.0 + scale * rng.normal(size=best.size))
            cand = np.maximum(cand, 1e-30)
            cand = normalize_vec(cand)
            sc = objective_norm(cand)
            if sc < best_score * 1.01:
                a = cand
            else:
                a = best.copy()
            no_improve = 0
            trust = 0.4
            K = K_init
        
        # Power transform escape
        if no_improve > 150 and remaining > 20:
            for g_exp in (0.7, 1.3, 1.5, 0.5):
                cand = np.maximum(a, 1e-30)**g_exp
                cand = normalize_vec(cand)
                sc = objective_norm(cand)
                if sc < best_score - 1e-12:
                    a = cand
                    best = cand.copy()
                    best_score = sc
                    no_improve = 0
                    break
        
        total_iters += 1
        if verbose and (total_iters % 50 == 0 or (time.monotonic() - last_print) > 30):
            print("  LS: it=%d t=%.0fs best=%.9f cur=%.9f K=%d noimp=%d lp=%s trust=%.3f"
                  % (total_iters, elapsed, best_score, 2.0*n*float(np.max(self_conv(a))),
                     K, no_improve, lp_enabled, trust))
            last_print = time.monotonic()
        
        if no_improve > 300:
            break
    
    return best


def coordinate_descent(a0, time_limit, rng_seed=777):
    """
    Coordinate descent: move mass between pairs of coordinates.
    Focus on positions that contribute to the peak.
    """
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    best = a.copy()
    best_score = objective_norm(a)
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 150:
        q = self_conv(a)
        top_lags = np.argsort(q)[::-1][:min(5, q.size)]
        
        # Compute contribution score for each position
        contrib = np.zeros(n, dtype=np.float64)
        for k in top_lags:
            k = int(k)
            i0 = max(0, k - n + 1)
            i1 = min(n, k + 1)
            if i0 < i1:
                start = k - (i1 - 1)
                stop = k - i0 + 1
                contrib[i0:i1] += a[start:stop][::-1]
        
        # Find best donor (high contrib, high value) and best receiver (low contrib)
        src_order = np.argsort(contrib * a)[::-1]
        dst_order = np.argsort(contrib)
        
        improved = False
        for si in src_order[:8]:
            if improved or time.monotonic() - t0 > time_limit:
                break
            si = int(si)
            if a[si] < 1e-10:
                continue
            for di in dst_order[:8]:
                di = int(di)
                if di == si:
                    continue
                for frac in (0.05, 0.1, 0.15, 0.25):
                    amt = a[si] * frac
                    if amt < 1e-15:
                        continue
                    b = a.copy()
                    b[si] -= amt
                    b[di] += amt
                    b = np.maximum(b, 0)
                    bs = b.sum()
                    if bs < 1e-15:
                        continue
                    b /= bs
                    sc = objective_norm(b)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best = b.copy()
                        a = b
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 30 == 0:
                # Random move
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    b = a.copy()
                    amt = a[i] * 0.03
                    b[i] -= amt
                    b[j] += amt
                    b = np.maximum(b, 0)
                    bs = b.sum()
                    if bs > 1e-15:
                        b /= bs
                        sc = objective_norm(b)
                        if sc < best_score - 1e-13:
                            best_score = sc
                            best = b.copy()
                            a = b
                            stall = 0
    
    return best


def power_transform_search(a0, time_limit):
    """Try various power transforms and blending with uniform."""
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    best = a.copy()
    best_score = objective_norm(a)
    
    base = np.maximum(a, 1e-30)
    for g in np.arange(0.3, 4.0, 0.15):
        if time.monotonic() - t0 > time_limit:
            break
        b = base ** g
        b = normalize_vec(b)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()
    
    # Blend with uniform
    u = np.ones(n, dtype=np.float64) / n
    for w in np.arange(0.01, 0.6, 0.03):
        if time.monotonic() - t0 > time_limit:
            break
        b = (1-w) * best + w * u
        b = normalize_vec(b)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()
    
    # Blend with reverse
    rev = best[::-1].copy()
    for w in np.arange(0.05, 0.5, 0.05):
        if time.monotonic() - t0 > time_limit:
            break
        b = (1-w) * best + w * rev
        b = normalize_vec(b)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()
    
    return best


def trim_search(a0, time_limit):
    """Try trimming from both ends."""
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    best = a.copy()
    best_score = objective_norm(a)
    
    max_side = min(n // 2, 5000)
    # Fibonacci-like samples
    samples = list(range(0, max_side + 1, max(1, max_side // 20)))
    if max_side not in samples:
        samples.append(max_side)
    
    for l in samples:
        if time.monotonic() - t0 > time_limit:
            break
        if l >= n - 5:
            continue
        b = normalize_vec(a[l:])
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()
    
    for r in samples:
        if time.monotonic() - t0 > time_limit:
            break
        if r >= n - 5:
            continue
        b = normalize_vec(a[:-r] if r > 0 else a)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()
    
    # Fine-tune around best
    cur_n = best.size
    for dl in range(-50, 51, 5):
        if time.monotonic() - t0 > time_limit:
            break
        new_n = cur_n + dl
        if new_n < 100 or new_n > n:
            continue
        b = interpolate_seq(best, new_n)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()
    
    return best


def propose_candidate():

    # Brief polish on top 3 starts
    for idx in range(min(3, len(kept))):
        elapsed = time.monotonic() - t_global
        if total_budget - elapsed < 60:
            break
        t_lim = min(30.0, (starts_budget - (time.monotonic() - t_global)) * 0.3)
        if t_lim < 10:
            t_lim = 10
        t_lim = min(t_lim, total_budget - (time.monotonic() - t_global) - 10)
        if t_lim < 5:
            break
        out = local_search_improved(kept[idx], t_lim, rng_seed=100 + idx, verbose=False)
        sc = objective_norm(out)
        if sc < best_score - 1e-12:
            best_score = sc
            best_seq = out.copy()
            print("Phase1 start%d: score=%.9f n=%d t=%.0fs" % (idx, sc, out.size, time.monotonic() - t_global))
    
    # Phase 2: Main local search on best
    elapsed = time.monotonic() - t_global
    remaining = total_budget - elapsed
    main_budget = min(1000.0, 0.55 * remaining)
    if main_budget > 60:
        print("Phase 2: main local search, budget=%.0fs" % main_budget)
        out = local_search_improved(best_seq, main_budget, rng_seed=42, verbose=True)
        sc = objective_norm(out)
        if sc < best_score - 1e-12:
            best_score = sc
            best_seq = out.copy()
            print("Phase2 main: score=%.9f" % sc)
    
    # Phase 3: Try second diverse start with remaining time
    elapsed = time.monotonic() - t_global
    remaining = total_budget - elapsed
    if len(kept) >= 2 and remaining > 200:
        cand2 = kept[1]
        too_similar = (cand2.size == best_seq.size and float(np.dot(cand2, best_seq)) > 0.99)
        if too_similar and len(kept) >= 3:
            cand2 = kept[2]
        if not too_similar:
            t2 = min(250.0, 0.3 * remaining)
            if t2 > 50:
                out = local_search_improved(cand2, t2, rng_seed=200, verbose=False)
                sc = objective_norm(out)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_seq = out.copy()
                    print("Phase3 second: score=%.9f" % sc)
    
    # Phase 4: Coordinate descent polish
    elapsed = time.monotonic() - t_global
    remaining = total_budget - elapsed
    if remaining > 60:
        t4 = min(120.0, 0.3 * remaining)
        if t4 > 20:
            out = coordinate_descent(best_seq, t4, rng_seed=999)
            sc = objective_norm(out)
            if sc < best_score - 1e-12:
                best_score = sc
                best_seq = out.copy()
                print("Phase4 coord: score=%.9f" % sc)
    
    # Phase 5: Power transform + trim
    elapsed = time.monotonic() - t_global
    remaining = total_budget - elapsed
    if remaining > 40:
        t5 = min(60.0, 0.4 * remaining)
        if t5 > 10:
            out = power_transform_search(best_seq, t5)
            sc = objective_norm(out)
            if sc < best_score - 1e-12:
                best_score = sc
                best_seq = out.copy()
                print("Phase5 power: score=%.9f" % sc)
    
    elapsed = time.monotonic() - t_global
    remaining = total_budget - elapsed
    if remaining > 40:
        t6 = min(60.0, 0.4 * remaining)
        if t6 > 10:
            out = trim_search(best_seq, t6)
            sc = objective_norm(out)
            if sc < best_score - 1e-12:
                best_score = sc
                best_seq = out.copy()
                print("Phase6 trim: score=%.9f" % sc)
    
    # Phase 6: Multi-length exploration
    elapsed = time.monotonic() - t_global
    remaining = total_budget - elapsed
    if remaining > 200:
        cur_n = best_seq.size
        for dn in (-500, -1000, 500, 1000):
            elapsed = time.monotonic() - t_global
            if total_budget - elapsed < 80:
                break
            nn = cur_n + dn
            if nn < 3000 or nn > 9000:
                continue
            a_interp = interpolate_seq(best_seq, nn)
            t_lim = min(80.0, 0.2 * (total_budget - elapsed))
            if t_lim < 20:
                continue
            out = local_search_improved(a_interp, t_lim, rng_seed=500 + dn, verbose=False)
            sc = objective_norm(out)
            if sc < best_score - 1e-12:
                best_score = sc
                best_seq = out.copy()
                print("Phase6 n=%d: score=%.9f" % (nn, sc))
    
    # Phase 7: Final coordinate + mirror polish
    elapsed = time.monotonic() - t_global
    remaining = total_budget - elapsed
    if remaining > 50:
        t7 = min(80.0, 0.4 * remaining)
        if t7 > 15:
            out = coordinate_descent(best_seq, t7, rng_seed=31415)
            sc = objective_norm(out)
            if sc < best_score - 1e-12:
                best_score = sc
                best_seq = out.copy()
    
    # Phase 8: Final micro-optimization with very short LP
    elapsed = time.monotonic() - t_global
    remaining = total_budget - elapsed
    if remaining > 30:
        t8 = min(50.0, 0.5 * remaining)
        if t8 > 10:
            out = local_search_improved(best_seq, t8, rng_seed=7777, verbose=False)
            sc = objective_norm(out)
            if sc < best_score - 1e-12:
                best_score = sc
                best_seq = out.copy()
    
    # Final validation
    if best_seq is None:
        x = np.linspace(0, 1, 6500)
        best_seq = normalize_vec(np.sin(np.pi * x)**2)
    
    out_arr = np.asarray(best_seq, dtype=np.float64)
    out_arr = np.where(np.isfinite(out_arr), out_arr, 0.0)
    out_arr = np.maximum(out_arr, 0.0)
    out_arr = np.minimum(out_arr, 1000.0)
    s = float(out_arr.sum())
    if s < 0.01:
        x = np.linspace(0, 1, 6500)
        out_arr = np.sin(np.pi * x)**2
        s = float(out_arr.sum())
    out_arr = out_arr / s
    
    # Final score check
    final_score = objective_norm(out_arr)
    try:
        official = evaluate_safe(out_arr)
        if np.isfinite(official):
            final_score = min(final_score, official)
    except Exception:
        pass
    
    elapsed = time.monotonic() - t_global
    print("FINAL: score=%.10f n=%d time=%.1fs" % (final_score, out_arr.size, elapsed))
    
    return [float(v) for v in out_arr]
