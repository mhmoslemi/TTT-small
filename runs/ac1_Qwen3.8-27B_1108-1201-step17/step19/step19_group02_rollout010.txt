def fast_convolve(a):

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

    return fast_convolve(a).max()

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

# Convex line search using ternary search
def line_search(y, g, y_conv, yg_conv, gg_conv, init_val):
    # f(alpha) = max_k [(1-a)^2 * yy[k] + 2a(1-a) * yg[k] + a^2 * gg[k]]
    # This is convex in alpha in [0,1]
    lo, hi = 0.0, 1.0
    best_alpha = 0.0
    best_val = init_val
    
    for _ in range(40):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        
        # Compute max at m1
        val_m1 = ((1-m1)**2 * y_conv + 2*m1*(1-m1) * yg_conv + m1**2 * gg_conv).max()
        val_m2 = ((1-m2)**2 * y_conv + 2*m2*(1-m2) * yg_conv + m2**2 * gg_conv).max()
        
        if val_m1 < val_m2:
            hi = m2
            if val_m1 < best_val:
                best_val = val_m1
                best_alpha = m1
        else:
            lo = m1
            if val_m2 < best_val:
                best_val = val_m2
                best_alpha = m2
    
    # Also check alpha = 1
    val_1 = gg_conv.max()
    if val_1 < best_val:
        best_alpha = 1.0
        best_val = val_1
    
    return best_alpha, best_val

def lp_fw_step(y, active_set, y_conv, M_curr, solver, n):
    """Solve LP: min M s.t. sum(g)=1, g>=0, conv(y,g)[k]<=M for k in active_set"""
    K = len(active_set)
    if K == 0:
        return None, y_conv.max()
    
    # Build constraint matrix: for each active k, conv(y,g)[k] = sum_i y[i]*g[k-i] <= M
    # conv(y,g)[k] = sum_{i: 0<=i<n, 0<=k-i<n} y[i] * g[k-i]
    # Let j = k-i, so i = k-j, and 0 <= k-j < n means k-n < j < k
    # Also 0 <= j < n
    # So j ranges from max(0, k-n+1) to min(n-1, k)
    # And coefficient of g[j] is y[k-j]
    
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(active_set):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start < j_end:
            js = np.arange(j_start, j_end)
            coeffs[jj, js] = y[k_int - js]
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, M_curr
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, M_curr
    
    g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
    g_sum = g_val.sum()
    if g_sum < 1e-15:
        return None, M_curr
    g_val = g_val / g_sum
    
    return g_val, M_curr

def subgradient_mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent using exact subgradient of max(conv(a,a))"""

    gamma = 2.0
    step_count = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        x_conv = fast_convolve(x)
        M = x_conv.max()
        
        # Find all indices achieving (near) max
        tol = M * 0.01 + 1e-12
        max_indices = np.where(x_conv >= M - tol)[0]

        grad = np.zeros(n)

            i_end = min(n, k_int + 1)

            grad[idx] += 2.0 * x[k_int - idx]
        grad /= len(max_indices)
        
        # Mirror descent step (multiplicative weights / exponential)
        step_count += 1
        eta = gamma / np.sqrt(step_count)
        log_x = np.log(np.maximum(x, 1e-30)) - eta * grad
        log_x -= log_x.max()
        x_new = np.exp(log_x)
        x_new = np.maximum(x_new, 0)
        s_new = x_new.sum()

def coordinate_polish(a0, time_limit, verbose=True):
    """Peak-aware coordinate descent"""
    n = len(a0)
    a = norm_sum1(a0)

    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    no_improve = 0
    max_no_improve = 500
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        a_conv = fast_convolve(a)
        top_k = np.argsort(a_conv)[::-1][:30]
        improved = False
        
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            max_idx_local = np.argmax(contrib)
            src = int(idx[max_idx_local])
            src_val = a[src]
            if src_val < 1e-15:
                continue
            
            # Find best destination: minimize impact on all top peaks
            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]
            
            # Avoid indices in peak regions
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                in_peak[s2:e2] = True
            
            candidates = np.where(~in_peak)[0]

                a_trial[dst] = a[dst] + amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t

                    a = a_trial
                    best_m = M_trial

                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = compute_max_conv(a)

def parametric_init(n, time_limit, t0):
    """Generate diverse parametric initializations"""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []
    
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))
    
    # Product of powers
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        for beta in [0.5, 1.0, 1.5, 2.0, 3.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            candidates.append(norm_sum1(a))
    
    # Gaussian bumps
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sigma in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            candidates.append(norm_sum1(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))))
    
    # Cosine windows
    candidates.append(norm_sum1(np.sin(np.pi * x) ** 2))
    candidates.append(norm_sum1(np.sin(np.pi * x) ** 4))
    candidates.append(norm_sum1(0.5 + 0.5 * np.cos(2 * np.pi * x)))  # Hann
    candidates.append(norm_sum1(0.54 + 0.46 * np.cos(2 * np.pi * x)))  # Hann proper
    candidates.append(norm_sum1(0.42 + 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)))  # Blackman
    
    # Sum of bumps
    for c1 in [0.2, 0.3]:
        for c2 in [0.7, 0.8]:
            for sigma in [0.1, 0.15, 0.2]:
                a = np.exp(-((x - c1) ** 2) / (2 * sigma ** 2)) + np.exp(-((x - c2) ** 2) / (2 * sigma ** 2))
                candidates.append(norm_sum1(a))
    
    # Cosine harmonics
    for freq in [1, 2, 3, 4]:
        for ph in [0.0, np.pi/2]:
            a = np.maximum(1.0 + np.cos(2 * np.pi * freq * x + ph), 0.01)
            candidates.append(norm_sum1(a))
    
    # Random smooth
    rng = np.random.default_rng(2024)
    for trial in range(100):
        a = rng.random(n)
        # Smooth it
        kernel = np.exp(-np.arange(-20, 21) ** 2 / 20.0)
        a = np.convolve(a, kernel / kernel.sum(), mode='same')
        a = a ** rng.uniform(0.3, 5.0)
        candidates.append(norm_sum1(a))
    
    best_a = None
    best_val = float('inf')
    for a in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        v = compute_max_conv(a)
        if v < best_val:
            best_val = v
            best_a = a.copy()
    
    if best_a is None:
        best_a = norm_sum1(np.sin(np.pi * x) ** 2)
        best_val = compute_max_conv(best_a)
    
    return best_a, best_val

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def solve_lp(y, active_set, n, solver):
    K = len(active_set)
    if K == 0:
        return None

    coeffs = np.zeros((K, n))
    for jj, k in enumerate(active_set):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start < j_end:
            js = np.arange(j_start, j_end)
            coeffs[jj, js] = y[k_int - js]

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
    g_sum = g_val.sum()
    if g_sum < 1e-15:
        return None
    return g_val / g_sum


def ternary_line_search(y_conv, yg_conv, gg_conv, M_curr):
    lo, hi = 0.0, 1.0
    best_alpha = 0.0
    best_val = M_curr

    for _ in range(35):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0

        val_m1 = ((1.0 - m1) ** 2 * y_conv + 2.0 * m1 * (1.0 - m1) * yg_conv + m1 ** 2 * gg_conv).max()
        val_m2 = ((1.0 - m2) ** 2 * y_conv + 2.0 * m2 * (1.0 - m2) * yg_conv + m2 ** 2 * gg_conv).max()

        if val_m1 < val_m2:
            hi = m2
            if val_m1 < best_val:
                best_val = val_m1
                best_alpha = m1
        else:
            lo = m1
            if val_m2 < best_val:
                best_val = val_m2
                best_alpha = m2

    val_1 = gg_conv.max()
    if val_1 < best_val:
        best_alpha = 1.0
        best_val = val_1

    return best_alpha, best_val


def lp_fw(a0, time_limit, solver, initial_K=200, max_K=2000, verbose=True):
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    y_conv = fast_convolve(y)
    best_m = y_conv.max()
    t0 = time.monotonic()

    max_K = min(max_K, max(100, int(2500000.0 / n)))
    initial_K = min(initial_K, max_K)

    active_set = sorted(np.argsort(y_conv)[::-1][:initial_K].tolist())
    no_improve = 0
    max_no_improve = 600
    iter_count = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: stagnation at iter " + str(iter_count))
            break
        iter_count += 1
        M_curr = y_conv.max()

        # Update active set: near-max + top-K
        threshold = M_curr * 0.97
        near_max = np.where(y_conv >= threshold)[0]
        top_k_idx = np.argsort(y_conv)[::-1][:max_K]
        new_active = set(near_max.tolist()) | set(top_k_idx.tolist())

        if len(new_active) > max_K:
            active_list = sorted(new_active)
            sorted_keys = np.argsort([y_conv[k] for k in active_list])[::-1][:max_K]
            active_set = [active_list[i] for i in sorted_keys]
        else:
            active_set = sorted(new_active)

        K = len(active_set)
        if K == 0:
            break

        g_dir = solve_lp(y, active_set, n, solver)
        if g_dir is None:
            no_improve += 1
            continue

        # Compute convolutions for line search
        fft_size = 1 << (2 * n - 2).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        yg_conv = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]
        gg_conv = np.fft.irfft(Gf * Gf, fft_size)[:2 * n - 1]

        # Add violated constraints
        violations = np.where(yg_conv > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = yg_conv[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)
            if len(active_set) > max_K:
                active_list = sorted(active_set)
                sorted_keys = np.argsort([y_conv[k] for k in active_list])[::-1][:max_K]
                active_set = [active_list[i] for i in sorted_keys]

        # Convex line search
        alpha, _ = ternary_line_search(y_conv, yg_conv, gg_conv, M_curr)
        if iter_count <= 2:
            alpha = 1.0
        alpha = max(0.01, min(1.0, alpha))

        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_new
        y = y_new
        y_conv = fast_convolve(y)
        M_new = y_conv.max()

        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = y.copy()
            no_improve = 0
            stagnation = 0
        else:
            no_improve += 1
            stagnation += 1

        # Periodic perturbation
        if stagnation > 150 and stagnation % 75 == 0:
            rng = np.random.default_rng(iter_count * 7 + 3)
            noise = rng.normal(0, 0.03 / (1.0 + stagnation * 0.01), n) * best_a
            y_pert = best_a + noise
            y_pert = np.maximum(y_pert, 0)
            sp = y_pert.sum()
            if sp > 1e-15:
                y_pert = y_pert / sp
                if compute_max_conv(y_pert) < best_m * 0.999:
                    y = y_pert
                    y_conv = fast_convolve(y)
                    best_m = y_conv.max()
                    best_a = y.copy()
                    no_improve = 0
                    stagnation = 0
                    active_set = sorted(np.argsort(y_conv)[::-1][:initial_K].tolist())

        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", |S|=" + str(K))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", iters=" + str(iter_count))
    return best_a, best_m


def subgradient_md(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    a_conv = fast_convolve(a)
    best_m = a_conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    gamma = 1.5
    step_count = 0
    no_improve = 0
    max_no_improve = 4000

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        x_conv = fast_convolve(x)
        M = x_conv.max()

        # Near-max subgradient
        tol = max(M * 0.02, 1e-12)
        max_indices = np.where(x_conv >= M - tol)[0]
        if len(max_indices) == 0:
            max_indices = np.array([np.argmax(x_conv)])

        grad = np.zeros(n)
        for k in max_indices[:100]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * x[k_int - idx]
        grad /= len(max_indices)

        step_count += 1
        eta = gamma / (np.sqrt(step_count) + 1.0)
        log_x = np.log(np.maximum(x, 1e-30)) - eta * grad
        log_x -= log_x.max()
        x_new = np.exp(log_x)
        x_new = np.maximum(x_new, 0)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / s_new

        x_new_conv = fast_convolve(x_new)
        M_new = x_new_conv.max()

        if M_new < best_m - 1e-13:
            best_m = M_new
            best_a = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        x = x_new

        if no_improve > 500 and no_improve % 500 == 0:
            gamma = max(gamma * 0.85, 0.1)
            noise = rng.normal(0, 0.01, n) * best_a
            x_p = best_a + noise
            x_p = np.maximum(x_p, 0)
            sp = x_p.sum()
            if sp > 1e-15:
                x_p = x_p / sp
                M_p = compute_max_conv(x_p)
                if M_p < best_m:
                    x = x_p
                    best_m = M_p
                    best_a = x_p.copy()
                    no_improve = 0

        if verbose and step_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SGD step=" + str(step_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

    J = 2.0 * n * best_m
    if verbose:
        print("    SGD done: J=" + repr(J))
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = compute_max_conv(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    no_improve = 0
    max_no_improve = 600

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        a_conv = fast_convolve(a)
        top_k = np.argsort(a_conv)[::-1][:25]
        improved = False

        for k in top_k[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-14:
                continue

            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i)
                in_peak[s2:e2] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < n // 4:
                candidates = np.arange(n)
            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])
            if dst == src:
                continue

            for frac in [0.08, 0.15, 0.25, 0.4, 0.6, 0.8]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] = a[src] - amt
                a_trial[dst] = a[dst] + amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = compute_max_conv(a_trial)
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break

        if not improved:
            no_improve += 1
            for _ in range(15):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.01, 0.08)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = compute_max_conv(a)
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            no_improve = 0
                            improved = True
                            break

        if verbose and no_improve % 150 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", no_imp=" + str(no_improve))

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Total time budget: " + repr(time_budget) + "s")

    # Get initial sequence from parent
    best_seq = None
    best_score = float('inf')
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            best_seq = p_norm
            best_score = evaluate_safe(best_seq)
            print("Parent: n=" + str(len(p)) + ", score=" + repr(best_score))
    except Exception:
        pass

    if best_seq is None:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: Quick parametric at current n
    print("\n=== Phase 1: Parametric (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.02, 20.0)
    if t1 > 5:
        a_p, _ = parametric_init(n0, t1, t_global)
        best_seq, best_score = try_update_state(best_seq, best_score, a_p, "parametric", t_global)

    # Phase 2: LP-FW main at current n
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: LP-FW main (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 60:
        a_out, _ = lp_fw(norm_sum1(best_seq), t2, solver, initial_K=300, max_K=2000, verbose=True)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, "lpfw_main", t_global)

    # Phase 3: Subgradient MD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Subgrad MD (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 30:
        a_out, _ = subgradient_md(norm_sum1(best_seq), t3, verbose=True)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, "sgd", t_global)

    # Phase 4: LP-FW refine with larger active set
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW refine (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.18, 240.0)
    if t4 > 30:
        a_out, _ = lp_fw(norm_sum1(best_seq), t4, solver, initial_K=400, max_K=2500, verbose=True)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, "lpfw_refine", t_global)

    # Phase 5: Try different n values
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Multi-n exploration ===")
    try_ns = []
    for delta in [500, -300, 800, -600]:
        nn = n_best + delta
        if 3000 <= nn <= 12000 and nn != n_best:
            try_ns.append(nn)

    for new_n in try_ns[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 180:
            break
        print("  Trying n=" + str(new_n))
        a_long = interpolate_seq(best_seq, new_n)

        t5a = min(remaining * 0.35, 200.0)
        if t5a > 50:
            a_out, _ = lp_fw(a_long, t5a, solver, initial_K=250, max_K=1800, verbose=False)
            best_seq, best_score = try_update_state(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t5b = min(remaining * 0.25, 80.0)
            if t5b > 20:
                a_out, _ = subgradient_md(norm_sum1(best_seq), t5b, verbose=False)
                best_seq, best_score = try_update_state(best_seq, best_score, a_out, "sgd_n" + str(new_n), t_global)

    # Phase 6: Final LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Final LP-FW (n=" + str(n_best) + ") ===")
    if remaining > 60:
        t6 = min(remaining * 0.40, 220.0)
        if t6 > 30:
            a_out, _ = lp_fw(norm_sum1(best_seq), t6, solver, initial_K=500, max_K=2500, verbose=True)
            best_seq, best_score = try_update_state(best_seq, best_score, a_out, "lpfw_final", t_global)

    # Phase 7: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Coord Polish (n=" + str(n_best) + ") ===")
    if remaining > 25:
        t7 = min(remaining * 0.55, 120.0)
        if t7 > 15:
            a_out, _ = coordinate_polish(norm_sum1(best_seq), t7, verbose=True)
            best_seq, best_score = try_update_state(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
