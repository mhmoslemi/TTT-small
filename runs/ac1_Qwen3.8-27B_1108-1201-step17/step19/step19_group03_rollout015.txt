import numpy as np
import time

def fast_conv(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 1 << (2*n - 2).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]

def compute_subgrad(a, conv_val, smooth_k=20, beta=3.0):
    """Compute smoothed subgradient of 2n*max(conv(a,a)).
    
    Uses top-k smoothing: weight each lag proportional to conv_val[k]^beta
    for the top k lags.
    """
    n = len(a)
    conv_size = 2*n - 1
    
    # Find top-k lags
    k_max = min(smooth_k, conv_size)
    top_idx = np.argsort(conv_val)[::-1][:k_max]
    top_vals = np.maximum(conv_val[top_idx], 1e-30)
    
    # Weights (softmax-like)
    w = top_vals ** beta
    w = w / w.sum()
    
    # Build weight vector (sparse: only k_max non-zero entries)
    w_vec = np.zeros(conv_size)
    w_vec[top_idx] = w
    
    # Compute cross-correlation: g[i] = sum_k w_vec[k] * a[k-i] (where valid)
    # This is ifft(fft(w_vec) * conj(fft(a)))
    size = 1 << (conv_size - 1).bit_length()
    W = np.fft.rfft(w_vec, size)
    A = np.fft.rfft(a, size)
    g = np.fft.irfft(W * np.conj(A), size)[:n]
    
    # The subgradient of 2n*max(conv) is 2n * g (approximately)
    return 2.0 * g  # factor of n is in the objective, but we optimize on simplex so it's a constant scaling

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def nesterov_subgrad(a0, time_limit, smooth_k=30, beta=4.0, lr=0.01, 
                       momentum=0.9, restart_every=2000, n_restarts=10,
                       rng_seed=0, verbose=False):
    """Nesterov-accelerated subgradient descent on the simplex."""
    n = len(a0)
    a = project_simplex(a0.copy())
    v = np.zeros(n)  # velocity for Nesterov
    best_a = a.copy()
    best_conv = fast_conv(a).max()
    best_score = 2.0 * n * best_conv
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        
        # Nesterov lookahead
        y = a + momentum * v
        y = project_simplex(y)
        
        # Compute convolution and subgradient
        conv_y = fast_conv(y)
        g = compute_subgrad(y, conv_y, smooth_k=smooth_k, beta=beta)
        
        # Project gradient onto tangent space (sum = 0)
        g_tangent = g - g.mean()
        
        # Adaptive learning rate
        lr_t = lr / np.sqrt(1.0 + t * 0.01)
        
        # Update
        v = momentum * v - lr_t * g_tangent
        a_new = a + v
        a_new = project_simplex(a_new)
        
        # Check improvement
        conv_new = fast_conv(a_new)
        M_new = conv_new.max()
        if M_new < best_conv:
            best_conv = M_new
            best_score = 2.0 * n * M_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Periodic restart
        if t % restart_every == 0:
            # Perturb from best
            noise = rng.normal(0, 0.1/np.sqrt(n), n)
            a_pert = best_a + noise
            a = project_simplex(a_pert)
            v = np.zeros(n)
            smooth_k = min(smooth_k + 5, 2*n-1)
        
        if verbose and t % 100 == 0:
            print(f"    Nesterov t={t} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score
