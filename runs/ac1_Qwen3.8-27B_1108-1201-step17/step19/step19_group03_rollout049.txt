def fast_conv(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgrad(c, a, top_k=50, beta=5.0):
    """Compute weighted subgradient of max(c)."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = np.maximum(c[top_idx], 1e-30)
    w = top_vals ** beta
    w = w / w.sum()
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        g[idx] += 2.0 * wi * a[j_idx]
    return g

def build_constraint_matrix(a, lags):
    """Build the constraint matrix T where T[i,j] = a[lags[i]-j] if valid."""
    n = len(a)
    K = len(lags)
    T = np.zeros((K, n), dtype=np.float64)
    for ii in range(K):
        k = int(lags[ii])
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_start < j_end:
            T[ii, j_start:j_end] = a[k - np.arange(j_start, j_end)]
    return T

        top_k = min(100, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        top_w = w[top_idx]
        top_w = top_w / top_w.sum()
        for wi, k in zip(top_w, top_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start < j_end:
                idx = np.arange(j_start, j_end)
                g[idx] += 2.0 * wi * a[ki - idx]

        # Nesterov momentum
        v = beta_mom * v - eta * g
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_conv(a_new).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 1.0)
        else:
            stall += 1
            if stall > 50:
                eta *= 0.7
            if stall > 150:
                pert = rng.normal(0, 0.005, n) * best_a
                a_p = np.maximum(best_a + pert, 0.0)
                sp = a_p.sum()
                if sp > 1e-15:
                    a_p /= sp
                    M_p = fast_conv(a_p).max()
                    if M_p < best_M:
                        a = a_p
                        best_M = M_p
                        best_a = a.copy()
                        stall = 0
                        eta = 0.3
                        v = np.zeros(n)
                    else:
                        a = a_new
                        v = np.zeros(n)

        if verbose and it % 200 == 0:
            J = 2.0 * n * best_M
            print(f"    SmoothGrad it={it} ({elapsed:.0f}s): J={J:.6f}, tau={tau_cur:.1f}, eta={eta:.4f}")

    return best_a


def coord_descent(a0, time_limit, verbose=False, rng_seed=0):
    """Coordinate descent polish: move mass from high-contributing positions to low-sensitivity positions."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 200
    n_top = 30

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        top_k = np.argsort(c)[::-1][:n_top]

        # Compute sensitivity: for each position, how much does it contribute to peak lags
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:15]:
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            idx = np.arange(j_start, j_end)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved = False
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            idx = np.arange(j_start, j_end)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            if a[src] < 1e-12:
                continue

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            top_cands = candidates[np.argsort(sens[candidates])[:5]]

            found = False
            for dst in top_cands:
                if dst == src or found:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.1, 0.15]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] = a[src] - amt
                    a_trial[dst] = a[dst] + amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial /= st
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_M - 1e-16:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a /= s
                    M_new = fast_conv(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 50 == 0:
            J = 2.0 * n * best_M
            print(f"    Coord ({time.monotonic()-t0:.0f}s): J={J:.6f}, stall={stall}")

    return best_a


def pocs_init(n, n_trials=5, time_per_trial=20.0, seed=0):
    """POCS-style initialization: alternate flat-spectrum and non-negativity."""
    rng = np.random.default_rng(seed)
    best_a = None
    best_M = float('inf')
    fft_size = 1 << (2 * n - 2).bit_length()

    for trial in range(n_trials):
        t_trial = time.monotonic() + time_per_trial
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        A_spec = np.ones(fft_size // 2 + 1, dtype=np.complex128) * np.exp(1j * phases)
        A_full = np.zeros(fft_size, dtype=np.complex128)
        A_full[:fft_size // 2 + 1] = A_spec
        for i in range(1, fft_size // 2):
            A_full[fft_size - i] = np.conj(A_spec[i])
        a = np.fft.irfft(A_full, fft_size)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            continue
        a /= s

        M = fast_conv(a).max()
        for it in range(5000):
            if time.monotonic() > t_trial:
                break
            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            alpha = 0.05 + 0.3 * (it / 5000.0)
            new_mag = (1 - alpha) * mag + alpha * mag_mean
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn < 1e-15:
                continue
            a_new /= sn
            M_new = fast_conv(a_new).max()
            if M_new < M:
                M = M_new
                a = a_new
            else:
                a = a_new

        if M < best_M:
            best_M = M
            best_a = a.copy()

    return best_a


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm1(a)


def make_asymmetric(n, alpha=0.5):
    """Asymmetric shape designed to minimize convolution peak."""
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-alpha * x) + 0.5 * np.exp(-alpha * (1 - x)) + 0.1 * np.cos(np.pi * x)
    a = np.maximum(a, 0.0)
    return norm1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Solver: {solver}, Budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_obj(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p.copy()
            p_score = eval_obj(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 6000
        best_seq = make_u_shape(n_base, 2.0)
        best_score = eval_obj(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 0: POCS for diverse init (if parent is far from optimal) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t0_phase = min(remaining * 0.05, 60.0)
    if t0_phase > 30:
        print(f"\n=== Phase 0: POCS init ({t0_phase:.0f}s) ===")
        a_pocs = pocs_init(n0, n_trials=4, time_per_trial=t0_phase / 5.0, seed=42)
        try_update(a_pocs, "pocs")

    # === Phase 1: LP-FW main (largest time allocation) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 400.0)
    print(f"\n=== Phase 1: LP-FW main ({t1:.0f}s, n={n0}) ===")
    if t1 > 30:
        a_out = lp_frank_wolfe(best_seq, t1, solver, K_init=150, K_growth=10,
                                K_max=min(3000, 2 * n0 - 1), verbose=True, rng_seed=0)
        try_update(a_out, "lp_main1")

    # === Phase 2: Smooth gradient for exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 2: Smooth gradient ({t2:.0f}s) ===")
    if t2 > 30:
        a_out = smooth_gradient(best_seq, t2, verbose=True, rng_seed=100)
        try_update(a_out, "smooth")

    # === Phase 3: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.3, 350.0)
    print(f"\n=== Phase 3: LP-FW r2 ({t3:.0f}s) ===")
    if t3 > 30:
        a_out = lp_frank_wolfe(best_seq, t3, solver, K_init=300, K_growth=20,
                                K_max=min(5000, 2 * len(best_seq) - 1), verbose=True, rng_seed=42)
        try_update(a_out, "lp_main2")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.1, 120.0)
    print(f"\n=== Phase 4: Coord descent ({t4:.0f}s) ===")
    if t4 > 20:
        a_out = coord_descent(best_seq, t4, verbose=True, rng_seed=777)
        try_update(a_out, "coord1")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Length exploration ({remaining:.0f}s remaining) ===")
    n_best = len(best_seq)
    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 2000, -2000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)
        t_trial = min(remaining * 0.3, 200.0)
        if t_trial > 30:
            a_out = lp_frank_wolfe(a_interp, t_trial, solver, K_init=150,
                                    K_growth=10, K_max=min(2000, 2 * new_n - 1),
                                    verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"lp_n{new_n}")

    # === Phase 6: Final LP-FW polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.3, 200.0)
    print(f"\n=== Phase 6: Final LP-FW ({t6:.0f}s, n={n_best}) ===")
    if t6 > 20:
        a_out = lp_frank_wolfe(best_seq, t6, solver, K_init=500, K_growth=30,
                                K_max=min(6000, 2 * n_best - 1), verbose=True, rng_seed=999)
        try_update(a_out, "lp_final")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.4, 100.0)
    print(f"\n=== Phase 7: Final coord ({t7:.0f}s) ===")
    if t7 > 15:
        a_out = coord_descent(best_seq, t7, verbose=False, rng_seed=1234)
        try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        final_score = eval_obj(best_seq)
        if final_score < best_score:
            best_score = final_score
        # Cross-check with evaluate_sequence
        check_score = eval_safe(best_seq)
        if np.isfinite(check_score) and check_score < best_score:
            best_score = check_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 6000
        best_seq = make_u_shape(n_fb, 2.0)
        best_score = eval_obj(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
