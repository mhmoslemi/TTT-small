    """Compute gradient of log-sum-exp(s * conv(a,a)) with respect to a."""
    n = len(a)
    ac = fast_convolve(a)  # length 2n-1
    
    # Softmax weights
    ac_shifted = ac - ac.max()  # for numerical stability
    exp_ac = np.exp(s * ac_shifted)
    p = exp_ac / exp_ac.sum()  # length 2n-1
    
    # Gradient: grad_j = 2 * sum_k p_k * a_{k-j} for valid k-j
    # = 2 * (p cross-correlate a)_j
    # FFT: IFFT(FFT(p, N) * conj(FFT(a, N)))[:n]
    # N >= len(p) + len(a) - 1 = 3n - 2

    A = np.fft.rfft(a, N)
    grad = 2.0 * np.fft.irfft(P * np.conj(A), N)[:n]
    
    return grad
