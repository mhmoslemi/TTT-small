def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def make_ramp(n, alpha):
    """Create normalized (i+1)^alpha sequence."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** alpha
    return a / a.sum()


def make_tapered_ramp(n, alpha, beta):
    """Create a[i] = (i+1)^alpha * (1 - (i/n)^beta) normalized."""
    x = np.arange(1, n + 1, dtype=np.float64)
    t = np.arange(n, dtype=np.float64) / n
    a = x ** alpha * (1.0 - 0.3 * t ** beta)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return make_ramp(n, alpha)
    return a / s


def make_two_factor(n, alpha, nramp):

def compute_gradient_at_peaks(a, conv, n_peaks=30):
    """Subgradient using top peaks with softmax weighting."""
    n = len(a)
    top_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
    # Softmax weights over top peaks
    vals = conv[top_idx]
    max_v = vals.max()
    exp_v = np.exp((vals - max_v) / max(1e-10, max_v * 0.01))
    w = exp_v / exp_v.sum()

    grad = np.zeros(n)
    for idx_i, k in enumerate(top_idx):
        ki = int(k)
        wi = w[idx_i]
        if wi < 1e-12:
            continue
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        j_range = np.arange(j_lo, j_hi + 1)
        pair = ki - j_range
        grad[j_range] += wi * 2.0 * a[pair]
    return grad

def spectral_flatten(a0, time_limit, verbose=False):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(20000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_mean = P.mean()
        if P_max < 1e-30:
            break

        # Flatten: push power from peaks to mean
        alpha = 0.3 + 0.7 * min(1.0, it / 500.0)
        P_new = P + alpha * (P_mean - P)
        P_new = np.maximum(P_new, 1e-30)

        mag_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        mag_ratio = np.clip(mag_ratio, 0.5, 2.0)
        A_new = A * mag_ratio
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            blend = 0.2
            a = (1 - blend) * a + blend * a_new
            a = np.maximum(a, 0.0)
            a = a / max(a.sum(), 1e-15)
            M_curr = fast_autocorr(a).max()
            if M_curr < best_m:
                best_m = M_curr
                best_a = a.copy()

        if verbose and it % 500 == 0:
            print(f"      Spec it={it} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=False, rng_seed=42):
    """Direct coordinate moves to reduce peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 100:
        conv = fast_autocorr(a)
        k_max = int(np.argmax(conv))

        # Find pair with max contribution at k_max
        j_lo = max(0, k_max - n + 1)
        j_hi = min(n - 1, k_max)
        if j_lo > j_hi:
            stall += 1
            continue
        idx = np.arange(j_lo, j_hi + 1)
        contrib = a[idx] * a[k_max - idx]
        src = idx[np.argmax(contrib)]

        if a[src] < 1e-12:
            stall += 1
            continue

        # Find low-cost destination
        # Cost: how much a[j] contributes to top peaks
        top_ks = np.argpartition(conv, -10)[-10:]
        costs = np.zeros(n)
        for k2 in top_ks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            if lo2 <= hi2:
                valid = np.arange(lo2, hi2 + 1)
                costs[valid] += a[k2i - valid]

        # Pick destination with low cost and low value
        sorted_idx = np.argsort(costs * 0.1 + a)
        dst = sorted_idx[0]
        if dst == src:
            dst = sorted_idx[1] if len(sorted_idx) > 1 else int(rng.integers(n))

        moved = False
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_t = fast_autocorr(a_trial).max()
            if M_t < best_m - 1e-16:
                a = a_trial
                best_m = M_t
                best_a = a.copy()
                moved = True
                stall = 0
                break

        if not moved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.01, 0.05)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss
