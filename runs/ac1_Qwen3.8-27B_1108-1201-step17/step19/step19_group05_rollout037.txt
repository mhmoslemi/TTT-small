import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def compute_conv(a):
    """Compute full linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * n - 1]


def smoothed_max_val(x, tau):
    """Compute smooth max: (1/tau)*log(sum(exp(tau*x)))."""
    x_max = x.max()
    if x_max * tau < -700:
        return x_max
    return x_max / tau + np.log(np.exp(tau * (x - x_max)).sum()) / tau


def compute_gradient_log_simplex(a, conv, tau):
    """
    Compute gradient of 2n*smoothed_max(conv) w.r.t. log(a) parameters.
    Gradient = 4n * (g - a*B) where g_k = sum_l w_l * a[k-l], B = sum_l w_l * conv[l]
    """
    n = len(a)
    x = tau * conv
    x_max = x.max()
    w = np.exp(x - x_max)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    B = np.dot(w, conv)

    # g_j = sum_k w_k * a[k-j] (cross-correlation of w and a)
    N = 1 << (2 * n - 1).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    g = np.fft.irfft(np.conj(W) * A, N)[:n]

    grad = 4.0 * n * (g - a * B)
    return grad

    """Objective and gradient for L-BFGS (parameterization: a = softmax(log_a))."""
    log_a_shifted = log_a - log_a.max()
    a = np.exp(log_a_shifted)

        return 1e10, np.zeros_like(log_a)

    val = 2.0 * n * smoothed_max_val(conv, tau)
    grad = compute_gradient_log_simplex(a, conv, tau)

def spectral_flatten(a, n_iter=20, alpha=0.5):
    """Gerchberg-Saxton spectral flattening."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    for _ in range(n_iter):
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_mean = P.mean()
        if P_max < 1e-30:
            break
        # Push power from peaks toward mean
        P_target = (1 - alpha) * P + alpha * P_mean
        P_target = np.maximum(P_target, 1e-30)
        ratio = np.sqrt(P_target / np.maximum(P, 1e-30))
        ratio = np.clip(ratio, 0.3, 3.0)
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        a = a_new
    return a

def interpolate_seq(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss

            if j_lo > j_hi:
                continue
            j_range = np.arange(j_lo, j_hi + 1)
            pair = ki - j_range
            grad[j_range] += wi * 2.0 * a[pair]

        eta = 5.0 / (np.sqrt(t) + 1.0)
        log_a = np.log(a) - eta * grad
        a_new = np.exp(log_a - log_a.max())
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new /= s
        a = a_new

        M_new = conv.max()
        M_x = compute_conv(a).max()
        M_eff = min(M_new, M_x)
        if M_eff < best_m - 1e-16:
            best_m = M_eff
            best_a = (a if M_x <= M_new else a.copy()).copy()

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search function."""
    t_global = time.monotonic()
    time_budget = min(budget_s - 20.0, 1780.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = safe_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = safe_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    # Generate initial candidates
    candidates = []
    if parent is not None:
        candidates.append((parent, "parent"))

    # Ramps of various types and lengths
    for n_try in [4000, 6000, 8000]:
        candidates.append((make_ramp(n_try, 1.0), f"ramp_n{n_try}"))
    candidates.append((make_ramp(6000, 0.8), "ramp_0.8"))
    candidates.append((make_ramp(6000, 1.2), "ramp_1.2"))
    candidates.append((make_symmetric_ramp(6000), "sym_ramp"))

    # Evaluate and sort
    print(f"\nEvaluating {len(candidates)} initial candidates...")
    scored = []
    for a, label in candidates:
        sc = safe_eval(a)
        if np.isfinite(sc):
            scored.append((sc, a, label))
            print(f"  [{label}]: n={len(a)}, score={sc:.6f}")
    scored.sort(key=lambda x: x[0])

    # Phase 1: L-BFGS on top candidates with adaptive temperature
    print(f"\n{'='*60}")
    print("Phase 1: L-BFGS Optimization (adaptive temperature)")
    print(f"{'='*60}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_top = min(4, len(scored))
    t_per = max(60.0, min(300.0, remaining * 0.45 / n_top))

    for i in range(n_top):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget - 100:
            break
        t_avail = min(t_per, time_budget - elapsed - 100)
        if t_avail < 30:
            break
        sc0, a0, label = scored[i]
        print(f"\n  L-BFGS [{label}] init={sc0:.6f}, time={t_avail:.0f}s")
        a_out, _ = l_bfgs_optimize(a0, tau_start=50.0, tau_end=5.0,
                                    time_limit=t_avail, rng_seed=42 + i * 7)
        try_update(a_out, f"lbfgs_{label}")

    # Phase 2: Spectral flattening + MD on best
    print(f"\n{'='*60}")
    print("Phase 2: Spectral + Mirror Descent Polish")
    print(f"{'='*60}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100 and best_seq is not None:
        a_in = best_seq / best_seq.sum()

        # Spectral flattening
        t_spec = min(remaining * 0.15, 120.0)
        print(f"\n  Spectral flattening (t={t_spec:.0f}s)")
        # Run spectral with time limit
        a_spec = a_in.copy()
        t0_spec = time.monotonic()
        n_iter_spec = 0
        while time.monotonic() - t0_spec < t_spec:
            a_spec = spectral_flatten(a_spec, n_iter=5, alpha=0.3)
            n_iter_spec += 1
        try_update(a_spec, "spectral")

        # Mirror descent on best so far
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t_md = min(remaining * 0.20, 180.0)
            print(f"\n  Mirror descent (t={t_md:.0f}s)")
            a_md, _ = mirror_descent_quick(best_seq / best_seq.sum(), t_md,
                                           n_peaks=50, rng_seed=123)
            try_update(a_md, "mirror_descent")

    # Phase 3: More L-BFGS rounds with different params
    print(f"\n{'='*60}")
    print("Phase 3: Aggressive L-BFGS")
    print(f"{'='*60}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        # Round with lower starting temperature
        t3 = min(remaining * 0.25, 250.0)
        print(f"\n  L-BFGS aggressive (t={t3:.0f}s)")
        a_out, _ = l_bfgs_optimize(a_in, tau_start=20.0, tau_end=2.0,
                                    time_limit=t3, rng_seed=999)
        try_update(a_out, "lbfgs_agg")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            # Try different initial perturbation
            rng = np.random.default_rng(777)
            a_pert = a_in + rng.normal(0, 0.02, len(a_in)) * a_in
            a_pert = np.maximum(a_pert, 1e-30)
            a_pert /= a_pert.sum()
            t3b = min(remaining * 0.20, 150.0)
            print(f"\n  L-BFGS perturbed (t={t3b:.0f}s)")
            a_out, _ = l_bfgs_optimize(a_pert, tau_start=30.0, tau_end=3.0,
                                        time_limit=t3b, rng_seed=555)
            try_update(a_out, "lbfgs_pert")

    # Phase 4: Direct peak reduction
    print(f"\n{'='*60}")
    print("Phase 4: Direct Peak Reduction")
    print(f"{'='*60}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        t4 = min(remaining * 0.15, 100.0)
        print(f"\n  Peak reduction (t={t4:.0f}s)")
        t0_pr = time.monotonic()
        a_pr = a_in.copy()
        # Run peak reduction in chunks to check time
        while time.monotonic() - t0_pr < t4:
            a_pr = direct_peak_reduction(a_pr, n_steps=50,
                                         rng=np.random.default_rng(int(time.time() % 10000)))
        try_update(a_pr, "peak_reduce")

    # Phase 5: Multi-length exploration
    print(f"\n{'='*60}")
    print("Phase 5: Multi-length Optimization")
    print(f"{'='*60}")

    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is not None and remaining > 80:
        n_best = len(best_seq)
        for new_n in [int(n_best * 1.3), int(n_best * 0.7), n_best + 2000]:
            if new_n < 1000 or new_n > 15000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            a_interp = interpolate_seq(best_seq, new_n)
            try_update(a_interp, f"interp_n{new_n}")

            t5 = min(remaining * 0.25, 120.0)
            if t5 > 20:
                print(f"\n  L-BFGS n={new_n} (t={t5:.0f}s)")
                a_out, _ = l_bfgs_optimize(a_interp, tau_start=40.0, tau_end=5.0,
                                            time_limit=t5, rng_seed=2000 + new_n % 100)
                try_update(a_out, f"lbfgs_n{new_n}")

    # Phase 6: Final combined polish
    print(f"\n{'='*60}")
    print("Phase 6: Final Polish")
    print(f"{'='*60}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40 and best_seq is not None:
        a_in = best_seq / best_seq.sum()

        # Quick spectral
        t6a = min(remaining * 0.20, 60.0)
        if t6a > 10:
            t0_6a = time.monotonic()
            a_6 = a_in.copy()
            while time.monotonic() - t0_6a < t6a:
                a_6 = spectral_flatten(a_6, n_iter=3, alpha=0.4)
            try_update(a_6, "spec_final")

        # Quick L-BFGS with very low tau (sharp max)
        remaining = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining * 0.30, 100.0)
        if t6b > 15:
            print(f"\n  Final L-BFGS sharp (t={t6b:.0f}s)")
            a_out, _ = l_bfgs_optimize(best_seq / best_seq.sum(),
                                        tau_start=10.0, tau_end=1.0,
                                        time_limit=t6b, rng_seed=4242)
            try_update(a_out, "lbfgs_sharp")

        # Peak reduction
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6c = min(remaining * 0.20, 50.0)
            t0_6c = time.monotonic()
            a_6c = best_seq / best_seq.sum()
            while time.monotonic() - t0_6c < t6c:
                a_6c = direct_peak_reduction(a_6c, n_steps=30,
                                              rng=np.random.default_rng(int(time.time() % 9999)))
            try_update(a_6c, "peak_final")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        fs = safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("ERROR: No valid sequence found! Using fallback ramp.")
        best_seq = make_ramp(5000, 1.0)
        best_score = safe_eval(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
