def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    return np.fft.irfft(A * A, fft_size)[:size]

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def build_lp_rhs(a, active_clist, n):
    """Build constraint matrix efficiently: row k is a shifted/reversed a."""
    K = len(active_clist)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(active_clist):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[ki - idx]
    return coeffs

def bisection_line_search_v2(a, b, n, size, fft_size, num_iters=30):
    """Proper bisection on the convex function f(t) = max_k quad_k(t)/(sum(t))^2."""
    A_fft = np.fft.rfft(a, fft_size)
    B_fft = np.fft.rfft(b, fft_size)
    sa = a.sum()
    sb = b.sum()

    lo = 0.0
    hi = 1.0

    for _ in range(num_iters):
        m1 = lo + 2.0*(hi - lo)/3.0
        m2 = hi - 2.0*(hi - lo)/3.0

        # Evaluate at m1
        t_fft = (1.0 - m1) * A_fft + m1 * B_fft
        c1 = np.fft.irfft(t_fft * t_fft, fft_size)[:size]
        s1 = (1.0 - m1)*sa + m1*sb
        v1 = c1.max() / (s1*s1) if s1 > 1e-15 else 1e30

        # Evaluate at m2
        t_fft = (1.0 - m2) * A_fft + m2 * B_fft
        c2 = np.fft.irfft(t_fft * t_fft, fft_size)[:size]
        s2 = (1.0 - m2)*sa + m2*sb
        v2 = c2.max() / (s2*s2) if s2 > 1e-15 else 1e30

        if v1 < v2:
            hi = m2
        else:
            lo = m1

    best_t = (lo + hi) / 2.0
    # Also check a few other candidates
    extra = [0.0, 1.0, best_t, 0.5, 1.0/3.0, 2.0/3.0]
    best_val = 1e30
    for t in extra:
        t = max(0.0, min(1.0, t))
        t_fft = (1.0 - t) * A_fft + t * B_fft
        c = np.fft.irfft(t_fft * t_fft, fft_size)[:size]
        s = (1.0 - t)*sa + t*sb
        v = c.max() / (s*s) if s > 1e-15 else 1e30
        if v < best_val:
            best_val = v
            best_t = t

    return best_t, best_val


def lp_fw_optimize(a0, time_limit, solver, init_K=400, max_K=5000,

    t0 = time.monotonic()

    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:min(init_K, size)])
    no_improve = 0
    stall_limit = 2000
    prev_lp_sol = None

    M_var = None

            break

            break

        if M_curr < best_M - 1e-16:

        else:
            no_improve += 1

        clist = sorted(active_set)
        if len(clist) > max_K:

            clist = sorted(active_set)

        # Prune: remove constraints with very large slack
        if len(clist) > init_K:
            slacks = np.array([ac[k] - M_curr for k in clist])  # negative = active
            threshold = 0.01 * M_curr  # keep if within 1% of peak

                             for j, k in enumerate(clist)])
            # Actually keep all that are within 5% of max
            rel_slack = (M_curr - np.array([ac[k] for k in clist])) / M_curr
            keep_mask = rel_slack < 0.05
            if keep_mask.sum() > 20:
                active_set = set(int(clist[j]) for j in range(len(clist)) if keep_mask[j])
                # Always keep the top few
                top_idx = np.argsort(ac)[::-1][:50]
                for k in top_idx:
                    active_set.add(int(k))
                clist = sorted(active_set)

        K = len(clist)
        if K == 0:
            break

        coeffs = build_lp_rhs(y, clist, n)

        # LP: min M' s.t. sum(g)=1, g>=0, coeffs@M' <= M_curr
        if g_var is None or g_var.shape != (n,):
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()

        if prev_lp_sol is not None:
            g_var.value = prev_lp_sol
            M_var.value = M_curr

        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)

                lp_ok = True
        except Exception:

            if no_improve > stall_limit // 2:
                break
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)

        g_dir = g_val / g_sum

def spectral_polish(a0, time_limit, verbose=True):
    """FFT-based alternating projection to flatten the convolution."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    stall = 0
    for it in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 200:
            break

        # Compute current convolution
        A = np.fft.rfft(a, fft_size)
        P = A.real**2 + A.imag**2
        c = np.fft.irfft(P, fft_size)[:size]
        M = c.max()

        # Target: reduce peak to M * (1 - reduction)
        # Clamp the power spectrum to reduce peak
        reduction = 1.0 - np.exp(-0.02 * (it + 1))  # gradual
        target_M = M * (1.0 - reduction * 0.3)

        # Modified approach: reduce the power at frequencies that contribute to peak
        # Simpler: iteratively clip the time-domain convolution and re-FFT
        c_clipped = np.minimum(c, target_M + 0.5 * (M - target_M))
        # Ensure c_clipped[0] and c_clipped[-1] are handled (they're a[0]^2, a[n-1]^2)

        # Update power spectrum: P_new = fft(c_clipped) (but this changes the phase)
        # Instead, scale P to reduce peak
        # Use a "water-filling" approach on P
        peak_idx = c.argmax()

        # Simple: reduce a at positions that contribute most to the peak
        ki = int(peak_idx)
        js = max(0, ki - n + 1)
        je = min(n - 1, ki)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[ki - idx]
        total_contrib = contrib.sum()
        if total_contrib < 1e-20:
            break

        # Reduce mass from top contributors
        top_frac = 0.005
        sort_idx = np.argsort(contrib)[::-1][:5]
        for si in sort_idx:
            src = int(idx[si])
            amt = a[src] * top_frac
            if amt < 1e-15:
                continue
            # Move to position with least impact on all top peaks
            ac_full = fast_autocorr(a)
            top_k = np.argsort(ac_full)[::-1][:10]
            scores = np.zeros(n)
            for k2 in top_k:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores2 = a[idx2] * a[k2i - idx2].sum()
                    scores[k2i - idx2] += a[idx2]
            scores[src] = 1e10
            dst = int(np.argmin(scores))
            if dst == src:
                continue

            a_new = a.copy()
            a_new[src] -= amt
            a_new[dst] += amt
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            M_new = fast_autocorr(a_new).max()
            if M_new < best_M - 1e-16:
                best_M = M_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1

        if verbose and it % 50 == 0:
            print("    Spectral it=%d (%.1fs) J=%.6f stall=%d" %
                  (it, time.monotonic()-t0, 2*n*best_M, stall))

    return best_a, 2.0 * n * best_M

        n_top = min(15, 2*n-1)

            # Score all positions for destination
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2] * 0.5
                    scores[k2i - idx2] += a[idx2] * 0.5
            scores[src] = 1e10

            # Also add "self-energy" penalty (a[j]^2 contributes to ac[0] and ac[2n-2])
            scores += a * a * 0.1

            dst = int(np.argmin(scores))
            if dst == src:
                dst = int((src + n // 2) % n)

            found = False
            for frac in fracs:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t / st
                Mt = fast_autocorr(a_t).max()
                if Mt < best_M - 1e-16:
                    a = a_t
                    best_M = Mt
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 300:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_M:
                            best_M = Mn
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 200 == 0 and stall > 50:
                for _ in range(3):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.008)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        st = a.sum()
                        if st > 1e-15:
                            a = a / st
                            Mn = fast_autocorr(a).max()
                            if Mn < best_M:
                                best_M = Mn
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 500 == 0:
            print("    Coord (%.1fs) J=%.6f stall=%d" %
                  (time.monotonic()-t0, 2*n*best_M, stall))

    return best_a, 2.0 * n * best_M


def multi_peak_deflation(a0, time_limit, n_targets=8, verbose=True):
    """Deflate multiple peaks simultaneously with careful redistribution."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for round_i in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        targets = np.argsort(ac)[::-1][:n_targets]
        M_curr = ac.max()

        # For each target peak, identify the top contributor and try to move it
        improved_any = False
        for k in targets[:4]:
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            ci_max = contrib.max()
            if ci_max < 1e-18:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination: minimize max increase across all targets
            scores = np.zeros(n)
            for k2 in targets:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    for jj in range(len(idx2)):
                        scores[idx2[jj]] += a[k2i - idx2[jj]]
                        scores[k2i - idx2[jj]] += a[idx2[jj]]
            scores[src] = 1e10
            dst = int(np.argmin(scores))
            if dst == src:
                dst = int((src + n // 3) % n)
                if dst == src:
                    continue

            for frac in [0.003, 0.005, 0.008, 0.012]:
                amt = src_val * frac
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    break
                a_t = a_t / st
                Mt = fast_autocorr(a_t).max()
                if Mt < best_M - 1e-16:
                    a = a_t
                    best_M = Mt
                    best_a = a.copy()
                    improved_any = True
                    break

        if not improved_any:
            # Random perturbation to escape
            if rng.random() < 0.3:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.002, 0.006)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_M:
                            best_M = Mn
                            best_a = a.copy()
                            improved_any = True

        if verbose and round_i % 100 == 0:
            print("    Deflate r=%d (%.1fs) J=%.6f" %
                  (round_i, time.monotonic()-t0, 2*n*best_M))

    return best_a, 2.0 * n * best_M


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 100.0, 1700.0)
    solver = get_solver()
    print("Budget: %.1fs" % time_budget)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            print("Parent: n=%d score=%.6f" % (len(p), best_score))
    except Exception:
        best_seq = None

    if best_seq is None:
        n_init = 11700
        x = np.linspace(0, 1, n_init)
        best_seq = normalize(np.sin(np.pi * x)**2)
        best_score = eval_safe(best_seq)
        print("Default init n=%d score=%.6f" % (n_init, best_score))

    n0 = len(best_seq)

    # Phase 1: LP-FW main optimization
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: LP-FW main (n=%d) ===" % n0)
    t1 = min(remaining * 0.40, 600.0)
    if t1 > 60:
        a_out, v_out = lp_fw_optimize(normalize(best_seq), t1, solver,
                                      init_K=600, max_K=5000, add_per=50, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  LP-FW1 improved: %.6f" % best_score)

    # Phase 2: Continue LP-FW with larger active set
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P2: LP-FW cont (n=%d) ===" % n_cur)
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 40:
        a_out, v_out = lp_fw_optimize(normalize(best_seq), t2, solver,
                                      init_K=800, max_K=6000, add_per=60, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  LP-FW2 improved: %.6f" % best_score)

    # Phase 3: Spectral / deflation polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P3: Multi-peak deflation (n=%d) ===" % n_cur)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_out, v_out = multi_peak_deflation(normalize(best_seq), t3, n_targets=10, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  Deflation improved: %.6f" % best_score)

    # Phase 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P4: Coord descent (n=%d) ===" % n_cur)
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        a_out, v_out = coord_descent_polish(normalize(best_seq), t4, seed=7, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  Coord improved: %.6f" % best_score)

    # Phase 5: Another LP-FW round (if time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if n_cur <= 15000:
        print("\n=== P5: LP-FW final (n=%d) ===" % n_cur)
        t5 = min(remaining * 0.30, 300.0)
        if t5 > 40:
            a_out, v_out = lp_fw_optimize(normalize(best_seq), t5, solver,
                                          init_K=1000, max_K=7000, add_per=80, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  LP-FW3 improved: %.6f" % best_score)

    # Phase 6: Final coord descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P6: Final coord polish (n=%d) ===" % n_cur)
    t6 = min(remaining * 0.30, 200.0)
    if t6 > 20:
        a_out, v_out = coord_descent_polish(normalize(best_seq), t6, seed=99, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  Final coord improved: %.6f" % best_score)

    # Phase 7: Deflation again
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        print("\n=== P7: Final deflation (n=%d) ===" % n_cur)
        t7 = min(remaining * 0.5, 150.0)
        if t7 > 20:
            a_out, v_out = multi_peak_deflation(normalize(best_seq), t7, n_targets=12, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  Final deflate improved: %.6f" % best_score)

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=%.6f n=%d t=%.1fs" % (best_score, len(best_seq), elapsed))
    else:
        print("\nERROR fallback")
        n_fb = 11700
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x)**2)
        best_score = eval_safe(best_seq)

    return [float(x) for x in best_seq]
