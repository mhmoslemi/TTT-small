def _lbfgs_direction(g, S, Y, rho):
    m=len(S)
    if m==0: return -g
    r = g.copy()
    alphas = np.zeros(m)
    for i in range(m-1,-1,-1):
        ai = np.dot(S[i], r) * rho[i]
        alphas[i] = ai
        r = r - ai * Y[i]
    p = -r
    for i in range(m):
        bi = np.dot(Y[i], p) * rho[i]
        p = p + (bi - alphas[i]) * S[i]
    return p
```

```python
"""
Minimize J(a) = 2*n*max(convolve(a,a))/(sum(a)**2) over non-negative sequences.

Strategy:
- Work with normalized sequences (sum = 1), so J = 2*n*max(a*a).
- Use an FFT-based smooth log-sum-exp approximation of max(a*a).
- Optimize in soft-max logit coordinates with a compact custom L-BFGS.
- Use several beta values (much more aggressively focused on the true maximum
  than typical tiny beta schedules).
- Start from the supplied parent if present, plus diverse explicit initializers
  based on the square-root-of-flat-window construction (central binomial
  coefficients) and simple envelope blends.
- Explore nearby lengths and do a final polish on the global best.
"""

import numpy as np
import time
import math
import operator

_T0 = None
_TIME_BUDGET = 1770.0
_GLOBAL_BEST_SEQ = None
_GLOBAL_BEST_SCORE = float("inf")


def _remaining():
    if _T0 is None:
        return float(_TIME_BUDGET)
    return max(0.0, float(_TIME_BUDGET) - (time.monotonic() - _T0))


def _norm_pos(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.reshape(-1)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones_like(a, dtype=np.float64) / max(1, len(a))
    return a / s


def _next_pow2(x):
    if x <= 1:
        return 1
    return 1 << int((x - 1).bit_length())


def _fft_len(n):
    return _next_pow2(max(2, 2 * int(n) - 1))


def _score_fixed(a, n, L, size):
    a = np.asarray(a, dtype=np.float64)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)
    c = c[:size]
    c = np.maximum(c, 0.0)
    return float(2.0 * float(n) * float(np.max(c)))


def _a_from_v(v, n):
    if len(v) != n - 1:
        return np.ones(n, dtype=np.float64) / float(n)
    z = np.empty(n, dtype=np.float64)
    z[:-1] = v
    z[-1] = 0.0
    if not np.all(np.isfinite(z)):
        return np.ones(n, dtype=np.float64) / float(n)
    z = np.clip(z, -700.0, 700.0)
    m = float(np.max(z))
    e = np.exp(z - m)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def _v_from_a(a):
    a = _norm_pos(a)
    n = len(a)
    a = np.maximum(a, 1e-300)
    v = np.log(a[:-1]) - np.log(a[-1])
    v = np.where(np.isfinite(v), v, 0.0)
    return np.clip(v, -100.0, 100.0)


def _value(v, beta, n, L, size):
    a = _a_from_v(v, n)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)
    c = c[:size]
    c = np.maximum(c, 0.0)
    cm = float(np.max(c))
    if beta <= 1.0:
        beta = 1.0
    shifted = beta * (c - cm)
    e = np.exp(shifted)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        s = 1.0
    f = cm + math.log(s) / beta
    if not np.isfinite(f):
        f = cm
    return float(f), cm


def _value_grad(v, beta, n, L, size):
    a = _a_from_v(v, n)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)
    c = c[:size]
    c = np.maximum(c, 0.0)
    cm = float(np.max(c))
    if beta <= 1.0:
        beta = 1.0
    shifted = beta * (c - cm)
    e = np.exp(shifted)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        w = np.ones(size, dtype=np.float64) / float(size)
        s = 1.0
    else:
        w = e / s
    f = cm + math.log(s) / beta
    if not np.isfinite(f):
        f = cm

    a_rev = a[::-1]
    W = np.fft.rfft(w, L)
    AR = np.fft.rfft(a_rev, L)
    conv = np.fft.irfft(W * AR, L)
    grad_a = 2.0 * conv[n - 1:2 * n - 1]

    dot = float(np.dot(a, grad_a))
    gfull = grad_a - a * dot
    g = gfull[:-1].copy()
    if not np.all(np.isfinite(g)):
        g = np.zeros(n - 1, dtype=np.float64)
    return float(f), g, cm


def _lbfgs_direction(g, S, Y, rho):
    m = len(S)
    if m == 0:
        return -g
    r = g.copy()
    alphas = np.zeros(m, dtype=np.float64)
    for i in range(m - 1, -1, -1):
        si = S[i]
        yi = Y[i]
        ai = float(np.dot(si, r) * rho[i])
        alphas[i] = ai
        r = r - ai * yi
    p = -r
    for i in range(m):
        bi = float(np.dot(Y[i], p) * rho[i])
        p = p + (bi - alphas[i]) * S[i]
    return p


def _line_search(v, p, g, f_base, cm_base, beta, n, L, size, gp):
    pnorm = float(np.linalg.norm(p))
    if (not np.isfinite(pnorm)) or pnorm <= 1e-300:
        return f_base, cm_base, v, False
    alpha = 1.0 / (1.0 + 0.25 * pnorm)
    best_f = float("inf")
    best_cm = cm_base
    best_v = v

    for _ in range(40):
        if (not np.isfinite(alpha)) or alpha <= 1e-14:
            break
        v_new = v + alpha * p
        if not np.all(np.isfinite(v_new)):
            alpha *= 0.5
            continue
        f_new, cm_new = _value(v_new, beta, n, L, size)
        if np.isfinite(f_new):
            if f_new < best_f - 1e-18 or cm_new < best_cm - 1e-18:
                best_f = f_new
                best_cm = cm_new
                best_v = v_new
            if f_new <= f_base + 1e-4 * alpha * gp:
                return f_new, cm_new, v_new, True
            if cm_new < cm_base - 1e-12 and f_new <= f_base + 1e-8:
                return f_new, cm_new, v_new, True
        alpha *= 0.5

    if np.isfinite(best_f) and best_f < f_base - 1e-13:
        return best_f, best_cm, best_v, True
    if (
        np.isfinite(best_f)
        and best_cm < cm_base - 1e-11
        and best_f < f_base + 1e-8
    ):
        return best_f, best_cm, best_v, True
    return f_base, cm_base, v, False

                break

            if not (np.isfinite(f) and np.all(np.isfinite(g))):
                break

            exact_new = 2.0 * n * cm
            if np.isfinite(exact_new) and exact_new < best_exact - 1e-15:
                best_exact = exact_new
                best_a = _a_from_v(v_new, n).copy()

            sv = v_new - v_old
            yg = g - g_old
            yts = float(np.dot(yg, sv))
            if np.isfinite(yts) and yts > 1e-14 and float(np.linalg.norm(sv)) > 1e-14:
                S.append(sv)
                Y.append(yg)
                rho.append(1.0 / yts)
                if len(S) > max_memory:
                    S.pop(0)
                    Y.pop(0)
                    rho.pop(0)
            else:
                if len(S) > 0:
                    S.pop(0)
                    Y.pop(0)
                    rho.pop(0)

            v = v_new

            if verbose and (it == 0 or it % 100 == 0):
                el = time.monotonic() - _T0
                print(
                    f"      iter={it:4d} beta={beta:9.1f} J={2.0*n*cm:.7f} "
                    f"best={best_exact:.7f} t={el:.0f}s",
                    flush=True,
                )

        v = _v_from_a(best_a)

    return best_a, best_exact


def _central_binomial(n):
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64) / float(max(1, n))
    b = np.empty(n, dtype=np.float64)
    b[0] = 1.0
    if n > 1:
        k = np.arange(n - 1, dtype=np.float64)
        b[1:] = np.cumprod((2.0 * k + 1.0) / (2.0 * k + 2.0))
    return _norm_pos(b)


def _interpolate_sequence(a, new_n):
    a = _norm_pos(a)
    old_n = len(a)
    new_n = int(new_n)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    return _norm_pos(np.interp(new_x, old_x, a))


def _candidate_pool(n, ref, budget_s=60.0):
    n = int(n)
    L = _fft_len(n)
    size = 2 * n - 1
    i = np.arange(n, dtype=np.float64)
    b = _central_binomial(n)

    cands = []
    cands.append(b)
    cands.append(b[::-1])

    for p in [-0.75, -0.5, -0.25, 0.0, 0.25, 0.5, 0.75, 1.0, 1.5, 2.0, 2.5]:
        if p < 0.0:
            w = np.exp(p * np.log(i + 0.5))
        else:
            w = np.exp(p * np.log(i + 1.0))
        cands.append(b * w)

    for rate in [-1.0, -0.5, 0.0, 0.5, 1.0, 2.0, 3.0, 4.0]:
        cands.append(b * np.exp(rate * i / max(1, n - 1)))

    for alpha in [0.1, 0.25, 0.5, 1.0, 2.0]:
        cands.append(b + alpha * b[::-1])

    prod = b * b[::-1]
    cands.append(prod)
    cands.append(np.sqrt(np.maximum(prod, 0.0)))

    if ref is not None:
        ref = _norm_pos(ref)
        if len(ref) == n:
            for t in [0.2, 0.4, 0.6, 0.8]:
                cands.append((1.0 - t) * b + t * ref)
                tail_boosted = _norm_pos(b * np.exp(1.0 * i / max(1, n - 1)))
                cands.append((1.0 - t) * tail_boosted + t * ref)

    scored = []
    t0 = time.monotonic()
    for cand in cands:
        if time.monotonic() - t0 > max(1.0, float(budget_s)):
            break
        a = _norm_pos(cand)
        sc = _score_fixed(a, n, L, size)
        if np.isfinite(sc):
            scored.append((float(sc), a))

    scored.sort(key=operator.itemgetter(0))
    return scored


def _consider(a, label):
    global _GLOBAL_BEST_SEQ, _GLOBAL_BEST_SCORE
    if a is None:
        return
    try:
        a2 = _norm_pos(a)
        n = len(a2)
        if n < 2:
            return
        L = _fft_len(n)
        size = 2 * n - 1
        sc = _score_fixed(a2, n, L, size)
    except Exception:
        return
    if np.isfinite(sc) and sc < _GLOBAL_BEST_SCORE - 1e-12:
        _GLOBAL_BEST_SCORE = float(sc)
        _GLOBAL_BEST_SEQ = a2
        el = time.monotonic() - _T0 if _T0 is not None else 0.0
        print(f"  *** NEW BEST [{label}]: J={sc:.8f}, n={n}, t={el:.0f}s", flush=True)


def propose_candidate(budget_s=1800.0):
    global _T0, _TIME_BUDGET, _GLOBAL_BEST_SEQ, _GLOBAL_BEST_SCORE

    _T0 = time.monotonic()
    _TIME_BUDGET = max(60.0, min(1770.0, float(budget_s) - 30.0))
    _GLOBAL_BEST_SEQ = None
    _GLOBAL_BEST_SCORE = float("inf")

    print(f"Total internal time budget: {_TIME_BUDGET:.0f}s", flush=True)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.maximum(p, 0.0)
            if p.sum() > 1e-12:
                parent = p / p.sum()
    except Exception:
        parent = None

    if parent is not None:
        if len(parent) > 9000:
            parent = _interpolate_sequence(parent, 8000)
        elif len(parent) < 1000:
            parent = _interpolate_sequence(parent, 5000)
        _consider(parent, "parent")
        n_primary = len(parent) if parent is not None else 5200
        if _GLOBAL_BEST_SEQ is None:
            _GLOBAL_BEST_SEQ = _central_binomial(n_primary)
            _consider(_GLOBAL_BEST_SEQ, "central-fallback")
    else:
        n_primary = 5200
        _consider(_central_binomial(n_primary), "central-init")

    if _GLOBAL_BEST_SEQ is None:
        _consider(_central_binomial(4000), "safe-fallback")

    print(
        f"Initial best: J={_GLOBAL_BEST_SCORE:.8f}, n={len(_GLOBAL_BEST_SEQ)}",
        flush=True,
    )

    # Diverse inits at the primary length.
    pool = []
    if _remaining() > 40:
        pool = _candidate_pool(
            n_primary,
            _GLOBAL_BEST_SEQ,
            budget_s=min(70.0, _remaining() * 0.05),
        )

    top = [(_GLOBAL_BEST_SCORE, _GLOBAL_BEST_SEQ.copy())]
    for sc, a in pool[:3]:
        top.append((float(sc), a.copy()))

    # Main optimization on the top few diverse starts.
    main_budget = _TIME_BUDGET * 0.62
    fracs = [0.48, 0.30, 0.14, 0.08]
    for idx, (sc0, a0) in enumerate(top):
        if _remaining() < 60:
            break
        frac = fracs[idx] if idx < len(fracs) else 0.05
        alloc = min(main_budget * frac, _remaining() - 30.0)
        if alloc < 35:
            continue
        label = f"start{idx} J0={sc0:.7f}"
        print(f"\nOptimizing {label} with {alloc:.0f}s", flush=True)
        try:
            a_out, sc_out = _optimize(a0, alloc, verbose=True)
            _consider(a_out, label + " out")
        except Exception as e:
            print("  optimization failed:", e, flush=True)

    # Length exploration.
    if _remaining() > 180:
        print("\n=== Length exploration ===", flush=True)
        candidates_n = []
        for delta in [-2000, -1000, 1000, 2000]:
            candidates_n.append(n_primary + delta)
        for rn in [4096, 4608, 5120, 6144, 7168, 8192]:
            candidates_n.append(rn)

        for new_n in candidates_n:
            if _remaining() < 120:
                break
            new_n = int(new_n)
            if not (2500 <= new_n <= 9000):
                continue
            if new_n == n_primary:
                continue

            a_interp = _interpolate_sequence(_GLOBAL_BEST_SEQ, new_n)
            L = _fft_len(new_n)
            size = 2 * new_n - 1
            sc_int = _score_fixed(a_interp, new_n, L, size)

            a_cent = _central_binomial(new_n)
            sc_cent = _score_fixed(a_cent, new_n, L, size)

            if sc_cent < sc_int:
                a_start = a_cent
                sc_start = sc_cent
            else:
                a_start = a_interp
                sc_start = sc_int

            if not np.isfinite(sc_start):
                continue

            alloc = min(140.0, _remaining() * 0.45)
            if alloc < 40:
                continue

            label = f"n={new_n} J0={sc_start:.7f}"
            print(f"  trying {label}, alloc={alloc:.0f}s", flush=True)
            try:
                a_out, sc_out = _optimize(a_start, alloc, verbose=False)
                _consider(a_out, label + " out")
            except Exception as e:
                print("  length optimization failed:", e, flush=True)

    # Final polish.
    if _remaining() > 90:
        alloc = min(260.0, _remaining() * 0.70)
        print(f"\n=== Final polish: {alloc:.0f}s ===", flush=True)
        try:
            a_out, sc_out = _optimize(_GLOBAL_BEST_SEQ, alloc, verbose=True)
            _consider(a_out, "final-polish")
        except Exception as e:
            print("final polish failed:", e, flush=True)

    if _GLOBAL_BEST_SEQ is None:
        _consider(_central_binomial(4000), "hard-fallback")

    final = [float(x) for x in _GLOBAL_BEST_SEQ]

    # One direct validation with the official evaluator if available.
    try:
        direct = float(evaluate_sequence(final))
        if np.isfinite(direct):
            print(f"Direct evaluate_sequence: J={direct:.8f}, n={len(final)}", flush=True)
            if direct < _GLOBAL_BEST_SCORE - 1e-12:
                _GLOBAL_BEST_SCORE = direct
                _GLOBAL_BEST_SEQ = _norm_pos(final)
                final = [float(x) for x in _GLOBAL_BEST_SEQ]
        else:
            print("Direct evaluation invalid; using fallback.", flush=True)
            fallback = _central_binomial(4000)
            return [float(x) for x in fallback]
    except Exception:
        pass

    print(f"RETURNING: J={_GLOBAL_BEST_SCORE:.8f}, n={len(final)}", flush=True)
    return final
