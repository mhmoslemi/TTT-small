"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:

import cvxpy as cp
from scipy import sparse


def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_gradient(a, beta, top_k=None):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (3 * n - 3).bit_length()
    
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]

        top_idx = np.argsort(c)[::-1][:top_k]

    A_rev = np.fft.rfft(a_rev, fft_size)

def fista_optimize(a0, time_limit, beta_start=5.0, beta_end=100.0,
                   eta0=0.8, top_k_frac=0.05, verbose=True):

    y = x.copy()
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_threshold = 10000
    
    top_k = max(50, int(top_k_frac * (2 * n - 1)))
    
    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        progress = min(iter_count / max(time_limit * 150, 1), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        grad, c_y, _ = softmax_gradient(y, beta, top_k=top_k)
        eta = eta0 / (np.sqrt(t) + 1.0)

        log_y -= log_y.max()
        z = np.exp(log_y)
        z = np.maximum(z, 1e-30)
        z = z / z.sum()
        
        # Nesterov momentum
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = z + momentum * (z - y)
            x_new = np.maximum(x_new, 1e-30)
            x_new = x_new / x_new.sum()
        else:
            x_new = z.copy()

            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1

            rng = np.random.default_rng(iter_count)
            pert = rng.normal(0, 0.01, n) * best_x
            x_restart = norm_sum1(best_x + pert)
            x_new = x_restart
            y = x_restart.copy()
            stall = 0

        x = x_new

            print(f"    FISTA iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}")

def build_constraint_matrix(constraint_list, n, y):
    """Build sparse constraint matrix for B(y, g) <= M."""
    K = len(constraint_list)
    rows = []
    cols = []
    vals = []
    
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            rows.extend([jj] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(y[k_int - idx].tolist())
    
    return sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))


def lp_frank_wolfe(a0, time_limit, solver, initial_K=300, max_K=2000, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()
    
    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)
    
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
        
        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()
        
        # Adaptive constraint management
        constraint_list = sorted(constraint_set)
        
        # Prune: keep only constraints that are close to binding
        if len(constraint_list) > initial_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            # Keep constraints with small margin (close to binding)
            keep_count = min(len(constraint_list), max_K)
            keep_idx = np.argsort(margins)[:keep_count]
            constraint_set = set(int(constraint_list[i]) for i in keep_idx)
            constraint_list = sorted(constraint_set)
        
        # Add violated constraints
        viol = np.where(ac > M_curr - 1e-10)[0]
        # Actually, ac values above M_curr shouldn't exist since M_curr = max(ac)
        # Instead, add constraints near the peak
        near_peak = np.where(ac > 0.95 * M_curr)[0]
        for k in near_peak:
            if k not in constraint_set:
                constraint_set.add(int(k))
        
        constraint_list = sorted(constraint_set)
        K = len(constraint_list)
        if K == 0:
            break
        
        if K > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep_idx = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep_idx)
            constraint_list = sorted(constraint_set)
            K = len(constraint_list)
        
        # Build sparse constraint matrix
        C = build_constraint_matrix(constraint_list, n, y)
        
        # Solve LP: min M s.t. sum(g)=1, g>=0, C@g <= M
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, C @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Fallback: reduce constraints and retry
            constraint_list = constraint_list[:min(initial_K, K)]
            constraint_set = set(constraint_list)
            C = build_constraint_matrix(constraint_list, n, y)
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [cp.sum(g_var) == 1, C @ g_var <= M_var]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # FW step with Nesterov acceleration
        alpha = 2.0 / (iter_count + 3.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if iter_count == 0:
            x_new = y_new.copy()
        else:
            beta_t = (iter_count) / (iter_count + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")
    
    return best_a, best_m


def parametric_init(n, best_ref=None):
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')
    
    candidates = []
    
    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    
    for alpha in [0.5, 0.8, 1.0, 1.2]:
        for beta_p in [0.5, 0.8, 1.0, 1.2]:
            candidates.append(np.maximum(i_arr + 1.0, 0.5) ** alpha *
                              np.maximum(n - i_arr, 0.5) ** beta_p)
    
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    
    for rate in [1, 2, 3, 5]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1.0)))
    
    for freq in [1, 2, 3]:
        candidates.append(np.maximum(np.sin(np.pi * freq * x) ** 2, 0.01))
    
    # Cosine bump
    candidates.append(np.cos(np.pi * (x - 0.5)) ** 2)
    # Raised cosine
    candidates.append(0.5 + 0.5 * np.cos(2 * np.pi * x))
    
    if best_ref is not None and len(best_ref) == n:
        ref_norm = norm_sum1(best_ref)
        for w in [0.3, 0.5, 0.7]:
            candidates.append(w * ref_norm + (1.0 - w) * norm_sum1((i_arr + 1.0) ** 0.8))
    
    for a_cand in candidates:
        a_norm = norm_sum1(a_cand)
        ac = fast_autocorr(a_norm)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_a = a_norm
    
    return best_a, best_val


def get_solver():
    for s in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS]:
        try:
            x = cp.Variable(30, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): SCS")
    return cp.SCS


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
