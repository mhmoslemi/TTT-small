def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def build_constraint_matrix_vec(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def cluster_active_set(ac, n, max_clusters=8, max_per_cluster=50):
    """Cluster the peak lags into groups and select representatives."""
    K = len(ac)
    # Get the top peaks
    top_idx = np.argsort(ac)[::-1][:max_per_cluster * max_clusters]
    top_vals = ac[top_idx]
    
    # Simple clustering: sort peaks and group nearby ones
    # Peaks at lags k and k' are "nearby" if |k - k'| < n/min(2, max_clusters)
    threshold = n // max(2, max_clusters)
    
    clustered = []
    used = set()
    for i in range(len(top_idx)):
        if i in used:
            continue
        cluster = [top_idx[i]]
        used.add(i)
        for j in range(i + 1, len(top_idx)):
            if j in used:
                continue
            if abs(int(top_idx[j]) - int(top_idx[i])) < threshold:
                cluster.append(top_idx[j])
                used.add(j)
        if len(cluster) > max_per_cluster:
            cluster = cluster[:max_per_cluster]
        clustered.append(cluster)
        if len(clustered) >= max_clusters:
            break
    
    # Also add the absolute top peaks
    all_selected = set()
    for cluster in clustered:
        for k in cluster:
            all_selected.add(int(k))
    
    # Ensure the top few peaks are always included
    for k in top_idx[:10]:
        all_selected.add(int(k))
    
    return sorted(all_selected)


def lp_fw_improved(a0, time_limit, solver, initial_K=300, max_K=3000,

        K = len(active_set)

            prob.solve(solver=solver, verbose=False)
        except Exception:

def subgradient_refinement(a0, time_limit, top_k=50, verbose=False):
    """Subgradient method with adaptive step size for the peak function."""
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_m = fast_convolve(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    gamma = 5.0
    stall = 0
    max_stall = 3000
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break
        
        # Compute subgradient of max convolution
        ac = fast_convolve(x)
        peak_idx = np.argmax(ac)
        
        # The subgradient of (a*a)_k at a is 2 * a shifted by k
        k = int(peak_idx)
        grad = np.zeros(n)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        grad[idx] = 2.0 * x[k - idx]
        
        # Also include top few peaks for a smoother gradient
        top_idx = np.argsort(ac)[::-1][:top_k]
        weights = np.maximum(ac[top_idx] - ac[peak_idx] + 1e-10, 1e-10)
        weights = weights / weights.sum()
        
        grad_smooth = np.zeros(n)
        for wi, ki in zip(weights, top_idx):
            ki = int(ki)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            if s < e:
                idx2 = np.arange(s, e)
                grad_smooth[idx2] += wi * 2.0 * x[ki - idx2]
        
        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_x = np.log(np.maximum(x, 1e-30)) - eta * grad_smooth
        max_log = log_x.max()
        x_new = np.exp(log_x - max_log)
        x_new = np.maximum(x_new, 1e-30)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / s_new
        
        x = x_new
        M_new = fast_convolve(x).max()
        
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
    
    J = 2.0 * n * best_m
    if verbose:
        print("    Subgrad done: J=" + repr(J))
    return best_a, best_m


def perturbation_escape(a0, time_limit, rng_seed=42, verbose=False):

def coordinate_polish(a0, time_limit, rng_seed=99, verbose=False):
    """Greedy coordinate descent with peak-aware mass transfer."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            # Find the source (highest contributor to this peak)
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find the best destination (lowest contribution to all peaks)
            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]
            
            # Destination should be far from all peaks
            dst = int(np.argmin(scores))
            
            if dst == src:
                continue
            
            # Try transferring mass
            found = False
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break
            
            if not found and stall > 200 and rng.random() < 0.3:
                # Random transfer
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if not improved:
            stall += 1
    
    J = 2.0 * n * best_m
    if verbose:
        print("    Coord done: J=" + repr(J))
    return best_a, best_m

def symmetrize(a):
    """Enforce a[i] = a[n-1-i]."""
    n = len(a)
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    for i in range(half):
        a_sym[i] = a[i] + a[n-1-i]
        a_sym[n-1-i] = a_sym[i]
    return a_sym


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def select_active_set(ac, n, max_K, mode="cluster"):
    """Select active constraints from the autocorrelation."""
    size = len(ac)
    if mode == "top":
        return sorted(np.argsort(ac)[::-1][:max_K].tolist())
    elif mode == "cluster":
        # Cluster peaks: group nearby lags
        top_idx = np.argsort(ac)[::-1][:max_K * 2]
        # Use a sliding window approach
        threshold = max(1, n // 8)
        selected = []
        for k in top_idx:
            k_int = int(k)
            # Check if this lag is already "covered" by a selected lag
            covered = any(abs(k_int - s) < threshold for s in selected)
            if not covered:
                selected.append(k_int)
            if len(selected) >= max_K:
                break
        # Always include the top 5 peaks
        for k in top_idx[:5]:
            k_int = int(k)
            if k_int not in selected:
                selected.append(k_int)
                if len(selected) > max_K:
                    selected.pop()
        return sorted(selected[:max_K])
    else:
        return sorted(np.argsort(ac)[::-1][:max_K].tolist())
