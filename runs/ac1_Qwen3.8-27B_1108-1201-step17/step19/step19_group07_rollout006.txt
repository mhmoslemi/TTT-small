    # Let me think more carefully.
    # c[k] = sum_j a[j]*a[k-j], so dc[k]/da[i] = a[k-i] (if 0<=k-i<n)
    # grad[i] = sum_k w[k] * dc[k]/da[i] = sum_k w[k] * a[k-i] (if 0<=k-i<n)
    # Let m = k-i, then k = m+i, m in [0, n-1]:
    # grad[i] = sum_{m=0}^{n-1} w[m+i] * a[m]
    # This is a "cross-correlation" of w and a.
    
    # Using FFT: grad[i] = sum_m w[m+i]*a[m]
    # This is the (i)-th element of the cross-correlation of a and w.
    # cross_corr[i] = sum_m a[m]*w[m+i]
    
    # We can compute this as: rev_a * w (convolution), taken at appropriate indices
    # (rev_a * w)[n-1+i] = sum_j rev_a[j]*w[n-1+i-j] = sum_j a[n-1-j]*w[n-1+i-j]
    # Let m = n-1-j: = sum_m a[m]*w[i+m] for valid m
    # So grad[i] = (rev_a * w)[n-1+i]
    
    rev_a = a[::-1]
    size2 = 2 * n - 1
    fft_size3 = 1 << (size2 - 1).bit_length()
    Ra = np.fft.rfft(rev_a, fft_size3)
    W2 = np.fft.rfft(w, fft_size3)
    corr = np.fft.irfft(Ra * W2, fft_size3)[:size2]
    grad = corr[n-1:2*n-1]
    
    return 2.0 * beta * grad, c, w


def smooth_gradient_descent(a0, time_limit, beta=50.0, verbose=True):
    """Polish using L-BFGS-B on the smooth surrogate."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = evaluate_safe(a)
    t0 = time.monotonic()
    
    def objective(log_params):
        a_try = np.exp(log_params)
        a_try = np.maximum(a_try, 1e-30)
        a_try = a_try / a_try.sum()
        ac = fast_autocorr(a_try)
        # Smooth max: logsumexp(beta * ac)
        ac_max = ac.max()
        smooth_max = (1.0 / beta) * np.log(np.sum(np.exp(beta * (ac - ac_max)))) + ac_max
        return 2.0 * n * smooth_max
    
    def gradient(log_params):
        a_try = np.exp(log_params)
        a_try = np.maximum(a_try, 1e-30)
        a_try = a_try / a_try.sum()
        grad_ac, ac, w = softmax_jacobian(a_try, beta, top_k=200)
        # grad of smooth_max w.r.t. a
        # smooth_max = mean_weighted = sum_k w_k * ac_k (where w_k = softmax(beta*ac))
        # d smooth_max / d a_i = sum_k w_k * ac_k' / da_i = (as computed in softmax_jacobian)
        # But we have the gradient of logsumexp, which is sum_k w_k * da/d a_i
        # Actually softmax_jacobian returns 2*beta*grad which is the grad of logsumexp(beta*ac)
        # We want grad of (1/beta)*logsumexp(beta*ac) + ac_max = sum_k w_k * ac_k
        # d/d a_i [sum_k w_k * ac_k] = ... this gets complicated with the w_k dependence
        # For simplicity, just use the gradient of logsumexp as a proxy
        # scale: d/d a_i [(1/beta) logsumexp(beta * ac)] = sum_k w_k * dac_k/da_i
        grad_smooth = grad_ac / beta  # this is sum_k w_k * dac_k/da_i
        # But we also need to account for the normalization (sum a = 1)
        # Using chain rule through the exp and normalization:
        # d objective / d log_params_i = d objective / d a_i * a_i
        return grad_smooth * a_try
    
    log_a = np.log(np.maximum(a, 1e-30))
    
    result = sp_minimize(
        objective, log_a, method='L-BFGS-B', jac=gradient,
        options={'maxiter': 50, 'maxfun': 2000, 'ftol': 1e-12, 'gtol': 1e-8}
    )
    
    if result.fun < best_J:
        a_result = np.exp(result.x)
        a_result = np.maximum(a_result, 1e-30)
        a_result = a_result / a_result.sum()
        J_result = evaluate_safe(a_result)
        if np.isfinite(J_result) and J_result < best_J:
            best_a = a_result
            best_J = J_result
    
    elapsed = time.monotonic() - t0
    print(f"    Smooth GD: J={best_J:.6f} ({elapsed:.0f}s)")
    return best_a, best_J


def peak_shave(a, n_shaves=20, fraction=0.01):
    """Reduce elements that contribute most to top autocorrelation peaks."""
    a = a.copy()
    n = len(a)
    for _ in range(n_shaves):
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:5]
        for k in top_k:
            k = int(k)
            # Find elements that contribute most to conv[k]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            contribs = a[idx] * a[k - idx]
            # Reduce the top contributors
            top_contrib = np.argsort(contribs)[::-1][:10]
            for tc in top_contrib:
                a[idx[tc]] *= (1 - fraction)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f
