def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A,fft_size)[:size]

def fast_conv(a, b):
    """Full convolution of a and b using FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:n]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    """Project z onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def _interp_to(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _compute_grad_correct(y, beta, top_k=None):
    """Compute correct gradient of (1/beta)*log(sum top_k exp(beta*b)) w.r.t. a.
    Returns (grad, p, b) where grad has the correct 2x factor."""
    n = len(y)
    b = _fast_autocorr(y)
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    # Gradient = 2 * c where c[j] = sum_m a[m]*p[m+j]
    # c = conv(a_rev, p)[n-1:2n-1]
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    c = conv[n - 1:2 * n - 1]
    grad = 2.0 * c
    return grad, p, b


def _smoothed_value(y, beta, top_k=None):
    """Compute the smoothed objective value (not the true max)."""
    n = len(y)
    b = _fast_autocorr(y)
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = b[idx]
    else:
        b_use = b
    b_max = b_use.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    return (1.0 / beta) * np.log(exp_vals.sum()) + b_max


def _nesterov_v2(a0, time_limit, beta_start=20.0, beta_end=2000.0,
                 beta_growth=1.005, top_k_ratio=0.04,
                 restart_every=4000, verbose=True):
    """Nesterov with corrected gradient and adaptive step size."""
    n = len(a0)
    x = _norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = _fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(12345)
    top_k = max(30, int(n * top_k_ratio))
    best_smoothed = _smoothed_value(x, beta, top_k)

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-18)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, b = _compute_grad_correct(y, beta, top_k)
        b_max = b.max()

        # Adaptive step size: smaller when beta is large (sharper gradient)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            x_new = x.copy()
        else:
            eta = 1.0 / max(grad_norm * 1.1, 0.01)
            eta = min(eta, 0.5 / max(1.0, beta * 0.01))
            z = y - eta * grad
            x_new = _simplex_project(z)

        # Track best on TRUE objective
        b_new = _fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-14:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        # Also track smoothed for convergence detection
        sw = _smoothed_value(x_new, beta, top_k)
        if sw < best_smoothed - 1e-10:
            best_smoothed = sw

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 60 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.03 / (1.0 + 0.15 * (t // restart_every))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 8.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest2 restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest2 t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + str(round(beta, 1)))

    return best_a, best_max_b


def _lp_fw_v2(a0, time_limit, solver, initial_K=300, max_K=2500,
              add_per_iter=25, verbose=True):
    """LP-Frank-Wolfe with improved active set management."""
    n = len(a0)
    y = _norm_sum1(a0)
    best_a = y.copy()
    best_m = _fast_autocorr(y).max()
    t0 = time.monotonic()
    y_ac = _fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(y_ac)[::-1][:min(initial_K, 2 * n - 1)])
    no_improve = 0
    max_no_improve = 800

    for t in range(1, 3000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW2: converged at iter " + str(t))
            break

        ac = _fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set size
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix A @ g <= M
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(y_ac)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-6)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5),
                  max(0.02, default_alpha * 2.0), 1.0 / (t + 1)]
        alphas = [max(0.01, min(1.0, a)) for a in alphas]
        alphas = sorted(set(alphas))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = _fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y
        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW2 iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def _pressure_descent(a0, time_limit, verbose=True):
    """Pressure-based peak-shaving coordinate descent."""
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    max_stall = 600
    n_peaks = min(30, 2 * n - 1)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = _fast_autocorr(a)
        top_idx = np.argsort(ac)[::-1][:n_peaks]

        # Compute pressure
        pressure = np.zeros(n)
        for k in top_idx[:15]:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            w = ac[k_int] / ac[top_idx[:15]].sum()
            pressure[idx] += w * a[k_int - idx]
            pressure[k_int - idx] += w * a[idx]

        # Find highest pressure position to reduce
        max_p_idx = int(np.argmax(pressure))
        if a[max_p_idx] < 1e-12:
            stall += 1
            continue

        # Find safest destination (lowest pressure, not in peak zone)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_idx[:10]:
            k_int = int(k)
            s2 = max(0, k_int - n + 1)
            e2 = min(n - 1, k_int)
            in_peak[s2:e2 + 1] = True

        candidates = np.where(~in_peak)[0]
        if len(candidates) < 50:
            candidates = np.argsort(pressure)[:n // 4]
        candidates = candidates[candidates != max_p_idx]
        if len(candidates) == 0:
            stall += 1
            continue

        dst = int(candidates[np.argmin(pressure[candidates])])

        # Try different fractions
        improved = False
        for frac in [0.02, 0.05, 0.1, 0.2, 0.35]:
            amt = a[max_p_idx] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[max_p_idx] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = _fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-15:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                stall = 0
                improved = True
                break

        if not improved:
            # Try random moves
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.08)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-1:
                        a_trial = a_trial / s_t
                        M_trial = _fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-15:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            stall = 0
                            improved = True
                            break

        if not improved:
            stall += 1
            if stall % 200 == 0:
                # Perturb for escape
                pert = rng.normal(0, 0.01, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-10)
                a_r = a_r / a_r.sum()
                a = a_r
                stall = max(0, stall - 100)

        if verbose and stall % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Pressure (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def _parametric_init(n, time_limit):
    """Generate diverse parametric initializations and return the best."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')

    candidates = []
    # Sine powers
    for p in [1, 2, 3, 4, 6]:
        candidates.append(np.sin(np.pi * x) ** p)
    # Power laws
    for p in [0.3, 0.5, 1.0, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    # Asymmetric
    for p in [0.5, 1.0, 2.0]:
        candidates.append(np.exp(-p * x))
        candidates.append(np.exp(-p * (1.0 - x)))
    # Gaussian bumps
    for c in [0.3, 0.5, 0.7]:
        for sig in [0.1, 0.2, 0.3]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sig ** 2)))
    # Cosine
    for freq in [1, 2, 3]:
        candidates.append(np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x)) / 2.0, 0.01))
    # Product of sine and exponential
    candidates.append(np.sin(np.pi * x) * np.exp(-2.0 * x))
    candidates.append(np.sin(np.pi * x) * np.exp(-2.0 * (1.0 - x)))
    candidates.append(np.sin(np.pi * x) ** 2 * np.exp(-1.0 * np.abs(x - 0.5) * 4.0))

    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a = _norm_sum1(a_raw)
        m = _fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    # Random
    rng = np.random.default_rng(42)
    for _ in range(80):
        if time.monotonic() - t0 > time_limit:
            break
        alpha = rng.uniform(0.5, 4.0)
        a = _norm_sum1(rng.random(n) ** alpha)
        m = _fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    # Structured random (block-constant)
    for n_blocks in [5, 10, 20, 50]:
        if time.monotonic() - t0 > time_limit:
            break
        for _ in range(10):
            block_vals = rng.random(n_blocks) ** rng.uniform(0.5, 3.0)
            a = np.repeat(block_vals, n // n_blocks + 1)[:n]
            a = _norm_sum1(a)
            m = _fast_autocorr(a).max()
            if m < best_val:
                best_val = m
                best_a = a.copy()

    return best_a, best_val


def _spectral_flatten(a0, time_limit, verbose=True):
    """Iteratively flatten the magnitude spectrum to reduce circular autocorrelation peaks."""
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(314)

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        A = np.fft.rfft(a, 2 * n)
        mag = np.abs(A)
        # Target: flat magnitude
        target = mag.mean()
        # Compute circular autocorrelation
        circ_ac = np.fft.irfft(A * A, 2 * n)[:n]
        max_circ = circ_ac.max()

        # Gradient of max circular autocorrelation
        peak_idx = int(np.argmax(circ_ac))
        # d/d a[i] of circ_ac[peak] = 2 * a[(peak-i) % n] (approximately)
        gradient = 2.0 * a[(peak_idx - np.arange(n)) % n]

        # Subtract gradient and project
        eta = 0.01 / (1.0 + t * 0.001)
        a_new = a - eta * gradient
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        m = _fast_autocorr(a_new).max()
        if m < best_m - 1e-14:
            best_m = m
            best_a = a_new.copy()
        a = a_new

        if t % 5000 == 0:
            # Small perturbation
            pert = rng.normal(0, 0.005, n) * a
            a = np.maximum(a + pert, 0)
            a = a / a.sum()

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def _try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = _eval_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + repr(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search: multi-phase optimization with diverse strategies."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = _get_solver()
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = _eval_safe(parent)
            if np.isfinite(p_score):
                best_seq = _norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(round(p_score, 6)))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = _norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = _eval_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(round(best_score, 6)))

    n0 = len(best_seq)

    # Phase 1: Parametric + diverse init
    print("\n=== Phase 1: Parametric init (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 40.0)
    if t1 > 10:
        a_param, _ = _parametric_init(n0, t1)
        best_seq, best_score = _try_update(best_seq, best_score, a_param, "param", t_global)

    # Phase 2: Nesterov (main)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov v2 (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 60:
        a_out, _ = _nesterov_v2(_norm_sum1(best_seq), t2,
                                beta_start=20.0, beta_end=3000.0,
                                beta_growth=1.005, top_k_ratio=0.04,
                                restart_every=5000, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "nest2_n" + str(n_best), t_global)

    # Phase 3: LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP-FW v2 (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 40 and n_best <= 8000:
        a_out, _ = _lp_fw_v2(_norm_sum1(best_seq), t3, solver,
                             initial_K=400, max_K=2500,
                             add_per_iter=25, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "lp2_n" + str(n_best), t_global)

    # Phase 4: Pressure descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Pressure descent (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_out, _ = _pressure_descent(_norm_sum1(best_seq), t4, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "pressure_n" + str(n_best), t_global)

    # Phase 5: Spectral flatten
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Spectral flatten (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.08, 120.0)
    if t5 > 20:
        a_out, _ = _spectral_flatten(_norm_sum1(best_seq), t5, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "spectral_n" + str(n_best), t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = [n_best + 500, n_best + 1000, n_best + 2000]
    longer_lengths = [l for l in longer_lengths if l <= 9000]

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print("\n=== Phase 6: Scale n=" + str(new_n) + " ===")
        a_long = _interp_to(best_seq, new_n)

        t6a = min(remaining * 0.35, 200.0)
        if t6a > 30:
            a_out, _ = _nesterov_v2(a_long, t6a,
                                    beta_start=25.0, beta_end=2000.0,
                                    beta_growth=1.005, top_k_ratio=0.03,
                                    restart_every=4000, verbose=True)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, "nest2_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 8000:
            t6b = min(remaining * 0.35, 180.0)
            if t6b > 30:
                a_long2 = _interp_to(best_seq, new_n)
                a_out, _ = _lp_fw_v2(_norm_sum1(a_long2), t6b, solver,
                                     initial_K=400, max_K=2000,
                                     add_per_iter=20, verbose=True)
                best_seq, best_score = _try_update(best_seq, best_score, a_out, "lp2_n" + str(new_n), t_global)

    # Phase 7: Second round of Nesterov with higher beta
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Nesterov high-beta (n=" + str(n_best) + ") ===")
    if remaining > 60:
        t7 = min(remaining * 0.25, 200.0)
        if t7 > 30:
            a_out, _ = _nesterov_v2(_norm_sum1(best_seq), t7,
                                    beta_start=100.0, beta_end=5000.0,
                                    beta_growth=1.008, top_k_ratio=0.02,
                                    restart_every=3000, verbose=True)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, "nest2_hb_n" + str(n_best), t_global)

    # Phase 8: Final pressure polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final pressure (n=" + str(n_best) + ") ===")
    if remaining > 20:
        t8 = min(remaining * 0.30, 120.0)
        if t8 > 15:
            a_out, _ = _pressure_descent(_norm_sum1(best_seq), t8, verbose=True)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, "pressure_final", t_global)

    # Phase 9: Final LP burst
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40 and n_best <= 8000:
        print("\n=== Phase 9: Final LP burst (n=" + str(n_best) + ") ===")
        t9 = min(remaining * 0.70, 180.0)
        if t9 > 20:
            a_out, _ = _lp_fw_v2(_norm_sum1(best_seq), t9, solver,
                                 initial_K=500, max_K=3000,
                                 add_per_iter=30, verbose=True)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, "lp_burst", t_global)

    # Phase 10: Tiny final Nesterov polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t10 = min(remaining * 0.80, 100.0)
        if t10 > 10:
            print("\n=== Phase 10: Tiny polish (n=" + str(n_best) + ") ===")
            a_out, _ = _nesterov_v2(_norm_sum1(best_seq), t10,
                                    beta_start=500.0, beta_end=10000.0,
                                    beta_growth=1.01, top_k_ratio=0.01,
                                    restart_every=2000, verbose=False)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, "polish", t_global)

    # Final validation
    if best_seq is not None:
        fs = _eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = _norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = _eval_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
