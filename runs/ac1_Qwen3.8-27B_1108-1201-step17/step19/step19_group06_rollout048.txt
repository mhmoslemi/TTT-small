"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using soft-max gradient descent
with Nesterov acceleration, adaptive temperature, and multi-start strategy.
"""

import numpy as np
import time


def fast_conv(a, b):
    """Compute full linear convolution via FFT."""
    out_len = len(a) + len(b) - 1
    N = 1 << max(1, (out_len - 1).bit_length())
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(A * B, N)[:out_len]


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def score(a):
    """Compute the objective from a normalized sequence (sum=1)."""
    n = len(a)
    b = fast_conv(a, a)
    return 2.0 * n * b.max()


def compute_grad(a, p):
    """
    Compute gradient of soft-max of conv(a,a) w.r.t. a.
    
    a: array of length n (the sequence)
    p: array of length 2n-1 (softmax weights over convolution lags)
    
    Returns: array of length n (gradient)
    """
    n = len(a)
    # S1[i] = sum_{j=0}^{n-1} a[j] * p[i+j]  for i = 0, ..., n-1
    # This equals conv(flip(a), p)[n-1 + i]
    flip_a = a[::-1]
    cp1 = fast_conv(flip_a, p)  # length: n + (2n-1) - 1 = 3n-2
    S1 = cp1[n-1 : 2*n-1]  # n elements
    
    # S2[i] = sum_{j=0}^{n-1} a[j] * p[i-j]  (valid when 0 <= i-j < 2n-1) for i = 0, ..., n-1
    # This equals conv(a, p)[i]
    cp2 = fast_conv(a, p)  # length: n + (2n-1) - 1 = 3n-2
    S2 = cp2[0 : n]  # n elements
    
    return S1 + S2


def softmax_grad_descent(a0, time_limit, beta_init=3.0, lr_init=0.5, seed=42, verbose=True):
    """
    Soft-max gradient descent with Nesterov acceleration for minimizing
    max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = normalize(a0.astype(np.float64))
    best_a = a.copy()
    best_max_b = fast_conv(a, a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    beta = beta_init
    lr = lr_init
    momentum = 0.9
    v = np.zeros(n)  # momentum buffer
    stall = 0
    stall_threshold = 2000
    
    # Precompute FFT size for gradient
    out_len = 3 * n - 2
    N = 1 << max(1, (out_len - 1).bit_length())
    
    for t in range(1, 10**9):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution
        b = fast_conv(a, a)
        
        # Compute softmax weights
        b_shifted = b - b.max()
        expo = beta * b_shifted
        expo = np.clip(expo, -50, 50)
        p = np.exp(expo)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.ones_like(b) / len(b)
        else:
            p = p / p_sum
        
        # Compute gradient
        flip_a = a[::-1]
        A = np.fft.rfft(flip_a, N)
        P = np.fft.rfft(p, N)
        S1 = np.fft.irfft(A * P, N)[n-1 : 2*n-1]
        
        A_orig = np.fft.rfft(a, N)
        S2 = np.fft.irfft(A_orig * P, N)[0 : n]
        
        grad = S1 + S2
        
        # Nesterov momentum
        v = momentum * v + (1 - momentum) * grad
        v_norm = np.linalg.norm(v)
        if v_norm > 1e-30:
            v = v / v_norm
        
        # Adaptive learning rate
        eta = lr / (1.0 + 0.05 * np.sqrt(t))
        
        # Gradient descent step (additive)
        a_new = a - eta * v
        a_new = normalize(a_new)
        
        # Check improvement
        b_new = fast_conv(a_new, a_new)
        max_new = b_new.max()
        
        if max_new < best_max_b * (1 - 1e-10):
            best_max_b = max_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Adaptive beta (increase to focus on peaks)
        if t % 500 == 0:
            beta = min(beta * 1.05, 50.0)
        
        # Adaptive lr
        if t % 1000 == 0:
            lr *= 0.95
        
        # Restart if stuck
        if stall > stall_threshold:
            stall = 0
            stall_threshold = int(stall_threshold * 1.5)
            # Perturb best solution
            pert_type = int(rng.integers(3))
            if pert_type == 0:
                # Gaussian perturbation
                a_pert = best_a * (1.0 + rng.normal(0, 0.1, n))
            elif pert_type == 1:
                # Blend with random
                a_rand = rng.exponential(1.0, n)
                w = float(rng.uniform(0.1, 0.3))
                a_pert = (1 - w) * best_a + w * a_rand
            else:
                # Blend with uniform
                w = float(rng.uniform(0.05, 0.15))
                a_pert = (1 - w) * best_a + w * np.ones(n) / n
            
            a = normalize(a_pert)
            v = np.zeros(n)
            lr = max(lr * 0.8, 0.01)
            beta = max(beta - 5.0, beta_init)
            
            if verbose:
                J = 2.0 * n * best_max_b
                print(f"      Restart at t={t} ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    
    if verbose:
        J = 2.0 * n * best_max_b
        print(f"    SGD done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_max_b

        # Gradient
        flip_a = a[::-1]
        A = np.fft.rfft(flip_a, N)
        P = np.fft.rfft(p, N)
        S1 = np.fft.irfft(A * P, N)[n-1 : 2*n-1]
        A_orig = np.fft.rfft(a, N)
        S2 = np.fft.irfft(A_orig * P, N)[0 : n]
        grad = S1 + S2
        
        # Mirror descent (multiplicative) update
        eta = lr / (1.0 + 0.03 * np.sqrt(t))
        log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = normalize(a_new)
        
        b_new = fast_conv(a_new, a_new)
        max_new = b_new.max()
        
        if max_new < best_max_b * (1 - 1e-10):
            best_max_b = max_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        if t % 1000 == 0:
            beta = min(beta * 1.03, 40.0)
            lr *= 0.97
        
        if stall > stall_threshold:
            stall = 0
            a_pert = best_a * (1.0 + rng.normal(0, 0.05, n))
            a_pert += 0.05 * np.ones(n) / n
            a = normalize(a_pert)
            lr = max(lr * 0.8, 0.02)
            beta = max(beta - 3.0, beta_init)
            stall_threshold = int(stall_threshold * 1.3)
    
    if verbose:
        J = 2.0 * n * best_max_b
        print(f"    Mirror done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_max_b


def parametric_init(n, seed=0):
    """Generate diverse parametric initializations and return the best."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    
    candidates = []
    
    # Uniform
    candidates.append(np.ones(n))
    
    # Random exponential
    for _ in range(5):
        candidates.append(rng.exponential(1.0, n))
    
    # Power laws
    for p in [0.3, 0.5, 1.0, 2.0]:
        candidates.append((i_arr + 1) ** p)
        candidates.append((n - i_arr) ** p)
    
    # Gaussian
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sig in [0.1, 0.2, 0.3]:
            candidates.append(np.exp(-((x - c)**2) / (2 * sig**2)))
    
    # Bimodal
    for sig in [0.1, 0.15, 0.2]:
        candidates.append(np.exp(-((x - 0.25)**2) / (2*sig**2)) + np.exp(-((x - 0.75)**2) / (2*sig**2)))
    
    # Sine
    for f in [1, 2, 3, 5]:
        candidates.append(np.sin(np.pi * f * x)**2 + 0.1)
    
    # Exponential decay
    for r in [0.5, 1.0, 2.0, 5.0]:
        candidates.append(np.exp(-r * x))
        candidates.append(np.exp(r * (x - 1)))
    
    # Evaluate
    best_c = None
    best_v = float('inf')
    for c in candidates:
        c = normalize(np.array(c, dtype=np.float64))
        b = fast_conv(c, c)
        v = b.max()
        if v < best_v:
            best_v = v
            best_c = c.copy()
    
    return best_c


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f)
    return normalize(f_int)
