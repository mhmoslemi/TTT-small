def make_parabolic_bump(n):
    """Parabolic bump: zero at edges, peaked in middle."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.maximum((x - center) ** 2 - center ** 2, 0.0)
    a = -a  # flip to make it a bump
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_cosine_bump(n):
    """Cosine bump: zero at edges."""
    x = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * x / (n - 1 + 1e-10)) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_gaussian_bump(n, sigma_frac=0.3):
    """Gaussian bump centered in middle."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a / a.sum()


def make_asymmetric(n, asym=1.5):
    """Asymmetric shape: steeper on one side."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    left = ((center - x) / (n * 0.4)) ** 2
    right = ((x - center) / (n * 0.4 / asym)) ** 2
    a = np.exp(-0.5 * np.minimum(left, right * asym))
    a = np.maximum(a, 0.0)
    return a / a.sum()


def make_allpass_mag(n, rng):
    """Generate magnitude of allpass filter impulse response."""
    # Random allpass filter of order 4
    order = 4
    zeros = np.exp(1j * rng.uniform(0.05, 0.45, order))
    # Build transfer function H(z) = prod (z - z_i) / (z - 1/conj(z_i))
    # Impulse response via IFFT of H evaluated on unit circle
    freqs = np.linspace(0, 2 * np.pi, 4 * n, endpoint=False)
    H = np.ones(len(freqs), dtype=np.complex128)
    for z in zeros:
        num = np.exp(1j * freqs) - z
        den = np.exp(1j * freqs) - 1.0 / np.conj(z)
        H = H * num / den
    # Inverse FFT to get impulse response
    h = np.fft.ifft(H).real
    # Take magnitude and keep longest part
    m = min(n, len(h))
    a = np.abs(h[:m])
    if len(a) < n:
        a = np.pad(a, (0, n - len(a)))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    """Two separate Gaussian bumps."""
    x = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.2, 0.35)
    c2 = n * rng.uniform(0.65, 0.8)
    s1 = n * rng.uniform(0.08, 0.15)
    s2 = n * rng.uniform(0.08, 0.15)
    a = np.exp(-0.5 * ((x - c1) / s1) ** 2) + np.exp(-0.5 * ((x - c2) / s2) ** 2)
    return a / a.sum()


def make_random_smooth(n, rng):
    """Random smooth non-negative sequence via random Fourier features."""
    # Generate random low-frequency signal
    k_max = rng.integers(3, 15)
    freqs = rng.uniform(0, k_max, size=k_max)
    phases = rng.uniform(0, 2 * np.pi, size=k_max)
    amps = rng.uniform(0.5, 2.0, size=k_max)
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for f, p, amp in zip(freqs, phases, amps):
        a += amp * np.cos(2 * np.pi * f * x / n + p)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def compute_subgradient(a, conv, n_peaks=100):
    """Compute subgradient of max(conv) using top peaks with softmax weights."""
    n = len(a)
    top_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
    vals = conv[top_idx]
    max_v = vals.max()
    if max_v < 1e-30:
        return np.zeros(n)
    scale = max(max_v * 0.005, 1e-10)
    w = np.exp((vals - max_v) / scale)
    w = w / w.sum()

    g = np.zeros(n)
    for idx_i in range(len(top_idx)):
        ki = int(top_idx[idx_i])
        wi = w[idx_i]
        if wi < 1e-14:
            continue
        # d(a*a)[k]/da[j] = a[k-j] (if valid) + a[j-k] (if valid)
        # First term: a[k-j] valid when 0 <= k-j <= n-1
        j_lo1 = max(0, ki - n + 1)
        j_hi1 = min(n - 1, ki)
        if j_lo1 <= j_hi1:
            jr = np.arange(j_lo1, j_hi1 + 1)
            g[jr] += wi * a[ki - jr]
        # Second term: a[j-k] valid when 0 <= j-k <= n-1
        j_lo2 = max(0, ki)
        j_hi2 = min(n - 1, ki + n - 1)
        if j_lo2 <= j_hi2:
            jr = np.arange(j_lo2, j_hi2 + 1)
            g[jr] += wi * a[jr - ki]
    return g


def subgradient_optimize(a0, time_limit, n_peaks=100, momentum=0.9, verbose=False):
    """Subgradient descent with momentum and Polyak step size on the simplex."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    # Lower bound for Polyak: 1/(2n-1)
    lb = 1.0 / (2 * n - 1)

    v = np.zeros(n)  # momentum
    t = 0
    stall = 0
    restart_count = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_autocorr(a)
        M = conv.max()
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Restart if stalled too long
        if stall > 800:
            restart_count += 1
            strength = min(0.3, 0.05 + 0.05 * restart_count)
            pert = rng.normal(0, 1, n) * best_a * strength
            a_r = best_a + pert
            a_r = np.maximum(a_r, 0.0)
            a_r = a_r / a_r.sum()
            a = a_r
            v = np.zeros(n)
            stall = 0

        g = compute_subgradient(a, conv, n_peaks=n_peaks)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Polyak step size
        gap = M - lb
        if gap > 1e-15:
            eta = 2.0 * gap / (g_norm * g_norm + 1e-30)
        else:
            eta = 1.0 / (g_norm + 1e-15)
        eta = min(eta, 5.0)  # cap step size

        # Momentum update
        v = momentum * v - eta * g
        a_new = a + v

        # Project onto simplex (a >= 0, sum a = 1)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        a = a_new

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD t={t} ({elapsed:.0f}s): M={best_M:.8f}, J={2*n*best_M:.6f}")

    return best_a, best_M


def mirror_descent_optimize(a0, time_limit, n_peaks=80, lr0=3.0, restart_every=2000,
                            rng_seed=42, verbose=False):
    """Mirror descent with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    restart_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-20)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        conv = fast_autocorr(y)
        g = compute_subgradient(y, conv, n_peaks=n_peaks)

        eta = lr0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_M - 1e-16:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            restart_count += 1
            strength = min(0.15, 0.03 + 0.02 * min(restart_count, 5))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-15)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

        if verbose and t % 5000 == 0:
            print(f"    MD t={t} ({time.monotonic()-t0:.0f}s): J={2*n*best_M:.6f}")

    return best_a, best_M


def coordinate_descent_v2(a0, time_limit, verbose=False, rng_seed=42):
    """Targeted coordinate descent: move mass from peak contributors to low-cost positions."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        conv = fast_autocorr(a)
        k_max = int(np.argmax(conv))

        # Find source: index with max contribution to peak at k_max
        j_lo = max(0, k_max - n + 1)
        j_hi = min(n - 1, k_max)
        if j_lo > j_hi:
            stall += 1
            continue
        idx = np.arange(j_lo, j_hi + 1)
        contrib = a[idx] * a[k_max - idx]
        src = idx[np.argmax(contrib)]

        if a[src] < 1e-12:
            stall += 1
            continue

        # Find destination: minimize increase in top peaks
        top_ks = np.argpartition(conv, -20)[-20:]
        costs = np.zeros(n)
        for k2 in top_ks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            if lo2 <= hi2:
                valid = np.arange(lo2, hi2 + 1)
                costs[valid] += a[k2i - valid]

        # Also penalize moving to positions that already have high value
        costs += 0.5 * a
        sorted_idx = np.argsort(costs)
        dst = sorted_idx[0]
        if dst == src:
            dst = sorted_idx[1] if len(sorted_idx) > 1 else int(rng.integers(n))

        moved = False
        for frac in [0.02, 0.05, 0.1, 0.15, 0.25, 0.4]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_t = fast_autocorr(a_trial).max()
            if M_t < best_M - 1e-16:
                a = a_trial
                best_M = M_t
                best_a = a.copy()
                moved = True
                stall = 0
                break

        if not moved:
            stall += 1
            # Random perturbation to escape local minimum
            if stall % 10 == 0:
                n_pert = rng.integers(max(1, n // 10), max(2, n // 5))
                pert_idx = rng.choice(n, size=n_pert, replace=False)
                a[pert_idx] *= rng.uniform(0.8, 1.2, n_pert)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_M


def spectral_equalize(a0, time_limit, verbose=False):
    """Equalize the power spectrum to reduce time-domain convolution peaks."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_mean = P.mean()
        if P_max < 1e-30:
            break

        # Flatten: push power from peaks toward mean
        alpha = 0.2 + 0.6 * min(1.0, it / 300.0)
        P_new = P + alpha * (P_mean - P)
        P_new = np.maximum(P_new, 1e-30)

        # Update magnitude while preserving phase
        mag_old = np.sqrt(P)
        mag_new = np.sqrt(P_new)
        ratio = mag_new / np.maximum(mag_old, 1e-30)
        ratio = np.clip(ratio, 0.5, 2.0)
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_M - 1e-16:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend to stay in good region
            a = 0.8 * a + 0.2 * a_new
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M_curr = fast_autocorr(a).max()
            if M_curr < best_M:
                best_M = M_curr
                best_a = a.copy()

        if verbose and it % 200 == 0:
            print(f"    Spec it={it} ({time.monotonic()-t0:.0f}s): J={2*n*best_M:.6f}")

    return best_a, best_M


def peak_deflation(a0, time_limit, verbose=False, rng_seed=123):
    """Directly deflate the maximum peak by redistributing mass."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    while time.monotonic() - t0 < time_limit and stall < 500:
        conv = fast_autocorr(a)
        k_peak = int(np.argmax(conv))
        peak_val = conv[k_peak]

        # Identify the top contributors to this peak
        j_lo = max(0, k_peak - n + 1)
        j_hi = min(n - 1, k_peak)
        if j_lo > j_hi:
            stall += 1
            continue
        idx = np.arange(j_lo, j_hi + 1)
        contrib = a[idx] * a[k_peak - idx]
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break

        # Move mass from top contributors to positions far from all top peaks
        top_peaks = np.argpartition(conv, -5)[-5:]

        # Find safe destination (low contribution to all top peaks)
        costs = np.zeros(n)
        for kp in top_peaks:
            kpi = int(kp)
            lo = max(0, kpi - n + 1)
            hi = min(n - 1, kpi)
            if lo <= hi:
                v = np.arange(lo, hi + 1)
                costs[v] += a[kpi - v]
        costs += 0.3 * a  # prefer low-value positions

        # Source: top contributor to the peak
        top_srcs = idx[np.argsort(contrib)[-3:]]
        # Destination: lowest cost position
        dst = int(np.argmin(costs))

        src = int(top_srcs[0])
        if src == dst:
            dst = int(np.argmin(np.where(costs == np.inf, 1e10, costs) ))
            if src == dst:
                dst = (dst + 1) % n

        # Try moving
        moved = False
        for frac in [0.01, 0.03, 0.08, 0.15, 0.3]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] = max(0, a_trial[src] - amt)
            a_trial[dst] += amt
            a_trial = a_trial / a_trial.sum()
            M_t = fast_autocorr(a_trial).max()
            if M_t < best_M - 1e-16:
                a = a_trial
                best_M = M_t
                best_a = a.copy()
                moved = True
                stall = 0
                break

        if not moved:
            stall += 1
            if stall % 20 == 0:
                # Larger random perturbation
                n_pert = rng.integers(5, max(6, n // 20))
                pidx = rng.choice(n, size=n_pert, replace=False)
                a[pidx] *= rng.uniform(0.5, 1.5, n_pert)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_M


def interpolate_seq(a, new_n):
    """Interpolate to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = np.asarray(a, dtype=np.float64)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def propose_candidate(budget_s=1800.0):
    """
    Multi-strategy search: diverse initializations + subgradient + mirror descent +
    coordinate descent + peak deflation + spectral equalization.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = safe_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = safe_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    rng = np.random.default_rng(2024)

    # Phase 1: Generate and quickly evaluate diverse initializations
    print("=== Phase 1: Initialization ===")
    n_values = [3000, 4000, 5000, 6200, 8000, 10000]
    initializations = []

    for n in n_values:
        # Different shapes
        initializations.append((make_ramp(n, 1.0), n, f"ramp_n{n}"))
        initializations.append((make_parabolic_bump(n), n, f"para_n{n}"))
        initializations.append((make_cosine_bump(n), n, f"cos_n{n}"))
        initializations.append((make_gaussian_bump(n, 0.25), n, f"gauss_n{n}"))
        initializations.append((make_asymmetric(n, 1.5), n, f"asym_n{n}"))
        initializations.append((make_allpass_mag(n, rng), n, f"allpass_n{n}"))
        initializations.append((make_two_bump(n, rng), n, f"2bump_n{n}"))
        initializations.append((make_random_smooth(n, rng), n, f"rnd_n{n}"))

    # Add parent if available
    if parent is not None:
        initializations.insert(0, (parent / parent.sum(), len(parent), "parent"))

    # Quick evaluation of all initializations
    scored_inits = []
    for a, n, label in initializations:
        sc = safe_eval(a)
        if np.isfinite(sc):
            scored_inits.append((sc, a, label))
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()

    scored_inits.sort(key=lambda x: x[0])
    print(f"  Top 5 inits: {[f'{l}:{s:.4f}' for s, _, l in scored_inits[:5]]}")

    # Phase 2: Subgradient optimization on top candidates
    print("\n=== Phase 2: Subgradient Optimization ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_cands = min(8, len(scored_inits))
    t_per = max(30.0, remaining * 0.35 / n_cands)

    for i in range(n_cands):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget * 0.45:
            break
        t_avail = min(t_per, time_budget * 0.45 - elapsed)
        if t_avail < 15:
            break
        sc0, a0, label = scored_inits[i]
        print(f"  SGD [{label}] (t={t_avail:.0f}s, init={sc0:.4f})...")
        a_out, _ = subgradient_optimize(a0, t_avail, n_peaks=100, momentum=0.85, verbose=False)
        try_update(a_out, f"sgd_{label}")

    # Phase 3: Mirror descent on best candidates
    print("\n=== Phase 3: Mirror Descent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and best_seq is not None:
        # Run MD on best and a few others
        md_candidates = []
        if best_seq is not None:
            md_candidates.append((best_seq / best_seq.sum(), "best"))
        for i in range(min(3, len(scored_inits))):
            md_candidates.append((scored_inits[i][1], scored_inits[i][2]))

        t_per_md = max(20.0, remaining * 0.20 / len(md_candidates))
        for a0, label in md_candidates:
            elapsed = time.monotonic() - t_global
            if elapsed > time_budget * 0.70:
                break
            t_avail = min(t_per_md, time_budget * 0.70 - elapsed)
            if t_avail < 10:
                break
            print(f"  MD [{label}] (t={t_avail:.0f}s)...")
            a_out, _ = mirror_descent_optimize(a0, t_avail, n_peaks=80, lr0=3.0,
                                               restart_every=2000, rng_seed=42 + len(label),
                                               verbose=False)
            try_update(a_out, f"md_{label}")

    # Phase 4: Peak deflation
    print("\n=== Phase 4: Peak Deflation ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40 and best_seq is not None:
        t4 = min(remaining * 0.15, 200.0)
        a_in = best_seq / best_seq.sum()
        print(f"  PeakDeflation (t={t4:.0f}s)...")
        a_out, _ = peak_deflation(a_in, t4, verbose=False, rng_seed=123)
        try_update(a_out, "peakdef")

        # Second pass with different seed
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t4b = min(remaining * 0.10, 100.0)
            a_out, _ = peak_deflation(a_in, t4b, verbose=False, rng_seed=999)
            try_update(a_out, "peakdef2")

    # Phase 5: Coordinate descent
    print("\n=== Phase 5: Coordinate Descent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40 and best_seq is not None:
        t5 = min(remaining * 0.12, 150.0)
        a_in = best_seq / best_seq.sum()
        print(f"  CoordDesc (t={t5:.0f}s)...")
        a_out, _ = coordinate_descent_v2(a_in, t5, verbose=False, rng_seed=42)
        try_update(a_out, "coord")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t5b = min(remaining * 0.08, 80.0)
            a_out, _ = coordinate_descent_v2(a_in, t5b, verbose=False, rng_seed=777)
            try_update(a_out, "coord2")

    # Phase 6: Spectral equalization
    print("\n=== Phase 6: Spectral Equalization ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40 and best_seq is not None:
        t6 = min(remaining * 0.10, 120.0)
        a_in = best_seq / best_seq.sum()
        print(f"  Spectral (t={t6:.0f}s)...")
        a_out, _ = spectral_equalize(a_in, t6, verbose=False)
        try_update(a_out, "spectral")

    # Phase 7: Final intensive optimization on best
    print("\n=== Phase 7: Final Polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        n = len(a_in)
