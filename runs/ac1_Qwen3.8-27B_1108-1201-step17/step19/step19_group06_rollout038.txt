```python
import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
```

```python
def adam_log_optimize(a0, time_limit, lr=0.01, beta=10.0, top_k=50, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    log_a = np.log(np.maximum(a, 1e-30))
    
    m = np.zeros(n)  # ADAM first moment
    v = np.zeros(n)  # ADAM second moment
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        a = norm_sum1(np.exp(log_a))
        
        ac = fast_autocorr(a)
        
        # Softmax gradient
        k_alarm = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_alarm]
        top_vals = ac[top_idx]
        max_v = top_vals.max()
        expo = beta * (top_vals - max_v)
        expo = np.clip(expo, -50, 50)
        weights = np.exp(expo)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones(k_alarm) / k_alarm
        else:
            weights = weights / w_sum
        
        # Gradient in original space
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k_int = int(k)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo >= i_hi:
                continue
            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            valid = (j_idx >= 0) & (j_idx < n)
            grad[idx[valid]] += 2.0 * w * a[j_idx[valid]]
        
        # Gradient in log-space: dL/d(log_a[i]) = a[i] * dL/da[i]
        log_grad = a * grad
        
        # ADAM update
        m = b1 * m + (1 - b1) * log_grad
        v = b2 * v + (1 - b2) * log_grad**2
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        
        # Learning rate with cosine annealing
        progress = min(1.0, t / (time_limit * 100))  # rough estimate
        lr_t = lr * 0.5 * (1 + np.cos(np.pi * progress))
        
        log_a = log_a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Check improvement
        a_new = norm_sum1(np.exp(log_a))
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a_new.copy()
    
    return best_a, best_m
```

```python
def compute_softmax_grad(a, ac, top_k, beta):
    """Compute gradient using softmax weighting over top autocorrelation lags."""
    n = len(a)
    k_alarm = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_alarm]
    top_vals = ac[top_idx]
    max_v = top_vals.max()
    expo = beta * (top_vals - max_v)
    expo = np.clip(expo, -50, 50)
    weights = np.exp(expo)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_alarm) / k_alarm
    else:
        weights = weights / w_sum
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n, k_int + 1)
        if i_lo >= i_hi:
            continue
        idx = np.arange(i_lo, i_hi)
        j_idx = k_int - idx
        valid = (j_idx >= 0) & (j_idx < n)
        grad[idx[valid]] += 2.0 * w * a[j_idx[valid]]
    return grad


def compute_deficit_grad(a, ac, target, top_k):
    """Compute gradient penalizing autocorrelation values above target level."""
    n = len(a)
    excess = np.maximum(ac - target, 0.0)
    k_alarm = min(top_k, len(ac))
    top_idx = np.argsort(excess)[::-1][:k_alarm]
    top_vals = excess[top_idx]
    if top_vals.max() < 1e-15:
        return np.zeros(n)
    expo = 5.0 * (top_vals - top_vals.max())
    expo = np.clip(expo, -50, 50)
    weights = np.exp(expo)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_alarm) / k_alarm
    else:
        weights = weights / w_sum
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        i_lo = max(0, k_int - n + 1)

def adam_optimize(a0, time_limit, lr=0.02, beta_init=8.0, top_k=50,
                  beta_end=3.0, dual_weight=0.3, rng_seed=42, verbose=False):
    """ADAM gradient descent in log-space with adaptive parameters."""
    n = len(a0)
    a = norm_sum1(a0)
    log_a = np.log(np.maximum(a, 1e-30))
    log_a -= log_a.max()  # numerical stability
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    total_steps_est = int(time_limit * 80)  # rough estimate
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    stall_threshold = max(500, total_steps_est // 20)
    rng = np.random.default_rng(rng_seed)
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        a = norm_sum1(np.exp(log_a))
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Adaptive beta: decrease over time for sharper focus
        progress = min(1.0, t / total_steps_est)
        beta = beta_init + (beta_end - beta_init) * progress
        top_k_curr = min(top_k + int(20 * progress), 200)
        
        # Combined gradient: softmax + deficit
        target = 1.0 / (2 * n - 1)  # ideal flat level (sum=1)
        grad_smooth = compute_softmax_grad(a, ac, top_k_curr, beta)
        grad_deficit = compute_deficit_grad(a, ac, target, min(top_k_curr, 80))
        
        dw = dual_weight * progress  # increase deficit weight over time
        grad = (1.0 - dw) * grad_smooth + dw * grad_deficit
        
        # Log-space gradient
        log_grad = a * grad
        
        # ADAM update
        m = b1 * m + (1.0 - b1) * log_grad
        v = b2 * v + (1.0 - b2) * log_grad ** 2
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        
        # Cosine annealed learning rate
        lr_t = lr * 0.5 * (1.0 + np.cos(np.pi * min(1.0, t / total_steps_est)))
        
        log_a = log_a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic reshaping to avoid getting stuck
        if t % 2000 == 0:
            a_new = norm_sum1(np.exp(log_a))
            a_new = 0.95 * a_new + 0.05 * norm_sum1(best_a + rng.normal(0, 0.02, n) * best_a)
            log_a = np.log(np.maximum(a_new, 1e-30))
            log_a -= log_a.max()
        
        # Check improvement
        a_eval = norm_sum1(np.exp(log_a))
        m_new = fast_autocorr(a_eval).max()
        if m_new < best_m - 1e-10:
            best_m = m_new
            best_a = a_eval.copy()
            stall = 0
        else:
            stall += 1
            if stall > stall_threshold:
                # Aggressive restart
                a_new = norm_sum1(best_a * (1.0 + rng.normal(0, 0.1, n)))
                # Preserve overall shape
                x = np.linspace(0, 1, n)
                shape = np.sin(np.pi * x) ** 0.5
                a_new = norm_sum1(0.8 * a_new + 0.2 * shape)
                log_a = np.log(np.maximum(a_new, 1e-30))
                log_a -= log_a.max()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                stall = 0
                lr *= 0.9
                if verbose:
                    print(f"      ADAM restart at t={t0 and int(time.monotonic()-t0)}s, lr={lr:.4f}, best_m={2*n*best_m:.6f}")
    
    return best_a, 2.0 * n * best_m


def spectral_shape(a0, time_limit, lr=0.1, verbose=False):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    rng = np.random.default_rng(123)
    stall = 0
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_mean = P.mean()
        
        if P_max < 1e-30:
            break
        
        # Flatten spectrum: push energy from peaks to valleys
        P_ratio = P / np.maximum(P_mean, 1e-30)
        # Reduce peaks, boost valleys
        A_new = A.copy()
        mask_peak = P > 1.5 * P_mean
        mask_valley = P < 0.5 * P_mean
        A_new[mask_peak] *= (1.0 - lr * 0.5 * (P[mask_peak] / P_max - 1.0))
        A_new[mask_valley] *= (1.0 + lr * 0.3)
        
        # Keep phase
        phase = np.angle(A)
        mag_new = np.abs(A_new)
        A_combined = mag_new * np.exp(1j * phase)
        
        a_new = np.fft.irfft(A_combined, fft_size)[:n]
        a_new = norm_sum1(a_new)
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-11:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 500:
                a = norm_sum1(best_a + rng.normal(0, 0.01, n) * best_a)
                stall = 0
                lr *= 0.85
    
    if verbose:
        print(f"    Spectral done: J={2*n*best_m:.6f}, time={time.monotonic()-t0:.0f}s")
    return best_a, 2.0 * n * best_m


def coord_descent(a0, time_limit, rng_seed=77, verbose=False):
    """Coordinate descent: move mass from high-impact to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)

        M_curr = ac.max()
        if M_curr < best_m - 1e-12:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break
        
        # Focus on top lags
        n_top = min(20, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:n_top]
        improved = False

            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k_pos)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo >= i_hi:
                continue
            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            valid = (j_idx >= 0) & (j_idx < n)
            idx = idx[valid]
            j_idx = j_idx[valid]
            if len(idx) < 2:
                continue
            contrib = a[idx] * a[j_idx]
            
            # Find the highest contributing pair
            max_pi = int(np.argmax(contrib))
            src = int(idx[max_pi])
            src_val = a[src]
            if src_val < 1e-14:
                continue
            
            # Find destination with minimal impact
            impact = np.zeros(n)
            for k2 in top_lags[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                if s2 < e2:
                    ii = np.arange(s2, e2)
                    jj = k2i - ii
                    valid2 = (jj >= 0) & (jj < n)
                    impact[ii[valid2]] += a[jj[valid2]]
            impact[src] = np.inf
            dst = int(np.argmin(impact))
            
            if a[dst] > 0.9 * src_val:

            for frac in [0.05, 0.1, 0.2, 0.3]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] = src_val - amt
                a_trial[dst] += amt
                a_trial = norm_sum1(a_trial)
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-13:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved and stall > 500:
            # Random escape
            n_moves = int(rng.integers(3, 15))
            for _ in range(n_moves):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    amt = a[i] * float(rng.uniform(0.01, 0.05))
                    a[i] -= amt
                    a[j] += amt
            a = norm_sum1(a)
            m_new = fast_autocorr(a).max()
            if m_new < best_m:
                best_m = m_new
                best_a = a.copy()
                stall = 0
    
    if verbose:
        print(f"    Coord descent done: J={2*n*best_m:.6f}, time={time.monotonic()-t0:.0f}s")
    return best_a, 2.0 * n * best_m


def make_raised_cosine(n, alpha=1.0):

def make_gaussian(n, center=0.5, sigma=0.25):

def make_multi_gaussian(n, rng, num_bumps=3):

def make_beta_dist(n, alpha=2.0, beta=2.0):

    x = np.linspace(0.01, 0.99, n)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss

        t5 = min(remaining * 0.4, 120.0)
        if t5 > 20:
            a_out, _ = adam_optimize(a_interp, t5, lr=0.02, beta_init=10.0,
                                       top_k=50, beta_end=5.0, dual_weight=0.3,
                                       rng_seed=new_n % 10000, verbose=False)
            check_update(a_out, f"adam_len{new_n}")
        
        # Quick coord descent
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t5b = min(remaining * 0.3, 80.0)
            if t5b > 10:
                a_out, _ = coord_descent(best_seq, t5b, rng_seed=new_n % 1000, verbose=False)
                check_update(a_out, f"coord_len{new_n}")
    
    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_final}, budget={remaining:.0f}s) ===")
    
    t6 = min(remaining * 0.5, 200.0)
    if t6 > 15:
        a_out, _ = adam_optimize(best_seq, t6, lr=0.008, beta_init=20.0,
                                   top_k=100, beta_end=8.0, dual_weight=0.4,
                                   rng_seed=999, verbose=False)
        check_update(a_out, f"final_adam_n{n_final}")
    
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t7 = min(remaining * 0.5, 80.0)
        if t7 > 10:
            a_out, _ = coord_descent(best_seq, t7, rng_seed=777, verbose=False)
            check_update(a_out, f"final_coord_n{len(best_seq)}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = make_raised_cosine(n_fb, 1.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
```
