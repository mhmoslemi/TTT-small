def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_at_peak(a, ac, top_k=80, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        if j_start >= j_end:
            continue
        idx = np.arange(j_start, j_end)
        grad[idx] += 2.0 * wi * a[ki - idx]
    return grad


def nesterov_mirror_descent(a0, time_limit, top_k=60, beta=5.0, c0=5.0,
                            momentum=0.9, restart_every=5000, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    mu = momentum

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_gradient_at_peak(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every // 2:
            cur_beta = max(cur_beta * 0.92, 2.0)
            cur_top_k = min(cur_top_k + 5, 200)

        if stall_count > restart_every:
            stall_count = 0
            noise_scale = 0.02 + 0.03 * np.log1p(stall_count / 100)
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 200)
            cur_beta = max(cur_beta - 1.0, 2.5)

    return best_a, best_m


def lp_frank_wolfe_adaptive(a0, time_limit, solver, top_K_init=200,
                            active_tolerance=0.05, verbose=False):
    """LP-FW with adaptive active set: only keep constraints near the peak."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    iter_count = 0

    K = top_K_init
    active_set = None
    max_K = min(2 * n - 1, 5000)

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Update active set: lags where ac[k] >= M_curr * (1 - active_tolerance)
        threshold = M_curr * (1.0 - active_tolerance)
        active_candidates = np.where(ac >= threshold)[0]
        # Also include top K by value
        top_K_idx = np.argsort(ac)[::-1][:K]
        combined = np.unique(np.concatenate([active_candidates, top_K_idx]))
        combined = combined[:min(max_K, len(combined))]

        K_sel = len(combined)
        if K_sel == 0:
            break

        # Build constraint matrix
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(combined):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 20, max_K)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Line search: try multiple step sizes
        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj

        alphas = [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0),
                  0.5, 0.3, 0.2, 0.8, 1.0]
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1

        # Adaptive K: grow slowly, shrink if too large
        if iter_count % 5 == 0:
            K = min(K + 15, max_K)

        # Gradually tighten active tolerance as we progress
        if iter_count > 50:
            active_tolerance = max(0.02, active_tolerance * 0.99)

        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

    return best_a, best_m


def peak_shaving(a0, time_limit, verbose=False, rng_seed=100):
    """Directly target the top peaks by moving mass away from contributing positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:15]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            # Find the position with maximum contribution
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find destination: position NOT contributing to any top peak
            in_peak = np.zeros(n, dtype=bool)
            for kk in top_k:
                kik = int(kk)
                ss = max(0, kik - n + 1)
                ee = min(n - 1, kik)
                in_peak[ss:ee+1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            # Pick destination with lowest current value among candidates
            best_dst = candidates[np.argmin(a[candidates])]

            for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation when stuck
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

    return best_a, best_m


def smooth_gradient_optimize(a0, time_limit, verbose=False, tau_init=20.0,

        progress = min(1.0, elapsed / min(time_limit, 400.0))
        tau = tau_init * (tau_final / tau_init) ** progress

        m_new = fast_autocorr(a_new).max()

        else:
            stall += 1
            if stall > 80:
                eta *= 0.7
            if stall > 150:
                pert = rng.normal(0, 0.005, n) * a

    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    fft_size = 1 << (2 * n - 2).bit_length()

    for it in range(n_max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: flat magnitude
        mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
        alpha = 0.05 + 0.3 * min(1.0, it / 500)
        new_mag = (1 - alpha) * mag + alpha * mag_mean

        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def make_cosine(n):
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = a ** 2
    return norm_sum1(a)


def make_cosine_mod(n, alpha=0.5):
    x = np.arange(n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * x / n)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_v_shape(n, sharpness=2.0):
    """V-shape: high at ends, low in middle (like U but sharper)"""
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * np.abs(x) ** 1.5
    return norm_sum1(a)


def make_quartic(n, sharpness=2.0):
    """Quartic U-shape"""
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 4
    return norm_sum1(a)


def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:10]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-contribution positions
            in_peak = np.zeros(n, dtype=bool)
            for kk in top_k:
                kik = int(kk)
                ss = max(0, kik - n + 1)
                ee = min(n - 1, kik)
                in_peak[ss:ee+1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            # Try multiple destinations
            cand_sorted = candidates[np.argsort(a[candidates])[:5]]
            for dst in cand_sorted:
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 6000
        best_seq = make_u_shape(n_base, sharpness=2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Quick LP-FW on parent ===
    print(f"\n=== Phase 1: LP-FW quick (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 200.0)
    if t1 > 30 and n0 <= 7000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_adaptive(a_in, t1, solver, top_K_init=200,
                                            active_tolerance=0.08, verbose=True)
        try_update(a_out, "lp1")

    # === Phase 2: Nesterov MD with diverse inits ===
    print(f"\n=== Phase 2: Nesterov MD multi-init (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)

    if t2 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_u_shape(n0, sharpness=1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, sharpness=3.0))
        labels.append("u3.0")
        inits.append(make_v_shape(n0, sharpness=2.0))
        labels.append("v2.0")
        inits.append(make_quartic(n0, sharpness=2.0))
        labels.append("quartic")
        inits.append(make_bimodal(n0, width=0.2))
        labels.append("bimodal")
        inits.append(make_cosine_mod(n0, alpha=0.4))
        labels.append("cosmod")
        inits.append(spectral_init(n0, n_modes=30, seed=42))
        labels.append("spectral")

        # Blend of parent and U-shape
        if parent is not None:
            blend = 0.5 * norm_sum1(parent) + 0.5 * make_u_shape(n0, 2.5)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Use current best as init
        inits.append(norm_sum1(best_seq))
        labels.append("best_sofar")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 50)], 4))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(15.0, t2 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  MD init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = nesterov_mirror_descent(init, t_avail, top_k=60, beta=5.5,
                                                c0=5.0, momentum=0.9,
                                                restart_every=4000, rng_seed=42 + idx)
            try_update(a_out, f"md_{lbl}")

    # === Phase 3: LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW main (n={n_best}) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40 and n_best <= 7000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_adaptive(a_in, t3, solver, top_K_init=300,
                                            active_tolerance=0.05, verbose=True)
        try_update(a_out, "lp_main")

    # === Phase 4: Smoothed gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Smooth gradient (n={n_best}) ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_gradient_optimize(a_in, t4, verbose=False,
                                             tau_init=15.0, tau_final=150.0,
                                             eta_init=0.8, rng_seed=100)
        try_update(a_out, "smooth")

    # === Phase 5: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW r2 (n={n_best}) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 40 and n_best <= 7000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_adaptive(a_in, t5, solver, top_K_init=400,
                                            active_tolerance=0.03, verbose=True)
        try_update(a_out, "lp_r2")

    # === Phase 6: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Peak shaving (n={n_best}) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving(a_in, t6, verbose=False, rng_seed=200)
        try_update(a_out, "shave")

    # === Phase 7: POCS spectral ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: POCS (n={n_best}) ===")
    t7 = min(remaining * 0.08, 100.0)
    if t7 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_pocs(a_in, t7, verbose=False, rng_seed=50)
        try_update(a_out, "pocs")

    # === Phase 8: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Length exploration ===")

    trial_lengths = []
    for delta in [300, -300, 600, -600, 1000, -1000, 1500, -1500, 2000, -2000]:
        new_n = n_best + delta
        if 1000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t8a = min(remaining * 0.35, 120.0)
        if t8a > 20:
            a_out, _ = nesterov_mirror_descent(a_long, t8a, top_k=60, beta=5.5,
                                                c0=5.0, momentum=0.9,
                                                restart_every=4000, rng_seed=new_n % 1000)
            try_update(a_out, f"md_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and new_n <= 7000:
            t8b = min(remaining * 0.4, 100.0)
            if t8b > 20:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                a_out, _ = lp_frank_wolfe_adaptive(a_in, t8b, solver,
                                                    top_K_init=200,
                                                    active_tolerance=0.05, verbose=False)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 9: Final LP-FW + polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final optimization (n={n_best}) ===")
    if remaining > 30:
        t9 = min(remaining * 0.35, 120.0)
        if t9 > 20 and n_best <= 7000:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_adaptive(a_in, t9, solver, top_K_init=500,
                                                active_tolerance=0.02, verbose=True)
            try_update(a_out, "lp_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t9b = min(remaining * 0.3, 80.0)
            if t9b > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = nesterov_mirror_descent(a_in, t9b, top_k=80, beta=7.0,
                                                    c0=6.0, momentum=0.92,
                                                    restart_every=3000, rng_seed=999)
                try_update(a_out, "md_final")

    # === Phase 10: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 10: Final polish (n={n_best}) ===")
        t10 = min(remaining * 0.6, 60.0)
        if t10 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t10, verbose=False, rng_seed=789)
            try_update(a_out, "coord_final")

        # One more peak shaving pass
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t11 = min(remaining * 0.5, 30.0)
            if t11 > 5:
                a_in = norm_sum1(best_seq)
                a_out, _ = peak_shaving(a_in, t11, verbose=False, rng_seed=333)
                try_update(a_out, "shave_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = make_u_shape(n_fb, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
