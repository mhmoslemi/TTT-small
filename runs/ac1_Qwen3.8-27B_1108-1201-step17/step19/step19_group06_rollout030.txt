Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact FFT-based soft-max gradient (cross-correlation trick)
2. Mirror descent with adaptive temperature and Nesterov momentum
3. L-BFGS-B refinement with soft-max smoothing
4. Multi-start with diverse parametric initializations
5. Length exploration via interpolation
6. Coordinate descent polish

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score_normalized(a):
    """Compute 2*n*max_b for a sequence with sum=1."""
    n = len(a)
    b = fast_conv(a)
    return 2.0 * n * b.max()


def sofmax_weights(b, T, top_k=None):
    """Compute soft-max weights over autocorrelation values."""
    if top_k is not None and top_k < len(b):
        idx = np.argpartition(b, -(len(b) - top_k))  # Hmm, this is wrong
        # Get top_k indices
        idx = np.argsort(b)[-top_k:]
        w = np.zeros_like(b)
        b_top = b[idx]
        w[idx] = np.exp(T * b_top - T * np.max(b_top))
        w /= w.sum()
    else:
        w = np.exp(T * b - T * np.max(b))
        w /= w.sum()
    return w


def compute_gradient(a, T, top_k=None):
    """
    Compute gradient of 2*n*logsumexp(T*b)/T with respect to a.
    Uses FFT cross-correlation trick for O(n log n) computation.
    
    Returns (gradient, max_b, weights).
    """
    n = len(a)
    b = fast_conv(a)
    w = sofmax_weights(b, T, top_k)
    
    # Gradient[m] = 2 * sum_j w[m+j] * a[j] = 2 * cross_correlation(w, a)[m]
    # Need FFT size >= len(w) + n - 1 = (2n-1) + n - 1 = 3n-2
    N = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    
    grad = 2.0 * R
    return grad, b.max(), w


def mirror_descent_sofmax(a_init, time_limit, T_init=5.0, T_max=200.0,
                           lr_init=0.1, top_k=None, verbose=True, seed=42):
    """
    Mirror descent with soft-max gradient and adaptive temperature.
    Multiplicative updates maintain non-negativity.
    """
    n = len(a_init)
    a = normalize(a_init)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    T = T_init
    lr = lr_init
    stall = 0
    momentum = 0.8
    a_prev = a.copy()
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        # Nesterov extrapolation in multiplicative space
        ratio = a / np.maximum(a_prev, 1e-30)
        ratio = np.clip(ratio, 1e-2, 1e2)
        a_nest = a * np.power(ratio, momentum)
        a_nest = normalize(a_nest)
        
        # Compute gradient
        grad, max_b, w = compute_gradient(a_nest, T, top_k)
        
        # Adaptive learning rate
        eta = lr / (1.0 + 0.1 * np.sqrt(it))
        
        # Multiplicative update
        log_a = np.log(np.maximum(a_nest, 1e-30)) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        a_prev = a.copy()
        a = a_new
        it += 1
        
        # Check improvement
        sc = score_normalized(a)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive temperature increase
        if it % 2000 == 0:
            T = min(T * 1.3, T_max)
        
        # Verbose output
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD iter={it} T={T:.1f} score={best_score:.6f} time={elapsed:.0f}s")
        
        # Restart if stuck
        if stall > 8000:
            stall = 0
            T = max(T_init, T * 0.5)  # Lower temperature for exploration

            # Perturb best

            a_prev = a.copy()
            if verbose:
                print(f"      MD restart T={T:.1f} lr={lr:.4f}")

def lbfgs_refine(a_init, time_limit, T=50.0, max_iter=5000, verbose=True):
    """
    Refine using L-BFGS-B with soft-max smoothed objective.
    """
    n = len(a_init)
    a0 = normalize(a_init)
    t0 = time.monotonic()
    
    def objective(x):
        if time.monotonic() - t0 > time_limit:
            return np.inf
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return np.inf
        x = x / s
        b = fast_conv(x)
        # Soft-max: (1/T) * log(sum(exp(T*b)))
        Tb = T * b - T * b.max()
        log_sum_exp = Tb.max() + np.log(np.sum(np.exp(Tb)))
        return 2.0 * n * (log_sum_exp / T)
    
    def gradient(x):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return np.zeros(n)
        x = x / s
        grad, _, _ = compute_gradient(x, T, top_k=None)
        return grad
    
    # Use L-BFGS-B with bounds [0, inf]
    bounds = [(0, None)] * n
    result = minimize(objective, a0, method='L-BFGS-B', jac=gradient,
                      bounds=bounds, options={'maxiter': max_iter, 
                                               'maxfun': 20000,
                                               'factr': 1e4,
                                               'gtol': 1e-6,
                                               'iprint': -1})

    sc = score_normalized(a_out)
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"      L-BFGS: score={sc:.6f} time={elapsed:.0f}s success={result.success}")
    
    return a_out, sc


def coord_descent_polish(a_init, time_limit, seed=123, verbose=True):
    """
    Coordinate descent: iteratively move mass from high-contributing
    positions to low-impact positions to reduce the max autocorrelation.
    """
    n = len(a_init)
    a = normalize(a_init)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        M = b.max()
        
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
        
        # Find the lag with maximum correlation
        k_star = int(np.argmax(b))
        
        # Find the most contributing pair at this lag
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n, k_star + 1)
        idx = np.arange(i_lo, i_hi)
        j_idx = k_star - idx
        contrib = a[idx] * a[j_idx]
        
        # Try moving mass from the highest contributing pair
        order = np.argsort(contrib)[::-1]
        improved = False
        
        for oi in order[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[oi])
            if a[src] < 1e-14:
                continue
            
            # Find best destination: minimize impact on top lags
            top_lags = np.argsort(b)[::-1][:10]
            scores = np.zeros(n)
            for k2 in top_lags:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                ii = np.arange(s2, e2)
                jj = k2i - ii
                scores[ii] += a[jj] * b[k2i]
                scores[jj] += a[ii] * b[k2i]
            scores[src] = np.inf
            dst = int(np.argmin(scores))
            
            if dst == src:
                continue
            
            # Try several transfer amounts
            for frac in [0.01, 0.03, 0.05, 0.1]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] = max(0, a_trial[src] - amt)
                a_trial[dst] += amt
                a_trial = normalize(a_trial)
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation
            npert = int(rng.integers(3, 10))
            for _ in range(npert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = float(rng.uniform(0.001, 0.01))
                    a[i] -= a[i] * frac
                    a[j] += a[i] * frac + a[i] * frac  # approximate
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    if verbose:
        print(f"      Coord: score={2*n*best_m:.6f} time={time.monotonic()-t0:.0f}s")
    return best_a, 2.0 * n * best_m

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact O(n log n) soft-max gradient via FFT cross-correlation
2. Mirror descent with adaptive temperature continuation
3. L-BFGS-B refinement with multi-T continuation
4. Multi-start with diverse parametric initializations
5. Length exploration via interpolation
6. Targeted coordinate descent polish
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), N)
    return np.fft.irfft(A * A, N)[:size]


def normalize(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score_norm(a):
    n = len(a)
    b = fast_conv(a)
    return 2.0 * n * b.max()


def compute_grad_sofmax(a, T, top_k=None):
    """Exact gradient of 2*n*logsumexp(T*b)/T via FFT cross-correlation."""
    n = len(a)
    b = fast_conv(a)
    
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[-top_k:]
        w = np.zeros(len(b))
        b_sub = b[idx]
        w[idx] = np.exp(T * b_sub - T * b_sub.max())
        w /= w.sum()
    else:
        w = np.exp(T * b - T * b.max())
        w /= w.sum()
    
    # Gradient[m] = 2 * sum_j w[m+j] * a[j]
    N = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    
    return 2.0 * R, b.max()


def md_sofmax(a_init, time_limit, T_start=5.0, T_end=150.0,
              lr_init=0.08, top_k=None, seed=42, verbose=False):
    """Mirror descent with soft-max gradient, adaptive T."""
    n = len(a_init)
    a = normalize(a_init)
    best_a = a.copy()
    best_sc = score_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    T = T_start
    lr = lr_init
    stall = 0
    mom = 0.8
    a_prev = a.copy()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov in multiplicative space
        ratio = np.clip(a / np.maximum(a_prev, 1e-30), 1e-2, 1e2)
        a_nest = normalize(a * np.power(ratio, mom))
        
        grad, _ = compute_grad_sofmax(a_nest, T, top_k)
        
        # Adaptive lr
        grad_scale = np.max(np.abs(grad))
        if grad_scale < 1e-30:
            break
        eta = lr / (1.0 + 0.05 * np.sqrt(it)) / grad_scale * 0.5
        
        log_a = np.log(np.maximum(a_nest, 1e-30)) - eta * grad / max(grad_scale, 1e-30)
        # Actually, let me just use eta directly
        log_a = np.log(np.maximum(a_nest, 1e-30)) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        a_prev = a.copy()
        a = a_new
        it += 1
        
        sc = score_norm(a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Increase T
        progress = min(1.0, it / 20000.0)
        T = T_start * (T_end / T_start) ** progress
        
        # Restart
        if stall > 6000:
            stall = 0
            T = max(T_start, T * 0.4)
            lr = max(0.01, lr * 0.8)
            w = float(rng.uniform(0.1, 0.4))
            pert = best_a * (1.0 + rng.normal(0, w, n))
            a = normalize(best_a + w * pert)
            a_prev = a.copy()
            if verbose:
                print(f"      MD restart T={T:.1f} lr={lr:.4f} it={it}")
        
        if verbose and it % 10000 == 0:
            print(f"      MD it={it} T={T:.1f} sc={best_sc:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_sc


def lbfgs_refine(a_init, time_limit, T_list=None, seed=0, verbose=False):
    """L-BFGS-B with continuation over T values."""
    n = len(a_init)
    a = normalize(a_init)
    t0 = time.monotonic()
    
    if T_list is None:
        T_list = [10, 25, 50, 100]
    
    for T in T_list:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def obj(x):
            if time.monotonic() - t0 > time_limit:
                return np.inf
            x = np.maximum(x, 1e-30)
            s = x.sum()
            if s < 1e-15:
                return np.inf
            x = x / s
            b = fast_conv(x)
            Tb = T * b
            Tb -= Tb.max()
            lse = Tb.max() + np.log(np.sum(np.exp(Tb)))
            return 2.0 * n * lse / T
        
        def grad(x):
            if time.monotonic() - t0 > time_limit:
                return np.zeros(n)
            x = np.maximum(x, 1e-30)
            s = x.sum()
            if s < 1e-15:
                return np.zeros(n)
            x = x / s
            g, _ = compute_grad_sofmax(x, T)
            return g
        
        try:
            res = minimize(obj, a, method='L-BFGS-B', jac=grad,
                          bounds=[(0, None)] * n,
                          options={'maxiter': 2000, 'maxfun': 10000,
                                   'iprint': -1, 'gtol': 1e-8})
        except Exception:
            pass
        
        a = normalize(res.x) if hasattr(res, 'x') and res.x is not None else a
        sc = score_norm(a)
        if verbose:
            print(f"      L-BFGS T={T}: sc={sc:.6f} t={time.monotonic()-t0:.0f}s")
    
    return a, score_norm(a)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact O(n log n) soft-max gradient via FFT cross-correlation
2. Mirror descent with adaptive temperature continuation
3. L-BFGS-B refinement with multi-T continuation
4. Multi-start with diverse parametric initializations
5. Length exploration via interpolation
6. Targeted coordinate descent polish

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), N)
    return np.fft.irfft(A * A, N)[:size]


def normalize(a):
    """Clip non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score_norm(a):
    """Score for normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(a)
    b = fast_conv(a)
    return 2.0 * n * b.max()


def compute_grad_sofmax(a, T, top_k=None):
    """
    Exact gradient of 2*n*logsumexp(T*b)/T w.r.t. a.
    Uses FFT cross-correlation: grad[m] = 2 * sum_j w[m+j]*a[j].
    Returns (gradient_vector, max_b_value).
    """
    n = len(a)
    b = fast_conv(a)

    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[-top_k:]
        w = np.zeros(len(b))
        b_sub = b[idx]
        w[idx] = np.exp(T * b_sub - T * b_sub.max())
        w /= w.sum()
    else:
        w = np.exp(T * b - T * b.max())
        w /= w.sum()

    # Cross-correlation: R[m] = sum_j w[m+j] * a[j], need N >= 3n-2
    N = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]

    return 2.0 * R, b.max()


def md_sofmax(a_init, time_limit, T_start=5.0, T_end=100.0,
              lr_init=0.06, top_k=None, seed=42, verbose=False):
    """Mirror descent (multiplicative) with soft-max gradient, adaptive T."""
    n = len(a_init)
    a = normalize(a_init)
    best_a = a.copy()
    best_sc = score_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    T = T_start
    lr = lr_init
    stall = 0
    mom = 0.75
    a_prev = a.copy()
    it = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov in multiplicative space
        ratio = np.clip(a / np.maximum(a_prev, 1e-30), 1e-3, 1e3)
        a_nest = normalize(a * np.power(ratio, mom))

        grad, _ = compute_grad_sofmax(a_nest, T, top_k)

        # Adaptive learning rate based on gradient magnitude
        gmax = np.max(np.abs(grad))
        if gmax < 1e-30:
            break
        eta = lr / (1.0 + 0.03 * np.sqrt(it))

        # Multiplicative update
        log_a = np.log(np.maximum(a_nest, 1e-30)) - eta * grad / gmax
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        a_prev = a.copy()
        a = a_new
        it += 1

        sc = score_norm(a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Continuation: increase T over time
        progress = min(1.0, it / 15000.0)
        T = T_start * (T_end / T_start) ** progress

        # Restart if stuck
        if stall > 5000:
            stall = 0
            T = max(T_start, T * 0.3)
            lr = max(0.005, lr * 0.75)
            mom = max(0.5, mom - 0.05)
            w = float(rng.uniform(0.05, 0.3))
            pert = best_a * (1.0 + rng.normal(0, w, n))
            a = normalize((1 - w) * best_a + w * normalize(pert))
            a_prev = a.copy()
            if verbose:
                print(f"      MD restart [{time.monotonic()-t0:.0f}s] T={T:.1f} lr={lr:.4f} sc={best_sc:.6f}")

        if verbose and it % 20000 == 0:
            print(f"      MD it={it} T={T:.1f} sc={best_sc:.6f} t={time.monotonic()-t0:.0f}s")

    return best_a, best_sc


def lbfgs_refine(a_init, time_limit, verbose=False):
    """L-BFGS-B refinement with continuation over T values."""
    n = len(a_init)
    a = normalize(a_init)
    t0 = time.monotonic()
    T_list = [8, 20, 50, 80]

    for T in T_list:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def obj_wrapper(x):
            if time.monotonic() - t0 > time_limit:
                return np.inf
            x = np.maximum(x, 1e-20)
            s = x.sum()
            if s < 1e-15:
                return np.inf
            xn = x / s
            b = fast_conv(xn)
            Tb = T * b
            Tb_shift = Tb - Tb.max()
            lse = Tb_shift.max() + np.log(np.sum(np.exp(Tb_shift)))
            return 2.0 * n * lse / T

        def grad_wrapper(x):
            if time.monotonic() - t0 > time_limit:
                return np.zeros(n)
            x = np.maximum(x, 1e-20)
            s = x.sum()
            if s < 1e-15:
                return np.zeros(n)
            xn = x / s
            g, _ = compute_grad_sofmax(xn, T)
            # Adjust for normalization: d(x/s)/dx = (I*s - x*1^T)/s^2
            # Approximate: just return g (scale-invariance means this works in practice)
            return g

        try:
            res = minimize(obj_wrapper, a, method='L-BFGS-B', jac=grad_wrapper,
                           bounds=[(0, None)] * n,
                           options={'maxiter': 1500, 'maxfun': 8000,
                                    'iprint': -1, 'gtol': 1e-7})
            a_new = normalize(res.x)
            sc_new = score_norm(a_new)
            if sc_new < score_norm(a):
                a = a_new
        except Exception:
            pass

        sc = score_norm(a)
        if verbose:
            print(f"      L-BFGS T={T}: sc={sc:.6f} t={time.monotonic()-t0:.0f}s")

    return a, score_norm(a)


def coord_polish(a_init, time_limit, seed=123, verbose=False):
    """Targeted coordinate descent on the peak autocorrelation lag."""
    n = len(a_init)
    a = normalize(a_init)
    best_a = a.copy()
    best_sc = score_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        M = b.max()
        sc = 2.0 * n * M

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break

        k_star = int(np.argmax(b))

        # Get contributing pairs at the peak lag
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n, k_star + 1)
        idx = np.arange(i_lo, i_hi)
        j_idx = k_star - idx
        contrib = a[idx] * a[j_idx]

        # Try top contributing pairs
        order = np.argsort(contrib)[::-1]
        improved = False

        for oi in order[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[oi])
            if a[src] < 1e-14:
                continue

            # Score destinations by impact on top lags
            top_lags = np.argsort(b)[::-1][:12]
            scores = np.zeros(n)
            for k2 in top_lags:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                ii = np.arange(s2, e2)
                jj = k2i - ii
                scores[ii] += a[jj] * b[k2i]
                scores[jj] += a[ii] * b[k2i]
            scores[src] = np.inf
            dst = int(np.argmin(scores))

            if dst == src:
                continue

            for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] = max(0.0, a_trial[src] - amt)
                a_trial[dst] += amt
                a_trial = normalize(a_trial)
                m_trial = fast_conv(a_trial).max()
                if 2.0 * n * m_trial < best_sc - 1e-14:
                    a = a_trial
                    best_sc = 2.0 * n * m_trial
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Small random perturbation to escape
            npert = int(rng.integers(3, 8))
            for _ in range(npert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = float(rng.uniform(0.001, 0.008))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            sc_new = score_norm(a)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a.copy()
                stall = 0

    if verbose:
        print(f"      Coord: sc={best_sc:.6f} t={time.monotonic()-t0:.0f}s")
    return best_a, best_sc


def parametric_candidates(n, rng):
    """Generate diverse parametric initializations."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    cands = []

    # Uniform
    cands.append(np.ones(n, dtype=np.float64))

    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        cands.append((i_arr + 1.0) ** p)
        cands.append((n - i_arr) ** p)

    # Gaussians
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sig in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            cands.append(np.exp(-((x - c) ** 2) / (2 * sig ** 2)))

    # Bimodal
    for sig in [0.05, 0.1, 0.15, 0.25]:
        cands.append(np.exp(-((x - 0.25) ** 2) / (2 * sig ** 2)) +
                     np.exp(-((x - 0.75) ** 2) / (2 * sig ** 2)))
        cands.append(np.exp(-((x - 0.2) ** 2) / (2 * sig ** 2)) +
                     np.exp(-((x - 0.8) ** 2) / (2 * sig ** 2)))

    # Cosine windows
    for alpha in [1.0, 2.0, 3.0, 5.0]:
        cands.append(np.cos(np.pi * alpha * (x - 0.5)) ** 2 + 0.01)

    # Sine
    for f in [1, 2, 3, 5]:
        cands.append(np.sin(np.pi * f * x) ** 2 + 0.01)
        cands.append(np.sin(np.pi * x * f + float(rng.uniform(0, 2 * np.pi))) ** 2 + 0.01)

    # Exponential
    for r in [1, 2, 5, 10]:
        cands.append(np.exp(-r * x))
        cands.append(np.exp(r * (x - 1)))

    # Random smooth (band-limited noise)
    for _ in range(5):
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        sig = np.zeros(n)
        for fi, pi in zip(freqs, phases):
            sig += np.sin(2 * np.pi * fi * x + pi)
        cands.append(np.abs(sig) + 0.1)

    # Triangle
    cands.append(1.0 - np.abs(2.0 * x - 1.0))
    cands.append(np.abs(2.0 * x - 1.0))

    return [normalize(c) for c in cands]


def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss

    print(f"\n=== Phase 4: Final polish (n={n_best}, budget={remaining:.0f}s) ===")

    t4 = min(remaining * 0.4, 150.0)
    if t4 > 10:
        a_out, sc_out = coord_polish(best_seq, t4, seed=42, verbose=True)
        check_update(a_out, "final_coord")

    # One more MD pass with very high T for fine-tuning
    remaining = time_budget - (time.monotonic() - t_global)
    t4b = min(remaining * 0.3, 80.0)
    if t4b > 10:
        a_out, _ = md_sofmax(best_seq, t4b, T_start=50.0, T_end=300.0,
                              lr_init=0.01, top_k=20, seed=99, verbose=False)
        check_update(a_out, "final_md_hight")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using uniform fallback.")
        n_fb = 5000
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")

    return [float(x) for x in best_seq]
