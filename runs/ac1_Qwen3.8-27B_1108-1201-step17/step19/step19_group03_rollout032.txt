    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize a to be non-negative with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def solve_primal_lp(a, M, constraint_idx, solver):
    """Solve the restricted primal LP: max sum(b) s.t. conv(a,b)[k] <= M for k in constraint_idx, b >= 0"""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix (sparse would be better but let's use dense for now)
    # T[k, j] = a[k-j] if 0 <= k-j < n
    constraints = []
    for k in constraint_idx:
        ki = int(k)
        # conv(a,b)[ki] = sum_j a[ki-j] * b[j] for valid j
        # Valid j: 0 <= j < n and 0 <= ki-j < n
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        if j_start >= j_end:
            continue
        j_idx = np.arange(j_start, j_end)
        row = a[ki - j_idx]
        constraints.append(row @ b_var <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    return np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

def solve_dual_lp(a, M, solver):
    """Solve the dual LP: min M*sum(y) s.t. (a*y_pad)[j] >= 1 for j=0..n-1, y >= 0"""
    n = len(a)
    size = 2 * n - 1
    y_var = cp.Variable(size, nonneg=True)
    
    # Constraint: sum_i a[i] * y[i+j] >= 1 for j = 0, ..., n-1
    # This is (a * y)[j] >= 1, where y is treated as a sequence of length 'size'
    constraints = []
    for j in range(n):
        # (a * y_pad)[j] = sum_{i=0}^{n-1} a[i] * y[i+j]
        # Valid: i+j < size, i.e., i < size - j
        i_end = min(n, size - j)
        if i_end <= 0:
            continue
        i_idx = np.arange(i_end)
        row = a[i_idx]
        y_idx = i_idx + j
        constraints.append(row @ y_var[y_idx] >= 1.0)
    
    prob = cp.Problem(cp.Minimize(M * cp.sum(y_var)), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    return np.maximum(np.asarray(y_var.value, dtype=np.float64), 0.0)

def dual_subgradient_step(a, y, M, eta, fft_size):
    """One step of dual subgradient method."""
    n = len(a)
    size = 2 * n - 1
    
    # Compute A*y = conv(a, y_pad)[:n]
    A_pad = np.zeros(fft_size)
    A_pad[:n] = a
    Ap = np.fft.rfft(A_pad, fft_size)
    Y = np.fft.rfft(y, fft_size)
    Ay = np.fft.irfft(Ap * Y, fft_size)[:n]
    
    # Violated constraints: Ay[j] < 1
    violated = (Ay < 1.0).astype(np.float64)
    
    # Compute A^T * violated
    # (A^T v)[k] = sum_j v[j] * A[j,k] = sum_j v[j] * a[k-j] for valid k-j
    # = conv(v, a)[k]
    V_pad = np.zeros(fft_size)
    V_pad[:n] = violated
    Vp = np.fft.rfft(V_pad, fft_size)
    At_v = np.fft.irfft(Vp * Ap, fft_size)[:size]
    
    # Subgradient: M*1 - A^T * v
    grad = M * np.ones(size) - At_v
    
    # Update
    y_new = np.maximum(0, y - eta * grad)
    return y_new, Ay

def lp_fw_optimize(a0, time_limit, solver, verbose=False, n_dual_iters=200, top_K_init=100):
    """LP Frank-Wolfe with dual-guided constraint selection and Nesterov acceleration."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Nesterov momentum
    x = a.copy()
    x_prev = a.copy()
    t_nest = 1.0
    
    iter_count = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov extrapolation
        t_nest_new = (1 + np.sqrt(1 + 4 * t_nest ** 2)) / 2
        beta = (t_nest - 1) / t_nest_new
        y_nest = x + beta * (x - x_prev)
        y_nest = np.maximum(y_nest, 0)
        s_y = y_nest.sum()
        if s_y < 1e-15:
            y_nest = x.copy()
        else:
            y_nest = y_nest / s_y
        
        # Compute current M
        ac = fast_autocorr(y_nest)
        M_curr = ac.max()
        
        # Solve dual LP (or use subgradient approximation)
        if M_curr > 0 and M_curr < 1e-10:
            break
            
        # Get constraint set from dual
        # For large n, solving the full dual LP is expensive
        # Use subgradient to get approximate dual
        y_dual = np.ones(size) / (M_curr * size)
        eta_dual = 2.0 / (M_curr * np.sqrt(iter_count + 1))
        
        for _ in range(n_dual_iters):
            if time.monotonic() - t0 > time_limit:
                break
            y_dual, Ay = dual_subgradient_step(y_nest, y_dual, M_curr, eta_dual, fft_size)
            eta_dual *= 0.999  # slowly decrease
        
        # Identify tight constraints from dual
        y_threshold = np.percentile(y_dual[y_dual > 0], 50) if np.any(y_dual > 0) else y_dual.max() * 0.5
        tight_k = np.where(y_dual > y_threshold)[0]
        
        # Also include the top-K by autocorrelation
        top_k_ac = np.argsort(ac)[::-1][:top_K_init]
        constraint_idx = np.unique(np.concatenate([tight_k, top_k_ac]))
        constraint_idx = constraint_idx[constraint_idx < size]
        
        # Solve restricted primal LP
        b_val = solve_primal_lp(y_nest, M_curr, constraint_idx, solver)
        
        if b_val is None:
            # Fallback: use subgradient direction
            eta = 2.0 / (np.sqrt(iter_count + 1) * (M_curr + 1e-10))
            # Gradient of max(conv) w.r.t. a
            # Approximate: use top few lags
            top_idx = np.argsort(ac)[::-1][:20]
            g = np.zeros(n)
            for k in top_idx:
                ki = int(k)
                j_start = max(0, ki - n + 1)
                j_end = min(n, ki + 1)
                j_idx = np.arange(j_start, j_end)
                g[j_idx] += a[ki - j_idx]
            a_new = np.maximum(x - eta * g, 0)
            a_new = a_new / a_new.sum()
        else:
            b_sum = b_val.sum()
            if b_sum < 1e-15:
                a_new = x.copy()
            else:
                g_dir = b_val / b_sum
                # FW step with Nesterov
                alpha = 2.0 / (t_nest_new + 2)
                a_new = (1 - alpha) * y_nest + alpha * g_dir
                a_new = np.maximum(a_new, 0)
                s_a = a_new.sum()
                if s_a < 1e-15:
                    a_new = x.copy()
                else:
                    a_new = a_new / s_a
        
        x_prev = x
        x = a_new
        t_nest = t_nest_new
        
        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_M:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled too long
        if stall > 200:
            x = best_a.copy()
            x_prev = x.copy()
            t_nest = 1.0
            stall = 0
            top_K_init = min(top_K_init + 50, size)
        
        iter_count += 1
        
        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): M={best_M:.8f}, stall={stall}, K={len(constraint_idx)}")
    
    return best_a, best_M

def coordinate_peel(a0, time_limit, verbose=False, n_trials_per_lag=5):
    """Coordinate descent that peels mass from peak-contributing positions."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 50:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find top lags
        top_lags = np.argsort(ac)[::-1][:15]
        
        improved = False
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            j_idx = np.arange(j_start, j_end)
            contributors = a[j_idx] * a[ki - j_idx]
            
            if contributors.max() < 1e-20:
                continue
            
            # Find the coordinate that contributes most to this lag
            src_offset = np.argmax(contributors)
            src = j_idx[src_offset]
            src_val = a[src]
            
            if src_val < 1e-12:
                continue
            
            # Find destination: a position that minimally contributes to top lags
            # Compute "sensitivity" of each position to top lags
            sens = np.zeros(n)
            for kk in top_lags[:8]:
                kki = int(kk)
                js = max(0, kki - n + 1)
                je = min(n, kki + 1)
                jj = np.arange(js, je)
                sens[jj] += a[kki - jj]
            
            # Destinations with low sensitivity
            low_sens = np.argsort(sens)[:10]
            candidates = [d for d in low_sens if d != src and a[d] < best_a[d] * 2]
            
            if not candidates:
                candidates = [d for d in low_sens if d != src]
            
            found = False
            for dst in candidates[:5]:
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M - 1e-15:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Peel ({elapsed:.0f}s): M={best_M:.8f}, stall={stall}")
    
    return best_a, best_M

def spectral_refine(a0, time_limit, verbose=False):
    """Refine using spectral projection toward flat spectrum."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 2).bit_length()
    stall = 0
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 100:
            break
        
        # Compute spectrum
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        
        # Target: flat magnitude (within the support)
        mag_mean = np.mean(mag)
        
        # Blend toward flat
        alpha = 0.05 + 0.1 * (it / min(it + 1, 100))
        new_mag = (1 - alpha) * mag + alpha * mag_mean
        # Don't go below zero
        new_mag = np.maximum(new_mag, 0)
        
        A_new = new_mag * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            # Try smaller step
            a_small = (1 - 0.5*alpha) * a + 0.5*alpha * a_new
            a_small = np.maximum(a_small, 0)
            s_s = a_small.sum()
            if s_s > 1e-15:
                a_small = a_small / s_s
                M_small = fast_autocorr(a_small).max()
                if M_small < best_M:
                    best_M = M_small
                    best_a = a_small.copy()
                    a = a_small
                    stall = 0
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral it={it} ({elapsed:.0f}s): M={best_M:.8f}, stall={stall}")
    
    return best_a, best_M

def make_u_shape(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = 1.0 + sharpness * (2*x - 1) ** 2
    return normalize(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_conv(a):
    """Full convolution of a with itself via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):

def solve_lp_highs(a, M, constraint_idx):
    """Solve max sum(b) s.t. conv(a,b)[k] <= M for k in constraint_idx, b >= 0 using HiGHS."""
    n = len(a)
    m = len(constraint_idx)
    if m == 0:
        return None

    # Build constraint matrix: T[i, j] = a[ki - j] if 0 <= ki-j < n
    T = np.zeros((m, n))
    rhs = np.zeros(m)
    for i, k in enumerate(constraint_idx):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        if j_start >= j_end:
            continue
        j_idx = np.arange(j_start, j_end)
        T[i, j_start:j_end] = a[ki - j_idx]
        rhs[i] = M

    # Filter out redundant zero rows
    nonzero_rows = np.any(T != 0, axis=1)
    if not np.any(nonzero_rows):
        return None
    T = T[nonzero_rows]
    rhs = rhs[nonzero_rows]

    # max sum(b) s.t. T@b <= rhs, b >= 0
    # -> min -sum(b) s.t. T@b <= rhs, b >= 0
    c = -np.ones(n)
    res = linprog(c, A_ub=T, b_ub=rhs, bounds=[(0, None)] * n, method='highs')

    if res.status != 0:
        return None
    return np.maximum(res.x, 0.0)


def solve_lp_cvxpy(a, M, constraint_idx, solver):
    """Solve max sum(b) s.t. conv(a,b)[k] <= M for k in constraint_idx, b >= 0 using cvxpy."""
    n = len(a)
    m = len(constraint_idx)
    if m == 0:
        return None

    T = np.zeros((m, n))
    for i, k in enumerate(constraint_idx):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        if j_start >= j_end:
            continue
        j_idx = np.arange(j_start, j_end)
        T[i, j_start:j_end] = a[ki - j_idx]

    nonzero_rows = np.any(T != 0, axis=1)
    if not np.any(nonzero_rows):
        return None
    T = T[nonzero_rows]

    b_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    prob.solve(solver=solver, verbose=False)

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    return np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)


def dual_subgradient(a, y_dual, M, n_iters, eta_init, fft_size):
    """Run dual subgradient iterations to identify tight constraints."""
    n = len(a)
    size = 2 * n - 1
    eta = eta_init

    A_pad = np.zeros(fft_size)
    A_pad[:n] = a
    Ap = np.fft.rfft(A_pad, fft_size)

    for _ in range(n_iters):
        Y = np.fft.rfft(y_dual, fft_size)
        Ay = np.fft.irfft(Ap * Y, fft_size)[:n]

        violated = (Ay < 1.0).astype(np.float64)
        if not np.any(violated):
            break

        V_pad = np.zeros(fft_size)
        V_pad[:n] = violated
        Vp = np.fft.rfft(V_pad, fft_size)
        At_v = np.fft.irfft(Vp * Ap, fft_size)[:size]

        grad = M * np.ones(size) - At_v
        y_dual = np.maximum(0, y_dual - eta * grad)
        eta *= 0.995

    return y_dual, Ay


def lp_fw_optimize(a0, time_limit, solver, verbose=False, n_dual_iters=150, top_K_init=100, use_highs=True):
    """LP Frank-Wolfe with dual-guided constraint selection and Nesterov acceleration."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    x = a.copy()
    x_prev = a.copy()
    t_nest = 1.0
    iter_count = 0
    stall = 0
    cur_K = top_K_init

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        # Nesterov extrapolation
        t_nest_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / t_nest_new
        y_nest = x + beta * (x - x_prev)
        y_nest = np.maximum(y_nest, 0.0)
        s_y = y_nest.sum()
        if s_y < 1e-15:
            y_nest = x.copy()
        else:
            y_nest = y_nest / s_y

        ac = fast_conv(y_nest)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Dual subgradient to identify tight constraints
        y_dual = np.ones(size) / (M_curr * size)
        eta_dual = 2.0 / (M_curr * np.sqrt(iter_count + 1.0))
        y_dual, Ay = dual_subgradient(y_nest, y_dual, M_curr, n_dual_iters, eta_dual, fft_size)

        # Select constraint set
        dual_thresh = np.percentile(y_dual[y_dual > 0], 60) if np.sum(y_dual > 0) > 10 else 0
        tight_k = np.where(y_dual > dual_thresh)[0]
        top_k_ac = np.argsort(ac)[::-1][:cur_K]
        constraint_idx = np.unique(np.concatenate([tight_k, top_k_ac]))
        constraint_idx = constraint_idx[constraint_idx < size]

        # Cap constraint set size for speed
        max_constraints = 400
        if len(constraint_idx) > max_constraints:
            # Keep top by dual value and top by ac value
            dual_vals = y_dual[constraint_idx]
            ac_vals = ac[constraint_idx]
            half = max_constraints // 2
            top_dual = constraint_idx[np.argsort(dual_vals)[::-1][:half]]
            top_ac = constraint_idx[np.argsort(ac_vals)[::-1][:half]]
            constraint_idx = np.unique(np.concatenate([top_dual, top_ac]))

        # Solve restricted primal LP
        if use_highs:
            b_val = solve_lp_highs(y_nest, M_curr, constraint_idx)
        else:
            b_val = solve_lp_cvxpy(y_nest, M_curr, constraint_idx, solver)

        if b_val is None or b_val.sum() < 1e-15:
            # Fallback: simple gradient step
            top_idx = np.argsort(ac)[::-1][:20]
            g = np.zeros(n)
            for k in top_idx:
                ki = int(k)
                j_start = max(0, ki - n + 1)
                j_end = min(n, ki + 1)
                j_idx = np.arange(j_start, j_end)
                g[j_idx] += 2.0 * y_nest[ki - j_idx]
            eta = 1.0 / (np.sqrt(iter_count + 1.0) * (g.max() + 1e-10))
            a_new = np.maximum(y_nest - eta * g, 0.0)
            a_new = a_new / a_new.sum()
        else:
            b_sum = b_val.sum()
            g_dir = b_val / b_sum
            alpha = 2.0 / (t_nest_new + 2.0)
            # Line search: try a few step sizes
            best_alpha = alpha
            best_trial_M = fast_conv(y_nest).max()
            for trial_alpha in [alpha, min(1.0, alpha * 1.5), min(1.0, alpha * 2.0), 0.5, 0.3]:
                a_trial = (1.0 - trial_alpha) * y_nest + trial_alpha * g_dir
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_trial_M:
                    best_trial_M = m_trial
                    best_alpha = trial_alpha

            a_new = (1.0 - best_alpha) * y_nest + best_alpha * g_dir
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = x.copy()
            else:
                a_new = a_new / s_a

        x_prev = x
        x = a_new
        t_nest = t_nest_new

        M_new = fast_conv(x).max()
        if M_new < best_M - 1e-16:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive K growth
        if iter_count % 50 == 0:
            cur_K = min(cur_K + 20, size)

        # Restart if stalled
        if stall > 150:
            x = best_a.copy() + np.random.default_rng(iter_count).normal(0, 0.01, n) * best_a
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            t_nest = 1.0
            stall = 0
            cur_K = min(cur_K + 50, size)

        iter_count += 1

        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): M={best_M:.8f}, K={len(constraint_idx)}, stall={stall}")

    return best_a, best_M


def coordinate_peel(a0, time_limit, verbose=False):
    """Targeted coordinate descent peeling mass from peak-contributing positions."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 80:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:12]

        # Compute sensitivity of each position to top lags
        sens = np.zeros(n)
        for k in top_lags[:8]:
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n, ki + 1)
            jj = np.arange(js, je)
            sens[jj] += a[ki - jj]

        improved = False
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n, ki + 1)
            jj = np.arange(js, je)
            contrib = a[jj] * a[ki - jj]

            if contrib.max() < 1e-20:
                continue

            src = jj[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-sensitivity destinations
            low_sens_idx = np.argsort(sens)[:15]
            candidates = [int(d) for d in low_sens_idx if int(d) != src and a[int(d)] < src_val * 0.5]
            if not candidates:
                candidates = [int(d) for d in low_sens_idx if int(d) != src]

            found = False
            for dst in candidates[:5]:
                for frac in [0.005, 0.01, 0.015, 0.02, 0.03, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_M - 1e-15:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and (stall % 20 == 0 or not improved == False and stall == 1):
            elapsed = time.monotonic() - t0
            print(f"    Peel ({elapsed:.0f}s): M={best_M:.8f}, stall={stall}")

    return best_a, best_M


def spectral_refine(a0, time_limit, verbose=False):
    """Spectral domain refinement toward flat power spectrum."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 2).bit_length()
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit and stall < 150:
        it += 1
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        mag_mean = np.mean(mag[mag > 1e-12]) if np.any(mag > 1e-12) else 1.0

        alpha = 0.03 + 0.07 * min(1.0, it / 200.0)
        new_mag = (1.0 - alpha) * mag + alpha * mag_mean
        new_mag = np.maximum(new_mag, 0.0)

        A_new = new_mag * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        M_new = fast_conv(a_new).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            # Try smaller step
            a_small = (1.0 - 0.5 * alpha) * a + 0.5 * alpha * a_new
            a_small = np.maximum(a_small, 0.0)
            s_s = a_small.sum()
            if s_s > 1e-15:
                a_small = a_small / s_s
                M_s = fast_conv(a_small).max()
                if M_s < best_M:
                    best_M = M_s
                    best_a = a_small.copy()
                    a = a_small
                    stall = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral it={it} ({elapsed:.0f}s): M={best_M:.8f}, stall={stall}")

    return best_a, best_M


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = 1.0 + sharpness * (2.0 * x - 1.0) ** 2
    return normalize(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
