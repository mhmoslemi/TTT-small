def fast_autocorr(a):

    a = np.asarray(a, dtype=np.float64)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

        # = a_{ki-j} (when i=j, if 0<=ki-j<n)
        # + a_j (when ki-i=j, i.e., i=ki-j, if 0<=ki-j<n and ki-j != j)
        # = a_{ki-j} + a_j * 1[ki-j != j and 0<=ki-j<n]
        # But since we're summing over i, and the term a_{ki'} a_j appears when i = ki - j:
        # d/d(a_j) of a_{ki-j}*a_j = a_{ki-j} (derivative w.r.t. first factor, i=j)
        # + a_{ki-j} (derivative w.r.t. second factor, when i=ki-j, giving a_{ki-j}*a_j)
        # Wait, let me be precise:
        # conv_k = sum_{i} a_i * a_{k-i} (over valid i)
        # d/d(a_j): 
        #   Term where i=j: a_j * a_{k-j} -> derivative is a_{k-j}
        #   Term where k-i=j, i.e., i=k-j: a_{k-j} * a_j -> derivative is a_{k-j}
        #   If j = k-j (i.e., k=2j): only one term a_j^2, derivative is 2*a_j
        #   Otherwise: derivative is a_{k-j} + a_{k-j} = 2*a_{k-j}... 
        #   Wait no. If i=j gives term a_j*a_{k-j}, derivative w.r.t. a_j is a_{k-j}.
        #   If i=k-j gives term a_{k-j}*a_j, derivative w.r.t. a_j is a_{k-j}.
        #   These are the SAME term only if j = k-j, i.e., k=2j.
        #   So for k != 2j: derivative = a_{k-j} + a_{k-j} = 2*a_{k-j}? 
        #   No! The term a_i * a_{k-i} with i=j is a_j * a_{k-j}.
        #   The term a_i * a_{k-i} with i=k-j is a_{k-j} * a_j.
        #   These are different terms in the sum (different i values) unless j = k-j.
        #   So the derivative is: a_{k-j} (from i=j term) + a_{k-j} (from i=k-j term)
        #   = 2*a_{k-j} if both i=j and i=k-j are valid and j != k-j.
        #   = a_{k-j} if only one is valid (e.g., j near boundary).
        #   = 2*a_j if k=2j (same term, derivative of a_j^2).
        
        # Simplified: for j in [j_lo, j_hi], the index ki-j is in [0, n-1].
        # The "other" index j-ki: is it valid? j-ki >= 0 means j >= ki.
        # And j-ki < n means j < ki + n, which is always true since j < n and ki < 2n.
        # So the "other" term is valid when j >= ki.
        
        pair_idx = ki - j_range  # This is in [0, n-1] by construction of j_range
        grad[j_range] += wi * a[pair_idx]  # Contribution from i=j term
        
        # Contribution from i=k-j term (valid when j >= ki, i.e., k-i = k-j >= 0... 
        # wait, i = k-j, and we need 0 <= i < n, so 0 <= k-j < n, i.e., k-n < j <= k)
        # j >= k means k-i = k-j <= 0, so i = k-j <= 0, valid only if i=0, j=k.
        # Hmm, I think I'm overcomplicating this.
        
        # Let me just use the direct formula:
        # conv_k = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a_i * a_{k-i}
        # d/d(a_j) = a_{k-j} * 1[max(0,k-n+1) <= j <= min(n-1,k)]
        #           + a_j * 1[max(0,k-n+1) <= k-j <= min(n-1,k)]  (from the a_{k-i} factor)
        #           Wait, the second term: d/d(a_j) of a_{k-i} when k-i=j, i.e., i=k-j.
        #           This is valid when 0 <= k-j < n AND max(0,k-n+1) <= k-j <= min(n-1,k).
        #           k-j >= 0 means j <= k. k-j < n means j > k-n, i.e., j >= k-n+1.
        #           So valid when max(0, k-n+1) <= j <= min(n-1, k).
        #           But that's the SAME range as the first term!
        #

        #              = 2*a_j if k = 2j (degenerate case)
        #              = a_{k-j} if only the first condition holds
        #              = a_j if only the second condition holds
        #
        # But we showed both conditions give the same range [j_lo, j_hi].
        # So for all j in [j_lo, j_hi]: derivative = a_{k-j} + a_{k-j} = 2*a_{k-j}
        # UNLESS k = 2j, in which case it's 2*a_j (same thing since k-j=j).
        #
        # Wait, that gives derivative = 2*a_{k-j} for all j in range. Let me verify:
        # a = [1, 2, 3], k=1: conv_1 = a_0*a_1 + a_1*a_0 = 2+2 = 4
        # d/d(a_0) = a_1 + a_1 = 4? Check: if a_0 -> a_0+eps, conv_1 = (a_0+eps)*a_1 + a_1*a_0 = 2*a_0*a_1 + eps*a_1
        # So d/d(a_0) = a_1 = 2. Not 2*a_1 = 4!
        
        # I see my error. Let me redo:
        # conv_k = sum_i a_i * a_{k-i}
        # If I perturb a_j by eps:
        #   The term with i=j: (a_j + eps) * a_{k-j} = a_j*a_{k-j} + eps*a_{k-j}
        #   The term with i=k-j (if k-j != j and valid): a_{k-j} * (a_j + eps) = a_{k-j}*a_j + eps*a_{k-j}
        #   If k-j == j: the term is (a_j+eps)^2 = a_j^2 + 2*eps*a_j
        # So derivative = a_{k-j} (from i=j) + a_{k-j} (from i=k-j, if k-j != j and valid)
        #              = 2*a_{k-j} if k != 2j and both indices valid
        #              = 2*a_j if k = 2j
        #              = a_{k-j} if only i=j is valid (i=k-j not in range)
        #
        # For a=[1,2,3], k=1: j_lo=max(0,1-2)=0, j_hi=min(2,1)=1
        #   j=0: k-j=1, valid. i=k-j=1, valid (0<=1<=1). k!=2j (1!=0). deriv = 2*a_1 = 4? 
        #   But we computed d/d(a_0) = a_1 = 2. Contradiction!
        #
        # Let me recompute by hand: conv_1 = a_0*a_1 + a_1*a_0 = 1*2 + 2*1 = 4.
        # d/d(a_0): treat a_1 as constant. = a_1 + a_1 = 2*2 = 4. 
        # Actually yes! Because both terms a_0*a_1 and a_1*a_0 contain a_0.
        # So d/d(a_0) = 2*a_1 = 4. And d/d(a_1) = a_0 + a_0 = 2*a_0 = 2.
        # Let me verify: if a_0 -> a_0+eps: conv_1 = (a_0+eps)*a_1 + a_1*(a_0+eps) = 2*a_0*a_1 + 2*eps*a_1
        # d/d(a_0) = 2*a_1 = 4. ✓
        
        # OK so the correct formula is: for j in [j_lo, j_hi]:
        #   If k-j is in [0, n-1] (which it always is by construction):
        #     The i=j term gives: a_{k-j}
        #     The i=k-j term (valid if k-j in [j_lo, j_hi] AND k-j != j): gives a_j
        #     Combined: a_{k-j} + a_j * 1[k-j in [j_lo, j_hi], k != 2j]
        #   
        # But k-j is always in [0, n-1] for j in [j_lo, j_hi].
        # Is k-j in [j_lo, j_hi]? j_lo = max(0, k-n+1), j_hi = min(n-1, k).
        # k-j >= j_lo means k-j >= max(0, k-n+1), i.e., j <= min(k, n-1) = j_hi. ✓ (always true)
        # k-j <= j_hi means k-j <= min(n-1, k), i.e., j >= max(0, k-n+1) = j_lo. ✓ (always true)
        # So k-j is ALWAYS in [j_lo, j_hi] for j in [j_lo, j_hi]!
        #
        # Therefore: for all j in [j_lo, j_hi], k != 2j:
        #   derivative = a_{k-j} + a_j
        # For k = 2j:
        #   derivative = 2*a_j (which equals a_{k-j} + a_j since k-j=j)
        #
        # So in ALL cases: derivative = a_{k-j} + a_{k-j}... NO!
        # derivative = a_{k-j} + a_j (NOT a_{k-j} + a_{k-j})
        # The first term is a_{k-j} (from i=j), the second is a_j (from i=k-j).
        
        # So: grad[j] += wi * (a[ki - j] + a[j]) for all j in valid range.
        # Wait, but a[j] is the SAME for all j in the range. Let me re-verify:
        # conv_1 = a_0*a_1 + a_1*a_0. d/d(a_0) = a_1 + a_1 = 2*a_1. 
        # My formula: a_{1-0} + a_0 = a_1 + a_0 = 3. WRONG!
        #
        # I see the issue. The "i=k-j" term: when i=k-j, the term in the sum is a_{k-j}*a_j.
        # d/d(a_j) of a_{k-j}*a_j = a_{k-j} (treating a_{k-j} as constant since we're differentiating w.r.t. a_j).
        # NOT a_j!
        #
        # So: derivative = a_{k-j} (from i=j term: d/d(a_j) of a_j*a_{k-j})
        #                + a_{k-j} (from i=k-j term: d/d(a_j) of a_{k-j}*a_j)
        #                = 2*a_{k-j} (when k != 2j and both valid)
        #                = 2*a_j (when k = 2j)
        #
        # Let me verify: a=[1,2,3], k=1, j=0: derivative = 2*a_{1-0} = 2*a_1 = 4. ✓
        # j=1: derivative = 2*a_{1-1} = 2*a_0 = 2. ✓ (d/d(a_1) of a_0*a_1+a_1*a_0 = a_0+a_0 = 2) ✓
        
        # GREAT. So the formula is simply: grad[j] += 2 * wi * a[ki - j] for j in valid range.
        # (When k=2j, a[ki-j] = a[j], so it's 2*wi*a[j], which is correct.)
        
        grad[j_range] += 2.0 * wi * a[pair_idx]
    
    return grad


def mirror_descent_v2(a0, time_limit, n_peaks=50, temperature=0.01, lr0=2.0,
                      restart_every=2000, rng_seed=42, verbose=False):
    """Improved mirror descent with adaptive step size and better subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_J = objective_value(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    restart_count = 0
    step = lr0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        if t % 50 == 0:
            prev_beta = (t - 2.0) / (t + 1.0)
            beta_t = (t - 1.0) / (t + 2.0)
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            prev_beta = (t - 2.0) / (t + 1.0)

        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-20)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        conv = fast_autocorr(y)
        g = subgradient_all_peaks(y, conv, n_peaks=n_peaks, temperature=temperature)

        # Adaptive step size
        eta = step / (np.sqrt(t) + 1.0)
        
        # Mirror step (exponential)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Evaluate
        M_x = fast_autocorr(x).max()
        J_x = 2.0 * n * M_x  # since sum = 1
        if J_x < best_J - 1e-12:
            best_J = J_x
            best_a = x.copy()
            stall = 0
            # Increase step slightly on improvement
            step = min(step * 1.02, lr0 * 3.0)
        else:
            stall += 1
            # Decrease step on stall
            step = max(step * 0.995, lr0 * 0.3)

        if stall > restart_every:
            stall = 0
            restart_count += 1
            strength = min(0.15, 0.03 + 0.02 * min(restart_count, 5))
            pert = rng.normal(0, 1, n) * strength * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-15)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            step = lr0
            if verbose and restart_count <= 5:
                print(f"      MD2 restart {restart_count} t={t}: J={best_J:.6f}")

        if verbose and t % 20000 == 0:
            print(f"      MD2 t={t} ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, step={step:.3f}")

    return best_a, best_J


def coordinate_descent_v2(a0, time_limit, verbose=False, rng_seed=42):
    """
    Coordinate descent that targets the peak lag specifically.
    Tries to move mass from high-contributing positions to low-cost positions.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_J = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 50:
        conv = fast_autocorr(a)
        k_max = int(np.argmax(conv))
        k_mean = (a.sum() ** 2) / (2 * n - 1)
        
        # Find the positions that contribute most to the peak
        j_lo = max(0, k_max - n + 1)
        j_hi = min(n - 1, k_max)
        idx = np.arange(j_lo, j_hi + 1)
        contrib = 2.0 * a[idx] * a[k_max - idx]  # derivative contribution
        
        # Source: position with highest contribution
        src = idx[np.argmax(contrib)]
        
        # Compute "cost" of each position: how much it contributes to top lags
        n_top = min(10, conv_len := len(conv))
        top_ks = np.argpartition(conv, -n_top)[-n_top:]
        costs = np.zeros(n)
        for k2 in top_ks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            if lo2 <= hi2:
                valid = np.arange(lo2, hi2 + 1)
                costs[valid] += 2.0 * a[k2i - valid] * a[valid]
        
        # Destination: low cost, prefer positions far from peak contribution
        # Weight by distance from peak to encourage spreading
        dist = np.abs(np.arange(n) - src).astype(float)
        score = costs + 0.01 * a - 0.001 * dist
        sorted_idx = np.argsort(score)
        
        moved = False
        for dst in sorted_idx[:5]:
            if dst == src:
                continue
            for frac in [0.02, 0.05, 0.1, 0.15, 0.25, 0.4]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_t = fast_autocorr(a_trial).max()
                J_t = 2.0 * n * M_t
                if J_t < best_J - 1e-10:
                    a = a_trial
                    best_J = J_t
                    best_m = M_t
                    best_a = a.copy()
                    moved = True
                    stall = 0
                    break
            if moved:
                break
        
        if not moved:
            stall += 1
            # Random move to escape local minimum
            n_tries = 3
            for _ in range(n_tries):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                amt = a[i] * rng.uniform(0.02, 0.1)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                J_new = 2.0 * n * M_new
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
                    stall = 0
                    break

    return best_a, best_J


def peak_shaving(a0, time_limit, verbose=False):
    """
    Iteratively shave the peak of the autocorrelation by identifying
    the dominant contributing pairs and reducing them.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_J = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        k_max = int(np.argmax(conv))
        k_mean = 1.0 / (2 * n - 1)  # mean when sum(a)=1
        excess = conv[k_max] - k_mean

        if excess < 1e-10:
            break

        # Find top contributing pairs at the peak
        j_lo = max(0, k_max - n + 1)
        j_hi = min(n - 1, k_max)
        idx = np.arange(j_lo, j_hi + 1)
        contrib = a[idx] * a[k_max - idx]
        total_contrib = contrib.sum()
        
        if total_contrib < 1e-15:
            break

        # Try reducing the top contributor and spreading to random positions
        top_k_pairs = min(5, len(idx))
        top_local = np.argsort(contrib)[-top_k_pairs:]
        
        for tl in top_local:
            src = idx[tl]
            # Find positions that DON'T contribute to the peak
            # and have low overall autocorrelation impact
            avoid = set()
            for k2 in np.argpartition(conv, -5)[-5:]:
                k2i = int(k2)
                lo2 = max(0, k2i - n + 1)
                hi2 = min(n - 1, k2i)
                if lo2 <= hi2:
                    avoid.update(range(int(lo2), int(hi2) + 1))
            
            candidates = [j for j in range(n) if j not in avoid and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            
            # Pick a random candidate with low value
            cands_arr = np.array(candidates)
            weights = 1.0 / (a[cands_arr] + 0.1)
            weights /= weights.sum()
            dst = cands_arr[rng.choice(len(cands_arr), p=weights)]
            
            for frac in [0.01, 0.03, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                M_t = fast_autocorr(a_trial).max()
                J_t = 2.0 * n * M_t
                if J_t < best_J - 1e-10:
                    a = a_trial
                    best_J = J_t
                    best_a = a.copy()
                    break
    
    if verbose:
        print(f"      PeakShaving ({time.monotonic()-t0:.0f}s): J={best_J:.6f}")
    return best_a, best_J


def spectral_concentrate(a0, time_limit, verbose=False):
    """
    Spectral method: concentrate power at DC to flatten autocorrelation.
    Uses iterative projection between time and frequency domains.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_J = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_dc = P[0]
        
        if P_max < 1e-30:
            break
        
        # We want to increase P_dc relative to other frequencies
        # This makes the autocorrelation flatter
        # Target: P[0] should dominate
        target_P = P.copy()
        # Reduce non-DC power, boost DC
        non_dc = P.copy()
        non_dc[0] = 0
        non_dc_max = non_dc.max()
        
        if non_dc_max > 1e-30:
            # Attenuate non-DC components (make spectrum more DC-concentrated)
            alpha = min(1.0, 0.05 + 0.05 * it / 1000.0)
            attenuation = 1.0 - alpha * (non_dc / non_dc_max)
            P_new = P * np.ones_like(P)
            P_new[1:] *= attenuation[1:]
            P_new[0] = P[0] + (1.0 - attenuation[1:].mean()) * P[0] * 0.1  # boost DC slightly
        
        # Take sqrt to get magnitude, reconstruct
        mag_new = np.sqrt(np.maximum(P_new, 0))
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        J_new = 2.0 * n * M_new
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend
            blend = 0.1
            a = (1 - blend) * a + blend * a_new
            a = np.maximum(a, 0.0)
            a = a / max(a.sum(), 1e-15)
            J_curr = 2.0 * n * fast_autocorr(a).max()
            if J_curr < best_J:
                best_J = J_curr
                best_a = a.copy()

        if verbose and it % 2000 == 0:
            print(f"      SpecConc it={it} ({time.monotonic()-t0:.0f}s): J={best_J:.6f}")

    return best_a, best_J


def cosine_taper(n, alpha):
    """Create a cosine-tapered constant: a[i] = 1 + alpha*cos(pi*i/(n-1))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def multi_cosine(n, n_harmonics=5, amplitude=0.1):
    """Create a multi-harmonic perturbation of constant."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for m in range(1, n_harmonics + 1):
        a += (amplitude / m) * np.cos(2 * np.pi * m * i / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def safe_eval(a):
    """Safely compute the objective via evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search function. Uses multi-phase optimization:
    1. Generate diverse initializations
    2. Mirror descent (exponential gradient) with adaptive subgradients
    3. Coordinate descent targeting peaks
    4. Peak shaving
    5. Spectral concentration
    6. Multi-length exploration
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = safe_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = safe_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Generate initializations
    candidates = []
    
    # Various n values
    n_values = [4000, 5000, 6000, 7000, 8000]
    if parent is not None:
        n_parent = len(parent)
        n_values = sorted(set(n_values + [n_parent, n_parent + 1000, n_parent - 1000]))
        n_values = [n for n in n_values if 2000 <= n <= 12000]

    print(f"Testing n values: {n_values}")
    
    for n_try in n_values:
        # Constant
        candidates.append((np.ones(n_try) / n_try, f"const_n{n_try}"))
        # Cosine taper
        candidates.append((cosine_taper(n_try, 0.3), f"cos03_n{n_try}"))
        candidates.append((cosine_taper(n_try, -0.3), f"cosm03_n{n_try}"))
        # Multi-harmonic
        candidates.append((multi_cosine(n_try, 3, 0.1), f"mc3_n{n_try}"))
        candidates.append((multi_cosine(n_try, 5, 0.08), f"mc5_n{n_try}"))
    
    # Linear ramp (different starting point)
    for n_try in [5000, 7000]:
        x = np.arange(1, n_try + 1, dtype=np.float64)
        ramp = x / x.sum()
        candidates.append((ramp, f"ramp_n{n_try}"))
        # Anti-ramp (decreasing)
        candidates.append((ramp[::-1].copy(), f"aramp_n{n_try}"))
    
    # If parent exists
    if parent is not None:
        p_norm = parent / parent.sum()
        candidates.insert(0, (p_norm, "parent"))
        # Perturbed parent
        rng_init = np.random.default_rng(0)
        for seed in range(3):
            pert = rng_init.normal(0, 0.1, len(p_norm)) * p_norm
            p_pert = p_norm + pert
            p_pert = np.maximum(p_pert, 0)
            p_pert = p_pert / p_pert.sum()
            candidates.append((p_pert, f"parent_pert{seed}"))

    # Evaluate all initializations
    print(f"\nInitial candidates: {len(candidates)}")
    scored = []
    for a, label in candidates:
        sc = safe_eval(a)
        if np.isfinite(sc):
            scored.append((sc, a, label))
            print(f"  Init [{label}]: {sc:.6f}")
    
    scored.sort(key=lambda x: x[0])
    
    if best_seq is None and scored:
        best_seq = scored[0][1].copy()
        best_score = scored[0][0]
        print(f"  Setting initial best: {best_score:.6f}")

    # === Phase 1: Mirror Descent on top candidates ===
    print(f"\n=== Phase 1: Mirror Descent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_cands = min(8, len(scored))
    t_per = max(30.0, min(180.0, remaining * 0.35 / max(1, n_cands)))
    
    for i in range(n_cands):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget:
            break
        t_avail = min(t_per, time_budget - elapsed)
        if t_avail < 15:
            break
        sc0, a0, label = scored[i]
        print(f"  MD [{label}] (t={t_avail:.0f}s, init={sc0:.6f})...")
        a_out, _ = mirror_descent_v2(a0, t_avail, n_peaks=50, temperature=0.01,
                                     lr0=2.5, restart_every=2000, rng_seed=42 + i, verbose=(i == 0))
        try_update(a_out, f"md1_{label}")
    
    # === Phase 2: Coordinate Descent on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Coordinate Descent ===")
    if remaining > 60 and best_seq is not None:
        t2 = min(remaining * 0.15, 200.0)
        a_in = best_seq / best_seq.sum()
        a_out, _ = coordinate_descent_v2(a_in, t2, verbose=False, rng_seed=42)
        try_update(a_out, "coord1")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t2b = min(remaining * 0.10, 120.0)
            a_out, _ = coordinate_descent_v2(a_in, t2b, verbose=False, rng_seed=777)
            try_update(a_out, "coord2")
    
    # === Phase 3: Peak Shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Peak Shaving ===")
    if remaining > 40 and best_seq is not None:
        t3 = min(remaining * 0.12, 150.0)
        a_in = best_seq / best_seq.sum()
        a_out, _ = peak_shaving(a_in, t3, verbose=True)
        try_update(a_out, "peakshave")
    
    # === Phase 4: Mirror Descent (aggressive) on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Aggressive MD ===")
    if remaining > 60 and best_seq is not None:
        t4 = min(remaining * 0.15, 200.0)
        a_in = best_seq / best_seq.sum()
        a_out, _ = mirror_descent_v2(a_in, t4, n_peaks=80, temperature=0.005,
                                     lr0=4.0, restart_every=1500, rng_seed=7, verbose=True)
        try_update(a_out, "md4_agg")
    
    # === Phase 5: Spectral Concentration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Spectral ===")
    if remaining > 40 and best_seq is not None:
        t5 = min(remaining * 0.10, 100.0)
        a_in = best_seq / best_seq.sum()
        a_out, _ = spectral_concentrate(a_in, t5, verbose=True)
        try_update(a_out, "spectral")
    
    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-length ===")
    if best_seq is not None and remaining > 80:
        n_best = len(best_seq)
        explore_ns = [n_best + 1000, n_best + 2000, n_best - 1000, 8000, 10000]
        explore_ns = [n for n in explore_ns if 2000 <= n <= 12000 and n != n_best]
        explore_ns = list(set(explore_ns))
        
        for new_n in explore_ns[:3]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            a_interp = interpolate_to_length(best_seq, new_n)
            try_update(a_interp, f"interp_n{new_n}")
            
            t6 = min(remaining * 0.20, 120.0)
            if t6 > 20:
                a_out, _ = mirror_descent_v2(a_interp, t6, n_peaks=50, temperature=0.01,
                                             lr0=3.0, restart_every=2000, rng_seed=100 + new_n % 100, verbose=False)
                try_update(a_out, f"md_n{new_n}")
    
    # === Phase 7: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final Polish ===")
    if remaining > 30 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        
        # MD with very fine parameters
        t7a = min(remaining * 0.30, 100.0)
        if t7a > 15:
            a_out, _ = mirror_descent_v2(a_in, t7a, n_peaks=100, temperature=0.003,
                                         lr0=1.5, restart_every=5000, rng_seed=555, verbose=True)
            try_update(a_out, "md_final")
        
        # Coordinate descent
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.30, 80.0)
            a_in2 = best_seq / best_seq.sum()
            a_out, _ = coordinate_descent_v2(a_in2, t7b, verbose=False, rng_seed=12345)
            try_update(a_out, "coord_final")
        
        # Peak shaving
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7c = min(remaining * 0.30, 60.0)
            a_in3 = best_seq / best_seq.sum()
            a_out, _ = peak_shaving(a_in3, t7c, verbose=True)
            try_update(a_out, "shave_final")
    
    # === Phase 8: Fresh start with best structure ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Fresh Optimization ===")
    if remaining > 60 and best_seq is not None:
        # Use the best sequence's "shape" but re-optimize from scratch
        a_in = best_seq / best_seq.sum()
        t8 = min(remaining * 0.5, 300.0)
        a_out, _ = mirror_descent_v2(a_in, t8, n_peaks=60, temperature=0.01,
                                     lr0=3.0, restart_every=3000, rng_seed=999, verbose=True)
        try_update(a_out, "fresh_md")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t8b = min(remaining * 0.4, 150.0)
            a_out, _ = coordinate_descent_v2(best_seq / best_seq.sum(), t8b, verbose=False, rng_seed=4242)
            try_update(a_out, "fresh_coord")

    # Final validation
    if best_seq is not None:
        fs = safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        best_seq = np.ones(5000) / 5000
        best_score = safe_eval(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
