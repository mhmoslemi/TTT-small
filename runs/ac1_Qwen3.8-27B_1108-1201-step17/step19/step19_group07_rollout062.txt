import numpy as np

def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum to 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_gradient(a, beta, top_k=None):
    """Compute softmax-weighted gradient for mirror descent."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]

    if top_k is not None and top_k < size:
        top_idx = np.argsort(c)[::-1][:top_k]
        c_top = c[top_idx]
        c_max = c_top.max()
        exp_c = np.zeros(size)
        exp_c[top_idx] = np.exp(beta * (c_top - c_max))
        w = exp_c / exp_c.sum()
    else:
        c_max = c.max()
        exp_c = np.exp(beta * (c - c_max))
        w = exp_c / exp_c.sum()

    # Gradient: grad_j = 2 * sum_k w_k * a_conv[j] where a_conv[k] = sum_j a_j * a[k-j]
    # Actually grad_j = sum_k w_k * (partial (a*a)_k / partial a_j) = sum_k w_k * 2 * a[k-j]
    # = 2 * (w * a_reversed)[j] (appropriate slicing)
    a_rev = a[::-1]
    W = np.fft.rfft(w, fft_size)
    A_rev = np.fft.rfft(a_rev, fft_size)
    conv_wa = np.fft.irfft(W * A_rev, fft_size)
    grad = 2.0 * conv_wa[n - 1:2 * n - 1]

    return grad, c

    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_threshold = 5000

    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        progress = min(iter_count / max(time_limit * 100, 1), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        grad, c_y = softmax_gradient(y, beta, top_k=top_k)

        eta = eta0 / (np.sqrt(t) + 1.0)

        # Mirror descent step (exponential family)
        log_y = np.log(np.maximum(y, 1e-30)) - eta * grad
        log_y -= log_y.max()
        z = np.exp(log_y)
        z = np.maximum(z, 1e-30)
        z = z / z.sum()

        # Nesterov momentum
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = z + momentum * (z - y)
            x_new = np.maximum(x_new, 1e-30)
            x_new = x_new / x_new.sum()
        else:
            x_new = z.copy()

        c_new = fast_autocorr(x_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > stall_threshold:
            rng = np.random.default_rng(iter_count)
            pert = rng.normal(0, 0.01, n) * best_x
            x_restart = norm_sum1(best_x + pert)
            x_new = x_restart
            y = x_restart.copy()
            stall = 0
            stall_threshold = max(stall_threshold * 0.7, 1000)

        y = x_new
        x = x_new
        iter_count += 1

        if verbose and iter_count % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}")

    return best_x, best_m


def build_constraint_matrix(constraint_list, y, n):
    """Build the constraint matrix rows for the given constraints."""
    K = len(constraint_list)
    # For constraint k: (g*y)_k = sum_j g_j * y[k-j] <= M
    # So row k of the matrix has entries: (row k)_j = y[k-j] if 0<=k-j<n, else 0
    A = np.zeros((K, n))
    for ii, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            A[ii, idx] = y[k_int - idx]
    return A


def lp_frank_wolfe_improved(a0, time_limit, solver, 
                              initial_K=200, max_K=6000,
                              verbose=True):
    """
    Improved LP-based Frank-Wolfe with:
    - Growing constraint set
    - Predictive constraint addition
    - Armijo line search
    - Nesterov acceleration
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()  # fast iterate (Nesterov)
    
    size = 2 * n - 1
    
    # Initialize constraint set with top-K
    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()
    
    init_K = min(initial_K, size)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)
    
    iter_count = 0
    failed_solves = 0
    nesterov_beta = 0.0
    
    while time.monotonic() - t0 < time_limit:
        # Check time
        if time.monotonic() - t0 > time_limit:
            break
        
        # Update best
        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()
        
        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
        
        # Manage constraint set
        constraint_list = sorted(constraint_set)
        
        # Prune: remove constraints with too much slack
        if len(constraint_list) > initial_K:
            slacks = np.array([M_curr - ac_y[k] for k in constraint_list])
            # Keep constraints with slack < 0.5 * M_curr (i.e., within 50% of the max)
            # Plus always keep the top init_K
            keep_mask = slacks < 0.5 * M_curr
            # Also keep the top initial_K by value
            top_mask = np.zeros(len(constraint_list), dtype=bool)
            top_positions = np.argsort(ac_y[constraint_list])[::-1][:initial_K]
            top_mask[top_positions] = True
            keep_mask = keep_mask | top_mask
            constraint_set = set(int(constraint_list[i]) for i in range(len(constraint_list)) if keep_mask[i])
            constraint_list = sorted(constraint_set)
        
        # Cap at max_K
        if len(constraint_list) > max_K:
            # Keep the ones with smallest slack
            slacks = np.array([M_curr - ac_y[k] for k in constraint_list])
            keep = np.argsort(slacks)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            constraint_set = set(int(k) for k in init_idx[:100])
            constraint_list = sorted(constraint_set)
            K = len(constraint_list)
        
        # Build LP
        A_mat = build_constraint_matrix(constraint_list, y, n)
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A_mat @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 10:
                # Reset constraint set
                ac_y = fast_autocorr(y)
                top_K = min(200, size)
                top_idx = np.argsort(ac_y)[::-1][:top_K]
                constraint_set = set(int(k) for k in top_idx)
                constraint_list = sorted(constraint_set)
                failed_solves = 0

            continue
        
        failed_solves = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violated and near-violated constraints
        Yf = np.fft.rfft(y, 2 << (size - 1).bit_length())
        Gf = np.fft.rfft(g_dir, 2 << (size - 1).bit_length())
        conv_yg = np.fft.irfft(Yf * Gf, 2 << (size - 1).bit_length())[:size]

        if len(violations) > 0:

            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                constraint_set.add(int(k))

        # Only add a few to avoid explosion
        if len(near_bound) > 0:
            # Add the top 20 by value that aren't already in the set
            nb_sorted = near_bound[np.argsort(conv_yg[near_bound])[::-1]]
            added = 0
            for k in nb_sorted:
                if int(k) not in constraint_set and added < 20:
                    constraint_set.add(int(k))
                    added += 1
        
        # Armijo line search
        # Current objective at y: M_y = max_k (y*y)_k
        # Direction: d = g_dir - y
        # We want to find alpha in (0,1] such that:
        # max_k ((1-alpha)*y + alpha*g_dir * (y*y)_k <= M_curr * (1 - c*alpha*(M_curr - M_star))
        # Simpler: use the LP's M_var as a bound
        
        M_lp = max(0, float(M_var.value)) if M_var.value is not None else M_curr

        if iter_count == 0:
            alpha = 0.5
        else:
            alpha = 2.0 / (iter_count + 2.0)
        
        # FW step
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        # Nesterov acceleration
        if iter_count == 0:
            x_new = y_new.copy()
        else:
            nesterov_beta = (iter_count - 1.0) / (iter_count + 2.0)
            x_new = y_new + nesterov_beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, |S|={K}, M_lp={M_lp:.2e}")

    return best_a, best_m


def get_solver():
    """Find a working LP solver."""
    for s in [cp.CLARABEL, cp.PDLP, cp.MOSEK, cp.ECOS, cp.SCS, cp.SCS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): CLARABEL")
    return cp.CLARABEL


def parametric_init(n, best_ref=None):
    """Try diverse parametric families."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')

    candidates = []

    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr + 1.0) ** p)

    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.15, 0.2, 0.25, 0.3, 0.4]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))

    for rate in [1, 2, 3, 5]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1.0)))

    for freq in [1, 2, 3]:
        candidates.append(np.maximum(np.sin(np.pi * freq * x) ** 2, 0.01))

    # "Flat convolution" inspired: a_j ~ 1/sqrt(j*(n-1-j)) in the middle
    for w in [0.1, 0.2, 0.3, 0.5]:
        mid = (n - 1) / 2.0
        candidates.append(1.0 / (1.0 + w * ((i_arr - mid) / mid) ** 2))

    if best_ref is not None and len(best_ref) == n:
        ref_norm = norm_sum1(best_ref)
        for w in [0.3, 0.5, 0.7]:
            candidates.append(w * ref_norm + (1.0 - w) * norm_sum1((i_arr + 1.0) ** 0.7))

    for a_cand in candidates:
        a_norm = norm_sum1(a_cand)
        ac = fast_autocorr(a_norm)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_a = a_norm

    return best_a, best_val


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

   - Multiple LP phases with increasing constraint budgets
3. Final FISTA polishing
"""

import numpy as np
import time
import cvxpy as cp


def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum to 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_gradient(a, beta, top_k=None):
    """Compute softmax-weighted gradient for mirror descent."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]

    if top_k is not None and top_k < size:
        top_idx = np.argsort(c)[::-1][:top_k]
        c_top = c[top_idx]
        c_max = c_top.max()
        exp_c = np.zeros(size)
        exp_c[top_idx] = np.exp(beta * (c_top - c_max))
        w = exp_c / exp_c.sum()
    else:
        c_max = c.max()
        exp_c = np.exp(beta * (c - c_max))
        w = exp_c / exp_c.sum()

    a_rev = a[::-1]
    W = np.fft.rfft(w, fft_size)
    A_rev = np.fft.rfft(a_rev, fft_size)
    conv_wa = np.fft.irfft(W * A_rev, fft_size)
    grad = 2.0 * conv_wa[n - 1:2 * n - 1]

    return grad, c


def fista_optimize(a0, time_limit, beta_start=3.0, beta_end=80.0,
                   eta0=0.8, top_k=300, verbose=False):
    """FISTA with adaptive softmax gradient and restarts."""
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_threshold = 4000

    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        progress = min(iter_count / max(time_limit * 80, 1), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        grad, c_y = softmax_gradient(y, beta, top_k=top_k)

        eta = eta0 / (np.sqrt(t) + 1.0)

        # Mirror descent step (exponential family)
        log_y = np.log(np.maximum(y, 1e-30)) - eta * grad
        log_y -= log_y.max()
        z = np.exp(log_y)
        z = np.maximum(z, 1e-30)
        z = z / z.sum()

        # Nesterov momentum
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = z + momentum * (z - y)
            x_new = np.maximum(x_new, 1e-30)
            x_new = x_new / x_new.sum()
        else:
            x_new = z.copy()

        c_new = fast_autocorr(x_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > stall_threshold:
            rng = np.random.default_rng(iter_count)
            pert = rng.normal(0, 0.01, n) * best_x
            x_restart = norm_sum1(best_x + pert)
            x_new = x_restart
            y = x_restart.copy()
            stall = 0
            stall_threshold = max(int(stall_threshold * 0.7), 800)

        y = x_new
        x = x_new
        iter_count += 1

        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}")

    return best_x, best_m


def build_constraint_matrix(constraint_list, y, n):
    """Build the constraint matrix: row k has y[k-j] at position j."""
    K = len(constraint_list)
    A = np.zeros((K, n))
    for ii, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            A[ii, idx] = y[k_int - idx]
    return A


def lp_frank_wolfe_improved(a0, time_limit, solver,
                            initial_K=300, max_K=5000,
                            verbose=True):
    """
    Improved LP-based Frank-Wolfe with:
    - Growing constraint set with predictive addition
    - Slack-based pruning
    - Nesterov acceleration
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    size = 2 * n - 1
    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()

    init_K = min(initial_K, size)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0
    failed_solves = 0
    lp_solve_times = []

    while time.monotonic() - t0 < time_limit:
        # Update best from both iterates
        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()

        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        # Manage constraint set
        constraint_list = sorted(constraint_set)

        # Prune: keep constraints with meaningful slack
        if len(constraint_list) > initial_K * 2:
            slacks = np.array([M_curr - ac_y[k] for k in constraint_list])
            # Keep if slack < 70% of M_curr (i.e., constraint is somewhat tight)
            keep_mask = slacks < 0.7 * M_curr
            # Always keep top initial_K
            vals = ac_y[constraint_list]
            top_positions = np.argsort(vals)[::-1][:initial_K]
            top_mask = np.zeros(len(constraint_list), dtype=bool)
            top_mask[top_positions] = True
            keep_mask = keep_mask | top_mask
            constraint_set = set(int(constraint_list[i]) for i in range(len(constraint_list)) if keep_mask[i])
            constraint_list = sorted(constraint_set)

        # Cap at max_K
        if len(constraint_list) > max_K:
            slacks = np.array([M_curr - ac_y[k] for k in constraint_list])
            keep = np.argsort(slacks)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            init_idx_2 = np.argsort(ac_y)[::-1][:min(100, size)]
            constraint_set = set(int(k) for k in init_idx_2)
            constraint_list = sorted(constraint_set)
            K = len(constraint_list)

        # Build and solve LP
        t_lp_start = time.monotonic()
        A_mat = build_constraint_matrix(constraint_list, y, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A_mat @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 15:
                top_K = min(200, size)
                top_idx = np.argsort(ac_y)[::-1][:top_K]
                constraint_set = set(int(k) for k in top_idx)
                constraint_list = sorted(constraint_set)
                failed_solves = 0
            continue

        t_lp_end = time.monotonic()
        lp_solve_times.append(t_lp_end - t_lp_start)
        if len(lp_solve_times) > 100:
            lp_solve_times = lp_solve_times[-100:]

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            failed_solves += 1
            continue

        failed_solves = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violated and near-bound constraints for predictive addition
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:size]

        # Add violated constraints
        violations = np.where(conv_yg > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_yg[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                constraint_set.add(int(k))

        # Predictive: add near-bound constraints
        threshold = M_curr * 0.92
        near_bound = np.where((conv_yg > threshold) & (conv_yg <= M_curr + 1e-10))[0]
        if len(near_bound) > 0:
            nb_sorted = near_bound[np.argsort(conv_yg[near_bound])[::-1]]
            added = 0
            for k in nb_sorted:
                if int(k) not in constraint_set and added < 30:
                    constraint_set.add(int(k))
                    added += 1

        # Step size: standard FW with Nesterov
        if iter_count == 0:
            alpha = 0.5
        else:
            alpha = 2.0 / (iter_count + 2.0)

        # FW step on y
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Nesterov acceleration for x
        if iter_count == 0:
            x_new = y_new.copy()
        else:
            nesterov_beta = (iter_count - 1.0) / (iter_count + 2.0)
            x_new = y_new + nesterov_beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            avg_lp = np.mean(lp_solve_times) if lp_solve_times else 0
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, |S|={K}, lp_t={avg_lp:.2f}s")

    return best_a, best_m


def get_solver():
    """Find a working LP solver."""
    for s in [cp.CLARABEL, cp.PDLP, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): CLARABEL")
    return cp.CLARABEL


def parametric_init(n, best_ref=None):
    """Try diverse parametric families and return the best."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')

    candidates = []

    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr + 1.0) ** p)

    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))

    for rate in [1, 2, 3, 5, 8]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1.0)))

    for freq in [1, 2, 3, 4]:
        candidates.append(np.maximum(np.sin(np.pi * freq * x) ** 2, 0.01))

    # "Flat convolution" inspired shapes
    mid = (n - 1) / 2.0
    for w in [0.05, 0.1, 0.2, 0.3, 0.5, 1.0]:
        candidates.append(1.0 / (1.0 + w * ((i_arr - mid) / mid) ** 2))

    # Beta-like
    for alpha_p in [0.3, 0.5, 0.7, 1.0]:
        for beta_p in [0.3, 0.5, 0.7, 1.0]:
            candidates.append(np.maximum(i_arr + 0.5, 0.1) ** alpha_p *
                              np.maximum(n - i_arr - 0.5, 0.1) ** beta_p)

    if best_ref is not None and len(best_ref) == n:
        ref_norm = norm_sum1(best_ref)
        for w in [0.2, 0.4, 0.6, 0.8]:
            for p in [0.5, 0.7, 1.0]:
                candidates.append(w * ref_norm + (1.0 - w) * norm_sum1((i_arr + 1.0) ** p))

    for a_cand in candidates:
        a_norm = norm_sum1(a_cand)
        ac = fast_autocorr(a_norm)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_a = a_norm

    return best_a, best_val


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main optimization pipeline:
    1. Load parent / parametric init
    2. FISTA warmup (60s)
    3. LP-FW main phase (900s) with growing constraints
    4. LP-FW polish (350s) with larger constraint budget
    5. FISTA final polish (60s)
    6. Length exploration (remaining)
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_start = len(parent) if parent is not None else 5000
    if parent is None:
        best_seq = norm_sum1((np.arange(n_start, dtype=np.float64) + 1.0) ** 0.7)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_start}, score={best_score:.6f}")

    # === Phase 1: Parametric init (quick) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 25.0)
    print(f"\n=== Phase 1: Parametric init (t={t1:.0f}s) ===")
    if t1 > 5:
        a_param, _ = parametric_init(n_start, best_seq)
        check_and_update(a_param, "parametric")

    # === Phase 2: FISTA warmup ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 2: FISTA warmup (n={n_best}, t={t2:.0f}s) ===")
    if t2 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_optimize(a_in, t2, beta_start=3.0, beta_end=60.0,
                                  eta0=0.7, top_k=400, verbose=True)
        check_and_update(a_out, "fista_warmup")

    # === Phase 3: Main LP-FW (largest time allocation) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.50, 850.0)
    print(f"\n=== Phase 3: LP-FW main (n={n_best}, t={t3:.0f}s) ===")
    if t3 > 30 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_improved(a_in, t3, solver,
                                           initial_K=300, max_K=4000,
                                           verbose=True)
        check_and_update(a_out, "lp_main")

    # === Phase 4: LP-FW polish with larger constraint set ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 4: LP-FW polish (n={n_best}, t={t4:.0f}s) ===")
    if t4 > 30 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_improved(a_in, t4, solver,
                                           initial_K=800, max_K=7000,
                                           verbose=True)
        check_and_update(a_out, "lp_polish")

    # === Phase 5: FISTA final polish (higher beta) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.08, 60.0)
    print(f"\n=== Phase 5: FISTA final (t={t5:.0f}s) ===")
    if t5 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_optimize(a_in, t5, beta_start=30.0, beta_end=250.0,
                                  eta0=0.3, top_k=500, verbose=False)
        check_and_update(a_out, "fista_final")

    # === Phase 6: Short LP-FW one more time ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40 and n_best <= 8000:
        t6 = min(remaining * 0.15, 80.0)
        print(f"\n=== Phase 6: LP-FW final (t={t6:.0f}s) ===")
        if t6 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_improved(a_in, t6, solver,
                                               initial_K=1500, max_K=8000,
                                               verbose=False)
            check_and_update(a_out, "lp_final")

    # === Phase 7: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 80:
        print(f"\n=== Phase 7: Length exploration (remaining={remaining:.0f}s) ===")
        for delta in [200, -200, 400, -400]:
            new_n = n_best + delta
            if 4000 <= new_n <= 7000 and remaining > 50:
                a_long = interpolate_sequence(best_seq, new_n)
                t7 = min(remaining * 0.25, 70.0)
                if t7 > 15:
                    a_out, _ = fista_optimize(a_long, t7, beta_start=5.0, beta_end=80.0,
                                              eta0=0.7, top_k=300, verbose=False)
                    check_and_update(a_out, f"fista_n{new_n}")
                remaining = time_budget - (time.monotonic() - t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
