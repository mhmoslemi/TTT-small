            no_improve = 0

        if verbose and it % 10000 == 0:
            print(f"    IRLS it={it} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m


def compute_gradient_fft(a, conv, temp=0.01):
    """
    Compute soft-max weighted gradient using FFT for O(n log n) complexity.
    grad_j = sum_k w_k * a[k-j] = cross-correlation of w and a
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()

    # Compute soft-max weights
    max_conv = conv.max()
    if max_conv < 1e-30:
        return np.zeros(n)

    # Soft-max: w_k = exp(conv_k / (temp * max_conv)) / sum
    scaled = conv / (temp * max_conv)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum

    # Cross-correlation: grad_j = sum_k w_k * a[k-j]
    # This is (w * a)[j + n - 1] in the full convolution
    # Or equivalently: IFFT(FFT(w_padded) * FFT(a_padded))
    # More precisely: conv(w, a) at the right offset

    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)

    # cross-correlation = IFFT(conj(FFT(w)) * FFT(a))
    # Actually: (w * a)_k = sum_j w_j a_{k-j}
    # We want: grad_j = sum_k w_k a_{k-j} = (w * a)_{j} ... hmm
    # Let's think: sum_k w_k a_{k-j} = (a * w_rev)_j where w_rev is reversed
    # Or: = (w * a)_{?}

    # Actually: sum_k w_k a_{k-j} is the cross-correlation of w and a at lag j
    # = IFFT(conj(FFT(w)) * FFT(a))[j + offset]

    cross = np.fft.irfft(np.conj(W) * A, fft_size)
    # The cross-correlation at lag j is cross[j] (for j >= 0)
    # But we need to be careful about the offset
    # For sequences of length n zero-padded to fft_size:
    # cross[j] = sum_k w[k] a[k-j] for j = 0, ...
    # But we need sum_k w[k] a[k-j] where k-j is in [0, n-1]
    # This is cross[j] for j = 0, ..., n-1 (picking the right part)

    grad = cross[:n].copy()
    # Wait, I need to double-check the convention
    # np.correlate(w, a, mode='full')[n-1+j] = sum_k w[k] a[k-j+n-1]
    # Hmm, let me just verify: sum_k w[k]*a[k-j] for fixed j
    # = convolution of w with a at position ? 
    # (w*a)[m] = sum_k w[k] a[m-k]
    # We want sum_k w[k] a[k-j] = (w*a)[?] where m-k = k-j => m = 2k-j... no that's wrong
    # sum_k w[k] a[k-j]: let l = k-j, then k = l+j, sum_l w[l+j] a[l] = (w_rev * a)[j]
    # where w_rev[k] = w[n-1-k]
    # Actually this is getting complicated. Let me just use the direct FFT approach.

    # sum_k w[k] * a[k-j] = IFFT(conj(FFT(w)) * FFT(a)) evaluated at position j
    # This is the cross-correlation r_wa[j] = sum_k w[k+j] a[k]
    # Hmm, that's not quite what we want either.

    # Let me just be explicit: we want grad[j] = sum_{k: k-j in [0,n-1]} w[k] * a[k-j]
    # = sum_{l=0}^{n-1} w[l+j] * a[l] (where w is zero-padded)
    # This is the cross-correlation of w and a at lag j: r_wa[j]
    # r_wa[j] = sum_l w[l+j] a[l] = IFFT(conj(FW) * FA)[j] ... 
    # Actually: IFFT(conj(FW) * FA)[m] = sum_l FW^*[l] FA[l] e^{2pi iml/N} ... no
    # 
    # The standard result: if X = FFT(x), Y = FFT(y), then
    # IFFT(conj(X) * Y)[n] = sum_k x[k] y[k+n] (circular cross-correlation)
    # With zero-padding to fft_size >= 2n-1, this gives the linear cross-correlation.
    # So cross[j] = sum_k w[k] a[k+j] (linear, with zero-padding)
    # We want sum_k w[k] a[k-j] = cross[-j] = cross[fft_size - j]
    # OR: cross[j] with w and a swapped = sum_k a[k] w[k+j] = sum_l w[l] a[l-j] (letting l=k+j)
    # = sum_l w[l] a[l-j] which is what we want!

    # So: grad[j] = IFFT(FFT(a) * conj(FFT(w)))[j] for j = 0, ..., n-1
    # Let me redo:
    cross2 = np.fft.irfft(A * np.conj(W), fft_size)
    grad = cross2[:n].copy()

    # Clamp small negatives from FFT roundoff
    grad = np.maximum(grad, 0.0)
    return grad


def mirror_descent_fast(a0, time_limit, temp0=0.02, temp1=0.002, lr=3.0, seed=42, verbose=False):
    """
    Nesterov-accelerated mirror descent with FFT-based gradient computation.
    Uses adaptive softmax temperature for the max approximation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1).bit_length()

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 8000
    t_count = 0

    while time.monotonic() - t0 < time_limit:
        t_count += 1

        # Adaptive temperature
        progress = min(1.0, t_count / 30000.0)
        temp = temp0 * (temp1 / temp0) ** progress

        # Nesterov momentum
        beta_t = (t_count - 1.0) / (t_count + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-20)
        y = y / y.sum()

        # Compute autocorrelation and gradient
        A_fft = np.fft.rfft(y, fft_size)
        conv = np.fft.irfft(A_fft * A_fft, fft_size)[:2 * n - 1]

        # Soft-max gradient
        max_conv = conv.max()
        if max_conv < 1e-30:
            break
        scaled = conv / (temp * max_conv)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        w = w / w_sum

        # FFT-based gradient
        W_fft = np.fft.rfft(w, fft_size)
        grad = np.fft.irfft(A_fft * np.conj(W_fft), fft_size)[:n]
        grad = np.maximum(grad, 0.0)

        # Mirror descent update (exponential family)
        eta = lr / (np.sqrt(t_count) + 1.0)
        log_a = np.log(y) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        # Evaluate
        M = conv.max()
        M_x = np.fft.irfft(np.fft.rfft(x, fft_size) ** 2, fft_size)[:2 * n - 1].max()
        M_eff = min(M, M_x)
        if M_eff < best_m - 1e-16:
            best_m = M_eff
            best_a = (x if M_x <= M else y).copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            strength = 0.05
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-20)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

        if verbose and t_count % 5000 == 0:
            print(f"    MD t={t_count} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}, stall={stall}")

    return best_a, best_m


def coordinate_descent_fast(a0, time_limit, seed=42, verbose=False):
    """
    Coordinate descent: find the worst peak, identify the contributing pair,
    and redistribute mass to a low-cost location.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_autocorr(a)
        k_max = int(np.argmax(conv))

        # Find the pair (i, k_max-i) with maximum contribution at k_max
        j_lo = max(0, k_max - n + 1)
        j_hi = min(n - 1, k_max)
        if j_lo > j_hi:
            stall += 1
            continue

        idx = np.arange(j_lo, j_hi + 1)
        contrib = a[idx] * a[k_max - idx]
        src = int(idx[np.argmax(contrib)])
        src_val = a[src]

        if src_val < 1e-12:
            # Try random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.02, 0.1)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
            continue

        # Find low-cost destination using FFT-based cost computation
        # Cost of putting mass at position d: how much it contributes to top peaks
        top_k = np.argpartition(conv, -20)[-20:]
        top_k = top_k[top_k >= 0]

        # Compute cost vector: cost[d] = sum over top peaks of a[peak - d]
        costs = np.zeros(n)
        for k2 in top_k:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            if lo2 <= hi2:
                valid = np.arange(lo2, hi2 + 1)
                pair = k2i - valid
                costs[valid] += a[pair]

        # Pick destination with minimum (cost + current_value)
        combined = costs + 0.5 * a
        sorted_idx = np.argsort(combined)
        dst = int(sorted_idx[0])
        if dst == src:
            dst = int(sorted_idx[1]) if len(sorted_idx) > 1 else int(rng.integers(n))

        # Try several fractions
        moved = False
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            M_t = fast_autocorr(a_trial).max()
            if M_t < best_m - 1e-16:
                a = a_trial
                best_m = M_t
                best_a = a.copy()
                moved = True
                stall = 0
                break

        if not moved:
            stall += 1
            # Random perturbation for exploration
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.01, 0.05)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def lp_direction(a0, M_limit, seed=42):
    """
    Find a direction g that maximizes sum(g) subject to g >= 0 and (a0*g)_k <= M_limit.
    Uses a greedy coordinate approach as an approximation to the LP.
    """
    n = len(a0)
    f = np.asarray(a0, dtype=np.float64)
    g = np.zeros(n)

    # Greedy: repeatedly add mass to the coordinate that allows the most
    # while keeping (f*g)_k <= M_limit
    conv_fg = fast_autocorr(f)  # This is f*f, we need f*g
    # Actually we need f*g, not f*f. Let me compute incrementally.

    # For each coordinate j, the contribution to (f*g)_k is f[k-j]
    # So adding delta to g[j] adds delta * f[k-j] to (f*g)_k for all k
    # The constraint is: (f*g)_k + delta * f[k-j] <= M_limit for all k
    # Max delta for coordinate j: min_k (M_limit - (f*g)_k) / f[k-j] over k where f[k-j] > 0

    fg = np.zeros(2 * n - 1)
    total_added = 0.0
    max_iters = 500
    batch_size = 50

    for it in range(max_iters):
        # For each candidate coordinate, compute max increment
        best_j = -1
        best_delta = 0.0

        # Sample a subset of coordinates for efficiency
        candidates = np.random.default_rng(seed + it).choice(n, size=min(batch_size, n), replace=False)

        for j in candidates:
            # Adding delta to g[j] changes (f*g) by delta * f shifted
            # (f*g)_k gets delta * f[k-j] added
            # We need: fg[k] + delta * f[k-j] <= M_limit for all k
            # Max delta = min_{k: f[k-j] > 0} (M_limit - fg[k]) / f[k-j]

            # The shift: f[k-j] is non-zero for k in [j, j+n-1]
            k_lo = j
            k_hi = j + n - 1
            k_lo = max(k_lo, 0)
            k_hi = min(k_hi, 2 * n - 2)

            if k_lo > k_hi:
                delta_j = 1e10
            else:
                # f[k-j] for k in [k_lo, k_hi] is f[k_j_lo], ..., f[k_hi-j]
                f_vals = f[k_lo - j:k_hi - j + 1]
                fg_vals = fg[k_lo:k_hi + 1]
                positive = f_vals > 1e-15
                if not np.any(positive):
                    delta_j = 1e10
                else:
                    available = (M_limit - fg_vals[positive]) / f_vals[positive]
                    delta_j = np.max(0.0, np.min(available))

            if delta_j > best_delta:
                best_delta = delta_j
                best_j = int(j)

        if best_j < 0 or best_delta < 1e-15:
            break

        # Add to g
        g[best_j] += best_delta * 0.5  # Use half the max for safety
        total_added += best_delta * 0.5

        # Update fg = f * g
        fg = np.convolve(f, g)[:2 * n - 1]

        if total_added > 1e-3:
            break

    if total_added < 1e-10:
        return None

    g = g / g.sum()
    return g


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search function: multi-phase optimization combining HIO spectral flattening,
    FFT-based mirror descent, LP-inspired direction finding, and coordinate descent.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = safe_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = safe_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Generate initial candidates
    candidates = []

    if parent is not None:
        candidates.append((parent.copy(), "parent"))

    # Various initialization shapes
    n_base = len(parent) if parent is not None else 6000
    for alpha in [0.8, 1.0, 1.5, 2.0]:
        candidates.append((make_ramp(n_base, alpha), f"ramp_a{alpha}"))
    candidates.append((make_uniform(n_base), "uniform"))
    candidates.append((make_gaussian(n_base), "gaussian"))
    candidates.append((make_cosine_taper(n_base), "cosine"))

    # Different lengths
    for n_try in [n_base - 1000, n_base + 1000, n_base + 2000]:
        if n_try >= 2000:
            candidates.append((make_ramp(n_try, 1.0), f"ramp_n{n_try}"))
            candidates.append((make_uniform(n_try), f"uniform_n{n_try}"))

    print(f"Initial candidates: {len(candidates)}")

    # Evaluate all candidates
    scored = []
    for a, label in candidates:
        sc = safe_eval(a)
        if np.isfinite(sc):
            scored.append((sc, a, label))
            print(f"  Init [{label}]: {sc:.6f}")
    scored.sort(key=lambda x: x[0])

    # Phase 1: HIO Spectral Flattening on top candidates
    print(f"\n=== Phase 1: HIO Spectral Flattening ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_cands_p1 = min(4, len(scored))
    t_per_p1 = max(30.0, min(250.0, remaining * 0.35 / max(1, n_cands_p1)))

    for i in range(n_cands_p1):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget:
            break
        t_avail = min(t_per_p1, time_budget - elapsed)
        if t_avail < 15:
            break
        sc0, a0, label = scored[i]
        print(f"  HIO [{label}] (t={t_avail:.0f}s, init={sc0:.6f})...")
        a_out, _ = spectral_hio(a0, t_avail, seed=42 + i * 7, verbose=False)
        try_update(a_out, f"hio_{label}")

    # Phase 2: IRLS Spectral Flattening
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: IRLS Spectral Flattening ===")
    if remaining > 60 and best_seq is not None:
        t2 = min(remaining * 0.12, 150.0)
        a_in = best_seq / best_seq.sum()
        a_out, _ = spectral_irls(a_in, t2, seed=99, verbose=False)
        try_update(a_out, "irls")

    # Phase 3: Mirror Descent (FFT-based gradient)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Mirror Descent ===")
    if remaining > 60 and best_seq is not None:
        n_cands_p3 = min(3, len(scored))
        t_per_p3 = max(20.0, min(200.0, remaining * 0.25 / max(1, n_cands_p3)))

        for i in range(n_cands_p3):
            elapsed = time.monotonic() - t_global
            if elapsed > time_budget:
                break
            t_avail = min(t_per_p3, time_budget - elapsed)
            if t_avail < 15:
                break
            a_in = scored[i][1] / scored[i][1].sum()
            label = scored[i][2]
            print(f"  MD [{label}] (t={t_avail:.0f}s)...")
            a_out, _ = mirror_descent_fast(a_in, t_avail, temp0=0.02, temp1=0.003,
                                           lr=3.0, seed=100 + i * 13, verbose=False)
            try_update(a_out, f"md_{label}")

        # MD on current best
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t3b = min(remaining * 0.15, 120.0)
            a_in = best_seq / best_seq.sum()
            print(f"  MD [best] (t={t3b:.0f}s)...")
            a_out, _ = mirror_descent_fast(a_in, t3b, temp0=0.015, temp1=0.002,
                                           lr=2.5, seed=777, verbose=False)
            try_update(a_out, "md_best")

    # Phase 4: LP-inspired direction + blend
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Direction + Blend ===")
    if remaining > 30 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        conv_in = fast_autocorr(a_in)
        M_in = conv_in.max()

        # Try LP direction finding
        t_lp = min(remaining * 0.1, 60.0)
        if t_lp > 10:
            t_lp_start = time.monotonic()
            while time.monotonic() - t_lp_start < t_lp:
                g = lp_direction(a_in, M_in, seed=int(rng_state := (time.monotonic() * 1000) % 100000))
                if g is not None:
                    # Blend
                    for t_blend in [0.1, 0.2, 0.3, 0.5]:
                        a_blend = (1 - t_blend) * a_in + t_blend * g
                        a_blend = np.maximum(a_blend, 0.0)
                        a_blend = a_blend / a_blend.sum()
                        try_update(a_blend, f"lp_blend_{t_blend}")
                        if a_blend.sum() > 0 and safe_eval(a_blend) < best_score:
                            a_in = a_blend / a_blend.sum()
                            conv_in = fast_autocorr(a_in)
                            M_in = conv_in.max()
                            break

    # Phase 5: HIO on best (second pass with different params)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: HIO Second Pass ===")
    if remaining > 40 and best_seq is not None:
        t5 = min(remaining * 0.20, 200.0)
        a_in = best_seq / best_seq.sum()
        print(f"  HIO [best, different params] (t={t5:.0f}s)...")
        a_out, _ = spectral_hio(a_in, t5, alpha0=0.5, alpha1=0.99, seed=12345, verbose=False)
        try_update(a_out, "hio2")

    # Phase 6: Coordinate Descent Polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Coordinate Descent ===")
    if remaining > 30 and best_seq is not None:
        t6 = min(remaining * 0.15, 120.0)
        a_in = best_seq / best_seq.sum()
        print(f"  Coord (t={t6:.0f}s)...")
        a_out, _ = coordinate_descent_fast(a_in, t6, seed=42, verbose=False)
        try_update(a_out, "coord1")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6b = min(remaining * 0.1, 80.0)
            a_out, _ = coordinate_descent_fast(a_in, t6b, seed=999, verbose=False)
            try_update(a_out, "coord2")

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Multi-length ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for new_n in [n_best + 500, n_best + 1500, n_best - 500, n_best + 3000]:
            if new_n < 2000 or new_n > 15000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            a_interp = interpolate_to_length(best_seq, new_n)
            try_update(a_interp, f"interp_n{new_n}")

            t7 = min(remaining * 0.2, 100.0)
            if t7 > 15:
                a_out, _ = spectral_hio(a_interp, t7, seed=200 + new_n % 100, verbose=False)
                try_update(a_out, f"hio_n{new_n}")

    # Phase 8: Final combined polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final Polish ===")
    if remaining > 30 and best_seq is not None:
        a_in = best_seq / best_seq.sum()

        # HIO + MD pipeline
        t8a = min(remaining * 0.3, 60.0)
        t8b = min(remaining * 0.4, 80.0)

        if t8a > 10:
            a_out, _ = spectral_hio(a_in, t8a, alpha0=0.6, alpha1=0.99, seed=555, verbose=False)
            try_update(a_out, "final_hio")
            a_in = a_out / max(a_out.sum(), 1e-15)

        if t8b > 10:
            a_out, _ = mirror_descent_fast(a_in, t8b, temp0=0.01, temp1=0.001,
                                           lr=2.0, seed=7777, verbose=False)
            try_update(a_out, "final_md")

    # Phase 9: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15 and best_seq is not None:
        t9 = min(remaining * 0.4, 60.0)
        a_in = best_seq / best_seq.sum()
        a_out, _ = coordinate_descent_fast(a_in, t9, seed=31415, verbose=False)
        try_update(a_out, "final_coord")

    # Final validation
    if best_seq is not None:
        fs = safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        best_seq = make_ramp(6000, 1.0)
        best_score = safe_eval(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
