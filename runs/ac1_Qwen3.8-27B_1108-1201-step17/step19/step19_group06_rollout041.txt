        res = linprog(c, A_ub=A, b_ub=b_ub, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None and res.x.sum() > 1e-15:
            return res.x
    except Exception:
        pass
    return None


def generate_initialization(n, method='sine'):
    """Generate initial sequences with different window shapes."""
    x = np.linspace(0, 1, n)
    if method == 'uniform':
        return np.ones(n) / n
    elif method == 'sine':
        return norm_sum1(np.sin(np.pi * (np.arange(n) + 0.5) / n))
    elif method == 'raised_cosine':
        return norm_sum1(0.5 * (1 + np.cos(2 * np.pi * (np.arange(n) + 0.5) / n)))
    elif method == 'tapered':
        j = np.arange(n, dtype=np.float64)
        edge = min(n // 4, 500)
        w = np.ones(n)
        w[:edge] = 0.5 * (1 + np.cos(np.pi * (edge - np.arange(edge)) / edge))
        w[-edge:] = 0.5 * (1 + np.cos(np.pi * (edge - np.arange(edge)[::-1]) / edge))
        return norm_sum1(w)
    elif method == 'gaussian':
        c = 0.5
        sig = 0.3
        return norm_sum1(np.exp(-((x - c) ** 2) / (2 * sig ** 2)))
    elif method == 'bimodal':
        return norm_sum1(np.exp(-((x - 0.3) ** 2) / 0.04) + np.exp(-((x - 0.7) ** 2) / 0.04))
    elif method == 'power':
        return norm_sum1((np.arange(n) + 1.0) ** 0.5)
    elif method == 'random_smooth':
        rng = np.random.default_rng(42)
        coarse = rng.uniform(0.5, 1.5, max(n // 50, 10))
        fine = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(coarse)), coarse)
        return norm_sum1(fine)
    else:
        return np.ones(n) / n


def lp_optimize(f0, time_budget, n, verbose=True):
    """Main LP-based optimization loop with adaptive stepping."""
    f = norm_sum1(f0)
    best_f = f.copy()
    best_score = score_from_ac(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    t_step = 0.25
    lp_count = 0

    if verbose:
        print(f"  LP start: score={best_score:.6f}")

    for it in range(100000):
        elapsed = time.monotonic() - t0
        if elapsed > time_budget:
            break

        ac = fast_autocorr(f)
        M = ac.max()

        # Solve LP (with decreasing time limit as we converge)
        remaining = time_budget - elapsed
        t_lp = min(15.0, max(3.0, remaining * 0.01))
        b_opt = solve_lp_direction(f, M, n, time_limit=t_lp)

        if b_opt is None:
            stall += 1
            if stall > 10:
                f = norm_sum1(best_f * (1 + rng.normal(0, 0.03, n)))
                stall = 0
            continue

        lp_count += 1
        g = norm_sum1(b_opt)

        # Check meaningful difference
        diff = np.abs(g - f).max()
        if diff < 1e-5:
            stall += 1
            if stall > 3:
                # Perturb to escape
                noise_scale = min(0.05, 0.01 + 0.005 * stall)
                f = norm_sum1(f + rng.normal(0, noise_scale, n))
                t_step = 0.1
            continue

        # Backtracking line search
        improved = False
        for t in [t_step, t_step * 0.5, t_step * 0.25, t_step * 0.125, 0.05]:
            f_trial = norm_sum1((1 - t) * f + t * g)
            s = score_from_ac(f_trial)
            if s < best_score - 1e-10:
                f = f_trial
                best_score = s
                best_f = f_trial.copy()
                improved = True
                stall = 0
                t_step = min(t * 1.2, 0.5)
                break

        if not improved:
            stall += 1
            t_step = max(t_step * 0.7, 0.02)
            if stall > 30:
                f = norm_sum1(best_f + rng.normal(0, 0.02, n))
                stall = 0
                t_step = 0.15

        if verbose and (it % 20 == 0 or it < 5):
            print(f"  LP iter {it} ({elapsed:.0f}s, {lp_count} LPs): score={best_score:.6f}")

    if verbose:
        print(f"  LP done ({time.monotonic()-t0:.0f}s, {lp_count} LPs): score={best_score:.6f}")
    return best_f, best_score


def mirror_descent(a0, time_limit, lr_init=0.08, beta_init=3.0, top_k_init=40,
                   restart_every=3000, verbose=False, rng_seed=42):
    """Mirror descent with soft-max gradient over top autocorrelation lags."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    a_prev = a.copy()
    momentum = 0.8
    lr = lr_init
    beta = beta_init
    top_k = top_k_init
    stall = 0

    for t in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ratio = a / np.maximum(a_prev, 1e-30)
        ratio = np.clip(ratio, 0.01, 100.0)
        a_nest = a * np.power(ratio, momentum)
        a_nest = norm_sum1(a_nest)

        ac = fast_autocorr(a_nest)
        k_alarm = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_alarm]
        top_vals = ac[top_idx]
        max_v = top_vals.max()
        expo = beta * (top_vals - max_v)
        expo = np.clip(expo, -50, 50)
        weights = np.exp(expo)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones(k_alarm) / k_alarm
        else:
            weights = weights / w_sum

        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k_int = int(k)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo >= i_hi:
                continue
            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            valid = (j_idx >= 0) & (j_idx < n)
            grad[idx[valid]] += 2.0 * w * a_nest[j_idx[valid]]

        eta = lr / (1.0 + 0.03 * np.sqrt(t))
        log_a = np.log(np.maximum(a_nest, 1e-30)) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        a_prev = a
        a = a_new

        ac_new = fast_autocorr(a)
        m_new = ac_new.max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if t % 500 == 0:
            top_k = min(top_k + 3, 120)

        if stall > restart_every:
            stall = 0
            pert = norm_sum1(best_a * (1 + rng.normal(0, 0.05, n)))
            a = norm_sum1(0.7 * best_a + 0.3 * pert)
            a_prev = a.copy()
            lr *= 0.85
            beta = max(beta - 0.3, 1.5)

    if verbose:
        print(f"  MD done ({time.monotonic()-t0:.0f}s): score={2*n*best_m:.6f}")
    return best_a, best_m


def coordinate_descent(a0, time_limit, rng_seed=77, verbose=False):
    """Batch coordinate descent for fine-tuning."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-13:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break

        n_top = min(20, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k_pos in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k_pos)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo >= i_hi:
                continue
            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            n_pairs = min(5, len(idx))
            pair_order = np.argsort(contrib)[::-1][:n_pairs]

            for pi in pair_order:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[pi])
                src_val = a[src]
                if src_val < 1e-14:
                    continue

                # Find low-impact destination
                scores = np.zeros(n)
                for k2 in top_lags[:6]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n, k2i + 1)
                    if s2 < e2:
                        ii = np.arange(s2, e2)
                        jj = k2i - ii
                        scores[ii] += a[jj] * ac[k2i]
                        scores[jj] += a[ii] * ac[k2i]
                scores[src] = np.inf
                dst = int(np.argmin(scores))

                if dst == src:
                    continue

                for frac in [0.03, 0.08, 0.15, 0.25]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and stall % 200 == 0:
            n_moves = int(rng.integers(3, 10))
            for _ in range(n_moves):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = float(rng.uniform(0.01, 0.05))
                    a[i] -= a[i] * frac
                    a[j] += a[i] * frac * 0.5
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    if verbose:
        print(f"  Coord done ({time.monotonic()-t0:.0f}s): score={2*n*best_m:.6f}")
    return best_a, best_m


def spectral_flux(a0, time_limit, lr_init=0.2, verbose=False):
    """Spectral domain optimization to flatten autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    fft_size = 1 << (2 * n - 2).bit_length()
    lr = lr_init
    stall = 0
    beta = 2.0

    for t in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        if P_max < 1e-30:
            break

        # Penalize spectral peaks to flatten autocorrelation
        P_dev = P - P.mean()
        expo = beta * P_dev / max(P_max, 1e-30)
        expo = np.clip(expo, -15, 15)
        w_spec = np.exp(expo)
        A_new = A * np.exp(-0.3 * lr / (1 + 0.1 * np.sqrt(t)) * (w_spec - 1.0) / max(w_spec.mean(), 1e-10))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = norm_sum1(a_new)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 2000:
                a = norm_sum1(best_a * (1 + rng.normal(0, 0.02, n)))
                lr *= 0.8
                beta = max(beta - 0.3, 0.5)
                stall = 0

    if verbose:
        print(f"  Spectral done: score={2*n*best_m:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator using LP-based optimization with multiple
    strategies: LP iteration, mirror descent, coordinate descent, spectral
    filtering, and length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 40.0, 1760.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence if available
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 3000
        best_seq = generate_initialization(n0, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = evaluate_safe(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")

    # === Phase 1: Best initialization search ===
    elapsed = time.monotonic() - t_global
    print(f"\n=== Phase 1: Initialization (t={elapsed:.0f}s) ===")
    n0 = len(best_seq)

    init_methods = ['sine', 'raised_cosine', 'tapered', 'bimodal', 'random_smooth', 'gaussian']
    for method in init_methods:
        if time.monotonic() - t_global > time_budget * 0.05:
            break
        init = generate_initialization(n0, method)
        check_update(init, f"init_{method}")

    # Try parent length if different
    if parent_loaded and n0 != len(best_seq):
        init_p = generate_initialization(len(best_seq), 'sine')
        check_update(init_p, f"init_sine_n{len(best_seq)}")

    n_active = len(best_seq)

    # === Phase 2: LP-based optimization (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP optimization (n={n_active}, budget={remaining:.0f}s) ===")

    # Choose n for LP (keep it reasonable for solve time)
    n_lp = n_active
    if n_lp > 4000:
        n_lp = 4000
    if n_lp < 2000:
        n_lp = 2000

    # Interpolate to LP length if needed
    f_for_lp = best_seq.copy() if len(best_seq) == n_lp else interpolate_sequence(best_seq, n_lp)

    t_lp_budget = min(remaining * 0.55, 900.0)
    if t_lp_budget > 60:
        a_out, _ = lp_optimize(f_for_lp, t_lp_budget, n_lp, verbose=True)
        check_update(a_out, f"lp_n{n_lp}")

    # Second LP pass with different init
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        t2 = min(remaining * 0.3, 300.0)
        f_init2 = generate_initialization(n_lp, 'tapered')
        a_out, _ = lp_optimize(f_init2, t2, n_lp, verbose=False)
        check_update(a_out, f"lp2_n{n_lp}")

    # === Phase 3: Mirror descent refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Mirror descent (n={n_best}, budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_out, _ = mirror_descent(best_seq, t3, lr_init=0.05, beta_init=4.0,
                                  top_k_init=60, restart_every=2000, verbose=True)
        check_update(a_out, f"md_n{n_best}")

    # === Phase 4: Spectral flux ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Spectral (n={n_best}, budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.08, 120.0)
    if t4 > 20:
        a_out, _ = spectral_flux(best_seq, t4, lr_init=0.15, verbose=False)
        check_update(a_out, f"spectral_n{n_best}")

    # === Phase 5: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Coord descent (n={n_best}, budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 15:
        a_out, _ = coordinate_descent(best_seq, t5, rng_seed=42, verbose=True)
        check_update(a_out, f"coord_n{n_best}")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration (budget={remaining:.0f}s) ===")

    length_tries = []
    for delta in [-500, -200, 200, 500, 1000]:
        nn = n_best + delta
        if 1500 <= nn <= 6000 and nn != n_best:
            length_tries.append(nn)
    for rn in [2000, 3000, 4000, 5000]:
        if rn != n_best and 1500 <= rn <= 6000:
            length_tries.append(rn)
    length_tries = sorted(set(length_tries))

    for new_n in length_tries[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")

        a_interp = interpolate_sequence(best_seq, new_n)
        check_update(a_interp, f"interp_n{new_n}")

        # Quick MD on new length
        t_md = min(remaining * 0.3, 100.0)
        if t_md > 20:
            a_out, _ = mirror_descent(a_interp, t_md, lr_init=0.04, beta_init=3.0,
                                      top_k_init=50, restart_every=2000, verbose=False)
            check_update(a_out, f"md_n{new_n}")

        # Quick coord descent
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t_cd = min(remaining * 0.3, 80.0)
            if t_cd > 10:
                a_out, _ = coordinate_descent(best_seq, t_cd, rng_seed=new_n % 1000, verbose=False)
                check_update(a_out, f"coord_n{new_n}")

    # === Phase 7: Final intensive polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_final}, budget={remaining:.0f}s) ===")

    # Fine MD
    t_fin = min(remaining * 0.3, 150.0)
    if t_fin > 20:
        a_out, _ = mirror_descent(best_seq, t_fin, lr_init=0.02, beta_init=6.0,
                                  top_k_init=100, restart_every=1500, verbose=False)
        check_update(a_out, f"final_md_n{n_final}")

    # Final coord
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t_cd = min(remaining * 0.4, 100.0)
        if t_cd > 10:
            a_out, _ = coordinate_descent(best_seq, t_cd, rng_seed=999, verbose=False)
            check_update(a_out, f"final_coord_n{n_final}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        best_seq = generate_initialization(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
