    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_m = fft_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    lr = lr_init
    beta = beta_init
    K = K_init
    stall = 0
    total_steps = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fft_autocorr(a)
        M = c.max()
        
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive parameters based on progress
        progress = min(1.0, total_steps / 20000.0)
        beta = beta_init + (beta_final - beta_init) * progress
        K = int(K_init + (K_final - K_init) * progress)
        
        # Compute gradient
        grad = compute_soft_gradient(a, c, beta, K)
        
        # Backtracking line search
        lr_cur = lr
        improved = False
        for attempt in range(8):
            eta = lr_cur / (1.0 + 0.03 * np.sqrt(total_steps + 1))
            # Multiplicative update
            log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            a_new = np.maximum(a_new, 1e-15)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new / s
            
            c_new = fft_autocorr(a_new)
            M_new = c_new.max()
            
            if M_new < M - 1e-16:
                a = a_new
                improved = True
                break
            else:
                lr_cur *= 0.5
        
        if not improved:
            # Accept the best trial
            a = a_new
            lr *= 0.8
        
        total_steps += 1
        
        # Restart when stuck
        if stall > restart_every:
            stall = 0
            # Informed restart: perturb in gradient direction
            perturb_type = int(rng.integers(5))
            if perturb_type == 0:
                x = np.linspace(0, 1, n)
                c_pos = float(rng.uniform(0.2, 0.8))
                sig = float(rng.uniform(0.1, 0.4))
                pert = np.exp(-((x - c_pos)**2) / (2 * sig**2))
            elif perturb_type == 1:
                i_arr = np.arange(n, dtype=np.float64)
                p = float(rng.uniform(0.3, 2.5))
                pert = (i_arr + 1.0) ** p
            elif perturb_type == 2:
                x = np.linspace(0, 1, n)
                freq = int(rng.integers(1, 6))
                phase = float(rng.uniform(0, 2 * np.pi))
                pert = np.sin(np.pi * freq * x + phase)**2
            elif perturb_type == 3:
                pert = best_a * (1.0 + rng.normal(0, 0.08, n))
            else:
                # Use gradient direction (uphill in smoothed objective)
                pert = best_a * (1.0 + 0.1 * grad / max(grad.max(), 1e-10))
                pert = np.maximum(pert, 0)
            
            pert = normalize_nonneg(pert)
            w = float(rng.uniform(0.1, 0.5))
            a = normalize_nonneg((1 - w) * best_a + w * pert)
            lr = max(lr * 0.7, 0.01)
            
            if verbose:
                J = 2.0 * n * best_m
                print(f"      MD restart t={total_steps} ({time.monotonic()-t0:.0f}s): J={J:.6f}, beta={beta:.1f}, K={K}")
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"      MD done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def spectral_equalize(a0, time_limit, alpha=0.1, verbose=False):
    """
    Iteratively modify the spectrum to flatten the autocorrelation.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    
    best_a = a.copy()
    best_m = fft_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        A = np.fft.rfft(a, N)
        P = np.abs(A)**2
        
        # Current autocorrelation
        c = np.fft.irfft(P, N)[:size]
        M = c.max()
        
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                a = normalize_nonneg(best_a * (1.0 + rng.normal(0, 0.02, n)))
                stall = 0
                alpha *= 0.8
                continue
        
        # Find peak lag
        k_peak = int(np.argmax(c))
        
        # Modify spectrum to reduce the peak
        # The peak at lag k_peak means there's coherence at frequency related to k_peak
        # We reduce P at frequencies that contribute to this peak
        freqs = np.arange(len(P)) * k_peak
        phase_contribution = np.cos(2.0 * np.pi * freqs / N)
        
        # Damp P where it contributes positively to the peak
        damp = 1.0 - alpha * np.maximum(phase_contribution, 0)
        damp = np.maximum(damp, 0.5)  # Don't over-damp
        
        A_new = A * np.sqrt(damp)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = normalize_nonneg(a_new)
        
        c_new = fft_autocorr(a_new)
        if c_new.max() < M:
            a = a_new
        else:
            # Try the opposite: boost where it contributes negatively
            damp2 = 1.0 + alpha * np.maximum(-phase_contribution, 0)
            A_new = A * np.sqrt(damp2)
            a_new = np.fft.irfft(A_new, N)[:n]
            a_new = normalize_nonneg(a_new)
            c_new = fft_autocorr(a_new)
            if c_new.max() < M:
                a = a_new
            else:
                alpha *= 0.9
                stall += 1
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"      Spectral done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def coordinate_descent(a0, time_limit, rng_seed=77, verbose=False):
    """
    Fine-grained coordinate descent: try all pairs for mass transfer.
    Uses top lags to evaluate which transfers are most promising.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_m = fft_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    size = 2 * n - 1
    
    while time.monotonic() - t0 < time_limit:
        c = fft_autocorr(a)
        M = c.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30000:
                break
        
        # Get top lags for evaluation
        n_top = min(20, size)
        top_lags = np.argsort(c)[::-1][:n_top]
        
        # Try random pairs of positions
        n_trials = 200
        improved = False
        
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break
            
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-15:
                continue
            
            # Try transferring small fraction
            fracs = [0.005, 0.01, 0.02, 0.05, 0.1]
            best_frac = None
            best_M_trial = M
            
            for frac in fracs:
                amt = a[i] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                
                c_trial = fft_autocorr(a_trial)
                M_trial = c_trial.max()
                
                if M_trial < best_M_trial:
                    best_M_trial = M_trial
                    best_frac = (i, j, frac)
            
            if best_frac is not None and best_M_trial < M - 1e-15:
                i, j, frac = best_frac
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                M = best_M_trial
                improved = True
                if M < best_m - 1e-15:
                    best_m = M
                    best_a = a.copy()
                    stall = 0
        
        if not improved:
            # Random perturbation to escape
            if stall % 100 == 0:
                n_moves = int(rng.integers(3, 10))
                for _ in range(n_moves):
                    src = int(rng.integers(n))
                    dst = int(rng.integers(n))
                    if src != dst and a[src] > 1e-12:
                        amt = a[src] * float(rng.uniform(0.001, 0.01))
                        a[src] -= amt
                        a[dst] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                c_new = fft_autocorr(a)
                if c_new.max() < best_m:
                    best_m = c_new.max()
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"      Coord descent done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def targeted_coord_descent(a0, time_limit, rng_seed=55, verbose=False):
    """
    More targeted coordinate descent: focus on positions that contribute
    to the highest autocorrelation lags.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_m = fft_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    size = 2 * n - 1
    
    while time.monotonic() - t0 < time_limit:
        c = fft_autocorr(a)
        M = c.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20000:
                break
        
        # Identify the peak lag and contributing positions
        k_peak = int(np.argmax(c))
        i_lo = max(0, k_peak - n + 1)
        i_hi = min(n, k_peak + 1)
        peak_positions = set(range(i_lo, i_hi))
        
        # Also get second peak
        c_copy = c.copy()
        c_copy[k_peak] = -np.inf
        k_second = int(np.argmax(c_copy))
        i_lo2 = max(0, k_second - n + 1)
        i_hi2 = min(n, k_second + 1)
        peak_positions.update(range(i_lo2, i_hi2))
        
        # Try transfers involving peak positions
        improved = False
        n_peak = len(peak_positions)
        
        for trial in range(300):
            if time.monotonic() - t0 > time_limit:
                break
            
            if rng.random() < 0.6:
                # Source from peak position
                src = int(rng.choice(list(peak_positions)))
                dst = int(rng.integers(n))
            else:
                src = int(rng.integers(n))
                dst = int(rng.choice(list(peak_positions)))
            
            if src == dst or a[src] < 1e-15:
                continue
            
            # Try transfer
            frac = float(rng.uniform(0.001, 0.05))
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            
            c_trial = fft_autocorr(a_trial)
            M_trial = c_trial.max()
            
            if M_trial < M - 1e-15:
                a = a_trial
                M = M_trial
                if M < best_m:
                    best_m = M
                    best_a = a.copy()
                improved = True
        
        if not improved and stall > 500:
            # Perturb
            n_moves = int(rng.integers(3, 8))
            for _ in range(n_moves):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    amt = a[i] * float(rng.uniform(0.001, 0.02))
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0)
            a = a / a.sum()
            c_new = fft_autocorr(a)
            if c_new.max() < best_m:
                best_m = c_new.max()
                best_a = a.copy()
                stall = 0
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"      Targeted CD done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def parametric_init(n, time_limit=15.0):
    """Generate and evaluate parametric initializations."""
    t0 = time.monotonic()
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    
    candidates = []
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(normalize_nonneg((i_arr + 1.0) ** p))
        candidates.append(normalize_nonneg((n - i_arr) ** p))
    
    # Gaussians
    for c_pos in [0.25, 0.5, 0.75]:
        for sig in [0.05, 0.1, 0.2, 0.3, 0.5]:
            candidates.append(normalize_nonneg(np.exp(-((x - c_pos)**2) / (2 * sig**2))))
    
    # Bimodal
    for sig in [0.05, 0.1, 0.15, 0.2]:
        candidates.append(normalize_nonneg(
            np.exp(-((x - 0.3)**2) / (2*sig**2)) + np.exp(-((x - 0.7)**2) / (2*sig**2))))
    
    # Sine-squared
    for f in [1, 2, 3, 4, 5]:
        candidates.append(normalize_nonneg(np.sin(np.pi * f * x)**2))
    
    # Exponential
    for r in [0.5, 1, 2, 5, 10]:
        candidates.append(normalize_nonneg(np.exp(-r * x)))
        candidates.append(normalize_nonneg(np.exp(-r * (1 - x))))
    
    # W-shape
    for w in [0.1, 0.2, 0.3]:
        candidates.append(normalize_nonneg(
            np.exp(-((x - 0.2)**2) / w**2) + np.exp(-((x - 0.8)**2) / w**2) + 0.1))
    
    # Triangular
    candidates.append(normalize_nonneg(1.0 - np.abs(2*x - 1)))
    
    # Evaluate
    best_cand = None
    best_val = float('inf')
    for c in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        v = compute_score(c)
        if v < best_val:
            best_val = v
            best_cand = c.copy()
    
    return best_cand if best_cand is not None else np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def evaluate_safe(a):
    """Safely call the evaluation function."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def alternating_optimize(a0, time_limit, verbose=False):
    """
    Alternate between mirror descent and coordinate descent.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_m = fft_autocorr(a).max()
    t0 = time.monotonic()
    
    md_count = 0
    cd_count = 0
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        # Alternate: 2/3 MD, 1/3 CD
        if (md_count + cd_count) % 3 == 0:
            # Mirror descent phase
            t_phase = min(remaining * 0.4, 120.0)
            if t_phase < 5:
                break
            a_out, m_out = mirror_descent_optimize(
                best_a, t_phase, beta_init=3.0, beta_final=10.0,
                K_init=50, K_final=200, lr_init=0.05,
                restart_every=2000, rng_seed=int(time.monotonic()*1000) % 10000,
                verbose=False)
            md_count += 1
        else:
            # Coordinate descent phase
            t_phase = min(remaining * 0.3, 60.0)
            if t_phase < 5:
                break
            a_out, m_out = targeted_coord_descent(
                best_a, t_phase, rng_seed=int(time.monotonic()*1000) % 10000,
                verbose=False)
            cd_count += 1
        
        if m_out < best_m - 1e-15:
            best_m = m_out
            best_a = a_out.copy()
            if verbose:
                J = 2.0 * n * best_m
                print(f"      Alt improve ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"      Alternating done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Multi-phase optimization: parametric init -> spectral shaping ->
    mirror descent -> coordinate descent -> alternating -> length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent sequence
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize_nonneg(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n0 = 5000
        best_seq = parametric_init(n0, time_limit=10.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = evaluate_safe(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")
    
    # Also try uniform as baseline
    a_uni = np.ones(n0) / n0
    sc_uni = evaluate_safe(a_uni)
    if np.isfinite(sc_uni) and sc_uni < best_score:
        best_score = sc_uni
        best_seq = a_uni.copy()
        print(f"  Uniform: {sc_uni:.6f}")
    
    # Try a few parametric inits for this length
    param_best = parametric_init(n0, time_limit=10.0)
    sc_param = evaluate_safe(param_best)
    if np.isfinite(sc_param) and sc_param < best_score:
        best_score = sc_param
        best_seq = param_best.copy()
        print(f"  Parametric init: {sc_param:.6f}")
    
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Spectral equalization (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.10, 120.0)
    if t1 > 10:
        a_out, _ = spectral_equalize(best_seq, t1, alpha=0.05, verbose=True)
        check_update(a_out, "spectral")
    
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Mirror descent (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 20:
        a_out, _ = mirror_descent_optimize(
            best_seq, t2, beta_init=2.0, beta_final=8.0,
            K_init=30, K_final=120, lr_init=0.08,
            restart_every=3000, rng_seed=42, verbose=True)
        check_update(a_out, "md_main")
    
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Targeted coord descent (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 15:
        a_out, _ = targeted_coord_descent(best_seq, t3, rng_seed=123, verbose=True)
        check_update(a_out, "targeted_cd")
    
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Second MD pass (finer) (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_out, _ = mirror_descent_optimize(
            best_seq, t4, beta_init=5.0, beta_final=12.0,
            K_init=80, K_final=200, lr_init=0.03,
            restart_every=2000, rng_seed=777, verbose=False)
        check_update(a_out, "md_fine")
    
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Alternating optimization (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 20:
        a_out, _ = alternating_optimize(best_seq, t5, verbose=True)
        check_update(a_out, "alternating")
    
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration (budget={remaining:.0f}s) ===")
    
    length_tries = []
    for delta in [-500, -200, 200, 500, 1000, -1000]:
        nn = n_best + delta
        if 2000 <= nn <= 10000 and nn != n_best:
            length_tries.append(nn)
    for rn in [3000, 4000, 5000, 6000, 7000, 8000]:
        if rn != n_best and 2000 <= rn <= 10000:
            length_tries.append(rn)
    length_tries = sorted(set(length_tries))
    
    for new_n in length_tries[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")
        
        a_interp = interpolate_sequence(best_seq, new_n)
        sc_interp = evaluate_safe(a_interp)
        if np.isfinite(sc_interp) and sc_interp < best_score:
            best_score = sc_interp
            best_seq = a_interp.copy()
            print(f"    Interpolated: {sc_interp:.6f}")
        
        # Quick optimization on new length
        t5a = min(remaining * 0.3, 100.0)
        if t5a > 10:
            a_out, _ = mirror_descent_optimize(
                a_interp, t5a, beta_init=3.0, beta_final=8.0,
                K_init=40, K_final=100, lr_init=0.05,
                restart_every=2000, rng_seed=new_n % 1000, verbose=False)
            check_update(a_out, f"md_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t5b = min(remaining * 0.2, 60.0)
            if t5b > 10:
                a_out, _ = targeted_coord_descent(a_interp, t5b, rng_seed=new_n % 2000, verbose=False)
                check_update(a_out, f"cd_n{new_n}")
    
    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}, budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.40, 200.0)
    if t7 > 10:
        a_out, _ = mirror_descent_optimize(
            best_seq, t7, beta_init=8.0, beta_final=15.0,
            K_init=100, K_final=250, lr_init=0.02,
            restart_every=1500, rng_seed=999, verbose=False)
        check_update(a_out, "final_md")
    
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t8 = min(remaining * 0.5, 100.0)
        if t8 > 5:
            a_out, _ = coordinate_descent(best_seq, t8, rng_seed=42, verbose=False)
            check_update(a_out, "final_cd")
    
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t9 = min(remaining * 0.6, 80.0)
        if t9 > 5:
            a_out, _ = targeted_coord_descent(best_seq, t9, rng_seed=2024, verbose=False)
            check_update(a_out, "final_tcd")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
