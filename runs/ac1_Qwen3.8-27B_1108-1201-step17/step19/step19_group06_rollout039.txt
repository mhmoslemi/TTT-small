    n = len(a)

import numpy as np
import time

def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def evaluate_safe(a):
    """Safely call the evaluation function."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def score_from_ac(a):
    """Compute score from a normalized sequence (sum=1)."""
    n = len(a)
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max())

def compute_gradient(a, beta):
    """
    Compute gradient of (1/beta) * log(sum_k exp(beta * c[k])) w.r.t. a.
    Uses FFT for efficiency.
    """
    n = len(a)
    L = 2 * n - 1
    ac = fast_autocorr(a)
    
    # Softmax weights
    max_ac = ac.max()
    expo = beta * (ac - max_ac)
    expo = np.clip(expo, -50.0, 50.0)
    w = np.exp(expo)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(L, dtype=np.float64) / L
    else:
        w = w / w_sum
    
    # Gradient: g[i] = sum_k w[k] * (a[k-i] + a[i-k]) / beta
    # Term 1: c1[i] = sum_k w[k] * a[k-i] = cross-correlation of w and a at "negative" lag i
    # Term 2: c2[i] = sum_k w[k] * a[i-k] = convolution of w and a at index i
    
    fft_size = 1 << (L + n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    
    # c2 = conv(w, a)
    c2 = np.fft.irfft(W * A, fft_size)[:n]
    
    # c1 = cross-correlation (negative lags)
    # xcorr_wa = ifft(conj(W) * A)
    # xcorr_wa[i] = sum_k w[k] * a[k+i] (positive lags)
    # c1[i] = sum_k w[k] * a[k-i] = xcorr_wa[fft_size - i]
    xcorr_wa = np.fft.irfft(np.conj(W) * A, fft_size)
    c1 = np.empty(n, dtype=np.float64)
    c1[0] = xcorr_wa[0]
    if n > 1:
        c1[1] = xcorr_wa[fft_size - 1]
        if n > 2:
            c1[2:] = xcorr_wa[fft_size - 2 : fft_size - n : -1]
    
    grad = (c1 + c2) / beta
    return grad, ac

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_int / ss
