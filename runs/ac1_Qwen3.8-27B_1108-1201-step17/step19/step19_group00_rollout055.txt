def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def downsample_seq(f, new_n):
    """Downsample by averaging blocks."""
    old_n = len(f)
    if old_n <= new_n:
        return interpolate_seq(f, new_n)
    s = f.sum()
    f_norm = f / s
    block_size = old_n / new_n
    result = np.zeros(new_n)
    for i in range(new_n):
        start = int(i * block_size)
        end = int((i + 1) * block_size)
        if end > old_n:
            end = old_n
        if start < end:
            result[i] = f_norm[start:end].sum()
    result = np.maximum(result, 0.0)
    ss = result.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return result / ss

def build_lp_matrix(a, clist, n):
    K = len(clist)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj in range(K):
        k = int(clist[jj])
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k - idx]
    return coeffs

def lp_fw_enhanced(a0, time_limit, solver, init_K=300, max_K=2500, add_per=30, seed=0):
    """Enhanced LP-Frank-Wolfe with dual-guided active set and adaptive steps."""

    active_set.add(0)
    active_set.add(size - 1)
    active_set.add(n - 1)

        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break

        M_curr = ac.max()

            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set size
        clist = sorted(active_set)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            keep = np.argsort(margins)[:max_K]
            # Always keep top 5 peaks and endpoints
            top_peaks = set(int(k) for k in np.argsort(ac)[::-1][:5])
            top_peaks.add(0)
            top_peaks.add(size - 1)
            top_peaks.add(n - 1)

            clist = sorted(active_set)
            # Hard cap

                clist = sorted(active_set)

        K = len(clist)
        if K == 0:
            break

        # Solve LP
        coeffs = build_lp_matrix(y, clist, n)
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        lp_ok = False
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                lp_ok = True
        except Exception:
            if len(active_set) > init_K // 2:
                active_set = set(int(k) for k in np.argsort(ac)[::-1][:init_K // 2])
                active_set.add(0)
                active_set.add(size - 1)
                active_set.add(n - 1)
                clist = sorted(active_set)

        if not lp_ok:
            no_improve += 5
            continue

        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations to add to active set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per]]

                active_set.add(int(k))

        best_alpha_recent = 0.7 * best_alpha_recent + 0.3 * best_alpha

def grad_descent_softmax(a0, time_limit, top_k=60, temp=40.0, c0=2.0, seed=123):
    """Soft-max gradient descent with momentum for long sequences."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    x_prev = x.copy()
    stall = 0
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for t in range(1, 3000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 5000:
            break

        # Momentum
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        # Compute autocorrelation
        Yf = np.fft.rfft(y, fft_size)
        ac = np.fft.irfft(Yf * Yf, fft_size)[:size]

        # Soft top-k weighting
        k_max = min(top_k, size)
        idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[idx]
        if top_vals[0] > 0:
            w = np.exp(temp * (top_vals - top_vals[0]))
            w = w / w.sum()
        else:
            w = np.ones(k_max) / k_max

        # Gradient computation
        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            idx_arr = np.arange(i_s, i_e)
            grad[idx_arr] += 2.0 * wi * y[ki - idx_arr]

        # Multiplicative update
        eta_t = c0 / (np.sqrt(max(t, 1)) + 1.0)
        log_a = np.log(y) - eta_t * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        val = obj_val(x)
        if val < best_val - 1e-13:
            best_val = val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Periodic random restart from best
        if stall > 2000 and stall % 2000 == 0:
            noise_scale = 0.02 * max(1, stall // 2000 - 1)
            noise = noise_scale * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            stall = 0

        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print("  GradDes t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(best_val))

    return best_a, best_val

def peak_deflation(a0, time_limit, n_targets=8, seed=42):
    """Target specific peak lags and reduce their contributions."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    size = 2 * n - 1

    for rnd in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        targets = np.argsort(ac)[::-1][:n_targets]
        improved = False

        for k in targets:
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            idx = np.arange(i_s, i_e)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[ki - idx]

            for _ in range(5):
                if contrib.max() < 1e-18:
                    break
                src = int(idx[np.argmax(contrib)])
                src_val = a[src]
                if src_val < 1e-12:
                    break

                # Score destinations by their contribution to top peaks
                scores = np.zeros(n)
                for k2 in targets[:5]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                        scores[k2i - idx2] += a[idx2]
                scores[src] = 1e10

                dst = int(np.argmin(scores))
                if dst == src:
                    break

                # Try multiple fractions
                found = False
                for frac in [0.002, 0.005, 0.01, 0.02]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-14:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        contrib = a[idx] * a[ki - idx]
                        found = True
                        break
                if not found:
                    break

        if not improved:
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.001, 0.004)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    Mn = fast_autocorr(a).max()
                    if Mn < best_m:
                        best_m = Mn
                        best_a = a.copy()

        if rnd % 100 == 0:
            print("  Deflate r=" + str(rnd) + " (" + str(round(time.monotonic()-t0, 1)) + "s) J=" + repr(2*n*best_m))

    return best_a, 2.0 * n * best_m

def coord_descent_polish(a0, time_limit, seed=77):

    max_stall = 600
    size = 2 * n - 1

    fracs = [0.002, 0.004, 0.008, 0.015, 0.03, 0.05, 0.08]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(15, size)
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination
            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]
            scores[src] = 1e10

            # Prefer positions with low score
            candidates = np.argsort(scores)[:n // 2]
            candidates = candidates[candidates != src]
            if len(candidates) < 10:
                candidates = np.arange(n)
                candidates = candidates[candidates != src]

            dst = int(candidates[0])
            found = False
            for frac in fracs:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t / st
                Mt = fast_autocorr(a_t).max()
                if Mt < best_m - 1e-14:
                    a = a_t
                    best_m = Mt
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 100:
                # Random move
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0 and stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.002, 0.01)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

        if stall % 200 == 0:
            print("  Coord (" + str(round(time.monotonic()-t0, 1)) + "s) J=" + repr(2*n*best_m) + " stall=" + str(stall))

    return best_a, 2.0 * n * best_m

def generate_diverse_init(n, best_ref):
    """Generate diverse initializations and return the best one."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    bi = normalize(best_ref) if best_ref is not None else None

    candidates = []

    # Tapered uniform (low endpoints)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append(normalize(np.sin(np.pi * x)**p))
        candidates.append(normalize((x * (1-x))**p))

    # Gaussian bumps
    for c in [0.3, 0.5, 0.7]:
        for sg in [0.1, 0.15, 0.2, 0.3]:
            candidates.append(normalize(np.exp(-((x - c)**2)/(2*sg*sg))))

    # Double Gaussian
    for c1 in [0.25, 0.35]:
        for c2 in [0.65, 0.75]:
            for sg in [0.08, 0.12, 0.18]:
                cands = np.exp(-((x-c1)**2)/(2*sg*sg)) + np.exp(-((x-c2)**2)/(2*sg*sg))
                candidates.append(normalize(cands))

    # Cosine windows
    for f in [1, 2, 3]:
        for phase in [0, np.pi/2, np.pi]:
            cands = np.maximum(1.0 + 0.8*np.cos(2*np.pi*f*x + phase), 0.01)
            candidates.append(normalize(cands))

    # Power law
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(normalize((i_arr + 1.0)**p / (n + 1.0 - i_arr)**p))
        candidates.append(normalize((1.0 + (i_arr/(n-1.0))**p) * (1.0 + ((n-1.0-i_arr)/(n-1.0))**p)))

    # Random smooth (spline-like)
    rng = np.random.default_rng(42)
    for _ in range(5):
        n_ctrl = rng.integers(5, 15)
        ctrl_x = np.sort(rng.uniform(0, 1, n_ctrl))
        ctrl_y = rng.uniform(0.1, 1.0, n_ctrl)
        ctrl_y[0] = 0.01
        ctrl_y[-1] = 0.01
        smooth = np.interp(x, ctrl_x, ctrl_y)
        candidates.append(normalize(smooth))

    # Blend with best reference
    if bi is not None:
        for w in [0.3, 0.5, 0.7]:
            for cand in candidates[:6]:
                candidates.append(normalize(w*bi + (1.0-w)*cand))

    best_a = bi if bi is not None else np.ones(n)/n
    best_val = obj_val(best_a)

    for c in candidates:
        v = obj_val(c)
        if v < best_val:
            best_val = v
            best_a = c.copy()

    return best_a, best_val

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 100.0, 1700.0)
    solver = get_solver()
    print("Total budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            print("Parent: n=" + str(len(p)) + " score=" + repr(best_score))
    except Exception:
        best_seq = None

    if best_seq is None:
        n_init = 8000
        x = np.linspace(0, 1, n_init)
        best_seq = normalize(np.sin(np.pi * x)**2)
        best_score = eval_safe(best_seq)
        print("Default init n=" + str(n_init) + " score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: Diverse initialization
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Diverse init (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.03, 40.0)
    if t1 > 5:
        a_p, v_p = generate_diverse_init(n0, best_seq)
        sc = eval_safe(a_p)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_p
            print("  New best init: " + repr(best_score))

    # Phase 2: Main LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P2: LP-FW (n=" + str(n_cur) + ") ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 60:
        a_out, v_out = lp_fw_enhanced(normalize(best_seq), t2, solver,
                                      init_K=400, max_K=3000, add_per=40, seed=1)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  LP-FW improved: " + repr(best_score))

    # Phase 3: Continue LP-FW with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P3: LP-FW cont (n=" + str(n_cur) + ") ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40:
        a_out, v_out = lp_fw_enhanced(normalize(best_seq), t3, solver,
                                      init_K=500, max_K=3500, add_per=50, seed=2)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  LP-FW2 improved: " + repr(best_score))

    # Phase 4: Gradient descent at longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = min(len(best_seq) + 2000, 14000)
    print("\n=== P4: GradDes n=" + str(n_long) + " ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 40:
        a_long = interpolate_seq(best_seq, n_long)
        a_out, v_out = grad_descent_softmax(a_long, t4, top_k=60, temp=50.0, c0=2.0, seed=55)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  GradDes improved: " + repr(best_score))

    # Phase 5: Try even longer
    remaining = time_budget - (time.monotonic() - t_global)
    n_long2 = min(len(best_seq) + 3000, 20000)
    if n_long2 != len(best_seq):
        print("\n=== P5: GradDes n=" + str(n_long2) + " ===")
        t5 = min(remaining * 0.12, 200.0)
        if t5 > 30:
            a_long2 = interpolate_seq(best_seq, n_long2)
            a_out, v_out = grad_descent_softmax(a_long2, t5, top_k=80, temp=70.0, c0=1.5, seed=66)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  GradDes2 improved: " + repr(best_score))

    # Phase 6: Peak deflation
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P6: Deflation (n=" + str(n_cur) + ") ===")
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 15:
        a_out, v_out = peak_deflation(normalize(best_seq), t6, n_targets=8, seed=33)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  Deflate improved: " + repr(best_score))

    # Phase 7: Final LP-FW if length allows
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if n_cur <= 10000:
        print("\n=== P7: Final LP-FW (n=" + str(n_cur) + ") ===")
        t7 = min(remaining * 0.18, 200.0)
        if t7 > 30:
            a_out, v_out = lp_fw_enhanced(normalize(best_seq), t7, solver,
                                          init_K=600, max_K=4000, add_per=50, seed=3)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  Final LP improved: " + repr(best_score))
    else:
        print("\n=== P7: Final GradDes (n=" + str(n_cur) + ") ===")
        t7 = min(remaining * 0.18, 200.0)
        if t7 > 30:
            a_out, v_out = grad_descent_softmax(normalize(best_seq), t7, top_k=100, temp=80.0, c0=1.0, seed=77)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  Final Grad improved: " + repr(best_score))

    # Phase 8: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P8: Coord polish (n=" + str(n_cur) + ") ===")
    t8 = min(remaining * 0.25, 180.0)
    if t8 > 15:
        a_out, v_out = coord_descent_polish(normalize(best_seq), t8, seed=99)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  Coord improved: " + repr(best_score))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + " n=" + str(len(best_seq)) + " t=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: using fallback")
        n_fb = 8000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x)**2)
        best_score = eval_safe(best_seq)

    return [float(x) for x in best_seq]
