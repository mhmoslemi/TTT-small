def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

def simplex_project(z):
    """Project z onto the probability simplex."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    rho_vals = np.nonzero(cond)[0]
    if len(rho_vals) == 0:
        return np.ones(n) / n
    r = rho_vals[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_peg_subgrad(a, top_k):
    """Compute the 'pressure' subgradient: sum over top-k peaks of a[k-i]."""
    n = len(a)
    c = fast_autocorr(a)
    size = 2 * n - 1
    idx = np.argsort(c)[::-1][:top_k]
    # Weight by how close to max
    c_max = c[idx[0]]
    c_min_top = c[idx[-1]]
    if c_max - c_min_top < 1e-15:
        weights = np.ones(top_k) / top_k
    else:
        weights = (c[idx] - c_min_top) / (c_max - c_min_top)
        weights = weights / weights.sum()
    
    sg = np.zeros(n)
    for k, w in zip(idx, weights):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            i_idx = np.arange(i_start, i_end)
            sg[i_idx] += w * a[k_int - i_idx]
    return sg, c

def lp_minimize_peak(a0, time_limit, solver, verbose=True):
    """LP-based approach: minimize peak M subject to sum(a)=1, a>=0, (a*a)_k <= M for active k."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()

    size = 2 * n - 1
    # Start with top K peaks as constraints
    K = min(500, size)
    active = set(int(k) for k in np.argsort(c)[::-1][:K])
    
    no_improve = 0
    max_no_improve = 500
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        M_curr = c.max()
        
        # Build constraint matrix for active set
        constraints_list = sorted(active)
        if len(constraints_list) > 2000:
            # Prune: keep only tightest constraints
            margins = np.array([M_curr - c[k] for k in constraints_list])
            keep = np.argsort(margins)[:2000]
            active = set(int(constraints_list[i]) for i in keep)
            constraints_list = sorted(active)
        
        K_curr = len(constraints_list)
        if K_curr == 0:
            break
        
        # Build sparse constraint: for each active lag k, (a*a)_k <= M
        # (a*a)_k = sum_i a[i] * a[k-i] (bilinear!)
        # Can't directly use in LP. Need linearization.
        # Alternative: fix a to current value in one factor
        # (a*a)_k ≈ a_curr[k-i] * a[i] + a[i] * a_curr[k-i] - a_curr[k-i]*a_curr[k-i] (linearization)
        
        # Actually, let's use the Frank-Wolfe approach:
        # Minimize M subject to sum(g)=1, g>=0, and (a_curr * g)_k <= M for active k
        # This is an LP!
        
        coeffs = np.zeros((K_curr, n))
        for jj, k in enumerate(constraints_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = a_curr[k_int - idx] if 'a_curr' in dir() else a[k_int - idx]
        
        # Hmm, I need a_curr. Let me restructure.
        a_curr = a.copy()
        coeffs = np.zeros((K_curr, n))
        for jj, k in enumerate(constraints_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = a_curr[k_int - idx]
        
        # LP: min M s.t. sum(g)=1, g>=0, coeffs @ g <= M
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        prob = cp.Problem(cp.Minimize(M_var),
                         [cp.sum(g_var) == 1, coeffs @ g_var <= M_var])
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            no_improve += 10
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 10
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            no_improve += 10
            continue
        g_dir = g_val / g_sum
        
        # Find violations
        conv_ag = np.fft.irfft(np.fft.rfft(a_curr, 1<<(2*n-2).bit_length()) * 
                               np.fft.rfft(g_dir, 1<<(2*n-2).bit_length()), 
                               1<<(2*n-2).bit_length())[:size]
        violations = np.where(conv_ag > M_var.value + 1e-6)[0]
        for v in violations[:50]:
            active.add(int(v))
        
        # Line search
        best_alpha = 0.5
        best_trial = float('inf')
        for alpha in [0.1, 0.3, 0.5, 0.7, 0.9]:
            trial = (1 - alpha) * a_curr + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_t = fast_autocorr(trial).max()
            if M_t < best_trial:
                best_trial = M_t
                best_alpha = alpha
        
        a_new = (1 - best_alpha) * a_curr + best_alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a_curr.copy()
        else:
            a_new = a_new / s_new
        
        c = fast_autocorr(a_new)
        m_new = c.max()
        
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP t=%d (%.1fs): J=%.6f |S|=%d" % (t, elapsed, J, len(active)))
    
    return best_a, best_m

def fast_conv(a):

def evaluate_safe(a):

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def simplex_project(z):
    """Project z onto probability simplex {x >= 0, sum(x) = 1}."""

    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    rho_vals = np.nonzero(cond)[0]
    if len(rho_vals) == 0:
        return np.ones(n) / n
    r = rho_vals[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def softmax_grad(a, beta, top_k_frac=0.1, min_top_k=10):
    """Compute smoothed gradient of max(conv(a,a)) using softmax over peaks."""
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -400, 400)
    exp_c = np.exp(shifted)
    p = exp_c / exp_c.sum()
    
    # Gradient: sum_k p_k * a[k-i] (with valid index bounds)
    grad = np.zeros(n)
    # Only compute for significant p values
    threshold = p.max() * 0.01
    significant = np.where(p > threshold)[0]
    for k in significant:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            i_idx = np.arange(i_start, i_end)
            grad[i_idx] += p[k] * a[k_int - i_idx]
    return grad, c

def nesterov_smoothed(a0, time_limit, beta_start=15.0, beta_end=1200.0,
                       beta_growth=1.004, eta_scale=1.0, restart_every=2500,
                       top_k_frac=0.08, verbose=True, seed=42):
    """Nesterov accelerated gradient on log-sum-exp smoothed objective."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fast_conv(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        # Compute smoothed gradient
        grad, c = softmax_grad(y, beta, top_k_frac)
        
        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            grad_norm = 1.0
        eta = eta_scale / (beta * grad_norm)
        eta = min(eta, 0.3)
        
        # Gradient step + projection
        z = y - eta * grad
        x_new = simplex_project(z)
        
        # Track best
        c_new = fast_conv(x_new)
        max_new = c_new.max()
        if max_new < best_m - 1e-15:
            best_m = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        # Adaptive beta increase
        if t % 60 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        # Restart mechanism with progressive noise
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            restart_count = t // restart_every
            noise = 0.04 / (1.0 + 0.1 * restart_count)
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.75, 8.0)
            if verbose and restart_count % 3 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    NG restart t=%d (%.1fs): J=%.6f beta=%.0f" % (t, elapsed, J, beta))
        
        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    NG t=%d (%.1fs): J=%.6f beta=%.0f" % (t, elapsed, J, beta))
    
    return best_a, best_m

def nesterov_multi_beta(a0, time_limit, verbose=True):
    """Run Nesterov with multiple beta schedules and keep the best."""
    n = len(a0)
    configs = [
        {"beta_start": 10.0, "beta_end": 500.0, "beta_growth": 1.006, "eta_scale": 1.2, "seed": 1},
        {"beta_start": 30.0, "beta_end": 1500.0, "beta_growth": 1.004, "eta_scale": 0.8, "seed": 2},
        {"beta_start": 50.0, "beta_end": 3000.0, "beta_growth": 1.003, "eta_scale": 0.6, "seed": 3},
    ]
    
    best_a = a0.copy()
    best_m = fast_conv(a0).max()
    t0 = time.monotonic()
    
    # Allocate time among configs (decreasing)
    weights = [0.5, 0.3, 0.2]
    for i, cfg in enumerate(configs):
        t_remaining = time_limit - (time.monotonic() - t0)
        if t_remaining < 20:
            break
        t_alloc = t_remaining * weights[i]
        if t_alloc < 15:
            continue
        print("    NG-multi config %d: beta_start=%.0f, t_alloc=%.0fs" % (i, cfg["beta_start"], t_alloc))
        a_out, m_out = nesterov_smoothed(a0, t_alloc, verbose=(i == 0), **cfg)
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
    
    return best_a, best_m

def pair_coord_descent(a0, time_limit, verbose=True):
    """Coordinate descent exploiting bilinear structure of (a*a)_k."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    stall = 0
    max_stall = 400
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        size = 2 * n - 1
        n_top = min(25, size)
        top_peaks = np.argsort(c)[::-1][:n_top]
        
        improved = False
        
        for k in top_peaks[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_contrib = contrib.max()
            
            if max_contrib < 1e-20:
                continue
            
            # Find the contributing pair
            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            if partner < 0 or partner >= n:
                continue
            
            # Try to reduce this pair's contribution
            # Option 1: Reduce the larger element
            if a[src] >= a[partner]:
                red = src
                keep = partner
            else:
                red = partner
                keep = src
            
            if a[red] < 1e-12:
                continue
            
            # Compute pressure field to find safe destinations
            pressure = np.zeros(n)
            for k2 in top_peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
            
            # Mark peak zones
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2+1] = True
            
            # Find low-pressure destinations outside peak zones
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)
            
            cand_press = pressure[candidates]
            order = np.argsort(cand_press)[:8]
            
            found = False
            for di in order:
                dst = int(candidates[di])
                if dst == red:
                    continue
                # Try different transfer fractions
                for frac in [0.1, 0.2, 0.35, 0.5, 0.7]:
                    amt = a[red] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[red] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break
            
            if not found and stall > 150:
                # Random exploration
                j = int(rng.integers(n))
                if j != red and a[red] > 1e-10:
                    frac = rng.uniform(0.02, 0.15)
                    amt = a[red] * frac
                    a[red] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                            break
        
        if not improved:
            stall += 1
            if stall % 80 == 0:
                # Broader random perturbation
                for _ in range(8):
                    i1 = int(rng.integers(n))
                    i2 = int(rng.integers(n))
                    if i1 != i2 and a[i1] > 1e-10:
                        frac = rng.uniform(0.005, 0.08)
                        amt = a[i1] * frac
                        a[i1] -= amt
                        a[i2] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_conv(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                break
        
        if verbose and stall % 300 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Pair-CD (%.1fs): J=%.6f stall=%d" % (elapsed, J, stall))
    
    return best_a, best_m

def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2500, verbose=True):
    """LP Frank-Wolfe: minimize max convolution peak."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    c = fast_conv(a)
    size = 2 * n - 1
    active = set(int(k) for k in np.argsort(c)[::-1][:initial_K])
    
    no_improve = 0
    max_no_improve = 800
    fft_size = 1 << (size - 1).bit_length()
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: converged at iter %d" % t)
            break
        
        c = fast_conv(a)
        M_curr = c.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Manage active set size
        if len(active) > max_K:
            margins = np.array([M_curr - c[k] for k in active])
            keep = np.argsort(margins)[:max_K]
            active = set(int(list(active)[i]) for i in keep)
        
        constraint_list = sorted(active)
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix: (a_curr * g)_k <= M for each active k
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = a[k_int - idx]
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        prob = cp.Problem(cp.Minimize(M_var),
                         [cp.sum(g_var) == 1, coeffs @ g_var <= M_var])
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            no_improve += 20
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 20
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            no_improve += 20
            continue
        g_dir = g_val / g_sum
        
        # Find violations (where a*g exceeds current M)
        A_f = np.fft.rfft(a, fft_size)
        G_f = np.fft.rfft(g_dir, fft_size)
        conv_ag = np.fft.irfft(A_f * G_f, fft_size)[:size]
        
        M_lp = float(M_var.value) if M_var.value is not None else M_curr
        violations = np.where(conv_ag > M_lp + 1e-5)[0]
        if len(violations) > 0:
            viol_amt = conv_ag[violations] - M_lp
            top_v = violations[np.argsort(viol_amt)[::-1][:40]]
            for v in top_v:
                active.add(int(v))
        
        # Adaptive line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.05, default_alpha * 0.5),
                  max(0.05, default_alpha * 2.0), 0.3, 0.5]
        alphas = [max(0.02, min(1.0, a)) for a in alphas]
        alphas = list(dict.fromkeys(alphas))  # deduplicate preserving order
        
        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * a + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_conv(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        if t <= 2:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * a + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = a.copy()
            else:
                y_new = y_new / s_y
        
        a = y_new
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=%d (%.1fs): J=%.6f |S|=%d" % (t, elapsed, J, len(active)))
    
    return best_a, best_m

def spectral_refine(a0, time_limit, verbose=True):
    """Frequency-domain refinement: minimize peak of |FFT(a)|^2 inverse."""
    n = len(a0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(200)
    
    beta = 30.0
    eta = 0.05
    stall = 0
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_conv(a)
        c_max = c.max()
        
        # Softmax weights over peaks
        shifted = beta * (c - c_max)
        shifted = np.clip(shifted, -400, 400)
        p = np.exp(shifted)
        p = p / p.sum()
        
        # Compute gradient in time domain
        grad = np.zeros(n)
        threshold = p.max() * 0.005
        sig_idx = np.where(p > threshold)[0]
        for k in sig_idx:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                i_idx = np.arange(i_start, i_end)
                grad[i_idx] += p[k] * a[k_int - i_idx]
        
        # Frequency domain step
        A = np.fft.rfft(a, fft_size)
        G = np.fft.rfft(grad, fft_size)
        
        # Damped frequency update
        A_new = A - eta * G
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        c_new = fast_conv(a_new)
        m_new = c_new.max()
        
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.02, 0.3)
        else:
            stall += 1
            eta *= 0.92
            if stall % 300 == 0:
                beta = min(beta * 1.15, 500.0)
                eta = 0.05
                # Random perturbation
                if stall > 1000:
                    pert = rng.normal(0, 0.02, n) * best_a
                    a_r = best_a + pert
                    a_r = np.maximum(a_r, 1e-10)
                    a_r = a_r / a_r.sum()
                    a = a_r
                    stall = 0
        
        a = a_new
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spec t=%d (%.1fs): J=%.6f beta=%.0f eta=%.4f" % (t, elapsed, J, beta, eta))
    
    return best_a, best_m

def structured_inits(n, verbose=False):
    """Generate diverse structured initializations and return best."""
    x = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)
    candidates = []
    
    # Standard windows
    candidates.append(("hann", 0.5 * (1 + np.cos(2 * np.pi * x))))
    candidates.append(("hamming", 0.54 + 0.46 * np.cos(2 * np.pi * x)))
    candidates.append(("blackman", 0.42 + 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)))
    candidates.append(("flattop", 0.2156 - 0.4166*np.cos(2*np.pi*x) + 0.2773*np.cos(4*np.pi*x) - 0.0836*np.cos(6*np.pi*x) + 0.0069*np.cos(8*np.pi*x)))
    candidates.append(("sine2", np.sin(np.pi * x) ** 2))
    candidates.append(("sine4", np.sin(np.pi * x) ** 4))
    candidates.append(("sine6", np.sin(np.pi * x) ** 6))
    candidates.append(("sine8", np.sin(np.pi * x) ** 8))
    
    # Power laws from center
    for p in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(("pow%.1f" % p, np.maximum(1.0 - np.abs(x - 0.5) * 2, 0.01) ** p))
    
    # Asymmetric
    for p in [1.0, 2.0, 3.0]:
        candidates.append(("asym%.0f" % p, (x + 0.1) ** p))
        candidates.append(("asym_r%.0f" % p, (1.0 - x + 0.1) ** p))
    
    # Two-hump
    for c in [0.2, 0.3, 0.4]:
        for sig in [0.05, 0.1, 0.15]:
            hump = np.exp(-((x - c)**2) / (2*sig*sig)) + np.exp(-((x - (1-c))**2) / (2*sig*sig))
            candidates.append(("2h_c%.1f_s%.2f" % (c, sig), hump))
    
    # Oscillatory
    for freq in [2, 4, 8]:
        candidates.append(("osc%d" % freq, (1 + np.cos(2*np.pi*freq*x))**2))
    
    # Prolate-like (product of sine and window)
    candidates.append(("prolate1", np.sin(np.pi*x)**2 * (0.5 + 0.5*np.cos(2*np.pi*x))))
    candidates.append(("prolate2", np.sin(np.pi*x)**3 * (0.5 + 0.5*np.cos(2*np.pi*x))))
    
    best_a = None
    best_m = float('inf')
    
    for name, raw in candidates:
        a = norm_sum1(raw)
        m = fast_conv(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()
            if verbose:
                print("    Structured '%s': J=%.6f" % (name, 2.0 * n * m))
    
    # Random inits
    rng = np.random.default_rng(12345)
    for trial in range(100):
        alpha = rng.uniform(0.3, 4.0)
        a = norm_sum1(rng.random(n) ** alpha)
        m = fast_conv(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()
            if verbose:
                print("    Random init %d: J=%.6f" % (trial, 2.0 * n * m))
    
    return best_a, best_m

def multi_start_optimize(a0, time_limit, solver, verbose=True):
    """Run multiple optimization passes with different initializations."""
    n = len(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_m = fast_conv(a0).max()
    
    # Generate diverse starts
    starts = [a0.copy()]
    
    # Perturbed versions
    rng = np.random.default_rng(777)
    for noise_level in [0.05, 0.1, 0.2]:
        pert = rng.normal(0, noise_level, n) * best_a
        a_pert = best_a + pert
        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = a_pert / a_pert.sum()
        starts.append(a_pert)
    
    # Structured
    a_struct, _ = structured_inits(n, verbose=False)
    if a_struct is not None:
        starts.append(a_struct)
    
    # Allocate time
    n_starts = len(starts)
    t_per_start = (time_limit - 60) / n_starts
    
    for si, start in enumerate(starts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break
        
        t_alloc = min(t_per_start, remaining - 10)
        if t_alloc < 20:
            continue
        
        if verbose:
            m_start = fast_conv(start).max()
            print("    Multi-start %d/%d: initial J=%.6f, t=%.0fs" % (si, n_starts, 2.0*n*m_start, t_alloc))
        
        # Nesterov
        t_neg = t_alloc * 0.6
        a_out, m_out = nesterov_smoothed(start, t_neg, beta_start=20, beta_end=800,
                                         beta_growth=1.005, eta_scale=1.0,
                                         restart_every=2000, verbose=False, seed=si+100)
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
        
        # LP-FW on best so far
        remaining = time_limit - (time.monotonic() - t0)
        if remaining > 30:
            t_lp = min(remaining * 0.4, 80.0)
            if t_lp > 15 and n <= 7000:
                a_out2, m_out2 = lp_fw(best_a, t_lp, solver, initial_K=200, max_K=1500, verbose=False)
                if m_out2 < best_m:
                    best_m = m_out2
                    best_a = a_out2.copy()
        
        if verbose:
            J = 2.0 * n * best_m
            print("    Multi-start %d done: best J=%.6f" % (si, J))
    
    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """
    Main search function: multi-strategy optimization to minimize
    2*n*max(convolve(a,a))/(sum(a))^2.
    
    Strategy:
    1. Structured initialization
    2. Nesterov with multi-beta schedules
    3. LP Frank-Wolfe
    4. Pair coordinate descent
    5. Spectral refinement
    6. Multi-start with diverse perturbations
    7. Multi-length exploration
    8. Final polish passes
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    
    # Get solver
    solver = None
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                print("Solver: " + str(s))
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS
        print("Solver (fallback): ECOS")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                print("Parent: n=%d, score=%.6f" % (len(parent), p_score))
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=%d, score=%.6f" % (len(best_seq), best_score))
    
    n0 = len(best_seq)
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print("  *** NEW BEST [%s]: %.6f (n=%d, t=%.0fs)" % (label, sc, len(a_norm), elapsed))
    
    # Phase 1: Structured initialization
    print("\n=== Phase 1: Structured inits (n=%d) ===" % n0)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 5:
        a_struct, m_struct = structured_inits(n0, verbose=True)
        if a_struct is not None:
            try_update(a_struct, "structured")
    
    # Phase 2: Nesterov multi-beta
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov multi-beta (n=%d) ===" % n_best)
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 60:
        a_out, _ = nesterov_multi_beta(norm_sum1(best_seq), t2, verbose=True)
        try_update(a_out, "nest_multi")
    
    # Phase 3: LP Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP-FW (n=%d) ===" % n_best)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 40 and n_best <= 8000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t3, solver,
                         initial_K=400, max_K=2500, verbose=True)
        try_update(a_out, "lp_fw")
    
    # Phase 4: Pair coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Pair CD (n=%d) ===" % n_best)
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 20:
        a_out, _ = pair_coord_descent(norm_sum1(best_seq), t4, verbose=True)
        try_update(a_out, "pair_cd")
    
    # Phase 5: Spectral refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Spectral (n=%d) ===" % n_best)
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 25:
        a_out, _ = spectral_refine(norm_sum1(best_seq), t5, verbose=True)
        try_update(a_out, "spectral")
    
    # Phase 6: Multi-start with diverse perturbations
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-start (n=%d) ===" % n_best)
    t6 = min(remaining * 0.18, 250.0)
    if t6 > 60:
        a_out, _ = multi_start_optimize(norm_sum1(best_seq), t6, solver, verbose=True)
        try_update(a_out, "multi_start")
    
    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = []
    for delta in [500, 1000, 1500, 2000]:
        ln = n_best + delta
        if ln <= 9000:
            longer_lengths.append(ln)
    
    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("\n=== Phase 7: Scale to n=%d ===" % new_n)
        a_long = interpolate_sequence(best_seq, new_n)
        
        # Nesterov on longer
        t7a = min(remaining * 0.35, 150.0)
        if t7a > 30:
            a_out, _ = nesterov_smoothed(a_long, t7a, beta_start=20, beta_end=800,
                                         beta_growth=1.005, eta_scale=0.9,
                                         restart_every=2000, verbose=True, seed=new_n % 1000)
            try_update(a_out, "nest_n%d" % new_n)
        
        # LP on longer
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 8000:
            t7b = min(remaining * 0.35, 120.0)
            if t7b > 25:
                a_long2 = interpolate_sequence(best_seq, new_n)
                a_out, _ = lp_fw(a_long2, t7b, solver, initial_K=300, max_K=2000, verbose=True)
                try_update(a_out, "lp_n%d" % new_n)
    
    # Phase 8: Final Nesterov with high beta
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final Nesterov high-beta (n=%d) ===" % n_best)
    if remaining > 50:
        t8 = min(remaining * 0.30, 200.0)
        if t8 > 25:
            a_out, _ = nesterov_smoothed(norm_sum1(best_seq), t8,
                                         beta_start=80, beta_end=3000,
                                         beta_growth=1.006, eta_scale=0.5,
                                         restart_every=1500, verbose=True, seed=999)
            try_update(a_out, "nest_final")
    
    # Phase 9: Final coordinate
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 9: Final pair-CD (n=%d) ===" % n_best)
    if remaining > 30:
        t9 = min(remaining * 0.35, 150.0)
        if t9 > 15:
            a_out, _ = pair_coord_descent(norm_sum1(best_seq), t9, verbose=True)
            try_update(a_out, "cd_final")
    
    # Phase 10: Final LP burst
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30 and n_best <= 8000:
        print("\n=== Phase 10: Final LP burst (n=%d) ===" % n_best)
        t10 = min(remaining * 0.60, 150.0)
        if t10 > 20:
            a_out, _ = lp_fw(norm_sum1(best_seq), t10, solver,
                             initial_K=500, max_K=3000, verbose=True)
            try_update(a_out, "lp_final")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=%.6f, n=%d, time=%.1fs" % (best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
    
    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
