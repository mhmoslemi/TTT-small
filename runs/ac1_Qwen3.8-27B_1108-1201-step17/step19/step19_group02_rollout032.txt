"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse parametric initialization at multiple n values
2. LSE-smoothed mirror descent with FFT-based gradient computation
3. LP-based Frank-Wolfe with margin-based adaptive active set
4. Exact line-search coordinate descent with peak-aware mass transfer
5. Multi-n exploration with interpolation and re-optimization
"""

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None

    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER

            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s

        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _MODULE_SOLVER


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_max_conv(a):
    return fast_convolve(a).max()


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lse_gradient_weights(conv, mu):
    """Compute softmax weights for LSE smoothing."""
    v = conv / mu
    v_max = v.max()
    w = np.exp(v - v_max)
    ws = w.sum()
    if ws < 1e-30:
        return np.ones(len(conv)) / len(conv)
    return w / ws


def compute_gradient_fft(y, w, fft_size):
    """
    Compute gradient of sum_k w[k]*(y*y)[k] w.r.t. y using FFT.
    grad[j] = 2 * sum_k w[k] * y[k-j] (with y out of range = 0)
    """
    n = len(y)
    # Create reflected y: y_ref[0] = y[0], y_ref[fft_size-i] = y[i] for i=1..n-1
    y_ref = np.zeros(fft_size)
    y_ref[0] = y[0]
    if n > 1:
        y_ref[fft_size - n + 1:] = y[:n - 1][::-1]

    W = np.fft.rfft(np.concatenate([w, np.zeros(fft_size - len(w))]), fft_size)
    Yref = np.fft.rfft(y_ref, fft_size)
    grad = 2.0 * np.fft.irfft(W * Yref, fft_size)[:n]
    return grad


def lse_mirror_descent(a0, time_limit, mu_init=3e-4, mu_min=3e-6,
                       mu_decay=0.9995, lr=1.5, momentum=0.8,
                       restart_every=4000, verbose=True):
    """
    Mirror descent with LSE-smoothed max of convolution.
    Uses exponentiated gradient for simplex constraint and FFT for gradients.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = compute_max_conv(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    mu = mu_init
    x = a.copy()
    vel = np.zeros(n)
    stall = 0
    restart_count = 0

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fft_size_grad = 1 << (3 * n).bit_length()  # Larger for gradient computation

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Momentum
        y = x + momentum * vel
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Convolution
        Yf = np.fft.rfft(y, fft_size)
        conv = np.fft.irfft(Yf * Yf, fft_size)[:size]

        # Softmax weights
        w = lse_gradient_weights(conv, mu)

        # Gradient via FFT
        grad = compute_gradient_fft(y, w, fft_size_grad)

        # Exponentiated gradient update
        eta = lr * mu
        scaled = -eta * grad
        scaled -= scaled.max()
        a_new = y * np.exp(scaled)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        vel = 0.9 * vel + 0.1 * (a_new - x)
        x = a_new

        M_new = compute_max_conv(x)
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        mu = max(mu * mu_decay, mu_min)

        if stall > restart_every:
            stall = 0
            restart_count += 1
            noise = 0.06 / (1.0 + 0.15 * restart_count)
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            x = x_r / s_r if s_r > 1e-15 else best_a.copy()
            vel = np.zeros(n)
            mu = max(mu * 5.0, mu_init * 0.3)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    LSE-MD restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LSE-MD t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", mu=" + repr(mu))

    return best_a, best_m


def build_lp_matrix(y, active_set, n):
    """Build constraint matrix for LP-FW."""
    K = len(active_set)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(active_set):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_frank_wolfe(a0, time_limit, solver, initial_K=300, max_K=3000,
                   margin_frac=0.02, add_per_iter=40, verbose=True):
    """
    LP-based Frank-Wolfe for minimax convolution.
    Uses margin-based active set management.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = compute_max_conv(a)
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_convolve(y)
    M_curr = ac_y.max()

    # Initial active set: top K by value
    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())

    no_improve = 0
    max_no_improve = 1500
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: stagnation at iter " + str(iter_count))
            break
        iter_count += 1

        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update active set: keep constraints within margin of max
        margin = M_curr * margin_frac
        active_indices = np.where(ac >= M_curr - margin)[0]
        # Also always keep the top max_K
        top_idx = np.argsort(ac)[::-1][:max_K]
        combined = np.unique(np.concatenate([active_indices, top_idx]))
        active_set = sorted(combined[:max_K].tolist())

        K = len(active_set)
        if K == 0:
            break

        coeffs = build_lp_matrix(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Reduce active set and retry
            active_set = sorted(active_set[:initial_K])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations to add to active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(list(new_set)[:max_K])

        # Step size selection: try multiple alphas
        default_alpha = 2.0 / (iter_count + 2.0)
        alphas = sorted(set([
            max(0.01, min(1.0, default_alpha)),
            max(0.01, min(1.0, default_alpha * 0.5)),
            max(0.01, min(1.0, default_alpha * 2.0)),
            max(0.01, min(1.0, 0.1)),
            max(0.01, min(1.0, 0.3)),
            max(0.01, min(1.0, 0.5)),
        ]))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = compute_max_conv(trial)
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 2:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", |S|=" + str(len(active_set)) + ", M=" + repr(best_m))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", iters=" + str(iter_count))
    return best_a, best_m


def coordinate_descent(a0, time_limit, rng_seed=99, verbose=False):
    """
    Coordinate descent with exact line search for peak reduction.
    Focuses on transferring mass from high-contribution coordinates to low-contribution ones.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = compute_max_conv(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 600
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        M_curr = ac.max()

        improved = False

        # For each top peak, try to reduce it by moving mass
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination: minimize contribution to all top peaks
            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            # Exclude positions that are in any top peak
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < n // 4:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])

            if dst == src:
                continue

            # Try multiple transfer fractions
            found = False
            for frac in [0.05, 0.02, 0.01, 0.005, 0.03, 0.08]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = compute_max_conv(a_trial)
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break

            if not found and stall > 150:
                # Random perturbation
                for _ in range(5):
                    j = int(rng.integers(n))
                    if j != src and a[src] > 1e-10:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a[src] * frac
                        a[src] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = compute_max_conv(a)
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True
                                break

        if not improved:
            stall += 1
            if stall % 100 == 0:
                # Periodic random kicks
                for _ in range(3):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.01)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = compute_max_conv(a)
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", stall=" + str(stall))

    return best_a, best_m


def parametric_init(n, best_score):
    """Generate diverse parametric initializations and return the best."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')

    candidates = []

    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))

    # Two-sided power
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        for beta in [0.5, 1.0, 1.5, 2.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            candidates.append(norm_sum1(a))

    # Gaussians
    for c in [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9]:
        for sigma in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            candidates.append(norm_sum1(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))))

    # Double Gaussian
    for c1 in [0.1, 0.2, 0.3]:
        for c2 in [0.7, 0.8, 0.9]:
            for sigma in [0.05, 0.1, 0.15]:
                a = np.exp(-((x - c1) ** 2) / (2.0 * sigma * sigma)) + np.exp(-((x - c2) ** 2) / (2.0 * sigma * sigma))
                candidates.append(norm_sum1(a))

    # Sine/cosine based
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(norm_sum1(x ** p + (1.0 - x) ** p + 0.1))

    for freq in [1, 2, 3, 5]:
        for ph in [0.0, np.pi / 2.0]:
            a = np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x + ph)) / 2.0, 0.01)
            candidates.append(norm_sum1(a))

    # Exponential
    for rate in [0.5, 1, 2, 5, 10]:
        a = np.exp(-rate * x) + np.exp(rate * (x - 1.0)) + 0.01
        candidates.append(norm_sum1(a))

    # Sin^2
    candidates.append(norm_sum1(np.sin(np.pi * x) ** 2))
    candidates.append(norm_sum1(np.sin(np.pi * x) ** 4))

    # Linear
    candidates.append(norm_sum1(i_arr + 1.0))
    candidates.append(norm_sum1(n - i_arr))

    # S-shaped
    for s in [2, 5, 10]:
        candidates.append(norm_sum1(1.0 / (1.0 + np.exp(-s * (x - 0.5)))))

    # Random
    rng = np.random.default_rng(2024)
    for trial in range(100):
        a = rng.random(n)
        a = a ** rng.uniform(0.3, 4.0)
        candidates.append(norm_sum1(a))

    for a in candidates:
        if time.monotonic() - t0 > 30:
            break
        v = compute_max_conv(a)
        if v < best_val:
            best_val = v
            best_a = a.copy()

    return best_a, best_val

def try_update(best_seq, best_score, a, label, t_global):

        return best_seq, best_score

        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")

    if t1 > 10:
        a_param, _ = parametric_init(n0, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "parametric", t_global)

    n_best = len(best_seq)

    # Phase 2: LSE Mirror Descent
    print("\n=== Phase 2: LSE-MD (n=" + str(n_best) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        a_out, _ = lse_mirror_descent(norm_sum1(best_seq), t2,
                                      mu_init=4e-4, mu_min=3e-6,
                                      mu_decay=0.9997, lr=1.2,
                                      momentum=0.8, restart_every=3500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lse_md_n" + str(n_best), t_global)

    n_best = len(best_seq)

    # Phase 3: LP-FW main
    print("\n=== Phase 3: LP-FW (n=" + str(n_best) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.25, 420.0)
    if t3 > 60 and n_best <= 10000:
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t3, solver,
                                  initial_K=400, max_K=3500,
                                  margin_frac=0.015, add_per_iter=50, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(n_best), t_global)

    n_best = len(best_seq)

    # Phase 4: Multi-n exploration
    print("\n=== Phase 4: Multi-n exploration ===")
    try_ns = [n_best - 500, n_best + 500, n_best + 1000, n_best - 1000]
    try_ns = sorted(set([x for x in try_ns if 3000 <= x <= 12000]))

    for new_n in try_ns[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print("\n  Trying n=" + str(new_n) + " (remaining=" + str(round(remaining, 1)) + "s)")
        a_long = interpolate_seq(best_seq, new_n)

        t_lp = min(remaining * 0.4, 250.0)
        if t_lp > 40 and new_n <= 10000:
            a_out, _ = lp_frank_wolfe(a_long, t_lp, solver,
                                      initial_K=350, max_K=3000,
                                      margin_frac=0.02, add_per_iter=40, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t_md = min(remaining * 0.25, 120.0)
            if t_md > 20:
                a_out, _ = lse_mirror_descent(norm_sum1(best_seq), t_md,
                                              mu_init=3e-4, mu_min=3e-6,
                                              mu_decay=0.9998, lr=1.0,
                                              momentum=0.75, restart_every=2500, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lse_md_n" + str(new_n), t_global)

    n_best = len(best_seq)

    # Phase 5: Final LP-FW at best n
    print("\n=== Phase 5: Final LP-FW (n=" + str(n_best) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 90 and n_best <= 10000:
        t5 = min(remaining * 0.35, 350.0)
        if t5 > 60:
            a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t5, solver,
                                      initial_K=500, max_K=4000,
                                      margin_frac=0.01, add_per_iter=60, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    n_best = len(best_seq)

    # Phase 6: Final LSE-MD polish
    print("\n=== Phase 6: Final LSE-MD (n=" + str(n_best) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50:
        t6 = min(remaining * 0.25, 150.0)
        if t6 > 25:
            a_out, _ = lse_mirror_descent(norm_sum1(best_seq), t6,
                                          mu_init=2e-5, mu_min=2e-6,
                                          mu_decay=0.9999, lr=0.8,
                                          momentum=0.7, restart_every=5000, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lse_md_polish", t_global)

    n_best = len(best_seq)

    # Phase 7: Coordinate descent polish
    print("\n=== Phase 7: Coord Descent (n=" + str(n_best) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t7 = min(remaining * 0.5, 200.0)
        if t7 > 20:
            a_out, _ = coordinate_descent(norm_sum1(best_seq), t7, rng_seed=42, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
