def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_grad(a, beta, top_k=None):
    """Compute gradient of smoothed objective (1/beta)*log(sum exp(beta*b_k)) w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)

        p = np.zeros(len(b))
        p[idx] = np.exp(beta * (b[idx] - b[idx].max()))
        p[idx] /= p[idx].sum()
    else:

        b_shifted = np.clip(b_shifted, -300, 300)
        p = np.exp(b_shifted)
        p /= p.sum()

def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1)
    rho_f = np.where(cond)[0]
    if len(rho_f) == 0:
        return np.ones(n) / n
    rho = rho_f[-1] + 1
    theta = (cssv[rho-1] - 1) / rho
    return np.maximum(v - theta, 0.0)

def objective_and_grad_x(x, beta, top_k=None):
    """Objective and gradient in x-space (softmax parameterization)."""
    # Convert x to a via softmax

    a = np.exp(x_shifted)
    a_sum = a.sum()

            break
        
        # For each beta, run L-BFGS-B
        max_iter = 200
        result = minimize(
            lambda x: objective_and_grad_x(x, beta, top_k=max(20, n//10))[0],
            x,
            jac=lambda x: objective_and_grad_x(x, beta, top_k=max(20, n//10))[1],
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'maxfun': max_iter*10, 'gtol': 1e-8, 'iprint': 0}
        )
        
        # Update
        x = result.x
        a_new = np.exp(x - x.max())
        a_new /= a_new.sum()
        a_new = np.maximum(a_new, 0)
        a_new /= a_new.sum()
        
        M = fast_autocorr(a_new).max()
        J = 2.0 * n * M
        if J < best_J:
            best_J = J
            best_a = a_new.copy()
        
        a = a_new
        x = np.log(np.maximum(a, 1e-15))
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"  L-BFGS-B beta={beta:.1f} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_J / (2.0 * n)

"""
Minimizes 2*n*max(autocorr(a))/(sum(a))^2 using:
1. L-BFGS-B on log-sum-exp smoothed objective with softmax parameterization
2. LP-Frank-Wolfe with adaptive active set
3. Coordinate descent with exact autocorrelation evaluation
4. Multi-length exploration with interpolation
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1] + 1
    theta = (cssv[idx[-1]] - 1) / rho_val
    return np.maximum(v - theta, 0.0)

def compute_grad(a, beta, top_k=None):
    n = len(a)
    b = fast_autocorr(a)
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        p = np.zeros(len(b))
        p[idx] = np.exp(beta * (b[idx] - b[idx].max()))
        p[idx] /= p[idx].sum()
    else:
        shifted = beta * (b - b.max())
        shifted = np.clip(shifted, -300, 300)
        p = np.exp(shifted)
        p /= p.sum()
    a_flip = a[::-1]
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    Af = np.fft.rfft(a_flip, fft_size)
    conv = np.fft.irfft(P * Af, fft_size)[:conv_len]
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, p, b

def softmax_a(x):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    s = a.sum()
    return a / s

def obj_grad_x(x, beta, top_k):
    a = softmax_a(x)
    grad_a, p, b = compute_grad(a, beta, top_k)
    val = b.max() + np.log(np.sum(np.exp(beta * (b - b.max())))) / beta
    g_dot_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - g_dot_a)
    return val, grad_x

def lbfgs_stage(a0, time_limit, beta_start=20.0, beta_end=500.0, n_betas=8, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    betas = np.geomspace(beta_start, beta_end, n_betas)
    top_k = max(30, n // 8)
    
    x = np.log(np.maximum(a, 1e-30))
    
    for bi, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit:
            break
        max_iter = 150
        
        def fun(x_):
            return obj_grad_x(x_, beta, top_k)[0]
        
        def jac(x_):
            return obj_grad_x(x_, beta, top_k)[1]
        
        res = minimize(fun, x, jac=jac, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'gtol': 1e-7, 'iprint': 0})
        x = res.x
        a_new = softmax_a(x)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        M_new = fast_autocorr(a_new).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a_new.copy()
        a = a_new
        x = np.log(np.maximum(a, 1e-30))
        
        if verbose:
            el = time.monotonic() - t0
            print("  LBFGS b=%.1f (%.1fs): M=%.2e J=%.6f" % (beta, el, best_M, 2*n*best_M))
    
    return best_a, best_M

def lp_fw(a0, time_limit, solver, K_init=300, K_max=2500, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    ac = fast_autocorr(y)
    active = set(int(k) for k in np.argsort(ac)[::-1][:K_init])
    no_imp = 0
    max_no_imp = 800
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_imp > max_no_imp:
            break
        
        ac = fast_autocorr(y)
        Mc = ac.max()
        if Mc < best_M - 1e-15:
            best_M = Mc
            best_a = y.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        cl = sorted(active)
        if len(cl) > K_max:
            margins = np.array([Mc - ac[k] for k in cl])
            keep = np.argsort(margins)[:K_max]
            active = set(int(cl[i]) for i in keep)
            cl = sorted(active)
        
        K = len(cl)
        if K == 0:
            break
        
        C = np.zeros((K, n))
        for jj, k in enumerate(cl):
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            if s < e:
                idx = np.arange(s, e)
                C[jj, idx] = y[ki - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        prob = cp.Problem(cp.Minimize(M_var), [cp.sum(g_var) == 1, C @ g_var <= M_var])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active) > K_init // 2:
                active = set(int(k) for k in np.argsort(ac)[::-1][:K_init // 2])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        gv = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = gv.sum()
        if gs < 1e-15:
            continue
        g_dir = gv / gs
        
        # Find violations
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        ysg = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        viols = np.where(ysg > Mc + 1e-8)[0]
        if len(viols) > 0:
            amt = ysg[viols] - Mc
            top_v = viols[np.argsort(amt)[::-1][:25]]
            for k in top_v:
                active.add(int(k))
        
        # Line search
        alpha_def = 2.0 / (t + 2.0)
        alphas = [alpha_def, max(0.01, alpha_def * 0.5), max(0.01, alpha_def * 2.0)]
        best_alpha = alpha_def
        best_trial = float('inf')
        for al in alphas:
            trial = (1 - al) * y + al * g_dir
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            Mt = fast_autocorr(trial).max()
            if Mt < best_trial:
                best_trial = Mt
                best_alpha = al
        
        y = (1 - best_alpha) * y + best_alpha * g_dir
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        if verbose and t % 100 == 0:
            el = time.monotonic() - t0
            print("  LPFW t=%d (%.1fs): J=%.6f |S|=%d" % (t, el, 2*n*best_M, len(active)))
    
    return best_a, best_M

def coord_descent(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 2000:
        ac = fast_autocorr(a)
        # Get top peaks
        n_top = min(50, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:n_top]
        improved = False
        
        # For each top peak, try to reduce it
        for ki in top_idx[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(ki)
            # Find the pair (i, j) that contributes most to this peak
            s = max(0, k - n + 1)
            e = min(n - 1, k)
            if s > e:
                continue
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[k - idx]
            max_c = contrib.max()
            if max_c < 1e-20:
                continue
            
            src = int(idx[np.argmax(contrib)])
            partner = k - src
            
            # Try moving mass from src to various destinations
            # First try: random destinations
            for _ in range(5):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                    # Skip self-move
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    trial = a.copy()
                    trial[src] -= amt
                    trial[dst] += amt
                    trial = np.maximum(trial, 0.0)
                    st = trial.sum()
                    if st < 1e-15:
                        continue
                    trial = trial / st
                    Mt = fast_autocorr(trial).max()
                    if Mt < best_M - 1e-16:
                        a = trial
                        best_M = Mt
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
        
        if not improved:
            # Try random moves
            for _ in range(10):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.003, 0.03)
                    amt = a[i] * frac
                    trial = a.copy()
                    trial[i] -= amt
                    trial[j] += amt
                    trial = np.maximum(trial, 0.0)
                    st = trial.sum()
                    if st > 1e-15:
                        trial = trial / st
                        Mt = fast_autocorr(trial).max()
                        if Mt < best_M:
                            a = trial
                            best_M = Mt
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
            if not improved:
                stall += 1
        
        if verbose and stall % 500 == 0 and stall > 0:
            el = time.monotonic() - t0
            print("  Coord (%.1fs): J=%.6f stall=%d" % (el, 2*n*best_M, stall))
    
    return best_a, best_M
