```python
    if s < 1e-15:
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse initialization (parent + parametric families)
2. LP-based Frank-Wolfe with scipy HiGHS solver and efficient sparse constraints
3. Away-step acceleration for faster convergence
4. Adaptive active set management
5. Coordinate descent polishing
6. Multi-n exploration
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_obj_normalized(a):
    """Compute 2*n*max_conv assuming sum(a)=1."""
    ac = fast_convolve(a)
    return 2.0 * len(a) * ac.max()

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def build_sparse_constraint(y, active_set, n):
    """Build sparse matrix A where A[k, i] = y[k-i] for valid i."""
    K = len(active_set)
    rows = np.zeros(K * n, dtype=np.int32)  # over-allocate
    cols = np.zeros(K * n, dtype=np.int32)
    vals = np.zeros(K * n, dtype=np.float64)
    count = 0
    for jj, k in enumerate(active_set):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            v = y[k_int - idx]
            mask = v > 1e-18
            for ii in np.where(mask)[0]:
                rows[count] = jj
                cols[count] = i_start + ii
                vals[count] = v[ii]
                count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(K, n))

def solve_lp_highs(y, active_set, n):
    """Solve: min M s.t. A@g <= M, sum(g) = 1, g >= 0, M >= 0."""
    K = len(active_set)
    if K == 0:
        return None
    
    A_sparse = build_sparse_constraint(y, active_set, n)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    # A_ub: [A_sparse | -I_K] <= 0
    # Build as sparse
    A_ub_sparse = A_sparse.tocoo()
    r = list(A_ub_sparse.row)
    cidx = list(A_ub_sparse.col)
    v = list(A_ub_sparse.data)
    # Add -I_K in column n
    for jj in range(K):
        r.append(jj)
        cidx.append(n)
        v.append(-1.0)
    
    A_ub = csr_matrix((v, (r, cidx)), shape=(K, n + 1))
    b_ub = np.zeros(K)
    
    # A_eq: [1, 1, ..., 1, 0] = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    
    g = np.maximum(res.x[:n], 0.0)
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s

def lp_fw(a0, time_limit, initial_K=300, max_K=3000, add_per_iter=30, verbose=True):

            no_improve += 1

        if len(active_set) < max_K:

                if k not in active_set_set:

def coordinate_polish(a0, time_limit, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        top_k = np.argsort(ac)[::-1][:30]
        
        improved = False
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find destination: index with minimal contribution to top peaks
            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]
            
            # Prefer indices not in top peak regions
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2+1] = True
            
            candidates = np.where(~in_peak)[0] if (~in_peak).sum() > n // 4 else np.arange(n)
            dst = int(candidates[np.argmin(scores[candidates])])
            
            if dst == src:
                continue
            
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] = a_trial[src] - amt
                a_trial[dst] = a_trial[dst] + amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            if stall % 50 == 0:
                # Random perturbation
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.003, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
    
    return best_a, best_m
```

```python
        t += 1
        
        ac = fast_convolve(x)
        M_curr = ac.max()

            best_m = M_curr
            best_a = x.copy()

        else:
            stall += 1
            if stall > 5000:
                break

        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]

        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            continue
        w = w / w_sum

                # Also add the symmetric contribution
                sym_idx = np.arange(k_int, k_int + (i_end - i_start))
                valid = (sym_idx >= 0) & (sym_idx < n)
                if valid.any():
                    grad[sym_idx[valid]] += 2.0 * wi * x[idx[valid]]
        
        # Mirror descent update (exponential family)
```
