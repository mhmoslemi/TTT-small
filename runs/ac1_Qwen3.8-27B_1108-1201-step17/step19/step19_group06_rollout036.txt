    # = sum_{m=0}^{n-1} a[m] * w[2n-2-j-m] (since a_padded[m]=0 for m>=n)
    # Let p = 2n-2-j-m, so m = 2n-2-j-p:
    # = sum_p a[2n-2-j-p] * w[p] (where 0<=2n-2-j-p<n, i.e., p in [n-2-j, 2n-2-j])
    # This is getting too complicated.
    # 
    # Let me just use the direct approach with numpy's stride tricks or just a loop.
    # For n up to ~10000, O(n^2) with numpy vectorization should be OK.
    
    # Actually, I just realized: I can compute h using FFT by noting:
    # h[j] = <w[j:j+n], a> = <w, a_shifted> where a_shifted is a shifted version.
    # 
    # The vector h is the result of correlating w with a. In FFT terms:
    # h = ifft(fft(w, L) * conj(fft(a, L))) ... this gives the circular cross-correlation.
    # With sufficient padding (L >= 2n-1), this gives the linear cross-correlation.
    
    # The linear cross-correlation (w ⋆ a)[j] = sum_m w[m+j] * conj(a[m])
    # For real signals: = sum_m w[m+j] * a[m] = h[j]
    # 
    # Yes! So h is the linear cross-correlation of w and a.
    # We can compute it as: h_full = ifft(fft(w, L) * conj(fft(a, L)))[:2n-1]
    # Wait, cross-correlation in FFT: IFFT(FFT(x) * conj(FFT(y))) gives the cross-correlation.
    # (x ⋆ y)[n] = sum_m x[m+n] * conj(y[m])
    # 
    # So (w ⋆ a)[j] = sum_m w[m+j] * conj(a[m]) = sum_m w[m+j] * a[m] = h[j] ✓
    
    # Great! So:
    L = 1 << (3 * n - 2).bit_length()  # Sufficient padding for linear correlation
    W_fft = np.fft.rfft(w, L)
    A_fft = np.fft.rfft(a, L)
    h_full = np.fft.irfft(W_fft * np.conj(A_fft), L)
    h = h_full[:n]  # h[j] = (w ⋆ a)[j] = sum_m w[m+j] * a[m]
    
    # Gradient of LSE w.r.t. a[j] = 2 * beta * h[j]
    grad_lse = 2.0 * beta * h
    
    # Full gradient of obj = 2n * LSE / (beta * S^2)
    # d(obj)/da[j] = 2n/(beta*S^2) * grad_lse[j] - 2n*LSE*(2*a[j])/(beta*S^3)
    # = 2n/(beta*S^2) * (grad_lse[j] - 2*LSE*a[j]/S)
    
    grad = (2.0 * n / (beta * S * S)) * (grad_lse - 2.0 * LSE * a / S)
    
    return obj, grad

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smooth approximation with L-BFGS-B and FFT gradients
2. Adaptive beta scheduling for progressive sharpening
3. Multiplicative mirror descent on the simplex
4. Targeted coordinate descent on peak lags
5. Multi-start with diverse parametric initializations
6. Length exploration via interpolation
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Compute full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score_normalized(a):
    """Compute score for a normalized (sum=1) non-negative sequence."""
    n = len(a)
    c = fast_convolve(a)
    return float(2.0 * n * c.max())


def evaluate_safe(a):
    """Safely call the evaluation function."""
    try:
        a_list = [float(max(0.0, x)) for x in a]
        if not a_list:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def smooth_obj_grad(x, beta, n):
    """
    Compute smooth approximation of 2n*logsumexp(beta*c)/S^2 and gradient.
    c = conv(a,a), S = sum(a). a = max(x, 0).
    """
    a = np.maximum(x, 0.0)
    S = a.sum()
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c = fast_convolve(a)
    beta_c = beta * c
    m = beta_c.max()
    exp_shifted = np.exp(beta_c - m)
    LSE = m + np.log(exp_shifted.sum())
    w = exp_shifted / exp_shifted.sum()

    obj = 2.0 * n * LSE / (beta * S * S)

    # Gradient: need h[j] = sum_m w[j+m] * a[m] = cross-correlation (w * a)[j]
    # Using FFT: h = ifft(fft(w) * conj(fft(a)))
    L = 1 << (3 * n - 2).bit_length()
    W_fft = np.fft.rfft(w, L)
    A_fft = np.fft.rfft(a, L)
    h_full = np.fft.irfft(W_fft * np.conj(A_fft), L)
    h = h_full[:n]

    grad_lse = 2.0 * beta * h
    grad = (2.0 * n / (beta * S * S)) * (grad_lse - 2.0 * LSE * a / S)

    return obj, grad


def lgbf_optimize(a0, time_limit, beta_start=15.0, beta_end=500.0, n_betas=8,
                  max_iter=500, tol=1e-8, verbose=False):
    """
    Run L-BFGS-B with increasing beta for progressive sharpening.
    """
    n = len(a0)
    x0 = np.maximum(a0.astype(np.float64), 0.0)
    best_x = x0.copy()
    best_score = score_normalized(norm_sum1(x0))
    t0 = time.monotonic()

    betas = np.geomspace(beta_start, beta_end, n_betas)

    for bi, beta in enumerate(betas):
        time_remaining = time_limit - (time.monotonic() - t0)
        if time_remaining < 10:
            break

        def objfun(x):
            obj, _ = smooth_obj_grad(x, beta, n)
            return obj

        def objgrad(x):
            _, g = smooth_obj_grad(x, beta, n)
            return g

        try:
            res = minimize(
                objfun, x0, jac=objgrad, method='L-BFGS-B',
                bounds=[(0, None)] * n,
                options={'maxiter': max_iter, 'ftol': tol, 'gtol': 1e-6,
                         'maxls': 50, 'maxfun': 10000}
            )
            x_new = np.maximum(res.x, 0.0)
            S = x_new.sum()
            if S > 1e-10:
                x_new = x_new / S
                sc = score_normalized(x_new)
                if sc < best_score:
                    best_score = sc
                    best_x = x_new.copy()
                    if verbose:
                        print(f"    L-BFGS beta={beta:.1f}: score={sc:.6f} (iter {res.nit})")
                x0 = x_new
            else:
                x0 = best_x.copy()
        except Exception as e:
            if verbose:
                print(f"    L-BFGS beta={beta:.1f} failed: {e}")
            x0 = best_x.copy()

    return best_x, best_score


def mirror_descent_v2(a0, time_limit, lr_init=0.15, beta_init=5.0,
                      top_k_init=30, restart_every=3000, verbose=False, rng_seed=42):
    """
    Improved mirror descent with adaptive parameters and better restarts.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    a_prev = a.copy()
    momentum = 0.8
    lr = lr_init
    beta = beta_init
    top_k = top_k_init
    stall = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1

        # Nesterov in multiplicative space
        ratio = a / np.maximum(a_prev, 1e-30)
        ratio = np.clip(ratio, 0.01, 100.0)
        a_nest = a * np.power(ratio, momentum)
        a_nest = norm_sum1(a_nest)

        # Compute autocorrelation and gradient
        c = fast_convolve(a_nest)
        k_alarm = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k_alarm]
        top_vals = c[top_idx]

        max_v = top_vals.max()
        expo = beta * (top_vals - max_v)
        expo = np.clip(expo, -50, 50)
        weights = np.exp(expo)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones(k_alarm) / k_alarm
        else:
            weights = weights / w_sum

        # Compute gradient via FFT cross-correlation
        # grad[j] = 2 * sum_k w_k * a[k-j] = 2 * (w * a)[j] (cross-corr)
        L = 1 << (3 * n - 2).bit_length()
        W_fft = np.fft.rfft(weights, L)
        A_fft = np.fft.rfft(a_nest, L)
        h = np.fft.irfft(W_fft * np.conj(A_fft), L)[:n]
        grad = 2.0 * h

        # Adaptive learning rate
        eta = lr / (1.0 + 0.03 * np.sqrt(step))

        # Multiplicative update
        log_a = np.log(np.maximum(a_nest, 1e-30)) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        a_prev = a
        a = a_new

        # Check improvement
        c_new = fast_convolve(a)
        m_new = c_new.max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive top_k
        if step % 500 == 0:
            top_k = min(top_k + 3, 200)
            beta = min(beta + 0.1, 15.0)

        # Adaptive learning rate decay
        if step % 1000 == 0:
            lr *= 0.95

        # Restart when stuck
        if stall > restart_every:
            stall = 0
            pert_type = int(rng.integers(5))
            if pert_type == 0:
                x = np.linspace(0, 1, n)
                c_center = float(rng.uniform(0.2, 0.8))
                sig = float(rng.uniform(0.05, 0.4))
                pert = np.exp(-((x - c_center) ** 2) / (2 * sig ** 2))
            elif pert_type == 1:
                i_arr = np.arange(n, dtype=np.float64)
                p = float(rng.uniform(0.3, 3.0))
                pert = (i_arr + 1.0) ** p
            elif pert_type == 2:
                x = np.linspace(0, 1, n)
                freq = int(rng.integers(1, 7))
                phase = float(rng.uniform(0, np.pi))
                pert = (1 + np.pi * freq * x + phase) ** 2
            elif pert_type == 3:
                # Geometric
                r = float(rng.uniform(0.3, 0.7))
                pert = r ** np.arange(n, dtype=np.float64)
            else:
                pert = best_a * (1.0 + rng.normal(0, 0.1, n))

            pert = norm_sum1(pert)
            w_mix = float(rng.uniform(0.2, 0.6))
            a = norm_sum1((1 - w_mix) * best_a + w_mix * pert)
            a_prev = a.copy()
            lr = max(lr * 0.7, 0.01)
            beta = max(beta - 1.0, 2.0)

            if verbose:
                J = 2.0 * n * best_m
                print(f"    MD2 restart step={step} ({time.monotonic()-t0:.0f}s): "
                      f"J={J:.6f}, lr={lr:.4f}, beta={beta:.1f}, top_k={top_k}")

    if verbose:
        J = 2.0 * n * best_m
        print(f"    Mirror descent v2 done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def coord_descent_v2(a0, time_limit, rng_seed=77, verbose=False):
    """
    Targeted coordinate descent: move mass away from peak-contributing positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    trial_count = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        M_curr = c.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15000:
                # Perturb and reset
                n_moves = int(rng.integers(10, 50))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = float(rng.uniform(0.001, 0.02))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                    m_new = fast_convolve(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                        stall = 0
                continue

        # Find top lags
        n_top = min(20, 2 * n - 1)
        top_lags = np.argsort(c)[::-1][:n_top]

        improved = False

        for k_pos in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            if improved:
                break

            k_int = int(k_pos)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo >= i_hi:
                continue

            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            valid = (j_idx >= 0) & (j_idx < n)
            idx = idx[valid]
            j_idx = j_idx[valid]
            contrib = a[idx] * a[j_idx]

            if contrib.size == 0:
                continue

            # Find top contributing pairs
            n_pairs = min(5, len(idx))
            pair_order = np.argsort(contrib)[::-1][:n_pairs]

            for pi in pair_order:
                if time.monotonic() - t0 > time_limit:
                    break

                src = int(idx[pi])
                src_val = a[src]
                if src_val < 1e-14:
                    continue

                # Find destination: minimize impact on top lags
                dest_scores = np.zeros(n)
                for k2 in top_lags[:6]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n, k2i + 1)
                    if s2 < e2:
                        ii = np.arange(s2, e2)
                        jj = k2i - ii
                        v = (jj >= 0) & (jj < n)
                        dest_scores[ii[v]] += a[jj[v]] * c[k2i]
                        dest_scores[jj[v]] += a[ii[v]] * c[k2i]

                dest_scores[src] = np.inf
                # Prefer positions with low values (less likely to create new peaks)
                dest_scores += 0.1 * a
                dst = int(np.argmin(dest_scores))

                if dst == src:
                    continue

                # Try transfer
                for frac in [0.03, 0.08, 0.15, 0.25, 0.4]:
                    trial_count += 1
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_convolve(a_trial).max()
                    if m_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break

        if not improved and stall > 5000 and stall % 1000 == 0:
            # Random perturbation
            n_moves = int(rng.integers(5, 30))
            for _ in range(n_moves):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = float(rng.uniform(0.002, 0.01))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-10:
                a = a / s

    if verbose:
        J = 2.0 * n * best_m
        print(f"    Coord descent v2 done ({time.monotonic()-t0:.0f}s): J={J:.6f}, trials={trial_count}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def parametric_init(n, time_limit=20.0):
    """Generate diverse parametric initializations and return the best."""
    t0 = time.monotonic()
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)

    candidates = []

    # Uniform
    candidates.append(np.ones(n))

    # Geometric (various ratios)
    for r in [0.3, 0.4, 0.5, 0.6, 0.7, 0.8]:
        candidates.append(r ** i_arr)
        candidates.append(r ** (n - 1 - i_arr))

    # Power laws
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)

    # Gaussians
    for c in [0.3, 0.5, 0.7]:
        for sig in [0.1, 0.2, 0.3, 0.5]:
            candidates.append(np.exp(-((x - c) ** 2) / (2 * sig ** 2)))

    # Bimodal
    for sig in [0.1, 0.2, 0.3]:
        candidates.append(
            np.exp(-((x - 0.25) ** 2) / (2 * sig ** 2)) +
            np.exp(-((x - 0.75) ** 2) / (2 * sig ** 2)))

    # Raised cosine variants
    candidates.append(1 + np.cos(2 * np.pi * x))
    candidates.append(1 + np.cos(4 * np.pi * x))

    # Exponential
    for r in [1, 2, 5, 10]:
        candidates.append(np.exp(-r * x))
        candidates.append(np.exp(r * x))

    # Sine
    for f in [1, 2, 3, 5]:
        candidates.append(np.sin(np.pi * f * x) ** 2)

    # Wedge (asymmetric linear)
    candidates.append(1.0 + 2.0 * x)
    candidates.append(3.0 - 2.0 * x)
    candidates.append(1.0 + 5.0 * x)

    # Evaluate all
    best_cand = None
    best_val = float('inf')
    for c in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        c_norm = norm_sum1(c)
        v = score_normalized(c_norm)
        if v < best_val:
            best_val = v
            best_cand = c_norm.copy()

    return best_cand if best_cand is not None else np.ones(n) / n
