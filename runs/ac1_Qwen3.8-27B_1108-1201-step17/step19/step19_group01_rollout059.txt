def fast_autocorr(a):
    """Compute full linear autocorrelation convolve(a,a) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]

def norm1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def simplex_project(z):
    """Project onto the probability simplex (sum=1, non-negative)."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_smoothed_grad(a, beta, top_k):
    """
    Compute gradient of (1/beta)*log(sum_{k in topK} exp(beta*b_k)) w.r.t. a.
    Returns (gradient, weights, autocorrelation).
    """
    n = len(a)
    b = fast_autocorr(a)
    size = len(b)

    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
    else:
        b_use = b

    b_max = b_use.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros(size)
        weights[np.argmax(b_use)] = 1.0
    else:
        weights /= wsum

    # grad[i] = 2/beta * sum_j a[j] * w[j+i]
    # = 2/beta * (a * w_flip)[size-1-i]
    w_flip = weights[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a, fft_size)
    Wf = np.fft.rfft(w_flip, fft_size)
    conv = np.fft.irfft(Ar * Wf, fft_size)[:conv_len]
    grad = 2.0 * conv[size - 1 - np.arange(n)] / beta
    # grad[i] = 2*conv[size-1-i]/beta
    grad = 2.0 * conv[size - 1 : size - 1 - n : -1] / beta

    return grad, weights, b


def nesterov_optimize(a0, time_limit, beta_start, beta_end, beta_growth,
                      eta_base, top_k_ratio, restart_every, seed, verbose):
    """Nesterov accelerated projected gradient with adaptive beta and restarts."""
    n = len(a0)
    x = norm1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, b = compute_smoothed_grad(y, beta, top_k)

        eta = eta_base / max(1.0, beta * p.max() * 1.5)
        eta = min(eta, 0.1)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 60 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.015 / (1.0 + 0.05 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 5.0)

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("  Nest t={} ({}s): J={}, beta={}".format(t, round(elapsed,1), round(J, 6), round(beta, 1)))

    return best_a, best_max_b

        # Decide which to reduce (the one with more "leverage")

            reduce_i = src_i
        else:
            reduce_i = src_partner

        if a[reduce_i] < 1e-12:
            stall += 1
            continue

        # Find a safe destination: minimize impact on all top peaks

def anti_palindromic_init(n, seed=0):
    """Create an initialization with anti-palindromic bias."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    # Base: slight ramp (increasing)
    base = 0.3 + 0.4 * x + 0.3 * np.sin(np.pi * x)
    # Add anti-palindromic component: large on left, small on right (or vice versa)
    anti = 0.3 * (2.0 * x - 1.0)  # -0.3 to +0.3
    # Add some randomness
    noise = 0.05 * rng.standard_normal(n)
    a = base + anti + noise
    a = np.maximum(a, 0.01)
    return norm1(a)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {} (n={}, t={}s)".format(label, round(sc, 6), len(a), round(elapsed, 1)))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main search function. Uses multi-strategy optimization:
    1. Multi-length initialization with anti-palindromic bias
    2. Nesterov accelerated gradient (top-K smoothing)
    3. Peak surgery coordinate descent
    4. Block operations
    5. Final coordinate polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print("Time budget: {:.1f}s".format(time_budget))

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm1(parent)
                best_score = p_score
                print("Parent: n={}, score={}".format(len(parent), round(p_score, 6)))
    except Exception:
        pass

    # Phase 0: If no parent, create initial from parametric search
    if best_seq is None:
        print("\n=== Phase 0: Parametric init ===")
        n_init = 5000
        a_init = parametric_init(n_init, 30.0, norm1(np.ones(n_init)), 1.0)
        best_seq = a_init
        best_score = evaluate_safe(a_init)
        print("Init: n={}, score={}".format(n_init, round(best_score, 6)))

    n0 = len(best_seq)

    # Phase 1: Nesterov on current best length
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Nesterov (n={}) ===".format(n0))
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 60:
        a_out, _ = nesterov_optimize(norm1(best_seq), t1,
                                     beta_start=10.0, beta_end=800.0,
                                     beta_growth=1.005, eta_base=0.7,
                                     top_k_ratio=0.03, restart_every=2000,
                                     seed=1, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 2: Peak surgery
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: Peak Surgery (n={}) ===".format(len(best_seq)))
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_out, _ = peak_surgery(norm1(best_seq), t2, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "surg1", t_global)

    # Phase 3: Nesterov again with higher beta (finer smoothing)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 3: Nesterov fine (n={}) ===".format(n_cur))
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 60:
        a_out, _ = nesterov_optimize(norm1(best_seq), t3,
                                     beta_start=50.0, beta_end=2000.0,
                                     beta_growth=1.006, eta_base=0.5,
                                     top_k_ratio=0.02, restart_every=1500,
                                     seed=2, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest3", t_global)

    # Phase 4: Block operations
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 4: Block Ops (n={}) ===".format(len(best_seq)))
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 20:
        a_out, _ = block_operations(norm1(best_seq), t4, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "block1", t_global)

    # Phase 5: Try longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 1000
    if n_long <= 10000 and remaining > 200:
        print("\n=== Phase 5: Scale up to n={} ===".format(n_long))
        a_long = interpolate_sequence(best_seq, n_long)

        t5a = min(remaining * 0.25, 300.0)
        if t5a > 60:
            a_out, _ = nesterov_optimize(a_long, t5a,
                                         beta_start=15.0, beta_end=600.0,
                                         beta_growth=1.005, eta_base=0.7,
                                         top_k_ratio=0.03, restart_every=2000,
                                         seed=3, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_long", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        t5b = min(remaining * 0.10, 120.0)
        if t5b > 30:
            a_out, _ = peak_surgery(norm1(best_seq), t5b, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "surg_long", t_global)

    # Phase 6: Try shorter length (sometimes shorter is better)
    remaining = time_budget - (time.monotonic() - t_global)
    n_short = max(3000, len(best_seq) - 1500)
    if remaining > 150 and n_short != len(best_seq):
        print("\n=== Phase 6: Scale down to n={} ===".format(n_short))
        a_short = interpolate_sequence(best_seq, n_short)

        t6a = min(remaining * 0.25, 200.0)
        if t6a > 40:
            a_out, _ = nesterov_optimize(a_short, t6a,
                                         beta_start=15.0, beta_end=600.0,
                                         beta_growth=1.005, eta_base=0.7,
                                         top_k_ratio=0.03, restart_every=2000,
                                         seed=4, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_short", t_global)

    # Phase 7: Final Nesterov polish on best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print("\n=== Phase 7: Final Nesterov (n={}) ===".format(n_final))
    t7 = min(remaining * 0.30, 300.0)
    if t7 > 40:
        a_out, _ = nesterov_optimize(norm1(best_seq), t7,
                                     beta_start=100.0, beta_end=5000.0,
                                     beta_growth=1.008, eta_base=0.4,
                                     top_k_ratio=0.01, restart_every=1500,
                                     seed=5, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

    # Phase 8: Final peak surgery
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 8: Final Surgery (n={}) ===".format(len(best_seq)))
    t8 = min(remaining * 0.20, 150.0)
    if t8 > 20:
        a_out, _ = peak_surgery(norm1(best_seq), t8, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "surg_final", t_global)

    # Phase 9: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 9: Coord Polish (n={}) ===".format(len(best_seq)))
    t9 = min(remaining * 0.30, 120.0)
    if t9 > 15:
        a_out, _ = coordinate_polish(norm1(best_seq), t9, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Phase 10: One more Nesterov burst if time remains
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print("\n=== Phase 10: Burst Nesterov (n={}) ===".format(len(best_seq)))
        t10 = min(remaining * 0.70, 180.0)
        if t10 > 30:
            a_out, _ = nesterov_optimize(norm1(best_seq), t10,
                                         beta_start=200.0, beta_end=8000.0,
                                         beta_growth=1.01, eta_base=0.3,
                                         top_k_ratio=0.01, restart_every=1000,
                                         seed=6, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "burst", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={}, n={}, time={}s".format(round(best_score, 6), len(best_seq), round(elapsed, 1)))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
