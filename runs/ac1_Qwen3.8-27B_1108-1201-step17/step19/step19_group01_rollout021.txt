def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size, n=size)


def norm_sum1(a):
    """Normalize to sum=1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate the objective."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:

def simplex_project(z):
    """Project onto the probability simplex."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def spectral_shape(a0, time_limit, target_reduction=0.1, verbose=True):
    """
    Iterative spectral shaping: clip autocorrelation peaks and reconstruct.
    This is similar to the Gerchberg-Saxton / IRAM algorithm.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation
        B = fast_autocorr(a)
        B_max = B.max()
        
        if B_max < best_m - 1e-10:
            best_m = B_max
            best_a = a.copy()
        
        # Target: reduce peak by a factor
        target = best_m * (1.0 - target_reduction * 0.1)
        # Or use a fixed target based on the mean
        B_mean = B.sum() / len(B)  # = 1/(2n-1) for sum=1
        target = max(B_mean * 1.5, B_max * 0.95)
        
        # Clip peaks in the autocorrelation
        B_clipped = np.minimum(B, target)
        
        # Reconstruct: A_freq = sqrt(B_freq) (magnitudes only)
        B_freq = np.fft.rfft(B_clipped, fft_size)
        # A_freq magnitudes
        A_mag = np.sqrt(np.maximum(B_freq, 0))
        # Phase: use original phase
        A_orig = np.fft.rfft(np.pad(a, (0, fft_size - n)), fft_size)
        A_phase = np.angle(A_orig)
        A_new = A_mag * np.exp(1j * A_phase)
        
        # Inverse FFT
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        a = a_new
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral it={it} ({elapsed:.1f}s): J={J:.6f}, B_max={best_m:.8f}")
    
    return best_a, best_m


def spectral_shape_v2(a0, time_limit, verbose=True):
    """
    Improved spectral shaping: directly optimize the power spectrum.
    Work with |A|^2 and enforce a >= 0 via projection.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Initialize power spectrum
    A = np.fft.rfft(np.pad(a, (0, fft_size - n)), fft_size)
    W = np.abs(A) ** 2  # Power spectrum
    
    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Inverse FFT of W to get autocorrelation
        B = np.fft.irfft(W, fft_size, n=size)
        B_max = B.max()
        
        if B_max < best_m - 1e-10:
            best_m = B_max
            best_a = a.copy()
        
        # Modify W to reduce the peak
        # The peak in B is at some lag k_peak
        k_peak = int(np.argmax(B))
        
        # The contribution of frequency m to B[k_peak] is W[m]*cos(2*pi*m*k_peak/fft_size)
        # To reduce B[k_peak], we want to reduce W[m] where cos is large at k_peak
        # and increase where cos is small (or negative)
        
        # Simple approach: reduce W where it contributes most to the peak
        cos_vals = np.cos(2 * np.pi * np.arange(len(W)) * k_peak / fft_size)
        # Reduce W proportional to cos_vals (where positive)
        reduction = 0.01 * np.maximum(cos_vals, 0)
        W = np.maximum(W * (1 - reduction), 1e-20)
        
        # Reconstruct a
        A_mag = np.sqrt(W)
        A_phase = np.angle(A)
        A = A_mag * np.exp(1j * A_phase)
        a_new = np.fft.irfft(A, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
        
        # Update phase
        A = np.fft.rfft(np.pad(a, (0, fft_size - n)), fft_size)
        W = np.abs(A) ** 2
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SpecV2 it={it} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2000, verbose=True):
    """
    LP-Frank-Wolfe method for minimizing max autocorrelation.
    At each step, solve an LP to find the best direction g such that
    (f*g)[k] <= M for all active k, then take a line search step.
    """
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    # Initial active set: top K autocorrelation peaks
    ac = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])
    
    no_improve = 0
    max_no_improve = 800
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print(f"    LP-FW: converged at iter {t}")
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Manage active set size
        K = len(active_set)
        if K > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep_idx = np.argsort(margins)[:max_K]
            active_set = set(int(list(active_set)[i]) for i in keep_idx)
        
        constraint_list = sorted(active_set)
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix (sparse-friendly)
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k - idx]
        
        # Solve LP: minimize M subject to sum(g)=1, g>=0, coeffs@g <= M
        try:
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K // 2])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find new violations to add to active set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:20]]
            for k in top_v:
                active_set.add(int(k))
        
        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.01, default_alpha * 0.5),
                  max(0.01, default_alpha * 1.5), 1.0 / (t + 1), 2.0 / (t + 1)]
        alphas = list(set([max(0.01, min(1.0, a)) for a in alphas]))
        
        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        if t <= 3:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y
        y = y_new
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f} |S|={len(active_set)}")
    
    return best_a, best_m


def nesterov_smoothed(a0, time_limit, beta_start=20.0, beta_end=500.0,
                       beta_growth=1.005, eta_base=1.0, top_k_ratio=0.05,
                       restart_every=2000, verbose=True):
    """Nesterov accelerated projected gradient on log-sum-exp smoothed objective."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(123)
    top_k = max(20, int(n * top_k_ratio))
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        # Compute smoothed gradient
        b = fast_autocorr(y)
        if top_k < len(b):
            idx = np.argsort(b)[::-1][:top_k]
            b_use = np.zeros(len(b))
            b_use[idx] = b[idx]
            b_max = b[idx].max()
        else:
            b_use = b
            b_max = b.max()
        
        shifted = beta * (b_use - b_max)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        p = exp_vals / exp_vals.sum()
        
        # Gradient via convolution
        a_rev = y[::-1]
        conv_len = n + len(p) - 1
        conv_fft = 1 << (conv_len - 1).bit_length()
        Ar = np.fft.rfft(a_rev, conv_fft)
        Pr = np.fft.rfft(p, conv_fft)
        conv = np.fft.irfft(Ar * Pr, conv_fft)
        grad = conv[n - 1:2 * n - 1] / beta
        
        # Step size
        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)
        
        # Gradient step + projection
        z = y - eta * grad
        x_new = simplex_project(z)
        
        # Track best
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-14:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        # Adaptive beta
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        # Restart
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.02 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 5.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print(f"    Nest restart t={t} ({elapsed:.1f}s): J={J:.6f}")
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print(f"    Nest t={t} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}")
    
    return best_a, best_max_b


def coord_descent(a0, time_limit, verbose=True):
    """Peak-targeted coordinate descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    max_stall = 500
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False
        
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            
            # Determine which to reduce
            if a[src] >= (a[partner] if 0 <= partner < n else 0):
                reduce_idx = src
            else:
                reduce_idx = partner
            
            if a[reduce_idx] < 1e-12:
                continue
            
            # Find low-pressure destination
            pressure = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]
            
            exclude = set(int(tk) for tk in top_k[:8])
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                exclude.update(range(s2, e2 + 1))
            
            candidates = np.array([i for i in range(n) if i not in exclude], dtype=int)
            if len(candidates) < 10:
                candidates = np.arange(n)
            
            cand_pressure = pressure[candidates]
            dst = int(candidates[np.argmin(cand_pressure)])
            if dst == reduce_idx:
                mask = candidates != reduce_idx
                if mask.sum() > 0:
                    dst = int(candidates[mask][np.argmin(cand_pressure[mask])])
                else:
                    continue
            
            # Try fractions
            found = False
            for frac in [0.02, 0.05, 0.1, 0.2]:
                amt = a[reduce_idx] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            
            if not found and stall > 100:
                # Random move
                j = int(rng.integers(n))
                if j != reduce_idx and a[reduce_idx] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[reduce_idx] * frac
                    a[reduce_idx] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if not improved:
            stall += 1
            if stall % 50 == 0:
                # Broader random moves
                for _ in range(3):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.003, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
        
        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f} stall={stall}")
    
    return best_a, best_m


def simulated_annealing(a0, time_limit, T_start=0.01, T_end=0.001, verbose=True):
    """Simulated annealing to escape local minima."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    T = T_start
    steps = 0
    n_iter = 50000
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Generate random perturbation
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        
        # Move a small fraction of mass from i to j
        frac = rng.uniform(0.001, 0.05)
        amt = a[i] * frac
        if amt < 1e-15:
            continue
        
        a_trial = a.copy()
        a_trial[i] -= amt
        a_trial[j] += amt
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s
        
        M_trial = fast_autocorr(a_trial).max()
        delta = M_trial - best_m
        
        if delta < 0 or rng.random() < np.exp(-abs(delta) / max(T, 1e-10)):
            a = a_trial
            if M_trial < best_m:
                best_m = M_trial
                best_a = a_trial.copy()
        
        T = T_start * (T_end / T_start) ** (it / n_iter)
        steps += 1
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SA it={it} ({elapsed:.1f}s): J={J:.6f}, T={T:.6f}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n, time_limit, best_init, best_score):
    """Try parametric initializations."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score
    
    candidates = []
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    # Symmetric
    for p in [0.5, 1.0, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p * (n - i_arr) ** p)
    # Gaussian
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.3]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    # Cosine
    for freq in [1, 2, 3]:
        candidates.append(np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x)) / 2.0, 0.01))
    # Sine
    for p in [1, 2, 3, 4]:
        candidates.append(np.sin(np.pi * x) ** p)
    # Exponential
    for rate in [1, 2, 3]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(-rate * (1 - x)))
    # Two bumps
    for c1, c2 in [(0.3, 0.7), (0.2, 0.8), (0.4, 0.6)]:
        for sigma in [0.05, 0.1, 0.15]:
            candidates.append(np.exp(-((x - c1) ** 2) / (2 * sigma ** 2)) + np.exp(-((x - c2) ** 2) / (2 * sigma ** 2)))
    
    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(a_raw)
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()
    
    # Random
    rng = np.random.default_rng(99)
    for _ in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(rng.random(n) ** rng.uniform(0.5, 3.0))
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()
    
    return best_a, best_val

    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    solver = get_solver()
    print(f"Time budget: {time_budget:.1f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        parent = None
    
    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
