    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def exact_subgrad(a, k):
    """Subgradient of (a*a)_k w.r.t. a: grad[j] = 2*a[k-j] for valid j."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo <= hi:
        j_vals = np.arange(lo, hi + 1)
        grad[j_vals] = 2.0 * a[k - j_vals]
    return grad

def softmax_mirror_step(a, grad, eta):
    """Mirror descent step with entropic regularizer."""
    log_a = np.log(np.maximum(a, 1e-30))
    log_new = log_a - eta * grad
    log_new -= log_new.max()
    new = np.exp(log_new)
    s = new.sum()
    if s < 1e-30:
        return a.copy()
    return new / s

def stochastic_subgrad_descent(a0, time_limit, beta_start=5.0, beta_end=200.0,
                                eta0=0.5, verbose=True):
    """Stochastic subgradient mirror descent with exact gradients."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_c = fast_autocorr(a)
    best_m = best_c.max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_threshold = 5000
    
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        progress = min(iter_count / max(time_limit * 100, 1), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        eta = eta0 / (np.sqrt(t) + 1.0)
        
        # Compute autocorrelation
        c = fast_autocorr(a)
        
        # Pick k stochastically
        c_max = c.max()
        # Subsample to top positions for efficiency
        top_k = min(100, 2 * n - 1)
        top_idx = np.argpartition(-c, top_k)[:top_k]
        c_top = c[top_idx]
        exp_vals = np.exp(beta * (c_top - c_top.max()))
        probs = exp_vals / exp_vals.sum()
        k_star = top_idx[rng.choice(len(top_idx), p=probs)]
        
        # Exact subgradient
        grad = exact_subgrad(a, int(k_star))
        
        # Mirror descent step
        a_new = softmax_mirror_step(a, grad, eta)
        
        # Check improvement
        c_new = fast_autocorr(a_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > stall_threshold:
            # Perturb and restart
            pert = rng.normal(0, 0.01, n) * best_a
            a = norm_sum1(best_a + pert)
            stall = 0
            stall_threshold = max(stall_threshold * 0.85, 1500)
        else:
            a = a_new
        
        iter_count += 1
        if verbose and iter_count % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  SSD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, k*={k_star}")
    
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, initial_K=200, max_K=2000, alpha_init=0.3, verbose=True):
    """LP-Frank-Wolfe with scipy HiGHS solver."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_c = fast_autocorr(a)
    best_m = best_c.max()
    t0 = time.monotonic()
    
    # Initial active set: top K peaks
    size = 2 * n - 1
    init_idx = np.argsort(best_c)[::-1][:initial_K]
    active_set = list(init_idx.astype(int))
    
    iter_count = 0
    failed = 0
    
    while time.monotonic() - t0 < time_limit and failed < 20:
        if len(active_set) > max_K:
            # Prune: keep constraints with smallest margin
            c_curr = fast_autocorr(a)
            margins = np.array([c_curr.max() - c_curr[k] for k in active_set])
            keep = np.argsort(margins)[:max_K]
            active_set = [active_set[i] for i in keep]
        
        K = len(active_set)
        if K == 0:
            break
        
        # Build sparse constraint matrix
        # Row for constraint k: A[k, j] = a[k-j] if 0 <= k-j < n
        rows = []
        cols = []
        vals = []
        for ii, k in enumerate(active_set):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                j_range = np.arange(lo, hi + 1)
                rows.extend([ii] * len(j_range))
                cols.extend(j_range.tolist())
                vals.extend(a[k - j_range].tolist())
        
        A = sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))
        
        # LP: min M s.t. A*g <= M, sum(g) = 1, g >= 0
        # Variables: g_0, ..., g_{n-1}, M (total n+1 variables)
        # c vector: [0, ..., 0, 1] (minimize M)
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        
        # Inequality constraints: A*g - M <= 0
        # A_ineq = [A, -e_M]
        A_ineq = sparse.csr_matrix((
            vals + [-1.0] * K,
            (rows + list(range(K)), cols + [n] * K)
        ), shape=(K, n + 1))
        
        # Equality constraint: sum(g) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        # Bounds: g_i >= 0, M >= 0
        bounds = [(0, None)] * n + [(0, None)]
        
        try:
            res = linprog(c_vec, A_ub=A_ineq, b_ub=np.zeros(K),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception:
            failed += 1
            active_set = active_set[:min(initial_K, len(active_set))]
            continue
        
        if res.status != 0:
            failed += 1
            continue
        
        failed = 0
        g = np.maximum(res.x[:n], 0.0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            continue
        g = g / g_sum
        
        # Compute new autocorrelation for checking violations
        fft_size = 1 << (size - 1).bit_length()
        Af = np.fft.rfft(a, fft_size)
        Gf = np.fft.rfft(g, fft_size)
        ag = np.fft.irfft(Af * Gf, fft_size)[:size]
        
        # Add violated constraints
        current_M = ag.max()
        violations = np.where(ag > current_M + 1e-10)[0]
        if len(violations) > 0:
            new_k = violations[np.argmax(ag[violations])]  # most violated
            if int(new_k) not in active_set:
                active_set.append(int(new_k))
        
        # Frank-Wolfe update
        alpha = alpha_init * (2.0 / (iter_count + 2.0))
        # Actually, for FW: alpha_t = 2/(t+2) for the strong Wolfe condition
        # But we want to be more conservative for polishing
        alpha = min(0.1, 2.0 / (iter_count + 20.0))
        
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        c_new = fast_autocorr(a_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
        
        a = a_new
        iter_count += 1
        
        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, |S|={len(active_set)}, M={best_m:.8f}")
    
    return best_a, best_m

def parametric_init(n, best_ref=None):
    """Try diverse parametric families and return the best."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')
    
    candidates = []
    
    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr + 1.0) ** p)
    
    for alpha in [0.5, 1.0, 1.5]:
        for beta_p in [0.5, 1.0, 1.5]:
            candidates.append(np.maximum(i_arr + 1.0, 0.5) ** alpha *
                              np.maximum(n - i_arr, 0.5) ** beta_p)
    
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    
    for rate in [1, 2, 3, 5]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1.0)))
    
    for freq in [1, 2, 3]:
        candidates.append(np.maximum(np.sin(np.pi * freq * x) ** 2, 0.01))
    
    if best_ref is not None and len(best_ref) == n:
        ref_norm = norm_sum1(best_ref)
        for w in [0.3, 0.5, 0.7]:
            candidates.append(w * ref_norm + (1.0 - w) * norm_sum1((i_arr + 1.0) ** 0.8))
    
    for a_cand in candidates:
        a_norm = norm_sum1(a_cand)
        ac = fast_autocorr(a_norm)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_a = a_norm
    
    return best_a, best_val

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
