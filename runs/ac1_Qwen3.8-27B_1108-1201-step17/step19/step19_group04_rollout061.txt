Improved search for nonnegative step-function heights minimizing

Key improvements over previous version:
1. LP-based Frank-Wolfe with cutting-plane (growing active set) for tighter directions
2. Variance reduction steps (minimizing ||auto-conv||_2^2 via spectral gradient)

import math
import time
import numpy as np
from scipy.optimize import linprog


def _next_pow2(m):
    m = max(1, int(m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = _next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]


def _conv_ab(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = _next_pow2(m)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones_like(a) / a.size
    return a / s


def _objective(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float('inf')
    c = _self_conv(a)
    return 2.0 * float(n) * float(c.max())


def _variance_gradient(a):
    """Gradient of sum_j |F[j]|^4 w.r.t. a (for variance reduction direction)."""
    n = a.size
    m = 2 * n - 1
    size = _next_pow2(m)
    F = np.fft.fft(a, size)
    X = np.abs(F) ** 2 * F
    X[0] = 0.0
    grad_full = size * np.real(np.fft.ifft(X))
    return grad_full[:n]


def _solve_lp_single(a, active_idx, c_active):
    """Solve single LP: min T s.t. 2*(a*g)[k]-T <= c[k] for active k, sum g=1, g>=0."""
    n = a.size
    K = len(active_idx)
    if K == 0:
        return None, 0.0
    if K * (n + 1) > 12_000_000:
        return None, 0.0

    A = np.zeros((K, n + 1), dtype=np.float64)
    two_a = 2.0 * a
    for r in range(K):
        k = int(active_idx[r])
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            A[r, start:stop] = two_a[i0:i1][::-1]
        A[r, n] = -1.0

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = linprog(c_obj, A_ub=A, b_ub=c_active.astype(np.float64),
                      A_eq=A_eq, b_eq=np.array([1.0]),
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
    except Exception:
        return None, 0.0

    if res.status != 0 or not getattr(res, 'success', False):
        return None, 0.0

    g = np.maximum(res.x[:n], 0.0)
    s = float(g.sum())
    if s < 1e-10:
        return None, 0.0
    g = g / s
    T = float(res.x[n])
    return g, T


def _solve_lp_cutting_plane(a, c, c_max, max_rounds=4, init_K=200, max_K=600):
    """LP with cutting plane: grow active set to cover violated constraints."""
    n = a.size

    active = order[:init_K].copy()

    best_T = float('inf')

            best_g = g

        c_g = _conv_ab(a, g)

        if len(new_active) <= max_K:

        else:

def _line_search(a, d, c0, c0_max):
    """Minimize max of conv((1-a)*a+a*d, (1-a)*a+a*d) over alpha in [0,1]."""
    cross = _conv_ab(a, d)
    dd = _self_conv(d)

    def val(al):
        a1 = 1.0 - al
        arr = a1 * a1 * c0 + 2.0 * al * a1 * cross + al * al * dd
        return float(arr.max())

    best_alpha = 0.0
    best_val = c0_max

    grid = [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    for al in grid:
        v = val(al)
        if v < best_val:
            best_val = v
            best_alpha = al

    if best_alpha > 0.0:
        lo = max(0.0, best_alpha - 0.15)
        hi = min(1.0, best_alpha + 0.15)
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        a1 = hi - gr * (hi - lo)
        b1 = lo + gr * (hi - lo)
        fa = val(a1)
        fb = val(b1)
        for _ in range(28):
            if fa < fb:
                hi = b1
                b1 = a1
                fb = fa
                a1 = hi - gr * (hi - lo)
                fa = val(a1)
            else:
                lo = a1
                a1 = b1
                fa = fb
                b1 = lo + gr * (hi - lo)
                fb = val(b1)
        mid = 0.5 * (lo + hi)
        vm = val(mid)
        if vm < best_val:
            best_val = vm
            best_alpha = mid

        return a, c0_max

    new_a = np.maximum(new_a, 0.0)
    s = float(new_a.sum())
    if s < 1e-15:
        return a, c0_max
    new_a = new_a / s
    return new_a, best_val


def _spectral_shaping(a, alpha_pow=0.85):
    """Compress the power spectrum to reduce auto-conv peaks."""
    n = a.size
    m = 2 * n - 1
    size = _next_pow2(m)
    F = np.fft.fft(a, size)
    amp = np.abs(F)
    phase = np.angle(F)
    # Compress amplitude
    amp_new = amp ** alpha_pow
    # Keep DC fixed
    amp_new[0] = amp[0]
    F_new = amp_new * np.exp(1j * phase)
    # Enforce Hermitian symmetry
    F_new = (F_new + np.conj(F_new[::-1])) / 2.0
    F_new[0] = amp[0]
    z = np.real(np.fft.ifft(F_new))
    a_new = z[:n]
    a_new = np.maximum(a_new, 0.0)
    s = float(a_new.sum())
    if s < 1e-10:
        return a
    return a_new / s


def _interpolate_seq(a, new_n):
    a = _normalize(a)
    old_n = a.size

    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, a)
    return _normalize(g)

                d = -grad
                d = np.maximum(d, 0.0)
                s_d = float(d.sum())
                if s_d > 1e-10:
                    d = d / s_d
                    # Blend with current for stability
                    d = 0.4 * d + 0.6 * a
                    d = _normalize(d)
                    new_a2, new_sc2 = _line_search(a, d, c, c_max)
                    if new_sc2 < score - 1e-13:
                        a = new_a2
                        improved = True
                        no_improve = 0

            # Try spectral shaping
            if not improved and no_improve % 12 == 6:
                for ap in (0.7, 0.8, 0.9):
                    a_shaped = _spectral_shaping(a, ap)
                    sc_sh = _objective(a_shaped)
                    if sc_sh < score - 1e-13:
                        new_a3, new_sc3 = _line_search(a, a_shaped, c, c_max)
                        if new_sc3 < score - 1e-13:
                            a = new_a3
                            improved = True
                            no_improve = 0
                            break

            # Random perturbation if still stuck
            if not improved and no_improve > 50:
                rng = np.random.default_rng(hash(iter_no) % 2**31)
                for scale in (0.005, 0.01, 0.02, 0.05):
                    pert = rng.normal(0, scale, n)
                    d = a * (1.0 + pert)
                    d = np.maximum(d, 0.0)
                    s_d = float(d.sum())
                    if s_d < 1e-10:
                        continue
                    d = d / s_d
                    new_a4, new_sc4 = _line_search(a, d, c, c_max)
                    if new_sc4 < score - 1e-13:
                        a = new_a4
                        improved = True
                        no_improve = 0
                        break

        # Restart if very stuck
        if no_improve > 200:
            a = best.copy()
            no_improve = 0
            K = K_init

        iter_no += 1
        if verbose and (iter_no % 40 == 0 or (time.monotonic() - t0) % 40.0 < 0.5):
            el = time.monotonic() - t0
            print("  LP-FW: it=%d t=%.1fs best=%.9f cur=%.9f K=%d noimp=%d lp_t=%.2f"
                  % (iter_no, el, best_score, score, K, no_improve, lp_time))

    return best


def _coord_polish(a0, time_limit=60.0):
    """Coordinate descent polish: move mass from high-conv contributors to low ones."""
    t0 = time.monotonic()
    a = _normalize(a0)
    n = a.size
    best = a.copy()
    best_score = _objective(a)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 150:
        c = _self_conv(a)
        c_max = float(c.max())
        top_lags = np.argsort(c)[::-1][:min(10, len(c))]

        # Score each position by its contribution to top lags
        contrib = np.zeros(n)
        for k in top_lags[:6]:
            k = int(k)
            i0 = max(0, k - n + 1)
            i1 = min(n, k + 1)
            if i0 < i1:
                contrib[i0:i1] += a[k - (i1 - 1):k - i0 + 1][::-1]

        src = np.argsort(contrib)[::-1][:10]
        dst = np.argsort(contrib)[:10]

        improved = False
        for s_idx in src:
            if improved or time.monotonic() - t0 > time_limit:
                break
            for d_idx in dst:
                if improved:
                    break
                if int(s_idx) == int(d_idx):
                    continue
                v = float(a[s_idx])
                if v < 1e-12:
                    continue
                for frac in (0.05, 0.1, 0.2, 0.35):
                    amt = v * frac
                    b = a.copy()
                    b[s_idx] -= amt
                    b[d_idx] += amt
                    b = np.maximum(b, 0.0)
                    s_b = float(b.sum())
                    if s_b < 1e-15:
                        continue
                    b = b / s_b
                    sc = _objective(b)
                    if sc < best_score - 1e-13:
                        best = b.copy()
                        best_score = sc
                        a = b
                        improved = True
                        stall = 0
                        break

        if not improved:
            stall += 1

    return best


def _transform_search(a0, time_limit=30.0):
    """Try simple transforms: power, blend with uniform, reverse."""
    t0 = time.monotonic()
    a = _normalize(a0)
    n = a.size
    best = a.copy()
    best_score = _objective(a)

    for g in (0.3, 0.5, 0.7, 1.5, 2.0, 3.0, 5.0):
        if time.monotonic() - t0 > time_limit:
            break
        b = _normalize(a ** g)
        sc = _objective(b)
        if sc < best_score - 1e-12:
            best = b
            best_score = sc

    for w in (0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5):
        if time.monotonic() - t0 > time_limit:
            break
        b = _normalize((1 - w) * a + w / n)
        sc = _objective(b)
        if sc < best_score - 1e-12:
            best = b
            best_score = sc

    rev = a[::-1]
    for w in (0.1, 0.2, 0.3, 0.5, 0.7):
        if time.monotonic() - t0 > time_limit:
            break
        b = _normalize((1 - w) * a + w * rev)
        sc = _objective(b)
        if sc < best_score - 1e-12:
            best = b
            best_score = sc

    return best


def propose_candidate():
    t_global = time.monotonic()
    total_budget = 1650.0

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        p = p.ravel()
        if p.size > 0 and p.sum() > 1e-12:
            parent = p
    except Exception:
        pass

    if parent is not None:
        best_a = _normalize(parent)
        best_score = _objective(best_a)
        print("Parent: n=%d score=%.9f" % (best_a.size, best_score))
    else:
        n_def = 6700
        x = np.linspace(0, 1, n_def)
        best_a = _normalize(np.sin(np.pi * x) ** 2)
        best_score = _objective(best_a)
        print("Default start: n=%d score=%.9f" % (best_a.size, best_score))

    n0 = best_a.size

    # Generate diverse starts
    starts = _make_diverse_starts(n0, best_a)
    kept = _reduce_starts(starts, max_kept=5)

    # Ensure best is in the list
    matched = False
    for k in kept:
        if k.size == best_a.size:
            dot = float(np.dot(k, best_a))
            nrm = float(np.linalg.norm(k) * np.linalg.norm(best_a))
            if nrm > 1e-15 and dot / nrm > 0.999:
                matched = True
                break
    if not matched:
        kept.insert(0, best_a.copy())

    # Phase 1: Main LP-FW on the best start
    remaining = total_budget - (time.monotonic() - t_global)
    lp_time = min(1000.0, 0.55 * remaining)
    print("\nPhase 1: LP-FW on primary start (budget=%.0fs)" % lp_time)
    out = _lp_fw_search(best_a, lp_time, verbose=True)
    sc = _objective(out)
    if sc < best_score - 1e-12:
        best_score = sc
        best_a = out
        print("  Phase 1 improved: %.9f" % best_score)

    # Phase 2: Try second diverse start
    remaining = total_budget - (time.monotonic() - t_global)
    if len(kept) >= 2 and remaining > 250:
        cand2 = kept[1]
        # Check diversity
        dot = float(np.dot(cand2, best_a))
        nrm = float(np.linalg.norm(cand2) * np.linalg.norm(best_a))
        if nrm > 1e-15 and dot / nrm < 0.99:
            t2 = min(300.0, 0.22 * remaining)
            print("\nPhase 2: LP-FW on second start (budget=%.0fs)" % t2)
            out2 = _lp_fw_search(cand2, t2, verbose=False)
            sc2 = _objective(out2)
            if sc2 < best_score - 1e-12:
                best_score = sc2
                best_a = out2
                print("  Phase 2 improved: %.9f" % best_score)

    # Phase 3: Spectral shaping + re-optimize on best
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 150:
        print("\nPhase 3: Spectral shaping + refine (budget=%.0fs)" % remaining)
        for ap in (0.6, 0.7, 0.8, 0.9):
            a_sh = _spectral_shaping(best_a, ap)
            sc_sh = _objective(a_sh)
            if sc_sh < best_score - 1e-12:
                t_ref = min(200.0, 0.3 * remaining)
                out_sh = _lp_fw_search(a_sh, t_ref, verbose=False)
                sc_ref = _objective(out_sh)
                if sc_ref < best_score - 1e-12:
                    best_score = sc_ref
                    best_a = out_sh
                    print("  Spectral shaping (alpha=%.1f) improved: %.9f" % (ap, best_score))
                break

    # Phase 4: Try different lengths
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 200:
        print("\nPhase 4: Length exploration (budget=%.0fs)" % remaining)
        for dn in (-500, 500, 1000, -1000):
            remaining = total_budget - (time.monotonic() - t_global)
            if remaining < 100:
                break
            nn = n0 + dn
            if nn < 3000 or nn > 10000:
                continue
            a_new = _interpolate_seq(best_a, nn)
            sc_new = _objective(a_new)
            t_try = min(150.0, 0.3 * remaining)
            out_try = _lp_fw_search(a_new, t_try, verbose=False)
            sc_try = _objective(out_try)
            if sc_try < best_score - 1e-12:
                best_score = sc_try
                best_a = out_try
                print("  Length %d improved: %.9f" % (nn, best_score))

    # Phase 5: Final transforms
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t_tf = min(30.0, remaining * 0.5)
        out_tf = _transform_search(best_a, t_tf)
        sc_tf = _objective(out_tf)
        if sc_tf < best_score - 1e-12:
            best_score = sc_tf
            best_a = out_tf
            print("  Transform improved: %.9f" % best_score)

    # Phase 6: Coordinate polish
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t_cp = min(60.0, remaining * 0.7)
        out_cp = _coord_polish(best_a, t_cp)
        sc_cp = _objective(out_cp)
        if sc_cp < best_score - 1e-12:
            best_score = sc_cp
            best_a = out_cp
            print("  Coord polish improved: %.9f" % best_score)

    # Phase 7: Final LP-FW push
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 60:
        t_final = min(remaining - 10, 120.0)
        out_fin = _lp_fw_search(best_a, t_final, verbose=False)
        sc_fin = _objective(out_fin)
        if sc_fin < best_score - 1e-12:
            best_score = sc_fin
            best_a = out_fin
            print("  Final LP-FW improved: %.9f" % best_score)

    # Final validation
    out_arr = np.asarray(best_a, dtype=np.float64)
    out_arr = np.where(np.isfinite(out_arr), out_arr, 0.0)
    out_arr = np.maximum(out_arr, 0.0)
    out_arr = np.minimum(out_arr, 1000.0)
    s = float(out_arr.sum())
    if s < 0.01:
        x = np.linspace(0, 1, 6700)
        out_arr = np.sin(np.pi * x) ** 2
        s = float(out_arr.sum())
    out_arr = out_arr / s

    final_score = _objective(out_arr)
    # Cross-check with official evaluator
    try:
        official = evaluate_sequence([float(v) for v in out_arr])
        if np.isfinite(official):
            final_score = min(final_score, official)
    except Exception:
        pass

    elapsed = time.monotonic() - t_global
    print("\nFINAL: score=%.10f n=%d time=%.1fs" % (final_score, out_arr.size, elapsed))

    return [float(v) for v in out_arr]
