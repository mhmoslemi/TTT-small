Hmm, but this is very similar to the previous `spectral_flatten`. Let me think of something genuinely different.

**Key new idea: Active-set penalty method**

Instead of using a soft-max over top lags (which is what the previous code does), I'll use an explicit penalty on the maximum:

`penalty(a, tau) = sum_k max(0, conv[k] - tau)^2`

where `tau` starts high and gets tightened. The gradient of this penalty is:
`d/da[j] penalty = 2 * sum_{k: conv[k] > tau} (conv[k] - tau) * grad_conv[k][j]`

This focuses all the "gradient" on the active set of lags that exceed the threshold.

**Another new idea: Frequency domain random restarts**

Instead of perturbations in the time domain, I'll do perturbations in the frequency domain:
- Add random phase noise to the spectrum
- Or modulate the amplitude at random frequencies

This can escape local minima in a different way than time-domain perturbations.

**Another idea: Multi-resolution approach**

1. Solve at a coarser resolution (e.g., n/4) to get a good shape
2. Upsample and refine at n/2
3. Upsample and refine at n

This is like "coarse-to-fine" optimization.

Let me write a complete, well-structured solution:

def make_gaussian(n, center=0.5, width=0.3):
    x = np.linspace(0, 1, n)
    return norm1(np.exp(-0.5 * ((x - center) / width) ** 2))

def make_bimodal(n, c1=0.25, c2=0.75, width=0.15):
    x = np.linspace(0, 1, n)
    g1 = np.exp(-0.5 * ((x - c1) / width) ** 2)
    g2 = np.exp(-0.5 * ((x - c2) / width) ** 2)

def make_power_law(n, p=1.0, reverse=False):
    i = np.arange(n, dtype=np.float64) + 1.0
    if reverse:
        i = n + 1.0 - i
    return norm1(i ** p)

def make_sine(n, freq=1, phase=0):
    x = np.linspace(0, 1, n)
    return norm1(np.sin(np.pi * freq * x + phase) ** 2)

def make_difference_set(n, rng):
    """Create a sequence inspired by difference sets.
    Place 'mass' at positions that form a quasi-difference-set pattern."""
    # Use a structured pattern: place peaks at intervals related to sqrt(n)
    n_peaks = max(3, int(np.sqrt(n) * 1.5))
    # Generate quasi-random peak positions (Fibonacci-like spacing)
    golden = 0.618033988749895
    positions = []
    for k in range(n_peaks):
        pos = int((k * golden) % 1.0 * n)
        positions.append(pos)
    
    a = np.zeros(n)
    for pos in positions:
        width = max(1, n // n_peaks // 3)
        for d in range(-width, width + 1):
            idx = (pos + d) % n
            a[idx] += np.exp(-0.5 * (d / max(1, width//2)) ** 2)
    return norm1(a)

def make_random_smooth(n, rng, smoothness=0.1):

# --- Optimization: Active-set penalty gradient ---

def active_set_gradient(conv, a, tau, top_k=50):
    """
    Compute gradient of the active-set penalty:
    penalty = sum_k max(0, conv[k] - tau)^2
    Returns gradient vector.
    """
    n = len(a)
    conv_size = len(conv)
    active = conv > tau
    n_active = active.sum()
    
    if n_active == 0:
        # Use top-k as fallback
        if top_k < conv_size:
            idx = np.argpartition(conv, conv_size - top_k)[conv_size - top_k:]
            active = np.zeros(conv_size, dtype=bool)
            active[idx] = True
        else:
            active = np.ones(conv_size, dtype=bool)
    
    grad = np.zeros(n)
    # Only process top 30 active lags for speed
    active_idx = np.where(active)[0]
    if len(active_idx) > 30:
        active_idx = active_idx[np.argsort(conv[active_idx])[::-1][:30]]
    
    for k in active_idx:
        k = int(k)
        weight = conv[k] - tau
        # d(conv[k])/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
        # First term: j ranges over [max(0, k-n+1), min(n-1, k)]
        j1_lo = max(0, k - n + 1)
        j1_hi = min(n, k + 1)
        if j1_lo < j1_hi:
            j1 = np.arange(j1_lo, j1_hi)
            grad[j1] += weight * a[k - j1]
        
        # Second term: j ranges over [max(0, k), min(n-1, k+n-1)]
        j2_lo = max(0, k)
        j2_hi = min(n, k + n)
        if j2_lo < j2_hi:
            j2 = np.arange(j2_lo, j2_hi)
            # But we need to avoid double-counting when k-j == j-k, i.e., j = k/2
            # Actually the two terms are: a[k-j] for the "first a" and a[j-k] for the "second a"
            # They are the same when k-j = j-k, i.e., j = k/2
            # But they're at different j values generally, so just add
            grad[j2] += weight * a[j2 - k]
    
    return 2.0 * grad  # Factor of 2 from the squared penalty

def active_set_descent(a0, time_limit, tau_init=None, verbose=True):
    """
    Proximal gradient descent on the active-set penalty.
    Adaptive threshold that tightens over time.
    """
    n = len(a0)
    a = norm1(a0.copy())
    conv = fast_conv_square(a)
    M = conv.max()
    mean_conv = conv.mean()
    
    if tau_init is None:
        tau = M * 0.8
    
    best_a = a.copy()
    best_obj = compute_obj(n, conv)
    t0 = time.monotonic()
    
    lr = 0.05
    momentum = 0.8
    v = np.zeros(n)  # momentum vector
    
    stall = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Adaptive threshold: tighten slowly
        tau_target = best_obj / (2.0 * n) * 1.05  # Just above current best
        tau = tau * 0.999 + tau_target * 0.001  # Slow adaptation
        tau = max(tau, mean_conv * 0.5)  # Don't go below half the mean
        tau = min(tau, M * 1.1)  # Don't go above current max
        
        # Compute gradient
        grad = active_set_gradient(conv, a, tau)
        
        # Nesterov momentum
        v = momentum * v + grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Evaluate
        conv_new = fast_conv_square(a_new)
        obj_new = compute_obj(n, conv_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            M = conv_new.max()
            stall = 0
            lr = min(lr * 1.01, 0.3)  # Slowly increase step
        else:
            stall += 1
            a = a  # Don't update
            if stall % 50 == 0:
                lr *= 0.9
                tau *= 1.05  # Relax threshold
            if stall > 5000:
                break
        
        if it % 10000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      AS t={it} ({elapsed:.0f}s): obj={best_obj:.6f}, tau={tau:.8f}, M={M:.8f}")
    
    if verbose:
        print(f"    Active-set done ({time.monotonic()-t0:.0f}s): obj={best_obj:.6f}")
    return best_a, best_obj

def spectral_annealing(a0, time_limit, T_init=0.1, T_final=0.001, verbose=True):
    """
    Simulated annealing in the frequency domain.
    Moves: add weighted random noise to the amplitude spectrum.
    """
    n = len(a0)
    a = norm1(a0.copy())
    best_a = a.copy()
    best_obj = compute_obj(n, fast_conv_square(a))
    t0 = time.monotonic()
    
    fft_size = 1 << (2*n - 2).bit_length()
    n_freqs = fft_size // 2 + 1
    
    T = T_init
    cooling = (T_final / T_init) ** (1.0 / max(1, int(time_limit * 100)))
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Generate spectral perturbation
        # Add random amplitude modulation
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Random amplitude perturbation (Gaussian in log domain)
        log_mag = np.log(np.maximum(mag, 1e-30))
        noise = np.random.randn(n_freqs) * T
        log_mag_new = log_mag + noise
        noise[0] = 0  # Keep DC component
        
        mag_new = np.exp(log_mag_new)
        A_new = mag_new * np.exp(1j * phase)
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            T *= cooling
            continue
        a_new = a_new / s
        
        obj_new = compute_obj(n, fast_conv_square(a_new))
        
        # Accept/reject
        if obj_new < best_obj or np.random.random() < np.exp(-(obj_new - best_obj) / max(T, 1e-10)):
            a = a_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
        
        T = max(T * cooling, T_final)
    
    if verbose:
        print(f"    Spectral annealing done ({time.monotonic()-t0:.0f}s): obj={best_obj:.6f}")
    return best_a, best_obj

def mirror_descent_v2(a0, time_limit, verbose=True):
    """
    Improved mirror descent with:
    - Adaptive soft-max temperature
    - Geometry-aware step size
    - Periodic re-normalization
    """
    n = len(a0)

    beta = 5.0  # Soft-max temperature (higher = more focused)
    top_k = 50
    lr = 0.05
    momentum = 0.7
    v_log = np.zeros(n)  # Momentum in log-space
    log_a = np.log(np.maximum(a, 1e-30))
    
    stall = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        a = np.exp(log_a - log_a.max())
        a = norm1(a)
        
        conv = fast_conv_square(a)
        
        # Soft-max weights over top lags
        k_max = top_k
        top_idx = np.argpartition(conv, -k_max)[-k_max:] if k_max < len(conv) else np.arange(len(conv))
        top_vals = conv[top_idx]
        
        max_v = top_vals.max()
        expo = beta * (top_vals - max_v)
        expo = np.clip(expo, -30, 30)
        w = np.exp(expo)
        w = w / w.sum()
        
        # Gradient in log-space: d(log(a_i)) = -grad_i / a_i
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k = int(k)
            j1_lo = max(0, k - n + 1)
            j1_hi = min(n, k + 1)
            if j1_lo < j1_hi:
                j1 = np.arange(j1_lo, j1_hi)
                grad[j1] += wi * a[k - j1]
            j2_lo = max(0, k)
            j2_hi = min(n, k + n)
            if j2_lo < j2_hi:
                j2 = np.arange(j2_lo, j2_hi)
                grad[j2] += wi * a[j2 - k]
        
        # Log-space gradient: dL/d(log_a_i) = a_i * dL/da_i = a_i * grad_i
        log_grad = a * grad
        
        # Nesterov in log-space
        v_log = momentum * v_log + log_grad
        lr_eff = lr / (1.0 + 0.01 * np.sqrt(it))
        log_a -= lr_eff * v_log
        log_a -= log_a.max()  # Numerical stability
        
        # Project
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(n, fast_conv_square(a_new))
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall % 200 == 0:
                lr *= 0.9
                beta = max(beta * 0.95, 1.0)
                top_k = min(top_k + 5, 200)
            if stall > 3000:
                # Restart with perturbation
                rng = np.random.default_rng(it)
                a_new = norm1(a_new * (1 + rng.normal(0, 0.1, n)))
                log_a = np.log(np.maximum(a_new, 1e-30))
                v_log = np.zeros(n)
                stall = 0
                lr = 0.05
                beta = 5.0
                top_k = 50
        
        if it % 10000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      MD2 t={it} ({elapsed:.0f}s): obj={best_obj:.6f}, beta={beta:.1f}, top_k={top_k}")
    
    if verbose:
        print(f"    Mirror descent v2 done ({time.monotonic()-t0:.0f}s): obj={best_obj:.6f}")

# --- Interpolation ---

def interpolate_seq(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = norm1(f)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    return norm1(f_int)

3. Improved mirror descent with soft-max gradients in log-space
4. Iterative spectral shaping via amplitude compression
5. Spectral-domain simulated annealing
6. Diverse structural initialization (difference sets, Gaussians, etc.)
7. Multi-phase orchestration with length exploration
"""

import numpy as np
import time


def fast_conv_square(a):
    """Compute np.convolve(a, a) via zero-padded FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_obj(n, conv):
    """Objective value given n and convolution (assumes sum(a)=1)."""
    return 2.0 * n * conv.max()


def eval_seq(a):
    """Safely call the external evaluation function."""
    try:
        a_list = [float(max(0.0, x)) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def gradient_from_weights(a, weights):
    """
    Compute sum_k weights[k] * d(conv[k])/d(a[j]) for all j.
    Since d(conv[k])/d(a[j]) = 2*a[k-j] (if 0<=k-j<n, else 0),
    we need: 2 * sum_k weights[k] * a[k-j] for each j.
    This equals 2 * cross_corr(weights, a)[j].
    Using FFT: cross_corr(w, a)[j] = conv(w, a_rev)[j + n - 1]
    where a_rev = a[::-1].
    """
    n = len(a)
    fw = len(weights)
    a_rev = a[::-1]
    total_len = fw + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    conv_result = np.fft.irfft(W * AR, fft_size)[:total_len]
    grad = conv_result[n - 1:n - 1 + n]
    return 2.0 * grad


def make_gaussian(n, center=0.5, width=0.3):
    x = np.linspace(0, 1, n)
    return norm1(np.exp(-0.5 * ((x - center) / width) ** 2))


def make_bimodal(n, c1=0.25, c2=0.75, width=0.15):
    x = np.linspace(0, 1, n)
    return norm1(np.exp(-0.5 * ((x - c1) / width) ** 2) + np.exp(-0.5 * ((x - c2) / width) ** 2))


def make_power_law(n, p=1.0, reverse=False):
    i = np.arange(n, dtype=np.float64) + 1.0
    if reverse:
        i = n + 1.0 - i
    return norm1(i ** p)


def make_sine(n, freq=1, phase=0.0):
    x = np.linspace(0, 1, n)
    return norm1(np.sin(np.pi * freq * x + phase) ** 2)


def make_golden_pattern(n):
    """Sequence with peaks at golden-ratio-spaced positions."""
    golden = 0.618033988749895
    n_peaks = max(3, int(np.sqrt(n) * 2))
    a = np.zeros(n)
    width = max(1, n // (n_peaks * 4))
    for k in range(n_peaks):
        pos = int((k * golden) % 1.0 * n)
        for d in range(max(0, pos - width), min(n, pos + width + 1)):
            a[d] += np.exp(-0.5 * (d - pos) ** 2 / max(1, (width // 2) ** 2))
    return norm1(a)


def make_random_smooth(n, rng, n_ctrl=20):
    xs = np.sort(rng.uniform(0, 1, n_ctrl))
    ys = rng.uniform(0.2, 1.0, n_ctrl)
    xs_ext = np.concatenate([[0.0], xs, [1.0]])
    ys_ext = np.concatenate([[ys[0]], ys, [ys[0]]])
    x = np.linspace(0, 1, n)
    a = np.interp(x, xs_ext, ys_ext)
    return norm1(a)


def diverse_init(n, budget=10.0):
    """Try diverse initializations and return the best one."""
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    candidates = [
        np.ones(n) / n,
        make_gaussian(n, 0.5, 0.2),
        make_gaussian(n, 0.5, 0.3),
        make_gaussian(n, 0.5, 0.4),
        make_gaussian(n, 0.5, 0.5),
        make_bimodal(n, 0.25, 0.75, 0.1),
        make_bimodal(n, 0.25, 0.75, 0.2),
        make_bimodal(n, 0.2, 0.8, 0.15),
        make_bimodal(n, 0.3, 0.7, 0.1),
        make_power_law(n, 0.3),
        make_power_law(n, 0.5),
        make_power_law(n, 1.0),
        make_power_law(n, 1.5),
        make_power_law(n, 0.5, reverse=True),
        make_sine(n, 1, 0),
        make_sine(n, 2, 0),
        make_sine(n, 3, np.pi / 4),
        make_golden_pattern(n),
    ]
    for _ in range(4):
        candidates.append(make_random_smooth(n, rng, n_ctrl=max(10, n // 200)))

    best = None
    best_val = float('inf')
    for c in candidates:
        if time.monotonic() - t0 > budget:
            break
        v = compute_obj(n, fast_conv_square(c))
        if v < best_val:
            best_val = v
            best = c.copy()
    if best is None:
        best = np.ones(n) / n
        best_val = compute_obj(n, fast_conv_square(best))
    return best, best_val


def spectral_shaping(a0, time_limit, verbose=True):
    """
    Iterative spectral shaping: compress the amplitude spectrum to flatten
    the autocorrelation. Uses adaptive exponent and blending.
    """
    n = len(a0)
    a = norm1(a0.copy())
    conv = fast_conv_square(a)
    best_a = a.copy()
    best_obj = compute_obj(n, conv)
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 2).bit_length()
    alpha = 0.6
    blend = 0.4
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        max_mag = mag.max()
        if max_mag < 1e-30:
            break
        mag_norm = mag / max_mag
        new_mag = np.power(mag_norm, alpha) * max_mag
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a_cand = (1 - blend) * a + blend * a_new
        a_cand = norm1(a_cand)
        conv_c = fast_conv_square(a_cand)
        obj_c = compute_obj(n, conv_c)
        if obj_c < best_obj - 1e-15:
            best_obj = obj_c
            best_a = a_cand.copy()
            a = a_cand
            conv = conv_c
            stall = 0
            alpha = min(alpha + 0.005, 0.92)
            blend = min(blend + 0.002, 0.7)
        else:
            stall += 1
            if stall % 200 == 0:
                blend = max(blend * 0.9, 0.05)
                alpha = max(alpha - 0.01, 0.3)
            if stall > 5000:
                break
        if it % 20000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      Spec t={it} ({elapsed:.0f}s): obj={best_obj:.6f}, alpha={alpha:.3f}")

    if verbose:
        print(f"    Spectral shaping done ({time.monotonic() - t0:.0f}s): obj={best_obj:.6f}")
    return best_a, best_obj


def active_set_dd(a0, time_limit, verbose=True):
    """
    Proximal gradient on active-set penalty: P = sum_k max(0, conv[k]-tau)^2
    with adaptive threshold tau that tightens toward the current best.
    """
    n = len(a0)
    a = norm1(a0.copy())
    conv = fast_conv_square(a)
    best_a = a.copy()
    best_obj = compute_obj(n, conv)
    t0 = time.monotonic()

    M = conv.max()
    conv_mean = conv.mean()
    tau = M * 0.9
    lr = 0.02
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Adapt threshold: tighten toward the current max
        tau_target = best_obj / (2.0 * n) * 1.01
        tau = 0.998 * tau + 0.002 * tau_target
        tau = max(tau, conv_mean * 0.3)
        tau = min(tau, M * 1.05)

        # Compute gradient via FFT cross-correlation
        w = np.maximum(conv - tau, 0.0)
        if w.sum() < 1e-30:
            top_idx = np.argpartition(conv, -30)[-30:] if n > 30 else np.arange(len(conv))
            w = np.zeros_like(conv)
            w[top_idx] = conv[top_idx] - conv[top_idx].min() + 1e-10
        grad = gradient_from_weights(a, w)

        # Nesterov step
        v = momentum * v + grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            lr *= 0.5
            continue
        a_new = a_new / s
        conv_new = fast_conv_square(a_new)
        M_new = conv_new.max()
        obj_new = compute_obj(n, conv_new)

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            M = M_new
            conv_mean = conv.newことです.mean() if hasattr(conv, 'preserved') else conv_new.mean()
            conv_mean = conv_new.mean()
            stall = 0
            lr = min(lr * 1.005, 0.15)
        else:
            stall += 1
            if stall % 100 == 0:
                lr *= 0.92
                tau = min(tau * 1.03, M * 1.1)
            if stall > 4000:
                np.random.seed(it)
                a = norm1(a * (1.0 + np.random.randn(n) * 0.05))
                v = np.zeros(n)
                lr = 0.02
                momentum = 0.7
                stall = 0
                tau = fast_conv_square(a).max() * 0.9

        if it % 20000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      AS t={it} ({elapsed:.0f}s): obj={best_obj:.6f}, tau={tau:.1e}")

    if verbose:
        print(f"    Active-set DD done ({time.monotonic() - t0:.0f}s): obj={best_obj:.6f}")
    return best_a, best_obj


def mirror_descent_v2(a0, time_limit, verbose=True):
    """
    Mirror descent in log-space with soft-max gradient over top autocorrelation lags.
    Uses Nesterov momentum and adaptive beta (temperature).
    """
    n = len(a0)
    a = norm1(a0.copy())
    conv = fast_conv_square(a)
    best_a = a.copy()
    best_obj = compute_obj(n, conv)
    t0 = time.monotonic()

    beta = 4.0
    top_k = 40
    lr = 0.04
    momentum = 0.75
    log_a = np.log(np.maximum(a, 1e-30))
    v_log = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        a = np.exp(log_a - log_a.max())
        a = norm1(a)
        conv = fast_conv_square(a)

        # Soft-max weights
        if top_k < len(conv):
            top_idx = np.argpartition(conv, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(conv))
        top_vals = conv[top_idx]
        max_v = top_vals.max()
        expo = beta * (top_vals - max_v)
        expo = np.clip(expo, -40, 40)
        w = np.exp(expo)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(w) / len(w)
        else:
            w = w / w_sum

        # Build full weight vector for FFT gradient computation
        full_w = np.zeros(len(conv))
        for wi, ki in zip(w, top_idx):
            full_w[int(ki)] = wi

        # Gradient: sum_k full_w[k] * d(conv[k])/d(a) = gradient_from_weights(a, full_w)
        grad_full = gradient_from_weights(a, full_w)

        # Log-space gradient: dL/d(log_a_j) = a_j * dL/da_j
        log_grad = a * grad_full

        # Nesterov in log-space
        v_log = momentum * v_log + log_grad
        lr_eff = lr / (1.0 + 0.005 * np.sqrt(it))
        log_a = log_a - lr_eff * v_log
        log_a = log_a - log_a.max()

        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        conv_new = fast_conv_square(a_new)
        obj_new = compute_obj(n, conv_new)

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall % 300 == 0:
                lr *= 0.93
                beta = max(beta * 0.97, 1.0)
                top_k = min(top_k + 5, 150)
            if stall > 3000:
                rng = np.random.default_rng(it)
                pert_type = it % 3
                if pert_type == 0:
                    a_new = norm1(a_new + rng.normal(0, 0.03, n) * a_new)
                elif pert_type == 1:
                    x = np.linspace(0, 1, n)
                    c = rng.uniform(0.2, 0.8)
                    sig = rng.uniform(0.05, 0.2)
                    pert = np.exp(-0.5 * ((x - c) / sig) ** 2)
                    a_new = norm1(0.8 * a_new + 0.2 * pert)
                else:
                    a_new = norm1(a_new * np.power(rng.uniform(0.5, 1.5, n), 0.3))
                log_a = np.log(np.maximum(a_new, 1e-30))
                v_log = np.zeros(n)
                stall = 0
                lr = 0.04
                beta = 4.0
                top_k = 40

        if it % 20000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      MD2 t={it} ({elapsed:.0f}s): obj={best_obj:.6f}, beta={beta:.1f}, top_k={top_k}")

    if verbose:
        print(f"    Mirror descent v2 done ({time.monotonic() - t0:.0f}s): obj={best_obj:.6f}")
    return best_a, best_obj


def spectral_annealing(a0, time_limit, T_init=0.03, T_final=0.0005, verbose=True):
    """
    Simulated annealing with moves in the frequency domain.
    Each move: add Gaussian noise to log-amplitude spectrum at random frequencies.
    """
    n = len(a0)
    a = norm1(a0.copy())
    conv = fast_conv_square(a)
    best_a = a.copy()
    best_obj = compute_obj(n, conv)
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 2).bit_length()
    n_freqs = fft_size // 2 + 1
    rng = np.random.default_rng(777)

    # Estimate total iterations
    est_iters = int(time_limit * 200)
    cooling = (T_final / T_init) ** (1.0 / max(1, est_iters))
    T = T_init
    it = 0
    accepted = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        log_mag = np.log(np.maximum(mag, 1e-30))

        # Perturb random subset of frequencies
        n_perturb = max(3, n_freqs // 20)
        freqs = rng.integers(1, n_freqs, size=n_perturb)
        log_mag[freqs] += rng.normal(0, T, size=n_perturb)

        mag_new = np.exp(log_mag)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            T = max(T * cooling, T_final)
            continue
        a_new = a_new / s
        conv_new = fast_conv_square(a_new)
        obj_new = compute_obj(n, conv_new)

        delta = obj_new - best_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            a = a_new
            accepted += 1
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

        T = max(T * cooling, T_final)

        if it % 50000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      Anneal t={it} ({elapsed:.0f}s): obj={best_obj:.6f}, T={T:.4e}, acc={accepted/it:.2f}")

    if verbose:
        print(f"    Spectral annealing done ({time.monotonic() - t0:.0f}s): obj={best_obj:.6f}")
    return best_a, best_obj


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = norm1(f)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    return norm1(f_int)


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator. Multi-phase optimization:
    1. Diverse initialization (possibly from parent)
    2. Spectral shaping (coarse)
    3. Active-set penalty descent
    4. Mirror descent v2
    5. Spectral annealing
    6. Length exploration
    7. Final polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Load parent if available
    n_try = 5500
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm1(p)
            n_try = len(parent)
            sc = eval_seq(parent)
            if np.isfinite(sc):
                best_seq = parent
                best_score = sc
                parent_loaded = True
                print(f"Parent loaded: n={n_try}, score={sc:.6f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq, best_obj_init = diverse_init(n_try, budget=8.0)
        best_score = eval_seq(best_seq)
        print(f"Fresh init: n={n_try}, score={best_score:.6f} (internal obj={best_obj_init:.6f})")

    n0 = len(best_seq)

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = eval_seq(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")

    # Phase 1: Spectral shaping
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Spectral shaping (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.12, 180.0)
    if t1 > 15:
        a_out, _ = spectral_shaping(best_seq, t1, verbose=True)
        check_update(a_out, "spec_coarse")

    # Phase 2: Active-set penalty descent
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Active-set DD (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 20:
        a_out, _ = active_set_dd(best_seq, t2, verbose=True)
        check_update(a_out, "active_set")

    # Phase 3: Mirror descent v2
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Mirror descent v2 (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 20:
        a_out, _ = mirror_descent_v2(best_seq, t3, verbose=True)
        check_update(a_out, "md_v2")

    # Phase 4: Spectral annealing
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Spectral annealing (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 15:
        a_out, _ = spectral_annealing(best_seq, t4, T_init=0.02, T_final=0.0003, verbose=True)
        check_update(a_out, "anneal")

    # Phase 5: Length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Length exploration (current n={n_cur}, budget={remaining:.0f}s) ===")

    length_options = []
    for delta in [-300, -500, 300, 500, 800, -800, 1000, -1000]:
        nn = n_cur + delta
        if 2000 <= nn <= 12000 and nn != n_cur:
            length_options.append(nn)
    for rn in [3000, 4000, 5000, 6000, 7000, 8000]:
        if rn != n_cur and 2000 <= rn <= 12000:
            length_options.append(rn)
    length_options = sorted(set(length_options))

    for new_n in length_options[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_interp = interpolate_seq(best_seq, new_n)

        t5 = min(remaining * 0.35, 150.0)
        if t5 > 20:
            a_out, _ = mirror_descent_v2(a_interp, t5, verbose=False)
            check_update(a_out, f"md_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t5b = min(remaining * 0.3, 100.0)
            if t5b > 15:
                a_out, _ = active_set_dd(a_interp, t5b, verbose=False)
                check_update(a_out, f"as_n{new_n}")

    # Phase 6: Final polish (cycle through optimizers)
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_final}, budget={remaining:.0f}s) ===")

    # Alternate optimizers for final squeeze
    polish_rounds = [
        ("md_polish", mirror_descent_v2),
        ("as_polish", active_set_dd),
        ("spec_polish", spectral_shaping),
    ]

    for label, func in polish_rounds:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 20:
            break
        t6 = min(remaining * 0.35, 150.0)
        if t6 > 10:
            a_out, _ = func(best_seq, t6, verbose=False)
            check_update(a_out, label)

    # Final validation
    if best_seq is not None:
        fs = eval_seq(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = np.ones(5000) / 5000
        best_score = eval_seq(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")

    return [float(x) for x in best_seq]
