```
def interpolate_sequence(f, new_n):
   f=np.asarray(f,float); s=f.sum(); if s<1e-15: ones
   f=f/s
   x_old=np.linspace(0.0,1.0,len(f))
   x_new=np.linspace(0.0,1.0,new_n)
   new=np.interp(x_new,x_old,f)
   new=np.maximum(new,0)
   sn=new.sum()
   if sn<1e-15: ones
   return new/sn
```

```python
"""
Search for a non-negative step-function coefficient sequence minimizing
    2 * n * max(np.convolve(a, a)) / sum(a)**2
by exploiting scale invariant normalization, FFT-based convolution, and a
combination of:
  * smoothed-exponentiated-gradient descent on a top-peak log-sum-exp model,
  * a multiplicative-weights saddle-point direction for the linearized
    Frank-Wolfe subproblem min_{g in simplex} max_k (a*g)_k,
  * exact local coordinate “mass transfer” polishing with O(n) trial updates,
  * multi-length homotopy / interpolation exploration.

The implementation is self-contained and uses only numpy/time plus the
globally available evaluate_sequence.
"""

import numpy as np
import time


def _next_pow2(size):
    n = 1
    while n < size:
        n <<= 1
    return n


def _safe_norm(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e-6, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = _next_pow2(size)
    X = np.fft.rfft(a, N)
    out = np.fft.irfft(X * X, N)[:size]
    return out


def _cross_conv(x, y):
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    size = x.size + y.size - 1
    N = _next_pow2(size)
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    return np.fft.irfft(X * Y, N)[:size]


def _cross_conv_with_fft(X, N, y, size):
    Y = np.fft.rfft(y, N)
    return np.fft.irfft(X * Y, N)[:size]


def _gradient_from_weights(a, p):
    """
    Gradient of sum_k p_k * (a*a)_k, where (a*a) is linear convolution.
   """
    n = a.size
    L = 2 * n - 1
    ar = np.ascontiguousarray(a[::-1])
    size = L + n - 1
    N = _next_pow2(size)
    P = np.fft.rfft(p, N)
    AR = np.fft.rfft(ar, N)
    conv = np.fft.irfft(P * AR, N)[n - 1:2 * n - 1]
    grad = 2.0 * n * conv
    grad = np.maximum(grad, 0.0)
    grad[~np.isfinite(grad)] = 0.0
    return grad


def _mirror_trial(a, grad, eta):
    loga = np.log(np.maximum(a, 1e-12)) - eta * grad
    if not np.all(np.isfinite(loga)):
        ok = np.isfinite(loga)
        if not np.any(ok):
            return None
        loga = np.where(ok, loga, loga[ok].min())
    loga -= float(loga.max())
    trial = np.exp(loga)
    s = float(trial.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None
    return trial / s


def _eval_local(a):
    if a is None:
        return float("inf"), None
    a = _safe_norm(a)
    n = a.size
    if n == 0:
        return float("inf"), a
    c = _self_conv(a)
    m = float(c.max())
    if not np.isfinite(m):
        return float("inf"), a
    return 2.0 * n * m, a


def _safe_eval(a):
    try:
        arr = np.asarray(a, dtype=np.float64)
        arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
        arr = np.maximum(arr, 0.0)
        seq = [float(x) for x in arr]
        val = evaluate_sequence(seq)
        if np.isfinite(val):
            return float(val)
        return float("inf")
    except Exception:
        return float("inf")


def _eg_optimizer(a0, time_limit, beta_start=5.0, beta_end=50.0,
                  top_k_ratio=0.1, max_top_k=3000, eta_init=0.05, label="eg"):
    a = _safe_norm(a0)
    n = a.size
    c = _self_conv(a)
    cur_m = n * float(c.max())
    best_m = cur_m
    best_a = a.copy()

    if time_limit <= 0.0:
        return best_a, 2.0 * best_m

    t0 = time.monotonic()
    L = 2 * n - 1
    top_k = int(max(1, min(max_top_k, max(10, L * top_k_ratio))))
    if top_k >= L:
        top_k = L

    top_idx = None
    eta = float(eta_init)
    beta = float(beta_start)
    stall = 0
    it = 0
    rng = np.random.default_rng(777)

    while True:
        if time.monotonic() - t0 >= time_limit:
            break
        it += 1

        if top_idx is None or it % 10 == 0:
            if top_k >= L:
                top_idx = np.arange(L)
            else:
                top_idx = np.argpartition(c, -top_k)[-top_k:]

        vals = c[top_idx]
        maxv = float(vals.max())
        shift = beta * (vals - maxv)
        w = np.exp(np.clip(shift, -30.0, 0.0))
        sw = float(w.sum())
        p = np.zeros(L, dtype=np.float64)
        if sw > 1e-12:
            p[top_idx] = w / sw
        else:
            p[top_idx] = 1.0 / float(len(top_idx))

        grad = _gradient_from_weights(a, p)
        gmax = float(np.max(grad))
        if not np.isfinite(gmax) or gmax <= 1e-300:
            break

        old_m = cur_m
        accepted = False
        scale = max(1.0, gmax / 2.0)

        for multiplier in (1.0, 0.5, 2.0, 0.25, 4.0):
            eta_try = max(1e-5, min(0.5, eta * multiplier / scale))
            trial = _mirror_trial(a, grad, eta_try)
            if trial is None:
                continue
            c2 = _self_conv(trial)
            m2 = n * float(c2.max())
            if m2 < old_m - 1e-14:
                a = trial
                c = c2
                cur_m = m2
                eta = eta_try
                accepted = True
                break

        beta = min(float(beta_end), beta * 1.01)

        if accepted:
            if cur_m < best_m - 1e-15:
                best_m = cur_m
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            if stall > 250:
                noise = 0.01 + 0.03 * min(1.0, stall / 1500.0)
                pert = best_a * (1.0 + noise * rng.normal(n)) + 1e-9
                pert = np.maximum(pert, 0.0)
                ps = float(pert.sum())
                if ps > 1e-12:
                    a = pert / ps
                else:
                    a = best_a.copy()
                c = _self_conv(a)
                cur_m = n * float(c.max())
                eta = max(0.002, min(0.2, eta * 0.7))
                beta = max(float(beta_start), beta * 0.6)
                stall = 0
                top_idx = None

        if it % 800 == 0:
            print("  [" + label + "] iter=" + str(it) +
                  " score=" + repr(2.0 * cur_m) +
                  " best=" + repr(2.0 * best_m))

    if cur_m < best_m:
        best_a = a.copy()
        best_m = cur_m
    return best_a, 2.0 * best_m


def _minimax_direction(a, time_limit, R_max=2000, eta_q=0.8, eta_lam=1.0,
                       beta_init=8.0):
    a = _safe_norm(a)
    n = a.size
    L = 2 * n - 1
    if time_limit <= 0.0:
        return a.copy()

    t0 = time.monotonic()
    N = _next_pow2(L)
    X = np.fft.rfft(a, N)

    c = _self_conv(a)
    c_s = n * c
    max_c = float(c_s.max())
    base_lam = np.exp(np.clip(beta_init * (c_s - max_c), -30.0, 0.0))
    bs = float(base_lam.sum())
    if bs <= 1e-12:
        base_lam = np.ones(L, dtype=np.float64) / L
    else:
        base_lam = base_lam / bs

    lam = base_lam.copy()
    q = a.copy()
    logq = np.log(np.maximum(q, 1e-12))
    q_avg = np.zeros(n, dtype=np.float64)
    count = 0

    while True:
        if time.monotonic() - t0 >= time_limit:
            break
        if count >= R_max:
            break
        count += 1

        cum = np.empty(L + 1, dtype=np.float64)
        cum[0] = 0.0
        np.cumsum(lam, out=cum[1:])
        Lcoef = cum[n:2 * n] - cum[:n]
        h = n * a * Lcoef
        h = np.clip(h, 0.0, 8.0)

        logq -= eta_q * h
        logq -= float(logq.max())
        q = np.exp(logq)
        qs = float(q.sum())
        if not np.isfinite(qs) or qs <= 1e-12:
            break
        q = q / qs
        q_avg += q

        F = _cross_conv_with_fft(X, N, q, L)
        F_s = n * F
        maxF = float(F_s.max())
        shift = eta_lam * (F_s - maxF)
        lam = lam * np.exp(np.clip(shift, -30.0, 0.0))

        lam = 0.8 * lam + 0.2 * base_lam
        s = float(lam.sum())
        if not np.isfinite(s) or s <= 1e-100:
            lam = base_lam.copy()
        else:
            lam = lam / s

    if count == 0:
        return a.copy()
    q_avg /= count
    return _safe_norm(q_avg)


def _line_search_dir(a, d, current_m):
    a = _safe_norm(a)
    d = _safe_norm(d)
    n = a.size
    best = a.copy()
    best_m = current_m
    alphas = (0.05, 0.1, 0.2, 0.35, 0.5, 0.75, 1.0)
    for alpha in alphas:
        if alpha >= 1.0:
            trial = d.copy()
        else:
            trial = (1.0 - alpha) * a + alpha * d
        trial = _safe_norm(trial)
        c = _self_conv(trial)
        m = n * float(c.max())
        if m < best_m - 1e-14:
            best = trial
            best_m = m
    return best, best_m


def _fw_minimax_polish(a0, time_limit, inner_time=6.0, inner_R=1500, label="fw"):
    a = _safe_norm(a0)
    n = a.size
    c = _self_conv(a)
    cur_m = n * float(c.max())
    best_m = cur_m
    best_a = a.copy()

    if time_limit <= 0.0:
        return best_a, 2.0 * best_m

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while True:
        rem = time_limit - (time.monotonic() - t0)
        if rem <= 1.0:
            break
        if no_improve >= 4:
            break
        it += 1

        tin = min(float(inner_time), max(0.75, rem * 0.25))
        q = _minimax_direction(a, tin, R_max=inner_R, eta_q=0.8, eta_lam=1.0, beta_init=8.0)
        trial, m = _line_search_dir(a, q, cur_m)

        if m < cur_m - 1e-14:
            a = trial
            cur_m = m
            no_improve = 0
        else:
            no_improve += 1

        if cur_m < best_m - 1e-15:
            best_m = cur_m
            best_a = a.copy()
            no_improve = 0

        if it % 2 == 0:
            print("  [" + label + "] dir-iter=" + str(it) +
                  " score=" + repr(2.0 * cur_m) +
                  " best=" + repr(2.0 * best_m))

    if cur_m < best_m:
        best_a = a.copy()
        best_m = cur_m
    return best_a, 2.0 * best_m


def _coordinate_polish(a0, time_limit, label="coord"):
    a = _safe_norm(a0)
    n = a.size
    c = _self_conv(a)
    cur_m = n * float(c.max())
    best_m = cur_m
    best_a = a.copy()

    if time_limit <= 0.0:
        return best_a, 2.0 * best_m

    t0 = time.monotonic()
    L = 2 * n - 1

    if n <= 7000:
        top_p = 24
        src_q = 12
        dst_q = 12
        fracs = (0.005, 0.01, 0.02, 0.05, 0.10)
    elif n <= 11000:
        top_p = 20
        src_q = 10
        dst_q = 10
        fracs = (0.01, 0.02, 0.05, 0.10)
    else:
        top_p = 16
        src_q = 8
        dst_q = 8
        fracs = (0.02, 0.05, 0.10)

    no_improve = 0
    it = 0
    accepted = 0
    rng = np.random.default_rng(12345)

    while True:
        if time.monotonic() - t0 >= time_limit:
            break
        if no_improve >= 40:
            break

        it += 1

        if accepted % 10 == 0:
            c = _self_conv(a)
            cur_m = n * float(c.max())
            if cur_m < best_m:
                best_m = cur_m
                best_a = a.copy()

        if top_p >= L:
            top_idx = np.arange(L)
        else:
            top_idx = np.argpartition(c, -top_p)[-top_p:]

        vals = c[top_idx]
        maxv = float(vals.max())
        w = np.exp(np.clip(50.0 * (vals - maxv), -30.0, 0.0))
        sw = float(w.sum())
        if sw <= 1e-12:
            w = np.ones_like(w) / w.size

        grad = np.zeros(n, dtype=np.float64)
        for ii in range(len(top_idx)):
            k = int(top_idx[ii])
            wk = float(w[ii])
            if wk <= 0.0:
                continue
            js = k - n + 1
            if js < 0:
                js = 0
            je = k
            if je >= n:
                je = n - 1
            if js <= je:
                idx = np.arange(js, je + 1)
                grad[js:je + 1] += 2.0 * n * wk * a[k - idx]

        if float(np.max(grad)) <= 0.0:
            no_improve += 1
            continue

        if src_q >= n:
            src = np.argsort(grad)[::-1]
        else:
            src = np.argpartition(grad, -src_q)[-src_q:]
            src = src[np.argsort(grad[src])[::-1]]
        if dst_q >= n:
            dst = np.argsort(grad)
        else:
            dst = np.argpartition(grad, dst_q)[:dst_q]
            dst = dst[np.argsort(grad[dst])]

        best_i = -1
        best_j = -1
        best_delta = 0.0
        best_pair_m = cur_m
        t_batch = time.monotonic()

        for i in src:
            i = int(i)
            ai = a[i]
            if ai <= 1e-12:
                continue
            gi = grad[i]
            for j in dst:
                j = int(j)
                if i == j:
                    continue
                if gi <= grad[j]:
                    continue
                for frac in fracs:
                    delta = ai * frac
                    if delta <= 1e-14:
                        continue
                    cT = c.copy()
                    cT[i:i + n] -= 2.0 * delta * a
                    cT[j:j + n] += 2.0 * delta * a
                    cT[2 * i] += delta * delta
                    cT[i + j] -= 2.0 * delta * delta
                    cT[2 * j] += delta * delta
                    m = n * float(cT.max())
                    if m < best_pair_m - 1e-15:
                        best_pair_m = m
                        best_i = i
                        best_j = j
                        best_delta = delta

            if time.monotonic() - t_batch > 1.0:
                break

        if best_i >= 0:
            a[best_i] -= best_delta
            a[best_j] += best_delta
            a = np.maximum(a, 0.0)
            s = float(a.sum())
            if s > 1e-12:
                a = a / s
            c = _self_conv(a)
            cur_m = n * float(c.max())
            accepted += 1
            no_improve = 0
            if cur_m < best_m - 1e-15:
                best_m = cur_m
                best_a = a.copy()
        else:
            no_improve += 1
            if no_improve >= 5:
                noise = 0.005
                pert = best_a * (1.0 + noise * rng.normal(n)) + 1e-9
                pert = np.maximum(pert, 0.0)
                ps = float(pert.sum())
                if ps > 1e-12:
                    a = pert / ps
                else:
                    a = best_a.copy()
                c = _self_conv(a)
                cur_m = n * float(c.max())
                no_improve = 0

        if it % 20 == 0:
            print("  [" + label + "] batch=" + str(it) +
                  " score=" + repr(2.0 * cur_m) +
                  " best=" + repr(2.0 * best_m))

    if cur_m < best_m:
        best_a = a.copy()
        best_m = cur_m
    return best_a, 2.0 * best_m


def _interpolate_sequence(f, new_n):
    f = _safe_norm(f)
    old_n = f.size
    if old_n == new_n:
        return f.copy()
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) / new_n
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-12:
        return np.ones(new_n, dtype=np.float64) / new_n
    return g / s


def _rand_smooth_init(n, seed):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    f = np.sin(np.pi * x) ** 2 + 0.3
    for _ in range(3):
        freq = int(rng.integers(1, 6))
        amp = rng.uniform(0.08, 0.45)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        f += amp * np.cos(2.0 * np.pi * freq * x + phase)
    f += rng.normal(0.0, 0.02, n)
    return _safe_norm(f)


def _spectral_flat_init(n):
    L = 2 * n - 1
    N = _next_pow2(L)
    target = np.zeros(N, dtype=np.float64)
    target[:L] = 1.0
    C = np.fft.fft(target)
    mag = np.sqrt(np.abs(C))
    phase = np.exp(0.5j * np.angle(C))
    A = mag * phase
    full = np.abs(np.fft.ifft(A))
    a = np.maximum(full[:n], 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return _safe_norm(np.ones(n, dtype=np.float64))
    return a / s


def _update_best(best_a, best_score, a, score, label):
    if a is None:
        return best_a, best_score
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0 or not np.isfinite(score):
        return best_a, best_score
    if score < best_score - 1e-12:
        a = _safe_norm(a).copy()
        print("  *** NEW BEST [" + label + "]: score=" + repr(score) +
              " n=" + str(a.size))
        return a, float(score)
    return best_a, best_score
```
