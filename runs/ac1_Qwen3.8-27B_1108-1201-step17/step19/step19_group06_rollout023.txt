def soft_max_objective(x, tau, n):
    """
    x: unconstrained parameters
    a = softmax(x) gives the non-negative sequence
    Returns soft-max of autocorrelation
    """
    a = softmax(x)  # sum to 1, all positive
    ac = fast_autocorr(a)
    max_ac = ac.max()
    expo = tau * (ac - max_ac)  # shift for numerical stability
    return (1/tau) * (max_ac + np.log(np.sum(np.exp(expo))))

def soft_max_grad(x, tau, n):
    """Gradient of soft-max objective w.r.t. x"""
    a = softmax(x)
    ac = fast_autocorr(a)
    max_ac = ac.max()
    expo = tau * (ac - max_ac)
    exp_vals = np.exp(expo)
    w = exp_vals / exp_vals.sum()  # weights for soft-max
    
    # Gradient of soft-max w.r.t. a
    # dF/da[j] = 2 * sum_k w[k] * a[k-j] * I(0<=k-j<n)
    g = np.zeros(n)
    # Only use lags with significant weight
    thresh = w.max() * 1e-6
    significant = w > thresh
    for k in np.where(significant)[0]:
        wk = w[k]
        if wk < 1e-15:
            continue
        # g[j] += 2 * wk * a[k-j] for 0 <= k-j < n
        j_min = max(0, k - n + 1)
        j_max = min(n, k + 1)
        for j in range(j_min, j_max):
            g[j] += 2.0 * wk * a[k - j]
    
    # Chain rule through softmax
    # dF/dx[j] = a[j] * (g[j] - sum_i g[i]*a[i])
    g_dot_a = np.dot(g, a)
    dx = a * (g - g_dot_a)

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_autocorr(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

                    jj = k2i - ii
                    scores[ii] += ac[k2i] * a[jj]
                    scores[jj] += ac[k2i] * a[ii]

                scores[src] = np.inf
                dst = int(np.argmin(scores))

                if dst == src:
                    continue

                # Try transfer
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial /= s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Random perturbation to escape
            if stall % 200 == 0:
                n_moves = int(rng.integers(3, 10))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = float(rng.uniform(0.005, 0.05))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a /= a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        print(f"      Coord descent: score={2*n*best_m:.8f} ({time.monotonic()-t0:.0f}s)")
    return best_a, 2.0 * n * best_m


def simulated_annealing(a_init, time_limit, seed=123, verbose=False):
    """Simple simulated annealing with pairwise moves."""
    n = len(a_init)
    a = norm_sum1(a_init)
    best_a = a.copy()
    best_m = fast_autocorr(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    T_init = best_m * 0.01
    T_final = best_m * 0.0001
    total_iters = 20000

    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        frac = it / total_iters
        T = T_init * (T_final / T_init) ** frac

        # Random move: transfer mass from one position to another
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j or a[i] < 1e-14:
            continue
        frac_move = float(rng.uniform(0.01, 0.1))
        amt = a[i] * frac_move

        a_trial = a.copy()
        a_trial[i] -= amt
        a_trial[j] += amt
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial /= s

        m_trial = fast_autocorr(a_trial).max()
        delta = m_trial - best_m

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-30)):
            a = a_trial
            if m_trial < best_m:
                best_m = m_trial
                best_a = a_trial.copy()

    if verbose:
        print(f"      SimAnneal: score={2*n*best_m:.8f} ({time.monotonic()-t0:.0f}s)")
    return best_a, 2.0 * n * best_m


def parametric_candidates(n, rng):
    """Generate diverse parametric candidates of length n."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []

    # Uniform
    candidates.append(np.ones(n))

    # Power laws
    for p in [0.2, 0.5, 0.8, 1.0, 1.5, 2.0, -0.5, -1.0]:
        base = np.maximum(i_arr + 1.0, 0.1) ** p
        candidates.append(np.maximum(base, 0))
        base2 = np.maximum(n - i_arr, 0.1) ** p
        candidates.append(np.maximum(base2, 0))

    # Gaussians
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sig in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            candidates.append(np.exp(-((x - c) ** 2) / (2 * sig ** 2)))

    # Bimodal
    for sig in [0.05, 0.1, 0.15, 0.2]:
        for c1, c2 in [(0.2, 0.8), (0.25, 0.75), (0.3, 0.7)]:
            candidates.append(
                np.exp(-((x - c1) ** 2) / (2 * sig ** 2)) +
                np.exp(-((x - c2) ** 2) / (2 * sig ** 2)))

    # Sines
    for f in range(1, 5):
        candidates.append(np.sin(np.pi * f * x) ** 2)
        candidates.append(np.cos(np.pi * f * x) ** 2)

    # Exponential
    for r in [0.5, 1, 2, 3, 5]:
        candidates.append(np.exp(-r * x))
        candidates.append(np.exp(r * x))

    # Random smooth
    for _ in range(5):
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        amps = rng.uniform(0.5, 1.0, size=4)
        sig = np.zeros(n)
        for f, ph, am in zip(freqs, phases, amps):
            sig += am * np.sin(2 * np.pi * f * x + ph)
        candidates.append(np.maximum(1.0 + sig, 0.01))

    return [norm_sum1(c) for c in candidates]


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = norm_sum1(f)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f)
    return norm_sum1(f_int)


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining L-BFGS, coordinate descent,
    simulated annealing, and length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')
    rng = np.random.default_rng(2024)

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = evaluate_safe(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a_n)}, t={elapsed:.0f}s)")

    # Load parent sequence
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 5000
        cands = parametric_candidates(n0, rng)
        best_c = None
        best_v = float('inf')
        for c in cands:
            v = 2.0 * n0 * fast_autocorr(c).max()
            if v < best_v:
                best_v = v
                best_c = c
        best_seq = best_c.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Generated init: n={n0}, score={best_score:.8f}")

    n0 = len(best_seq)
    print(f"Starting optimization: n={n0}, score={best_score:.8f}")

    # === Phase 1: L-BFGS with continuation on parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 1: L-BFGS continuation (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, sc = run_lbfgs_continuation(best_seq, t1, tau_start=3.0, tau_end=40.0,
                                           n_cont=6, top_k=80, max_iter=200,
                                           seed=1, verbose=True)
        check_update(a_out, "lbfgs_p1")

    # === Phase 2: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 2: Coordinate descent (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, sc = coordinate_descent(best_seq, t2, seed=77, verbose=True)
        check_update(a_out, "coord_p2")

    # === Phase 3: L-BFGS with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 3: L-BFGS variant (budget={t3:.0f}s) ===")
    if t3 > 30:
        # Perturb best
        perturbed = norm_sum1(best_seq * (1 + rng.normal(0, 0.05, len(best_seq))))
        a_out, sc = run_lbfgs_continuation(perturbed, t3, tau_start=5.0, tau_end=50.0,
                                           n_cont=5, top_k=60, max_iter=150,
                                           seed=42, verbose=True)
        check_update(a_out, "lbfgs_p3")

    # === Phase 4: Simulated annealing ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: Simulated annealing (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, sc = simulated_annealing(best_seq, t4, seed=123, verbose=True)
        check_update(a_out, "simann_p4")

    # === Phase 5: Multi-restart from diverse inits ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-restart (budget={remaining:.0f}s) ===")
    n_restarts = 5
    per_restart = remaining / n_restarts * 0.7

    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        tr = min(per_restart, remaining * 0.3)
        if tr < 20:
            break

        # Different initialization strategies
        if r == 0:
            # Slightly perturb best
            init = norm_sum1(best_seq * (1 + rng.normal(0, 0.1, len(best_seq))))
        elif r == 1:
            # Mix with uniform
            w = float(rng.uniform(0.1, 0.3))
            init = norm_sum1((1 - w) * best_seq + w * np.ones(n0))
        elif r == 2:
            # Fresh parametric candidate
            cands = parametric_candidates(n0, np.random.default_rng(100 + r))
            best_c = min(cands, key=lambda c: fast_autocorr(c).max())
            init = best_c
        elif r == 3:
            # Smoothed version (blur)
            kernel_size = int(n0 * 0.01) + 1
            kernel = np.ones(kernel_size) / kernel_size
            blurred = np.convolve(best_seq, kernel, mode='same')
            init = norm_sum1(blurred)
        else:
            # Frequency-domain perturbation
            freqs = np.fft.rfft(best_seq)
            magnitudes = np.abs(freqs)
            magnitudes *= (1 + rng.normal(0, 0.1, len(magnitudes)))
            magnitudes = np.maximum(magnitudes, 0)
            phase = np.angle(freqs)
            new_freqs = magnitudes * np.exp(1j * phase)
            init = norm_sum1(np.fft.irfft(new_freqs, n0 * 2)[:n0])

        # L-BFGS on this init
        a_out, sc = run_lbfgs_continuation(init, tr, tau_start=3.0, tau_end=35.0,
                                           n_cont=4, top_k=70, max_iter=100,
                                           seed=200 + r, verbose=False)
        check_update(a_out, f"restart_{r}")

        # Quick coord descent follow-up
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            tr2 = min(remaining * 0.15, 60.0)
            if tr2 > 10:
                a_out2, sc2 = coordinate_descent(a_out, tr2, seed=300 + r, verbose=False)
                check_update(a_out2, f"restart_{r}_coord")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration (budget={remaining:.0f}s) ===")

    length_tries = []
    for delta in [-1000, -500, 500, 1000, 1500, 2000, 3000]:
        nn = n_best + delta
        if 1000 <= nn <= 10000 and nn != n_best:
            length_tries.append(nn)
    for rn in [2000, 3000, 4000, 5000, 6000, 7000, 8000]:
        if rn != n_best and 1000 <= rn <= 10000:
            length_tries.append(rn)
    length_tries = sorted(set(length_tries))

    n_len_tried = 0
    for new_n in length_tries:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100 or n_len_tried >= 4:
            break

        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_interp = interpolate_sequence(best_seq, new_n)

        # Quick L-BFGS
        tr = min(remaining * 0.35, 120.0)
        if tr > 20:
            a_out, sc = run_lbfgs_continuation(a_interp, tr, tau_start=3.0, tau_end=30.0,
                                               n_cont=4, top_k=60, max_iter=100,
                                               seed=500 + new_n, verbose=False)
            check_update(a_out, f"len_{new_n}")

            # Quick coord
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40:
                tr2 = min(remaining * 0.2, 50.0)
                if tr2 > 10:
                    a_out2, sc2 = coordinate_descent(a_out, tr2, seed=600 + new_n, verbose=False)
                    check_update(a_out2, f"len_{new_n}_coord")

        n_len_tried += 1

    # === Phase 7: Final intensive polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}, budget={remaining:.0f}s) ===")

    if remaining > 50:
        # L-BFGS with very high tau (sharp max approximation)
        tr = min(remaining * 0.4, 150.0)
        a_out, sc = run_lbfgs_continuation(best_seq, tr, tau_start=10.0, tau_end=60.0,
                                           n_cont=4, top_k=40, max_iter=200,
                                           seed=999, verbose=True)
        check_update(a_out, "final_lbfgs")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        # Final coordinate descent
        tr = min(remaining * 0.5, 100.0)
        a_out, sc = coordinate_descent(best_seq, tr, seed=42, verbose=True)
        check_update(a_out, "final_coord")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        # One more L-BFGS from slightly perturbed best
        perturbed = norm_sum1(best_seq * (1 + rng.normal(0, 0.02, len(best_seq))))
        tr = min(remaining * 0.6, 60.0)
        a_out, sc = run_lbfgs_continuation(perturbed, tr, tau_start=8.0, tau_end=50.0,
                                           n_cont=3, top_k=50, max_iter=100,
                                           seed=777, verbose=False)
        check_update(a_out, "final_perturb")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.8f}")

    return [float(x) for x in best_seq]
