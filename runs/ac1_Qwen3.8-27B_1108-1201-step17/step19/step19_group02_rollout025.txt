def smoothed_gradient_descent(a0, time_limit, eps_start=0.01, eps_end=0.001, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    t0 = time.monotonic()
    
    eps = eps_start
    epoch_length = 500
    epochs = 20
    
    for epoch in range(epochs):
        if time.monotonic() - t0 > time_limit:
            break
        eps = eps_start * (eps_end / eps_start) ** (epoch / epochs)
        
        for t in range(epoch_length):
            if time.monotonic() - t0 > time_limit:
                break
            
            c = fast_convolve(a)
            # Smoothed max gradient
            c_scaled = c / eps
            c_max = c_scaled.max()
            exp_c = np.exp(c_scaled - c_max)
            c_norm = exp_c / exp_c.sum()
            
            # Gradient of smoothed max conv = sum_k c_norm[k] * d(conv)/d(a)
            # d(conv[k])/d(a_i) = a[k-i] (for valid i)
            grad = np.zeros(n)
            for k in np.where(c_norm > 1e-10)[0]:
                k_i = int(k)
                i_start = max(0, k_i - n + 1)
                i_end = min(n, k_i + 1)
                if i_start < i_end:
                    idx = np.arange(i_start, i_end)
                    grad[idx] += 2.0 * c_norm[k_i] * a[k_i - idx]
            
            # Gradient step with projection
            eta = eps  # step size proportional to epsilon
            a_new = a - eta * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s
            
            M_new = fast_convolve(a_new).max()
            if M_new < best_M:
                best_M = M_new
                best_a = a_new.copy()
            
            a = a_new
    
    return best_a, best_M

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_conv_gradient(w, a):
    """Compute grad_j = 2 * sum_k w[k] * a[k-j] for j=0..n-1 using FFT cross-correlation.
    w has length 2n-1 (convolution peaks), a has length n."""
    n = len(a)
    size_w = len(w)
    N = 1 << (size_w + n - 1).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * corr[:n]


def smoothed_gradient_descent(a0, time_limit, verbose=True):
    """Minimize max(a*a) using log-sum-exp smoothed gradient descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    t0 = time.monotonic()

    # Parameters
    n_epochs = 30
    iters_per_epoch = 200
    eps_start = 0.005
    eps_end = 0.0003
    base_lr = 0.8
    momentum = 0.75
    vel = np.zeros(n)

    for epoch in range(n_epochs):
        if time.monotonic() - t0 > time_limit:
            break
        eps = eps_start * (eps_end / eps_start) ** (epoch / max(n_epochs - 1, 1))

        for t in range(iters_per_epoch):
            if time.monotonic() - t0 > time_limit:
                break

            # Compute convolution
            c = fast_convolve(a)
            M_curr = c.max()

            # Smoothed max weights (softmax)
            c_scaled = c / eps
            c_max = c_scaled.max()
            exp_c = np.exp(c_scaled - c_max)
            w = exp_c / exp_c.sum()

            # Compute gradient via FFT cross-correlation
            grad = compute_conv_gradient(w, a)

            # Momentum update
            vel = momentum * vel - base_lr * grad
            a_new = a + vel
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
                vel = np.zeros(n)
            else:
                a_new = a_new / s_new

            M_new = fast_convolve(a_new).max()
            if M_new < best_M - 1e-16:
                best_M = M_new
                best_a = a_new.copy()

            a = a_new

        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print("    SGD epoch=" + str(epoch) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", eps=" + repr(eps))

    return best_a, best_M


def build_constraint_matrix_vec(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs

                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            # Find best destination (lowest score, not in top peak region)
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < n // 4:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])

            if dst == src:
                continue

            found = False
            for frac in step_scales:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 150:
                # Random perturbation to escape
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 80 == 0:
                # Larger random moves
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.003, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", stall=" + str(stall))

    return best_a, best_m


def sova_update(a0, time_limit, n_rounds=500, verbose=True):
    """SOVA (Successively Over-Flow Algorithm) for fast peak reduction."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    t0 = time.monotonic()

    # Start with a target slightly below current max
    target = best_M * 0.95
    total_rounds = n_rounds

    for rnd in range(total_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        M_curr = c.max()
        if M_curr < best_M - 1e-16:
            best_M = M_curr
            best_a = a.copy()

        # Find peaks above target
        peaks = np.where(c > target)[0]
        if len(peaks) == 0:
            # Lower the target
            target = M_curr * 0.95
            if target < best_M * 0.5:
                break
            continue

        for k in peaks[:20]:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            # Find the element with maximum contribution and reduce it
            max_idx_local = np.argmax(contrib)
            src = int(idx[max_idx_local])
            src_val = a[src]
            if src_val < 1e-14:
                continue

            # How much to reduce: enough to bring c[k] down to target
            overflow = c[k_int] - target
            if overflow <= 0:
                continue
            # Reduction needed: reduce a[src] by delta, effect on c[k] is 2*delta*a[k-src] (approximately)
            partner = k_int - src
            partner_val = a[partner] if 0 <= partner < n else 0.0
            if partner_val < 1e-14:
                delta = src_val * 0.05
            else:
                delta = min(src_val * 0.5, overflow / (2.0 * partner_val + 1e-30))

            if delta < 1e-15:
                continue

            # Find destination: element with minimum "exposure" to other peaks
            exposure = np.zeros(n)
            for k2 in peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    exposure[idx2] += a[k2i - idx2]
                    exposure[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < n // 4:
                candidates = np.arange(n)
            dst = int(candidates[np.argmin(exposure[candidates])])

            if dst == src:
                continue

            a[src] -= delta
            a[dst] += delta
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a < 1e-15:
                a = best_a.copy()
            else:
                a = a / s_a

        # Gradually tighten target
        target = max(best_M * 0.5, M_curr * 0.97)

        if verbose and rnd % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print("    SOVA round=" + str(rnd) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

    return best_a, best_M


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Total time budget: " + repr(time_budget) + "s")

    # Get starting point
    best_seq = None
    best_score = float('inf')
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                best_seq = p_norm
                best_score = sc
                print("Parent: n=" + str(len(p)) + ", score=" + repr(sc))
    except Exception:
        pass

    if best_seq is None:
        n = 6000
        x = np.linspace(0, 1, n)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: SOVA for quick improvement
    print("\n=== Phase 1: SOVA (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 120.0)
    if t1 > 15:
        a_out, _ = sova_update(best_seq, t1, n_rounds=300, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sova", t_global)

    # Phase 2: Smoothed Gradient Descent
    print("\n=== Phase 2: Smoothed GD (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 30:
        a_out, _ = smoothed_gradient_descent(best_seq, t2, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sgd", t_global)

    # Phase 3: LP-FW main
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP-FW (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.30, 450.0)
    g_warm = None
    if t3 > 30 and n_best <= 10000:
        a_out, _, g_warm = lp_fw(norm_sum1(best_seq), t3, solver,
                                  initial_K=600, max_K=5000, add_per_iter=60, verbose=True,
                                  warm_start_g=g_warm)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1", t_global)

    # Phase 4: More smoothed GD (different hyperparams)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: SGD fine (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_out, _ = smoothed_gradient_descent(best_seq, t4, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sgd_fine", t_global)

    # Phase 5: Try different n values
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Multi-n exploration ===")
    try_ns = []
    for dn in [-1000, -500, 500, 1000, 1500, 2000]:
        new_n = n_best + dn
        if 3000 <= new_n <= 12000:
            try_ns.append(new_n)

    for new_n in try_ns[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print("\n    Trying n=" + str(new_n) + " (remaining=" + str(round(remaining, 1)) + "s)")
        a_long = interpolate_seq(best_seq, new_n)

        t5a = min(remaining * 0.30, 180.0)
        if t5a > 20 and new_n <= 10000:
            a_out, _ = smoothed_gradient_descent(a_long, t5a, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "sgd_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 8000:
            t5b = min(remaining * 0.35, 200.0)
            a_out, _, _ = lp_fw(norm_sum1(best_seq), t5b, solver,
                                  initial_K=500, max_K=4000, add_per_iter=50, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

    # Phase 6: Final LP-FW with warm start
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Final LP-FW (n=" + str(n_best) + ") ===")
    if remaining > 60 and n_best <= 10000:
        t6 = min(remaining * 0.40, 300.0)
        if t6 > 30:
            a_out, _, _ = lp_fw(norm_sum1(best_seq), t6, solver,
                                 initial_K=800, max_K=6000, add_per_iter=80, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # Phase 7: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final Coord (n=" + str(n_best) + ") ===")
    if remaining > 20:
        t7 = min(remaining * 0.60, 200.0)
        if t7 > 15:
            a_out, _ = coordinate_polish(norm_sum1(best_seq), t7, rng_seed=123, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Phase 8: One more SOVA pass
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t8 = min(remaining * 0.50, 100.0)
        if t8 > 10:
            a_out, _ = sova_update(best_seq, t8, n_rounds=200, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "sova_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
