    Compute d(sum_k w[k]*r[k])/da[i] = 2*E[i]
    where E[i] = sum_j w[i+j]*a[j]
    Using FFT cross-correlation.

    E = np.fft.irfft(W * A.conj(), N)[:n]
    return 2.0 * E

def peak_shave(a0, time_limit, max_iters=10000, verbose=True, rng_seed=77):
    """
    Targeted coordinate descent: identify worst lag, move mass to reduce it.
    """
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                # Random perturbation
                n_moves = int(rng.integers(10, 50))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = float(rng.uniform(0.001, 0.02))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = norm_sum1(a)
                stall = 0
                continue
        
        # Find worst lag
        k_worst = int(np.argmax(ac))
        i_lo = max(0, k_worst - n + 1)
        i_hi = min(n, k_worst + 1)
        idx = np.arange(i_lo, i_hi)
        j_idx = k_worst - idx
        contrib = a[idx] * a[j_idx]
        
        # Find the position with highest contribution to this lag
        # and move mass from it to a position with low "damage"
        top_pi = int(np.argmax(contrib))
        src = int(idx[top_pi])
        src_val = a[src]
        
        if src_val < 1e-15:
            continue
        
        # Compute "damage score" for each position
        # (contribution to top few lags)
        top_lags = np.argsort(ac)[::-1][:10]
        damage = np.zeros(n)
        for k2 in top_lags:
            k2i = int(k2)
            s2 = max(0, k2i - n + 1)
            e2 = min(n, k2i + 1)
            ii = np.arange(s2, e2)
            jj = k2i - ii
            damage[ii] += ac[k2i] * a[jj]
            damage[jj] += ac[k2i] * a[ii]
        
        # Choose destination: low damage, not the source
        damage[src] = np.inf
        dst = int(np.argmin(damage))
        
        # Try moving a fraction
        for frac in [0.01, 0.03, 0.05, 0.1]:
            amt = src_val * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = norm_sum1(a_trial)
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_m - 1e-15:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                break
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    PeakShave done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m

def make_templates(n, rng):
    """Generate diverse initial templates of length n."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    templates = []
    
    # Uniform
    templates.append(np.ones(n))
    
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        templates.append((i_arr + 1.0) ** p)
        templates.append((n - i_arr + 1.0) ** p)
    
    # Gaussians
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sig in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]:
            templates.append(np.exp(-((x - c)**2) / (2*sig**2)))
    
    # Bimodal
    for sig in [0.05, 0.1, 0.15, 0.2]:
        for c in [0.25, 0.35]:
            templates.append(
                np.exp(-((x-c)**2)/(2*sig**2)) + np.exp(-((x-(1-c))**2)/(2*sig**2)))
    
    # Sine waves
    for f in [1, 2, 3, 4, 5]:
        templates.append(np.sin(np.pi * f * x + rng.uniform(0, np.pi))**2 + 0.01)
    
    # Exponential
    for r in [0.5, 1, 2, 3, 5]:
        templates.append(np.exp(-r * x))
        templates.append(np.exp(r * x))
    
    # Random smooth
    for _ in range(5):
        freq = int(rng.integers(1, 8))
        phase = rng.uniform(0, np.pi)
        templates.append(np.sin(np.pi * freq * x + phase)**2 + 0.01)
    
    # Rician-like / "comet"
    for c in [0.1, 0.3, 0.5]:
        templates.append(np.exp(-abs(x - c) * rng.uniform(2, 10)))
    
    return [norm_sum1(t) for t in templates]

        if time.monotonic() - t0 > time_limit * 0.85:
            break
        r = rng.random(n)
        r = norm_sum1(r)
        v = score_local(r)
        if v < best_v:
            best_v = v
            best_t = r.copy()
    
    return best_t if best_t is not None else np.ones(n) / n

def downsample(a, factor):
    """Downsample by averaging blocks."""
    n = len(a)
    new_n = n // factor
    if new_n < 10:
        return a.copy()
    a_padded = np.pad(a, (0, factor - n % factor))
    a_block = a_padded[:new_n * factor].reshape(new_n, factor)
    return a_block.mean(axis=1) / a_block.mean(axis=1).sum() * n / new_n
    # Actually just normalize:
    result = a_block.mean(axis=1)
    return norm_sum1(result)

def upsample(a, factor):
    """Upsample by linear interpolation."""
    n = len(a)
    new_n = n * factor
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return norm_sum1(np.interp(x_new, x_old, a))
