Actually, I realize the simplest correct formula is:
$$grad_j = 2 \sum_{k=0}^{2n-2} w_k \cdot a_{k-j} \cdot \mathbf{1}[0 \leq k-j < n]$$

Let $m = k - j$, so $k = m + j$:
$$grad_j = 2 \sum_{m=0}^{n-1} w_{m+j} \cdot a_m \cdot \mathbf{1}[m+j \leq 2n-2]$$

This is the cross-correlation of $a$ and $w$ (padded to length $2n-1$) evaluated at lag $j$.

Using the standard cross-correlation formula:
$$(a \star w)(j) = \sum_m a(m) \cdot w(m+j)$$

In FFT: $(a \star w) = \text{IFFT}(\text{FFT}(a) \cdot \overline{\text{FFT}(w)})$

Wait, I need to be careful with conventions. The standard result is:
$$\text{IFFT}(\text{FFT}(a) \cdot \overline{\text{FFT}(b)})[n] = \sum_m a[m] \cdot b[n+m]$$

So if I set $a = a$ (padded) and $b = w$ (padded), then:
$$\text{IFFT}(\text{FFT}(a\_padded) \cdot \overline{\text{FFT}(w\_padded)})[j] = \sum_m a[m] \cdot w[m+j]$$

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_obj(a):
    ac = fast_convolve(a)
    return 2.0 * len(a) * ac.max()


def try_update(best_seq, best_score, a, label, t_global):

def project_simplex(v):
    """Project v onto probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1).astype(np.float64)
    cond = u - cssv / ind > 0
    i = np.max(np.where(cond)[0])
    theta = cssv[i] / (i + 1)
    return np.maximum(v - theta, 0.0)


def compute_gradient_hard(a, ac):
    """Compute gradient of max_k (a*a)_k using hard max."""
    n = len(a)
    k_star = int(np.argmax(ac))
    grad = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(n - 1, k_star)
    if j_start <= j_end:
        idx = np.arange(j_start, j_end + 1)
        grad[idx] = 2.0 * a[k_star - idx]
    return grad


def compute_gradient_soft(a, ac, beta=8.0):
    """Compute gradient using softmax-weighted average of autocorrelation peaks."""
    n = len(a)
    size = 2 * n - 1
    shifted = ac - ac.max()
    weights = np.exp(beta * shifted)
    total = weights.sum()
    if total < 1e-30:
        top_k = int(np.argmax(ac))
        return compute_gradient_hard(a, ac)
    weights /= total

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(weights, fft_size)
    corr = np.fft.irfft(A * np.conj(W), fft_size)[:n]
    return 2.0 * corr


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k = int(constraint_list[jj])
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k - idx]
    return coeffs


def lp_fw(a0, time_limit, solver, initial_K=500, max_K=4000, add_per_iter=50,
          verbose=True):
    """Improved LP-based Frank-Wolfe with aggressive active set management."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_convolve(a)
    best_m = ac.max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_convolve(y)
    active_set = list(np.argsort(ac_y)[::-1][:min(initial_K, len(ac_y))])

    no_improve = 0
    max_no_improve = 800
    iter_count = 0
    lp_fail_count = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: stagnation at iter %d, stopping" % iter_count)
            break
        iter_count += 1
        lp_fail_count = 0

        M_curr = fast_convolve(y).max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Prune active set if too large
        if len(active_set) > max_K:
            ac_check = fast_convolve(y)
            margins = np.array([M_curr - ac_check[k] for k in active_set])
            keep = np.argsort(margins)[:max_K]
            active_set = [active_set[i] for i in keep]

        K = len(active_set)
        if K == 0:
            break

        coeffs = build_constraint_matrix(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = list(active_set[:initial_K])
            lp_fail_count += 1
            if lp_fail_count > 5:
                break
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail_count += 1
            if lp_fail_count > 5:
                break
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations: find where y*g_dir exceeds M_curr
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)

        # Alpha selection: try multiple step sizes
        default_alpha = 2.0 / (iter_count + 2.0)
        alphas = [default_alpha, default_alpha * 0.5, default_alpha * 1.5,
                  1.0 / (iter_count + 1.0), 0.1, 0.3, 0.05]
        alphas = [max(0.01, min(1.0, a)) for a in alphas]
        # Remove duplicates
        seen = set()
        unique_alphas = []
        for a in alphas:
            if a not in seen:
                seen.add(a)
                unique_alphas.append(a)

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in unique_alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 2:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        # Periodic active set refresh
        if iter_count % 200 == 0 and verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=%d (%.1fs): J=%.6f, |S|=%d" % (iter_count, elapsed, J, len(active_set)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW done (%.1fs): J=%.6f, iters=%d" % (elapsed, J, iter_count))
    return best_a, best_m


def projected_gradient(a0, time_limit, beta=8.0, verbose=False):
    """Projected gradient with Barzilai-Borwein step sizes."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_convolve(a)
    best_m = ac.max()
    t0 = time.monotonic()

    a_prev = a.copy()
    grad_prev = None
    bb_step = 0.01 / n
    stall = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_convolve(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 5000:
            break

        # Compute gradient
        if t % 50 == 0:
            grad = compute_gradient_soft(a, ac, beta=beta)
        else:
            grad = compute_gradient_hard(a, ac)

        # BB step size
        if grad_prev is not None:
            s = a - a_prev
            y_diff = grad - grad_prev
            s_norm = np.dot(s, s)
            s_y = np.dot(s, y_diff)
            if s_y > 1e-15:
                bb_step = s_norm / s_y
                bb_step = max(1e-6, min(bb_step, 0.1 / n))
            elif s_norm > 1e-15:
                s_norm_grad = np.dot(s, grad)
                if s_norm_grad > 1e-15:
                    bb_step = s_norm / (s_norm_grad * n)
                    bb_step = max(1e-6, min(bb_step, 0.1 / n))

        # Backtracking line search
        alpha = bb_step
        for _ in range(20):
            a_trial = a - alpha * grad
            a_trial = project_simplex(a_trial)
            M_trial = fast_convolve(a_trial).max()
            if M_trial < M_curr + 1e-12:
                break
            alpha *= 0.5

        a_prev = a.copy()
        grad_prev = grad.copy()
        a = a - alpha * grad
        a = project_simplex(a)

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PG iter=%d (%.1fs): J=%.6f, alpha=%.2e" % (t, elapsed, J, alpha))

    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=42, verbose=False):
    """Coordinate descent with peak-aware mass transfer."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_convolve(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 600

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score: how much each position contributes to peaks
            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            candidates = np.where(scores == scores.min())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            dst = int(candidates[np.argmin(scores[candidates])])

            if dst == src:
                continue

            found = False
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 100:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.005, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 50 == 0:
                for _ in range(3):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.01)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (%.1fs): J=%.6f, stall=%d" % (elapsed, J, stall))

    return best_a, best_m


def parametric_init(n, time_limit, best_init, best_score):
    """Try diverse parametric initializations."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score

    candidates = []

    # Product of raised cosines
    for a1 in [1.0, 1.5, 2.0]:

    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.25, 0.3]:
            candidates.append(norm_sum1(np.exp(-((x - c) ** 2) / (2.0 * sigma ** 2))))
    # Double bump
    for sigma in [0.08, 0.1, 0.15]:
        candidates.append(norm_sum1(np.exp(-((x - 0.3) ** 2) / (2 * sigma**2)) +
                                     np.exp(-((x - 0.7) ** 2) / (2 * sigma**2))))
    # Triangular
    candidates.append(norm_sum1(1.0 - np.abs(2.0 * x - 1.0)))
    # Power laws
    for p in [0.5, 1.0, 1.5, 2.0]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))

    rng = np.random.default_rng(123)
    for _ in range(100):
        if time.monotonic() - t0 > time_limit:
            break
        a = rng.random(n)
        a = a ** rng.uniform(0.5, 3.0)
        candidates.append(norm_sum1(a))

    for a in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        v = compute_obj(a)
        if v < best_val:
            best_val = v
            best_a = a.copy()

    return best_a, best_val


def propose_candidate(budget_s=1800.0):
    """Main entry point: search for the best sequence within the time budget."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Total time budget: %.1fs" % time_budget)

    # Get initial solution
    best_seq = None
    best_score = float('inf')
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                best_seq = p_norm
                best_score = sc
                print("Parent: n=%d, score=%.6f" % (len(p), sc))
    except Exception:
        pass

    if best_seq is None:
        n = 5000
        x = np.linspace(0, 1, n)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=%d, score=%.6f" % (len(best_seq), best_score))

    n0 = len(best_seq)

    # Phase 1: Quick parametric search at current scale
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Parametric (n=%d) ===" % n0)
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 5:
        a_param, _ = parametric_init(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "parametric", t_global)

    # Phase 2: LP-FW main (most important phase - give it the most time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: LP-FW (n=%d) ===" % n_best)
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 60 and n_best <= 9000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t2, solver,
                         initial_K=600, max_K=4000, add_per_iter=50, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n%d" % n_best, t_global)

    # Phase 3: Projected gradient refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Projected Gradient (n=%d) ===" % n_best)
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 30:
        a_out, _ = projected_gradient(norm_sum1(best_seq), t3, beta=8.0, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pg_n%d" % n_best, t_global)

    # Phase 4: LP-FW again with refreshed active set
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW refine (n=%d) ===" % n_best)
    t4 = min(remaining * 0.22, 300.0)
    if t4 > 60 and n_best <= 9000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t4, solver,
                         initial_K=800, max_K=5000, add_per_iter=60, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2_n%d" % n_best, t_global)

    # Phase 5: Multi-scale - try nearby n values
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Multi-scale ===")
    try_ns = []
    for delta in [300, -300, 600, -600, 900]:
        nn = n_best + delta
        if 3000 <= nn <= 10000 and nn not in try_ns:
            try_ns.append(nn)

    for new_n in try_ns[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print("  Trying n=%d" % new_n)
        a_long = interpolate_seq(best_seq, new_n)

        t5a = min(remaining * 0.4, 180.0)
        if t5a > 40 and new_n <= 9000:
            a_out, _ = lp_fw(a_long, t5a, solver,
                             initial_K=500, max_K=3500, add_per_iter=40, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n%d" % new_n, t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t5b = min(remaining * 0.25, 80.0)
            if t5b > 20:
                a_out, _ = projected_gradient(norm_sum1(best_seq), t5b, beta=6.0, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "pg_n%d" % new_n, t_global)

    # Phase 6: Final LP-FW push
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Final LP-FW (n=%d) ===" % n_best)
    if remaining > 80 and n_best <= 9000:
        t6 = min(remaining * 0.5, 350.0)
        if t6 > 40:
            a_out, _ = lp_fw(norm_sum1(best_seq), t6, solver,
                             initial_K=1000, max_K=5000, add_per_iter=80, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # Phase 7: Final projected gradient
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final PG (n=%d) ===" % n_best)
    if remaining > 30:
        t7 = min(remaining * 0.35, 100.0)
        if t7 > 15:
            a_out, _ = projected_gradient(norm_sum1(best_seq), t7, beta=10.0, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "pg_final", t_global)

    # Phase 8: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Coord Polish (n=%d) ===" % n_best)
    if remaining > 20:
        t8 = min(remaining * 0.6, 80.0)
        if t8 > 10:
            a_out, _ = coordinate_polish(norm_sum1(best_seq), t8, rng_seed=77, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=%.6f, n=%d, time=%.1fs" % (best_score, len(best_seq), elapsed))
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=%.6f" % best_score)

    return [float(x) for x in best_seq]
