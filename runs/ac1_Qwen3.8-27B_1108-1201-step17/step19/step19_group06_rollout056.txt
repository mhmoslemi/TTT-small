def subgradient_refine(a0, time_limit, n_iters=10000, verbose=False):
    """
    Subgradient method for exact max (not smoothed).
    Uses projected subgradient descent on the simplex.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    lr = 1.0
    decay = 0.999
    
    for t in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
        
        # Find active lags (within tolerance of max)
        tol = 1e-8 * max(M, 1e-10)
        active = ac >= (M - tol)
        active_idx = np.where(active)[0]
        n_active = len(active_idx)
        
        # Subgradient: average of gradients of active constraints
        # d(C[k])/da[j] = 2*a[k-j] if 0<=k-j<n
        grad = np.zeros(n)
        for k_int in active_idx:
            j_lo = max(0, k_int - n + 1)
            j_hi = min(n - 1, k_int)
            if j_lo <= j_hi:
                m = k_int - np.arange(j_lo, j_hi + 1)
                grad[j_lo:j_hi+1] += 2.0 * a[m]
        grad /= n_active
        
        # Projected gradient step on simplex (approximately)
        # First, do the step
        a_new = a - lr * grad
        # Project onto simplex: a >= 0, sum(a) = 1
        # Use the water-filling algorithm
        a_new = a_new - a_new.min() if a_new.min() < 0 else a_new
        s = a_new.sum()
        if s > 0:
            a_new = a_new / s
        else:
            a_new = a.copy()
        
        # Ensure non-negativity after projection
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 0:
            a_new = a_new / s
        
        a = a_new
        lr *= decay
    
    if verbose:
        print(f"    Subgradient done ({time.monotonic()-t0:.0f}s): max_ac={best_m:.8f}")
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=False):
    """
    Fine-grained coordinate descent.
    For the top autocorrelation peaks, try redistributing mass.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)
    
    iter_count = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        ac = fast_autocorr(a)
        M_cur = ac.max()
        
        if M_cur < best_m - 1e-15:
            best_m = M_cur
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break
        
        # Get top lags
        n_top = min(20, 2*n-1)
        top_lags = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        for k_pos in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k_pos)
            
            # d(C[k])/da[j] = 2*a[k-j] for valid j
            # We want to reduce C[k] by moving mass.
            # The "cost" of having mass at position j for lag k is proportional to a[k-j].
            # So we want to move mass from high-cost positions to low-cost positions.
            
            j_lo = max(0, k_int - n + 1)
            j_hi = min(n - 1, k_int)
            if j_lo > j_hi:
                continue
            
            # Cost at each position j for this lag
            m = k_int - np.arange(j_lo, j_hi + 1)
            cost = a[m]  # This is a[k-j] for j in [j_lo, j_hi]
            
            # Find position with highest a[j]*cost (most contributing to this lag)
            contrib = a[j_lo:j_hi+1] * cost
            if contrib.max() < 1e-15:
                continue
            src = j_lo + int(np.argmax(contrib))
            src_val = a[src]
            if src_val < 1e-14:
                continue
            
            # Find destination: minimize impact on all top lags
            # Impact of moving to position d: for each top lag k2, the increase is
            # 2 * a2[k2-d] (approximately, for small transfers)
            min_impact = float('inf')
            best_dst = -1
            for d in range(n):
                if d == src:
                    continue
                impact = 0.0
                for k2 in top_lags[:5]:
                    k2i = int(k2)
                    if 0 <= k2i - d < n:
                        impact += a[k2i - d] * ac[k2i]
                if impact < min_impact:
                    min_impact = impact
                    best_dst = d
            
            if best_dst < 0:
                continue
            
            # Try small transfer
            for frac in [0.01, 0.03, 0.05, 0.1]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial /= s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved and stall > 100:
            # Random perturbation
            n_pert = int(rng.integers(3, 10))
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    amt = a[i] * float(rng.uniform(0.001, 0.01))
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 0:
                a /= s

    if verbose:
        J = 2.0 * n * best_m
        print(f"    Coord polish done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss

def make_beta_schedule(start_beta, end_beta, n_steps):
    """Create a geometric beta schedule from start to end."""
    if n_steps <= 1:
        return [end_beta]
    return np.geomspace(start_beta, end_beta, n_steps)

Chain rule:
df/dx_j = (df/da) · (da/dx)

da/dx_j: da_i/dx_j = a_i(δ_ij - a_j) (standard softmax Jacobian)

So: df/dx_j = Σ_i (df/da_i) * a_i(δ_ij - a_j) = a_j * df/da_j - a_j * Σ_i (df/da_i) * a_i

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on log-parametrization with log-sum-exp smoothed objective
2. Adaptive beta annealing (simulated-annealing style)
3. Subgradient refinement for exact max handling
4. Block coordinate descent polish
5. Multi-length exploration with interpolation
6. Multiple diverse initializations with restarts
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely call the evaluation function."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def score_from_ac(a):
    """Compute score from a normalized sequence (sum=1)."""
    n = len(a)
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max())


def softmax_from_x(x):
    """Compute softmax of x (numerically stable)."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(x) / len(x)
    return a / s


def logsumexp_objective_and_grad(x, n, beta):
    """
    Compute objective and gradient of the log-sum-exp smoothed objective.
    Objective: (2n/beta) * log(sum_k exp(beta * C_k))
    where C_k = conv(a,a)[k], a = softmax(x)
    
    Returns (objective_value, gradient_wrt_x)
    """
    a = softmax_from_x(x)
    ac = fast_autocorr(a)
    
    # Log-sum-exp of beta * ac
    beta_ac = beta * ac
    max_val = beta_ac.max()
    exp_vals = np.exp(beta_ac - max_val)
    log_sum = max_val + np.log(exp_vals.sum())
    obj_val = (2.0 * n / beta) * log_sum
    
    # Softmax weights over lags
    weights = exp_vals / exp_vals.sum()
    
    # Gradient of sum_k w_k * C_k w.r.t. a:
    # d(C_k)/da[j] = 2*a[k-j] if 0 <= k-j < n
    # grad_a[j] = sum_k w_k * 2*a[k-j] (valid k-j)
    
    # For efficiency, only process lags with significant weight
    threshold = 1e-12
    significant = weights > threshold
    sig_idx = np.where(significant)[0]
    sig_w = weights[sig_idx]
    
    grad_a = np.zeros(n)
    for k_int, w_val in zip(sig_idx, sig_w):
        # d(C_k)/da[j] = 2*a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
        j_lo = max(0, k_int - n + 1)
        j_hi = min(n - 1, k_int)
        if j_lo <= j_hi:
            m = k_int - np.arange(j_lo, j_hi + 1)
            grad_a[j_lo:j_hi + 1] += 2.0 * w_val * a[m]
    
    # Chain rule for log-parametrization:
    # df/dx_j = a_j * (grad_a_j - sum_i a_i * grad_a_i)
    mean_grad = np.dot(a, grad_a)
    grad_x = a * (grad_a - mean_grad)
    
    return obj_val, grad_x


def lbgfs_optimize(a0, time_limit, beta_schedule, verbose=True, maxiter_per_beta=300):
    """
    Optimize using L-BFGS on log-parametrization with adaptive beta schedule.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = score_from_ac(a)
    t0 = time.monotonic()
    
    x = np.log(np.maximum(a, 1e-30))
    
    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        elapsed_beta = time.monotonic() - t0
        remaining_beta = time_limit - elapsed_beta
        if remaining_beta < 2.0:
            break
        
        def obj_func(x_eval):
            val, _ = logsumexp_objective_and_grad(x_eval, n, beta)
            return val
        
        def grad_func(x_eval):
            _, g = logsumexp_objective_and_grad(x_eval, n, beta)
            return g
        
        try:
            result = minimize(
                obj_func, x, jac=grad_func, method='L-BFGS-B',
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-14, 'gtol': 1e-10,
                         'maxfun': maxiter_per_beta * 2}
            )
            x = result.x
        except Exception as e:
            if verbose:
                print(f"      L-BFGS error at beta={beta}: {e}")
            break
        
        a_new = softmax_from_x(x)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        
        sc = score_from_ac(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            if verbose:
                print(f"      beta={beta:.0f}: score={sc:.8f} (improved)")
        else:
            if verbose:
                print(f"      beta={beta:.0f}: score={sc:.8f} (no improvement)")
        
        a = a_new
    
    return best_a, best_score


def subgradient_refine(a0, time_limit, verbose=False):
    """
    Subgradient method with adaptive step size for exact max.
    Uses the Moreau-Yosida regularization idea with decreasing step.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Adaptive learning rate
    lr = 1.0 / np.sqrt(n)
    n_iters = 0
    patience = 0
    best_patience = 0
    
    while time.monotonic() - t0 < time_limit:
        n_iters += 1
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            patience = 0
            # Decrease lr when improving
            lr *= 0.995
        else:
            patience += 1
            if patience > 2000:
                break
            if patience % 500 == 0:
                lr *= 0.9
        
        # Find active lags (within relative tolerance of max)
        tol = 1e-6 * max(M, 1e-15)
        active_mask = ac >= (M - tol)
        active_idx = np.where(active_mask)[0]
        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(ac))])
        n_active = len(active_idx)
        
        # Subgradient: average gradient of active constraints
        grad = np.zeros(n)
        for k_int in active_idx:
            j_lo = max(0, k_int - n + 1)
            j_hi = min(n - 1, k_int)
            if j_lo <= j_hi:
                m = k_int - np.arange(j_lo, j_hi + 1)
                grad[j_lo:j_hi + 1] += 2.0 * a[m]
        grad /= n_active
        
        # Projected gradient step on simplex
        # Center the gradient (project onto tangent space of simplex)
        grad_centered = grad - grad.mean()
        
        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        else:
            a_new = a.copy()
        
        a = a_new
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Subgradient done ({time.monotonic()-t0:.0f}s, {n_iters} iters): J={J:.6f}")
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False):
    """
    Fine-grained coordinate descent: for top autocorrelation peaks,
    redistribute mass from high-impact to low-impact positions.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()

            best_m = M_cur
            best_a = a.copy()
            stall = 0

            stall += 1
            if stall > 10000:
                break

        n_top = min(25, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:n_top]

            # Positions contributing to this lag
            j_lo = max(0, k_int - n + 1)
            j_hi = min(n - 1, k_int)

                continue
