def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def topk_gradient(a, ac, top_k=80, beta=8.0):
    n = len(a)
    k_sel = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_sel]
    top_vals = ac[top_idx]
    w = np.exp(beta * (top_vals - top_vals[0]))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_sel) / k_sel
    else:
        w = w / w_sum
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        if ki - (n - 1) <= 0:
            i_start = 0
            i_end = ki + 1
        else:
            i_start = ki - (n - 1)
            i_end = n
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        g[idx] += wi * a[j_idx]
    return g

    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = topk_gradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every // 2:
            cur_beta = max(cur_beta * 0.92, 3.0)
            cur_top_k = min(cur_top_k + 5, 300)

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 200)

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Grad t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def spectral_project(a, blend=0.3):
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    target = np.median(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
    new_mag = (1 - blend) * mag + blend * target
    A_new = new_mag * np.exp(1j * np.angle(A))
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s


def lp_frank_wolfe(a0, time_limit, solver, top_K_init=300, top_K_max=None,
                   verbose=False, max_iters=200):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = min(3000, 2 * n - 1)
    K = top_K_init
    iter_count = 0
    prev_b = None

    while time.monotonic() - t0 < time_limit and iter_count < max_iters:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        if prev_b is not None and len(prev_b) == n:
            b_var.value = prev_b
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            iter_count += 1
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 50, top_K_max)
            iter_count += 1
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            K = min(K + 100, top_K_max)
            iter_count += 1
            continue
        g_dir = b_val / b_sum
        prev_b = b_val.copy()

        base_obj = best_m
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = fast_autocorr(a).max()

        # Backtracking line search
        alpha = min(1.0, 2.0 / (iter_count + 2.0))
        for _ in range(15):
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                alpha *= 0.5
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha
                break
            alpha *= 0.5

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1
        if iter_count % 5 == 0:
            K = min(K + 50, top_K_max)

        if verbose and iter_count % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

    return best_a, best_m


def hybrid_optimize(a0, time_limit, solver, verbose=False, rng_seed=0):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    it = 0
    stall = 0
    max_stall = 500

    t_lp_interval = max(5, int(120.0 / max(1, (time_limit / 30))))
    last_lp_time = t0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        it += 1
        elapsed = time.monotonic() - t0

        # Every t_lp_interval iterations, do an LP step
        if it % t_lp_interval == 0 and (elapsed - last_lp_time) > 15 and n <= 5000:
            last_lp_time = elapsed
            remaining_lp = min(60.0, (time_limit - elapsed) * 0.3)
            if remaining_lp > 5:
                a_lp, m_lp = lp_frank_wolfe(a, remaining_lp, solver,
                                             top_K_init=min(400, 2*n-1),
                                             top_K_max=min(2000, 2*n-1),
                                             verbose=False, max_iters=20)
                if m_lp < best_m:
                    best_m = m_lp
                    best_a = a_lp.copy()
                    a = a_lp.copy()
                    stall = 0
                continue

        # Spectral projection every 50 iterations
        if it % 50 == 0:
            a_spec = spectral_project(a, blend=0.2)
            m_spec = fast_autocorr(a_spec).max()
            if m_spec < best_m:
                best_m = m_spec
                best_a = a_spec.copy()
                a = a_spec
                stall = 0
            else:
                stall += 1
            continue

        # Gradient step
        ac = fast_autocorr(a)
        top_k = min(60 + it // 100, 200)
        beta = max(5.0 - it * 0.002, 2.0)
        g = topk_gradient(a, ac, top_k=top_k, beta=beta)

        eta = 3.0 / (np.sqrt(it + 1) + 1.0)
        log_a = np.log(np.maximum(a, 1e-15)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            a = a_new

        if stall > 200:
            pert = rng.normal(0, 0.015, n) * best_a
            a_p = np.maximum(best_a + pert, 1e-12)
            a = a_p / a_p.sum()
            stall = 100

        if verbose and it % 500 == 0:
            J = 2.0 * n * best_m
            print(f"    Hybrid it={it} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 200
    n_top = 30

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_top]

        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:10]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            top_cands = candidates[np.argsort(sens[candidates])[:5]]

            found = False
            for dst in top_cands:
                if dst == src or found:
                    continue
                for frac in [0.005, 0.002, 0.01, 0.02, 0.001, 0.03]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-18:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

            if not found and stall > 50:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.008)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def make_cosine(n):
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = a ** 2
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 6200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # Phase 1: POCS spectral init
    print(f"\n=== Phase 1: POCS (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    if t1 > 30:
        a_pocs, _ = pocs_optimize(best_seq, t1, n_trials=4, rng_seed=7)
        try_update(a_pocs, "pocs")

    # Phase 2: Multi-init gradient optimization
    print(f"\n=== Phase 2: Gradient multi-init (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 60:
        inits = []
        labels = []
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        inits.append(make_u_shape(n0, 2.0))
        labels.append("u2")
        inits.append(make_u_shape(n0, 1.0))
        labels.append("u1")
        inits.append(make_bimodal(n0, 0.12))
        labels.append("bimodal")
        inits.append(make_cosine(n0))
        labels.append("cos")
        inits.append(norm_sum1(best_seq))
        labels.append("best")

        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 50)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(20.0, t2 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  Grad '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = gradient_optimize(init, t_avail, top_k=80, beta=7.0,
                                          c0=4.0, restart_every=3000,
                                          rng_seed=100 + idx)
            try_update(a_out, f"grad_{lbl}")

    # Phase 3: Hybrid gradient + spectral + LP
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Hybrid (n={n_best}) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = hybrid_optimize(a_in, t3, solver, verbose=True, rng_seed=50)
        try_update(a_out, "hybrid")

    # Phase 4: LP-FW focused
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW (n={n_best}) ===")
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True,
                                   top_K_init=min(400, 2 * n_best - 1),
                                   top_K_max=min(3000, 2 * n_best - 1),
                                   max_iters=80)
        try_update(a_out, "lp1")

    # Phase 5: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord polish (n={n_best}) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=False, rng_seed=456)
        try_update(a_out, "coord1")

    # Phase 6: Length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")
    trial_lengths = []
    for delta in [300, -300, 600, -600, 1000, -1000, 500, -500]:
        new_n = n_best + delta
        if 1000 <= new_n <= 7500:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.35, 120.0)
        if t6a > 15:
            a_out, _ = gradient_optimize(a_long, t6a, top_k=70, beta=6.0,
                                          c0=4.0, restart_every=4000, rng_seed=new_n % 1000)
            try_update(a_out, f"grad_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 5000:
            t6b = min(remaining * 0.4, 100.0)
            if t6b > 15:
                a_in2 = norm_sum1(interpolate_sequence(best_seq, new_n))
                a_out, _ = lp_frank_wolfe(a_in2, t6b, solver, verbose=False,
                                           top_K_init=min(300, 2 * new_n - 1),
                                           top_K_max=min(2000, 2 * new_n - 1),
                                           max_iters=30)
                try_update(a_out, f"lp_n{new_n}")

    # Phase 7: Final hybrid on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final hybrid (n={n_best}) ===")
    if remaining > 40:
        t7 = min(remaining * 0.4, 150.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = hybrid_optimize(a_in, t7, solver, verbose=False, rng_seed=777)
            try_update(a_out, "hybrid_final")

    # Phase 8: Final LP
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30 and n_best <= 5000:
        print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
        t8 = min(remaining * 0.5, 120.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t8, solver, verbose=False,
                                       top_K_init=min(500, 2 * n_best - 1),
                                       top_K_max=min(4000, 2 * n_best - 1),
                                       max_iters=40)
            try_update(a_out, "lp_final")

    # Phase 9: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 9: Final coord (n={len(best_seq)}) ===")
        t9 = min(remaining * 0.8, 80.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t9, verbose=False, rng_seed=999)
            try_update(a_out, "coord_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
