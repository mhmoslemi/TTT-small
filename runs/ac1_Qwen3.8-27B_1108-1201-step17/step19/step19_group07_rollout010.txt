def fft_autocorr(a):
    """Compute full autocorrelation a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(A * A, fft_size)[:size]
    return corr


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def eval_obj(a):
    """Compute J = 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    corr = fft_autocorr(a)
    return 2.0 * n * corr.max() / (s * s)

def softmax_weights(corr, beta, top_k=None):
    """Compute softmax weights over autocorrelation values."""
    n_c = len(corr)
    if top_k is not None and top_k < n_c:
        idx = np.argpartition(corr, -top_k)[-top_k:]
        c_vals = corr[idx]
        c_max = c_vals.max()
        exps = np.exp(beta * (c_vals - c_max))
        s = exps.sum()
        w = np.zeros(n_c)
        if s > 1e-300:
            w[idx] = exps / s
        else:
            w[c_vals.argmax() + (idx[0] if len(idx) > 0 else 0)] = 1.0
            w = np.zeros(n_c)
            w[idx[np.argmax(c_vals)]] = 1.0

        c_max = corr.max()
        exps = np.exp(beta * (corr - c_max))
        s = exps.sum()
        if s < 1e-300:
            w = np.zeros(n_c)
            w[corr.argmax()] = 1.0
        else:
            w = exps / s
    return w

    w_rev = w[::-1]
    conv_len = n + size - 1  # = 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    Wr = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(A * Wr, fft_size)[:conv_len]
    
    # grad[l] = 2 * conv[2n-2-l], l=0..n-1
    # indices: conv[2n-2], conv[2n-3], ..., conv[n-1]
    rev = conv[n-1:2*n-1][::-1]
    grad = 2.0 * rev
    return grad, corr, w


def fista_optimize(a0, time_limit, beta_start=5.0, beta_end=100.0,
                   eta0=1.0, use_top_k=True, top_k_frac=0.05,
                   verbose=True, seed=42):
    """FISTA with softmax-smoothed gradient and Nesterov acceleration."""
    n = len(a0)
    x = normalize(a0)
    y = x.copy()
    best_x = x.copy()
    best_m = fft_autocorr(x).max()
    t0 = time.monotonic()
    t = 0
    stall = 0
    stall_threshold = max(2000, n // 3)
    rng = np.random.default_rng(seed)
    
    top_k = None
    if use_top_k:
        top_k = max(50, int(top_k_frac * (2 * n - 1)))
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / max(time_limit, 1.0), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Adaptive top_k: increase as we converge
        if use_top_k:
            tk = max(30, int(top_k * (1.0 - 0.5 * progress)))
        else:
            tk = None
        
        grad, corr_y, _ = smooth_gradient_fft(y, beta, top_k=tk)
        
        # Step size: decreases with t, scaled by beta
        eta = eta0 / (t ** 0.5 + 1.0) * min(1.0, 50.0 / (beta + 1.0))
        
        # Mirror descent in log-space
        log_y = np.log(np.maximum(y, 1e-300)) - eta * grad
        log_y -= log_y.max()
        z = np.exp(log_y)
        s = z.sum()
        if s < 1e-300:
            z = np.ones(n) / n
        else:
            z = z / s
        
        # Nesterov momentum
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = z + momentum * (z - y)
            x_new = np.maximum(x_new, 1e-300)
            sx = x_new.sum()
            if sx < 1e-300:
                x_new = z.copy()
            else:
                x_new = x_new / sx
        else:
            x_new = z.copy()
        
        # Track best
        corr_new = fft_autocorr(x_new)
        m_new = corr_new.max()
        if m_new < best_m - 1e-18:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart on stall
        if stall > stall_threshold:
            pert_size = 0.01 + 0.03 * rng.random()
            mask = rng.random(n) < 0.3
            pert = rng.normal(0, pert_size, n)
            pert[~mask] = 0
            x_restart = normalize(best_x + pert)
            x_new = x_restart
            y = x_restart.copy()
            stall = 0
            stall_threshold = max(1000, stall_threshold * 0.9)
        
        y = x_new
        if verbose and t % 5000 == 0:
            J = 2.0 * n * best_m
            print(f"      FISTA t={t} ({elapsed:.0f}s/budget): J={J:.6f}, beta={beta:.1f}, m={best_m:.2e}")
    
    return best_x, best_m


def lp_polish(a0, time_limit, solver, initial_K=300, max_K=3000, verbose=True):
    """LP-based Frank-Wolfe polishing: minimize max conv subject to sum=1, a>=0."""
    n = len(a0)
    if n > 8000:
        return a0.copy(), fft_autocorr(a0).max()
    
    a = normalize(a0)
    best_a = a.copy()
    corr = fft_autocorr(a)
    best_m = corr.max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    # Initial active set: top K lags
    init_K = min(initial_K, 2 * n - 1)
    active_idx = np.argsort(corr)[::-1][:init_K]
    active_set = set(int(k) for k in active_idx)
    
    t = 0
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        corr_y = fft_autocorr(y)
        M_curr = corr_y.max()
        
        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
        
        # Update active set
        if len(active_set) < max_K:
            top_now = np.argsort(corr_y)[::-1][:max_K]
            for k in top_now:
                active_set.add(int(k))
        elif len(active_set) > max_K:
            constraint_list = sorted(active_set)
            margins = np.array([M_curr - corr_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
        
        constraint_list = sorted(active_set)
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        # Solve LP: min sum(g) s.t. sum(g)=1, coeffs@g <= M, g>=0
        # Actually: min M' s.t. coeffs@g <= M', sum(g)=1, g>=0
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            g_dir = a.copy()
        else:
            g_dir = g_val / g_sum
        
        # Frank-Wolfe step with Nesterov
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW t={t} ({elapsed:.0f}s): J={J:.6f}, |S|={K}")
    
    return best_a, best_m


def parametric_init(n, seed=0):
    """Generate diverse parametric initial sequences and return the best."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = np.inf
    
    candidates = []
    
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 2.5]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    
    # Beta-like (product of powers)
    for a_exp in [0.3, 0.5, 0.7, 1.0, 1.5]:
        for b_exp in [0.3, 0.5, 0.7, 1.0, 1.5]:
            candidates.append(np.maximum(i_arr + 1, 0.1) ** a_exp *
                              np.maximum(n - i_arr, 0.1) ** b_exp)
    
    # Gaussian
    for c in [0.25, 0.35, 0.5, 0.65, 0.75]:
        for sigma in [0.08, 0.12, 0.15, 0.2, 0.25, 0.35]:
            candidates.append(np.exp(-((x - c) ** 2) / (2 * sigma ** 2)))
    
    # Exponential
    for rate in [1, 2, 3, 5, 8]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1.0)))
    
    # Bump + tail
    for c in [0.3, 0.4, 0.5]:
        for w in [0.1, 0.15, 0.2, 0.3]:
            candidates.append(np.exp(-((x - c) ** 2) / (2 * 0.15 ** 2)) +
                              0.1 * np.exp(-rate * x))
    
    # Trigonometric
    for freq in [1, 2, 3, 5]:
        for phase in [0, 0.3, 0.7]:
            candidates.append(np.maximum(np.sin(np.pi * freq * (x + phase)) ** 2, 1e-6))
    
    # Asymmetric: large left, small right (or vice versa)
    for skew in [1.5, 2.0, 3.0, 5.0]:
        candidates.append(1.0 / (1.0 + skew * x))
        candidates.append(1.0 / (1.0 + skew * (1.0 - x)))
    
    for a_cand in candidates:
        a_cand = np.maximum(a_cand, 0.0)
        sc = a_cand.sum()
        if sc < 1e-15:
            continue
        a_norm = a_cand / sc
        m = fft_autocorr(a_norm).max()
        if m < best_val:
            best_val = m
            best_a = a_norm.copy()
    
    if best_a is None:
        best_a = np.ones(n) / n
        best_val = fft_autocorr(best_a).max()
    
    return best_a, best_val


def interpolate_resample(f, new_n):
    """Resample sequence to new length via linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def get_best_solver():
    """Find a working LP solver."""
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Using LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver fallback: ECOS")
    return cp.ECOS


def coordinate_perturb(a, rng, frac=0.1, scale=0.02):
    """Perturb a random subset of coordinates."""
    n = len(a)
    mask = rng.random(n) < frac
    pert = rng.normal(0, scale, n)
    pert[~mask] = 0.0
    a_new = a + pert * a  # relative perturbation
    return normalize(a_new)


def propose_candidate(budget_s=1800.0):
    """Main search function: finds best non-negative sequence minimizing J."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 40.0, 1760.0)
    solver = get_best_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = np.inf
    
    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = eval_safe(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={sc:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            p_score = eval_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    # Determine starting length
    if parent is not None:
        n_start = len(parent)
    else:
        n_start = 5000
        best_seq, _ = parametric_init(n_start)
        best_score = eval_safe(best_seq)
        print(f"Default init: n={n_start}, score={best_score:.6f}")
    
    rng = np.random.default_rng(123)
    
    # === Phase 1: Parametric search at n_start ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1_budget = min(remaining * 0.03, 30.0)
    print(f"\n=== Phase 1: Parametric init (n={n_start}, budget={t1_budget:.0f}s) ===")
    if t1_budget > 5:
        a_param, _ = parametric_init(n_start, seed=777)
        check_update(a_param, "parametric")
        # Also try at nearby lengths
        for delta_n in [500, -500]:
            new_n = n_start + delta_n
            if 2000 <= new_n <= 7000:
                a_p2, _ = parametric_init(new_n, seed=999)
                check_update(a_p2, f"param_n{new_n}")
    
    # === Phase 2: Main FISTA optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2_budget = min(remaining * 0.30, 450.0)
    print(f"\n=== Phase 2: FISTA main (n={n_best}, budget={t2_budget:.0f}s) ===")
    if t2_budget > 30:
        a_in = normalize(best_seq)
        a_out, _ = fista_optimize(a_in, t2_budget, beta_start=5.0, beta_end=80.0,
                                  eta0=0.8, use_top_k=True, top_k_frac=0.08,
                                  verbose=True, seed=42)
        check_update(a_out, "fista_main")
    
    # === Phase 3: FISTA with different params (higher beta range) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3_budget = min(remaining * 0.18, 250.0)
    print(f"\n=== Phase 3: FISTA round 2 (n={n_best}, budget={t3_budget:.0f}s) ===")
    if t3_budget > 20:
        a_in = normalize(best_seq)
        a_out, _ = fista_optimize(a_in, t3_budget, beta_start=15.0, beta_end=200.0,
                                  eta0=0.5, use_top_k=True, top_k_frac=0.04,
                                  verbose=True, seed=88)
        check_update(a_out, "fista_r2")
    
    # === Phase 4: LP-FW polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4_budget = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 4: LP-FW polish (n={n_best}, budget={t4_budget:.0f}s) ===")
    if t4_budget > 30 and n_best <= 7000:
        a_in = normalize(best_seq)
        a_out, _ = lp_polish(a_in, t4_budget, solver, initial_K=400, max_K=2000, verbose=True)
        check_update(a_out, "lp_polish")
    
    # === Phase 5: FISTA after LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5_budget = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 5: FISTA post-LP (n={n_best}, budget={t5_budget:.0f}s) ===")
    if t5_budget > 15:
        a_in = normalize(best_seq)
        a_out, _ = fista_optimize(a_in, t5_budget, beta_start=30.0, beta_end=300.0,
                                  eta0=0.3, use_top_k=True, top_k_frac=0.03,
                                  verbose=False, seed=55)
        check_update(a_out, "fista_post_lp")
    
    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")
    
    length_candidates = []
    for target_n in [4000, 5000, 6000, 7000, 8000, 9000]:
        if target_n != n_best and 3000 <= target_n <= 10000:
            length_candidates.append(target_n)
    # Sort by distance to current best (try closest first)
    length_candidates.sort(key=lambda x: abs(x - n_best))
    
    for new_n in length_candidates[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_interp = interpolate_resample(best_seq, new_n)
        
        t6a = min(remaining * 0.4, 150.0)
        if t6a > 20:
            a_out, _ = fista_optimize(a_interp, t6a, beta_start=8.0, beta_end=100.0,
                                      eta0=0.7, use_top_k=True, top_k_frac=0.06,
                                      verbose=False, seed=new_n % 1000)
            check_update(a_out, f"fista_n{new_n}")
    
    # === Phase 7: LP-FW on best found so far ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW (n={n_best}, remaining={remaining:.0f}s) ===")
    if remaining > 40 and n_best <= 7000:
        t7 = min(remaining * 0.4, 180.0)
        if t7 > 20:
            a_in = normalize(best_seq)
            a_out, _ = lp_polish(a_in, t7, solver, initial_K=500, max_K=2500, verbose=True)
            check_update(a_out, "lp_final")
    
    # === Phase 8: Final FISTA polish with very high beta ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t8 = min(remaining * 0.5, 80.0)
        if t8 > 10:
            a_in = normalize(best_seq)
            a_out, _ = fista_optimize(a_in, t8, beta_start=50.0, beta_end=500.0,
                                      eta0=0.2, use_top_k=True, top_k_frac=0.02,
                                      verbose=False, seed=7777)
            check_update(a_out, "fista_very_sharp")
    
    # === Phase 9: Random restarts from best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 9: Random restarts (n={n_best}, remaining={remaining:.0f}s) ===")
        t9_total = min(remaining * 0.5, 150.0)
        t9_each = t9_total / 3
        for restart_i in range(3):
            remaining_chk = time_budget - (time.monotonic() - t_global)
            if remaining_chk < 20:
                break
            a_pert = coordinate_perturb(normalize(best_seq), rng, frac=0.15, scale=0.05)
            a_pert = coordinate_perturb(a_pert, rng, frac=0.1, scale=0.03)
            t9 = min(remaining_chk * 0.3, 50.0)
            if t9 > 10:
                a_out, _ = fista_optimize(a_pert, t9, beta_start=10.0, beta_end=100.0,
                                          eta0=0.6, use_top_k=True, top_k_frac=0.05,
                                          verbose=False, seed=1000 + restart_i)
                check_update(a_out, f"restart_{restart_i}")
    
    # Final validation
    if best_seq is not None:
        final_score = eval_safe(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: J={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 5000
        best_seq, _ = parametric_init(n_fb)
        best_score = eval_safe(best_seq)
        print(f"Fallback: J={best_score:.6f}")
    
    return [float(x) for x in best_seq]
