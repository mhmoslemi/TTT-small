def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize so a >= 0 and sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project z onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate sequence f to new length new_n, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(y, beta, top_k=None):
    """Compute gradient of smoothed max autocorrelation w.r.t. y."""
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    # Gradient: sum_k p_k * d(autocorr[k])/d(y)
    # d(autocorr[k])/d(y[i]) = y[k-i] + y[i-k] (with boundary)
    # Which is: (y * p_reversed) convolution
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, p, b


def nesterov_bt(a0, time_limit, beta_start=20.0, beta_end=800.0,

    rng = np.random.default_rng(123)
    top_k = max(30, int(n * top_k_ratio))
    eta = 1.0  # step size for backtracking
    
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        # Compute gradient
        grad, p, b = compute_smoothed_grad(y, beta, top_k)
        
        # Backtracking line search
        f_curr = fast_autocorr(y).max()
        improved = False
        for ls in range(10):
            z = y - eta * grad
            x_new = simplex_project(z)
            f_new = fast_autocorr(x_new).max()
            if f_new < f_curr - 0.01 * eta * np.sum(grad * (y - x_new)) + 1e-12:

                break
            eta *= 0.5
            if eta < 1e-6:
                break
        
        if not improved:
            x_new = simplex_project(y - eta * grad)

        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:

            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.5, 0.5)
        else:
            stall += 1
            
        x_prev = x
        x = x_new
        t_nest = t_new
        
        # Adaptive beta
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)

            x = x_r

def spectral_shaping(a0, time_limit, n_iter_max=2000, verbose=True):

    """

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1 + 1).bit_length()
    stall = 0
    rng = np.random.default_rng(777)
    
    for it in range(1, n_iter_max + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute spectrum
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        
        # Compute autocorrelation
        ac = fast_autocorr(a)
        peak_val = ac.max()
        peak_idx = ac.argmax()
        
        # Strategy: reduce the power at frequencies that contribute to the peak
        # The contribution of frequency f to autocorrelation[k] is |A(f)|^2 * exp(2*pi*i*f*k/n)
        # To reduce autocorrelation[peak_idx], we want to reduce |A(f)|^2 where
        # exp(2*pi*i*f*peak_idx/n) is in phase with the peak
        
        # Simpler approach: apply a "flatness" pressure
        # Reduce peaks in P, boost valleys
        P_mean = P.mean()
        P_max = P.max()
        
        if P_max > P_mean * 1.01:
            # Compress the power spectrum
            alpha = 0.1 + 0.3 * min(1.0, it / 200.0)  # gradually increase pressure
            P_flat = P_mean + (P - P_mean) * (1.0 - alpha)
            P_flat = np.maximum(P_flat, P.min() * 0.5)
            
            # New amplitude with same phase
            A_new = np.sqrt(P_flat) * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n].real
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                m_new = fast_autocorr(a_new).max()
                if m_new < best_m - 1e-12:
                    best_m = m_new
                    best_a = a_new.copy()
                    stall = 0
                    a = a_new
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1
        
        # Periodic restart with time-domain noise
        if stall > 300:
            stall = 0
            noise = 0.02
            a_r = best_a + rng.normal(0, noise, n) * best_a
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            a = a_r
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral it=" + str(it) + " (" + str(elapsed) + "s): J=" + repr(J))
    
    return best_a, best_m


def spectral_shaping_v2(a0, time_limit, verbose=True):
    """
    Improved spectral shaping: directly targets the autocorrelation peak
    by modifying the power spectrum at frequencies that contribute most to it.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n).bit_length()
    stall = 0
    rng = np.random.default_rng(888)
    
    freqs = np.fft.rfftfreq(fft_size) * fft_size  # in units of n
    
    for it in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        phase = np.angle(A)
        
        ac = fast_autocorr(a)
        peak_val = ac.max()
        
        # Find top K peaks
        K_peaks = min(5, len(ac))
        top_peaks = np.argsort(ac)[::-1][:K_peaks]
        
        # For each peak at lag k, the frequency contribution is P[f] * cos(2*pi*f*k/n)
        # We want to reduce P[f] where it contributes positively to the peaks
        reduction = np.zeros_like(P)
        for k in top_peaks:
            k_int = int(k)
            # Contribution of freq f to autocorr[k] is proportional to P[f]*cos(2*pi*f*k_int/n)
            contrib = P * np.cos(2.0 * np.pi * freqs * k_int / fft_size)
            # Only reduce where contribution is positive
            reduction += np.maximum(contrib, 0)
        
        # Normalize reduction
        red_max = reduction.max()
        if red_max > 1e-15:
            reduction = reduction / red_max
        
        # Apply reduction
        alpha = 0.05 + 0.15 * min(1.0, it / 500.0)
        P_new = P * (1.0 - alpha * reduction)
        P_new = np.maximum(P_new, P.min() * 0.1)
        
        # Preserve DC component (sum constraint)
        P_new[0] = P[0]
        
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n].real
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
            m_new = fast_autocorr(a_new).max()
            if m_new < best_m - 1e-14:
                best_m = m_new
                best_a = a_new.copy()
                a = a_new.copy()
                stall = 0
            else:
                stall += 1
                # Try smaller step
                if stall > 50:
                    alpha *= 0.5
        else:
            stall += 1
        
        if stall > 500:
            stall = 0
            a_r = best_a + rng.normal(0, 0.015, n) * best_a
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            a = a_r
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral2 it=" + str(it) + " (" + str(elapsed) + "s): J=" + repr(J))
    
    return best_a, best_m


def lp_fw_v2(a0, time_limit, solver, initial_K=300, max_K=2000,
             add_per_iter=50, verbose=True):
    """
    Improved LP-Frank-Wolfe with:
    - Lazy catalog (only evaluate a subset of constraints each iteration)
    - Better line search (BB step + backtracking)
    - Adaptive M (slightly below current max to force progress)
    """
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    # Lazy catalog: maintain a set of constraint indices
    ac = fast_autocorr(y)
    catalog = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])
    
    no_improve = 0
    max_no_improve = 800
    prev_g = None
    bb_step = 2.0 / n  # Barzilai-Borwein step
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update catalog: check random subset for violations
        all_lags = np.arange(len(ac_y))
        sample_size = min(200, len(all_lags))
        if t % 10 == 0:
            # Periodically check all lags (expensive but thorough)
            violations = np.where(ac_y > M_curr + 1e-7)[0]  # should be empty
            # Actually check which lags have high autocorr of y with potential g
            # For now, just ensure catalog covers top lags
            top_lags = np.argsort(ac_y)[::-1][:min(500, len(ac_y))]
            for k in top_lags[:200]:
                catalog.add(int(k))
        elif t % 5 == 0:
            # Add a few new candidate lags
            candidates = rng_choice = np.random.choice(len(ac_y), size=sample_size, replace=False)
            for k in candidates:
                if ac_y[k] > M_curr * 0.7:  # Only add "active" lags
                    catalog.add(int(k))
        
        # Prune catalog if too large
        if len(catalog) > max_K:
            margins = np.array([M_curr - ac_y[k] for k in catalog])
            keep = np.argsort(margins)[:max_K]
            catalog = set(int(list(catalog)[i]) for i in keep)
        
        if len(catalog) < 5:
            continue
        
        constraint_list = sorted(catalog)
        K = len(constraint_list)
        
        # Build constraint matrix: coeffs[j, i] = y[k_j - i] for valid i
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        # Solve LP: minimize M s.t. coeffs @ g <= M, sum(g) = 1, g >= 0
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Prune catalog and retry
            if len(catalog) > initial_K:
                catalog = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Line search with BB step
        if prev_g is not None:
            s_vec = y - prev_g
            y_vec = g_dir - prev_g
            ss = np.sum(s_vec * s_vec)
            sy = np.sum(s_vec * y_vec)
            if ss > 1e-15:
                bb_step = max(0.01, min(1.0, sy / ss))
        
        alpha_candidates = [bb_step, 2.0 / (t + 2.0), 1.0 / (t + 1), 0.05, 0.1, 0.2]
        alpha_candidates = list(set(max(0.01, min(1.0, a)) for a in alpha_candidates))
        
        best_alpha = bb_step
        best_trial_m = float('inf')
        for alpha in sorted(alpha_candidates, reverse=True):
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_m:
                best_trial_m = M_trial
                best_alpha = alpha
        
        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        prev_g = y.copy()
        y = y_new
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW2 iter=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J) + " |C|=" + str(len(catalog)))
    
    return best_a, best_m


def block_coord_descent(a0, time_limit, block_size=50, verbose=True):
    """
    Block coordinate descent: moves mass from peak-contributing blocks
    to low-pressure blocks, using quadratic approximation for step size.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(444)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 2000:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:min(20, len(ac))]
        improved = False
        
        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            # Find the top contributing pair
            top_pairs = np.argsort(contrib)[::-1][:block_size]
            
            for pi in top_pairs[:10]:
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(idx[pi])
                j = int(k_int - i)
                if j < 0 or j >= n:
                    continue
                
                # Move mass from the larger of a[i], a[j] to a random low-pressure position
                if a[i] >= a[j]:
                    src = i
                else:
                    src = j
                
                if a[src] < 1e-10:
                    continue
                
                # Find low-pressure destination
                # Pressure at position p = sum over top peaks of contribution to that peak
                pressure = np.zeros(n)
                for k2 in top_k[:10]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        pressure[idx2] += a[k2i - idx2]
                        pressure[k2i - idx2] += a[idx2]
                
                # Try a few random destinations with low pressure
                n_tries = 5
                for _ in range(n_tries):
                    # Sample from low-pressure regions
                    low_press = np.argsort(pressure)[:n // 10]
                    dst = int(low_press[rng.integers(len(low_press))])
                    if dst == src:
                        continue
                    
                    # Try different step sizes
                    for frac in [0.02, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-14:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-14:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                
                if improved:
                    break
            
            if improved:
                stall = 0
                break
        
        if not improved:
            stall += 1
            # Random exploration when stalled
            if stall % 50 == 0:
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                improved = True
        
        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    BlockCD (" + str(elapsed) + "s): J=" + repr(J) + " stall=" + str(stall))
    
    return best_a, best_m


def diverse_init(n, time_limit, current_best, current_score):
    """Generate diverse initial candidates."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = current_best.copy()
    best_val = current_score
    rng = np.random.default_rng(2024)
    
    candidates = []
    
    # Parametric families
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
        candidates.append((i_arr + 1.0) ** p * (n - i_arr + 1.0) ** 0.5)
    
    # Symmetric
    mid = (n - 1) / 2.0
    for sigma in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
        candidates.append(np.exp(-((i_arr - mid) ** 2) / (2.0 * (sigma * n) ** 2)))
    
    # Modulated
    for freq in [1, 2, 3, 5]:
        for amp in [0.3, 0.5, 0.7]:
            candidates.append(1.0 + amp * np.sin(2.0 * np.pi * freq * x))
            candidates.append(1.0 + amp * np.cos(2.0 * np.pi * freq * x))
    
    # Product of sines
    for f1 in [1, 2, 3]:
        for f2 in [3, 5, 7]:
            candidates.append(np.abs(np.sin(np.pi * f1 * x)) * np.abs(np.sin(np.pi * f2 * x)))
    
    # Random smooth
    for _ in range(100):
        n_knots = rng.integers(3, 20)
        knots = np.sort(rng.random(n_knots))
        heights = rng.random(n_knots) ** rng.uniform(0.5, 4.0)
        candidates.append(np.interp(x, knots, heights))
    
    # Random with structure
    for _ in range(100):
        a_rand = rng.random(n)
        # Smooth it
        kernel_size = rng.integers(3, 20)
        kernel = np.ones(kernel_size) / kernel_size
        a_smooth = np.convolve(a_rand, kernel, mode='same')
        candidates.append(a_smooth)
    
    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a_raw = np.asarray(a_raw, dtype=np.float64)
        if a_raw.shape != (n,):
            continue
        a = norm_sum1(a_raw)
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()
    
    return best_a, best_val


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(elapsed) + "s)")
    return best_seq, best_score

def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize so a >= 0 and sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project z onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate sequence f to new length new_n, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(y, beta, top_k=None):
    """Compute gradient of smoothed max autocorrelation w.r.t. y."""
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, p, b

                    continue

                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                improved = True
                                stall = 0

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    BlockCD (" + str(elapsed) + "s): J=" + repr(J) + " stall=" + str(stall))

    return best_a, best_m


def diverse_init(n, time_limit, current_best, current_score):
    """Generate diverse initial candidates to find a good starting point."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = current_best.copy()
    best_val = current_score
    rng = np.random.default_rng(2024)

    candidates = []

    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)

    mid = (n - 1) / 2.0
    for sigma in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
        candidates.append(np.exp(-((i_arr - mid) ** 2) / (2.0 * (sigma * n) ** 2)))

    for freq in [1, 2, 3, 5, 7]:
        for amp in [0.3, 0.5, 0.7]:
            candidates.append(1.0 + amp * np.sin(2.0 * np.pi * freq * x))

    for f1 in [1, 2, 3]:
        for f2 in [3, 5, 7]:
            candidates.append(np.abs(np.sin(np.pi * f1 * x)) * np.abs(np.sin(np.pi * f2 * x)))

    for _ in range(80):
        n_knots = rng.integers(3, 25)
        knots = np.sort(rng.random(n_knots))
        heights = rng.random(n_knots) ** rng.uniform(0.5, 4.0)
        candidates.append(np.interp(x, knots, heights))

    for _ in range(80):
        a_rand = rng.random(n)
        kernel_size = rng.integers(3, 25)
        kernel = np.ones(kernel_size) / kernel_size
        a_smooth = np.convolve(a_rand, kernel, mode='same')
        candidates.append(a_smooth)

    # Walsh-Hadamard inspired
    for _ in range(30):
        a_w = np.abs(np.random.source_rng() if False else rng.standard_normal(n))
        # Apply a random low-pass filter
        A_f = np.fft.rfft(a_w)
        cutoff = rng.integers(3, n // 4)
        A_f[cutoff:] = 0
        a_filt = np.fft.irfft(A_f, n)
        candidates.append(np.abs(a_filt))

    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a_raw = np.asarray(a_raw, dtype=np.float64)
        if a_raw.shape != (n,):
            continue
        a = norm_sum1(a_raw)
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    return best_a, best_val


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(elapsed) + "s)")
    return best_seq, best_score

    t_global = time.monotonic()

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):

                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: Diverse initialization
    print("\n=== Phase 1: Diverse Init (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 40.0)
    if t1 > 10:
        a_param, _ = diverse_init(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "diverse", t_global)

    # Phase 2: Nesterov with backtracking
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov-BT (n=" + str(n_best) + ") ===")
