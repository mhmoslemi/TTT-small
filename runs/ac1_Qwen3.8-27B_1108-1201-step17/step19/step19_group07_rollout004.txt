```python
def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}"""
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, s=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - s)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_grad(a, beta, n_active=50):
    """Compute smoothed gradient of logsumexp(beta * autocorr) / beta"""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    
    # Softmax weights on the autocorrelation lags
    c_max = c.max()
    # Only use top lags for efficiency and focus
    if n_active < size:
        # Use top k
        top_k = np.argsort(c)[::-1][:n_active]
        c_work = np.zeros(size)
        c_work[top_k] = c[top_k]
        c_work_max = c_work[top_k].max()
        exp_c = np.exp(beta * (c_work - c_work_max))
        exp_c[c_work == 0] = np.exp(-beta * c_work_max)  # small weight for others
        w = exp_c / exp_c.sum()
    else:
        w = np.exp(beta * (c - c_max))
        w = w / w.sum()
    
    # Gradient: 2 * (w * a)

    # This is a convolution (with flipped a)

    W = np.fft.rfft(w, fft_size)
    Af = np.fft.rfft(a_flip, fft_size)
    grad = 2.0 * np.fft.irfft(W * Af, fft_size)[:n]
    
    return grad, c
```

2. Solve the QP: min max(c_k for k in S) subject to a >= 0, sum(a) = 1
3. Check if any non-active lag violates the max
4. If yes, add it to S and goto 2
5. If no, we have the optimal solution for the active set

Step 2 is a QP (quadratic objective, linear constraints). For |S| active constraints, the QP is:
min M
subject to sum_i a_i * a_{k-i} <= M for k in S  (these are QUADRATIC, not linear!)
a >= 0, sum(a) = 1

This is a QP with quadratic constraints, which CVXPY might handle (as a SOCP or general QP). But it's nonconvex (quadratic constraints).

sum_i (a_i^0 * a_{k-i} + a_i * a_{k-i}^0) <= M + current_c_k - M
which simplifies to 2*sum_i a_i * a_{k-i}^0 <= M (approximately)

This is back to the LP formulation. So the Frank-Wolfe LP is essentially a linearized active-set method.

```python
"""

1. Nesterov-accelerated projected gradient with simplex projection (FFT-accelerated)

import numpy as np

import cvxpy as cp


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s} (O(n log n) water-filling)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def softmax_gradient(a, beta, n_active=50):
    """Compute gradient of (1/beta)*logsumexp(beta*autocorr(a)) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]

    c_max = c.max()
    if n_active < size and n_active > 0:
        top_idx = np.argsort(c)[::-1][:n_active]
        c_w = np.full(size, c_min_val(c, c_max))
        c_w[top_idx] = c[top_idx]
    else:
        c_w = c.copy()

    c_w_max = c_w.max()
    exp_c = np.exp(beta * (c_w - c_w_max))
    exp_c = np.maximum(exp_c, 1e-300)
    w = exp_c / exp_c.sum()

    # grad[j] = 2 * sum_k w[k] * a[k-j] = 2 * (w * a_flip)[j]
    a_flip = a[::-1]
    W = np.fft.rfft(w, fft_size)
    Af = np.fft.rfft(a_flip, fft_size)
    grad = 2.0 * np.fft.irfft(W * Af, fft_size)[:n]

    return grad, c


def c_min_val(c, c_max):
    return c.min()


def fista_simplex(a0, time_limit, beta_start=5.0, beta_end=120.0,
                  n_active=40, eta0=1.0, verbose=True):
    """Nesterov-accelerated projected gradient with simplex projection."""
    n = len(a0)
    x = project_simplex(a0, 1.0)
    y = x.copy()
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    t = 1.0
    t_fista = 1.0
    eta = eta0
    stall = 0
    stall_limit = 500

    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        progress = min(iter_count / max(time_limit * 80.0, 1.0), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        grad, c = softmax_gradient(y, beta, n_active)
        M_curr = c.max()

        # Adaptive step size (backtracking)
        alpha = 1.0
        f_y = M_curr
        for _ in range(30):
            x_try = project_simplex(y - alpha * grad, 1.0)
            f_try = fast_autocorr(x_try).max()
            if f_try <= f_y - 1e-14 or alpha < 1e-8:
                break
            alpha *= 0.7

        x_new = project_simplex(y - alpha * grad, 1.0)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista))
        momentum = (t_fista - 1.0) / t_new if t_fista > 1.0 else 0.0
        z = x_new + momentum * (x_new - y)
        z = project_simplex(z, 1.0)

        # Evaluate
        m_new = fast_autocorr(z).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_x = z.copy()
            stall = 0
        else:
            stall += 1

        # Periodic restart to escape plateaus
        if stall > stall_limit:
            rng = np.random.default_rng(iter_count * 7 + 13)
            pert_std = 0.05 + 0.03 * (stall / stall_limit)
            pert = rng.normal(0, pert_std, n) * best_x
            x_restart = project_simplex(best_x + pert, 1.0)
            z = x_restart
            best_m = fast_autocorr(z).max()
            best_x = z.copy()
            stall = 0
            stall_limit = max(200, int(stall_limit * 0.85))

        y = z
        t_fista = t_new
        iter_count += 1

        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA-PG iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, alpha={alpha:.3f}, stall={stall}")

    return best_x, best_m


def lp_frank_wolfe(a0, time_limit, solver, initial_K=200, max_K=3000, verbose=True):
    """LP-based Frank-Wolfe with adaptive constraint set."""
    n = len(a0)
    y = normalize1(a0)
    x = y.copy()
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    init_K = min(initial_K, 2 * n - 1)
    ac_y = fast_autocorr(y)
    constraint_set = set(int(k) for k in np.argsort(ac_y)[::-1][:init_K])

    t_iter = 1
    t_nest = 1.0

    while time.monotonic() - t0 < time_limit and t_iter < 200000:
        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-14:
            best_m = ac_x.max()
            best_a = x.copy()

        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix: row k has (y * ones)[k-i] at position i
        # (y_dir)[i] coefficient for lag k: sum_i y_{k-i} * g_i
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            constraint_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])
            t_iter += 1
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            break
        g_dir = g_val / g_sum

        # Check for violated constraints (add new ones)
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-6)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                constraint_set.add(int(k))

        # Nesterov step
        alpha = 2.0 / (t_nest + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        momentum = (t_nest - 1.0) / (t_nest + 2.0) if t_nest > 1.0 else 0.0
        x_new = y_new + momentum * (y_new - y)
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y_new.copy()
        else:
            x_new = x_new / sx

        y = y_new
        x = x_new
        t_nest += 1.0
        t_iter += 1

        if verbose and t_iter % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t_iter} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    return best_a, best_m


def parametric_init(n, best_ref=None):
    """Try diverse parametric families."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')
    candidates = []

    for p in [0.3, 0.5, 0.7, 0.8, 1.0, 1.2, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr + 1.0) ** p)

    for alpha in [0.5, 0.8, 1.0, 1.2]:
        for beta_p in [0.5, 0.8, 1.0, 1.2]:
            candidates.append(np.maximum(i_arr + 1.0, 0.5) ** alpha *
                              np.maximum(n - i_arr, 0.5) ** beta_p)

    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))

    for rate in [0.5, 1, 2, 3, 5]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1.0)))

    for freq in [1, 2, 3, 5]:
        candidates.append(np.maximum(np.sin(np.pi * freq * x) ** 2, 0.005))

    # Raised cosine variants
    candidates.append(0.5 * (1 + np.cos(2 * np.pi * x)))
    candidates.append(0.5 * (1 + np.cos(np.pi * x)))

    if best_ref is not None and len(best_ref) == n:
        ref_norm = normalize1(best_ref)
        for w in [0.3, 0.5, 0.7]:
            candidates.append(w * ref_norm + (1 - w) * normalize1((i_arr + 1.0) ** 0.7))

    for a_cand in candidates:
        a_norm = project_simplex(a_cand, 1.0)
        ac = fast_autocorr(a_norm)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_a = a_norm

    return best_a, best_val


def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(30, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize1(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    return project_simplex(f_interp, 1.0)


def random_perturb(a, std=0.03, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    pert = rng.normal(0, std, n)
    return project_simplex(a + pert * a, 1.0)

            a_out, _ = fista_simplex(a_in, t9, beta_start=50.0, beta_end=300.0,
                                     n_active=20, eta0=0.4, verbose=False)
            check_and_update(a_out, "fista_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = project_simplex(0.5 * (1 + np.cos(np.pi * x)), 1.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
```
