def next_pow2(m):
    m = int(max(1, m))

def normalize_vec(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:

    if a.size == 0:

        return float('inf')
    q = self_conv(a)
    return 2.0 * float(a.size) * float(np.max(q))


def evaluate_safe(a):
    try:
        x = [float(v) for v in np.asarray(a, dtype=np.float64)]
        val = evaluate_sequence(x)
        if np.isfinite(val):
            return float(val)
    except Exception:
        pass
    return float('inf')


def quad_max_alpha(alpha, q0, c, qd):
    a1 = 1.0 - alpha
    val = (a1 * a1) * q0 + (2.0 * alpha * a1) * c + (alpha * alpha) * qd
    return float(np.max(val))


def golden_alpha_max(q0, c, qd, a, b, iters=38):
    phi = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = b - phi * (b - a)
    d1 = a + phi * (b - a)
    f1 = quad_max_alpha(c1, q0, c, qd)
    f2 = quad_max_alpha(d1, q0, c, qd)
    for _ in range(iters):
        if f1 < f2:
            b = d1
            d1 = c1
            f2 = f1
            c1 = b - phi * (b - a)
            f1 = quad_max_alpha(c1, q0, c, qd)
        else:
            a = c1
            c1 = d1
            f1 = f2
            d1 = a + phi * (b - a)
            f2 = quad_max_alpha(d1, q0, c, qd)
    return 0.5 * (a + b)


def line_search_direction(a, d, q0, T=None):
    a = np.asarray(a, dtype=np.float64)
    d = np.asarray(d, dtype=np.float64)
    q0 = np.asarray(q0, dtype=np.float64)
    M0 = float(np.max(q0))
    dot = float(np.dot(a, d))
    if dot > 1.0 - 1e-7:
        return a.copy(), M0
    c = conv_ab(a, d)
    qd = self_conv(d)
    best_alpha = 0.0
    best_M = M0
    alphas = [1e-4, 5e-4, 1e-3, 2e-3, 5e-3, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    if T is not None and np.isfinite(T) and T < 0.0 and M0 > 1e-15:
        t1 = min(0.35, max(5e-4, 0.6 * (-float(T)) / M0))
        alphas.append(max(1e-5, 0.4 * t1))
        alphas.append(min(0.95, 1.2 * t1))
    uniq = []
    seen = set()
    for al in alphas:
        al = float(min(1.0, max(0.0, al)))
        key = round(al, 12)
        if key not in seen:
            seen.add(key)
            uniq.append(al)
    for al in uniq:
        val = quad_max_alpha(al, q0, c, qd)
        if val < best_M:
            best_M = val
            best_alpha = al
    if best_M < M0 - 1e-13 and 0.0 < best_alpha < 1.0:
        al2 = golden_alpha_max(q0, c, qd, 0.0, 1.0, 36)
        val2 = quad_max_alpha(al2, q0, c, qd)
        if val2 < best_M:
            best_M = val2
            best_alpha = al2
    if best_alpha <= 0.0 or best_M >= M0 - 1e-14:
        return a.copy(), M0
    new = (1.0 - best_alpha) * a + best_alpha * d
    new = np.maximum(new, 0.0)
    s = float(new.sum())
    if s <= 1e-15:
        return a.copy(), M0
    new = new / s
    return new, best_M


def solve_lp_direction(y, active, q_active, n_max_vars=8000):
    if not HAVE_LP:
        return None, None
    y = np.asarray(y, dtype=np.float64)
    active = np.asarray(active, dtype=int)
    q_active = np.asarray(q_active, dtype=np.float64)
    n = y.size
    K = active.size
    if K <= 0 or n > n_max_vars:
        return None, None
    if K * (n + 1) > 8_000_000:
        return None, None
    try:
        A = np.zeros((K, n + 1), dtype=np.float64)
    except Exception:
        return None, None
    yy = 2.0 * y
    for r, k in enumerate(active):
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            A[r, i0:i1] = yy[start:stop][::-1]
    A[:, n] = -1.0
    b = np.maximum(q_active, 0.0)
    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0
    Aeq = np.zeros((1, n + 1), dtype=np.float64)
    Aeq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)
    bounds = [(0.0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 120})
    except Exception:
        return None, None
    status = getattr(res, 'status', None)
    success = getattr(res, 'success', False)
    x = getattr(res, 'x', None)
    if (status != 0 or not success) or x is None:
        return None, None
    g = np.asarray(x[:n], dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-10:
        return None, None
    g = g / s
    T = float(x[n])
    if not np.isfinite(T):
        T = 0.0
    return g, T


def subgradient_at(y, q, active, weights=None):
    """Compute subgradient of max_k q[k] at point y using active set."""
    y = np.asarray(y, dtype=np.float64)
    n = y.size
    active = np.asarray(active, dtype=int)
    if weights is None:
        weights = np.ones(active.size, dtype=np.float64)
        sw = float(weights.sum())
        if sw <= 0:
            weights = np.ones(active.size, dtype=np.float64) / max(active.size, 1)
            sw = 1.0
        else:
            weights = weights / sw
    v = np.zeros(n, dtype=np.float64)
    for idx, k in enumerate(active):
        k = int(k)
        w = float(weights[idx])
        if w <= 0:
            continue
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            start = k - (i1 - 1)
            stop = k - i0 + 1
            v[i0:i1] += w * 2.0 * y[start:stop][::-1]
    return v


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x) = target_sum, x >= 0}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.zeros(n, dtype=np.float64)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s > 1e-15:
        out = out * (target_sum / s)
    return out


def nesterov_subgradient_step(a, v, momentum):
    """Step: a_new = project(a - step_size * v + momentum correction)."""
    n = a.size
    grad = v.copy()
    grad -= np.mean(grad)
    norm_g = float(np.linalg.norm(grad))
    if norm_g < 1e-15:
        return a.copy()
    grad = grad / norm_g
    step = 0.5 / (1.0 + 0.1 * np.sqrt(n))
    candidate = a - step * grad
    candidate = np.maximum(candidate, 0.0)
    s = float(candidate.sum())
    if s <= 1e-15:
        return a.copy()
    candidate = candidate / s
    return candidate


def lp_fw_search(a0, time_limit, verbose=True, K_start=200, K_max=800):
    """LP-based Frank-Wolfe search."""
    if time_limit <= 0:
        return normalize_vec(a0)
    t0 = time.monotonic()
    y = normalize_vec(a0)
    n = y.size
    if n <= 5:
        return y
    best = y.copy()
    best_score = objective_norm(y)
    K = K_start
    use_lp = HAVE_LP
    lp_fails = 0
    no_improve = 0
    last_print = t0
    it = 0
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining <= 1.0:
            break
        qy = self_conv(y)
        M_y = float(np.max(qy))
        score_y = 2.0 * n * M_y
        if score_y < best_score - 1e-12:
            best = y.copy()
            best_score = score_y
        if no_improve > 300 and time.monotonic() - t0 > 0.85 * time_limit:
            break
        Kcur = int(min(K, len(qy), 1200))
        order = np.argsort(qy)[::-1]
        act = order[:Kcur]
        q_active = qy[act]
        g_dir = None
        T = None
        if use_lp and Kcur >= 15 and remaining > 10:
            tstep = time.monotonic()
            g_dir, T = solve_lp_direction(y, act, q_active, n)
            spend = time.monotonic() - tstep
            if g_dir is None:
                lp_fails += 1
                if lp_fails >= 5:
                    use_lp = False
            else:
                lp_fails = 0
                if spend > 100.0:
                    use_lp = False
                elif spend > max(20.0, 0.15 * remaining):
                    K = max(100, K - 30)
        if g_dir is None:
            # Fallback: subgradient-based direction
            M_val = float(np.max(qy))
            thresh = M_val * 0.95
            act_sg = np.where(qy >= thresh)[0]
            if act_sg.size == 0:
                act_sg = order[:50]
            wts = qy[act_sg]
            wts = np.maximum(wts, 1e-10)
            wts = wts / float(wts.sum())
            v = subgradient_at(y, qy, act_sg, wts)
            v -= np.mean(v)
            j = int(np.argmin(v))
            g_dir = np.zeros(n, dtype=np.float64)
            g_dir[j] = 1.0
            T = None
        new, Mnew = line_search_direction(y, g_dir, qy, T)
        if Mnew < M_y - 1e-13:
            y = new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve in (1, 3, 8, 20):
                # Try subgradient direction
                M_val = float(np.max(qy))
                thresh = M_val * 0.92
                act_sg = np.where(qy >= thresh)[0]
                if act_sg.size == 0:
                    act_sg = order[:Kcur]
                wts = (qy[act_sg] - M_val + M_val * 0.05)
                wts = np.maximum(wts, 1e-10)
                wts = wts / float(wts.sum())
                v = subgradient_at(y, qy, act_sg, wts)
                v -= np.mean(v)
                j = int(np.argmin(v))
                g_dir2 = np.zeros(n, dtype=np.float64)
                g_dir2[j] = 1.0
                new2, M2 = line_search_direction(y, g_dir2, qy, None)
                if M2 < M_y - 1e-13:
                    y = new2
                    no_improve = 0
            if no_improve > 0 and no_improve % 20 == 0:
                K = min(K + 30, K_max)
            if no_improve > 250:
                break
        it += 1
        if verbose and it % 20 == 0:
            print("    LP-FW: it=%d t=%.1fs best=%.9f cur=%.9f K=%d lp=%s imp=%d"
                  % (it, time.monotonic() - t0, best_score, score_y, K, use_lp, no_improve))
    return best


def nesterov_sg_search(a0, time_limit, verbose=True):
    """Nesterov-accelerated subgradient method."""
    if time_limit <= 0:
        return normalize_vec(a0)
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 5:
        return a
    best = a.copy()
    best_score = objective_norm(a)
    y_nest = a.copy()
    momentum = 0.0
    step_decay = 1.0
    it = 0
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        if it == 0:
            y_nest = a.copy()
        qy = self_conv(y_nest)
        M_y = float(np.max(qy))
        score_y = 2.0 * n * M_y
        if score_y < best_score - 1e-12:
            best = y_nest.copy()
            best_score = score_y
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200 and time.monotonic() - t0 > 0.8 * time_limit:
            break
        if no_improve > 50:
            step_decay *= 0.95
        thresh = M_y * 0.90
        act = np.where(qy >= thresh)[0]
        if act.size == 0:
            act = np.argsort(qy)[::-1][:80]
        wts = qy[act]
        wts = np.maximum(wts, 1e-10)
        wts = wts ** 2
        sw = float(wts.sum())
        if sw > 0:
            wts = wts / sw
        else:
            wts = np.ones(act.size) / act.size
        v = subgradient_at(y_nest, qy, act, wts)
        v -= np.mean(v)
        norm_v = float(np.linalg.norm(v))
        if norm_v < 1e-15:
            break
        step = step_decay * 0.3 / (1.0 + 0.05 * np.sqrt(n))
        new_mom = (it + 1.0) / (it + 1.0 + 3.0)
        candidate = y_nest - step * v / max(norm_v, 1e-10)
        candidate = np.maximum(candidate, 0.0)
        sc = float(candidate.sum())
        if sc <= 1e-15:
            candidate = a.copy()
            sc = 1.0
        candidate = candidate / sc
        a_new = (1.0 - new_mom) * candidate + new_mom * (candidate - (y_nest - a))
        a_new = np.maximum(a_new, 0.0)
        sc2 = float(a_new.sum())
        if sc2 <= 1e-15:
            a_new = candidate
            sc2 = 1.0
        a_new = a_new / sc2
        a = a_new
        y_nest = candidate
        it += 1
        if verbose and it % 100 == 0:
            print("    NestSG: it=%d t=%.1fs best=%.9f step=%.4f noimp=%d"
                  % (it, time.monotonic() - t0, best_score, step_decay, no_improve))
    return best


def smooth_polish(a0, time_limit, rng_seed=42, verbose=False):
    """Mirror descent polish."""
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 1:
        return a
    best = a.copy()
    best_score = objective_norm(a)
    trust = 0.4
    focus = 3.0
    fails = 0
    rng = np.random.default_rng(rng_seed)
    while time.monotonic() - t0 < time_limit:
        q = self_conv(a)
        M = float(np.max(q))
        score = 2.0 * n * M
        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            fails = 0
        else:
            fails += 1
        if fails > 500:
            break
        if fails > 30 and fails % 20 == 0:
            focus = min(focus * 1.1, 50.0)
            trust = max(trust * 0.82, 0.02)
        if fails > 100 and fails % 100 == 0:
            scale_noise = 0.015 / (1.0 + 0.05 * (fails // 100))
            cand = best + scale_noise * rng.normal(size=best.size) * best
            cand = np.maximum(cand, 1e-12)
            cand = normalize_vec(cand)
            sc = objective_norm(cand)
            if np.isfinite(sc) and sc <= best_score + 1e-7:
                a = cand
                fails = 0
                trust = 0.3
            else:
                a = best.copy()
        m = float(q.max())
        if m <= 1e-15:
            break
        scale = 0.5 * m
        gamma = focus / scale
        vals = q - m
        vals = gamma * vals
        vals = np.clip(vals, -50.0, 0.0)
        e = np.exp(vals)
        s = float(e.sum())
        if s <= 1e-300:
            break
        w = e / s
        size = next_pow2(w.size + n - 1)
        fw = np.fft.rfft(w, size)
        fa = np.fft.rfft(a, size)
        c = np.fft.irfft(fw * fa.conj(), size)[:n]
        c = np.maximum(c, 0.0)
        cmax = float(c.max())
        cmin = float(c.min())
        rng_c = cmax - cmin
        if rng_c <= 1e-15:
            break
        move = trust * (cmax - c) / rng_c
        move = np.clip(move, -50.0, 50.0)
        b = a * np.exp(move)
        b = normalize_vec(b)
        Mb = float(np.max(self_conv(b)))
        score_b = 2.0 * n * Mb
        if score_b < best_score - 1e-13:
            best = b.copy()
            best_score = score_b
            a = b
            fails = 0
            trust = min(trust * 1.08, 4.0)
        elif score_b < score - 1e-14:
            a = b
            fails = 0
            trust = min(trust * 1.05, 4.0)
        else:
            fails += 1
            trust = max(trust * 0.75, 0.02)
            focus = min(focus * 1.08, 50.0)
        if verbose and (time.monotonic() - t0) % 30.0 < 0.02:
            print("    smooth: t=%.1fs score=%.9f focus=%.1f trust=%.3f fails=%d"
                  % (time.monotonic() - t0, best_score, focus, trust, fails))
    return best


def coordinate_polish(a0, time_limit, rng_seed=123):
    """Targeted coordinate descent to reduce peaks."""
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n <= 10 or time_limit <= 0:
        return a
    best = a.copy()
    best_score = objective_norm(a)
    stall = 0
    rng = np.random.default_rng(rng_seed)
    while time.monotonic() - t0 < time_limit and stall < 300:
        q = self_conv(a)
        top_lags = np.argsort(q)[::-1][:min(12, len(q))]
        scores = np.zeros(n, dtype=np.float64)
        for kk in top_lags[:6]:
            k = int(kk)
            i0 = max(0, k - n + 1)
            i1 = min(n, k + 1)
            if i0 < i1:
                start = k - (i1 - 1)
                stop = k - i0 + 1
                scores[i0:i1] += a[start:stop][::-1]
        src = np.argsort(scores)[::-1][:15]
        dst = np.argsort(scores)[:15]
        improved = False
        for s_idx in range(len(src)):
            if time.monotonic() - t0 > time_limit:
                break
            if improved:
                break
            d = int(src[s_idx])
            for j_idx in range(len(dst)):
                if improved:
                    break
                j = int(dst[j_idx])
                if d == j:
                    continue
                v = float(a[d])
                if v <= 1e-15:
                    continue
                for frac in (0.03, 0.08, 0.15, 0.25):
                    if time.monotonic() - t0 > time_limit:
                        break
                    amt = v * frac
                    if amt <= 1e-18:
                        continue
                    b = a.copy()
                    b[d] -= amt
                    b[j] += amt
                    b = np.maximum(b, 0.0)
                    ss = float(b.sum())
                    if ss <= 1e-15:
                        continue
                    b = b / ss
                    sc = objective_norm(b)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best = b.copy()
                        a = b
                        improved = True
                        break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 30 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    b = a.copy()
                    amt = a[i] * 0.05
                    b[i] -= amt
                    b[j] += amt
                    b = np.maximum(b, 0.0)
                    ss = float(b.sum())
                    if ss > 1e-15:
                        b = b / ss
                        sc = objective_norm(b)
                        if sc < best_score - 1e-13:
                            best_score = sc
                            best = b.copy()
                            a = b
                            stall = 0
    return best


def interpolate_sequence(f, new_n):
    f = normalize_vec(f)
    old_n = f.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return f.copy()
    if old_n <= 1:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f)
    return normalize_vec(g)


def transform_search(a0, time_limit):
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    best = a.copy()
    best_score = objective_norm(a)
    gammas = (0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0)
    for g in gammas:
        if time.monotonic() - t0 > time_limit:
            break
        b = np.maximum(a, 1e-30) ** g
        b = normalize_vec(b)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()
    u = np.ones(n, dtype=np.float64) / float(n)
    for w in (0.03, 0.08, 0.15, 0.25, 0.4):
        if time.monotonic() - t0 > time_limit:
            break
        b = normalize_vec((1.0 - w) * a + w * u)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()
    rev = a[::-1].copy()
    for w in (0.1, 0.25, 0.5, 0.75):
        if time.monotonic() - t0 > time_limit:
            break
        b = normalize_vec((1.0 - w) * a + w * rev)
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()
    return best


def trim_search(a0, time_limit):
    t0 = time.monotonic()
    a = normalize_vec(a0)
    n = a.size
    if n < 30 or time_limit <= 0:
        return a
    best = a.copy()
    best_score = objective_norm(a)
    max_side = min(n // 4, 5000)
    samples = []
    for v in (0, 1, 2, 5, 10, 20, 50, 100, 200, 500, 1000, 2000, 3000, 4000):
        if 0 <= v <= max_side:
            samples.append(v)
    samples.append(max_side)
    samples = list(set(samples))
    samples.sort()
    for l in samples:
        if time.monotonic() - t0 > time_limit:
            break
        if n - l < 20:
            continue
        b = normalize_vec(a[l:])
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()
    for r in samples:
        if time.monotonic() - t0 > time_limit:
            break
        if n - r < 20:
            continue
        b = normalize_vec(a[:n - r])
        sc = objective_norm(b)
        if sc < best_score - 1e-12:
            best_score = sc
            best = b.copy()
    for l in samples[:8]:
        if time.monotonic() - t0 > time_limit:
            break
        for r in samples[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            if l + r >= n - 20:
                continue
            b = normalize_vec(a[l:n - r])
            sc = objective_norm(b)
            if sc < best_score - 1e-12:
                best_score = sc
                best = b.copy()
    return best


def make_diverse_starts(n, parent):
    out = []
    if parent is not None:
        p = normalize_vec(parent)
        if p.size != n:
            p = interpolate_sequence(p, n)
    else:
        x = np.linspace(0.0, 1.0, n)
        p = normalize_vec(np.sin(np.pi * x) ** 2)
    out.append(p.copy())
    out.append(p[::-1].copy())
    x = np.arange(n, dtype=np.float64) / float(max(n - 1, 1))
    for aexp in (0.5, 1.0, 1.5, 2.0, 3.0):
        out.append(normalize_vec((x + 1e-8) ** aexp))
        out.append(normalize_vec(((1.0 - x) + 1e-8) ** aexp))
    for aexp in (0.5, 1.0, 2.0):
        for bexp in (0.5, 1.0, 2.0):
            seq = ((x + 1e-8) ** aexp) * (((1.0 - x) + 1e-8) ** bexp)
            out.append(normalize_vec(seq))
    for r in (0.5, 1.0, 2.0, 5.0, 10.0):
        seq = np.exp(-r * x) + np.exp(-r * (1.0 - x))
        out.append(normalize_vec(seq))
    for ctr in (0.3, 0.5, 0.7):
        for wd in (0.05, 0.1, 0.15, 0.25, 0.4):
            seq = np.exp(-((x - ctr) ** 2) / (2.0 * wd * wd))
            out.append(normalize_vec(seq))
    for c1, c2 in ((0.25, 0.75), (0.2, 0.8), (0.35, 0.65)):
        for wd in (0.08, 0.15, 0.25):
            seq = (np.exp(-((x - c1) ** 2) / (2.0 * wd * wd)) +
                   np.exp(-((x - c2) ** 2) / (2.0 * wd * wd)))
            out.append(normalize_vec(seq))
    rng = np.random.default_rng(2024)
    for _ in range(20):
        seq = np.zeros(n, dtype=np.float64)
        for __ in range(int(rng.integers(1, 5))):
            ctr = rng.uniform(0.05, 0.95)
            wd = rng.uniform(0.04, 0.4)
            ht = rng.uniform(0.3, 1.5)
            seq += ht * np.exp(-((x - ctr) ** 2) / (2.0 * wd * wd))
        seq += 1e-8
        out.append(normalize_vec(seq))
    base = np.maximum(p, 1e-30)
    for g in (0.4, 0.7, 1.4, 2.0, 3.0):
        out.append(normalize_vec(base ** g))
    return out


def best_diverse_starts(starts, max_kept=6):
    scored = []
    for s in starts:
        s = normalize_vec(s)
        sc = objective_norm(s)
        if np.isfinite(sc):
            scored.append((sc, s))
    scored.sort(key=lambda x: x[0])
    kept = []
    for sc, seq in scored:
        dup = False
        for kseq in kept:
            if seq.size == kseq.size:
                dp = float(np.dot(seq / np.linalg.norm(seq), kseq / np.linalg.norm(kseq)))
                if dp > 0.99:
                    dup = True
                    break
        if not dup:
            kept.append(seq.copy())
        if len(kept) >= max_kept:
            break
    return kept


def update_best(best_seq, best_score, a, label, t0):
    if a is None:
        return best_seq, best_score
    an = normalize_vec(a)
    if an.size == 0:
        return best_seq, best_score
    sc = objective_norm(an)
    if np.isfinite(sc) and sc < best_score - 1e-12:
        best_score = sc
        best_seq = an.copy()
        try:
            print("*** NEW BEST [%s]: %.9f n=%d t=%.1fs"
                  % (label, sc, an.size, time.monotonic() - t0))
        except Exception:
            pass
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    total_budget = min(1800.0, float(budget_s)) - 60.0
    total_budget = max(120.0, total_budget)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if p.ndim != 1:
            p = p.ravel()
        if p.size > 0 and p.sum() > 1e-12:
            parent = p
    except Exception:
        parent = None

    if parent is not None:
        best_seq = normalize_vec(parent)
        best_score = objective_norm(best_seq)
        n0 = best_seq.size
        if n0 > 9000:
            best_seq = interpolate_sequence(best_seq, 8000)
            best_score = objective_norm(best_seq)
            n0 = best_seq.size
        print("Parent: n=%d objective=%.9f" % (n0, best_score))
    else:
        n0 = 6700
        x = np.linspace(0.0, 1.0, n0)
        best_seq = normalize_vec(np.sin(np.pi * x) ** 2)
        best_score = objective_norm(best_seq)
        print("Default: n=%d objective=%.9f" % (n0, best_score))

    # Generate diverse starts and pick best few
    starts = make_diverse_starts(n0, best_seq)
    kept = best_diverse_starts(starts, max_kept=6)
    if best_seq is not None:
        matched = False
        for k in kept:
            if k.size == best_seq.size and float(np.dot(k, best_seq)) > 0.999:
                matched = True
                break
        if not matched:
            kept.insert(0, best_seq.copy())

    # Phase 1: LP-FW on best start (largest chunk of time)
    remaining = total_budget - (time.monotonic() - t_global)
    t_lp1 = min(650.0, 0.42 * remaining)
    if t_lp1 > 60:
        print("Phase 1: LP-FW main (t=%.0fs)" % t_lp1)
        out = lp_fw_search(best_seq, t_lp1, verbose=True, K_start=180, K_max=700)
        best_seq, best_score = update_best(best_seq, best_score, out, "LP_main", t_global)

    # Phase 2: Nesterov subgradient to escape local minima
    remaining = total_budget - (time.monotonic() - t_global)
    t_ng = min(200.0, 0.14 * remaining)
    if t_ng > 40:
        print("Phase 2: Nesterov SG (t=%.0fs)" % t_ng)
        out = nesterov_sg_search(best_seq, t_ng, verbose=True)
        best_seq, best_score = update_best(best_seq, best_score, out, "NestSG", t_global)

    # Phase 3: Smooth polish
    remaining = total_budget - (time.monotonic() - t_global)
    t_sp = min(200.0, 0.13 * remaining)
    if t_sp > 30:
        print("Phase 3: Smooth polish (t=%.0fs)" % t_sp)
        out = smooth_polish(best_seq, t_sp, rng_seed=777, verbose=True)
        best_seq, best_score = update_best(best_seq, best_score, out, "smooth1", t_global)

    # Phase 4: LP-FW again (second round, refined)
    remaining = total_budget - (time.monotonic() - t_global)
    t_lp2 = min(300.0, 0.20 * remaining)
    if t_lp2 > 40:
        print("Phase 4: LP-FW round 2 (t=%.0fs)" % t_lp2)
        out = lp_fw_search(best_seq, t_lp2, verbose=True, K_start=150, K_max=500)
        best_seq, best_score = update_best(best_seq, best_score, out, "LP_r2", t_global)

    # Phase 5: Try second diverse start
    remaining = total_budget - (time.monotonic() - t_global)
    if len(kept) >= 2 and remaining > 200:
        cand2 = kept[1]
        same = False
        if cand2.size == best_seq.size and float(np.dot(cand2, best_seq)) > 0.99:
            if len(kept) >= 3:
                cand2 = kept[2]
                if cand2.size == best_seq.size and float(np.dot(cand2, best_seq)) > 0.99:
                    cand2 = None
            else:
                cand2 = None
        if cand2 is not None:
            t2 = min(180.0, 0.22 * remaining)
            if t2 > 40:
                print("Phase 5: LP-FW on diverse start (t=%.0fs)" % t2)
                out = lp_fw_search(cand2, t2, verbose=False, K_start=150, K_max=500)
                best_seq, best_score = update_best(best_seq, best_score, out, "LP_diverse", t_global)

    # Phase 6: Coordinate polish
    remaining = total_budget - (time.monotonic() - t_global)
    t_cp = min(120.0, 0.10 * remaining)
    if t_cp > 25:
        print("Phase 6: Coordinate polish (t=%.0fs)" % t_cp)
        out = coordinate_polish(best_seq, t_cp, rng_seed=31415)
        best_seq, best_score = update_best(best_seq, best_score, out, "coord", t_global)

    # Phase 7: Transform
    remaining = total_budget - (time.monotonic() - t_global)
    t_tf = min(60.0, 0.06 * remaining)
    if t_tf > 15:
        out = transform_search(best_seq, t_tf)
        best_seq, best_score = update_best(best_seq, best_score, out, "transform", t_global)

    # Phase 8: Trim
    remaining = total_budget - (time.monotonic() - t_global)

        out = coordinate_polish(best_seq, t_fc, rng_seed=555)
        best_seq, best_score = update_best(best_seq, best_score, out, "coord_final", t_global)

    # Final validation
    if best_seq is None:
        x = np.linspace(0.0, 1.0, 6700)
        best_seq = normalize_vec(np.sin(np.pi * x) ** 2)

    out_arr = np.asarray(best_seq, dtype=np.float64)
    out_arr = np.where(np.isfinite(out_arr), out_arr, 0.0)
    out_arr = np.maximum(out_arr, 0.0)
    out_arr = np.minimum(out_arr, 1000.0)
    s = float(out_arr.sum())
    if s <= 0.01:
        x = np.linspace(0.0, 1.0, 6700)
        out_arr = np.sin(np.pi * x) ** 2
        s = float(out_arr.sum())
    out_arr = out_arr / s

    final_score = objective_norm(out_arr)
    try:
        official = evaluate_safe(out_arr)
        if np.isfinite(official):
            final_score = min(final_score, official)
    except Exception:
        pass

    elapsed = time.monotonic() - t_global
    print("FINAL: internal=%.10f n=%d time=%.1fs" % (final_score, out_arr.size, elapsed))
    return [float(v) for v in out_arr]
