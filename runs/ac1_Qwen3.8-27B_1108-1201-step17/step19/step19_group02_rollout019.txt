    """Compute the smooth approximation to max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    # Shift to avoid overflow in exp
    shifted = tau * (conv - conv.max())
    log_sum = np.log(np.sum(np.exp(shifted))) + tau * conv.max()
    return log_sum / tau  # This is (1/tau) * log(sum(exp(tau*conv)))

def smooth_gradient(a, tau):

    n = len(a)

    shifted = tau * (conv - conv.max())
    weights = np.exp(shifted)
    weights /= weights.sum()  # softmax weights

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_gradient(a, tau):
    """Compute gradient of (1/tau)*log(sum(exp(tau*conv(a,a)))) efficiently via FFT."""
    n = len(a)
    conv = fast_convolve(a)
    # Softmax weights on convolution values
    shifted = tau * (conv - conv.max())
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones_like(conv) / len(conv)
    else:
        w /= w_sum
    # Cross-correlation: grad[i] = 2 * sum_j a[j] * w[i+j] for valid indices
    # This equals 2 * conv(a, flip(w))[2n-2-i] for i=0..n-1
    L = 3 * n - 2
    fft_size_gc = 1 << (L - 1).bit_length()
    Aw = np.fft.rfft(a, fft_size_gc)
    Fw = np.fft.rfft(w[::-1], fft_size_gc)
    conv_aw = np.fft.irfft(Aw * Fw, fft_size_gc)
    # Extract: grad[i] = 2 * conv_aw[2n-2-i] for i=0..n-1
    grad = 2.0 * conv_aw[2 * n - 2 : n - 2 : -1]
    return grad, conv

def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2500, add_per_iter=40,
          verbose=True):
    """LP-based Frank-Wolfe for minimizing max convolution."""
    n = len(a0)
    y = np.asarray(a0, dtype=np.float64).copy()
    y = np.maximum(y, 0.0)
    s = y.sum()
    if s < 1e-15:
        y = np.ones(n) / n
    else:
        y = y / s

    best_a = y.copy()
    best_m = fast_convolve(y).max()
    t0 = time.monotonic()

    ac_y = fast_convolve(y)
    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())

    no_improve = 0
    max_no_improve = 800
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            break
        iter_count += 1

        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set size
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set if k < len(ac)])
            if len(margins) > 0:
                keep = np.argsort(margins)[:max_K // 2]
                new_set = set(active_set[i] for i in keep if i < len(active_set))
                active_set = sorted(new_set)

        # Build constraint matrix
        K = len(active_set)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(active_set):
            k_int = int(k)
            if k_int >= len(ac):
                continue
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = sorted(active_set[:initial_K])
            no_improve += 50
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 20
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations and add to active set
        size = 2 * n - 1
        fft_size_vs = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size_vs)
        Gf = np.fft.rfft(g_dir, fft_size_vs)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size_vs)[:size]

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)

        # Line search for step size
        default_alpha = 2.0 / (iter_count + 2.0)
        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in [default_alpha, default_alpha * 0.5, default_alpha * 1.5,
                      0.05, 0.1, 0.2, 0.3, 0.5]:
            alpha = max(0.01, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 2:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.1f}s): J={J:.6f}, |S|={len(active_set)}")

    best_val = current_score

    candidates = []
    # Power law families
    for p in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(np.maximum(i_arr + 1.0, 0.1) ** p)
        candidates.append(np.maximum(n - i_arr, 0.1) ** p)
    # Gaussian bumps
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sigma in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    # Sine-based
    for freq in [1, 2, 3]:
        for phase in [0.0, np.pi / 4, np.pi / 2]:
            candidates.append(np.maximum(1.0 + np.cos(2 * np.pi * freq * x + phase), 0.01))
    # Sine squared
    candidates.append(np.sin(np.pi * x) ** 2)
    candidates.append(np.sin(np.pi * x) ** 3)
    candidates.append(np.sin(np.pi * x) ** 0.5)
    # Mixed
    for w in [0.1, 0.3, 0.5]:
        candidates.append(np.sin(np.pi * x) ** 2 + w * x * (1 - x))
    # Exponential
    for rate in [1, 2, 5, 10]:
        candidates.append(np.exp(-rate * x) + np.exp(rate * (x - 1)))
    # Two-sided power
    for p in [0.5, 1.0, 2.0]:
        candidates.append((x + 0.01) ** p + (1 - x + 0.01) ** p)

    # Normalize and evaluate
    for cand in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        c = np.maximum(cand, 0.0)
        sc = c.sum()
        if sc < 1e-10:
            continue
        c = c / sc
        obj = 2.0 * n * fast_convolve(c).max()
        if obj < best_val:
            best_val = obj
            best_a = c.copy()

    return best_a, best_val


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, t_global):
    """Try to update the best sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    M = fast_convolve(a_norm).max()
    score = 2.0 * len(a_norm) * M
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"*** NEW BEST: {score:.6f} (n={len(a)}, t={elapsed:.1f}s)")
    return best_seq, best_score
