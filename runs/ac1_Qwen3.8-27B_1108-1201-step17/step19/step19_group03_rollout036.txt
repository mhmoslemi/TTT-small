def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(p, ac, top_k=50, power=3.0):
    """Compute weighted subgradient of max_k ac[k] at point p."""
    n = len(p)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx].copy()
    top_vals = np.maximum(top_vals, 1e-30)
    w = top_vals ** power
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        grad[idx] += 2.0 * wi * p[j_idx]
    return grad

    t = 0
    stall = 0
    stall_max = 5000
    cur_top_k = top_k
    cur_power = power
    gamma = 1.0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        ac = fast_autocorr(p)
        F_curr = ac.max()
        
        if F_curr < best_F - 1e-15:
            best_F = F_curr
            best_p = p.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > stall_max:
            pert = rng.normal(0, 0.01, n) * best_p
            p = np.maximum(best_p + pert, 0.0)
            p = p / p.sum()
            stall = 0
            cur_top_k = min(cur_top_k + 20, 200)
            cur_power = max(cur_power - 0.5, 1.5)
            gamma = 1.0
            continue
        
        g = compute_subgradient(p, ac, top_k=cur_top_k, power=cur_power)
        
        # FW step: move toward vertex with min gradient
        v_idx = np.argmin(g)
        e_v = np.zeros(n)
        e_v[v_idx] = 1.0
        
        # Adaptive step size
        gamma_t = 2.0 / (t + 2.0)
        gamma = min(1.0, gamma * 0.999 + gamma_t * 0.001)
        
        p_new = (1.0 - gamma) * p + gamma * e_v
        p_new = np.maximum(p_new, 0.0)
        s = p_new.sum()
        if s < 1e-15:
            p_new = p.copy()
        else:
            p_new = p_new / s
        
        # Line search: try a few step sizes
        best_m = fast_autocorr(p_new).max()
        best_p_new = p_new
        for mult in [0.5, 1.5, 2.0, 0.3]:
            gamma_try = gamma * mult
            if gamma_try <= 0 or gamma_try > 1:
                continue
            p_try = (1.0 - gamma_try) * p + gamma_try * e_v
            p_try = np.maximum(p_try, 0.0)
            s_try = p_try.sum()
            if s_try < 1e-15:
                continue
            p_try = p_try / s_try
            m_try = fast_autocorr(p_try).max()
            if m_try < best_m:
                best_m = m_try
                best_p_new = p_try
        
        p = best_p_new
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_F
            print(f"    FW t={t} ({elapsed:.0f}s): J={J:.6f}, F={best_F:.2e}, stall={stall}")
    
    return best_p, best_F


def nesterov_subgrad(p0, time_limit, top_k=80, power=4.0, c0=2.0, verbose=False, rng_seed=0):
    """Nesterov-accelerated subgradient with simplex projection and AdaGrad."""
    n = len(p0)
    p = norm_sum1(p0)
    best_p = p.copy()
    best_F = fast_autocorr(p).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    p_prev = p.copy()
    G = np.ones(n) * 1e-8  # AdaGrad accumulator
    eta = c0
    stall = 0
    stall_max = 3000
    cur_top_k = top_k
    cur_power = power
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = p + beta_t * (p - p_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = p.copy()
        else:
            y = y / s_y
        
        ac = fast_autocorr(y)
        F_curr = ac.max()
        
        if F_curr < best_F - 1e-15:
            best_F = F_curr
            best_p = y.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > stall_max:
            pert = rng.normal(0, 0.02, n) * best_p
            p_r = np.maximum(best_p + pert, 1e-10)
            p_r = p_r / p_r.sum()
            p = p_r
            p_prev = p_r.copy()
            stall = 0
            eta = c0
            G = np.ones(n) * 1e-8
            cur_top_k = min(cur_top_k + 30, 200)
            cur_power = max(cur_power - 0.5, 2.0)
            continue
        
        g = compute_subgradient(y, ac, top_k=cur_top_k, power=cur_power)
        
        # AdaGrad step size
        eta_t = eta / (np.sqrt(G).mean() + 1e-8)
        
        # Projection onto simplex
        p_trial = y - eta_t * g
        p_new = project_simplex(p_trial)
        
        # Update AdaGrad
        G += g ** 2
        
        p_prev = p
        p = p_new
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_F
            print(f"    NSub t={t} ({elapsed:.0f}s): J={J:.6f}, eta={eta_t:.2e}, stall={stall}")
    
    return best_p, best_F


def lp_fw_optimize(p0, time_limit, solver, top_K_init=200, top_K_max=None, verbose=False):
    """LP-based Frank-Wolfe: solve LP to find best direction."""
    n = len(p0)
    p = norm_sum1(p0)
    best_p = p.copy()
    best_F = fast_autocorr(p).max()
    t0 = time.monotonic()
    
    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(p)
        M_curr = ac.max()
        
        if M_curr < best_F - 1e-15:
            best_F = M_curr
            best_p = p.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 50:
            K = min(K + 100, top_K_max)
            stall = 25
        
        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]
        
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = p[ki - j]
        
        rhs = M_curr - ac[binding_idx]
        rhs = np.maximum(rhs, 0.0)
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= rhs, cp.sum(b_var) <= 1.0]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        
        g_dir = b_val / b_sum
        
        # Line search
        base_obj = M_curr
        best_alpha = 2.0 / (it + 2.0)
        best_trial_F = fast_autocorr((1 - best_alpha) * p + best_alpha * g_dir).max()
        for alpha in [best_alpha * 0.5, best_alpha, best_alpha * 1.5, best_alpha * 2.0, 0.1, 0.2, 0.3]:
            if alpha <= 0 or alpha > 1:
                continue
            p_trial = (1.0 - alpha) * p + alpha * g_dir
            p_trial = np.maximum(p_trial, 0.0)
            s_t = p_trial.sum()
            if s_t < 1e-15:
                continue
            p_trial = p_trial / s_t
            F_trial = fast_autocorr(p_trial).max()
            if F_trial < best_trial_F:
                best_trial_F = F_trial
                best_alpha = alpha
        
        p = (1.0 - best_alpha) * p + best_alpha * g_dir
        p = np.maximum(p, 0.0)
        s_p = p.sum()
        if s_p < 1e-15:
            break
        p = p / s_p
        
        it += 1
        if it % 5 == 0:
            K = min(K + 50, top_K_max)
        
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_F
            print(f"    LP-FW it={it} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")
    
    return best_p, best_F


def coord_polish(p0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent: move mass from high-contribution coordinates to low ones."""
    n = len(p0)
    p = norm_sum1(p0)
    best_p = p.copy()
    best_F = fast_autocorr(p).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 80
    n_top = 20
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(p)
        top_k = np.argsort(ac)[::-1][:n_top]
        
        # Compute sensitivity: which coordinates contribute most to peaks
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:10]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += p[ki - idx]
        
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = p[idx] * p[ki - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = p[src]
            if src_val < 1e-12:
                continue
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            top_cands = candidates[np.argsort(cand_sens)[:3]]
            
            found = False
            for dst in top_cands:
                if dst == src or found:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    p_trial = p.copy()
                    p_trial[src] -= amt
                    p_trial[dst] += amt
                    p_trial = np.maximum(p_trial, 0.0)
                    s_t = p_trial.sum()
                    if s_t < 1e-15:
                        continue
                    p_trial = p_trial / s_t
                    F_trial = fast_autocorr(p_trial).max()
                    if F_trial < best_F - 1e-16:
                        p = p_trial
                        best_F = F_trial
                        best_p = p.copy()
                        improved = True
                        found = True
                        break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and p[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = p[i] * frac
                p[i] -= amt
                p[j] += amt
                p = np.maximum(p, 0.0)
                p = p / p.sum()
                F_new = fast_autocorr(p).max()
                if F_new < best_F:
                    best_F = F_new
                    best_p = p.copy()
                    stall = 0
                    improved = True
    
    return best_p, best_F


def smooth_optimize(p0, time_limit, verbose=False, rng_seed=100):
    """Smooth gradient approach: use softmax-weighted all lags."""
    n = len(p0)
    p = norm_sum1(p0)
    best_p = p.copy()
    best_F = fast_autocorr(p).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    eta = 0.5
    stall = 0
    max_stall = 400
    tau = 10.0
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / min(time_limit, 600.0))
        tau_cur = tau * (5.0) ** progress
        
        ac = fast_autocorr(p)
        ac_max = ac.max()
        ac_shifted = -tau_cur * (ac_max - ac)
        exp_ac = np.exp(ac_shifted)
        w = exp_ac / exp_ac.sum()
        
        # Weighted gradient
        g = np.zeros(n)
        # Use all lags weighted (approximate with top 200 for speed)
        top_200 = np.argsort(ac)[::-1][:200]
        w_top = w[top_200]
        w_top = w_top / w_top.sum()
        for wi, k in zip(w_top, top_200):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            g[idx] += 2.0 * wi * p[j_idx]
        
        g_tangent = g - g.mean()
        p_trial = p - eta * g_tangent
        p_new = project_simplex(p_trial)
        
        F_new = fast_autocorr(p_new).max()
        if F_new < best_F - 1e-15:
            best_F = F_new
            best_p = p_new.copy()
            stall = 0
            eta = min(eta * 1.01, 2.0)
        else:
            stall += 1
            if stall > 50:
                eta *= 0.8
            
            p = p_new
            if stall > 200:
                pert = rng.normal(0, 0.005, n) * p
                p_p = np.maximum(p + pert, 0.0)
                p_p = p_p / p_p.sum()
                F_p = fast_autocorr(p_p).max()
                if F_p < best_F:
                    best_F = F_p
                    best_p = p_p.copy()
                    stall = 0
                    eta = 0.5
                else:
                    p = p_new
        total_iters += 1
        
        if verbose and total_iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_F
            print(f"    Smooth it={total_iters} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.4f}, stall={stall}")
    
    return best_p, best_F


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def make_cosine(n, alpha=0.5):
    x = np.arange(n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * x / n)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def fw_optimize_advanced(p0, time_limit, top_k=80, power=3.0, verbose=False, rng_seed=0):
    """Advanced FW with pairwise steps and adaptive parameters."""
    n = len(p0)
    p = norm_sum1(p0)
    best_p = p.copy()
    best_F = fast_autocorr(p).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    t = 0
    stall = 0
    stall_max = 8000
    cur_top_k = top_k
    cur_power = power
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        ac = fast_autocorr(p)
        F_curr = ac.max()
        
        if F_curr < best_F - 1e-15:
            best_F = F_curr
            best_p = p.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > stall_max:
            # Restart with perturbation
            pert = rng.normal(0, 0.015, n) * best_p
            p = np.maximum(best_p + pert, 1e-10)
            p = p / p.sum()
            stall = 0
            t = max(t // 2, 1)
            cur_top_k = min(cur_top_k + 30, 250)
            cur_power = max(cur_power - 0.5, 1.5)
            continue
        
        g = compute_subgradient(p, ac, top_k=cur_top_k, power=cur_power)
        
        # FW step with step size
        gamma_t = 2.0 / (t + 2.0)
        
        # Standard FW: move to min gradient vertex
        v_idx = np.argmin(g)
        
        # Also try pairwise: swap mass between v_idx and max gradient coordinate
        v_idx2 = np.argmax(g)
        
        best_F_trial = F_curr
        best_p_trial = p.copy()
        best_gamma = gamma_t
        
        # Standard FW
        e_v = np.zeros(n)
        e_v[v_idx] = 1.0
        p_try = (1 - gamma_t) * p + gamma_t * e_v
        p_try = np.maximum(p_try, 0.0)
        s_try = p_try.sum()
        if s_try > 1e-15:
            p_try = p_try / s_try
            F_try = fast_autocorr(p_try).max()
            if F_try < best_F_trial:
                best_F_trial = F_try
                best_p_trial = p_try.copy()
                best_gamma = gamma_t
        
        # Pairwise FW
        if v_idx2 != v_idx and p[v_idx2] > 1e-10:
            e_v2 = np.zeros(n)
            e_v2[v_idx2] = 1.0
            for gamma_try in [gamma_t, min(1.0, gamma_t * 2.0), gamma_t * 0.5]:
                p_try2 = (1 - gamma_try) * p + gamma_try * e_v2
                p_try2 = np.maximum(p_try2, 0.0)
                s_try2 = p_try2.sum()
                if s_try2 > 1e-15:
                    p_try2 = p_try2 / s_try2
                    F_try2 = fast_autocorr(p_try2).max()
                    if F_try2 < best_F_trial:
                        best_F_trial = F_try2
                        best_p_trial = p_try2.copy()
        
        # Away-step FW: remove mass from max gradient coordinate
        if p[v_idx2] > 1e-10:
            gamma_away = min(0.3, gamma_t)
            p_away = p - gamma_away * p[v_idx2] / max(np.sum(p > 1e-10), 1) * np.eye(n)[v_idx2]
            # Simpler: just reduce the coordinate
            p_away = p.copy()
            amt = p[v_idx2] * gamma_away
            p_away[v_idx2] -= amt
            p_away = np.maximum(p_away, 0.0)
            s_away = p_away.sum()
            if s_away > 1e-15:
                p_away = p_away / s_away
                F_away = fast_autocorr(p_away).max()
                if F_away < best_F_trial:
                    best_F_trial = F_away
                    best_p_trial = p_away.copy()
        
        p = best_p_trial
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_F
            print(f"    FW-adv t={t} ({elapsed:.0f}s): J={J:.6f}, K={cur_top_k}, stall={stall}")
    
    return best_p, best_F

        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_optimize(a_in, t4, verbose=True, rng_seed=300)
        try_update(a_out, "smooth")
    
    # === Phase 5: LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW (n={n_best}) ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 40 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(300, 2 * n_best - 1)
        top_K_max = min(3000, 2 * n_best - 1)
        a_out, _ = lp_fw_optimize(a_in, t5, solver, verbose=True,
                                   top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp_fw")
    
    # === Phase 6: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Coord polish (n={len(best_seq)}) ===")
    t6 = min(remaining * 0.08, 130.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_polish(a_in, t6, verbose=True, rng_seed=456)
        try_update(a_out, "coord")
    
    # === Phase 7: Second round of FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.10, 170.0)
    if t7 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = fw_optimize_advanced(a_in, t7, top_k=100, power=2.5, verbose=False, rng_seed=777)
        try_update(a_out, "fw_round2")
    
    # === Phase 8: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Length exploration ===")
    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 1500, -1500, 300, -300]:
        new_n = n_best + delta
        if 1000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]
    
    for new_n in trial_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)...")
        a_long = interpolate_sequence(best_seq, new_n)
        
        t8a = min((remaining - 10) * 0.4, 100.0)
        if t8a > 20:
            a_out, _ = fw_optimize_advanced(a_long, t8a, top_k=60, power=3.0, verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"fw_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t8b = min(remaining * 0.3, 80.0)
            if t8b > 20:
                a_out, _ = coord_polish(np.asarray(interpolate_sequence(best_seq, new_n)), t8b, verbose=False, rng_seed=new_n + 1)
                try_update(a_out, f"coord_n{new_n}")
    
    # === Phase 9: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 9: Final polish ===")
        t9 = min(remaining * 0.8, 60.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_polish(a_in, t9, verbose=False, rng_seed=789)
            try_update(a_out, "final_coord")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
