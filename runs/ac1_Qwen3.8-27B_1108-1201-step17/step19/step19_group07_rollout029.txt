def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def softmax_gradient(a, beta, top_k=None):

def fista_optimize(a0, time_limit, beta_start=5.0, beta_end=200.0,
                   eta0=0.8, top_k=500, verbose=True):
    """FISTA optimization with adaptive softmax gradient."""
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    stall_threshold = 5000
    best_eta = eta0
    
    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        elapsed_frac = min(iter_count / max(time_limit * 100, 1), 1.0)
        beta = beta_start * (beta_end / beta_start) ** (elapsed_frac ** 0.7)
        
        # Periodic boost
        if iter_count % 4000 == 3000:
            eta_mult = 2.5
        else:
            eta_mult = 1.0
        
        eta = best_eta * eta_mult / (np.sqrt(t) * 0.5 + 1.0)
        
        k_use = min(top_k, 2 * n - 1)
        grad, c_y, _ = softmax_gradient(y, beta, top_k=k_use)
        
        # Ensure gradient has non-negative entries (it should, since a ≥ 0 and w ≥ 0)
        grad = np.maximum(grad, 0)
        
        # Mirror descent step (log-barrier for positivity)
        log_y = np.log(np.maximum(y, 1e-30)) - eta * grad
        log_y -= log_y.max()
        z = np.exp(log_y)
        z = np.maximum(z, 1e-15)
        z = z / z.sum()
        
        # Nesterov momentum
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = z + momentum * (z - y)
            x_new = np.maximum(x_new, 1e-15)
            x_new = x_new / x_new.sum()
        else:
            x_new = z.copy()
        
        c_new = fast_autocorr(x_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
            # Adaptively increase eta if we're making progress
            if iter_count % 1000 == 0:
                best_eta = min(best_eta * 1.1, 2.0)
        else:
            stall += 1
            if stall > 0 and stall % 1000 == 0:
                best_eta = max(best_eta * 0.95, 0.1)
        
        if stall > stall_threshold:
            rng = np.random.default_rng(iter_count * 7919)
            pert = rng.normal(0, 0.05, n) * best_x
            x_restart = norm_sum1(best_x + pert)
            x_new = x_restart
            y = x_restart.copy()
            stall = 0
            stall_threshold = max(stall_threshold * 0.85, 2000)
            best_eta = eta0
        
        y = x_new
        x = x_new
        iter_count += 1
        
        if verbose and iter_count % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, stall={stall}")
    
    return best_x, best_m
