def get_solver():

    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(100, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(cp.abs(x))), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):

                return s
        except Exception:

    _MODULE_SOLVER = cp.ECOS

    return _MODULE_SOLVER


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_obj(a):
    ac = fast_convolve(a)
    return 2.0 * len(a) * ac.max()

    return best_seq, best_score


def parametric_u_init(n, time_limit, best_seq, best_score):

    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_seq.copy()
    best_val = best_score

    candidates = []
    # U-shapes: (x^2 + (1-x)^2 + eps)
    for eps in [0.01, 0.05, 0.1, 0.2, 0.5]:
        candidates.append(norm_sum1(x**2 + (1.0 - x)**2 + eps))
    # Power U-shapes
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        candidates.append(norm_sum1(x**p + (1.0 - x)**p + 1e-6))
    # W-shapes
    for a in [1.0, 2.0, 3.0]:
        for b in [0.5, 1.0, 2.0]:
            candidates.append(norm_sum1((x - 1.0/3.0)**2 + (x - 2.0/3.0)**2 + b*(x**2 + (1-x)**2) + a*1e-6))
    # Asymmetric
    for asym in [0.3, 0.4, 0.6, 0.7]:
        candidates.append(norm_sum1(x**2 + ((1.0 - asym)*(1.0 - x) + asym*x)**2 + 0.1))
    # Sinusoidal variants
    for freq in [1, 2, 3]:
        for phase in [0, np.pi/4, np.pi/2]:
            candidates.append(norm_sum1(np.maximum(np.cos(freq * np.pi * x + phase)**2, 0.001)))
    # Previous-style: sin^2
    candidates.append(norm_sum1(np.sin(np.pi * x)**2))
    # Exponential edge concentration
    for rate in [1.0, 2.0, 3.0, 5.0, 10.0]:
        candidates.append(norm_sum1(np.exp(-rate * x) + np.exp(rate * (x - 1.0)) + 0.01))
    # Flat with edge bumps
    for bump in [0.5, 1.0, 2.0, 5.0]:
        candidates.append(norm_sum1(0.1 + bump * (x < 0.1).astype(float) + bump * (x > 0.9).astype(float)))

def focus_descent(a0, time_limit, top_k=5, momentum=0.8, verbose=True):
    """Mirror descent focusing on the worst lags."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    vel = np.zeros(n)
    c0 = 3.0
    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        y = a + momentum * vel
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y / s_y

        ac = fast_convolve(y)
        
        # Compute gradient from top k lags
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        
        # Softmax weights with temperature
        beta = 20.0
        max_tv = top_vals.max()
        w = np.exp(beta * (top_vals - max_tv))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            if wi < 1e-10:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                grad[idx] += 2.0 * wi * y[k_int - idx]

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        vel = 0.9 * vel + 0.1 * (a_new - a)
        a = a_new

        M_new = fast_convolve(a).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            noise = 0.03 / (1.0 + stall / 3000.0)
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r > 1e-15:
                a = x_r / s_r
            vel = np.zeros(n)
            stall = 0
            top_k = min(top_k + 5, 20)

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print("    Focus iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

    return best_a, best_M


def build_constraint_matrix_vec(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw(a0, time_limit, solver, initial_K=400, max_K=3000, add_per_iter=40, verbose=True):
    """Improved LP-Frank-Wolfe with periodic active set recomputation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_convolve(y)
    # Initialize active set: top K by value
    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())

    no_improve = 0
    max_no_improve = 800
    iter_count = 0
    full_recompute_every = 200

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: stagnation at iter " + str(iter_count))
            break
        iter_count += 1

        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic full active set recomputation
        if iter_count % full_recompute_every == 0:
            threshold = M_curr * 0.90
            above = np.where(ac >= threshold)[0]
            if len(above) > max_K:
                above = np.argsort(ac)[::-1][:max_K]
            active_set = sorted(above.tolist())
            if verbose:
                print("    LP-FW: recomputed active set, |S|=" + str(len(active_set)))

        # Trim active set if too large
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep = np.argsort(margins)[:max_K]
            active_set = sorted([active_set[i] for i in keep])

        K = len(active_set)
        if K == 0:
            break

        coeffs = build_constraint_matrix_vec(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = sorted(active_set[:initial_K])
            no_improve += 50
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 10
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations to add to active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)

        # Line search with more candidates
        default_alpha = 2.0 / (iter_count + 2.0)
        alphas = sorted(set([
            max(0.01, min(1.0, default_alpha)),
            max(0.01, min(1.0, default_alpha * 0.5)),
            max(0.01, min(1.0, default_alpha * 0.75)),
            max(0.01, min(1.0, default_alpha * 1.5)),
            max(0.01, min(1.0, default_alpha * 2.0)),
            max(0.01, min(1.0, 1.0 / (iter_count + 1.0))),
            max(0.01, min(1.0, 0.05)),
            max(0.01, min(1.0, 0.15)),
            max(0.01, min(1.0, 0.3)),
        ]))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 2:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and iter_count % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", |S|=" + str(len(active_set)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", iters=" + str(iter_count))
    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=99, verbose=False):
    """Peak-aware coordinate polish with multi-step line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 1000
    step_scales = [0.002, 0.004, 0.007, 0.01, 0.02, 0.04, 0.06, 0.1]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score destinations by how little they contribute to top lags
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < n // 4:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])

            if dst == src:
                continue

            found = False
            for frac in step_scales:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 300:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.015)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 200 == 0:
                for _ in range(8):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

                    if time.monotonic() - t0 > time_limit:
                        break

        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", stall=" + str(stall))

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search function trying multiple approaches and n values."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Total time budget: " + repr(time_budget) + "s")

    # Get initial best
    best_seq, best_score = get_best(height_sequence_1, t_global)
    n0 = len(best_seq)

    # Phase 1: Parametric U-shape search (quick)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Parametric (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.04, 40.0)
    if t1 > 10:
        a_param, _ = parametric_u_init(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "parametric", t_global)

    # Phase 2: Focus descent (fast initial improvement)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Focus Descent (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.12, 120.0)
    if t2 > 15:
        a_out, _ = focus_descent(norm_sum1(best_seq), t2, top_k=5, momentum=0.8, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "focus_n" + str(n_best), t_global)

    # Phase 3: LP-FW main
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP-FW main (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.30, 420.0)
    if t3 > 30 and n_best <= 10000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t3, solver,
                         initial_K=500, max_K=3500, add_per_iter=50, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1_n" + str(n_best), t_global)

    # Phase 4: Focus descent again (escape from LP-FW local min)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Focus Descent 2 (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.08, 80.0)
    if t4 > 15:
        a_out, _ = focus_descent(norm_sum1(best_seq), t4, top_k=8, momentum=0.7, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "focus2_n" + str(n_best), t_global)

    # Phase 5: LP-FW again
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: LP-FW refine (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 30 and n_best <= 10000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t5, solver,
                         initial_K=600, max_K=4000, add_per_iter=60, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2_n" + str(n_best), t_global)

    # Phase 6: Multi-n exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-n exploration ===")
    
    # Try several n values around the current best
    n_candidates = [n_best - 500, n_best + 500, n_best + 1000, n_best - 1000]
    # Also try some "round" numbers
    for round_n in [5000, 5500, 6000, 6500, 7000, 7500, 8000, 8500, 9000]:
        if abs(round_n - n_best) > 2000:
            continue
        if round_n not in n_candidates:
            n_candidates.append(round_n)
    
    n_candidates = sorted(set([l for l in n_candidates if 3000 <= l <= 12000]))
    
    # Sort by proximity to current best (try nearby first)
    n_candidates.sort(key=lambda x: abs(x - n_best))
    
    for new_n in n_candidates[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        if new_n == n_best:
            continue
        print("  Trying n=" + str(new_n) + " (remaining=" + str(round(remaining, 1)) + "s)")
        a_long = interpolate_seq(best_seq, new_n)
        
        # Quick LP-FW
        t6a = min(remaining * 0.30, 180.0)
        if t6a > 30 and new_n <= 10000:
            a_out, _ = lp_fw(a_long, t6a, solver,
                             initial_K=400, max_K=3000, add_per_iter=40, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t6b = min(remaining * 0.20, 60.0)
            if t6b > 15:
                a_out, _ = focus_descent(norm_sum1(best_seq), t6b, top_k=5, momentum=0.8, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "focus_n" + str(new_n), t_global)

    # Phase 7: Final LP-FW on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final LP-FW (n=" + str(n_best) + ") ===")
    if remaining > 60 and n_best <= 10000:
        t7 = min(remaining * 0.40, 300.0)
        if t7 > 30:
            a_out, _ = lp_fw(norm_sum1(best_seq), t7, solver,
                             initial_K=700, max_K=4500, add_per_iter=70, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # Phase 8: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final Coord (n=" + str(n_best) + ") ===")
    if remaining > 30:
        t8 = min(remaining * 0.70, 180.0)
        if t8 > 15:
            a_out, _ = coordinate_polish(norm_sum1(best_seq), t8, rng_seed=42, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
