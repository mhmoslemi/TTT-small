    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def pocs_optimize(a0, time_limit, verbose=False, n_iters_max=500000, rng_seed=0):
    """POCS: alternate between flat-spectrum projection and non-negativity projection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    fft_size = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    A_spec = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
    A_full = np.zeros(fft_size, dtype=np.complex128)
    A_full[:fft_size // 2 + 1] = A_spec
    for i in range(1, fft_size // 2):
        A_full[fft_size - i] = np.conj(A_spec[i])
    a = np.fft.irfft(A_full, fft_size)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = a0.copy()
    else:
        a = a / s

    ac = fast_autocorr(a)
    M = ac.max()
    if M < best_m:
        best_m = M
        best_a = a.copy()

    for it in range(n_iters_max):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        target_mag = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
        alpha_blend = 0.1 + 0.5 * (it / min(it + 1, 100))
        new_mag = (1 - alpha_blend) * mag + alpha_blend * target_mag
        A_new = new_mag * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
        a = a_new

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    POCS it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=False, top_K_init=200, top_K_max=None):
    """
    Improved LP-Frank-Wolfe with:
    - Proper quadratic line search
    - Adaptive K with smoother increase
    - Better handling of degeneracy
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build constraint matrix: T[i,j] = a[ki - j] for valid j
        T_rows = []
        valid_mrows = []
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start >= j_end:
                continue
            row = np.zeros(n)
            row[j_start:j_end] = a[ki - np.arange(j_start, j_end)]
            T_rows.append(row)
            valid_mrows.append(ii)

        if len(T_rows) == 0:
            break

        T_sel = np.array(T_rows)
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            K = min(K + 50, top_K_max)
            continue
        g_dir = b_val / b_sum

        # Proper line search: try to find best alpha in (0, 1]
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = M_curr
        
        # Quadratic line search
        alphas_to_try = [2.0 / (iter_count + 2.0)]
        # Add some neighborhood points
        base = 2.0 / (iter_count + 2.0)
        for mult in [0.5, 0.7, 1.0, 1.3, 1.7, 2.0, 3.0]:
            alphas_to_try.append(min(1.0, base * mult))
        alphas_to_try.append(0.1)
        alphas_to_try.append(0.05)
        
        # Remove duplicates and sort
        alphas_to_try = sorted(set(np.round(alphas_to_try, 6).tolist()))
        
        for alpha in alphas_to_try:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        iter_count += 1

        # Adapt K: increase gradually, faster when stagnating
        if iter_count % 5 == 0:
            if stagnation > 20:
                K = min(K + 100, top_K_max)
            else:
                K = min(K + 20, top_K_max)

        if stagnation > 50:
            # Reset K and try different approach
            K = max(top_K_init, K // 2)
            stagnation = 0

        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW v2 iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}, stag={stagnation}")

    return best_a, best_m


def smooth_gradient_optimize(a0, time_limit, verbose=False, tau_init=10.0, tau_final=100.0,
                              eta_init=1.0, rng_seed=100):
    """Smoothed gradient descent with adaptive parameters."""
    from scipy.signal import correlate as sp_correlate
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = eta_init
    stall = 0
    max_stall = 300
    total_iters = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / min(time_limit, 600.0))
        tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(a)
        ac_max = ac.max()
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()

        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        cross = sp_correlate(p, a, mode='valid')
        g = a * S + cross
        g_tangent = g - g.mean()

        # Try multiple step sizes
        best_trial_m = fast_autocorr(a).max()
        best_step = a.copy()
        for eta_try in [eta, eta * 0.5, eta * 0.1, eta * 2.0]:
            a_trial = a - eta_try * g_tangent
            a_trial = project_simplex(a_trial)
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_step = a_trial
                eta = eta_try

        a = best_step
        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                eta *= 0.5
            if stall > 100:
                pert = rng.normal(0, 0.003, n) * a
                a_pert = a + pert
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                m_pert = fast_autocorr(a_pert).max()
                if m_pert < best_m:
                    a = a_pert
                    best_m = m_pert
                    best_a = a.copy()
                    stall = 0
                    eta = eta_init * 0.3
                else:
                    a = project_simplex(a - 0.001 * g_tangent)

        total_iters += 1

        if verbose and total_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}, stall={stall}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Targeted coordinate descent for final polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 50
    n_top = 30

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        # Compute sensitivity: for each index, how much does it contribute to top autocorrelation peaks?
        sens = np.zeros(n)
        for k in top_k[:15]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[idx] * a[ki - idx]

        improved = False

        # Try moving mass from high-sensitivity positions to low-sensitivity positions
        high_sens = np.argsort(sens)[::-1]
        low_sens = np.argsort(sens)

        for src in high_sens[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue
            for dst in low_sens[:5]:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 15 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def pocs_optimize(a0, time_limit, verbose=False, rng_seed=0):
    """POCS: alternate between flat-spectrum projection and non-negativity projection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    fft_size = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    A_spec = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
    A_full = np.zeros(fft_size, dtype=np.complex128)
    A_full[:fft_size // 2 + 1] = A_spec
    for i in range(1, fft_size // 2):
        A_full[fft_size - i] = np.conj(A_spec[i])
    a = np.fft.irfft(A_full, fft_size)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = a0.copy()
    else:
        a = a / s

    ac = fast_autocorr(a)
    M = ac.max()
    if M < best_m:
        best_m = M
        best_a = a.copy()

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        target_mag = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
        alpha_blend = 0.1 + 0.5 * (it / min(it + 1, 100))
        new_mag = (1 - alpha_blend) * mag + alpha_blend * target_mag
        A_new = new_mag * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
        a = a_new

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    POCS it={it} ({elapsed:.0f}s): J={J:.6f}")

def lp_frank_wolfe_reduced(a0, time_limit, solver, verbose=False,
                           active_threshold=1e-5, top_K_init=300, top_K_max=None):
    """
    LP-Frank-Wolfe that only optimizes over the active set (nonzero entries).

        if M_curr < 1e-16:

        n_active = len(active)
        if n_active < 20:

            n_active = n

        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build reduced constraint matrix
        active_pos = np.searchsorted(active, np.arange(n))
        T_rows = []
        for k in binding_idx:

            j_start = max(0, ki - n + 1)

            mask = (active >= j_start) & (active < j_end)

                continue
            active_in_range = active[mask]
            row = np.zeros(n_active)
            for pos, j in zip(np.where(mask)[0], active_in_range):
                row[pos] = a[ki - j]
            T_rows.append(row)

        if len(T_rows) == 0:
            K = min(K + 50, top_K_max)
            continue

        T_sel = np.array(T_rows)

        b_var = cp.Variable(n_active, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]

            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            K = min(K + 50, top_K_max)
            iter_count += 1
            continue
        g_dir = b_val / b_sum

        base_alpha = 2.0 / (iter_count + 2.0)
        best_alpha = base_alpha
        best_trial_m = M_curr

        alphas = sorted(set([base_alpha, min(1.0, base_alpha * 0.5), min(1.0, base_alpha * 1.5),
                             0.1, 0.05]))
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        iter_count += 1
        if iter_count % 5 == 0:
            K = min(K + 40, top_K_max)
        if stagnation > 40:
            K = max(top_K_init, K // 2)
            stagnation = 0

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

    return best_a, best_m


def smooth_gradient_optimize(a0, time_limit, verbose=False, tau_init=10.0, tau_final=100.0,
                              eta_init=1.0, rng_seed=100):
    """Smoothed gradient descent with proper step size selection."""
    from scipy.signal import correlate as sp_correlate
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = eta_init
    stall = 0
    max_stall = 200
    total_iters = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1.0))
        tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(a)
        ac_max = ac.max()
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()

        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        cross = sp_correlate(p, a, mode='valid')
        g = a * S + cross
        g_tangent = g - g.mean()

        # Try multiple step sizes
        current_m = fast_autocorr(a).max()
        best_step = a.copy()
        best_trial_m = current_m
        for eta_try in [eta, eta * 0.5, eta * 0.2, eta * 2.0]:
            a_trial = a - eta_try * g_tangent
            a_trial = project_simplex(a_trial)
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_step = a_trial
                eta = eta_try

        a = best_step
        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                eta *= 0.7
            if stall > 60:
                pert = rng.normal(0, 0.003, n) * a
                a_pert = a + pert
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                m_pert = fast_autocorr(a_pert).max()
                if m_pert < best_m:
                    a = a_pert
                    best_m = m_pert
                    best_a = a.copy()
                    stall = 0
                    eta = eta_init * 0.3
                else:
                    a = project_simplex(a - 0.001 * g_tangent)

        total_iters += 1

        if verbose and total_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}, stall={stall}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Targeted coordinate descent for final polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 40
    n_top = 25

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_top]

        # Compute sensitivity
        sens = np.zeros(n)
        for k in top_k[:12]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[idx] * a[ki - idx]

        improved = False
        high_sens = np.argsort(sens)[::-1]
        low_sens = np.argsort(sens)

        for src in high_sens[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue
            for dst in low_sens[:4]:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator:
    - Reduced LP-FW (fast, many iterations)
    - Smooth gradient (smooths the landscape)
    - Full LP-FW (converges on full space)
    - Coordinate polish (final micro-optimization)
    - Length exploration
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Quick POCS ===
    print(f"\n=== Phase 1: Quick POCS (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 60.0)
    if t1 > 15:
        a_pocs, _ = pocs_optimize(best_seq, t1, verbose=False, rng_seed=7)
        try_update(a_pocs, "pocs")

    # === Phase 2: Reduced LP-FW (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Reduced LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        top_K_init = min(400, 2 * n_best - 1)
        top_K_max = min(2 * n_best - 1, 5000)
        a_out, _ = lp_frank_wolfe_reduced(a_in, t2, solver, verbose=True,
                                           active_threshold=1e-5,
                                           top_K_init=top_K_init,
                                           top_K_max=top_K_max)
        try_update(a_out, "lp_reduced_1")

    # === Phase 3: Smooth gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smooth Gradient (n={n_best}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_gradient_optimize(a_in, t3, verbose=True,
                                             tau_init=8.0, tau_final=60.0,
                                             eta_init=1.0, rng_seed=100)
        try_update(a_out, "smooth")

    # === Phase 4: Full LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Full LP-FW (n={n_best}) ===")
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        top_K_init = min(500, 2 * n_best - 1)
        top_K_max = min(2 * n_best - 1, 6000)
        a_out, _ = lp_frank_wolfe_full(a_in, t4, solver, verbose=True,
                                        top_K_init=top_K_init,
                                        top_K_max=top_K_max)
        try_update(a_out, "lp_full")

    # === Phase 5: Second reduced LP-FW (with lower threshold) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Reduced LP-FW r2 (n={n_best}) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        top_K_init = min(600, 2 * n_best - 1)
        top_K_max = min(2 * n_best - 1, 8000)
        a_out, _ = lp_frank_wolfe_reduced(a_in, t5, solver, verbose=True,
                                           active_threshold=1e-4,
                                           top_K_init=top_K_init,
                                           top_K_max=top_K_max)
        try_update(a_out, "lp_reduced_2")

    # === Phase 6: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coordinate Polish (n={n_best}) ===")
    t6 = min(remaining * 0.08, 120.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t6, verbose=True, rng_seed=456)
        try_update(a_out, "coord")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")

    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 1500]:
        new_n = n_best + delta
        if 1200 <= new_n <= 7500:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best][:2]

    for new_n in trial_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        t7a = min(remaining * 0.5, 150.0)
        if t7a > 25:
            a_in = norm_sum1(a_long)
            top_K_init = min(400, 2 * new_n - 1)
            top_K_max = min(2 * new_n - 1, 4000)
            a_out, _ = lp_frank_wolfe_reduced(a_in, t7a, solver, verbose=False,
                                               active_threshold=1e-5,
                                               top_K_init=top_K_init,
                                               top_K_max=top_K_max)
            try_update(a_out, f"lp_n{new_n}")

    # === Phase 8: Final LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP-FW (n={n_best}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.6, 250.0)
        a_in = norm_sum1(best_seq)
        top_K_init = min(800, 2 * n_best - 1)
        top_K_max = min(2 * n_best - 1, 10000)
        a_out, _ = lp_frank_wolfe_full(a_in, t8, solver, verbose=True,
                                        top_K_init=top_K_init,
                                        top_K_max=top_K_max)
        try_update(a_out, "lp_final")

    # === Phase 9: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 9: Final Polish (n={n_best}) ===")
        t9 = min(remaining * 0.8, 80.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t9, verbose=False, rng_seed=789)
            try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
