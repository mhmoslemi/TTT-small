"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFN (Fast Frank-Wolfe) with adaptive constraint management
2. Nesterov-accelerated iterations
3. Multi-scale optimization with periodic restarts

import numpy as np
import time
import cvxpy as cp


def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def get_solver():

    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:

def solve_lp(n, y, constraint_list, M_curr, solver):
    """
    Solve the LP: min M s.t. (y*b)_k <= M for k in constraint_list, sum(b)=1, b>=0
    Returns the optimal b direction and M value.
    """
    K = len(constraint_list)
    if K == 0:
        return None, M_curr
    
    # Build constraint matrix
    # (y*b)_k = sum_i y[i+k-n+1] * b[i] for appropriate i
    # More precisely: (y*b)_k = sum_{i=max(0,k-n+1)}^{min(n-1,k)} y[i] * b[k-i]
    # Wait, let me be more careful.
    # conv(y, b)[k] = sum_i y[i] * b[k-i] where b is defined for indices 0..n-1
    # = sum_{i=max(0,k-n+1)}^{min(n-1,k)} y[i] * b[k-i]
    
    # Actually, np.convolve(y, b)[k] = sum_i y[i] * b[k-i]
    # For k from 0 to 2n-2
    # The constraint is conv(y, b)[k] <= M
    
    # Let me use a different parameterization:
    # conv(y, b)[k] = sum_{j: 0<=j<n, 0<=k-j<n} y[j] * b[k-j]
    # = sum_{j: max(0,k-n+1) <= j <= min(n-1,k)} y[j] * b[k-j]
    
    # For the LP, we need the coefficient of b[i] in conv(y,b)[k]:
    # b[i] appears when k-j = i, i.e., j = k-i
    # So the coefficient of b[i] in row k is y[k-i] if 0 <= k-i < n, else 0.
    
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k = int(k)
        # b[i] coefficient is y[k-i] for 0 <= k-i < n
        # i.e., k-n < i <= k, and 0 <= i < n
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            idx = np.arange(i_min, i_max + 1)
            coeffs[jj, idx] = y[k - idx]
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(b_var) == 1, coeffs @ b_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, M_curr
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, M_curr
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    M_val = float(M_var.value) if M_var.value is not None else M_curr
    
    return b_val, M_val


def ffni_optimize(a0, time_limit, solver, initial_K=500, max_K=5000, verbose=True):
    """
    Frank-Wolfe with Nesterov acceleration for the autocorrelation minimization.
    """
    n = len(a0)
    y = norm_sum1(a0).copy()  # Nesterov iterate
    x = y.copy()  # primal iterate
    best_a = y.copy()
    
    ac = fast_autocorr(y)
    best_m = ac.max()
    t0 = time.monotonic()
    
    # Initial constraint set: top K lags
    size = 2 * n - 1
    init_idx = np.argsort(ac)[::-1][:min(initial_K, size)]
    constraint_set = set(int(k) for k in init_idx)
    
    iter_count = 0
    stall = 0
    stall_threshold = 200
    
    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        
        # Evaluate current point
        ac_x = fast_autocorr(x)
        m_x = ac_x.max()
        if m_x < best_m - 1e-14:
            best_m = m_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Check time
        if time.monotonic() - t0 > time_limit:
            break
        
        # Get current M from y
        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        
        # Update constraint set: add top violators of (y*b) against M_curr
        # First solve LP with current constraint set
        constraint_list = sorted(constraint_set)
        
        # Trim if too large
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac_y[k] for k in constraint_list])
            # Keep most binding (smallest margin)
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        # Solve LP
        b_dir, M_lp = solve_lp(n, y, constraint_list, M_curr, solver)
        
        if b_dir is None:
            # LP failed, try with fewer constraints
            constraint_list = sorted(constraint_set)[:100]
            b_dir, M_lp = solve_lp(n, y, constraint_list, M_curr, solver)
            if b_dir is None:
                iter_count += 1
                continue
        
        b_sum = b_dir.sum()
        if b_sum < 1e-15:
            b_dir = np.ones(n) / n
        else:
            b_dir = b_dir / b_sum
        
        # Check for new violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Bf = np.fft.rfft(b_dir, fft_size)
        conv_yb = np.fft.irfft(Yf * Bf, fft_size)[:size]
        
        violations = np.where(conv_yb > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_yb[violations] - M_curr
            # Add top violators
            top_v = violations[np.argsort(viol_amt)[::-1][:min(50, len(violations))]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # Also periodically add some random constraints to avoid missing important ones
        if iter_count % 100 == 50:
            random_lags = np.random.choice(size, min(20, size), replace=False)
            for k in random_lags:
                constraint_set.add(int(k))
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FFN iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}, M={M_curr:.6f}")
        
        # Restart if stalled
        if stall > stall_threshold:
            rng = np.random.default_rng(iter_count)
            pert = rng.normal(0, 0.01, n) * best_a
            x_restart = norm_sum1(best_a + pert)
            y = x_restart.copy()
            x = x_restart.copy()
            stall = 0
            stall_threshold = max(stall_threshold * 0.9, 50)
            # Reset constraint set
            ac_r = fast_autocorr(x_restart)
            init_idx = np.argsort(ac_r)[::-1][:min(initial_K, size)]
            constraint_set = set(int(k) for k in init_idx)
    
    return best_a, best_m

        c_max = ac_y.max()
        exp_c = np.exp(beta * (ac_y - c_max))
        w = exp_c / exp_c.sum()
        
        # Gradient: conv with w (reversed)
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Wf = np.fft.rfft(w, fft_size)
        Yf_rev = np.fft.rfft(y[::-1], fft_size)
        grad = 2.0 * np.fft.irfft(Wf * Yf_rev, fft_size)[n-1:2*n-1]
        
        # Step
        eta = eta0 / (np.sqrt(t) + 1.0)
        y_new = y - eta * grad
        y_new = np.maximum(y_new, 1e-30)
        y_new = y_new / y_new.sum()
        
        # Nesterov momentum
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = y_new + momentum * (y_new - y)
            x_new = np.maximum(x_new, 1e-30)
            x_new = x_new / x_new.sum()
        else:
            x_new = y_new.copy()
        
        m_new = fast_autocorr(x_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 3000:
            rng = np.random.default_rng(iter_count)
            pert = rng.normal(0, 0.01, n) * best_x
            x_restart = norm_sum1(best_x + pert)
            y = x_restart.copy()
            x_new = x_restart
            stall = 0
        
        y = x_new
        x = x_new
        iter_count += 1
        
        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}")
    
    return best_x, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Accelerated Frank-Wolfe (FFN) with adaptive constraint management
2. Interleaved FISTA with softmax gradient for local minimum escape

import numpy as np
import time
import cvxpy as cp


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS


def build_constraint_matrix(constraint_list, y, n):
    """Build the constraint matrix efficiently."""
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k = int(constraint_list[jj])
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            coeffs[jj, i_min:i_max+1] = y[k-i_min : k-i_max-1 : -1]
    return coeffs


def solve_lp_fast(n, y, constraint_list, solver):
    """Solve the LP and return the optimal direction."""
    K = len(constraint_list)
    if K == 0:
        return np.ones(n) / n
    
    coeffs = build_constraint_matrix(constraint_list, y, n)
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(b_var) == 1, coeffs @ b_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

    Accelerated Frank-Wolfe for autcorrelation minimization.
    Uses adaptive constraint set management.

    x = y.copy()

            stall = 0
        else:
            stall += 1

            break
        
        # Get current autocorrelation at y
        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()

        Yf = np.fft.rfft(y, fft_size)

        conv_yb = np.fft.irfft(Yf * Bf, fft_size)[:size]
        
        violations = np.where(conv_yb > M_curr + 1e-10)[0]

            constraint_set_arr = np.array(constraint_set, dtype=int)

        y_new = (1.0 - alpha) * y + alpha * b_dir

        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize array to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def get_solver():

            prob.solve(solver=s, verbose=False)

def build_constraint_matrix(constraint_list, y, n):
    """Build the LP constraint matrix: row k has y[k-i] at column i."""
    K = len(constraint_list)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj in range(K):
        k = int(constraint_list[jj])
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            coeffs[jj, i_min:i_max + 1] = y[k - i_min:k - i_max - 1:-1]
    return coeffs


def solve_lp_direction(n, y, constraint_list, solver):
    """Solve the LP to get the FW direction. Returns normalized b or None."""
    K = len(constraint_list)
    if K == 0:
        return np.ones(n) / n
    coeffs = build_constraint_matrix(constraint_list, y, n)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(b_var) == 1, coeffs @ b_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def subgrad_run(a0, time_limit, verbose=True):
    """
    Direct subgradient method on max autocorrelation.
    Projects onto the simplex after each step.
    """
    n = len(a0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    x = norm_sum1(a0).copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()

    for t in range(500000):
        if time.monotonic() - t0 > time_limit - 3:
            break
        ac = fast_autocorr(x)
        m = ac.max()
        if m < best_m - 1e-14:
            best_m = m
            best_a = x.copy()
        # Subgradient: sum of gradients at near-max lags
        eps = max(1e-8, 0.01 * m)
        active = np.where(ac >= m - eps)[0]
        grad = np.zeros(n)
        for k in active[:20]:  # Limit to 20 active lags for speed
            k = int(k)
            i_vals = np.arange(n)
            j_vals = k - i_vals
            valid = (j_vals >= 0) & (j_vals < n)
            if not valid.any():
                continue
            contrib = np.zeros(n)
            contrib[valid] = 2.0 * x[j_vals[valid]]
            diag = valid & (j_vals == i_vals)
            if diag.any():
                contrib[diag] = x[j_vals[diag]]
            grad += contrib
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm
        # Step with diminishing step size
        step = 0.05 / (1.0 + 0.05 * t)
        # Try multiple step sizes
        best_x = x
        best_val = m
        for mult in [0.5, 1.0, 2.0, 0.25]:
            x_trial = x - step * mult * grad
            x_trial = np.maximum(x_trial, 0.0)
            st = x_trial.sum()
            if st < 1e-15:
                continue
            x_trial = x_trial / st
            val = fast_autocorr(x_trial).max()
            if val < best_val:
                best_val = val
                best_x = x_trial
        x = best_x
        if verbose and (t + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SubGrad iter={t+1} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

        a_out, _ = ffni_run(norm_sum1(best_seq), t1, solver,
                            initial_K=500, max_K=3000, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Phase 1 improved: {best_score:.6f}")

    # === Phase 2: FFN with large constraints ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.28, 380.0)
    print(f"\n=== Phase 2: FFN K=2000 (n={len(best_seq)}, t={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = ffni_run(norm_sum1(best_seq), t2, solver,
                            initial_K=2000, max_K=5000, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Phase 2 improved: {best_score:.6f}")

    # === Phase 3: FISTA escape ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 120.0)
    print(f"\n=== Phase 3: FISTA escape (n={len(best_seq)}, t={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = fista_run(norm_sum1(best_seq), t3,
                             beta_start=10.0, beta_end=200.0, eta0=0.4, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Phase 3 improved: {best_score:.6f}")

    # === Phase 4: FFN with very large K (fine polish) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.22, 300.0)
    print(f"\n=== Phase 4: FFN K=4000 polish (n={len(best_seq)}, t={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = ffni_run(norm_sum1(best_seq), t4, solver,
                            initial_K=4000, max_K=6000, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Phase 4 improved: {best_score:.6f}")

    # === Phase 5: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.08, 80.0)
    print(f"\n=== Phase 5: Subgradient (n={len(best_seq)}, t={t5:.0f}s) ===")
    if t5 > 10:
        a_out, _ = subgrad_run(norm_sum1(best_seq), t5, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Phase 5 improved: {best_score:.6f}")

    # === Phase 6: Final FFN ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.30, 300.0)
    print(f"\n=== Phase 6: Final FFN (n={len(best_seq)}, t={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = ffni_run(norm_sum1(best_seq), t6, solver,
                            initial_K=3000, max_K=7000, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Phase 6 improved: {best_score:.6f}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        x_fb = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x_fb) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
