def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(y, beta, top_k=None):
    """Compute gradient of (1/beta)*log(sum top_k exp(beta*b)) w.r.t. a."""
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    # Gradient: (1/beta) * sum_k p_k * a_{k-j}
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, p, b


def nesterov_projected(a0, time_limit, beta_start=15.0, beta_end=800.0,
                       beta_growth=1.004, eta_base=0.8, top_k_ratio=0.05,
                       restart_every=3000, verbose=True):
    """Nesterov accelerated projected gradient with adaptive beta and restarts."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(99)
    top_k = max(20, int(n * top_k_ratio))

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Compute gradient
        grad, p, b = compute_smoothed_grad(y, beta, top_k)

        # Step size
        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.15)

        # Gradient step + projection
        z = y - eta * grad
        x_new = simplex_project(z)

        # Track best on true objective
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        # Adaptive beta increase
        if t % 80 == 0:
            beta = min(beta * beta_growth, beta_end)

        # Restart mechanism
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.02 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 5.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest restart t=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J))

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J) + ", beta=" + repr(beta))

    return best_a, best_max_b


def mirror_descent(a0, time_limit, beta=50.0, eta_start=0.05, eta_decay=0.9995,
                   restart_every=2000, verbose=True):
    """Mirror descent (multiplicative weights) on log-sum-exp smoothed objective."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_max_b = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = eta_start
    stall = 0
    rng = np.random.default_rng(200)

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        grad, p, b = compute_smoothed_grad(a, beta, top_k=None)

        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        b_new = fast_autocorr(a_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        eta *= eta_decay

        if stall > restart_every:
            stall = 0
            eta = eta_start
            pert = rng.normal(0, 0.015, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            a = a_r
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    MD restart t=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    MD t=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J))

    return best_a, best_max_b


def lp_fw_improved(a0, time_limit, solver, initial_K=400, max_K=3000,
                   add_per_iter=30, verbose=True):
    """Improved LP-Frank-Wolfe with better active set and line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 1200
    warm_start_prob = None
    warm_start_vars = {}

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: converged at iter " + str(t))
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations to add to active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5),
                  max(0.02, default_alpha * 1.5), 1.0 / (t + 1), 2.0 / (t + 1)]
        alphas = [max(0.01, min(1.0, a)) for a in alphas]
        alphas = list(set(alphas))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def coordinate_polish_v2(a0, time_limit, verbose=True):
    """Peak-targeted coordinate descent with adaptive strategy."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(40, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            src_val = a[src]
            partner_val = a[partner] if 0 <= partner < n else 0.0

            # Determine which to reduce
            if src_val >= partner_val:
                reduce_idx = src
                keep_idx = partner
            else:
                reduce_idx = partner
                keep_idx = src

            if a[reduce_idx] < 1e-12:
                continue

            # Find safe destination: low gradient positions
            # Compute "pressure" at each position
            pressure = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]

            # Exclude positions that are part of top peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)

            cand_pressure = pressure[candidates]
            dst = int(candidates[np.argmin(cand_pressure)])

            if dst == reduce_idx:
                dst = int(candidates[np.argmin(cand_pressure[candidates != reduce_idx])])
            if dst == reduce_idx:
                continue

            # Try different fractions
            found = False
            for frac in [0.01, 0.02, 0.04, 0.08, 0.15]:
                amt = a[reduce_idx] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 200:
                # Random exploration
                j = int(rng.integers(n))
                if j != reduce_idx and a[reduce_idx] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[reduce_idx] * frac
                    a[reduce_idx] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0:
                # Broader random moves
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:

                                stall = 0

        if verbose and stall % 400 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord v2 (" + str(elapsed) + "s): J=" + repr(J) + " stall=" + str(stall))

    return best_a, best_m


def parametric_init(n, time_limit, best_init, best_score):
    """Quick parametric search for good initialization."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score

    candidates = []
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    # Symmetric
    for p in [0.5, 1.0, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p * (n - i_arr) ** p)
        candidates.append(x ** p + (1.0 - x) ** p)
    # Gaussian
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    # Cosine
    for freq in [1, 2, 3]:
        candidates.append(np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x)) / 2.0, 0.01))
    # Sine
    for p in [1, 2, 3, 4]:
        candidates.append(np.sin(np.pi * x) ** p)

    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(a_raw)
        b = fast_autocorr(a)
        m = b.max()
        if m < best_val:
            best_val = m
