def norm_sum1(a):

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, ac, tau, top_k=100):
    """Gradient of (1/tau)*logsumexp(tau*ac[top_k]) w.r.t. a."""
    n = len(a)
    k_sel = np.argsort(ac)[::-1][:top_k]
    vals = ac[k_sel]
    max_val = vals.max()
    exp_vals = np.exp((vals - max_val) / tau)
    w = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for wi, k in zip(w, k_sel):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad / tau


def power_gradient(a, ac, top_k, beta):
    """Gradient using power-weighted top-k."""
    n = len(a)
    k_sel = np.argsort(ac)[::-1][:top_k]
    vals = np.maximum(ac[k_sel], 1e-30)
    w = vals ** beta
    w = w / w.sum()
    grad = np.zeros(n)
    for wi, k in zip(w, k_sel):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def mirror_descent_optimize(a0, time_limit, verbose=False, rng_seed=0, mode="lse"):
    """Adaptive mirror descent with log-sum-exp or power-weighted gradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    if mode == "lse":
        tau_start = 8.0
        tau_end = 0.5
        eta = 0.4
    else:
        eta = 0.5
        beta_start = 8.0
        beta_end = 3.0

    top_k_start = 60
    top_k_end = 120
    stall = 0
    max_stall = 400
    it = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1.0))

        ac = fast_autocorr(a)
        M = ac.max()

        if mode == "lse":
            tau = tau_start * (tau_end / tau_start) ** progress
            top_k = int(top_k_start + (top_k_end - top_k_start) * progress)
            grad = lse_gradient(a, ac, tau, top_k)
        else:
            beta = beta_start + (beta_end - beta_start) * progress
            top_k = int(top_k_start + (top_k_end - top_k_start) * progress)
            grad = power_gradient(a, ac, top_k, beta)

        # Multiplicative (mirror) update
        log_val = np.log(np.maximum(a, 1e-15)) - eta * grad
        max_log = log_val.max()
        a_new = np.exp(log_val - max_log)
        a_new = a_new / a_new.sum()

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0

        else:
            stall += 1
            if stall > 100:
                eta *= 0.85
            if stall > 200:

                pert = rng.normal(0, scale, n) * best_a
                a_p = np.maximum(best_a + pert, 1e-10)
                a = a_p / a_p.sum()
                stall = 100
            else:
                a = a_new

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD it={it} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K_init=200, top_K_max=None):
    """LP-based Frank-Wolfe with adaptive constraint count."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build constraint matrix
        T = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 20, top_K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Line search
        base_m = fast_autocorr(a).max()
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_m
        for alpha in [best_alpha, best_alpha * 1.5, best_alpha * 2.0, 0.1, 0.2, 0.3, 0.5, 0.7]:
            alpha = min(alpha, 1.0)
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1
        if iter_count % 8 == 0:
            K = min(K + 25, top_K_max)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW it={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m

def spectral_refine(a0, time_limit, verbose=False, rng_seed=0):

        mag = np.abs(A)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_u_shape(n, sharpness=2.0):

def make_bimodal(n, width=0.15):

def make_edge_conc(n, power=3.0):
    """Edge-concentrated: a[i] ~ (distance to center)^power"""
    x = np.linspace(-1, 1, n)
    a = np.abs(x) ** power + 0.01
    return norm_sum1(a)


def make_poly(n, coeffs):
    """Polynomial shape: a[i] = sum c_j * (2*i/(n-1) - 1)^j"""
    x = 2.0 * np.arange(n) / max(n - 1, 1) - 1.0
    a = np.ones(n)
    for c, p in zip(coeffs, range(len(coeffs))):
        a += c * x ** p
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5500
        best_seq = make_u_shape(n_base, sharpness=2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Quick mirror descent on parent ===
    print(f"\n=== Phase 1: Mirror Descent (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 180.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_optimize(a_in, t1, verbose=True, rng_seed=1, mode="lse")
        try_update(a_out, "md_lse_1")

    # === Phase 2: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 40 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        K_init = min(300, 2 * n_best - 1)
        K_max = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, top_K_init=K_init, top_K_max=K_max)
        try_update(a_out, "lp_1")

    # === Phase 3: Targeted coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord Descent (n={n_best}) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = targeted_coord_descent(a_in, t3, verbose=True, rng_seed=42)
        try_update(a_out, "coord_1")

    # === Phase 4: Power-weighted mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Power MD (n={n_best}) ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_optimize(a_in, t4, verbose=True, rng_seed=2, mode="power")
        try_update(a_out, "md_power_1")

    # === Phase 5: Second LP round ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW r2 (n={n_best}) ===")
    t5 = min(remaining * 0.13, 200.0)
    if t5 > 40 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        K_init = min(500, 2 * n_best - 1)
        K_max = min(5000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True, top_K_init=K_init, top_K_max=K_max)
        try_update(a_out, "lp_2")

    # === Phase 6: Spectral refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Spectral (n={n_best}) ===")
    t6 = min(remaining * 0.08, 120.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_refine(a_in, t6, verbose=True, rng_seed=10)
        try_update(a_out, "spectral_1")

    # === Phase 7: Second coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Coord r2 (n={n_best}) ===")
    t7 = min(remaining * 0.08, 120.0)
    if t7 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = targeted_coord_descent(a_in, t7, verbose=True, rng_seed=123)
        try_update(a_out, "coord_2")

    # === Phase 8: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Length Exploration ===")

    trial_lengths = []
    for delta in [300, -300, 600, -600, 1000, -1000, 1500, -1500, 2000, -2000, 3000, -3000]:
        new_n = n_best + delta
        if 1000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)...")
        a_interp = interpolate_sequence(best_seq, new_n)

        # Quick MD
        t8a = min(remaining * 0.25, 100.0)
        if t8a > 15:
            a_out, _ = mirror_descent_optimize(a_interp, t8a, verbose=False, rng_seed=new_n % 1000, mode="lse")
            try_update(a_out, f"md_n{new_n}")

        # LP if time allows
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 5000:
            t8b = min(remaining * 0.3, 80.0)
            if t8b > 15:
                a_in = interpolate_sequence(best_seq, new_n)
                K_init = min(200, 2 * new_n - 1)
                K_max = min(2000, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe(a_in, t8b, solver, verbose=False, top_K_init=K_init, top_K_max=K_max)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 9: Final refinement on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final Refinement (n={n_best}) ===")
    if remaining > 30:
        t9a = min(remaining * 0.25, 100.0)
        if t9a > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = mirror_descent_optimize(a_in, t9a, verbose=True, rng_seed=999, mode="lse")
            try_update(a_out, "md_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and n_best <= 6000:
            t9b = min(remaining * 0.3, 100.0)
            if t9b > 20:
                a_in = norm_sum1(best_seq)
                K_init = min(800, 2 * n_best - 1)
                K_max = min(8000, 2 * n_best - 1)
                a_out, _ = lp_frank_wolfe(a_in, t9b, solver, verbose=True, top_K_init=K_init, top_K_max=K_max)
                try_update(a_out, "lp_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t9c = min(remaining * 0.3, 80.0)
            if t9c > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = targeted_coord_descent(a_in, t9c, verbose=True, rng_seed=777)
                try_update(a_out, "coord_final")

    # === Phase 10: Final polish with very small steps ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 10: Final Polish (n={n_best}) ===")
        t10 = min(remaining * 0.8, 60.0)
        if t10 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = targeted_coord_descent(a_in, t10, verbose=False, rng_seed=555)
            try_update(a_out, "polish")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5500
        best_seq = make_u_shape(n_fb, 2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
