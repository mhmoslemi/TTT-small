def build_T_sparse(a, k_indices):
    """Build sparse matrix T where T[k_idx, j] = a[k_idx - j] if 0 <= k_idx-j < n."""

    cols = []

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def lp_fw_optimize(a0, time_limit, solver, K_init=100, K_max=None, K_growth=100,
                   verbose=True, rng_seed=42):
    """LP-Frank-Wolfe optimization with adaptive K."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    if K_max is None:
        K_max = 2 * n - 1
    
    K = K_init
    it = 0
    alpha = 1.0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Select top K constraints
        K_sel = min(K, 2 * n - 1)
        if K_sel < 2 * n - 1:
            k_indices = np.argsort(ac)[::-1][:K_sel]
        else:
            k_indices = np.arange(2 * n - 1)
        
        b_dir = lp_fw_step(a, M_curr, k_indices, solver)
        if b_dir is None:
            K = min(K + 200, K_max)
            continue
        
        # Line search for step size
        best_alpha = 2.0 / (it + 2.0)
        best_m_trial = M_curr
        for trial_alpha in [best_alpha, min(1.0, best_alpha * 2), min(1.0, best_alpha * 3), 0.5, 0.3, 0.2, 0.1, 0.7, 0.8]:
            a_trial = (1 - trial_alpha) * a + trial_alpha * b_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_m_trial:
                best_m_trial = m_trial
                best_alpha = trial_alpha
        
        a = (1 - best_alpha) * a + best_alpha * b_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a
        
        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()
        
        it += 1
        
        # Grow K periodically
        if it % 10 == 0:
            K = min(K + K_growth, K_max)
        
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW it={it} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent polish: move mass from high-impact positions to low-impact ones."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 50
    n_top = 20
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]
        
        # Compute sensitivity: for each position, how much does it contribute to peaks?
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:10]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            idx = np.arange(s, e)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]
        
        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s_idx = max(0, ki - n + 1)
            e_idx = min(n, ki + 1)
            idx = np.arange(s_idx, e_idx)
            contrib = a[idx] * a[ki - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find low-sensitivity destination
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            top_cands = candidates[np.argsort(cand_sens)[:5]]
            
            found = False
            for dst in top_cands:
                if dst == src or found:
                    continue
                for frac in [0.01, 0.005, 0.02, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        else:
            stall = 0
        
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def spectral_init(n, seed=0):
    """Initialize with flat spectrum + random phases."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    A_spec = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
    A_full = np.zeros(fft_size, dtype=np.complex128)
    A_full[:fft_size//2 + 1] = A_spec
    for i in range(1, fft_size // 2):
        A_full[fft_size - i] = np.conj(A_spec[i])
    a = np.fft.irfft(A_full, fft_size)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_cosine(n):
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = a ** 2
    return norm_sum1(a)


def make_u_shape(n, sharpness=2.0):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def evaluate_safe(a):

        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def get_T_structure(n, K_idx):
    """Get or cache the row/col structure of the Toeplitz block."""
    key = (n, tuple(K_idx.tolist()))
    if key in _T_structure_cache:
        return _T_structure_cache[key]
    rows = []
    cols = []
    for ii in range(len(K_idx)):
        ki = int(K_idx[ii])
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        length = j_end - j_start
        if length > 0:
            rows.extend([ii] * length)
            cols.extend(range(j_start, j_end))
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    _T_structure_cache[key] = (rows, cols)
    return (rows, cols)


def build_T(a, k_indices):
    """Build sparse Toeplitz matrix T[k,j] = a[k-j] for valid (k,j)."""
    n = len(a)
    K = len(k_indices)
    rows, cols = get_T_structure(n, k_indices)
    # Compute values: for entry (row ii, col j), value = a[k_indices[ii] - j]
    k_arr = np.asarray(k_indices, dtype=np.int64)
    k_expanded = np.repeat(k_arr, np.diff(np.concatenate([[-1], np.cumsum([min(n, int(k_indices[ii]+1)) - max(0, int(k_indices[ii]-n+1))] for ii in range(K))]))])
    # Recompute properly
    vals = np.empty(len(rows), dtype=np.float64)
    for ii in range(K):
        ki = int(k_indices[ii])
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        length = j_end - j_start
        if length > 0:
            mask = (rows == ii)
            vals[mask] = a[ki - cols[mask]]
    coo = coo_matrix((vals, (rows, cols)), shape=(K, n))
    return coo.tocsr()


def build_T_simple(a, k_indices):
    """Simpler (slower) build for correctness."""
    n = len(a)
    K = len(k_indices)
    r = []
    c = []
    v = []
    for ii in range(K):
        ki = int(k_indices[ii])
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        for j in range(j_start, j_end):
            r.append(ii)
            c.append(j)
            v.append(a[ki - j])
    r = np.array(r, dtype=np.int32)
    c = np.array(c, dtype=np.int32)
    v = np.array(v, dtype=np.float64)
    return coo_matrix((v, (r, c)), shape=(K, n)).tocsr()


def solve_lp_fw(a, M, k_indices, solver):
    """Solve: max sum(b) s.t. T@b <= M, b >= 0. Return b/sum(b) or None."""
    n = len(a)
    K = len(k_indices)
    if K == 0:
        return None
    T = build_T_simple(a, k_indices)
    b = cp.Variable(n, nonneg=True)
    constraints = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def lp_fw_optimize(a0, time_limit, solver, K_init=50, K_max=2000, K_growth=80,
                   verbose=True, rng_seed=42):
    """LP-Frank-Wolfe with adaptive constraint set size."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    K = K_init
    K_max = min(K_max, 2 * n - 1)
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 5:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if K_sel < 2 * n - 1:
            k_indices = np.argsort(ac)[::-1][:K_sel]
        else:
            k_indices = np.arange(2 * n - 1)

        b_dir = solve_lp_fw(a, M_curr, k_indices, solver)
        if b_dir is None:
            K = min(K + K_growth, K_max)
            it += 1
            continue

        # Line search
        best_alpha = 2.0 / (it + 2.0)
        best_m_trial = M_curr
        alphas = [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2),
                  0.5, 0.3, 0.2, 0.1, 0.7, 0.8, 0.9]
        for trial_alpha in alphas:
            if trial_alpha <= 0 or trial_alpha > 1:
                continue
            a_trial = (1 - trial_alpha) * a + trial_alpha * b_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_m_trial:
                best_m_trial = m_trial
                best_alpha = trial_alpha

        a = (1 - best_alpha) * a + best_alpha * b_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        it += 1
        if it % 8 == 0:
            K = min(K + K_growth, K_max)

        # Restart if stuck
        if no_improve > 30:
            pert = rng.normal(0, 0.02, n) * best_a
            a_r = np.maximum(best_a + pert, 1e-10)
            a = a_r / a_r.sum()
            no_improve = 0
            K = max(K // 2, K_init)

        if verbose and it % 10 == 0:
            J = 2.0 * n * best_m
            print(f"    LP-FW it={it} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}, noimp={no_improve}")

    return best_a, best_m


def softmax_gradient_step(a, tau, step_size):
    """One smooth gradient step using soft-max approximation."""
    n = len(a)
    ac = fast_autocorr(a)
    ac_max = ac.max()
    p = np.exp((ac - ac_max) / tau)
    p = p / p.sum()

    # Gradient: g[i] = sum_k p[k] * a[k-i] (for valid k-i)
    g = np.zeros(n)
    # Use FFT for efficiency
    # g = IFFT(FFT(p[0:n]) * FFT(a)) approximately (circular)
    # But we need linear. Let's do it directly for the top peaks.
    top_k = np.argsort(p)[::-1][:50]
    for k in top_k:
        ki = int(k)
        pk = p[ki]
        if pk < 1e-10:
            continue
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        for j in range(j_start, j_end):
            g[j] += pk * a[ki - j]

    return g


def smooth_grad_optimize(a0, time_limit, verbose=False, rng_seed=100):
    """Smooth gradient descent with simplex projection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    def project_simplex(v):
        nn = len(v)
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, nn + 1) > (cssv - 1))[0]
        if len(rho) == 0:
            return np.ones(nn) / nn
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)

    it = 0
    stall = 0
    tau = 5.0
    eta = 1.0

    while time.monotonic() - t0 < time_limit and stall < 200:
        if time.monotonic() - t0 > time_limit - 3:
            break
        g = softmax_gradient_step(a, tau, eta)
        g_t = g - g.mean()  # project off constant
        a_trial = a - eta * g_t
        a_new = project_simplex(a_trial)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.05, 3.0)
        else:
            stall += 1
            if stall % 20 == 0:
                eta *= 0.8
                tau *= 0.9
            if stall > 50:
                pert = rng.normal(0, 0.01, n) * best_a
                a_p = np.maximum(best_a + pert, 0)
                a = a_p / a_p.sum()
                stall = 0

        a = a_new
        it += 1
        tau = max(tau * 0.995, 0.5)

        if verbose and it % 100 == 0:
            J = 2.0 * n * best_m
            print(f"    SmoothGrad it={it} ({time.monotonic()-t0:.0f}s): J={J:.6f}, eta={eta:.4f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent: move mass from high-contribution to low-contribution positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 60
    n_top = 15

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_top]

        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:8]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            idx = np.arange(s, e)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s_idx = max(0, ki - n + 1)
            e_idx = min(n, ki + 1)
            idx = np.arange(s_idx, e_idx)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            order = np.argsort(sens[candidates])
            for dst_idx in order[:5]:
                dst = candidates[dst_idx]
                if dst == src:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.1, 0.03]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        else:
            stall = 0

    return best_a, best_m


def spectral_init(n, seed=0):
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    A_spec = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
    A_full = np.zeros(fft_size, dtype=np.complex128)
    A_full[:fft_size // 2 + 1] = A_spec
    for i in range(1, fft_size // 2):
        A_full[fft_size - i] = np.conj(A_spec[i])
    a = np.fft.irfft(A_full, fft_size)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_cosine(n):
    x = np.linspace(0, 1, n)
    a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 2
    return norm_sum1(a)


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search: LP-Frank-Wolfe (adaptive K) as primary optimizer with
    smooth-gradient, coordinate polish, and length exploration as supplements.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 6000
        best_seq = make_cosine(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: LP-FW Main (50% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.50, 880.0)
    print(f"\n=== Phase 1: LP-FW Main (n={n0}, t={t1:.0f}s) ===")
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        K_max = min(3000, 2 * n0 - 1)
        a_out, _ = lp_fw_optimize(a_in, t1, solver, K_init=40, K_max=K_max,
                                   K_growth=60, verbose=True, rng_seed=42)
        try_update(a_out, "lp_main")

    # === Phase 2: Smooth gradient from best ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 2: Smooth Gradient (t={t2:.0f}s) ===")
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_grad_optimize(a_in, t2, verbose=True, rng_seed=200)
        try_update(a_out, "smooth1")

    # === Phase 3: LP-FW from alt init ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 3: LP-FW Alt (t={t3:.0f}s) ===")
    if t3 > 50:
        # Create diverse alt init
        a_spec = spectral_init(n0, seed=314)
        a_cos = make_cosine(n0)
        a_alt = 0.4 * norm_sum1(best_seq) + 0.3 * a_spec + 0.3 * a_cos
        a_alt = norm_sum1(a_alt)
        a_out, _ = lp_fw_optimize(a_alt, t3, solver, K_init=50,
                                   K_max=min(2500, 2 * n0 - 1), K_growth=80,
                                   verbose=True, rng_seed=555)
        try_update(a_out, "lp_alt")

    # === Phase 4: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 4: Coord Polish (t={t4:.0f}s) ===")
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, verbose=False, rng_seed=789)
        try_update(a_out, "coord1")

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length Exploration ===")
    trial_lengths = []
    for delta in [400, -400, 800, -800, 1200, -1200, 2000, -2000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best][:3]

    for new_n in trial_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)
        t5a = min(remaining * 0.4, 120.0)
        if t5a > 30:
            a_out, _ = lp_fw_optimize(a_interp, t5a, solver, K_init=40,
                                       K_max=min(1500, 2 * new_n - 1), K_growth=80,
                                       verbose=False, rng_seed=new_n % 997)
            try_update(a_out, f"lp_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t5b = min(remaining * 0.3, 60.0)
            if t5b > 15:
                a_out, _ = coordinate_polish(a_interp, t5b, verbose=False, rng_seed=new_n % 500)
                try_update(a_out, f"coord_n{new_n}")

    # === Phase 6: Final LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.20, 200.0)
    print(f"\n=== Phase 6: Final LP-FW (n={n_best}, t={t6:.0f}s) ===")
    if t6 > 30:
        a_in = norm_sum1(best_seq)
        K_init = min(300, 2 * n_best - 1)
        K_max = min(4000, 2 * n_best - 1)
        a_out, _ = lp_fw_optimize(a_in, t6, solver, K_init=K_init, K_max=K_max,
                                   K_growth=200, verbose=True, rng_seed=1111)
        try_update(a_out, "lp_final")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t7 = min(remaining * 0.5, 80.0)
        print(f"\n=== Phase 7: Final Polish (n={n_best}, t={t7:.0f}s) ===")
        if t7 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t7, verbose=False, rng_seed=444)
            try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 6000
        best_seq = make_cosine(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
