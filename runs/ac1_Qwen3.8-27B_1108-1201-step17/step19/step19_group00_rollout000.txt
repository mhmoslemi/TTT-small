import time

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    # Find threshold
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    rho = np.where(v_sorted > (cumsum - target) / np.arange(1, n+1))[0]
    if len(rho) == 0:
        return np.full(n, target/n)
    rho = rho[-1]
    tau = (cumsum[rho] - target) / (rho + 1)
    return np.maximum(v - tau, 0.0)

def subgradient_conv(a, k):
    """Subgradient of (a*a)[k] w.r.t. a: 2*a[k-i] for valid i."""
    n = len(a)
    grad = np.zeros(n)
    ki = int(k)
    i_start = max(0, ki - n + 1)
    i_end = min(n, ki + 1)
    if i_start < i_end:
        idx = np.arange(i_start, i_end)
        grad[idx] = 2.0 * a[ki - idx]
    return grad

def obj_val(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_conv(a)
    return 2.0 * n * c.max() / (s*s)

import numpy as np

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target=1.0):
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    rho = np.where(v_sorted > (cumsum - target) / np.arange(1, n + 1))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = int(rho[-1])
    tau = (cumsum[rho] - target) / (rho + 1)
    return np.maximum(v - tau, 0.0)

def obj_val(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_conv(a)
    return 2.0 * n * c.max() / (s * s)

def normalize(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def mirror_descent_step(y, grad, eta, a):
    """Multiplicative weight update: a_new = a * exp(-eta*grad) normalized."""
    log_ratio = -eta * grad

    w = np.exp(log_ratio)
    a_new = a * w

def subgradient_conv(a, active_lags, weights=None):
    """Weighted sum of subgradients 2*a[k-i] for valid i, over active lags."""
    n = len(a)
    grad = np.zeros(n)
    if weights is None:

    for k, w in zip(active_lags, weights):
        ki = int(k)
        i_s = max(0, ki - n + 1)
        i_e = min(n, ki + 1)
        if i_s < i_e:
            idx = np.arange(i_s, i_e)
            grad[idx] += w * 2.0 * a[ki - idx]
    return grad
