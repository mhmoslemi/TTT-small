    grad = 2.0 * corr[:n]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full soft-max gradient via FFT (O(n log n) per step)
2. Adaptive temperature beta scheduling
3. Mirror descent (multiplicative) updates with momentum
4. Frequency-domain spectral flattening
5. Multi-phase: explore -> exploit -> polish
6. Diverse restarts and length exploration

import numpy as np
import time


def _next_pow2(n):
    return 1 << (n - 1).bit_length()


def _fast_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = _next_pow2(size)
    A = np.fft.rfft(a.astype(np.float64), N)
    return np.fft.irfft(A * A, N)[:size]


def _norm_nonneg(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _eval_safe(a):
    """Safely call the evaluation function."""
    try:
        a_list = [float(x) for x in a]
        val = float(evaluate_sequence(a_list))
        return val
    except Exception:
        return float('inf')


def _score_from_ac(a):
    """Compute 2*n*max(ac)/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    ac = _fast_conv(a)
    return float(2.0 * n * ac.max() / (s * s))


def _softmax_gradient(a, beta, fft_size_sm, fft_size_grad):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*b_k)) via FFT.
    Returns (grad, b_max, p) where p is the softmax weight vector.
    """
    n = len(a)
    b = _fast_conv(a)
    b_max = b.max()
    # Numerically stable softmax
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = e.sum()
    if e_sum < 1e-30:
        p = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        p = e / e_sum

    # Gradient: grad[j] = 2 * sum_m a[m] * p[m+j] for j=0..n-1
    # c[j] = sum_m a[m] * p[m+j] = conv(a, p[::-1])[2n-2-j]
    q = p[::-1]
    A = np.fft.rfft(a, fft_size_grad)
    Q = np.fft.rfft(q, fft_size_grad)
    conv_result = np.fft.irfft(A * Q, fft_size_grad)

    # c[j] = conv_result[2n-2-j] for j=0..n-1
    # So c = conv_result[n-1 : 2n-1][::-1]
    c = conv_result[n - 1:2 * n - 1][::-1]
    grad = 2.0 * c
    return grad, b_max, p


def _mirror_descent_full(a0, time_limit, beta_init=3.0, beta_end=40.0,
                         lr_init=0.3, momentum_init=0.3, rng_seed=42,
                         verbose=True):
    """
    Full soft-max mirror descent with adaptive beta and momentum.
    """
    n = len(a0)
    fft_size_sm = _next_pow2(2 * n - 1)
    fft_size_grad = _next_pow2(3 * n - 2)

    a = _norm_nonneg(a0.copy())
    best_a = a.copy()
    ac_best = _fast_conv(a)
    best_m = ac_best.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    beta = beta_init
    lr = lr_init
    momentum = np.zeros(n)
    mu = momentum_init
    stall = 0
    best_beta = beta

    total_iters = 0
    for t in range(1, 10**9):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive beta: increase gradually
        progress = min(1.0, t / max(1, int(time_limit * 2000)))
        beta = beta_init + (beta_end - beta_init) * progress * 0.8
        # Oscillate slightly to avoid getting stuck
        beta = beta * (1.0 + 0.1 * np.sin(0.01 * t))
        beta = max(beta, beta_init)

        # Compute gradient
        grad, m_curr, p = _softmax_gradient(a, beta, fft_size_sm, fft_size_grad)

        # Normalize gradient for stable updates
        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad_norm = grad / gn
        else:
            grad_norm = grad

        # Momentum update
        mu = 0.95 * mu + 0.05 * 0.3
        momentum = mu * momentum + (1 - mu) * grad_norm

        # Adaptive learning rate
        lr = lr_init / (1.0 + 0.002 * t)

        # Multiplicative (mirror) update
        log_update = -lr * momentum
        log_update -= log_update.max()  # numerical stability
        log_update = np.clip(log_update, -50, 50)
        a_new = a * np.exp(log_update)
        a_new = _norm_nonneg(a_new)

        total_iters += 1

        # Check improvement
        ac_new = _fast_conv(a_new)
        m_new = ac_new.max()
        if m_new < best_m - 1e-13:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            best_beta = beta
        else:
            stall += 1
            # Still update a (for momentum continuity) but with damping
            a = _norm_nonneg(0.99 * a_new + 0.01 * a)
            if stall > 3000:
                # Restart with perturbation
                pert_type = int(rng.integers(5))
                if pert_type == 0:
                    # Gaussian bump
                    x = np.linspace(0, 1, n)
                    c = float(rng.uniform(0.2, 0.8))
                    sig = float(rng.uniform(0.05, 0.3))
                    pert = np.exp(-((x - c) ** 2) / (2 * sig ** 2))
                elif pert_type == 1:
                    # Random smooth
                    freqs = rng.uniform(0, 5, min(5, n))
                    phases = rng.uniform(0, 2 * np.pi, min(5, n))
                    x = np.linspace(0, 1, n)
                    pert = 1.0
                    for f, ph in zip(freqs[:3], phases[:3]):
                        pert += 0.3 * np.sin(2 * np.pi * f * x + ph)
                    pert = np.maximum(pert, 0.01)
                elif pert_type == 2:
                    # Power law
                    i_arr = np.arange(n, dtype=np.float64)
                    p_exp = float(rng.uniform(-2, 2))
                    pert = (i_arr + 1.0) ** p_exp
                    pert = np.maximum(pert, 0.0)
                elif pert_type == 3:
                    # Symmetric random
                    half = n // 2
                    h = np.abs(rng.normal(1.0, 0.3, half + 1))
                    pert = np.concatenate([h, h[-2::-1]])[:n]
                else:
                    # Small noise on best
                    pert = best_a * (1.0 + rng.normal(0, 0.05, n))

                pert = _norm_nonneg(pert)
                w = float(rng.uniform(0.2, 0.5))
                a = _norm_nonneg((1 - w) * best_a + w * pert)
                momentum = np.zeros(n)
                mu = momentum_init
                lr = lr_init
                stall = 0
                beta = max(beta_init, beta * 0.8)

                if verbose:
                    J = 2.0 * n * best_m
                    print(f"    MD restart t={t} ({time.monotonic()-t0:.0f}s): "
                          f"J={J:.6f}, beta={beta:.1f}, iters={total_iters}")

        # Periodic verbose output
        if verbose and t % 10000 == 0:
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({time.monotonic()-t0:.0f}s): J={J:.6f}, "
                  f"beta={beta:.1f}, stall={stall}")

    if verbose:
        J = 2.0 * n * best_m
        print(f"    MD done ({time.monotonic()-t0:.0f}s): J={J:.6f}, iters={total_iters}")
    return best_a, best_m


def _spectral_flatten(a0, time_limit, lr_init=0.5, beta_spec=2.0, rng_seed=55,
                      verbose=False):
    """
    Flatten the power spectrum to reduce autocorrelation peaks.
    Works in frequency domain: push energy from spectral peaks to valleys.
    """
    n = len(a0)
    a = _norm_nonneg(a0.copy())
    best_a = a.copy()
    best_m = _fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    fft_size = _next_pow2(2 * n - 1)
    lr = lr_init
    stall = 0

    for t in range(1, 10**9):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute spectrum
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        if P_max < 1e-30:
            break

        # Soft-max weighting on spectrum
        P_shifted = beta_spec * (P - P_max)
        P_shifted = np.clip(P_shifted, -50, 50)
        w = np.exp(P_shifted)
        w_mean = w.mean()

        # Damp spectral peaks, boost valleys
        factor = 1.0 - lr / (1.0 + 0.01 * t) * (w / w_mean - 1.0)
        factor = np.clip(factor, 0.5, 1.5)
        A_new = A * factor

        # Transform back and project
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = _norm_nonneg(a_new)

        m_new = _fast_conv(a_new).max()
        if m_new < best_m - 1e-13:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 2000:
                # Perturb and restart
                a = _norm_nonneg(best_a * (1.0 + rng.normal(0, 0.02, n)))
                stall = 0
                lr *= 0.8
                beta_spec = max(1.0, beta_spec * 0.9)

    if verbose:
        J = 2.0 * n * best_m
        print(f"    Spectral done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def _coord_descent_polish(a0, time_limit, rng_seed=77, verbose=False):
    """
    Coordinate descent: try moving mass from high-impact to low-impact positions.
    """
    n = len(a0)
    a = _norm_nonneg(a0.copy())
    best_a = a.copy()
    best_m = _fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        ac = _fast_conv(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-13:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10000:
                break

        # Find the top autocorrelation lag
        k_peak = int(np.argmax(ac))
        i_lo = max(0, k_peak - n + 1)
        i_hi = min(n, k_peak + 1)

        if i_lo >= i_hi:
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                amt = a[i] * 0.01
                a[i] -= amt
                a[j] += amt
                a = _norm_nonneg(a)
            continue

        idx = np.arange(i_lo, i_hi)
        j_idx = k_peak - idx
        contrib = a[idx] * a[j_idx]

        # Find top contributing pair
        top_pi = int(np.argmax(contrib))
        src = int(idx[top_pi])

        # Find best destination: minimize impact on peak
        # Impact of increasing a[dst]: contributes to ac at lags dst, k_peak-dst, etc.
        impact = np.zeros(n)
        for k in range(max(0, k_peak - n + 1), min(2 * n - 1, k_peak + n)):
            ii = np.arange(max(0, k - n + 1), min(n, k + 1))
            jj = k - ii
            impact[ii] += a[jj] * ac[k]
            impact[jj] += a[ii] * ac[k]

        impact[src] = np.inf
        dst = int(np.argmin(impact))

        if src == dst or a[dst] > 0.9 * a[src]:
            # Try random small move
            dst = int(rng.integers(n))
            if dst == src:
                continue

        # Try transfer
        for frac in [0.01, 0.02, 0.03, 0.05, 0.08, 0.12]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = _norm_nonneg(a_trial)
            m_trial = _fast_conv(a_trial).max()
            if m_trial < best_m - 1e-14:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                stall = 0
                break
        else:
            if stall % 500 == 0:
                # Random perturbation to escape
                n_moves = int(rng.integers(3, 10))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        amt = a[i] * float(rng.uniform(0.005, 0.02))
                        a[i] -= amt
                        a[j] += amt
                a = _norm_nonneg(a)
                m_new = _fast_conv(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        J = 2.0 * n * best_m
        print(f"    Coord done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a


def _interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def _diverse_init(n, rng):
    """Generate a diverse set of initial sequences."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []

    # Uniform
    candidates.append(np.ones(n) / n)

    # Slight asymmetric
    for alpha in [0.1, 0.2, 0.3]:
        candidates.append(_norm_nonneg(np.ones(n) + alpha * np.sin(np.pi * x)))
        candidates.append(_norm_nonneg(np.ones(n) + alpha * np.cos(np.pi * x)))

    # Gaussian centered
    for c in [0.3, 0.5, 0.7]:
        for sig in [0.15, 0.25, 0.35]:
            candidates.append(_norm_nonneg(
                np.exp(-((x - c) ** 2) / (2 * sig ** 2)) + 0.3))

    # Bimodal
    for gap in [0.3, 0.4, 0.5]:
        for sig in [0.08, 0.12, 0.18]:
            c1 = 0.5 - gap / 2
            c2 = 0.5 + gap / 2
            candidates.append(_norm_nonneg(
                np.exp(-((x - c1) ** 2) / (2 * sig ** 2)) +
                np.exp(-((x - c2) ** 2) / (2 * sig ** 2)) + 0.2))

    # Power laws (slight)
    for p in [0.3, -0.3, 0.7, -0.7]:
        candidates.append(_norm_nonneg((i_arr + 1.0) ** p + 0.5))

    # Random smooth
    for _ in range(5):
        freqs = np.sort(rng.uniform(0, 4, 4))
        amps = rng.uniform(0.1, 0.5, 4)
        phases = rng.uniform(0, 2 * np.pi, 4)
        sig = np.ones(n)
        for f, a_, ph in zip(freqs, amps, phases):
            sig += a_ * np.sin(2 * np.pi * f * x + ph)
        candidates.append(_norm_nonneg(sig))

    return candidates

        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")

        a_interp = _interpolate_sequence(best_seq, new_n)
        t5a = min(remaining * 0.4, 180.0)
        if t5a > 30:
            a_out, _ = _mirror_descent_full(
                a_interp, t5a,
                beta_init=5.0, beta_end=35.0,
                lr_init=0.25, momentum_init=0.3,
                rng_seed=new_n % 1000, verbose=False)
            try_update(a_out, "md_n{}".format(new_n))

    # === Phase 6: Diverse starts (different initializations) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Diverse starts (budget={remaining:.0f}s) ===")
    if remaining > 120:
        n_div = len(best_seq)
        cands = _diverse_init(n_div, rng)
        # Score them all quickly
        cand_scores = []
        for c in cands:
            s = _score_from_ac(c)
            cand_scores.append(s)
        # Sort by score
        order = np.argsort(cand_scores)
        # Try top 3 diverse initializations
        for idx in order[:3]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            c = cands[idx]
            # Blend with current best
            blend = _norm_nonneg(0.5 * best_seq + 0.5 * c)
            t6a = min(remaining * 0.3, 150.0)
            if t6a > 20:
                a_out, _ = _mirror_descent_full(
                    blend, t6a,
                    beta_init=5.0, beta_end=40.0,
                    lr_init=0.2, momentum_init=0.3,
                    rng_seed=int(rng.integers(10000)), verbose=False)
                try_update(a_out, "div_start")

    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}, budget={remaining:.0f}s) ===")
    if remaining > 40:
        t7 = min(remaining * 0.5, 200.0)
        if t7 > 30:
            a_out, _ = _mirror_descent_full(
                best_seq, t7,
                beta_init=20.0, beta_end=80.0,
                lr_init=0.08, momentum_init=0.5,
                rng_seed=999, verbose=True)
            try_update(a_out, "final_md")

    # Final coord descent
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t8 = min(remaining * 0.6, 80.0)
        if t8 > 10:
            a_out = _coord_descent_polish(best_seq, t8, rng_seed=42)
            try_update(a_out, "final_coord")

    # Final validation
    if best_seq is not None:
        fs = _eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = np.ones(n_fb) / n_fb
        best_score = _eval_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")

    return [float(x) for x in best_seq]
