    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]

    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]

    rho = rho[-1]

    return np.maximum(v - theta, 0.0)

def subgradient_descent(a0, time_limit, temperature=1.0, eta0=0.5, verbose=False, rng_seed=0):
    """Subgradient descent with adaptive step size."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    eta = eta0
    stall = 0
    momentum = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        g, ac, ac_max = subgradient_autocorr(a, temperature)
        
        # Adam update
        m_hat = beta1 * m_hat + (1 - beta1) * g
        v_hat = beta2 * v_hat + (1 - beta2) * g ** 2
        m_hat_hat = m_hat / (1 - beta1 ** t)
        v_hat_hat = v_hat / (1 - beta2 ** t)
        
        a_trial = a - eta * m_hat_hat / (np.sqrt(v_hat_hat) + 1e-8)
        a_new = project_simplex(a_trial)
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.01, 2.0)
        else:
            stall += 1
            if stall > 100:
                eta *= 0.8
            if stall > 300:
                # Restart with perturbation
                noise = rng.normal(0, 0.01, n) * best_a
                a = np.maximum(best_a + noise, 0)
                a = a / a.sum()
                stall = 0
                eta = eta0 * 0.5
                m_hat = np.zeros(n)
                v_hat = np.zeros(n)
                t = 0
        
        a = a_new
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SGD it={t} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.4f}, stall={stall}")
    
    return best_a, best_m

def norm_sum1(a):

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    w = w / w.sum()
    cross_corr = np.correlate(w, a, mode='valid')
    g = 2.0 * cross_corr
    return g, ac, ac_max

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SGD it={t} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.4f}, temp={temperature:.2f}")

    return best_a, best_m


def fast_coordinate_descent(a0, time_limit, verbose=False, rng_seed=0, n_candidates=6):
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    sweep = 0
    # Precompute prefix and suffix max of ac for fast background computation
    # prefix_max[i] = max(ac[0], ..., ac[i])
    # suffix_max[i] = max(ac[i], ac[i+1], ..., ac[2n-2])

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False

        # Focus on coordinates near peak lags
        peak_lags = np.argsort(ac)[::-1][:min(50, 2 * n - 1)]
        hot_indices = set()
        for k in peak_lags[:20]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            hot_indices.update(range(s, e))
        hot_list = list(hot_indices)
        cold_list = [i for i in range(n) if i not in hot_indices]

        order = hot_list + list(rng.permutation(cold_list))

        for i in order:
            v = a[i]
            if v < 1e-15:
                continue

            # Background max: ac[k] for k not in [i, i+n-1]
            # k in [i, i+n-1] is the "relevant" range
            relevant_lo = i
            relevant_hi = min(i + n - 1, 2 * n - 2)

            if relevant_lo > 0 and relevant_hi < 2 * n - 2:
                C = max(ac[:relevant_lo].max(), ac[relevant_hi + 1:].max())
            elif relevant_lo > 0:
                C = ac[:relevant_lo].max()
            elif relevant_hi < 2 * n - 2:
                C = ac[relevant_hi + 1:].max()
            else:
                C = 0.0

            # ac_relevant = ac[i : i+n]
            ac_rel = ac[i:i + n]

            # Candidates for new value
            fracs = np.array([0.0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.6, 2.0])
            candidates = v * fracs
            candidates = np.clip(candidates, 0.0, 1.0)
            u_vals = candidates - v  # displacement

            # For each candidate t = v + u:
            # new_ac[k] = ac[k] + 2*u*a[k-i] for k in [i, i+n-1], k != 2i
            # new_ac[2i] = ac[2i] + 2*v*u + u^2
            # Peak = max(C, max_j (ac[i+j] + 2*u*a[j]) for j!=i, ac[2i] + 2*v*u + u^2)

            # Vectorized: shape (n_cands, n)
            linear_parts = ac_rel[None, :] + 2.0 * np.outer(u_vals, a)
            # Fix position j=i (which corresponds to k=2i):
            linear_parts[:, i] = ac[2 * i] + 2.0 * v * u_vals + u_vals ** 2

            peaks = np.maximum(C, linear_parts.max(axis=1))

            best_idx = np.argmin(peaks)
            best_peak = peaks[best_idx]

            if best_peak < best_m - 1e-15:
                u_best = u_vals[best_idx]
                # Update ac incrementally
                ac[i:i + n] += 2.0 * u_best * a
                # Fix the 2i position (already has 2*u*a[i] = 2*u*v added, need to add u^2)
                ac[2 * i] += u_best ** 2
                a[i] = v + u_best
                best_m = best_peak
                best_a = a.copy()
                improved = True

        if not improved:
            stall += 1
            if stall > 40:
                # Try perturbation
                for _ in range(20):
                    idx = int(rng.integers(n))
                    if a[idx] < 1e-10:
                        continue
                    frac = rng.uniform(0.05, 0.3)
                    amt = a[idx] * frac
                    a[idx] -= amt
                    j = int(rng.integers(n))
                    a[j] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    ac = fast_autocorr(a)
                    m = ac.max()
                    if m < best_m:
                        best_m = m
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
                    else:
                        a[idx] += amt
                        a[j] -= amt
                        a = a / a.sum()
                        ac = fast_autocorr(a)
                if stall > 100:
                    break
    if_verbose = verbose
    if if_verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordD final ({elapsed:.0f}s): J={J:.6f}, sweeps={sweep}")

    return best_a, best_m


def coordinate_descent_focused(a0, time_limit, verbose=False, rng_seed=0):
    """Coordinate descent that focuses on peak-contributing coordinates."""
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        improved = False

        # Find top peak lags and their contributing indices
        top_lags = np.argsort(ac)[::-1][:30]
        target_indices = set()
        for k in top_lags:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            for j in range(s, e):
                if a[j] > 1e-10:
                    target_indices.add(j)

        # Try to reduce each peak by adjusting contributing coordinates
        for k in top_lags:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            idx = np.arange(s, e)
            # Contribution: a[j] * a[ki-j]
            contrib = a[idx] * a[ki - idx]
            total_k = contrib.sum()
            if total_k < 1e-15:
                continue

            # The coordinate with largest contribution
            max_j = idx[np.argmax(contrib)]
            max_contrib = contrib.max()
            if max_contrib < 1e-15:
                continue

            # Try reducing this coordinate
            v = a[max_j]
            # Compute what the new peak would be
            # We'll try a few reduction factors
            for frac in [0.7, 0.8, 0.9]:
                new_v = v * frac
                delta = new_v - v
                # Update: ac[ki] changes by 2*delta*a[ki-max_j] (if ki-max_j != max_j)
                # or delta*(2*v + delta) if ki = 2*max_j
                if ki == 2 * max_j:
                    new_ac_ki = ac[ki] + 2 * v * delta + delta ** 2
                else:
                    new_ac_ki = ac[ki] + 2 * delta * a[ki - max_j]

                # Check if this reduces the overall peak
                # The new peak is at most max(ac.max(), new_ac_ki) but could be lower
                # Actually other ac values also change, but this is a quick check
                if new_ac_ki < ac.max() - 1e-14 or (new_ac_ki < best_m - 1e-14 and ac[ki] >= best_m * 0.99):
                    # Full update
                    u = delta
                    ac[s:e] += 2.0 * u * a[ki - idx]  # approximate (uses old a)
                    # Fix the diagonal if needed
                    for j in idx:
                        if 2 * j == ki:
                            ac[ki] += u ** 2
                            break
                    a[max_j] = new_v

                    # Recompute new peak (approximate)
                    new_peak = ac.max()
                    if new_peak < best_m - 1e-15:
                        best_m = new_peak
                        best_a = a.copy()
                        improved = True
                    break

        if not improved:
            stall += 1
            if stall > 50:
                # Perturb
                idx = int(rng.integers(n))
                if a[idx] > 1e-10:
                    amt = a[idx] * 0.1
                    a[idx] -= amt
                    j = int(rng.integers(n))
                    a[j] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    ac = fast_autocorr(a)
                    if ac.max() < best_m:
                        best_m = ac.max()
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[idx] += amt
                        a[j] -= amt
                        a = a / a.sum()
                        ac = fast_autocorr(a)
                if stall > 150:
                    break
        else:
            stall = 0

        if verbose and iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordFocus it={iteration} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K_init=200):
    n = len(a0)
    if n > 5000:
        return a0.copy(), fast_autocorr(a0).max()
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build constraint matrix T
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, 2 * n - 1)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 50, 2 * n - 1)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            K = min(K + 50, 2 * n - 1)
            continue
        g_dir = b_val / b_sum

        # Line search
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = M_curr
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), 0.5, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1
        if iter_count % 8 == 0:
            K = min(K + 20, 2 * n - 1)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def spectral_pocs(a0, time_limit, verbose=False, n_trials=3, rng_seed=0):
    n = len(a0)
    best_a = a0.copy()
    best_m = fast_autocorr(a0).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit - 5:
            break
        t_trial = min((time_limit - (time.monotonic() - t0)) / (n_trials - trial), 30.0)
        rng = np.random.default_rng(rng_seed + trial)

        # Initialize with random phases
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        A_spec = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
        A_full = np.zeros(fft_size, dtype=np.complex128)
        A_full[:fft_size // 2 + 1] = A_spec
        for i in range(1, fft_size // 2):
            A_full[fft_size - i] = np.conj(A_spec[i])
        a = np.fft.irfft(A_full, fft_size)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = a0.copy()
        else:
            a = a / s

        # POCS iterations
        for it in range(100000):
            if time.monotonic() - t0 > time_limit:
                break
            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            alpha_blend = 0.05 + 0.3 * min(1.0, it / 200)
            new_mag = (1 - alpha_blend) * mag + alpha_blend * mag_mean
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s2 = a_new.sum()
            if s2 < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s2

            M_new = fast_autocorr(a_new).max()
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = a_new.copy()
            a = a_new

    return best_a, best_m


def smooth_gradient_descent(a0, time_limit, verbose=False, rng_seed=100):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    tau = 10.0
    eta = 0.5
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        ac_max = ac.max()

        # Smooth weights
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()

        # Gradient: g[i] = 2 * sum_k p[k] * a[k-i] (with proper bounds)
        g = 2.0 * np.correlate(p, a, mode='valid')

        # Project gradient onto tangent space
        g_tangent = g - g.mean()

        a_trial = a - eta * g_tangent
        a_new = project_simplex(a_trial)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 2.0)
            tau = min(tau * 1.005, 100.0)
        else:
            a = a_new
            stall += 1
            if stall > 100:
                eta *= 0.7
            if stall > 300:
                pert = rng.normal(0, 0.01, n) * best_a
                a_p = np.maximum(best_a + pert, 0)
                a = a_p / a_p.sum()
                stall = 0
                eta = 0.3
                tau = 5.0

        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            if int(time.monotonic() - t0) % 10 == 0:
                print(f"    SmoothGD ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.3f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Total time budget: {time_budget:.0f}s")

    solver = get_solver()

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Quick subgradient from multiple inits ===
    print(f"\n=== Phase 1: Subgradient (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 180.0)
    if t1 > 30:
        inits = [norm_sum1(best_seq)]
        labels = ["parent_best"]
        inits.append(make_u_shape(n0, 2.0))
        labels.append("u2")
        inits.append(make_bimodal(n0, 0.2))
        labels.append("bimodal")
        inits.append(make_cosine_sq(n0))
        labels.append("cosine_sq")

        t_per = max(10.0, t1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            a_out, _ = subgradient_descent(init, t_avail, temperature=3.0, eta0=0.5,
                                           verbose=False, rng_seed=100 + idx)
            try_update(a_out, f"sgd_{lbl}")

    # === Phase 2: Coordinate Descent (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Coordinate Descent (n={n_best}) ===")
    t2 = min(remaining * 0.30, 480.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = fast_coordinate_descent(a_in, t2, verbose=True, rng_seed=42)
        try_update(a_out, "coord1")

    # === Phase 3: LP-Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.15, 240.0)
    if t3 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True, top_K_init=300)
        try_update(a_out, "lpfw1")

    # === Phase 4: More Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinate Descent r2 (n={n_best}) ===")
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = fast_coordinate_descent(a_in, t4, verbose=True, rng_seed=123)
        try_update(a_out, "coord2")

    # === Phase 5: Spectral POCS ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Spectral POCS (n={n_best}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_pocs(a_in, t5, verbose=True, n_trials=3, rng_seed=7)
        try_update(a_out, "pocs")

    # === Phase 6: Smoothed gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Smooth Gradient (n={n_best}) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_gradient_descent(a_in, t6, verbose=True, rng_seed=200)
        try_update(a_out, "smooth")

    # === Phase 7: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: LP-FW r2 (n={n_best}) ===")
    t7 = min(remaining * 0.08, 120.0)
    if t7 > 20 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=False, top_K_init=500)
        try_update(a_out, "lpfw2")

    # === Phase 8: Final coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Coord (n={n_best}) ===")
    t8 = min(remaining * 0.20, 200.0)
    if t8 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = fast_coordinate_descent(a_in, t8, verbose=True, rng_seed=789)
        try_update(a_out, "coord_final")

    # === Phase 9: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Length exploration ===")
    if remaining > 100:
        trial_lengths = []
        for delta in [500, -500, 1000, -1000, 2000, -2000, 3000, -3000]:
            new_n = n_best + delta
            if 1000 <= new_n <= 8000:
                trial_lengths.append(new_n)
        trial_lengths = list(dict.fromkeys(trial_lengths))
        trial_lengths = [l for l in trial_lengths if l != n_best]

        for new_n in trial_lengths[:3]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            print(f"  Trying n={new_n}...")
            a_long = interpolate_sequence(best_seq, new_n)

            t9a = min(remaining * 0.4, 120.0)
            if t9a > 15:
