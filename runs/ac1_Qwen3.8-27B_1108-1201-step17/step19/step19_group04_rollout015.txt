def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def self_conv(a):
    """Compute full self-convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)[:m]


def conv_ab(a, b):
    """Compute cross-convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = next_pow2(m)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:m]


def objective_score(a):
    """Compute 2*n*max(conv(a,a)) / (sum(a))^2."""

    Find s = argmin grad[i], return direction e_s and step size.

    s = int(np.argmin(grad))

    new = np.maximum(new, 0.0)
    ss = float(new.sum())
    if ss > 1e-15:

    return new, s
