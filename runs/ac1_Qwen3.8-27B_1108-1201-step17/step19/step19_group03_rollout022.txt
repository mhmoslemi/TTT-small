"""

import time
import cvxpy as cp

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgradient_softmax(a, ac, num_peaks=200, temperature=0.05):
    """Compute softmax-weighted subgradient of max_k ac[k] / S^2."""
    n = len(a)
    # Use top num_peaks with softmax weighting
    sorted_idx = np.argsort(ac)[::-1]
    top_idx = sorted_idx[:min(num_peaks, len(ac))]
    top_vals = ac[top_idx]
    
    # Softmax weights
    if temperature > 0:
        log_w = (top_vals - top_vals[0]) / temperature
        log_w[np.isnan(log_w)] = -100
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(w)) / len(w)
        else:
            w = w / w_sum
    else:
        w = np.ones(len(top_idx)) / len(top_idx)
    
    # Compute gradient: g[i] = (2/S) * sum_k w_k * a[k-i]
    # This is a weighted cross-correlation
    # Weighted correlation: sum_k w_k * a[k-i] for each i
    # We can compute this via FFT or direct summation
    
    # Direct computation for the top peaks
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n - 1, ki)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        g[idx] += wi * a[ki - idx]
    
    # The subgradient of max ac[k] is the gradient at the max point
    # For the smoothed version: g[i] = (2/S) * (weighted_correlation[i] - mean_ac)
    # where weighted_correlation[i] = sum_k w_k * a[k-i]
    # and we subtract the "average" to make it a tangent step
    
    S = a.sum()
    mean_contrib = np.sum(w * ac[top_idx])  # weighted average of ac values = "smoothed max"
    g = (2.0 / S) * (g - mean_contrib / S * np.ones(n))
    # Actually, the subgradient of f(a) = ac[k*]/S^2 is:
    # df/da[i] = (a[k*-i] * S^2 - ac[k*] * 2S) / S^4 = (a[k*-i] - ac[k*]/S) / S^2 * 2S
    # = 2(a[k*-i] - M/S) / S
    # For multi-peak: g[i] = sum_k w_k * 2(a[k-i] - ac[k]/S) / S
    # = (2/S) * (sum_k w_k * a[k-i] - sum_k w_k * ac[k]/S)
    # = (2/S) * (g_raw[i] - weighted_mean_ac / S)
    
    # Hmm, this doesn't quite work because ac[k] = sum_j a[j]*a[k-j], not a single a[k-i].
    # Let me just use: g[i] = (2/S) * sum_k w_k * a[k-i] as the direction to decrease.
    # We want to decrease a[i] where g[i] is large.
    
    return g


def lp_frank_wolfe_step(a, ac, K, M, solver):
    """
    Solve the LP for Frank-Wolfe direction.
    max sum(b) s.t. T@b <= M, b >= 0
    where T[k,j] = a[binding_idx[k] - j]
    
    Uses dual formulation for efficiency.
    """
    n = len(a)
    binding_idx = np.argsort(ac)[::-1][:K]
    
    # Build T matrix (K x n)
    T = np.zeros((K, n))
    for ii in range(K):
        k = int(binding_idx[ii])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            T[ii, j_start:j_end+1] = a[k - (j_start + np.arange(j_end - j_start + 1))]
    
    # Dual: min M * sum(y) s.t. T.T @ y >= 1, y >= 0
    y = cp.Variable(K, nonneg=True)
    obj = cp.Minimize(M * cp.sum(y))
    constraint = T.T @ y >= np.ones(n)
    prob = cp.Problem(obj, [constraint])
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    # Get primal solution from dual variables
    # For constraint T.T @ y >= 1 in a min problem, dual value is <= 0
    # The primal b = -dual_value (component-wise, clipped to >= 0)
    try:
        b_raw = -np.asarray(constraint.dual_value, dtype=np.float64)
    except Exception:
        return None
    
    b = np.maximum(b_raw, 0.0)
    b_sum = b.sum()
    if b_sum < 1e-15:
        return None
    return b / b_sum


def lp_frank_wolfe_primal(a, ac, K, M, solver):
    """
    Solve the primal LP directly.
    max sum(b) s.t. T@b <= M, b >= 0
    """
    n = len(a)
    binding_idx = np.argsort(ac)[::-1][:K]
    
    T = np.zeros((K, n))
    for ii in range(K):
        k = int(binding_idx[ii])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            T[ii, j_start:j_end+1] = a[k - (j_start + np.arange(j_end - j_start + 1))]
    
    b_var = cp.Variable(n, nonneg=True)
    obj = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(obj, [T @ b_var <= M])
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b.sum()
    if b_sum < 1e-15:
        return None
    return b / b_sum


def pocs_init(n, time_limit, n_trials=3, rng_seed=0):
    """POCS initialization with random phases."""
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    best_a = np.ones(n) / n
    best_m = fast_autocorr(best_a).max()
    
    for trial in range(n_trials):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        t_trial = min((time_limit - elapsed) / (n_trials - trial), 30.0)
        rng = np.random.default_rng(rng_seed + trial)
        
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        A_spec = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
        A_full = np.zeros(fft_size, dtype=np.complex128)
        A_full[:fft_size // 2 + 1] = A_spec
        for i in range(1, fft_size // 2):
            A_full[fft_size - i] = np.conj(A_spec[i])
        a = np.fft.irfft(A_full, fft_size)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s
        
        # POCS iterations
        t_pocs_start = time.monotonic()
        for it in range(100000):
            if time.monotonic() - t_pocs_start > t_trial:
                break
            
            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            alpha_blend = 0.1 + 0.4 * min(1.0, it / 100)
            new_mag = (1 - alpha_blend) * mag + alpha_blend * mag_mean
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s
            
            a = a_new
            m = fast_autocorr(a).max()
            if m < best_m:
                best_m = m
                best_a = a.copy()
    
    return best_a


def mirror_descent_step(a, ac, num_peaks, temperature, eta):
    """One step of mirror descent with softmax-weighted subgradient."""
    n = len(a)
    S = a.sum()
    
    # Compute weighted subgradient
    sorted_idx = np.argsort(ac)[::-1]
    top_idx = sorted_idx[:min(num_peaks, len(ac))]
    top_vals = ac[top_idx]
    
    # Softmax weights
    log_w = (top_vals - top_vals[0]) / temperature
    log_w[~np.isfinite(log_w)] = -200
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(w)) / len(w)
    else:
        w = w / w_sum
    
    # g_raw[i] = sum_k w_k * a[k-i]
    g_raw = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n - 1, ki)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        g_raw[idx] += wi * a[ki - idx]
    
    # The "peaked" components should be decreased
    # Mirror descent: a_new[i] = a[i] * exp(-eta * g_raw[i])
    log_a = np.log(np.maximum(a, 1e-30)) - eta * g_raw
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    a_new = a_new / a_new.sum()
    
    return a_new


def line_search(a, direction, ac_current, M_current):
    """Backtracking line search for Frank-Wolfe step."""
    best_alpha = 2.0 / 3.0
    best_m = M_current
    
    for alpha in [best_alpha, 0.5, 0.3, 0.7, 0.2, 0.1, 0.8, 0.4]:
        a_trial = (1.0 - alpha) * a + alpha * direction
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s
        m_trial = fast_autocorr(a_trial).max()
        if m_trial < best_m:
            best_m = m_trial
            best_alpha = alpha
    
    return best_alpha, best_m


def main_optimize(a0, time_limit, solver, verbose=True):
    """Main optimization loop combining mirror descent and LP-FW."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    # Parameters
    num_peaks_grad = 150  # peaks used in gradient
    temperature = 0.03
    eta_base = 1.0
    K_lp_init = 100
    K_lp_max = 3000
    lp_every = 20  # do LP every this many gradient steps
    
    K_lp = K_lp_init
    lp_count = 0
    grad_count = 0
    stall = 0
    best_iter = 0
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Decide: gradient step or LP step
        use_lp = (iteration % lp_every == 0) or (lp_count == 0 and iteration > 10)
        
        if use_lp and K_lp < 2 * n - 1:
            # LP-Frank-Wolfe step
            direction = lp_frank_wolfe_step(a, ac, K_lp, M_curr, solver)
            if direction is None:
                # Fallback to gradient
                use_lp = False
            else:
                alpha, M_new = line_search(a, direction, ac, M_curr)
                a = (1.0 - alpha) * a + alpha * direction
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                lp_count += 1
        else:
            # Mirror descent step
            eta = eta_base / (1.0 + 0.01 * grad_count)
            a_new = mirror_descent_step(a, ac, num_peaks_grad, temperature, eta)
            M_new = fast_autocorr(a_new).max()
            if M_new < M_curr * 0.999:
                a = a_new
                grad_count += 1
            else:
                # Try smaller step
                eta_small = eta * 0.3
                a_new2 = mirror_descent_step(a, ac, num_peaks_grad, temperature, eta_small)
                M_new2 = fast_autocorr(a_new2).max()
                if M_new2 < M_curr:
                    a = a_new2
                    grad_count += 1
                else:
                    a = a_new  # accept anyway (helps escape)
                    grad_count += 1
        
        # Update best
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
            best_iter = iteration
            stall = 0
        else:
            stall += 1
        
        # Adapt K_lp
        if iteration % 50 == 0:
            K_lp = min(K_lp + 50, K_lp_max)
        
        # Restart if stalled
        if stall > 200:
            stall = 0
            # Spectral restart
            fft_size = 1 << (2 * n - 2).bit_length()
            A = np.fft.rfft(best_a, fft_size)
            mag = np.abs(A)
            phases = np.angle(A)
            # Add small random phase perturbation
            phase_pert = rng.normal(0, 0.05, len(phases))
            A_new = mag * np.exp(1j * (phases + phase_pert))
            a_restart = np.fft.irfft(A_new, fft_size)[:n]
            a_restart = np.maximum(a_restart, 0.0)
            s = a_restart.sum()
            if s > 1e-15:
                a_restart = a_restart / s
                m_restart = fast_autocorr(a_restart).max()
                if m_restart < best_m:
                    a = a_restart
                    best_m = m_restart
                    best_a = a.copy()
                else:
                    # Blend
                    a = 0.7 * best_a + 0.3 * a_restart
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    m_blend = fast_autocorr(a).max()
                    if m_blend < best_m:
                        best_m = m_blend
                        best_a = a.copy()
            eta_base = max(0.5, eta_base * 0.8)
            temperature = max(0.01, temperature * 0.8)
        
        # Cooling over time
        if iteration % 500 == 0:
            tempscale = 0.95 ** (iteration / 500)
            temperature = max(0.01, 0.05 * tempscale)
        
        if verbose and iteration % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  Main it={iteration} ({elapsed:.0f}s): J={J:.6f}, K_lp={K_lp}, stall={stall}")
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=123):
    """Coordinate descent polish targeting top autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:20]
        
        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find low-impact destination
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2+1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            # Move mass from src to a low-impact location
            for dst in candidates[np.argsort(a[candidates])[:5]]:
                if dst == src:
                    continue
                for frac in [0.01, 0.005, 0.02, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_m


def make_analytic_init(n):
    """Create an analytic initialization based on known good structures."""
    # A "triangular" shape has lower peak autocorrelation than uniform
    # For non-negative sequences, a shape that's high in the middle and low at edges
    # tends to have lower off-peak autocorrelation
    x = np.arange(n) / (n - 1)
    # Try a raised cosine
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = a ** 2
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
