Accelerated FW (also known as the "catalyzed" FW or "STAR" method) maintains two iterates: $y_t$ (the "actual" iterate) and $x_t$ (the "virtual" iterate). At each step:

2. Update $x_{t+1} = (1 - \alpha_t) x_t + \alpha_t g_t$
3. Update $y_{t+1} = (1 - \beta_t) y_t + \beta_t x_{t+1}$

where $\alpha_t$ and $\beta_t$ are chosen based on the duality gap.

Actually, for the Frank-Wolfe method on the simplex, the accelerated version (Lan 2016) works as follows:

Initialize: $y_0 = x_0$, pick $g_0$ (FW step at $y_0$), set $x_0 = g_0$

For $t = 1, 2, \ldots$:
- Solve FW subproblem at $y_{t-1}$: $g_t = \arg\min_{g \in \Delta} \nabla f(y_{t-1})^T g$
- Gap: $\Delta_t = f(y_{t-1}) - \min_{g \in \Delta} \nabla f(y_{t-1})^T g$

- $y_t = \frac{t}{t+2} y_{t-1} + \frac{2}{t+2} x_t$

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw_accelerated(a0, time_limit, solver, initial_K=300, max_K=3500,
                      add_per_iter=40, verbose=True):
    """LP-based Frank-Wolfe with dual-driven active set and accelerated updates."""
    n = len(a0)
    y = norm_sum1(a0)
    x_virt = y.copy()
    best_a = y.copy()
    best_m = fast_convolve(y).max()
    t0 = time.monotonic()

    ac_y = fast_convolve(y)
    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())

    no_improve = 0
    max_no_improve = 1500
    iter_count = 0
    lp_fail_count = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            break
        if lp_fail_count > 30:
            break
        iter_count += 1

        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep = np.argsort(margins)[:max_K]
            active_set = sorted([active_set[i] for i in keep])

        K = len(active_set)
        if K == 0:
            break

        coeffs = build_constraint_matrix(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        cons_list = [cp.sum(g_var) == 1]
        for jj in range(K):
            cons_list.append(coeffs[jj] @ g_var <= M_var)
        cons_list.append(M_var >= 0)

        prob = cp.Problem(cp.Minimize(M_var), cons_list)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail_count += 1
            active_set = sorted(active_set[:max(initial_K, len(active_set) // 2)])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail_count += 1
            if lp_fail_count > 10:
                active_set = sorted(np.argsort(fast_convolve(y))[::-1][:initial_K].tolist())
                lp_fail_count = 0
            continue

        lp_fail_count = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Update active set: add violated constraints
        conv_ys = fast_cross_convolve(g_dir, y)
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            n_add = min(add_per_iter, len(violations))
            top_v = violations[np.argsort(viol_amt)[::-1][:n_add]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)

        # Also try to access dual values for pruning
        # (CVXPY may or may not provide these depending on solver)

        # Step size selection with accelerated scheme
        t = iter_count
        alpha_virt = 2.0 / (t + 2.0)

        # Try multiple step sizes
        alphas = [max(0.01, min(1.0, alpha_virt)),
                  max(0.01, min(1.0, alpha_virt * 0.5)),
                  max(0.01, min(1.0, alpha_virt * 1.5)),
                  max(0.01, min(1.0, 1.0 / (t + 1.0))),
                  max(0.01, min(1.0, 0.1)),
                  max(0.01, min(1.0, 0.3))]

        best_alpha = alpha_virt
        best_trial_M = M_curr
        best_trial = y.copy()

        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
                best_trial = trial.copy()

        # Accelerated update
        x_virt = (1.0 - best_alpha) * x_virt + best_alpha * g_dir
        x_virt = np.maximum(x_virt, 0.0)
        s_v = x_virt.sum()
        if s_v < 1e-15:
            x_virt = y.copy()
        else:
            x_virt = x_virt / s_v

        y_new = (1.0 - best_alpha) * y + best_alpha * x_virt
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y = best_trial.copy()
        else:
            y = y_new / s_y

        # If accelerated point is worse, use the best trial
        M_y = fast_convolve(y).max()
        if M_y > best_trial_M + 1e-12:
            y = best_trial.copy()

        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    ACCEL-LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", |S|=" + str(len(active_set)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    ACCEL-LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", iters=" + str(iter_count))
    return best_a, best_m

def compute_soft_gradient(y, ac, top_k, beta):
    n = len(y)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        grad[idx] += 2.0 * wi * y[k_int - idx]
    return grad


def soft_mirror_descent(a0, time_limit, top_k=80, beta=5.0, c0=4.0,
                        restart_every=5000, momentum=0.7, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(7)

    x = a.copy()
    vel = np.zeros(n)
    gamma = c0
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    restart_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        y = x + momentum * vel
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        ac = fast_convolve(y)
        grad = compute_soft_gradient(y, ac, cur_top_k, cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        vel = 0.9 * vel + 0.1 * (a_new - x)
        x = a_new

        M_new = fast_convolve(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            restart_count += 1
            noise_level = 0.03 / (1.0 + 0.1 * restart_count)
            pert = rng.normal(0, noise_level, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r < 1e-15:
                x_r = best_a.copy()
            else:
                x_r = x_r / s_r
            x = x_r
            vel = np.zeros(n)
            cur_top_k = min(cur_top_k + 10, 150)
            cur_beta = max(cur_beta - 0.3, 2.0)
            gamma = max(gamma * 0.92, 0.5)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    SSD restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

    return best_a, best_m

    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 600
    step_scales = [0.002, 0.004, 0.008, 0.015, 0.03, 0.06, 0.12, 0.2]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(50, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        # Focus on the top contributing lags
        for k in top_k[:25]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination: minimize impact on all top lags
            scores = np.zeros(n)
            for k2 in top_k[:15]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:15]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < n // 5:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])

            if dst == src:
                continue

            found = False
            for frac in step_scales:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 150:
                # Random kick
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.008)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0:
                # Periodic random perturbations
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.01)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", stall=" + str(stall))

    return best_a, best_m


def parametric_init(n, time_limit, best_init, best_score):
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score

    candidates = []
    for p in [0.2, 0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))
    for alpha in [0.3, 0.5, 1.0, 1.5, 2.0]:
        for beta in [0.3, 0.5, 1.0, 1.5, 2.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            candidates.append(norm_sum1(a))
    for c in [0.15, 0.3, 0.5, 0.7, 0.85]:
        for sigma in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]:
            candidates.append(norm_sum1(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))))
    for c1 in [0.1, 0.2, 0.3, 0.4]:
        for c2 in [0.6, 0.7, 0.8, 0.9]:
            for sigma in [0.05, 0.1, 0.15]:
                a = np.exp(-((x - c1) ** 2) / (2.0 * sigma * sigma)) + np.exp(-((x - c2) ** 2) / (2.0 * sigma * sigma))
                candidates.append(norm_sum1(a))
    for freq in [1, 2, 3, 4]:
        for ph in [0.0, np.pi / 2.0, np.pi / 3.0]:
            a = np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x + ph)) / 2.0, 0.01)
            candidates.append(norm_sum1(a))
    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        candidates.append(norm_sum1(x ** p + (1.0 - x) ** p))

    for a in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        v = compute_obj(a)
        if v < best_val:
            best_val = v
            best_a = a.copy()

    return best_a, best_val


def get_initial(t_global):
    """Get the best initial sequence from available sources."""
    best_seq = None
    best_score = float('inf')

    # Try the parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                best_seq = p_norm
                best_score = sc
                print("Parent: n=" + str(len(p)) + ", score=" + repr(sc))
    except Exception:
        pass

    if best_seq is None:
        n = 5000
        x = np.linspace(0, 1, n)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    return best_seq, best_score


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Total time budget: " + repr(time_budget) + "s")

    best_seq, best_score = get_initial(t_global)
    n0 = len(best_seq)

    # Phase 1: Quick parametric scan (5% of time)
    print("\n=== Phase 1: Parametric (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 50.0)
    if t1 > 10:
        a_param, _ = parametric_init(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "parametric", t_global)

    # Phase 2: Main LP-FW with acceleration (40% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: ACCEL LP-FW (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.40, 600.0)
    if t2 > 60 and n_best <= 9000:
        a_out, _ = lp_fw_accelerated(norm_sum1(best_seq), t2, solver,
                                     initial_K=400, max_K=4000, add_per_iter=50, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "accel_lpfw", t_global)

    # Phase 3: Standard LP-FW to continue from where accelerated left off (20% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP-FW continue (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40 and n_best <= 9000:
        a_out, _ = lp_fw_standard(norm_sum1(best_seq), t3, solver,
                                  initial_K=500, max_K=3500, add_per_iter=40, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_cont", t_global)

    # Phase 4: Soft mirror descent (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: SSD (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_out, _ = soft_mirror_descent(norm_sum1(best_seq), t4, top_k=100, beta=5.0,
                                       c0=4.0, restart_every=4000, momentum=0.7, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ssd", t_global)

    # Phase 5: Try alternative n values
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    try_ns = []
    for delta in [-500, 300, 800, -1000]:
        cand = n_best + delta
        if 3000 <= cand <= 10000:
            try_ns.append(cand)

    for new_n in try_ns[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 200:
            break
        print("\n=== Phase 5: Scale n=" + str(new_n) + " ===")
        a_long = interpolate_seq(best_seq, new_n)

        t5 = min(remaining * 0.30, 250.0)
        if t5 > 40 and new_n <= 9000:
            a_out, _ = lp_fw_standard(a_long, t5, solver,
                                      initial_K=400, max_K=3000, add_per_iter=40, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80:
            t5b = min(remaining * 0.20, 100.0)
            if t5b > 20:
                a_out, _ = soft_mirror_descent(norm_sum1(best_seq), t5b, top_k=80, beta=4.0,
                                               c0=3.5, restart_every=3000, momentum=0.6, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "ssd_n" + str(new_n), t_global)

    # Phase 6: Final LP-FW (15% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Final LP-FW (n=" + str(n_best) + ") ===")
    if remaining > 80 and n_best <= 9000:
        t6 = min(remaining * 0.15, 200.0)
        if t6 > 40:
            a_out, _ = lp_fw_accelerated(norm_sum1(best_seq), t6, solver,
                                         initial_K=600, max_K=4500, add_per_iter=60, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_lpfw", t_global)

    # Phase 7: Coordinate descent polish (remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final Coord (n=" + str(n_best) + ") ===")
    if remaining > 30:
        t7 = min(remaining * 0.80, 180.0)
        if t7 > 15:
            a_out, _ = coordinate_polish(norm_sum1(best_seq), t7, rng_seed=42, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
