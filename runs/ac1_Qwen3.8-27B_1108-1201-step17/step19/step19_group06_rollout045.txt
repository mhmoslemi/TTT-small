    N = 2*n - 1
    a_padded = np.zeros(N)

        target_mag = np.sqrt(np.sum(a**2))
        phase = A / np.abs(A)  # careful with zeros

        A_new = target_mag * phase

        if s < 1e-15:

        a = a_new
    
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gerchberg-Saxton algorithm for flat-spectrum initialization
2. Soft-max mirror descent with Nesterov momentum
3. Proactive gradient descent on top autocorrelation lags
4. Coordinate descent polish
5. Multi-length exploration

import numpy as np
import time


def fast_autocorr(a):
    """Compute full autocorrelation (linear convolution a*a) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely call the evaluation function."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def score_from_ac(a):
    """Compute score from a normalized sequence (sum=1)."""
    n = len(a)
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max())


def gs_flat(a0, n_iters, damping=0.5, verbose=False):
    """
    Gerchberg-Saxton algorithm to flatten the autocorrelation.
    Alternates between flat magnitude in frequency and non-negativity in time.
    """
    n = len(a0)
    N = 2 * n - 1
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = score_from_ac(a)
    
    for it in range(n_iters):
        if verbose and it % 2000 == 0:
            print(f"    GS iter {it}: score={best_score:.6f}")
        
        a_padded = np.zeros(N)
        a_padded[:n] = a
        
        A = np.fft.fft(a_padded)
        mag = np.abs(A)
        
        # Target: flat magnitude = sqrt(sum(a^2))
        target_mag = np.sqrt(np.sum(a * a))
        if target_mag < 1e-30:
            break
        
        # Enforce flat magnitude, keep phase
        phase = np.ones(N, dtype=np.complex128)
        nonzero = mag > 1e-30
        phase[nonzero] = A[nonzero] / mag[nonzero]
        
        A_new = target_mag * phase
        a_new_full = np.fft.ifft(A_new).real
        a_new = a_new_full[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        a_new = norm_sum1(a_new)
        
        score = score_from_ac(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        # Damped update
        a = (1.0 - damping) * a + damping * a_new
    
    return best_a, best_score


def gs_soft(a0, n_iters, beta=5.0, damping=0.5, verbose=False):
    """
    Soft version of GS: instead of exact flat magnitude,
    use exponential weighting to soften peaks.
    """
    n = len(a0)
    N = 2 * n - 1
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = score_from_ac(a)
    
    for it in range(n_iters):
        if verbose and it % 2000 == 0:
            print(f"    GS-soft iter {it}: score={best_score:.6f}")
        
        a_padded = np.zeros(N)
        a_padded[:n] = a
        
        A = np.fft.fft(a_padded)
        P = np.abs(A) ** 2
        
        # Soft flatten: reduce peaks, boost valleys
        P_max = P.max()
        P_mean = P.mean()
        if P_max < 1e-30:
            break
        
        # Multiplicative adjustment
        adjustment = np.exp(-beta * (P - P_mean) / P_max)
        adjustment = np.clip(adjustment, 0.01, 100.0)
        
        A_new = A * np.sqrt(adjustment)
        a_new_full = np.fft.ifft(A_new).real
        a_new = a_new_full[:n]
        
        a_new = np.maximum(a_new, 0.0)
        a_new = norm_sum1(a_new)
        
        score = score_from_ac(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = (1.0 - damping) * a + damping * a_new
    
    return best_a, best_score


def soft_max_mirror_descent(a0, time_limit, lr_init=0.1, beta_init=3.0,
                            top_k_init=30, restart_every=3000, verbose=False, rng_seed=42):
    """
    Mirror descent (multiplicative updates) with soft-max gradient
    on top autocorrelation lags.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    a_prev = a.copy()
    momentum = 0.8
    lr = lr_init
    beta = beta_init
    top_k = top_k_init
    stall = 0
    
    for t in range(1, 500000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov in multiplicative space
        ratio = a / np.maximum(a_prev, 1e-30)
        ratio = np.clip(ratio, 0.01, 100.0)
        a_nest = a * np.power(ratio, momentum)
        a_nest = norm_sum1(a_nest)
        
        # Compute autocorrelation
        ac = fast_autocorr(a_nest)
        
        # Soft-max gradient on top lags
        k_alarm = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_alarm]
        top_vals = ac[top_idx]
        
        max_v = top_vals.max()
        expo = beta * (top_vals - max_v)
        expo = np.clip(expo, -50, 50)
        weights = np.exp(expo)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones(k_alarm) / k_alarm
        else:
            weights = weights / w_sum
        
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k_int = int(k)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo >= i_hi:
                continue
            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            valid = (j_idx >= 0) & (j_idx < n)
            grad[idx[valid]] += 2.0 * w * a_nest[j_idx[valid]]
        
        # Adaptive learning rate
        eta = lr / (1.0 + 0.03 * np.sqrt(t))
        
        # Multiplicative update
        log_a = np.log(np.maximum(a_nest, 1e-30)) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        a_prev = a
        a = a_new
        
        ac_new = fast_autocorr(a)
        m_new = ac_new.max()
        if m_new < best_m - 1e-13:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if t % 500 == 0:
            top_k = min(top_k + 3, 200)
        
        if stall > restart_every:
            stall = 0
            pert_type = int(rng.integers(3))
            if pert_type == 0:
                pert = best_a * (1.0 + rng.normal(0, 0.05, n))
            elif pert_type == 1:
                x = np.linspace(0, 1, n)
                c = float(rng.uniform(0.2, 0.8))
                sig = float(rng.uniform(0.1, 0.4))
                pert = np.exp(-((x - c)**2) / (2 * sig**2))
            else:
                pert = np.ones(n) / n
            pert = norm_sum1(pert)
            w = float(rng.uniform(0.1, 0.3))
            a = norm_sum1((1 - w) * best_a + w * pert)
            a_prev = a.copy()
            lr = max(lr * 0.9, 0.01)
            beta = max(beta - 0.3, 1.5)
            
            if verbose:
                J = 2.0 * n * best_m
                print(f"    MD restart t={t} ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    MD done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=99, verbose=False):
    """
    Coordinate descent: try to reduce top autocorrelation peaks
    by adjusting individual coordinates.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                # Random perturbation to escape
                n_moves = int(rng.integers(10, 50))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = float(rng.uniform(0.002, 0.02))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                stall = 0
                continue
        
        # Find the top lags and try to reduce them
        n_top = min(20, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        for k_pos in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            
            k_int = int(k_pos)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo >= i_hi:
                continue
            
            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            
            # Try reducing the largest contributing pairs
            n_pairs = min(5, len(idx))
            pair_order = np.argsort(contrib)[::-1][:n_pairs]
            
            for pi in pair_order:
                src = int(idx[pi])
                src_val = a[src]
                if src_val < 1e-14:

                # Try moving a small amount to a random position
                for dst in rng.choice(n, size=5, replace=False):
                    dst = int(dst)
                    if dst == src:
                        continue
                    
                    for frac in [0.01, 0.03, 0.05, 0.1]:
                        amt = src_val * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        m_trial = fast_autocorr(a_trial).max()
                        if m_trial < best_m - 1e-15:
                            a = a_trial
                            best_m = m_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Coord polish done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def parametric_init(n):

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss
