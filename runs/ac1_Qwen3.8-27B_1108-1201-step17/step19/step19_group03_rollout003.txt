def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K_init=500, top_K_max=None, max_iter=500):

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    iter_count = 0
    stall = 0
    max_stall = 30

        M_curr = ac.max()
        if M_curr < 1e-16:

        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 100, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 100, top_K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            K = min(K + 100, top_K_max)
            continue
        g_dir = b_val / b_sum

        # Aggressive line search
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = M_curr
        alphas = [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0),
                  min(1.0, best_alpha * 3.0), 0.5, 0.3, 0.2, 0.1, 0.7, 0.8]
        for alpha in alphas:
            if time.monotonic() - t0 > time_limit - 2:
                break
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:

            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

            stall = 0

        if iter_count % 5 == 0:
            K = min(K + 150, top_K_max)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

    return best_a, best_m


def peak_shaving(a0, time_limit, n_top_lags=30, verbose=False, rng_seed=0):
    """Directly target the top lags by redistributing mass from responsible pairs."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:n_top_lags]

        # For each top lag, find the coefficient pair with maximum contribution
        improvements = 0
        for lag in top_lags:
            if time.monotonic() - t0 > time_limit - 1:
                break
            ki = int(lag)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            contrib = a[idx] * a[j_idx]
            if contrib.max() < 1e-20:
                continue
            src_i = idx[np.argmax(contrib)]

            # Find target: coefficient NOT involved in any top lag
            involved = np.zeros(n, dtype=bool)
            for tl in top_lags[:10]:
                tls = max(0, int(tl) - n + 1)
                tle = min(n - 1, int(tl))
                involved[tls:tle+1] = True
                involved2 = int(tl) - np.arange(tls, tle+1)
                involved2 = involved2[(involved2 >= 0) & (involved2 < n)]
                if len(involved2) > 0:
                    involved[involved2] = True

            candidates = np.where(~involved)[0]
            if len(candidates) == 0:
                candidates = np.where(a < a.mean())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            # Move a small amount from src to a low-contribution candidate
            src_val = a[src_i]
            if src_val < 1e-14:
                continue

            for frac in [0.02, 0.01, 0.03, 0.005, 0.05]:
                amt = src_val * frac
                if amt < 1e-16:
                    continue
                # Try moving to the candidate with lowest sensitivity
                cand_sens = np.zeros(len(candidates))
                for ci, cand in enumerate(candidates[:20]):
                    for tl in top_lags[:5]:
                        ki2 = int(tl)
                        if 0 <= ki2 - cand < n and 0 <= cand < n:
                            # Check if cand cancels with ki2
                            j2 = ki2 - cand
                            if 0 <= j2 < n:
                                cand_sens[ci] += a[j2]
                dst = candidates[np.argmin(cand_sens)]
                
                a_trial = a.copy()
                a_trial[src_i] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improvements += 1
                    break

        iteration += 1
        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave it={iteration} ({elapsed:.0f}s): J={J:.6f}, improved={improvements}")

        if improvements == 0 and iteration % 100 == 0:
            # Perturb and restart
            pert = rng.normal(0, 0.005, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()

        if improvements == 0 and iteration > 2000:
            break

    return best_a, best_m


def log_nesterov_gradient(a0, time_limit, verbose=False, top_k=80, beta_init=4.0,
                           c0=5.0, restart_every=3000, rng_seed=42):
    """Nesterov-accelerated gradient in log-space with soft-subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = np.log(a + 1e-15)
    x_prev = x.copy()
    x_ymm = x.copy()
    beta_t = 1.0
    gamma = c0
    stall = 0
    cur_top_k = top_k
    cur_beta = beta_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        beta_t = (1.0 + np.sqrt(1.0 + 4.0 * beta_t**2)) / 2.0
        y = x + (beta_t - 1.0) / (beta_t + 1.0) * (x - x_prev)
        y_ymm = x + ((beta_t - 1.0) / (beta_t + 1.0)) ** 2 * (x - x_ymm) if it > 1 else x
        x_ymm = x_prev

        a_y = np.exp(y - y.max())
        s_y = a_y.sum()
        if s_y < 1e-15:
            a_y = a.copy()
        else:
            a_y = a_y / s_y

        ac = fast_autocorr(a_y)
        # Soft subgradient
        top_idx = np.argsort(ac)[::-1][:cur_top_k]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w = w / w_sum

        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            g[idx] += 2.0 * wi * a_y[j_idx]

        eta = gamma / (np.sqrt(it) + 1.0)
        x_new = y - eta * g
        x_prev = x
        x = x_new

        a_new = np.exp(x - x.max())
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every // 2:
            cur_beta = max(cur_beta * 0.9, 2.0)
            cur_top_k = min(cur_top_k + 10, 200)

        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            a_r = np.maximum(best_a + pert, 1e-10)
            a_r = a_r / a_r.sum()
            x = np.log(a_r)
            x_prev = x.copy()
            x_ymm = x.copy()
            cur_top_k = min(cur_top_k + 20, 250)

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LogNest it={it} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent polish with targeted moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 200
    n_top = 30

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit - 1:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            j_idx = ki - idx
            contrib = a[idx] * a[j_idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # try moving to random non-involved positions
            for _ in range(5):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.01, 0.005, 0.02, 0.03]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def spectral_pocs(a0, time_limit, verbose=False, n_trials=3, rng_seed=0):
    """POCS with flat-spectrum targeting."""
    n = len(a0)
    best_a = norm_sum1(a0)
    best_m = fast_autocorr(best_a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit - 5:
            break
        rng = np.random.default_rng(rng_seed + trial)
        a = best_a.copy()

        for it in range(200000):
            if time.monotonic() - t0 > time_limit - 5:
                break
            # Frequency domain: push toward flat magnitude
            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            blend = 0.05 + 0.15 * min(1.0, it / 5000)
            new_mag = (1 - blend) * mag + blend * mag_mean
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]

            # Time domain: non-negativity + normalization
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new / s

            M_new = fast_autocorr(a_new).max()
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = a_new.copy()
            a = a_new

        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    POCS trial {trial} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 6200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: LP-FW (most impactful) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP-FW (n={n0}) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        top_K_init = min(800, 2 * n0 - 1)
        top_K_max = min(2 * n0 - 1, 8000)
        a_out, _ = lp_frank_wolfe(a_in, t1, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max, max_iter=300)
        try_update(a_out, "lp1")

    # === Phase 2: Log Nesterov Gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Log Nesterov (n={n_best}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = log_nesterov_gradient(a_in, t2, verbose=True, top_k=80,
                                          beta_init=4.0, c0=5.0, rng_seed=42)
        try_update(a_out, "lognest1")

    # === Phase 3: Peak Shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Peak Shaving (n={n_best}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving(a_in, t3, n_top_lags=40, verbose=True, rng_seed=0)
        try_update(a_out, "peak1")

    # === Phase 4: LP-FW Round 2 (larger K) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW r2 (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 50 and n_best <= 7000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(1500, 2 * n_best - 1)
        top_K_max = min(2 * n_best - 1, 10000)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max, max_iter=200)
        try_update(a_out, "lp2")

    # === Phase 5: Coordinate Polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord Polish (n={n_best}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=False, rng_seed=456)
        try_update(a_out, "coord1")

    # === Phase 6: Multi-restart LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-restart LP-FW (n={n_best}) ===")
    t6_total = min(remaining * 0.15, 250.0)
    if t6_total > 80:
        t6_per = t6_total / 3
        for restart in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            rng = np.random.default_rng(1000 + restart)
            pert = rng.normal(0, 0.008, n_best) * best_seq
            a_pert = np.maximum(best_seq + pert, 1e-10)
            a_pert = a_pert / a_pert.sum()
            a_in = a_pert
            top_K = min(2000, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe(a_in, min(t6_per, remaining - 30), solver, verbose=False,
                                        top_K_init=min(1000, 2 * n_best - 1),
                                        top_K_max=top_K, max_iter=100)
            try_update(a_out, f"lp_restart{restart}")

    # === Phase 7: POCS Spectral ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: POCS (n={n_best}) ===")
    t7 = min(remaining * 0.08, 120.0)
    if t7 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_pocs(a_in, t7, verbose=True, n_trials=2, rng_seed=7)
        try_update(a_out, "pocs")

    # === Phase 8: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Length exploration ===")
    trial_lengths = [n_best - 200, n_best - 400, n_best + 200, n_best + 400, n_best + 600]
    trial_lengths = [l for l in trial_lengths if 4000 <= l <= 8000 and l != n_best]
    trial_lengths = list(dict.fromkeys(trial_lengths))[:3]

    for new_n in trial_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t8a = min(remaining * 0.4, 150.0)
        if t8a > 30:
            a_out, _ = log_nesterov_gradient(a_long, t8a, verbose=False, top_k=60,
                                              beta_init=4.0, c0=5.0, rng_seed=new_n % 1000)
            try_update(a_out, f"nest_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t8b = min(remaining * 0.5, 120.0)
            if t8b > 30:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                top_K = min(3000, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe(a_in, t8b, solver, verbose=False,
                                            top_K_init=min(800, 2 * new_n - 1),
                                            top_K_max=top_K, max_iter=100)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 9: Final aggressive optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final optimization (n={n_best}) ===")
    if remaining > 40:
        t9 = min(remaining * 0.35, 180.0)
        if t9 > 30:
            a_in = norm_sum1(best_seq)
            top_K = min(2 * n_best - 1, 10000)
            a_out, _ = lp_frank_wolfe(a_in, t9, solver, verbose=False,
                                        top_K_init=min(2000, 2 * n_best - 1),
                                        top_K_max=top_K, max_iter=200)
            try_update(a_out, "lp_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        t9b = min(remaining * 0.35, 150.0)
        if t9b > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_shaving(a_in, t9b, n_top_lags=50, verbose=False, rng_seed=777)
            try_update(a_out, "peak_final")

    # === Phase 10: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 10: Final polish (n={n_best}) ===")
        t10 = min(remaining * 0.7, 100.0)
        if t10 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t10, verbose=False, rng_seed=999)

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 6200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
