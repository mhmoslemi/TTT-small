def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_power_grad(y, power, top_k=None):
    """Compute gradient of (sum top_k b_i^power)^{1/power} w.r.t. a."""
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
    else:
        b_use = b
    b_p = np.maximum(b_use, 0.0) ** power
    S = b_p.sum()
    if S < 1e-30:
        return np.zeros(n), b_use
    # d/d a of S^{1/power} = (1/power) * S^{1/power - 1} * power * sum_i b_i^{power-1} * db_i/da
    # = S^{1/power - 1} * sum_i b_i^{power-1} * db_i/da
    weights = np.zeros(len(b_use))
    pos = b_use > 1e-15
    weights[pos] = b_use[pos] ** (power - 1)
    # db_k/da_j involves a[k-j] for valid j, so grad_j = sum_k weights[k] * a[k-j]
    a_rev = y[::-1]
    conv_len = n + len(weights) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Wr = np.fft.rfft(weights, fft_size)
    conv = np.fft.irfft(Ar * Wr, fft_size)
    grad = conv[n - 1:2 * n - 1] * (S ** (1.0 / power - 1.0))
    return grad, b_use


def compute_lse_grad(y, beta, top_k=None):
    """Compute gradient of (1/beta)*log(sum top_k exp(beta*b)) w.r.t. a."""
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, p, b

        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        if power_mode:
            grad, b = compute_power_grad(y, power, top_k)
            L = power * (fast_autocorr(y).max() ** (power - 1)) * max(1.0, y.max()) + 1.0
        else:
            grad, p, b = compute_lse_grad(y, beta, top_k)
            L = beta * p.max() * 2.0 * max(1.0, y.max()) + 1.0

        # Backtracking line search
        eta_trial = min(eta * 2.0, 0.5)
        eta_trial = max(eta_trial, 1e-6)
        func_y = (fast_autocorr(y).max())
        max_back = 5
        found_step = False
        for _ in range(max_back):
            z = y - eta_trial * grad
            x_candidate = simplex_project(z)
            max_c = fast_autocorr(x_candidate).max()
            if max_c < func_y - 1e-15 * eta_trial * np.linalg.norm(grad, ord=np.inf):
                found_step = True
                break
            eta_trial *= 0.5
        if not found_step:
            x_candidate = x
            eta = max(eta * 0.7, 1e-8)
        else:
            eta = eta_trial
            x_candidate = simplex_project(y - eta * grad)

        max_new = fast_autocorr(x_candidate).max()
        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_candidate.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_candidate
        t_nest = t_new

        # Adaptive parameter increase
        if t % 100 == 0:
            if power_mode:
                power = min(power * power_growth, power_end)
            else:
                beta = min(beta * beta_growth, beta_end)

        # Restart
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.03 / (1.0 + 0.2 * (t // restart_every))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if power_mode:
                power = max(power * 0.5, power_start)
            else:
                beta = max(beta * 0.5, beta_start * 0.5)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            param_str = "beta=" + repr(round(beta, 1)) if not power_mode else "pwr=" + repr(round(power, 1))
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " " + param_str)

    return best_a, best_max_b


def peak_pair_cd(a0, time_limit, verbose=True, seed=123):
    """Peak-pair directional coordinate descent: targets top peaks simultaneously."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 1500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_peaks = min(15, len(ac))
        top_idx = np.argsort(ac)[::-1][:n_peaks]

        improved = False
        for k in top_idx[:8]:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-20:
                continue

            # Find the top contributing pairs
            pair_order = np.argsort(contrib)[::-1][:5]
            for pi in pair_order[:3]:
                src = int(idx[pi])
                partner = k_int - src
                if partner < 0 or partner >= n:
                    continue
                if a[src] < 1e-15 and a[partner] < 1e-15:
                    continue

                # Determine which to reduce (reduce the larger, or the one with more "pressure")
                # Compute pressure: how much does this position contribute to other top peaks?
                pressure_src = 0.0
                pressure_part = 0.0
                for k2 in top_idx[:6]:
                    k2i = int(k2)
                    if 0 <= k2i - src < n:
                        pressure_src += a[src] * max(a[k2i - src], 0.0)
                    if 0 <= k2i - partner < n:
                        pressure_part += a[partner] * max(a[k2i - partner], 0.0)

                if pressure_src >= pressure_part:
                    reduce_idx = src
                    keep_idx = partner
                else:
                    reduce_idx = partner
                    keep_idx = src

                if a[reduce_idx] < 1e-12:
                    continue

                # Find destination: position with lowest total pressure
                # Sample candidate positions
                n_cand = min(200, n // 5)
                cand_positions = rng.choice(n, size=n_cand, replace=False)
                pressures = np.zeros(len(cand_positions))
                for k2 in top_idx[:6]:
                    k2i = int(k2)
                    offsets = k2i - cand_positions
                    valid = (offsets >= 0) & (offsets < n)
                    pressures[valid] += a[cand_positions[valid]] * a[offsets[valid]]
                    valid2 = (cand_positions >= 0) & (cand_positions < n) & ((k2i - cand_positions) >= 0) & ((k2i - cand_positions) < n)
                    pressures[valid2] += a[k2i - cand_positions[valid2]] * a[cand_positions[valid2]]

                # Exclude the reduce position and its partner
                valid_cand = np.where(cand_positions != reduce_idx)[0]
                if len(valid_cand) == 0:
                    continue
                valid_cand = np.where(cand_positions[valid_cand] != keep_idx)[0]
                if len(valid_cand) == 0:
                    continue
                dst = int(cand_positions[valid_cand[np.argmin(pressures[valid_cand])]])

                # Try moving a fraction
                found = False
                for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                    amt = a[reduce_idx] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_idx] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break

            if time.monotonic() - t0 > time_limit:
                break

        if not improved:
            stall += 1
            # Random exploration when stalled
            if stall % 50 == 0:
                for _ in range(3):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True

        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PCD (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def lp_fw(a0, time_limit, solver, initial_K=400, max_K=2500,
          add_per_iter=25, verbose=True):
    """LP-Frank-Wolfe with active set management."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 800

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K // 2])
            no_improve += 10
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 10
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            no_improve += 10
            continue
        g_dir = g_val / g_sum

        # Find violations
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5),
                  max(0.02, default_alpha * 1.5), 1.0 / (t + 1)]
        alphas = [max(0.01, min(1.0, a)) for a in alphas]
        alphas = list(set(round(a, 6) for a in alphas))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = a.copy()
            else:
                y = y / s_y

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def parametric_init(n, time_limit, best_init, best_score, seed=7):
    """Quick parametric search for good initialization."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score
    rng = np.random.default_rng(seed)

    candidates = []
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr + 1.0) ** p)
    for p in [0.5, 1.0, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p * (n - i_arr + 1.0) ** p)
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.3]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    for freq in [1, 2, 3, 4]:
        candidates.append(np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x)) / 2.0, 0.01))
    for p in [1, 2, 3, 4, 6]:
        candidates.append(np.sin(np.pi * x) ** p)
    # Gaussian mixtures
    for _ in range(10):
        c1 = rng.uniform(0.2, 0.8)
        c2 = 1.0 - c1
        s1 = rng.uniform(0.05, 0.2)
        s2 = rng.uniform(0.05, 0.2)
        w1 = rng.uniform(0.3, 0.7)
        candidates.append(w1 * np.exp(-((x - c1) ** 2) / (2 * s1 ** 2)) +
                         (1 - w1) * np.exp(-((x - c2) ** 2) / (2 * s2 ** 2)))

    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(a_raw)
        b = fast_autocorr(a)
        m = b.max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    # Random structured
    for _ in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        a = rng.random(n)
        a = a ** rng.uniform(0.5, 4.0)
        a = norm_sum1(a)
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    return best_a, best_val


def propose_candidate(budget_s=1800.0):
    """Main search: multi-method optimization with length exploration."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(round(p_score, 6)))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(round(best_score, 6)))

    n0 = len(best_seq)

    # Phase 1: Parametric search at current length
    print("\n=== Phase 1: Parametric (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 40.0)
    if t1 > 10:
        a_param, _ = parametric_init(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "param", t_global)

    # Phase 2: Main Nesterov (LSE mode)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov LSE (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 60:
        a_out, _ = nesterov_improved(norm_sum1(best_seq), t2,
                                     beta_start=20.0, beta_end=600.0,
                                     beta_growth=1.003, top_k_ratio=0.04,
                                     power_mode=False, restart_every=2000,
                                     verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_lse_n" + str(n_best), t_global)

    # Phase 3: Peak-pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Peak-pair CD (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 30:
        a_out, _ = peak_pair_cd(norm_sum1(best_seq), t3, verbose=True, seed=123)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd_n" + str(n_best), t_global)

    # Phase 4: LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 40 and n_best <= 7500:
        a_out, _ = lp_fw(norm_sum1(best_seq), t4, solver,
                         initial_K=500, max_K=2500,
                         add_per_iter=25, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(n_best), t_global)

    # Phase 5: Nesterov (Power mode) - different smoothing
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Nesterov Power (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 40:
        a_out, _ = nesterov_improved(norm_sum1(best_seq), t5,
                                     power_mode=True,
                                     power_start=2.0, power_end=40.0,
                                     power_growth=1.003, top_k_ratio=0.03,
                                     restart_every=1800, verbose=True, seed=77)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_pow_n" + str(n_best), t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-length exploration ===")

    # Try a range of lengths centered around current best
    length_offsets = [-500, -200, 200, 500, 800, 1200]
    tried_lengths = set()

    for offset in length_offsets:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        new_n = n_best + offset
        if new_n < 2000 or new_n > 9000 or new_n in tried_lengths:
            continue
        tried_lengths.add(new_n)

        print("\n  --- Trying n=" + str(new_n) + " ---")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.20, 150.0)
        if t6a > 30:
            a_out, _ = nesterov_improved(a_long, t6a,
                                         beta_start=25.0, beta_end=500.0,
                                         beta_growth=1.004, top_k_ratio=0.04,
                                         power_mode=False, restart_every=1500,
                                         verbose=False, seed=new_n % 1000)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t6b = min(remaining * 0.15, 100.0)
            if t6b > 20:
                a_long2 = interpolate_sequence(best_seq, new_n)
                a_out, _ = peak_pair_cd(a_long2, t6b, verbose=False, seed=new_n % 2000 + 500)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd_n" + str(new_n), t_global)

    # Phase 7: Final Nesterov polish at best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final Nesterov (n=" + str(n_best) + ") ===")
    if remaining > 60:
        t7 = min(remaining * 0.30, 250.0)
        if t7 > 30:
            a_out, _ = nesterov_improved(norm_sum1(best_seq), t7,
                                         beta_start=50.0, beta_end=1000.0,
                                         beta_growth=1.005, top_k_ratio=0.02,
                                         power_mode=False, restart_every=1500,
                                         verbose=True, seed=999)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final_n" + str(n_best), t_global)

    # Phase 8: Final LP-FW burst
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40 and n_best <= 7500:
        print("\n=== Phase 8: Final LP-FW (n=" + str(n_best) + ") ===")
        t8 = min(remaining * 0.40, 200.0)
        if t8 > 20:
            a_out, _ = lp_fw(norm_sum1(best_seq), t8, solver,
                             initial_K=600, max_K=3000,
                             add_per_iter=30, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Phase 9: Final peak-pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print("\n=== Phase 9: Final PCD (n=" + str(n_best) + ") ===")
        t9 = min(remaining * 0.50, 150.0)
        if t9 > 15:
            a_out, _ = peak_pair_cd(norm_sum1(best_seq), t9, verbose=True, seed=5555)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd_final", t_global)

    # Final validation
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = np.minimum(best_seq, 1000.0)
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
