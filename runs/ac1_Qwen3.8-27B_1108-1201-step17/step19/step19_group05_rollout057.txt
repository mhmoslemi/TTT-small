"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using ramp-shaped initializations,
Nesterov mirror descent with peak-subgradient, spectral flattening, and
coordinate descent polishing. The linear ramp gives J -> 4/3 asymptotically,
far better than the parent's 1.504. We optimize the shape to go below 4/3.

import numpy as np
import time


def fast_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def make_ramp(n, alpha=1.0):
    """Create normalized (i+1)^alpha sequence."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** alpha
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_damped_ramp(n, alpha=1.0, damping=0.0):
    """Create normalized (i+1)^alpha * exp(-damping * i/n) sequence."""
    x = np.arange(1, n + 1, dtype=np.float64)
    t = np.arange(n, dtype=np.float64) / n
    a = x ** alpha * np.exp(-damping * t)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_tapered_ramp(n, alpha=1.0, beta=2.0, strength=0.3):
    """Create normalized (i+1)^alpha * (1 - strength*(i/n)^beta) sequence."""
    x = np.arange(1, n + 1, dtype=np.float64)
    t = np.arange(n, dtype=np.float64) / n
    a = x ** alpha * (1.0 - strength * t ** beta)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return make_ramp(n, alpha)
    return a / s


def make_bump(n, shape='sine'):
    """Create a bump-shaped sequence (high in middle, low at ends)."""
    x = np.arange(n, dtype=np.float64)
    if shape == 'sine':
        a = np.sin(np.pi * (x + 0.5) / n) ** 2
    elif shape == 'parab':
        t = 2.0 * x / max(1, n - 1) - 1.0
        a = 1.0 - t ** 2
    elif shape == 'cos':
        a = 0.5 * (1.0 + np.cos(np.pi * x / max(1, n - 1)))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    return a / s


def compute_subgrad_peaks(a, conv, n_peaks=50, tau=None):
    """Compute subgradient using top peaks or softmax weighting."""
    n = len(a)
    if tau is not None and tau > 0:
        # Soft gradient using log-sum-exp style
        max_v = conv.max()
        exp_v = np.exp((conv - max_v) / tau)
        w = exp_v / exp_v.sum()
        active = np.where(w > 1e-10 / n)[0]
    else:
        top_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
        vals = conv[top_idx]
        max_v = vals.max()
        exp_v = np.exp((vals - max_v) / max(1e-10, max_v * 0.01))
        w = exp_v / exp_v.sum()
        active = top_idx
        w = np.array([w[list(active).index(k)] if k in list(active) else 0 for k in active])
        # Simpler: just use the weights as computed
        w_map = {int(k): w[i] for i, k in enumerate(top_idx)}
        w = np.array([w_map.get(int(k), 0.0) for k in active])

    grad = np.zeros(n)
    for k in active:
        ki = int(k)
        wi = w_map.get(ki, w[list(active).index(k)] if hasattr(w, '__index__') else 0.0)
        if tau is not None:
            wi = w[list(active).index(k)] if k in list(active) else 0
        else:
            pass
        # Compute contribution
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        j_range = np.arange(j_lo, j_hi + 1)
        pair = ki - j_range
        grad[j_range] += wi * a[pair]
    return grad


def compute_subgrad_simple(a, conv, n_peaks=50, tau=None):
    """Simpler subgradient computation."""
    n = len(a)
    if tau is not None and tau > 0:
        max_v = conv.max()
        exp_v = np.exp((conv - max_v) / tau)
        w = exp_v / exp_v.sum()
        # Use all positions with significant weight
        threshold = w.max() * 1e-8
        active_mask = w > threshold
        active_idx = np.where(active_mask)[0]
        active_w = w[active_mask]
    else:
        top_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
        vals = conv[top_idx]
        max_v = vals.max()
        exp_v = np.exp((vals - max_v) / max(1e-10, max_v * 0.01))
        w = exp_v / exp_v.sum()
        active_idx = top_idx
        active_w = w

    grad = np.zeros(n)
    for i in range(len(active_idx)):
        ki = int(active_idx[i])
        wi = active_w[i]
        if wi < 1e-15:
            continue
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        j_range = np.arange(j_lo, j_hi + 1)
        pair = ki - j_range
        grad[j_range] += wi * 2.0 * a[pair]
    return grad


def mirror_descent(a0, time_limit, n_peaks=50, lr0=2.0, restart_every=2000,
                   rng_seed=42, verbose=False, soft_tau=None):
    """Nesterov-accelerated mirror descent on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    restart_count = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-20)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        conv = fast_conv(y)
        g = compute_subgrad_simple(y, conv, n_peaks=n_peaks, tau=soft_tau)

        eta = lr0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = conv.max()
        M_x = fast_conv(x).max()
        M_eff = min(M_new, M_x)
        if M_eff < best_m - 1e-16:
            best_m = M_eff
            best_a = (x if M_x <= M_new else y).copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            restart_count += 1
            strength = min(0.15, 0.02 + 0.01 * min(restart_count, 8))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-15)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

        if verbose and t % 5000 == 0:
            print(f"      MD t={t} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}, stall={stall}")

    return best_a, best_m


def spectral_flatten(a0, time_limit, alpha_start=0.2, verbose=False):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_mean = P.mean()
        if P_max < 1e-30:
            break

        alpha = alpha_start + (1.0 - alpha_start) * min(1.0, it / 2000.0) * 0.5
        P_new = P + alpha * (P_mean - P)
        P_new = np.maximum(P_new, 1e-30)

        mag_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        mag_ratio = np.clip(mag_ratio, 0.3, 3.0)
        A_new = A * mag_ratio
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            blend = 0.1 + 0.3 * min(1.0, it / 1000.0)
            a = (1 - blend) * a + blend * a_new
            a = np.maximum(a, 0.0)
            a = a / max(a.sum(), 1e-15)
            M_curr = fast_conv(a).max()
            if M_curr < best_m:
                best_m = M_curr
                best_a = a.copy()

        if verbose and it % 500 == 0:
            print(f"      Spec it={it} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=42, verbose=False):
    """Direct coordinate moves to reduce peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        conv = fast_conv(a)
        k_max = int(np.argmax(conv))

        j_lo = max(0, k_max - n + 1)
        j_hi = min(n - 1, k_max)
        if j_lo > j_hi:
            stall += 1
            continue
        idx = np.arange(j_lo, j_hi + 1)
        contrib = a[idx] * a[k_max - idx]
        src = idx[np.argmax(contrib)]

        if a[src] < 1e-12:
            stall += 1
            continue

        # Find good destination
        top_ks = np.argpartition(conv, -15)[-15:]
        costs = np.zeros(n)
        for k2 in top_ks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            if lo2 <= hi2:
                valid = np.arange(lo2, hi2 + 1)
                costs[valid] += a[k2i - valid]

        sorted_idx = np.argsort(costs * 0.1 + a)
        dst = sorted_idx[0]
        if dst == src:
            dst = sorted_idx[1] if len(sorted_idx) > 1 else int(rng.integers(n))

        moved = False
        for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_t = fast_conv(a_trial).max()
            if M_t < best_m - 1e-16:
                a = a_trial
                best_m = M_t
                best_a = a.copy()
                moved = True
                stall = 0
                break

        if not moved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.01, 0.05)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m
