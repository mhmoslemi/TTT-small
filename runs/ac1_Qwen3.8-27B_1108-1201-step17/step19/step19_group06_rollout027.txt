import time

def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

    """Score for a normalized (sum=1) non-negative sequence."""
    n = len(a)
    c = fast_conv(a)
    return float(2.0 * n * c.max())

def subgradient_step(a, c, beta, top_frac=0.5):
    """
    Compute the subgradient of max(c) using soft-max weighting.
    Returns gradient vector of shape (n,).
    """
    n = len(a)
    M = c.max()
    # Select lags above threshold
    threshold = M * (1.0 - top_frac * 0.5)  # e.g., top 50% of range
    active = np.where(c >= threshold)[0]

    if w_sum < 1e-30:

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FFT-based subgradient descent with backtracking line search
2. Adaptive soft-max active set selection over autocorrelation lags
3. Vectorized gradient computation via per-lag slice updates
4. Polyak momentum with adaptive learning rate
5. Multi-scale restarts with structured diverse perturbations
6. Coordinate descent local refinement
7. Length exploration via interpolation
8. Spectral shaping in frequency domain

import numpy as np
import time


def fft_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_score(a):
    """Score = 2*n*max(conv) for normalized (sum=1) sequence."""
    n = len(a)
    c = fft_conv(a)
    return float(2.0 * n * c.max())


def compute_gradient(a, c, beta, top_k):
    """
    Compute subgradient of max(c) using soft-max weighting over top-k lags.
    Vectorized per-lag slice updates.
    """
    n = len(a)
    k_alarm = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_alarm]
    top_vals = c[top_idx]

    # Soft-max weights
    max_v = top_vals.max()
    expo = beta * (top_vals - max_v)
    expo = np.clip(expo, -50, 50)
    weights = np.exp(expo)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_alarm) / k_alarm
    else:
        weights = weights / w_sum

    # Compute gradient via vectorized per-lag updates
    grad = np.zeros(n)
    for k_int, w in zip(top_idx, weights):
        k_int = int(k_int)
        # grad[j] += 2*w*a[k-j] for valid j
        # a[k-j] valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_lo = max(0, k_int - n + 1)
        j_hi = min(n, k_int + 1)
        if j_lo >= j_hi:
            continue
        # a[k-j] for j in [j_lo, j_hi): indices k-j_lo, k-j_lo-1, ..., k-(j_hi-1)
        a_idx_hi = k_int - j_lo
        a_idx_lo = k_int - (j_hi - 1)
        grad[j_lo:j_hi] += 2.0 * w * a[a_idx_lo:a_idx_hi + 1][::-1]

    return grad


def subgradient_optimize(a0, time_limit, lr_init=0.05, beta_init=3.0, top_k_init=30,
                         momentum_init=0.8, verbose=True, rng_seed=42):
    """
    Subgradient descent with backtracking line search, momentum, and adaptive restarts.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    grad_avg = np.zeros(n)
    lr = lr_init
    beta = beta_init
    top_k = top_k_init
    momentum = momentum_init
    stall = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        c = fft_conv(a)
        M_curr = c.max()
        score_curr = 2.0 * n * M_curr

        if score_curr < best_score - 1e-13:
            best_score = score_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive parameters
        if stall > 0 and stall % 500 == 0:
            top_k = min(top_k + 3, 150)
        if stall > 5000:
            # Restart
            stall = 0
            lr = lr_init * rng.uniform(0.5, 1.5)
            beta = beta_init * rng.uniform(0.5, 2.0)
            top_k = top_k_init
            momentum = momentum_init
            grad_avg = np.zeros(n)

            # Diverse perturbation
            pert_type = int(rng.integers(6))
            x = np.linspace(0, 1, n)
            if pert_type == 0:
                # Raised cosine
                pert = 0.5 * (1 + np.cos(2 * np.pi * x))
            elif pert_type == 1:
                # Gaussian
                c_pos = float(rng.uniform(0.3, 0.7))
                sig = float(rng.uniform(0.1, 0.4))
                pert = np.exp(-((x - c_pos) ** 2) / (2 * sig ** 2))
            elif pert_type == 2:
                # Bimodal
                sig = float(rng.uniform(0.08, 0.25))
                pert = (np.exp(-((x - 0.25) ** 2) / (2 * sig ** 2)) +
                        np.exp(-((x - 0.75) ** 2) / (2 * sig ** 2)))
            elif pert_type == 3:
                # Periodic (sine-squared)
                freq = int(rng.integers(1, 6))
                phase = float(rng.uniform(0, np.pi))
                pert = np.sin(np.pi * freq * x + phase) ** 2
            elif pert_type == 4:
                # Power law
                i_arr = np.arange(n, dtype=np.float64)
                p = float(rng.uniform(0.3, 3.0))
                pert = (i_arr + 1.0) ** p * (n - i_arr + 1.0) ** p
            else:
                # Random smooth
                pert = best_a * (1.0 + rng.normal(0, 0.1, n))
                for _ in range(min(3, n // 100)):
                    idx = int(rng.integers(n))
                    win = int(rng.integers(10, max(20, n // 50)))
                    pert[max(0, idx - win):min(n, idx + win)] += float(rng.uniform(0.1, 0.5))

            pert = normalize_nonneg(pert)
            w_mix = float(rng.uniform(0.1, 0.5))
            a = normalize_nonneg((1 - w_mix) * best_a + w_mix * pert)
            grad_avg = np.zeros(n)
            if verbose:
                print(f"      Restart@{iters} ({time.monotonic()-t0:.0f}s): score={best_score:.6f}, beta={beta:.1f}, lr={lr:.4f}")

        # Compute gradient
        grad = compute_gradient(a, c, beta, top_k)

        # Momentum (Polyak)
        grad_avg = momentum * grad_avg + (1.0 - momentum) * grad

        # Backtracking line search (try 3 step sizes)
        best_step_score = score_curr
        best_step_a = a.copy()
        for trial in range(3):
            lr_try = lr * (0.5 ** trial)
            a_trial = a - lr_try * grad_avg
            a_trial = normalize_nonneg(a_trial)
            c_trial = fft_conv(a_trial)
            score_trial = 2.0 * n * c_trial.max()
            if score_trial < best_step_score:
                best_step_score = score_trial
                best_step_a = a_trial
                if trial == 0:
                    lr = min(lr * 1.1, 0.5)  # Can increase lr
                    break
        else:
            # None improved, use smallest step
            a_trial = a - lr * 0.125 * grad_avg
            a_trial = normalize_nonneg(a_trial)
            best_step_a = a_trial
            lr *= 0.7

        a = best_step_a
        iters += 1

        if verbose and iters % 5000 == 0:
            print(f"      it={iters} ({time.monotonic()-t0:.0f}s): score={best_score:.6f}, "
                  f"beta={beta:.1f}, lr={lr:.5f}, top_k={top_k}, stall={stall}")

    if verbose:
        print(f"      Subgrad done ({time.monotonic()-t0:.0f}s): best={best_score:.6f}, iters={iters}")
    return best_a, best_score


def spectral_shaped_init(n, rng):
    """Generate a sequence with controlled spectral shape."""
    fft_size = 1 << (2 * n - 2).bit_length()
    # Design a power spectrum
    freqs = np.fft.rfftfreq(fft_size)
    # Low-pass with some structure
    alpha = float(rng.uniform(1.0, 5.0))
    P = np.exp(-alpha * (freqs * fft_size / 2.0) ** 2)
    # Add some randomness
    P *= (1.0 + float(rng.uniform(0.1, 0.3)) * rng.random(fft_size // 2 + 1))
    # Take square root and inverse FFT
    A = np.sqrt(P) * np.exp(1j * rng.uniform(0, 2 * np.pi, fft_size // 2 + 1))
    A[0] = np.abs(A[0])  # Ensure DC is positive (for non-negativity bias)
    a = np.fft.irfft(A, fft_size)[:n]
    return normalize_nonneg(a)


def coordinate_refine(a0, time_limit, rng_seed=77, verbose=True):
    """
    Local refinement: try small adjustments to individual elements
    guided by the gradient.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        c = fft_conv(a)
        score_curr = 2.0 * n * c.max()

        if score_curr < best_score - 1e-13:
            best_score = score_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                # Perturb
                n_pert = int(rng.integers(3, min(20, n // 10)))
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = float(rng.uniform(0.005, 0.05))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = normalize_nonneg(a)
                stall = 0

        # Compute gradient to find most "expensive" positions
        beta = 5.0
        top_k = min(50, 2 * n - 1)
        grad = compute_gradient(a, c, beta, top_k)

        # Find positions with highest gradient (most impactful to reduce)
        n_try = min(30, n)
        # Combine gradient magnitude with current value
        impact = np.abs(grad) * np.maximum(a, 1e-10)
        top_positions = np.argsort(impact)[::-1][:n_try]

        improved = False
        for pos in top_positions:
            if time.monotonic() - t0 > time_limit:
                break
            pos = int(pos)
            val = a[pos]
            if val < 1e-12:
                continue

            # Try reducing this position and distributing to low-impact positions
            low_impact = np.argsort(impact)
            # Pick a few target positions
            n_targets = 3
            for frac in [0.05, 0.1, 0.2]:
                amt = val * frac
                a_trial = a.copy()
                a_trial[pos] -= amt
                # Distribute to low-impact positions
                for t_idx in low_impact[:n_targets]:
                    t_idx = int(t_idx)
                    if t_idx != pos:
                        a_trial[t_idx] += amt / n_targets
                a_trial = normalize_nonneg(a_trial)
                c_trial = fft_conv(a_trial)
                score_trial = 2.0 * n * c_trial.max()
                if score_trial < best_score - 1e-12:
                    a = a_trial
                    best_score = score_trial
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Also try direct perturbation
        if not improved and iters % 10 == 0:
            pos = int(top_positions[int(rng.integers(len(top_positions)))])
            for frac in [0.02, 0.05, 0.1]:
                amt = a[pos] * frac
                a_trial = a.copy()
                a_trial[pos] -= amt
                # Spread uniformly
                a_trial[:] += amt / n
                a_trial = normalize_nonneg(a_trial)
                c_trial = fft_conv(a_trial)
                score_trial = 2.0 * n * c_trial.max()
                if score_trial < best_score - 1e-12:
                    a = a_trial
                    best_score = score_trial
                    best_a = a.copy()
                    improved = True
                    break

        iters += 1

    if verbose:
        print(f"      Coord done ({time.monotonic()-t0:.0f}s): best={best_score:.6f}, iters={iters}")
    return best_a, best_score


def interpolate_sequence(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_int = np.interp(x_new, x_old, a_norm)
    a_int = np.maximum(a_int, 0.0)
    ss = a_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_int / ss


def parametric_init(n, rng_seed=0):
    """Try several parametric initializations and return the best."""
    rng = np.random.default_rng(rng_seed)
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []

    # Raised cosine
    candidates.append(normalize_nonneg(0.5 * (1 + np.cos(2 * np.pi * x))))
    # Hamming
    candidates.append(normalize_nonneg(0.54 - 0.46 * np.cos(2 * np.pi * x)))
    # Hanning
    candidates.append(normalize_nonneg(0.5 * (1 - np.cos(2 * np.pi * x))))
    # Gaussian center
    candidates.append(normalize_nonneg(np.exp(-((x - 0.5) ** 2) / 0.08)))
    # Bimodal
    candidates.append(normalize_nonneg(
        np.exp(-((x - 0.25) ** 2) / 0.02) + np.exp(-((x - 0.75) ** 2) / 0.02)))
    # Power law symmetric
    for p in [0.5, 1.0, 1.5, 2.0]:
        candidates.append(normalize_nonneg((i_arr + 1.0) ** p * (n - i_arr + 1.0) ** p))
    # Sine squared
    for f in [1, 2, 3]:
        candidates.append(normalize_nonneg(np.sin(np.pi * f * x) ** 2))
    # Exponential decay (both sides)
    for r in [1, 2, 3]:
        candidates.append(normalize_nonneg(np.exp(-r * x) + np.exp(-r * (1 - x))))
    # Uniform
    candidates.append(np.ones(n) / n)

    best = None
    best_sc = float('inf')
    for c in candidates:
        sc = compute_score(c)
        if sc < best_sc:
            best_sc = sc
            best = c.copy()

    # Also try spectral
    for _ in range(5):
        c = spectral_shaped_init(n, rng)
        sc = compute_score(c)
        if sc < best_sc:
            best_sc = sc
            best = c.copy()

    return best, best_sc
