def simplex_project(z):
    """Project z onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def grad_qk(a, k):
    """Gradient of q_k(a) = sum_i a[i]*a[k-i] w.r.t. a[j] = 2*a[k-j] (0 if out of range)."""
    n = len(a)
    # a[k-j] for j=0..n-1, where a[x]=0 if x<0 or x>=n
    # k-j ranges from k to k-(n-1)
    j = np.arange(n)
    idx = k - j
    grad = np.zeros(n)
    valid = (idx >= 0) & (idx < n)
    grad[valid] = 2.0 * a[idx[valid]]
    return grad

def grad_qk_fast(a, k, n):
    """Fast gradient: grad[j] = 2*a[k-j] if 0<=k-j<n else 0."""
    # grad is a shifted version of a
    # grad[j] = 2*a[k-j] for j where k-j is in [0, n-1]
    # i.e., j in [max(0, k-n+1), min(k, n-1)]
    j_start = max(0, k - n + 1)
    j_end = min(k, n - 1)
    if j_start > j_end:
        return np.zeros(n)
    grad = np.zeros(n)
    # j goes from j_start to j_end
    # k-j goes from k-j_start to k-j_end
    idx_start = k - j_end  # minimum index in a
    idx_end = k - j_start  # maximum index in a
    grad[j_start:j_end+1] = 2.0 * a[idx_start:idx_end+1][::-1]

def compute_subgrad(a, q, K=50, beta=1.0):
    """Compute weighted subgradient using top-K peaks."""
    n = len(a)
    size = 2 * n - 1
    
    # Find top K peaks
    if K < size:
        top_idx = np.argpartition(q, -K)[-K:]
        top_vals = q[top_idx]
        # Softmax weights
        shifted = beta * (top_vals - top_vals.max())
        w = np.exp(shifted)
        w /= w.sum()
    else:
        top_idx = np.arange(size)
        top_vals = q
        shifted = beta * (top_vals - top_vals.max())
        w = np.exp(shifted)
        w /= w.sum()
    
    # Create weight vector
    w_full = np.zeros(size)
    w_full[top_idx] = w
    
    # grad[j] = 2 * sum_k w_full[k] * a[k-j] (for valid k-j)
    # This is 2 * convolve(reversed(a), w_full)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Wr = np.fft.rfft(w_full, fft_size)
    conv = np.fft.irfft(Ar * Wr, fft_size)

            print("  SG t=" + str(t) + " (" + repr(el) + "s): J=" + repr(2.0 * n * best_m) + " beta=" + repr(beta) + " K=" + str(K))

def _lp_fw(a0, tlim, solver, K0=200, Kmax=1500, verbose=True):
    """LP-Frank-Wolfe with targeted active set."""
    n = len(a0)
    a = _norm1(a0)
    best_a = a.copy()
    best_m = _autocorr(a).max()
    t0 = time.monotonic()
    
    ac = _autocorr(a)
    active = set(int(k) for k in np.argsort(ac)[::-1][:K0])
    no_imp = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > tlim:
            break
        if no_imp > 600:
            if verbose:
                print("  LP-FW: converged iter " + str(t))
            break
        
        ac = _autocorr(a)
        Mc = ac.max()
        
        if Mc < best_m - 1e-16:
            best_m = Mc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Manage active set
        cl = sorted(active)
        if len(cl) > Kmax:
            margins = np.array([Mc - ac[k] for k in cl])
            keep = np.argsort(margins)[:Kmax]
            active = set(int(cl[i]) for i in keep)
            cl = sorted(active)
        
        K = len(cl)
        if K == 0:
            break
        
        # Build constraint matrix
        C = np.zeros((K, n))
        for jj, k in enumerate(cl):
            ks = max(0, k - n + 1)
            ke = min(n - 1, k)
            if ks <= ke:
                idx = np.arange(ks, ke + 1)
                C[jj, idx] = a[k - idx]
        
        g = cp.Variable(n, nonneg=True)
        Mv = cp.Variable()
        cons = [cp.sum(g) == 1, C @ g <= Mv]
        prob = cp.Problem(cp.Minimize(Mv), cons)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active) > K0 // 2:
                ac2 = _autocorr(a)
                active = set(int(k) for k in np.argsort(ac2)[::-1][:K0 // 2])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        gv = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        gs = gv.sum()
        if gs < 1e-15:
            continue
        gdir = gv / gs
        
        # Find violations
        sz = 2 * n - 1
        fs = 1 << (sz - 1).bit_length()
        Af = np.fft.rfft(a, fs)
        Gf = np.fft.rfft(gdir, fs)
        cv = np.fft.irfft(Af * Gf, fs)[:sz]
        viol = np.where(cv > Mc + 1e-7)[0]
        if len(viol) > 0:
            va = cv[viol] - Mc
            top_v = viol[np.argsort(va)[::-1][:20]]
            for k in top_v:
                active.add(int(k))
        
        # Line search
        alphas = [2.0 / (t + 2.0), 1.0 / (t + 1), 0.5, 0.1, 0.05]
        alphas = [max(0.01, min(1.0, x)) for x in alphas]
        alphas = list(dict.fromkeys(alphas))  # remove duplicates preserving order
        
        ba = alphas[0]
        bm = float('inf')
        for al in alphas:
            tr = (1.0 - al) * a + al * gdir
            tr = np.maximum(tr, 0.0)
            ts = tr.sum()
            if ts < 1e-15:
                continue
            tr = tr / ts
            mt = _autocorr(tr).max()
            if mt < bm:
                bm = mt
                ba = al
        
        if t == 1:
            a = gdir.copy()
        else:
            a = (1.0 - ba) * a + ba * gdir
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
        
        if verbose and t % 100 == 0:
            el = time.monotonic() - t0
            print("  LP iter=" + str(t) + " (" + repr(el) + "s): J=" + repr(2.0 * n * best_m) + " |S|=" + str(len(active)))
    
    return best_a
