def compute_r(f, lags, weights):

        return np.array([weights[0]*f[0]])

    nz=weights>0
    if not np.any(nz): return np.zeros(n)
    lags=lags[nz]; weights=weights[nz]
    if len(lags)*n <= 250000:
        r=np.zeros(n)
        for kk, ww in zip(lags, weights):
            ks=int(kk)
            start=ks-n+1
            if start<0: start=0
            end=ks+1
            if end>n: end=n
            if start<end:
                a_idx=ks-start
                b_idx=ks-(end-1)
                r[start:end] += ww * f[b_idx:a_idx+1][::-1]
        r=np.maximum(r,0)
        return r
    else:

        fft_size=1 << (size-1).bit_length()

        F=np.fft.rfft(rev, fft_size)
        W=np.fft.rfft(w_full, fft_size)
        corr=np.fft.irfft(F*W, fft_size)
        r=corr[n-1:n-1+n]
        r=np.maximum(r,0)

def interpolate_seq(f, new_n):
    old=norm1(f)
    old_n=old.size
    if new_n <=0: return old.copy()
    if old_n == new_n: return old.copy()
    if old_n == 1: return np.ones(new_n, dtype=np.float64)
    xo=np.linspace(0.0,1.0,old_n)
    xn=np.linspace(0.0,1.0,new_n)
    g=np.interp(xn, xo, old)
    g=np.maximum(g,0.0)
    s=g.sum()
    if s<=1e-15: return np.ones(new_n)/new_n
    return g/s
```

def build_active_matrix(f, lags):
    n=f.size; K=len(lags); C=np.zeros((K,n),float)
    for row, kk in enumerate(lags):
       ks=int(kk); start=ks-n+1
       if start<0: start=0
       end=ks+1
       if end>n: end=n
       if start<end:
          a_idx=ks-start; b_idx=ks-(end-1)
          C[row,start:end]=f[b_idx:a_idx+1][::-1]
    return C

"""
Searches for non-negative step-function heights minimizing

    2*n*max(np.convolve(a,a)) / (sum(a))**2

The code starts from height_sequence_1 when available, tries several
lengths and parametric probes, then refines with:
  * stochastic mirror descent on the log-simplex driven by active
    convolution peaks;
  * an optional Frank-Wolfe linear oracle (cvxpy, skipped if unavailable);
  * pairwise peak-shaving polish;
  * final exact validation via the provided evaluate_sequence().
"""

import numpy as np
import time
import math

try:
    import cvxpy as cp
except Exception:
    cp = None

_LP_READY = False
_LP_SOLVER = None


def _next_pow2(x):
    return 1 << max(0, int(x - 1).bit_length())


def safe_parse_seq(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.zeros(0, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    return arr


def norm1(a):
    arr = safe_parse_seq(a)
    if arr.size == 0:
        return np.zeros(0, dtype=np.float64)
    arr = np.maximum(arr, 0.0)
    s = arr.sum()
    if s <= 1e-15:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    arr /= s
    return arr


def fast_convolve(a):
    arr = safe_parse_seq(a)
    n = arr.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    F = np.fft.rfft(arr, fft_size)
    conv = np.fft.irfft(F * F, fft_size)
    conv = conv[:size]
    return np.maximum(conv, 0.0)


def objective_score(a):
    arr = norm1(a)
    n = arr.size
    if n == 0:
        return float("inf")
    c = fast_convolve(arr)
    if c.size == 0:
        return float("inf")
    return float(2.0 * n * np.max(c))


def safe_eval_final(seq):
    try:
        vals = [float(x) for x in seq]

        return float(sc)
    except Exception:

def softmax_stable(logx):
    arr = safe_parse_seq(logx)
    if arr.size == 0:
        return arr
    arr = np.where(np.isfinite(arr), arr, -700.0)
    m = float(np.max(arr))
    e = np.exp(arr - m)
    s = e.sum()
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones_like(arr, dtype=np.float64) / float(arr.size)
    return e / s


def interpolate_seq(f, new_n):
    old = norm1(f)
    old_n = old.size
    new_n = int(new_n)
    if new_n <= 0:
        return old.copy()
    if old_n == new_n:
        return old.copy()
    if old_n == 1:
        return np.ones(new_n, dtype=np.float64)
    xo = np.linspace(0.0, 1.0, old_n)
    xn = np.linspace(0.0, 1.0, new_n)
    g = np.interp(xn, xo, old)
    g = np.maximum(g, 0.0)
    s = g.sum()
    if s <= 1e-15:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    return g / s


def top_indices(arr, k):
    n = arr.size
    k = int(k)
    if k <= 0:
        return np.empty(0, dtype=int)
    if k >= n:
        return np.arange(n, dtype=int)
    part = np.argpartition(arr, n - k)[-k:]
    part = part[np.argsort(arr[part])[::-1]]
    return part.astype(int, copy=False)


def choose_active(c, M, top_k, n, rng):
    L = c.size
    top_k = int(min(max(top_k, 1), L))
    order = top_indices(c, top_k)
    vals = c[order]
    scale = float(M) if M > 0 else 1.0
    ratio = np.maximum(vals / scale, 0.0)
    power = float(rng.uniform(1.0, 3.0))
    weights = np.power(ratio, power)
    center = 0.5 * (L - 1)
    pos = (order.astype(np.float64) - center) / (0.5 * n + 1.0)
    bias = float(rng.uniform(-1.0, 1.0))
    weights = weights * np.exp(bias * pos)
    weights = weights + 1e-6 * rng.random(top_k)
    weights = np.maximum(weights, 0.0)
    s = weights.sum()
    if s <= 1e-300:
        weights = np.ones(top_k, dtype=np.float64) / float(top_k)
    else:
        weights /= s
    return order, weights


def compute_r(f, lags, weights):
    """
    r_j = sum_k weights[k] * f[k-j], for valid k-j.
    This is the weighted coverage contribution under active lags.
    """
    f = safe_parse_seq(f)
    n = f.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    lags = np.asarray(lags, dtype=int).ravel()
    weights = safe_parse_seq(weights)
    if lags.size == 0 or weights.size == 0:
        return np.zeros(n, dtype=np.float64)
    if lags.size != weights.size:
        m = min(lags.size, weights.size)
        lags = lags[:m]
        weights = weights[:m]
    nz = weights > 1e-300
    if not np.any(nz):
        return np.zeros(n, dtype=np.float64)
    lags = lags[nz]
    weights = weights[nz]

    if n == 1:
        return np.array([weights[0] * f[0]], dtype=np.float64)

    # Direct loop is faster for very small active sets.
    if int(lags.size) * int(n) <= 250000:
        r = np.zeros(n, dtype=np.float64)
        for kk, ww in zip(lags, weights):
            ks = int(kk)
            start = ks - n + 1
            if start < 0:
                start = 0
            end = ks + 1
            if end > n:
                end = n
            if start < end:
                a_idx = ks - start
                b_idx = ks - (end - 1)
                r[start:end] += ww * f[b_idx:a_idx + 1][::-1]
        return np.maximum(r, 0.0)

    # FFT path for larger active sets.
    # convolve(rev_f, w_full); desired slice is [n-1, n).
    rev = np.flip(f)
    w_full = np.zeros(2 * n - 1, dtype=np.float64)
    w_full[lags] = weights
    length = n + (2 * n - 1) - 1
    fs = _next_pow2(length)
    F = np.fft.rfft(rev, fs)
    W = np.fft.rfft(w_full, fs)
    corr = np.fft.irfft(F * W, fs)
    r = corr[n - 1:n - 1 + n]
    return np.maximum(r, 0.0)


def mirror_search(a0, t_limit, seed=123456, verbose=True):
    a = norm1(a0)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64), 0.0, float("inf")

    t0 = time.monotonic()
    c0 = fast_convolve(a)
    M0 = float(np.max(c0))
    best = a.copy()
    best_M = M0
    cur_M = M0

    rng = np.random.default_rng(seed)
    step = 5.0
    min_step = 0.2
    max_step = 256.0
    top_k = min(36, 2 * n - 1)
    max_top = min(128, 2 * n - 1)
    no_best = 0
    no_cur = 0
    it = 0
    log_eps = 1e-8

    while time.monotonic() - t0 < float(t_limit):
        it += 1
        c = fast_convolve(a)
        M = float(np.max(c))
        if M < best_M - 1e-15:
            best = a.copy()
            best_M = M
            no_best = 0
        else:
            no_best += 1

        if no_best > 350 or no_cur > 150:
            logbest = np.log(best + 1e-8)
            sigma = 0.18 + 0.22 / (1.0 + min(10.0, float(it) / 500.0))
            logp = logbest + sigma * rng.standard_normal(n)
            a = softmax_stable(logp)
            cur_M = float(np.max(fast_convolve(a)))
            no_best = 0
            no_cur = 0
            top_k = min(36, 2 * n - 1)
            step = 5.0
            if verbose:
                print("  mirror reheat t=%.1f best=%.6f" % (time.monotonic() - t0, 2.0 * n * best_M))
            continue

        loga = np.log(a + log_eps)
        trial_M = M
        trial_q = a
        improved_trial = False

        n_trials = 1 if no_cur < 50 else 2
        for _sel in range(n_trials):
            lags, w = choose_active(c, M, top_k, n, rng)
            r = compute_r(a, lags, w)
            rmax = float(np.max(r))
            if not np.isfinite(rmax) or rmax <= 1e-300:
                continue
            rs = r / rmax

            for factor in (0.5, 1.0, 2.0):
                st = step * factor
                if st < min_step:
                    st = min_step
                if st > max_step:
                    st = max_step
                logq = loga - st * rs
                q = softmax_stable(logq)
                if q.size != n:
                    continue
                q = np.maximum(q, 0.0)
                qs = q.sum()
                if qs <= 1e-300:
                    continue
                q = q / qs
                mq = float(np.max(fast_convolve(q)))
                if mq < trial_M - 1e-15:
                    trial_M = mq
                    trial_q = q.copy()
                    improved_trial = True

        if improved_trial and trial_M < cur_M - 1e-15:
            a = trial_q
            cur_M = trial_M
            no_cur = 0
            step = min(max_step, step * 1.25)
        else:
            # Mild exploration: accept a plausible non-improving move sometimes.
            if improved_trial and no_cur > 40 and trial_M < best_M * 1.05 and rng.random() < 0.35:
                a = trial_q
                cur_M = trial_M
            no_cur += 1
            step = max(min_step, step * 0.75)
            if no_cur > 60:
                top_k = min(max_top, top_k + 2)

        if verbose and (it % 300 == 0):
            print(
                "  mirror n=%d it=%d t=%.1f best=%.6f cur=%.6f step=%.2f top=%d"

            )

    M_end = float(np.max(fast_convolve(a)))
    if M_end < best_M:
        best = a.copy()
        best_M = M_end
    best = norm1(best)
    best_M = float(np.max(fast_convolve(best)))
    return best, float(2.0 * best.size * best_M), best_M


def build_active_matrix(f, lags):
    f = safe_parse_seq(f)
    n = f.size
    lags = np.asarray(lags, dtype=int).ravel()
    K = lags.size
    C = np.zeros((K, n), dtype=np.float64)
    for row, kk in enumerate(lags):
        ks = int(kk)
        start = ks - n + 1
        if start < 0:
            start = 0
        end = ks + 1
        if end > n:
            end = n
        if start < end:
            a_idx = ks - start
            b_idx = ks - (end - 1)
            C[row, start:end] = f[b_idx:a_idx + 1][::-1]
    return C


def get_lp_solver():
    global _LP_READY, _LP_SOLVER
    if _LP_READY:
        return _LP_SOLVER
    _LP_READY = True
    if cp is None:
        _LP_SOLVER = None
        print("  cvxpy not available")
        return None

    solver_names = []
    for nm in ("PDLP", "CLARABEL", "ECOS", "SCS"):
        s = getattr(cp, nm, None)
        if s is not None:
            solver_names.append(s)

    for s in solver_names:
        try:
            x = cp.Variable(30, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                _LP_SOLVER = s
                print("  LP solver chosen: " + str(s))
                return s
        except Exception:
            pass

    _LP_SOLVER = None
    print("  No LP solver usable")
    return None


def lp_fw_search(a0, t_limit, solver=None, verbose=True):

        return np.zeros(0, dtype=np.float64), 0.0, float("inf")

    t0 = time.monotonic()
    c0 = fast_convolve(a)
    M0 = float(np.max(c0))
    best = a.copy()
    best_M = M0

    if solver is None or cp is None:
        return best, float(2.0 * n * best_M), best_M

    max_K = 220
    active_lags = top_indices(c0, min(180, c0.size))
    active_lags = [int(x) for x in active_lags]
    rng = np.random.default_rng(555)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < float(t_limit):
        if it > 0 and time.monotonic() - t0 >= float(t_limit):
            break

        c = fast_convolve(a)
        M = float(np.max(c))
        if M < best_M - 1e-15:
            best = a.copy()
            best_M = M
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 120:
            break

        # Active set: old constraints plus current top peaks.
        top_now = top_indices(c, 160)
        old_keep = active_lags[:100]
        active = list(dict.fromkeys(old_keep + [int(x) for x in top_now]))
        if len(active) > max_K:
            act_arr = np.asarray(active, dtype=int)
            act_score = c[act_arr]
            keep = top_indices(act_score, max_K)
            active = [int(act_arr[i]) for i in keep]

        if len(active) == 0:
            break

        C = build_active_matrix(a, np.asarray(active, dtype=int))

        try:
            gvar = cp.Variable(n, nonneg=True)
            tvar = cp.Variable()
            constraints = [cp.sum(gvar) == 1, C @ gvar <= tvar]
            prob = cp.Problem(cp.Minimize(tvar), constraints)
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break

        if prob.status not in ("optimal", "optimal_inaccurate"):
            no_improve += 2
            if no_improve > 160:
                break
            continue

        gval = np.asarray(gvar.value, dtype=np.float64).ravel()
        if gval.size != n:
            break
        gval = np.where(np.isfinite(gval), gval, 0.0)
        gval = np.maximum(gval, 0.0)
        gsum = gval.sum()
        if gsum <= 1e-15 or not np.isfinite(gsum):
            continue
        gdir = gval / gsum

        best_alpha_M = M
        best_trial = a
        accepted = False
        for alpha in (0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0):
            trial = (1.0 - alpha) * a + alpha * gdir
            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s <= 1e-15:
                continue
            trial = trial / s
            Mt = float(np.max(fast_convolve(trial)))
            if Mt < best_alpha_M - 1e-15:
                best_alpha_M = Mt
                best_trial = trial
                accepted = True

        if accepted:
            a = best_trial
            no_improve = 0
        else:
            if it < 2:
                a = norm1(gdir)
            no_improve += 1
            if no_improve > 160:
                break

        add = top_indices(fast_convolve(a), 80)
        new_active = list(dict.fromkeys(active + [int(x) for x in add]))
        if len(new_active) > max_K:
            new_arr = np.asarray(new_active, dtype=int)
            new_score = c[new_arr]
            keep = top_indices(new_score, max_K)
            new_active = [int(new_arr[i]) for i in keep]
        active_lags = new_active

        it += 1
        if verbose and (it % 3 == 0 or it == 1):
            print(
                "  LP it=%d t=%.1f best=%.6f |K|=%d"
                % (it, time.monotonic() - t0, 2.0 * n * best_M, len(active_lags))
            )

    M_end = float(np.max(fast_convolve(a)))
    if M_end < best_M:
        best = a.copy()
        best_M = M_end
    best = norm1(best)
    best_M = float(np.max(fast_convolve(best)))
    return best, float(2.0 * best.size * best_M), best_M


def pair_polish(a0, t_limit, seed=31337, verbose=True):
    a = norm1(a0)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64), 0.0, float("inf")

    t0 = time.monotonic()
    c0 = fast_convolve(a)
    M0 = float(np.max(c0))
    best = a.copy()
    best_M = M0
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < float(t_limit) and stall < 500:
        c = fast_convolve(a)
        M = float(np.max(c))
        if M < best_M - 1e-15:
            best = a.copy()
            best_M = M
            stall = 0
        else:
            stall += 1

        if stall > 220:
            loga = np.log(a + 1e-8)
            logp = loga + 0.15 * rng.standard_normal(n)
            a = softmax_stable(logp)
            stall = 0
            continue

        lags, w = choose_active(c, M, 64, n, rng)
        r = compute_r(a, lags, w)
        mask = np.where(a > 1e-10, 1.0, 0.0)
        src_score = r * mask
        src = int(np.argmax(src_score))
        if a[src] <= 1e-10:
            stall += 1
            continue

        dst_score = r.copy()
        dst_score[src] = np.inf
        dst = int(np.argmin(dst_score))
        if r[dst] >= r[src] * (1.0 - 1e-8):
            stall += 1
            continue

        improved = False
        for frac in (0.001, 0.003, 0.01, 0.03, 0.1):
            amt = a[src] * frac
            if amt <= 1e-16 or not np.isfinite(amt):
                continue
            trial = a.copy()
            trial[src] -= amt
            trial[dst] += amt
            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s <= 1e-15:
                continue
            trial = trial / s
            Mt = float(np.max(fast_convolve(trial)))
            if Mt < best_M - 1e-15:
                a = trial
                best = trial.copy()
                best_M = Mt
                improved = True
                stall = 0
                break

        if improved:
            continue

        # Secondary local attempts.
        if stall % 5 == 0:
            srcs = top_indices(src_score, 8)
            dsts = np.argsort(r)[:8]
            found = False
            for si in srcs:
                si = int(si)
                if a[si] <= 1e-10:
                    continue
                for di in dsts:
                    di = int(di)
                    if di == si or r[di] >= r[si] * (1.0 - 1e-8):
                        continue
                    amt = a[si] * 0.002
                    if amt <= 1e-12:
                        amt = 1e-8
                    if amt > a[si]:
                        amt *= 0.01
                    trial = a.copy()
                    trial[si] -= amt
                    trial[di] += amt
                    trial = np.maximum(trial, 0.0)
                    s = trial.sum()
                    if s <= 1e-15:
                        continue
                    trial = trial / s
                    Mt = float(np.max(fast_convolve(trial)))
                    if Mt < best_M - 1e-15:
                        a = trial
                        best = trial.copy()
                        best_M = Mt
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break

        if not improved:
            stall += 1

    M_end = float(np.max(fast_convolve(a)))
    if M_end < best_M:
        best = a.copy()
        best_M = M_end
    best = norm1(best)
    best_M = float(np.max(fast_convolve(best)))
    return best, float(2.0 * best.size * best_M), best_M


def parametric_candidates(n, rng, max_count=40):
    out = []
    if n < 50:
        return out
    x = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)

    out.append((norm1(np.ones(n, dtype=np.float64)), "uni"))
    out.append((norm1(np.sin(np.pi * x) ** 2), "sin"))

    for p in (0.5, 1.0, 2.0, 3.0):
        out.append((norm1((i + 1.0) ** p), "dpow%.1f" % p))
        out.append((norm1((n - i) ** p), "upow%.1f" % p))

    for r in (1.0, 2.0, 5.0, 10.0):
        a = np.exp(-r * x) + np.exp(r * (x - 1.0))
        out.append((norm1(a), "edge%d" % int(r)))

    for c in (0.2, 0.35, 0.5, 0.65, 0.8):
        for sig in (0.1, 0.2):
            a = np.exp(-((x - c) ** 2) / (2.0 * sig * sig))
            out.append((norm1(a), "g%.2f_%.2f" % (c, sig)))

    kernel = np.ones(5, dtype=np.float64) / 5.0
    while len(out) < max_count:
        a = rng.random(n, dtype=np.float64)
        a = np.convolve(a, kernel, mode="same")
        out.append((norm1(a), "rand%d" % len(out)))

    return out


def extract_parent_sequence():
    val = globals().get("height_sequence_1", None)
    if val is None:
        return None, float("inf")
    try:
        arr = norm1(val)
        if arr.size == 0:
            return None, float("inf")
        sc = objective_score(arr)
        if np.isfinite(sc):
            return arr, float(sc)
    except Exception:
        pass
    return None, float("inf")


def remaining_time(t_global, t_budget):
    return max(0.0, float(t_budget) - (time.monotonic() - t_global))


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    MAX_N = 12000
    LP_MAX_N = 7000
    print("magic mirror search start, budget=%.1f" % time_budget)

    base_seq, base_score = extract_parent_sequence()
    if base_seq is None or not np.isfinite(base_score):
        n_base = 5000
        x = np.linspace(0.0, 1.0, n_base)
        base_seq = norm1(np.sin(np.pi * x) ** 2 + 0.05 * np.ones(n_base, dtype=np.float64))
        base_score = objective_score(base_seq)

    if base_seq.size > MAX_N:
        base_seq = interpolate_seq(base_seq, 10000)
        base_score = objective_score(base_seq)
    n_base = base_seq.size

    pool = []
    idx = 0
    pool.append((base_score, n_base, idx, base_seq, "base"))
    idx += 1

    # Length variants of the parent.
    for f in (0.8, 1.1, 1.25, 1.5):
        new_n = int(round(n_base * f))
        new_n = int(min(max(new_n, 2500), MAX_N))
        if abs(new_n - n_base) > 200:
            seq = interpolate_seq(base_seq, new_n)
            sc = objective_score(seq)
            if np.isfinite(sc):
                pool.append((sc, new_n, idx, seq, "len%d" % new_n))
                idx += 1

    # Cheap parametric probe at shorter length; upscale the best.
    rem = remaining_time(t_global, time_budget)
    if rem > 600:
        n_param = int(min(2000, max(800, n_base // 4)))
        rngp = np.random.default_rng(2024)
        cands = parametric_candidates(n_param, rngp, 36)
        param_pool = []
        for seq, lab in cands:
            sc = objective_score(seq)
            if np.isfinite(sc):
                param_pool.append((sc, n_param, len(param_pool), seq, lab))
        if param_pool:
            param_pool.sort()
            _, _, _, pseq, plab = param_pool[0]
            p_up = interpolate_seq(pseq, n_base)
            sc_up = objective_score(p_up)
            if np.isfinite(sc_up):
                pool.append((sc_up, n_base, idx, p_up, "param_" + plab))
                idx += 1

    if not pool:
        base_seq = norm1(np.sin(np.pi * np.linspace(0.0, 1.0, 5000)) ** 2)
        base_score = objective_score(base_seq)
        pool.append((base_score, base_seq.size, 0, base_seq, "fallback"))

    pool.sort()
    best_score = pool[0][0]
    best_seq = pool[0][3]
    print("initial best %.6f n=%d label=%s" % (best_score, best_seq.size, pool[0][-1]))

    # Quick probe of alternate starting lengths if competitive.
    rem = remaining_time(t_global, time_budget)
    for item in pool[1:4]:
        if rem < 120:
            break
        sc, nn, _, seq, lab = item
        if sc <= best_score + 0.03 and nn != best_seq.size:
            t_q = min(60.0, rem * 0.15)
            out_seq, out_score, _ = mirror_search(seq, t_q, seed=987650, verbose=True)
            if out_score < best_score:
                best_score = out_score
                best_seq = out_seq
            else:
                # Still use the run's own best point; it should be the initial or better.
                best_seq = out_seq
                best_score = min(best_score, out_score)
            rem = remaining_time(t_global, time_budget)

    # Main mirror descent pass.
    rem = remaining_time(t_global, time_budget)
    t_mirror = min(rem * 0.30, 300.0)
    if t_mirror > 30:
        out_seq, out_score, _ = mirror_search(best_seq, t_mirror, seed=42, verbose=True)
        best_seq = out_seq
        best_score = min(best_score, out_score)
        print("after mirror1 best=%.6f n=%d" % (best_score, best_seq.size))

    # Optional LP-based Frank-Wolfe pass (only when small enough and time allows).
    rem = remaining_time(t_global, time_budget)
    if rem > 500 and best_seq.size <= LP_MAX_N:
        solver = get_lp_solver()
        if solver is not None:
            t_lp = min(rem * 0.20, 180.0)
            if t_lp > 60:
                try:
                    out_seq, out_score, _ = lp_fw_search(best_seq, t_lp, solver, verbose=True)
                    if np.isfinite(out_score):
                        best_seq = out_seq
                        best_score = min(best_score, out_score)
                except Exception:
                    pass
                print("after LP best=%.6f n=%d" % (best_score, best_seq.size))

    # Second mirror pass with a different seed.
    rem = remaining_time(t_global, time_budget)
    t2 = min(rem * 0.50, 420.0)
    if t2 > 30:
        out_seq, out_score, _ = mirror_search(best_seq, t2, seed=101, verbose=True)
        best_seq = out_seq
        best_score = min(best_score, out_score)
        print("after mirror2 best=%.6f n=%d" % (best_score, best_seq.size))

    # Local pair polishing.
    rem = remaining_time(t_global, time_budget)
    t_pair = min(rem * 0.60, 240.0)
    if t_pair > 20:
        try:
            out_seq, out_score, _ = pair_polish(best_seq, t_pair, seed=31337, verbose=True)
            best_seq = out_seq
            best_score = min(best_score, out_score)
        except Exception:
            pass
        print("after polish best=%.6f n=%d" % (best_score, best_seq.size))

    # Try a modest length increase if there is time; denser discretization sometimes helps.
    rem = remaining_time(t_global, time_budget)
    if rem > 200 and best_seq.size < MAX_N:
        new_n = min(MAX_N, best_seq.size + max(500, int(0.25 * best_seq.size)))
        if new_n > best_seq.size:
            seq_hi = interpolate_seq(best_seq, new_n)
            t_hi = min(rem * 0.40, 240.0)
            try:
                out_seq, out_score, _ = mirror_search(seq_hi, t_hi, seed=5555, verbose=True)
                if np.isfinite(out_score) and out_score < best_score:
                    best_score = out_score
                    best_seq = out_seq
            except Exception:
                pass
            print("after hi-res probe best=%.6f n=%d" % (best_score, best_seq.size))

    # Final exact evaluation of a few candidate lengths.
    final_pool = []
    final_pool.append((best_score, best_seq.size, 0, best_seq, "final"))
    for item in pool[:3]:
        sc, nn, _, seq, lab = item
        final_pool.append((sc, nn, 1, seq, lab))
    final_pool.sort()

    selected = []
    seen_n = []
    for item in final_pool:
        nn = item[1]
        if nn not in seen_n:
            selected.append(item)
            seen_n.append(nn)
        if len(selected) >= 3:
            break

    actual_best_score = float("inf")
    actual_seq = best_seq
    for item in selected:
        seq = item[3]
        try:
            vals = [float(x) for x in seq]
            sc = safe_eval_final(vals)
            if np.isfinite(sc) and sc < actual_best_score:
                actual_best_score = sc
                actual_seq = seq
                print("candidate exact score %.6f n=%d label=%s" % (sc, seq.size, item[-1]))
        except Exception:
            pass

    if not np.isfinite(actual_best_score):
        actual_seq = best_seq

    print("FINAL returned n=%d fast_best=%.8f exact_best=%.8f time=%.1f" %
          (actual_seq.size, best_score, actual_best_score, time.monotonic() - t_global))
    return [float(x) for x in actual_seq]
