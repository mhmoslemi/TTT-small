    B = fast_conv(a)  # length 2n-1
    
    # Softmax weights (temperature-based)
    B_shifted = B - B.max()
    W = np.exp(B_shifted / tau)
    W_sum = W.sum()
    if W_sum < 1e-30:
        W = np.zeros_like(B)
        W[B.argmax()] = 1.0
    else:
        W = W / W_sum
    
    # Subgradient: g[m] = 2 * sum_j W[m+j] * a[j]
    # This is cross-correlation of W and a
    # Using FFT:
    L = 1 << (2 * n - 1).bit_length()
    # cross_corr[m] = sum_j a[j] * W[m+j] for m = 0..n-1
    # This is ifft(fft(W, L) * conj(fft(a, L))) with appropriate slicing
    # Actually: (W corr a)[m] = sum_j W[m+j] * a[j]
    # = ifft(fft(W) * conj(fft(a)))... let me verify
    
    # scipy.signal.correlate(W, a, 'full')[m] = sum_j W[m+j] * a[j] for valid m
    # But 'full' gives length (2n-1) + n - 1... let me just use the FFT formula.
    
    # correlate(W, a, 'valid')[m] = sum_j W[m+j] * a[j] for m = 0..n-1
    # In terms of convolution: correlate(W, a) = conv(W[::-1], a) shifted
    # Let's just use scipy:
    g = 2.0 * sp_signal.correlate(W, a, mode='valid')
    
    return B, W, g


def nesterov_gradient_descent(a0, time_limit, tau=1.0, verbose=False, seed=0):
    """Nesterov accelerated gradient descent with line search."""
    n = len(np.asarray(a0))
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    S_target = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a = a / s * S_target
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = 2.0 * n * best_M / S_target**2
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()  # current point
    x_prev = a.copy()  # previous point (for Nesterov)
    v = np.zeros(n)  # momentum
    
    eta = 1.0  # initial step size
    tau_cur = tau
    stall = 0
    max_stall = 500
    it = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        it += 1
        
        # Nesterov lookahead
        beta = (1.0 + np.sqrt(1.0 + 4.0 * (it-1)**2)) / 2.0
        beta = min(beta, (it + 1.0) / (it - 0.5))  # cap momentum
        y = x + (beta - 1.0) / beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum() * S_target
        
        # Compute gradient
        B = fast_conv(y)
        B_max = B.max()
        
        # Adaptive temperature: decrease over time
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        tau_it = tau * (1.0 - 0.5 * progress)
        tau_it = max(tau_it, 0.01)
        
        B_shifted = B - B_max
        W = np.exp(B_shifted / tau_it)
        W = W / W.sum()
        
        # Subgradient
        g = 2.0 * sp_signal.correlate(W, y, mode='valid')
        
        # Normalize gradient (to handle scale invariance)
        g_norm = g / (g.max() + 1e-30)
        
        # Line search
        x_prev = x
        found = False
        for trial_eta in [eta, eta*0.5, eta*0.25, eta*0.1, eta*0.05, eta*0.01]:
            x_trial = x - trial_eta * g_norm
            x_trial = np.maximum(x_trial, 0.0)
            s_t = x_trial.sum()
            if s_t < 0.01:
                continue
            x_trial = x_trial / s_t * S_target
            M_trial = fast_conv(x_trial).max()
            J_trial = 2.0 * n * M_trial / S_target**2
            
            if J_trial < best_J:
                best_J = J_trial
                best_M = M_trial
                best_a = x_trial.copy()
            
            if M_trial < B_max - 1e-10:
                x = x_trial
                stall = 0
                found = True
                break
            elif trial_eta < eta * 0.1:
                # Can't reduce further, accept anyway to avoid infinite loop
                x = x_trial
                break
        
        if not found and time.monotonic() - t0 < time_limit:
            stall += 1
            if stall > 100:
                eta *= 0.8
            if stall > 200:
                # Perturbation
                pert = rng.normal(0, 0.01, n) * best_a
                x_p = np.maximum(best_a + pert, 0.0)
                s_p = x_p.sum()
                if s_p > 0.01:
                    x = x_p / s_p * S_target
                    x_prev = x.copy()
                    stall = 50
                    eta = 1.0
        
        # Increase eta if making progress
        if stall == 0 and eta < 5.0:
            eta *= 1.01
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NAG it={it} ({elapsed:.0f}s): J={best_J:.6f}, eta={eta:.4f}, stall={stall}")
    
    return best_a, best_J


def lp_frank_wolfe(a0, time_limit, solver_name, verbose=False, top_K_init=300, top_K_max=None):
    """LP-based Frank-Wolfe."""
    import cvxpy as cp
    n = len(np.asarray(a0))
    S_target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S_target / n
    else:
        a = a / s * S_target
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = 2.0 * n * best_M / S_target**2
    
    t0 = time.monotonic()
    if top_K_max is None:
        top_K_max = min(2 * n - 1, 5000)
    K = top_K_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        B = fast_conv(a)
        M_curr = B.max()
        
        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_k = np.arange(2 * n - 1)
        else:
            binding_k = np.argsort(B)[::-1][:K_sel]
        
        # Build constraint matrix: T[i, j] = contribution of a[j] to B[k_i]
        # B[k] = sum_j a[j] * a[k-j] (valid j)
        # For LP: find b maximizing sum(b) s.t. sum_j b[j] * a[k_i - j] <= M_curr
        T = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_k):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start >= j_end:
                continue
            T[ii, j_start:j_end] = a[ki - np.arange(j_start, j_end)]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum * S_target  # scale to match normalization
        
        # Line search
        base_obj = M_curr
        best_alpha = 2.0 / (it + 2.0)
        best_trial_M = base_obj
        for alpha in [best_alpha, min(1.0, best_alpha * 2.0), 0.5, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t * S_target
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 0.01:
            break
        a = a / s_a * S_target
        
        M_new = fast_conv(a).max()
        J_new = 2.0 * n * M_new / S_target**2
        if J_new < best_J:
            best_J = J_new
            best_M = M_new
            best_a = a.copy()
        
        it += 1
        if it % 10 == 0:
            K = min(K + 20, top_K_max)
        
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW it={it} ({elapsed:.0f}s): J={best_J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")
    
    return best_a, best_J


def make_ramp(n, power=2.0):
    """Asymmetric ramp: high on left, low on right (or vice versa)."""
    x = np.linspace(0, 1, n)
    a = (1 - x) ** power
    return a / a.sum() * np.sqrt(2.0 * n)


def make_shifted_gaussian(n, center=0.3, width=0.2):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2) + 1e-6
    return a / a.sum() * np.sqrt(2.0 * n)


def make_squared_ramp(n, alpha=0.5):
    x = np.linspace(0, 1, n)
    a = (1 - alpha * x) ** 2
    a = np.maximum(a, 0.0)
    return a / a.sum() * np.sqrt(2.0 * n)


def make_sinusoidal_asym(n, freq=1.0, decay=2.0):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x * freq) ** 2 * np.exp(-decay * x)
    a = np.maximum(a, 1e-6)
    return a / a.sum() * np.sqrt(2.0 * n)


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(f)
    S = f.sum()
    if old_n == new_n:
        return f.copy()
    if S < 1e-15:
        return np.ones(new_n) / new_n * np.sqrt(2.0 * new_n)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n * np.sqrt(2.0 * new_n)
    return f_interp / ss * S


def reflect_and_blend(a):
    """Try reflected version and blend."""
    a_ref = a[::-1].copy()
    blend = 0.5 * a + 0.5 * a_ref
    s = blend.sum()
    S_target = np.sqrt(2.0 * len(a))
    return blend / s * S_target


def coordinate_polish(a0, time_limit, verbose=False, seed=123):
    """Move mass from high-contribution positions to low-contribution ones."""
    n = len(np.asarray(a0))
    S_target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S_target / n
    else:
        a = a / s * S_target
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = 2.0 * n * best_M / S_target**2
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        B = fast_conv(a)
        M_curr = B.max()
        top_k = np.argsort(B)[::-1][:15]
        
        # Compute sensitivity: how much each position contributes to top peaks
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:8]:
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]
        
        improved = False
        
        # Try moving mass from high-sensitivity positions to low-sensitivity ones
        if in_peak.sum() > 0 and (n - in_peak.sum()) > 0:
            src_candidates = np.where(in_peak)[0]
            src_candidates = src_candidates[np.argsort(sens[src_candidates])[::-1][:5]]
            dst_candidates = np.where(~in_peak)[0]
            dst_candidates = dst_candidates[np.argsort(sens[dst_candidates])[:5]]
            
            for src in src_candidates:
                if time.monotonic() - t0 > time_limit:
                    break
                if a[src] < 1e-10:
                    continue
                for dst in dst_candidates:
                    for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 0.01:
                            continue
                        a_trial = a_trial / s_t * S_target
                        M_trial = fast_conv(a_trial).max()
                        J_trial = 2.0 * n * M_trial / S_target**2
                        if J_trial < best_J - 1e-8:
                            a = a_trial
                            best_a = a.copy()
                            best_M = M_trial
                            best_J = J_trial
                            improved = True
                            break
                    if improved:
                        break
            
            if improved:
                stall = 0
                continue
        
        # Random perturbation
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i != j and a[i] > 1e-10:
            frac = rng.uniform(0.002, 0.02)
            amt = a[i] * frac
            a[i] -= amt
            a[j] += amt
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a > 0.01:
                a = a / s_a * S_target
                M_new = fast_conv(a).max()
                J_new = 2.0 * n * M_new / S_target**2
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
                    stall = 0
                    continue
        
        stall += 1
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")
    
    return best_a, best_J
