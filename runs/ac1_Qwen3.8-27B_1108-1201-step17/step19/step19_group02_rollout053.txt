            a = np.maximum(1 + np.cos(2 * np.pi * freq * x + ph), 0.01)
            candidates.append(norm_sum1(a))
    # Power-law ramps
    for p in [0.3, 0.5, 1.0, 2.0, 3.0]:
        candidates.append(norm_sum1((i_arr + 1) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))

    # Random perturbations of best
    rng = np.random.default_rng(42)
    for trial in range(100):
        if time.monotonic() - t0 > time_limit * 0.7:
            break
        noise = rng.normal(0, 0.1, n) * best_init
        a = best_init + noise
        candidates.append(norm_sum1(a))

    for a in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        v = compute_objective(a)
        if v < best_val:
            best_val = v
            best_a = a.copy()
            print("    Param: J=" + repr(v))

    print("  Parametric done: best J=" + repr(best_val))
    return best_a, best_val


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw(a0, time_limit, solver, initial_K=200, max_K=2000, verbose=True):
    """LP-based Frank-Wolfe with lazy active set and exact line search."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_convolve(y).max()
    t0 = time.monotonic()

    ac_y = fast_convolve(y)
    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())

    no_improve = 0
    max_no_improve = 800
    iter_count = 0
    last_lp_iter = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: stagnation at iter " + str(iter_count))
            break
        iter_count += 1

        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update active set: keep most binding constraints
        if iter_count % 20 == 0 or len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep_count = min(len(active_set), max_K)
            keep = np.argsort(margins)[:keep_count]
            active_set = sorted([active_set[i] for i in keep])
            # Add new top constraints
            top_k_all = np.argsort(ac)[::-1][:max_K]
            new_ones = [k for k in top_k_all if k not in active_set]
            active_set = sorted(active_set + new_ones[:max_K - len(active_set)])
            if len(active_set) > max_K:
                active_set = active_set[:max_K]

        K = len(active_set)
        if K == 0:
            break

        # Solve LP: minimize M s.t. sum(g)=1, g>=0, (y*g)_k <= M for k in active_set
        coeffs = build_constraint_matrix(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if g_var.value is None:
                continue
            # Try with fewer constraints
            active_set = sorted(active_set[:max(50, K // 2)])
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Exact line search along FW direction
        # Minimize max_k ((1-t)*y + t*g_dir) * ((1-t)*y + t*g_dir) for t in [0,1]
        # For efficiency, just check a few values
        alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9]
        # Add adaptive alpha based on progress
        default_alpha = 2.0 / (iter_count + 2.0)
        alphas = list(set(alphas + [max(0.01, min(1.0, default_alpha)),
                                     max(0.01, min(1.0, default_alpha * 0.5)),
                                     max(0.01, min(1.0, default_alpha * 2.0))]))

        best_alpha = 0.1
        best_trial_M = fast_convolve((1 - 0.1) * y + 0.1 * g_dir).max()
        for alpha in sorted(alphas):
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        # Add violated constraints to active set
        ac_new = fast_convolve(y)
        violations = np.where(ac_new > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = ac_new[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)
            if len(active_set) > max_K:
                active_set = active_set[:max_K]

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " |S|=" + str(len(active_set)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " iters=" + str(iter_count))
    return best_a, best_m


def smoothed_pg(a0, time_limit, mu_init=0.005, mu_end=0.0005, verbose=True):
    """
    Log-sum-exp smoothed projected gradient descent.
    Smoothed objective: f_mu(a) = 2n * mu * log(sum_k exp((a*a)_k / mu))
    Gradient computed via FFT.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = compute_objective(a)
    t0 = time.monotonic()

    mu = mu_init
    mu_ratio = (mu_end / mu_init) ** (1.0 / max(1, int(time_limit / 0.1)))

    # Momentum
    momentum = 0.85
    vel = np.zeros(n)

    # Learning rate schedule
    base_lr = 0.5 / np.sqrt(n)

    stall = 0
    max_stall = 3000

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break

        # Update mu (geometric decay)
        mu *= mu_ratio
        mu = max(mu, mu_end)

        # Compute gradient of smoothed objective
        ac = fast_convolve(a)
        # Softmax weights
        ac_norm = ac / mu - ac.max() / mu
        w = np.exp(ac_norm)
        w_sum = w.sum()
        w = w / w_sum

        # Gradient: 4n * corr(w, a) where corr[i] = sum_j w[i+j] * a[j]
        # corr[i] = (w * a_rev)[i + n - 1]
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        a_rev = a[::-1]
        W = np.fft.rfft(w, fft_size)
        AR = np.fft.rfft(a_rev, fft_size)
        conv_w_ar = np.fft.irfft(W * AR, fft_size)
        grad_inner = conv_w_ar[n - 1: 2 * n - 1]
        grad = 4.0 * n * grad_inner

        # Gradient step with momentum
        lr = base_lr / (1.0 + 0.01 * t)
        vel = momentum * vel - lr * grad
        z = a + vel
        z = project_simplex(z)

        # Check improvement
        M_new = fast_convolve(z).max()
        if M_new < best_obj / (2.0 * n) - 1e-15:
            best_obj = 2.0 * n * M_new
            best_a = z.copy()
            stall = 0
        else:
            # Try with smaller step
            z2 = a - 0.1 * lr * grad
            z2 = project_simplex(z2)
            M_new2 = fast_convolve(z2).max()
            if M_new2 < best_obj / (2.0 * n) - 1e-15:
                best_obj = 2.0 * n * M_new2
                best_a = z2.copy()
                a = z2.copy()
                vel = np.zeros(n)
                stall = 0
            else:
                stall += 1
                a = z  # Still update to keep exploring

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_obj / (2.0 * n)
            print("    PG iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * fast_convolve(best_a).max()) + " mu=" + repr(mu))

    return best_a, best_obj


def peak_local_search(a0, time_limit, verbose=True):
    """
    Targeted local search: for each peak, move mass from peak-contributing
    positions to positions that don't contribute to any top peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(30, len(ac))
        top_peaks = np.argsort(ac)[::-1][:n_top]

        # Identify "safe" positions: those that don't pair with any high-value
        # position at a top peak
        danger = np.zeros(n)
        for k in top_peaks[:10]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            for i in range(i_start, i_end + 1):
                partner = k_int - i
                if 0 <= partner < n:
                    danger[i] = max(danger[i], a[i] * a[partner])
                    danger[partner] = max(danger[partner], a[partner] * a[i])

        # Find positions with high a-value that are dangerous (contribute to peaks)
        # and positions with low danger (safe destinations)
        sorted_by_val = np.argsort(a)[::-1]
        sorted_by_danger = np.argsort(danger)

        improved_this_round = False

        for src_idx_i in range(min(15, n)):
            if time.monotonic() - t0 > time_limit:
                break
            src = int(sorted_by_val[src_idx_i])
            if a[src] < 1e-12:
                continue

            # Find the best destination: low danger, not a peak contributor
            for dst_idx_i in range(min(50, n)):
                dst = int(sorted_by_danger[dst_idx_i])
                if dst == src:
                    continue
                if danger[dst] > 1e-15:
                    break

                # Try moving a fraction of mass
                for frac in [0.01, 0.05, 0.1, 0.2, 0.3]:
                    amt = a[src] * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_convolve(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved_this_round = True
                        stall = 0
                        break
                if improved_this_round:
                    # Re-sort after modification
                    sorted_by_val = np.argsort(a)[::-1]
                    danger = np.zeros(n)
                    ac = fast_convolve(a)
                    top_peaks = np.argsort(ac)[::-1][:10]
                    for k in top_peaks:
                        k_int = int(k)
                        i_start = max(0, k_int - n + 1)
                        i_end = min(n - 1, k_int)
                        for i in range(i_start, i_end + 1):
                            partner = k_int - i
                            if 0 <= partner < n:
                                danger[i] = max(danger[i], a[i] * a[partner])
                                danger[partner] = max(danger[partner], a[partner] * a[i])
                    sorted_by_danger = np.argsort(danger)
                    break

        if not improved_this_round:
            stall += 1
            # Random perturbation when stuck
            if stall % 50 == 0:
                n_perturb = max(5, n // 100)
                sources = rng.choice(n, size=n_perturb, replace=False)
                for src in sources:
                    if a[src] < 1e-10:
                        continue
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Peak search (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " stall=" + str(stall))

    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=99, verbose=True):
    """
    Fine-grained coordinate descent: move small amounts of mass between pairs
    of positions to reduce the peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 1000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            # Find the top contributor to this peak
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find safe destinations (low contribution to all top peaks)
            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            candidates = np.where(scores < np.median(scores))[0]
            if len(candidates) < 10:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])

            if dst == src:
                continue

            found = False
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break

            if found:
                stall = 0

        if not improved:
            stall += 1
            # Random perturbations
            if stall % 100 == 0:
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " stall=" + str(stall))

    return best_a, best_m


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Total time budget: " + repr(time_budget) + "s")

    best_seq, best_score = get_best(height_sequence_1, t_global)
    n0 = len(best_seq)

    # Phase 1: Parametric search
    print("\n=== Phase 1: Parametric (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 60.0)
    if t1 > 15:
        a_param, _ = parametric_init(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "parametric", t_global)

    # Phase 2: LP-FW main optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: LP-FW (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 40 and n_best <= 9000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t2, solver,
                         initial_K=300, max_K=2000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1_n" + str(n_best), t_global)

    # Phase 3: Smoothed projected gradient
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Smoothed PG (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_out, _ = smoothed_pg(norm_sum1(best_seq), t3, mu_init=0.003, mu_end=0.0003, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pg_n" + str(n_best), t_global)

    # Phase 4: LP-FW second pass
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW refine (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.25, 350.0)
    if t4 > 40 and n_best <= 9000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t4, solver,
                         initial_K=400, max_K=2500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2_n" + str(n_best), t_global)

    # Phase 5: Peak-directed local search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Peak search (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.15, 180.0)
    if t5 > 30:
        a_out, _ = peak_local_search(norm_sum1(best_seq), t5, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_n" + str(n_best), t_global)

    # Phase 6: Try nearby n values
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    try_ns = [n_best - 500, n_best + 500, n_best + 1000]
    try_ns = [l for l in try_ns if 3000 <= l <= 10000]

    for new_n in try_ns[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 200:
            break
        print("\n=== Phase 6: Scale n=" + str(new_n) + " ===")
        a_long = interpolate_seq(best_seq, new_n)

        t6a = min(remaining * 0.4, 250.0)
        if t6a > 40 and new_n <= 10000:
            a_out, _ = lp_fw(a_long, t6a, solver,
                             initial_K=300, max_K=2000, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80:
            t6b = min(remaining * 0.3, 120.0)
            if t6b > 20:
                a_out, _ = peak_local_search(norm_sum1(best_seq), t6b, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_n" + str(new_n), t_global)

    # Phase 7: Final LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final LP-FW (n=" + str(n_best) + ") ===")
    if remaining > 60 and n_best <= 10000:
        t7 = min(remaining * 0.4, 250.0)
        if t7 > 30:
            a_out, _ = lp_fw(norm_sum1(best_seq), t7, solver,
                             initial_K=500, max_K=2500, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # Phase 8: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final Coord (n=" + str(n_best) + ") ===")
    if remaining > 20:
        t8 = min(remaining * 0.7, 150.0)
        if t8 > 15:
            a_out, _ = coordinate_polish(norm_sum1(best_seq), t8, rng_seed=42, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Phase 9: Final peak search
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t9 = min(remaining * 0.6, 100.0)
        if t9 > 15:
            a_out, _ = peak_local_search(norm_sum1(best_seq), t9, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
