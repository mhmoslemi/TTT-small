    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

`sum λ_k c_k(a) = sum_k λ_k sum_i a_i a_{k-i} = a^T (sum_k λ_k Q_k) a` where `Q_k` is the matrix for lag `k`.

The inner minimization `min_{a: sum a=1, a>=0} a^T Q a` where `Q = sum_k λ_k Q_k` is a quadratic program. If `Q` is positive definite (or at least positive semidefinite on the simplex), this is convex.

The matrix `Q = sum_k λ_k Q_k` where `(Q_k)_{ij} = 1` if `i+j=k`. So `Q_{ij} = λ_{i+j}`. This is a **Hankel matrix**!

So the dual problem is: `max_{λ >= 0, sum λ = 1} min_{a: sum a=1, a>=0} a^T H(λ) a` where `H(λ)_{ij} = λ_{i+j}`.

The inner problem: `min a^T H a` s.t. `sum a = 1, a >= 0`. The KKT conditions give: `2Ha + μ*1 + ν = 0` with `ν >= 0, ν_i a_i = 0`.

If `H` is positive definite, the maximum over `a` (ignoring non-negativity) is at `a = -μ*1/(2...)`, which doesn't work because `a` must be non-negative.

This is getting too theoretical. Let me just write practical code.

Let me focus on making the existing algorithms more effective:

1. **Better step size for Nesterov**: Use a backtracking line search.
2. **Smarter peak sampling**: Instead of sampling from `softmax(β*c)`, sample from `softmax(β*(c - mean(c)))` or from the top-k peaks uniformly.
3. **Multi-scale optimization**: First optimize a "blurred" version of `a` (e.g., averaged in blocks of size B), then refine.
4. **Adaptive length**: Start with a shorter sequence (faster convergence), then interpolate to longer and refine.

def compute_subgrad(a, c, beta, n_samples=None):
    """
    Compute (stochastic) subgradient of (1/beta)*log(sum exp(beta*c_k)) w.r.t. a.
    Uses peak sampling for efficiency.
    Returns gradient vector of length n.
    """
    n = len(a)
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    
    if n_samples is not None and n_samples < len(p):
        # Sample
        idx = np.random.choice(len(p), size=n_samples, replace=False, p=p)
        p_sparse = np.zeros(len(p))
        p_sparse[idx] = p[idx] / p[idx].sum()
        p = p_sparse
    
    # grad_j = (2/beta) * sum_k p_k * a[k-j] (cross-correlation)
    size = n + len(p) - 1
    fft_size = 1 << (size - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    # cross-correlation: ifft(P * conj(A))
    grad_full = np.fft.irfft(P * np.conj(A), fft_size)
    grad = (2.0 / beta) * grad_full[:n]
    return grad, p

        # Track best
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_m - 1e-16:
            best_m = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        # Beta annealing
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        # Restart mechanism
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.03 / (1.0 + 0.05 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 5.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    Nest restart t=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J))

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Nest t=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J) + " beta=" + repr(round(beta, 1)))

    return best_a, best_m


# ===================== Peak-Shaving Coordinate Descent =====================

def peak_shave(a0, time_limit, verbose=True, seed=123):
    """
    Targeted coordinate descent: for each peak lag, find the worst contributing
    pair and move mass to a low-pressure location.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 3000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_autocorr(a)
        M = c.max()

        # Find top peaks
        n_peaks = min(15, len(c))
        peak_lags = np.argsort(c)[::-1][:n_peaks]
        improved = False

        for k in peak_lags[:10]:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_pos = int(np.argmax(contrib))
            max_idx = int(idx[max_pos])
            partner = k_int - max_idx

            # Determine which to reduce (the larger one)
            if a[max_idx] >= (a[partner] if 0 <= partner < n else 0):
                reduce_i = max_idx
            else:
                reduce_i = partner

            if a[reduce_i] < 1e-12:
                continue

            # Compute pressure: how much each position contributes to peaks
            pressure = np.zeros(n)
            for k2 in peak_lags[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]

            # Mark peak-contributing positions
            in_peak = np.zeros(n, dtype=bool)
            for k2 in peak_lags[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            # Find low-pressure destination
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 30:
                candidates = np.arange(n)
            mask = candidates != reduce_i
            valid_cands = candidates[mask]
            if len(valid_cands) < 10:
                valid_cands = np.arange(n)
                v

                mask2 = valid_cands != reduce_i
                valid_cands = valid_cands[mask2]
            valid_press = pressure[valid_cands]
            dst = int(valid_cands[np.argmin(valid_press)])

            # Try different fractions
            found = False
            for frac in [0.05, 0.10, 0.20, 0.35, 0.50]:
                amt = a[reduce_i] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[reduce_i] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 200:
                j = int(rng.integers(n))
                if j != reduce_i and a[reduce_i] > 1e-10:
                    frac = rng.uniform(0.01, 0.08)
                    amt = a[reduce_i] * frac
                    a[reduce_i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 200 == 0:
                for _ in range(15):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        frac = rng.uniform(0.005, 0.06)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakShave (" + str(round(elapsed)) + "s): J=" + repr(J) + " stall=" + str(stall))

    return best_a, best_m


# ===================== LP-Frank-Wolfe =====================

def lp_fw(a0, time_limit, solver, initial_K=400, max_K=2500, verbose=True):
    """LP-Frank-Wolfe: maximize sum(g) s.t. conv(y,g)_k <= M for active lags."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 1000

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW converged at iter " + str(t))
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set size
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix: row k has y[k-i] at position i
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations to add to active set
        conv_ys = fast_conv(y, g_dir)
        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:40]]
            for k in top_v:
                active_set.add(int(k))

        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5), 1.0 / (t + 1)]
        alphas = [max(0.01, min(1.0, a_)) for a_ in alphas]
        alphas = list(set(alphas))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(round(elapsed)) + "s): J=" + repr(J) + " |S|=" + str(len(active_set)))

    return best_a, best_m


# ===================== Spectral Power Flattening =====================

def spectral_flatten(a0, time_limit, verbose=True, seed=77):
    """
    Iteratively flatten the power spectrum |A|^2 to reduce autocorrelation peaks.
    Uses magnitude-only modification with non-negativity projection.
    """
    n = len(a0)
    a = norm_sum1(a0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    gamma = 1.3  # flattening exponent (1/gamma applied to power)
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autocorr(a)
        M = c.max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()

        # Compute power spectrum
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        phase = np.angle(A)

        # Flatten: raise power to 1/gamma (reduces dynamic range)
        power_safe = np.maximum(power, 1e-20)
        power_flat = power_safe ** (1.0 / gamma)
        # Preserve DC
        power_flat[0] = power[0]

        # Reconstruct with new magnitude
        A_new = np.sqrt(power_flat) * np.exp(1j * phase)
        A_new[0] = A[0]

        # Inverse FFT
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        # Project to simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        # Accept if improvement, otherwise blend
        c_new = fast_autocorr(a_new)
        if c_new.max() < M:
            a = a_new
        else:
            # Blend with small step
            blend = 0.3
            a_blend = blend * a_new + (1.0 - blend) * a
            a_blend = np.maximum(a_blend, 0.0)
            s_b = a_blend.sum()
            if s_b > 1e-15:
                a_blend = a_blend / s_b
                if fast_autocorr(a_blend).max() < M:
                    a = a_blend

        # Gradually increase flattening
        if t % 50 == 0:
            gamma = min(gamma * 1.01, 3.0)

        # Periodic perturbation to escape local minima
        if t % 200 == 0:
            noise = rng.normal(0, 0.01, n) * a
            a_p = a + noise
            a_p = np.maximum(a_p, 0.0)
            s_p = a_p.sum()
            if s_p > 1e-15:
                a_p = a_p / s_p
                if fast_autocorr(a_p).max() < best_m:
                    best_m = fast_autocorr(a_p).max()
                    best_a = a_p.copy()
                    a = a_p

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral t=" + str(t) + " (" + str(round(elapsed)) + "s): J=" + repr(J) + " gamma=" + repr(round(gamma, 2)))

    return best_a, best_m


# ===================== Parametric Initialization =====================

def parametric_search(n, time_limit, best_a, best_m):
    """Try diverse parametric families for good initialization."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    centered = i_arr - (n - 1) / 2.0
    norm_c = centered / (n / 2.0)

    candidates = []

    # Power laws (various exponents and orientations)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
        candidates.append(np.abs(norm_c) ** p)

    # Symmetric products
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p * (n - i_arr) ** p)

    # Gaussian
    for sigma in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
        candidates.append(np.exp(-norm_c ** 2 / (2.0 * sigma * sigma)))
        for center in [0.3, 0.4, 0.6, 0.7]:
            candidates.append(np.exp(-((x - center) ** 2) / (2.0 * sigma * sigma)))

    # Raised cosine
    for p in [1, 2, 3, 4, 6]:
        candidates.append(np.maximum(np.cos(np.pi * norm_c / 2.0), 0) ** (2 * p))

    # Sine
    for p in [1, 2, 3, 4]:
        candidates.append(np.sin(np.pi * x) ** p)

    # Multi-peak
    for n_peaks in [2, 3, 4]:
        centers = np.linspace(0.2, 0.8, n_peaks)
        for width in [0.05, 0.1, 0.15]:
            sig = np.zeros(n)
            for c in centers:
                sig += np.exp(-((x - c) ** 2) / (2.0 * width * width))
            candidates.append(sig)

    # Lorentzian
    for width in [0.05, 0.1, 0.2, 0.3]:
        candidates.append(1.0 / (1.0 + (norm_c / width) ** 2))

    # B-spline-like
    for order in [2, 4, 6]:
        candidates.append(np.maximum(1.0 - np.abs(norm_c), 0) ** order)

    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(a_raw)
        m = fast_autocorr(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()

    # Random initialization
    rng = np.random.default_rng(42)
    for _ in range(100):
        if time.monotonic() - t0 > time_limit:
            break
        alpha = rng.uniform(0.5, 3.0)
        a = rng.random(n) ** alpha
        # Optionally add structure
        if rng.random() < 0.3:
            a *= np.exp(-0.5 * (norm_c / 0.3) ** 2)
        a = norm_sum1(a)
        m = fast_autocorr(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()

    return best_a, best_m


# ===================== Update Helper =====================

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed)) + "s)")
    return best_seq, best_score

    longer_options = [l for l in longer_options if l <= 10000]

    for new_n in longer_options[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print("  Exploring n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick Nesterov
        t8a = min(remaining * 0.35, 180.0)
        if t8a > 30:
            a_out, _ = nesterov_opt(a_long, t8a,
                                    beta_start=20.0, beta_end=600.0,
                                    beta_growth=1.005, eta_base=0.8,
                                    restart_every=2000, verbose=False, seed=300)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n" + str(new_n), t_global)

        # Peak shave on longer
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t8b = min(remaining * 0.30, 120.0)
            if t8b > 15:
                a_out, _ = peak_shave(interpolate_sequence(best_seq, new_n), t8b,
                                      verbose=False, seed=500)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_n" + str(new_n), t_global)

    # Try shorter length (might be better ratio)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100 and n_best > 4000:
        short_n = max(3000, n_best - 2000)
        print("  Trying shorter n=" + str(short_n))
        a_short = interpolate_sequence(best_seq, short_n)
        t8c = min(remaining * 0.30, 200.0)
        if t8c > 30:
            a_out, _ = nesterov_opt(a_short, t8c,
                                    beta_start=20.0, beta_end=800.0,
                                    beta_growth=1.005, eta_base=0.9,
                                    restart_every=2000, verbose=False, seed=600)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_short", t_global)

    # ===== Phase 9: Final LP burst =====
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 9: Final LP burst (n=" + str(n_best) + ") ===")
    if remaining > 50 and n_best <= 8000:
        t9 = min(remaining * 0.50, 250.0)
        if t9 > 30:
            a_out, _ = lp_fw(norm_sum1(best_seq), t9, solver,
                             initial_K=600, max_K=3500, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # ===== Phase 10: Final peak shave =====
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 10: Final peak shave (n=" + str(n_best) + ") ===")
    if remaining > 20:
        t10 = min(remaining * 0.60, 150.0)
        if t10 > 10:
            a_out, _ = peak_shave(norm_sum1(best_seq), t10, verbose=True, seed=789)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_final", t_global)

    # ===== Final validation =====
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed)) + "s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    # Ensure all values are valid
    result = np.asarray(best_seq, dtype=np.float64)
    result = np.clip(result, 0.0, 1000.0)
    result[~np.isfinite(result)] = 0.0
    s = result.sum()
    if s < 0.01:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        result = norm_sum1(np.sin(np.pi * x) ** 2) * 1.0

    return [float(x) for x in result]
