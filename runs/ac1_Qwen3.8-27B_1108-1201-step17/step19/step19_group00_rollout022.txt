def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    return a / s

def obj_val(a):

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def get_solver():

def build_constraint_rows(a, active_lags, n):
    """Build sparse constraint rows for given lags. Returns list of (row_array, lag)."""
    rows = []
    for k in active_lags:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            row = a[k - idx]
            rows.append((row, idx))
    return rows

def lp_solve(a, active_lags, solver):
    """Solve the LP: min M s.t. sum(g)=1, g>=0, A@g <= M"""
    n = len(a)
    rows = build_constraint_rows(a, active_lags, n)
    if not rows:
        return None, None
    
    K = len(rows)
    A_mat = np.zeros((K, n), dtype=np.float64)
    for jj, (row, idx) in enumerate(rows):
        A_mat[jj, idx] = row
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [cp.sum(g) == 1, A_mat @ g <= M]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
            g_sum = g_val.sum()
            if g_sum < 1e-15:
                return None, None
            return g_val / g_sum, float(M.value)
    except Exception:
        pass
    return None, None

def golden_section_search(a, direction, f_obj, lo=1e-6, hi=1.0, n_iter=12):
    """Find best alpha in [lo, hi] for a + alpha * direction (projected to simplex)."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    a1 = hi - gr * (hi - lo)
    a2 = lo + gr * (hi - lo)
    
    def eval_alpha(alpha):
        trial = np.maximum(a + alpha * direction, 0.0)
        s = trial.sum()
        if s < 1e-15:
            return 1e10
        trial = trial / s
        return f_obj(trial)
    
    f1 = eval_alpha(a1)
    f2 = eval_alpha(a2)
    
    for _ in range(n_iter):
        if f1 < f2:
            hi = a2
            a2 = a1
            f2 = f1
            a1 = hi - gr * (hi - lo)
            f1 = eval_alpha(a1)
        else:
            lo = a1
            a1 = a2
            f1 = f2
            a2 = lo + gr * (hi - lo)
            f2 = eval_alpha(a2)
    
    best_a = (lo + hi) / 2.0
    trial = np.maximum(a + best_a * direction, 0.0)
    s = trial.sum()
    if s < 1e-15:
        return a.copy()
    return trial / s

    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1

        if M_curr < best_M - 1e-14:

        else:
            no_improve += 1
        
        # Solve LP
        g_dir, lp_M = lp_solve(y, active, solver)
        if g_dir is None:
            if no_improve > stall_limit // 3:
                break
            # Try with fewer constraints
            if len(active) > init_K:
                active = list(np.argsort(ac)[::-1][:init_K].astype(int))
            continue

        Gf = np.fft.rfft(g_dir, 1 << (2*n-2).bit_length())

def projected_gradient(a0, time_limit, verbose=True):
    """Projected subgradient method with AdaGrad-like step size."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    # AdaGrad accumulator
    grad_sq = np.zeros(n, dtype=np.float64)
    base_lr = 0.5
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(x)
        M_curr = ac.max()
        
        # Find active lags (within 1% of max)
        threshold = M_curr * 0.999
        active_lags = np.where(ac >= threshold)[0]
        
        # Compute subgradient
        grad = np.zeros(n, dtype=np.float64)
        for k in active_lags:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            grad[idx] += x[ki - idx]
        grad /= len(active_lags)
        
        # Full gradient of J (with sum normalization)
        # dJ/da[i] = 2n*(grad[i] - 2*M*x[i]) / (sum(x))^2
        # For sum(x)=1: 2n*(grad[i] - 2*M*x[i])
        full_grad = grad - 2.0 * M_curr * x
        
        # AdaGrad step
        grad_sq += full_grad * full_grad
        lr = base_lr / (np.sqrt(grad_sq) + 1e-8)
        
        x_new = x - lr * full_grad
        x_new = np.maximum(x_new, 0.0)
        x_new_sum = x_new.sum()
        if x_new_sum < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / x_new_sum
        
        val = obj_val(x_new)
        if val < best_val - 1e-12:
            best_val = val
            best_a = x_new.copy()
        
        x = x_new
        
        # Periodic restart with noise
        if t % 5000 == 0 and t > 0:
            noise = 0.01 * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 0.0)
            sr = x_r.sum()
            if sr > 1e-15:
                x = x_r / sr
            grad_sq = np.zeros(n, dtype=np.float64)
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  PG it={t} ({elapsed:.0f}s) J={best_val:.6f}")
    
    return best_a, best_val

    print(f"\n=== P1: LP-FW n={n0} ===")
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 60:
        a_out, v_out = lp_fw_optimize(normalize(best_seq), t1, solver,
                                       init_K=25, max_K=1500, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  P1 new best: {best_score:.6f}")
    
    # Phase 2: Targeted deflation
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== P2: Deflation n={len(best_seq)} ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_out, v_out = targeted_deflation(normalize(best_seq), t2, n_targets=10, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  P2 new best: {best_score:.6f}")
    
    # Phase 3: LP-FW continued (refreshed active set)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== P3: LP-FW cont n={n_cur} ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 50:
        a_out, v_out = lp_fw_optimize(normalize(best_seq), t3, solver,
                                       init_K=15, max_K=800, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  P3 new best: {best_score:.6f}")
    
    # Phase 4: Try different length
    remaining = time_budget - (time.monotonic() - t_global)
    n_alt = min(len(best_seq) + 2000, 16000)
    if n_alt != len(best_seq):
        print(f"\n=== P4: Alt length n={n_alt} ===")
        t4 = min(remaining * 0.15, 250.0)
        if t4 > 40:
            a_alt = interpolate_seq(best_seq, n_alt)
            a_out, v_out = lp_fw_optimize(a_alt, t4, solver,
                                           init_K=25, max_K=1200, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print(f"  P4 new best: {best_score:.6f} at n={n_alt}")
    
    # Phase 5: Projected gradient (complementary method)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== P5: Projected Gradient n={n_cur} ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 40:
        a_out, v_out = projected_gradient(normalize(best_seq), t5, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  P5 new best: {best_score:.6f}")
    
    # Phase 6: Final deflation
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== P6: Final Deflation n={n_cur} ===")
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 20:
        a_out, v_out = targeted_deflation(normalize(best_seq), t6, n_targets=12, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  P6 new best: {best_score:.6f}")
    
    # Phase 7: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== P7: Coord Polish n={n_cur} ===")
    t7 = min(remaining * 0.20, 250.0)
    if t7 > 20:
        a_out, v_out = coord_descent_polish(normalize(best_seq), t7, seed=99, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  P7 new best: {best_score:.6f}")
    
    # Phase 8: One more LP-FW if time remains
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 60:
        print(f"\n=== P8: Final LP-FW n={n_cur} (remaining={remaining:.0f}s) ===")
        t8 = remaining * 0.85
        a_out, v_out = lp_fw_optimize(normalize(best_seq), t8, solver,
                                       init_K=20, max_K=1000, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  P8 new best: {best_score:.6f}")
    
    # Final validation
    if best_seq is not None:
        final_score = eval_safe(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.1f}s")
    else:
        print("\nERROR - using fallback")
        n_fb = 10000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.exp(-4 * (x - 0.5)**2))
        best_score = eval_safe(best_seq)
    
    return [float(x) for x in best_seq]
