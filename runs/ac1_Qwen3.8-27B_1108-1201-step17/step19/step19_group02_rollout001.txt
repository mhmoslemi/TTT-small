def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_system_matrix(active_set, a, n):
    """Build constraint matrix rows: (a*g)[k] for k in active_set."""
    K = len(active_set)
    A_mat = np.zeros((K, n))
    for jj, k in enumerate(active_set):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            A_mat[jj, idx] = a[k_int - idx]
    return A_mat


def solve_lp_direction(a, active_set, n, solver):
    """Solve the LP to find the Frank-Wolfe direction."""
    K = len(active_set)
    if K == 0:
        return None
    A_mat = build_system_matrix(active_set, a, n)
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1]
    for jj in range(K):
        constraints.append(A_mat[jj] @ g_var <= M_var)
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
    g_sum = g_val.sum()
    if g_sum < 1e-15:
        return None
    return g_val / g_sum

            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M - 1e-14:
                best_trial_M = M_trial
                best_alpha = ta

        a_new = (1.0 - best_alpha) * a + best_alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) +
                  "s): J=" + repr(round(J, 8)) + ", |S|=" + str(len(active_set)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" +
              repr(round(J, 8)) + ", iters=" + str(iter_count))
    return best_a, best_m


def subgradient_descent(a0, time_limit, top_k=30, momentum=0.5,
                        restart_every=3000, verbose=False):
    """Subgradient method with Polyak step size and momentum."""
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_convolve(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    vel = np.zeros(n)
    stall = 0
    lb = 1.0  # Lower bound on max_conv (for Polyak step)

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_convolve(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
            lb = max(1.0 / n, M_curr * 0.9)
        else:
            stall += 1

        # Compute subgradient: weighted combination of top-k peak gradients
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = ac[top_idx]
        w = top_vals / top_vals.sum()

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            if j_start < j_end:
                idx = np.arange(j_start, j_end)
                grad[idx] += 2.0 * wi * a[k_int - idx]

        # Polyak step size
        gap = M_curr - lb
        eta = gap / max(np.dot(grad, grad), 1e-30)
        eta = min(eta, 0.5)
        eta = max(eta, 1e-5 / t)

        # Gradient step with momentum
        vel = momentum * vel - eta * grad
        a_new = a + vel
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        # Restart with noise
        if stall > restart_every:
            stall = 0
            noise = 0.03 / (1.0 + stall)
            pert = rng.normal(0, noise * 0.1, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-10)
            sr = a_r.sum()
            if sr > 1e-15:
                a = a_r / sr
            vel = np.zeros(n)

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SGD t=" + str(t) + " (" + str(round(elapsed, 1)) +
                  "s): J=" + repr(round(J, 8)) + ", stall=" + str(stall))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    SGD done (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 8)))
    return best_a, best_m


def coordinate_polish_v2(a0, time_limit, n_top_peaks=30, rng_seed=77, verbose=False):
    """Targeted coordinate descent: move mass from peak-contributing indices."""
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_convolve(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 1500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_pk = min(n_top_peaks, 2 * n - 1)
        top_peaks = np.argsort(ac)[::-1][:n_pk]
        peak_vals = ac[top_peaks]
        M_curr = peak_vals[0]

        # For each top peak, find the index contributing most and move mass
        improved = False

        for pk_i in range(min(15, n_pk)):
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(top_peaks[pk_i])
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            if j_start >= j_end:
                continue

            idx = np.arange(j_start, j_end)
            contrib = a[idx] * a[k_int - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-15:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-14:
                continue

            # Find destination: index with minimal contribution across all top peaks
            scores = np.zeros(n)
            for pk_j in range(min(10, n_pk)):
                k2 = int(top_peaks[pk_j])
                s2 = max(0, k2 - n + 1)
                e2 = min(n, k2 + 1)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2)
                    scores[idx2] += a[k2 - idx2]
                    scores[k2 - idx2] += a[idx2]

            # Also penalize indices already large
            scores += 0.5 * a

            # Exclude indices in any peak
            in_peak = np.zeros(n, dtype=bool)
            for pk_j in range(n_pk):
                k2 = int(top_peaks[pk_j])
                s2 = max(0, k2 - n + 1)
                e2 = min(n, k2 + 1)
                in_peak[s2:e2] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            # Also exclude src
            candidates = candidates[candidates != src]
            if len(candidates) == 0:
                continue

            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])

            # Try multiple transfer fractions with exact evaluation
            best_frac = 0
            best_trial_m = M_curr
            for frac in [0.005, 0.01, 0.02, 0.04, 0.06, 0.1, 0.15, 0.2, 0.3]:
                amt = src_val * frac
                if amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_trial_m - 1e-15:
                    best_trial_m = M_trial
                    best_frac = frac
                    best_trial = a_trial

            if best_frac > 0:
                a = best_trial
                best_m = best_trial_m < M_curr ? best_trial_m : best_m
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    M_new = fast_convolve(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 100 == 0:
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" +
                  repr(round(J, 8)) + ", stall=" + str(stall))

    return best_a, best_m


def reweight_flatten(a0, time_limit, iters_per_pass=5000, verbose=False):
    """Iteratively reweight: multiply by exp(-beta * conv_contribution) to flatten peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(iters_per_pass):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_convolve(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()

        # Compute per-index contribution to max peaks
        top_k = np.argsort(ac)[::-1][:min(20, 2 * n - 1)]
        w = ac[top_k]
        w = w / w.sum()

        contrib = np.zeros(n)
        for wi, k in zip(w, top_k):
            k_int = int(k)
            j_s = max(0, k_int - n + 1)
            j_e = min(n, k_int + 1)
            if j_s < j_e:
                idx = np.arange(j_s, j_e)
                contrib[idx] += wi * a[k_int - idx]

        # Multiplicative update: reduce high-contribution indices
        beta = 0.02
        a_new = a * np.exp(-beta * contrib / max(contrib.max(), 1e-15))
        a_new = np.maximum(a_new, 1e-15)
        sn = a_new.sum()
        if sn > 1e-15:
            a_new = a_new / sn
            a = a_new

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search: multi-phase optimization to minimize the evaluation function."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Total time budget: " + repr(round(time_budget, 1)) + "s")

    # --- Initialization ---
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 100 and np.isfinite(p).all() and p.sum() > 1e-10:
            p = p / p.sum()
            best_seq = p
            best_score = evaluate_safe(best_seq)
            print("Parent: n=" + str(len(p)) + ", score=" + repr(best_score))
    except Exception:
        pass

    if best_seq is None:
        n_init = 5000
        x = np.linspace(0, 1, n_init)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(n_init) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # --- Phase 1: Quick parametric exploration ---
    print("\n=== Phase 1: Parametric (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t1 = min(remaining * 0.03, 40.0)
        x = np.linspace(0, 1, n0)
        i_arr = np.arange(n0, dtype=np.float64)
        candidates = []

        # Various parametric families
        for p_exp in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            candidates.append(norm_sum1((i_arr + 1.0) ** p_exp))
            candidates.append(norm_sum1((n0 - i_arr) ** p_exp))

        for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
            for sig in [0.08, 0.15, 0.25, 0.4]:
                candidates.append(norm_sum1(np.exp(-((x - c) ** 2) / (2 * sig * sig))))

        # Asymmetric shapes
        for a1 in [0.5, 1.0, 1.5, 2.0]:
            for a2 in [0.5, 1.0, 1.5, 2.0]:
                a = (i_arr + 1) ** a1 * (n0 - i_arr + 1) ** a2
                candidates.append(norm_sum1(a))

        # Bessel-like
        for beta in [2, 4, 6, 10]:
            a = np.ones(n0)
            for m in range(beta + 1):
                a *= (x * 4.0) ** m / (m + 1)
            candidates.append(norm_sum1(np.maximum(a, 1e-6)))

        for a_cand in candidates:
            if time.monotonic() - t_global > time_budget * 0.97:
                break
            J = compute_obj_normalized(a_cand)
            sc = evaluate_safe(a_cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_cand.copy()
                print("  Parametric: J=" + repr(round(sc, 8)))

    # --- Phase 2: LP-FW main optimization ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: LP-FW main (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 60 and n_best <= 10000:
        a_out, _ = lp_fw_optimized(norm_sum1(best_seq), t2, solver,
                                   initial_K=200, max_K=1200, add_per_iter=20,
                                   verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("*** NEW BEST [lpfw1]: " + repr(round(best_score, 8)))

    # --- Phase 3: Subgradient descent ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Subgradient (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 30:
        a_out, _ = subgradient_descent(norm_sum1(best_seq), t3, top_k=30,
                                       momentum=0.6, restart_every=4000, verbose=False)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("*** NEW BEST [sgd]: " + repr(round(best_score, 8)))

    # --- Phase 4: LP-FW second pass with refined active set ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW refine (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.25, 350.0)
    if t4 > 60 and n_best <= 10000:
        a_out, _ = lp_fw_optimized(norm_sum1(best_seq), t4, solver,
                                   initial_K=300, max_K=1500, add_per_iter=25,
                                   verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("*** NEW BEST [lpfw2]: " + repr(round(best_score, 8)))

    # --- Phase 5: Reweighting to flatten ---

    print("\n=== Phase 5: Reweight (n=" + str(n_best) + ") ===")

                                    iters_per_pass=10000, verbose=False)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("*** NEW BEST [reweight]: " + repr(round(best_score, 8)))

    # --- Phase 6: Multi-n exploration ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    try_ns = []
    for delta in [-500, -200, 200, 500, 800]:
        new_n = n_best + delta
        if 3000 <= new_n <= 10000:
            try_ns.append(new_n)

    print("\n=== Phase 6: Multi-n ===")
    for new_n in try_ns[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print("  Trying n=" + str(new_n) + "")
        a_long = interpolate_seq(best_seq, new_n)
        t6a = min(remaining * 0.30, 180.0)
        if t6a > 40 and new_n <= 10000:
            a_out, _ = lp_fw_optimized(a_long, t6a, solver,
                                       initial_K=200, max_K=1200, add_per_iter=15,
                                       verbose=False)

            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print("*** NEW BEST [n=" + str(new_n) + "]: " + repr(round(best_score, 8)))

        remaining = time_budget - (time.monotonic() - t_global)

                                               momentum=0.5, restart_every=3000, verbose=False)

                    best_seq = a_out.copy()
                    print("*** NEW BEST [sgd_n=" + str(new_n) + "]: " + repr(round(best_score, 8)))

    # --- Phase 7: Final LP-FW ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final LP-FW (n=" + str(n_best) + ") ===")
    if remaining > 60 and n_best <= 10000:
        t7 = min(remaining * 0.35, 250.0)
        if t7 > 30:
            a_out, _ = lp_fw_optimized(norm_sum1(best_seq), t7, solver,
                                       initial_K=400, max_K=1800, add_per_iter=30,
                                       verbose=True)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print("*** NEW BEST [lpfw_final]: " + repr(round(best_score, 8)))

    # --- Phase 8: Final coordinate polish ---

    n_best = len(best_seq)
    print("\n=== Phase 8: Final Coord (n=" + str(n_best) + ") ===")
    if remaining > 20:
