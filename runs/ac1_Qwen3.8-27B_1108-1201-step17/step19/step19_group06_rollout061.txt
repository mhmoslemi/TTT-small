2. Compute c = convolve(a, a) via FFT.
3. Identify the top M lags (where c is largest).
4. For each of these lags, compute the "sensitivity" of c[k] to changes in a[i].
5. Propose moves: for each pair (i, j) where moving mass from j to i would reduce the max over the top lags, evaluate the exact new objective.
6. Accept the best move.
7. Repeat.

The key advantage: exact evaluation means we always make the best possible single-pair move. The disadvantage: it's expensive (O(n^2) pairs per iteration, each requiring O(n log n) to evaluate).

But we can be smart about it:
- Only consider pairs where the sensitivity suggests improvement.
- Use a limited number of candidate pairs per iteration.
- Use the FFT to quickly compute the effect of a move.

Actually, for a single pair move (a[i] += δ, a[j] -= δ), the new convolution is:
c_new[k] = c[k] + δ*(a[k-i] + a[k-j]) - δ*(a[k-i] + a[k-j]) + δ^2*(δ_{k,i+j} + ...)

Wait, let me be more precise. Using the linearity of convolve:
c_new = convolve(a + δ*e_i - δ*e_j, a + δ*e_i - δ*e_j)
= c + 2δ*convolve(e_i, a) - 2δ*convolve(e_j, a) + δ^2*(2*convolve(e_i, e_j) - convolve(e_i, e_i) - convolve(e_j, e_j))

Where e_i is the unit vector with 1 at position i.

convolve(e_i, a)[k] = a[k-i] (if 0 <= k-i < n, else 0).
convolve(e_i, e_j)[k] = 1 if k = i+j, else 0.
convolve(e_i, e_i)[k] = 1 if k = 2i, else 0.

So:
c_new[k] = c[k] + 2δ*(a[k-i] - a[k-j]) + δ^2*(2*δ_{k,i+j} - δ_{k,2i} - δ_{k,2j})

where a[k-i] is 0 if k-i is out of [0, n-1].

This is great! The first-order term is O(1) per k (just looking up a[k-i] and a[k-j]), and the second-order term only affects 3 specific k values.

def interp_seq(a, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(a)

        return a.copy()

    if s < 1e-15:

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact first-order pair-move local search with adaptive step sizes
2. Iterative spectral flattening (DFT-domain)
3. Multi-restart with diverse perturbations  
4. Length exploration via interpolation
5. Coarse-to-fine optimization strategy

import numpy as np
import time


def fast_conv(a):
    """Compute linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def eval_safe(a):
    """Safely call evaluate_sequence."""
    try:
        return float(evaluate_sequence([float(x) for x in a]))
    except Exception:
        return float('inf')


def interp_seq(a, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    an = a / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    out = np.interp(x_new, x_old, an)
    return norm1(out)

        # For scoring:
        # "cost of being a source" at p: sum_k w_k * a[k-p] (we want this LOW)
        # "benefit of being a destination" at p: sum_k w_k * a[k-p] (we want this HIGH)
        # Wait, these are the same! That can't be right.
        
        # Let me reconsider. Δc[k] = 2δ * (a[k-src] - a[k-dst])
        # For the max to decrease, we need a[k-src] < a[k-dst] for the peak k.
        # So: src should be at a position where a[k-src] is SMALL for peak k.
        #     dst should be at a position where a[k-dst] is LARGE for peak k.
        
        # Define: for each position p, profile[p] = sum_{k in active} a[k-p] (with bounds)
        # Then: src wants LOW profile, dst wants HIGH profile.
        
        profile = np.zeros(n)
        for k in active:
            ki = int(k)
            i_lo = max(0, ki - n + 1)
            i_hi = min(n, ki + 1)
            if i_lo < i_hi:
                # For lag ki, the term a[ki-p] is valid when 0 <= ki-p < n
                # i.e., ki-n < p <= ki, i.e., max(0,ki-n+1) <= p <= min(n-1, ki)
                p_lo = max(0, ki - n + 1)
                p_hi = min(n - 1, ki)
                if p_lo <= p_hi:
                    p_idx = np.arange(p_lo, p_hi + 1)
                    j_idx = ki - p_idx
                    profile[p_idx] += a[j_idx]
        
        # Source quality: want high a[p] (have mass) and low profile[p] (low impact)
        src_score = a / (profile + 1e-12)
        # Destination quality: want high profile[p]
        dst_score = profile.copy()
        
        # Get top candidates
        top_src = np.argsort(src_score)[::-1][:25]
        top_dst = np.argsort(dst_score)[::-1][:25]
        
        best_new_m = cur_m
        best_a_new = None
        best_c_new = None
        
        # Evaluate candidate pairs
        n_candidates = 0
        for si in top_src:
            si = int(si)
            if a[si] < 1e-12:
                continue
            for di in top_dst:
                di = int(di)
                if di == si:
                    continue
                
                # Quick first-order check
                # At the main peak:
                k_peak = int(active[0])
                a_kp_s = a[k_peak - si] if 0 <= k_peak - si < n else 0.0
                a_kp_d = a[k_peak - di] if 0 <= k_peak - di < n else 0.0
                if a_kp_d <= a_kp_s:
                    continue  # Won't help the main peak
                
                # Try a step size
                # Heuristic: delta ≈ 0.1 * a[si], but adjust based on the "gap"
                base_delta = 0.1 * a[si]
                
                for frac_mult in [0.5, 1.0, 2.0]:
                    delta = base_delta * frac_mult
                    if delta > 0.3 * a[si]:
                        break
                    
                    a_new = a.copy()
                    a_new[si] += delta
                    a_new[di] -= delta
                    if a_new[di] < -1e-12:
                        continue
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new < 1e-15:
                        continue
                    a_new = a_new / s_new
                    c_new = fast_conv(a_new)
                    new_m = c_new.max()
                    n_candidates += 1
                    
                    if new_m < best_new_m:
                        best_new_m = new_m
                        best_a_new = a_new.copy()
                        best_c_new = c_new.copy()
        
        # Apply best move
        if best_a_new is not None and best_new_m < cur_m - 1e-13:
            a = best_a_new
            c = best_c_new
            cur_m = best_new_m
            stall = 0
            if cur_m < best_m - 1e-13:
                best_m = cur_m
                best_a = a.copy()
                if verbose and iter_count % 50 == 0:
                    J = 2.0 * n * best_m
                    print(f"      ELS iter={iter_count}, candidates={n_candidates}: J={J:.6f}")
        else:
            stall += 1
            if stall > 200:
                # Perturb to escape
                stall = 0
                w = float(rng.uniform(0.03, 0.15))
                # Different types of perturbations
                pert_type = rng.integers(3)
                if pert_type == 0:
                    pert = norm1(rng.chisquare(2.0, n))
                elif pert_type == 1:
                    x = np.linspace(0, 1, n)
                    c0 = float(rng.uniform(0.2, 0.8))
                    sig = float(rng.uniform(0.1, 0.4))
                    pert = norm1(np.exp(-((x - c0)**2) / (2 * sig**2)))
                else:
                    pert = norm1(np.ones(n) + rng.standard_normal(n) * 0.3)
                
                a_try = norm1((1 - w) * a + w * pert)
                c_try = fast_conv(a_try)
                if c_try.max() <= cur_m * 1.01:  # Allow slight worsening
                    a = a_try
                    c = c_try
                    cur_m = c_try.max()
    
    return best_a, 2.0 * n * best_m


def spectral_flatten(a, time_limit, rng, verbose=False):
    """
    Iteratively flatten the power spectrum to reduce autocorrelation peaks.
    Works in the DFT domain with periodic non-negative projection.
    """
    n = len(a)
    a = norm1(a.copy())
    c = fast_conv(a)
    best_a = a.copy()
    best_m = c.max()
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 2).bit_length()
    lr = 0.5
    beta = 2.0
    stall = 0
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute spectrum
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_min = P.min()
        P_mean = P.mean()
        
        if P_max < 1e-30:
            break
        
        # Flattening: modify A to reduce spectral peaks
        # Target: make P more uniform
        # A_new = A * sqrt(target / P)
        # But we need to handle the DC term and Nyquist carefully
        
        target = P_mean
        ratio = np.sqrt(target / (P + 1e-30))
        ratio = np.clip(ratio, 0.3 / lr, 3.0 / lr)  # Limit step size
        
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        c_new = fast_conv(a_new)
        new_m = c_new.max()
        
        if new_m < best_m - 1e-12:
            best_m = new_m
            best_a = a_new.copy()
            a = a_new
            c = c_new
            stall = 0
        else:
            stall += 1
            if stall % 100 == 0:
                lr *= 0.9
                beta = max(beta * 0.95, 0.5)
        
        if stall > 500:
            # Reset with perturbation
            stall = 0
            w = float(rng.uniform(0.05, 0.2))
            pert = norm1(rng.chisquare(2.0, n))
            a = norm1((1 - w) * best_a + w * pert)
            lr = 0.3
        
        if verbose and it % 200 == 0:
            J = 2.0 * n * best_m
            print(f"      SF iter={it}: J={J:.6f}, lr={lr:.3f}")
    
    return best_a, 2.0 * n * best_m
