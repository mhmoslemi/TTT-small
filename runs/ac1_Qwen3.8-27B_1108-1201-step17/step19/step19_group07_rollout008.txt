"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe with Nesterov acceleration and full constraint set
2. FISTA polishing with adaptive beta
3. Multi-length exploration
4. Adaptive constraint management

import numpy as np
import time
import scipy.sparse as sp
import cvxpy as cp


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS


def precompute_sparsity(n):
    """Precompute the sparsity pattern for the constraint matrix."""
    size = 2 * n - 1
    rows_list = []
    cols_list = []
    for k in range(size):
        j_start = max(0, k - n + 1)
        j_end = n - 1
        js = np.arange(j_start, j_end + 1)
        rows_list.extend([k] * len(js))
        cols_list.extend(js.tolist())
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def compute_constraint_values(y, n, rows, cols):
    """Compute A[k,j] = y[k-j] + y[j-k] for all nonzero (k,j) pairs."""
    t1_idx = rows - cols  # k-j, always in [0, n-1]
    t2_idx = cols - rows  # j-k, in [-(n-1), n-1]
    t1 = y[t1_idx]
    t2 = np.where(t2_idx >= 0, y[t2_idx], 0.0)
    return t1 + t2


def lp_frank_wolfe(a0, time_limit, solver, max_constraints=None, verbose=True):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    rows, cols = precompute_sparsity(n)
    y = a.copy()
    x = a.copy()
    
    iter_count = 0
    last_print = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute autocorrelation at y
        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        
        # Update best
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
        
        b_x = fast_autocorr(x)
        if b_x.max() < best_m - 1e-15:
            best_m = b_x.max()
            best_a = x.copy()
        
        # Build constraint matrix
        vals = compute_constraint_values(y, n, rows, cols)
        A = sp.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [
            cp.sum(g_var) == 1,
            A @ g_var - M_var <= b_y  # A@g - M <= b_y[k] for each k
        ]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            t_lp_start = time.monotonic()
            prob.solve(solver=solver, verbose=False)
            t_lp_end = time.monotonic()
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Evaluate direction
        b_g = fast_autocorr(g_dir)
        M_g = b_g.max()
        
        # Nesterov step
        if iter_count == 0:
            y_new = g_dir.copy()
            x_new = g_dir.copy()
        else:
            alpha = 2.0 / (iter_count + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / sy
            
            beta_t = iter_count / (iter_count + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        if verbose and iter_count - last_print >= 5:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            lp_time = t_lp_end - t_lp_start if 't_lp_end' in dir() else 0
            print(f"  LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, LP_time={lp_time:.1f}s, M_y={M_curr:.8f}, M_g={M_g:.8f}")
            last_print = iter_count
    
    return best_a, best_m
