And $\frac{\partial f}{\partial x_j} = \sum_i \frac{\partial f}{\partial a_i} \cdot a_i(\delta_{ij} - a_j) = a_j \left(\frac{\partial f}{\partial a_j} - \sum_i \frac{\partial f}{\partial a_i} a_i\right)$

This automatically ensures $a \ge 0$ and $\sum a = 1$.

The only downside is that $a_i$ can never be exactly 0 (it's always positive). For our purposes, this should be fine since the optimum likely has all $a_i > 0$ (or very close to 0 for some).

"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on log-sum-exp smooth approximation (FFT-accelerated gradients)

import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp


def fast_conv(a, b):
    """Full convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def auto_conv(a):
    """Self-convolution a * a."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def logsumexp_gradient(x, beta):
    """
    Compute gradient and value of log-sum-exp approximation of max(auto_conv(softmax(x))).
    
    Parameters:
    - x: logits of length n (will be converted to simplex via softmax)
    - beta: sharpness parameter
    
    Returns:
    - grad: gradient w.r.t. x, shape (n,)
    - val: scalar value
    """
    n = len(x)
    
    # Softmax
    x_shift = x - x.max()
    a = np.exp(x_shift)
    a_sum = a.sum()
    a = a / a_sum
    
    # Auto-convolution
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    
    # Log-sum-exp of beta * c
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    p = exp_bc / exp_bc.sum()
    val = (bc_max + np.log(exp_bc.sum())) / beta
    
    # Gradient of f_beta w.r.t. a: 2 * cross_correlate(p, a)
    # cross_correlate(p, a)[i] = sum_k p[k] * a[k-i] (with a[j]=0 outside [0,n-1])
    fft_size_g = 1 << (size + n - 1).bit_length()
    P = np.fft.rfft(p, fft_size_g)
    A_full = np.fft.rfft(a, fft_size_g)
    # We want g[i] = sum_k p[k] a[k-i]
    # This is: ifft(FFT(p) * conj(FFT(a))) gives sum_k p[k+i] a[k] = sum_j p[j] a[j-i]
    # So corr = ifft(P * conj(A_full)) and corr[i] = sum_j p[j] a[j-i]
    corr = np.fft.irfft(P * np.conj(A_full), fft_size_g)
    grad_a = 2.0 * corr[:n]
    
    # Chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    
    return grad_x, val


def lbfgs_optimize(x0, beta_start, beta_end, time_limit, verbose=False):
    """
    Run L-BFGS-B on the log-sum-exp objective with adaptive beta.
    """
    n = len(x0)
    t0 = time.monotonic()
    best_x = x0.copy()
    best_val = logsumexp_gradient(x0, beta_start)[1]
    
    # Schedule: increase beta over time
    n_inner = 0
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)
        # Exponential schedule for beta
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Run L-BFGS for a short time
        inner_time = min(10.0, (time_limit - elapsed) * 0.3)
        
        def func_grad(x):
            g, v = logsumexp_gradient(x, beta)
            return v, g
        
        # L-BFGS-B (unconstrained since softmax handles constraints)
        result = minimize(
            func_grad, best_x,
            method='L-BFGS-B',
            jac=True,
            options={
                'maxiter': 200,
                'gtol': 1e-8,
                'maxfun': 300,
                'iprint': -1,
            }
        )
        
        new_val = result.fun
        if new_val < best_val:
            best_val = new_val
            best_x = result.x.copy()
        
        n_inner += 1
        if verbose and n_inner % 5 == 0:
            print(f"    L-BFGS round {n_inner}: beta={beta:.1f}, val={best_val:.8f}, t={elapsed:.0f}s")
    
    # Convert back to simplex
    a = np.exp(best_x - best_x.max())
    a = a / a.sum()
    return a, best_val


def lp_frank_wolfe(a0, time_limit, solver, initial_K=200, max_K=3000, verbose=True):
    """Improved LP Frank-Wolfe with better constraint management."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = auto_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()  # Nesterov point
    x = a.copy()  # Actual iterate
    
    # Initialize constraint set
    ac_y = auto_conv(y)
    M_curr = ac_y.max()
    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)
    
    failed_solves = 0
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        ac = auto_conv(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
        
        ac_x = auto_conv(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()
        
        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]
        
        # Solve LP: min M s.t. coeffs_matrix @ g <= M, g >= 0, sum(g) = 1
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 10:
                # Reset constraint set
                ac_y = auto_conv(y)
                init_idx = np.argsort(ac_y)[::-1][:init_K]
                constraint_set = set(int(k) for k in init_idx)
                failed_solves = 0
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            failed_solves += 1
            if failed_solves > 10:
                constraint_list = sorted(constraint_set)[:init_K]
                constraint_set = set(constraint_list)
                failed_solves = 0
            continue
        
        failed_solves = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violated constraints and add them
        conv_ys = fast_conv(y, g_dir)
        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")
    
    return best_a, best_m


def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(30, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS


def parametric_init(n, best_ref=None):
    """Generate diverse parametric initializations."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')
    
    candidates = []
    
    # Symmetric power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 2.5, 3.0]:
        sym = (i_arr + 1.0) ** p
        sym = sym + sym[::-1]
        candidates.append(sym)
    
    # Beta-distribution-like
    for alpha in [0.5, 0.8, 1.0, 1.5, 2.0]:
        for beta_p in [0.5, 0.8, 1.0, 1.5, 2.0]:
            c = np.maximum(i_arr + 1.0, 0.3) ** alpha * np.maximum(n - i_arr, 0.3) ** beta_p
            candidates.append(c)
    
    # Gaussian centered
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    
    # Sine-based (symmetric)
    for freq in [1, 2, 3]:
        candidates.append(np.abs(np.sin(np.pi * freq * x / 2.0)) ** 2)
        candidates.append(1.0 + np.cos(np.pi * x))
    
    # Tent / triangle
    candidates.append(1.0 - np.abs(2.0 * x - 1.0))
    
    # If reference available, blend
    if best_ref is not None and len(best_ref) == n:
        ref_norm = best_ref / best_ref.sum()
        for w in [0.3, 0.5, 0.7, 0.9]:
            for p in [0.5, 1.0, 1.5]:
                shape = (i_arr + 1.0) ** p
                shape = shape + shape[::-1]
                shape = shape / shape.sum()
                candidates.append(w * ref_norm + (1.0 - w) * shape)
    
    for a_cand in candidates:
        a_cand = np.maximum(a_cand, 0.0)
        s = a_cand.sum()
        if s < 1e-15:
            continue
        a_norm = a_cand / s
        ac = auto_conv(a_norm)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_a = a_norm.copy()
    
    if best_a is None:
        best_a = np.ones(n) / n
        best_val = auto_conv(best_a).max()
    
    return best_a, best_val


def random_smooth_init(n, rng):
    """Generate a random smooth non-negative sequence."""
    # Random frequency-domain: flat magnitude with random phase
    fft_size = 1 << (2 * n - 1).bit_length()
    phase = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    # Use a low-pass filter to ensure smoothness
    freq = np.arange(fft_size // 2 + 1)
    cutoff = rng.uniform(0.3, 1.0) * (fft_size // 2)
    magnitude = np.exp(-0.5 * (freq / cutoff) ** 2)
    spectrum = magnitude * np.exp(1j * phase)
    spectrum[0] = magnitude[0]  # Real DC
    a_full = np.fft.irfft(spectrum, fft_size)[:n]
    a_full = np.maximum(a_full, 0.0)
    s = a_full.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_full / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
