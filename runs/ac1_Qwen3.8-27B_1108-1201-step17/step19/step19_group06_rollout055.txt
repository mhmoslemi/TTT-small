    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score = 2n * max(conv(a,a)) / sum(a)^2. Assumes a >= 0."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_conv(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def spectral_equalization(a, time_limit, alpha_init=0.7, alpha_end=0.3,
                          n_max_iter=2000000, rng_seed=42, verbose=True):
    """
    Iteratively flatten the power spectrum of the sequence.
    P_target = P_mean^alpha * P^(1-alpha)
    This reduces the dynamic range of the spectrum, making the convolution flatter.
    """
    n = len(a)
    a = normalize(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    fft_size = 1 << (2 * n - 2).bit_length()
    stall = 0
    stall_threshold = 5000
    count = 0
    
    while time.monotonic() - t0 < time_limit:
        alpha = alpha_init + (alpha_end - alpha_init) * min(1.0, count / 50000.0)
        
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        P_safe = np.maximum(P, 1e-30)
        
        # Target spectrum: geometric interpolation between P and flat
        log_P = np.log(P_safe)
        log_target = alpha * np.log(P_mean) + (1 - alpha) * log_P
        P_target = np.exp(log_target)
        
        # Scale spectrum
        ratio = np.sqrt(P_target / P_safe)
        ratio = np.clip(ratio, 0.01, 100.0)
        A_new = A * ratio
        
        # Recover time domain
        a_new = np.fft.irfft(A_new, fft_size)[:n].real
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            # Failed, perturb and retry
            a = normalize(best_a + rng.normal(0, 0.01, n) * best_a)
            count += 1
            continue
        a_new = a_new / s
        
        # Evaluate periodically
        if count % 50 == 0:
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                stall = 0
                if verbose and count % 5000 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"      SpecEq: score={sc:.6f}, alpha={alpha:.3f}, t={elapsed:.0f}s")
            else:
                stall += 1
            
            if stall > stall_threshold:
                # Adaptive restart
                stall = 0
                alpha = max(0.1, alpha - 0.2)
                # Perturb
                w = float(rng.uniform(0.05, 0.3))
                a = normalize((1 - w) * best_a + w * rng.random(n))
                stall_threshold = max(2000, stall_threshold // 2)
        else:
            a = a_new
        
        count += 1
    
    if verbose:
        print(f"      Spectral eq done ({time.monotonic()-t0:.0f}s): score={best_score:.6f}, iters={count}")
    return best_a, best_score


def freq_gradient_descent(a, time_limit, lr_init=0.5, beta=3.0,
                          verbose=True, rng_seed=123):
    """
    Gradient descent in the frequency domain.
    Minimizes max(conv(a,a)) by pushing spectral energy from peaks to valleys.
    """
    n = len(a)
    a = normalize(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    fft_size = 1 << (2 * n - 2).bit_length()
    lr = lr_init
    stall = 0
    
    count = 0
    while time.monotonic() - t0 < time_limit:
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_mean = P.mean()
        
        if P_max < 1e-30:
            break
        
        # Compute soft-max weights over spectral peaks
        dev = (P - P_mean) / max(P_max - P_mean, 1e-30)
        expo = beta * dev
        expo = np.clip(expo, -30, 30)
        w = np.exp(expo)
        w = w / (w.sum() + 1e-30)
        
        # Fade the spectrum at peaks
        fade = 1.0 - lr / (1.0 + 0.1 * np.sqrt(count)) * (w / (w.mean() + 1e-30))
        fade = np.clip(fade, 0.1, 10.0)
        A_new = A * fade
        
        a_new = np.fft.irfft(A_new, fft_size)[:n].real
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a = normalize(best_a + rng.normal(0, 0.02, n) * best_a)
            count += 1
            continue
        a_new = a_new / s
        
        if count % 20 == 0:
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
                if stall > 3000:
                    a = normalize(best_a * (1 + rng.normal(0, 0.03, n)))
                    lr *= 0.7
                    beta = max(1.0, beta - 0.5)
                    stall = 0
        a = a_new
        count += 1
    
    if verbose:
        print(f"      FreqGD done ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")
    return best_a, best_score


def coord_descent_mass_transfer(a, time_limit, verbose=True, rng_seed=77):
    """
    Coordinate descent: find top convolution peaks, then transfer mass
    from high-contributing positions to low-impact positions.
    """
    n = len(a)
    a = normalize(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    fft_size = 1 << (2 * n - 2).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv(a)
        M = conv.max()
        
        if M < best_score / (2.0 * n) - 1e-15:
            best_a = a.copy()
            best_score = 2.0 * n * M
            stall = 0
        else:
            stall += 1
            if stall > 50000:
                break
        
        # Get top lags in convolution
        n_top = min(20, len(conv))
        top_lags = np.argsort(conv)[::-1][:n_top]
        
        improved = False
        
        for k_pos in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k_pos)
            # Find contributing pairs
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo >= i_hi:
                continue
            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            
            # Top contributing pair
            pi = int(np.argmax(contrib))
            src = int(idx[pi])
            
            if a[src] < 1e-12:
                continue
            
            # Find best destination: minimize impact on top lags
            scores = np.zeros(n)
            for k2 in top_lags[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                if s2 < e2:
                    ii = np.arange(s2, e2)
                    jj = k2i - ii
                    scores[ii] += a[jj] * conv[k2i]
                    scores[jj] += a[ii] * conv[k2i]
            
            scores[src] = np.inf
            dst = int(np.argmin(scores))
            
            if dst == src:
                continue
            
            # Try transfer fractions
            for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = normalize(a_trial)
                
                # Quick score check via FFT
                A = np.fft.rfft(a_trial, fft_size)
                conv_t = np.fft.irfft(A * A, fft_size)[:2*n-1]
                sc = 2.0 * n * conv_t.max()
                
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a_trial.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved and stall % 1000 == 0:
            # Random perturbation
            n_moves = int(rng.integers(3, 15))
            for _ in range(n_moves):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = float(rng.uniform(0.002, 0.02))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = normalize(a)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall = 0
    
    if verbose:
        print(f"      Coord descent done ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")
    return best_a, best_score


def simulated_annealing(a, time_limit, t_init=0.1, t_min=0.001,
                        verbose=True, rng_seed=55):
    """
    Simulated annealing with random coordinate moves.
    """
    n = len(a)
    a = normalize(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    fft_size = 1 << (2 * n - 2).bit_length()
    temp = t_init
    stall = 0
    count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Temperature schedule
        temp = max(t_min, t_init * np.exp(-0.0001 * count))
        
        # Random move: pick 2-5 coordinates and redistribute
        n_moves = int(rng.integers(1, 5))
        a_trial = a.copy()
        for _ in range(n_moves):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            # Move some mass from i to j (or vice versa)
            if rng.random() < 0.5:
                amt = a[i] * float(rng.uniform(0.01, 0.2))
                a_trial[i] -= amt
                a_trial[j] += amt
            else:
                amt = a[j] * float(rng.uniform(0.01, 0.2))
                a_trial[j] -= amt
                a_trial[i] += amt
        
        a_trial = normalize(a_trial)
        
        # Quick score
        A = np.fft.rfft(a_trial, fft_size)
        conv_t = np.fft.irfft(A * A, fft_size)[:2*n-1]
        sc_trial = 2.0 * n * conv_t.max()
        
        delta = sc_trial - best_score
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_trial
            if sc_trial < best_score:
                best_score = sc_trial
                best_a = a_trial.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        if stall > 20000:
            # Restart from best with perturbation
            w = float(rng.uniform(0.05, 0.2))
            a = normalize((1 - w) * best_a + w * rng.random(n))
            stall = 0
            temp = min(t_init, temp * 5)
        
        count += 1
    
    if verbose:
        print(f"      SimAnneal done ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")
    return best_a, best_score


def generate_diverse_inits(n, rng):
    """Generate diverse initial sequences of length n."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Raised cosine
    inits.append(normalize(0.5 + 0.5 * np.cos(np.pi * x)))
    
    # Linear ramp
    inits.append(normalize(np.linspace(0.1, 1.0, n)))
    inits.append(normalize(np.linspace(1.0, 0.1, n)))
    
    # Triangular
    inits.append(normalize(1.0 - np.abs(2 * x - 1)))
    
    # Power laws
    for p in [0.5, 1.0, 2.0, 3.0]:
        inits.append(normalize((i_arr + 1) ** p))
        inits.append(normalize((n - i_arr) ** p))
    
    # Gaussian
    for c in [0.3, 0.5, 0.7]:
        for sig in [0.15, 0.25, 0.4]:
            inits.append(normalize(np.exp(-((x - c)**2) / (2 * sig**2))))
    
    # Bimodal
    inits.append(normalize(
        np.exp(-((x - 0.25)**2) / (2 * 0.15**2)) + np.exp(-((x - 0.75)**2) / (2 * 0.15**2))))
    
    # Sine
    for f in [1, 2, 3]:
        inits.append(normalize(np.sin(np.pi * f * x) ** 2))
    
    # Chirp-based (flat spectrum)
    k = np.arange(n)
    chirp_phase = np.pi * k**2 / n
    a_chirp = np.fft.irfft(np.exp(1j * chirp_phase), 1 << (2*n-2).bit_length())[:n].real
    inits.append(normalize(a_chirp))
    inits.append(normalize(np.abs(a_chirp)))
    
    # Random phase with flat magnitude
    fft_size = 1 << (2*n-2).bit_length()
    n_freq = fft_size // 2 + 1
    for _ in range(3):
        phases = rng.random(n_freq) * 2 * np.pi
        # Enforce Hermitian symmetry for real output
        A = np.ones(n_freq)
        A[1:] = np.exp(1j * phases[1:])
        A[-1] = 1.0  # Nyquist
        # Make it Hermitian
        A_full = np.zeros(fft_size, dtype=complex)
        A_full[:n_freq] = A
        A_full[n_freq:] = np.conj(A_full[1:n_freq][::-1])
        a_r = np.fft.ifft(A_full)[:n].real
        inits.append(normalize(a_r))
    
    # Random non-negative
    for _ in range(3):
        inits.append(normalize(rng.random(n)))
    
    return inits


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    return normalize(f_int)


def multi_start_optimize(a_init, time_limit, verbose=True):
    """Run multiple optimization methods on a given initialization."""
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    t0 = time.monotonic()
    
    if verbose:
        print(f"    Initial score: {best_score:.6f}")
    
    remaining = time_limit - (time.monotonic() - t0)
    
    # Phase A: Spectral equalization (aggressive)
    t_a = min(remaining * 0.3, 300.0)
    if t_a > 10:
        a_out, sc = spectral_equalization(a_init, t_a, alpha_init=0.8, alpha_end=0.4,
                                         rng_seed=1, verbose=verbose)
        if sc < best_score:
            best_score = sc
            best_a = a_out
            print(f"    After SpecEq: {best_score:.6f}")
    
    remaining = time_limit - (time.monotonic() - t0)
    
    # Phase B: Frequency gradient descent
    t_b = min(remaining * 0.25, 200.0)
    if t_b > 10:
        a_out, sc = freq_gradient_descent(best_a, t_b, lr_init=0.3, beta=4.0,
                                         verbose=verbose, rng_seed=2)
        if sc < best_score:
            best_score = sc
            best_a = a_out
            print(f"    After FreqGD: {best_score:.6f}")
    
    remaining = time_limit - (time.monotonic() - t0)
    
    # Phase C: Spectral equalization (gentle, from current best)
    t_c = min(remaining * 0.25, 200.0)
    if t_c > 10:
        a_out, sc = spectral_equalization(best_a, t_c, alpha_init=0.4, alpha_end=0.1,
                                         rng_seed=3, verbose=verbose)
        if sc < best_score:
            best_score = sc
            best_a = a_out
            print(f"    After SpecEq2: {best_score:.6f}")
    
    remaining = time_limit - (time.monotonic() - t0)
    
    # Phase D: Coordinate descent
    t_d = min(remaining * 0.2, 150.0)
    if t_d > 10:
        a_out, sc = coord_descent_mass_transfer(best_a, t_d, verbose=verbose, rng_seed=4)
        if sc < best_score:
            best_score = sc
            best_a = a_out
            print(f"    After CoordDescent: {best_score:.6f}")
    
    remaining = time_limit - (time.monotonic() - t0)
    
    # Phase E: Simulated annealing
    t_e = min(remaining * 0.2, 120.0)
    if t_e > 10:
        a_out, sc = simulated_annealing(best_a, t_e, t_init=0.05, verbose=verbose, rng_seed=5)
        if sc < best_score:
            best_score = sc
            best_a = a_out
            print(f"    After SimAnneal: {best_score:.6f}")
    
    return best_a, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Multi-phase optimization with diverse starts and multiple techniques.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    rng = np.random.default_rng(2024)
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            p_score = compute_score(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    n0 = len(best_seq) if best_seq is not None else 5000
    
    if best_seq is None:
        best_seq = np.ones(n0) / n0
        best_score = compute_score(best_seq)
        print(f"Default uniform init: n={n0}, score={best_score:.6f}")
    
    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = normalize(a)
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Phase 1: Multi-start with diverse initializations at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Multi-start (n={n0}, budget={remaining:.0f}s) ===")
    
    inits = generate_diverse_inits(n0, rng)
    # Add parent as first candidate
    inits.insert(0, best_seq.copy())
    
    # Quick evaluation of all inits
    init_scores = []
    for i, ini in enumerate(inits):
        sc = compute_score(ini)
        init_scores.append(sc)
        if sc < best_score:
            best_score = sc
            best_seq = ini.copy()
    
    # Sort by score, take top candidates
    init_order = np.argsort(init_scores)
    n_to_try = min(8, len(inits))
    
    # Allocate time for multi-start
    t_per_start = min(remaining * 0.5 / n_to_try, 120.0)
    
    for rank in range(n_to_try):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        idx = int(init_order[rank])
        a_init = inits[idx]
        init_sc = init_scores[idx]
        
        if init_sc > best_score * 1.1:
            continue  # Skip if too far from best
        
        print(f"  Start {rank+1}/{n_to_try}: init_score={init_sc:.6f}, t_alloc={t_per_start:.0f}s")
        a_out, sc = multi_start_optimize(a_init, t_per_start, verbose=False)
        check_update(a_out, f"start{rank+1}_n{n0}")
    
    # === Phase 2: Focused optimization on best found ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Focused optimization (n={n_best}, budget={remaining:.0f}s) ===")
    
    t_focused = min(remaining * 0.35, 400.0)
    if t_focused > 20:
        a_out, sc = multi_start_optimize(best_seq, t_focused, verbose=True)
        check_update(a_out, f"focused_n{n_best}")
    
    # === Phase 3: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Length exploration (budget={remaining:.0f}s) ===")
    
    length_tries = sorted(set([
        n_best - 1000, n_best - 500, n_best + 500, n_best + 1000,
        3000, 4000, 5000, 6000, 7000, 8000
    ]))
    length_tries = [ln for ln in length_tries if 2000 <= ln <= 10000 and ln != n_best]
    
    for new_n in length_tries[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")
        
        a_interp = interpolate_seq(best_seq, new_n)
        t_len = min(remaining * 0.3, 120.0)
        if t_len > 15:
            a_out, sc = multi_start_optimize(a_interp, t_len, verbose=False)
            check_update(a_out, f"len_n{new_n}")
    
    # === Phase 4: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Final polish (n={n_best}, budget={remaining:.0f}s) ===")
    
    if remaining > 20:
        # Gentle spectral equalization
        t_poly = min(remaining * 0.4, 150.0)
        a_out, sc = spectral_equalization(best_seq, t_poly, alpha_init=0.3,
                                          alpha_end=0.05, rng_seed=99, verbose=True)
        check_update(a_out, f"polish_spec_n{n_best}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t_coord = min(remaining * 0.4, 80.0)
            a_out, sc = coord_descent_mass_transfer(best_seq, t_coord, verbose=False, rng_seed=42)
            check_update(a_out, f"polish_coord_n{n_best}")
    
    # === Phase 5: One more round of diverse search if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 5: Additional diverse search (budget={remaining:.0f}s) ===")
        n_best = len(best_seq)
        
        # Try a few more random inits
        for ri in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            a_rand = normalize(best_seq * (1 + rng.normal(0, 0.2, n_best)))
            t_r = min(remaining * 0.25, 80.0)
            a_out, sc = multi_start_optimize(a_rand, t_r, verbose=False)
            check_update(a_out, f"diverse{ri}_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = compute_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = np.ones(5000) / 5000
        best_score = compute_score(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")
    
    # Ensure we return valid values (clip to [0, 1000])
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    
    return [float(x) for x in best_seq]
