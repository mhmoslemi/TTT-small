def smoothed_grad_descent(a0, time_limit, tau_init=100.0, tau_final=500.0, 
                           eta_init=0.1, c=1.0, verbose=True):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature
        progress = min(t / 10000.0, 1.0)
        tau = tau_init * (tau_final / tau_init) ** progress
        
        # Compute gradient
        ac = fft_autocorr(a)
        max_ac = ac.max()
        exp_ac = np.exp(tau * (ac - max_ac))
        w = exp_ac / exp_ac.sum()
        
        A = np.fft.rfft(a, fft_size)
        W = np.fft.rfft(w, fft_size)
        corr = np.fft.irfft(np.conj(A) * W, fft_size)
        grad = 2.0 * corr[:n]
        
        # Adaptive step size
        eta = eta_init / (1.0 + 0.1 * np.sqrt(t))
        
        # Multiplicative update
        log_a = np.log(a + 1e-30) - eta * grad
        a_new = np.exp(log_a - log_a.max())
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        a = a_new
        
        # Check improvement
        val = 2.0 * n * fft_autocorr(a).max()
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
        
        if verbose and t % 100 == 0:
            print(f"  SGD t={t} tau={tau:.1f} val={best_val:.6f}")
    
    return best_a, best_val

1. FFT-based smoothed gradient descent (fast bulk optimization)
2. LP-based Frank-Wolfe with adaptive active set (refinement)
3. Exact coordinate descent polishing (final precision)
4. Multi-length exploration and parameterized global search
5. Hybrid scheduling to maximize quality within time budget
"""

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None

def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  Solver fallback: ECOS")
    return _MODULE_SOLVER

def fft_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    ac = fft_autocorr(a)
    s = a.sum()
    if s < 1e-15:
        return 1e10
    return 2.0 * n * ac.max() / (s * s)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_lp_constraint_matrix(a, active_set, n):
    K = len(active_set)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(active_set):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]
    return coeffs

def smoothed_grad_descent(a0, time_limit, verbose=True):
    """Fast smoothed-gradient descent with multiplicative updates and adaptive temperature."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    stall = 0
    stall_limit = 3000
    tau = 50.0
    tau_max = 500.0
    eta = 0.5
    rng = np.random.default_rng(123)
    n_iter = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > stall_limit:
            break

        n_iter += 1
        ac = fft_autocorr(a)
        ac_max = ac.max()
        exp_ac = np.exp(tau * (ac - ac_max))
        w = exp_ac / exp_ac.sum()

        A = np.fft.rfft(a, fft_size)
        W = np.fft.rfft(w, fft_size)
        corr = np.fft.irfft(np.conj(A) * W, fft_size)
        grad = 2.0 * corr[:n]

        grad_mean = grad.mean()
        grad_centered = grad - grad_mean

        eta_t = min(eta, 0.8 / (np.sqrt(1.0 + 0.05 * np.linalg.norm(grad_centered, ord=np.inf) + 1e-10)))
        log_a = np.log(a + 1e-30) - eta_t * grad_centered
        a_new = np.exp(log_a - log_a.max())
        a_new = np.maximum(a_new, 1e-25)
        ss = a_new.sum()
        if ss < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / ss
        a = a_new

        if n_iter % 5 == 0:
            val = 2.0 * n * fft_autocorr(a).max()
            if val < best_val - 1e-13:
                best_val = val
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        tau = min(tau * 1.002, tau_max)

        if stall > stall_limit // 2 and stall % 500 == 0:
            noise = rng.standard_normal(n) * 0.02 * best_a
            a_r = best_a + noise
            a_r = np.maximum(a_r, 1e-12)
            a_r = a_r / a_r.sum()
            a = a_r
            stall = 0
            tau = 50.0
            eta = 0.5

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print("    SGD t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(best_val) + " tau=" + repr(round(tau, 1)))

    return best_a, best_val

def lp_fw(a0, time_limit, solver, init_K=400, max_K=4000, add_per=40, verbose=True):
    """LP-based Frank-Wolfe with adaptive constraint growth and trisection line search."""
    n = len(a0)
    y = normalize(a0).copy()
    best_a = y.copy()
    best_m = fft_autocorr(y).max()
    t0 = time.monotonic()

    ac_y = fft_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:init_K])
    no_improve = 0
    stall_limit = 2000

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break

        ac = fft_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        clist = sorted(active_set)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = sorted(active_set)

        # Also add some new constraints near the peak
        if len(clist) < max_K:
            peak_region = np.where(ac > M_curr - 0.05 * M_curr)[0]
            for k in peak_region:

                    active_set.add(int(k))
            clist = sorted(active_set)
