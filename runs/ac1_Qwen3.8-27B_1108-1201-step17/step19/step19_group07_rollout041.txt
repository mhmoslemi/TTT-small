FFT_SIZE_FACTOR = 1  # next power of 2

def fast_conv(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))² using:
1. Multi-scale FISTA warm-start (when no good parent)
2. FISTA with adaptive softmax top-K gradient and Nesterov momentum
3. LP-based Frank-Wolfe for polishing
4. Final refinement with high-beta FISTA
Key improvements over previous: better multi-scale init, adaptive K schedule,
improved step size, more aggressive LP-FW phase, length exploration.

import numpy as np
import time
import cvxpy as cp


def fast_conv(a):
    """Full convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize_a(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_grad_c(a, beta, top_k):
    """Compute smoothed gradient and convolution via softmax-over-top-K."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]

    if top_k < size:
        top_idx = np.argpartition(-c, min(top_k, size) - 1)[:top_k]
        c_top = c[top_idx]
        c_max = c_top.max()
        exp_v = np.exp(beta * (c_top - c_max))
        w = np.zeros(size)
        w[top_idx] = exp_v
        w /= w.sum()
    else:
        c_max = c.max()
        w = np.exp(beta * (c - c_max))
        w /= w.sum()

    # grad[i] = 2 * sum_k w[k] * a[k-i] (with a out-of-range = 0)
    # = 2 * conv(w, a_rev)[n-1 : 2n-1]
    conv_len = size + n - 1
    conv_fft = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, conv_fft)
    Ar = np.fft.rfft(a[::-1], conv_fft)
    conv_w = np.fft.irfft(W * Ar, conv_fft)
    grad = 2.0 * conv_w[n - 1:2 * n - 1]

    return grad, c, w

            best_a = x.copy()

        # Find violated constraints and add them
        size = 2 * n - 1
        fft_sz = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_sz)
        Gf = np.fft.rfft(g_dir, fft_sz)
        conv_ys = np.fft.irfft(Yf * Gf, fft_sz)[:size]

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:min(30, len(violations))]]
            for k in top_v:
                constraint_set.add(int(k))

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        if verbose and t - last_print >= 10:
            last_print = t
            elapsed = time.monotonic() - t0
            print(f"    LP-FW it={t} ({elapsed:.1f}s): J={2*n*best_m:.6f}, |S|={len(constraint_set)}")

    return best_a, best_m


def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            xv = cp.Variable(20, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(xv)), [cp.sum(xv) == 1, xv <= 0.3])
            p.solve(solver=s, verbose=False)
            if p.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize_a(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    return normalize_a(f_new)


def parametric_candidates(n):
    """Generate diverse parametric initializations."""
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    cands = []

    # Half-sine
    cands.append(np.sin(np.pi * (x + 0.5) / (n + 1)) ** 2)
    # Raised cosine
    cands.append(0.5 * (1 + np.cos(2 * np.pi * x)))
    # Power laws
    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        cands.append((i + 1) ** p)
        cands.append((n - i) ** p)
    # Gaussian
    for sigma in [0.1, 0.2, 0.3, 0.5]:
        cands.append(np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2)))
    # Bimodal
    for sigma in [0.08, 0.12, 0.15, 0.2]:
        cands.append(np.exp(-((x - 0.25) ** 2) / (2 * sigma ** 2)) +
                     np.exp(-((x - 0.75) ** 2) / (2 * sigma ** 2)))
    # U-shaped
    for t in [0.5, 1.0, 2.0]:
        cands.append(1.0 / (1.0 + t * (x - 0.5) ** 2))
    # Parabolic
    cands.append(1.0 - (2 * x - 1) ** 2)
    cands.append((2 * x - 1) ** 2)

    return cands


def best_parametric(n):
    """Find best parametric initialization."""
    best_a = None
    best_val = float('inf')
    for cand in parametric_candidates(n):
        a = normalize_a(cand)
        c = fast_conv(a)
        v = c.max()
        if v < best_val:
            best_val = v
            best_a = a.copy()
    return best_a, best_val


def propose_candidate(budget_s=1800.0):
    """Main search function."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Solver: {solver}, budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        try:
            sc = float(evaluate_sequence([float(v) for v in a_n]))
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            el = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={el:.0f}s)")

    # Phase 0: Load parent or initialize
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 100 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize_a(p)
            ps = float(evaluate_sequence([float(v) for v in parent]))
            if np.isfinite(ps):
                best_seq = parent.copy()
                best_score = ps
                print(f"Parent loaded: n={len(parent)}, score={ps:.6f}")
    except Exception:
        pass

    if parent is None:
        # Multi-scale initialization
        print("No parent, using multi-scale init...")
        n_small = 300
        a_small, _ = best_parametric(n_small)
        check_update(a_small, "init_small")
        t_ms = time_budget * 0.05
        a_small, _ = fista_run(a_small, t_ms, beta_start=3, beta_end=30,
                               k_start=5, k_end=50, eta0=1.0, verbose=False)
        check_update(a_small, "fista_small")

        n_mid = 1500
        a_mid = interpolate_seq(a_small, n_mid)
        t_mid = time_budget * 0.08
        a_mid, _ = fista_run(a_mid, t_mid, beta_start=5, beta_end=50,
                             k_start=10, k_end=100, eta0=0.8, verbose=False)
        check_update(a_mid, "fista_mid")

        n_full = 5000
        best_seq = interpolate_seq(a_mid, n_full)
        best_score = float(evaluate_sequence([float(v) for v in best_seq]))
        print(f"Multi-scale init done: n={n_full}, score={best_score:.6f}")

    n_best = len(best_seq)
    print(f"\n=== Main FISTA (n={n_best}) ===")

    # Phase 1: Main FISTA optimization
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 600.0)
    a_in = normalize_a(best_seq)
    a_out, _ = fista_run(a_in, t1, beta_start=5.0, beta_end=100.0,
                         k_start=15, k_end=300, eta0=1.2, verbose=True)
    check_update(a_out, "fista_main")

    # Phase 2: Second FISTA with different params (higher beta)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_in = normalize_a(best_seq)
        a_out, _ = fista_run(a_in, t2, beta_start=20.0, beta_end=200.0,
                             k_start=30, k_end=500, eta0=0.6, verbose=True)
        check_update(a_out, "fista_r2")

    # Phase 3: LP-FW polishing
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 30 and n_best <= 8000:
        a_in = normalize_a(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, init_K=300, max_K=2000, verbose=True)
        check_update(a_out, "lp_polish")

    # Phase 4: FISTA after LP
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_in = normalize_a(best_seq)
        a_out, _ = fista_run(a_in, t4, beta_start=50.0, beta_end=400.0,
                             k_start=50, k_end=800, eta0=0.3, verbose=False)
        check_update(a_out, "fista_post_lp")

    # Phase 5: Final LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.20, 200.0)
    if t5 > 30 and n_best <= 8000:
        a_in = normalize_a(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, init_K=400, max_K=2500, verbose=True)
        check_update(a_out, "lp_final")

    # Phase 6: Try slightly different lengths if time permits
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_best = len(best_seq)
        for delta in [500, -500, 1000]:
            new_n = n_best + delta
            if 3000 <= new_n <= 8000 and remaining > 60:
                print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
                a_long = interpolate_seq(best_seq, new_n)
                t_probe = min(remaining * 0.3, 120.0)
                a_out, _ = fista_run(a_long, t_probe, beta_start=10, beta_end=100,
                                     k_start=20, k_end=200, eta0=0.8, verbose=False)
                check_update(a_out, f"n{new_n}")
                remaining = time_budget - (time.monotonic() - t_global)

    # Final validation
    if best_seq is not None:
        fs = float(evaluate_sequence([float(v) for v in best_seq]))
        if np.isfinite(fs):
            best_score = fs
        el = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={el:.0f}s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize_a(np.sin(np.pi * x) ** 2)
        best_score = float(evaluate_sequence([float(v) for v in best_seq]))
        print(f"Fallback: score={best_score:.6f}")

    return [float(v) for v in best_seq]
