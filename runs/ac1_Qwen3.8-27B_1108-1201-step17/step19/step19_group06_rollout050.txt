import numpy as np
import time


def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_grad_smoothed(a, ac, top_k, beta):
    """
    Compute weighted gradient using soft-max over top peaks.
    Returns gradient vector of shape (n,).
    """
    n = len(a)
    k_sel = min(top_k, 2 * n - 1)
    top_idx = np.argsort(ac)[::-1][:k_sel]
    top_vals = ac[top_idx]

    max_v = top_vals.max()
    expo = beta * (top_vals - max_v)
    expo = np.clip(expo, -80, 0)
    weights = np.exp(expo)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_sel) / k_sel
    else:
        weights = weights / w_sum

    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        if k_int < 0 or k_int >= 2 * n - 1:
            continue
        lo = max(0, k_int - n + 1)
        hi = min(k_int, n - 1)
        if lo > hi:
            continue
        i_arr = np.arange(lo, hi + 1)
        j_arr = k_int - i_arr
        # d(conv[k])/da[i] includes a[j] for each pair (i,j)
        grad[i_arr] += w * a[j_arr]
        neq = j_arr != i_arr
        if neq.any():
            grad[j_arr[neq]] += w * a[i_arr[neq]]
        eq = j_arr == i_arr
        if eq.any():
            grad[i_arr[eq]] += w * a[i_arr[eq]]

    return grad

def parametric_inits(n):
    """Generate various parametric initializations."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    cands = []

    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        cands.append(normalize((i_arr + 1) ** p))
        cands.append(normalize((n - i_arr) ** p))

    # Gaussians
    for c in [0.25, 0.35, 0.5, 0.65, 0.75]:

    cands.append(normalize(np.sin(np.pi * x) ** 4))
    cands.append(normalize(0.5 * (1 + np.cos(2 * np.pi * x))))
    cands.append(normalize(0.5 * (1 + np.cos(2 * np.pi * x)) ** 2))

    cands.append(normalize(0.42 + 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)))

    # Exponential
    for r in [0.5, 1, 2, 3]:
        cands.append(normalize(np.exp(-r * x)))

    cands.append(normalize(1 - np.abs(2 * x - 1)))

def interpolate_seq(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n + 1)
    f_padded = np.concatenate([f_norm, [0.0]])
    x_new = np.linspace(0.0, 1.0, new_n + 1)
    f_new = np.interp(x_new, x_old, f_padded)[:-1]
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / ss

                x = np.linspace(0, 1, n)
                d = float(rng.uniform(0.15, 0.35))
                sig = float(rng.uniform(0.1, 0.25))
                pert = np.exp(-((x - 0.5 + d) ** 2) / (2 * sig ** 2)) + \
                       np.exp(-((x - 0.5 - d) ** 2) / (2 * sig ** 2))
            elif pert_type == 3:
                i_arr = np.arange(n, dtype=np.float64)
                p = float(rng.uniform(0.3, 2.5))
                pert = (i_arr + 1) ** p
                pert = pert + (n - i_arr) ** p  # symmetric power law
            elif pert_type == 4:
                # Add noise to best
                pert = best_a * (1.0 + rng.normal(0, 0.08, n))
            else:
                # Random sparse perturbation
                pert = best_a.copy()
                n_sparse = n // 10
                idx = rng.choice(n, n_sparse, replace=False)
                pert[idx] *= rng.uniform(0.5, 2.0, n_sparse)

            pert = normalize(pert)
            w = float(rng.uniform(0.15, 0.5))
            a = normalize((1 - w) * best_a + w * pert)
            grad_ema *= 0.5
            lr = max(lr * 0.9, 0.01)
            beta = max(beta - 1.0, 1.5)
            temperature = max(temperature * 0.8, 0.005)

            if verbose and t % 10000 == 0:
                elapsed = time.monotonic() - t0
                print(f"      restart t={t} ({elapsed:.0f}s): score={2*n*best_m:.6f}, "
                      f"top_k={top_k}, beta={beta:.1f}, lr={lr:.4f}")

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Main opt done ({elapsed:.0f}s): score={2.0*n*best_m:.6f}")
    return best_a, best_score


def coord_descent_fine(a0, time_budget, rng_seed=99, verbose=True):
    """
    Fine coordinate descent: for each top peak, try moving small amounts
    of mass to reduce it.
    """
    n = len(a0)
    a = normalize(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_budget:
        ac = fast_autocorr(a)
        m_curr = ac.max()

        if m_curr < best_m - 1e-14:
            best_m = m_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30000:
                break

        # Get top lags
        n_top = min(20, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k_pos in top_lags[:10]:
            if time.monotonic() - t0 > time_budget:
                break
            k_int = int(k_pos)
            lo = max(0, k_int - n + 1)
            hi = min(k_int, n - 1)
            if lo > hi:
                continue
            i_arr = np.arange(lo, hi + 1)
            j_arr = k_int - i_arr
            contrib = a[i_arr] * a[j_arr]
            n_pairs = min(5, len(i_arr))
            pair_order = np.argsort(contrib)[::-1][:n_pairs]

            for pi in pair_order:
                if time.monotonic() - t0 > time_budget:
                    break
                src = int(i_arr[pi])
                src_val = a[src]
                if src_val < 1e-14:
                    continue

                # Find best destination: minimize impact on top peaks
                scores = np.zeros(n)
                for k2 in top_lags[:6]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(k2i, n - 1)
                    if s2 <= e2:
                        ii = np.arange(s2, e2 + 1)
                        jj = k2i - ii
                        scores[ii] += a[jj]
                        if (jj != ii).any():
                            eq = jj == ii
                            scores[jj[~eq]] += a[ii[~eq]]
                            scores[jj[eq]] += a[ii[eq]]

                scores[src] = 1e10
                dst = int(np.argmin(scores))
                if dst == src:
                    continue

                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and stall % 2000 == 0:
            # Random perturbation
            n_moves = int(rng.integers(3, 10))
            for _ in range(n_moves):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = float(rng.uniform(0.005, 0.02))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = normalize(a)
            m_new = fast_autocorr(a).max()
            if m_new < best_m:
                best_m = m_new
                best_a = a.copy()
                stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Coord descent done ({elapsed:.0f}s): score={2.0*n*best_m:.6f}")
    return best_a, 2.0 * n * best_m


def evaluate_safe(a):
    """Safely call the evaluation function."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with multi-phase optimization.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)

    best_seq = None
    best_score = float('inf')

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = evaluate_safe(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")

    # Load parent sequence
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 4000

    # === Phase 1: Try multiple initializations ===
    print(f"\n=== Phase 1: Initialization (t={time.monotonic()-t_global:.0f}s) ===")
    init_time = 30.0
    elapsed = time.monotonic() - t_global

    # Try parametric inits at the base length
    t1_start = time.monotonic()
    for n_try in [n_base]:
        if time.monotonic() - t1_start > init_time:
            break
        cands = parametric_inits(n_try)
        for c in cands:
            if time.monotonic() - t1_start > init_time:
                break
            sc = score_from_ac(c)
            if sc < best_score:
                best_score = sc
                best_seq = c.copy()
                elapsed = time.monotonic() - t_global
                print(f"  Better init: n={n_try}, score={sc:.6f} (t={elapsed:.0f}s)")

    # Also try a couple other lengths
    for n_alt in [n_base // 2, n_base + 500]:
        if n_alt < 500 or n_alt > 10000:
            continue
        if time.monotonic() - t1_start > init_time:
            break
        best_cand = None
        best_cand_sc = float('inf')
        cands = parametric_inits(n_alt)
        for c in cands:
            if time.monotonic() - t1_start > init_time:
                break
            sc = score_from_ac(c)
            if sc < best_cand_sc:
                best_cand_sc = sc
                best_cand = c
        if best_cand is not None and best_cand_sc < best_score:
            best_score = best_cand_sc
            best_seq = best_cand.copy()
            print(f"  Better init at n={n_alt}: score={best_cand_sc:.6f}")

    # === Phase 2: Main optimization on best sequence ===
    n_now = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Main optimization (n={n_now}, budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.45, 700.0)
    if t2 > 30:
        a_out, sc_out = main_optimize(best_seq, t2, rng_seed=42, verbose=True)
        check_update(a_out, f"main_n{n_now}")

    # === Phase 3: Second optimization pass with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_now = len(best_seq)
    print(f"\n=== Phase 3: Second pass (n={n_now}, budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30:
        # Start from best with slight perturbation
        rng3 = np.random.default_rng(123)
        a_pert = normalize(best_seq * (1.0 + rng3.normal(0, 0.02, n_now)))
        a_out, sc_out = main_optimize(a_pert, t3, rng_seed=777, verbose=True)
        check_update(a_out, f"pass2_n{n_now}")

    # === Phase 4: Coordinate descent fine-tuning ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_now = len(best_seq)
    print(f"\n=== Phase 4: Coord descent (n={n_now}, budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_out, sc_out = coord_descent_fine(best_seq, t4, rng_seed=55, verbose=True)
        check_update(a_out, f"coord_n{n_now}")

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_now = len(best_seq)
    print(f"\n=== Phase 5: Length exploration (budget={remaining:.0f}s) ===")
    length_tries = []
    for delta in [-1000, -500, -200, 200, 500, 1000, 2000]:
        nn = n_now + delta
        if 1000 <= nn <= 12000 and nn != n_now:
            length_tries.append(nn)
    for rn in [2000, 3000, 4000, 5000, 6000, 7000, 8000]:
        if rn != n_now and 1000 <= rn <= 12000:
            length_tries.append(rn)
    length_tries = sorted(set(length_tries))[:6]

    for new_n in length_tries:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_interp = interpolate_seq(best_seq, new_n)
        # Quick check
        sc_interp = score_from_ac(a_interp)
        if sc_interp < best_score:
            best_score = sc_interp
            best_seq = a_interp.copy()
            print(f"    Interpolated improves: {sc_interp:.6f}")

        t5a = min(remaining * 0.3, 100.0)
        if t5a > 15:
            a_out, sc_out = main_optimize(a_interp, t5a, rng_seed=new_n % 10000, verbose=False)
            check_update(a_out, f"len_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t5b = min(remaining * 0.2, 60.0)
            if t5b > 10:
                a_out, sc_out = coord_descent_fine(best_seq, t5b, rng_seed=new_n % 5000, verbose=False)
                check_update(a_out, f"coord_len_n{new_n}")

    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_now = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_now}, budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.5, 150.0)
    if t6 > 15:
        a_out, sc_out = main_optimize(best_seq, t6, rng_seed=31415, verbose=False)
        check_update(a_out, f"final_n{n_now}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t7 = min(remaining * 0.6, 60.0)
        if t7 > 8:
            a_out, sc_out = coord_descent_fine(best_seq, t7, rng_seed=271, verbose=False)
            check_update(a_out, f"final_coord_n{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")

    # Ensure all values are valid
    result = [float(max(0, min(1000, x))) for x in best_seq]
    return result
