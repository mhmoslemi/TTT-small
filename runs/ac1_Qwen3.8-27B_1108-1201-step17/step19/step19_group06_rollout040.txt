    # Need linear conv of p (len 2n-1) with a (len n): result length 3n-2
    L = 1 << (3 * n - 2 - 1).bit_length()

    P = np.fft.rfft(p, L)
    A = np.fft.rfft(a, L)

    # conv[p,a][i] = sum_k p[k]*a[i-k] for i=0..n-1
    conv_pa = np.fft.irfft(P * A, L)[:n]

    # xcor[p,a][i] = sum_k p[k]*a[k-i] = conv[flip(p),a][p_len-1-i]
    p_rev = p[::-1]
    P_rev = np.fft.rfft(p_rev, L)
    conv_prev_a = np.fft.irfft(P_rev * A, L)
    # xcor[i] = conv_prev_a[p_len-1-i] for i=0..n-1
    # Indices: p_len-1, p_len-2, ..., p_len-n
    xcor = conv_prev_a[p_len - n : p_len][::-1]

    g = 2.0 * a * (xcor + conv_pa)
    return g


def lse_adam_optimize(a0, time_limit, beta_start=8.0, beta_end=80.0,
                      lr_init=0.05, rng_seed=42, verbose=True):
    """
    LSE-Adam optimizer in log-space.
    Minimizes the log-sum-exp of autocorrelation values (smooth approximation of max).
    """
    n = len(a0)
    log_a = np.log(np.maximum(a0, 1e-30))
    best_a = norm_sum1(a0)
    best_score = score_normalized(best_a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    lr = lr_init
    stall = 0
    best_beta = beta_start

    total_iters = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        total_iters += 1

        # Adaptive beta (annealing)
        frac = min(1.0, t / 5000.0)
        beta = beta_start + (beta_end - beta_start) * frac

        # Compute gradient
        a = np.exp(log_a - log_a.max())
        a = a / a.sum()
        g = compute_log_grad(a, beta)

        # Adam update
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        lr_t = lr / (1.0 + 0.005 * np.sqrt(t))
        log_a = log_a - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        # Convert back and normalize
        a_new = np.exp(log_a - log_a.max())
        a_new = a_new / a_new.sum()

        sc = score_normalized(a_new)
        if sc < best_score - 1e-11:
            best_score = sc
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            # Diverse restart
            pert_type = int(rng.integers(6))
            if pert_type == 0:
                x = np.linspace(0, 1, n)
                c = float(rng.uniform(0.2, 0.8))
                sig = float(rng.uniform(0.05, 0.3))
                pert = np.exp(-((x - c) ** 2) / (2 * sig ** 2))
            elif pert_type == 1:
                i_arr = np.arange(n, dtype=np.float64)
                p = float(rng.uniform(0.3, 3.0))
                pert = (i_arr + 1.0) ** p
            elif pert_type == 2:
                x = np.linspace(0, 1, n)
                f = int(rng.integers(1, 6))
                phase = float(rng.uniform(0, np.pi))
                pert = np.sin(np.pi * f * x + phase) ** 2
            elif pert_type == 3:
                x = np.linspace(0, 1, n)
                sig = float(rng.uniform(0.08, 0.25))
                pert = (np.exp(-((x - 0.2) ** 2) / (2 * sig ** 2)) +
                        np.exp(-((x - 0.8) ** 2) / (2 * sig ** 2)))
            elif pert_type == 4:
                # Random smooth perturbation via spectral method
                freq = np.fft.rfftfreq(n)
                spec = np.exp(-freq * n * float(rng.uniform(0.5, 2.0)))
                spec[0] = 0  # Remove DC
                pert = np.fft.irfft(spec * rng.normal(0, 1, len(spec)), n)
                pert = np.abs(pert)
            else:
                # Small multiplicative perturbation of best
                pert = best_a * np.exp(rng.normal(0, 0.15, n))

            pert = norm_sum1(pert)
            w = float(rng.uniform(0.05, 0.35))
            a_mix = norm_sum1((1 - w) * best_a + w * pert)
            log_a = np.log(np.maximum(a_mix, 1e-30))
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            stall = 0
            lr = max(lr * 0.85, 0.005)
            beta = max(beta_start, beta - 5)

            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LSE-Adam restart ({elapsed:.0f}s): score={best_score:.6f}, beta={beta:.1f}, lr={lr:.4f}, type={pert_type}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam t={t} ({elapsed:.0f}s): best={best_score:.6f}, beta={beta:.1f}, stall={stall}")

    if verbose:
        print(f"    LSE-Adam done ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")
    return best_a, best_score


def coord_descent_polish(a0, time_limit, rng_seed=77, verbose=False):
    """
    Coordinate descent: for each top autocorrelation lag,
    try small transfers to reduce the peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = score_normalized(best_a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_conv(a)
        M_curr = c.max()
        sc_curr = 2.0 * n * M_curr

        if sc_curr < best_score - 1e-12:
            best_score = sc_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break

        # Get top lags
        n_top = min(20, len(c))
        top_lags = np.argsort(c)[::-1][:n_top]

        improved = False
        for k_pos in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k_pos)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo >= i_hi:
                continue

            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            valid = (j_idx >= 0) & (j_idx < n)
            idx_v = idx[valid]
            contrib = a[idx_v] * a[j_idx[valid]]

            # Find the pair with max contribution
            if len(contrib) == 0:
                continue
            pi = int(np.argmax(contrib))
            src = int(idx_v[pi])
            src_val = a[src]
            if src_val < 1e-14:
                continue

            # Try moving a small fraction to a random other position
            # that has low "impact" on the top lags
            n_tries = 5
            for _ in range(n_tries):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_conv(a_trial).max()
                    sc_trial = 2.0 * n * m_trial
                    if sc_trial < best_score - 1e-13:
                        a = a_trial
                        best_score = sc_trial
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved and stall % 1000 == 0:
            # Random perturbation to escape
            n_moves = int(rng.integers(3, 15))
            for _ in range(n_moves):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = float(rng.uniform(0.005, 0.05))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    if verbose:
        print(f"    Coord descent done ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def parametric_init(n, rng_seed=0):
    """Generate diverse parametric initializations and return the best."""
    rng = np.random.default_rng(rng_seed)
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)

    candidates = []
    # Uniform
    candidates.append(np.ones(n) / n)
    # Power laws
    for p in [0.5, 1.0, 1.5, 2.0]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))
    # Gaussians at various positions
    for c in [0.3, 0.5, 0.7]:
        for sig in [0.15, 0.25, 0.4]:
            candidates.append(norm_sum1(np.exp(-((x - c) ** 2) / (2 * sig ** 2))))
    # Bimodal
    for sig in [0.12, 0.2, 0.3]:
        candidates.append(norm_sum1(
            np.exp(-((x - 0.25) ** 2) / (2 * sig ** 2)) +
            np.exp(-((x - 0.75) ** 2) / (2 * sig ** 2))))
    # Cosh (U-shaped)
    for width in [1.0, 2.0, 4.0]:
        candidates.append(norm_sum1(np.cosh(width * (x - 0.5))))
    # Sine
    for f in [1, 2, 3]:
        candidates.append(norm_sum1(np.sin(np.pi * f * x) ** 2))
    # Triangle
    candidates.append(norm_sum1(1.0 - np.abs(2.0 * x - 1.0)))
    # Random smooth
    for _ in range(5):
        freq = rng.integers(1, 4)
        phase = float(rng.uniform(0, 2 * np.pi))
        amp = float(rng.uniform(0.5, 2.0))
        sig = float(rng.uniform(0.3, 1.0))
        candidates.append(norm_sum1(np.exp(amp * np.sin(freq * 2 * np.pi * x + phase) * sig / (x + 0.1))))

    best_c = None
    best_v = float('inf')
    for c in candidates:
        v = score_normalized(c)
        if v < best_v:
            best_v = v
            best_c = c.copy()
    return best_c if best_c is not None else np.ones(n) / n


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with multi-phase optimization.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 40.0, 1760.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = score_normalized(parent)
            # Verify with actual evaluator
            p_list = [float(x) for x in parent]
            eval_score = float(evaluate_sequence(p_list))
            if np.isfinite(eval_score):
                best_seq = parent.copy()
                best_score = eval_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={best_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if best_seq is None:
        n0 = 6000
        best_seq = parametric_init(n0)
        best_score = score_normalized(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = score_normalized(a_n)
        if np.isfinite(sc) and sc < best_score:
            # Verify with actual evaluator
            eval_sc = float(evaluate_sequence([float(x) for x in a_n]))
            if np.isfinite(eval_sc) and eval_sc < best_score:
                best_score = eval_sc
                best_seq = a_n.copy()
                elapsed = time.monotonic() - t_global
                print(f"  *** NEW BEST [{label}]: {best_score:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")
            elif np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_n.copy()
                elapsed = time.monotonic() - t_global
                print(f"  *** NEW BEST [{label}]: {best_score:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")

    # === Phase 1: LSE-Adam on parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LSE-Adam (n={n0}, budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 350.0)
    if t1 > 30:
        a_out, _ = lse_adam_optimize(best_seq, t1, beta_start=8.0, beta_end=60.0,
                                      lr_init=0.04, rng_seed=42, verbose=True)
        check_and_update(a_out, f"lse1_n{n0}")

    # === Phase 2: Second LSE-Adam pass with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LSE-Adam fine (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 280.0)
    if t2 > 20:
        a_out, _ = lse_adam_optimize(best_seq, t2, beta_start=30.0, beta_end=120.0,
                                      lr_init=0.02, rng_seed=123, verbose=False)
        check_and_update(a_out, f"lse2_n{len(best_seq)}")

    # === Phase 3: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coord descent (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 15:
        a_out, _ = coord_descent_polish(best_seq, t3, rng_seed=55, verbose=True)
        check_and_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 4: Another LSE pass with very high beta (near-max) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: High-beta LSE (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_out, _ = lse_adam_optimize(best_seq, t4, beta_start=50.0, beta_end=200.0,
                                      lr_init=0.015, rng_seed=200, verbose=False)
        check_and_update(a_out, f"lse_high_n{len(best_seq)}")

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration (budget={remaining:.0f}s) ===")

    length_tries = []
    for delta in [-1000, -500, 500, 1000, 2000, 3000]:
        nn = n_best + delta
        if 2000 <= nn <= 15000 and nn != n_best:
            length_tries.append(nn)
    for rn in [3000, 5000, 8000, 10000, 12000]:
        if rn != n_best and 2000 <= rn <= 15000:
            length_tries.append(rn)
    length_tries = sorted(set(length_tries))

    n_explored = 0
    for new_n in length_tries[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        n_explored += 1
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")

        a_interp = interpolate_sequence(best_seq, new_n)
        # Quick LSE optimization
        t5a = min(remaining * 0.4, 150.0)
        if t5a > 20:
            a_out, _ = lse_adam_optimize(a_interp, t5a, beta_start=10.0, beta_end=80.0,
                                          lr_init=0.04, rng_seed=new_n % 500, verbose=False)
            check_and_update(a_out, f"lse_len{new_n}")

        # Coord descent on new length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t5b = min(remaining * 0.3, 100.0)
            if t5b > 15:
                a_out, _ = coord_descent_polish(best_seq, t5b, rng_seed=new_n % 1000, verbose=False)
                check_and_update(a_out, f"coord_len{new_n}")

    # === Phase 6: Final polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_best}, budget={remaining:.0f}s) ===")

    # LSE with very high beta
    t6 = min(remaining * 0.35, 150.0)
    if t6 > 15:
        a_out, _ = lse_adam_optimize(best_seq, t6, beta_start=80.0, beta_end=300.0,
                                      lr_init=0.01, rng_seed=999, verbose=False)
        check_and_update(a_out, f"final_lse_n{n_best}")

    # Final coord descent
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t7 = min(remaining * 0.5, 100.0)
        if t7 > 10:
            a_out, _ = coord_descent_polish(best_seq, t7, rng_seed=42, verbose=True)
            check_and_update(a_out, f"final_coord_n{len(best_seq)}")

    # === Phase 7: Random seed exploration (try a few fresh starts) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 7: Fresh starts (budget={remaining:.0f}s) ===")
        n_fresh = min(8000, max(3000, n_best))
        t_fresh = min(remaining * 0.5, 200.0)
        per_seed = t_fresh / 3
        for seed in [7, 13, 99]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            a_init = parametric_init(n_fresh, rng_seed=seed * 100)
            # Blend with current best for warm start
            a_blend = norm_sum1(0.3 * interpolate_sequence(best_seq, n_fresh) +
                                0.7 * a_init)
            t_f = min(per_seed, remaining - 10)
            if t_f > 10:
                a_out, _ = lse_adam_optimize(a_blend, t_f, beta_start=10.0, beta_end=100.0,
                                              lr_init=0.04, rng_seed=seed, verbose=False)
                check_and_update(a_out, f"fresh_{seed}_n{n_fresh}")

    # Final validation
    if best_seq is not None:
        fs = score_normalized(best_seq)
        # Cross-check with evaluator
        try:
            eval_fs = float(evaluate_sequence([float(x) for x in best_seq]))
            if np.isfinite(eval_fs):
                fs = eval_fs
        except Exception:
            pass
        best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = np.ones(n_fb) / n_fb
        best_score = score_normalized(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")

    return [float(x) for x in best_seq]
