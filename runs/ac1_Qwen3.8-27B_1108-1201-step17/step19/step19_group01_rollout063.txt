Actually, let me think about this problem from the perspective of the DFT. The convolution theorem says:
DFT(a * a) = DFT(a) * DFT(a) = |A|^2 (for real a, this is |A|^2 since DFT(a*a) = A^2, and for real a, A^2 is not the same as |A|^2 in general... wait, no.

Actually, for real sequences: DFT(a * a) = DFT(a) · DFT(a) = A · A = A^2 (element-wise product).
And |A|^2 = A · conjugate(A).

So DFT(a * a) = A^2, not |A|^2. This is different from the autocorrelation!

The autocorrelation would be: DFT(autocorr(a)) = |A|^2.
But the convolution is: DFT(conv(a,a)) = A^2.

So the problem is about making A^2 (the square of the DFT, not the power spectrum) have a small maximum in the time domain.

By Parseval: sum_k |c[k]|^2 = (1/N) * sum_f |A(f)^2|^2 = (1/N) * sum_f |A(f)|^4

And the maximum of c is bounded by the L2 norm: max|c| <= ||c||_2 = sqrt((1/N) * sum_f |A(f)|^4)

So to minimize the peak of c, we want to minimize sum_f |A(f)|^4, which means making |A(f)| as uniform as possible (i.e., making |A(f)|^2 flat, which is the power spectrum flat).

But we also have the constraint that A(0) = sum(a) = 1 (fixed). So we want to minimize sum_f |A(f)|^4 subject to A(0) = 1 and a >= 0.

Without the a >= 0 constraint, the minimum of sum |A(f)|^4 subject to A(0) = 1 would be achieved by spreading the energy uniformly: |A(f)|^2 = 1/N for all f, giving sum |A(f)|^4 = N * (1/N)^2 = 1/N. Then ||c||_2 = sqrt(1/N^2) = 1/N, and max|c| <= 1/N.

But with a >= 0, we can't achieve this. The constraint a >= 0 in the time domain translates to a constraint on the DFT (it's not a simple pointwise constraint).

This gives us a useful upper bound: max|c| >= ||c||_2 / sqrt(2n-1) (by the relationship between L-infinity and L2 norms for sequences of length 2n-1).

Actually, ||c||_inf >= ||c||_2 / sqrt(2n-1).
||c||_2^2 = (1/N) * sum_f |A(f)|^4 (where N is the FFT size, which should be >= 2n-1)

||c||_2^2 = (1/(2n-1)) * sum_{f=0}^{2n-2} |A(f)|^4

import numpy as np
import time

def fast_convolve(a):
    """Compute full convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, p):
    """Compute sum_j a[j]*p[i+j] for i=0..n-1 using FFT. p has length 2n-1."""
    n = len(a)
    size_p = len(p)
    a_flip = a[::-1]
    conv_len = n + size_p - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Af = np.fft.rfft(a_flip, fft_size)
    Pf = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Af * Pf, fft_size)
    return conv[n - 1: 2 * n - 1]


def simplex_project(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_lse(a, beta, top_k=None):
    """
    Compute gradient of (1/beta)*log(sum_{top_k} exp(beta*c)) w.r.t. a.
    Returns (grad, p, c).
    """
    n = len(a)
    c = fast_convolve(a)
    size = len(c)
    if top_k is not None and top_k < size:
        idx = np.argpartition(c, -top_k)[-top_k:]
        mask = np.zeros(size, dtype=bool)
        mask[idx] = True
    else:
        mask = np.ones(size, dtype=bool)

    c_max = c[mask].max()
    shifted = np.full(size, -300.0)
    shifted[mask] = beta * (c[mask] - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    exp_vals[~mask] = 0.0
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros(size)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total

    grad = (2.0 / beta) * fast_cross_convolve(a, p)
    return grad, p, c


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def nesterov_v2(a0, time_limit, beta_start=10.0, beta_end=800.0,
                beta_growth=1.003, top_k_start=None, top_k_end=None,
                restart_every=2500, verbose=True):

    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_convolve(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(12345)

    if top_k_start is None:
        top_k_start = max(50, int(n * 0.1))

        top_k_end = max(5, int(n * 0.005))

    total_iters_est = 30000
    step_counter = 0

    while time.monotonic() - t0 < time_limit:
        step_counter += 1

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Adaptive top-K: decrease K as we progress
        progress = min(1.0, step_counter / total_iters_est)
        cur_k = int(top_k_start + (top_k_end - top_k_start) * progress)
        cur_k = max(5, cur_k)

        # Compute gradient

        # Backtracking line search
        g_norm = np.linalg.norm(grad)

            eta = 0.01
        else:

            L_est = 4.0 * beta * np.max(p) * 2.0 * np.max(y)
            eta = 1.0 / max(L_est, 1e-10)

        found_improve = False

            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new

def peak_shaving(a0, time_limit, verbose=True):

        peak_val = c[peak_k]

        # Find contributing coordinates for this peak
        js = max(0, peak_k - n + 1)

        if js > je:
            stall += 1
            continue
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()
        if total_contrib < 1e-20:
            stall += 1
            continue

        else:

            stall += 1

            a_trial = a_trial / s_t
            m_trial = fast_convolve(a_trial).max()
            if m_trial < best_m - 1e-15:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                stall = 0
                found = True
                break

        if not found:
            stall += 1
            # Random exploration when stuck
            if stall % 50 == 0:
                for _ in range(3):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            m_new = fast_convolve(a).max()
                            if m_new < best_m:
                                best_m = m_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakShave (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def spectral_shaping(a0, time_limit, verbose=True):
    """
    Spectral domain optimization: try to flatten |A(f)|^2 for f != 0.
    Uses proximal gradient in the DFT domain with non-negativity projection.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(314)

    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    eta = 0.1
    stall = 0

    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 3000:
            break

        # Compute DFT
        A = np.fft.rfft(a, fft_size)
        # A[0] is the DC component (sum = 1)
        # Objective: minimize sum_{f!=0} |A(f)|^4 (proxy for peak of convolution)
        # Gradient w.r.t. A: 4 * |A(f)|^2 * A(f) for f != 0

        # But we need gradient w.r.t. a, so:
        # grad_a = IDFT(4 * |A|^2 * A * mask) where mask[0] = 0
        A_mag_sq = np.abs(A) ** 2
        mask = np.ones(len(A))
        mask[0] = 0.0
        grad_A = 4.0 * A_mag_sq * A * mask
        grad_a = np.real(np.fft.irfft(grad_A, fft_size))[:n]

        # Also add a term to keep A[0] fixed (enforce sum=1)
        # The gradient step shouldn't change the mean too much
        grad_a -= grad_a.mean()

        # Gradient step with projection
        a_new = a - eta * grad_a
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        m_new = fast_convolve(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        eta *= 0.999

        if stall > 500 and stall % 500 == 0:
            eta = 0.05
            pert = rng.normal(0, 0.02, n)
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            a = a_r
            stall = 0

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def lp_fw(a0, time_limit, verbose=True):
    """
    LP-Frank-Wolfe with adaptive active set.
    """
    n = len(a0)
    if n > 8000:
        return a0, fast_convolve(a0).max()

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_convolve(y)
    initial_K = min(300, 2 * n - 1)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    max_K = min(2500, 2 * n - 1)
    add_per_iter = 20
    no_improve = 0
    max_no_improve = 800

    solver = None
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: converged at iter " + str(t))
            break

        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix: (y * g)[k] <= M for k in active_set
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # Solve LP: minimize M subject to y*g <= M (for active), g>=0, sum(g)=1
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations to add
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf)[:size]

        violations = np.where(conv_ys > M_curr + 1e-6)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.01, default_alpha * 0.5),
                  max(0.01, default_alpha * 2.0), 1.0 / (t + 1)]
        alphas = [max(0.01, min(1.0, a)) for a in alphas]
        alphas = list(set(alphas))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def parametric_init(n, time_limit, best_init, best_score):
    """Try various parametric sequences for good initialization."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score

    candidates = []
    # Power laws (asymmetric)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    # Skewed Gaussians
    for c in [0.3, 0.4, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.3]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    # Modulated
    for freq in [1, 2, 3]:
        for amp in [0.3, 0.5, 0.7]:
            candidates.append(np.maximum(1.0 + amp * np.sin(2.0 * np.pi * freq * x), 0.01))
    # Sine powers
    for p in [1, 2, 3, 4, 6]:
        candidates.append(np.sin(np.pi * x) ** p)
    # Combination: ramp * sine
    for p in [0.5, 1.0, 1.5, 2.0]:
        candidates.append(((i_arr + 1.0) / n) ** p * np.sin(np.pi * x))
        candidates.append((1.0 - i_arr / n) ** p * np.sin(np.pi * x))

    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a_raw = np.maximum(a_raw, 0.0)
        if a_raw.sum() < 1e-10:
            continue
        a = a_raw / a_raw.sum()
        m = fast_convolve(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    # Random
    rng = np.random.default_rng(42)
    for _ in range(80):
        if time.monotonic() - t0 > time_limit:
            break
        # Random smooth sequences
        freqs = rng.integers(1, 10, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amps = rng.random(5) ** 2
        a_raw = np.zeros(n)
        for f, ph, am in zip(freqs, phases, amps):
            a_raw += am * np.cos(2 * np.pi * f * x + ph)
        a_raw += 1.0  # Ensure positive
        a_raw = np.maximum(a_raw, 0.0)
        if a_raw.sum() < 1e-10:
            continue
        a = a_raw / a_raw.sum()
        m = fast_convolve(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    return best_a, best_val


def try_update(best_seq, best_score, a, label, t_global):
    """Evaluate and update best if improved."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        sc = evaluate_sequence([float(x) for x in a_norm])
    except Exception:
        return best_seq, best_score
    if np.isfinite(sc) and sc < best_score:
        best_score = float(sc)
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(best_score, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def coordinate_polish(a0, time_limit, verbose=True):
    """
    Adaptive coordinate descent: try moving mass from high-contributing
    positions to low-impact positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    max_stall = 1500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_convolve(a)
        n_top = min(30, len(c))
        top_k = np.argsort(c)[::-1][:n_top]
        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            src_val = a[src]
            partner_val = a[partner] if 0 <= partner < n else 0.0

            if src_val >= partner_val:
                reduce_idx = src
                keep_idx = partner
            else:
                reduce_idx = partner
                keep_idx = src

            if a[reduce_idx] < 1e-12:
                continue

            # Compute pressure for destination selection
            pressure = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)

            valid_cands = candidates[candidates != reduce_idx]
            if len(valid_cands) == 0:
                continue

            cand_press = pressure[valid_cands]
            # Pick a few candidates with lowest pressure
            n_cand = min(5, len(valid_cands))
            top_cands = valid_cands[np.argsort(cand_press)[:n_cand]]

            found = False
            for dst in top_cands:
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    amt = a[reduce_idx] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_idx] -= amt
                    a_trial[int(dst)] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_convolve(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                if found:
                    break

            if not found:
                break

        if not improved:
            stall += 1
            if stall % 100 == 0:
                # Broader random moves
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.04)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search function. Combines multiple optimization strategies
    with multi-length exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_sequence([float(x) for x in parent])
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = float(p_score)
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(round(best_score, 6)))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_sequence([float(v) for v in best_seq])
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(round(best_score, 6)))

    n0 = len(best_seq)

    # Phase 1: Parametric initialization
    print("\n=== Phase 1: Parametric (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 40.0)
    if t1 > 10:
        a_param, _ = parametric_init(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "param", t_global)

    # Phase 2: Nesterov v2 (main optimizer, first pass)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov V2 (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 60:
        a_out, _ = nesterov_v2(norm_sum1(best_seq), t2,
                               beta_start=10.0, beta_end=600.0,
                               beta_growth=1.003,
                               top_k_start=max(50, int(n_best * 0.08)),
                               top_k_end=max(10, int(n_best * 0.003)),
                               restart_every=2500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2_n" + str(n_best), t_global)

    # Phase 3: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Peak Shaving (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 30:
        a_out, _ = peak_shaving(norm_sum1(best_seq), t3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_n" + str(n_best), t_global)

    # Phase 4: LP-Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.18, 280.0)
    if t4 > 40 and n_best <= 8000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t4, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(n_best), t_global)

    # Phase 5: Nesterov v2 (second pass, higher beta)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Nesterov V2 pass2 (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.22, 350.0)
    if t5 > 60:
        a_out, _ = nesterov_v2(norm_sum1(best_seq), t5,
                               beta_start=50.0, beta_end=1500.0,
                               beta_growth=1.005,
                               top_k_start=max(20, int(n_best * 0.03)),
                               top_k_end=max(5, int(n_best * 0.002)),
                               restart_every=2000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2b_n" + str(n_best), t_global)

    # Phase 6: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Coordinate (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 20:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t6, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n" + str(n_best), t_global)

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Multi-length ===")

    lengths_to_try = []
    for delta in [500, 1000, -500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 10000:
            lengths_to_try.append(new_n)

    for new_n in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("  Trying n=" + str(new_n) + "...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.35, 200.0)
        if t7a > 40:
            a_out, _ = nesterov_v2(a_long, t7a,
                                   beta_start=20.0, beta_end=800.0,
                                   beta_growth=1.004,
                                   top_k_start=max(30, int(new_n * 0.05)),
                                   top_k_end=max(5, int(new_n * 0.003)),
                                   restart_every=2000, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t7b = min(remaining * 0.40, 120.0)
            if t7b > 20:
                a_long2 = interpolate_sequence(best_seq, new_n)
                a_out, _ = coordinate_polish(a_long2, t7b, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n" + str(new_n), t_global)

    # Phase 8: Final Nesterov polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final Nesterov (n=" + str(n_best) + ") ===")
    if remaining > 60:
        t8 = min(remaining * 0.35, 250.0)
        if t8 > 30:
            a_out, _ = nesterov_v2(norm_sum1(best_seq), t8,
                                   beta_start=100.0, beta_end=2000.0,
                                   beta_growth=1.006,
                                   top_k_start=max(10, int(n_best * 0.02)),
                                   top_k_end=max(3, int(n_best * 0.001)),
                                   restart_every=1500, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

    # Phase 9: Final peak shaving + coordinate
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 9: Final polish (n=" + str(n_best) + ") ===")
    if remaining > 60:
        t9 = min(remaining * 0.40, 150.0)
        if t9 > 20:
            a_out, _ = peak_shaving(norm_sum1(best_seq), t9, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_final", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t10 = min(remaining * 0.50, 100.0)
        if t10 > 15:
            a_out, _ = coordinate_polish(norm_sum1(best_seq), t10, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None:
        a_final = np.maximum(best_seq, 0.0)
        s_final = a_final.sum()
        if s_final > 1e-10:
            a_final = a_final / s_final
        try:
            final_score = evaluate_sequence([float(x) for x in a_final])
            if np.isfinite(final_score):
                best_score = float(final_score)
                best_seq = a_final
            else:
                print("WARNING: final score is not finite, using previous best")
        except Exception:
            print("WARNING: final evaluation failed")

        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_sequence([float(v) for v in best_seq])

    # Ensure we return a valid list of floats
    result = [float(max(0.0, min(1000.0, x))) for x in np.asarray(best_seq, dtype=np.float64)]
    return result
