def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:

        y = np.maximum(y, 0.0)
        s_y = y.sum()

def fast_autocorr(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_grad(y, beta, top_k=None):
    """Compute gradient of LSE-smoothed max of conv(y,y) w.r.t. y."""
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and top_k < len(b):
        idx = np.argpartition(b, -top_k)[-top_k:]
        b_mask = np.zeros(len(b))
        b_mask[idx] = 1.0
        b_use = b * b_mask
    else:
        b_use = b

    b_max = b_use.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = b_use / max(b_use.sum(), 1e-300)
    else:
        p = exp_vals / total

    # grad_i = 2 * sum_k p_k * a_{k-i} (valid indices)
    a_rev = y[::-1].copy()
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1: 2 * n - 1] * 2.0 / beta
    return grad, p, b

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    # Random with structure
    rng = np.random.default_rng(42)
    for _ in range(80):
        if time.monotonic() - t0 > time_limit:
            break
        # Random smooth function
        freqs = rng.integers(1, 10, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amps = rng.uniform(0.1, 1.0, size=5)
        a_raw = np.ones(n)
        for f_, p_, a_ in zip(freqs, phases, amps):
            a_raw += a_ * np.cos(2 * np.pi * f_ * x + p_)
        a_raw = np.maximum(a_raw, 0.01)
        a = norm_sum1(a_raw)
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    return best_a, best_val


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.1f}s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search: multi-phase optimization with diverse methods and multi-length exploration."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    print(f"Time budget: {time_budget:.1f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # Phase 1: Parametric search
    print("\n=== Phase 1: Parametric (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 40.0)
    if t1 > 10:
        a_param, _ = parametric_init(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "param", t_global)

    # Phase 2: Nesterov with backtracking
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Nesterov BT (n={n_best}) ===")
    t2 = min(remaining * 0.25, 360.0)
    if t2 > 60:
        a_out, _ = nesterov_bt(norm_sum1(best_seq), t2,
                               beta_start=15.0, beta_end=800.0,
                               beta_growth=1.003, top_k_ratio=0.03,
                               restart_every=3000, verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"nest_{n_best}", t_global)

    # Phase 3: Pressure-based local search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Pressure LS (n={n_best}) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 30:
        a_out, _ = pressure_local_search(norm_sum1(best_seq), t3, verbose=True, seed=123)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"press_{n_best}", t_global)

    # Phase 4: Nesterov again (from Phase 3 result)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Nesterov BT round 2 (n={n_best}) ===")
    t4 = min(remaining * 0.15, 240.0)
    if t4 > 40:
        a_out, _ = nesterov_bt(norm_sum1(best_seq), t4,
                               beta_start=30.0, beta_end=1200.0,
                               beta_growth=1.004, top_k_ratio=0.02,
                               restart_every=2500, verbose=True, seed=200)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"nest2_{n_best}", t_global)

    # Phase 5: Pairwise optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Pairwise (n={n_best}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_out, _ = pairwise_optimize(norm_sum1(best_seq), t5, verbose=True, seed=77)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"pair_{n_best}", t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-length exploration ===")

    # Try nearby and different lengths
    length_options = []
    for delta in [-500, -200, 200, 500, 1000, 1500, 2000]:
        ln = n_best + delta
        if 2000 <= ln <= 12000 and ln != n_best:
            length_options.append(ln)
    # Also try some completely different lengths
    for ln in [3000, 4000, 6000, 8000, 10000]:
        if ln not in length_options and abs(ln - n_best) > 1000:
            length_options.append(ln)

    length_options = sorted(set(length_options))[:6]  # Limit to 6 lengths

    for new_n in length_options:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n  --- Length {new_n} ---")
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick Nesterov
        t6a = min(remaining * 0.25, 150.0)
        if t6a > 30:
            a_out, _ = nesterov_bt(a_long, t6a,
                                   beta_start=20.0, beta_end=600.0,
                                   beta_growth=1.004, top_k_ratio=0.03,
                                   restart_every=2000, verbose=False, seed=new_n)
            best_seq, best_score = try_update(best_seq, best_score, a_out, f"nest_{new_n}", t_global)

        # Quick pressure search
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t6b = min(remaining * 0.20, 100.0)
            if t6b > 20:
                a_long2 = interpolate_sequence(best_seq, new_n)
                a_out, _ = pressure_local_search(a_long2, t6b, verbose=False, seed=new_n + 1)
                best_seq, best_score = try_update(best_seq, best_score, a_out, f"press_{new_n}", t_global)

    # Phase 7: Final Nesterov polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Nesterov (n={n_best}) ===")
    if remaining > 60:
        t7 = min(remaining * 0.30, 250.0)
        if t7 > 30:
            a_out, _ = nesterov_bt(norm_sum1(best_seq), t7,
                                   beta_start=50.0, beta_end=1500.0,
                                   beta_growth=1.005, top_k_ratio=0.02,
                                   restart_every=2000, verbose=True, seed=999)
            best_seq, best_score = try_update(best_seq, best_score, a_out, f"nest_final_{n_best}", t_global)

    # Phase 8: Final pressure polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Pressure (n={n_best}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.30, 150.0)
        if t8 > 15:
            a_out, _ = pressure_local_search(norm_sum1(best_seq), t8, verbose=True, seed=555)
            best_seq, best_score = try_update(best_seq, best_score, a_out, f"press_final", t_global)

    # Phase 9: Quick Nesterov if time remains
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 9: Quick Nest (n={n_best}) ===")
        t9 = min(remaining * 0.60, 120.0)
        if t9 > 20:
            a_out, _ = nesterov_bt(norm_sum1(best_seq), t9,
                                   beta_start=80.0, beta_end=2000.0,
                                   beta_growth=1.005, top_k_ratio=0.015,
                                   restart_every=1500, verbose=True, seed=1111)
            best_seq, best_score = try_update(best_seq, best_score, a_out, f"nest_final2", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.1f}s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
