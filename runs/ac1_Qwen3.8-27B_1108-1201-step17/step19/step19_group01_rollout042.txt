def interpolate_sequence(f, new_n):
    f=np.asarray(f,dtype=np.float64)
    old_n=len(f)
    if old_n == new_n: return f.copy()
    s=f.sum()
    if s<1e-15: return np.ones(new_n)/new_n
    fn=f/s
    old_x=np.linspace(0.0,1.0,old_n)
    new_x=np.linspace(0.0,1.0,new_n)
    out=np.interp(new_x, old_x, fn)

def waterfill_target(b, tau):
    size=len(b)
    bmax=float(np.max(b))
    if tau >= bmax:
       return b.copy()
    avg=1.0/size
    if tau <= avg*1.000001:
       # infeasible to get sum1 with max <= tau if tau<avg; return uniform
       return np.full(size, avg)
    lo = -bmax
    hi = bmax
    for _ in range(50):
       lam=(lo+hi)*0.5
       tt=np.clip(b-lam, 0.0, tau)
       s=float(np.sum(tt))
       if s > 1.0:
           lo=lam
       else:
           hi=lam
    lam=(lo+hi)*0.5
    tt=np.clip(b-lam, 0.0, tau)
    s=float(np.sum(tt))
    if s <= 1e-15:
       tt=np.full(size, avg)
    else:
       tt=tt/s
    # if normalization exceeded tau, do a few manual clips
    for _ in range(5):
       if np.max(tt) <= tau:
          break
       excess = tt > tau
       tt[excess] = tau
       s2=float(np.sum(tt))
       if s2 <= 0:
          tt=np.full(size, avg); break
       tt=tt/s2
    return tt

def waterfill_target(b, tau):
    size=len(b); avg=1.0/size
    if tau <= avg: return full(avg)
    t = np.minimum(b, tau)

    if deficit <= 1e-12:

"""
Search for non-negative step-function heights minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2.

The program uses:
- warm-starting from the provided height_sequence_1 when available;
- density-preserving interpolation to larger lengths;
- parametric initial shapes;
- Adam optimization in log-density space for an L2-to-flat-target surrogate;
- Adam optimization in log-density space for smoothed-max and mixed surrogates;
- a very fast exact single-coordinate-transfer coordinate polish that maintains
  the full convolution vector without recomputing FFTs for every candidate.
"""

import numpy as np
import time


def next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def norm_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(len(a))
    return a / s


def autocorr_fft(a, fft_size=None):
    n = len(a)
    size = 2 * n - 1
    if fft_size is None:
        fft_size = next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(fa * fa, fft_size)[:size]
    out = np.maximum(out, 0.0)
    return out


def cross_grad_fft(w, a, fft_size=None):
    n = len(a)
    wl = len(w)
    if fft_size is None:
        fft_size = next_pow2(n + wl - 1)
    ar = np.fft.rfft(a[::-1], fft_size)
    wr = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(ar * wr, fft_size)
    return conv[n - 1:2 * n - 1].copy()


def softmax_weights_add(b, beta):
    size = len(b)
    if size == 0:
        return np.array([], dtype=np.float64)
    M = float(np.max(b))
    z = beta * (b - M)
    z = z - float(np.max(z))
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    s = float(np.sum(e))
    if s < 1e-300:
        top = b > (M - 1e-12 * max(1.0, abs(M)))
        cnt = int(np.count_nonzero(top))
        if cnt == 0:
            return np.full(size, 1.0 / size, dtype=np.float64)
        return np.where(top, 1.0 / cnt, 0.0).astype(np.float64)
    return e / s


def poly_weights(b, p):
    size = len(b)
    if size == 0:
        return np.array([], dtype=np.float64)
    bpos = np.maximum(b, 0.0)
    bmax = float(np.max(bpos))
    if bmax <= 0.0:
        return np.full(size, 1.0 / size, dtype=np.float64)
    scaled = bpos / bmax
    scaled = np.maximum(scaled, 1e-300)
    z = p * np.log(scaled)
    z = z - float(np.max(z))
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    s = float(np.sum(e))
    if s < 1e-300:
        top = bpos > 0.5 * bmax
        cnt = int(np.count_nonzero(top))
        if cnt == 0:
            return np.full(size, 1.0 / size, dtype=np.float64)
        return np.where(top, 1.0 / cnt, 0.0).astype(np.float64)
    return e / s


def soft_max_x_to_a(x):
    y = x - float(np.max(x))
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    s = float(np.sum(e))
    if s < 1e-300:
        return np.full_like(x, 1.0 / float(len(x)), dtype=np.float64)
    return e / s


def interpolate_sequence(f, new_n):
    f = np.asarray(f, dtype=np.float64)
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    if new_n <= 0:
        return np.array([], dtype=np.float64)
    s = float(np.sum(np.maximum(f, 0.0)))
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    fn = np.maximum(f, 0.0) / s
    # Interpolate the density f(x) = old_n * a_i, then divide by new_n.
    f_hat = fn * float(old_n)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, f_hat)
    out = np.maximum(out, 0.0)
    out = out / float(new_n)
    s2 = float(np.sum(out))
    if s2 < 1e-15:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    return out / s2


def parametric_initial(n, current_best_score, budget_s=10.0):
    t0 = time.monotonic()
    best = np.ones(n, dtype=np.float64) / float(n)
    best_score = 2.0 * float(n) * float(np.max(autocorr_fft(best)))
    if current_best_score < best_score:
        best_score = current_best_score

    idx = np.arange(n, dtype=np.float64)
    t = (idx + 1.0) / float(n + 1)
    u = 1.0 - t
    x = np.linspace(0.0, 1.0, n)

    power_a = (-0.6, -0.4, -0.2, 0.0, 0.3, 0.7, 1.2, 2.0)
    power_b = (-0.2, 0.0, 0.2, 0.5, 1.0)

    for pa in power_a:
        if time.monotonic() - t0 > budget_s:
            break
        for pb in power_b:
            if time.monotonic() - t0 > budget_s:
                break
            try:
                base = np.power(t, pa) * np.power(u, pb)
            except Exception:
                continue
            base = np.nan_to_num(base, nan=0.0, posinf=0.0, neginf=0.0)
            base = np.maximum(base, 0.0)
            for flip in (False, True):
                shape = base[::-1] if flip else base
                arr = norm_sum1(shape)
                sc = 2.0 * float(n) * float(np.max(autocorr_fft(arr)))
                if sc < best_score:
                    best = arr
                    best_score = sc

    mix_a = (0.2, 0.4, 0.5)
    mix_b = (0.0, 0.2, 0.4, 0.6)
    for pa in mix_a:
        if time.monotonic() - t0 > budget_s:
            break
        for pb in mix_b:
            if time.monotonic() - t0 > budget_s:
                break
            try:
                if pb <= 0.0:
                    base = np.power(t, -pa) + 1.0 * np.power(u, 0.0)
                else:
                    base = np.power(t, -pa) + np.power(u, -pb)
            except Exception:
                continue
            base = np.nan_to_num(base, nan=0.0, posinf=0.0, neginf=0.0)
            base = np.maximum(base, 0.0)
            for flip in (False, True):
                shape = base[::-1] if flip else base
                arr = norm_sum1(shape)
                sc = 2.0 * float(n) * float(np.max(autocorr_fft(arr)))
                if sc < best_score:
                    best = arr
                    best_score = sc

    for pw in (1, 2, 3):
        if time.monotonic() - t0 > budget_s:
            break
        base = np.sin(np.pi * x) ** pw
        base = np.maximum(base, 0.0)
        for flip in (False, True):
            shape = base[::-1] if flip else base
            arr = norm_sum1(shape)
            sc = 2.0 * float(n) * float(np.max(autocorr_fft(arr)))
            if sc < best_score:
                best = arr
                best_score = sc

    return best, best_score


def pressure_from_active(active_idx, weights, a):
    n = len(a)
    P = np.zeros(n, dtype=np.float64)
    for k, w in zip(active_idx, weights):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            P[lo:hi + 1] += w * a[k - hi:k - lo + 1][::-1]
    return P


def estimate_move(a, active_b, active_idx, i, j):
    n = len(a)
    max_delta = a[i] * 0.999
    if max_delta <= 1e-15:
        return None
    ki = active_idx - i
    kj = active_idx - j
    L = np.zeros(len(active_idx), dtype=np.float64)
    vj = (kj >= 0) & (kj < n)
    if np.any(vj):
        L[vj] += 2.0 * a[kj[vj]]
    vi = (ki >= 0) & (ki < n)
    if np.any(vi):
        L[vi] -= 2.0 * a[ki[vi]]

    Q = np.zeros(len(active_idx), dtype=np.float64)
    Q[active_idx == 2 * i] += 1.0
    Q[active_idx == 2 * j] += 1.0
    Q[active_idx == i + j] -= 2.0

    fracs = np.array([0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 0.99])
    best_val = float("inf")
    best_delta = None

    for frac in fracs:
        d = max_delta * float(frac)
        nb = active_b + d * L + d * d * Q
        if nb.min() < 0.0:
            nb = np.maximum(nb, 0.0)
        val = float(np.max(nb)) if nb.size else float("inf")
        if val < best_val:
            best_val = val
            best_delta = d

    if best_delta is None:
        return None

    factors = np.array([0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0])
    for factor in factors:
        d = best_delta * float(factor)
        if d <= 1e-15 or d > max_delta:
            continue
        nb = active_b + d * L + d * d * Q
        if nb.min() < 0.0:
            nb = np.maximum(nb, 0.0)
        val = float(np.max(nb))
        if val < best_val:
            best_val = val
            best_delta = d

    return (best_val, float(best_delta))


def full_try_move(a, b, i, j, frac):
    if frac <= 0.0:
        return None
    delta = a[i] * frac
    if delta <= 1e-18:
        return None
    n = len(a)
    size = 2 * n - 1
    L = np.zeros(size, dtype=np.float64)
    L[i:i + n] -= 2.0 * a
    L[j:j + n] += 2.0 * a
    nb = b + delta * L
    d2 = delta * delta

    idx = 2 * i
    if 0 <= idx < size:
        nb[idx] += d2
    idx = 2 * j
    if 0 <= idx < size:
        nb[idx] += d2
    idx = i + j
    if 0 <= idx < size:
        nb[idx] -= 2.0 * d2

    if nb.min() < 0.0:
        nb = np.maximum(nb, 0.0)
    return (nb, float(np.max(nb)))


def fast_coordinate_polish(a0, time_limit, top_active=40, src_count=40, dst_count=40,
                           pair_candidates=80, full_candidates=8,
                           max_no_improve=800, verbose=False):
    a = norm_sum1(a0).copy()
    n = len(a)
    if n < 2 or time_limit <= 0.0:
        b = autocorr_fft(a)
        return a, 2.0 * float(n) * float(np.max(b))

    fft_b = next_pow2(2 * n - 1)
    b = autocorr_fft(a, fft_b)
    M = float(np.max(b))
    best_a = a.copy()
    best_M = M

    t0 = time.monotonic()
    rng = np.random.default_rng(2025)
    no_improve = 0
    recompute = 0
    move_count = 0
    last_print_move = 0

    active_idx = None
    active_b = None
    P = None
    pairs = []

    while time.monotonic() - t0 < time_limit:
        if no_improve >= max_no_improve:
            break

        if P is None or recompute >= 12 or no_improve > 40:
            M_cur = float(np.max(b))
            thr = M_cur - max(0.02 * M_cur, 1e-13)
            idx_above = np.nonzero(b >= thr)[0]
            if len(idx_above) == 0:
                idx_above = np.argsort(b)[::-1][:top_active]
            elif len(idx_above) > top_active:
                local = np.argsort(b[idx_above])[::-1][:top_active]
                idx_above = idx_above[local]
            active_idx = idx_above.astype(np.int64, copy=False)

            wv = np.maximum(b[active_idx], 0.0)
            wv = wv * wv
            ss = float(np.sum(wv))
            if ss < 1e-300:
                wv = np.ones(len(active_idx), dtype=np.float64)
            else:
                wv = wv / ss

            active_b = b[active_idx].copy()
            P = pressure_from_active(active_idx, wv, a)

            idx_topP = np.argsort(P)[::-1][:src_count]
            idx_lowP = np.argsort(P)[:dst_count]
            idx_topA = np.argsort(a)[::-1][:src_count]
            idx_lowA = np.argsort(a)[:dst_count]

            src_list = np.unique(np.concatenate((idx_topP, idx_topA))).astype(np.int64)
            dst_list = np.unique(np.concatenate((idx_lowP, idx_lowA))).astype(np.int64)
            src_list = src_list[a[src_list] > 1e-12]
            if len(src_list) == 0:
                break

            pairs = []
            for ii in src_list:
                pii = float(P[ii])
                for jj in dst_list:
                    if ii != jj:
                        pairs.append((pii - float(P[jj]), int(ii), int(jj)))

            if len(pairs) == 0:
                break

            pairs.sort(reverse=True)
            keep = pair_candidates + 20
            if len(pairs) > keep:
                pairs = pairs[:keep]

            for _ in range(12):
                si = int(src_list[int(rng.integers(len(src_list)))])
                sj = int(dst_list[int(rng.integers(len(dst_list)))])
                if si != sj:
                    pairs.append((float(P[si] - P[sj]) - 100.0, si, sj))

            recompute = 0

        improved = False
        estimates = []
        for _score, i, j in pairs:
            if a[i] <= 1e-12:
                continue
            res = estimate_move(a, active_b, active_idx, i, j)
            if res is not None:
                estimates.append((res[0], i, j, res[1]))

        if estimates:
            estimates.sort()
            tried = 0
            for _val, i, j, delta in estimates:
                if tried >= full_candidates:
                    break
                tried += 1
                frac = float(delta) / float(a[i])
                if frac <= 0.0 or frac >= 1.0:
                    continue

                res_full = full_try_move(a, b, i, j, frac)
                if res_full is None:
                    continue
                best_nb, best_nb_max = res_full
                best_frac = frac

                for factor in (0.5, 0.75, 1.25, 2.0, 0.25):
                    fr2 = frac * float(factor)
                    if fr2 <= 0.0 or fr2 >= 1.0:
                        continue
                    r2 = full_try_move(a, b, i, j, fr2)
                    if r2 is not None and r2[1] < best_nb_max:
                        best_nb = r2[0]
                        best_nb_max = r2[1]
                        best_frac = fr2

                if best_nb_max < M - 1e-14:
                    d = float(a[i]) * best_frac
                    a[i] -= d
                    a[j] += d
                    if a[i] < 0.0:
                        a[i] = 0.0
                    if a[j] < 0.0:
                        a[j] = 0.0

                    b = best_nb
                    M = best_nb_max
                    move_count += 1
                    recompute += 1
                    no_improve = 0

                    if M < best_M - 1e-15:
                        best_M = M
                        best_a = a.copy()
                        if verbose and (move_count == 1 or move_count - last_print_move >= 100):
                            elapsed = time.monotonic() - t0
                            print("    coord move %d (%.1fs): score=%.12f" %
                                  (move_count, elapsed, 2.0 * n * best_M))
                            last_print_move = move_count

                    if move_count % 400 == 0:
                        b2 = autocorr_fft(a, fft_b)
                        M2 = float(np.max(b2))
                        if M2 < M - 1e-12:
                            b = b2
                            M = M2
                            if M < best_M:
                                best_M = M
                                best_a = a.copy()
                        elif M2 > M + 1e-7:
                            a = best_a.copy()
                            b = autocorr_fft(a, fft_b)
                            M = float(np.max(b))

                    improved = True
                    break

        if not improved:
            no_improve += 1
            if verbose and no_improve % 500 == 0:
                print("    coord stall %d (%.1fs): score=%.12f" %
                      (no_improve, time.monotonic() - t0, 2.0 * n * M))

    return best_a, 2.0 * float(n) * float(best_M)


def waterfill_target(b, tau):
    size = len(b)
    if size == 0:
        return np.array([], dtype=np.float64)
    avg = 1.0 / float(size)
    tau = float(tau)
    if tau <= avg * 1.000001:
        return np.full(size, avg, dtype=np.float64)

    t = np.minimum(b, tau)
    t = np.maximum(t, 0.0)

    for _ in range(12):
        deficit = 1.0 - float(np.sum(t))
        if abs(deficit) < 1e-12:
            break
        if deficit > 0.0:
            mask = t < (tau - 1e-18)
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            add = deficit / float(cnt)
            t[mask] += add
            over = t >= tau
            if np.any(over):
                t[over] = tau
        else:
            s2 = float(np.sum(t))
            if s2 > 1e-300:
                t = t / s2
            continue

    s = float(np.sum(t))
    if s <= 1e-300:
        return np.full(size, avg, dtype=np.float64)
    t = t / s
    if t.max() > tau:
        t = np.minimum(t, tau)
        s2 = float(np.sum(t))
        if s2 > 1e-300:
            t = t / s2
    return t


def adam_log(a0, time_limit, mode="add", lr_start=0.05, lr_min=0.01,
             beta_scale_start=25.0, beta_scale_end=80.0, p_power=2.0,
             tau_target=0.0, verbose=False):
    a = norm_sum1(a0)
    n = len(a)
    if n < 2 or time_limit <= 0.0:
        b = autocorr_fft(a)
        return a, 2.0 * float(n) * float(np.max(b))

    fft_b = next_pow2(2 * n - 1)
    fft_g = next_pow2(3 * n - 2)

    x = np.log(np.maximum(a, 1e-30))
    x = x - float(np.mean(x))

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    lr = float(lr_start)

    b = autocorr_fft(a, fft_b)
    best_M = float(np.max(b))
    best_score = 2.0 * float(n) * best_M
    best = a.copy()

    stall = 0
    it = 0
    t0 = time.monotonic()
    rng = np.random.default_rng(77)

    if mode == "l2proj":
        if tau_target <= 0.0:
            tau_target = best_M * 0.98
        tau = max(tau_target, best_M * 0.999)

    while time.monotonic() - t0 < time_limit:
        a = soft_max_x_to_a(x)
        b = autocorr_fft(a, fft_b)
        M = float(np.max(b))
        score = 2.0 * float(n) * M
        it += 1

        if score < best_score - 1e-15:
            best_score = score
            best = a.copy()
            stall = 0
        else:
            stall += 1

        if mode == "l2proj":
            if M > tau:
                tau = max(tau_target, 0.98 * M)
            else:
                tau = max(tau_target, 0.9995 * tau)
            target = waterfill_target(b, tau)
            r = b - target
            g = cross_grad_fft(r, a, fft_g)
        elif mode == "add":
            beta = max(beta_scale_start, 40.0 / max(M, 1e-15))
            beta = min(beta, beta_scale_end / max(M, 1e-15))
            w = softmax_weights_add(b, beta)
            g = cross_grad_fft(w, a, fft_g)
        elif mode == "poly":
            w = poly_weights(b, p_power)
            g = cross_grad_fft(w, a, fft_g)
        else:
            beta = max(beta_scale_start, 30.0 / max(M, 1e-15))
            beta = min(beta, beta_scale_end / max(M, 1e-15))
            w1 = softmax_weights_add(b, beta)
            w2 = poly_weights(b, p_power)
            w = 0.5 * w1 + 0.5 * w2
            s = float(np.sum(w))
            if s > 0.0:
                w = w / s
            g = cross_grad_fft(w, a, fft_g)

        g = 2.0 * g / max(M, 1e-15)
        gx = a * (g - float(np.dot(a, g)))
        gx = np.nan_to_num(gx, nan=0.0, posinf=100.0, neginf=-100.0)
        gx = np.clip(gx, -10.0, 10.0)

        t += 1
        m = 0.9 * m + 0.1 * gx
        v = 0.999 * v + 0.001 * (gx * gx)
        mhat = m / (1.0 - 0.9 ** t)
        vhat = v / (1.0 - 0.999 ** t)
        vhat = np.maximum(vhat, 1e-20)

        x_new = x - lr * mhat / np.sqrt(vhat)
        x_new = np.nan_to_num(x_new, nan=0.0, posinf=0.0, neginf=0.0)
        x_new = np.clip(x_new, -25.0, 25.0)
        x_new = x_new - float(np.mean(x_new))
        x = x_new

        if it % 1000 == 1:
            lr = max(lr_min, lr * 0.999)

        if stall > 3000:
            noise = 0.05 / (1.0 + 0.25 * (stall // 3000))
            x = np.log(np.maximum(best, 1e-30))
            x = x - float(np.mean(x))
            x = x + rng.normal(0.0, noise, n)
            x = np.clip(x, -25.0, 25.0)
            x = x - float(np.mean(x))
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            lr = float(lr_start)
            stall = 0
            if verbose:
                print("    adam restart after stall, score=%.12f" % best_score)

        if verbose and it % 1000 == 0:
            print("    adam %d (%.1fs): score=%.12f" % (it, time.monotonic() - t0, best_score))

        if stall > 12000:
            break

    return best, best_score


def try_update(best_seq, best_score, a, label, t_start=None):
    if a is None:
        return best_seq, best_score
    arr = np.asarray(a, dtype=np.float64)
    if arr.size == 0:
        return best_seq, best_score
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    s = float(np.sum(arr))
    if s < 1e-8:
        return best_seq, best_score
    arr = arr / s
    b = autocorr_fft(arr)
    score = 2.0 * float(len(arr)) * float(np.max(b))
    if np.isfinite(score) and score < best_score:
        msg = "  *** NEW BEST [%s]: score=%.12f, n=%d" % (label, score, len(arr))
        if t_start is not None:
            msg += ", elapsed=%.1fs" % (time.monotonic() - t_start)
        print(msg)
        return arr, score
    return best_seq, best_score


def run_optimizer_phases(a, alloc, aim_score, verbose=True):
    n = len(a)
    t0 = time.monotonic()
    a2 = a.copy()

    t = min(alloc * 0.18, 70.0)
    if t > 10.0:
        a2, _ = fast_coordinate_polish(a2, t, max_no_improve=150, verbose=verbose and n <= 8500)

    rem = alloc - (time.monotonic() - t0)
    t = min(rem * 0.35, 150.0)
    if t > 15.0:
        aim_M = aim_score / (2.0 * float(n))
        min_M = 1.01 / (2.0 * float(n))
        a2, _ = adam_log(a2, t, mode="l2proj", tau_target=max(aim_M, min_M),
                         verbose=verbose and n <= 8500)

    rem = alloc - (time.monotonic() - t0)
    t = min(rem * 0.35, 110.0)
    if t > 15.0:
        a2, _ = adam_log(a2, t, mode="add", verbose=verbose and n <= 8500)

    rem = alloc - (time.monotonic() - t0)
    if rem > 10.0:
        a2, _ = fast_coordinate_polish(a2, rem, max_no_improve=1200,
                                       verbose=verbose and n <= 8500)

    return a2


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    total_budget = max(600.0, min(float(budget_s) - 45.0, 1755.0))

    best_seq = None
    best_score = float("inf")

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim != 1:
            p = p.ravel()
        p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
        p = np.clip(p, 0.0, 1000.0)
        if p.size > 0 and float(np.sum(p)) > 1e-12:
            best_seq = norm_sum1(p)
            b = autocorr_fft(best_seq)
            best_score = 2.0 * float(len(best_seq)) * float(np.max(b))
            print("Parent init: n=%d, score=%.12f" % (len(best_seq), best_score))
    except Exception:
        best_seq = None

    if best_seq is None:
        n_fb = 5000
        best_seq = np.ones(n_fb, dtype=np.float64) / float(n_fb)
        b = autocorr_fft(best_seq)
        best_score = 2.0 * float(n_fb) * float(np.max(b))
        print("Default init: n=%d, score=%.12f" % (n_fb, best_score))

    n0 = int(len(best_seq))
    max_n = max(10900, n0)

    rem = total_budget - (time.monotonic() - t_global)
    if rem > 80.0:
        for n_try in (n0, min(n0 + 1500, max_n)):
            if n_try == n0:
                a_p, _ = parametric_initial(n_try, best_score, min(20.0, rem * 0.04))
            else:
                a_p, _ = parametric_initial(n_try, best_score, min(20.0, rem * 0.04))
            best_seq, best_score = try_update(best_seq, best_score, a_p, "param", t_global)
            rem = total_budget - (time.monotonic() - t_global)
            if rem < 60.0:
                break

    lengths = [n0]
    for step in (700, 1400, 2200, 3500):
        lengths.append(n0 + step)
    lengths = sorted(set([l for l in lengths if 200 <= l <= min(max_n, n0 + 5000)]))
    if len(lengths) == 0:
        lengths = [n0]

    for idx, n in enumerate(lengths):
        rem = total_budget - (time.monotonic() - t_global)
        if rem < 250.0:
            break

        if idx == 0:
            alloc = min(rem * 0.40, 550.0)
        elif idx == 1:
            alloc = min(rem * 0.35, 420.0)
        elif idx == 2:
            alloc = min(rem * 0.30, 350.0)
        else:
            alloc = min(rem * 0.28, 300.0)
        alloc = max(alloc, 80.0)

        if len(best_seq) == n:
            a_init = best_seq.copy()
        else:
            a_init = interpolate_sequence(best_seq, n)

        b = autocorr_fft(a_init)
        sc = 2.0 * float(n) * float(np.max(b))
        if sc < best_score:
            best_seq = a_init
            best_score = sc
            print("Interpolated n=%d: score=%.12f" % (n, sc))

        aim = max(1.500, best_score - 0.015)
        a_out = run_optimizer_phases(a_init, alloc, aim, verbose=(idx < 2))
        best_seq, best_score = try_update(best_seq, best_score, a_out, "len%d" % n, t_global)

    rem = total_budget - (time.monotonic() - t_global)
    if rem > 120.0:
        t = min(rem * 0.75, 500.0)
        a_out, _ = fast_coordinate_polish(best_seq, t, max_no_improve=2500, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "final_coord", t_global)

    rem = total_budget - (time.monotonic() - t_global)
    if rem > 80.0 and len(best_seq) <= 10000:
        t = min(rem * 0.50, 180.0)
        aim = max(1.500, best_score - 0.005)
        a_out, _ = adam_log(best_seq, t, mode="mixed", verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "final_mixed", t_global)

    rem = total_budget - (time.monotonic() - t_global)
    if rem > 40.0:
        t = min(rem * 0.80, 200.0)
        a_out, _ = fast_coordinate_polish(best_seq, t, max_no_improve=1000, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "last_coord", t_global)

    arr = np.asarray(best_seq, dtype=np.float64)
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    s = float(np.sum(arr))
    if s < 1e-8:
        n_fb = 5000
        arr = np.ones(n_fb, dtype=np.float64) / float(n_fb)
        s = 1.0
    else:
        arr = arr / s

    # Final score via the same convolution formula; evaluate_sequence is not called
    # here because it can be expensive for long sequences and the returned vector is
    # already normalized and clamped to the feasible nonnegative domain.
    final_b = autocorr_fft(arr)
    final_score = 2.0 * float(len(arr)) * float(np.max(final_b))
    print("Final returned: n=%d, proxy_score=%.12f, total_time=%.1fs" %
          (len(arr), final_score, time.monotonic() - t_global))

    return [float(x) for x in arr]
