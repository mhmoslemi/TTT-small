def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def coord_descent_polish(a0, time_limit, n_sweeps=500):
    """Targeted coordinate descent focusing on peak-reducing moves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(4242)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Focus on coordinates that contribute to peaks
        c = fast_convolve(a)
        top_k = min(5, len(c))
        peaks = np.argpartition(c, -top_k)[-top_k:]
        
        # Identify important coordinates
        important = set()
        for pk in peaks:
            pk = int(pk)
            i_start = max(0, pk - n + 1)
            i_end = min(n - 1, pk)
            idx = np.arange(i_start, i_end + 1)
            contribs = a[idx] * a[pk - idx]
            top_c = np.argsort(contribs)[::-1][:min(15, len(contribs))]
            for ci in top_c:
                important.add(int(idx[ci]))
                important.add(int(pk - idx[ci]))
        
        # Also add some random coordinates
        random_coords = rng.choice(n, size=min(50, n), replace=False)
        
        all_coords = list(important) + [int(x) for x in random_coords if int(x) not in important]
        
        improved = False
        for i in all_coords:
            if time.monotonic() - t0 > time_limit:
                break
            if i < 0 or i >= n:
                continue
            old_val = a[i]
            if old_val < 1e-14:
                continue
            
            for factor in [0.7, 0.8, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2, 1.4]:
                new_val = old_val * factor
                if new_val < 0:
                    continue
                a[i] = new_val
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a.copy()
                    old_val = a[i]
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()
        
        if not improved and sweep > 10:
            break
    
    return best_a, best_score

def frank_wolfe(a0, time_limit, max_iters=5000, verbose=False):
    """Frank-Wolfe (conditional gradient) method for simplicity of implementation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient of max(conv(a,a))
        c = fast_convolve(a)
        k_peak = int(np.argmax(c))
        
        # Subgradient: for the peak at k_peak, grad[i] = 2*a[k_peak - i] for valid i
        grad = np.zeros(n)
        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        idx = np.arange(i_start, i_end + 1)
        grad[idx] = 2.0 * a[k_peak - idx]
        
        # But we should use a weighted subgradient (softmax over top peaks)
        top_k = min(10, len(c))
        peaks = np.argpartition(c, -top_k)[-top_k:]
        pv = c[peaks]
        temp = 0.01 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        w /= w.sum()
        
        grad = np.zeros(n)
        for wi, pk in zip(w, peaks):
            if wi < 1e-12:
                continue
            pk = int(pk)
            i_s = max(0, pk - n + 1)
            i_e = min(n - 1, pk)
            idx2 = np.arange(i_s, i_e + 1)
            grad[idx2] += wi * 2.0 * a[pk - idx2]
        
        # Frank-Wolfe: find vertex of simplex that minimizes <grad, v>
        # This is the e_i where grad[i] is minimum
        i_min = int(np.argmin(grad))
        e_i = np.zeros(n)
        e_i[i_min] = 1.0
        
        # Step size: gamma_t = 2/(t+2) for FW
        gamma = 2.0 / (it + 2)
        # Or use line search
        a_new = (1 - gamma) * a + gamma * e_i
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW it={it} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def lse_gradient(a, tau):
    """Compute gradient of (1/tau)*log(sum_k exp(tau*conv(a,a)[k])) and the conv values."""
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    exp_c = np.exp(tau * (c - max_c))
    p = exp_c / exp_c.sum()
    
    # grad[i] = 2 * sum_j a[j] * p[i+j] (cross-correlation of a and p)
    corr_size = n + (2 * n - 1) - 1
    corr_fft_size = 1 << (corr_size - 1).bit_length()
    A_corr = rfft(a, corr_fft_size)
    P_corr = rfft(p, corr_fft_size)
    corr = irfft(np.conj(A_corr) * P_corr, corr_fft_size)[:n]
    
    grad = 2.0 * corr
    return grad, c, p

def build_conv_matrix(a):
    """Build matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():

def solve_lp_br(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        rem = deadline - time.monotonic() - 700
        if rem < 40:
            break
        t_alloc = min(time_per, rem * 0.9)
        if t_alloc < 15:
            continue
        
        if n_try == n_parent:
            inits = [parent.copy()]
        else:
            inits = [interpolate_seq(parent, n_try)]
            # Add Fourier init
            nh = rng.integers(4, 15)
            amps = rng.random(nh) * 0.3
            phases = rng.uniform(0, 2 * np.pi, nh)
            freqs = np.arange(1, nh + 1)
            tt = np.arange(n_try, dtype=np.float64)
            af = np.full(n_try, 0.5)
            for j in range(nh):
                af += amps[j] * np.cos(2 * np.pi * freqs[j] * tt / n_try + phases[j])
            af = np.maximum(af, 0)
            af = af / af.sum()
            inits.append(af)
        
        for ii, ia in enumerate(inits):
            rem = deadline - time.monotonic() - 700
            if rem < 15:
                break
            t_sub = t_alloc / len(inits)
            a_opt, s_opt = lse_optimize(ia, t_sub, tau_init=30.0, tau_final=300.0, lr_init=0.006)
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"lse_n{n_try}_{ii}")
    
    print(f"  Phase 1 done ({time.monotonic()-t1:.0f}s), best={best_score:.6f}")
    
    # === Phase 2: Active peak refinement (120s) ===
    print(f"\n=== Phase 2: Active Peak Refine (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t2 = min(120.0, rem * 0.5)
            a2, s2 = active_peak_refine(best_seq, t2, n_active=10)
            best_seq, best_score = check_and_update(a2, best_seq, best_score, "active_peak")
    
    # === Phase 3: Pairwise mass transfer (120s) ===
    print(f"\n=== Phase 3: Pairwise Transfer (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t3 = min(120.0, rem * 0.5)
            a3, s3 = pairwise_transfer(best_seq, t3, rng)
            best_seq, best_score = check_and_update(a3, best_seq, best_score, "pairwise")
    
    # === Phase 4: LP Fictitious Play (250s) ===
    print(f"\n=== Phase 4: LP Fictitious Play (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Run on best length and one alternative
        fp_ns = [n_best]
        if n_best + 32 <= 2000:
            fp_ns.append(n_best + 32)
        
        for nf in fp_ns[:2]:
            rem = deadline - time.monotonic() - 200
            if rem < 40:
                break
            t4 = min(150.0, rem * 0.5)
            init_f = interpolate_seq(best_seq, nf) if nf != n_best else best_seq.copy()
            a4, s4, it4 = lp_fictitious_play(init_f, t4, solver=solver, verbose=True)
            best_seq, best_score = check_and_update(a4, best_seq, best_score, f"fp_n{nf}")
    
    # === Phase 5: Post-LP refinement (150s) ===
    print(f"\n=== Phase 5: Post-LP Refine (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t5 = min(80.0, rem * 0.5)
            a5, s5 = active_peak_refine(best_seq, t5, n_active=6)
            best_seq, best_score = check_and_update(a5, best_seq, best_score, "post_lp_active")
        
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t5b = min(70.0, rem * 0.5)
            a5b, s5b = pairwise_transfer(best_seq, t5b, rng)
            best_seq, best_score = check_and_update(a5b, best_seq, best_score, "post_lp_pairwise")
    
    # === Phase 6: Coordinate descent polish (100s) ===
    print(f"\n=== Phase 6: Coord Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t6 = min(100.0, rem * 0.5)
            a6, s6 = coord_descent_polish(best_seq, t6, n_sweeps=400)
            best_seq, best_score = check_and_update(a6, best_seq, best_score, "coord_descent")
    
    # === Phase 7: Final LSE with high tau (60s) ===
    print(f"\n=== Phase 7: High-Tau LSE (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t7 = min(60.0, rem * 0.6)
            a7, s7 = lse_optimize(best_seq, t7, tau_init=200.0, tau_final=800.0, lr_init=0.002)
            best_seq, best_score = check_and_update(a7, best_seq, best_score, "high_tau_lse")
    
    # === Phase 8: Final ultra-fine polish (30s) ===
    print(f"\n=== Phase 8: Ultra Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t8 = min(25.0, rem * 0.6)
            a8, s8 = coord_descent_polish(best_seq, t8, n_sweeps=200)
            best_seq, best_score = check_and_update(a8, best_seq, best_score, "ultra_cd")
        
        rem = deadline - time.monotonic() - 8
        if rem > 8:
            a8b, _ = pairwise_transfer(best_seq, min(12.0, rem * 0.7), rng)
            best_seq, best_score = check_and_update(a8b, best_seq, best_score, "ultra_pm")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
