```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w
```

```python
def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
    """Use spectral domain to construct a good sequence."""
    # Start with a target power spectrum
    # The "ideal" power spectrum for a flat convolution would be:
    # If b_k = C for all k in [n-1] (the "main region"), then
    # S(ω) = sum_k C * e^{ikω} = C * sum_{k=0}^{2n-2} e^{ikω}
    # = C * sin((2n-1)*ω/2) / sin(ω/2) (Dirichlet kernel)
    # This is not non-negative, so it's not a valid power spectrum.

    # Then b_k = n for all k (circular), which is the ideal.
    # For linear convolution, we need to handle the boundaries.

    N = 1 << (2*n-1).bit_length()  # FFT size
    S = np.ones(N) * n  # target power spectrum
```

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
