"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Alternating LP (best-response) optimization as the workhorse
2. Soft-max L-BFGS for polynomial-time polishing
3. Multi-start with perturbations
4. Multiple sequence lengths
5. Sparse LP (top constraints only) for speed

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a, top_k=None):
    n = len(a)
    n_rows = 2 * n - 1
    if top_k is not None and top_k < n_rows:
        # Only build top_k rows (closest to center)
        center = n - 1
        idxs = []
        for d in range(0, n_rows // 2 + 1):
            if center + d < n_rows:
                idxs.append(center + d)
            if center - d >= 0 and d > 0:
                idxs.append(center - d)
            if len(idxs) >= top_k:
                break
        idxs = idxs[:top_k]
    else:
        idxs = list(range(n_rows))
    
    T = np.zeros((len(idxs), n))
    for row_i, k in enumerate(idxs):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            T[row_i, j] = a[k - j]
    return T, idxs

def solve_lp_best_response(a, solver=None, top_k=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T, idxs = build_conv_matrix(a, top_k=top_k)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def get_lp_solver():
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    import cvxpy as cp
    for solver_name in [cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name

        except Exception:

    _lp_solver = cp.SCS
    return cp.SCS

def alternating_lp(a0, time_limit, solver=None, top_k=None, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP for best response to a
        b_new, M_b = solve_lp_best_response(a, solver, top_k=top_k)
        if b_new is None:
            break
        
        # Evaluate b_new
        score_b = 2.0 * n * fast_autocorr(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()
        
        # Use b_new as new a (full alternating)
        a = b_new
        
        if verbose and (it + 1) % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ALP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.5, maxiter=5000):
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    def make_obj(T_val):
        def obj(z_vec):
            a = softmax(z_vec)
            c = fast_autocorr(a)
            cT = T_val * c
            m = cT.max()
            log_se = m + np.log(np.exp(cT - m).sum())
            return (2.0 * n / T_val) * log_se
        return obj
    
    def make_jac(T_val):
        def jac(z_vec):
            a = softmax(z_vec)
            c = fast_autocorr(a)
            cT = T_val * c
            m = cT.max()
            wc = np.exp(cT - m)
            w = wc / wc.sum()
            grad_a = np.zeros(n)
            for k in range(len(c)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac
    
    n_stages = 4
    temps = np.geomspace(T_start, T_end, n_stages)
    z_cur = z.copy()
    iters_per = maxiter // n_stages
    
    for T_s in temps:
        if time.monotonic() - t0 > time_limit:
            break
        try:
            res = minimize(
                fun=make_obj(T_s),
                x0=z_cur,
                jac=make_jac(T_s),
                method='L-BFGS-B',
                options={'maxiter': iters_per, 'ftol': 1e-15, 'gtol': 1e-12,
                         'maxfun': iters_per * 30, 'disp': False}
            )
            z_cur = res.x
        except Exception:
            break
    
    return softmax(z_cur)

def peak_shave(a0, time_limit, max_steps=5000):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Alternating LP (exact best-response) as the primary workhorse
2. Soft-max L-BFGS with aggressive geometric annealing for polishing
3. Peak-shaving local search for final refinement
4. Multi-length exploration and diversity restarts
Key innovation: sparse LP (top-constraint selection) for faster iterations,

def fast_autocorr(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def get_lp_solver():

        except Exception:

def solve_lp_best_response(a, solver=None, top_k=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build constraint matrix (optionally sparse - only top_k constraints)
    n_rows = 2 * n - 1
    if top_k is not None and top_k < n_rows:
        # Select constraints closest to the center (where peaks tend to be)
        center = n - 1
        idxs = []
        for d in range(0, center + 1):
            if center + d < n_rows:
                idxs.append(center + d)
            if d > 0 and center - d >= 0:
                idxs.append(center - d)
            if len(idxs) >= top_k:
                break
        idxs = idxs[:top_k]
    else:
        idxs = list(range(n_rows))

    T = np.zeros((len(idxs), n))
    for row_i, k in enumerate(idxs):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            T[row_i, j] = a[k - j]

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def alternating_lp(a0, time_limit, solver=None, verbose=True, top_k_start=None, top_k_after=None, full_iters=10):
    """
    Alternating LP: alternate between solving best-response LPs.
    Uses full constraints initially, then sparse constraints for speed.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_k_start is None:
        top_k_start = 2 * n - 1
    if top_k_after is None:
        top_k_after = max(200, n)

            break

        a = b_new

            elapsed = time.monotonic() - t0
            print(f"    ALP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

def softmax_fn(z):
    """Convert log-domain to probability vector."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.3, maxiter=6000):
    """L-BFGS with soft-max smoothing and geometric temperature schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

    def make_obj(T_val):
        def obj(z_vec):
            a = softmax_fn(z_vec)
            c = fast_autocorr(a)
            cT = T_val * c
            m = cT.max()
            log_se = m + np.log(np.exp(cT - m).sum())
            return (2.0 * n / T_val) * log_se
        return obj

    def make_jac(T_val):
        def jac(z_vec):
            a = softmax_fn(z_vec)
            c = fast_autocorr(a)
            cT = T_val * c
            m = cT.max()
            wc = np.exp(cT - m)
            w = wc / wc.sum()
            grad_a = np.zeros(n)
            for k in range(len(c)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 5
    temps = np.geomspace(T_start, T_end, n_stages)
    z_cur = z.copy()
    iters_per = max(100, maxiter // n_stages)

    for T_s in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_obj(T_s),
                x0=z_cur,
                jac=make_jac(T_s),
                method='L-BFGS-B',
                options={'maxiter': iters_per, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_per * 40, 'disp': False}
            )
            z_cur = res.x
        except Exception:
            pass

    return softmax_fn(z_cur)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
