import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_convolve(a, b):
    """Compute conv(a, b) using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def lp_solve(a, M, n):
    """
    Solve: max sum(b) s.t. b >= 0, conv(a,b) <= M
    Returns optimal b.
    """
    # Constraint matrix: A[k, i] = a[k-i] if 0 <= k-i <= n-1, else 0
    # Number of constraints: 2n-1
    num_constraints = 2 * n - 1
    
    # Build constraint matrix (sparse would be better but for n=768 it's dense enough)
    # A is num_constraints x n
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = a[j]
    
    # Objective: minimize -sum(b)
    c = -np.ones(n)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    # Constraints: A @ b <= M
    res = linprog(c, A_ub=A, b_ub=np.full(num_constraints, M), bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        return np.full(n, M / (2*n))  # fallback: uniform-ish

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play as the
main optimization driver, combined with Adam log-space gradient descent for
polishing, peak-targeted local search, and multi-start exploration.
The LP approach directly optimizes the autocorrelation bound constraint,

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_lp_matrix_sparse(a, n):
    """
    Build sparse matrix A where A[k, i] = a[k-i] if 0 <= k-i < n else 0.
    This represents the constraint conv(a, b)[k] <= M for each k.
    """
    num_constraints = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for i in range(n):
        row_start = i
        row_end = i + n
        rows.extend(range(row_start, row_end))
        cols.extend([i] * n)
        vals.extend(a)
    return csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))


def lp_best_response(a, M, n, method='highs'):
    """
    Solve: max sum(b) s.t. b >= 0, conv(a,b) <= M elementwise.
    Returns optimal b vector.
    """
    A = build_lp_matrix_sparse(a, n)
    num_constraints = 2 * n - 1
    c = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method=method,
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def lp_fictitious_play(a0, time_limit, max_iter=2000, verbose=True, verbose_interval=20):
    """
    LP-based fictitious play: iteratively solve LP best response and mix.
    a0 must be normalized so sum(a0) = sqrt(2*n).

            break

def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def compute_grad_conv(a, w, n, fft_size):
    """
    Compute gradient of max(conv(a,a)) approximated by weighted sum.
    grad_a[i] = sum_j w[j+i] * a[j] (for the smooth surrogate).
    """
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    return corr


def adam_logspace(a0, time_limit, T_init=0.05, T_final=0.001, lr=0.01,
                  beta1=0.9, beta2=0.999, eps=1e-8, verbose=False,
                  verbose_interval=5000):
    """
    Adam optimizer in log-space with temperature annealing.
    a0 should be a probability vector (sums to 1).
    Returns (best_a_normalized_to_sum_1, best_score_unnormalized).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 5000

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Temperature: exponential decay
        T = T_init * (T_final / T_init) ** progress

        a_curr = softmax_from_z(z)
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()

        # Gradient in a-space
        grad_a = compute_grad_conv(a_curr, w, n, fft_size)

        # Transform to z-space
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = conv_new.max()

        if score < best_conv_max - 1e-14:
            best_conv_max = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.003
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = max(0.001, lr * 0.5)

        lr = max(0.0005, lr * (1.0 - 0.00005))

        if verbose and t % verbose_interval == 0:
            print(f"      Adam t={t} T={T:.6f} score={best_conv_max:.8f}")

    return best_a, 2.0 * n * best_conv_max

def coordinate_descent(a0, time_limit, n_sweeps=200):

    n = len(a0)

                continue

            best_local = a[i]
            best_local_score = best_conv_max

            for factor in [0.7, 0.8, 0.9, 1.1, 1.2, 1.3]:
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = fast_convolve(a_trial).max()
                if score < best_local_score - 1e-14:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_and_track(a, label, best_seq, best_score):
    """Evaluate a candidate probability vector and track best."""
    a = np.maximum(a, 0.0)

        best_score = score

        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def structured_initializations(n, rng):
    """Generate diverse structured initializations."""
    candidates = []
    x = np.linspace(0, 1, n)

    # Uniform
    candidates.append(('uniform', np.ones(n) / n))

    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x))
    cos_w = np.maximum(cos_w, 0.001)
    candidates.append(('cosine', cos_w / cos_w.sum()))

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    rc = np.maximum(rc, 0.001)
    candidates.append(('raised_cos', rc / rc.sum()))

    # Gaussian center
    g = np.exp(-20 * (x - 0.5) ** 2)
    candidates.append(('gauss_center', g / g.sum()))

    # Two peaks
    g2 = np.exp(-50 * (x - 0.25) ** 2) + np.exp(-50 * (x - 0.75) ** 2)
    candidates.append(('two_peaks', g2 / g2.sum()))

    # Sinc-like (positive part)
    t = np.linspace(-4, 4, n)
    sinc = np.sin(np.pi * t) / (np.pi * t + 1e-10)
    sinc = np.maximum(sinc, 0)
    if sinc.sum() > 1e-10:
        candidates.append(('sinc_pos', sinc / sinc.sum()))

    # Random smooth
    freqs = rng.integers(1, 10, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amps = rng.uniform(0.1, 1.0, size=5)
    smooth = np.ones(n)
    for fr, ph, am in zip(freqs, phases, amps):
        smooth += am * np.cos(2 * np.pi * fr * x + ph)
    smooth = np.maximum(smooth, 0.001)
    candidates.append(('smooth_rand', smooth / smooth.sum()))

    # Random
    rand = np.abs(rng.standard_normal(n)) + 0.01
    candidates.append(('random', rand / rand.sum()))

    return candidates

        for factor in [1.25, 1.5, 0.75, 2.0]:
            alt_n = int(n_parent * factor)
            if 256 <= alt_n <= 4096 and alt_n != n_parent:
                if alt_n not in alt_lengths:
                    alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:2]:
            rem = deadline - time.monotonic() - 10
            if rem < 30:
                break
            t_sub = min(rem * 0.5, 60)
            if t_sub < 10:
                continue
            print(f"  Trying n={alt_n}, budget={t_sub:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_res, score_res = adam_logspace(init_alt, t_sub,
                                              T_init=0.03, T_final=0.001,
                                              lr=0.005)
            best_seq, best_score = evaluate_and_track(a_res, f"alt_n{alt_n}",
                                                       best_seq, best_score)

    # =========================================================
    # Phase 6: Final intensive polish
    # =========================================================
    print(f"\n=== Phase 6: Final Polish (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 3

    if rem > 15 and best_seq is not None:
        # Final peak shave
        t_fin = min(rem * 0.5, 30)
        a_res = peak_shave(best_seq, t_fin, n_trials=5000)
        best_seq, best_score = evaluate_and_track(a_res, "final_ps", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK: {best_score:.6f}")

    return [float(x) for x in best_seq]
