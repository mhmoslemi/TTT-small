from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def smooth_objective(a, beta):
    """Log-sum-exp of convolution values (smooth max approximation)."""
    b = fast_convolve(a)
    max_b = b.max()
    shifted = beta * (b - max_b)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    return max_b * beta + np.log(np.sum(np.exp(shifted)))

def smooth_gradient(a, beta, top_k=50):
    """Gradient of the log-sum-exp objective."""
    b = fast_convolve(a)
    max_b = b.max()
    shifted = beta * (b - max_b)
    shifted = np.clip(shifted, -50, 50)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Gradient of log(sum exp(beta * b_k)) w.r.t. a_i
    # = sum_k weights[k] * beta * db_k/da_i
    # db_k/da_i = 2 * a_{k-i} (for valid k-i)
    
    n = len(a)
    grad = np.zeros(n)
    # Only use top_k weights for efficiency
    top_idx = np.argsort(weights)[::-1][:top_k]
    for k in top_idx:

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver_cached
    if '_lp_solver_cached' in globals() and _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.GLPK, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return cp.ECOS


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.MOSEK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """Alternating LP best responses with averaging."""
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = compute_score(a)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        a_avg = (iters * a_avg + a) / (iters + 1)

        candidates = [a, b, 0.5 * (a + b), a_avg, 0.7 * a + 0.3 * a_avg]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand_norm = cand / s
            sc = compute_score(cand_norm)
            if sc < best_score:
                best_score = sc
                best_a = cand_norm.copy()

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def lse_gradient_descent(a0, time_limit, beta_start=5.0, beta_end=200.0,
                          lr_start=0.02, verbose=False):
    """Log-sum-exp gradient descent with adaptive beta and learning rate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    total_iters = 5000

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / total_iters)
        beta = beta_start + (beta_end - beta_start) * progress * progress
        lr = lr_start * (1.0 - 0.5 * progress)

        b = fast_convolve(a)
        max_b = b.max()
        shifted = beta * (b - max_b)
        shifted = np.clip(shifted, -50, 50)
        weights = np.exp(shifted)
        weights /= weights.sum()

        # Compute gradient
        grad = np.zeros(n)
        top_k = min(100, len(b))
        top_idx = np.argsort(weights)[::-1][:top_k]
        for k in top_idx:
            if weights[k] < 1e-14:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                j = np.arange(i_start, i_end + 1)
                grad[j] += 2.0 * beta * weights[k] * a[k_int - j]

        # Projected gradient step
        a_new = a - lr * grad
        a_new = simplex_project(a_new)

        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 100:
            # Random restart within neighborhood
            noise = np.random.default_rng(it * 31 + 7).standard_normal(n) * 0.02
            a = simplex_project(0.9 * best_a + 0.1 * np.ones(n) / n + noise)
            no_improve = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD it={it} ({elapsed:.0f}s): beta={beta:.1f} best={best_score:.6f}")

    return best_a, best_score


def subgradient_for_max(a, top_k=30):
    """Compute a subgradient of max(conv(a,a)) at point a."""
    n = len(a)
    b = fast_convolve(a)
    k_actual = min(top_k, len(b))
    peaks = np.argpartition(b, -k_actual)[-k_actual:]
    pv = b[peaks]
    temp = 0.005 * max(pv.max(), 1e-10)
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j = np.arange(i_start, i_end + 1)
            sg[j] += 2.0 * w[idx] * a[k - j]
    return sg, peaks, w


def pairwise_mass_transfer(a0, time_limit, verbose=False):
    """Local search using pairwise mass transfer guided by subgradients."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0
    step_sizes = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]

    while time.monotonic() - t0 < time_limit:
        # Compute subgradient to identify important coordinates
        sg, peaks, w = subgradient_for_max(a, top_k=20)
        
        # Identify source (high subgradient) and target (low subgradient) coordinates
        # We want to move mass FROM high-contribution coords TO low-contribution coords
        # to flatten the convolution
        
        # Focus on the peak position
        peak_k = int(peaks[0])
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        contrib_indices = np.arange(i_start, i_end + 1)
        
        if len(contrib_indices) == 0:
            break
        
        # Compute contributions to the peak
        contribs = np.zeros(len(contrib_indices))
        for idx, i in enumerate(contrib_indices):
            j = peak_k - i
            if 0 <= j < n:
                contribs[idx] = a[i] * a[j]
        
        # Top contributors to peak
        top_contrib = np.argsort(contribs)[::-1][:10]
        
        # Also pick random pairs for exploration
        improved = False
        
        # Strategy 1: Reduce top contributors by transferring mass elsewhere
        for ci in top_contrib[:5]:
            i = int(contrib_indices[ci])
            j = peak_k - i
            
            # Find a good target: a coordinate with low subgradient
            # that won't create new peaks
            targets = []
            for t in range(n):
                if t == i or t == j:
                    continue
                if a[t] > 0 and sg[t] < np.median(sg):
                    targets.append(t)
            
            if not targets:
                targets = [t for t in range(n) if t != i and t != j]
            
            rng.shuffle(targets)
            
            for t in targets[:3]:
                for delta in [0.001, 0.002, 0.005, 0.01]:
                    if a[i] < delta:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[t] += delta
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if not improved and i != j and a[j] >= 1e-14:
                # Also try reducing the paired coordinate
                for t in targets[:2]:
                    for delta in [0.002, 0.005]:
                        if a[j] < delta:
                            continue
                        a_trial = a.copy()
                        a_trial[j] -= delta
                        a_trial[t] += delta
                        a_trial = np.maximum(a_trial, 0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        sc = compute_score(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break
        
        # Strategy 2: Random pairwise transfers
        if not improved:
            for _ in range(20):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                if a[i] < 1e-6:
                    continue
                for delta in [0.001, 0.002, 0.005, 0.01, 0.02]:
                    if a[i] < delta:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[j] += delta
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Perturb and restart
                noise = rng.standard_normal(n) * 0.005
                a = simplex_project(best_a + noise)
                no_improve = 0

        if verbose and (it if 'it' in dir() else 0) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Pairwise ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def coordinate_descent_v2(a0, time_limit, verbose=False):
    """Enhanced coordinate descent with adaptive scaling."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    # Get initial subgradient to prioritize coordinates
    sg, _, _ = subgradient_for_max(a, top_k=30)

    for sweep in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False

        # Process coordinates in order of subgradient magnitude
        order = np.argsort(sg)[::-1]

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-14:
                continue

            # Try scaling
            for factor in [0.7, 0.8, 0.9, 1.1, 1.25, 1.5]:
                new_val = old_val * factor
                a[i] = new_val
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s
                sc = compute_score(a)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a.copy()
                    old_val = a[i]
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()

            # Try small adjustments
            for factor in [0.95, 0.975, 1.025, 1.05]:
                a[i] = old_val * factor
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s
                sc = compute_score(a)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    old_val = a[i]
                else:
                    a[i] = old_val
                    a = a / a.sum()

        if not improved:
            # Update subgradient
            sg, _, _ = subgradient_for_max(a, top_k=30)
            if sweep > 3:
                # Try random coordinate
                for _ in range(50):
                    i = rng.integers(0, n)
                    if a[i] < 1e-14:
                        continue
                    old_val = a[i]
                    for factor in [0.9, 0.95, 1.05, 1.1]:
                        a[i] = old_val * factor
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        sc = compute_score(a)
                        if sc < best_score - 1e-14:
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                        a[i] = old_val
                        a = a / a.sum()
                    if improved:
                        break

        if not improved and sweep > 10:
            # Check if we should stop
            if sweep > 30:
                break

    return best_a, best_score


def peak_shave_direct(a0, time_limit):
    """Directly shave peaks by reducing the most contributing pairs."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for step in range(3000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        top_k = min(3, len(b))
        peaks = np.argpartition(b, -top_k)[-top_k:]

        improved = False
        for peak_k in peaks:
            peak_k = int(peak_k)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue

            contribs = a[idx] * a[peak_k - idx]
            if contribs.max() < 1e-15:
                continue

            top_contrib = np.argsort(contribs)[::-1][:8]

            for ci in top_contrib:
                i = int(idx[ci])
                j = peak_k - i
                if a[i] < 1e-14:
                    continue

                # Try reducing a[i]
                for factor in [0.85, 0.7, 0.5]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break

                # Try reducing both a[i] and a[j]
                if not improved and i != j and 0 <= j < n and a[j] >= 1e-14:
                    for f1, f2 in [(0.8, 0.8), (0.7, 0.9), (0.9, 0.7)]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        sc = compute_score(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                if improved:
                    break

        if not improved:
            break

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_trapezoidal(n, rng):
    """Trapezoidal initialization."""
    rise = int(n * rng.uniform(0.2, 0.4))
    fall = int(n * rng.uniform(0.2, 0.4))
    a = np.zeros(n)
    plateau = n - rise - fall
    if plateau < 1:
        plateau = 1
        rise = n // 3
        fall = n - rise - plateau
    for i in range(n):
        if i < rise:
            a[i] = i / rise
        elif i < rise + plateau:
            a[i] = 1.0
        else:
            a[i] = max(0, 1.0 - (i - rise - plateau + 1) / max(1, fall))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_bimodal(n, rng):
    """Bimodal initialization."""
    center1 = int(n * rng.uniform(0.2, 0.35))
    center2 = int(n * rng.uniform(0.65, 0.8))
    width = int(n * rng.uniform(0.05, 0.15))
    a = np.zeros(n)
    for i in range(n):
        a[i] = np.exp(-0.5 * ((i - center1) / width) ** 2) + np.exp(-0.5 * ((i - center2) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_symmetric_fourier(n, rng):
    """Symmetric Fourier initialization (a_i = a_{n-1-i})."""
    n_sym = n // 2 + 1
    a_sym = np.ones(n_sym) / n_sym
    n_harmonics = int(rng.integers(2, 10))
    for j in range(n_harmonics):
        freq = int(rng.integers(1, max(2, n_sym // 2)))
        amp = rng.random() * 0.2
        a_sym += amp * np.cos(2 * np.pi * freq * np.arange(n_sym) / n_sym)
    a_sym = np.maximum(a_sym, 0)
    # Build full symmetric sequence
    a = np.zeros(n)
    a[:n_sym] = a_sym
    a[n_sym:] = a_sym[max(0, n_sym - 1):0:-1] if n > n_sym else a_sym
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def nesterov_optimize(a0, time_limit, top_k=30, lr_init=0.005, verbose=False):
    """Nesterov-accelerated projected subgradient method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_k = (1 + np.sqrt(1 + 4 * (it - 1) ** 2)) / 2
        beta_k = (t_k - 1) / t_k

        sg, _, _ = subgradient_for_max(y, top_k=top_k)
        v = beta_k * v + sg
        norm_v = np.linalg.norm(v)
        if norm_v > 1e-15:
            v = v / norm_v

        y_new = y - lr * v
        y_projected = simplex_project(y_new)

        sc = compute_score(y_projected)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = y_projected.copy()
            no_improve = 0
        else:
            no_improve += 1

        y = y_projected

        if no_improve > 100:
            lr *= 0.8
            no_improve = 0

        if no_improve > 400:
            rng_local = np.random.default_rng(it * 13 + 7)
            noise = rng_local.standard_normal(n) * 0.015
            y = simplex_project(best_a + noise)
            v = np.zeros(n)
            lr = max(lr_init * 0.7, lr * 2)
            no_improve = 0

        lr *= 0.9998

        if verbose and it % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization: diverse init -> LSE-GD warmup -> LP fictitious play
    -> pairwise mass transfer -> coordinate descent -> peak shaving -> final polish.
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(2024)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Diverse initialization + LSE-GD warmup ===
    print("\n=== Phase 1: Diverse Init + LSE-GD ===")
    t1_start = time.monotonic()
    t_phase1 = 200.0

    n_values = [512, 640, 768, 896, 1024, 1152, 1280]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)
        n_values.insert(1, max(256, np_ + 64))

    candidates = []

    for n in n_values[:6]:
        elapsed_p1 = time.monotonic() - t1_start
        if elapsed_p1 > t_phase1:
            break
        t_per_n = max(10.0, (t_phase1 - elapsed_p1) / max(1, len(n_values[:6]) - n_values.index(n)))

        inits = []
        if parent is not None:
            inits.append(interpolate_seq(parent, n))
        inits.append(init_fourier(n, rng))
        inits.append(init_trapezoidal(n, rng))
        inits.append(init_bimodal(n, rng))
        inits.append(init_symmetric_fourier(n, rng))
        inits.append(np.ones(n) / n)

        for init in inits[:4]:  # Limit to 4 per length for time
            if time.monotonic() - t1_start > t_phase1:
                break
            t_alloc = min(15.0, (t_phase1 - (time.monotonic() - t1_start)))
            if t_alloc < 5:
                break
            a_opt, s_opt = lse_gradient_descent(init, t_alloc, beta_start=10, beta_end=150, lr_start=0.015)
            candidates.append((s_opt, a_opt, n))

    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates, best: {candidates[0][0]:.6f}")

    for s_c, a_c, n_c in candidates[:4]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}")

    print(f"\n=== Phase 2: LP FP (best={best_score:.6f}) ===")
    t2_start = time.monotonic()
    t_phase2 = 500.0

    fp_lengths = []
    if best_seq is not None:
        n_best = len(best_seq)
        fp_lengths.append(n_best)
    for n in [768, 1024, 640, 896]:
        if n not in fp_lengths:
            fp_lengths.append(n)
    fp_lengths = fp_lengths[:3]

    time_per_fp = t_phase2 / len(fp_lengths)

    for n_fp in fp_lengths:
        rem = deadline - time.monotonic() - 700
        if rem < 60:
            break
        t_alloc = min(time_per_fp, rem * 0.85)
        if t_alloc < 30:
            continue

        print(f"  FP n={n_fp}, budget={t_alloc:.0f}s")

            initial = interpolate_seq(best_seq, n_fp)
        elif candidates:
            initial = interpolate_seq(candidates[0][1], n_fp)
        else:
            initial = None

            n_fp, t_alloc, initial=initial, solver=solver, verbose=True
        )
        best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"fp_n{n_fp}")

        # Quick LSE-GD polish on FP result
        rem = deadline - time.monotonic() - 500
        if rem > 20:
            a_lse, s_lse = lse_gradient_descent(a_fp, min(rem * 0.4, 40.0),
                                                 beta_start=50, beta_end=300, lr_start=0.008)
            best_seq, best_score = check_and_update(a_lse, best_seq, best_score, f"lse_after_fp_n{n_fp}")

    # === Phase 3: Intensive local search on best ===
    print(f"\n=== Phase 3: Local Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Pairwise mass transfer
        rem = deadline - time.monotonic() - 400

            rem = deadline - time.monotonic() - 60
            if rem > 15:
                a_pol, _ = pairwise_mass_transfer(best_seq, min(rem * 0.5, 40.0))

    print(f"\n=== Phase 5: Alt Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:

    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")

        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))

        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
