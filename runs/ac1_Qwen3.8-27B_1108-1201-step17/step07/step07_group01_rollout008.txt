    n = len(a)
    g = np.zeros(n)

    return g

import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]


def normalize(a):
    """Normalize to non-negative simplex (sum=1)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def compute_grad_soft(a, b, T, n):
    """Compute gradient of soft-max smoothed objective using FFT."""
    scaled = b / T
    m = scaled.max()
    w = np.exp(scaled - m)
    w = w / w.sum()
    
    # g_j = 2n * sum_k w[k] * a[k-j] for valid (k,j)
    # = 2n * (a * w_rev)[2n-2-j]

    Wr = np.fft.rfft(w[::-1], fft_size)
    conv = np.fft.irfft(A * Wr, fft_size)
    return 2.0 * n * conv[n-1:2*n-1][::-1]

    j_max = min(n - 1, k_star)
    for j in range(j_min, j_max + 1):

    return 2.0 * n * g

**Alternative: Use the "active set" assumption.** Near the optimum, the set of active peaks (those achieving the maximum) is likely stable. If we assume a fixed active set, the problem becomes:
$$\min \max_{k \in K} (a*a)_k \text{ s.t. } \sum a = 1, a \geq 0$$

where $K$ is the active set. This can be reformulated as:
$$\min t \text{ s.t. } (a*a)_k \leq t \forall k \in K, \sum a = 1, a \geq 0$$

This is a SOCP (second-order cone program) if we write $(a*a)_k \leq t$ as a rotated SOCP constraint! Because $(a*a)_k = a^T T_k a$ is a quadratic form, and the constraint $a^T T_k a \leq t$ is a SOCP constraint if $T_k \succeq 0$.

But as we discussed, $T_k$ is not always PSD. So this doesn't directly work.

However, for the specific case where the peak is at $k = n-1$ (the center), we have:
$(a*a)_{n-1} = \sum_i a_i a_{n-1-i}$

This is $a^T S a$ where $S$ is the anti-identity matrix. $S$ is NOT PSD (it has eigenvalues ±1). So this constraint is non-convex.

But wait: for $k = 0$, $(a*a)_0 = a_0^2$, which IS convex (it's just a squared term).

For $k = 1$, $(a*a)_1 = 2a_0 a_1$, which is a bilinear term (non-convex).

So the problem is inherently non-convex for most peaks. We can't use convex optimization.
