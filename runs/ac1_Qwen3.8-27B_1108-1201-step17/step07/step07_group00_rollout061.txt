def solve_lp_direction(a, top_idx, M):
    n = len(a)
    K = len(top_idx)
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for i in range(K):
        k = top_idx[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = a[lo:hi+1]
    
    # Solve: max sum(g) s.t. A@g <= M, g >= 0
    # In linprog: min -sum(g) s.t. A@g <= M, 0 <= g <= inf
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(K, M), bounds=[(0, None)]*n, method='highs')
    
    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x, 0)
        return g
    return None

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated subgradient for quick warmup
2. LP-based direction finding with HiGHS solver (main work)

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    r = np.nonzero(cond)[0][-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    res = np.maximum(v - theta, 0.0)
    s = res.sum()
    if s > 1e-15:
        res /= s
    return res


def compute_subgradient(a, b, top_k, weight_mode='softmax'):
    """Compute subgradient of max(conv(a,a)) using top_k peaks."""
    n = len(a)
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)

    if weight_mode == 'uniform':
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        vals = b[top_idx]
        rng = vals.max() - vals.min()
        if rng < 1e-12:
            weights = np.ones(len(top_idx)) / len(top_idx)
        else:
            temp = rng * 0.5
            weights = np.exp((vals - vals.max()) / temp)
            ws = weights.sum()
            if ws < 1e-30:
                weights = np.ones(len(top_idx)) / len(top_idx)
            else:
                weights /= ws

    g = np.zeros(n)
    for i in range(len(top_idx)):
        k = int(top_idx[i])
        w = weights[i]
        if w < 1e-14:
            continue
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            g[lo1:hi1 + 1] += w * a[k - np.arange(lo1, hi1 + 1)]
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            g[lo2:hi2 + 1] += w * a[np.arange(lo2, hi2 + 1) - k]
    return g


def run_nesterov(a_init, time_limit, top_k=50, lr_init=0.005, momentum=0.85):
    """Nesterov-accelerated subgradient descent on the simplex."""
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    a_prev = a.copy()
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov look-ahead
        if it > 1:
            momentum_t = momentum * (it ** 0.25) / (1 + it ** 0.25)
            y = a + momentum_t * (a - a_prev)
        else:
            y = a.copy()
        y = np.maximum(y, 0.0)
        s = y.sum()
        if s < 1e-15:
            y = np.ones(n) / n
        else:
            y /= s

        b_y = fast_convolve(y)
        g_y = compute_subgradient(y, b_y, top_k, 'softmax')

        v = momentum * v - lr * g_y
        a_prev = a.copy()
        a_new = project_simplex(y + v)
        a = a_new

        if it % 50 == 0:
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
                lr = min(lr * 1.03, 0.02)
            else:
                no_improve += 1
                if no_improve > 8:
                    lr *= 0.92
                    no_improve = 0
                    if no_improve > 30:
                        a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
                        v = np.zeros(n)
                        no_improve = 0

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    nest iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} lr={lr:.5f} t={elapsed:.0f}s")

    return best_a, best_obj


def solve_lp_direction(a, top_idx, M):
    """Solve LP: max sum(g) s.t. (a*g)[k_i] <= M for i in range(K), g >= 0."""
    n = len(a)
    K = len(top_idx)
    if K == 0:
        return None

    # Build constraint matrix A (K x n)
    A = np.zeros((K, n))
    for i in range(K):
        k = int(top_idx[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = a[lo:hi + 1]

    # Primal: max 1^T g s.t. A g <= M*1, g >= 0
    # linprog minimizes: min -1^T g s.t. A g <= M*1, 0 <= g
    c = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30.0})
    except Exception:
        return None

    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if not np.all(np.isfinite(g)):
            return None
        return g
    return None


def line_search(a, h, current_obj, n_candidates=10):
    """Find best step size t in (0,1] for (1-t)*a + t*h."""
    best_t = 0.05
    best_obj = current_obj
    t_candidates = [0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]

    for t in t_candidates:
        f_trial = (1.0 - t) * a + t * h
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-12:
            continue
        f_trial /= s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t

    return best_t, best_obj


def run_lp_phase(a_init, time_limit, top_k_init=30, top_k_max=200,
                 label="lp", grow_every=40):
    """LP-based optimization with adaptive top_k."""
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    it = 0
    stuck = 0
    top_k = top_k_init
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive top_k growth
        if it % grow_every == 0 and top_k < top_k_max:
            top_k = min(top_k + 5, top_k_max)

        b = fast_convolve(a)
        M = b.max()

        # Get top peaks
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        top_idx = np.sort(top_idx)

        # Solve LP
        g = solve_lp_direction(a, top_idx, M)

        if g is None:
            stuck += 5
            if stuck > 20:
                break
            # Try with fewer peaks
            reduced = top_k // 2
            if reduced < 10:
                break
            top_idx_r = np.argpartition(b, -reduced)[-reduced:]
            g = solve_lp_direction(a, np.sort(top_idx_r), M)
            if g is None:
                if stuck > 30:
                    break
                continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 15:
                # Reset with random perturbation
                a = project_simplex(a + rng.standard_normal(n) * 0.005)
                stuck = 0
                top_k = min(top_k + 10, top_k_max)
            continue

        h = g / S_g

        # Line search
        best_t, best_t_obj = line_search(a, h, best_obj)

        if best_t <= 0.01:
            stuck += 1
            if stuck > 20:
                a = project_simplex(a + rng.standard_normal(n) * 0.003)
                stuck = 0
            continue

        # Update
        a = (1.0 - best_t) * a + best_t * h
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-12:
            a /= s

        cur_obj = fast_convolve(a).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} K={top_k} stuck={stuck} t={elapsed:.0f}s")

    return best_a, best_obj


def coordinate_polish(a_init, time_limit, n_sweep=6):
    """Coordinate descent: try scaling each element."""
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    best_obj = fast_convolve(a).max()
    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.0, 1.2, 1.5, 2.0, 3.0])
    improved = True

    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return a, best_obj
            if a[j] < 1e-15:
                continue
            best_j_val = a[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = a[j] * mult
                if new_val <= 0 or new_val > 2.0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                s = a_trial.sum()
                if s < 1e-12:
                    continue
                a_trial /= s
                obj = fast_convolve(a_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != a[j]:
                a[j] = best_j_val
                s = a.sum()
                if s > 1e-12:
                    a /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved = True
    return a, best_obj

    for nice_n in [512, 768, 1024, 1152, 1280, 1536, 2048, 2500]:
        if abs(nice_n - n_base) / n_base < 0.5:
            candidate_lengths.add(nice_n)

    # Sort by proximity to current best length
    candidate_lengths = sorted(candidate_lengths, key=lambda x: abs(x - n_base))
    # Skip the current length (already optimized)
    candidate_lengths = [cl for cl in candidate_lengths if cl != n_base]

    for target_n in candidate_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t4 = min(rem * 0.25, 200.0)
        if t4 < 30:
            break

        # Interpolate to new length
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        pk = max(20, target_n // 30)
        pk_m = min(200, max(80, target_n // 4))
        print(f"  Trying n={target_n}, budget={t4:.0f}s")
        f4, obj4 = run_lp_phase(f_interp, time_limit=t4,
                                top_k_init=pk, top_k_max=pk_m,
                                label=f"n{target_n}", grow_every=40)
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [n={target_n}]: {score4:.6f}")
        else:
            print(f"  n={target_n}: {score4:.6f} (no improvement)")

    # === Phase 5: Subgradient polish ===
    print(f"\n=== Phase 5: Subgradient polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 150.0)
    if t5 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t5:.0f}s, n={n_b}")
        f5, obj5 = run_subgradient_fast(best_seq, time_limit=t5,
                                        top_k=max(40, n_b // 10), lr_init=0.002)
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [sg_polish]: {score5:.6f}")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.1, 120.0)
    if t6 > 15:
        print(f"  Budget: {t6:.0f}s")
        f6, obj6 = coordinate_polish(best_seq, time_limit=t6)
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [coord]: {score6:.6f}")

    # === Phase 7: Final LP burst ===
    print(f"\n=== Phase 7: Final LP burst (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.35, 200.0)
    if t7 > 20:
        n_b = len(best_seq)
        pk7 = min(300, max(80, n_b // 4))
        print(f"  Budget: {t7:.0f}s, n={n_b}, top_k up to {pk7}")
        f7, obj7 = run_lp_phase(best_seq, time_limit=t7,
                                top_k_init=60, top_k_max=pk7,
                                label="lp_final", grow_every=25)
        score7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(score7) and score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST [lp_final]: {score7:.6f}")

    # === Final verification and cleanup ===
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        print("Warning: final sequence invalid, reverting to parent")
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    # Ensure all values are in valid range
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    if not np.isfinite(final_score):
        best_seq = parent.copy()
        final_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {final_score:.6f} (n={len(best_seq)})")
    print(f"Target: 1.5027")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
