import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0) * target_sum])
    # Project onto {x >= 0, sum(x) = 1} then scale
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n * target_sum
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    return proj * target_sum


def compute_score(f):
    """Compute the objective value for a normalized sequence."""
    n = len(f)
    b = fast_convolve(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * b.max() / (s * s)


def solve_lp_direction(f, n, peak_indices, M):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices.
    Returns g vector or None.
    """
    K = len(peak_indices)
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for row_i in range(K):
        k = peak_indices[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=500000, eps=1e-7)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=100000)
        except Exception:
            return None
    
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val


def get_peaks_above_threshold(b, frac=0.95):
    """Get indices of peaks above frac * max."""
    M = b.max()
    threshold = frac * M
    return np.where(b >= threshold)[0]


def run_lp_optimize(f_init, time_limit, label="lp", peak_frac=0.95, 
                    max_peaks=500, restart_patience=30):

    LP-based optimization with adaptive peak selection.
    """

    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()

    it = 0
    stuck = 0

        M = b.max()

            peaks = np.argpartition(b, -max_peaks)[-max_peaks:]
        
        g = solve_lp_direction(f, n, peaks, M)

            stuck += 1
            if stuck > restart_patience:
                # Perturb and restart
                break
            continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-4:
            stuck += 1
            if stuck > restart_patience:
                break

            if len(peaks) < 2 * n - 1:
                all_peaks = np.arange(len(b))
                g = solve_lp_direction(f, n, all_peaks, M)

                else:
                    continue
            else:
                continue

        best_t = 0.0
        best_t_obj = M

    return best_f, best_b


def mirror_descent_step(f, lr, b, n_top=100):
    """
    Mirror descent step with negative entropy.
    f_new[i] ∝ f[i] * exp(-lr * grad[i])
    where grad is the subgradient of max auto-conv.
    """
    n = len(f)
    
    # Compute subgradient: for each peak k, the gradient contribution to a[i]
    # is a[k-i] for valid i. Weight by importance of each peak.
    M = b.max()
    
    # Use softmax weighting of peaks
    scaled = b / (0.001 * M + 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w /= w_sum
    
    # Focus on top peaks
    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx].copy()
        ws = w_sel.sum()
        if ws > 1e-30:
            w_sel /= ws
        w = np.zeros(len(b))
        w[top_idx] = w_sel
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += w[k] * f[k - idx]
    
    # Mirror descent update
    f_new = f * np.exp(-lr * grad)
    s = f_new.sum()
    if s < 1e-30:
        return f.copy()
    f_new /= s
    return f_new

def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = np.array([0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0])
    
    improved = True
    passes = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        passes += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_b
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_b - 1e-13:
                    best_b = best_j_obj
                    improved = True
    
    return f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, n, peak_indices, M, eps=1e-8):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices.
    Returns g vector or None.
    """
    K = len(peak_indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solved = False
    for solver in [cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=500000, eps=eps)
            solved = True
            break
        except Exception:
            continue

    if not solved or g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def line_search(f, h, M_current, max_steps=15):

        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:

def mirror_descent_optimize(f_init, time_limit, lr_init=0.05, n_top=100, label="md"):

def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent on individual elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = np.array([0.6, 0.75, 0.85, 0.92, 1.08, 1.15, 1.3, 1.5, 2.0])

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_b - 1e-14:
                    best_b = best_j_obj
                    improved = True
        if not improved:
            break

    return f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

        f2, obj2 = run_lp_optimize(best_seq, time_limit=t2, label="lp2",
                                    initial_peak_frac=0.92, final_peak_frac=0.82,
                                    max_peaks=500, restart_patience=25, lp_eps=1e-9)
        score2 = 2.0 * n_b * obj2
        vs2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(vs2) and vs2 < best_score:
            best_score = vs2
            best_seq = f2.copy()
            print(f"  NEW BEST [lp2]: {vs2:.6f}")

    # === Phase 3: Mirror descent polish ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 150.0)
    if t3 > 25:
        n_b = len(best_seq)
        print(f"\n=== Phase 3: Mirror descent (n={n_b}, budget={t3:.0f}s) ===")
        f3, obj3 = mirror_descent_optimize(best_seq, time_limit=t3, lr_init=0.03,
                                            n_top=120, label="md")
        vs3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(vs3) and vs3 < best_score:
            best_score = vs3
            best_seq = f3.copy()
            print(f"  NEW BEST [md]: {vs3:.6f}")

    # === Phase 4: Multi-length exploration ===
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.85, 1.1, 1.3, 1.5]:
        tl = int(n_base * factor)
        tl = max(512, min(2500, tl))
        if tl != n_base and tl not in target_lengths:
            target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t4 = min(rem * 0.12, 120.0)
        if t4 < 20:
            break
        f_interp = interpolate_to_length(best_seq, target_n)
        print(f"\n=== Phase 4: LP n={target_n} (budget={t4:.0f}s) ===")
        f4, obj4 = run_lp_optimize(f_interp, time_limit=t4, label=f"lp_n{target_n}",
                                    initial_peak_frac=0.95, final_peak_frac=0.85,
                                    max_peaks=250, restart_patience=15, lp_eps=1e-8)
        vs4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(vs4) and vs4 < best_score:
            best_score = vs4
            best_seq = f4.copy()
            print(f"  NEW BEST [n{target_n}]: {vs4:.6f}")

    # === Phase 5: Subgradient polish ===
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 120.0)
    if t5 > 20:
        n_b = len(best_seq)
        print(f"\n=== Phase 5: Subgradient (n={n_b}, budget={t5:.0f}s) ===")
        f5, obj5 = subgradient_optimize(best_seq, time_limit=t5, lr_init=0.004,
                                         n_top=100, label="sg")
        vs5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(vs5) and vs5 < best_score:
            best_score = vs5
            best_seq = f5.copy()
            print(f"  NEW BEST [sg]: {vs5:.6f}")

    # === Phase 6: Coordinate descent ===
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 100.0)
    if t6 > 15:
        n_b = len(best_seq)
        print(f"\n=== Phase 6: Coordinate descent (n={n_b}, budget={t6:.0f}s) ===")
        f6, obj6 = coordinate_descent(best_seq, time_limit=t6, label="cd")
        vs6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(vs6) and vs6 < best_score:
            best_score = vs6
            best_seq = f6.copy()
            print(f"  NEW BEST [cd]: {vs6:.6f}")

    # === Phase 7: Final LP burst ===
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.30, 120.0)
    if t7 > 15:
        n_b = len(best_seq)
        print(f"\n=== Phase 7: Final LP (n={n_b}, budget={t7:.0f}s) ===")
        f7, obj7 = run_lp_optimize(best_seq, time_limit=t7, label="lp_final",
                                    initial_peak_frac=0.90, final_peak_frac=0.80,
                                    max_peaks=500, restart_patience=20, lp_eps=1e-9)
        vs7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(vs7) and vs7 < best_score:
            best_score = vs7
            best_seq = f7.copy()
            print(f"  NEW BEST [lp_final]: {vs7:.6f}")

    # === Phase 8: Final coordinate descent ===
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.20, 60.0)
    if t8 > 10:
        print(f"\n=== Phase 8: Final CD (budget={t8:.0f}s) ===")
        f8, obj8 = coordinate_descent(best_seq, time_limit=t8, label="cd_final")
        vs8 = float(evaluate_sequence(f8.tolist()))
        if np.isfinite(vs8) and vs8 < best_score:
            best_score = vs8
            best_seq = f8.copy()
            print(f"  NEW BEST [cd_final]: {vs8:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
