2. Smooth subgradient descent with adaptive step size and top-k weighting
3. Coordinate descent with pairwise optimal transfer
4. Multiple sequence lengths exploration
The LP fictitious play is the key novelty: at each step, solve an LP to find
the optimal "response" sequence, then update the current sequence toward it.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Algorithm from Duchi et al. 2008
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    # Ensure exact sum
    s = w.sum()
    if s > 1e-15:
        w = w / s
    else:
        w = np.ones(n) / n
    return w


def build_conv_matrix(f):
    """Build the convolution matrix A where A[k, j] = f[k-j] if valid."""
    n = len(f)
    size = 2 * n - 1
    A = np.zeros((size, n))
    for k in range(size):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    return A


def solve_best_response_lp(f, M):
    """
    Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0.
    Returns the optimal g (unnormalized).
    """
    n = len(f)
    size = 2 * n - 1
    
    # Build constraint matrix efficiently
    A = np.zeros((size, n))
    for j in range(n):
        # A[k, j] = f[k-j] for valid k-j
        lo = max(0, j)
        hi = min(size - 1, j + n - 1)
        for k in range(lo, hi + 1):
            A[k, j] = f[k - j]
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b = np.full(size, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x
        else:
            # Fallback: return current f
            return f.copy()
    except Exception:
        return f.copy()


def topk_subgradient(f, k=10, T=0.01):
    """
    Compute a weighted subgradient of max(conv) using top-k peaks.
    Returns gradient vector of shape (n,).
    """
    n = len(f)
    conv = fast_convolve(f)
    
    # Compute softmax weights over the convolution
    shifted = conv - conv.max()
    w = np.exp(shifted / T)
    w = w / w.sum()
    
    # Gradient: 2 * (w * f) cross-correlated
    # d(conv)_k / df_i = 2 * f_{k-i}
    # So grad_i = 2 * sum_k w_k * f_{k-i}
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    F = rfft(f, fft_size)
    grad = 2.0 * irfft(np.conj(W) * F, fft_size)[:n]
    
    return grad, conv.max()


def fictitious_play_lp(f0, time_limit, verbose=True):
    """
    LP-based fictitious play for minimizing max(auto-convolution).
    """
    n = len(f0)
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        return f0, compute_score(f0)
    f = f / s
    
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    # Step size schedule
    alpha = 0.5
    beta = 10.0
    t_max = 0.3
    stagnation_count = 0
    stall_threshold = 50
    
    # Track running average
    f_avg = f.copy()
    avg_count = 0
    
    iters = 0
    A_cached = None
    M_cached = -1
    
    # Precompute the convolution matrix structure (indices)
    # A[k, j] = f[k-j] if 0 <= k-j < n
    # We can avoid rebuilding A by computing (f*g) directly via FFT
    # But for the LP we need the matrix. Let's build it once and update f values.
    
    # Actually, let's use FFT for the convolution in the LP constraints evaluation
    # and only build the matrix when solving the LP.
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current max convolution
        conv = fast_convolve(f)
        M = conv.max()
        current_score = 2.0 * n * M
        
        # Update step size
        k = iters
        t = min(t_max, alpha / (k + beta))
        
        # Decrease step size if stagnating
        if stagnation_count > stall_threshold:
            t *= 0.5
            alpha *= 0.9
            if alpha < 0.01:
                alpha = 0.01
            stagnation_count = 0
        
        # Solve LP for best response
        if iters % 1 == 0:  # Solve LP every iteration
            g = solve_best_response_lp(f, M)
            g_sum = g.sum()
            if g_sum < 1e-15:
                g = f.copy()
                g_sum = 1.0
            g = g / g_sum
        
        # Update: f <- (1-t)*f + t*g
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        f_sum = f_new.sum()
        if f_sum < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / f_sum
        
        # Check improvement
        new_score = compute_score(f_new)
        
        # Update running average
        f_avg = f_avg + (f_new - f_avg) / (avg_count + 1)
        avg_count += 1
        
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_f = f_new.copy()
            stagnation_count = 0
            if verbose and iters % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    FP iter={iters}, score={best_score:.8f}, t={t:.4f}, elapsed={elapsed:.0f}s")
        else:
            stagnation_count += 1
        
        # Also check the average
        if avg_count % 100 == 0 and avg_count > 100:
            avg_score = compute_score(f_avg)
            if avg_score < best_score - 1e-12:
                best_score = avg_score
                best_f = f_avg.copy()
                if verbose:
                    print(f"    FP avg improved at iter={iters}: {best_score:.8f}")
        
        f = f_new
    
    return best_f, best_score


def smooth_gradient_descent(f0, time_limit, T=50.0, verbose=True):
    """
    Smooth gradient descent with adaptive step size using softmax-weighted subgradient.
    """
    n = len(f0)
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        return f0, compute_score(f0)
    f = f / s
    
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    eta = 0.02
    patience = 0
    best_patience_score = best_score
    
    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute smoothed subgradient
        grad, M = topk_subgradient(f, T=T)
        
        # Center the gradient (remove mean component to maintain sum=1)
        grad_centered = grad - grad.mean()
        
        # Gradient step
        f_new = f - eta * grad_centered
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        
        score = compute_score(f_new)
        
        # Adaptive step size
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            patience = 0
            eta = min(eta * 1.02, 0.1)
            if verbose and iters % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    SGD iter={iters}, score={best_score:.8f}, eta={eta:.4f}, elapsed={elapsed:.0f}s")
        else:
            patience += 1
            if patience > 50:
                eta *= 0.7
                patience = 0
                # Try to recover
                f_new = f  # revert
        
        f = f_new
        
        # Periodically try with different T
        if iters % 500 == 0:
            T = T * 1.5 if T < 500 else T * 0.7
    
    return best_f, best_score


def coordinate_descent(f0, time_limit, n_pairs=500, verbose=True):
    """
    Coordinate descent with pairwise optimal transfer.
    """
    n = len(f0)
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        return f0, compute_score(f0)
    f = f / s
    
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if f[i] < 1e-14:
                continue
            
            # Try multiple transfer fractions
            base = f[i]
            best_local = best_score
            best_frac = 0.0
            
            for frac in [0.1, 0.25, 0.5, 0.75, 0.05, 0.15]:
                t_val = frac * base
                if t_val >= f[i] - 1e-15:
                    t_val = f[i] * 0.49
                
                f_trial = f.copy()
                f_trial[i] -= t_val
                f_trial[j] += t_val
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                sc = compute_score(f_trial)
                
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_frac = t_val
                
                # Also try reverse
                if f[j] > 1e-14:
                    t_val2 = frac * f[j]
                    f_trial2 = f.copy()
                    f_trial2[j] -= t_val2
                    f_trial2[i] += t_val2
                    f_trial2 = np.maximum(f_trial2, 0)
                    ss2 = f_trial2.sum()
                    if ss2 < 1e-15:
                        continue
                    f_trial2 = f_trial2 / ss2
                    sc2 = compute_score(f_trial2)
                    if sc2 < best_local - 1e-14:
                        best_local = sc2
                        best_frac = -t_val2
            
            if best_frac != 0.0:
                f[i] -= best_frac
                f[j] += best_frac
                f = np.maximum(f, 0)
                ss = f.sum()
                f = f / ss
                best_score = best_local
                best_f = f.copy()
                improved = True
        
        if not improved and iters > 20:
            # Perturb
            noise = rng.standard_normal(n) * 0.002
            f = best_f + noise
            f = np.maximum(f, 0)
            f = f / f.sum()
            best_score = compute_score(f)
    
    return best_f, best_score


def multiplicative_weights(f0, time_limit, T=20.0, verbose=True):
    """
    FTRL with entropy regularization (multiplicative weights / mirror descent).
    Uses top-k weighted subgradient.
    """
    n = len(f0)
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        return f0, compute_score(f0)
    f = f / s
    
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    # Accumulated gradients
    G = np.zeros(n)
    eta = 0.1
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient
        grad, M = topk_subgradient(f, T=T)
        G += grad
        
        # FTRL update with entropy: f_new(i) = exp(-eta * G(i)) / Z
        # But this can become very peaked. Use reference to initial point:
        # f_new(i) = f0(i) * exp(-eta * G(i)) / Z
        
        # Actually, standard multiplicative weights:
        # f_new(i) proportional to f_old(i) * exp(-eta * grad(i))
        # This is simpler and maintains non-negativity automatically.
        
        scaled = f * np.exp(-eta * grad)
        scaled = np.maximum(scaled, 0)
        ss = scaled.sum()
        if ss < 1e-300:
            f_new = f.copy()
        else:
            f_new = scaled / ss
        
        score = compute_score(f_new)
        
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            eta = min(eta * 1.05, 1.0)
            if verbose and iters % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"    MW iter={iters}, score={best_score:.8f}, eta={eta:.4f}, elapsed={elapsed:.0f}s")
        else:
            eta *= 0.95
            f_new = f  # revert
        
        f = f_new
        
        # Occasionally change T
        if iters % 300 == 0:
            T = max(5.0, T * 0.8)
    
    return best_f, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def lbgfs_polish(f0, time_limit, T=300.0, verbose=True):
    """L-BFGS polish using softmax smoothing in log-space."""
    n = len(f0)
    a = np.maximum(f0, 1e-15)
    a = a / a.sum()
    z0 = np.log(a)
    
    t0 = time.monotonic()
    best_score = compute_score(a)
    best_a = a.copy()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        s = exp_z.sum()
        if s < 1e-300:
            a_local = np.ones(n) / n
        else:
            a_local = exp_z / s
        conv = fast_convolve(a_local)
        shifted = conv / T
        max_shifted = shifted.max()
        log_sum_exp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
        return (2.0 * n / T) * np.exp(T * log_sum_exp)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        s = exp_z.sum()
        if s < 1e-300:
            a_local = np.ones(n) / n
        else:
            a_local = exp_z / s
        conv = fast_convolve(a_local)
        shifted = conv / T
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        w = w / w.sum()
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a_local, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        S = np.dot(grad_a, a_local)
        grad_z = a_local * (grad_a - S)
        return (2.0 * n / T) * grad_z
    
    try:
        res = minimize(func, z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxfun': 50000, 'disp': False})
        if res.x is not None:
            z_out = res.x
            z_max = z_out.max()
            exp_z = np.exp(z_out - z_max)
            a_out = exp_z / exp_z.sum()
            sc = compute_score(a_out)
            if sc < best_score:
                best_score = sc
                best_a = a_out
    except Exception:
        pass
    
    return best_a, best_score


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = compute_score(a_norm)
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization:
    1. LP-based fictitious play (primary)
    2. Multiplicative weights / mirror descent
    3. Smooth gradient descent
    4. Coordinate descent
    5. L-BFGS polish
    6. Try different sequence lengths
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        t = np.linspace(0, 2 * np.pi * 3, n_init)
        parent = parent + 0.05 * np.cos(t)
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        best_score = compute_score(parent)
        best_seq = parent.copy()
        print(f"Initialized: n={n_init}, score={best_score:.8f}")
    
    n_parent = len(parent)
    
    # === Phase 1: LP-based Fictitious Play ===
    print(f"\n=== Phase 1: LP Fictitious Play (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 800
    if rem > 30:
        t_alloc = min(rem * 0.85, 700.0)
        print(f"  FP LP n={n_parent}, budget={t_alloc:.0f}s")
        a_fp, sc_fp = fictitious_play_lp(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_fp, "fp_lp", best_seq, best_score)
    
    # === Phase 2: Multiplicative Weights ===
    print(f"\n=== Phase 2: Multiplicative Weights (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_alloc = min(rem * 0.5, 300.0)
        print(f"  MW n={n_parent}, budget={t_alloc:.0f}s")
        a_mw, sc_mw = multiplicative_weights(best_seq, t_alloc, T=30.0, verbose=True)
        best_seq, best_score = check_and_update(a_mw, "mw", best_seq, best_score)
    
    # === Phase 3: Smooth Gradient Descent ===
    print(f"\n=== Phase 3: Smooth GD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(rem * 0.5, 250.0)
        print(f"  SGD n={n_parent}, budget={t_alloc:.0f}s")
        a_sgd, sc_sgd = smooth_gradient_descent(best_seq, t_alloc, T=50.0, verbose=True)
        best_seq, best_score = check_and_update(a_sgd, "sgd", best_seq, best_score)
    
    # === Phase 4: Try longer sequences ===
    print(f"\n=== Phase 4: Longer Sequences (best={best_score:.8f}) ===")
    for alt_n in [1024, 1536]:
        rem = deadline - time.monotonic() - 200
        if rem < 60:
            break
        t_alloc = min(rem * 0.35, 120.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  FP LP n={alt_n}, budget={t_alloc:.0f}s")
        a_alt, sc_alt = fictitious_play_lp(init_alt, t_alloc, verbose=False)
        best_seq, best_score = check_and_update(a_alt, f"fp_n{alt_n}", best_seq, best_score)
        
        # Quick SGD on the longer sequence
        rem2 = deadline - time.monotonic() - 150
        if rem2 > 20:
            t_alloc2 = min(rem2 * 0.3, 50.0)
            a_alt2, _ = smooth_gradient_descent(a_alt, t_alloc2, T=30.0)
            best_seq, best_score = check_and_update(a_alt2, f"sgd_n{alt_n}", best_seq, best_score)
    
    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 100
    if rem > 20:
        t_alloc = min(rem * 0.6, 150.0)
        print(f"  CD n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t_alloc, n_pairs=400, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "coord_desc", best_seq, best_score)
    
    # === Phase 6: L-BFGS Polish ===
    print(f"\n=== Phase 6: L-BFGS Polish (best={best_score:.8f}) ===")
    for T_val in [200, 1000]:
        rem = deadline - time.monotonic() - 50
        if rem < 15:
            break
        t_alloc = min(rem * 0.4, 40.0)
        print(f"  L-BFGS T={T_val}, budget={t_alloc:.0f}s")
        a_lb, _ = lbgfs_polish(best_seq, t_alloc, T=T_val)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T_val}", best_seq, best_score)
    
    # === Phase 7: Final Coordinate Descent ===
    rem = deadline - time.monotonic() - 20
    if rem > 10:
        t_alloc = min(rem * 0.7, 30.0)
        a_cd2, _ = coordinate_descent(best_seq, t_alloc, n_pairs=1000)
        best_seq, best_score = check_and_update(a_cd2, "final_cd", best_seq, best_score)
    
    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
    except Exception:
        pass
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
