```python
    """Compute auto-convolution of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_exp_gradient(a, T):
    """
    Compute gradient of F_T(a) = sum_k exp(b_k/T) where b = convolve(a,a).
    Returns gradient vector of length n.
    """

    b = compute_convolution(a)  # length 2n-1
    w = np.exp(b / T)  # length 2n-1
    # Gradient: (2/T) * cross_correlation(a, w)
    # cross_correlation(a, w)_j = sum_m a_m * w_{m+j}

    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1)
    return np.maximum(v - theta, 0.0)
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exponential surrogate
gradient descent with adaptive temperature scheduling, momentum,
LP-based refinement, and multi-length exploration.

Key innovations:
- Exponential surrogate provides smooth gradients (unlike subgradient)
- FFT-based gradient computation (O(n log n) per step)
- Adaptive temperature annealing
- LP refinement for principled peak reduction
- Multi-length exploration with interpolation
"""

import numpy as np
import time
from scipy.fft import fft, ifft, rfft, irfft
from scipy.optimize import linprog


def compute_convolution(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_exp_gradient(a, T):
    n = len(a)
    b = compute_convolution(a)
    max_b = float(b.max())
    w = np.exp(b / T)
    fft_size = 1 << (3 * n - 2).bit_length()
    A = fft(a, fft_size)
    W = fft(w, fft_size)
    grad = np.real(ifft(np.conj(A) * W, fft_size))[:n]
    grad *= 2.0 / T
    return grad, max_b


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def exp_surrogate_optimize(a0, time_limit, T_ratio_init=5.0, T_ratio_min=50.0,
                           lr_init=0.003, momentum=0.9,
                           restart_interval=3000, seed=42, verbose=False):
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = compute_convolution(a)
    max_b_init = float(b.max())
    T = max_b_init / T_ratio_init
    T_min = max_b_init / T_ratio_min

    best_a = a.copy()
    best_obj = max_b_init
    v = np.zeros(n)
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        grad, cur_max = compute_exp_gradient(a, T)
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-15:
            break

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        b_new = compute_convolution(a_new)
        obj_new = float(b_new.max())

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it += 1

        if it % 500 == 0:
            T = max(T_min, T * 0.97)

        if no_improve >= restart_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = max(lr * 0.5, 1e-6)
            no_improve = 0
        elif no_improve >= restart_interval // 2:
            lr *= 0.95
        elif no_improve >= restart_interval // 5:
            lr *= 0.99

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    iter={it}, score={score:.8f}, T/T_min={T/T_min:.2f}, lr={lr:.2e}, t={elapsed:.0f}s")

    return best_a, best_obj


def lp_refine(a0, max_iters=20, time_limit=60.0, verbose=False):
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = float(compute_convolution(a).max())

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = compute_convolution(a)
        M = float(b.max())

        # Build LP: max 1^T g  s.t.  A_lp @ g <= M, g >= 0
        # Row k: sum_{j} a_j * g_{k-j} <= M  (for valid indices)
        num_constraints = 2 * n - 1
        A_lp = np.zeros((num_constraints, n))
        for k in range(num_constraints):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A_lp[k, k - j] = a[j]

        c = -np.ones(n)
        try:
            res = linprog(c, A_ub=A_lp, b_ub=np.full(num_constraints, M),
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': max(1.0, time_limit - (time.monotonic() - t0))})
            if not res.success:
                break
            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            g_norm = g / g_sum
        except Exception:
            break

        # Line search for blending
        found = False
        for t in [0.3, 0.2, 0.15, 0.1, 0.05, 0.4, 0.5, 0.7, 1.0, 0.02, 0.01]:
            blend = (1.0 - t) * best_a + t * g_norm
            blend = np.maximum(blend, 0.0)
            ss = blend.sum()
            if ss < 1e-10:
                continue
            blend = blend / ss
            obj_blend = float(compute_convolution(blend).max())
            if obj_blend < best_obj - 1e-14:
                best_obj = obj_blend
                best_a = blend.copy()
                found = True
                break

        if not found:
            break
        if verbose:
            print(f"    LP iter {it}: score={2*n*best_obj:.8f}")

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp
```
