import numpy as np

    size = 2 * n - 1

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def multi_peak_gradient(a, b, threshold_frac):
    """Compute subgradient using all peaks within threshold of max."""
    n = len(a)
    b_max = b.max()
    if b_max < 1e-300:
        return np.zeros(n), 1
    threshold = b_max * (1.0 - threshold_frac)
    active_peaks = np.where(b >= threshold)[0]
    if len(active_peaks) == 0:

        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += a[k - idx]

            s1 = score_from_a(a1)

            s2 = score_from_a(a2)

            else:

        a_best_local = np.maximum(a_best_local, 0)
        a_best_local = a_best_local / a_best_local.sum()
        s_best_local = score_from_a(a_best_local)
        
        if s_best_local < current_best_score - 1e-14:
            return a_best_local, s_best_local, True
        return current_a, current_best_score, False

    no_improve_rounds = 0
    max_no_improve = 5

    while time.monotonic() - t0 < time_limit and no_improve_rounds < max_no_improve:
        no_improve_rounds = 0
        round_improved = False
        # Process coordinates in order of contribution to peaks
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        important = np.arange(js, je + 1)
        contribs = a[important] * a[peak_k - important]
        order = important[np.argsort(contribs)[::-1]]
        
        # Add some random coordinates
        n_random = max(1, n // 10)
        rng = np.random.default_rng(hash((round_no if 'round_no' in dir() else 0)) % 2**31)
        random_coords = rng.choice(n, size=n_random, replace=False)
        order = np.concatenate([order, random_coords])
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            a_new, score_new, improved = optimize_coordinate(int(i), a, best_score)
            if improved:
                a = a_new
                best_score = score_new
                best_a = a.copy()
                round_improved = True

        if not round_improved:
            no_improve_rounds += 1
            # Try all coordinates with smaller search range
            if no_improve_rounds <= 3:
                for i in range(n):
                    if time.monotonic() - t0 > time_limit:
                        break
                    old_val = a[i]
                    if old_val < 1e-15:
                        continue
                    for factor in [0.99, 0.995, 1.005, 1.01]:
                        a_trial = a.copy()
                        a_trial[i] = old_val * factor
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = score_from_a(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            break

        if time.monotonic() % 30 < 1:
            print(f"    {label} GS ({time.monotonic()-t0:.0f}s): best={best_score:.7f} "
                  f"no_improve={no_improve_rounds}")

    return best_a, best_score


def nesterov_log_space(a_init, time_limit, T_start=20.0, T_end=0.5,
                       lr_start=0.05, lr_end=0.001, max_iters=100000,
                       label=""):
    """Nesterov-accelerated gradient descent in log-space with soft-max smoothing."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    z = np.log(a)
    z = z - z.mean()
    v = np.zeros(n)  # velocity for Nesterov
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / max_iters)
        # Cosine annealing
        T = T_end + 0.5 * (T_start - T_end) * (1 + np.cos(np.pi * progress))
        lr = lr_end + 0.5 * (lr_start - lr_end) * (1 + np.cos(np.pi * progress))
        beta = 0.9 + 0.05 * progress  # momentum increases over time

        # Compute a from z
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()

        # Compute gradient
        b = fast_convolve(a)
        b_max = b.max()
        current_score = 2.0 * n * b_max

        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()

        # Soft-max weights
        scaled = b / T
        m = scaled.max()
        exp_scaled = np.exp(scaled - m)
        w = exp_scaled / exp_scaled.sum()

        # Gradient w.r.t. a: 4n * cross_corr(w, a)
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad_a = 4.0 * n * correl[:n]

        # Gradient w.r.t. z: a * (grad_a - dot(a, grad_a))
        mean_grad = np.dot(a, grad_a)
        grad_z = a * (grad_a - mean_grad)

        # Nesterov update
        v = beta * v - lr * grad_z
        z = z + v

        if it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} NG iter={it} ({elapsed:.0f}s): T={T:.2f} lr={lr:.5f} "
                  f"score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def peak_shave_aggressive(a_init, time_limit, label=""):
    """Aggressive peak shaving targeting top-3 peaks simultaneously."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    for step in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_sorted_idx = np.argsort(b)[::-1][:5]
        
        improved = False
        for peak_k in b_sorted_idx:
            peak_k = int(peak_k)
            if b[peak_k] * 2.0 * n < best_score * 0.95:
                continue  # This peak isn't near the max

            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_idx = np.argsort(contribs)[::-1][:8]

            for ci in top_idx:
                i = int(idx[ci])
                j = peak_k - i
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for factor in [0.97, 0.95, 0.93, 0.9, 0.85]:
                    a_trial = a.copy()
                    old_val = a_trial[i]
                    a_trial[i] = old_val * factor
                    diff = old_val - a_trial[i]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask].copy()
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = score_from_a(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try paired reduction
            peak_k = int(b_sorted_idx[0])
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_idx = np.argsort(contribs)[::-1][:5]
            
            found = False
            for ci1 in range(len(top_idx)):
                for ci2 in range(ci1 + 1, len(top_idx)):
                    i1 = int(idx[top_idx[ci1]])
                    i2 = int(idx[top_idx[ci2]])
                    if i1 == i2:
                        continue
                    for f1 in [0.95, 0.9]:
                        for f2 in [0.95, 0.9]:
                            a_trial = a.copy()
                            a_trial[i1] *= f1
                            a_trial[i2] *= f2
                            a_trial = np.maximum(a_trial, 0)
                            a_trial = a_trial / a_trial.sum()
                            sc = score_from_a(a_trial)
                            if sc < best_score - 1e-15:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                found = True
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break
            if not found:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n + 1)
    new_x = np.linspace(0, 1, new_n + 1)
    f_padded = np.concatenate([f_norm, [0]])
    f_interp = np.interp(new_x, old_x, f_padded)[:new_n]
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using the actual evaluate_sequence function."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining multi-peak mirror descent, golden-section
    coordinate search, Nesterov log-space gradient descent, and aggressive peak
    shaving. Uses cosine annealing, adaptive thresholds, and periodic restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initialized uniform: n={n_init}, score={best_score:.7f}")

    n_best = len(best_seq)
    print(f"\nStarting optimization: n={n_best}, score={best_score:.7f}")

    # === Phase 1: Multi-peak Mirror Descent ===
    print("\n=== Phase 1: Multi-Peak Mirror Descent ===")
    rem = deadline - time.monotonic() - 800
    if rem > 60:
        t_alloc = min(rem * 0.4, 300.0)
        print(f"  Budget: {t_alloc:.0f}s, n={n_best}")
        a_md, _ = mirror_descent_multi_peak(
            best_seq, t_alloc, label="MD1",
            T_threshold_start=0.03, T_threshold_end=0.003,
            eta_start=0.02, eta_end=0.003
        )
        best_seq, best_score = check_and_update(a_md, "MD1", best_seq, best_score)

    # === Phase 2: Nesterov Log-Space Gradient Descent ===
    print(f"\n=== Phase 2: Nesterov Log-Space (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 600
        if rem > 60:
            t_alloc = min(rem * 0.4, 250.0)
            print(f"  Budget: {t_alloc:.0f}s, T=[30,1]")
            a_ng, _ = nesterov_log_space(
                best_seq, t_alloc, T_start=30.0, T_end=1.0,
                lr_start=0.04, lr_end=0.002, label="NG1"
            )
            best_seq, best_score = check_and_update(a_ng, "NG1", best_seq, best_score)

    # === Phase 3: Golden-Section Coordinate Descent ===
    print(f"\n=== Phase 3: Golden-Section CD (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 40:
            t_alloc = min(rem * 0.45, 200.0)
            print(f"  Budget: {t_alloc:.0f}s")
            a_gs, _ = golden_section_coordinate(best_seq, t_alloc, label="GS1", n_factors=18)
            best_seq, best_score = check_and_update(a_gs, "GS1", best_seq, best_score)

    # === Phase 4: Aggressive Peak Shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 40:
            t_alloc = min(rem * 0.45, 150.0)
            print(f"  Budget: {t_alloc:.0f}s")
            a_ps, _ = peak_shave_aggressive(best_seq, t_alloc, label="PS1")
            best_seq, best_score = check_and_update(a_ps, "PS1", best_seq, best_score)

    # === Phase 5: Second round - Nesterov with lower T ===
    print(f"\n=== Phase 5: Nesterov Low-T (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 40:
            t_alloc = min(rem * 0.5, 120.0)
            print(f"  Budget: {t_alloc:.0f}s, T=[5,0.3]")
            a_ng2, _ = nesterov_log_space(
                best_seq, t_alloc, T_start=5.0, T_end=0.3,
                lr_start=0.02, lr_end=0.0005, label="NG2"
            )
            best_seq, best_score = check_and_update(a_ng2, "NG2", best_seq, best_score)

    # === Phase 6: Mirror Descent refinement ===
    print(f"\n=== Phase 6: MD Refinement (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 120
        if rem > 30:
            t_alloc = min(rem * 0.5, 80.0)
            print(f"  Budget: {t_alloc:.0f}s")
            a_md2, _ = mirror_descent_multi_peak(
                best_seq, t_alloc, label="MD2",
                T_threshold_start=0.01, T_threshold_end=0.001,
                eta_start=0.008, eta_end=0.001
            )
            best_seq, best_score = check_and_update(a_md2, "MD2", best_seq, best_score)

    # === Phase 7: Random restart + re-optimize ===
    print(f"\n=== Phase 7: Restart & Re-optimize (best={best_score:.7f}) ===")
    if best_seq is not None:
        for restart in range(3):
            rem = deadline - time.monotonic() - 60
            if rem < 40:
                break
            # Perturb
            scale = 0.05 * (restart + 1)
            rng_r = np.random.default_rng(seed + 100 + restart)
            a_perturbed = best_seq + rng_r.standard_normal(len(best_seq)) * scale
            a_perturbed = np.maximum(a_perturbed, 0)
            a_perturbed = a_perturbed / a_perturbed.sum()
            
            t_alloc = min(rem * 0.3, 50.0)
            print(f"  Restart {restart}: scale={scale:.3f}, budget={t_alloc:.0f}s")
            a_r, _ = mirror_descent_multi_peak(
                a_perturbed, t_alloc, label=f"R{restart}",
                T_threshold_start=0.02, T_threshold_end=0.003,
                eta_start=0.015, eta_end=0.002
            )
            best_seq, best_score = check_and_update(a_r, f"restart{restart}", best_seq, best_score)

    # === Phase 8: Golden-section final polish ===
    print(f"\n=== Phase 8: GS Final Polish (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 30:
            t_alloc = min(rem * 0.5, 60.0)
            print(f"  Budget: {t_alloc:.0f}s")
            a_gs2, _ = golden_section_coordinate(best_seq, t_alloc, label="GS2", n_factors=15)
            best_seq, best_score = check_and_update(a_gs2, "GS2", best_seq, best_score)

    # === Phase 9: Alternative lengths ===
    print(f"\n=== Phase 9: Alternative Lengths (best={best_score:.7f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = [int(n_best * 1.25), int(n_best * 1.5), int(n_best * 0.75)]
        alt_lengths = [n for n in alt_lengths if 256 <= n <= 4096 and n != len(best_seq)]
        alt_lengths = list(dict.fromkeys(alt_lengths))[:2]  # Remove dupes, keep order

        for alt_n in alt_lengths:
            rem = deadline - time.monotonic() - 30
            if rem < 30:
                break
            t_alloc = min(rem * 0.4, 60.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            print(f"  n={alt_n}, budget={t_alloc:.0f}s")
            a_alt, _ = mirror_descent_multi_peak(
                init_alt, t_alloc, label=f"alt{alt_n}",
                T_threshold_start=0.02, T_threshold_end=0.002,
                eta_start=0.02, eta_end=0.003
            )
            best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}", best_seq, best_score)

    # === Phase 10: Final micro-polish ===
    print(f"\n=== Phase 10: Final Micro-Polish (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            # Very short golden section
            t_alloc = min(rem * 0.6, 15.0)
            a_fin, _ = golden_section_coordinate(best_seq, t_alloc, label="finGS", n_factors=12)
            best_seq, best_score = check_and_update(a_fin, "finalGS", best_seq, best_score)

        rem = deadline - time.monotonic() - 3
        if rem > 3:
            a_fin2, _ = peak_shave_aggressive(best_seq, min(rem * 0.7, 8.0), label="finPS")
            best_seq, best_score = check_and_update(a_fin2, "finalPS", best_seq, best_score)

    # Fallback
    if best_seq is None:
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
