import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
from scipy.optimize import minimize
from time import monotonic
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via real FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.GUROBI, cp.MOSEK, cp.SCS, cp.ECOS, cp.GLPK, cp.CBC]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver]
    for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK, cp.GUROBI]:
        if alt not in solvers_to_try:
            solvers_to_try.append(alt)
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def compute_gs_target(n, N):
    """Compute target |A| for GS iterations aiming at flat linear convolution."""
    rect = np.zeros(N)
    rect[:2 * n - 1] = 1.0
    D = fft(rect)
    target_mag = np.sqrt(np.maximum(np.abs(D), 1e-30))
    target_mag[0] = 1.0  # DC must be 1 (sum a = 1)
    return target_mag


def gs_spectral(a0, time_limit, alpha_init=1.0, alpha_decay=0.9995, verbose=True):
    """Gerchberg-Saxton spectral flattening with correct Dirichlet-kernel target."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    N = 1 << (2 * n - 1).bit_length()
    target_mag = compute_gs_target(n, N)

    alpha = alpha_init
    t0 = monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    it = 0
    no_improve = 0

    while monotonic() - t0 < time_limit:
        it += 1
        A = fft(a, N)
        mag = np.maximum(np.abs(A), 1e-30)
        # Relaxed projection to target magnitude
        ratio = (target_mag / mag) ** alpha
        A_new = A * ratio
        a_new = np.real(ifft(A_new, N))[:n]
        a = np.maximum(a_new, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Decay alpha (more relaxed over time)
        alpha *= alpha_decay
        if alpha < 0.01:
            alpha = 0.01

        # If stuck, perturb and reset alpha
        if no_improve > 200:
            a = best_a + np.random.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            alpha = alpha_init
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = monotonic() - t0
            print(f"    GS iter={it} ({elapsed:.0f}s): score={best_score:.8f} alpha={alpha:.4f}")

    return best_a, best_score


def damped_lp_refine(a0, time_limit, solver=None, verbose=True):
    """Damped alternating LP refinement with adaptive step size."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()

    t0 = monotonic()

        it += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        # Adaptive step size
        alpha = min(1.0, 3.0 / (it + 10))
        a_candidate = (1 - alpha) * a + alpha * b_new
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s < 1e-15:
            break
        a_candidate = a_candidate / s
        score = 2.0 * n * fast_convolve(a_candidate).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_candidate.copy()
        a = a_candidate

        # Stagnation handling
        if it % 500 == 0 and it > 200:
            a = best_a + np.random.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and it % 100 == 0:

def subgradient_refine(a0, time_limit, T_init=0.002, T_decay=0.9998, momentum=0.85,
                       top_k=30, lr_init=0.003, verbose=True):
    """Adaptive subgradient with softmax peak weighting and line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    lr = lr_init
    T = T_init
    t0 = monotonic()
    step = 0
    no_improve = 0

    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = (pv - pv.max()) / T
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = momentum * v - lr * sg
        a_candidate = a + v

        # Project onto simplex
        u = np.sort(a_candidate)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_candidate - theta, 0.0)
        else:
            a_new = np.ones(n) / n
        a_new = a_new / a_new.sum()

        score = 2.0 * n * fast_convolve(a_new).max()
        step += 1

        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        T *= T_decay
        if T < 1e-5:
            T = 1e-5
        lr *= 0.9995

        if no_improve > 500:
            a = best_a + np.random.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 3, 0.004)
            T = max(T * 10, 0.005)
            no_improve = 0

        if verbose and step % 2000 == 0:
            elapsed = monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f} T={T:.6f} lr={lr:.6f}")

    return best_a, best_score


def peak_equalize(a0, time_limit, n_peaks=5, verbose=True):
    """Iteratively equalize the top peaks of the convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    stall = 0

    for it in range(100000):
        if monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        top_k = np.argsort(b)[::-1][:n_peaks]
        target_val = b[top_k].mean()

        # Find direction to equalize peaks
        # For each peak, compute the gradient and move against it proportional to (b[k] - target)
        direction = np.zeros(n)
        for k in top_k:
            if b[k] <= target_val:
                continue
            weight = b[k] - target_val
            js = max(0, int(k) - n + 1)
            je = min(n - 1, int(k))
            idx = np.arange(js, je + 1)
            direction[idx] -= 2.0 * weight * a[int(k) - idx]

        # Try multiple step sizes
        improved = False
        for step_factor in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_trial = a + step_factor * direction
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_trial.copy()
                a = a_trial
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall > 100:
                # Perturb and retry
                a = best_a + np.random.standard_normal(n) * 0.001
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if monotonic() - t0 > time_limit * 0.9:
                    break

        if verbose and it % 2000 == 0:
            elapsed = monotonic() - t0
            print(f"    PeakEq iter={it} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T=50, verbose=True):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a_z = exp_z / exp_z.sum()
        b = fast_convolve(a_z)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse / T

    def jac(z):
        if stopped[0]:
            return np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a_z = exp_z / exp_z.sum()
        b = fast_convolve(a_z)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()

        grad_a = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a_z[k - idx]

        grad_a *= 2.0 * n
        S = np.dot(grad_a, a_z)
        grad_z = a_z * (grad_a - a_z * S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': 45000, 'disp': False}
        )
        if not stopped[0]:
            z_shifted = res.x - res.x.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0
    except Exception:
        a_out = a0

    if verbose:
        elapsed = monotonic() - t0
        score = 2.0 * n * fast_convolve(a_out).max()
        print(f"    L-BFGS ({elapsed:.0f}s): score={score:.8f}")

    return a_out


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
