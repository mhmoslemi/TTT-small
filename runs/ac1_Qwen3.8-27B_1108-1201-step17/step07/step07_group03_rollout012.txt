import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    import cvxpy as cp
    if '_lp_solver_cached' in globals() and globals()['_lp_solver_cached'] is not None:

    for solver_name in [cp.MOSEK, cp.GUROBI, cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])

            if prob.status in ('optimal', 'optimal_inaccurate'):

def solve_lp_best_response(a_fixed, solver=None):

    if solver is None:

                break

                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def coord_descent_golden(a0, time_limit, max_sweeps=1000, tol=1e-10):

                        best_score = score
                        best_a = a.copy()
                        improved_any = True
                        break

            # Golden section search for optimal a[i]
            # Search range: [0, 2 * current value + 0.05/n]
            lo = 0.0

                    hi = d2
                    d2 = c
                    f_d = f_c
                    c = hi - golden * (hi - lo)
                    a_c = a.copy()
                    a_c[i] = c
                    a_c = a_c / a_c.sum()
                    f_c = 2.0 * n * fast_convolve(a_c).max()
                else:

                    c = d2
                    f_c = f_d

                    a_d = a.copy()
                    a_d[i] = d2
                    a_d = a_d / a_d.sum()
                    f_d = 2.0 * n * fast_convolve(a_d).max()

            a_new[i] = v_opt

        # Find coordinates that contribute to top peaks
        top_k = min(10, len(b))
        peaks = np.argpartition(b, -top_k)[-top_k:]

        # For each peak, find contributing coordinates
        relevant = set()
        for k in peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            relevant.update(range(js, je + 1))

        relevant = list(relevant)
        if not relevant:
            break

        rng.shuffle(relevant)
        improved = False

        for i in relevant:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue

            # Try reducing (most common case: we want to lower the peak)
            for factor in [0.85, 0.7, 0.5, 0.9, 0.95]:
                a[i] = old_val * factor
                a = np.maximum(a, 0)
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    break
                a = a / s2
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a.copy()
                    old_val = a[i]
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()
                    break

        if not improved and sweep > 10:
            # Try all coordinates with smaller steps
            perm = rng.permutation(n)
            any_improve = False
            for i in perm:
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = a[i]
                if old_val < 1e-15:
                    continue
                for factor in [0.95, 1.05, 0.9, 1.1]:
                    a[i] = old_val * factor
                    a = a.maximum(a, 0) if hasattr(a, 'maximum') else np.maximum(a, 0)
                    s2 = a.sum()
                    if s2 < 1e-15:
                        a[i] = old_val
                        break
                    a = a / s2
                    score = 2.0 * n * fast_convolve(a).max()
                    if score < best_score - 1e-14:
                        best_score = score
                        best_a = a.copy()
                        old_val = a[i]
                        improved = True
                        any_improve = True
                    else:
                        a[i] = old_val
                        a = a / a.sum()

            if not any_improve and sweep > 50:
                break

    return best_a, best_score

def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """Alternating LP best responses."""
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            break
        a_new, M_a = solve_lp_best_response(b_new, solver)
        if a_new is None:
            break

        candidates = [a_new, b_new, 0.5 * (a_new + b_new)]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand = cand / s
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score:
                best_score = score
                best_a = cand.copy()
            a = cand  # Use the best candidate as next starting point
            break  # Use first valid candidate

        if verbose and iters % 10 == 0:
            print(f"    FP it={iters} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters

def init_ramp(n, rng, direction=1):
    """Linear ramp (asymmetric)."""
    if direction == 1:
        a = np.arange(1, n + 1, dtype=np.float64)
    else:
        a = np.arange(n, 0, -1, dtype=np.float64)
    s = a.sum()
    return a / s

def init_exponential(n, rng, tau=None, left_decay=False):
    """Exponential decay (asymmetric)."""
    if tau is None:
        tau = n / rng.uniform(2, 8)
    j = np.arange(n, dtype=np.float64)
    if left_decay:
        a = np.exp(-j / tau)
    else:
        a = np.exp(-(n-1-j) / tau)
    s = a.sum()
    return a / s

def init_two_hump(n, rng):
    """Two humps at different heights (asymmetric)."""
    h1 = rng.uniform(0.5, 2.0)
    h2 = rng.uniform(0.3, 1.5)
    sigma1 = n * rng.uniform(0.05, 0.15)
    sigma2 = n * rng.uniform(0.05, 0.15)
    center1 = n * rng.uniform(0.15, 0.35)
    center2 = n * rng.uniform(0.55, 0.85)
    j = np.arange(n, dtype=np.float64)
    a = h1 * np.exp(-((j - center1)**2) / (2*sigma1**2)) + h2 * np.exp(-((j - center2)**2) / (2*sigma2**2))
    s = a.sum()
    return a / s

def init_random_asym(n, rng):
    """Truly random asymmetric initialization."""
    a = rng.random(n) + 0.01
    # Add some structure: multiply by a random smooth envelope
    env_scale = rng.uniform(0.5, 2.0)
    envelope = np.linspace(1.0, env_scale, n)
    a = a * envelope
    s = a.sum()
    return a / s

def init_fourier(n, rng):
    """Fourier superposition (may be asymmetric)."""
    n_harmonics = int(rng.integers(5, 25))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.6)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
