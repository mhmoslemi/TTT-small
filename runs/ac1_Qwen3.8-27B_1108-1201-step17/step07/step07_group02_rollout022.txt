def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(max(0, min(1000.0, x))) for x in a]
        s = sum(a_list)
        if s < 0.01:
            return float('inf')
        b = np.convolve(a_list, a_list)
        return float(2 * len(a_list) * b.max() / (s ** 2))
    except Exception:

def simplex_project(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_subgradient_nondiff(a, b, active_set):
    """Compute subgradient of max_k b_k at point a.
    
    ∂b_k/∂a[i] = 2*a[k-i] if k-i is valid and i != k-i
                 = a[i] if i == k-i (i.e., 2i = k)
                 = 0 otherwise
    """
    n = len(a)
    g = np.zeros(n)
    for k in active_set:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for i in range(js, je + 1):
            if 2 * i == k:
                g[i] += a[i]
            else:
                g[i] += 2 * a[k - i]
    g /= len(active_set)
    return g


def lbgfs_smoothed(a0, time_limit, T_stages=None, maxiter_per_stage=2000):
    """L-BFGS on log-sum-exp smoothed objective in log-space."""
    n = len(a0)
    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2]
    
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    n_stages = len(T_stages)
    iters = maxiter_per_stage // n_stages
    
    for stage_T in T_stages:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        
        def func(z_val, T_val=stage_T):
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            return (2.0 * n / T_val) * (max_Tb + np.log(np.exp(Tb - max_Tb).sum()))
        
        def jac(z_val, T_val=stage_T):
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            # grad w.r.t. a of log-sum-exp(T*b)
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                # ∂b_k/∂a[i] for i in idx
                ki = k - idx
                contrib = np.where(idx == ki, a[idx], 2.0 * a[ki])
                grad_a[idx] += w[k] * contrib
            # Convert to log-space: grad_z[i] = a[i] * (grad_a[i] - sum(a*grad_a))
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def nesterov_subgrad(a0, time_limit, K=5, step_scale=1.0):
    """Nesterov-accelerated subgradient descent on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t = 1.0
    a_prev = a.copy()
    t0 = time.monotonic()
    it = 0
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        
        y = a + beta * (a - a_prev)
        y = simplex_project(y)
        
        # Compute subgradient at y
        b_y = fast_convolve(y)
        max_b = b_y.max()
        # Active set: top K peaks
        sorted_idx = np.argsort(b_y)[::-1]
        active = sorted_idx[:K]
        
        g = compute_subgradient_nondiff(y, b_y, active)
        
        # Step size
        step = step_scale / np.sqrt(t_new + 1.0)
        
        a_new = simplex_project(y - step * g)
        
        # Evaluate
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        a_prev = a
        a = a_new
        t = t_new
        
        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov iter={it+1} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def block_coord_descent(a0, time_limit, block_size=16):
    """Block coordinate descent: optimize each block with L-BFGS while fixing others."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    n_blocks = (n + block_size - 1) // block_size
    
    for sweep in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        
        for b_idx in range(n_blocks):
            if time.monotonic() - t0 > time_limit:
                break
            
            start = b_idx * block_size
            end = min(start + block_size, n)
            block = a[start:end].copy()
            block_sum = block.sum()
            
            if block_sum < 1e-15:
                continue
            
            # Optimize this block: minimize max_b subject to sum(block) = block_sum, block >= 0
            # Use L-BFGS with constraint handled by softmax parameterization
            block_norm = block / block_sum  # probability vector for the block
            
            def block_func(z_block):
                b_trial = softmax_from_z(z_block)
                a_trial = a.copy()
                a_trial[start:end] = b_trial * block_sum
                b = fast_convolve(a_trial)
                return b.max()
            
            def block_jac(z_block):
                b_trial = softmax_from_z(z_block)
                a_trial = a.copy()
                a_trial[start:end] = b_trial * block_sum
                b = fast_convolve(a_trial)
                max_b = b.max()
                # Subgradient of max_b w.r.t. a_trial
                active = [np.argmax(b)]
                grad_a = np.zeros(n)
                for k in active:
                    js = max(0, k - n + 1)
                    je = min(n - 1, k)
                    for i in range(js, je + 1):
                        if 2 * i == k:
                            grad_a[i] += a_trial[i]
                        else:
                            grad_a[i] += 2 * a_trial[k - i]
                grad_block = grad_a[start:end]
                # Convert to log-space
                S = np.dot(grad_block, b_trial)
                return b_trial * (grad_block - S)
            
            z_block = z_from_a(block_norm)
            try:
                res = minimize(
                    fun=block_func, x0=z_block, jac=block_jac,
                    method='L-BFGS-B',
                    options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-8, 'disp': False}
                )
                b_new = softmax_from_z(res.x)
                a[start:end] = b_new * block_sum
            except Exception:
                pass
        
        # Renormalize
        s = a.sum()
        if s > 1e-15:
            a = a / s
        
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if (sweep + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BlockCD sweep={sweep+1} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def peak_shaving(a0, time_limit):
    """Aggressive peak shaving: target the top peaks and reduce contributing coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find interior peaks (exclude first and last which are always small)
        b_interior = b.copy()
        b_interior[0] = -np.inf
        b_interior[-1] = -np.inf
        peak_k = int(np.argmax(b_interior))
        peak_val = b[peak_k]
        
        # Find contributing coordinates
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        
        # Contributions to this peak
        contribs = np.abs(a[idx] * a[peak_k - idx])
        top_contrib = np.argsort(contribs)[::-1][:3]
        
        improved = False
        for ci in top_contrib:
            i = int(idx[ci])
            if a[i] < 1e-14:
                continue
            
            # Try reducing a[i] and redistributing
            for factor in [0.99, 0.97, 0.95, 0.90, 0.85]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                
                # Redistribute to non-contributing coordinates
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                # Also mask near-contributing coords
                for dj in range(-2, 3):
                    j2 = (peak_k - i) + dj
                    if 0 <= j2 < n:
                        mask[j2] = False
            
                if not mask.any():
                    continue
                
                others = a[mask]
                if others.sum() < 1e-15:
                    continue
                
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try reducing the peak contributor and its mirror
            if len(top_contrib) >= 1:
                i = int(idx[top_contrib[0]])
                j = peak_k - i
                if i != j:
                    for f1 in [0.95, 0.90]:
                        for f2 in [0.95, 0.90]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            a_trial = np.maximum(a_trial, 0)
                            ss = a_trial.sum()
                            if ss < 1e-15:
                                continue
                            a_trial = a_trial / ss
                            score = 2.0 * n * fast_convolve(a_trial).max()
                            if score < best_score - 1e-15:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
        
        if not improved:
            break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
