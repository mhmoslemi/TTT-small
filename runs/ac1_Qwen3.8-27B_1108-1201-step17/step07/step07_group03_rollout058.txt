"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth gradient descent with
adaptive temperature, Adam optimizer, multi-start initialization, and
coordinate descent polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a, b=None):
    """Linear convolution of a with b (or with itself)."""
    if b is None:
        b = a
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def fast_autoconvolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_smooth_grad(a, beta):
    """
    Compute the gradient of (1/beta)*log(sum(exp(beta*b))) where b = conv(a,a).
    Returns gradient of length n.
    """
    n = len(a)
    b = fast_autoconvolve(a)
    # Softmax with temperature beta
    b_max = b.max()
    log_z = b_max * beta + np.log(np.sum(np.exp(beta * (b - b_max))))
    w = np.exp(beta * (b - b_max)) / np.exp(log_z)  # softmax weights
    
    # grad[j] = 2 * sum_m a[m] * w[j+m] for valid m, j+m
    # This is cross-correlation: corr[j] = conv(a_rev, w)[j + n - 1]
    a_rev = a[::-1]
    c = fast_convolve(a_rev, w)
    # c has length n + (2n-1) - 1 = 3n - 2
    # We need c[n-1], ..., c[2n-2]
    grad = 2.0 * c[n-1:2*n-1]
    return grad, b.max()

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        rho_idx = np.nonzero(cond)[0][-1]
        theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def adam_optimize_smooth(a0, time_limit, beta_init=100.0, beta_max=5000.0,
                         lr_init=0.01, beta_growth=1.002, verbose=False):
    """
    Adam optimizer with smooth max approximation and simplex projection.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    beta = beta_init
    lr = lr_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_autoconvolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        beta = min(beta * beta_growth, beta_max)
        
        grad, peak = compute_smooth_grad(a, beta)
        
        # Normalize gradient (scale-invariant problem)
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm
        
        # Adam update
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = simplex_project(a_new)
        
        score = 2.0 * n * fast_autoconvolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.9
            no_improve = 0
        
        # Restarts
        if no_improve > 500:
            noise = np.random.default_rng(t * 31 + 7).standard_normal(n) * 0.05
            a = simplex_project(best_a + noise)
            m = np.zeros(n)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, lr * 2)
            no_improve = 0
            beta = beta_init
        
        lr *= 0.9995
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam it={t} ({elapsed:.0f}s): best={best_score:.6f} beta={beta:.0f} lr={lr:.6f}")

    return best_a, best_score

def coord_descent_polish(a0, time_limit, n_sweeps=500):
    """Coordinate descent polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autoconvolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.7, 0.85, 0.92, 0.96, 1.04, 1.08, 1.15, 1.3]:
                a[i] = old_val * factor
                a = np.maximum(a, 0)
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s_new
                score = 2.0 * n * fast_autoconvolve(a).max()
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a.copy()
                    old_val = a[i]
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()
        if not improved and sweep > 10:
            break

    return best_a, best_score

def peak_shave(a0, time_limit):
    """Targeted peak shaving by adjusting contributing coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autoconvolve(a).max()
    t0 = time.monotonic()

    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_autoconvolve(a)
        peak_k = int(np.argmax(b))
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        if contribs.max() < 1e-15:
            break
        
        # Try reducing the top contributors
        top_contrib = np.argsort(contribs)[::-1][:10]
        improved = False
        
        for ci in top_contrib:
            i = int(idx[ci])
            for factor in [0.85, 0.75, 0.65]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                score = 2.0 * n * fast_autoconvolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            # Try adding to non-contributing positions
            non_contrib = np.setdiff1d(np.arange(n), idx)
            if len(non_contrib) > 0:
                for _ in range(5):
                    i_add = int(rng.choice(non_contrib)) if hasattr(rng, 'choice') else int(np.random.randint(len(non_contrib)))
                    a_trial = a.copy()
                    a_trial[i_add] += a.max() * 0.1
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autoconvolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
        if not improved:
            break

    return best_a, best_score

def init_fourier(n, rng, n_harmonics=None):
    """Fourier superposition initialization."""
    if n_harmonics is None:
        n_harmonics = int(rng.integers(5, 25))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n

def init_sinc(n, rng):
    """Sinc-like initialization (squared Gaussian)."""
    center = n / 2.0
    sigma = rng.uniform(0.2, 0.5) * n

    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n

def init_symmetric_random(n, rng):

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    return f_interp / ss if ss > 1e-15 else np.ones(new_n) / new_n

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a_fixed, solver=None):
    """LP: given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max conv(a,b)."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative (mirror descent) method with smooth log-sum-exp gradient
2. Adam optimizer with simplex projection as secondary method
3. LP-based fictitious play for refinement
4. Multi-start with diverse initializations at multiple lengths
5. Coordinate descent and peak shaving for final polish
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a, b=None):
    """Linear convolution of a with b (or with itself)."""
    if b is None:
        b = a
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def fast_autoconvolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_smooth_grad(a, beta):
    """
    Compute gradient of (1/beta)*log(sum(exp(beta*b))) where b = conv(a,a).
    Uses FFT-based cross-correlation for O(n log n) computation.
    Returns (gradient, peak_value).
    """
    n = len(a)
    b = fast_autoconvolve(a)
    b_max = b.max()
    shifted = beta * (b - b_max)
    log_z = np.log(np.sum(np.exp(shifted)))
    w = np.exp(shifted - log_z)  # softmax weights, sums to 1

    # grad[j] = 2 * sum_m a[m] * w[j+m]
    # = 2 * conv(a_rev, w)[j + n - 1]
    a_rev = a[::-1]
    c = fast_convolve(a_rev, w)
    grad = 2.0 * c[n-1:2*n-1]
    return grad, b_max

def compute_mixed_grad(a, betas):
    """Compute mixed gradient from multiple temperature levels."""
    n = len(a)
    b = fast_autoconvolve(a)
    b_max = b.max()
    total_grad = np.zeros(n)
    for beta in betas:
        shifted = beta * (b - b_max)
        log_z = np.log(np.sum(np.exp(shifted)))
        w = np.exp(shifted - log_z)
        a_rev = a[::-1]
        c = fast_convolve(a_rev, w)
        total_grad += 2.0 * c[n-1:2*n-1]
    total_grad /= len(betas)
    return total_grad, b_max

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        idx = np.nonzero(cond)[0][-1]
        theta = (cssv[idx] - 1.0) / (idx + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def multiplicative_descent(a0, time_limit, beta_init=50.0, beta_max=3000.0,
                           lr_init=0.3, beta_growth=1.003, verbose=False):
    """
    Multiplicative (mirror descent) method with log-sum-exp smoothing.
    Update: a[j] *= exp(-lr * grad[j]), then normalize.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    beta = beta_init
    lr = lr_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_autoconvolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    ema_grad = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        beta = min(beta * beta_growth, beta_max)

        grad, peak = compute_smooth_grad(a, beta)

        # Smooth the gradient with EMA for stability
        ema_grad = 0.8 * ema_grad + 0.2 * grad

        # Normalize gradient to control step size
        g_norm = np.linalg.norm(ema_grad)
        if g_norm < 1e-15:
            break
        ema_grad = ema_grad / g_norm

        # Multiplicative update (mirror descent with exponential map)
        a = a * np.exp(-lr * ema_grad)
        a = np.maximum(a, 1e-15)  # Keep strictly positive
        a = a / a.sum()

        score = 2.0 * n * fast_autoconvolve(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 80:
            lr *= 0.85
            no_improve = 0

        # Reheating: periodically reduce beta and lr to escape local minima
        if no_improve > 400:
            beta = max(beta_init, beta * 0.3)
            lr = min(lr * 2.0, lr_init)
            noise = np.random.default_rng(it * 17 + 3).standard_normal(n) * 0.03
            a = a + noise * a
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            ema_grad = np.zeros(n)
            no_improve = 0

        lr *= 0.9997

        if verbose and it % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MultDesc it={it} ({elapsed:.0f}s): best={best_score:.6f} beta={beta:.0f} lr={lr:.4f}")

    return best_a, best_score

def adam_smooth_optimize(a0, time_limit, beta_init=100.0, beta_max=5000.0,
                         lr_init=0.02, beta_growth=1.002, verbose=False):
    """Adam optimizer with smooth max and simplex projection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    m = np.zeros(n)
    v = np.zeros(n)
    beta = beta_init
    lr = lr_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_autoconvolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        beta = min(beta * beta_growth, beta_max)

        grad, peak = compute_smooth_grad(a, beta)
        g_norm = np.linalg.norm(grad)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = simplex_project(a_new)

        score = 2.0 * n * fast_autoconvolve(a_new).max()

            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 100:
            lr *= 0.88
            no_improve = 0

        if no_improve > 500:
            noise = np.random.default_rng(it * 23 + 11).standard_normal(n) * 0.04
            a = simplex_project(best_a + noise)
            m = np.zeros(n)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, lr * 2)
            beta = beta_init
            no_improve = 0

        if verbose and it % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam it={it} ({elapsed:.0f}s): best={best_score:.6f} beta={beta:.0f} lr={lr:.5f}")

    return best_a, best_score

def mixed_temp_descent(a0, time_limit, betas=None, lr_init=0.2, verbose=False):
    """Multiplicative descent with mixed temperature gradient (ensemble)."""
    n = len(a0)
    if betas is None:
        betas = [50.0, 200.0, 800.0, 3000.0]

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    lr = lr_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_autoconvolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        grad, peak = compute_mixed_grad(a, betas)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm

        a = a * np.exp(-lr * grad)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        score = 2.0 * n * fast_autoconvolve(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            lr *= 0.85
            no_improve = 0

        if no_improve > 400:
            lr = min(lr * 2, lr_init)
            noise = np.random.default_rng(it * 37 + 5).standard_normal(n) * 0.04
            a = a + noise * a
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            no_improve = 0

        lr *= 0.9995

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MixedTemp it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.4f}")

    return best_a, best_score

def coord_descent_polish(a0, time_limit, n_sweeps=1000):
    """Coordinate descent: scale individual coordinates to reduce peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autoconvolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77777)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.7, 0.82, 0.9, 0.95, 1.05, 1.1, 1.18, 1.3]:
                a[i] = old_val * factor
                a = np.maximum(a, 0)
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s_new
                score = 2.0 * n * fast_autoconvolve(a).max()
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a.copy()
                    old_val = a[i]
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()
        if not improved and sweep > 20:
            break

    return best_a, best_score

def peak_shave(a0, time_limit):
    """Targeted peak shaving: reduce coordinates that contribute most to the peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autoconvolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(44444)

    for step in range(20000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_autoconvolve(a)
        top_k = min(3, len(b))
        peaks = np.argpartition(b, -top_k)[-top_k:]

        improved = False
        for peak_k in peaks:
            peak_k = int(peak_k)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]

            if contribs.max() < 1e-15:
                continue

            top_contrib = np.argsort(contribs)[::-1][:8]
            for ci in top_contrib:
                i = int(idx[ci])
                if a[i] < 1e-14:
                    continue
                for factor in [0.88, 0.8, 0.7]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    score = 2.0 * n * fast_autoconvolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try small random perturbations
            for _ in range(3):
                i = int(rng.integers(n))
                a_trial = a.copy()
                factor = rng.uniform(0.85, 1.15)
                a_trial[i] *= factor
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                score = 2.0 * n * fast_autoconvolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if not improved:
                break

    return best_a, best_score

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a_fixed, solver):
    """LP: given fixed a (sum=1, a>=0), find b>=0, sum(b)=1 minimizing max conv(a,b)."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """LP fictitious play (alternating best responses)."""
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        a = a / s if s > 1e-15 else np.ones(n) / n

    b = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autoconvolve(a).max()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new = lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new = lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        a_avg = 0.5 * (a + b)
        candidates = [a, b, a_avg]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand = cand / s
            score = 2.0 * n * fast_autoconvolve(cand).max()
            if score < best_score:
                best_score = score
                best_a = cand.copy()

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters

def damped_lp_refine(a0, time_limit, solver=None):
    """Damped LP refinement with decreasing step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    best_a = a.copy()
    best_score = 2.0 * n * fast_autoconvolve(a).max()
    t0 = time.monotonic()

    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        b_new = lp_best_response(a, solver)
        if b_new is None:
            continue
        alpha = min(1.0, 2.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_autoconvolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score

def get_lp_solver():
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def init_fourier(n, rng, n_harmonics=None):
    """Fourier superposition with random phases."""
    if n_harmonics is None:
        n_harmonics = int(rng.integers(5, 30))
    amplitudes = rng.random(n_harmonics) * 0.45
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n

def init_gaussian(n, rng):
    """Gaussian bump with random center and width."""
    center = rng.uniform(0.2, 0.8) * n
    sigma = rng.uniform(0.1, 0.45) * n
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((t - center) / sigma)**2)
    a += rng.random(n) * 0.08 * a.max()
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n

def init_bimodal(n, rng):
    """Bimodal: mass concentrated near the two ends."""
    sigma = rng.uniform(0.05, 0.2) * n
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * (t / sigma)**2) + np.exp(-0.5 * ((t - n + 1) / sigma)**2)
    a += rng.random(n) * 0.1 * a.max()
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n

def init_symmetric_rand(n, rng):
    """Symmetric random initialization."""
    half = (n + 1) // 2
    a_half = np.abs(rng.standard_normal(half)) + 0.1
    if n % 2 == 0:
        a = np.concatenate([a_half, a_half[:-1][::-1]])
    else:
        a = np.concatenate([a_half, a_half[:-1][::-1]])
    a = a[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n

def init_blocks(n, rng):
    """Block structure with random amplitudes and smoothing."""
    n_blocks = int(rng.integers(4, 20))
    block_size = max(2, n // n_blocks)
    amplitudes = rng.random(n_blocks) + 0.1
    a = np.zeros(n)
    for b_idx in range(n_blocks):
        start = b_idx * block_size
        end = min(start + block_size, n)
        a[start:end] = amplitudes[b_idx]
    ksize = min(5, block_size)
    if ksize > 1:
        kernel = np.ones(ksize) / ksize
        a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    return f_interp / ss if ss > 1e-15 else np.ones(new_n) / new_n

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate via evaluate_sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

            t_fin = min(rem * 0.4, 40.0)
            a_fin2, s_fin2 = adam_smooth_optimize(best_seq, t_fin,
                                                   beta_init=1000.0, beta_max=10000.0,
                                                   lr_init=0.005, beta_growth=1.002)
            best_seq, best_score = check_and_update(a_fin2, best_seq, best_score, "final_adam")

        # Final coord descent
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            a_fin3, s_fin3 = coord_descent_polish(best_seq, min(rem * 0.8, 20.0), n_sweeps=200)
            best_seq, best_score = check_and_update(a_fin3, best_seq, best_score, "final_cd")

        # Final peak shave
        rem = deadline - time.monotonic() - 5
        if rem > 3:
            a_fin4, s_fin4 = peak_shave(best_seq, min(rem * 0.8, 10.0))
            best_seq, best_score = check_and_update(a_fin4, best_seq, best_score, "final_ps")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
