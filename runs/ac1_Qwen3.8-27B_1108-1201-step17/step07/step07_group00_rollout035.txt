Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Uses multiplicative (mirror) descent with soft-max peak weighting as the primary

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def multiplicative_step(f, top_k, temperature, eta):
    """
    One multiplicative descent step.
    
    For each element f[j], compute its 'pressure' from the top peaks,
    then reduce it proportionally.
    
    Returns updated f (not yet renormalized).
    """
    n = len(f)
    b = fast_convolve(f)
    
    # Soft-max weights on top peaks
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    
    # Weight by soft-max
    logits = b[top_idx] / temperature
    logits -= logits.max()
    w = np.exp(logits)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(top_idx)) / len(top_idx)
    else:
        w = w / w_sum

    # pressure[j] = sum_k w[m] * f[k - j] where k = top_idx[m] and 0 <= k-j < n
    pressure = np.zeros(n)
    for m in range(len(top_idx)):
        if w[m] < 1e-12:
            continue
        k = top_idx[m]
        # f[j] contributes to b[k] via f[k-j], so for element j,
        # its contribution is f[k-j] (if valid)
        # Equivalently, the "blame" on position j from peak k is f[k-j]
        lo = max(0, k - n + 1)  # j range where k-j is valid
        hi = min(n - 1, k)

        contrib = f[k - j_range]
        pressure[j_range] += w[m] * contrib

    f_new = f / (1.0 + eta * pressure)
    return f_new

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Primary method: Multiplicative (mirror) descent with soft-max peak weighting,
which is O(n * top_k) per iteration and allows many more iterations than LP.
Complemented by LP-based polish, adaptive temperature, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_pressure(f, top_idx, weights):
    """
    Compute pressure vector: pressure[j] = sum_m weights[m] * f[top_idx[m] - j]
    for valid indices.
    """
    n = len(f)
    pressure = np.zeros(n)
    for m in range(len(top_idx)):
        w = weights[m]
        if w < 1e-12:
            continue
        k = int(top_idx[m])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j_range = np.arange(lo, hi + 1)
        pressure[j_range] += w * f[k - j_range]
    return pressure


def multiplicative_descent(f_init, time_limit, top_k_init=50, temp_init=1e-4, eta_init=0.5):
    """
    Run multiplicative (mirror) descent.
    Returns (best_f, best_score).
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj

    temperature = temp_init
    eta = eta_init
    top_k = top_k_init
    it = 0
    no_improve = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        it += 1

        b = fast_convolve(f)
        
        # Find top peaks
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))

        # Soft-max weights
        logits = b[top_idx] / temperature
        logits -= logits.max()
        w = np.exp(logits)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w = w / w_sum

        # Compute pressure
        pressure = compute_pressure(f, top_idx, w)

        # Multiplicative update (exact exponential form)
        # f[j] <- f[j] * exp(-eta * pressure[j])
        log_update = -eta * pressure
        # Clamp to avoid underflow/overflow
        log_update = np.clip(log_update, -50, 0)
        f = f * np.exp(log_update)
        s = f.sum()
        if s < 1e-10:
            f = best_f.copy()
            break
        f /= s

        # Check objective periodically
        if it % 3 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-11:
                best_obj = obj
                best_score = 2.0 * n * obj
                best_f = f.copy()
                no_improve = 0
                eta = min(eta * 1.01, 3.0)
            else:
                no_improve += 1
                if no_improve > 30:
                    f = best_f.copy()
                    eta *= 0.6
                    temperature = min(temperature * 1.5, 0.5)
                    top_k = min(top_k + 10, 500)
                    no_improve = 0
                    if eta < 0.005:
                        break
        # Adaptive schedule
        if it % 100 == 0:
            temperature = min(temperature * 1.05, 0.3)

        if it % 500 == 0:
            print(f"  md iter={it} score={best_score:.6f} eta={eta:.3f} temp={temperature:.6f} top_k={top_k}")

    return best_f, best_score


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_polish(f_init, time_limit, top_k=80):
    """
    LP-based polish: find the best non-negative direction that doesn't increase
    the max convolution at the top peaks.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f /= s
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        if M < 1e-15:
            break

        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        top_idx_sorted = np.sort(top_idx)
        K = len(top_idx_sorted)

        # Build constraint matrix
        A = np.zeros((K, n))
        for row_i in range(K):
            k = int(top_idx_sorted[row_i])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[row_i, lo:hi + 1] = f[lo:hi + 1]

        g = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-8)
        except Exception:
            break

        if g.value is None:
            break
        g_val = np.maximum(g.value, 0.0)
        if np.any(np.isnan(g_val)):
            break

        S_g = g_val.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 15:
                break
            continue

        # Line search
        h = g_val / S_g
        best_t = 0.05
        current_obj = fast_convolve(f).max()
        for t in [0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-10:
                continue
            f_trial /= st
            obj = fast_convolve(f_trial).max()
            if obj < current_obj:
                best_t = t
                current_obj = obj

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_score = 2.0 * n * obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 20 == 0:
            print(f"  lp iter={it} score={best_score:.6f} M={M:.8f}")

    return best_f, best_score


def subgradient_polish(f_init, time_limit, n_top=100):
    """
    Subgradient method with softmax weighting on top peaks.
    Uses projected gradient descent on the simplex.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f /= s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj

    v = np.zeros(n)
    momentum = 0.85
    lr = 0.3
    temp = 1e-4
    it = 0
    no_improve = 0
    rng = np.random.default_rng(99)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))

        logits = b[top_idx] / temp
        logits -= logits.max()
        w = np.exp(logits)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w = w / w_sum

        # Subgradient: grad[j] = sum_m w[m] * f[top_idx[m] - j]
        grad = np.zeros(n)
        for m in range(len(top_idx)):
            wm = w[m]
            if wm < 1e-12:
                continue
            k = int(top_idx[m])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            j_range = np.arange(lo, hi + 1)
            grad[j_range] += wm * f[k - j_range]

        # Projected gradient step with momentum
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        f = f_new

        if it % 5 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-11:
                best_obj = obj
                best_score = 2.0 * n * obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.03, 1.0)
            else:
                no_improve += 1
                if no_improve > 20:
                    f = best_f.copy()
                    v = np.zeros(n)
                    lr *= 0.7
                    temp = min(temp * 1.3, 0.5)
                    no_improve = 0
                    if lr < 0.01:
                        # Random restart
                        f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                        v = np.zeros(n)
                        lr = 0.3

        if it % 200 == 0:
            print(f"  sg iter={it} score={best_score:.6f} lr={lr:.4f}")

    return best_f, best_score

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Primary method: Primal-dual multiplicative descent with FFT-accelerated pressure
computation. The dual variables represent "penalty" for each convolution position,
and the primal (multiplicative) update reduces elements that contribute to penalized
peaks. Complemented by subgradient polish, LP-based direction finding, and
multi-length exploration.

Key innovations over previous approach:
- FFT-based pressure computation enables O(n log n) per iteration
- Primal-dual framework naturally balances peak reduction
- Annealed target level (from peak to mean) for progressive flattening
- Multiple method sequence to escape local minima
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_pressure_fft(lamda, f):
    """
    Compute pressure[j] = sum_k lamda[k] * f[k-j] for j in [0, n-1]
    using FFT for O(n log n) complexity.
    """
    n = len(f)
    f_flipped = f[::-1]
    conv_len = len(lamda) + n - 1
    N = 1 << (conv_len - 1).bit_length()
    L = rfft(lamda, N)
    F = rfft(f_flipped, N)
    conv = irfft(L * F, N)
    return conv[n - 1: 2 * n - 1]


def primal_dual_descent(f_init, time_limit, eta_p_init=0.8, eta_d_init=0.15,
                        alpha_init=0.8, label="pd"):
    """
    Primal-dual multiplicative descent.
    - Dual: lamda[k] increases where b[k] exceeds target
    - Primal: f[j] decreases where it contributes to penalized positions
    - Target anneals from peak level to mean level
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj

    M_len = 2 * n - 1
    lamda = np.zeros(M_len)
    eta_p = eta_p_init
    eta_d = eta_d_init
    alpha = alpha_init
    it = 0
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        b_max = b.max()
        b_mean = b.mean()

        # Anneal the target: from peak-focused to mean-focused
        target = b_mean + alpha * (b_max - b_mean)

        # Dual update
        lamda = np.maximum(lamda + eta_d * (b - target), 0.0)
        # Prevent lamda from growing too large
        lamda_max = lamda.max()
        if lamda_max > 1e4:
            lamda /= lamda_max * 0.01

        # Primal update (multiplicative)
        pressure = compute_pressure_fft(lamda, f)
        # Normalize pressure for stable updates
        p_max = pressure.max()
        if p_max > 1e-10:
            pressure_norm = pressure / p_max
        else:
            pressure_norm = pressure

        log_update = -eta_p * pressure_norm
        log_update = np.clip(log_update, -20.0, 0.0)
        f = f * np.exp(log_update)
        s = f.sum()
        if s < 1e-10:
            f = best_f.copy()
            lamda = np.zeros(M_len)
            break
        f /= s

        # Adaptive schedule
        alpha = max(alpha * 0.999, 0.05)
        if it % 200 == 0:
            eta_p *= 0.95
            eta_d *= 0.97

        # Check objective
        if it % 4 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-11:
                best_obj = obj
                best_score = 2.0 * n * obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 40:
                    # Restart: perturb and reset dual
                    f = best_f + rng.standard_normal(n) * 0.002
                    f = np.maximum(f, 0.0)
                    f /= f.sum()
                    lamda = np.zeros(M_len)
                    alpha = max(alpha * 1.5, 0.5)
                    eta_p = eta_p_init * 0.5
                    no_improve = 0
                    if it > 5000 and eta_p < 0.01:
                        break

        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} score={best_score:.6f} "
                  f"eta_p={eta_p:.3f} alpha={alpha:.4f} t={elapsed:.0f}s")

    return best_f, best_score


def multiplicative_descent_topk(f_init, time_limit, top_k_init=40,
                                temp_init=2e-4, eta_init=1.0, label="md"):
    """
    Multiplicative descent focusing on top-K peaks with softmax weighting.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj

    temperature = temp_init
    eta = eta_init
    top_k = top_k_init
    it = 0
    no_improve = 0
    rng = np.random.default_rng(333)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))

        # Soft-max weights
        logits = b[top_idx] / temperature
        logits -= logits.max()
        w = np.exp(logits)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w = w / w_sum

        # Compute pressure directly (top_k is small)
        pressure = np.zeros(n)
        for m in range(len(top_idx)):
            wm = w[m]
            if wm < 1e-12:
                continue
            k = int(top_idx[m])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            j_range = np.arange(lo, hi + 1)
            pressure[j_range] += wm * f[k - j_range]

        # Multiplicative update
        p_max = pressure.max()
        if p_max > 1e-10:
            pressure_norm = pressure / p_max
        else:
            pressure_norm = pressure

        log_update = -eta * pressure_norm
        log_update = np.clip(log_update, -20.0, 0.0)
        f = f * np.exp(log_update)
        s = f.sum()
        if s < 1e-10:
            f = best_f.copy()
            break
        f /= s

        # Adaptive schedule
        if it % 200 == 0:
            temperature = min(temperature * 1.08, 0.5)
            eta *= 0.97
            top_k = min(top_k + 10, 800)

        if it % 4 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-11:
                best_obj = obj
                best_score = 2.0 * n * obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 50:
                    f = best_f + rng.standard_normal(n) * 0.003
                    f = np.maximum(f, 0.0)
                    f /= f.sum()
                    temperature = min(temperature * 2.0, 1.0)
                    eta *= 0.5
                    no_improve = 0
                    if eta < 0.01:
                        break

        if it % 1000 == 0:
            print(f"    {label} iter={it} score={best_score:.6f} "
                  f"eta={eta:.3f} temp={temperature:.6f} top_k={top_k}")

    return best_f, best_score


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_polish(f_init, time_limit, n_top=100, label="sg"):
    """
    Projected subgradient descent with softmax peak weighting and momentum.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f /= s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj

    v = np.zeros(n)
    momentum = 0.8
    lr = 0.5
    temp = 5e-5
    it = 0
    no_improve = 0
    rng = np.random.default_rng(555)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))

        logits = b[top_idx] / temp
        logits -= logits.max()
        w = np.exp(logits)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w = w / w_sum

        # Subgradient
        grad = np.zeros(n)
        for m in range(len(top_idx)):
            wm = w[m]
            if wm < 1e-12:
                continue
            k = int(top_idx[m])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            j_range = np.arange(lo, hi + 1)
            grad[j_range] += wm * f[k - j_range]

        # Normalize gradient
        g_norm = grad.max()
        if g_norm > 1e-10:
            grad = grad / g_norm

        # Momentum + projected step
        v = momentum * v - lr * grad
        f = project_simplex(f + v)

        if it % 5 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-11:
                best_obj = obj
                best_score = 2.0 * n * obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.02, 2.0)
            else:
                no_improve += 1
                if no_improve > 25:
                    f = best_f.copy()
                    v = np.zeros(n)
                    lr *= 0.6
                    temp = min(temp * 1.4, 1.0)
                    no_improve = 0
                    if lr < 0.05:
                        f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                        v = np.zeros(n)
                        lr = 0.3

        if it % 100 == 0:
            temp = min(temp * 1.03, 0.3)
        if it % 500 == 0:
            print(f"    {label} iter={it} score={best_score:.6f} lr={lr:.4f}")

def lp_polish(f_init, time_limit, top_k=100, label="lp"):
    """
    LP-based direction finding: maximize sum(g) s.t. (f*g)[k] <= M for top peaks.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f /= s
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        if M < 1e-15:
            break

        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        top_idx_sorted = np.sort(top_idx)
        K = len(top_idx_sorted)

        # Build constraint matrix
        A_mat = np.zeros((K, n))
        for row_i in range(K):
            k = int(top_idx_sorted[row_i])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A_mat[row_i, lo:hi + 1] = f[lo:hi + 1]

        g_var = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)),
                          [g_var >= 0, A_mat @ g_var <= M])
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-8)
        except Exception:
            break

        if g_var.value is None:
            break
        g_val = np.maximum(g_var.value, 0.0)
        if np.any(np.isnan(g_val)):
            break

        S_g = g_val.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 12:
                # Try with more peaks
                top_k = min(top_k * 2, 2 * n - 1)
                stuck = 0
                continue
            continue

        # Line search
        h = g_val / S_g
        best_t = 0.05
        current_obj = fast_convolve(f).max()
        for t in [0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-10:
                continue
            f_trial /= st
            obj = fast_convolve(f_trial).max()
            if obj < current_obj:
                best_t = t
                current_obj = obj

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_score = 2.0 * n * obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 20 == 0:
            print(f"    {label} iter={it} score={best_score:.6f} top_k={top_k}")

    return best_f, best_score


def coordinate_polish(f_init, time_limit, label="cd"):
    """
    Coordinate descent: try scaling each element by various factors.
    Focus on elements with highest contribution to peaks.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f /= s
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()

    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0])
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        # Sort elements by magnitude (process large ones first)
        order = np.argsort(f)[::-1]
        # Only process top half
        order = order[:n // 2]

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(j)
            if f[j] < 1e-10:
                continue

            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                st = f_trial.sum()
                if st < 1e-10:
                    continue
                f_trial /= st
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-12:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != f[j]:
                f[j] = best_j_val
                st = f.sum()
                if st > 1e-10:
                    f /= st
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break
        if pass_count % 5 == 0:
            print(f"    {label} pass={pass_count} score={best_score:.6f}")

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization function.

    Strategy:
    1. Load parent sequence
    2. Phase 1: Primal-dual descent (45% time) - main workhorse
    3. Phase 2: Top-K multiplicative descent (15% time) - different dynamics
    4. Phase 3: Subgradient polish (10% time)
    5. Phase 4: LP polish (10% time)
    6. Phase 5: Multi-length exploration (10% time)
    7. Phase 6: Final coordinate + subgradient (10% time)
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: Primal-dual descent (main) ===
    print(f"\n=== Phase 1: Primal-dual descent (n={len(parent)}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.45, 700.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, score1 = primal_dual_descent(parent, time_limit=t1,
                                          eta_p_init=1.0, eta_d_init=0.2,
                                          alpha_init=0.9, label="pd1")
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [pd1]: {score1:.6f}")
        else:
            print(f"  pd1 result: {score1:.6f} (no improvement over {best_score:.6f})")

    # === Phase 2: Top-K multiplicative descent ===
    print(f"\n=== Phase 2: Top-K MD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 250.0)
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")
        f2, score2 = multiplicative_descent_topk(best_seq, time_limit=t2,
                                                  top_k_init=30, temp_init=1e-4,
                                                  eta_init=1.5, label="md2")
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [md2]: {score2:.6f}")
        else:
            print(f"  md2 result: {score2:.6f}")

    # === Phase 3: Subgradient polish ===
    print(f"\n=== Phase 3: Subgradient polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 200.0)
    if t3 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t3:.0f}s, n={n_b}")
        f3, score3 = subgradient_polish(best_seq, time_limit=t3,
                                         n_top=max(60, n_b // 6), label="sg3")
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [sg3]: {score3:.6f}")
        else:
            print(f"  sg3 result: {score3:.6f}")

    # === Phase 4: LP polish ===
    print(f"\n=== Phase 4: LP polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 200.0)
    if t4 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t4:.0f}s, n={n_b}")
        f4, score4 = lp_polish(best_seq, time_limit=t4, top_k=80, label="lp4")
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [lp4]: {score4:.6f}")
        else:
            print(f"  lp4 result: {score4:.6f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.85, 1.15, 1.3, 1.5]:
        tl = int(n_base * factor)
        tl = max(512, min(4000, tl))
        if tl != n_base and tl not in target_lengths:
            target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t5 = min(rem * 0.22, 150.0)
        if t5 < 15:
            break

        # Interpolate to new length
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  Primal-dual n={target_n}, budget={t5:.0f}s")
        f5, score5 = primal_dual_descent(f_interp, time_limit=t5 * 0.55,
                                          eta_p_init=0.8, eta_d_init=0.15,
                                          alpha_init=0.7, label=f"pd_n{target_n}")

        # Quick subgradient
        rem2 = deadline - time.monotonic()
        if rem2 > 10:
            f5b, score5b = subgradient_polish(f5, time_limit=min(rem2 * 0.4, 40.0),
                                               n_top=80, label=f"sg_n{target_n}")
            if np.isfinite(score5b) and score5b < score5:
                f5, score5 = f5b, score5b

        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [n{target_n}]: {score5:.6f}")
        else:
            print(f"  n{target_n} result: {score5:.6f}")

    # === Phase 6: Final refinement ===
    print(f"\n=== Phase 6: Final refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        # Subgradient with tight parameters
        t6 = min(rem * 0.5, 80.0)
        n_b = len(best_seq)
        print(f"  SG Budget: {t6:.0f}s, n={n_b}")
        f6, score6 = subgradient_polish(best_seq, time_limit=t6,
                                         n_top=max(80, n_b // 5), label="sg6")
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [sg6]: {score6:.6f}")

        # Coordinate descent
        rem2 = deadline - time.monotonic()
        if rem2 > 10:
            t6b = min(rem2 * 0.5, 50.0)
            f6b, score6b = coordinate_polish(best_seq, time_limit=t6b, label="cd6")
            if np.isfinite(score6b) and score6b < best_score:
                best_score = score6b
                best_seq = f6b.copy()
                print(f"  NEW BEST [cd6]: {score6b:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
