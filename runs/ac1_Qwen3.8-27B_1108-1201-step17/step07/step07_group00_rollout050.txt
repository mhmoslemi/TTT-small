"""
Optimized step function search using full-constraint LP with active set refinement,
smoothed gradient descent, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, indices):
    """Build the constraint matrix A where A[k_idx, j] = f[k-j] for valid k-j."""
    n = len(f)
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)

    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])

    cssv = np.cumsum(u)

def mirror_descent_step(f, lr, temp, n_active):
    """One mirror descent step with exponential weights on active peaks."""
    n = len(f)
    b = fast_convolve(f)
    
    # Compute weights on peaks
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Restrict to top n_active peaks
    if n_active < len(b):
        top_idx = np.argpartition(b, -n_active)[-n_active:]
        w_new = np.zeros(len(b))
        w_new[top_idx] = w[top_idx]
        w_new /= w_new.sum()
        w = w_new
    
    # Gradient: grad_j = 2 * sum_k w_k * f[k-j]
    grad = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-12:

        hi = min(n - 1, k)
        grad[lo:hi+1] += w[k] * f[k-hi:k-lo+1][::-1]
    grad *= 2
    
    # Mirror descent update (multiplicative weights)
    f_new = f * np.exp(-lr * grad)
    s = f_new.sum()
    if s < 1e-30:
        return f.copy()
    return f_new / s

    # irfft(conj(W) * F)[j] = sum_k w[k] f[j+k] (circular)

    # R[j] = sum_k w[k] f[k-j] = sum_k w[k] f[k-j]
    # = (w * f_rev_shifted)[...]

    # R[j] = sum_k w[k] f[k-j] for j = 0, ..., n-1
    # Note that f[k-j] is non-zero only for 0 <= k-j < n, i.e., j <= k < j+n

    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += w[k] * f[k-hi:k-lo+1][::-1]
    return 2.0 * grad

2. Compute `R = irfft(conj(rfft(w)) * rfft(f))`... 

Actually, let me look this up properly. The linear cross-correlation of `a` and `b` is:
`(a ⋆ b)[n] = sum_k a[k] b[k+n]`

In terms of FFT: `FFT(a ⋆ b) = conj(FFT(a)) * FFT(b)` (with zero-padding to avoid circular effects).

So: `R[n] = sum_k w[k] f[k+n] = (w ⋆ f)[n]`

def fast_autocorr(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_xcorr(a, b):
    """Linear cross-convolution via FFT."""
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
