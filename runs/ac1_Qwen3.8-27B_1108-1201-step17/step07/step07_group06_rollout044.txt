    # Alternative: note that corr[i] = sum_j a[j]*w[i+j] is the same as
    # the (i)-th element of the "anti-diagonal" product.
    # 
    # Actually, let me reconsider. We have w of length 2n-1 and a of length n.
    # corr[i] = sum_{j=0}^{n-1} a[j] * w[i+j], where i+j ranges from i to i+n-1.
    # For this to be valid, we need 0 <= i <= 2n-2-(n-1) = n-1.
    # So corr is defined for i = 0, ..., n-1. Good, that's what we need.
    
    # FFT approach: 
    # Reverse a: ar[n-1-j] = a[j]
    # Then corr[i] = sum_j ar[n-1-j] * w[i+j]
    # Let p = n-1-j: corr[i] = sum_p ar[p] * w[i+n-1-p]
    # = (ar * w)[i+n-1] where * denotes convolution
    # = convolve(ar, w)[i+n-1]
    # convolve(ar, w) has length n + (2n-1) - 1 = 3n-2
    # We need elements from index n-1 to 2n-2.
    
    # For FFT: we need fft_size >= 3n-2. Let's use a larger size.
    # 3n-2 for n=768 is 2302, next pow2 is 4096. That's 2x our current fft_size.
    # This will be slower but correct.
    
    # Alternative: use the relationship with our existing fft_size.
    # Actually, let me just compute grad directly without FFT for clarity,
    # or use a different approach.
    
    # SIMPLER APPROACH: Use the fact that:
    # sum_k w[k] * conv[k] = sum_k w[k] * sum_i a[i]*a[k-i]
    # = sum_i a[i] * sum_k w[k]*a[k-i]
    # = sum_i a[i] * corr[i]
    # where corr[i] = sum_k w[k]*a[k-i] (with appropriate bounds)
    # = sum_i a[i] * (cross-correlation of w and a at lag i)
    
    # So grad_a[i] = n * corr[i] where corr[i] = sum_{k} w[k] * a[k-i] * I(0<=k-i<n)
    #              = n * sum_{j} w[i+j] * a[j] * I(0<=i+j<2n-1)
    #              = n * sum_{j=0}^{n-1} w[i+j] * a[j]  (since i<n and j<n, i+j < 2n <= 2n-1 for i+j <= 2n-2)
    
    # OK so the formula is correct. Let me just compute it with a larger FFT.
    
    ax = n
    wlen = len(w)
    need = ax + wlen - 1
    fft2 = 1 << (need - 1).bit_length()
    Ar = rfft(a, fft2)
    Wf = rfft(w, fft2)
    # corr[i] = sum_j a[j]*w[i+j]
    # = convolve(a_reversed, w)[i + ax - 1]
    ar = a[::-1]
    Ar2 = rfft(ar, fft2)
    conv_full = irfft(Ar2 * Wf, fft2)[:need]
    corr = conv_full[ax-1:ax-1+ax]
    return n * corr


def soft_max_value(b, T):
    """Compute soft-max value: T * log(sum(exp(b/T)) - max(b)/T)... """
    max_b = np.max(b)
    shifted = b / T - max_b / T
    log_sum = max_b / T + np.log(np.sum(np.exp(shifted - np.max(shifted))) + np.exp(-np.max(shifted)))
    # Actually: log(sum(exp(b/T))) = max_b/T + log(sum(exp(b/T - max_b/T)))
    val = max_b + T * np.log(np.sum(np.exp(b / T - max_b / T)))
    return val


def topk_soft_max_weights(b, T, k=10):
    """
    Compute weights for top-k soft-max.
    Only the top k entries get significant weight.
    """
    n = len(b)
    # Sort indices by value
    # For efficiency, use partial sort
    if k >= n:
        k = n
    # Get top k indices
    top_idx = np.argpartition(b, -k)[-k:]
    # Soft-max within top k, but also include a small weight for the rest
    b_top = b[top_idx]
    max_b = np.max(b)
    
    weights = np.zeros(n)
    # High temperature for the rest (gives tiny weights)
    b_rest = np.delete(b, top_idx)
    rest_weight = np.exp(b_rest / (T * 10) - max_b / (T * 10))
    top_weight = np.exp(b_top / T - max_b / T)
    
    total = top_weight.sum() + rest_weight.sum()
    if total < 1e-300:
        return np.ones(n) / n
    
    weights[top_idx] = top_weight / total
    rest_idx = np.delete(np.arange(n), top_idx)
    weights[rest_idx] = rest_weight / total
    
    return weights


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def get_score_normalized(a):
    """Compute the objective score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def adam_topk_logspace(a0, time_limit, T_init=0.03, T_final=0.0005, lr=0.01,
                        beta1=0.9, beta2=0.999, eps=1e-8, k_top=8,
                        warm_restart_interval=2000, warm_restart_scale=0.5,
                        verbose=False, verbose_interval=1000):
    """
    Adam optimizer in log-space with top-k focused soft-max and cosine annealing
    with warm restarts.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    # Larger FFT for gradient computation
    need = 3 * n - 2
    grad_fft_size = 1 << (need - 1).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = get_score_normalized(a)
    no_improve = 0
    max_no_improve = 5000
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Cosine annealing with warm restarts
        # Temperature follows cosine schedule, resets periodically
        cycle = int(elapsed / max(1, warm_restart_interval))
        cycle_progress = (elapsed % max(1, warm_restart_interval)) / max(1, warm_restart_interval)
        # Cosine from T_init to T_final
        T = T_final + 0.5 * (T_init - T_final) * (1 + np.cos(np.pi * cycle_progress))
        # Scale down with warm restarts
        T *= warm_restart_scale ** cycle
        T = max(T, 0.0001)
        
        # Current a from z
        a_curr = softmax_from_z(z)
        
        # Compute soft-max and weights (top-k focused)
        b = fast_convolve(a_curr)
        max_b = b.max()
        
        if k_top > 0 and k_top < len(b):
            w = topk_soft_max_weights(b, T, k=k_top)
        else:
            exp_b = np.exp(b / T - max_b / T)
            w = exp_b / exp_b.sum()
        
        # Gradient: d/dz [n * soft_max] = a * (grad_a - a^T grad_a)
        grad_a = compute_grad_a(a_curr, w, n, grad_fft_size)
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate
        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()
        
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Warm restart: if stuck, perturb from best
        if no_improve > max_no_improve:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.01
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            t = 0  # Reset Adam counter for bias correction
        
        if verbose and t % verbose_interval == 0:
            print(f"      Adam t={t} T={T:.6f} score={best_score:.7f}")
    
    return best_a, best_score


def coordinate_descent_true(a0, time_limit, n_sweeps=50, trials_per_coord=15):
    """
    Coordinate descent on the TRUE (non-smooth) objective.
    For each coordinate, try several values and keep the best.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = get_score_normalized(a)
    t0 = time.monotonic()
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        sweep_improved = False
        order = np.random.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            
            others_sum = 1.0 - a[i]
            if others_sum < 1e-15:
                continue
                
            # Try several values for a[i]
            # Include: current, 0, max (1-others_sum is not possible since sum=1)
            # a[i] can be in [0, 1] but must keep others >= 0
            # Since sum=1 and others sum to others_sum, a[i] is determined
            # Actually we can set a[i] to anything in [0, 1] and rescale others
            # But that changes all other coordinates. Let's just try scaling.
            
            # More practical: try a[i] * factor for various factors, then renormalize
            best_local_score = best_score
            best_local_factor = 1.0
            
            # Generate candidate factors based on current value
            factors = [0.5, 0.7, 0.85, 1.0/(0.85), 1.0/(0.7), 1.0/(0.5)]
            # Also try setting to 0
            factors = [0.0] + factors
            
            for f in factors:
                a_trial = a.copy()
                a_trial[i] = a[i] * f
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = get_score_normalized(a_trial)
                if score < best_local_score - 1e-15:
                    best_local_score = score
                    best_local_factor = f
            
            if best_local_factor != 1.0:
                a[i] = a[i] * best_local_factor
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True
        
        if not sweep_improve and sweep > 5:
            break
    
    return best_a, best_score


def peak_equalize(a0, time_limit, n_iterations=2000):
    """
    Direct peak equalization: find the max convolution peak, then reduce it
    by adjusting the contributing elements while trying to keep other peaks low.
    Uses a "water-filling" approach.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = get_score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find the argmax (prefer non-zero lags for more interesting adjustments)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        
        # The peak at lag k is sum_i a[i]*a[k-i]
        # Find the top contributors
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break
        contribs = a[idx] * a[peak_k - idx]
        
        # Get top 3 contributors
        top_idx = np.argsort(contribs)[::-1][:3]
        
        improved = False
        for ci in top_idx:
            i = idx[ci]
            j = peak_k - i
            if i > n - 1 or j > n - 1 or i < 0 or j < 0:
                continue
            if i == j:
                # Diagonal: reduce a[i]
                for frac in [0.9, 0.8, 0.7]:
                    a_trial = a.copy()
                    a_trial[i] *= frac
                    a_trial = a_trial / a_trial.sum()
                    score = get_score_normalized(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
            else:
                # Off-diagonal: reduce a[i] and/or a[j]
                for fi, fj in [(0.9, 1.0), (1.0, 0.9), (0.95, 0.95), (0.8, 1.05), (1.05, 0.8)]:
                    a_trial = a.copy()
                    a_trial[i] *= fi
                    a_trial[j] *= fj
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    score = get_score_normalized(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try a random perturbation
            if it % 100 == 0:
                idx_pert = np.random.choice(n, size=max(1, n//100), replace=False)
                for ip in idx_pert:
                    a_trial = a.copy()
                    a_trial[ip] *= 0.8
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = get_score_normalized(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        break
    
    return best_a, best_score


def lbgfs_polish(a0, time_limit, T=50):
    """L-BFGS optimization with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    stopped = [False]
    
    def soft_max_obj(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        return 2.0 * n * (max_b + T * np.log(np.sum(np.exp(b / T - max_b / T))))
    
    def soft_max_grad(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        grad_a = compute_grad_a(a, w, n, 1 << (3*n-3).bit_length())
        S = np.dot(grad_a, a)
        return a * (grad_a - S)
    
    try:
        res = minimize(
            fun=soft_max_obj,
            x0=z0,
            jac=soft_max_grad,
            method='L-BFGS-B',
            options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-10,
                     'maxfun': 50000, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
    except Exception as e:
        print(f"      L-BFGS error: {e}")
        a_out = a0.copy()
    
    return a_out


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n + 1)[:-1]  # Center of each bin
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_and_track(a, label, best_seq, best_score, print_flag=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if print_flag:
            print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score

        t_sub = min(rem * 0.8, 100.0)
        a_cd, _ = coordinate_descent_true(best_seq, t_sub, n_sweeps=30, trials_per_coord=8)
        best_seq, best_score = evaluate_and_track(a_cd, "coord_desc", best_seq, best_score)
        print(f"  After coord descent: {best_score:.7f}")
    
    # Another round of peak equalization
    rem = deadline - time.monotonic() - 130
    if rem > 15:
        t_sub = min(rem * 0.8, 60.0)
        a_pe2, _ = peak_equalize(best_seq, t_sub, n_iterations=3000)
        best_seq, best_score = evaluate_and_track(a_pe2, "peak_eq2", best_seq, best_score)
        print(f"  After peak equalize 2: {best_score:.7f}")
    
    # === Phase 4: L-BFGS polish & final (10% time) ===
    print(f"\n{'='*60}")
    print(f"Phase 4: L-BFGS Polish & Final Adjustments")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 30
    
    if rem > 30:
        # L-BFGS at different temperatures
        for T in [100, 30, 10, 3]:
            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break
            t_sub = min(rem * 0.35, 45.0)
            a_lb = lbgfs_polish(best_seq, t_sub, T=T)
            best_seq, best_score = evaluate_and_track(a_lb, f"lbgfs_T{T}", best_seq, best_score)
            print(f"  L-BFGS T={T}: {best_score:.7f}")
    
    # Final Adam with very low temp
    rem = deadline - time.monotonic() - 10
    if rem > 20:
        t_sub = min(rem * 0.6, 40.0)
        a_fin, _ = adam_topk_logspace(
            best_seq, t_sub, T_init=0.008, T_final=0.0001,
            lr=0.003, k_top=3, warm_restart_interval=5000
        )
        best_seq, best_score = evaluate_and_track(a_fin, "final_adam", best_seq, best_score)
    
    # Final peak equalization
    rem = deadline - time.monotonic() - 5
    if rem > 10:
        a_fin_pe, _ = peak_equalize(best_seq, rem * 0.8, n_iterations=2000)
        best_seq, best_score = evaluate_and_track(a_fin_pe, "final_pe", best_seq, best_score)
    
    # === Final verification ===
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\n{'='*60}")
        print(f"FINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
        print(f"{'='*60}")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.7f}")
    
    return [float(x) for x in best_seq]
