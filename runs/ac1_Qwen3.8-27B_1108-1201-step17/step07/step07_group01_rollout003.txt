import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(f)
        M = float(b.max())
        current_score = 2.0 * n * M

        if current_score < best_score - 1e-14:
            best_score = current_score
            best_f = f.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Solve LP for direction
        g = solve_lp_direction(f, M)
        if g is None:
            if stagnant > 100:
                break
            # Fallback: use subgradient
            k_peak = int(np.argmax(b))
            g = np.zeros(n)
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            g[idx] = f[k_peak - idx]
            g = g / (g.sum() + 1e-30)
        else:
            g = g / g.sum()

        # Line search
        best_t = 0.0
        best_trial = best_score
        for t in line_search_steps:
            trial = (1 - t) * f + t * g
            ts = score_from_a(trial)
            if ts < best_trial - 1e-15:
                best_trial = ts
                best_t = t

        if best_t == 0.0:
            if stagnant > 200:
                break
            # Try very small step
            f = f - 0.001 * np.abs(f) * 0.1
            f = np.maximum(f, 0)
            f = f / (f.sum() + 1e-30)
            continue

        f = (1 - best_t) * f + best_t * g
        f = np.maximum(f, 0)
        f = f / (f.sum() + 1e-30)

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} LP iter={it} ({elapsed:.0f}s): score={current_score:.7f} "
                  f"best={best_score:.7f} t={best_t:.3f}")

    return best_f, best_score


def nesterov_smoothed(a_init, time_limit, T=10.0, lr=0.5, beta=0.9,
                       max_iters=50000, verbose=True, label=""):
    """Nesterov-accelerated gradient on log-barrier smoothed objective."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    z = np.log(a)
    z = z - z.mean()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        a = softmax_vec(z)
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()

        # Compute gradient in log-space
        scaled = b / T
        m = scaled.max()
        w = np.exp(scaled - m)
        w = w / w.sum()

        # grad_a = 4*n * cross_corr(w, a)
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)[:n]
        grad_a = 4.0 * n * correl

        # Project to tangent space
        grad_a = grad_a - np.dot(a, grad_a)
        grad_z = a * grad_a

        # Nesterov update
        v = beta * v + (1 - beta) * grad_z
        step = lr / (1.0 + 0.01 * it)
        z = z - step * v
        z = z - z.mean()

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} Nest iter={it} ({elapsed:.0f}s): T={T} "
                  f"score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def softmax_vec(z):
    """Compute softmax of vector z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def subgradient_descent(a_init, time_limit, eta_start=0.1, decay=0.999,
                        max_iters=50000, verbose=True, label=""):
    """Multiplicative subgradient descent on exact objective."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    eta = eta_start

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()

        k_peak = int(np.argmax(b))
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * a[k_peak - idx]
        grad = grad - np.dot(a, grad)

        log_update = np.clip(-eta * grad, -2.0, 2.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        a = a / (a.sum() + 1e-30)
        eta *= decay

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} SubGD iter={it} ({elapsed:.0f}s): "
                  f"score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def coordinate_polish(a_init, time_limit, verbose=False, label=""):
    """Coordinate descent with proportional redistribution."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    factors = [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 300:
        no_improve = 0
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for f in factors:
                a_trial = a.copy()
                old = a_trial[i]
                a_trial[i] = old * f
                if a_trial[i] < 0:
                    continue
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others_sum
                if a_trial.min() < -1e-10:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / (a_trial.sum() + 1e-30)
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-13:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            no_improve += 1

    return best_a, best_score


def peak_shave(a_init, time_limit, verbose=False, label=""):
    """Targeted peak shaving of top contributing pairs."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    for step in range(5000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        if 2.0 * n * b[peak_k] <= best_score + 1e-14:
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:8]

        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.9, 0.8, 0.7, 0.5]:
                a_trial = a.copy()
                old = a_trial[i]
                a_trial[i] = old * factor
                diff = old - a_trial[i]
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / (a_trial.sum() + 1e-30)
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try reducing both i and j simultaneously
            for ci in top_idx[:5]:
                i = int(idx[ci])
                j = peak_k - i
                if i == j or a[j] < 1e-14:
                    continue
                for f1 in [0.8, 0.6]:
                    for f2 in [0.8, 0.6]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / (a_trial.sum() + 1e-30)
                        sc = score_from_a(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using evaluate_sequence and update if better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP-based Frank-Wolfe direction finding,
    Nesterov-accelerated smoothed gradients, subgradient descent, and polishing.
    Uses multiple initializations to explore the landscape.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score

    if best_seq is None:
        n = 768
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback init: n={n}, score={best_score:.7f}")

    # === Phase 1: LP-based optimization from parent ===
    print(f"\n=== Phase 1: LP Optimization (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 30 and best_seq is not None:
        t_alloc = min(rem * 0.4, 250.0)
        print(f"  LP from parent: n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_lp, _ = lp_optimize(best_seq, t_alloc, max_iters=3000, label="LP1")
        best_seq, best_score = check_and_update(a_lp, "LP_parent", best_seq, best_score)

    # === Phase 2: Nesterov smoothed from current best ===
    print(f"\n=== Phase 2: Nesterov Smoothed (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            for T in [50, 20, 10]:
                rem = deadline - time.monotonic() - 150
                if rem < 15:
                    break
                t_alloc = min(rem * 0.3, 60.0)
                print(f"  Nesterov T={T}: budget={t_alloc:.0f}s")
                a_ns, _ = nesterov_smoothed(best_seq, t_alloc, T=T, lr=0.3,
                                            beta=0.95, label=f"Nest_T{T}")
                best_seq, best_score = check_and_update(a_ns, f"Nest_T{T}",
                                                         best_seq, best_score)

    # === Phase 3: Subgradient descent ===
    print(f"\n=== Phase 3: Subgradient Descent (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc = min(rem * 0.3, 80.0)
            print(f"  SubGD: budget={t_alloc:.0f}s")
            a_sg, _ = subgradient_descent(best_seq, t_alloc, eta_start=0.05,
                                          decay=0.9995, label="SubGD")
            best_seq, best_score = check_and_update(a_sg, "SubGD", best_seq, best_score)

    # === Phase 4: LP from perturbed best ===
    print(f"\n=== Phase 4: LP from Perturbed (best={best_score:.7f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for pert_scale in [0.1, 0.05]:
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            t_alloc = min(rem * 0.3, 100.0)
            noise = rng.standard_normal(n_best) * pert_scale
            perturbed = np.maximum(best_seq + noise, 0)
            perturbed = perturbed / (perturbed.sum() + 1e-30)
            print(f"  LP perturb={pert_scale}: budget={t_alloc:.0f}s")
            a_lp2, _ = lp_optimize(perturbed, t_alloc, max_iters=2000,
                                   label=f"LP_p{pert_scale}")
            best_seq, best_score = check_and_update(a_lp2, f"LP_perturb{pert_scale}",
                                                     best_seq, best_score)

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_alloc = min(rem * 0.4, 60.0)
            print(f"  PS: budget={t_alloc:.0f}s")
            a_ps, _ = peak_shave(best_seq, t_alloc, label="PS")
            best_seq, best_score = check_and_update(a_ps, "peak_shave",
                                                     best_seq, best_score)

    # === Phase 6: Coordinate polish ===
    print(f"\n=== Phase 6: Coordinate Polish (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 10:
            t_alloc = min(rem * 0.4, 40.0)
            print(f"  CD: budget={t_alloc:.0f}s")
            a_cd, _ = coordinate_polish(best_seq, t_alloc, label="CD")
            best_seq, best_score = check_and_update(a_cd, "coord_desc",
                                                     best_seq, best_score)

    # === Phase 7: Try different initializations ===
    print(f"\n=== Phase 7: Alternative Inits (best={best_score:.7f}) ===")
    n_best = len(best_seq) if best_seq is not None else 768
    alternatives = []

    # Windowed sequences
    for win_name, win_func in [("hamming", np.hamming), ("hanning", np.hanning)]:
        x = np.linspace(0, np.pi, n_best)
        if win_name == "hamming":
            w = 0.54 - 0.46 * np.cos(x)
        else:
            w = 0.5 - 0.5 * np.cos(x)
        w = np.maximum(w, 0.01)
        w = w / w.sum()
        alternatives.append((f"win_{win_name}", w))

    # Cosine-modulated
    x = np.arange(n_best) / n_best
    for freq in [0.5, 1.0, 1.5, 2.0]:
        w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x)
        w = np.maximum(w, 0.01)
        w = w / w.sum()
        alternatives.append((f"cos_f{freq}", w))

    # Random
    for rs in [123, 456]:
        w = rng.default_rng(rs).standard_normal(n_best)
        w = np.abs(w) ** 2
        w = w / (w.sum() + 1e-30)
        alternatives.append((f"rand_{rs}", w))

    # Interpolated parent to different lengths then back
    for alt_n in [int(n_best * 1.5), int(n_best * 0.7)]:
        if 64 <= alt_n <= 4096 and alt_n != n_best:
            interp = interpolate_sequence(best_seq, alt_n) if best_seq is not None else np.ones(alt_n) / alt_n
            interp_back = interpolate_sequence(interp, n_best)
            alternatives.append((f"interp_{alt_n}", interp_back))

    for name, init_seq in alternatives[:4]:
        rem = deadline - time.monotonic() - 60
        if rem < 30:
            break
        t_alloc = min(rem * 0.3, 80.0)
        print(f"  Alt init '{name}': budget={t_alloc:.0f}s")
        # Quick LP from this init
        a_alt, _ = lp_optimize(init_seq, t_alloc, max_iters=1000, label=name)
        best_seq, best_score = check_and_update(a_alt, f"alt_{name}",
                                                 best_seq, best_score)

    # === Phase 8: Final LP refinement ===
    print(f"\n=== Phase 8: Final LP Refinement (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem * 0.5, 60.0)
            print(f"  Final LP: budget={t_alloc:.0f}s")
            a_fin, _ = lp_optimize(best_seq, t_alloc, max_iters=1500,
                                   label="FinalLP")
            best_seq, best_score = check_and_update(a_fin, "final_LP",
                                                     best_seq, best_score)

    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_ps2, _ = peak_shave(best_seq, min(rem * 0.5, 20.0), label="PS2")
            best_seq, best_score = check_and_update(a_ps2, "final_PS",
                                                     best_seq, best_score)

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_cd2, _ = coordinate_polish(best_seq, min(rem * 0.5, 10.0), label="CD2")
            best_seq, best_score = check_and_update(a_cd2, "final_CD",
                                                     best_seq, best_score)

    # Fallback
    if best_seq is None:
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
