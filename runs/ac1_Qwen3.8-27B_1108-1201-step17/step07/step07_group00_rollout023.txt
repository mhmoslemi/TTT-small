MWU + LP hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_subgradient(p, q):
    """Compute g_i = sum_k q[k] * p[k-i] using FFT."""
    n = len(p)
    # g_i = convolve(q, p[::-1])[n-1+i] for i = 0, ..., n-1
    # Actually, let me be more careful.
    # We want g_i = sum_k q[k] * p[k-i] for i = 0, ..., n-1
    # where p[j] = 0 for j < 0 or j >= n.
    
    # Use direct convolution via FFT
    # convolve(q, rev_p)[m] = sum_k q[k] * rev_p[m-k]
    # where rev_p[j] = p[n-1-j]
    # So convolve(q, rev_p)[m] = sum_k q[k] * p[n-1-m+k]
    # We want this to equal p[k-i], so n-1-m+k = k-i => m = n-1+i
    
    rev_p = p[::-1]
    size = len(q) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    Q = rfft(q, fft_size)
    P = rfft(rev_p, fft_size)
    convoluted = irfft(Q * P, fft_size)[:size]
    # g[i] = convoluted[n-1+i] for i = 0, ..., n-1
    g = convoluted[n-1:n-1+n]
    return g

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Hybrid MWU + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative weight updates with soft peak selection as the main engine,
complemented by LP-based direction finding and coordinate polish.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgradient(p, q):
    """Compute g_i = sum_k q[k] * p[k-i] using FFT cross-correlation."""
    n = len(p)
    if n == 0:
        return np.array([])
    rev_p = p[::-1]
    size = len(q) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    Q = rfft(q, fft_size)
    P = rfft(rev_p, fft_size)
    convoluted = irfft(Q * P, fft_size)[:size]
    g = convoluted[n - 1:n - 1 + n]
    return g


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_solve_direction(f, n, top_k):
    """Solve LP to find best direction g with bounded convolution at top peaks."""
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M
    
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=cp.ECOS, max_iters=100000, abs_acc=1e-8, rel_acc=1e-8)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-8)
        except Exception:
            return None, M
    
    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M

"""
Hybrid MWU + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Primary engine: Multiplicative Weight Updates with soft peak selection and
temperature annealing. Secondary: LP-based direction finding. Polish: coordinate descent.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgradient(p, q):
    """Compute g_i = sum_k q[k] * p[k-i] using FFT cross-correlation."""
    n = len(p)
    if n == 0:
        return np.array([])
    rev_p = p[::-1]
    size = len(q) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    Q = rfft(q, fft_size)
    P = rfft(rev_p, fft_size)
    convoluted = irfft(Q * P, fft_size)[:size]
    g = convoluted[n - 1:n - 1 + n]
    return g


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def mwu_phase(p_init, time_limit, lr_init=0.4, beta_init=30.0,
              beta_growth=1.5, n_top=100, label="mwu"):
    """MWU optimization with temperature annealing and restarts."""
    t0 = time.monotonic()
    n = len(p_init)
    p = p_init.copy()
    best_p = p.copy()
    b = fast_convolve(p)
    best_obj = b.max()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(99)

    while time.monotonic() - t0 < time_limit:
        it += 1
        lr = lr_init / (1.0 + 0.003 * it)
        beta = beta_init + beta_growth * it

        b = fast_convolve(p)
        b_max = b.max()

        # Soft best response with top-K restriction
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))

        b_sel = b[top_idx]
        b_shifted = b_sel - b_sel.max()
        q_sel = np.exp(beta * b_shifted)
        q_sum = q_sel.sum()
        if q_sum < 1e-30:
            q_sel = np.ones(len(top_idx)) / len(top_idx)
        else:
            q_sel = q_sel / q_sum

        q = np.zeros(len(b))
        q[top_idx] = q_sel

        # Compute subgradient via FFT
        g = compute_subgradient(p, q)

        # Stabilize
        g_mean = g.mean()
        g = g - g_mean

        # MWU update with clipping for numerical stability
        log_update = -lr * g
        log_update = np.clip(log_update, -40.0, 40.0)
        p_new = p * np.exp(log_update)
        s = p_new.sum()
        if s < 1e-30:
            p_new = p * (1.0 + 1e-8 * rng.standard_normal(n))
            s = p_new.sum()
        p = p_new / s

        obj = fast_convolve(p).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive restart
        if no_improve > 200:
            p = best_p.copy()
            no_improve = 0
            lr_init = max(lr_init * 0.85, 0.01)

        # Periodic noise injection to escape local structure
        if it % 500 == 0 and no_improve > 100:
            p = best_p.copy()
            noise = rng.standard_normal(n) * 0.002
            p = p + noise
            p = np.maximum(p, 0.0)
            p = p / p.sum()

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.12f} "
                  f"score={2*n*best_obj:.6f} beta={beta:.0f} "
                  f"lr={lr:.5f} ntop={n_top} t={elapsed:.0f}s")

    return best_p, best_obj

def lp_solve_direction(f, n, top_k):
    """Solve LP for best direction g with bounded convolution at top peaks."""
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])

    g_val = None
    for solver in [cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, max_iters=100000)
            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                if not np.any(np.isnan(g_val)):
                    break
        except Exception:
            continue

    if g_val is None or np.any(np.isnan(g_val)):
        return None, M
    return g_val, M
