```python
from scipy.optimize import linprog

def lp_expansion(f, M):
    """Find g maximizing sum(g) subject to g>=0 and f*g <= M."""
    n = len(f)
    # Variables: g[0], ..., g[n-1]
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_i f[i]*g[k-i] (where 0<=i<n and 0<=k-i<n)
    # This is linear in g: sum_{j} f[k-j]*g[j] for valid j
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            if 0 <= k-j < n:
                A_ub[k, j] = f[k-j]
    
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None
```

```python
def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_a(a):
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())
```

  `= M(1-2t+t^2) + 2tM - 2t^2 M + t^2 * max(g*g)`
  `= M - 2tM + t^2 M + 2tM - 2t^2 M + t^2 * max(g*g)`
  `= M + t^2(max(g*g) - M)`

So the new peak is at most `M + t^2 (max(g*g) - M)`. If `max(g*g) > M`, the peak could increase by `t^2 * (max(g*g) - M)`. For small t, this is negligible.

But if we use `g*g <= M` as the constraint instead of `a*g <= M`, then the new peak is at most M (guaranteed decrease). However, this is a much more restrictive constraint, and the LP might not find a good g.

Actually, using `g*g <= M` is not linear in g! `g*g` is quadratic. So we can't use it directly in an LP.

The LP uses `a*g <= M` which IS linear in g (since a is fixed). This is the correct approach.

So the theoretical guarantee is: the new peak is at most `M + t^2 * (max(g*g) - M)`. For the first few iterations, t could be moderate (say 0.1-0.5), and the increase is small. As we iterate, M decreases, and the relative increase becomes smaller.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based successive expansion
combined with coordinate descent polishing. The LP finds the optimal direction
to expand the sequence while maintaining the auto-convolution bound, providing
a principled descent direction for reducing the peak.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def build_A_ub(f, n):
    """Build constraint matrix for LP: A_ub[k,j] = f[k-j] if valid."""
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            A_ub[k, js] = f[k - js]
    return A_ub


def lp_expand(f, M):
    """Solve LP: max sum(g) s.t. g>=0, f*g <= M componentwise."""
    n = len(f)
    A_ub = build_A_ub(f, n)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_optimize(a_init, time_limit, verbose=True):
    """Iterative LP-based optimization."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    # Pre-compute sparsity pattern (booleans for non-zero entries)
    # Actually, let's just build the matrix each time since a changes
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        M = fast_convolve(a).max()
        
        # Solve LP
        t_lp_start = time.monotonic()
        g = lp_expand(a, M)
        t_lp = time.monotonic() - t_lp_start
        
        if g is None or g.sum() < 1e-10:
            break
        
        g = g / g.sum()
        
        # Adaptive step size: try to make progress
        # Compute max(g*g) to estimate the quadratic term
        g_peak = fast_convolve(g).max()
        
        # Choose t such that the expected new peak <= M (or close)
        # New peak <= (1-t)^2 * M + 2t(1-t) * M + t^2 * g_peak
        #          = M * (1-t)^2 + 2t(1-t) * M + t^2 * g_peak
        #          = M - 2tM + t^2 M + 2tM - 2t^2 M + t^2 * g_peak
        #          = M + t^2 * (g_peak - M)
        # So if g_peak <= M, any t works. If g_peak > M, we need t <= sqrt(M/(g_peak-M))
        if g_peak > M:
            t_max = np.sqrt(M / (g_peak - M))
            t = min(0.5, t_max * 0.9)
        else:

        # Ensure t is in a reasonable range
        t = max(t, 0.01)

        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 1e-15)

            a = a_new

            a_new = (1 - t) * a + t * g

def coordinate_descent(a_init, time_limit, verbose=True):
    """Coordinate descent with proportional redistribution."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    factors = [0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5]
    consecutive_no_improve = 0
    max_no_improve = 300
    
    while time.monotonic() - t0 < time_limit and consecutive_no_improve < max_no_improve:
        consecutive_no_improve = 0
        improved_pass = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for f in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                new_val = old_val * f
                a_trial[i] = new_val
                diff = a_trial.sum() - 1.0
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others_sum
                if a_trial.min() < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved_pass = True
                    break
        if not improved_pass:
            consecutive_no_improve += 1
    
    return best_a, best_score


def peak_shave(a_init, time_limit, verbose=True):
    """Targeted peak shaving."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    for step in range(20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        
        # Also consider the main peak
        all_peaks = np.argsort(b)[::-1][:5]
        
        improved = False
        for peak_k in all_peaks:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_idx = np.argsort(contribs)[::-1][:5]
            
            for ci in top_idx:
                i = int(idx[ci])
                for factor in [0.95, 0.9, 0.85, 0.8, 0.7]:
                    a_trial = a.copy()
                    old_val = a_trial[i]
                    a_trial[i] = old_val * factor
                    diff = old_val - a_trial[i]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = score_from_a(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_a, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based successive expansion
(constraint generation LP to find optimal expansion direction) combined with
coordinate descent and peak shaving for final polishing. The LP approach
finds the best direction to grow the sequence while maintaining the
auto-convolution bound, providing a principled descent direction.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve_f(g, f):
    """Compute f*g (cross-convolution) via FFT, where f and g have same length."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def build_constraint_row(k, n, f):
    """Build a single constraint row for lag k."""
    row = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        js = np.arange(j_start, j_end + 1)
        row[js] = f[k - js]
    return row


def lp_expand_cg(f, M, max_cg=30, active_frac=0.3):
    """LP with constraint generation: max sum(g) s.t. g>=0, f*g <= M."""
    n = len(f)
    
    # Determine initial active constraints based on f*f peaks
    ff = fast_convolve(f)
    threshold = M * (1.0 - active_frac)
    active_k = np.where(ff >= threshold)[0]
    # Always include boundaries and center
    active_k = np.unique(np.concatenate([active_k, [0, n - 1, 2 * n - 2]]))
    
    # Build initial constraint matrix
    A_ub = np.zeros((len(active_k), n))
    for i, k in enumerate(active_k):
        A_ub[i] = build_constraint_row(int(k), n, f)
    b_ub = np.full(len(active_k), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    g = None
    for cg_iter in range(max_cg):
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        g = res.x
        
        # Check all constraints
        fg = cross_convolve_f(g, f)
        violated = np.where(fg > M + 1e-9)[0]
        if len(violated) == 0:
            break
        
        # Add violated constraints
        for k in violated:
            row = build_constraint_row(int(k), n, f)
            A_ub = np.vstack([A_ub, row.reshape(1, -1)])
            b_ub = np.append(b_ub, M)
    
    return g


def lp_optimize(a_init, time_limit, verbose=True):
    """Iterative LP-based optimization with adaptive step size."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    iteration = 0
    no_improve_count = 0
    max_no_improve = 50
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        M = fast_convolve(a).max()
        
        # Solve LP
        t_lp_start = time.monotonic()
        g = lp_expand_cg(a, M)
        t_lp = time.monotonic() - t_lp_start
        
        if g is None or g.sum() < 1e-10:
            break
        
        g = np.maximum(g, 0.0)
        g = g / g.sum()
        
        # Compute g*g peak to estimate step size
        g_peak = fast_convolve(g).max()
        
        # Adaptive step
        if g_peak > M:
            t_max = np.sqrt(M / (g_peak - M))
            t = min(0.5, t_max * 0.8)
        else:
            t = 0.5
        t = max(t, 0.01)
        
        # Try the update
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.1]:
            t_try = t * factor
            a_new = (1 - t_try) * a + t_try * g
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            new_score = score_from_a(a_new)
            if new_score < best_score - 1e-15:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        
        if not improved:
            no_improve_count += 1
            if no_improve_count > max_no_improve:
                break
            # Try a random perturbation to escape
            if no_improve_count % 10 == 0:
                rng = np.random.default_rng(iteration * 31)
                noise = rng.standard_normal(n) * 0.05
                a = best_a + noise
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
        else:
            no_improve_count = 0
        
        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={iteration} ({elapsed:.0f}s): best={best_score:.7f} "
                  f"t={t:.3f} lp_time={t_lp:.3f}s")
    
    return best_a, best_score


def mirror_descent_step(a, M, T, eta):
    """Single mirror descent step using soft-max weights."""
    n = len(a)
    b = fast_convolve(a)
    scaled = b / T
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()
    
    # Gradient: 2n * cross-correlation of w with a
    # grad[j] = 2n * sum_k w[k] * a[k-j] (for valid indices)
    grad = np.zeros(n)
    fft_size = 1 << (3 * n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    correl = irfft(np.conj(W) * A, fft_size)
    grad = 2.0 * n * correl[:n]
    
    log_update = -eta * grad
    log_update = np.clip(log_update, -3.0, 3.0)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-300)
    a_new = a_new / a_new.sum()
    return a_new


def mirror_descent_optimize(a_init, time_limit, T_start=0.01, T_end=0.001,
                            eta_start=0.02, eta_end=0.001, verbose=False):
    """Mirror descent with annealing as fallback/refinement."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    max_iters = 20000
    
    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        progress = min(1.0, it / (0.8 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress
        
        a = mirror_descent_step(a, best_score / (2 * n), T, eta)
        
        current_score = score_from_a(a)
        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()
    
    return best_a, best_score


def coordinate_descent(a_init, time_limit, verbose=False):
    """Coordinate descent with proportional redistribution."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    factors = [0.6, 0.7, 0.8, 0.9, 1.1, 1.2, 1.4]
    consecutive_no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - t0 < time_limit and consecutive_no_improve < max_no_improve:
        consecutive_no_improve = 0
        improved_pass = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for f in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * f
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others_sum
                if a_trial.min() < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved_pass = True
                    break
        if not improved_pass:
            consecutive_no_improve += 1
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
```
