def fast_convolve(a):

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score_normalized(a):

def adam_pnorm_optimize(a0, time_limit, p, lr_init=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimization with p-norm smoothing on the simplex."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        grad = pnorm_gradient(a, p)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr_init / (t ** 0.5) * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex (a >= 0, sum(a) = 1)
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / ss
        
        score = compute_score_normalized(a_new)
        a = a_new
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
        
        if t % 1000 == 0:
            print(f"    Adam-p={p} iter={t}, score={best_score:.8f}")
    
    return best_a, best_score

def fictitious_play(a0, time_limit, n_iter_max=5000):
    """
    Fictitious play / multiplicative weight iteration.
    At each step:
    1. Compute M = max(f*f)
    2. Solve LP: max sum(g) s.t. f*g <= M for all lags, g >= 0
    3. Normalize g to sum=1
    4. Update f = (1-t)*f + t*g
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    
    # We'll use a simplified version: instead of full LP, use multiplicative weight updates
    # This is faster and often works well
    lambda_vec = np.ones(2*n-1) / (2*n-1)  # uniform distribution over lags
    eta = 0.1  # step size
    
    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Multiplicative weight update on the lag distribution
        # lambda_k proportional to exp(-eta * (a*a)_k / M) ... 
        # Actually, the dual of the LP "max sum(g) s.t. (f*g)_k <= M, g>=0"
        # has variables lambda_k >= 0, sum lambda_k <= 1
        # The primal solution g is determined by the active constraints
        # 
        # Fictitious play: update lambda based on which constraints are violated
        # Since (a*a)_k varies, we update lambda to concentrate on high (a*a)_k
        # Then the next g should satisfy (a*g)_k <= M * lambda_k / sum(lambda) ... 
        
        # Simpler approach: use the current a to define weights, then find g
        # that equalizes (a*g)_k across the top lags
        
        # For the multiplicative weight approach:
        # weight_k = exp(eta * conv_k / M)
        # Then the "best response" g satisfies (a*g)_k = M * weight_k / sum(weights)
        # for active k, and (a*g)_k < M * weight_k / sum(weights) for inactive k
        
        # This is complex. Let's use a simpler gradient-based update instead.
        # The subgradient of max_k (a*a)_k at the current a is 2*a_{k*-i} where k* = argmax
        # We want to decrease this, so move a in the negative gradient direction
        # while staying on the simplex.
        
        k_star = np.argmax(conv)
        # Subgradient: d(a*a)_{k*}/da_i = 2*a_{k*-i} (for valid k*-i)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        subgrad = np.zeros(n)
        subgrad[idx] = 2 * a[k_star - idx]
        
        # Projected gradient step (stay on simplex)
        a_new = a - eta * subgrad
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss > 1e-15:
            a_new = a_new / ss
        
        score = compute_score_normalized(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        eta *= 0.999
        
        if it % 500 == 0:
            print(f"    FP iter={it}, score={best_score:.8f}, eta={eta:.6f}")
    
    return best_a, best_score

def lbgfs_multi_temperature(a0, time_limit, temps=[100, 300, 1000, 3000], max_per_T=2000):
    """L-BFGS with log-sum-exp smoothing at multiple temperatures."""
    n = len(a0)
    best_a = a0.copy()
    best_score = compute_score_normalized(a0)
    t0 = time.monotonic()
    
    for T in temps:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        t_alloc = min(rem * 0.4, 60.0)
        
        z0 = np.log(np.maximum(a0, 1e-30))
        
        def func(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a = exp_z / exp_z.sum()
            conv = fast_convolve(a)
            shifted = conv / T
            m = shifted.max()
            lse = m + np.log(np.exp(shifted - m).sum())
            return (2.0 * n / T) * np.exp(T * lse)
        
        def jac(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a = exp_z / exp_z.sum()
            conv = fast_convolve(a)
            shifted = conv / T
            m = shifted.max()
            w = np.exp(shifted - m)
            w = w / w.sum()
            size = 2 * n - 1
            fft_size = 1 << (size - 1).bit_length()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            return (2.0 * n / T) * grad_z
        
        try:
            res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_per_T, 'ftol': 1e-15, 'gtol': 1e-12})
            z_max = res.x.max()
            a_out = np.exp(res.x - z_max)
            a_out = a_out / a_out.sum()
            sc = compute_score_normalized(a_out)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_out.copy()
                a0 = a_out.copy()
        except Exception:
            pass
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:

import numpy as np
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score_normalized(a):
    """Compute score for normalized sequence (sum=1). Returns 2*n*max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def normalize_simplex(a):
    """Project onto probability simplex (non-negative, sum=1)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        return a / s
    return np.ones(len(a)) / len(a)


def lse_objective(z, n, T):
    """Log-sum-exp smoothed objective in log-space."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return 1e15
    a = exp_z / s
    conv = fast_convolve(a)
    shifted = conv / T
    m = shifted.max()
    lse = m + np.log(np.exp(shifted - m).sum())
    return (2.0 * n / T) * np.exp(T * lse)


def lse_gradient(z, n, T):
    """Gradient of log-sum-exp smoothed objective in log-space."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.zeros(n)
    a = exp_z / s
    conv = fast_convolve(a)
    shifted = conv / T
    m = shifted.max()
    w = np.exp(shifted - m)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def pnorm_value_grad(a, p):
    """Compute value and gradient of p-norm of auto-convolution."""
    n = len(a)
    conv = fast_convolve(a)
    conv_clip = np.maximum(conv, 1e-30)
    pth_powers = conv_clip ** p
    total = pth_powers.sum()
    if total < 1e-300:
        return 0.0, np.zeros(n)
    value = total ** (1.0 / p)
    weights = conv_clip ** (p - 1)
    size_c = len(conv)
    fft_size = 1 << (size_c + n - 1 - 1).bit_length()
    W = rfft(weights, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * np.conj(A), fft_size)[:n]
    norm_factor = total ** (1.0 - 1.0 / p)
    if norm_factor < 1e-300:
        return value, np.zeros(n)
    grad = 2.0 * cross / norm_factor
    return value, grad


def run_lbgfs(a0, time_limit, T, maxiter=10000):
    """Run L-BFGS with log-sum-exp smoothing for a given temperature."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-30))
    best_a = a0.copy()
    best_score = compute_score_normalized(a0)
    t0 = time.monotonic()

    # Check time budget rough estimate
    # Each L-BFGS iteration does ~4 function evals, each O(n log n)
    estimated_time_per_iter = 4 * (time.time() - time.time()) + 0.001  # rough
    max_iter_use = min(maxiter, max(100, int(time_limit / max(estimated_time_per_iter, 0.001))))

    try:
        res = minimize(
            fun=lse_objective, x0=z0, jac=lse_gradient,
            args=(n, T), method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        z_out = res.x
        z_max = z_out.max()
        a_out = np.exp(z_out - z_max)
        a_out = a_out / a_out.sum()
        sc = compute_score_normalized(a_out)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
    except Exception:
        pass

    return best_a, best_score, time.monotonic() - t0


def run_adam_pnorm(a0, time_limit, p, lr_init=0.005, beta1=0.9, beta2=0.999):
    """Adam optimizer with p-norm smoothing on the simplex."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    eps = 1e-8
    no_improve = 0
    stall_threshold = 2000

    while time.monotonic() - t0 < time_limit:
        t += 1
        val, grad = pnorm_value_grad(a, p)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_init / (t ** 0.5)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = normalize_simplex(a_new)

        sc = compute_score_normalized(a_new)
        a = a_new

        if sc < best_score - 1e-13:

            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

            rng = np.random.default_rng(t)
            noise = rng.standard_normal(n) * 0.005
            a = normalize_simplex(best_a + noise)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0

    return best_a, best_score


def run_subgradient_momentum(a0, time_limit, lr_init=0.02, momentum=0.9):
    """Subgradient descent with Nesterov momentum on the simplex."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    v = np.zeros(n)
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        # Nesterov lookahead
        a_la = normalize_simplex(a - momentum * v)

        # Compute subgradient (focus on top peaks)
        conv = fast_convolve(a_la)
        M = conv.max()
        # Weighted subgradient: weight peaks proportional to (conv_k / M)^2
        weights = (conv / M) ** 4
        w_sum = weights.sum()
        if w_sum < 1e-30:
            break
        weights = weights / w_sum

        # Gradient = sum_k weights_k * 2 * a_{k-i}
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(weights, fft_size)
        A = rfft(a_la, fft_size)
        grad = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]

        # Update
        v = momentum * v + lr_init / (iters ** 0.5) * grad
        a_new = normalize_simplex(a - v)

        sc = compute_score_normalized(a_new)
        a = a_new

        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000:
            # Perturb and restart
            rng = np.random.default_rng(iters)
            a = normalize_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_score


def run_peak_balance(a0, time_limit, n_top=8):
    """
    Peak balancing: iteratively try to equalize the top peaks.
    At each step, identify the highest and lowest among top peaks,
    and make moves to reduce the spread.
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        # Find top peaks
        top_indices = np.argpartition(conv, -n_top)[-n_top:]
        top_vals = conv[top_indices]
        spread = top_vals.max() - top_vals.min()

        if spread < 1e-10 * M:
            break

        k_hi = top_indices[np.argmax(top_vals)]
        k_lo = top_indices[np.argmin(top_vals)]

        # Get contributing pairs for k_hi
        js_hi = max(0, k_hi - n + 1)
        je_hi = min(n - 1, k_hi)
        idx_hi = np.arange(js_hi, je_hi + 1)
        contribs_hi = a[idx_hi] * a[k_hi - idx_hi]

        improved = False

        # Try reducing top contributors to k_hi
        top_contrib = idx_hi[np.argsort(contribs_hi)[::-1][:8]]
        for i in top_contrib:
            if a[i] < 1e-12:
                continue
            for frac in [0.02, 0.05, 0.1, 0.15, 0.2]:
                delta = frac * a[i]
                # Try redistributing to positions that help k_lo
                js_lo = max(0, k_lo - n + 1)
                je_lo = min(n - 1, k_lo)
                targets = list(range(js_lo, je_lo + 1))
                rng.shuffle(targets)

                for j in targets[:5]:
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[j] += delta
                    a_trial = normalize_simplex(a_trial)
                    sc = compute_score_normalized(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random perturbation focused on peak positions
            no_improve += 1
            if no_improve % 100 == 0:
                peak_pos = set()
                for pi in top_indices:
                    si = max(0, int(pi) - n + 1)
                    ei = min(n - 1, int(pi))
                    peak_pos.update(range(si, ei + 1))
                if peak_pos:
                    pos_list = list(peak_pos)
                    p = pos_list[rng.integers(0, len(pos_list))]
                    a_trial = a.copy()
                    delta = 0.05 * a[p] * rng.standard_normal()
                    a_trial[p] -= delta
                    j = rng.integers(0, n)
                    a_trial[j] += delta
                    a_trial = normalize_simplex(a_trial)
                    sc = compute_score_normalized(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        no_improve = 0

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving non-negativity and sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1.0, old_n)
    new_x = np.linspace(0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
