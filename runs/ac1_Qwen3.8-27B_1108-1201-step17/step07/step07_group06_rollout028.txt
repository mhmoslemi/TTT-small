    """Project v onto {x >= 0, sum(x) = 1}"""
    n = len(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play (best response
dynamics), Nesterov-accelerated subgradient descent with active-set refinement,
and targeted coordinate descent polishing. Key differences from the previous soft-max
Adam approach: (1) LP best-response provides guaranteed descent with exact active set,
(2) Nesterov subgradient uses exact (non-smoothed) subgradients for faster convergence,
(3) Polyak step size adapts to the optimization gap, (4) multi-scale exploration
across sequence lengths with interpolation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_constraint_matrix(a):
    """Build A_ub for LP: row k has A_ub[k,i] = a[k-i] if 0<=k-i<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi + 1] = a[k - lo:k - hi - 1:-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M elementwise, b >= 0. Returns normalized b."""
    n = len(a)
    A_ub = build_constraint_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0)
            s = b.sum()
            if s > 1e-10:
                return b / s
    except Exception:
        pass
    return np.ones(n) / n

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    return best_seq, best_score

    rem = deadline - time.monotonic() - 30.0
    if rem > 60:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [1.5, 2.0, 0.75, 1.25, 3.0]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 8192 and alt_n != n_best:
                if alt_n not in alt_lengths:
                    alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:4]:
            rem2 = deadline - time.monotonic() - 20.0
            if rem2 < 40:
                break
            t_sub = min(rem2 * 0.3, 90.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            # Quick NS on interpolated
            a_alt, _ = nesterov_subgradient(init_alt, t_sub, rel_tol=0.01, top_k=8, verbose=False)
            best_seq, best_score = evaluate_track(a_alt, f"alt_n{alt_n}", best_seq, best_score)
            print(f"  Alt n={alt_n}: {best_score:.6f}")

    # === Phase 5: Final Intensive Polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 10.0
    if rem > 20:
        t_sub = min(rem * 0.5, 60.0)
        a_fin, _ = coordinate_descent_peaks(best_seq, t_sub, top_peak_k=3, n_top_contribs=12)
        best_seq, best_score = evaluate_track(a_fin, "final_cd", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.6f}")

    return [float(x) for x in best_seq]
