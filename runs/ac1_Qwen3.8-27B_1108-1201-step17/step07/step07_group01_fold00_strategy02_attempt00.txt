def fast_convolve(a):
    """Compute linear auto-convolution efficiently via FFT."""
    n = len(a)
    conv_length = 2 * n - 1
    # Next power of 2 for FFT
    fft_size = 1 << (conv_length - 1).bit_length()
    ffted = rfft(a, fft_size)
    # Element-wise multiplication in frequency domain
    conv_fft = np.conj(ffted) * ffted
    # IFFT and crop to the required length
    conv_result = irfft(conv_fft, fft_size)[:conv_length]
    return conv_result
