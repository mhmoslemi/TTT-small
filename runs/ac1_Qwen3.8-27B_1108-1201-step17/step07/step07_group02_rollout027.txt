"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an aggressive LP-based alternating
method with adaptive damping, perturbation-based restarts, and L-BFGS refinement.
Key improvements over previous approach:
1. Gauss-Seidel style LP alternation (immediate use of new solution as anchor)
2. Adaptive damping with oscillation detection
3. Periodic perturbation to escape plateaus
4. Active-set QP refinement after LP steps
5. More time allocated to LP-based method
6. Multi-length exploration
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver_cached' in globals():
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def solve_lp_max_sum(a_fixed, M_bound, solver=None):
    """Given fixed a and bound M, find b >= 0 maximizing sum(b) s.t. conv(a,b) <= M."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_bound]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def lbgfs_soft_max(a0, time_limit, T_start=100.0, T_end=5.0, maxiter=5000):
    """L-BFGS refinement with soft-max smoothing and adaptive temperature."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    stopped = [False]

    def make_func(T_val):
        def func(z):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return 1e15
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return np.zeros_like(z)
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 5
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = maxiter // n_stages

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining_time = time_limit - (time.monotonic() - t0)
        if remaining_time < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxfun': iters_per_stage * 50, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = softmax_from_z(z)
    return a_out


def lp_alternate_optimize(a0, time_limit, solver=None, n_max_iters=50000, rng=None):
    """
    Aggressive LP-based alternating optimization with adaptive damping and perturbation.
    Uses Gauss-Seidel style with immediate feedback and oscillation detection.
    """
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Adaptive damping
    alpha = 1.0  # Start with full step
    recent_scores = []
    no_improve_count = 0
    perturbation_interval = 30

    last_perturb = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Every few iterations, add small perturbation if stuck
        if no_improve_count > perturbation_interval and it - last_perturb > perturbation_interval:
            noise = rng.normal(0, 0.01, size=n)
            a_perturbed = a + noise
            a_perturbed = np.maximum(a_perturbed, 0)
            sp = a_perturbed.sum()
            if sp > 1e-10:
                a_perturbed = a_perturbed / sp
                sc_pert = 2.0 * n * fast_convolve(a_perturbed).max()
                if sc_pert < best_score * 1.01:  # Only accept if not too bad
                    a = a_perturbed
                last_perturb = it
                no_improve_count = 0
                perturbation_interval = max(10, perturbation_interval // 2)

        # Solve LP best response
        b_new, M_lp = solve_lp_best_response(a, solver)
        if b_new is None:
            continue

        # Compute score of LP solution
        score_b = 2.0 * n * fast_convolve(b_new).max()

        # Adaptive mixing
        if score_b < best_score - 1e-10:
            a_new = (1 - alpha) * a + alpha * b_new
            a_new = np.maximum(a_new, 0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = b_new.copy()
                sa = 1.0
            a_new = a_new / sa
            score_new = 2.0 * n * fast_convolve(a_new).max()

            if score_new < best_score - 1e-12:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new
                no_improve_count = 0
                # Increase alpha if improving fast
                if score_new < score_b:
                    alpha = min(1.0, alpha * 1.05)
                else:
                    alpha = max(0.3, alpha * 0.95)
            else:
                no_improve_count += 1
                alpha = max(0.3, alpha * 0.9)

            # Check oscillation: track recent scores
            recent_scores.append(best_score)
            if len(recent_scores) > 20:
                recent_scores.pop(0)
            if len(recent_scores) >= 20:
                first_half = np.mean(recent_scores[:10])
                second_half = np.mean(recent_scores[10:])
                if second_half > first_half * 1.001:
                    # Oscillation detected, reduce alpha
                    alpha = max(0.2, alpha * 0.7)
        else:
            # LP solution is not better, reduce alpha
            alpha = max(0.2, alpha * 0.95)
            no_improve_count += 1

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-alt iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} alpha={alpha:.3f} M_lp={M_lp:.8f}")

    return best_a, best_score


def active_set_qp_refine(a0, time_limit, solver=None, n_iters=200):
    """
    QP refinement using only active constraints.
    At each step, identify which convolution constraints are near-active,
    then solve a QP with just those constraints.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find active and near-active constraints
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]

        if len(active) < 3:
            threshold = max_b * 0.99
            active = np.where(b >= threshold)[0]

        if len(active) < 3:
            active = np.where(b >= max_b * 0.95)[0]

        # Build reduced constraint matrix
        T_active = np.zeros((len(active), n))
        for idx, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                T_active[idx, j] = a[k - j]

        # Solve QP: minimize t s.t. T_active @ x <= t, sum(x)=1, x>=0
        x_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [T_active @ x_var <= t_var, cp.sum(x_var) == 1]
        # Add proximity to current solution
        obj = t_var + 0.01 * cp.sum_squares(x_var - a)
        prob = cp.Problem(cp.Minimize(obj), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        x_val = np.asarray(x_var.value, dtype=np.float64)
        x_val = np.maximum(x_val, 0)
        sx = x_val.sum()
        if sx < 1e-15:
            continue
        x_val = x_val / sx

        score_x = 2.0 * n * fast_convolve(x_val).max()
        if score_x < best_score - 1e-12:
            best_score = score_x
            best_a = x_val.copy()
            a = x_val

        if (it + 1) % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    QP-refine iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} active={len(active)}")

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=10000):
    """Aggressive local peak shaving with multiple coordinate adjustments."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Skip lag 0 (it's always the max for non-negative sequences often)
        # Actually, let's find the true max
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_contrib = np.argsort(contribs)[::-1][:8]

        improved = False
        for ci in top_contrib:
            i = idx[ci]
            j = peak_k - i
            if i == j:
                j = None
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.95, 0.92, 0.88, 0.82]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                # Redistribute diff to other elements proportionally
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                if j is not None and a[j] > 0:
                    # Give half to j, half to rest
                    a_trial[j] += diff * 0.3
                    mask[j] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * 0.7 * a_trial[mask] / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try simultaneous reduction of top contributors
            if len(top_contrib) >= 2:
                improved2 = False
                for ci1 in top_contrib[:4]:
                    for ci2 in top_contrib[:4]:
                        if ci1 == ci2:
                            continue
                        i1 = idx[ci1]
                        i2 = idx[ci2]
                        if a[i1] < 1e-14 or a[i2] < 1e-14:
                            continue
                        for f1 in [0.96, 0.93]:
                            for f2 in [0.96, 0.93]:
                                a_trial = a.copy()
                                a_trial[i1] *= f1
                                a_trial[i2] *= f2
                                a_trial = np.maximum(a_trial, 0)
                                sa = a_trial.sum()
                                if sa < 1e-15:
                                    continue
                                a_trial = a_trial / sa
                                score = 2.0 * n * fast_convolve(a_trial).max()
                                if score < best_score - 1e-14:
                                    a = a_trial
                                    best_score = score
                                    best_a = a.copy()
                                    improved = True
                                    improved2 = True
                                    break
                            if improved2:
                                break
                        if improved2:
                            break
                    if improved2:
                        break
            if not improved:
                # Try reducing all top 5 slightly
                a_trial = a.copy()
                for ci in top_contrib[:5]:
                    i = idx[ci]
                    a_trial[i] *= 0.97
                a_trial = np.maximum(a_trial, 0)
                sa = a_trial.sum()
                if sa > 1e-15:
                    a_trial = a_trial / sa
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                if not improved:
                    # Stuck, try random perturbation
                    mask = np.random.default_rng(step).choice(n, size=min(10, n), replace=False)
                    a_trial = a.copy()
                    for i in mask:
                        a_trial[i] *= np.random.default_rng(step + i).uniform(0.8, 1.2)
                    a_trial = np.maximum(a_trial, 0)
                    sa = a_trial.sum()
                    if sa > 1e-15:
                        a_trial = a_trial / sa
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score * 1.001:
                            a = a_trial  # Accept even if slightly worse (exploration)
                            if score < best_score:
                                best_score = score
                                best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely, returning inf on failure."""
    try:
        a_list = [float(max(0, min(1000, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def mirror_descent_fast(a0, time_limit, lr_init=0.05, n_steps=20000):
    """Fast mirror descent with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    z = z_from_a(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find active indices
        threshold = max_b - 0.05 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Mirror descent update
        z_new = z - lr * g
        a_new = softmax_from_z(z_new)
        score_new = 2.0 * n * fast_convolve(a_new).max()

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                lr *= 0.8
                no_improve = 0
        lr = max(lr, 1e-7)

        if step % 3000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP-based alternation, L-BFGS,
    active-set QP, and peak shaving for minimizing auto-convolution peak.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick L-BFGS to ensure good basin (150s) ===
    print(f"\n=== Phase 1: L-BFGS Init (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 20:
        print(f"  L-BFGS T: 50->5, n={n_best}, budget={t_alloc:.0f}s")
        a_lb = lbgfs_soft_max(best_seq, t_alloc, T_start=50.0, T_end=5.0, maxiter=4000)
        try_update(a_lb, "lbgfs_init")

    # === Phase 2: LP-based alternation (main workhorse, 700s) ===
    print(f"\n=== Phase 2: LP Alternation (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 700.0)
    if t_alloc > 30:
        print(f"  LP-alt n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_lp, _ = lp_alternate_optimize(best_seq, t_alloc, solver=solver, rng=rng)
        try_update(a_lp, "lp_alt")

        # Second round of LP alternation from the improved point
        rem = deadline - time.monotonic()
        t_alloc2 = min(rem * 0.15, 250.0)
        if t_alloc2 > 30:
            print(f"  LP-alt round 2, budget={t_alloc2:.0f}s")
            a_lp2, _ = lp_alternate_optimize(best_seq, t_alloc2, solver=solver, rng=rng)
            try_update(a_lp2, "lp_alt2")

    # === Phase 3: Active-set QP refinement (200s) ===
    print(f"\n=== Phase 3: Active-Set QP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        print(f"  QP-refine n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_qp, _ = active_set_qp_refine(best_seq, t_alloc, solver=solver)
        try_update(a_qp, "qp_refine")

    # === Phase 4: Mirror descent (150s) ===
    print(f"\n=== Phase 4: Mirror Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        print(f"  MirrorDesc n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_md, _ = mirror_descent_fast(best_seq, t_alloc, lr_init=0.03)
        try_update(a_md, "mirror_desc")

    # === Phase 5: L-BFGS low-temperature refinement (200s) ===
    print(f"\n=== Phase 5: L-BFGS Low-T (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        print(f"  L-BFGS T: 3->0.5, n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_lb2 = lbgfs_soft_max(best_seq, t_alloc, T_start=3.0, T_end=0.5, maxiter=5000)
        try_update(a_lb2, "lbgfs_lowT")

    # === Phase 6: Peak shaving (200s) ===
    print(f"\n=== Phase 6: Peak Shaving (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 15:
        print(f"  PeakShave n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_ps, _ = peak_shave_local(best_seq, t_alloc)
        try_update(a_ps, "peak_shave")

    # === Phase 7: Try alternative lengths (150s) ===
    print(f"\n=== Phase 7: Alternative Lengths (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = [int(n_best * 1.2), int(n_best * 0.85), int(n_best * 1.5)]
    alt_lengths = [x for x in alt_lengths if 384 <= x <= 3072 and x != n_best]
    for alt_n in alt_lengths[:2]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(rem * 0.4, 70.0)
        if t_alloc < 20:
            break
        print(f"  Alt length n={alt_n}, budget={t_alloc:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt = lbgfs_soft_max(init_alt, t_alloc, T_start=30.0, T_end=3.0, maxiter=3000)
        try_update(a_alt, f"alt_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None and len(best_seq) == alt_n:
            t_lp = min(rem * 0.4, 50.0)
            if t_lp > 15:
                a_al, _ = lp_alternate_optimize(best_seq, t_lp, solver=solver, rng=rng)
                try_update(a_al, f"lp_alt_n{alt_n}")

    # === Phase 8: Final LP + polish cycle (100s) ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t1 = min(rem * 0.4, 50.0)
        if t1 > 15:
            a_f1, _ = lp_alternate_optimize(best_seq, t1, solver=solver, rng=rng)
            try_update(a_f1, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 20:
        t2 = min(rem * 0.3, 40.0)
        if t2 > 10:
            a_f2, _ = peak_shave_local(best_seq, t2, n_steps=5000)
            try_update(a_f2, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 15:
        t3 = min(rem * 0.5, 30.0)
        a_f3 = lbgfs_soft_max(best_seq, t3, T_start=2.0, T_end=0.3, maxiter=2000)
        try_update(a_f3, "final_lbgfs")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
