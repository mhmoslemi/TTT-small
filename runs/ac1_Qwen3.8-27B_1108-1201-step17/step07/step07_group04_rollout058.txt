import cvxpy as cp
from scipy.fft import rfft, irfft
from time import monotonic

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        rho_hat = np.max(rho[cond])
        theta = (cssv[rho_hat - 1] - 1.0) / rho_hat
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

def get_lp_solver():
    global _lp_solver_cache
    if '_lp_solver_cache' in globals():
        return _lp_solver_cache
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(2, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cache = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cache = cp.SCS
    return cp.SCS

    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

    b_val = np.maximum(b_val, 0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated subgradient with adaptive softmax peak weighting
2. LP-based periodic direction correction
3. Iterated local search with simulated annealing perturbations
4. Final fine-tuning with ultra-small steps and coordinate checks
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        rho_hat = np.max(rho[cond])
        theta = (cssv[rho_hat - 1] - 1.0) / rho_hat
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver_result' in globals():
        return _lp_solver_result
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_result = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_result = cp.SCS
    return cp.SCS


def build_conv_matrix(a):
    """Build matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver):
    """Minimize M s.t. (a*b)[k] <= M, b >= 0, sum(b) = 1. Return optimal b."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def compute_subgradient(y, peaks, weights, n):
    """Compute weighted subgradient of max_k (y*y)[k] at given peaks."""
    grad = np.zeros(n)
    for w, k in zip(weights, peaks):
        if w < 1e-14:
            continue
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            other = k - j
            grad[j] += w * y[other]
            if other != j:
                grad[other] += w * y[j]
    grad *= 2.0
    return grad


def nesterov_subgradient(a0, time_limit, lr_scale=0.003, momentum=0.85,
                          peak_thresh=0.97, lp_interval=2000,
                          verbose=True, label=""):
    """
    Nesterov-accelerated subgradient with adaptive step size and
    periodic LP correction.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    y = a.copy()
    v = np.zeros(n)
    t0 = monotonic()
    step_count = 0
    no_improve = 0
    solver = get_lp_solver()
    last_lp_step = 0
    grad_norms = []

    while monotonic() - t0 < time_limit:
        step_count += 1

        # Compute convolution at lookahead point
        b_y = fast_convolve(y)
        max_val = b_y.max()
        if max_val < 1e-15:
            break

        # Select peaks above threshold
        thresh = peak_thresh * max_val
        peak_idx = np.where(b_y >= thresh)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b_y)])

        # Softmax weights on peaks
        peak_vals = b_y[peak_idx]
        temp = 0.02 * max_val + 1e-15
        w = np.exp((peak_vals - peak_vals.max()) / temp)
        sw = w.sum()
        if sw < 1e-15:
            w = np.zeros(len(peak_idx))
            w[0] = 1.0
        else:
            w = w / sw

        # Subgradient
        grad = compute_subgradient(y, peak_idx, w, n)
        gn = np.linalg.norm(grad)

        # Adaptive step size: c / sqrt(t+1), with c tuned to lr_scale
        lr = lr_scale / np.sqrt(step_count + 1)

        # Nesterov momentum update with Polyak restart
        if step_count > 10 and len(grad_norms) > 0:
            if gn > 1.5 * np.mean(grad_norms[-20:]):
                v = np.zeros(n)  # Restart momentum
        grad_norms.append(gn)
        if len(grad_norms) > 100:
            grad_norms.pop(0)

        v = momentum * v - lr * grad
        y_candidate = a + v
        a_new = project_simplex(y_candidate)

        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()
        a = a_new

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic LP correction
        if lp_interval > 0 and (step_count - last_lp_step) >= lp_interval and (monotonic() - t0) < time_limit - 5:
            b_lp = solve_lp_best_response(a, solver)
            if b_lp is not None:
                t_blend = 0.3
                a_trial = (1 - t_blend) * a + t_blend * b_lp
                a_trial = project_simplex(a_trial)
                score_trial = 2.0 * n * fast_convolve(a_trial).max()
                if score_trial < best_score - 1e-14:
                    best_score = score_trial
                    best_a = a_trial.copy()
                    a = a_trial
                    y = a_trial.copy()
                    v = np.zeros(n)
                    no_improve = 0
            last_lp_step = step_count

        # Restart if stagnating
        if no_improve > 800:
            a = best_a + np.random.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            v = np.zeros(n)
            no_improve = 0

        if verbose and step_count % 10000 == 0:
            elapsed = monotonic() - t0
            print(f"      {label} {elapsed:.0f}s step={step_count}: score={best_score:.8f} lr={lr:.8f}")

    return best_a, best_score


def iterated_local_search(a0, time_limit, n_restarts=50,
                          base_perturb=0.005, decay=0.95,
                          verbose=True):
    """
    Iterated local search: perturb, run short subgradient, keep if improved.
    Perturbation size anneals over time.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    perturb_size = base_perturb

    for restart in range(n_restarts):
        if monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (monotonic() - t0)
        if remaining < 10:
            break

        # Perturb
        noise = np.random.standard_normal(n) * perturb_size
        a_pert = a + noise
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        # Short subgradient from perturbed point
        alloc = min(remaining * 0.4, 30.0)
        a_trial, score_trial = nesterov_subgradient(
            a_pert, alloc, lr_scale=0.002, momentum=0.8,
            peak_thresh=0.96, lp_interval=0, verbose=False
        )

        if score_trial < best_score - 1e-14:
            best_score = score_trial
            best_a = a_trial.copy()
            a = a_trial.copy()
            if verbose:
                print(f"    ILS restart {restart}: improved to {best_score:.8f}")
        else:
            # Decay perturbation
            perturb_size *= decay

        if verbose and restart % 10 == 0:
            elapsed = monotonic() - t0
            print(f"    ILS {elapsed:.0f}s restart={restart}: best={best_score:.8f} perturb={perturb_size:.6f}")

    return best_a, best_score


def lp_polish(a0, time_limit, solver=None, verbose=True):
    """Alternating LP best response with damped updates."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    it = 0

    while monotonic() - t0 < time_limit:
        it += 1
        b = solve_lp_best_response(a, solver)
        if b is None:
            break

        # Damped update with decreasing step
        alpha = min(1.0, 3.0 / (it + 1))
        a_trial = (1 - alpha) * a + alpha * b
        a_trial = project_simplex(a_trial)
        score = 2.0 * n * fast_convolve(a_trial).max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
        elif it % 50 == 0:
            # Try smaller step
            alpha2 = alpha * 0.3
            a_trial2 = (1 - alpha2) * best_a + alpha2 * b
            a_trial2 = project_simplex(a_trial2)
            score2 = 2.0 * n * fast_convolve(a_trial2).max()
            if score2 < best_score - 1e-14:
                best_score = score2
                best_a = a_trial2.copy()
                a = a_trial2
            else:
                break  # No improvement, stop

        if verbose and it % 50 == 0:
            elapsed = monotonic() - t0
            print(f"    LP polish {elapsed:.0f}s iter={it}: score={best_score:.8f}")

    return best_a, best_score


def fine_tune(a0, time_limit, verbose=True):
    """Ultra-fine subgradient with very small steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    # Very small learning rate, no momentum, focus on exact max peak
    t0 = monotonic()
    step = 0
    lr = 0.0001
    no_improve = 0

    while monotonic() - t0 < time_limit:
        step += 1
        b = fast_convolve(a)
        max_k = np.argmax(b)
        max_val = b[max_k]

        # Use only the top few peaks with tighter threshold
        thresh = 0.995 * max_val
        peak_idx = np.where(b >= thresh)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([max_k])

        peak_vals = b[peak_idx]
        temp = 0.005 * max_val + 1e-15
        w = np.exp((peak_vals - peak_vals.max()) / temp)
        sw = w.sum()
        if sw < 1e-15:
            w = np.zeros(len(peak_idx))
            w[0] = 1.0
        else:
            w = w / sw

        grad = compute_subgradient(a, peak_idx, w, n)

        # Simple step (no momentum for precision)
        a_trial = a - lr * grad
        a_trial = project_simplex(a_trial)
        score = 2.0 * n * fast_convolve(a_trial).max()

        if score < best_score - 1e-15:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
            # Try half step
            a_trial2 = a - lr * 0.5 * grad
            a_trial2 = project_simplex(a_trial2)
            score2 = 2.0 * n * fast_convolve(a_trial2).max()
            if score2 < best_score - 1e-15:
                best_score = score2
                best_a = a_trial2.copy()
                a = a_trial2
                no_improve = 0

        if no_improve > 200:
            lr *= 0.7
            no_improve = 0
            if lr < 1e-8:
                break

        if verbose and step % 5000 == 0:
            elapsed = monotonic() - t0
            print(f"      FineTune {elapsed:.0f}s step={step}: score={best_score:.8f} lr={lr:.2e}")

    return best_a, best_score


def coordinate_refine(a0, time_limit, verbose=True):
    """Try adjusting individual coordinates to reduce the peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    improving = True

    while improving and (monotonic() - t0) < time_limit:
        improving = False
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # For each coordinate that contributes to the peak, try reducing it
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)

        for j in range(js, je + 1):
            if (monotonic() - t0) > time_limit:
                return best_a, best_score
            other = peak_k - j
            if a[j] < 1e-12:
                continue

            # Try reducing a[j] by various amounts, redistributing to others
            for frac in [0.9, 0.8, 0.95]:
                old_val = a[j]
                new_val = old_val * frac
                diff = old_val - new_val
                if diff < 1e-15:
                    continue

                a_trial = a.copy()
                a_trial[j] = new_val
                # Redistribute diff proportionally to other elements
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()

                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    best_score = score
                    best_a = a_trial.copy()
                    a = a_trial
                    improving = True
                    break

        if verbose:
            elapsed = monotonic() - t0
            print(f"    CoordRefine {elapsed:.0f}s: score={best_score:.8f} improving={improving}")

    return best_a, best_score
