Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using gradient-based optimization
in log-space with log-sum-exp smoothing and adaptive temperature annealing.
Uses L-BFGS-B as the primary optimizer, with FFT-based gradient computation.
Multi-start with diverse initializations and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_objective_and_grad(a, T):
    """Compute smoothed objective and its gradient with respect to a.
    a: length-n array, sum(a) = 1
    Returns: (objective_value, gradient)"""
    n = len(a)
    c = fast_convolve(a)
    
    # Log-sum-exp
    c_scaled = c / T
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    w = exp_c / exp_c.sum()
    obj = 2.0 * n * (c_max + T * np.log(exp_c.sum()))  # = 2n * logsumexp(c/T)
    
    # Gradient computation
    # term1[i] = sum_k w[k] * a[k-i] (cross-correlation)
    # term2[i] = a[i] * (sum of w[k] for valid k, k != 2i)
    
    # Cross-correlation via FFT
    N = 1 << (3 * n).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(np.conj(A) * W, N)
    term1 = R[:n]
    
    # Running sums for term2
    w_cs = np.concatenate([[0.0], np.cumsum(w)])
    k_starts = np.arange(n)
    k_ends = np.minimum(k_starts + n, 2*n-1)  # exclusive
    S = w_cs[k_ends] - w_cs[k_starts]
    S -= w[2 * np.arange(n)]
    
    term2 = a * S
    grad = (2.0 * n / T) * (term1 + term2)
    
    return obj, grad

def log_space_objective(z, T, n):
    """Objective in log-space."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    obj, _ = compute_objective_and_grad(a, T)
    return obj

def log_space_gradient(z, T, n):
    """Gradient in log-space."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    _, grad_a = compute_objective_and_grad(a, T)
    # Transform to z-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return grad_z

    n = len(a0)
    a = a0.copy()
    t0 = monotonic()
    
    for T in T_schedule:
        remaining = time_budget - (monotonic() - t0)
        if remaining < 5:
            break
        
        z0 = np.log(np.maximum(a, 1e-30))
        z0 -= z0.mean()
        
        # Time estimate per iteration: roughly 0.1ms for n=1000
        est_time_per_iter = n * 1e-4
        maxiter = min(maxiter_per_T, int(remaining * 0.8 / max(est_time_per_iter, 0.001)))
        maxiter = max(maxiter, 50)
        
        res = minimize(
            fun=lambda z: log_space_objective(z, T, n),
            x0=z0,
            jac=lambda z: log_space_gradient(z, T, n),
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 10, 'disp': False}
        )
        
        if res.x is not None:
            z_shifted = res.x - res.x.max()
            a = np.exp(z_shifted)
            a = a / a.sum()
        
        elapsed = monotonic() - t0
        score = 2.0 * n * fast_convolve(a).max()
        print(f"    T={T:.1f}: score={score:.6f} ({elapsed:.0f}s)")
    
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using gradient-based optimization
in log-space with log-sum-exp smoothing and adaptive temperature annealing.
Uses L-BFGS-B with FFT-based gradient computation for efficiency.
Multi-start with diverse initializations, peak-based local search,
and multi-length exploration to escape local optima.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def logsumexp_smooth(c, T):
    """Compute logsumexp(c/T) and the softmax weights w_k."""
    c_scaled = c / T
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    exp_sum = exp_c.sum()
    lse = c_max + T * np.log(exp_sum)
    w = exp_c / exp_sum
    return lse, w


def compute_obj_grad_logspace(z, T):
    """Compute smoothed objective and gradient in log-space.
    z: log-space variable, a = softmax(z)
    Returns (objective, gradient_wrt_z)"""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    c = fast_convolve(a)
    lse, w = logsumexp_smooth(c, T)
    obj = 2.0 * n * lse

    # Gradient wrt a: (4n/T) * cross_corr(w, a)
    # cross_corr[i] = sum_j w[i+j] * a[j]
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(np.conj(A) * W, N)
    grad_a = (4.0 * n / T) * R[:n]

    # Transform to z-space: grad_z = a * (grad_a - a * dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)

    return obj, grad_z


def lbgfs_run(a0, T_schedule, time_budget, maxiter_per_T=3000, verbose=True):
    """Run L-BFGS with temperature schedule in log-space."""
    n = len(a0)
    a = a0.copy()
    t0 = monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    for Ti in T_schedule:
        remaining = time_budget - (monotonic() - t0)
        if remaining < 5:
            break

        z0 = np.log(np.maximum(a, 1e-30))
        z0 -= z0.mean()

        est_time_per_iter = max(0.001, n * 5e-5)
        maxiter = min(maxiter_per_T, int(remaining * 0.85 / est_time_per_iter))
        maxiter = max(maxiter, 100)

        def func(z):
            return compute_obj_grad_logspace(z, Ti)[0]

        def jac(z):
            return compute_obj_grad_logspace(z, Ti)[1]

        try:
            res = minimize(
                fun=func, x0=z0, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': maxiter * 12, 'disp': False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                z_shifted = res.x - res.x.max()
                a = np.exp(z_shifted)
                a = a / a.sum()
        except Exception:
            pass

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose:
            elapsed = monotonic() - t0
            print(f"    T={Ti:.2f}: score={score:.6f}, best={best_score:.6f} ({elapsed:.0f}s)")

    return best_a, best_score


def peak_local_search(a0, time_budget, verbose=False):
    """Local search focusing on reducing the peak of the auto-convolution."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    no_improve = 0

    while monotonic() - t0 < time_budget:
        c = fast_convolve(a)
        k_peak = int(np.argmax(c))
        peak_val = c[k_peak]

        if peak_val <= best_score / (2.0 * n) + 1e-15:
            break

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_peak - idx]

        if contribs.sum() < 1e-30:
            break

        top_k = min(15, len(contribs))
        top_contribs = np.argsort(contribs)[::-1][:top_k]
        improved = False

        for ci in top_contribs:
            i = int(idx[ci])
            j = int(k_peak - i)
            if a[i] < 1e-15:
                continue

            for frac in [0.95, 0.9, 0.8, 0.7, 0.5]:
                a_trial = a.copy()
                delta = a_trial[i] * (1.0 - frac)
                a_trial[i] *= frac
                if delta < 1e-18:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                osum = others.sum()
                if osum < 1e-15:
                    continue
                a_trial[mask] = others + delta * others / osum
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial /= s

                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 300:
                # Random perturbation to escape
                for attempt in range(20):
                    a_trial = a + np.random.standard_normal(n) * 0.003
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial /= s
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score + 0.005:
                        a = a_trial
                        no_improve = 0
                        break
                else:
                    break
        else:
            no_improve = 0

        if verbose and no_improve % 500 == 0:
            elapsed = monotonic() - t0
            print(f"    PeakSearch ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def gradient_descent_smooth(a0, T, time_budget, lr=0.01, momentum=0.9):
    """Momentum gradient descent with adaptive learning rate in log-space."""
    n = len(a0)
    a = a0.copy()
    z = np.log(np.maximum(a, 1e-30))
    z -= z.mean()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    no_improve = 0
    lr_current = lr

    while monotonic() - t0 < time_budget:
        obj, grad = compute_obj_grad_logspace(z, T)
        v = momentum * v - lr_current * grad
        z_new = z + v
        z_shifted = z_new - z_new.max()
        a_new = np.exp(z_shifted)
        a_new = a_new / a_new.sum()

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            z = z_new
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            lr_current *= 0.95
            v = v * 0.5  # damp momentum

        if no_improve > 200:
            lr_current = lr * 2
            v = np.zeros(n)
            a = best_a + np.random.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            z = np.log(np.maximum(a, 1e-30))
            z -= z.mean()
            no_improve = 0

    return best_a, best_score


def make_initialization(n, method='uniform'):
    """Create various initializations for the optimization."""
    if method == 'uniform':
        a = np.ones(n) / n
    elif method == 'hann':
        t = np.arange(n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / (n - 1)))
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif method == 'raised_cosine':
        t = np.linspace(-np.pi, np.pi, n)
        a = np.cos(t / 2) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif method == 'random':
        a = np.random.dirichlet(np.ones(n))
    elif method == 'bump':
        # Central bump with decay
        center = n / 2.0
        sigma = n / 6.0
        t = np.arange(n)
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
        a = a / a.sum()
    elif method == 'two_bumps':
        # Two symmetric bumps
        center1 = n * 0.25
        center2 = n * 0.75
        sigma = n / 8.0
        t = np.arange(n)
        a = np.exp(-0.5 * ((t - center1) / sigma) ** 2) + np.exp(-0.5 * ((t - center2) / sigma) ** 2)
        a = a / a.sum()
    elif method == 'comb':
        # Alternating pattern with smooth interpolation
        a = np.ones(n)
        a[1::2] = 0.3
        # Smooth it
        kernel = np.ones(5) / 5
        a_smooth = np.convolve(a, kernel, mode='same')
        a = np.maximum(a_smooth, 0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    f_interp = np.interp(x_new, x_old, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_update(a, label, best_seq, best_score):
    """Evaluate candidate via evaluate_sequence and update best if improved."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. L-BFGS with temperature annealing from diverse initializations
    2. Momentum gradient descent for fine-tuning
    3. Peak-based local search
    4. Multi-length exploration
    5. Final polish
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={best_score:.6f}")
    else:
        best_seq = np.ones(1024) / 1024.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial uniform: n=1024, score={best_score:.6f}")

    n_parent = len(best_seq)

    # === Phase 1: L-BFGS on parent with full temperature schedule ===
    print(f"\n=== Phase 1: L-BFGS on parent (n={n_parent}) ===")
    rem = deadline - monotonic() - 400
    if rem > 30:
        T_schedule = [200, 100, 50, 25, 12, 8, 5, 3, 2, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1]
        t_alloc = min(rem, 500)
        a_opt, sc = lbgfs_run(best_seq, T_schedule, t_alloc, maxiter_per_T=2000, verbose=True)
        best_seq, best_score = check_update(a_opt, "L-BFGS_parent", best_seq, best_score)

    # === Phase 2: Momentum GD for further refinement ===
    print(f"\n=== Phase 2: Momentum GD (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 300
    if rem > 20:
        t_alloc = min(rem, 150)
        a_gd, _ = gradient_descent_smooth(best_seq, T=0.5, time_budget=t_alloc, lr=0.005, momentum=0.92)
        best_seq, best_score = check_update(a_gd, "momentum_gd", best_seq, best_score)

    # === Phase 3: Peak local search ===
    print(f"\n=== Phase 3: Peak Local Search (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 200
    if rem > 10:
        t_alloc = min(rem, 200)
        a_ps, _ = peak_local_search(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_update(a_ps, "peak_search", best_seq, best_score)

    # === Phase 4: L-BFGS second pass on refined sequence ===
    print(f"\n=== Phase 4: L-BFGS Polish (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 150
    if rem > 15:
        T_schedule2 = [10, 5, 3, 2, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.07]
        t_alloc = min(rem, 250)
        a_opt2, _ = lbgfs_run(best_seq, T_schedule2, t_alloc, maxiter_per_T=3000, verbose=True)
        best_seq, best_score = check_update(a_opt2, "L-BFGS_polish", best_seq, best_score)

    # === Phase 5: Try different initializations at same length ===
    print(f"\n=== Phase 5: Alternative Inits (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    inits_to_try = ['hann', 'raised_cosine', 'two_bumps', 'random', 'bump']
    for init_method in inits_to_try:
        rem = deadline - monotonic() - 80
        if rem < 30:
            break
        a_init = make_initialization(n_best, init_method)
        # Blend with current best to give it a head start
        a_init = 0.5 * a_init + 0.5 * best_seq
        a_init = np.maximum(a_init, 0)
        a_init = a_init / a_init.sum()

        t_alloc = min(rem, 120)
        T_schedule3 = [50, 20, 10, 5, 3, 2, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2]
        a_alt, _ = lbgfs_run(a_init, T_schedule3, t_alloc, maxiter_per_T=1500, verbose=False)
        best_seq, best_score = check_update(a_alt, f"init_{init_method}", best_seq, best_score)

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Length Exploration (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    alt_lengths = []
    for fac in [0.75, 1.0, 1.25, 1.5, 2.0]:
        alt_n = int(n_best * fac)
        if 256 <= alt_n <= 4096 and alt_n not in alt_lengths:
            alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:3]:
        rem = deadline - monotonic() - 60
        if rem < 40:
            break
        if alt_n == n_best:
            init_alt = best_seq.copy()
        else:
            init_alt = interpolate_sequence(best_seq, alt_n)

        t_alloc = min(rem, 200)
        T_schedule4 = [50, 20, 10, 5, 3, 2, 1.5, 1.0, 0.5, 0.3, 0.2]
        a_alt, _ = lbgfs_run(init_alt, T_schedule4, t_alloc, maxiter_per_T=2000, verbose=False)
        best_seq, best_score = check_update(a_alt, f"len_{alt_n}", best_seq, best_score)

        # Quick peak search on new length result
        rem = deadline - monotonic() - 40
        if rem > 15 and best_seq is not None and len(best_seq) == alt_n:
            a_ps2, _ = peak_local_search(best_seq, min(rem, 50), verbose=False)
            best_seq, best_score = check_update(a_ps2, f"peak_{alt_n}", best_seq, best_score)

    # === Phase 7: Final L-BFGS + peak search ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 10
    if rem > 10:
        T_final = [3, 2, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1]
        t_alloc = min(rem * 0.6, 60)
        a_final, _ = lbgfs_run(best_seq, T_final, t_alloc, maxiter_per_T=2000, verbose=False)
        best_seq, best_score = check_update(a_final, "final_lbgfs", best_seq, best_score)

    rem = deadline - monotonic() - 5
    if rem > 8:
        a_ps_final, _ = peak_local_search(best_seq, min(rem, 40), verbose=False)
        best_seq, best_score = check_update(a_ps_final, "final_peak", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
