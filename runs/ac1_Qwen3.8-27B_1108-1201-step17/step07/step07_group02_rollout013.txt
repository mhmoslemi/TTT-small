"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an LP best-response iteration
with Nesterov acceleration as the main workhorse, combined with L-BFGS soft-max
refinement, local peak shaving, and multi-length exploration.
Key innovations: aggressive LP iteration with convergence gap monitoring,
periodic L-BFGS resets during LP to avoid slow convergence, and adaptive
temperature scheduling for the soft-max approximation.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    global _lp_solver_cached
    if '_lp_solver_cached' in globals():
        return _lp_solver_cached
    preferred = [cp.PDLP, cp.GLOP, cp.CBC, cp.ECOS, cp.SCS, cp.MOSEK]
    for solver_name in preferred:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_min_max(a_fixed, solver=None, reltol=1e-8):
    """
    Given fixed a (non-negative, sum=1), solve:
    min M s.t. conv(a,b)[k] <= M for all k, sum(b)=1, b>=0
    Returns (b_opt, M_opt) or (None, None) on failure.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    T = build_conv_matrix(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = cp.Constant(np.ones(2 * n - 1))
    constraints = [T @ b_var <= M_var * ones, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver == cp.PDLP:
            prob.solve(solver=solver, reltol=reltol, abstol=1e-10)
        elif solver == cp.GLOP:
            prob.solve(solver=solver)
        else:
            prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.GLOP, cp.CBC, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, reltol=1e-8):
    """
    Nesterov-accelerated LP best-response iteration with convergence monitoring.
    Tracks the gap between LP value and actual max convolution to detect convergence.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    a_avg = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stall_count = 0
    last_improve = 0.0

    while time.monotonic() - t0 < time_limit:
        b_new, M_val = solve_lp_min_max(y, solver, reltol)
        if b_new is None:
            break

        # Nesterov step
        alpha = 2.0 / (it + 2)
        y_new = (1.0 - alpha) * y + alpha * b_new
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
            sy = 1.0
        y_new = y_new / sy

        a_avg = ((it + 1) * a_avg + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a_avg = np.maximum(a_avg, 0)
        sa = a_avg.sum()
        if sa < 1e-15:
            a_avg = y_new.copy()
            sa = 1.0
        a_avg = a_avg / sa

        y = y_new
        it += 1

        # Evaluate and track
        score_y = 2.0 * n * fast_convolve(y).max()
        score_avg = 2.0 * n * fast_convolve(a_avg).max()

        # Convergence gap: M_val should approach max_conv(sum-1 normalized)
        # M_val is max_k (y*b_opt)[k] with sum(b_opt)=1
        # At equilibrium, M_val = max_k (y*y)[k] when sum(y)=1
        gap = fast_convolve(y).max() - M_val
        gap = max(0, gap)

        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()
        if score_avg < best_score - 1e-12:
            best_score = score_avg
            best_a = a_avg.copy()
            last_improve = time.monotonic()

        # Stall detection: if no improvement for 500s and gap is tiny, stop
        if gap < 1e-10 and (time.monotonic() - last_improve) > 30:
            break

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it} ({elapsed:.0f}s): score={best_score:.6f}, gap={gap:.2e}, M_lp={M_val:.6f}")

    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def compute_soft_max_obj_grad_z(a, n, T_val):
    """Compute soft-max smoothed objective and gradient in log-space."""
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=1.0, maxiter=5000):
    """L-BFGS refinement with soft-max smoothing and geometric temperature schedule."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    stopped = [False]

    def make_func(T_val):

                stopped[0] = True
                return 1e15
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return np.zeros(n)
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 4
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()

            z = res.x
        except Exception:
            pass

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        a_ps, _ = peak_shave_local(best_seq, t_ps)
        try_update(a_ps, "peak_shave")

    rem = deadline - time.monotonic()
    t_cd = min(rem * 0.08, 80.0)
    if t_cd > 15:
        a_cd, _ = coordinate_descent_refine(best_seq, t_cd)
        try_update(a_cd, "coord_desc")

    # Another peak shaving round after coordinate descent
    rem = deadline - time.monotonic()
    t_ps2 = min(rem * 0.06, 60.0)
    if t_ps2 > 10:
        a_ps2, _ = peak_shave_local(best_seq, t_ps2)
        try_update(a_ps2, "peak_shave2")

    # === Phase 4: Alternative lengths (~200s) ===
    print(f"\n=== Phase 4: Alternative Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = [int(n_best * 1.25), int(n_best * 1.5), int(n_best * 0.8)]
    alt_lengths = sorted(set(x for x in alt_lengths if 256 <= x <= 3072 and x != n_best))
    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(rem * 0.3, 100.0)
        if t_alloc < 20:
            break
        print(f"  Alt n={alt_n}, budget={t_alloc:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        # Quick LP
        t_lp = t_alloc * 0.5
        a_alt, _ = nesterov_lp_refine(init_alt, t_lp, solver=solver)
        try_update(a_alt, f"lp_n{alt_n}")
        # Quick L-BFGS
        rem = deadline - time.monotonic()
        t_lb = min(rem * 0.2, t_alloc * 0.5)
        if t_lb > 15:
            a_alt2 = lbgfs_refine(best_seq if len(best_seq) == alt_n else init_alt, t_lb,
                                   T_start=10.0, T_end=2.0, maxiter=2000)
            try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 5: Final polish (~100s) ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_final = min(rem * 0.5, 60.0)
        a_final = lbgfs_refine(best_seq, t_final, T_start=2.0, T_end=0.2, maxiter=3000)
        try_update(a_final, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps_final = min(rem * 0.5, 40.0)
        a_ps_final, _ = peak_shave_local(best_seq, t_ps_final, n_steps=5000)
        try_update(a_ps_final, "final_ps")

    # Final LP verification
    rem = deadline - time.monotonic()
    if rem > 20:
        t_verify = min(rem * 0.7, 60.0)
        a_ver, _ = nesterov_lp_refine(best_seq, t_verify, solver=solver)
        try_update(a_ver, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
