"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam-based subgradient descent
with adaptive peak tracking, LP-based refinement (sparse HiGHS), mirror descent,
and multi-length exploration. Key innovations: Adam optimizer with per-coordinate
adaptive step sizes, sparse LP formulation, multiplicative-weights phase, and
aggressive multi-n search with interpolation.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgradient(a, w):
    """Compute subgradient g[i] = sum_j a[j]*w[i+j] (reflected convolution)."""
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    g = irfft(np.conj(A) * W, fft_size)[:n]
    return g


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_max_b(a):
    """Return max of auto-convolution for normalized sequence (sum=1)."""
    b = fast_convolve(a)
    return float(b.max())


def adam_subgradient(a0, time_limit, lr=0.003, beta1=0.9, beta2=0.999,
                     eps=1e-8, n_top=None, seed=42, restart_interval=2000):
    """
    Adam-based projected subgradient optimization.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_obj = eval_max_b(a)
    no_improve = 0
    kick_size = 0.005

    if n_top is None:
        n_top = max(15, n // 10)

    while time.monotonic() - t0 < time_limit:
        # Compute convolution and identify peaks
        b = fast_convolve(a)
        b_max = b.max()

        # Select active peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))

        top_vals = b[top_idx]
        threshold = b_max * 0.998
        active_mask = top_vals >= threshold
        if not active_mask.any():
            active_mask[0] = True

        w = np.zeros(2 * n - 1)
        active_indices = top_idx[active_mask]
        w[active_indices] = 1.0 / len(active_indices)

        # Compute subgradient
        g = compute_subgradient(a, w)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * g * g
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = eval_max_b(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive restart
        if no_improve >= restart_interval:
            perturbation = rng.standard_normal(n) * kick_size
            a = project_simplex(best_a + perturbation)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            kick_size *= 0.6
            if kick_size < 0.0005:
                kick_size = 0.0005
                lr = max(lr * 0.7, 0.0001)
        elif no_improve >= restart_interval // 2:
            lr *= 0.93

    return best_a, best_obj


def lp_refine_sparse(f, max_iters=40, time_limit=60.0, verbose=False):
    """
    LP-based refinement using sparse constraint matrix.
    Solves: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = eval_max_b(a)

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        # Build sparse constraint matrix
        num_constraints = 2 * n - 1
        rows = []
        cols = []
        data = []

        for k in range(num_constraints):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                rows.append(k)
                cols.append(k - j)
                data.append(a[j])

        A = csr_matrix((data, (rows, cols)), shape=(num_constraints, n))
        b_lp = np.full(num_constraints, M)
        c = -np.ones(n)

        try:
            res = linprog(c, A_ub=A, b_ub=b_lp,
                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                break

            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break

            g_norm = g / g_sum

            # Line search for blending
            best_t = 0.0
            best_blend_obj = best_obj
            t_candidates = [0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]
            for t in t_candidates:
                blend = (1.0 - t) * best_a + t * g_norm
                ss = blend.sum()
                if ss < 1e-10:
                    continue
                blend = blend / ss
                obj_blend = eval_max_b(blend)
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t

            if best_t <= 0.0:
                for t in [0.01, 0.02, 0.03, 0.05]:
                    blend = (1.0 - t) * best_a + t * g_norm
                    blend = blend / blend.sum()
                    obj_blend = eval_max_b(blend)
                    if obj_blend < best_obj - 1e-14:
                        best_t = t
                        best_blend_obj = obj_blend
                        break

            if best_t > 0.0:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                obj = eval_max_b(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    if verbose:
                        print(f"    LP iter {iteration}: t={best_t:.3f}, max_b={best_obj:.10f}, obj={2*n*best_obj:.8f}")
                else:
                    a = best_a.copy()
                    best_obj = eval_max_b(a)
            else:
                break

        except Exception:
            break

    return best_a, best_obj


def mirror_descent(a0, time_limit, lr=0.001, n_top=None, seed=42):
    """
    Mirror descent (multiplicative weights) on the simplex.
    Update: a_new[i] = a[i] * exp(-lr * g[i]) / Z
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = eval_max_b(a)
    no_improve = 0
    cur_lr = lr

    if n_top is None:
        n_top = max(15, n // 10)

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))

        top_vals = b[top_idx]
        threshold = b_max * 0.999
        active_mask = top_vals >= threshold
        if not active_mask.any():
            active_mask[0] = True

        w = np.zeros(2 * n - 1)
        active_indices = top_idx[active_mask]
        w[active_indices] = 1.0 / len(active_indices)

        g = compute_subgradient(a, w)

        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-300))
        new_log_a = log_a - cur_lr * g
        # Shift for numerical stability
        new_log_a -= new_log_a.max()
        a_new = np.exp(new_log_a)
        ss = a_new.sum()
        if ss < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / ss
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()

        obj_new = eval_max_b(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve >= 1000:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            no_improve = 0
            cur_lr *= 0.7
            if cur_lr < 1e-5:
                break

    return best_a, best_obj


def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def combined_optimize(a_init, time_limit, seed=42, verbose=False):
    """
    Combined optimization: Adam SG -> LP -> mirror descent -> LP.
    Returns (best_a, best_max_b).
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    # Phase A: Adam subgradient
    t_alloc = time_limit * 0.45
    if t_alloc > 5:
        a_adam, obj_adam = adam_subgradient(
            a, time_limit=t_alloc, lr=0.004, beta1=0.9, beta2=0.999,
            n_top=max(20, n // 8), seed=seed
        )
        if obj_adam < best_obj:
            best_obj = obj_adam
            best_a = a_adam.copy()
            if verbose:
                print(f"    Adam: max_b={best_obj:.10f}, obj={2*n*best_obj:.8f}")

    # Phase B: LP refinement
    rem = time_limit - (time.monotonic() - t0)
    t_lp = rem * 0.3
    if t_lp > 5:
        a_lp, obj_lp = lp_refine_sparse(best_a, max_iters=40, time_limit=t_lp, verbose=verbose)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            if verbose:
                print(f"    LP: max_b={best_obj:.10f}, obj={2*n*best_obj:.8f}")

    # Phase C: Mirror descent
    rem = time_limit - (time.monotonic() - t0)
    t_mirror = rem * 0.3
    if t_mirror > 5:
        a_md, obj_md = mirror_descent(best_a, time_limit=t_mirror, lr=0.002,
                                      n_top=max(20, n // 8), seed=seed + 777)
        if obj_md < best_obj:
            best_obj = obj_md
            best_a = a_md.copy()
            if verbose:
                print(f"    Mirror: max_b={best_obj:.10f}, obj={2*n*best_obj:.8f}")

    # Phase D: Final LP
    rem = time_limit - (time.monotonic() - t0)
    if rem > 5:
        a_lp2, obj_lp2 = lp_refine_sparse(best_a, max_iters=25, time_limit=rem, verbose=verbose)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()
            if verbose:
                print(f"    LP2: max_b={best_obj:.10f}, obj={2*n*best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses Adam subgradient, sparse LP refinement, mirror descent,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
            return True
        return False

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = make_uniform_init(1152)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1152, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Deep optimization on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Adam+LP+Mirror on parent (n={n_parent})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_alloc = min(rem, 500.0)
        print(f"  Allocated: {t_alloc:.0f}s")
        a_opt, obj_opt = combined_optimize(parent, t_alloc, seed=seed, verbose=True)
        update_best(a_opt, "phase1_parent")

    # === Phase 2: Second optimization with different hyperparams ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Alternative Adam (different lr/momentum)")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 400
    if rem > 20 and best_seq is not None:
        t_alloc = min(rem, 150.0)
        print(f"  Allocated: {t_alloc:.0f}s")
        # Run Adam with different parameters
        a2, obj2 = adam_subgradient(
            best_seq, time_limit=t_alloc * 0.6,
            lr=0.002, beta1=0.85, beta2=0.9995,
            n_top=max(25, len(best_seq) // 6), seed=seed + 111
        )
        if obj2 < eval_max_b(best_seq):
            update_best(a2, "phase2_adam_alt")
            a_lp2, obj_lp2 = lp_refine_sparse(a2, max_iters=30, time_limit=t_alloc * 0.4)
            update_best(a_lp2, "phase2_lp")

    # === Phase 3: Diverse initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Diverse Initializations (best={best_score:.6f})")
    print(f"{'='*60}")
    n_base = len(best_seq) if best_seq is not None else n_parent

    inits = {
        "gauss_0.2": make_gaussian_init(n_base, 0.2),
        "gauss_0.3": make_gaussian_init(n_base, 0.3),
        "cosine": make_cosine_init(n_base),
        "tukey_0.4": make_tukey_init(n_base, 0.4),
    }

    total_diverse = min(200.0, (deadline - time.monotonic() - 300.0) * 0.35)
    time_per_init = total_diverse / len(inits)

    for name, init_a in inits.items():
        rem = deadline - time.monotonic() - 250
        if rem < 15:
            break
        t_alloc = min(time_per_init, rem * 0.35)
        if t_alloc < 10:
            continue
        print(f"  {name} (t={t_alloc:.0f}s)")
        a_div, obj_div = combined_optimize(
            init_a, t_alloc, seed=seed + hash(name) % 100000, verbose=False
        )
        update_best(a_div, f"div_{name}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-length Exploration (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        target_lengths = []
        for factor in [0.7, 0.8, 1.0, 1.2, 1.5, 2.0, 0.5, 1.3]:
            tl = int(n_parent * factor)
            tl = max(256, min(3000, tl))
            if tl not in target_lengths:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_parent))

        for target_n in target_lengths[:6]:
            rem = deadline - time.monotonic() - 200
            if rem < 20:
                break

            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.2, 50.0)
            if t_alloc < 10:
                continue
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_len, obj_len = combined_optimize(
                f_interp, t_alloc, seed=seed + target_n, verbose=False
            )
            update_best(a_len, f"len_n{target_n}")

    # === Phase 5: Deep final optimization ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Deep Final Optimization (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_alloc = min(rem, 400.0)
            print(f"  Deep: n={n_best}, t={t_alloc:.0f}s")
            a_deep, obj_deep = combined_optimize(
                best_seq, t_alloc, seed=seed + 99999, verbose=True
            )
            update_best(a_deep, "deep")

    # === Phase 6: Micro-polish ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Micro-polish (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_micro = min(rem * 0.5, 30.0)
            a_micro, obj_micro = adam_subgradient(
                best_seq, time_limit=t_micro,
                lr=0.0003, beta1=0.95, beta2=0.9999,
                n_top=max(30, len(best_seq) // 5), seed=seed + 777777
            )
            update_best(a_micro, "micro_adam")

            rem = deadline - time.monotonic() - 15
            if rem > 5:
                a_micro_lp, obj_micro_lp = lp_refine_sparse(
                    best_seq, max_iters=20, time_limit=min(rem, 20.0)
                )
                update_best(a_micro_lp, "micro_lp")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
