def lbgfs_optimize(a0, time_limit, T_start=100.0, T_end=1.0, maxiter=5000):
    """L-BFGS with log-sum-exp smoothing and geometric temperature schedule."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    
    def make_func(T_val):
        def func(z):
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func
    
    def make_jac(T_val):
        def jac(z):
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            # Gradient in log-space
            # d/dz_i = a_i * (d/da_i - sum_j a_j * d/da_j)
            # d/da_i of (2n/T) * LSE(T*b) = 2n * sum_k w_k * (db_k/da_i)
            # db_k/da_i = a_{k-i} (for valid indices)
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac
    
    n_stages = 6
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = max(100, maxiter // n_stages)
    
    for stage_idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 100}
            )
            z = res.x
        except Exception as e:
            print(f"  L-BFGS stage {stage_idx} error: {e}")
            break
    
    return softmax_from_z(z)

def frank_wolfe(a0, time_limit, n_steps=10000):
    """Frank-Wolfe (conditional gradient) method on the simplex."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Compute subgradient: average of gradients at active (near-max) positions
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Frank-Wolfe: find s = e_i (standard basis) minimizing <g, s>
        i_min = np.argmin(g)
        s_vertex = np.zeros(n)
        s_vertex[i_min] = 1.0
        
        # Step size
        gamma = 2.0 / (step + 2.0)
        
        # Take step
        a_new = (1 - gamma) * a + gamma * s_vertex
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn > 1e-15:
            a_new = a_new / sn
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        a = a_new
        
        if step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW {elapsed:.0f}s: score={best_score:.6f}")
    
    return best_a, best_score

def peak_shave(a0, time_limit, n_steps=5000):
    """Aggressive peak shaving local search."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Find the peak (exclude endpoints which are just a_0^2 and a_{n-1}^2)
        b_inner = b.copy()
        b_inner[0] = -np.inf
        b_inner[-1] = -np.inf
        peak_k = int(np.argmax(b_inner))
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top = np.argsort(contribs)[::-1][:5]
        
        improved = False
        for ci in top:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85]:
                a_trial = a.copy()
                diff = a[i] * (1 - factor)
                a_trial[i] *= factor
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_trial[mask] += diff * a_trial[mask] / a_trial[mask].sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try reducing the peak contributors
            for ci in top[:3]:
                i = idx[ci]
                j = peak_k - i
                if i == j:
                    continue
                if a[j] < 1e-14:
                    continue
                for f in [0.95, 0.90]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial[j] *= f
                    a_trial /= a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            # Try random perturbation
            mask = np.random.random(n) > 0.5
            a_trial = a * (1 + 0.01 * (2 * mask - 1))
            a_trial = np.maximum(a_trial, 0)
            a_trial /= a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
        
        if not improved and step > 100:
            # Stuck, try a bigger jump
            i = np.random.randint(n)
            a_trial = a.copy()
            a_trial[i] *= 0.7
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:
                break  # Give up on peak shaving
    
    return best_a, best_score

def lp_refine(a0, time_limit, n_iters=2000):
    """LP-based alternating optimization."""
    import cvxpy as cp
    
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Find solver
    solver = None
    for s_name in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s_name
                break
        except:
            continue
    
    if solver is None:
        return best_a, best_score
    
    y = a.copy()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP: min M s.t. T @ b <= M * 1, sum(b) = 1, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        T = build_conv_matrix(y)
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        bs = b_val.sum()
        if bs < 1e-15:
            continue
        b_val = b_val / bs
        
        # Nesterov-style update
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_val
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            break
        y = y / ys
        
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. L-BFGS with log-sum-exp smoothing and geometric temperature annealing
2. Iterative perturbation + re-optimization to escape local minima
3. LP-based best-response alternating method with Nesterov averaging
4. Peak-shaving local search with dual-coordinate adjustments
5. Multi-length exploration with interpolation
Key innovations: aggressive low-temperature L-BFGS, structured perturbations,
and dual-aware gradient computation for faster convergence.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Evaluate sequence safely, returning inf on failure."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def simplex_project(x):
    """Project x onto the probability simplex."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def lbgfs_obj(z, T_val, n):
    """Compute soft-max smoothed objective value."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T_val) * log_sum_exp


def lbgfs_grad(z, T_val, n):
    """Compute gradient of soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_optimize(a0, time_limit, T_start=100.0, T_end=0.5, maxiter=8000):
    """L-BFGS with log-sum-exp smoothing and geometric temperature schedule."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    n_stages = max(3, int(np.log(T_start / max(T_end, 0.1)) / np.log(3.0) + 0.5))
    n_stages = min(n_stages, 10)
    temps = np.geomspace(T_start, max(T_end, 0.1), n_stages)
    iters_per_stage = max(50, maxiter // n_stages)

    for stage_idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_wrap(z_arg):
            return lbgfs_obj(z_arg, stage_T, n)

        def jac_wrap(z_arg):
            return lbgfs_grad(z_arg, stage_T, n)

        try:
            res = minimize(
                fun=func_wrap,
                x0=z,
                jac=jac_wrap,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 100}
            )
            z = res.x
        except Exception:
            break

    return softmax_from_z(z)


def _get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def lp_refine(a0, time_limit, solver=None, n_max_iters=3000):
    """LP-based alternating optimization with Nesterov dual averaging."""
    import cvxpy as cp

    if solver is None:
        solver = _get_lp_solver()

    n = len(a0)
    y = np.maximum(a0, 0).astype(np.float64)
    s = y.sum()
    y = y / s
    best_a = y.copy()
    best_score = 2.0 * n * fast_convolve(y).max()
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        T = build_conv_matrix(y)
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        bs = b_val.sum()
        if bs < 1e-15:
            continue
        b_val = b_val / bs

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_val
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            break
        y = y / ys

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=8000):
    """Aggressive peak shaving with multi-coordinate adjustments."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_inner = b.copy()
        b_inner[0] = -np.inf
        b_inner[-1] = -np.inf
        peak_k = int(np.argmax(b_inner))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top = np.argsort(contribs)[::-1][:6]

        improved = False
        for ci in top:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.96, 0.93, 0.90, 0.85]:
                a_trial = a.copy()
                diff = a[i] * (1 - factor)
                a_trial[i] *= factor
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            for ci in top[:4]:
                i = idx[ci]
                j = peak_k - i
                if i == j or a[j] < 1e-14 or a[i] < 1e-14:
                    continue
                for f in [0.95, 0.90, 0.85]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial[j] *= f
                    a_trial /= a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Random perturbation to escape
            n_perturb = max(3, n // 32)
            perturb_idx = rng.choice(n, size=n_perturb, replace=False)
            a_trial = a.copy()
            for pi in perturb_idx:

            a_trial = np.maximum(a_trial, 0)
            ts = a_trial.sum()

                a_trial = a_trial / ts
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if not improved and step > 200:
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def subgradient_step(a0, lr, n_active_frac=0.1):
    """One subgradient step with projection onto simplex."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - n_active_frac)
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(b)])

    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)

    a_new = a - lr * g
    a_new = simplex_project(a_new)
    return a_new


def subgradient_optimize(a0, time_limit, lr_init=0.05, n_steps=50000):

            break

        a_new = subgradient_step(a, lr)

        # Polyak averaging
        t = step + 1
        x_bar = ((t - 1) * x_bar + a_new) / t
        x_bar_score = 2.0 * n * fast_convolve(x_bar).max()

        if score_new < best_score - 1e-14:

        else:
            no_improve += 1
