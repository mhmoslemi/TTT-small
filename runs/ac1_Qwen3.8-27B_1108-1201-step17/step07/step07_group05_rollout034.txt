def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

                    best_local_score = f_opt
                    best_local_t = t_opt

            a[j] += t

            a = a / a.sum()
            conv = fast_convolve(a)

            best_a = a.copy()
            stagnant = 0

        if verbose and iters % 50 == 0:
            print(f"    CD iter={iters}, score={best_score:.8f}")

    return best_a, best_score

def fast_convolve(a):
    n = len(a)

    A = rfft(a, fft_size)

def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)

    if s < 1e-300:

            t_width = 0.02 * base
            t_search_lo = max(t_lo, t_center - t_width)
            t_search_hi = min(t_hi, t_center + t_width)

            for _ in range(18):
                m1 = t_search_lo + 0.382 * (t_search_hi - t_search_lo)
                m2 = t_search_hi - 0.382 * (t_search_hi - t_search_lo)
                f1 = eval_pairwise_move_max(conv, a, i, j, m1, n)
                f2 = eval_pairwise_move_max(conv, a, i, j, m2, n)
                if f1 < f2:
                    t_search_hi = m2
                else:
                    t_search_lo = m1

            t_opt = (t_search_lo + t_search_hi) / 2.0
            f_opt = eval_pairwise_move_max(conv, a, i, j, t_opt, n)
            if f_opt < local_best_score:
                local_best_score = f_opt
                local_best_t = t_opt

            if local_best_score < best_move_score:
                best_move_score = local_best_score
                best_move = (i, j, local_best_t)

        # Apply best move
        if best_move is not None:
            i, j, t = best_move
            a[i] -= t
            a[j] += t
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss > 1e-15:
                a = a / ss
            conv = fast_convolve(a)
            actual_score = 2.0 * n * conv.max()
            if actual_score < best_score - 1e-14:
                best_score = actual_score
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        # Perturb if stuck
        if stagnant > 30:
            scale = 0.003 * stagnant / 30.0
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            conv = fast_convolve(a)
            best_score = 2.0 * n * conv.max()
            best_a = a.copy()
            stagnant = 0

        if verbose and iters % 100 == 0:
            print(f"      CD iter={iters}, score={best_score:.8f}, stagnant={stagnant}")

    return best_a, best_score


def gradient_descent_simplex(a0, time_limit, lr_init=0.01, decay=0.9999, T=500, verbose=False):
    """Direct gradient descent with simplex projection."""
    n = len(a0)
    a = np.maximum(a0, 0).copy().astype(np.float64)
    ss = a.sum()
    a = a / ss
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = lr_init
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        shifted = T * conv
        shifted_max = shifted.max()
        exp_shifted = np.exp(shifted - shifted_max)
        w = exp_shifted / exp_shifted.sum()

        # Gradient: 4n * cross_correlation(w, a)
        fft_size = 1 << (2 * n - 2).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 4.0 * n * irfft(np.conj(W) * A, fft_size)[:n]

        # Step
        a_new = a - lr * grad_a
        # Project onto simplex {a >= 0, sum(a) = 1}
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        else:
            a_new = a

        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        elif score < best_score + lr * 0.1:
            a = a_new
        else:
            lr *= 0.95

        lr *= decay

        if verbose and iters % 200 == 0:
            print(f"    GD iter={iters}, lr={lr:.6f}, score={best_score:.8f}")

    return best_a, best_score


def peak_equalization(a0, time_limit, n_peaks=10, verbose=False):
    """Iteratively equalize top peaks by reducing contributors."""
    n = len(a0)
    a = np.maximum(a0, 0).copy().astype(np.float64)
    ss = a.sum()
    a = a / ss
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        top_k = min(n_peaks, 2 * n - 1)
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peak_indices]
        max_peak = peak_vals.max()
        min_peak = peak_vals.min()

        if max_peak - min_peak < 1e-10:
            break

        improved = False
        # For the max peak, find top contributors and try reducing them
        max_idx = int(np.argmax(conv))
        lo = max(0, max_idx - n + 1)
        hi = min(n - 1, max_idx)

        contribs = []
        for i in range(lo, hi + 1):
            j = max_idx - i
            if i != j:
                contribs.append((a[i] * a[j], i, j))
        contribs.sort(reverse=True)

        for (cv, i, j) in contribs[:8]:
            if a[i] < 1e-12:
                continue
            for frac in [0.02, 0.05, 0.1, 0.15]:
                delta = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= delta
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                os_sum = others.sum()
                if os_sum < 1e-15:
                    continue
                a_trial[mask] = others + delta * others / os_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try random perturbation on peak-contributing coords
            peak_positions = set()
            for pi in peak_indices[:5]:
                si = max(0, int(pi) - n + 1)
                ei = min(n - 1, int(pi))
                peak_positions.update(range(si, ei + 1))
            pos_list = list(peak_positions)
            if pos_list:
                found = False
                for _ in range(20):
                    p = pos_list[int(rng.integers(0, len(pos_list)))]
                    a_trial = a.copy()
                    delta = a[p] * 0.05 * rng.standard_normal()
                    a_trial[p] -= delta
                    mask2 = np.ones(n, dtype=bool)
                    mask2[p] = False
                    o2 = a_trial[mask2]
                    o2s = o2.sum()
                    if o2s > 1e-15:
                        a_trial[mask2] = o2 + delta * o2 / o2s
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        found = True
                        break
                if not found:
                    # Larger perturbation
                    noise = rng.standard_normal(n) * 0.005
                    a_trial = best_a + noise
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    a = a_trial
                    best_score = compute_score(a)
                    best_a = a.copy()

        if verbose and iters % 30 == 0:
            print(f"      PE iter={iters}, score={best_score:.8f}, spread={max_peak-min_peak:.2e}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
