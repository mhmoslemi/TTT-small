"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Aggressive L-BFGS with LogSumExp smoothing at very low temperatures (0.1-50)
2. LP-based alternating optimization with Nesterov acceleration
3. Targeted paired-coordinate peak shaving
4. Spectral perturbation for escaping local minima
5. Multi-length exploration

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver_cached' in globals():
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def softmax_obj_and_grad(a, T_val):
    """
    Compute LogSumExp-smoothed objective and gradient in log-space.
    f(z) = (2n/T) * logsumexp(T * conv(softmax(z), softmax(z)))
    """
    n = len(a)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    # Gradient w.r.t. a: sum_k w[k] * d(conv(a,a)[k])/da[i] = sum_k w[k] * 2*a[k-i]
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    # Project to log-space gradient
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_aggressive(a0, time_limit, temps, maxiter_per_stage=2000):
    """

    Goes from high to very low temperature for fine polishing.

    n = len(a0)
    z = z_from_a(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    for stage_i, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        print(f"    L-BFGS stage {stage_i}: T={T_val:.2f}, remaining={remaining:.0f}s")

        def func(z_in):
            a_in = softmax_from_z(z_in)
            f, _ = softmax_obj_and_grad(a_in, T_val)
            return f

        def jac(z_in):
            a_in = softmax_from_z(z_in)
            _, g = softmax_obj_and_grad(a_in, T_val)
            return g

        try:
            res = minimize(

def lp_alternating_nesterov(a0, time_limit, solver=None, max_iters=20000):
    """
    LP-based alternating optimization with Nesterov dual averaging.
    Converges to the saddle point of the minimax problem.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    y = a.copy()
    t0 = time.monotonic()
    running_sum = np.zeros(n)

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Nesterov step
        alpha = 2.0 / (it + 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        # Running average
        running_sum = running_sum * (it / (it + 1.0)) + b_new * (1.0 / (it + 1.0))

        # Evaluate both y and running average
        score_y = 2.0 * n * fast_convolve(y).max()
        score_avg = 2.0 * n * fast_convolve(running_sum).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if score_avg < best_score:
            best_score = score_avg
            best_a = running_sum.copy()

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={it} ({elapsed:.0f}s): best={best_score:.6f}, y={score_y:.6f}, avg={score_avg:.6f}")

    return best_a, best_score


def paired_peak_shave(a0, time_limit, n_rounds=50000):
    """
    Targeted peak shaving: identify peak contributions and
    systematically adjust pairs of coordinates to reduce the peak.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    npass = 0

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Exclude endpoints (always small) to find meaningful peak
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))

        if b[peak_k] <= b.max() * 0.9999:
            break  # No dominant peak

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Find top contributors
        n_top = min(8, len(contribs))
        top_ci = np.argsort(contribs)[::-1][:n_top]

        improved = False
        for ci in top_ci:
            i = idx[ci]
            j = peak_k - i
            if i == j:
                # Self-pair: reducing a[i] by delta, adding to some other position
                if a[i] < 1e-14:
                    continue
                for factor in [0.97, 0.94, 0.90, 0.85]:
                    delta = a[i] * (1.0 - factor)
                    if delta < 1e-16:
                        continue
                    # Distribute delta to positions not involved in this peak
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    # Also exclude positions that contribute to this peak
                    for ci2 in top_ci:
                        i2 = idx[ci2]
                        mask[i2] = False
                        mask[peak_k - i2] = False
                    if mask.sum() == 0:
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                    target_sum = a[mask].sum()
                    if target_sum < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[mask] += delta * a[mask] / target_sum
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial /= ss
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
            else:
                # Cross-pair: reduce a[i] or a[j]
                for target in [i, j]:
                    if a[target] < 1e-14:
                        continue
                    for factor in [0.98, 0.95, 0.90, 0.85]:
                        delta = a[target] * (1.0 - factor)
                        if delta < 1e-16:
                            continue
                        # Add to the other member of the pair
                        a_trial = a.copy()
                        a_trial[target] -= delta
                        a_trial[peak_k - target] += delta
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial /= ss
                        sc = 2.0 * n * fast_convolve(a_trial).max()
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break

        if not improved:
            # Try reducing all top contributors simultaneously
            a_trial = a.copy()
            for ci in top_ci[:4]:
                i = idx[ci]
                if i != peak_k - i:
                    a_trial[i] *= 0.96
                    a_trial[peak_k - i] *= 0.96
                else:
                    a_trial[i] *= 0.96
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss > 1e-15:
                a_trial /= ss
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
            if not improved:
                # Try random perturbation to escape
                if rnd % 200 == 0:
                    for _ in range(20):
                        a_trial = a.copy()
                        p = rng_int % n
                        q = (p + 1 + rng_int % (n - 1)) % n
                        if p == q:
                            continue
                        frac = 0.01 + 0.05 * (rng_int % 100) / 100.0
                        delta = a[p] * frac
                        a_trial[p] -= delta
                        a_trial[q] += delta
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss > 1e-15:
                            a_trial /= ss
                            sc = 2.0 * n * fast_convolve(a_trial).max()
                            if sc < best_score - 1e-15:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                improved = True
                                break
            if not improved and rnd > 1000:
                break

        if (rnd + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave round={rnd+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def spectral_perturb(a0, n_tries=200):
    """
    Perturb the sequence in the frequency domain to potentially escape local minima.
    Adds small random phase perturbations to the FFT.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    rng = np.random.default_rng(12345)

    fft_size = 1 << (2 * n - 2).bit_length()
    A = rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)

    for _ in range(n_tries):
        # Small random phase perturbation (preserve DC)
        dphase = rng.normal(0, 0.01, size=mag.shape)
        dphase[0] = 0
        new_A = mag * np.exp(1j * (phase + dphase))
        a_trial = irfft(new_A, fft_size)[:n]
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-10:
            continue
        a_trial /= ss
        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()

    return best_a


def coordinate_descent_refined(a0, time_limit, n_passes=50):
    """
    Golden-section coordinate descent on the simplex.
    For each coordinate, find the optimal value given others fixed.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue
            # When we set a[i] = t, the rest scale by (1-t)
            # conv(new)[k] = t^2 * a[i]*a[i] (if k=2i) + t*(1-t)*(a[i]*rest + rest*a[i]) + (1-t)^2 * rest_conv
            # This is quadratic in t, so we can find the exact minimum
            base = a.copy()
            base[i] = 0
            base_sum = base.sum()
            if base_sum < 1e-15:
                continue
            base = base / base_sum  # normalize rest to sum=1

            # conv(t*a[i]_e + (1-t)*base, itself)
            # = t^2 * e_i*e_i + t(1-t)*(e_i*base + base*e_i) + (1-t)^2 * (base*base)
            # For coordinate k: t^2 * [k==2i] + t(1-t)*(base[k-i] + base[k-i]) + (1-t)^2 * bb[k]
            # where [k==2i] = 1 if k=2i else 0, and bb = conv(base, base)

            bb = fast_convolve(base)
            # Objective: max_k [t^2 * delta_{k,2i} + 2t(1-t)*base[k-i] + (1-t)^2 * bb[k]]
            # We want to minimize this max.
            # Try a range of t values
            t_best = a[i]
            sc_best = best_score

            # Sample t values
            t_candidates = np.linspace(0.0, 1.0, 21)
            for t in t_candidates:
                a_trial = (1.0 - t) * base
                a_trial[i] = t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < sc_best:
                    sc_best = sc
                    t_best = t

            if t_best != a[i]:
                a = (1.0 - t_best) * base
                a[i] = t_best
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss > 1e-15:
                    a /= ss
                    sc = 2.0 * n * fast_convolve(a).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()

        if (p + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc pass={p+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0, min(1000, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
