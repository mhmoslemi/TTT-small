import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    # cross-correlation: ifft(conj(W) * F) gives sum_k w[k] * f[k+j]
    # We want sum_k w[k+j] * f[k], which is ifft(W * conj(F))[:n]... 
    # Let me verify: ifft(W * conj(F))[j] = sum_k W_k * conj(F_k) e^{2pi i jk/N}
    # = sum_k (sum_a w[a] e^{-2pi i ak/N}) (sum_b conj(f[b]) e^{-2pi i bk/N}) e^{2pi i jk/N}
    # = sum_{a,b} w[a] conj(f[b]) e^{-2pi i (a+b-j)k/N}
    # = sum_{a+b=j} w[a] conj(f[b])  (for circular, but we padded so it's linear)
    # Hmm, this gives sum_{a+b=j} w[a] f[b] (for real f)
    # That's the correlation at "negative lag" or something...
    
    # Let me use a different approach. 
    # grad[j] = sum_m w[j+m] * f[m]
    # Let's define: reverse f, call it fr[m] = f[n-1-m]
    # Then grad[j] = sum_m w[j+m] * fr[n-1-m]
    # Let p = n-1-m, m = n-1-p:
    # grad[j] = sum_p w[j+n-1-p] * fr[p]
    # = (w * fr)[j+n-1]  (convolution!)
    # where (w * fr)[k] = sum_p w[k-p] * fr[p]
    # So grad[j] = (w * fr)[j + n - 1]
    
    fr = f[::-1]
    conv_len = len_w + n - 1
    fft_N = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_N)
    FR = rfft(fr, fft_N)
    conv = irfft(W * FR, fft_N)[:conv_len]
    grad = conv[n - 1:n - 1 + n]
    return grad


def smooth_gradient_step(f, lr, temp, buf, momentum=0.9):
    """One step of smooth gradient descent with Nesterov momentum on the simplex."""
    n = len(f)
    b = fast_convolve(f)
    
    # Softmax weights (focus on high peaks)
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum
    
    # Compute gradient
    grad = compute_gradient(b, w, f)
    
    # Nesterov momentum
    buf[:] = momentum * buf - lr * grad
    f_new = project_simplex(f + buf)
    return f_new


def run_smooth_gradient(f_init, time_limit, lr_init=0.002, temp_init=0.005,
                        temp_max=0.5, label="sg"):
    """Smooth gradient descent with continuation on temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    buf = np.zeros(n)
    lr = lr_init
    temp = temp_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        f = smooth_gradient_step(f, lr, temp, buf)
        it += 1
        
        if it % 100 == 0:
            obj = fast_convolve(f).max()
            score = 2.0 * n * obj
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_score = score
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= 5:
                    temp = min(temp * 1.4, temp_max)
                    if no_improve >= 15:
                        lr *= 0.92
                        no_improve = 0
                    elif no_improve >= 25:
                        f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                        buf[:] = 0.0
                        no_improve = 0
                    else:
                        no_improve = 0
            
            if it % 500 == 0:
                print(f"  {label}: it={it} obj={best_obj:.10f} score={best_score:.6f} "
                      f"temp={temp:.5f} lr={lr:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_score


def lp_solve(f, peak_indices, M, eps=1e-10):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices.
    Returns (g, success).
    """
    n = len(f)
    K = len(peak_indices)
    
    if K == 0:
        return np.ones(n) / n, True
    
    A = np.zeros((K, n))
    for i in range(K):
        k = peak_indices[i]
        m_lo = max(0, k - n + 1)
        m_hi = min(k, n - 1)
        # A[i, m] = f[k-m] for m in [m_lo, m_hi]
        A[i, m_lo:m_hi + 1] = f[k - m_hi:k - m_lo + 1][::-1]
    
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=cp.SCS, max_iters=500000, eps=eps)
    except Exception:
        return None, False
    
    if g.value is None:
        return None, False
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, False
    return g_val, True


def run_lp_optimization(f_init, time_limit, top_k_init=25, top_k_max=250,
                        label="lp", seed=42):
    """
    LP-based optimization with adaptive peaks, binding constraint tracking,
    geometric line search, and restart strategy.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    it = 0
    stuck = 0
    top_k = top_k_init
    rng = np.random.default_rng(seed)
    
    # Track binding peaks
    binding_peaks = set()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Adaptive top_k
        if it % 60 == 0 and top_k < top_k_max:
            top_k = min(top_k + 15, top_k_max)
        
        b = fast_convolve(f)
        M = b.max()
        
        # Select peaks: top top_k + binding peaks
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        
        # Add binding peaks
        all_peaks = set(top_idx.tolist()) | binding_peaks
        # Limit total peaks
        if len(all_peaks) > top_k * 2:
            # Keep top (top_k * 2) by value
            peak_arr = np.array(list(all_peaks))
            peak_vals = b[peak_arr]
            keep = np.argsort(peak_vals)[-top_k * 2:]
            all_peaks = set(peak_arr[keep].tolist())
        
        peak_indices = np.array(sorted(all_peaks))
        
        g, success = lp_solve(f, peak_indices, M)
        if not success or g is None:
            stuck += 3
            if stuck > 40:
                f = project_simplex(f + rng.standard_normal(n) * 0.008)
                stuck = 0
                top_k = max(20, top_k // 2)
                binding_peaks = set()
            continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-4:
            stuck += 1
            if stuck > 40:
                f = project_simplex(f + rng.standard_normal(n) * 0.008)
                stuck = 0
                top_k = max(20, top_k // 2)
                binding_peaks = set()
            continue
        
        h = g / S_g
        
        # Identify binding constraints for next iteration
        A_rows = np.zeros((len(peak_indices), n))
        for i in range(len(peak_indices)):
            k = peak_indices[i]
            m_lo = max(0, k - n + 1)
            m_hi = min(k, n - 1)
            A_rows[i, m_lo:m_hi + 1] = f[k - m_hi:k - m_lo + 1][::-1]
        tightness = A_rows @ h  # Should be close to M/S_g for binding
        # Binding if A_rows @ h is close to the max constraint value
        max_tight = tightness.max()
        if max_tight > 0:
            binding_threshold = max_tight * 0.95
            binding_peaks = set(peak_indices[tightness >= binding_threshold].tolist())
            # Limit binding peaks
            if len(binding_peaks) > 30:
                # Keep only the worst 30
                bp_arr = np.array(list(binding_peaks))
                bp_vals = b[bp_arr]
                keep_bp = np.argsort(bp_vals)[-30:]
                binding_peaks = set(bp_arr[keep_bp].tolist())
        
        # Geometric line search
        cur_obj = M
        best_t = 0.0
        best_t_obj = cur_obj
        t = 0.4
        for _ in range(12):
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                t *= 0.6
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
            t *= 0.65
        
        if best_t < 0.005:
            stuck += 1
            if stuck > 40:
                f = project_simplex(f + rng.standard_normal(n) * 0.008)
                stuck = 0
                top_k = max(20, top_k // 2)
                binding_peaks = set()
            continue
        
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        
        cur_obj = fast_convolve(f).max()
        cur_score = 2.0 * n * cur_obj
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_score = cur_score
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
        
        if it % 80 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label}: it={it} obj={best_obj:.10f} score={best_score:.6f} "
                  f"top_k={top_k} bind={len(binding_peaks)} t={elapsed:.0f}s")
        
        if stuck > 50:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            stuck = 0
            top_k = max(20, top_k // 2)
            binding_peaks = set()
    
    return best_f, best_score


def coordinate_polish(f_init, time_limit):
    """Coordinate descent: try scaling each element independently."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = np.array([0.7, 0.85, 1.0, 1.15, 1.3, 1.5])
    
    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-15:
                continue
            best_j_obj = best_obj
            best_j_val = f[j]
            base_sum = f.sum()
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val < 0:
                    continue
                delta = new_val - f[j]
                new_sum = base_sum + delta
                if new_sum < 1e-10:
                    continue
                # Scale: f_trial = f / f.sum() * (f.sum() + delta) ... 
                # Actually simpler: just modify and renormalize
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_obj > best_j_obj:
                    best_obj = best_j_obj
                    improved = True
        if not improved:
            break
    
    return f, best_obj
