(2) Adaptive coordinate descent with peak-focused updates,

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_probs(b, T):
    """Compute softmax probabilities from values b with temperature T."""
    b_max = b.max()
    exp_b = np.exp((b - b_max) / T)
    s = exp_b.sum()
    if s < 1e-300:
        return np.ones_like(b) / len(b)
    return exp_b / s


def grad_conv_w(a, w, n, fft_size):
    """
    Gradient of n * sum_k w[k] * conv[k] w.r.t. a.
    conv[k] = sum_m a[m]*a[k-m]
    d conv[k]/d a[i] = 2*a[k-i] (with a[j]=0 for j outside [0,n-1])
    So grad[i] = 2 * sum_k w[k] * a[k-i]
    This is 2 * cross-correlation(w, a).
    """
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * corr


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def a_from_z(z):
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def nesterov_logspace(a0, time_limit, T_init=0.05, T_final=0.001,
                       lr_init=0.01, momentum=0.9, max_iter=1000000):
    """
    Nesterov-accelerated gradient descent in log-space with temperature annealing.
    Uses backtracking line search for step size.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = z_from_a(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 50000
    
    v = np.zeros(n)  # velocity for Nesterov
    lr = lr_init
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_init * (T_final / T_init) ** progress
        
        # Nesterov look-ahead
        z_look = z - momentum * v
        a_curr = a_from_z(z_look)
        
        # Compute soft-max weights and gradient
        b = fast_convolve(a_curr)
        w = softmax_probs(b, T)
        grad_a = grad_conv_w(a_curr, w, n, fft_size)
        
        # Transform to z-space
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)
        
        # Nesterov update
        v = momentum * v + lr * grad_z
        z_new = z - v
        
        # Evaluate
        a_new = a_from_z(z_new)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive restart
        if no_improve > max_no_improve:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.01
            v = np.zeros(n)
            no_improve = 0
            lr *= 0.5
        
        # Slow LR decay
        lr *= 0.99995
    
    return best_a, best_score

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    return irfft(A * A, fft_size)[:size]


def smax_weights(b, T):
    bm = b.max()
    e = np.exp((b - bm) / T)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(b) / len(b)
    return e / s


def grad_w(a, w, n, fft_size):
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return 2.0 * irfft(W * np.conj(A), fft_size)[:n]


def to_z(a):
    return np.log(np.maximum(a, 1e-300))


def from_z(z):
    zs = z - z.max()
    e = np.exp(zs)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    an = a / s
    return 2.0 * n * fast_convolve(an).max()


def adam_logspace_v2(a0, time_limit, T_init=0.05, T_final=0.001,

    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        if schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        elif schedule == 'hold_anneal':
            if progress < 0.3:
                T = T_init
            else:
                p2 = (progress - 0.3) / 0.7
                T = T_init * (T_final / T_init) ** p2
        else:
            T = T_init * (T_final / T_init) ** progress
        
        a_curr = from_z(z)
        b = fast_convolve(a_curr)
        w = smax_weights(b, T)
        ga = grad_w(a_curr, w, n, fft_size)
        S = np.dot(ga, a_curr)
        gz = a_curr * (ga - S)
        
        m = beta1 * m + (1 - beta1) * gz
        v = beta2 * v + (1 - beta2) * gz * gz
        mh = m / (1 - beta1**t)
        vh = v / (1 - beta2**t)

def lbgfs_refine(a0, time_limit, T=10.0, maxiter=2000):
    n = len(a0)
    z0 = to_z(a0)
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    
    def obj(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = from_z(z)
        b = fast_convolve(a)
        bm = b.max()
        e = np.exp((b - bm) / T)
        return 2.0 * n * T * (bm / T + np.log(e.sum()))
    
    def grad(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = from_z(z)
        b = fast_convolve(a)
        w = smax_weights(b, T)
        ga = grad_w(a, w, n, fft_size)
        S = np.dot(ga, a)
        return a * (ga - S)
    
    try:
        res = minimize(obj, z0, jac=grad, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-12, 'maxfun': maxiter * 30})
        a_out = from_z(res.x)
    except Exception:
        a_out = np.asarray(a0, dtype=np.float64).copy()
    
    return a_out

def interpolate_to_n(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi = fi / ss
    return fi
