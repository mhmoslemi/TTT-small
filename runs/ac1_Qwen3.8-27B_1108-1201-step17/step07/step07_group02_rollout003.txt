def peak_shave(a0, time_limit, n_steps=20000):
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autoconv(a)
        # Find the peak (excluding edges which are naturally low)
        b_internal = b.copy()
        b_internal[:max(1, n//10)] = -np.inf
        b_internal[-max(1, n//10):] = -np.inf
        peak_k = int(np.argmax(b_internal))
        
        if b[peak_k] < best_score / (2.0 * n) * 1.001:
            # Peak is already below best, try next
            break
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = idx[np.argsort(contribs)[::-1][:8]]
        
        improved = False
        for i in top_idx:
            if a[i] < 1e-14:
                continue
            for factor in [0.99, 0.97, 0.94, 0.9]:
                diff = a[i] * (1 - factor)
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] -= diff
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True

def lp_best_response(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build T matrix: T @ b = conv(a_fixed, b)
    # T[k, j] = a_fixed[k-j] if 0 <= k-j < n, else 0
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def lp_alternate(a0, time_limit, solver, n_iters=1000):
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP for best response to a
        b_new = lp_best_response(a, solver)
        if b_new is None:
            break
        
        # Blend
        alpha = 2.0 / (it + 2)
        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if (it + 1) % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP alt iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, np.linspace(0, 1, old_n), f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def get_lp_solver():
    if not HAS_CVXPY:

    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
