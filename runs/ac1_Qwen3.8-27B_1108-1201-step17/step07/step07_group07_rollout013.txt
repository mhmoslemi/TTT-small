import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def cross_convolve_fft(f, g):
    """Compute conv(f, g) where f has length n and g has length n."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

        hi = min(n - 1, kk)
        for j in range(lo, hi + 1):
            A_ub[i, kk - j] = f[j]
    
    b_ub = np.full(k, M_target)
    c = -np.ones(n)  # minimize -sum(g)

        # Subgradient: g[i] = sum_j w[i+j] * a[j]
        # Using FFT: g[i] = conv(a, w_flipped)[2n-2-i]
        w_flipped = w[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        A = rfft(a, fft_size)
        Wf = rfft(w_flipped, fft_size)
        conv_full = irfft(A * Wf, fft_size)
        g = conv_full[n-1:2*n-1][::-1]
        
        # Nesterov step
        v = momentum * v - lr * g
        a_new = project_simplex(a + v)
        
        obj_new = eval_max_b(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        T *= 0.995  # slowly decrease temperature
        
        if no_improve > 200:
            # Restart with smaller lr
            lr *= 0.8
            v = np.zeros(n)
            a = best_a.copy()
            no_improve = 0
        
        if lr < 1e-7:
            break
    
    return best_a, best_obj

def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_skewed_init(n, skew=0.3):
    """Initialize with a skewed (asymmetric) distribution."""
    x = np.linspace(0, 1, n)
    # Skewed beta-like distribution
    center = 0.5 + skew
    sigma = 0.2
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

    smooth gradient refinement, and multi-length exploration.

        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print(f"  UPDATE [{label}]: score={score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        parent = make_skewed_init(1152, 0.1)

    n_parent = len(parent)
    
    # === Phase 1: Aggressive LP on parent ===

        t_sg = min(rem * 0.7, 150.0)

        # Quick LP after gradient
        rem = deadline - time.monotonic() - 200

    # === Phase 3: Multi-length exploration ===

    if best_seq is not None:
