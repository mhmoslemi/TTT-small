"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play
as the primary optimization driver, combined with L-BFGS polishing
and coordinate descent for final refinement.

The LP-based approach: at each iteration, solve an LP to find the direction
that maximizes total mass while keeping the cross-convolution bounded.
ThisDirection, when mixed with the current sequence, reduces the auto-convolution
peak. This is a provably convergent scheme for this non-convex problem.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)


def smoothed_objective(z, T, n):
    """Soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    log_sum_exp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * log_sum_exp)


def smoothed_gradient(z, T, n):
    """Gradient of the soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lbgfs_optimize(a0, time_limit, T, maxiter=5000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    result = {'a': a0.copy(), 'score': compute_score(a0)}

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z)
        return smoothed_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 40, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < result['score']:
            result['a'] = a_out
            result['score'] = score
    except Exception:
        pass

    result['seconds'] = time.monotonic() - t0
    return result


def solve_lp_direction(f, M):
    """
    Solve the LP: maximize sum(g) subject to g >= 0 and (f*g)_k <= M for all k.
    Returns the optimal g vector.
    """
    n = len(f)
    # Build constraint matrix C where C[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # k ranges from 0 to 2n-2, j ranges from 0 to n-1
    # C is (2n-1) x n
    num_constraints = 2 * n - 1
    num_vars = n
    
    # Build the matrix in sparse form for efficiency
    # C[k, j] = f[k-j] if 0 <= k-j < n
    # For row k, non-zero entries are at columns j where 0 <= k-j < n
    # i.e., max(0, k-n+1) <= j <= min(n-1, k)
    
    from scipy.sparse import csr_matrix, coo_matrix
    
    rows = []
    cols = []
    vals = []
    
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    C = csr_matrix((vals, (rows, cols)), shape=(num_constraints, num_vars))
    
    # LP: max 1^T g s.t. C g <= M, g >= 0
    # linprog minimizes, so we negate the objective
    c_obj = -np.ones(n)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(
        c=c_obj,
        A_ub=C,
        b_ub=np.full(num_constraints, M),
        bounds=bounds,
        method='highs'
    )
    
    if res.status == 0:
        return np.maximum(res.x, 0)
    else:
        # Fallback: return uniform
        return np.ones(n) / n


def lp_fictitious_play(a0, time_limit, start_t=0.1, min_t=0.01, verbose=True):
    """
    LP-based fictitious play iteration.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    t = start_t
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Compute current peak
        conv = fast_convolve(a)
        M = conv.max()
        
        # Solve LP to get direction
        g = solve_lp_direction(a, M)
        g_sum = g.sum()
        
        if g_sum < 1e-15:
            break
        
        # Rescale g to match normalization
        g = g / g_sum
        
        # Update: a <- (1-t)*a + t*g
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        new_score = compute_score(a_new)
        
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            # Decrease step size slowly
            t = max(min_t, t * 0.999)
        else:
            # Try smaller step
            t = max(min_t, t * 0.8)
            if t < min_t:
                # Try a different approach: just use g directly if it's better
                g_score = compute_score(g)
                if g_score < best_score:
                    best_score = g_score
                    best_a = g.copy()
                    a = g.copy()
                    t = start_t
                else:
                    break
        
        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP iter={iters}, score={best_score:.8f}, t={t:.6f}, time={elapsed:.0f}s")
    
    return best_a, best_score, iters


def coordinate_descent(a0, time_limit, n_pairs_per_iter=200):
    """
    Coordinate descent using pairwise moves.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        pairs = []
        for _ in range(n_pairs_per_iter):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.append((i, j))

        for (i, j) in pairs:
            if a[i] < 1e-14:
                continue

            best_t = 0.0
            best_local_score = best_score

            base = min(a[i], a[j]) if a[j] > 0 else a[i]
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9, 0.01, 0.05]:
                t_val = frac * base
                if t_val >= a[i] - 1e-15:
                    t_val = a[i] * 0.5

                a_trial = a.copy()
                a_trial[i] -= t_val
                a_trial[j] += t_val
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local_score - 1e-14:
                    best_local_score = sc
                    best_t = t_val

                if a[j] > 1e-15:
                    t2 = frac * a[j]
                    a_trial2 = a.copy()
                    a_trial2[j] -= t2
                    a_trial2[i] += t2
                    a_trial2 = np.maximum(a_trial2, 0)
                    ss2 = a_trial2.sum()
                    if ss2 < 1e-15:
                        continue
                    a_trial2 = a_trial2 / ss2
                    sc2 = compute_score(a_trial2)
                    if sc2 < best_local_score - 1e-14:
                        best_local_score = sc2
                        best_t = -t2

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                ss = a.sum()
                a = a / ss
                best_score = best_local_score
                best_a = a.copy()
                improved = True

        if not improved and iters > 50:
            noise = rng.standard_normal(n) * 0.001
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def peak_equalization(a0, time_limit, n_peaks=10):
    """
    Iteratively try to equalize the top peaks of the auto-correlation.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        top_k = min(n_peaks, len(conv))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_values = conv[peak_indices]
        max_peak = peak_values.max()
        min_peak = peak_values.min()

        if max_peak - min_peak < 1e-8:
            break

        max_idx = int(np.argmax(conv))
        js = max(0, max_idx - n + 1)
        je = min(n - 1, max_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_idx - idx]

        top_contrib = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_contrib:
            i = idx[ci]
            if a[i] < 1e-14:
                continue

            for frac in [0.05, 0.1, 0.2, 0.3]:
                delta = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= delta
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                os = others.sum()
                if os < 1e-15:
                    continue
                a_trial[mask] = others + delta * others / os
                a_trial = np.maximum(a_trial, 0)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < result['score']:
            result['a'] = a_out
            result['score'] = score
    except Exception:
        pass

    result['seconds'] = time.monotonic() - t0
    return result


def build_conv_matrix(n):
    """Precompute the structure of the convolution constraint matrix."""
    num_constraints = 2 * n - 1
    rows = []
    cols = []
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_direction(f, M, row_idx, col_idx):
    """
    Solve LP: maximize sum(g) s.t. C*g <= M, g >= 0.
    C[k,j] = f[k-j] if valid.
    """
    n = len(f)
    num_constraints = 2 * n - 1
    
    # Build the values for the sparse matrix
    vals = f[f.shape[0] - 1 - (col_idx - (0))]  # This is wrong, let me fix
    
    # For entry (k, j), the value is f[k-j]
    # k = row_idx[i], j = col_idx[i]
    # value = f[row_idx[i] - col_idx[i]]
    vals = f[row_idx - col_idx]
    
    C = csr_matrix((vals, (row_idx, col_idx)), shape=(num_constraints, n))
    
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = linprog(
        c=c_obj,
        A_ub=C,
        b_ub=np.full(num_constraints, M),
        bounds=bounds,
        method='highs',
        options={'presolve': True}
    )
    
    if res.status == 0:
        return np.maximum(res.x, 0)
    else:
        return np.ones(n) / n


def lp_fictitious_play(a0, time_limit, verbose=True):
    """
    LP-based fictitious play with adaptive step size and momentum.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    
    # Precompute matrix structure
    row_idx, col_idx = build_conv_matrix(n)
    
    # Parameters
    t = 0.1  # step size
    min_t = 0.005
    max_t = 0.5
    momentum_buf = np.zeros(n)  # running average of directions
    
    stall_count = 0
    best_stall_score = best_score
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        elapsed = time.monotonic() - t0
        
        # Compute current peak
        conv = fast_convolve(a)
        M = conv.max()
        
        # Solve LP
        g = solve_lp_direction(a, M, row_idx, col_idx)
        g_sum = g.sum()
        
        if g_sum < 1e-15:
            break
        
        g = g / g_sum
        
        # Update momentum buffer
        momentum_buf = 0.9 * momentum_buf + 0.1 * g
        
        # Use momentum-smoothed direction
        dir = momentum_buf / momentum_buf.sum()
        
        # Try the step
        a_new = (1 - t) * a + t * dir
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = a
        else:
            a_new = a_new / a_new_sum
        
        new_score = compute_score(a_new)
        
        # Also try the raw LP direction
        g_score = compute_score(g)
        
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            # Slightly increase step if we're making good progress
            t = min(max_t, t * 1.01)
        elif g_score < best_score - 1e-12:
            # Jump to LP solution
            best_score = g_score
            best_a = g.copy()
            a = g.copy()
            stall_count = 0
            t = 0.1
        else:
            a = a_new  # Still move in the direction, even if not optimal
            stall_count += 1
            t = max(min_t, t * 0.95)
            
            # Check for stall
            if stall_count > 200:
                # Perturb and restart
                rng = np.random.default_rng(iters)
                perturbation = rng.standard_normal(n) * 0.01
                a = best_a + perturbation
                a = np.maximum(a, 0)
                a = a / a.sum()
                momentum_buf = np.zeros(n)
                t = 0.1
                stall_count = 0
        
        if verbose and iters % 200 == 0:
            print(f"  LP iter={iters}, score={best_score:.8f}, t={t:.4f}, "
                  f"stall={stall_count}, time={elapsed:.0f}s")
    
    return best_a, best_score, iters


def coordinate_descent(a0, time_limit, n_pairs_per_iter=300):
    """Coordinate descent using pairwise moves."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        for _ in range(n_pairs_per_iter):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if a[i] < 1e-14:
                continue

            best_t = 0.0
            best_local_score = best_score
            base = min(a[i], a[j]) if a[j] > 0 else a[i]
            
            for frac in [0.1, 0.3, 0.5, 0.7, 0.05]:
                t_val = frac * base
                if t_val >= a[i] - 1e-15:
                    t_val = a[i] * 0.5
                a_trial = a.copy()
                a_trial[i] -= t_val
                a_trial[j] += t_val
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local_score - 1e-14:
                    best_local_score = sc
                    best_t = t_val

                if a[j] > 1e-15:
                    t2 = frac * a[j]
                    a_trial2 = a.copy()
                    a_trial2[j] -= t2
                    a_trial2[i] += t2
                    a_trial2 = np.maximum(a_trial2, 0)
                    ss2 = a_trial2.sum()
                    if ss2 < 1e-15:
                        continue
                    a_trial2 = a_trial2 / ss2
                    sc2 = compute_score(a_trial2)
                    if sc2 < best_local_score - 1e-14:
                        best_local_score = sc2
                        best_t = -t2

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                ss = a.sum()
                a = a / ss
                best_score = best_local_score
                best_a = a.copy()
                improved = True

        if not improved and iters > 30:
            noise = rng.standard_normal(n) * 0.002
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score
