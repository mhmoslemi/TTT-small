"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2:
1. Diverse initialization with quick Nesterov optimization
2. LP-based fictitious play for basin refinement
3. Smooth continuation (log-sum-exp gradient) with backtracking line search
4. Sharp coordinate descent with golden-section search
5. Stochastic refinement with restarts
6. Final intensive polish
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def evaluate(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()

    """
    Compute gradient of (1/t) * log(sum_k exp(t * conv(a,a)[k]))
    using only the top_k peaks for efficiency.

    n = len(a)

    a = np.maximum(a, 0).astype(np.float64).copy()

    b = fast_convolve(a)  # conv(a, a)
    best_a = a.copy()
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue
            
            # The new convolution when changing a[i] by delta and renormalizing:
            # conv_new[k] = (b[k] + delta * (a[k-i] + a[i-k]) + delta^2 * I(k=0)) / (1+delta)^2
            # We want to minimize max_k conv_new[k]
            # This is a 1D optimization in delta
            
            # Precompute the "cross terms"
            # a[k-i] for k in [0, 2n-2]: valid when 0 <= k-i < n, i.e., i <= k < i+n
            # a[i-k] for k in [0, 2n-2]: valid when 0 <= i-k < n, i.e., i-n < k <= i
            
            cross = np.zeros(2 * n - 1)
            # a[k-i] term
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            if k_start <= k_end:
                cross[k_start:k_end+1] = a[0:n-(k_end-k_start)]  # a[k-i] for k in [k_start, k_end]
                # Actually: for k in [i, i+n-1], a[k-i] = a[0], a[1], ..., a[n-1]
                # So cross[i : i+n] = a[0:n] (capped at 2n-1)
                cross[i:min(i+n, 2*n-1)] = a[:min(n, 2*n-1-i)]
            
            # a[i-k] term
            k_start2 = max(0, i - n + 1)
            k_end2 = i
            if k_start2 <= k_end2:
                # For k in [i-n+1, i], a[i-k] = a[n-1], a[n-2], ..., a[0]
                # So cross[i-n+1 : i+1] += a[n-1 : -1 : -(i-k_start2+1)]
                # Simpler: for k, a[i-k] where i-k ranges from i-k_start2 to i-k_end2 = 0
                # i-k_start2 = i - max(0, i-n+1)
                # If i >= n-1: i-k_start2 = i-(i-n+1) = n-1, i-k_end2 = 0
                # If i < n-1: i-k_start2 = i-0 = i, i-k_end2 = 0
                cross[k_start2:k_end2+1] += a[i - np.arange(k_start2, k_end2+1)]

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP fictitious play, smooth continuation (log-sum-exp), 
coordinate descent with golden-section line search, and stochastic refinement.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def obj_value(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return float(2.0 * n * fast_convolve(a).max())

def get_lp_solver():
    import cvxpy as cp
    if '_lp_solver_global' in globals() and _lp_solver_global is not None:
        return _lp_solver_global
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_global = s
                return s
        except Exception:
            continue
    _lp_solver_global = cp.ECOS
    return cp.ECOS

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):

    if solver is None:
        solver = get_lp_solver()

    b_var = cp.Variable(n, nonneg=True)

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:

                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def nesterov_optimize(a0, time_limit, top_k=20, lr_init=0.006):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()

    no_improve = 0

    while time.monotonic() - t0 < time_limit:

        peaks = np.argpartition(b, -k_actual)[-k_actual:]

        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()

            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                j = np.arange(j_start, j_end + 1)
                sg[j] += 2.0 * w[idx] * a[k - j]
        v = beta_k * v + sg
        norm_v = np.linalg.norm(v)
        if norm_v > 1e-15:
            v = v / norm_v
        y_new = y - lr * v
        y_projected = simplex_project(y_new)
        score = obj_value(y_projected)
        if score < best_score - 1e-12:

def smooth_continuation(a0, time_limit, t_start=10.0, t_end=5000.0, t_factor=1.4, lr_init=0.01, top_k=200):
    """
    Smooth continuation method using log-sum-exp approximation.
    Gradually increases temperature to go from smooth to sharp.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = obj_value(a)
    t0 = time.monotonic()
    t = t_start
    lr = lr_init
    momentum = 0.85
    v = np.zeros(n)
    it = 0
    iters_per_temp = 100
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute smooth gradient
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        shifted = t * (pv - pv.max())
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Compute gradient
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-15:
                continue
            k = int(peaks[idx])
            wi = w[idx]
            j1s = max(0, k - n + 1)
            j1e = min(n - 1, k)
            if j1s <= j1e:
                j_arr = np.arange(j1s, j1e + 1)
                grad[j_arr] += wi * a[k - j_arr]
            j2s = max(0, k)
            j2e = n - 1
            if j2s <= j2e:
                j_arr = np.arange(j2s, j2e + 1)
                grad[j_arr] += wi * a[j_arr - k]
        
        # Momentum
        v = momentum * v + grad
        grad_norm = np.linalg.norm(v)
        if grad_norm > 1e-15:
            v = v / grad_norm
        
        # Backtracking line search
        a_trial = a.copy()
        lr_trial = lr
        obj_current = obj_value(a)
        found = False
        for bt in range(20):
            a_trial = a - lr_trial * v
            a_projected = simplex_project(a_trial)
            obj_new = obj_value(a_projected)
            if obj_new < obj_current - 1e-15:
                a = a_projected
                found = True
                break
            lr_trial *= 0.5
        
        if not found:
            # Stuck, try to restart velocity
            v = v * 0.5
            continue
        
        lr = max(lr * 1.02, lr_init * 0.5)  # Slowly increase lr
        if obj_new < best_score - 1e-13:
            best_score = obj_new
            best_a = a.copy()
        
        # Temperature increase
        if it % iters_per_temp == 0 and t < t_end:
            t *= t_factor
    
    return best_a, best_score
