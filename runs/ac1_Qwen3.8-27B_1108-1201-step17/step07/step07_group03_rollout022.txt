import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    """Find a working LP solver with timeout."""
    import cvxpy as cp
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    solved = [False]
    solver_name = [None]
    for sn in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK, cp.CBC]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=sn, timeout=10)
            if prob.status in ('optimal', 'optimal_inaccurate'):

                solved[0] = True
                break
        except Exception:
            continue
    if not solved[0]:
        solver_name[0] = cp.SCS
    _lp_solver = solver_name[0]
    return _lp_solver

def solve_lp_best_response(a_fixed, solver=None, top_k=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()

    M_var = cp.Variable()

    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, timeout=30)
    except Exception:

                continue
            try:

                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def nesterov_optimize(a0, time_limit, top_k=20, lr_init=0.005, verbose=False):
    """Nesterov-accelerated projected subgradient method with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    best_no_improve_score = best_score

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_k = (1 + np.sqrt(1 + 4 * (it - 1) ** 2)) / 2
        beta_k = (t_k - 1) / t_k

        b = fast_convolve(y)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        temp = 0.005 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j = np.arange(i_start, i_end + 1)
                sg[j] += 2.0 * w[idx] * y[k - j]

        v = beta_k * v + sg
        norm_v = np.linalg.norm(v)
        if norm_v > 1e-15:
            v = v / norm_v

        y_new = y - lr * v
        y_projected = simplex_project(y_new)

        score = 2.0 * n * fast_convolve(y_projected).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = y_projected.copy()
            no_improve = 0
        else:
            no_improve += 1

        y = y_projected
        lr *= 0.9999

        if no_improve > 200:
            lr *= 0.8
            no_improve = 0

        if no_improve > 500:
            noise = np.random.default_rng(it * 31 + 7).standard_normal(n) * 0.02
            y = simplex_project(best_a + noise)
            v = np.zeros(n)
            lr = max(lr_init * 0.7, lr * 1.5)
            no_improve = 0

    return best_a, best_score

                    break
                a_norm = a / s
                score = 2.0 * n * fast_convolve(a_norm).max()
                if score < best_score - 1e-14:
                    best_score = score
                    a = a_norm.copy()
                    old_val = a[i]
                    improved = True
                    break
                else:
                    a[i] = old_val
                    a = a / a.sum()

            # Occasionally try zeroing
            if rng.random() < 0.05 and old_val > 0.01:
                a[i] = 0.0
                s = a.sum()
                if s > 1e-15:
                    a_norm = a / s
                    score = 2.0 * n * fast_convolve(a_norm).max()
                    if score < best_score - 1e-14:
                        best_score = score
                        a = a_norm.copy()
                        improved = True
                    else:
                        a[i] = old_val
                        a = a / a.sum()

        if not improved and sweep > 10:
            # Try finer perturbations
            any_fine = False
            subset = rng.choice(n, size=min(n, 200), replace=False)
            for i in subset:
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = a[i]
                if old_val < 1e-14:
                    continue
                for factor in [0.97, 0.985, 1.015, 1.03]:
                    a[i] = old_val * factor
                    s = a.sum()
                    if s < 1e-15:
                        a[i] = old_val
                        break
                    a_norm = a / s
                    score = 2.0 * n * fast_convolve(a_norm).max()
                    if score < best_score - 1e-15:
                        best_score = score
                        a = a_norm.copy()
                        old_val = a[i]
                        improved = True
                        any_fine = True
                        break
                    else:
                        a[i] = old_val
                        a = a / a.sum()
            if not any_fine and sweep > 30:
                break

    return a, best_score

def peak_shave_refined(a0, time_limit):
    """Refined peak shaving: identify top peaks, adjust contributing coordinates jointly."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(4242)

    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        top_k = min(8, len(b))
        peaks = np.argpartition(b, -top_k)[-top_k:]

        improved = False
        for peak_k in peaks:
            peak_k = int(peak_k)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_idx = np.argsort(contribs)[::-1][:6]

            for ci in top_idx:
                i = int(idx[ci])
                j = peak_k - i
                if a[i] < 1e-14:
                    continue
                # Try reducing i (and possibly j)
                for f_i in [0.92, 0.85, 0.75]:
                    a_trial = a.copy()
                    a_trial[i] *= f_i
                    if i != j and a[j] > 1e-14:
                        for f_j in [0.95, 0.9]:
                            a_trial2 = a_trial.copy()
                            a_trial2[j] *= f_j
                            a_trial2 = np.maximum(a_trial2, 0)
                            st = a_trial2.sum()
                            if st > 1e-15:
                                a_trial2 = a_trial2 / st
                                sc = 2.0 * n * fast_convolve(a_trial2).max()
                                if sc < best_score - 1e-14:
                                    best_score = sc
                                    best_a = a_trial2.copy()
                                    a = a_trial2.copy()
                                    improved = True
                                    break
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st > 1e-15:
                        a_trial = a_trial / st
                        sc = 2.0 * n * fast_convolve(a_trial).max()
                        if sc < best_score - 1e-14:
                            best_score = sc
                            best_a = a_trial.copy()
                            a = a_trial.copy()
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try random small perturbations to escape
            if step % 50 == 0:
                mask = rng.random(n) < 0.02
                a_trial = best_a.copy()
                a_trial[mask] *= (1 + rng.standard_normal(int(mask.sum())) * 0.05)
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial = a_trial / st
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved = True
            if not improved and step > 200:
                break

    return best_a, best_score

def spectral_init(n, rng):
    """Initialize using spectral shaping: create flat-ish power spectrum."""
    # Create a random phase spectrum, take magnitude, shape it, inverse FFT
    fft_size = 1 << (2 * n).bit_length()
    # Random complex spectrum with controlled magnitude
    mag = rng.random(fft_size // 2 + 1) ** 0.5
    phase = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spec = mag * np.exp(1j * phase)
    # Make it Hermitian
    full_spec = np.zeros(fft_size, dtype=np.complex128)
    full_spec[:fft_size // 2 + 1] = spec
    for i in range(1, fft_size // 2):
        full_spec[fft_size - i] = np.conj(spec[i])

    a_raw = irfft(full_spec, fft_size)[:n]
    a = np.maximum(a_raw, 0)
    s = a.sum()
    if s < 1e-10:
        # Fallback: use magnitude directly
        a = np.abs(irfft(full_spec, fft_size)[:n])
        a = np.maximum(a, 0)
        s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_fourier_mix(n, rng):
    """Initialize with controlled Fourier superposition for low peak."""
    n_harmonics = int(rng.integers(5, 30))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = rng.choice(np.arange(1, n // 2 + 1), size=n_harmonics, replace=False)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_structured(n, rng):
    """Initialize with structured patterns (periodic with noise)."""
    period = int(rng.integers(2, max(3, n // 10)))
    pattern = rng.random(period)
    pattern = np.maximum(pattern, 0.05)
    a = np.tile(pattern, n // period + 1)[:n]
    # Add small noise to break exact periodicity
    a += rng.standard_normal(n) * 0.05
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def two_point_optimize(a, i, j, time_limit_per_call=0.5):
    """Optimize two coordinates simultaneously (small 2D local search)."""
    n = len(a)
    best = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    ai, aj = a[i], a[j]

    # Try a grid of (ai, aj) values
    for fi in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
        for fj in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
            a_try = a.copy()
            a_try[i] = ai * fi
            a_try[j] = aj * fj
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            sc = 2.0 * n * fast_convolve(a_try).max()
            if sc < best_score - 1e-15:
                best_score = sc
                best = a_try.copy()
    return best, best_score

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate via the actual evaluator and update best."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return best_seq, best_score
