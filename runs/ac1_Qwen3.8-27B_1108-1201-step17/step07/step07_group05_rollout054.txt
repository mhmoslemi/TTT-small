            k_minus_i = ks - i

            a_kj = np.zeros_like(ks, dtype=np.float64)
            mask_j = (k_minus_j >= 0) & (k_minus_j < n)
            a_kj[mask_j] = a[k_minus_j[mask_j]]

            a_ki = np.zeros_like(ks, dtype=np.float64)
            mask_i = (k_minus_i >= 0) & (k_minus_i < n)
            a_ki[mask_i] = a[k_minus_i[mask_i]]

            c_base = conv[ks]
            A_k = 2.0 * (a_kj - a_ki)

            # B_k: quadratic coefficient
            B_k = np.zeros_like(ks, dtype=np.float64)
            bk_2i = (ks == 2 * i) & (0 <= i < n)
            bk_2j = (ks == 2 * j) & (0 <= j < n)
            bk_ij = (ks == i + j) & (0 <= i < n) & (0 <= j < n)
            B_k = bk_2i.astype(np.float64) + bk_2j.astype(np.float64) - 2.0 * bk_ij.astype(np.float64)

            # Also: the non-affected peaks remain constant
            c_rest_max = 0.0
            # Check if there are peaks outside [k_lo, k_hi]
            if k_lo > 0:
                c_rest_max = max(c_rest_max, conv[:k_lo].max())
            if k_hi < 2 * n - 2:
                c_rest_max = max(c_rest_max, conv[k_hi + 1:].max())

            def max_at_t(t):
                """Compute max of c_k(t) for the given t."""
                vals = c_base + t * A_k + t * t * B_k
                return max(vals.max(), c_rest_max)

            # Golden section search on t in [-a[j], a[i]]
            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_lo >= t_hi:
                continue

            # First check: does moving in either direction help?
            score_0 = max_at_t(0.0)
            if score_0 < best_score / (2.0 * n) - 1e-12:
                continue  # Already below best (shouldn't happen usually)

            # Quick check at small steps
            step = (t_hi - t_lo) * 0.1
            score_fwd = max_at_t(min(step, t_hi))
            score_bwd = max_at_t(max(-step, t_lo))
            if score_fwd >= score_0 - 1e-14 and score_bwd >= score_0 - 1e-14:
                continue  # No improvement in either direction

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            t1 = t_hi - gr * (t_hi - t_lo)
            t2 = t_lo + gr * (t_hi - t_lo)
            f1 = max_at_t(t1)
            f2 = max_at_t(t2)

            for _ in range(40):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = t2
                    t2 = t1
                    f2 = f1
                    t1 = t_hi - gr * (t_hi - t_lo)
                    f1 = max_at_t(t1)
                else:
                    t_lo = t1
                    t1 = t2
                    f1 = f2
                    t2 = t_lo + gr * (t_hi - t_lo)
                    f2 = max_at_t(t2)

            t_opt = (t_lo + t_hi) / 2.0
            f_opt = max_at_t(t_opt)

            # Apply if improvement
            new_score = 2.0 * n * f_opt
            if new_score < best_score - 1e-13:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss < 1e-15:
                    a = best_a.copy()
                    break
                a = a / ss
                best_score = new_score
                best_a = a.copy()
                improvements += 1
                # Update conv for next iteration
                conv = fast_convolve(a)

        if improvements == 0 and sweep > 5:
            # Perturb to escape local minimum
            perturb_size = int(n * 0.05)
            idx = rng.choice(n, size=perturb_size, replace=False)
            noise = rng.standard_normal(perturb_size) * 0.005 * a[idx].mean()
            a_trial = a.copy()
            a_trial[idx] += noise
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss > 1e-15:
                a_trial = a_trial / ss
                sc_trial = compute_score(a_trial)
                if sc_trial < best_score * 1.01:  # Accept small degradation
                    a = a_trial
                else:
                    a = best_a.copy()
        else:
            a = best_a.copy()

        if sweep % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Pairwise sweep {sweep}: score={best_score:.10f}, "
                  f"improvements={improvements}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def gradient_descent_active(a0, time_limit, lr_init=0.1, decay=0.9995):
    """
    Gradient descent on the true (non-smooth) max objective using
    the subgradient at active peaks, with line search.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = lr_init
    rng = np.random.default_rng(555)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        c_max = conv.max()
        # Find active peaks (within 5% of max)
        threshold = c_max * 0.95
        active = np.where(conv >= threshold)[0]

        # Compute weighted subgradient
        if len(active) == 0:
            break
        active_weights = np.exp((conv[active] - c_max) / (c_max * 0.01 + 1e-10))
        active_weights /= active_weights.sum()

        # Gradient: sum of w_k * grad(c_k)
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            w_k = active_weights[list(active).index(k)]
            # grad(c_k) w.r.t. a_m = a_{k-m} for valid m
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            if m_lo > m_hi:
                continue
            ms = np.arange(m_lo, m_hi + 1)
            partners = k - ms
            grad[m_lo:m_hi + 1] += w_k * a[partners]

        # Project gradient (zero mean for sum-preserving)
        grad -= grad.mean()

        # Normalize
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm

        # Line search
        best_t = 0.0
        best_local = best_score
        for t_factor in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            a_trial = a + t_factor * grad
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            sc = compute_score(a_trial)
            if sc < best_local - 1e-14:
                best_local = sc
                best_t = t_factor

        if best_t == 0:
            stall += 1
            if stall > 100:
                # Random perturbation
                noise = rng.standard_normal(n) * lr * 0.5
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
            lr *= 0.9
        else:
            a = a + best_t * grad
            a = np.maximum(a, 0)
            a = a / a.sum()
            if best_local < best_score - 1e-14:
                best_score = best_local
                best_a = a.copy()
            stall = 0
            lr = min(lr * 1.05, 0.5)

        if time.monotonic() - t0 < time_limit and (int(time.monotonic() - t0) % 5 < 1):
            pass  # skip print for speed

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def cosine_window_seq(n, harmonics=5, rng=None):
    """Generate a cosine-modulated non-negative sequence."""
    if rng is None:
        rng = np.random.default_rng(0)
    a = np.ones(n) / n
    for h in range(1, harmonics + 1):
        amp = rng.random() * 0.05 / h
        phase = rng.random() * 2 * np.pi
        a += amp * np.cos(2 * np.pi * h * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    # Uniform with cosine modulation
    inits.append(cosine_window_seq(n, 5, rng))
    # Triangular
    a = 1.0 - 2.0 * np.abs(np.arange(n) / n - 0.5)
    a = np.maximum(a, 0)
    inits.append(a / a.sum())
    # Bump in center
    a = np.exp(-((np.arange(n) - n / 2) / (n / 4)) ** 2)
    inits.append(a / a.sum())
    # Two bumps
    a = np.exp(-((np.arange(n) - n * 0.3) / (n / 8)) ** 2) + \
        np.exp(-((np.arange(n) - n * 0.7) / (n / 8)) ** 2)
    inits.append(a / a.sum())
    # Random non-negative
    a = rng.random(n)
    inits.append(a / a.sum())
    return inits


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization:
    1. L-BFGS with adaptive temperature (multiple starting points)
    2. Gradient descent on true max with active set
    3. Exact pairwise transfer refinement
    4. Final L-BFGS polish at low temperature
    Tests multiple sequence lengths around 768.
    """
    global deadline_global
    rng = np.random.default_rng(seed)
    deadline_global = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        t = np.linspace(0, 2 * np.pi * 3, n_init)
        parent = parent + 0.05 * np.cos(t)
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.10f}")

    n_parent = len(parent)
    print(f"\n=== Starting optimization (target: 1.5027, current: {best_score:.10f}) ===")

    # === Phase 1: L-BFGS from parent with adaptive T ===
    print(f"\n=== Phase 1: L-BFGS Adaptive T from parent (n={n_parent}) ===")
    rem = deadline_global - time.monotonic()
    t_alloc = min(400.0, rem * 0.3)
    if t_alloc > 30:
        a1, s1 = lbgfs_adaptive(best_seq, t_alloc, T_start=10000, T_end=20,
                                maxiter_per_phase=3000, num_phases=5)
        best_seq, best_score = check_and_update(a1, "lbgfs_adaptive", best_seq, best_score)

    # === Phase 2: Pairwise refinement ===
    print(f"\n=== Phase 2: Pairwise Refinement (n={n_parent}, best={best_score:.10f}) ===")
    rem = deadline_global - time.monotonic()
    t_alloc = min(300.0, rem * 0.2)
    if t_alloc > 30:
        a2, s2 = pairwise_refine(best_seq, t_alloc, n_pairs_per_sweep=3000)
        best_seq, best_score = check_and_update(a2, "pairwise", best_seq, best_score)

    # === Phase 3: L-BFGS again (refined start) ===
    print(f"\n=== Phase 3: L-BFGS Refined (n={n_parent}, best={best_score:.10f}) ===")
    rem = deadline_global - time.monotonic()
    t_alloc = min(200.0, rem * 0.15)
    if t_alloc > 20:
        a3, s3 = lbgfs_adaptive(best_seq, t_alloc, T_start=2000, T_end=10,
                                maxiter_per_phase=2000, num_phases=4)
        best_seq, best_score = check_and_update(a3, "lbgfs_refined", best_seq, best_score)

    # === Phase 4: More pairwise ===
    print(f"\n=== Phase 4: Pairwise 2 (best={best_score:.10f}) ===")
    rem = deadline_global - time.monotonic()
    t_alloc = min(200.0, rem * 0.15)
    if t_alloc > 20:
        a4, s4 = pairwise_refine(best_seq, t_alloc, n_pairs_per_sweep=4000)
        best_seq, best_score = check_and_update(a4, "pairwise2", best_seq, best_score)

    # === Phase 5: Try diverse initializations at parent length ===
    print(f"\n=== Phase 5: Diverse Inits (best={best_score:.10f}) ===")
    inits = diverse_init(n_parent, rng)
    for ii, init in enumerate(inits[:3]):
        rem = deadline_global - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(100.0, rem * 0.1)
        print(f"  Diverse init {ii}: initial score={compute_score(init):.6f}")
        a5, s5 = lbgfs_adaptive(init, t_alloc, T_start=5000, T_end=50,
                                maxiter_per_phase=1500, num_phases=4)
        best_seq, best_score = check_and_update(a5, f"diverse_{ii}", best_seq, best_score)

    # === Phase 6: Try n = n_parent + 64 ===
    print(f"\n=== Phase 6: Try n={n_parent + 64} (best={best_score:.10f}) ===")
    rem = deadline_global - time.monotonic()
    if rem > 100:
        alt_n = n_parent + 64
        t_alloc = min(150.0, rem * 0.15)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  Interpolated to n={alt_n}, initial score={compute_score(init_alt):.6f}")
        a6, s6 = lbgfs_adaptive(init_alt, t_alloc, T_start=5000, T_end=50,
                                maxiter_per_phase=2000, num_phases=4)
        best_seq, best_score = check_and_update(a6, f"n{alt_n}", best_seq, best_score)

    # === Phase 7: Try n = n_parent - 64 ===
    print(f"\n=== Phase 7: Try n={n_parent - 64} (best={best_score:.10f}) ===")
    rem = deadline_global - time.monotonic()
    if rem > 80:
        alt_n = n_parent - 64
        t_alloc = min(100.0, rem * 0.1)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  Interpolated to n={alt_n}, initial score={compute_score(init_alt):.6f}")
        a7, s7 = lbgfs_adaptive(init_alt, t_alloc, T_start=5000, T_end=50,
                                maxiter_per_phase=1500, num_phases=3)
        best_seq, best_score = check_and_update(a7, f"n{alt_n}", best_seq, best_score)

    # === Phase 8: Final pairwise on best ===
    print(f"\n=== Phase 8: Final Pairwise Polish (best={best_score:.10f}) ===")
    rem = deadline_global - time.monotonic()
    t_alloc = min(200.0, rem * 0.3)
    if t_alloc > 20:
        a8, s8 = pairwise_refine(best_seq, t_alloc, n_pairs_per_sweep=5000)
        best_seq, best_score = check_and_update(a8, "final_pairwise", best_seq, best_score)

    # === Phase 9: Final L-BFGS at very low T ===
    print(f"\n=== Phase 9: Final L-BFGS Low-T (best={best_score:.10f}) ===")
    rem = deadline_global - time.monotonic()
    t_alloc = min(100.0, rem * 0.4)
    if t_alloc > 15:
        a9, s9 = lbgfs_adaptive(best_seq, t_alloc, T_start=500, T_end=5,
                                maxiter_per_phase=2000, num_phases=3)
        best_seq, best_score = check_and_update(a9, "final_lbgfs", best_seq, best_score)

    # === Phase 10: One more pairwise ===
    rem = deadline_global - time.monotonic()
    t_alloc = min(60.0, rem * 0.5)
    if t_alloc > 10:
        print(f"\n=== Phase 10: Final Pairwise 2 (best={best_score:.10f}) ===")
        a10, s10 = pairwise_refine(best_seq, t_alloc, n_pairs_per_sweep=5000)
        best_seq, best_score = check_and_update(a10, "final_pairwise2", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = parent
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    print(f"TARGET: 1.5027")
    print(f"{'='*60}")
    return [float(x) for x in best_seq]
