For n=2: conv[1:3][::-1] = [conv[2], conv[1]] = [w1*a1+w0*a0, w2*a1+w1*a0]. 
grad = 2 * [w1*a1+w0*a0, w2*a1+w1*a0]. ✓

Great, so:
```python
grad = 2.0 * conv[n-1 : 2*n-1][::-1]
```

This is correct!

Now let me put together the full algorithm. I'll use:
1. FFT-based convolution for evaluating the objective
2. FFT-based gradient computation for the smoothed objective
3. Nesterov accelerated gradient with adaptive temperature
4. LP fictitious play for the initial phase
5. Peak shaving for the final polish

Let me also think about the temperature schedule. Starting with T large (smooth, gradient points in a reasonable direction) and decreasing T (sharper, focuses on the actual max). A good schedule might be:
- T starts at max_b (the current peak value)
- T decreases geometrically: T_k = T_0 * (0.95)^k
- Or: T_k = max_b / (1 + k/10)

Actually, I think a better approach is:
- Start with T = max_b * 0.5 (so the softmax gives reasonable weights)
- Decrease T by factor 0.95 every 100 iterations
- Stop when T < max_b * 0.01 (at this point the softmax is essentially a hard max)

For the Nesterov acceleration:
- y_k = x_k + beta_k * (x_k - x_{k-1})
- x_{k+1} = project_simplex(y_k - alpha_k * grad(f(y_k)))
- alpha_k = 2 / (k + 2) (standard) or adaptive
- beta_k = (1 - 1/sqrt(k+3)) / (1 + 1/sqrt(k+1)) (standard Nesterov momentum)

Actually, for the simplex-constrained problem, the standard Nesterov acceleration with projection works well. Let me implement it.

One more consideration: the problem is to minimize f(a) = max_k (a*a)[k] subject to a >= 0, sum(a) = 1. The score is 2*n*f(a).

For the LP fictitious play, the best response to a is:
minimize M subject to (a*b)[k] <= M for all k, b >= 0, sum(b) = 1.

This is a standard LP with 2n-1 + 1 = 2n constraints and n+1 variables. For n=768, that's 1536 constraints and 769 variables. PDLP should handle this quickly.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based fictitious play (PDLP) for initial optimization
2. Nesterov accelerated gradient with softmax smoothing
3. Targeted peak shaving with line search
4. L-BFGS final polish
Explores multiple sequence lengths with adaptive time allocation.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic

# Core computation functions

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    cond = u * rho_candidates > (cssv - 1.0)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

def compute_grad_smooth(a, w):
    """Compute gradient of sum_k w[k]*(a*a)[k] wrt a using FFT.
    w is a weight vector of length 2n-1 (non-negative, sums to 1).
    Returns gradient of length n.
    """
    n = len(a)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    u = np.zeros(2 * n - 1)
    u[:] = w[::-1]
    U = rfft(u, fft_size)
    A = rfft(a, fft_size)
    conv = irfft(U * A, fft_size)[:3 * n - 2]
    return 2.0 * conv[n - 1:2 * n - 1][::-1]

def softmax_weights(b, T):
    """Compute softmax weights for convolution values b with temperature T."""
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted / T)
    w = exp_b / exp_b.sum()
    return w

_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                print(f"  LP solver: {solver_name}")
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [
        T_mat @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """LP-based fictitious play for auto-correlation minimization."""
    if solver is None:
        solver = get_lp_solver()
    t0 = monotonic()
    
    if initial is not None:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        a = a / s if s > 1e-15 else np.ones(n) / n
    else:
        a = np.ones(n) / n
    
    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    b_avg = b.copy().astype(np.float64)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    iters = 0
    stagnation = 0

    while monotonic() - t0 < time_limit:
        iters += 1
        
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        
        a_new, _ = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
        
        a_avg = (iters * a_avg + a) / (iters + 1)
        b_avg = (iters * b_avg + b) / (iters + 1)
        
        candidates = [a, b, 0.5*(a+b), a_avg, 0.5*(a_avg+b_avg)]
        improved = False
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand = cand / s
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = cand.copy()
                improved = True
                stagnation = 0
            else:
                stagnation += 1
        
        if stagnation > 300:
            a = a + np.random.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            b = b + np.random.standard_normal(n) * 0.002
            b = np.maximum(b, 0)
            b = b / b.sum()
            stagnation = 0
            
        if verbose and iters % 20 == 0:
            print(f"    FP iter={iters}: best={best_score:.6f}")

    return best_a, best_score, iters

def nesterov_accelerated(a0, time_limit, T0=None, T_min=None, verbose=False):
    """Nesterov accelerated gradient for smoothed max-convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    
    if T0 is None:
        T0 = fast_convolve(a).max() * 0.5
    if T_min is None:
        T_min = T0 * 0.01
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    T = T0
    
    x = a.copy()
    y = a.copy()
    k = 0
    t0 = monotonic()
    
    while monotonic() - t0 < time_limit:
        k += 1
        alpha = 2.0 / (k + 2)
        beta = ((k - 1) / (k + 2)) if k > 1 else 0.0
        
        # Nesterov extrapolation
        y_new = x + beta * (x - y)
        y_new = project_simplex(y_new)
        
        # Compute gradient at y_new
        b_y = fast_convolve(y_new)
        w = softmax_weights(b_y, T)
        grad = compute_grad_smooth(y_new, w)
        
        # Update
        x_new = project_simplex(y_new - alpha * grad)
        
        # Evaluate
        score = 2.0 * n * fast_convolve(x_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = x_new.copy()
        
        y = x
        x = x_new
        
        # Decrease temperature
        if k % 50 == 0:
            T = max(T * 0.92, T_min)
        
        if verbose and k % 200 == 0:
            print(f"    Nesterov k={k}: T={T:.2e} best={best_score:.6f}")
    
    return best_a, best_score

def lbgfs_polish(a0, time_limit, T=50, maxiter=3000):
    """L-BFGS in log-space with very small temperature for fine polish."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]
    
    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse / T
    
    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()
        
        grad_a = compute_grad_smooth(a, w)
        grad_a *= 2.0 * n / T
        
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0
    except Exception:
        a_out = a0
    
    return a_out

def peak_shave(a0, time_limit, max_steps=5000):
    """Targeted peak shaving with proper line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    failed = 0
    
    for step in range(max_steps):
        if monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        
        # Find contributing pairs
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        if contribs.sum() < 1e-20:
            break
        
        improved = False
        
        # Strategy 1: Reduce top contributor with redistribution
        top_idx = np.argsort(contribs)[::-1][:3]
        for ci in top_idx:
            i = int(idx[ci])
            old_val = a[i]
            if old_val < 1e-14:
                continue
            
            # Try different reduction factors
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7]:
                diff = old_val * (1 - factor)
                if diff < 1e-15:
                    continue
                
                a_trial = a.copy()
                a_trial[i] = old_val * factor
                
                # Redistribute to other positions (proportional)
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        # Strategy 2: Reduce symmetric pair
        if not improved:
            for ci in top_idx:
                i = int(idx[ci])
                j = peak_k - i
                if j < 0 or j >= n or i == j or a[j] < 1e-14:
                    continue
                for f in [0.9, 0.85, 0.8]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial[j] *= f
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        # Strategy 3: Small random perturbation focused on peak region
        if not improved:
            for attempt in range(3):
                # Perturb positions that contribute to the peak
                perturb_mask = np.zeros(n, dtype=bool)
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                perturb_mask[js:je+1] = True
                
                a_trial = a.copy()
                a_trial[perturb_mask] += np.random.standard_normal(perturb_mask.sum()) * 0.001
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            failed += 1
            if failed > 100:
                break
        else:
            failed = 0
    
    return best_a, best_score

def subgradient_momentum(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=50):
    """Subgradient method with momentum, focusing on top-K peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    lr = lr_init
    t0 = monotonic()
    no_improve = 0
    
    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        
        # Softmax weights over peaks
        temp = 0.01 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        w = w / w.sum()
        
        # Build effective weight vector
        w_full = np.zeros(2*n-1)
        w_full[peaks] = w
        
        # Compute gradient
        grad = compute_grad_smooth(a, w_full)
        
        # Update with momentum
        v = momentum * v - lr * grad
        a_candidate = a + v
        a_new = project_simplex(a_candidate)
        
        score = 2.0 * n * fast_convolve(a_new).max()
        
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9995
        if no_improve > 200:
            a = best_a + np.random.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 3, 0.005)
            no_improve = 0
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score

    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

1. LP-based fictitious play (PDLP) for initial optimization

3. L-BFGS polish in log-space at multiple temperatures

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

# Core computation functions
# ============================================================

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    cond = u * rho_candidates > (cssv - 1.0)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
    else:
        theta = 0.0
    result = np.maximum(v - theta, 0.0)
    return result

def compute_grad_smooth(a, w):
    """Compute gradient of sum_k w[k]*(a*a)[k] wrt a using FFT.
    w is a weight vector of length 2n-1 (non-negative, sums to 1).
    Returns gradient of length n: grad[i] = 2 * sum_k w[k]*a[k-i].
    """
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    u = np.zeros(2 * n - 1)
    u[:] = w[::-1]
    U = rfft(u, fft_size)
    A = rfft(a, fft_size)
    conv = irfft(U * A, fft_size)[:3 * n - 2]
    return 2.0 * conv[n - 1:2 * n - 1][::-1]

def softmax_weights(b, T):
    """Compute softmax weights for convolution values with temperature T."""
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted / T)
    total = exp_b.sum()
    if total < 1e-300:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
        return w
    return exp_b / total

def lbgfs_polish(a0, time_limit, T=50, maxiter=3000):
    """L-BFGS in log-space with temperature smoothing for fine polish."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def obj_func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse / T

    def obj_jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()

        grad_a = compute_grad_smooth(a, w)
        grad_a *= 2.0 * n / T

        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z

    try:
        res = minimize(
            fun=obj_func, x0=z0, jac=obj_jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0
    except Exception:
        a_out = a0

    return a_out

                a_trial[perturb_mask] += noise
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            failed += 1
            if failed > 80:
                break
        else:
            failed = 0

    return best_a, best_score

def damped_lp_refine(a0, time_limit, solver=None, max_iters=3000):
    """Damped alternating LP refinement with adaptive step size."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()

    for it in range(max_iters):
        if monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        alpha = min(1.0, 3.0 / (it + 3))
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
