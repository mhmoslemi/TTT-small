    b = fast_convolve(a)

    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()
    # Objective

    # Gradient: dF/da[i] = 4n * sum_k w[k] * a[k-i]

    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    correl = irfft(np.conj(W) * A, fft_size)
    grad = 4.0 * n * correl[:n]
    return obj, grad, w

def lbfgs_smoothed(a_init, time_limit, T, maxiter=5000, verbose=True, label=""):
    """L-BFGS on log-sum-exp smoothed objective in log-space."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    z0 = np.log(a)
    z0 = z0 - z0.mean()
    t0 = time.monotonic()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a_s = softmax_from_z(z)
        obj, _, _ = lse_smoothed_obj_and_grad(a_s, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        a_s = softmax_from_z(z)
        _, grad_a, _ = lse_smoothed_obj_and_grad(a_s, T, n)
        # Transform to z-space: grad_z[i] = a[i]*(grad_a[i] - a^T grad_a)
        mean_grad = np.dot(a_s, grad_a)
        grad_z = a_s * (grad_a - mean_grad)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        z_opt = res.x
        a_out = softmax_from_z(z_opt)
    except Exception as e:
        if verbose:
            print(f"    {label} L-BFGS error: {e}")
        a_out = a_init.copy()

    return a_out

    temperatures = [200, 100, 50, 20, 10, 5, 2, 1]
    for T in temperatures:
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        t_alloc = min(rem * 0.9 / len(temperatures), rem * 0.4)
        if t_alloc < 3:
            continue

        a_new = lbfgs_smoothed(a, t_alloc, T=T, maxiter=2000, verbose=False, label=f"{label}_T{T}")
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()

        if verbose:
            print(f"    {label} T={T}: score={compute_score(a):.7f} best={best_score:.7f}")

    return best_a, best_score


def random_restart_local(a_best, score_best, n_restarts=5, time_per=30):
    """Try random perturbations of the best solution followed by local optimization."""
    n = len(a_best)
    t0 = time.monotonic()
    best_a = a_best.copy()
    best_score = score_best

    for r in range(n_restarts):
        if time.monotonic() - t0 + time_per > 1800:
            break
        rng = np.random.default_rng(42 + r * 1000)
        # Gaussian perturbation
        noise_scale = [0.01, 0.02, 0.05, 0.1, 0.2][r % 5]
        noise = rng.standard_normal(n) * noise_scale
        a_pert = a_best + noise
        a_pert = project_simplex(a_pert)

        # Quick subgradient refinement
        t_rem = min(time_per, 1800 - (time.monotonic() - t0) - 60)
        if t_rem < 5:
            continue
        a_refined, _ = subgradient_descent(a_pert, t_rem, verbose=False, label=f"restart_{r}")
        sc = compute_score(a_refined)
        if sc < best_score:
            best_score = sc
            best_a = a_refined.copy()
            print(f"    Restart {r}: improved to {sc:.7f} (scale={noise_scale})")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def try_length(a_source, target_n, time_limit, verbose=True):
    """Interpolate to target_n and optimize."""
    a_init = interpolate_sequence(a_source, target_n)
    best_a = a_init.copy()
    best_score = compute_score(a_init)

    # Subgradient
    rem = time_limit - 0.5
    if rem > 10:
        t_alloc = rem * 0.5
        a_sg, _ = subgradient_descent(a_init, t_alloc, verbose=False, label=f"n{target_n}_SG")
        sc = compute_score(a_sg)
        if sc < best_score:
            best_score = sc
            best_a = a_sg.copy()

    # Pairwise CD
    rem = time_limit - (time_limit - t_alloc) - 0.5
    if rem > 5:
        a_pc, _ = pairwise_cd(best_a, rem, top_k=5, verbose=False, label=f"n{target_n}_PCD")
        sc = compute_score(a_pc)
        if sc < best_score:
            best_score = sc
            best_a = a_pc.copy()

    # L-BFGS
    rem = time_limit - (time_limit - t_alloc) - rem - 0.5
    if rem > 5:
        a_lb, _ = multi_scale_lbfgs(best_a, rem, verbose=False, label=f"n{target_n}_LB")
        sc = compute_score(a_lb)
        if sc < best_score:
            best_score = sc
            best_a = a_lb.copy()

    if verbose:
        print(f"  n={target_n}: best={best_score:.7f}")

    return best_a, best_score


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and update best if improved."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    
    Phases:
    1. Subgradient descent from parent (monotone decrease guaranteed)
    2. 2D pairwise coordinate descent on peak-contributing pairs
    3. L-BFGS with adaptive temperature (log-sum-exp smoothing)
    4. Random restarts to escape local minima
    5. Multi-length exploration
    6. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        # Fallback: create a reasonable initial sequence
        n_init = 512
        x = np.linspace(0, 1, n_init)
        parent = np.sin(np.pi * x) ** 0.5
        parent = parent / parent.sum()
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Generated initial: n={n_init}, score={best_score:.7f}")

    n_best = len(best_seq)

    # === Phase 1: Subgradient Descent ===
    print(f"\n=== Phase 1: Subgradient Descent (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 900
    if rem > 30:
        t_alloc = min(rem * 0.5, 300)
        print(f"  Budget: {t_alloc:.0f}s, n={n_best}")
        a_sg, _ = subgradient_descent(best_seq, t_alloc, verbose=True, label="SGD1")
        best_seq, best_score = check_and_update(a_sg, "SGD1", best_seq, best_score)

    # === Phase 2: Pairwise Coordinate Descent ===
    print(f"\n=== Phase 2: Pairwise CD (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 800
    if rem > 30:
        t_alloc = min(rem * 0.5, 300)
        print(f"  Budget: {t_alloc:.0f}s, n={len(best_seq)}")
        a_pc, _ = pairwise_cd(best_seq, t_alloc, top_k=10, verbose=True, label="PCD1")
        best_seq, best_score = check_and_update(a_pc, "PCD1", best_seq, best_score)

    # === Phase 3: L-BFGS with multiple temperatures ===
    print(f"\n=== Phase 3: L-BFGS Multi-T (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_alloc = min(rem * 0.6, 250)
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb, lb_score = multi_scale_lbfgs(best_seq, t_alloc, verbose=True, label="LBFGS")
        best_seq, best_score = check_and_update(a_lb, "LBFGS", best_seq, best_score)

    # === Phase 4: Second round of SGD + PCD from current best ===
    print(f"\n=== Phase 4: Refinement Round 2 (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 30:
        t_alloc_sg = min(rem * 0.3, 120)
        if t_alloc_sg > 10:
            a_sg2, _ = subgradient_descent(best_seq, t_alloc_sg, verbose=False, label="SGD2")
            best_seq, best_score = check_and_update(a_sg2, "SGD2", best_seq, best_score)

        rem = deadline - time.monotonic() - 400
        if rem > 20:
            t_alloc_pc = min(rem * 0.4, 150)
            a_pc2, _ = pairwise_cd(best_seq, t_alloc_pc, top_k=15, verbose=False, label="PCD2")
            best_seq, best_score = check_and_update(a_pc2, "PCD2", best_seq, best_score)

    # === Phase 5: Low-T L-BFGS for fine polishing ===
    print(f"\n=== Phase 5: Low-T L-BFGS (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 20:
        n_cur = len(best_seq)
        for T in [5, 2, 1]:
            rem = deadline - time.monotonic() - 300
            if rem < 10:
                break
            t_alloc = min(rem * 0.5, 80)
            a_lb2 = lbfgs_smoothed(best_seq, t_alloc, T=T, maxiter=3000, verbose=False, label=f"LB_T{T}")
            best_seq, best_score = check_and_update(a_lb2, f"LB_T{T}", best_seq, best_score)

    # === Phase 6: Random restarts ===
    print(f"\n=== Phase 6: Random Restarts (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t_alloc = min(rem * 0.7, 150)
        n_starts = max(2, int(t_alloc // 25))
        a_rr, rr_score = random_restart_local(best_seq, best_score, n_restarts=n_starts,
                                               time_per=min(25, t_alloc // n_starts))
        best_seq, best_score = check_and_update(a_rr, "restart", best_seq, best_score)

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length (best={best_score:.7f}) ===")
    n_best = len(best_seq)
    alt_lengths = []
    for factor in [1.25, 1.5, 0.8, 2.0, 0.75]:
        alt_n = int(n_best * factor)
        if 128 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
            alt_lengths.append(alt_n)
    alt_lengths.sort(key=lambda x: abs(x - n_best))

    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic() - 100
        if rem < 40:
            break
        t_alloc = min(rem * 0.5, 100)
        print(f"  Trying n={alt_n} (budget={t_alloc:.0f}s)")
        a_alt, alt_score = try_length(best_seq, alt_n, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)

    # === Phase 8: Final intense polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 15
    if rem > 15:
        # Quick SGD
        t1 = min(rem * 0.3, 40)
        if t1 > 5:
            a_f1, _ = subgradient_descent(best_seq, t1, verbose=False, label="FinSGD")
            best_seq, best_score = check_and_update(a_f1, "FinSGD", best_seq, best_score)

    rem = deadline - time.monotonic() - 10
    if rem > 10:
        # Quick PCD
        t2 = min(rem * 0.4, 30)
        if t2 > 5:
            a_f2, _ = pairwise_cd(best_seq, t2, top_k=20, verbose=False, label="FinPCD")
            best_seq, best_score = check_and_update(a_f2, "FinPCD", best_seq, best_score)

    rem = deadline - time.monotonic() - 10
    if rem > 10:
        # Final L-BFGS at very low T
        t3 = min(rem * 0.6, 20)
        if t3 > 5:
            a_f3 = lbfgs_smoothed(best_seq, t3, T=0.5, maxiter=5000, verbose=False, label="FinLB")
            best_seq, best_score = check_and_update(a_f3, "FinLB", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_seq = np.maximum(best_seq, 0.0)
    fs = float(evaluate_sequence(final_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
        best_seq = final_seq

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
