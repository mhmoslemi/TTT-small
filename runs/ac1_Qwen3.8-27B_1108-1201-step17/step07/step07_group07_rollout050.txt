import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_with(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def objective_normalized(a):
    """Compute 2n*max(conv(a,a)) for normalized a (sum=1)."""
    b = fast_convolve(a)
    return float(b.max())


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_minmax_step(a, max_step=0.3, verbose=False):
    """
    Solve min-max LP: find delta minimizing t s.t.
      2*conv(a, delta)[k] + conv(a,a)[k] <= t  for all k
      delta >= -a (so a+delta >= 0)
      sum(delta) = 0
      |delta| <= max_step (to keep steps bounded)
    
    Returns (delta, t_opt) or (None, None) on failure.
    """
    n = len(a)
    b_aa = fast_convolve(a)  # conv(a, a), length 2n-1
    
    # Variables: x[0..n-1] = delta, x[n] = t
    # Total: n+1 variables
    n_vars = n + 1
    
    # Objective: minimize t = x[n]
    c = np.zeros(n_vars)
    c[n] = 1.0
    
    # Constraints: 2*conv(a, delta)[k] <= t - b_aa[k]
    # i.e., -sum_i 2*a_{k-i} * delta_i + t >= b_aa[k]
    # i.e., sum_i 2*a_{k-i} * delta_i - t <= -b_aa[k]
    
    num_conv = 2 * n - 1
    A_ub = np.zeros((num_conv, n_vars))
    b_ub = np.full(num_conv, 0.0)
    
    for k in range(num_conv):
        A_ub[k, n] = -1.0  # coefficient of t
        b_ub[k] = -b_aa[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            A_ub[k, k - i] = 2.0 * a[i]
    
    # Constraint: sum(delta) = 0 -> handled as equality
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -a[i] <= delta[i] <= max_step, t >= 0
    bounds = []
    for i in range(n):
        bounds.append((-a[i], max_step))
    bounds.append((0.0, None))  # t >= 0
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            return None, None
        delta = res.x[:n]
        t_opt = res.x[n]
        return delta, t_opt
    except Exception:
        return None, None

def coordinate_descent(a0, time_limit, multipliers=None):
    """
    Coordinate descent on the simplex.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()

        multipliers = np.array([0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0])
    
    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 20.0:
                    continue
                
                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = objective_normalized(a_tr)
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True
    
    return best_a, best_obj


def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using iterative first-order min-max LP
(linearization of the quadratic convolution peak), FFT-accelerated subgradient
descent, coordinate descent polish, and multi-length exploration.

Key innovation over previous approach: A direct min-max LP that computes the
optimal perturbation direction to reduce the peak auto-convolution using
first-order approximation. This is more principled than subgradient methods
and converges faster to the local optimum.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def objective_normalized(a):
    """Compute 2n*max(conv(a,a)) for normalized a (sum=1)."""
    b = fast_convolve(a)
    return float(b.max())


def eval_score(a):
    """Compute the full score for a normalized sequence."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * float(b.max())


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_minmax_step(a, max_step=0.3, top_k=None):
    """
    Solve min-max LP: find delta minimizing t s.t.
      2*conv(a, delta)[k] + conv(a,a)[k] <= t  for all k (or top-k)
      delta >= -a (so a+delta >= 0)
      |delta|_inf <= max_step
      sum(delta) = 0

    Returns (delta, t_opt) or (None, None) on failure.
    """
    n = len(a)
    b_aa = fast_convolve(a)  # conv(a, a), length 2n-1
    num_conv = 2 * n - 1

    # Determine which constraints to include
    if top_k is not None and top_k < num_conv:
        top_idx = np.argpartition(b_aa, -top_k)[-top_k:]
        # Sort by value descending
        top_idx = top_idx[np.argsort(b_aa[top_idx])[::-1]]
    else:
        top_idx = np.arange(num_conv)

    n_vars = n + 1  # delta[0..n-1], t
    c = np.zeros(n_vars)
    c[n] = 1.0  # minimize t

    A_ub = np.zeros((len(top_idx), n_vars))
    b_ub = np.full(len(top_idx), 0.0)

    for row, k in enumerate(top_idx):
        A_ub[row, n] = -1.0
        b_ub[row] = -b_aa[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            A_ub[row, k - i] = 2.0 * a[i]

    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = [(-a[i], max_step) for i in range(n)]
    bounds.append((0.0, None))

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            return None, None
        delta = res.x[:n]
        t_opt = res.x[n]
        return delta, t_opt
    except Exception:
        return None, None


def lp_iterate(a, time_limit, max_iters=500, verbose=False):
    """
    Iteratively apply LP min-max steps to reduce max convolution.
    Returns (best_a, best_obj) where best_a is normalized to sum=1.
    """

    a_cur = a_cur / a_cur.sum()
    best_a = a_cur.copy()
    best_obj = objective_normalized(a_cur)

    step_size = 0.5
    no_improve = 0
    it_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Determine top_k: use all constraints for small n, top peaks for large n
        if n > 512:
            top_k = max(200, n)
        else:
            top_k = None  # use all

        delta, t_opt = lp_minmax_step(a_cur, max_step=step_size, top_k=top_k)
        if delta is None:
            step_size *= 0.5
            if step_size < 1e-7:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_trial_obj = best_obj
        alphas = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03]
        for alpha in alphas:
            trial = a_cur + alpha * delta
            trial = np.maximum(trial, 0.0)

            if s < 1e-10:

            trial = trial / s
            obj = objective_normalized(trial)
            if obj < best_trial_obj - 1e-14:
                best_trial_obj = obj
                best_alpha = alpha

        if best_alpha > 0:
            a_cur = a_cur + best_alpha * delta
            a_cur = np.maximum(a_cur, 0.0)
            a_cur = a_cur / a_cur.sum()
            cur_obj = objective_normalized(a_cur)

            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a_cur.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP iter {it}: obj={best_obj:.10f}, alpha={best_alpha:.2f}, "

                no_improve += 1
        else:
            no_improve += 1

            step_size *= 0.6
            no_improve = 0
            if step_size < 1e-7:
                break
        elif no_improve >= 5:
            step_size *= 0.85

        it_count += 1

    if verbose:
        print(f"    LP finished: {it_count} iters, best_obj={best_obj:.10f}")

def subgradient_fft(a0, time_limit, lr_init=0.004, momentum=0.92,
                    n_top=None, seed=42):
    """
    FFT-accelerated projected subgradient with Nesterov momentum.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = objective_normalized(a)
    cur_lr = lr_init
    no_improve = 0

    if n_top is None:
        n_top = max(15, n // 10)

    fft_size = 1 << (3 * n - 2).bit_length()

    while time.monotonic() - t0 < time_limit:
        A = rfft(a, fft_size)
        b_full = irfft(A * A, fft_size)
        b = b_full[:2 * n - 1]
        b_max = b.max()

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))

        top_vals = b[top_idx]
        threshold = b_max * 0.98
        active_mask = top_vals >= threshold
        if not active_mask.any():
            active_mask = np.zeros(len(top_idx), dtype=bool)
            active_mask[0] = True

        w = np.zeros(2 * n - 1)
        active_indices = top_idx[active_mask]
        weights = b[active_indices]
        w[active_indices] = weights / weights.sum()

        W = rfft(w, fft_size)
        g = irfft(np.conj(A) * W, fft_size)[:n]

        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)

        obj_new = objective_normalized(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve >= 3000:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.4, lr_init * 0.03)
            no_improve = 0
        elif no_improve >= 1500:
            cur_lr *= 0.93
        elif no_improve >= 500:
            cur_lr *= 0.97

    return best_a, best_obj


def coordinate_descent(a0, time_limit, multipliers=None):
    """Coordinate descent on the simplex."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = objective_normalized(a)

    if multipliers is None:
        multipliers = np.array([0.4, 0.6, 0.8, 0.9, 1.1, 1.25, 1.5, 2.0, 3.0])

    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 20.0:
                    continue

                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = objective_normalized(a_tr)
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True

    return best_a, best_obj


def make_gaussian_init(n, sigma):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def make_kaiser_init(n, beta=5.0):
    """Kaiser window initialization."""
    from scipy.special import i0
    n1 = n - 1
    if n1 == 0:
        return np.ones(1)
    alpha = np.linspace(-1.0, 1.0, n)
    a = i0(beta * np.sqrt(np.maximum(1 - alpha * alpha, 0))) / i0(beta)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = make_uniform_init(1152)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1152, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Focused LP on parent (primary method) ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Focused LP on parent (n={n_parent})")
    print(f"{'='*60}")
    t_lp1 = min(500.0, (deadline - time.monotonic()) * 0.35)
    print(f"  Time allocated: {t_lp1:.0f}s")
    a_lp, obj_lp = lp_iterate_focused(parent, time_limit=t_lp1, max_iters=500, verbose=True)
    update_best(a_lp, "lp1_focused")

    # === Phase 2: Full LP (all constraints) ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Full LP (best={best_score:.8f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t_lp2 = min(rem * 0.4, 250.0)
            print(f"  Time: {t_lp2:.0f}s, n={len(best_seq)}")
            a_lp2, obj_lp2 = lp_iterate(best_seq, time_limit=t_lp2, max_iters=200, verbose=True)
            update_best(a_lp2, "lp2_full")

    # === Phase 3: Subgradient ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Subgradient (best={best_score:.8f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_sg = min(rem * 0.4, 200.0)
            n_sg = len(best_seq)
            print(f"  SG: n={n_sg}, t={t_sg:.0f}s")
            a_sg, obj_sg = subgradient_fft(
                best_seq, time_limit=t_sg,
                lr_init=0.003, momentum=0.93,
                n_top=max(20, n_sg // 8), seed=42
            )
            update_best(a_sg, "sg")

    # === Phase 4: Another LP round from best ===
    print(f"\n{'='*60}")
    print(f"Phase 4: LP round 2 (best={best_score:.8f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 20:
            t_lp3 = min(rem * 0.5, 250.0)
            print(f"  LP: n={len(best_seq)}, t={t_lp3:.0f}s")
            a_lp3, obj_lp3 = lp_iterate_focused(best_seq, time_limit=t_lp3,
                                                 max_iters=300, verbose=True)
            update_best(a_lp3, "lp3")

    # === Phase 5: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-length (best={best_score:.8f})")
    print(f"{'='*60}")
    if best_seq is not None:
        target_lengths = []
        for factor in [0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 0.5, 1.1, 0.9, 1.4]:
            tl = int(n_parent * factor)
            tl = max(256, min(4096, tl))
            if tl not in target_lengths and tl != len(best_seq):
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - len(best_seq)))

        for target_n in target_lengths[:5]:
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            t_alloc = min(rem * 0.2, 100.0)
            if t_alloc < 15:
                continue

            f_interp = interpolate_sequence(best_seq, target_n)
            print(f"  n={target_n}, t={t_alloc:.0f}s")

            # LP first, then SG
            a_m, obj_m = lp_iterate_focused(f_interp, time_limit=t_alloc * 0.6,
                                             max_iters=80)
            rem2 = t_alloc * 0.4
            if rem2 > 10:
                a_m2, obj_m2 = subgradient_fft(a_m, time_limit=rem2, lr_init=0.003,
                                               momentum=0.9, n_top=max(10, target_n // 12), seed=target_n)
                if obj_m2 < obj_m:
                    a_m, obj_m = a_m2, obj_m2

            # Quick LP polish
            rem3 = deadline - time.monotonic() - 180
            if rem3 > 10 and rem3 < t_alloc:
                a_m3, obj_m3 = lp_iterate_focused(a_m, time_limit=min(rem3, 20.0), max_iters=30)
                if obj_m3 < obj_m:
                    a_m, obj_m = a_m3, obj_m3

            update_best(a_m, f"len_{target_n}")

    # === Phase 6: Coordinate descent polish ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Coordinate Descent (best={best_score:.8f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t_cd = min(rem * 0.5, 60.0)
            n_cd = len(best_seq)
            print(f"  CD: n={n_cd}, t={t_cd:.0f}s")
            a_cd, obj_cd = coordinate_descent(best_seq, time_limit=t_cd)
            update_best(a_cd, "cd")

    # === Phase 7: Final LP polish ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Final LP (best={best_score:.8f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_final = min(rem, 50.0)
            print(f"  Final LP: n={len(best_seq)}, t={t_final:.0f}s")
            a_final, obj_final = lp_iterate_focused(best_seq, time_limit=t_final,
                                                     max_iters=100, verbose=False)
            update_best(a_final, "final_lp")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
