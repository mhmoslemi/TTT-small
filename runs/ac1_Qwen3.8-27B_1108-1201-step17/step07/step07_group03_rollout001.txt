"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed projected gradient with momentum and temperature annealing
2. FFT-based gradient computation for the log-sum-exp surrogate
3. Coordinate descent polishing
4. Multi-start with diverse initializations and lengths
5. Adaptive restarts when progress stalls

import numpy as np
from scipy.fft import rfft, irfft
from scipy.signal import correlate
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def smooth_objective_and_grad(a, T):
    """
    Compute smoothed objective F_T(a) = (1/T)*log(sum_k exp(T*c_k))
    and its gradient, where c_k = (a*a)_k.
    
    Returns (F_val, grad) where grad has same shape as a.
    """
    n = len(a)
    c = fast_convolve(a)  # length 2n-1
    
    # Numerically stable log-sum-exp
    c_max = c.max()
    shifted = T * (c - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    
    if sum_exp < 1e-300:
        return 0.0, np.zeros(n)
    
    F_val = (c_max + np.log(sum_exp)) / T
    
    # Softmax weights
    w = exp_shifted / sum_exp
    
    # Gradient: grad[m] = 2 * sum_{j=0}^{n-1} w[m+j] * a[j]
    # This is 2 * correlate(w, a, mode='valid')
    # w has length 2n-1, a has length n
    # correlate gives length (2n-1) - n + 1 = n
    grad_half = correlate(w, a, mode='valid')
    grad = 2.0 * grad_half
    
    return F_val, grad


def smoothed_proj_grad_descent(a0, T, time_limit, lr_init=0.01, momentum=0.9,
                                backtracking=True, verbose=False):
    """
    Projected gradient descent on the smoothed objective with momentum.
    a is maintained on the simplex (sum=1, a>=0).

        a = a / s

    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)  # momentum
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    patience = 200
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        F_val, grad = smooth_objective_and_grad(a, T)
        
        # Momentum
        v = momentum * v + grad
        
        # Gradient step
        a_new = a - lr * v
        
        # Project onto simplex
        a_new = simplex_project(a_new)

def init_symmetric(n, rng):
    """Generate a symmetric random initialization (a[i] = a[n-1-i])."""
    half_n = (n + 1) // 2

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def multi_temperature_lp_polish(a0, time_limit):
    """
    Alternating approach: use the LP best-response idea but with damping.
    Given a, find b that minimizes max_k (a*b)_k via bisection + LP.
    Then update a = (1-t)*a + t*b.
    """
    from scipy.optimize import linprog
    
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Find best response b: minimize M s.t. (a*b)_k <= M, sum(b)=1, b>=0
        # This is: min M s.t. T @ b <= M*1, sum(b)=1, b>=0
        # Variables: [b_0, ..., b_{n-1}, M] (n+1 variables)
        # Constraints: T @ b - M <= 0 (2n-1 constraints), sum(b)=1
        
        # Build constraint matrix
        # (2n-1) + 1 = 2n inequalities (the last being -M <= 0 i.e. M >= 0)
        # Actually let's use: A_ub @ x <= b_ub
        # x = [b_0,...,b_{n-1}, M]
        # For each k: sum_j T[k,j]*b_j - M <= 0
        # Plus: -M <= 0 (M >= 0)
        
        # T is (2n-1) x n where T[k,j] gives (a*b)_k coefficient of b_j
        # (a*b)_k = sum_j a[k-j]*b_j (with appropriate bounds)
        # T[k,j] = a[k-j] if 0<=k-j<n else 0
        
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = a
        
        # A_ub: (2n) x (n+1)
        A_ub = np.zeros((2 * n, n + 1))
        A_ub[:2*n-1, :n] = T
        A_ub[:2*n-1, n] = -1.0  # -M
        A_ub[2*n-1, n] = -1.0   # -M <= 0
        b_ub = np.zeros(2 * n)
        
        # Equality: sum(b) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        # Objective: minimize M
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        bounds = [(0, None)] * n + [(0, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')

                b_new = np.maximum(b_new, 0)

                if bs > 1e-10:

                else:
                    b_new = a.copy()
            else:
                b_new = a.copy()
        except Exception:
            b_new = a.copy()
        
        # Damped update
        alpha = min(1.0, 2.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:

            best_a = a_new.copy()

    # === Phase 5: Second round of annealed PGD on best ===
    print(f"\n=== Phase 5: Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_ref = min(rem * 0.5, 80.0)
            temps_refine = [0.1, 0.05, 0.02, 0.01]
            a_ref, s_ref = annealed_optimization(best_seq, t_ref, temperatures=temps_refine)
            best_seq, best_score = check_and_update(a_ref, best_seq, best_score, "refine_pgd")
            
            # Quick CD
            rem2 = deadline - time.monotonic() - 60
            if rem2 > 10:
                a_cd2, s_cd2 = coordinate_descent_polish(best_seq, min(rem2*0.5, 30), n_sweeps=200)
                best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "cd_refine")
    
    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_fin, s_fin = coordinate_descent_polish(best_seq, min(rem * 0.5, 40), n_sweeps=500)
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_cd")
        
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_fin2, s_fin2 = peak_shave(best_seq, min(rem * 0.6, 20))
            best_seq, best_score = check_and_update(a_fin2, best_seq, best_score, "final_ps")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Shouldn't happen, but just in case
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
