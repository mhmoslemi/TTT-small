    # d/da[2] = a[0] + a[0] = 2*a[0]. Formula: 2*a[2-2] = 2*a[0]. Correct.
    # 
    # Great! So the gradient is:
    # grad_a[i] = sum_k w[k] * 2*a[k-i] (where a[m]=0 for m not in [0,n))
    # = 2 * (w * a)_reversed[i]... this is a cross-correlation.
    # Specifically: grad_a[i] = 2 * sum_k w[k] * a[k-i] where we sum over valid k.
    # This is the cross-correlation of w and a, evaluated at lag i.
    
    # The cross-correlation (w ⋆ a)[i] = sum_k w[k] * a[k-i]
    # Using FFT: this is irfft(conj(FFT(w)) * FFT(a))
    # But we need the linear (not circular) version.
    
    # Actually, let me just use the formula directly with the FFT trick.
    # cross_corr[i] = sum_k w[k] * a[k-i] for k where 0<=k-i<n
    # = sum_{j=0}^{n-1} w[j+i] * a[j] (substituting j=k-i)
    # This is a cross-correlation that can be computed via FFT.
    
    # For linear cross-correlation: irfft(conj(RFFT(w)) * RFFT(a)) gives circular.
    # For linear, we need zero-padding.
    
    # Actually, the standard trick: cross_corr[i] = sum_j a[j] * w[j+i]
    # This equals the convolution of a with reversed w, shifted.
    # Or: it's the "full" cross-correlation.
    
    # Let me just use: cross_corr = irfft(conj(RFFT(w, size)) * RFFT(a, size))[:n]
    # where size is large enough for linear ops.
    
    # Actually for auto-correlation of a: (a⋆a)[i] = sum_j a[j]*a[j+i] = sum_j a[j]*a[j-i] (by symmetry for real)
    # And this equals conv[2*(n-1)-i] or conv[n-1+i]... 
    # conv[n-1+i] = sum_j a[j]*a[n-1+i-j]. Let m = n-1-j: = sum_m a[n-1-m]*a[m+i]
    # Hmm this is getting confusing. Let me just use the direct FFT approach.
    
    # The simplest correct approach: 
    # grad_a[i] = 2 * sum_{k=max(0,i)}^{min(n-1, 2n-2-i)} w[k] * a[k-i]
    # But with the FFT zero-padding trick:
    # Let A = rfft(a, fft_size), W = rfft(w, fft_size) with fft_size >= 3n-2
    # Then irfft(conj(W) * A, fft_size)[:n] gives sum_k w[k]*a[k-i] for i=0,...,n-1
    # (this is the cross-correlation w ⋆ a at lags 0 to n-1)
    
    # Let me verify: (w ⋆ a)[i] = sum_k w[k] * conj(a[k-i]) = sum_k w[k] * a[k-i] (real case)
    # With FFT: IFFT(conj(FFT(w)) * FFT(a)) = w ⋆ a (circular). With enough zero-padding, it's linear.
    
    # So: grad_a = 2 * n * T * 2 * (w ⋆ a)[:n] = 4*n*T*(w⋆a)[:n]
    # Wait, let me re-derive. obj = 2*n*T * (max_s/T + log(mean(exp((s-max_s)/T))))
    # where s = conv(a,a).
    # d obj / d a[i] = 2*n * sum_k w[k] * d conv[k]/da[i] = 2*n * sum_k w[k] * 2*a[k-i] (valid)
    # = 4*n * (w ⋆ a)[i]
    
    # Then the gradient in z-space: grad_z = a * (grad_a - dot(a, grad_a))
    
    grad_a = 4.0 * n * correl
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)
    return grad_z


def lbgfs_optimize(a_init, time_limit, T, maxiter=5000, verbose=False):
    """L-BFGS optimization in log-space with given temperature."""
    n = len(a_init)
    a = normalize(a_init)
    z0 = np.log(np.maximum(a, 1e-300))
    z0 = z0 - z0.mean()
    t0 = time.monotonic()
    call_count = [0]

    def func(z):
        call_count[0] += 1
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return log_space_obj(z, T, n)

    def jac(z):
        call_count[0] += 1
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        return log_space_grad(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 40, 'disp': False}
        )
        z_opt = res.x
        a_out = softmax_from_z(z_opt)
    except Exception:
        a_out = a_init.copy()

    if verbose:
        print(f"      L-BFGS T={T}: {call_count[0]} calls, score={score_norm(a_out):.7f}")
    return normalize(a_out)


def peak_shave_iterative(a_init, time_limit, max_iters=100000):
    """Iterative peak shaving: reduce the top contributing pair to the active peak."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()

        # Find the peak (prefer non-endpoint peaks)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b[peak_k]

        if peak_val <= best_score / (2.0 * n) * 1.0001:
            break  # At or near the optimum

        # Find contributing pairs
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing the top contributor
        top_idx = np.argsort(contribs)[::-1][:10]
        improved = False

        for ci in top_idx:
            i = int(idx[ci])
            j = peak_k - i
            if a[i] < 1e-12:
                continue
            # Try reducing a[i] by various amounts, redistributing to a[j] or others
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7, 0.6]:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * factor
                diff = old_val - a_trial[i]
                if diff < 1e-15:
                    continue
                # Redistribute proportionally to others
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_norm(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        # Also try the symmetric: reduce a[j] instead
        if not improved:
            for ci in top_idx[:5]:
                i = int(idx[ci])
                j = peak_k - i
                if i == j or a[j] < 1e-12:
                    continue
                for factor in [0.95, 0.9, 0.85, 0.8]:
                    a_trial = a.copy()
                    old_val = a_trial[j]
                    a_trial[j] = old_val * factor
                    diff = old_val - a_trial[j]
                    if diff < 1e-15:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = score_norm(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Stuck - try a small random perturbation
            if it > 100:
                rng = np.random.default_rng(it)
                noise = rng.standard_normal(n) * 0.01
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
            else:
                break

    return best_a, best_score


def coordinate_descent_v2(a_init, time_limit, max_no_improve=500):
    """Coordinate descent: for each coordinate, find optimal 1D adjustment."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    no_improve = 0

    factors = [0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5, 2.0]

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve = 0
        improved_any = False
        # Random order for diversity
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for f in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                new_val = old_val * f
                a_trial[i] = new_val
                diff = new_val - old_val
                if abs(diff) < 1e-15:
                    continue
                # Redistribute to others proportionally
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others_sum
                if a_trial.min() < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_norm(a_trial)
                if sc < best_score - 1e-13:
                    best_score = sc
                    a = a_trial
                    best_a = a.copy()
                    improved_any = True
                    break

        if not improved_any:
            no_improve += 1

    return best_a, best_score


def generate_chirp_init(n, seed=0):
    """Generate a chirp-based initialization with good autocorrelation properties."""
    rng = np.random.default_rng(seed)
    # Real chirp: cos(pi * j^2 / n) gives good autocorrelation
    j = np.arange(n)
    # Multiple chirp frequencies for diversity
    freq = rng.choice([1, 2, 3, 4])
    base = np.cos(np.pi * (j * j) / (n * freq))
    # Add small random perturbation
    noise = rng.standard_normal(n) * 0.1
    a = base + noise
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def generate_smooth_random(n, seed=0):
    """Generate a smooth random initialization."""
    rng = np.random.default_rng(seed)
    # Generate in frequency domain for smoothness
    freq = rng.standard_normal(n // 2 + 1)
    freq[:n // 10] *= 0.1  # Suppress high frequencies
    a = np.zeros(n)
    # IFFT to get smooth signal
    from scipy.fft import irfft as _irfft
    a = _irfft(freq, n)
    a = np.abs(a) + 0.1
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate using actual evaluate_sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def multi_temp_lbgfs(a_init, time_limit, label=""):
    """Run L-BFGS with decreasing temperatures."""
    n = len(a_init)
    current = a_init.copy()
    t0 = time.monotonic()
    
    # Temperature schedule: start high, end very low
    temps = [1000, 500, 200, 100, 50, 20, 10, 5, 2, 1, 0.5, 0.2]
    
    for T in temps:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        # Allocate more time to mid-range temperatures
        if T > 200:
            t_alloc = min(remaining * 0.15, 30)
            maxiter = 2000
        elif T > 20:
            t_alloc = min(remaining * 0.15, 50)
            maxiter = 3000
        elif T > 2:
            t_alloc = min(remaining * 0.2, 60)
            maxiter = 4000
        else:
            t_alloc = min(remaining * 0.3, 80)
            maxiter = 5000
        
        if t_alloc < 3:
            break
            
        current = lbgfs_optimize(current, t_alloc, T, maxiter=maxiter)
        sc = score_norm(current)
        if sc < 10:  # Sanity check
            print(f"    {label} T={T}: score={sc:.7f} ({time.monotonic()-t0:.0f}s elapsed)")
    
    return current


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization with diverse initializations and aggressive
    multi-temperature L-BFGS followed by targeted peak reduction.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score

    # Determine base length
    if parent is not None:
        base_n = len(parent)
    else:
        base_n = 768
        best_seq = np.ones(base_n) / base_n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={base_n}, score={best_score:.7f}")

    # === Phase 1: Diverse initializations ===
    print(f"\n=== Phase 1: Initializations (n={base_n}) ===")
    candidates = [best_seq.copy()]
    
    # Chirp-based
    candidates.append(generate_chirp_init(base_n, seed=1))
    candidates.append(generate_chirp_init(base_n, seed=2))
    candidates.append(generate_chirp_init(base_n, seed=3))
    
    # Smooth random
    candidates.append(generate_smooth_random(base_n, seed=10))
    candidates.append(generate_smooth_random(base_n, seed=20))
    
    # Slightly perturbed parent
    for s in [42, 99]:
        rng_p = np.random.default_rng(s)
        perturbed = best_seq + rng_p.standard_normal(base_n) * 0.05
        candidates.append(normalize(perturbed))
    
    # Evaluate all candidates
    for i, cand in enumerate(candidates):
        sc = score_norm(cand)
        print(f"  Candidate {i}: score={sc:.7f}")
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    # === Phase 2: Multi-temperature L-BFGS on best candidate ===
    print(f"\n=== Phase 2: Multi-Temp L-BFGS (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 60:
        t_alloc = min(rem * 0.6, 500.0)
        print(f"  Main L-BFGS: budget={t_alloc:.0f}s, n={len(best_seq)}")
        a_opt = multi_temp_lbgfs(best_seq, t_alloc, label="main")
        best_seq, best_score = check_and_update(a_opt, "main_lbgfs", best_seq, best_score)

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_alloc = min(rem * 0.4, 150.0)
        print(f"  PS: budget={t_alloc:.0f}s")
        a_ps, _ = peak_shave_iterative(best_seq, t_alloc)
        best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 4: Second round L-BFGS from improved point ===
    print(f"\n=== Phase 4: L-BFGS Round 2 (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 30:
        t_alloc = min(rem * 0.5, 200.0)
        # Focus on lower temperatures
        n = len(best_seq)
        for T in [3, 1.5, 0.7, 0.3]:
            rem = deadline - time.monotonic() - 100
            if rem < 10:
                break
            t_a = min(rem * 0.25, 50.0)
            if t_a < 3:
                break
            a_r2 = lbgfs_optimize(best_seq, t_a, T, maxiter=4000)
            best_seq, best_score = check_and_update(a_r2, f"lbgfs_T{T}", best_seq, best_score)

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 80
    if rem > 20:
        t_alloc = min(rem * 0.4, 100.0)
        print(f"  CD: budget={t_alloc:.0f}s")
        a_cd, _ = coordinate_descent_v2(best_seq, t_alloc)
        best_seq, best_score = check_and_update(a_cd, "coord_descent", best_seq, best_score)

    # === Phase 6: Additional peak shaving ===
    print(f"\n=== Phase 6: Peak Shaving 2 (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 60
    if rem > 20:
        t_alloc = min(rem * 0.5, 80.0)
        a_ps2, _ = peak_shave_iterative(best_seq, t_alloc)
        best_seq, best_score = check_and_update(a_ps2, "peak_shave2", best_seq, best_score)

    # === Phase 7: Try alternative lengths ===
    print(f"\n=== Phase 7: Alternative Lengths (best={best_score:.7f}) ===")
    n_best = len(best_seq)
    alt_lengths = []
    for factor in [1.25, 1.5, 0.75, 2.0]:
        alt_n = int(n_best * factor)
        if 256 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:

    for alt_n in alt_lengths[:2]:
        rem = deadline - time.monotonic() - 60
        if rem < 60:

        t_alloc = min(rem * 0.35, 120.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt = multi_temp_lbgfs(init_alt, t_alloc, label=f"alt_{alt_n}")
        best_seq, best_score = check_and_update(a_alt, f"alt_{alt_n}", best_seq, best_score)

    # === Phase 8: Final polishing ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.7f}) ===")
    n_final = len(best_seq)
    
    # Very low temperature L-BFGS
    rem = deadline - time.monotonic() - 30
    if rem > 15:
        for T in [0.5, 0.1]:
            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break
            t_a = min(rem * 0.4, 30.0)
            a_fin = lbgfs_optimize(best_seq, t_a, T, maxiter=3000)
            best_seq, best_score = check_and_update(a_fin, f"final_T{T}", best_seq, best_score)

    # Final peak shave
    rem = deadline - time.monotonic() - 15
    if rem > 10:
        a_fin_ps, _ = peak_shave_iterative(best_seq, min(rem * 0.5, 20.0))
        best_seq, best_score = check_and_update(a_fin_ps, "final_ps", best_seq, best_score)

    # Final coordinate descent
    rem = deadline - time.monotonic() - 8
    if rem > 5:
        a_fin_cd, _ = coordinate_descent_v2(best_seq, min(rem * 0.6, 10.0), max_no_improve=200)
        best_seq, best_score = check_and_update(a_fin_cd, "final_cd", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(base_n) / base_n
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
