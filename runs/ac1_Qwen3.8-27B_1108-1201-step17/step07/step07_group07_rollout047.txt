with adaptive step sizes, FFT-accelerated subgradient descent, and coordinate
descent polish. Key improvement: many LP iterations with line search for
optimal blending, combined with smoothed gradient methods.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def eval_max_b(a):
    """Return max of auto-convolution for input array a."""
    b = fast_convolve(a)
    return float(b.max())

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_direction(f, n_top=None):
    """
    Compute subgradient of max(f*f) at point f.
    Returns (g, b_max) where g is the subgradient (length n) and b_max is the max.
    """
    n = len(f)
    b = fast_convolve(f)
    b_max = b.max()
    
    if n_top is None:
        n_top = max(10, n // 10)
    
    # Find peaks within 0.1% of max
    threshold = b_max * 0.999
    peak_indices = np.where(b >= threshold)[0]
    
    # Create weight vector for the active set
    w = np.zeros(2 * n - 1)
    w[peak_indices] = 1.0 / len(peak_indices)
    
    # Subgradient: g[i] = sum_k w[k] * f[k-i] (from the convolution structure)
    # (f*f)[k] = sum_i f[i] * f[k-i]
    # d(f*f)/df[i] |_k = f[k-i]
    # So g[i] = sum_k w[k] * f[k-i]
    # This is a cross-correlation of w and f
    # g[i] = sum_k w[k] * f[k-i] = (w * f^R)[i] where f^R is reversed
    # Actually: g[i] = sum_k w[k] * f[k-i]
    # Let j = k-i: g[i] = sum_j w[i+j] * f[j]
    # This is the cross-correlation of w and f, evaluated at lag i
    
    # Use FFT: g = ifft(conj(FFT(f)) * FFT(w))
    # Wait, let me be more careful.
    # g[i] = sum_j w[i+j] * f[j] for j = 0..n-1 and i+j = 0..2n-2
    # This is like: g = cross_correlation(w, f)
    # cross_correlation(w, f)[i] = sum_j w[i+j] * conj(f[j])
    # For real signals: sum_j w[i+j] * f[j]
    
    fft_size = 1 << (3 * n - 2).bit_length()
    F = rfft(f, fft_size)
    W = rfft(w, fft_size)
    # g[i] = sum_j w[i+j] * f[j]
    # = sum_j f[j] * w[i+j]
    # This is the "anti-convolution" or cross-correlation
    # = ifft(conj(F) * W) ... let me verify
    # ifft(conj(F) * W) = cross_correlation(F, W) in some convention
    # Let's just use: ifft(F * W_rev) where W_rev is w reversed and zero-padded
    
    # Actually, simplest: use the definition directly with FFT
    # g[i] = sum_j f[j] * w[i+j]
    # Pad w to length 3n-1: w_pad[0..2n-2] = w, rest = 0
    # Then g[i] = sum_j f[j] * w_pad[i+j]
    # This is the "cross-correlation" which can be computed as:
    # g = ifft(FFT(f) * conj(FFT(w_pad)))
    # Wait, cross-correlation r_xy[i] = sum_j x[j] * conj(y[j+i])
    # We want g[i] = sum_j f[j] * w[i+j]
    # Let y = w, x = f: r_fy[i] = sum_j f[j] * y[j+i] = sum_j f[j] * w[j+i]
    # But we want w[i+j] not w[j+i]. These are the same! (i+j = j+i)
    # So g = cross_correlation(f, w) = ifft(FFT(f) * conj(FFT(w)))
    # Hmm wait, let me double-check with a small example.
    # f = [a, b], w = [p, q, r] (length 3 = 2*2-1)
    # g[0] = f[0]*w[0] + f[1]*w[1] = a*p + b*q
    # g[1] = f[0]*w[1] + f[1]*w[2] = a*q + b*r
    # 
    # FFT approach: 
    # FFT(f) = [a+b, a-b] (length 2), padded to fft_size
    # FFT(w) = [p+q+r, ...] (length 3), padded to fft_size
    # ifft(FFT(f) * conj(FFT(w))):
    # This gives circular cross-correlation, which for our purposes (with sufficient padding)
    # gives the linear cross-correlation.
    
    # Actually, I think the cleanest way is:
    # g[i] = sum_j f[j] * w[i+j]
    # = (f * w_flipped)[sum] ... no
    # Let me just compute it directly for small n, or use a different FFT trick.
    
    # Alternative: g = ifft(conj(FFT(f)) * FFT(w)) gives:
    # [g]_i = sum_j conj(f_j) * w_{i-j} (circular)
    # That's not quite what we want.
    
    # Let me use: g[i] = sum_j f[j] * w[i+j]
    # Define v[k] = w[k] for k=0..2n-2, 0 otherwise. Then:
    # g[i] = sum_j f[j] * v[i+j]
    # This is the "cross-correlation" of f and v at lag i, but with a sign flip on the lag.
    # Standard cross-correlation: (f ⊛ v)[i] = sum_j f[j] * conj(v[j+i])
    # So g[i] = (f ⊛ v)[i] for real signals = sum_j f[j] * v[j+i]
    # And cross-correlation via FFT: ifft(conj(FFT(f)) * FFT(v))
    
    # Wait no. The standard definition:
    # (f ⊛ v)[i] = sum_j conj(f[j]) * v[j+i]
    # For real f: = sum_j f[j] * v[j+i]
    # FFT: ifft(conj(FFT(f)) * FFT(v))
    
    # So g = ifft(conj(FFT(f)) * FFT(w_padded))
    # where w_padded has length >= 3n-1 to avoid circular aliasing.
    
    f_pad = np.zeros(fft_size)
    f_pad[:n] = f
    w_pad = np.zeros(fft_size)
    w_pad[:2*n-1] = w
    
    F_full = rfft(f_pad, fft_size)  # rfft of length fft_size
    W_full = rfft(w_pad, fft_size)
    
    g = irfft(np.conj(F_full) * W_full, fft_size)[:n]
    
    return g, b_max

def lp_peak_reduction(f, time_limit, max_iters=500, verbose=True):
    """
    Aggressive LP-based peak reduction.
    f is normalized to sum=1.
    Returns (best_f, best_max_b).
    """
    t0 = time.monotonic()
    n = len(f)
    f_cur = f.copy().astype(np.float64)
    S = f_cur.sum()
    if S < 1e-15:
        f_cur = np.ones(n) / n
        S = 1.0
    else:
        f_cur = f_cur / S
    
    best_f = f_cur.copy()
    best_obj = eval_max_b(f_cur)
    
    t_candidates = [1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
    
    iter_count = 0
    no_improve_count = 0
    
    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current max
        b_cur = fast_convolve(f_cur)
        M = b_cur.max()
        
        if M < best_obj - 1e-12:
            best_obj = M
            best_f = f_cur.copy()
        
        # Build LP constraint matrix
        # A[k,j] = f_cur[j] if 0 <= k-j <= n-1 (i.e., A[k,j] = f_cur[k-j] when valid)
        A = np.zeros((2*n-1, n))
        for j in range(n):
            A[j:j+n, j] = f_cur
        
        b_lp = np.full(2*n-1, M)
        c = -np.ones(n)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_lp, 
                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                break
            
            g = res.x
            g_sum = g.sum()
            if g_sum <= 1.0 + 1e-10:
                # No improvement possible from this direction
                # Try a smaller M to force a different direction
                no_improve_count += 1
                if no_improve_count > 10:
                    # Perturb and restart
                    break
                continue
            
            g_norm = g / g_sum
            
            # Line search for best blend
            best_t = 0.0
            best_blend_obj = best_obj
            for t in t_candidates:
                blend = (1.0 - t) * best_f + t * g_norm
                blend = np.maximum(blend, 0.0)
                s = blend.sum()
                if s < 1e-10:
                    continue
                blend = blend / s
                obj_blend = eval_max_b(blend)
                if obj_blend < best_blend_obj - 1e-14:
                    best_blend_obj = obj_blend
                    best_t = t
            
            if best_t > 0.0:
                f_cur = (1.0 - best_t) * best_f + best_t * g_norm
                f_cur = np.maximum(f_cur, 0.0)
                f_cur = f_cur / f_cur.sum()
                obj_cur = eval_max_b(f_cur)
                
                if obj_cur < best_obj - 1e-14:
                    best_obj = obj_cur
                    best_f = f_cur.copy()
                    no_improve_count = 0
                    if verbose and iter_count % 20 == 0:
                        print(f"    LP iter {iter_count}: t={best_t:.2f}, max_b={best_obj:.10f}, g_sum={g_sum:.4f}")
                else:
                    f_cur = best_f.copy()
                    no_improve_count += 1
            else:
                no_improve_count += 1
                if no_improve_count > 20:
                    break
        except Exception as e:
            if verbose:
                print(f"    LP error at iter {iter_count}: {e}")
            break
        
        iter_count += 1
    
    return best_f, best_obj

def smoothed_gradient_descent(f0, time_limit, alpha_init=100.0, lr=0.01, verbose=False):
    """
    Gradient descent on the smoothed objective:
    F_alpha(f) = (1/alpha) * log(sum_k exp(alpha * b_k))
    where b_k = (f*f)_k and sum(f) = 1, f >= 0.
    """
    t0 = time.monotonic()
    n = len(f0)
    f = f0.copy().astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_obj = eval_max_b(f)
    
    alpha = alpha_init
    cur_lr = lr
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        # Softmax over b
        b_shifted = b - b.max()
        weights = np.exp(alpha * b_shifted)
        weights = weights / weights.sum()
        
        # Gradient of (1/alpha)*log(sum exp(alpha*b_k)) w.r.t. f
        # = sum_k weights[k] * db_k/df
        # db_k/df[i] = f[k-i] (for valid indices)
        # So grad[i] = sum_k weights[k] * f[k-i]
        # Same as subgradient direction but with soft weights
        
        fft_size = 1 << (3 * n - 2).bit_length()
        f_pad = np.zeros(fft_size)
        f_pad[:n] = f
        w_pad = np.zeros(fft_size)
        w_pad[:2*n-1] = weights
        
        F_full = rfft(f_pad, fft_size)
        W_full = rfft(w_pad, fft_size)
        grad = irfft(np.conj(F_full) * W_full, fft_size)[:n]
        
        # Normalize gradient for step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        grad = grad / grad_norm
        
        # Step
        f_new = f - cur_lr * grad
        f_new = project_simplex(f_new)
        
        obj_new = eval_max_b(f_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        it += 1
        
        # Decay
        if no_improve >= 100:
            cur_lr *= 0.5
            no_improve = 0
        elif no_improve >= 50:
            cur_lr *= 0.8
        if no_improve >= 500:
            break
        
        # Increase alpha gradually (sharpen the approximation to max)
        if it % 200 == 0:
            alpha *= 1.1
    
    return best_f, best_obj

def coordinate_descent_polish(f0, time_limit, verbose=False):
    """
    Coordinate descent: adjust one element at a time.
    """
    t0 = time.monotonic()
    n = len(f0)
    f = f0.copy().astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_obj = eval_max_b(f)
    
    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0])
    
    improved = True
    passes = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        passes += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            old_val = f[j]
            if old_val < 1e-14:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0 or scale > 50:
                    continue
                
                f_tr = f.copy()
                f_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_tr[mask] *= scale
                f_tr = f_tr / f_tr.sum()
                
                val = eval_max_b(f_tr)
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
    
    return best_f, best_obj

def subgradient_optimize(f0, time_limit, lr_init=0.003, momentum=0.93, n_top=None, seed=42):
    """
    FFT-accelerated projected subgradient with momentum.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(f0)
    f = f0.copy().astype(np.float64)
    f = f / f.sum()
    
    v = np.zeros(n)
    best_f = f.copy()
    best_obj = eval_max_b(f)
    cur_lr = lr_init
    no_improve = 0
    it = 0
    
    if n_top is None:
        n_top = max(15, n // 8)
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        b_max = b.max()
        
        # Find active peaks
        threshold = b_max * 0.998
        peak_idx = np.where(b >= threshold)[0]
        if len(peak_idx) < n_top:
            # Also include top n_top peaks
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            peak_idx = np.unique(np.concatenate([peak_idx, top_idx]))
        
        w = np.zeros(2*n-1)
        w[peak_idx] = 1.0 / len(peak_idx)
        
        # Compute subgradient
        fft_size = 1 << (3*n-2).bit_length()
        f_pad = np.zeros(fft_size)
        f_pad[:n] = f
        w_pad = np.zeros(fft_size)
        w_pad[:2*n-1] = w
        F_full = rfft(f_pad, fft_size)
        W_full = rfft(w_pad, fft_size)
        grad = irfft(np.conj(F_full) * W_full, fft_size)[:n]
        
        # Momentum step
        v = momentum * v - cur_lr * grad
        f_new = project_simplex(f + v)
        
        obj_new = eval_max_b(f_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        it += 1
        
        if no_improve >= 500:
            cur_lr *= 0.5
            v = np.zeros(n)
            no_improve = 0
            if cur_lr < lr_init * 0.01:
                # Restart
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                v = np.zeros(n)
                cur_lr = lr_init * 0.5
        elif no_improve >= 200:
            cur_lr *= 0.85
    
    return best_f, best_obj

def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_uniform_init(n):
    return np.ones(n) / n

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: aggressive LP peak reduction + smoothed gradient
    + subgradient + coordinate descent. Focuses on maximizing LP iterations
    with proper line search for optimal blending.
    """
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        parent = make_cosine_init(1152)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using cosine n=1152, score={best_score:.6f}")
    
    n_parent = len(parent)
    print(f"\n{'='*60}")
    print(f"Starting optimization (budget={budget_s:.0f}s, n={n_parent})")
    print(f"{'='*60}")
    
    # === Phase 1: Aggressive LP (main workhorse) ===
    print(f"\nPhase 1: Aggressive LP (n={n_parent})")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_lp = min(rem, 900.0)
        print(f"  Time allocated: {t_lp:.0f}s")
        a_lp, obj_lp = lp_peak_reduction(parent, time_limit=t_lp, max_iters=2000, verbose=True)
        update_best(a_lp, "lp_phase1")
    
    # === Phase 2: Subgradient from LP result ===
    print(f"\nPhase 2: Subgradient (best={best_score:.6f})")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_sg = min(rem, 200.0)
            print(f"  Time allocated: {t_sg:.0f}s")
            a_sg, obj_sg = subgradient_optimize(best_seq, time_limit=t_sg, 
                                                 lr_init=0.002, momentum=0.94, seed=seed)
            update_best(a_sg, "sg_phase2")
    
    # === Phase 3: More LP from subgradient result ===
    print(f"\nPhase 3: LP Round 2 (best={best_score:.6f})")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_lp2 = min(rem, 400.0)
            print(f"  Time allocated: {t_lp2:.0f}s")
            a_lp2, obj_lp2 = lp_peak_reduction(best_seq, time_limit=t_lp2, 
                                                max_iters=1000, verbose=True)
            update_best(a_lp2, "lp_phase3")
    
    # === Phase 4: Smoothed gradient ===
    print(f"\nPhase 4: Smoothed Gradient (best={best_score:.6f})")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_smooth = min(rem, 150.0)
            print(f"  Time allocated: {t_smooth:.0f}s")
            a_sm, obj_sm = smoothed_gradient_descent(best_seq, time_limit=t_smooth,
                                                      alpha_init=50.0, lr=0.005)
            update_best(a_sm, "smooth_phase4")
    
    # === Phase 5: Final LP ===
    print(f"\nPhase 5: Final LP (best={best_score:.6f})")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_lp3 = min(rem, 200.0)
            print(f"  Time allocated: {t_lp3:.0f}s")
            a_lp3, obj_lp3 = lp_peak_reduction(best_seq, time_limit=t_lp3,
                                                max_iters=500, verbose=True)
            update_best(a_lp3, "lp_phase5")
    
    # === Phase 6: Coordinate descent polish ===
    print(f"\nPhase 6: CD Polish (best={best_score:.6f})")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
