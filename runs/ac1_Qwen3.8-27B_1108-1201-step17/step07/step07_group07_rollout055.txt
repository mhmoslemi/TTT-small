def compute_gradient_power_sum(a, p, N=None):
    """
    Compute gradient of f_p(a) = (sum_i b_i^p)^(1/p) where b = conv(a,a).
    Returns (objective_value, gradient) with gradient having same shape as a.
    """
    n = len(a)
    if N is None:
        size = 2 * n - 1
        N = 1 << (size - 1).bit_length()
    
    # Compute b = conv(a, a)
    A = rfft(a, N)
    b_full = irfft(A * A, N)
    b = b_full[:2*n-1]
    
    # Compute power sum objective
    b_p = b ** p
    sum_bp = b_p.sum()
    if sum_bp < 1e-300:
        return 0.0, np.zeros(n)
    
    obj = sum_bp ** (1.0/p)
    
    # Gradient: d(obj)/d(a_k) = (1/p) * sum_bp^(1/p - 1) * sum_i p * b_i^(p-1) * a_{i-k}
    # = sum_bp^(1/p - 1) * sum_i b_i^(p-1) * a_{i-k}
    # The inner sum is cross-correlation of w = b^(p-1) (length 2n-1) with a (length n)
    
    w = b ** (p - 1)  # length 2n-1
    
    # Cross-correlation: g_k = sum_j w[k+j] * a_j for j=0..n-1

    A2 = rfft(a, N2)

    grad = (sum_bp ** (1.0/p - 1.0)) * g
    
    return obj, grad


def adam_step(params, grads, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step. params is a dict of name->array. Returns updated params."""
    for name in params:
        g = grads[name]
        if 'm' not in state[name]:
            state[name]['m'] = np.zeros_like(params[name])
            state[name]['v'] = np.zeros_like(params[name])
            state[name]['t'] = 0
        state[name]['t'] += 1
        t = state[name]['t']
        state[name]['m'] = beta1 * state[name]['m'] + (1 - beta1) * g
        state[name]['v'] = beta2 * state[name]['v'] + (1 - beta2) * g**2
        m_hat = state[name]['m'] / (1 - beta1**t)
        v_hat = state[name]['v'] / (1 - beta2**t)
        params[name] = params[name] - lr * m_hat / (np.sqrt(v_hat) + eps)
    return params

        a_candidate = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project to simplex
        a_candidate = project_simplex(a_candidate)

        # Evaluate
        max_b_new = fast_autoconv(a_candidate).max()

        if max_b_new < best_max_b - 1e-14:
            best_max_b = max_b_new
            best_a = a_candidate.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_candidate

        # Adaptive restart
        if no_improve >= 500:
            a = project_simplex(best_a + np.random.default_rng(seed + t).standard_normal(n) * kick)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = max(lr * 0.5, lr_init * 0.01)
            no_improve = 0
            kick *= 0.7

        # Learning rate decay
        if no_improve >= 200:
            lr *= 0.95
        if no_improve >= 350:
            lr *= 0.98

    return best_a, best_max_b


def lp_refine(f, max_iters=40, time_limit=60.0):
    """
    LP-based refinement: at each step solve max sum(g) s.t. conv(f,g)<=M, g>=0,
    blend with current f to reduce peak.
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = fast_autoconv(a).max()

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autoconv(a)
        M = b.max()

        # Build constraint matrix
        num_constraints = 2 * n - 1
        A_ub = np.zeros((num_constraints, n))
        for k in range(num_constraints):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A_ub[k, k - hi:k - lo + 1] = a[lo:hi + 1][::-1]

        b_ub = np.full(num_constraints, M)
        c = -np.ones(n)

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0.0, None)] * n, method='highs')
        except Exception:
            break

        if not res.success:
            break

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-10:
            break

        g_norm = g / g_sum

        # Line search
        best_t = 0.0
        best_blend_obj = best_obj
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            blend = (1.0 - t) * best_a + t * g_norm
            blend = np.maximum(blend, 0.0)
            s = blend.sum()
            if s < 1e-10:
                continue
            blend = blend / s
            obj = fast_autoconv(blend).max()
            if obj < best_blend_obj - 1e-14:
                best_blend_obj = obj
                best_t = t

        if best_t > 0:
            a = (1.0 - best_t) * best_a + best_t * g_norm
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            obj = fast_autoconv(a).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
            else:
                a = best_a.copy()
        else:
            break

    return best_a, best_obj


def dual_peak_equalize(a, n_iters=200, n_peaks=20, decay=0.95):
    """
    Iterative peak equalization using dual variables.
    At each step, compute softmax weights over top peaks,
    take a gradient step in the direction that reduces weighted peaks.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_autoconv(a).max()

    N_conv = 1 << (2 * n - 2).bit_length()
    N_grad = 1 << (3 * n - 2).bit_length()

    for it in range(n_iters):
        A = rfft(a, N_conv)
        b = irfft(A * A, N_conv)[:2 * n - 1]
        b_max = b.max()

        # Softmax weights over peaks (temperature-based)
        T = max(1.0, n_peaks / (b_max * 10))
        log_w = b / T
        log_w -= log_w.max()
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient: g_k = sum_i w_i * a_{i-k}
        A2 = rfft(a, N_grad)
        W2 = rfft(w, N_grad)
        g = irfft(np.conj(A2) * W2, N_grad)[:n]

        # Step: reduce peaks
        lr = 0.002 * decay ** (it / 50.0)
        a_candidate = project_simplex(a - lr * g)
        a = a_candidate

        obj = fast_autoconv(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit, multipliers=None):
    """Coordinate descent: scale one element, rescale others proportionally."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_autoconv(a).max()

    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0])

    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0.0 or scale > 50.0:
                    continue

                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = fast_autoconv(a_tr).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

    return best_a, best_obj


def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def propose_candidate():
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses GS initialization, Adam with power-sum, LP refinement,

    deadline = time.monotonic() + 1770.0  # Leave 30s buffer

        a = np.maximum(a, 0.0)

            return

        a_list = a_norm.tolist()
        try:
            score = float(evaluate_sequence(a_list))
        except Exception:
            return

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:

    if parent is None:
        parent = make_uniform_init(1152)

        A = fft(a_gs_parent, N)

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        # Blend with current to stay close to parent
        a_gs_parent = 0.5 * a_gs_parent + 0.5 * a_new
        a_gs_parent = np.maximum(a_gs_parent, 0.0)
