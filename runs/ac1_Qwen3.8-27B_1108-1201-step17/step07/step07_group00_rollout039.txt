import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, peak_indices, M):
    """Solve LP to find best direction."""
    n = len(f)
    K = len(peak_indices)
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for row in range(K):
        k = peak_indices[row]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row, lo:hi+1] = f[lo:hi+1]
    
    c = -np.ones(n)
    b_ub = np.full(K, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        g = res.x
        g = np.maximum(g, 0)
        return g
    return None

import numpy as np

from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, peak_indices):
    """Build LP constraint matrix A where A[row, col] = f[col] for valid entries."""
    n = len(f)
    K = len(peak_indices)
    A = np.zeros((K, n))
    for row in range(K):
        k = peak_indices[row]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp(f, M, peak_indices):
    """Solve LP: max sum(g) s.t. g >= 0, A@g <= M."""
    n = len(f)
    A = build_constraint_matrix(f, peak_indices)
    c = -np.ones(n)
    b_ub = np.full(len(peak_indices), M)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    except Exception:
        return None
    
    if not res.success or res.x is None:
        return None
    
    g = np.maximum(res.x, 0.0)
    if np.any(np.isnan(g)):
        return None
    return g

"""
LP-based optimization with corrected constraint matrix and HiGHS solver.
Key fix: The previous LP used an incorrect constraint matrix (A[row,i]=f[i] 
instead of A[row,i]=f[k-i]). This solved a different problem and limited
improvement. Now uses correct banded matrix, adaptive threshold, multi-phase
search with subgradient polish and coordinate descent.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, peak_indices):
    """Build LP constraint matrix: A[row, i] = f[k - i] for valid i."""
    n = len(f)
    K = len(peak_indices)
    A = np.zeros((K, n))
    for row in range(K):
        k = peak_indices[row]
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            A[row, idx] = f[k - idx]
    return A


def solve_lp(f, M, peak_indices):
    """Solve LP: max sum(g) s.t. g >= 0, A@g <= M."""
    n = len(f)
    A = build_constraint_matrix(f, peak_indices)
    c = -np.ones(n)
    b_ub = np.full(len(peak_indices), M)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    g = np.maximum(res.x, 0.0)
    if np.any(np.isnan(g)) or np.any(np.isinf(g)):
        return None
    return g


def get_peak_indices(b, M, threshold_frac, max_count=None):
    """Get indices where convolution exceeds threshold, with optional cap."""
    thresh = threshold_frac * M
    idx = np.where(b >= thresh)[0]
    if len(idx) == 0:
        idx = np.array([b.argmax()])
    if max_count is not None and len(idx) > max_count:
        # Keep the largest ones
        vals = b[idx]
        top = np.argpartition(vals, -max_count)[-max_count:]
        idx = idx[top]
    return idx


def line_search(f, direction, current_obj):
    """Find best step size for convex combination with direction."""
    n = len(f)
    dir_sum = direction.sum()
    if dir_sum < 1e-15:
        return 0.0

    h = direction / dir_sum
    best_t = 0.0
    best_obj = current_obj

    for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9]:
        f_trial = (1.0 - t) * f + t * h
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial = f_trial / s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t

    return best_t


def run_lp_phase(f_init, time_limit, label="lp", init_thresh=0.95, min_thresh=0.60, max_constr=None):
    """LP-based optimization with corrected constraint matrix and adaptive threshold."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    best_f = f.copy()
    b = fast_convolve(f)
    best_obj = b.max()
    it = 0
    stuck = 0
    thresh = init_thresh
    rng = np.random.default_rng(2024)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        peak_idx = get_peak_indices(b, M, thresh, max_count=max_constr)
        g = solve_lp(f, M, peak_idx)

        if g is None:
            stuck += 2
            if stuck > 15:
                break
            continue

        g_sum = g.sum()

        if g_sum <= 1.0 + 1e-8:
            # LP at bound - can't improve with current constraints
            if thresh > min_thresh + 0.05:
                thresh = max(thresh - 0.05, min_thresh)
                stuck = max(stuck - 1, 0)
                continue
            else:
                stuck += 2
                if stuck > 8:
                    # Perturb and reset threshold
                    f = project_simplex(f + rng.standard_normal(n) * 0.008)
                    thresh = min_thresh
                    stuck = 0
                    best_f = f.copy()
                    best_obj = fast_convolve(f).max()
                    continue

        # Line search
        t = line_search(f, g, M)

        if t < 1e-6:
            stuck += 1
            if stuck > 5:
                thresh = max(thresh - 0.03, min_thresh)
            if stuck > 25:
                break
            continue

        # Update
        f = (1.0 - t) * f + t * g / g_sum
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-15:
            f = f / s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
            # Gradually tighten for better directions
            if it % 25 == 0 and thresh > min_thresh:
                thresh = max(thresh - 0.02, min_thresh)
        else:
            stuck += 1
            if stuck > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                stuck = 0
                if it % 100 == 0:
                    thresh = max(thresh - 0.05, min_thresh)

        if it % 15 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={2*n*best_obj:.6f} "
                  f"th={thresh:.2f} stuck={stuck} t={elapsed:.0f}s")

    return best_f, best_obj


def run_subgradient(f_init, time_limit, label="sg"):
    """Subgradient with soft-max weighting on convolution peaks."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    v = np.zeros(n)
    lr = 0.004
    temp = 0.0005
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Soft-max weights (concentrate on peaks)
        scaled = b / (temp * M)
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w = w / w_sum

        # Compute gradient: g_j = sum_k w_k * f[k-j] (conv of w with reversed f)
        # g_j = sum_{k=j}^{j+n-1} w_k * f_{k-j} = sum_{i=0}^{n-1} w_{j+i} * f_i
        g = np.zeros(n)
        # Use FFT: g = cross-correlation of w and f
        # g_j = sum_i w_{j+i} * f_i = (w * reverse(f))[j + n - 1]... 
        # Actually let's use the direct but vectorized approach for n~1000
        for j in range(n):
            hi = min(len(b), j + n)
            if j < len(b):
                seg_w = w[j:hi]
                seg_f = f[:len(seg_w)]
                g[j] = np.dot(seg_w, seg_f)

        # Update with momentum
        v = 0.92 * v - lr * g
        f_new = project_simplex(f + v)
        f = f_new

        if it % 50 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.04, 0.015)
            else:
                no_improve += 1
                if no_improve > 4:
                    lr *= 0.92
                    no_improve = 0

            if it % 250 == 0:
                temp = min(temp * 1.08, 0.05)
                print(f"    {label} it={it} obj={best_obj:.10f} score={2*n*best_obj:.6f} "
                      f"temp={temp:.5f} lr={lr:.4f} t={time.monotonic()-t0:.0f}s")

            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                v = np.zeros(n)
                no_improve = 0

    return best_f, best_obj


def coordinate_polish(f_init, time_limit, label="cd"):
    """Coordinate descent with multiple multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    best_obj = fast_convolve(f).max()
    multipliers = [0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0, 3.0]

    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s_trial = f_trial.sum()
                if s_trial < 1e-10:
                    continue
                f_trial = f_trial / s_trial
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-12:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f = f / s
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    improved = True

    return f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: LP with corrected constraint matrix (HiGHS solver),
    adaptive threshold, subgradient polish, multi-length exploration, coordinate descent.
    
    Key improvement over previous: Fixed the LP constraint matrix which was using
    A[row,i]=f[i] instead of the correct A[row,i]=f[k-i].
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024.0
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Using uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: LP on parent length (main work) ===
    n_parent = len(parent)
    print(f"\n=== Phase 1: LP main n={n_parent} (corrected matrix) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.42, 650.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = run_lp_phase(parent, time_limit=t1, label="lp_main",
                                init_thresh=0.97, min_thresh=0.65)
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [lp_main]: {score1:.6f}")
        else:
            print(f"  lp_main score={score1:.6f} (no improvement over {best_score:.6f})")

    # === Phase 2: Deeper LP with lower threshold ===
    print(f"\n=== Phase 2: LP deep (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.22, 320.0)
    if t2 > 30:
        n_b = len(best_seq)
        print(f"  Budget: {t2:.0f}s, n={n_b}")
        f2, obj2 = run_lp_phase(best_seq, time_limit=t2, label="lp_deep",
                                init_thresh=0.90, min_thresh=0.50)
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [lp_deep]: {score2:.6f}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.6f}) ===")
    n_base = len(best_seq)
    candidates_n = []
    for factor in [0.7, 0.85, 1.15, 1.3, 1.5]:
        tn = int(n_base * factor)
        tn = max(256, min(5000, tn))
        if tn != n_base and tn not in candidates_n:
            candidates_n.append(tn)
    candidates_n.sort(key=lambda x: abs(x - n_base))

    for target_n in candidates_n[:3]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t3 = min(rem * 0.18, 120.0)
        if t3 < 20:
            break

        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp = f_interp / s
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  LP n={target_n}, budget={t3:.0f}s")
        f3, obj3 = run_lp_phase(f_interp, time_limit=t3, label=f"lp_n{target_n}",
                                init_thresh=0.93, min_thresh=0.60)
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [n{target_n}]: {score3:.6f}")

    # === Phase 4: Subgradient polish ===
    print(f"\n=== Phase 4: Subgradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 100.0)
    if t4 > 15:
        n_b = len(best_seq)
        print(f"  Budget: {t4:.0f}s, n={n_b}")
        f4, obj4 = run_subgradient(best_seq, time_limit=t4, label="sg")
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [sg]: {score4:.6f}")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 80.0)
    if t5 > 10:
        print(f"  Budget: {t5:.0f}s")
        f5, obj5 = coordinate_polish(best_seq, time_limit=t5, label="cd")
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [cd]: {score5:.6f}")

    # === Phase 6: Final LP burst ===
    print(f"\n=== Phase 6: Final LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = max(rem, 0)
    if t6 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t6:.0f}s, n={n_b}")
        f6, obj6 = run_lp_phase(best_seq, time_limit=t6, label="lp_final",
                                init_thresh=0.85, min_thresh=0.45)
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [lp_final]: {score6:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
