"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
(1) Spectral initialization via random-phase flat-spectrum sequences,
(2) Fienup-style error reduction (iterative projection between frequency-domain
    flatness and time-domain non-negativity),
(3) Adam gradient descent in log-space with soft-max temperature annealing,
(4) Simulated annealing for global exploration,
(5) Peak-targeted local search and coordinate descent for final polishing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def get_score(a):
    """Compute the objective score for a sequence (assumes sum(a)=1)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def fienup_iteration(a, fft_size, n_flatten=0):
    """
    One Fienup-style iteration:
    1. Go to frequency domain
    2. Flatten the magnitude (partial flatten to avoid killing information)
    3. Come back to time domain
    4. Enforce non-negativity
    5. Renormalize
    """
    n = len(a)
    A = rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Partial flattening: interpolate between current magnitude and flat
    target_mag = np.ones_like(mag) * mag.mean()
    # Blend factor: 0 = no change, 1 = fully flat
    blend = 0.3
    new_mag = (1 - blend) * mag + blend * target_mag
    
    # Also try to reduce peaks specifically
    # Identify frequency bins that contribute most to time-domain peaks
    A_new = new_mag * np.exp(1j * phase)
    a_new = irfft(A_new, fft_size)[:2*n-1]
    
    # Enforce non-negativity and renormalize
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a.copy()
    
    return a_new

    """
    Fienup-style iterative optimization with adaptive blending.
    """

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    a = a / a.sum()
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    
    best_a = a.copy()
    best_score = get_score(a)
    no_improve = 0
    max_no_improve = 100
    
    for i in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive blend: start aggressive, become more gentle
        progress = i / n_iter
        current_blend = blend * (1.0 - 0.5 * progress)
        
        a = fienup_iteration(a, fft_size)
        
        # Also do a direct time-domain projection
        # Reduce the peak-contributing elements
        conv = fast_convolve(a)
        peak_idx = np.argmax(conv)
        
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:

            a = best_a + np.random.standard_normal(n) * 0.01
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
    
    return best_a, best_score

        # Let me think more carefully.
        # c[k] = sum_i a[i] * a[k-i]
        # dc[k]/da[m] = sum_i [delta(i,m)*a[k-i] + a[i]*delta(k-i,m)]
        # = a[k-m] + a[k-m] = 2*a[k-m] (when m != k-m)
        # = a[k-m] = a[m] (when m = k-m, i.e., k = 2m) -- wait no
        # Actually dc[k]/da[m] = a[k-m] + (a[m] if k-m != m else a[k-m])
        # Hmm, let me just use the standard result.
        # For c = conv(a, a), dc[k]/da[m] = a[k-m] + a[k-m] = 2*a[k-m] (for valid k-m)
        # But this is the derivative with respect to the "right" copy of a.
        # The total derivative (both copies are the same a) is:
        # dc[k]/da[m] = a[k-m] + a[m-(k-m)] ... no.
        # 
        # c[k] = sum_i a[i] * a[k-i]
        # dc[k]/da[m] = (d/dam a[m]) * a[k-m] + sum_i a[i] * (d/dam a[k-i])
        # = a[k-m] + a[k-m] = 2*a[k-m]
        # (The second term: d/dam a[k-i] = delta(k-i, m) = delta(i, k-m), so the sum picks out i=k-m, giving a[k-m])
        # Wait, that gives dc[k]/da[m] = a[k-m] + a[k-m] = 2*a[k-m]
        # But only when 0 <= k-m < n (i.e., k-m is a valid index)
        
        # So grad_obj[m] = 2n * sum_k w[k] * dc[k]/da[m] = 2n * sum_k w[k] * 2*a[k-m] (for valid k-m)
        # = 4n * sum_{k: 0<=k-m<n, 0<=k<2n-1} w[k] * a[k-m]
        # Let j = k-m, then k = j+m, and we need 0 <= j < n and 0 <= j+m < 2n-1
        # So j ranges from max(0, -(m)) to min(n-1, 2n-2-m)
        # Since m >= 0 and m < n, this is j from 0 to min(n-1, 2n-2-m)
        # For m < n-1: j from 0 to n-1 (all valid)
        # For m = n-1: j from 0 to n-1 (since 2n-2-(n-1) = n-1)
        # So actually for all m in [0, n-1], j ranges from 0 to n-1
        # Wait: 2n-2-m >= n-1 iff m <= n-1, which is always true.
        # And j >= 0 is always satisfied.
        # So grad_obj[m] = 4n * sum_{j=0}^{n-1} w[j+m] * a[j]
        # 
        # Hmm wait, I need to be more careful. k ranges from 0 to 2n-2.
        # For a given m, k-m ranges from -m to 2n-2-m.
        # Valid j = k-m requires 0 <= j < n, so 0 <= k-m < n, i.e., m <= k < m+n.
        # And k also needs to satisfy 0 <= k < 2n-1.
        # So k ranges from max(0, m) to min(2n-2, m+n-1).
        # Since m < n: max(0,m) = m, min(2n-2, m+n-1) = m+n-1 (since m+n-1 <= 2n-2 iff m <= n-1, true)
        # So k ranges from m to m+n-1, and j = k-m ranges from 0 to n-1.
        # grad_obj[m] = 4n * sum_{j=0}^{n-1} w[m+j] * a[j]
        
        # This is a cross-correlation between w (shifted) and a!
        # Specifically, it's like convolving w[m:] with a, but only the first n elements.
        # We can compute this efficiently using FFT.
        
        # Actually, let me use a simpler approach: 
        # grad_obj = 4*n * cross_correlation(w_padded, a)
        # where we take appropriate slices.
        
        # Using the previous approach:
        W = rfft(w, fft_size)
        A = rfft(a_curr, fft_size)
        # We need sum_{j=0}^{n-1} w[m+j] * a[j] for each m
        # This is like a cross-correlation. Let's think of it as:
        # For each m, result[m] = sum_j w[m+j]*a[j]
        # This is the (n-1-m)th element of conv(w, a_reversed) ... 
        # Actually, let's use the direct FFT approach.
        # 
        # Define: for each m in [0,n-1], we want sum_{j=0}^{n-1} a[j] * w[m+j]
        # This is the cross-correlation of a and w at lag m.
        # cross_corr[m] = sum_j a[j] * w[j+m]
        # 
        # In Fourier: cross_corr = IFFT(FFT(a) * conj(FFT(w)))... no that's not right either.
        # 
        # Let me just use the previous method which was:
        # corr = irfft(W * conj(A), fft_size)[:n]
        # This gives sum_j w[j] * a[m-j] for m in [0,n-1]... 
        # Hmm, I need to be more careful.
        
        # Let me just compute it directly for small n or use a different FFT trick.
        # For efficiency, let's compute:
        # result[m] = sum_{j=0}^{n-1} a[j] * w[m+j]
        # 
        # Pad both to length fft_size:
        # a_padded = [a[0], ..., a[n-1], 0, ..., 0]
        # w_padded = [w[0], ..., w[2n-2], 0, ..., 0]
        # 
        # We want the "anti-diagonal" product sum.
        # result[m] = a[0]*w[m] + a[1]*w[m+1] + ... + a[n-1]*w[m+n-1]
        # 
        # This is the (n-1-th) element of conv(a_reversed, w_padded)?
        # conv(a_rev, w)[k] = sum_i a_rev[i] * w[k-i] = sum_i a[n-1-i] * w[k-i]
        # For k = m+n-1: sum_i a[n-1-i] * w[m+n-1-i]
        # Let j = n-1-i: sum_j a[j] * w[m+j]  -- YES!
        # So result = conv(a_reversed, w)[n-1 : 2n-1]
        
        a_rev = a_curr[::-1]
        # conv(a_rev, w) has length n + (2n-1) - 1 = 3n-2
        # We want elements at positions n-1 to 2n-2 (inclusive), which is n elements
        conv_size = n + (2*n-1) - 1  # = 3n-2
        conv_fft_size = 1 << (conv_size - 1).bit_length()
        Ar = rfft(a_rev, conv_fft_size)
        Wc = rfft(w, conv_fft_size)
        conv_aw = irfft(Ar * Wc, conv_fft_size)
        grad_a = 4.0 * n * conv_aw[n-1:n-1+n]
        
        # Transform to z-space
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate
        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive restart
        if no_improve > max_no_improve:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.01
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = last_lr * 0.8
        
        # Slow LR decay
        lr = max(0.001, last_lr * 0.9999)
        last_lr = lr
        
        if verbose and t % verbose_interval == 0:
            print(f"      Adam t={t} T={T:.5f} lr={lr:.6f} score={best_score:.6f}")
    
    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def peak_shave(a0, time_limit, n_trials=10000):
    """
    Peak-targeted local search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find the peak (exclude edges which are small)
        b_search = b.copy()
        b_search[:max(1, n//10)] = 0
        b_search[-max(1, n//10):] = 0
        peak_k = int(np.argmax(b_search))
        peak_val = b[peak_k]
        
        # Elements contributing to this peak: a[i] * a[peak_k - i] for valid i
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        idx = np.arange(i_start, i_end + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing the top contributors
        n_top = min(5, len(idx))
        top_local = np.argsort(contribs)[::-1][:n_top]
        
        improved = False
        for ci in top_local:
            i = int(idx[ci])
            j = int(peak_k - i)
            if i >= n or j >= n or i < 0 or j < 0:
                continue
            if i == j or a[j] < 1e-15 or a[i] < 1e-15:
                continue
            
            for f1 in [0.9, 0.85, 0.8, 0.7]:
                a_trial = a.copy()
                a_trial[i] *= f1
                # Redistribute the removed mass
                removed = a[i] * (1.0 - f1)
                a[i] = a_trial[i]
                # Add to a[j] or redistribute
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += removed * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a_trial.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try random perturbation
            if trial % 100 == 0:
                a = best_a + rng.standard_normal(n) * 0.01
                a = np.maximum(a, 0)
                a = a / a.sum()
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, n_sweeps=100):
    """
    Coordinate descent with multiplicative updates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        order = rng.permutation(n)
        sweep_improved = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            
            best_local_factor = 1.0
            best_local_score = best_score
            
            for factor in [0.7, 0.8, 0.9, 1.1, 1.2, 1.3]:
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_local_score:
                    best_local_score = score
                    best_local_factor = factor
            
            if best_local_factor != 1.0:
                a[i] = a[i] * best_local_factor
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True
        
        if not sweep_improved:
            break
    
    return best_a, best_score


def spectral_initialize(n, rng):
    """
    Generate a sequence with approximately flat spectrum, then project to non-negative.
    """
    fft_size = 1 << (2 * n - 2).bit_length()
    
    # Random phase, flat magnitude
    n_freq = fft_size // 2 + 1
    phase = rng.uniform(0, 2 * np.pi, n_freq)
    mag = np.ones(n_freq)
    # Modify DC component
    mag[0] = np.sqrt(n)  # To ensure reasonable mean
    A = mag * np.exp(1j * phase)
    # Make it real: A[k] = conj(A[n-k])
    A_full = np.zeros(fft_size, dtype=complex)
    A_full[:n_freq] = A
    for k in range(1, fft_size // 2 + 1):
        A_full[fft_size - k] = np.conj(A[k] if k < n_freq else A[fft_size - k] if fft_size - k < n_freq else 0])
    # Actually let's just use rfft/irfft properly
    a = irfft(A, fft_size)[:n]
    
    # Project to non-negative
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        # Retry or use uniform
        a = np.ones(n) / n
    
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def simulated_annealing(a0, time_limit, T_init=0.1, T_final=0.001, 
                        n_steps_per_temp=10, alpha=0.999):
    """
    Simulated annealing in the simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = get_score(a)
    T = T_init
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        # Propose a move: randomly select 2-5 elements and perturb them
        n_perturb = rng.integers(2, min(6, n))
        indices = rng.choice(n, size=n_perturb, replace=False)
        
        a_trial = a.copy()
        for i in indices:
            scale = rng.uniform(0.5, 1.5)
            a_trial[i] *= scale
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s
        
        new_score = get_score(a_trial)
        delta = new_score - best_score
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_trial
            if new_score < best_score:
                best_score = new_score
                best_a = a_trial.copy()
        
        step += 1
        T = max(T_final, T_init * (alpha ** step))
    
    return best_a, best_score


def evaluate_and_track(a, label, best_seq, best_score, print_flag=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if print_flag:
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
(1) Spectral initialization via random-phase flat-spectrum sequences,
(2) Fienup-style iterative projection (frequency-domain flattening + time-domain non-negativity),
(3) Adam gradient descent in log-space with soft-max temperature annealing,
(4) Simulated annealing for global exploration,
(5) Pairwise peak shaving and coordinate descent for final polishing.
Key improvements: spectral inits, pairwise local search, aggressive multi-start,
and two-phase temperature scheduling.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def get_score(a):
    """Compute the objective score for a sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def compute_grad_a(a, w, n, fft_size):
    """
    Compute gradient of 2n * softmax(conv(a,a), T) w.r.t. a.
    grad_a[m] = 4n * sum_{j=0}^{n-1} w[m+j] * a[j]
    Uses FFT-based cross-correlation via convolution with reversed a.
    """
    a_rev = a[::-1]
    conv_size = n + (2 * n - 1) - 1
    conv_fft_size = 1 << (conv_size - 1).bit_length()
    Ar = rfft(a_rev, conv_fft_size)
    Wc = rfft(w, conv_fft_size)
    conv_aw = irfft(Ar * Wc, conv_fft_size)
    return 4.0 * n * conv_aw[n - 1:n - 1 + n]


def adam_logspace(a0, time_limit, T_init=0.05, T_final=0.002, lr=0.01,
                  beta1=0.9, beta2=0.999, eps=1e-8, schedule='hold_anneal',
                  verbose=False, verbose_interval=5000):
    """
    Adam optimizer in log-space with temperature annealing for soft-max smoothing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    s = a.sum()
    a = a / s
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 4000
    last_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        if schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        elif schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'hold_anneal':
            if progress < 0.25:
                T = T_init
            else:
                p2 = (progress - 0.25) / 0.75
                T = T_init * (T_final / T_init) ** p2
        elif schedule == 'cosine':
            T = T_final + 0.5 * (T_init - T_final) * (1 + np.cos(np.pi * progress))
        else:
            T = T_init * (T_final / T_init) ** progress

        a_curr = softmax_from_z(z)

        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()

        grad_a = compute_grad_a(a_curr, w, n, fft_size)

        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.008
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = last_lr * 0.75

        lr = max(0.001, last_lr * 0.99995)
        last_lr = lr

        if verbose and t % verbose_interval == 0:
            print(f"      Adam t={t} T={T:.5f} lr={lr:.6f} score={best_score:.6f}")

    return best_a, best_score


def fienup_optimize(a0, time_limit, blend=0.4):
    """
    Fienup-style iterative optimization:
    Alternate between frequency-domain magnitude flattening and time-domain non-negativity.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    a = a / a.sum()
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()

    best_a = a.copy()
    best_score = get_score(a)
    no_improve = 0
    max_no_improve = 200
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        progress = iteration / (time_limit * 500)
        
        # Adaptive blend: start moderate, decrease over time
        current_blend = blend * max(0.3, 1.0 - 0.5 * progress)

        # Frequency domain
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Partial flattening
        target_mag = np.full_like(mag, mag.mean())
        new_mag = (1.0 - current_blend) * mag + current_blend * target_mag

        # Also reduce the largest frequency components (contribute most to peaks)
        n_top = max(1, n // 4)
        top_indices = np.argsort(mag)[::-1][:n_top]
        new_mag[top_indices] = np.minimum(new_mag[top_indices], mag.mean() * 1.5)

        A_new = new_mag * np.exp(1j * phase)
        a_new = irfft(A_new, fft_size)[:n]

        # Enforce non-negativity and renormalize
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        else:
            a_new = a.copy()

        a = a_new
        score = 2.0 * n * fast_convolve(a).max()

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            a = best_a + np.random.standard_normal(n) * 0.02
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0

    return best_a, best_score


def peak_shave(a0, time_limit, n_trials=20000):
    """
    Peak-targeted local search with pairwise optimization.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)

        # Find the dominant peak (exclude very small edge values)
        margin = max(1, n // 20)
        b_search = b.copy()
        b_search[:margin] = b[margin]
        b_search[-margin:] = b[-margin]
        peak_k = int(np.argmax(b_search))
        peak_val = b[peak_k]

        # Elements contributing to this peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        idx = np.arange(i_start, i_end + 1)
        if len(idx) == 0:
            continue
        contribs = a[idx] * a[peak_k - idx]

        n_top = min(4, len(idx))
        top_local = np.argsort(contribs)[::-1][:n_top]

        improved = False

        # Try single-element reductions
        for ci in top_local:
            i = int(idx[ci])
            if a[i] < 1e-15:
                continue
            for factor in [0.85, 0.75, 0.65]:
                a_trial = a.copy()
                removed = a[i] * (1.0 - factor)
                a_trial[i] *= factor
                # Redistribute to non-peak-contributing elements
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                if mask.sum() < n // 2:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += removed * a_trial[mask] / others_sum
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a_trial.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pairwise: reduce one, increase its partner (peak-contributing pair)
            for ci in top_local[:2]:
                i = int(idx[ci])
                j = int(peak_k - i)
                if i == j or i >= n or j >= n or i < 0 or j < 0:
                    continue
                if a[i] < 1e-15 or a[j] < 1e-15:
                    continue
                for f1 in [0.7, 0.5, 0.3]:
                    for f2 in [1.1, 1.3, 1.5]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a_trial.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                continue

        # Stuck: random perturbation
        if trial > 50 and trial % 200 == 0:
            a = best_a + rng.standard_normal(n) * 0.008
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_sweeps=150):
    """
    Coordinate descent with multiplicative updates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue

            best_local_factor = 1.0
            best_local_score = best_score

            for factor in [0.7, 0.8, 0.9, 1.1, 1.2, 1.4]:
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_local_score:
                    best_local_score = score
                    best_local_factor = factor

            if best_local_factor != 1.0:
                a[i] = a[i] * best_local_factor
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True

        if not sweep_improved:
            break

    return best_a, best_score


def simulated_annealing(a0, time_limit, T_init=0.05, T_final=0.0005, alpha=0.9997):
    """
    Simulated annealing on the probability simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = get_score(a)
    T = T_init
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    step = 0

    while time.monotonic() - t0 < time_limit:
        n_perturb = int(rng.integers(2, min(8, n)))
        indices = rng.choice(n, size=n_perturb, replace=False)

        a_trial = a.copy()
        for i in indices:
            scale = rng.uniform(0.4, 1.6)
            a_trial[i] *= scale
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-15:
            step += 1
            T = max(T_final, T * alpha)
            continue
        a_trial = a_trial / ss

        new_score = get_score(a_trial)
        delta = new_score - best_score

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            a = a_trial
            if new_score < best_score:
                best_score = new_score
                best_a = a_trial.copy()

        step += 1
        T = max(T_final, T * alpha)

    return best_a, best_score


def spectral_initialize(n, rng):
    """
    Generate a non-negative sequence with approximately flat periodic spectrum.
    """
    fft_size = 1 << (2 * n - 2).bit_length()
    n_freq = fft_size // 2 + 1

    # Random phases with flat magnitude
    phase = rng.uniform(0, 2 * np.pi, n_freq)
    mag = np.ones(n_freq) * np.sqrt(n)
    phase[0] = 0  # DC component real
    mag[0] = np.sqrt(n)

    A = mag * np.exp(1j * phase)
    a = irfft(A, fft_size)[:n]

    # Project to non-negative
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        # Retry with different phases
        for _ in range(10):
            phase = rng.uniform(0, 2 * np.pi, n_freq)
            phase[0] = 0
            A = mag * np.exp(1j * phase)
            a = irfft(A, fft_size)[:n]
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
                break
        else:
            a = np.ones(n) / n

    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_and_track(a, label, best_seq, best_score, print_flag=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if print_flag:
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
