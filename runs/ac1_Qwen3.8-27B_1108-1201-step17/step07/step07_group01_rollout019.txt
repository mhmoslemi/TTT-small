def subgradient_optimize(a_init, time_limit, verbose=True, label=""):
    """Subgradient method focusing on peak lags."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    eta = 0.1
    decay = 0.999
    
    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        
        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
        
        # Find top peaks (excluding lag 0 and 2n-2 which are just a[0]^2 and a[n-1]^2)
        b_work = b.copy()
        b_work[0] = -1
        b_work[-1] = -1
        top_peaks = np.argsort(b_work)[::-1][:5]
        
        # Compute subgradient: average of peak gradients
        grad = np.zeros(n)
        for k in top_peaks:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            grad[idx] += 2.0 * a[k - idx]
        
        grad /= len(top_peaks)
        
        # Project to tangent space
        grad = grad - np.dot(a, grad)
        
        # Multiplicative update
        log_update = -eta * grad
        log_update = np.clip(log_update, -2.0, 2.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        a = a / s
        
        eta *= decay
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} SGD iter={it} ({elapsed:.0f}s): score={current_score:.7f} best={best_score:.7f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

LP-based optimization with peak-focused subgradient refinement for minimizing
2*n*max(auto_conv(a))/(sum(a))^2. Uses convolution-structured LPs to find
descent directions, combined with adaptive step sizes, random restarts,
and multi-n exploration.

import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def build_conv_matrix(a, n, selected_rows=None):

    if selected_rows is None:
        selected_rows = range(2 * n - 1)
    A_mat = np.zeros((len(list(selected_rows)), n))
    for row_idx, k in enumerate(selected_rows):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        A_mat[row_idx, i_start:i_end + 1] = a[i_start:i_end + 1]
    return A_mat


def lp_solve_direction(a, M, n_top=None, full_constraints=False):
    """Solve LP: max sum(g) s.t. a*g <= M (on selected rows), g >= 0."""
    n = len(a)
    b = fast_convolve(a)

    if full_constraints:
        rows = list(range(2 * n - 1))
    elif n_top is not None:
        top_indices = np.argsort(b)[::-1][:n_top]
        rows = sorted(top_indices.tolist())
    else:
        rows = list(range(2 * n - 1))

    A_mat = build_conv_matrix(a, n, rows)
    c = -np.ones(n)
    b_vec = M * np.ones(len(rows))

    try:
        res = linprog(c, A_ub=A_mat, b_ub=b_vec,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a_init, time_limit, n_top=80, max_iters=50000,
                verbose=True, label=""):
    """LP-based optimization with adaptive step size."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stuck = 0
    rng = np.random.default_rng(777)

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        current_score = 2.0 * n * M

        if current_score < best_score - 1e-13:
            best_score = current_score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Solve LP for direction
        g = lp_solve_direction(a, M, n_top=n_top)
        if g is None or g.sum() < 1e-12:
            if stuck > 20:
                break
            continue

        g = g / g.sum()

        # Line search
        improved = False
        step_sizes = [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]
        for t in step_sizes:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-14:
                continue
            a_trial = a_trial / s
            sc = score_normalized(a_trial)
            if sc < current_score - 1e-13:
                a = a_trial
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    stuck = 0
                improved = True
                break

        if not improved and stuck > 30:
            # Perturb from best
            noise_scale = min(0.05, 0.01 + 0.001 * (stuck // 50))
            noise = rng.standard_normal(n) * noise_scale
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} LP it={it} ({elapsed:.0f}s): "
                  f"cur={current_score:.7f} best={best_score:.7f} stuck={stuck}")

    return best_a, best_score


def subgradient_refine(a_init, time_limit, verbose=True, label=""):
    """Subgradient method with peak focusing and multiplicative updates."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.05
    decay = 0.9995
    stuck = 0
    rng = np.random.default_rng(999)

    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-13:
            best_score = current_score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Focus on top peaks
        b_work = b.copy()
        b_work[0] = -1.0
        b_work[-1] = -1.0
        n_peaks = min(8, 2 * n - 3)
        top_peaks = np.argsort(b_work)[::-1][:n_peaks]

        # Compute gradient (average of peak subgradients)
        grad = np.zeros(n)
        for k in top_peaks:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= n_peaks

        # Project to simplex tangent
        grad = grad - np.dot(a, grad)

        # Multiplicative update with clipping
        log_up = -eta * grad
        log_up = np.clip(log_up, -1.5, 1.5)
        a = a * np.exp(log_up)
        a = np.maximum(a, 1e-300)
        a = a / a.sum()

        eta *= decay

        if stuck > 200:
            noise = rng.standard_normal(n) * 0.02
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck = 0
            eta = 0.05

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} SGD it={it} ({elapsed:.0f}s): "
                  f"cur={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def peak_shave_refine(a_init, time_limit, verbose=True, label=""):
    """Targeted peak shaving: directly reduce top contributing pairs."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(555)

    for it in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-13:
            best_score = current_score
            best_a = a.copy()

        # Find top 3 peaks (excluding endpoints)
        b_work = b.copy()
        b_work[0] = -1.0
        b_work[-1] = -1.0
        peaks = np.argsort(b_work)[::-1][:3]

        improved = False
        for k in peaks:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            contribs = a[idx] * a[k - idx]

            # Top 5 contributing pairs
            top_ci = np.argsort(contribs)[::-1][:5]
            for ci in top_ci:
                i = int(idx[ci])
                j = k - i
                if i == j:
                    for f in [0.9, 0.8, 0.7]:
                        a_trial = a.copy()
                        a_trial[i] *= f
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                else:
                    # Try reducing both
                    for f1 in [0.9, 0.8]:
                        for f2 in [0.9, 0.8]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            a_trial = np.maximum(a_trial, 0)
                            a_trial = a_trial / a_trial.sum()
                            sc = score_normalized(a_trial)
                            if sc < best_score - 1e-13:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try random coordinate perturbation
            if it % 100 == 0:
                for _ in range(20):
                    i = rng.integers(0, n)
                    for f in [0.5, 0.7, 0.9, 1.2, 1.5, 2.0]:
                        a_trial = a.copy()
                        a_trial[i] *= f
                        a_trial = np.maximum(a_trial, 0)
                        s = a_trial.sum()
                        if s < 1e-14:
                            continue
                        a_trial = a_trial / s
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if not improved and it % 500 == 0:
                noise = rng.standard_normal(n) * 0.03
                a = best_a + noise
                a = np.maximum(a, 1e-15)
                a = a / a.sum()

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} PS it={it} ({elapsed:.0f}s): "
                  f"best={best_score:.7f}")

    return best_a, best_score


def mirror_descent_refine(a_init, time_limit, verbose=True, label=""):
    """Mirror descent with softmax smoothing of the max objective."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    T_start = 0.01
    T_end = 0.001
    eta = 0.03
    max_iters = 30000
    rng = np.random.default_rng(333)
    stuck = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (0.8 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        eta_cur = 0.03 * (0.005 / 0.03) ** progress

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-13:
            best_score = current_score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Softmax weights
        scaled = b / T
        m = scaled.max()
        exp_s = np.exp(scaled - m)
        w = exp_s / exp_s.sum()

        # Gradient: 2n * cross_corr(w, a)
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad = 2.0 * n * correl[:n]

        # Multiplicative update
        log_up = -eta_cur * grad
        log_up = np.clip(log_up, -2.0, 2.0)
        a = a * np.exp(log_up)
        a = np.maximum(a, 1e-300)
        a = a / a.sum()

        if stuck > 300:
            noise = rng.standard_normal(n) * 0.015
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck = 0

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} MD it={it} ({elapsed:.0f}s): "
                  f"cur={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP-based descent, mirror descent,
    peak shaving, and multi-n exploration to minimize the auto-correlation
    peak ratio.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 12.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default: n=768, score={best_score:.7f}")

    n0 = len(best_seq)

    # === Phase 1: LP-based optimization (primary, full constraints) ===
    print(f"\n=== Phase 1: LP Full (n={n0}, best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 720
    if rem > 60:
        t_alloc = min(rem * 0.55, 350.0)
        print(f"  LP full: budget={t_alloc:.0f}s")
        a_lp, _ = lp_optimize(best_seq, t_alloc, n_top=None, verbose=True,
                              label="LP_full", max_iters=20000)
        best_seq, best_score = check_and_update(a_lp, "LP_full", best_seq, best_score)

    # === Phase 2: LP with top constraints (faster iterations) ===
    print(f"\n=== Phase 2: LP Top-80 (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 40:
        t_alloc = min(rem * 0.5, 200.0)
        print(f"  LP top80: budget={t_alloc:.0f}s")
        a_lp2, _ = lp_optimize(best_seq, t_alloc, n_top=80, verbose=True,
                               label="LP_80", max_iters=30000)
        best_seq, best_score = check_and_update(a_lp2, "LP_80", best_seq, best_score)

    # === Phase 3: Mirror descent ===
    print(f"\n=== Phase 3: Mirror Descent (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 350
    if rem > 30:
        t_alloc = min(rem * 0.5, 150.0)
        print(f"  MD: budget={t_alloc:.0f}s")
        a_md, _ = mirror_descent_refine(best_seq, t_alloc, label="MD")
        best_seq, best_score = check_and_update(a_md, "MD", best_seq, best_score)

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shave (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 250
    if rem > 20:
        t_alloc = min(rem * 0.5, 120.0)
        print(f"  PS: budget={t_alloc:.0f}s")
        a_ps, _ = peak_shave_refine(best_seq, t_alloc, label="PS")
        best_seq, best_score = check_and_update(a_ps, "PS", best_seq, best_score)

    # === Phase 5: LP again from improved point ===
    print(f"\n=== Phase 5: LP Refinement (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 180
    if rem > 30:
        t_alloc = min(rem * 0.5, 120.0)
        print(f"  LP top100: budget={t_alloc:.0f}s")
        a_lp3, _ = lp_optimize(best_seq, t_alloc, n_top=100, verbose=True,
                               label="LP_ref", max_iters=20000)
        best_seq, best_score = check_and_update(a_lp3, "LP_ref", best_seq, best_score)

    # === Phase 6: Subgradient polish ===
    print(f"\n=== Phase 6: Subgradient (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 120
    if rem > 20:
        t_alloc = min(rem * 0.5, 80.0)
        print(f"  SGD: budget={t_alloc:.0f}s")
        a_sg, _ = subgradient_refine(best_seq, t_alloc, label="SGD")
        best_seq, best_score = check_and_update(a_sg, "SGD", best_seq, best_score)

    # === Phase 7: Try larger n ===
    print(f"\n=== Phase 7: Larger n (best={best_score:.7f}) ===")
    for alt_n in [1024]:
        rem = deadline - time.monotonic() - 80
        if rem < 60:
            break
        t_alloc = min(rem * 0.4, 120.0)
        print(f"  n={alt_n}: budget={t_alloc:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = lp_optimize(init_alt, t_alloc, n_top=80, label=f"LP_n{alt_n}",
                               max_iters=15000)
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)

    # === Phase 8: Final LP polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 10
    if rem > 10:
        t_alloc = min(rem * 0.7, 60.0)
        a_fin, _ = lp_optimize(best_seq, t_alloc, n_top=120, label="LP_fin",
                               max_iters=10000)
        best_seq, best_score = check_and_update(a_fin, "LP_fin", best_seq, best_score)

    # Final peak shave
    rem = deadline - time.monotonic() - 3
    if rem > 5 and best_seq is not None:
        a_ps2, _ = peak_shave_refine(best_seq, min(rem, 8.0), label="PS_fin")
        best_seq, best_score = check_and_update(a_ps2, "PS_fin", best_seq, best_score)

    if best_seq is None:
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
