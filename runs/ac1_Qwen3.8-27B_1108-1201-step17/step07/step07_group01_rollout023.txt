import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute score assuming sum(a)=1: returns 2*n*max(conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def compute_smooth_max_grad(a, T):
    """Compute gradient of T*log(sum exp(b_k/T)) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    scaled = b / T
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()
    # Gradient of smooth max: sum_k w_k * grad(b_k)
    # grad(b_k)[i] = 2*a[k-i] if 0<=k-i<n else 0
    # So gradient[i] = 2 * sum_k w_k * a[k-i] (for valid k-i)
    # = 2 * (w * a_flipped)[i] where we handle the linear convolution indexing
    # More precisely: gradient[i] = 2 * sum_k w[k] * a[k-i] for 0<=k-i<n
    # = 2 * cross_correlate(w, a)[i]
    # cross_correlate(w, a)[i] = sum_k w[k] * a[k-i]
    # This is the correlation of w (length 2n-1) with a (length n)
    fft_size = 1 << (3 * n).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # cross_correlate(w, a)[i] = sum_k w[k] * a[k-i]
    # = (w * reversed_a)[i + n - 1] using convolution
    # Or: = irfft(conj(W) * A) gives cross correlation
    correl = irfft(np.conj(W) * A, fft_size)[:n]
    grad = 2.0 * n * correl
    return grad, b, w


def smooth_max_value(b, T):
    """Compute T*log(sum exp(b_k/T))."""
    scaled = b / T
    m = scaled.max()
    return T * (m + np.log(np.exp(scaled - m).sum()))


def mirror_update(a, grad, eta):
    """Multiplicative (mirror) update on the simplex."""
    log_update = -eta * grad
    log_update = np.clip(log_update, -4.0, 4.0)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-300)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def backtrack_line_search_mirror(a, grad, eta0, T, max_backtrack=20):
    """Backtracking line search for mirror descent steps."""
    n = len(a)
    current_score = score_normalized(a)
    eta = eta0
    for bt in range(max_backtrack):
        a_trial = mirror_update(a, grad, eta)
        trial_score = score_normalized(a_trial)
        if trial_score < current_score - 1e-12:
            return a_trial, trial_score
        eta *= 0.5
    return a, current_score


def spectral_shaping(a, n_iterations=50, alpha=0.5):
    """Gerchberg-Saxton style spectral shaping to flatten the auto-convolution spectrum."""
    n = len(a)
    a_curr = a.copy()
    # Target: |FFT(a)|^2 flat => circular auto-correlation is delta
    # For linear conv, approximate by zero-padding
    fft_size = 1 << (2 * n).bit_length()
    for it in range(n_iterations):
        A = rfft(a_curr, fft_size)
        power = np.abs(A) ** 2
        # Target power: flat = total_power / fft_size
        total_power = power.sum()
        target_power = np.ones_like(power) * (total_power / len(power))
        # Gradually move toward flat spectrum
        new_power = (1 - alpha) * power + alpha * target_power
        # Update amplitude, keep phase
        new_A = np.sqrt(new_power) * np.exp(1j * np.angle(A))
        a_curr = irfft(new_A, fft_size)[:n]
        a_curr = np.maximum(a_curr, 0)
        s = a_curr.sum()
        if s > 1e-15:
            a_curr = a_curr / s
    return a_curr


def multi_peak_balance_grad(a, top_k=5, T=0.001):
    """Compute gradient weighting the top-k peaks more heavily."""
    n = len(a)
    b = fast_convolve(a)
    # Get top-k peak indices
    top_indices = np.argsort(b)[::-1][:top_k]
    # Weight by how close to max
    T_local = max(T, (b.max() - b.min()) * 0.1)
    weights = np.exp((b[top_indices] - b.max()) / T_local)
    weights = weights / weights.sum()
    # Compute weighted gradient
    grad = np.zeros(n)
    for i, k in enumerate(top_indices):
        if weights[i] < 1e-6:
            continue
        # grad of b_k: component j = 2*a[k-j] if 0<=k-j<n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += weights[i] * 2.0 * a[k - idx]
    return grad * n


def coordinate_refine(a, n_trials_per_coord=8, time_limit=30.0):
    """Refine by trying multiplicative changes to individual coordinates."""
    n = len(a)
    a_curr = a.copy()
    best_score = score_normalized(a_curr)
    t0 = time.monotonic()
    factors = [0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.3]
    improved_total = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a_curr[i] < 1e-12:
                continue
            for f in factors:
                a_trial = a_curr.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * f
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others_sum
                if a_trial.min() < -1e-10:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-13:
                    best_score = sc
                    a_curr = a_trial
                    improved = True
                    improved_total += 1
                    break
        if not improved:
            break

    return a_curr, best_score


def peak_shave_multi(a, top_k=3, time_limit=60.0):
    """Simultaneously shave the top-k peaks by reducing their top contributors."""
    n = len(a)
    a_curr = a.copy()
    best_score = score_normalized(a_curr)
    t0 = time.monotonic()

    for step in range(5000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a_curr)
        # Get top-k peaks
        peak_indices = np.argsort(b)[::-1][:top_k]
        if b[peak_indices[0]] <= best_score / (2.0 * n) + 1e-14:
            break

        improved = False
        for pk in peak_indices:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a_curr[idx] * a_curr[pk - idx]
            top_idx = np.argsort(contribs)[::-1][:4]

            for ci in top_idx:
                i = int(idx[ci])
                if a_curr[i] < 1e-14:
                    continue
                for factor in [0.95, 0.9, 0.85, 0.8]:
                    a_trial = a_curr.copy()
                    old_val = a_trial[i]
                    a_trial[i] = old_val * factor
                    diff = old_val - a_trial[i]
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-14:
                        a_curr = a_trial
                        best_score = sc
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try paired reduction
            for pk in peak_indices[:2]:
                js = max(0, pk - n + 1)
                je = min(n - 1, pk)
                idx = np.arange(js, je + 1)
                contribs = a_curr[idx] * a_curr[pk - idx]
                top_idx = np.argsort(contribs)[::-1][:3]
                for ci1 in top_idx[:2]:
                    for ci2 in top_idx[:2]:
                        if ci1 == ci2:
                            continue
                        i1 = int(idx[ci1])
                        i2 = int(idx[ci2])
                        if a_curr[i1] < 1e-14 or a_curr[i2] < 1e-14:
                            continue
                        a_trial = a_curr.copy()
                        a_trial[i1] *= 0.9
                        a_trial[i2] *= 0.9
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-14:
                            a_curr = a_trial
                            best_score = sc
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break

    return a_curr, best_score


def main_optimization(a_init, time_limit, T_start=0.05, T_end=0.001,
                      eta_start=0.01, label=""):
    """Main adaptive mirror descent with backtracking and periodic restarts."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = eta_start
    stuck_count = 0
    last_restart = 0
    restart_interval = 3000
    T = T_start

    for it in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive temperature: decrease slowly
        progress = min(1.0, it / 50000.0)
        T = T_start * (T_end / T_start) ** progress

        # Compute gradient
        grad, b, w = compute_smooth_max_grad(a, T)

        # Project gradient onto tangent space of simplex
        grad_proj = grad - np.dot(a, grad)

        # Backtracking line search
        a_trial, trial_score = backtrack_line_search_mirror(a, grad_proj, eta, T)

        if trial_score < best_score - 1e-14:
            best_score = trial_score
            best_a = a_trial.copy()
            stuck_count = 0
        else:
            stuck_count += 1
            # If stuck, try smaller eta
            eta *= 0.7
            if eta < eta_start * 0.001:
                eta = eta_start * 0.1

        a = a_trial

        # Adaptive eta: increase if making good progress
        if stuck_count == 0:
            eta = min(eta * 1.05, eta_start)

        # Periodic restart to escape local minima
        if it - last_restart > restart_interval and stuck_count > 500:
            rng = np.random.default_rng(it)
            noise_scale = 0.03 * (1.0 + 0.5 * np.sin(it * 0.01))
            noise = rng.standard_normal(n) * noise_scale
            a_restart = best_a + noise
            a_restart = np.maximum(a_restart, 1e-10)
            a_restart = a_restart / a_restart.sum()
            restart_score = score_normalized(a_restart)
            if restart_score < best_score * 1.05:  # accept if not too bad
                a = a_restart
            else:
                a = best_a.copy()
            last_restart = it
            stuck_count = 0
            eta = eta_start * 0.5

        # Periodic spectral shaping restart
        if it % 10000 == 5000 and time.monotonic() - t0 < time_limit * 0.8:
            a_spectral = spectral_shaping(best_a, n_iterations=20, alpha=0.3)
            spec_score = score_normalized(a_spectral)
            if spec_score < best_score:
                best_a = a_spectral.copy()
                best_score = spec_score
                a = a_spectral.copy()
                last_restart = it
                stuck_count = 0

        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} MD iter={it} ({elapsed:.0f}s): T={T:.6f} eta={eta:.6f} "
                  f"score={trial_score:.8f} best={best_score:.8f} stuck={stuck_count}")

    return best_a, best_score


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using the actual evaluate_sequence function."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
