def softmax_from_z(z):

def mirror_descent_optimize(a_init, time_limit, T_start=0.01, T_end=0.0005,

def coordinate_descent(a_init, time_limit, verbose=False, label=""):
    """Coordinate descent with proportional redistribution."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    factors = [0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.3]
    consecutive_no_improve = 0
    max_no_improve = 50
    
    while time.monotonic() - t0 < time_limit and consecutive_no_improve < max_no_improve:
        consecutive_no_improve = 0
        improved_pass = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for f in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                new_val = old_val * f
                if new_val < 0:
                    continue
                a_trial[i] = new_val
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others_sum
                if a_trial.min() < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved_pass = True
                    break
        
        if not improved_pass:
            consecutive_no_improve += 1
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                         options={'presolve': True})
        if result.success and result.x is not None:
            return np.maximum(result.x, 0)
    except Exception:
        pass

    try:
        import cvxpy as cp
        b = cp.Variable(n, nonneg=True)
        constraints = [A @ b <= M]
        objective = cp.Maximize(cp.sum(b))
        prob = cp.Problem(objective, constraints)
        for solver in [cp.ECOS, cp.CBC, cp.GLPK]:
            try:
                prob.solve(solver=solver)
                if prob.status in ['optimal', 'optimal_inaccurate'] and b.value is not None:
                    if np.isfinite(b.value).all():
                        return np.maximum(b.value, 0)
            except Exception:
                continue
    except Exception:
        pass

    return None


def lp_guided_optimize(a_init, time_limit, max_iters=500, verbose=True, label=""):
    """LP-guided optimization with adaptive step size and stall detection."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    last_lp_time = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_conv = fast_convolve(a)
        M = b_conv.max()
        current_score = 2.0 * n * M

        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 30:
            break

        # Solve LP
        lp_t0 = time.monotonic()
        direction = lp_find_direction(a, M)
        last_lp_time = time.monotonic() - lp_t0

        if direction is None:
            break

        dir_sum = direction.sum()
        if dir_sum < 1e-10:
            break

        # Check if direction is basically the same as current (no improvement possible)
        g = direction / dir_sum
        cos_sim = np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-15)
        if cos_sim > 0.9999:
            stall_count += 10
            continue

        # Line search with more granular steps
        best_t = 0.0
        best_t_score = current_score
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            sc = score_normalized(a_trial)
            if sc < best_t_score - 1e-15:
                best_t_score = sc
                best_t = t

        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
        else:
            stall_count += 5

        if verbose and (it % 20 == 0 or it < 5):
            elapsed = time.monotonic() - t0
            print(f"    {label} LP iter={it} ({elapsed:.1f}s, lp={last_lp_time:.3f}s): "
                  f"t={best_t:.3f} cos={cos_sim:.6f} score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def softmax_from_z(z):
    """Convert z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def mirror_descent_optimize(a_init, time_limit, T_start=0.01, T_end=0.0005,
                            eta_start=0.02, eta_end=0.0005, max_iters=30000,
                            verbose=False, label=""):
    """Adaptive mirror descent with temperature annealing."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stuck_count = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (0.7 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        # Softmax smoothing
        scaled = b / T
        m = scaled.max()
        exp_scaled = np.exp(scaled - m)
        w = exp_scaled / exp_scaled.sum()

        # Gradient via cross-correlation
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad_a = 4.0 * n * correl[:n]

        # Multiplicative update
        log_update = -eta * grad_a
        log_update = np.clip(log_update, -2.5, 2.5)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        if stuck_count > 300:
            rng = np.random.default_rng(it * 7919 + 13)
            noise = rng.standard_normal(n) * 0.015
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck_count = 0

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} MD iter={it} ({elapsed:.1f}s): T={T:.5f} "
                  f"score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def peak_shave(a_init, time_limit, verbose=False, label=""):
    """Targeted peak shaving with multi-variable moves."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))

        if b[peak_k] <= best_score / (2.0 * n) + 1e-15:
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:10]

        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7, 0.6, 0.5]:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * factor
                diff = old_val - a_trial[i]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try reducing both i and j simultaneously
            for ci in top_idx[:5]:
                i = int(idx[ci])
                j = peak_k - i
                if i == j or a[j] < 1e-14:
                    continue
                for f1 in [0.9, 0.8, 0.7]:
                    for f2 in [0.9, 0.8, 0.7]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break

    return best_a, best_score


def coordinate_descent(a_init, time_limit, verbose=False, label=""):
    """Coordinate descent with proportional redistribution and bisection refinement."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    coarse_factors = [0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5]
    fine_factors = [0.93, 0.96, 0.99, 1.01, 1.04, 1.07]
    consecutive_no_improve = 0
    max_no_improve = 80

    while time.monotonic() - t0 < time_limit and consecutive_no_improve < max_no_improve:
        consecutive_no_improve = 0
        improved_pass = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for f in coarse_factors + fine_factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                new_val = old_val * f
                if new_val < 0:
                    continue
                a_trial[i] = new_val
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others_sum
                if a_trial.min() < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved_pass = True
                    break

        if not improved_pass:
            consecutive_no_improve += 1

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using the actual evaluate_sequence function."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP-guided direction finding (primary novel method)
    2. Mirror descent with temperature annealing
    3. Peak shaving
    4. Second LP round
    5. Coordinate descent polishing
    6. Alternative length exploration
    7. Final refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(512) / 512
        best_score = score_normalized(best_seq)
        print(f"No parent, starting with uniform n=512, score={best_score:.7f}")

    # === Phase 1: LP-guided optimization (primary) ===
    print(f"\n=== Phase 1: LP-Guided (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 60:
            t_alloc = min(rem * 0.55, 450.0)
            print(f"  LP1: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_lp, _ = lp_guided_optimize(best_seq, t_alloc, max_iters=500, verbose=True, label="LP1")
            best_seq, best_score = check_and_update(a_lp, "LP1", best_seq, best_score)

    # === Phase 2: Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.7f}) ===")
    if best_seq is not None:
        configs = [
            (0.01, 0.001, 0.02, 0.0005, "MD1"),
            (0.005, 0.0003, 0.01, 0.0002, "MD2"),
        ]
        for T_s, T_e, eta_s, eta_e, lbl in configs:
            rem = deadline - time.monotonic() - 350
            if rem < 20:
                break
            t_alloc = min(rem * 0.25, 150.0)
            print(f"  {lbl}: budget={t_alloc:.0f}s")
            a_md, _ = mirror_descent_optimize(
                best_seq, t_alloc, T_start=T_s, T_end=T_e,
                eta_start=eta_s, eta_end=eta_e, label=lbl
            )
            best_seq, best_score = check_and_update(a_md, lbl, best_seq, best_score)

    # === Phase 3: Peak Shaving ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 15:
            t_alloc = min(rem * 0.3, 100.0)
            print(f"  PS: budget={t_alloc:.0f}s")
            a_ps, _ = peak_shave(best_seq, t_alloc)
            best_seq, best_score = check_and_update(a_ps, "PS", best_seq, best_score)

    # === Phase 4: Second LP round ===
    print(f"\n=== Phase 4: LP Round 2 (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 60:
            t_alloc = min(rem * 0.45, 300.0)
            print(f"  LP2: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_lp2, _ = lp_guided_optimize(best_seq, t_alloc, max_iters=400, verbose=True, label="LP2")
            best_seq, best_score = check_and_update(a_lp2, "LP2", best_seq, best_score)

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_alloc = min(rem * 0.4, 80.0)
            print(f"  CD: budget={t_alloc:.0f}s")
            a_cd, _ = coordinate_descent(best_seq, t_alloc)
            best_seq, best_score = check_and_update(a_cd, "CD", best_seq, best_score)

    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.7f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [1.25, 1.5, 0.8, 2.0, 0.7, 1.33, 0.6, 1.75]:
            alt_n = int(round(n_best * factor))
            if 64 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in sorted(alt_lengths, key=lambda x: abs(x - n_best))[:4]:
            rem = deadline - time.monotonic() - 80
            if rem < 30:
                break
            print(f"  Trying n={alt_n} (rem={rem:.0f}s)")
            t_alloc = min(rem * 0.25, 50.0)
            init_alt = interpolate_sequence(best_seq, alt_n)

            # Quick MD
            a_md_alt, _ = mirror_descent_optimize(
                init_alt, t_alloc, T_start=0.01, T_end=0.001,
                eta_start=0.02, eta_end=0.0005, label=f"alt_{alt_n}"
            )
            best_seq, best_score = check_and_update(a_md_alt, f"alt_{alt_n}", best_seq, best_score)

            # Quick LP on best alt
            if len(best_seq) == alt_n:
                rem = deadline - time.monotonic() - 50
                if rem > 30:
                    t_alloc_lp = min(rem * 0.5, 40.0)
                    a_lp_alt, _ = lp_guided_optimize(
                        best_seq, t_alloc_lp, max_iters=50, verbose=False, label=f"LP_{alt_n}"
                    )
                    best_seq, best_score = check_and_update(a_lp_alt, f"LP_{alt_n}", best_seq, best_score)

    # === Phase 7: Final refinement ===
    print(f"\n=== Phase 7: Final Refinement (best={best_score:.7f}) ===")
    if best_seq is not None:
        # Final LP
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem * 0.5, 50.0)
            a_fin, _ = lp_guided_optimize(best_seq, t_alloc, max_iters=80, verbose=False, label="FinLP")
            best_seq, best_score = check_and_update(a_fin, "FinLP", best_seq, best_score)

        # Final peak shave
        rem = deadline - time.monotonic() - 15
        if rem > 8:
            a_fin_ps, _ = peak_shave(best_seq, min(rem * 0.6, 20.0))
            best_seq, best_score = check_and_update(a_fin_ps, "FinPS", best_seq, best_score)

        # Final CD
        rem = deadline - time.monotonic() - 8
        if rem > 5:
            a_fin_cd, _ = coordinate_descent(best_seq, min(rem * 0.7, 10.0))
            best_seq, best_score = check_and_update(a_fin_cd, "FinCD", best_seq, best_score)

    # === Phase 8: One more LP pass if time allows ===
    print(f"\n=== Phase 8: Final LP Pass (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 15:
            a_last, _ = lp_guided_optimize(best_seq, min(rem * 0.9, 20.0),
                                           max_iters=30, verbose=False, label="LastLP")
            best_seq, best_score = check_and_update(a_last, "LastLP", best_seq, best_score)

    # Fallback
    if best_seq is None:
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
