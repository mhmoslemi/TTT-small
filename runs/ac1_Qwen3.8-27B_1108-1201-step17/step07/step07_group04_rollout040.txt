def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def get_lp_solver():

            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=s)

                _lp_solver = s
                return s

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a

            i_lo = 0
            i_hi = min(n - 1, 2 * n - 2 - j)
            if i_hi >= i_lo:
                grad_a[i_lo:i_hi + 1] += 2.0 * a[j] * w[i_lo + j:i_hi + 1 + j]
        # Transform to z-space: d/dz = a * (grad_a - a * dot(grad_a, a))
        S = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - a * S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0
    except Exception:
        a_out = a0
    return a_out

def subgradient_optimize(a0, time_limit, lr_init=0.003, momentum=0.9):
    """Subgradient method with momentum and adaptive peak weighting."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    v = np.zeros(n)
    lr = lr_init
    t0 = monotonic()
    no_improve = 0
    step = 0
    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        # Focus on top peaks with exponential weighting
        top_k = max(20, n // 8)
        peaks = np.argpartition(b, -top_k)[-top_k:]
        pv = b[peaks]
        temp = 0.003 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        w = w / w.sum()
        sg = np.zeros(n)
        for ii in range(len(peaks)):
            if w[ii] < 1e-12:
                continue
            k = int(peaks[ii])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[ii] * a[k - idx]
        v = momentum * v - lr * sg
        a_candidate = a + v
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_candidate = np.maximum(a_candidate, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            a_candidate = np.ones(n) / n
        else:
            a_candidate = a_candidate / sc
        score = score_of(a_candidate)
        step += 1
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_candidate.copy()
            a = a_candidate
            no_improve = 0
        else:
            no_improve += 1
        lr *= 0.9995
        if no_improve > 400:
            a = best_a + np.random.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 2, 0.002)
            no_improve = 0
        if step % 2000 == 0:
            print(f"    SubGrad ({monotonic()-t0:.0f}s): score={best_score:.8f} lr={lr:.6f}")
    return best_a, best_score

def peak_shave(a0, time_limit):
    """Aggressive local peak shaving targeting top contributors."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    t0 = monotonic()
    failed = 0
    while monotonic() - t0 < time_limit and failed < 80:
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        if b[peak_k] <= best_score / (2.0 * n) + 1e-15:
            break
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total_contrib = contribs.sum()
        if total_contrib < 1e-20:
            break
        top_c = np.argsort(contribs)[::-1][:8]
        improved = False
        # Strategy 1: Reduce single top contributor
        for ci in top_c:
            i = int(idx[ci])
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.8, 0.7, 0.6]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                a_trial = a_trial / st
                score = score_of(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        # Strategy 2: Reduce pair
        if not improved:
            for ci in top_c[:5]:
                i = int(idx[ci])
                j = int(peak_k - i)
                if j == i or j < 0 or j >= n or a[j] < 1e-14:
                    continue
                for f1 in [0.9, 0.85, 0.8]:
                    for f2 in [0.9, 0.85, 0.8]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial = a_trial / st
                        score = score_of(a_trial)
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        # Strategy 3: Small random perturbation
        if not improved:
            for _ in range(8):
                a_trial = a + np.random.standard_normal(n) * 0.0008
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                score = score_of(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        if improved:
            failed = 0
        else:
            failed += 1
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n + 1)
    f_ext = np.append(f_norm, 0)
    new_x = np.linspace(0, 1, new_n + 1)
    f_interp = np.interp(new_x[:-1], old_x, f_ext)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, label, best_seq, best_score):
    """Evaluate candidate and update best if improved."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
