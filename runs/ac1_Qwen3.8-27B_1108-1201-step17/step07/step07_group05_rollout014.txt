"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play
combined with L-BFGS smoothing and coordinate descent polish.
The LP-fictitious play is the main novelty: at each iteration, we solve
an LP to find the direction that maximizes sum(b) subject to a*b <= M,
which naturally equalizes the auto-correlation peaks.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)


def smoothed_objective(z, T, n):
    """Soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    log_sum_exp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * log_sum_exp)


def smoothed_gradient(z, T, n):
    """Gradient of the soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lbgfs_optimize(a0, time_limit, T, maxiter=5000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    result = {'a': a0.copy(), 'score': compute_score(a0)}

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z)
        return smoothed_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 40, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < result['score']:
            result['a'] = a_out
            result['score'] = score
    except Exception as e:
        pass

    result['seconds'] = time.monotonic() - t0
    return result


def build_lp_matrix(a):
    """Build the constraint matrix for the LP.
    A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    This gives (a*b)[k] = A[k,:] @ b
    """
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A[k, j_start:j_end+1] = a[k - j_end : k - j_start + 1][::-1]
    return A


def lp_fictitious_play(a0, time_limit, t_init=0.3, t_decay=0.9995, min_t=0.005):

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    t = t_init
    iters = 0
    
    # Pre-build the LP matrix (it changes with a, so rebuild each time)
    # Actually the matrix depends on a, so we need to rebuild each iteration
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Build LP matrix: A_ub[k, j] = a[k-j] if valid

        for k in range(2*n-1):
            j_start = max(0, k - n + 1)

def coordinate_descent(a0, time_limit, n_pairs_per_iter=300):
    """Coordinate descent using pairwise moves."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        for _ in range(n_pairs_per_iter):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue

            base = min(a[i], max(a[j], 1e-8))
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                t = frac * base
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a_trial.copy()
                    improved = True
                    break

        if not improved and iters > 100:
            noise = rng.standard_normal(n) * 0.002
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_score = compute_score(a)
            best_a = a.copy()

    return best_a, best_score


def peak_equalization(a0, time_limit, n_peaks=10):
    """Iteratively try to equalize the top peaks of the auto-correlation."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_idx = int(np.argmax(conv))
        max_peak = conv[max_idx]
        
        # All peaks above 0.999 * max
        active_peaks = np.where(conv > 0.999 * max_peak)[0]
        
        if len(active_peaks) < 2:
            break
        
        # For the max peak, find contributing pairs
        js = max(0, max_idx - n + 1)
        je = min(n - 1, max_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_idx - idx]
        top_contrib = np.argsort(contribs)[::-1][:3]

        improved = False
        for ci in top_contrib:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for frac in [0.05, 0.1, 0.2]:
                delta = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= delta
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                os = others.sum()
                if os < 1e-15:
                    continue
                a_trial[mask] = others + delta * others / os
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Random perturbation
            i = rng.integers(0, n)
            if a[i] > 1e-14:
                a_trial = a.copy()
                delta = a[i] * 0.05 * rng.standard_normal()
                a_trial[i] -= delta
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                os = others.sum()
                if os > 1e-15:
                    a_trial[mask] = others + delta * others / os
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play as the
main workhorse, combined with L-BFGS soft-max smoothing and coordinate descent.
The key innovation is the LP-fictitious play: at each iteration, we solve an LP
to find the direction b that maximizes sum(b) subject to cross-correlation
a*b <= M (current peak), then move toward b. This naturally equalizes the
auto-correlation peaks toward the theoretical minimum.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)

    """Soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T

    log_sum_exp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * log_sum_exp)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lbgfs_optimize(a0, time_limit, T, maxiter=5000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    result = {'a': a0.copy(), 'score': compute_score(a0)}

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < result['score']:
            result['a'] = a_out
            result['score'] = score
    except Exception:
        pass

    result['seconds'] = time.monotonic() - t0
    return result


def build_conv_matrix(a):
    """Build matrix A such that A @ b = a * b (cross-correlation).
    A[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def lp_fictitious_play(a0, time_limit, t_init=0.4, t_decay=0.9997, min_t=0.005,
                       restart_interval=200):
    """
    LP-based fictitious play to equalize auto-correlation peaks.
    At each iteration: solve LP max sum(b) s.t. a*b <= M, b >= 0,
    then update a = (1-t)*a + t*b.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0), 0
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    t = t_init
    iters = 0
    no_improve = 0
    rng = np.random.default_rng(2024)

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        if M < 1e-15:
            break

        A_ub = build_conv_matrix(a)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          bounds=(0, None), method='highs',
                          options={'presolve': True, 'time_limit': 5.0})
            if res.success and res.x is not None:
                b = np.maximum(res.x, 0)
                b_sum = b.sum()
                if b_sum > 1e-15:
                    b = b / b_sum
                    # Try the step
                    a_new = (1.0 - t) * a + t * b
                    a_new = np.maximum(a_new, 0)
                    ss = a_new.sum()
                    if ss > 1e-15:
                        a_new = a_new / ss
                        sc = compute_score(a_new)
                        if sc < best_score - 1e-15:
                            best_score = sc
                            best_a = a_new.copy()
                            no_improve = 0
                        a = a_new
                        if sc >= best_score:
                            no_improve += 1
                        else:
                            no_improve = 0
            else:
                no_improve += 1
        except Exception:
            pass

        t *= t_decay
        t = max(t, min_t)

        # Restart if stuck for too long
        if no_improve > restart_interval:
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss > 1e-15:
                a = a / ss
            no_improve = 0
            t = max(t, 0.1)

        if iters % 100 == 0:
            elapsed = time.monotonic() - t0
            remaining = time_limit - elapsed
            print(f"  LP-FP iter={iters}, t={t:.4f}, best={best_score:.8f}, "
                  f"elapsed={elapsed:.0f}s, rem={remaining:.0f}s")

    return best_a, best_score, iters


def coordinate_descent(a0, time_limit, n_pairs_per_iter=400):
    """Coordinate descent with pairwise transfer moves."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        for _ in range(n_pairs_per_iter):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue

            base = min(a[i], max(a[j], 1e-8))
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9, 0.02]:
                t = frac * base
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a_trial.copy()
                    no_improve = 0
                    break
            else:
                no_improve += 1

        if no_improve > 50:
            a = best_a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_score = compute_score(a)
            best_a = a.copy()
            no_improve = 0

    return best_a, best_score


def peak_equalization(a0, time_limit, n_peaks=10):
    """Iteratively reduce the maximum auto-correlation peak."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        js = max(0, max_idx - n + 1)
        je = min(n - 1, max_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_idx - idx]
        top_contrib = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_contrib:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for frac in [0.03, 0.05, 0.1, 0.15]:
                delta = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= delta
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                os = others.sum()
                if os < 1e-15:
                    continue
                a_trial[mask] = others + delta * others / os
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            i = rng.integers(0, n)
            if a[i] > 1e-14:
                a_trial = a.copy()
                delta = a[i] * 0.05 * rng.standard_normal()
                a_trial[i] -= delta
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                os = others.sum()
                if os > 1e-15:
                    a_trial[mask] = others + delta * others / os
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate via the true evaluator and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization using LP-based fictitious play as the core
    method, with L-BFGS and coordinate descent for refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized uniform: n={n_init}, score={best_score:.8f}")

    print(f"\nOptimization starting: n={n_parent}, score={best_score:.8f}")
    print(f"Time budget: {budget_s:.0f}s, deadline in {deadline - time.monotonic():.0f}s\n")

    # === Phase 1: Quick L-BFGS initialization (60s) ===
    rem = deadline - time.monotonic()

    # === Phase 3: L-BFGS refinement (250s) ===

    for T_val in [1000, 300, 100]:

        if rem < 40:

        t_alloc = min(80.0, rem * 0.12)

        res = lbgfs_optimize(best_seq, t_alloc, T=T_val, maxiter=5000)
        best_seq, best_score = check_and_update(res['a'], f"lbfgs3_T{T_val}",
                                                best_seq, best_score)

    # === Phase 4: LP Fictitious Play Round 2 (400s) ===

    print(f"\n{'='*60}")

    print(f"{'='*60}")
    return [float(x) for x in best_seq]
