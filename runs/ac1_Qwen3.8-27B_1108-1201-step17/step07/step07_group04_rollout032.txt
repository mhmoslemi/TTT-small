"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using adaptive-temperature L-BFGS,

- Multi-temperature L-BFGS with adaptive schedule

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score_norm(a):
    """Score for normalized sequence (sum=1)."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving non-negativity and normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def lse_objective(z, T, n, deadline):
    """Soft-max objective in log-space for scipy minimize."""
    if monotonic() > deadline:
        return 1e15, None
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    exp_b = np.exp(scaled)
    lse = max_b + T * np.log(exp_b.sum())
    return 2.0 * n * lse / T


def lse_gradient(z, T, n, deadline):
    """Gradient of soft-max objective in log-space."""
    if monotonic() > deadline:
        return np.zeros(n)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    exp_b = np.exp(scaled)
    w = exp_b / exp_b.sum()

    # g[i] = (2n/T) * sum_k w[k] * (a[k-i] + a[i-k]) with bounds
    # Compute via direct summation (O(n^2) but n~768 so fast)
    grad_a = np.zeros(n)
    nz = np.nonzero(w > 1e-14)[0]
    for k in nz:
        wk = w[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * wk * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    grad_z *= 2.0 * n / T
    return grad_z

    """Run L-BFGS in log-space with given temperature T."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-25)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    deadline = monotonic() + time_limit

            x0=z0,

    a = np.maximum(a0.copy(), 1e-20)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_norm(a)
    t0 = monotonic()
    no_improve = 0
    rng = np.random.RandomState(123)

    for pass_num in range(max_passes):
        if monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_val = b.max()
        peak_k = int(np.argmax(b))

        # If we've essentially reached the uniform lower bound, stop
        if peak_val <= 1.0 / (2 * n - 1) + 1e-12:
            break

        improved = False
        # Process coordinates in random order
        order = rng.permutation(n)

        for i in order:
            if monotonic() - t0 > time_limit:
                break
            if no_improve > 2000:
                break

            if a[i] < 1e-15:
                continue

            # First-order analysis: d[k] = a[k-i] + a[i-k] - 2*b[k]
            # Change in b[k] ≈ delta * d[k]
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                val = 0.0
                ki = k - i
                if 0 <= ki < n:
                    val += a[ki]
                ik = i - k
                if 0 <= ik < n:
                    val += a[ik]
                d[k] = val - 2.0 * b[k]

            d_peak = d[peak_k]
            if abs(d_peak) < 1e-14:
                # Try small positive and negative steps
                for delta_frac in [0.01, -0.01, 0.02, -0.02]:
                    delta = delta_frac * a[i]
                    if a[i] + delta < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[i] += delta
                    a_trial = a_trial / a_trial.sum()
                    score = compute_score_norm(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        no_improve = 0
                        break
                continue

            # Determine optimal direction
            if d_peak < 0:
                # Increasing a[i] reduces the peak
                # Find max delta such that no other peak exceeds current max
                delta_max = a[i] * 10  # generous upper bound
                for k in range(2 * n - 1):
                    if d[k] > 1e-14 and b[k] < peak_val:
                        bound = (peak_val - b[k]) / d[k]
                        if bound < delta_max:
                            delta_max = bound
                # Try a fraction of delta_max
                for frac in [0.5, 0.3, 0.1, 0.05]:
                    delta = frac * delta_max
                    if delta > a[i]:
                        delta = a[i] * 0.5
                    a_trial = a.copy()
                    a_trial[i] += delta
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = compute_score_norm(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        no_improve = 0
                        break
            else:
                # d_peak > 0: decreasing a[i] reduces the peak
                delta_min = -a[i] * 0.9  # can't go below 0
                for k in range(2 * n - 1):
                    if d[k] < -1e-14 and b[k] < peak_val:
                        bound = (peak_val - b[k]) / d[k]
                        if bound > delta_min:
                            delta_min = bound
                for frac in [0.5, 0.3, 0.1, 0.05]:
                    delta = frac * delta_min
                    if delta < -a[i]:
                        delta = -a[i] * 0.5
                    a_trial = a.copy()
                    a_trial[i] += delta
                    a_trial = np.maximum(a_trial, 0)
                    if a_trial.sum() < 1e-15:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = compute_score_norm(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        no_improve = 0
                        break

        if not improved:
            no_improve += n
            # Periodic perturbation to escape local minima
            if no_improve > 5000:
                a = best_a + rng.standard_normal(n) * 0.001
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0

        if pass_num % 100 == 0:
            elapsed = monotonic() - t0
            print(f"    CD pass={pass_num} ({elapsed:.0f}s): score={best_score:.8f} peak={peak_val:.6e}")

    return best_a, best_score


def peak_shave_refined(a0, time_limit):
    """
    Refined peak shaving: identifies top contributing pairs to the peak
    and makes targeted adjustments.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 1e-20)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_norm(a)
    t0 = monotonic()
    stale = 0

    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        # Find top 3 peaks
        top_peaks = np.argsort(b)[::-1][:3]

        improved = False
        for pk in top_peaks:
            if monotonic() - t0 > time_limit:
                break
            pk = int(pk)
            if b[pk] < best_score / (2.0 * n) * 0.999:
                continue

            # Find contributing pairs
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]

            if contribs.sum() < 1e-20:
                continue

            # Sort contributions descending
            top_idx = np.argsort(contribs)[::-1][:8]

            for ci in top_idx:
                i = int(idx[ci])
                j = pk - i
                if j < 0 or j >= n or j == i:
                    continue
                # Try reducing a[i] slightly
                for f in [0.97, 0.95, 0.93]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial = a_trial / a_trial.sum()
                    score = compute_score_norm(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        stale = 0
                        break
                if improved:
                    break

                # Try reducing a[j] slightly
                for f in [0.97, 0.95, 0.93]:
                    a_trial = a.copy()
                    a_trial[j] *= f
                    a_trial = a_trial / a_trial.sum()
                    score = compute_score_norm(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        stale = 0
                        break
                if improved:
                    break

            if improved:
                break

        if not improved:
            stale += 1
            if stale > 20:
                # Random perturbation
                a = best_a + np.random.standard_normal(n) * 0.0005
                a = np.maximum(a, 0)
                a = a / a.sum()
                stale = 0
            elif stale > 100:
                break

    return best_a, best_score


def multiplicative_update(a0, time_limit, eta=0.01):
    """
    Multiplicative (Weiszfeld-like) update for peak reduction.
    Updates a[i] *= exp(-eta * contribution_to_peak) then renormalizes.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_norm(a)
    t0 = monotonic()
    it = 0

    while monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        max_b = b.max()
        # Soft weights over peaks
        T_local = 0.01 * max_b
        scaled = (b - max_b) / T_local
        w = np.exp(scaled)
        w = w / w.sum()

        # Compute pressure on each a[i]
        pressure = np.zeros(n)
        nz = np.nonzero(w > 1e-12)[0]
        for k in nz:
            wk = w[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            pressure[idx] += wk * a[k - idx]

        # Multiplicative update
        a_new = a * np.exp(-eta * pressure / (pressure.max() + 1e-20))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        score = compute_score_norm(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step size
            eta *= 0.9

        if it % 500 == 0:
            print(f"    MU iter={it}: score={best_score:.8f} eta={eta:.5f}")

        if eta < 1e-6:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Multi-length quick evaluation with structural priors
    2. Adaptive-temperature L-BFGS (increasing T)
    3. Peak-aware coordinate descent
    4. Multiplicative update refinement
    5. Final peak shaving
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = float(evaluate_sequence(best_seq.tolist()))
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial: n=768, score={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 1: Multi-length exploration (60s) ===
    print(f"\n=== Phase 1: Multi-length exploration ===")
    n_candidates = sorted(set([
        n_base, n_base - 32, n_base + 32, n_base - 64, n_base + 64,
        n_base - 128, n_base + 128, 640, 800, 960
    ]), key=lambda x: abs(x - n_base))
    n_candidates = [n for n in n_candidates if 256 <= n <= 2048][:5]

    t_p1 = min(60.0, (deadline - monotonic()) * 0.08)
    t_per_n = t_p1 / len(n_candidates)

    for n_try in n_candidates:
        if monotonic() > deadline - 1200:
            break
        if n_try == n_base:
            init = best_seq.copy()
        else:
            init = interpolate_seq(best_seq, n_try)

        # Quick L-BFGS with moderate temperature
        a_opt = lbgfs_run(init, t_per_n, T=100, maxiter=2000)
        s = a_opt.sum()
        if s > 1e-10:
            a_opt = a_opt / s
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  Phase1 n={n_try}: NEW BEST {best_score:.8f}")
            else:
                print(f"  Phase1 n={n_try}: score={sc:.8f}")

    # === Phase 2: Adaptive-temperature L-BFGS (250s) ===
    print(f"\n=== Phase 2: Adaptive L-BFGS (best={best_score:.8f}) ===")
    t_p2 = min(250.0, (deadline - monotonic()) * 0.18)
    temps = [10, 25, 50, 100, 200, 400, 800]
    t_per_T = t_p2 / len(temps)

    for T in temps:
        if monotonic() > deadline - 800:
            break
        a_opt = lbgfs_run(best_seq, t_per_T, T=T, maxiter=2000)
        s = a_opt.sum()
        if s > 1e-10:
            a_opt = a_opt / s
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  L-BFGS T={T}: NEW BEST {best_score:.8f}")
            else:
                print(f"  L-BFGS T={T}: score={sc:.8f}")

    # === Phase 3: Coordinate descent (700s) ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.8f}) ===")
    t_p3 = min(700.0, (deadline - monotonic()) * 0.45)
    a_cd, score_cd = coordinate_descent(best_seq, t_p3)
    sc = float(evaluate_sequence(a_cd.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_cd.copy()
        print(f"  CD: NEW BEST {best_score:.8f}")
    else:
        print(f"  CD: score={sc:.8f}")

    # === Phase 4: Multiplicative update (200s) ===
    print(f"\n=== Phase 4: Multiplicative Update (best={best_score:.8f}) ===")
    t_p4 = min(200.0, (deadline - monotonic()) * 0.15)
    if t_p4 > 10:
        a_mu, score_mu = multiplicative_update(best_seq, t_p4, eta=0.02)
        sc = float(evaluate_sequence(a_mu.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_mu.copy()
            print(f"  MU: NEW BEST {best_score:.8f}")
        else:
            print(f"  MU: score={sc:.8f}")

    # === Phase 5: Second round of coordinate descent (300s) ===
    print(f"\n=== Phase 5: CD Round 2 (best={best_score:.8f}) ===")
    t_p5 = min(300.0, (deadline - monotonic()) * 0.25)
    if t_p5 > 10:
        a_cd2, score_cd2 = coordinate_descent(best_seq, t_p5)
        sc = float(evaluate_sequence(a_cd2.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_cd2.copy()
            print(f"  CD2: NEW BEST {best_score:.8f}")
        else:
            print(f"  CD2: score={sc:.8f}")

    # === Phase 6: Final peak shaving (100s) ===
    print(f"\n=== Phase 6: Final Peak Shave (best={best_score:.8f}) ===")
    t_p6 = min(100.0, (deadline - monotonic()) * 0.12)
    if t_p6 > 5:
        a_ps, score_ps = peak_shave_refined(best_seq, t_p6)
        sc = float(evaluate_sequence(a_ps.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_ps.copy()
            print(f"  PS: NEW BEST {best_score:.8f}")
        else:
            print(f"  PS: score={sc:.8f}")

    # === Phase 7: Final L-BFGS polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    t_p7 = min(80.0, (deadline - monotonic()) - 10)
    if t_p7 > 10:
        for T_final in [500, 2000]:
            if monotonic() > deadline - 15:
                break
            a_f = lbgfs_run(best_seq, max(5, t_p7 / 2), T=T_final, maxiter=1500)
            s = a_f.sum()
            if s > 1e-10:
                a_f = a_f / s
                sc = float(evaluate_sequence(a_f.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_f.copy()
                    print(f"  Final T={T_final}: NEW BEST {best_score:.8f}")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    best_seq = np.maximum(best_seq, 0.0)
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
