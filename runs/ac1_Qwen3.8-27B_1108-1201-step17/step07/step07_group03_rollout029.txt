"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based fictitious play for global optimization
2. Smoothed gradient (log-sum-exp) with projected gradient method
3. Exact 1D coordinate descent with ternary search
4. Multi-length exploration

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2"""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    return 2.0 * len(a_norm) * fast_convolve(a_norm).max()

def simplex_project(v):
    """Project vector v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():
    import cvxpy as cp
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.PDLP, cp.SCS, cp.GLPK, cp.MOSEK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """Alternating LP best responses."""
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        a_avg = (iters * a_avg + a) / (iters + 1)

        for cand in [a, b, 0.5*(a+b), a_avg]:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand = cand / s
            sc = 2.0 * n * fast_convolve(cand).max()
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        if verbose and iters % 50 == 0:
            print(f"    FP it={iters} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters

def smoothed_gradient(a, t):
    """Compute gradient of smoothed max: (1/t)log(sum exp(t*c_k)) where c = conv(a,a)."""
    n = len(a)
    c = fast_convolve(a)
    c_shift = c - c.max()
    exp_t = np.exp(t * c_shift)
    p = exp_t / exp_t.sum()
    
    # grad = 2 * sum_k p[k] * a[k-m] (valid range)
    # = 2 * (conv(a_flip, p))[n-1 : 2n-1]
    a_flip = a[::-1]
    conv_ap = irfft(rfft(a_flip, 1<<(3*n-1).bit_length()) * rfft(p, 1<<(3*n-1).bit_length()), 1<<(3*n-1).bit_length())[:3*n-2]
    
    # Actually let me just use np.convolve for clarity (slower but correct)
    # conv_ap_full = np.convolve(a_flip, p, mode='full')  # length 3n-2
    # result = conv_ap_full[n-1 : 2*n-1]
    
    # Let me use FFT properly
    L = 1 << (3*n - 2 - 1).bit_length()
    A = rfft(a_flip, L)
    P = rfft(p, L)
    conv_full = irfft(A * P, L)[:3*n-2]
    result = conv_full[n-1:2*n-1]
    
    return 2.0 * result, p, c

def projected_gradient_smoothed(a0, time_limit, t_init=0.05, t_decay=0.8, verbose=False):
    """Projected gradient on the log-sum-exp smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    t = t_init
    lr = 0.01
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, p, c = smoothed_gradient(a, t)
        
        # Projected gradient step
        a_new = simplex_project(a - lr * grad)
        
        # Check improvement
        sc = 2.0 * n * fast_convolve(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        # Adaptive learning rate
        if it % 200 == 0:
            lr *= 0.95
        if it % 2000 == 0:
            t *= t_decay
        
        a = a_new
    
    return best_a, best_score

def coord_descent_1d(a0, time_limit, n_sweeps=100, verbose=False):
    """
    Coordinate descent where for each coordinate, we solve the exact 1D
    subproblem using ternary search on [0,1].
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            
            x_old = a[i]
            if x_old < 1e-14:
                continue
            
            # b = (a - x_old * e_i) / (1 - x_old)
            b = a.copy()
            b[i] = 0
            b = b / (1 - x_old)  # sum(b) = 1
            
            c_b = fast_convolve(b)  # length 2n-1
            
            # Quadratic coefficients for each lag k:
            # g_k(x) = x^2 * delta_{k,2i} + 2x(1-x)*b[k-i] + (1-x)^2 * c_b[k]
            # = x^2*(delta + c_b[k] - 2*b[k-i]) + x*(2*b[k-i] - 2*c_b[k]) + c_b[k]
            nai = np.clip(np.arange(len(c_b)), 0, n-1)  # for b[k-i]
            # b[k-i] is valid when 0 <= k-i <= n-1
            b_shifted = np.zeros(len(c_b))
            valid = (np.arange(len(c_b)) - i >= 0) & (np.arange(len(c_b)) - i < n)
            b_shifted[valid] = b[np.arange(len(c_b))[valid] - i]
            
            A_coeff = c_b.copy()  # c_b[k] part
            A_coeff[2*i] += 1.0  # delta_{k,2i}
            A_coeff -= 2.0 * b_shifted
            
            B_coeff = 2.0 * b_shifted - 2.0 * c_b
            C_coeff = c_b.copy()
            
            # f(x) = max_k (A_coeff[k]*x^2 + B_coeff[k]*x + C_coeff[k])
            
            def eval_f(x):
                vals = A_coeff * x * x + B_coeff * x + C_coeff
                return vals.max()
            
            # Ternary search on [0, 1]
            lo, hi = 0.0, 1.0
            for _ in range(60):
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                if eval_f(m1) < eval_f(m2):
                    hi = m2
                else:
                    lo = m1
            
            x_star = (lo + hi) / 2
            
            # Update
            a_new = (1 - x_star) * b
            a_new[i] = x_star
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                improved = True
        
        if not improved and sweep > 3:
            break
        if verbose and sweep % 10 == 0:
            print(f"    CD sweep={sweep} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based fictitious play for global optimization
2. Exact 1D coordinate descent with ternary search (solves each 1D subproblem optimally)
3. Log-sum-exp smoothed projected gradient for additional refinement
4. Multi-length exploration and iterative refinement
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    return 2.0 * len(a_norm) * fast_convolve(a_norm).max()


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    import cvxpy as cp
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.PDLP, cp.SCS, cp.GLPK, cp.MOSEK, cp.GUROBI]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        a_avg = (iters * a_avg + a) / (iters + 1)

        for cand in [a, b, 0.5 * (a + b), a_avg]:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand = cand / s
            sc = 2.0 * n * fast_convolve(cand).max()
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        if verbose and iters % 30 == 0:
            print(f"    FP it={iters} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def smoothed_subgradient(a, t):
    """Compute subgradient-like direction for smoothed max of conv(a,a)."""
    n = len(a)
    c = fast_convolve(a)
    c_shift = c - c.max()
    exp_t = np.exp(t * c_shift)
    p = exp_t / exp_t.sum()

    a_flip = a[::-1]
    L = 1 << (3 * n - 2 - 1).bit_length()
    A = rfft(a_flip, L)
    P = rfft(p, L)
    conv_full = irfft(A * P, L)[:3 * n - 2]
    result = conv_full[n - 1:2 * n - 1]
    return 2.0 * result, p, c


def projected_gradient_smoothed(a0, time_limit, t_init=0.03, t_decay=0.75, verbose=False):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()

        a = a / s

    t0 = time.monotonic()
    t = t_init
    lr = 0.005
    it = 0
    stalling = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, p, c = smoothed_subgradient(a, t)
        a_new = simplex_project(a - lr * grad)
        sc = 2.0 * n * fast_convolve(a_new).max()
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            stalling = 0
        else:
            stalling += 1
        a = a_new

        if stalling > 100:
            lr *= 0.8
            stalling = 0
        if it % 1000 == 0:
            t *= t_decay
            lr = max(lr, 0.001)

        if verbose and it % 500 == 0:
            print(f"    PG it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.6f} t={t:.4f}")

    return best_a, best_score


def coord_descent_1d(a0, time_limit, n_sweeps=200, verbose=False):
    """
    Exact 1D coordinate descent: for each coordinate i, solve the 1D
    subproblem via ternary search on [0,1].
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = 0
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[i]
            if x_old < 1e-14:
                continue
            if x_old > 1.0 - 1e-10:
                continue

            b_rest = a.copy()
            b_rest[i] = 0.0
            b_rest = b_rest / (1.0 - x_old)

            c_b = fast_convolve(b_rest)
            len_c = len(c_b)

            b_shifted = np.zeros(len_c)
            k_arr = np.arange(len_c)
            valid = (k_arr - i >= 0) & (k_arr - i < n)
            b_shifted[valid] = b_rest[k_arr[valid] - i]

            A_c = c_b - 2.0 * b_shifted
            if 2 * i < len_c:
                A_c[2 * i] += 1.0
            B_c = 2.0 * b_shifted - 2.0 * c_b
            C_c = c_b

            lo = 0.0
            hi = 1.0
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(A_c * m1 * m1 + B_c * m1 + C_c)
                f2 = np.max(A_c * m2 * m2 + B_c * m2 + C_c)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

            x_star = (lo + hi) / 2.0
            if abs(x_star - x_old) < 1e-10:
                continue

            a_new = (1.0 - x_star) * b_rest
            a_new[i] = x_star
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new

            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                improved += 1

        if verbose and sweep % 5 == 0:
            print(f"    CD sweep={sweep} ({time.monotonic()-t0:.0f}s): best={best_score:.6f} imp={improved}")

        if improved == 0 and sweep > 5:
            break

    return best_a, best_score


def pairwise_coord_descent(a0, time_limit, n_passes=20, verbose=False):
    """Try pairwise adjustments for the top contributing coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        top_k = min(8, len(c))
        peaks = np.argpartition(c, -top_k)[-top_k:]

        for pk in peaks:
            if time.monotonic() - t0 > time_limit:
                break
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top_ci = np.argsort(contribs)[::-1][:3]

            for ci in top_ci:
                i = int(idx[ci])
                j = pk - i
                if i == j:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue

                for f1 in [0.7, 0.8, 0.9, 1.1, 1.2, 1.3]:
                    for f2 in [0.7, 0.8, 0.9, 1.1, 1.2, 1.3]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        sc = 2.0 * n * fast_convolve(a_trial).max()
                        if sc < best_score - 1e-13:
                            best_score = sc
                            best_a = a_trial.copy()
                            a = a_trial

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f
