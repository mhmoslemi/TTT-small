Uses: (1) diverse initialization, (2) L-BFGS with analytic gradient on 
log-sum-exp smoothed objective, (3) coordinate descent with 1D minimization,
(4) multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_gradient_z(z, T):
    """Compute gradient of LSE_T objective in z-space.
    a = softmax(z), objective = T * log(sum exp(c/T)) + c.max()
    Actually let's use: objective = T * log(sum exp((c - c_max)/T)) + c_max
    which equals the LSE (log-sum-exp) of c.
    """
    n = len(z)
    # Compute a from z
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    # Compute auto-correlation
    c = fast_convolve(a)
    
    # Compute weights for LSE
    c_shifted = (c - c.max()) / T
    w = np.exp(c_shifted)
    w = w / w.sum()
    
    # Compute g_i = sum_j w[j+i] * a_j (cross-correlation)
    # Using FFT: g = IFFT(FFT(w_padded) * conj(FFT(a_padded)))[:n]
    L = 1 << (3 * n).bit_length()  # next power of 2 >= 3n
    W = rfft(w, L)
    A = rfft(a, L)
    g = irfft(W * np.conj(A), L)[:n]
    
    # S = sum_j g_j * a_j
    S = np.dot(g, a)
    
    # grad_z_i = 2 * a_i * (g_i - S)
    grad = 2.0 * a * (g - S)
    return grad, a, c.max()

def generate_windows(n):
    """Generate diverse window function initializations."""
    x = np.linspace(0, 1, n)
    windows = {}

    # Uniform
    windows['uniform'] = np.ones(n) / n

    # Hann
    w = np.sin(np.pi * x) ** 2
    windows['hann'] = w / w.sum()

    # Hamming
    w = 0.54 - 0.46 * np.cos(2 * np.pi * x)
    w = np.maximum(w, 0)
    windows['hamming'] = w / w.sum()

    # Blackman
    w = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    w = np.maximum(w, 0)
    windows['blackman'] = w / w.sum()

    # Raised cosine (cosine taper)
    w = np.cos(np.pi * (x - 0.5))
    w = np.maximum(w, 0)
    windows['cosine'] = w / w.sum()

    # Super-Gaussian
    sigma = 0.25
    w = np.exp(-((x - 0.5) / sigma) ** 4)
    windows['super_gauss'] = w / w.sum()

    # Sine (half period)
    w = np.sin(np.pi * x)
    w = np.maximum(w, 0)
    windows['sine'] = w / w.sum()

    # Flat top approximation
    w = np.ones(n)
    taper = max(1, n // 10)
    for i in range(taper):
        f = np.sin(np.pi * i / (2 * taper)) ** 2
        w[i] = f
        w[n - 1 - i] = f
    windows['flattop'] = w / w.sum()

    return windows


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def lse_obj_grad(z, T, L_conv, L_corr):
    """Compute LSE_T(c) and its gradient in z-space where a=softmax(z)."""

                s_pert = 2.0 * n * fast_convolve(a_pert).max()
                if s_pert < best_score - 1e-14:
                    a = a_pert
                    c = fast_convolve(a)
                    best_score = s_pert
                    best_a = a.copy()
                    no_improve_count = 0
                elif outer > 20:
                    break

    return best_a, best_score
