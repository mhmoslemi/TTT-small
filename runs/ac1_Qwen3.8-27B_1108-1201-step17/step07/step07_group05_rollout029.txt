def active_set_qp(a, active_indices, delta):
    n = len(a)
    d = cp.Variable(n)
    c = fast_convolve(a)

    # c_k(a+d) ≈ c_k(a) + sum_i 2*(a_{k-i})*d_i (for valid indices)
    # This is c_k(a) + (grad_k)^T d
    
    constraints = [cp.sum(d) == 0]
    # Non-negativity: a_i + d_i >= 0 => d_i >= -a_i
    constraints.append(d >= -a)

    constraints.append(cp.norm(d, 2) <= delta)
    
    # Objective: minimize max of (c_k(a) + grad_k^T d) for k in active
    obj_terms = []
    for k in active_indices:

Trust region approach:
1. Solve QP with radius delta
2. Compute actual decrease: `decrease = f(a) - f(a + d)`
3. Compute predicted decrease: `predicted = c_max - (c_max + min_k (g_k^T d))` (approximately)
4. If `decrease / predicted > 0.5`, expand delta
5. Otherwise, shrink delta

This is the standard trust region approach.

import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def conv_gradient(a, w):
    """Compute sum_k w_k * grad(c_k) = 2 * cross_correlation(w, a)"""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    return grad

def project_simplex(v):
    """Project v onto the simplex {x : x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def nesterov_bb_optimize(a0, time_limit, T=100.0, verbose=False):

    n = len(a0)

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    y = a.copy()
    beta = 0.5  # momentum

    s_prev = None
    y_grad_prev = None
    lr = 1.0 / (2.0 * n)  # initial step size
    
    iters = 0
    stalled = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1

        conv_y = fast_convolve(y)
        # Softmax weights
        shifted = conv_y / T
        shifted_max = shifted.max()
        w = np.exp(shifted - shifted_max)
        w /= w.sum()
        
        grad = conv_gradient(y, w)
        
        # BB step size update
        if s_prev is not None and y_grad_prev is not None:
            sy = np.dot(s_prev, y_grad_prev)
            if abs(sy) > 1e-15:
                yy = np.dot(y_grad_prev, y_grad_prev)
                ss = np.dot(s_prev, s_prev)
                if yy > 1e-15:
                    # Alternate between BB+ and BB-
                    if iters % 2 == 0:
                        lr_new = ss / yy  # BB-
                    else:
                        lr_new = sy / yy if sy > 0 else ss / yy  # BB+
                    lr = np.clip(lr_new, 1e-8, 1.0)

        # Or use heavy ball: beta = 0.9

        # Actually, for Nesterov:

        a_new = project_simplex(y_new)
        # Actually, let me use a simpler momentum approach:

"""

- Barzilai-Borwein adaptive step size for superlinear convergence

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, lsq_linear
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def conv_gradient_from_weights(a, w):
    """
    Compute sum_k w_k * grad(c_k) where c_k = conv(a,a)_k.
    grad(c_k)[i] = 2*a[k-i] if 0<=k-i<n else 0.
    Result = 2 * cross_correlation(w, a) (truncated to n).
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    return grad


def project_simplex(v):
    """Project v onto the probability simplex {x : x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_weights(conv, T):
    """Compute softmax weights over convolution values."""
    shifted = conv / T
    shifted_max = shifted.max()
    w = np.exp(shifted - shifted_max)
    s = w.sum()
    if s < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return w / s


def nesterov_simplex(a0, time_limit, T_start=5.0, T_end=0.01,
                     alpha0=0.9, verbose=False):
    """
    Nesterov-accelerated gradient descent on the simplex with BB step size.
    Optimizes the log-sum-exp approximation of 2n*max(conv).
    """
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    total_time = time_limit

    # State
    v = np.zeros(n)  # momentum
    s_prev = None    # BB: previous step direction
    g_prev = None    # BB: previous gradient
    lr = 1.0 / (2.0 * np.linalg.norm(fast_convolve(a)) + 1e-10)
    iters = 0
    stale_counter = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / total_time)

        # Adaptive temperature
        T = T_start * (T_end / T_start) ** progress

        # Compute gradient
        conv = fast_convolve(a)
        w = softmax_weights(conv, T)
        grad = conv_gradient_from_weights(a, w)

        # BB step size
        if s_prev is not None and g_prev is not None:
            sy = np.dot(s_prev, g_prev)
            yy = np.dot(g_prev, g_prev)
            ss = np.dot(s_prev, s_prev)
            if yy > 1e-20:
                if sy > 1e-20:
                    lr_bb_plus = sy / yy
                    lr_bb_minus = ss / yy
                    lr_new = lr_bb_plus if (iters % 4 < 2) else lr_bb_minus
                else:
                    lr_new = lr
                lr = np.clip(lr_new, np.min(a + 1e-10) * 0.1, 10.0 * np.max(a + 1e-10))
                lr = max(lr, 1e-12)

        # Heavy ball momentum
        momentum = alpha0 + (1 - alpha0) * progress * 0.5  # increase momentum slightly
        v = momentum * v + (1 - momentum) * grad
        a_new = project_simplex(a - lr * v)

        # BB state update
        s_curr = a_new - a
        s_prev = s_curr
        g_prev = grad
        a = a_new

        # Track best
        score = compute_score(a)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            stale_counter = 0
        else:
            stale_counter += 1

        # Perturb if stuck
        if stale_counter > 2000:
            noise_scale = 0.0005 * np.max(a)
            a = best_a + noise_scale * np.random.randn(n)
            a = project_simplex(a)
            v = np.zeros(n)
            s_prev = None
            g_prev = None
            lr = 1.0 / (2.0 * n)
            stale_counter = 0

        if verbose and iters % 500 == 0:
            print(f"    NBB iter={iters}, T={T:.4f}, lr={lr:.6e}, score={best_score:.8f}")

    return best_a, best_score


def active_set_qp_refine(a0, time_limit, K=12, delta_init=0.005, verbose=False):
    """
    Trust-region QP refinement using active peak set.
    At each iteration: identify top K peaks, solve SOCP for best direction.
    """
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    delta = delta_init
    iters = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        iters += 1
        if time.monotonic() - t0 < time_limit:
            conv = fast_convolve(a)

            # Top K peaks
            top_k_idx = np.argsort(conv)[::-1][:K]
            c_active = conv[top_k_idx]

            # Compute gradient matrix: G[i,j] = grad(c_{top_k_idx[i]})[j]
            G = np.zeros((K, n))
            for i, k in enumerate(top_k_idx):
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for j in range(lo, hi + 1):
                    G[i, j] = 2.0 * a[k - j]

            # Solve trust region QP
            d = np.zeros(n)
            try:
                d_var = cp.Variable(n)
                t_var = cp.Variable()
                constraints = [
                    cp.sum(d_var) == 0,
                    d_var >= -a * 0.99,  # allow small violations for feasibility
                    cp.norm(d_var, 2) <= delta,
                ]
                for i in range(K):
                    constraints.append(c_active[i] + G[i] @ d_var <= t_var)
                prob = cp.Problem(cp.Minimize(t_var), constraints)
                prob.solve(solver=cp.ECOS, max_iter=3000)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    d = d_var.value
                    if d is None:
                        d = np.zeros(n)
            except Exception:
                # Fallback: least squares equalization
                try:
                    c_target = c_active.mean()
                    b = c_target - c_active
                    # Solve G @ d = b in least squares with sum(d)=0 constraint
                    A_mat = np.vstack([G, np.ones((1, n))])
                    b_vec = np.append(b, 0.0)
                    res = np.linalg.lstsq(A_mat, b_vec, rcond=None)
                    d = res[0]
                    d = d / (np.linalg.norm(d) + 1e-15) * delta * 0.5
                except Exception:
                    d = np.zeros(n)

            if np.max(np.abs(d)) < 1e-15:
                # Try a perturbation
                d = rng.standard_normal(n) * delta * 0.1
                d -= d.mean()

            # Take step
            a_new = project_simplex(a + d)
            score_new = compute_score(a_new)

            # Trust region update
            actual_decrease = best_score - score_new
            if score_new < best_score - 1e-14:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new
                delta = min(delta * 1.3, 0.05)
            else:
                delta *= 0.5
                if delta < 1e-10:
                    # Reset with perturbation
                    a = best_a
                    delta = delta_init
                    a = a + 0.001 * rng.standard_normal(n)
                    a = project_simplex(a)

        if verbose and iters % 5 == 0:
            print(f"    ASQP iter={iters}, score={best_score:.8f}, delta={delta:.6f}")

    return best_a, best_score


def lbgfs_logspace(a0, time_limit, T=50.0, maxiter=3000):
    """L-BFGS in log-space with log-sum-exp smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-15)
    a_safe = a_safe / a_safe.sum()
    z0 = np.log(a_safe)
    t0 = time.monotonic()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()
        conv = fast_convolve(a)
        shifted = conv / T
        sm = shifted.max()
        lse = sm + np.log(np.exp(shifted - sm).sum())
        return (2.0 * n / T) * np.exp(T * lse)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()
        conv = fast_convolve(a)
        shifted = conv / T
        sm = shifted.max()
        w = np.exp(shifted - sm)
        w = w / w.sum()
        grad_a = conv_gradient_from_weights(a, w)
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return (2.0 * n / T) * grad_z

    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-10, 'maxfun': maxiter * 30})
        z_max = res.x.max()
        a_out = np.exp(res.x - z_max)
        a_out = a_out / a_out.sum()
        score = compute_score(a_out)
        return a_out, score
    except Exception:
        return a0, compute_score(a0)


def cosine_init(n):
    """Initialize with a raised cosine window."""
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def chirp_init(n, freq=1.0):

    a = np.maximum(a, 1e-10)
    return a / a.sum()


def random_spectral_init(n, seed=0):

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    for alt_n in nearby:
        if alt_n in seen or alt_n < 128 or alt_n > 4096:
            continue
        seen.add(alt_n)
        rem = deadline - time.monotonic() - 50
        if rem < 30:
            break
        t_alloc = min(rem * 0.25, 60.0)
        a_alt = interpolate_sequence(best_seq, alt_n)
        # Quick Nesterov
        a_alt, _ = nesterov_simplex(a_alt, min(t_alloc * 0.6, 30.0),
                                    T_start=5.0, T_end=0.1)
        best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}",
                                                best_seq, best_score)

    # === Phase 6: Final Combined Polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 20
    if rem > 10:
        # One more Nesterov round
        t_alloc = min(rem * 0.3, 50.0)
        a_fin1, _ = nesterov_simplex(best_seq, t_alloc, T_start=2.0, T_end=0.005,
                                     alpha0=0.9)
        best_seq, best_score = check_and_update(a_fin1, "final_nest", best_seq, best_score)

    rem = deadline - time.monotonic() - 15
    if rem > 5:
        t_alloc = min(rem * 0.5, 30.0)
        a_fin2, _ = active_set_qp_refine(best_seq, t_alloc, K=15,
                                         delta_init=0.001)
        best_seq, best_score = check_and_update(a_fin2, "final_qp", best_seq, best_score)

    rem = deadline - time.monotonic() - 10
    if rem > 5:
        t_alloc = min(rem * 0.7, 20.0)
        a_fin3, _ = lbgfs_logspace(best_seq, t_alloc, T=100.0, maxiter=1000)
        best_seq, best_score = check_and_update(a_fin3, "final_lbgfs", best_seq, best_score)

    # === Final Verification ===
    if best_seq is None:
        best_seq = parent
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = min(best_score, final_score)

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
