**Key algorithmic ideas:**

1. **Multiplicative Weight Descent (MWD)**: 
   - Maintain a on the simplex.
   - At each step, compute the "loss" for each coordinate (its contribution to the peak).

3. **Restarts and perturbations**:
   - If no improvement for a while, perturb the current solution.
   - Try different β values.

5. **Pairwise coordinate moves**:

import numpy as np

    size = 2 * n - 1

def compute_grad_direct(a, p):
    """grad[i] = sum_j p[i+j]*a[j] for valid indices. p has length 2n-1."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        max_j = min(n, 2 * n - 1 - i)
        if max_j > 0:
            grad[i] = np.dot(p[i:i + max_j], a[:max_j])
    return grad

def mwd_optimize(a0, time_limit, beta_start=20.0, beta_end=500.0, 
                 lr_init=0.1, lr_decay=0.999, verbose=False):
    """
    Multiplicative Weight Descent with adaptive softmax temperature.
    Minimize max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = (it / max(1, it))  # just for tracking
        
        # Compute convolution
        b = fast_convolve(a)  # length 2n-1
        
        # Adaptive beta: increase over time
        progress = min(1.0, it / 5000.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute softmax weights
        b_max = b.max()
        log_p = beta * (b - b_max)
        log_p = log_p - log_p.max()  # numerical stability
        p = np.exp(log_p)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(2 * n - 1)
            p[np.argmax(b)] = 1.0
        else:
            p = p / p_sum
        
        # Compute gradient
        grad = compute_grad_direct(a, p)
        
        # Multiplicative update
        lr = lr_init * (lr_decay ** it)
        lr = max(lr, 1e-6)
        
        # a_new[i] = a[i] * exp(-lr * grad[i])
        updates = -lr * grad
        updates -= updates.max()  # numerical stability
        a_new = a * np.exp(updates)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            break
        a_new = a_new / a_new_sum
        
        # Check score
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # If stuck, try a different beta or perturb
        if no_improve > 200:
            # Perturb
            rng_local = np.random.default_rng(it * 31 + 7)
            noise = rng_local.standard_normal(n) * 0.02
            a = a + noise
            a = np.maximum(a, 0)
            s2 = a.sum()
            if s2 > 1e-10:
                a = a / s2
            else:
                a = best_a.copy()
            no_improve = 0
        
        if verbose and it % 500 == 0:
            elapsed_t = time.monotonic() - t0
            print(f"    MWD it={it} ({elapsed_t:.0f}s): beta={beta:.1f} best={best_score:.6f} no_improve={no_improve}")
    
    return best_a, best_score

        for eq_idx, i in enumerate(support):
            for k in active_sorted:
                j = k - i
                if 0 <= j < n and best_a[j] > threshold:  # j is in support
                    # Find the column index of a_j in the unknowns
                    col_idx = np.where(support == j)[0]
                    if len(col_idx) > 0:
                        A_mat[eq_idx, col_idx[0]] += 1.0 / K
            A_mat[eq_idx, m] = -1.0  # coefficient of λ
        
        # Last equation: sum a_i = 1
        for col_idx in range(m):
            A_mat[m, col_idx] = 1.0
        rhs[m] = 1.0
        
        # Solve
        try:
            x = np.linalg.solve(A_mat, rhs)
        except np.linalg.LinAlgError:
            continue
        
        # Extract solution
        a_new = np.zeros(n)
        for idx, i in enumerate(support):
            a_new[i] = max(0, x[idx])
        lam = x[m]
        
        # Check if solution is valid
        if a_new.sum() < 0.99:  # should be close to 1
            continue
        a_new = a_new / a_new.sum()
        
        # Check that the peaks are indeed at the active set
        b_new = fast_convolve(a_new)
        max_val = b_new.max()
        active_max = b_new[active_sorted].max()
        if abs(max_val - active_max) > 1e-6:
            continue  # Active set changed
        
        # Check KKT: for i in support, (G p)_i should be close to λ
        # And for i not in support, (G p)_i should be <= λ
        p = np.ones(K) / K
        grad = compute_grad_direct(a_new, p)
        support_grad = grad[support]
        nonsupport_grad = grad[~np.isin(np.arange(n), support)] if len(support) < n else np.array([])
        
        if np.max(np.abs(support_grad - lam)) > 1e-4:
            continue  # KKT not satisfied
        
        if len(nonsupport_grad) > 0 and nonsupport_grad.max() > lam + 1e-4:
            continue  # KKT violated
        
        # Score
        score = 2.0 * n * b_new.max()
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative Weight Descent (MWD) with adaptive softmax beta annealing
2. LP-based fictitious play for basin finding
3. KKT-condition-based polishing (solving linear systems for active peak sets)
4. Pairwise coordinate descent for fine refinement
5. Multi-length exploration and diverse initializations

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_grad_direct(a, p):
    """grad[i] = sum_j p[i+j]*a[j] for valid indices. p has length 2n-1."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        max_j = min(n, 2 * n - 1 - i)
        if max_j > 0:
            grad[i] = np.dot(p[i:i + max_j], a[:max_j])
    return grad

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.PDLP, cp.SCS, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def mwd_optimize(a0, time_limit, beta_start=15.0, beta_end=800.0,
                 lr_init=0.08, lr_decay=0.9995, verbose=False):
    """
    Multiplicative Weight Descent with adaptive softmax temperature.
    Minimize max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        b = fast_convolve(a)

        progress = min(1.0, it / 8000.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        b_max = b.max()
        log_p = beta * (b - b_max)
        log_p -= log_p.max()
        p = np.exp(log_p)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(2 * n - 1)
            p[np.argmax(b)] = 1.0
        else:
            p = p / p_sum

        grad = compute_grad_direct(a, p)

        lr = lr_init * (lr_decay ** it)
        lr = max(lr, 1e-7)

        updates = -lr * grad
        updates -= updates.max()
        a_new = a * np.exp(updates)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            break
        a_new = a_new / a_new_sum

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 300:
            rng_local = np.random.default_rng(it * 31 + 7)
            sigma = 0.01 + 0.02 * min(1.0, no_improve / 1000.0)
            noise = rng_local.standard_normal(n) * sigma
            a = a + noise
            a = np.maximum(a, 0)
            s2 = a.sum()
            if s2 > 1e-10:
                a = a / s2
            else:
                a = best_a.copy()
            no_improve = 0

        if verbose and it % 1000 == 0:
            elapsed_t = time.monotonic() - t0
            print(f"    MWD it={it} ({elapsed_t:.0f}s): beta={beta:.1f} best={best_score:.6f} stuck={no_improve}")

            continue

            A_mat[eq_idx, m] = -1.0

            A_mat[m, col_idx] = 1.0
        rhs[m] = 1.0

        try:
            x = np.linalg.solve(A_mat, rhs)

                            if len(col_idx) > 0:

                    A2[eq_idx, m_iter] = -1.0

def single_cd(a0, time_limit, verbose=False):
    """Single coordinate descent with scaling."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(4242)

    for sweep in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        perm = rng.permutation(n)
        improved = False
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.7, 0.85, 0.92, 0.96, 1.04, 1.08, 1.15, 1.3]:
                new_val = old_val * factor
                if new_val < 0:
                    continue
                a[i] = new_val
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    continue
                a_norm = a / s
                score = 2.0 * n * fast_convolve(a_norm).max()
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a_norm.copy()
                    a = a_norm
                    old_val = a[i]
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()
            a = a / a.sum()
        if not improved and sweep > 10:
            break

    return best_a, best_score

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        a_avg = (iters * a_avg + a) / (iters + 1)

        candidates = [a, b, 0.5 * (a + b), a_avg]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand_norm = cand / s
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

def damped_lp_refine(a0, time_limit, solver=None):
    """Damped alternating LP refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        alpha = min(1.0, 3.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_non_square(n, rng):
    """Initialize with non-symmetric structure to break the center peak."""
    a = np.zeros(n)
    n_segments = int(rng.integers(8, 30))
    seg_len = max(2, n // n_segments)
    for seg in range(n_segments):
        start = seg * seg_len
        end = min(start + seg_len, n)
        a[start:end] = rng.random() + 0.05
    # Add a "gap" in the middle to reduce center convolution
    mid = n // 2
    gap = max(3, n // 20)
    a[mid - gap:mid + gap] *= 0.3
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def anti_symmetric_init(n, rng):
    """Initialize with anti-symmetric structure (a[i] large where a[n-1-i] is small)."""
    a = np.zeros(n)
    for i in range(n):
        if i < n // 2:
            a[i] = rng.random() * 0.8 + 0.2
            a[n - 1 - i] = rng.random() * 0.2
        else:
            a[i] = rng.random() * 0.8 + 0.2
            a[n - 1 - i] = rng.random() * 0.2
    # Ensure non-negativity
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
