def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    """Linear cross-convolution via FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    A = rfft(a, size)
    B = rfft(b, size)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP-based alternating optimization
2. Multiplicative weights (exponentiated gradient) for fast best-response approximation
3. L-BFGS-B with adaptive soft-max smoothing and geometric cooling
4. Symmetric-constrained optimization to reduce search space
5. Peak shaving and coordinate descent for final polish
6. Multi-length exploration for global improvement
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def autoconv_fft(a):
    """Fast linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_conv_fft(a, b):
    """Fast linear cross-convolution via FFT."""
    out_len = len(a) + len(b) - 1
    fft_size = 1 << (out_len - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:out_len]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_conv_matrix_fast(a):
    """Build convolution matrix using FFT-free approach for speed."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, cached."""
    if hasattr(get_lp_solver, 'cached'):
        return get_lp_solver.cached
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                get_lp_solver.cached = solver_name
                return solver_name
        except Exception:
            continue
    get_lp_solver.cached = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, a>=0), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_fast(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt)
                    break
                except Exception:
                    continue
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def mpw_best_response(fixed, current, n_steps=50, eta=0.05, delta_frac=0.02):
    """
    Multiplicative weights (mirror descent) to approximate best response.
    Given fixed 'fixed', find 'other' minimizing max_k conv(fixed, other)[k].
    Returns the optimized 'other' (normalized to sum=1).
    """
    n = len(fixed)
    b = np.maximum(current, 0).astype(np.float64)
    s = b.sum()
    if s < 1e-15:
        b = np.ones(n) / n
    else:
        b = b / s
    log_b = np.log(np.maximum(b, 1e-300))

    for step in range(n_steps):
        c = cross_conv_fft(fixed, b)
        max_c = c.max()
        if max_c < 1e-15:
            break
        threshold = max_c * (1.0 - delta_frac)
        A = np.where(c >= threshold)[0]
        if len(A) == 0:
            A = np.array([np.argmax(c)])

        g = np.zeros(n)
        for k in A:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += fixed[k - idx]
        g /= len(A)

        log_b = log_b - eta * g
        log_b = log_b - log_b.max()
        b = np.exp(log_b)
        b = np.maximum(b, 0)
        s2 = b.sum()
        if s2 < 1e-300:
            b = np.ones(n) / n
        else:
            b = b / s2

    return b


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=20000):
    """
    Nesterov-accelerated LP-based alternating optimization.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * autoconv_fft(a).max()
    t0 = time.monotonic()
    y = a.copy()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa

        score_a = 2.0 * n * autoconv_fft(a).max()
        score_y = 2.0 * n * autoconv_fft(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} M_lp={M_b:.6f}")

    return best_a, best_score

def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def lbfgs_soft_max(a0, time_limit, T_start=50.0, T_end=0.5, maxiter=5000):
    """L-BFGS refinement with soft-max smoothing and geometric temperature schedule."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z)
            b = autoconv_fft(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a = softmax_from_z(z)
            b = autoconv_fft(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 6
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = max(maxiter // n_stages, 100)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 80, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = softmax_from_z(z)
    return a_out


def peak_shave_local(a0, time_limit, n_steps=10000):
    """Aggressive peak shaving local search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * autoconv_fft(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 200:
            break

        b = autoconv_fft(a)
        mask = np.ones(len(b), dtype=bool)
        mask[0] = False
        mask[-1] = False
        b_masked = b.copy()
        b_masked[~mask] = -np.inf
        peak_k = int(np.argmax(b_masked))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        order = np.argsort(contribs)[::-1]

        improved = False
        for ci in order[:8]:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.95, 0.90, 0.85, 0.75]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                mask_i = np.ones(n, dtype=bool)
                mask_i[i] = False
                others = a_trial[mask_i]
                if others.sum() < 1e-15:
                    continue
                a_trial[mask_i] = others + diff * others / others.sum()
                score = 2.0 * n * autoconv_fft(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
            if improved:
                break

        if not improved:
            if len(order) >= 2:
                for ci in order[:4]:
                    i = idx[ci]
                    j = peak_k - i
                    if i == j or a[j] < 1e-14:
                        continue
                    for f1 in [0.97, 0.93]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f1
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * autoconv_fft(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
                    if improved:
                        break
            if not improved:
                no_improve += 1
                if no_improve > 50:
                    top3 = order[:3]
                    a_trial = a.copy()
                    for ci in top3:
                        i = idx[ci]
                        a_trial[i] *= 0.97
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * autoconv_fft(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1

    return best_a, best_score


def coord_descent_fast(a0, time_limit, n_passes=50):
    """Fast coordinate descent with random ordering."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * autoconv_fft(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            base = a.copy()
            base[i] = 0
            base_sum = base.sum()
            if base_sum < 1e-15:
                continue
            best_t = a[i]
            best_local = best_score
            for factor in [0.8, 0.9, 1.0, 1.1, 1.25, 1.5]:
                trial_t = a[i] * factor
                if trial_t < 0:
                    continue
                a_trial = base * (1.0 - trial_t / (1.0 + trial_t)) 
                # Simpler: set a[i] = trial_t, scale others to keep sum=1
                others_scale = (1.0 - trial_t) / base_sum if base_sum > 1e-15 else 0
                if others_scale < 0:
                    continue
                a_trial = base * others_scale
                a_trial[i] = trial_t
                sc = 2.0 * n * autoconv_fft(a_trial).max()
                if sc < best_local:
                    best_local = sc
                    best_t = trial_t
                    best_trial = a_trial.copy()
            if best_t != a[i]:
                a = best_trial
                if best_local < best_score:
                    best_score = best_local
                    best_a = a.copy()

        if p % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={p} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def spectral_flatten(a, n_iter=100, rng=None):
    """
    Try to flatten the auto-convolution spectrum using gradient steps in log-domain.
    Minimizes sum of (conv - mean(conv))^2 as a proxy.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()
    z = np.log(a)
    if rng is None:
        rng = np.random.default_rng(0)

    for it in range(n_iter):
        c = autoconv_fft(a)
        target = c.mean()
        diff = c - target
        # Gradient of sum(diff^2) w.r.t. a: 2 * T^T @ diff where T is conv matrix
        grad = np.zeros(n)
        for k in range(len(c)):
            if abs(diff[k]) < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * diff[k] * a[k - idx]
        # Gradient step in log-domain
        lr = 0.01 / (1.0 + it * 0.01)
        z = z - lr * (grad / a - np.dot(grad, a) / a.sum())
        a_new = softmax_from_z(z)
        # Check if it improved the max
        c_new = autoconv_fft(a_new)
        if c_new.max() < c.max():
            a = a_new
    return a
