"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fictitious play with LP-based best responses.
The key algorithm: at each iteration, given current f, solve the LP
  max sum(g) s.t. (f*g)_k <= M for all k, g >= 0
where M = max(f*f). The optimal g is the "best response" in a two-player game.
Mix f towards g and repeat. This converges to a local minimum.
Combined with local optimization (L-BFGS, coordinate descent) for final polishing.

import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize

    A = rfft(a, fft_size)
    B = rfft(b, fft_size)

def compute_score_normalized(a):

def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix where row k gives (f*g)_k."""
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A


def solve_lp_best_response(f, M):
    """
    Solve: max sum(g) s.t. ConvMatrix(f) @ g <= M * ones, g >= 0
    Returns (g, sum_g) or (None, None) if failed.
    """
    n = len(f)
    A_ub = build_conv_matrix(f, n)
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)  # minimize -sum(g)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    return None, None


def fictitious_play(f0, time_limit, t_mix=0.1, verbose=True):
    """
    Fictitious play with LP best responses.
    """
    n = len(f0)
    f = np.maximum(f0.copy(), 0)
    s = f.sum()
    if s < 1e-15:
        return f0, compute_score_normalized(f0)
    f = f / s
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    t0 = time.monotonic()
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        M = fast_convolve(f).max()
        
        g, sum_g = solve_lp_best_response(f, M)
        
        if g is None:
            if verbose:
                print(f"  LP failed at iter {iters}")
            break
        
        if sum_g <= 1e-15:
            break
        
        # Rescale g to sum = 1
        g = g / sum_g
        
        # Mix
        f_new = (1 - t_mix) * f + t_mix * g
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new = f_new / s_new
        
        score = compute_score_normalized(f_new)
        if score < best_score - 1e-14:
            best_score = score
            best_f = f_new.copy()
        
        f = f_new
        
        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  FP iter={iters}, score={best_score:.8f}, sum_g={sum_g:.6f}, elapsed={elapsed:.1f}s")
    
    return best_f, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-30))


def lbgfs_optimize(a0, time_limit, T=500, maxiter=3000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score_normalized(a0)
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_s = shifted.max()
        lse = max_s + np.log(np.exp(shifted - max_s).sum())
        return (2.0 * n / T) * np.exp(T * lse)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_s = shifted.max()
        w = np.exp(shifted - max_s)
        w = w / w.sum()
        # Gradient of smoothed objective w.r.t. a
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        # Project to log-space
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return (2.0 * n / T) * grad_z
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 40, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score_normalized(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs_per_iter=200):

                    a_trial2 = a_trial2 / ss2

            if best_t != 0.0:

                best_score = best_local

def peak_equalization(a0, time_limit, n_peaks=10):
    """Iteratively reduce the maximum peak by adjusting contributing elements."""
    n = len(a0)

    best_score = compute_score_normalized(a)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_idx = int(np.argmax(conv))
        
        # Find contributing pairs
        js = max(0, max_idx - n + 1)
        je = min(n - 1, max_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_idx - idx]
        top_contrib = np.argsort(contribs)[::-1][:min(10, len(idx))]
        
        improved = False
        for ci in top_contrib:
            i = idx[ci]
            if a[i] < 1e-14:

                a_trial = a.copy()
                a_trial[i] -= delta
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask].copy()
                os = others.sum()
                if os < 1e-15:
                    continue
                a_trial[mask] = others + delta * others / os
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score_normalized(a_trial)
                if sc < best_score - 1e-14:

                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:

            top_k_idx = np.argpartition(conv, -5)[-5:]

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fictitious play with LP-based best responses.
At each iteration, given current f, solve the LP:
  max sum(g) s.t. (f*g)_k <= M for all k, g >= 0
where M = max(f*f). The optimal g is the "best response." Mix f towards g and repeat.
Combined with L-BFGS, peak equalization, and coordinate descent for polishing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score_normalized(a):
    """Compute 2n * max(conv(a,a)) for a with sum=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_best_response(f, M):
    """Solve: max sum(g) s.t. ConvMatrix(f) @ g <= M * ones, g >= 0."""
    n = len(f)
    A_ub = build_conv_matrix(f, n)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return None, None


def solve_lp_reduced(f, M, active_indices):
    """Solve LP with only active constraints (cutting plane acceleration)."""
    n = len(f)
    if len(active_indices) == 0:
        active_indices = list(range(min(20, 2 * n - 1)))
    A_sub = np.zeros((len(active_indices), n))
    for idx, k in enumerate(active_indices):
        for j in range(n):
            ii = k - j
            if 0 <= ii < n:
                A_sub[idx, j] = f[ii]
    b_sub = np.full(len(active_indices), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_sub, b_ub=b_sub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return None, None


def fictitious_play(f0, time_limit, t_mix_init=0.3, verbose=True):
    """Fictitious play with LP best responses and adaptive mixing."""
    n = len(f0)
    f = np.maximum(f0.copy(), 0)
    s = f.sum()
    if s < 1e-15:
        return f0, compute_score_normalized(f0)
    f = f / s

    best_f = f.copy()
    best_score = compute_score_normalized(f)
    t0 = time.monotonic()
    iters = 0
    t_mix = t_mix_init
    prev_sum_g = 1.0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv_ff = fast_convolve(f)
        M = conv_ff.max()

        g, sum_g = solve_lp_best_response(f, M)
        if g is None or sum_g <= 1e-15:
            break

        # Adaptive mixing: decrease t as we converge
        improvement = max(0, sum_g - 1.0)
        t_mix = max(0.01, t_mix_init * (0.95 ** iters) + 0.01 * min(1.0, improvement * 10))

        g_norm = g / sum_g
        f_new = (1 - t_mix) * f + t_mix * g_norm
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new = f_new / s_new

        score = compute_score_normalized(f_new)

            best_f = f_new.copy()

            print(f"    FP iter={iters}, score={best_score:.8f}, sum_g={sum_g:.6f}, t={t_mix:.4f}, elapsed={elapsed:.1f}s")

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-30))


def lbgfs_optimize(a0, time_limit, T=500, maxiter=3000):
    """L-BFGS in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score_normalized(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_s = shifted.max()
        lse = max_s + np.log(np.exp(shifted - max_s).sum())
        return (2.0 * n / T) * np.exp(T * lse)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_s = shifted.max()
        w = np.exp(shifted - max_s)
        w = w / w.sum()
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return (2.0 * n / T) * grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 40, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score_normalized(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs_per_iter=200):
    """Coordinate descent with pairwise moves."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        improved = False
        for _ in range(n_pairs_per_iter):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue
            best_t = 0.0
            best_local = best_score
            base = min(a[i], a[j]) if a[j] > 0 else a[i]
            for frac in [0.1, 0.3, 0.5, 0.01, 0.05, 0.7]:
                t = frac * base
                if t >= a[i] - 1e-15:
                    t = a[i] * 0.5
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score_normalized(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t
                if a[j] > 1e-15:
                    t2 = frac * a[j]
                    a_trial2 = a.copy()
                    a_trial2[j] -= t2
                    a_trial2[i] += t2
                    a_trial2 = np.maximum(a_trial2, 0)
                    ss2 = a_trial2.sum()
                    if ss2 < 1e-15:
                        continue
                    a_trial2 = a_trial2 / ss2
                    sc2 = compute_score_normalized(a_trial2)
                    if sc2 < best_local - 1e-14:
                        best_local = sc2
                        best_t = -t2
            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
        if not improved:
            noise = rng.standard_normal(n) * 0.002
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
    return best_a, best_score


def peak_equalization(a0, time_limit, n_peaks=15):
    """Iteratively reduce the maximum peak."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_idx = int(np.argmax(conv))
        js = max(0, max_idx - n + 1)
        je = min(n - 1, max_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_idx - idx]
        top_contrib = np.argsort(contribs)[::-1][:min(10, len(idx))]

        improved = False
        for ci in top_contrib:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for frac in [0.03, 0.05, 0.1, 0.15]:
                delta = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= delta
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask].copy()
                os = others.sum()
                if os < 1e-15:
                    continue
                a_trial[mask] = others + delta * others / os
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            top_k_idx = np.argpartition(conv, -min(n_peaks, len(conv)))[-min(n_peaks, len(conv)):]
            peak_positions = set()
            for pi in top_k_idx:
                si = max(0, int(pi) - n + 1)
                ei = min(n - 1, int(pi))
                peak_positions.update(range(si, ei + 1))
            if peak_positions:
                pos_list = list(peak_positions)
                for _ in range(30):
                    p = pos_list[rng.integers(0, len(pos_list))]
                    a_trial = a.copy()
                    delta = a[p] * 0.05 * rng.standard_normal()
                    a_trial[p] -= delta
                    mask2 = np.ones(n, dtype=bool)
                    mask2[p] = False
                    o2 = a_trial[mask2].copy()
                    o2s = o2.sum()
                    if o2s > 1e-15:
                        a_trial[mask2] = o2 + delta * o2 / o2s
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_score_normalized(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
    return best_seq, best_score


def multiplicative_updates(a0, time_limit, verbose=False):
    """
    Multiplicative update rule inspired by NMF.
    Update: a_i <- a_i * (target_i) / (actual_i)
    where actual_i relates to the gradient of the objective.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    lr = 0.1

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        # Gradient of max(conv) approximated by softmax-weighted convolution
        T = 50
        shifted = conv / T
        max_s = shifted.max()
        w = np.exp(shifted - max_s)
        w = w / w.sum()

        # Gradient w.r.t. a_i: 2 * sum_k w_k * a_{|k-i|} type cross-correlation
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]

        # Multiplicative update: a_i *= (1 + lr) / (1 + lr * grad_a_i / (2n))
        denom = 1.0 + lr * grad_a / (2.0 * n)
        denom = np.maximum(denom, 1e-10)
        a = a * (1.0 + lr) / denom
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        score = compute_score_normalized(a)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        lr *= 0.999
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    MU score={best_score:.8f}, lr={lr:.4f}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization:
    1. Fictitious play with LP best responses (primary novel approach)
    2. L-BFGS in log-space with soft-max smoothing
    3. Peak equalization
    4. Coordinate descent
    5. Try nearby lengths
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score_normalized(parent)
        print(f"Initialized uniform: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # ============ Phase 1: Fictitious Play (main contribution) ============
    print(f"\n=== Phase 1: Fictitious Play with LP (best={best_score:.8f}) ===")
    t1_budget = min(500.0, (deadline - time.monotonic()) * 0.30)

    # First: standard FP with full LP
    rem = deadline - time.monotonic() - (budget_s * 0.55)
    t_alloc = min(t1_budget * 0.4, max(20, rem))
    print(f"  FP (full LP), n={n_parent}, budget={t_alloc:.0f}s")
    a_fp, sc_fp = fictitious_play(best_seq, t_alloc, t_mix_init=0.3, verbose=True)
    best_seq, best_score = check_and_update(a_fp, best_seq, best_score)
    print(f"    score={best_score:.8f}")

    # Second: FP with cutting plane (faster, can do more iterations)
    rem = deadline - time.monotonic() - (budget_s * 0.45)
    t_alloc = min(t1_budget * 0.6, max(20, rem))
    print(f"  FP (cutting plane), n={n_parent}, budget={t_alloc:.0f}s")
    a_fp2, sc_fp2 = fictitious_play_with_cutting_plane(best_seq, t_alloc, verbose=True)
    best_seq, best_score = check_and_update(a_fp2, best_seq, best_score)
    print(f"    score={best_score:.8f}")

    # ============ Phase 2: L-BFGS Polish ============
    print(f"\n=== Phase 2: L-BFGS (best={best_score:.8f}) ===")
    for T_val in [200, 800, 4000]:
        rem = deadline - time.monotonic() - (budget_s * 0.25)
        if rem < 20:
            break
        t_alloc = min(rem * 0.15, 80.0)
        print(f"  L-BFGS T={T_val}, budget={t_alloc:.0f}s")
        a_lb, sc_lb = lbgfs_optimize(best_seq, t_alloc, T=T_val)
        best_seq, best_score = check_and_update(a_lb, best_seq, best_score)
        print(f"    score={best_score:.8f}")

    # ============ Phase 3: Multiplicative Updates ============
    print(f"\n=== Phase 3: Multiplicative Updates (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - (budget_s * 0.15)
    if rem > 20:
        t_alloc = min(rem * 0.2, 80.0)
        print(f"  MU budget={t_alloc:.0f}s")
        a_mu, sc_mu = multiplicative_updates(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_mu, best_seq, best_score)
        print(f"    score={best_score:.8f}")

    # ============ Phase 4: Peak Equalization ============
    print(f"\n=== Phase 4: Peak Equalization (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - (budget_s * 0.10)
    if rem > 15:
        t_alloc = min(rem * 0.3, 100.0)
        print(f"  PE budget={t_alloc:.0f}s")
        a_pe, sc_pe = peak_equalization(best_seq, t_alloc, n_peaks=20)
        best_seq, best_score = check_and_update(a_pe, best_seq, best_score)
        print(f"    score={best_score:.8f}")

    # ============ Phase 5: Coordinate Descent ============
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - (budget_s * 0.06)
    if rem > 15:
        t_alloc = min(rem * 0.3, 80.0)
        print(f"  CD budget={t_alloc:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t_alloc, n_pairs_per_iter=400)
        best_seq, best_score = check_and_update(a_cd, best_seq, best_score)
        print(f"    score={best_score:.8f}")

    # ============ Phase 6: Try nearby lengths ============
    print(f"\n=== Phase 6: Nearby Lengths (best={best_score:.8f}) ===")
    nearby_ns = [n_parent + 1, n_parent - 1, n_parent + 2, n_parent - 2, n_parent + 4]
    for alt_n in nearby_ns:
        if alt_n < 128 or alt_n > 3000:
            continue
        rem = deadline - time.monotonic() - (budget_s * 0.03)
        if rem < 30:
            break
        t_alloc = min(rem * 0.2, 50.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  FP+LBFGS n={alt_n}, budget={t_alloc:.0f}s")
        a_alt, _ = fictitious_play(init_alt, t_alloc * 0.5, t_mix_init=0.2)
        a_alt2, _ = lbgfs_optimize(a_alt, t_alloc * 0.5, T=1000)
        best_seq, best_score = check_and_update(a_alt2, best_seq, best_score)
        print(f"    score={best_score:.8f}")

    # ============ Phase 7: Final L-BFGS + PE ============
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 10
    if rem > 10:
        t_alloc = min(rem * 0.5, 40.0)
        a_final, _ = lbgfs_optimize(best_seq, t_alloc, T=2000)
        best_seq, best_score = check_and_update(a_final, best_seq, best_score)

    rem = deadline - time.monotonic() - 5
    if rem > 5:
        t_alloc = min(rem * 0.8, 20.0)
        a_final2, _ = peak_equalization(best_seq, t_alloc, n_peaks=25)
        best_seq, best_score = check_and_update(a_final2, best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            print(f"\nFINAL VERIFIED SCORE: {fs:.8f} (n={len(best_seq)})")
            if fs < best_score:
                best_score = fs
    else:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"FINAL: {best_score:.8f}")
    return [float(x) for x in best_seq]
