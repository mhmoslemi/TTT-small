"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP Fictitious Play for initialization
2. Successive Linear Programming (linearized minimax) for refinement
3. QP refinement for final polish
4. Peak shaving and L-BFGS for last-mile improvement
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution a * b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.MOSEK, cp.GUROBI, cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def solve_successive_lp(a, solver=None):
    """
    Solve the linearized minimax LP:
    min t
    s.t. b_k + 2*(a*d)_k <= t for all k
         sum(d) = 0
         d_i >= -a_i for all i
    Returns (d, t) or (None, None) on failure.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a)
    b = fast_convolve(a)
    
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    
    # T @ d = cross_convolve(a, d) where T is built from a
    T = build_conv_matrix(a)
    
    constraints = [
        b + 2 * (T @ d_var) <= t_var,
        cp.sum(d_var) == 0,
        d_var >= -a
    ]
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    d_val = np.asarray(d_var.value, dtype=np.float64)
    t_val = float(t_var.value)
    return d_val, t_val

    t0 = monotonic()

        d, t_pred = solve_successive_lp(a, solver)

        # Line search

            a_trial = np.maximum(a_trial, 0)

            if s_trial < 1e-15:

            if score < best_score - 1e-14:

        if not improved:

def fictitious_play(n, time_limit, initial=None, solver=None, verbose=True):

        else:
            a = a / s

    b = a.copy()
    best_score = float('inf')
    best_a = a.copy()
    iters = 0
    stagnation = 0

    while monotonic() - t0 < time_limit:
        iters += 1

        if b_new is not None:
            b = b_new

        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:

            a = a_new

        a_eval = np.maximum(a, 0)
        s = a_eval.sum()
        if s > 1e-15:
            a_eval = a_eval / s
            max_conv = fast_convolve(a_eval).max()
            score = 2.0 * n * max_conv
            if score < best_score - 1e-12:
                best_score = score
                best_a = a_eval.copy()
                stagnation = 0
            else:
                stagnation += 1

        if stagnation > 100:
            noise = np.random.standard_normal(n) * 0.002
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            stagnation = 0

        if verbose and iters % 20 == 0:
            elapsed = monotonic() - t0
            print(f"  FP n={n} iter={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def lbgfs_smooth(a0, time_limit, T=50, maxiter=5000):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse / T

    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()

        grad_a = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]

        grad_a *= 2.0 * n
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10,
                     'maxfun': maxiter * 10, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0
    except Exception:
        a_out = a0

    return a_out


def peak_shave(a0, time_limit, n_steps=5000):
    """Aggressive local peak shaving."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    failed = 0

    for step in range(n_steps):
        if monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        if contribs.sum() < 1e-20:
            break

        top_contrib = np.argsort(contribs)[::-1][:10]
        improved = False

            i = int(idx[ci])
            old_val = a[i]
            if old_val < 1e-14:

                new_val = old_val * factor

                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def solve_successive_lp_topk(a, solver=None, top_k=50):
    """
    Solve the linearized minimax LP using only top-k peaks.
    This is faster and focuses on the most important constraints.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a)
    b = fast_convolve(a)
    
    # Use only top-k peaks as constraints
    k_actual = min(top_k, len(b))
    peak_indices = np.argpartition(b, -k_actual)[-k_actual:]
    
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    
    T = build_conv_matrix(a)
    
    constraints = []
    for k in peak_indices:
        constraints.append(b[k] + 2.0 * (T[k] @ d_var) <= t_var)
    
    # Also add constraints for the current max (safety)
    max_k = int(np.argmax(b))
    if max_k not in peak_indices:
        constraints.append(b[max_k] + 2.0 * (T[max_k] @ d_var) <= t_var)
    
    constraints.append(cp.sum(d_var) == 0)
    constraints.append(d_var >= -a)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    d_val = np.asarray(d_var.value, dtype=np.float64)
    t_val = float(t_var.value)
    return d_val, t_val

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fictitious_play(n, time_limit, initial=None, solver=None, verbose=True):
    """Run fictitious play for auto-correlation minimization."""
    if solver is None:
        solver = get_lp_solver()
    t0 = monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    best_score = float('inf')
    best_a = a.copy()
    iters = 0
    stagnation = 0

    while monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new

        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a_avg = (iters * a_avg + a_new) / (iters + 1)
            a = a_new

        a_eval = np.maximum(a_avg, 0)
        s = a_eval.sum()
        if s > 1e-15:
            a_eval = a_eval / s
            max_conv = fast_convolve(a_eval).max()
            score = 2.0 * n * max_conv
            if score < best_score - 1e-12:
                best_score = score
                best_a = a_eval.copy()
                stagnation = 0
            else:
                stagnation += 1

        if stagnation > 150:
            noise = np.random.standard_normal(n) * 0.003
            a_avg = a_avg + noise
            a_avg = np.maximum(a_avg, 0)
            a_avg = a_avg / a_avg.sum()
            a = a_avg.copy()
            b = a_avg.copy()
            stagnation = 0

        if verbose and iters % 20 == 0:
            elapsed = monotonic() - t0
            print(f"  FP n={n} iter={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def solve_slp_direction(a, solver=None, top_k=None):
    """
    Solve the linearized minimax LP to get a descent direction.
    min t s.t. b_k + 2*(T@d)_k <= t for all k (or top-k)
              sum(d) = 0
              d >= -a
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a)
    b = fast_convolve(a)
    n_convs = len(b)

    d_var = cp.Variable(n)
    t_var = cp.Variable()
    T = build_conv_matrix(a)

    if top_k is not None and top_k < n_convs:
        peak_indices = np.argpartition(b, -top_k)[-top_k:]
        constraints = [b[k] + 2.0 * (T[k] @ d_var) <= t_var for k in peak_indices]
    else:
        constraints = [b[k] + 2.0 * (T[k] @ d_var) <= t_var for k in range(n_convs)]

    constraints.append(cp.sum(d_var) == 0)
    constraints.append(d_var >= -a)

    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    d_val = np.asarray(d_var.value, dtype=np.float64)
    t_val = float(t_var.value)
    return d_val, t_val


def successive_lp_optimize(a0, time_limit, solver=None, top_k=None, max_iters=1000, verbose=True):
    """
    Successive LP: iteratively solve linearized LP, line search, update.
    This is the main optimization engine.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    no_improve = 0
    it = 0

    while monotonic() - t0 < time_limit and it < max_iters:
        it += 1

        # Choose top_k adaptively: start small, grow if stuck
        if top_k is None:
            eff_top_k = max(30, n // 10)
        else:
            eff_top_k = top_k

        d, t_pred = solve_slp_direction(a, solver, top_k=eff_top_k)
        if d is None:
            # Try with full constraints
            d, t_pred = solve_slp_direction(a, solver, top_k=None)
            if d is None:
                break

        # Line search with geometrically decreasing steps
        improved = False
        alphas = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.01]
        for alpha in alphas:
            a_trial = a + alpha * d
            a_trial = np.maximum(a_trial, 0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            a_trial = a_trial / s_trial
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve > 10:
                # Try with full constraints
                d_full, _ = solve_slp_direction(a, solver, top_k=None)
                if d_full is not None:
                    for alpha in [0.05, 0.03, 0.01]:
                        a_trial = a + alpha * d_full
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break

            if not improved and no_improve > 30:
                # Perturb to escape
                a = best_a + np.random.standard_normal(n) * 0.002
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0

            if not improved and no_improve > 80:
                # Aggressive perturbation
                a = best_a + np.random.standard_normal(n) * 0.01
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0

        if verbose and it % 20 == 0:
            elapsed = monotonic() - t0
            print(f"    SLP iter={it} ({elapsed:.0f}s): score={best_score:.6f} no_imp={no_improve}")

    return best_a, best_score


def lbgfs_smooth(a0, time_limit, T=50, maxiter=5000):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse / T

    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()

        grad_a = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]

        grad_a *= 2.0 * n
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10,
                     'maxfun': maxiter * 10, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0
    except Exception:
        a_out = a0

    return a_out


def peak_shave(a0, time_limit, n_steps=5000):
    """Aggressive local peak shaving with multiple strategies."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    failed = 0

    for step in range(n_steps):
        if monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        if contribs.sum() < 1e-20:
            break

        top_contrib = np.argsort(contribs)[::-1][:10]
        improved = False

        # Strategy 1: Reduce single contributor and redistribute
        for ci in top_contrib:
            i = int(idx[ci])
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.98, 0.96, 0.93, 0.9, 0.85]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Strategy 2: Reduce pair
        if not improved:
            for ci in top_contrib[:6]:
                i = int(idx[ci])
                j = int(peak_k - i)
                if i == j or a[j] < 1e-14:
                    continue
                for f1 in [0.97, 0.94, 0.9]:
                    for f2 in [0.97, 0.94, 0.9]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Strategy 3: Small random search
        if not improved:
            for attempt in range(5):
                a_trial = a + np.random.standard_normal(n) * 0.0008
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            failed += 1
            if failed > 150:
                break
        else:
            failed = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using evaluate_sequence and update best if improved."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def multi_start_explore(base_seq, solver, time_per_start, n_starts=3):
    """Run SLP from perturbed versions of base_seq to escape local minima."""
    n = len(base_seq)
    best_a = base_seq.copy()
    best_score = 2.0 * n * fast_convolve(base_seq).max()

    for i in range(n_starts):
        t_start = monotonic()
        pert_size = 0.01 + 0.01 * i
        a_pert = base_seq + np.random.standard_normal(n) * pert_size
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        t_remaining = time_per_start - (monotonic() - t_start)
        if t_remaining < 5:
            break

        a_local, s_local = successive_lp_optimize(a_pert, t_remaining, solver=solver, top_k=max(20, n // 12), max_iters=200, verbose=False)
        if s_local < best_score:
            best_a = a_local
            best_score = s_local

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining:
    1. LP Fictitious Play for initialization
    2. Successive LP (linearized minimax) as primary refinement
    3. L-BFGS smooth polish
    4. Peak shaving for micro-optimization
    5. Multi-start and length exploration
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial uniform: n=768, score={best_score:.6f}")

    n_parent = len(best_seq)

    # === Phase 1: Fictitious Play at parent length ===
    print(f"\n=== Phase 1: Fictitious Play (n={n_parent}) ===")
    rem = deadline - monotonic() - 500
    if rem > 30:
        t_fp = min(rem * 0.4, 350.0)
        a_fp, score_fp, iters = fictitious_play(n_parent, t_fp, initial=best_seq, solver=solver, verbose=True)
        best_seq, best_score = check_and_update(a_fp, f"FP_n{n_parent}", best_seq, best_score)

    # === Phase 2: Successive LP (top-k) - main refinement ===
    print(f"\n=== Phase 2: Successive LP top-k (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 400
        if rem > 30:
            t_slp = min(rem * 0.55, 350.0)
            n_best = len(best_seq)
            top_k = max(40, n_best // 8)
            print(f"  SLP-topk: n={n_best}, top_k={top_k}, budget={t_slp:.0f}s")
            a_slp, score_slp = successive_lp_optimize(best_seq, t_slp, solver=solver, top_k=top_k, max_iters=2000, verbose=True)
            best_seq, best_score = check_and_update(a_slp, "SLP_topk", best_seq, best_score)

    # === Phase 3: Successive LP (full constraints) ===
    print(f"\n=== Phase 3: Successive LP full (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 250
        if rem > 20:
            t_slp2 = min(rem * 0.5, 180.0)
            n_best = len(best_seq)
            print(f"  SLP-full: n={n_best}, budget={t_slp2:.0f}s")
            a_slp2, _ = successive_lp_optimize(best_seq, t_slp2, solver=solver, top_k=None, max_iters=200, verbose=True)
            best_seq, best_score = check_and_update(a_slp2, "SLP_full", best_seq, best_score)

    # === Phase 4: L-BFGS with decreasing temperatures ===
    print(f"\n=== Phase 4: L-BFGS Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        for T in [30, 80, 200]:
            rem = deadline - monotonic() - 150
            if rem < 10:
                break
            t_lb = min(rem * 0.25, 50.0)
            print(f"  L-BFGS T={T}, budget={t_lb:.0f}s")
            a_lb = lbgfs_smooth(best_seq, t_lb, T=T)
            best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T}", best_seq, best_score)

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 100
        if rem > 10:
            t_ps = min(rem * 0.6, 70.0)
            a_ps, _ = peak_shave(best_seq, t_ps)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 6: SLP again after polishing ===
    print(f"\n=== Phase 6: SLP Round 2 (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 80
        if rem > 20:
            t_slp3 = min(rem * 0.55, 120.0)
            n_best = len(best_seq)
            top_k = max(30, n_best // 10)
            print(f"  SLP round2: n={n_best}, top_k={top_k}, budget={t_slp3:.0f}s")
            a_slp3, _ = successive_lp_optimize(best_seq, t_slp3, solver=solver, top_k=top_k, max_iters=500, verbose=True)
            best_seq, best_score = check_and_update(a_slp3, "SLP_r2", best_seq, best_score)

    # === Phase 7: Multi-start exploration ===
    print(f"\n=== Phase 7: Multi-Start (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 60
        if rem > 30:
            t_ms = min(rem * 0.5, 100.0)
            print(f"  Multi-start: budget={t_ms:.0f}s")
            a_ms, _ = multi_start_explore(best_seq, solver, t_ms, n_starts=3)
            best_seq, best_score = check_and_update(a_ms, "multi_start", best_seq, best_score)

    # === Phase 8: Try alternative lengths ===
    print(f"\n=== Phase 8: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [0.8, 1.25, 1.5]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:2]:
            rem = deadline - monotonic() - 40
            if rem < 25:
                break
            t_alt = min(rem * 0.4, 80.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _, _ = fictitious_play(alt_n, t_alt, initial=init_alt, solver=solver, verbose=False)
            best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}", best_seq, best_score)

            # Quick SLP on new length
            if best_seq is not None and len(best_seq) == alt_n:
                rem = deadline - monotonic() - 20
                if rem > 15:
                    a_slp_alt, _ = successive_lp_optimize(best_seq, min(rem * 0.5, 30.0), solver=solver, top_k=max(20, alt_n // 12), max_iters=100, verbose=False)
                    best_seq, best_score = check_and_update(a_slp_alt, f"SLP_alt_n{alt_n}", best_seq, best_score)

    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 10
        if rem > 8:
            a_final = lbgfs_smooth(best_seq, min(rem * 0.35, 25.0), T=20, maxiter=3000)
            best_seq, best_score = check_and_update(a_final, "final_lbgfs", best_seq, best_score)

        rem = deadline - monotonic() - 5
        if rem > 5:
            a_ps_final, _ = peak_shave(best_seq, min(rem * 0.6, 12.0))
            best_seq, best_score = check_and_update(a_ps_final, "final_ps", best_seq, best_score)

        # One final SLP burst
        rem = deadline - monotonic() - 2
        if rem > 8:
            a_final_slp, _ = successive_lp_optimize(best_seq, min(rem * 0.5, 8.0), solver=solver, top_k=max(20, len(best_seq) // 12), max_iters=50, verbose=False)
            best_seq, best_score = check_and_update(a_final_slp, "final_slp", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
