def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""

    b = fast_convolve(a)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max(), b


def softmax_z(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def smooth_obj(z, T, n):
    p = softmax_z(z)
    b = fast_convolve(p)
    scaled = b / T
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return 2.0 * n * T * lse


def smooth_grad(z, T, n):
    p = softmax_z(z)
    b = fast_convolve(p)
    scaled = b / T
    m = scaled.max()
    w = np.exp(scaled - m)
    w = w / w.sum()
    # grad of 2*n*T*lse w.r.t. p[i] = 2*n * sum_k w_k * p[k-i]
    # Then chain rule for z: grad_z = p * (grad_p - p^T grad_p)
    n2 = n  # use n for cross-correlation size
    fft_size = 1 << (3 * n - 1).bit_length()
    W = rfft(w, fft_size)
    P = rfft(p, fft_size)
    correl = irfft(np.conj(W) * P, fft_size)
    grad_p = 2.0 * n * correl[:n]
    grad_z = p * (grad_p - np.dot(p, grad_p))
    return grad_z

def solve_lambda_qp(p, peaks, n, max_peaks=20):
    """Solve min lambda^T G^T G lambda s.t. sum(lambda)=1, lambda>=0"""
    m = len(peaks)
    if m == 1:
        return np.array([1.0])
    if m > max_peaks:
        m = max_peaks
    
    # Build G: G[i, j] = p[peaks[j] - i] if valid
    G = np.zeros((n, m))
    for j in range(m):
        k = int(peaks[j])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        G[idx, j] = p[k - idx]
    
    GtG = G.T @ G  # m x m
    # min lambda^T GtG lambda s.t. 1^T lambda = 1, lambda >= 0
    # KKT: 2*GtG*lambda + 1*nu - mu = 0, mu >= 0, lambda >= 0, lambda_i * mu_i = 0
    
    # For small m, use SLSQP
    if m <= 15:
        def obj(lam):
            return lam @ GtG @ lam
        def jac(lam):
            return 2.0 * GtG @ lam
        res = minimize(obj, np.ones(m)/m, jac=jac, method='SLSQP',
                       bounds=[(0,1)]*m,
                       constraints=[{'type':'eq','fun':lambda x: x.sum()-1.0,
                                     'jac':lambda x: np.ones(m)}],
                       options={'maxiter': 200, 'ftol': 1e-12})
        lam = np.maximum(res.x, 0)
        lam = lam / lam.sum()
        return lam
    else:
        # Projected gradient for larger m
        lam = np.ones(m) / m
        for _ in range(100):
            grad = 2.0 * GtG @ lam
            lam = lam - 0.01 * grad
            lam = np.maximum(lam, 0)
            lam = lam / lam.sum()
        return lam

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:

    s = f.sum()

    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi = fi / ss
    return fi

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using optimal subgradient multiplicative weights
(minimum-norm element in the subdifferential), adaptive-temperature L-BFGS in log-space,
active-set QP for optimal peak weighting, and multi-length exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n*max(conv(a,a)) assuming sum(a)=1. Returns score and correlation."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max(), b


def softmax_z(z):
    """Stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def smooth_obj(z, T, n):
    """Smoothed objective: 2n*T*logsumexp(b/T) where p = softmax(z)."""
    p = softmax_z(z)
    b = fast_convolve(p)
    scaled = b / T
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return 2.0 * n * T * lse


def smooth_grad(z, T, n):
    """Gradient of smoothed objective w.r.t. z."""
    p = softmax_z(z)
    b = fast_convolve(p)
    scaled = b / T
    m = scaled.max()
    w = np.exp(scaled - m)
    w = w / w.sum()
    # grad_p[i] = 2n * sum_k w[k] * p[k-i]
    fft_size = 1 << (3 * n - 1).bit_length()
    W = rfft(w, fft_size)
    P = rfft(p, fft_size)
    correl = irfft(np.conj(W) * P, fft_size)
    grad_p = 2.0 * n * correl[:n]
    grad_z = p * (grad_p - np.dot(p, grad_p))
    return grad_z


def lbgfs_smooth(p_init, time_limit, T=50.0, maxiter=3000):
    """L-BFGS on softmax parameterization with soft-max smoothing."""
    n = len(p_init)
    p = np.maximum(p_init, 1e-15).astype(np.float64)
    p = p / p.sum()
    z0 = np.log(p)
    z0 -= z0.mean()
    t0 = time.monotonic()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smooth_obj(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        return smooth_grad(z, T, n)

    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10,
                                'maxfun': maxiter * 80, 'disp': False})
        return softmax_z(res.x)
    except Exception:
        return p_init.copy()


def solve_lambda_qp_fast(G, m):
    """Solve min lambda^T G^T G lambda s.t. sum(lambda)=1, lambda>=0.
    Uses active set method for small m."""
    A = G.T @ G  # m x m
    lam = np.ones(m) / m
    active = np.ones(m, dtype=bool)

    for _ in range(m + 1):
        # Solve on active set: min lam_a^T A_aa lam_a s.t. sum(lam_a) = 1
        idx = np.where(active)[0]
        A_aa = A[np.ix_(idx, idx)]
        # KKT: 2*A_aa*lam_a + nu*1 = 0, 1^T lam_a = 1
        # lam_a = -nu * inv(2*A_aa) * 1
        try:
            v = np.linalg.solve(2.0 * A_aa, np.ones(len(idx)))
            denom = v.sum()
            if abs(denom) < 1e-15:
                break
            nu = -1.0 / denom
            lam_a = nu * v
        except np.linalg.LinAlgError:
            break

        # Check if all active lambda are positive
        if np.all(lam_a > -1e-10):
            lam = np.zeros(m)
            lam[idx] = np.maximum(lam_a, 0)
            lam = lam / lam.sum()
            return lam

        # Find the most violated (most negative) and deactivate it
        worst_pos = np.argmin(lam_a)
        active[idx[worst_pos]] = False

        # If nothing active, return uniform
        if not active.any():
            return np.ones(m) / m

    # Fallback
    return np.ones(m) / m


def multiplicative_weights_opt(p_init, time_limit, eps_start=0.10, eps_end=0.001,
                                eta_start=0.08, eta_end=0.001,
                                max_iters=50000, verbose=True, label=""):
    """
    Multiplicative weights with optimal (minimum-norm) subgradient.
    Epsilon anneals from eps_start to eps_end to go from broad to narrow active set.
    """
    n = len(p_init)
    p = np.maximum(p_init.copy(), 1e-15).astype(np.float64)
    p = p / p.sum()
    best_p = p.copy()
    best_score, _ = compute_score(p)
    t0 = time.monotonic()
    lr_decay = (eta_end / eta_start) ** (1.0 / max(1, max_iters))
    eps_decay = (eps_end / eps_start) ** (1.0 / max(1, max_iters))
    eta = eta_start
    eps = eps_start
    stuck = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(p)
        cur = 2.0 * n * b.max()

        if cur < best_score - 1e-14:
            best_score = cur
            best_p = p.copy()
            stuck = 0
        else:
            stuck += 1

        # Find active peaks
        b_max = b.max()
        threshold = b_max * (1.0 - eps)
        active_idx = np.where(b >= threshold)[0]

        # Limit to top 12 peaks
        if len(active_idx) > 12:
            top = np.argsort(b[active_idx])[::-1][:12]
            active_idx = active_idx[top]

        # Compute optimal subgradient
        if len(active_idx) == 1:
            k = int(active_idx[0])
            g = np.zeros(n)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            ii = np.arange(js, je + 1)
            g[ii] = p[k - ii]
        else:
            # Build G matrix
            G = np.zeros((n, len(active_idx)))
            for j, k in enumerate(active_idx):
                k = int(k)
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                ii = np.arange(js, je + 1)
                G[ii, j] = p[k - ii]
            lam = solve_lambda_qp_fast(G, len(active_idx))
            g = G @ lam

        # Multiplicative update
        upd = -eta * g
        upd = np.clip(upd, -4.0, 4.0)
        p = p * np.exp(upd)
        p = np.maximum(p, 1e-300)
        s = p.sum()
        if s < 1e-15:
            p = best_p.copy()
        else:
            p = p / s

        eta *= lr_decay
        eps *= eps_decay

        # Stuck detection with perturbation
        if stuck > 400:
            rng = np.random.default_rng(it * 31 + 7)
            p = best_p + rng.standard_normal(n) * 0.015
            p = np.maximum(p, 1e-15)
            p = p / p.sum()
            stuck = 0
            eta = min(eta * 3.0, eta_start * 0.4)

        if verbose and it % 10000 == 0:
            el = time.monotonic() - t0
            print(f"    {label} iter={it} ({el:.0f}s): eta={eta:.5f} eps={eps:.5f} "
                  f"score={cur:.7f} best={best_score:.7f} |S|={len(active_idx)}")

    return best_p, best_score


def soft_max_mw(p_init, time_limit, T_start=0.5, T_end=0.01,
                eta_start=0.05, eta_end=0.001,
                max_iters=30000, verbose=True, label=""):

    best_p = p.copy()

    t0 = time.monotonic()
    lr_decay = (eta_end / eta_start) ** (1.0 / max(1, max_iters))
    T_decay = (T_end / T_start) ** (1.0 / max(1, max_iters))
    eta = eta_start
    T = T_start
    stuck = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(p)
        cur = 2.0 * n * b.max()

        if cur < best_score - 1e-14:
            best_score = cur
            best_p = p.copy()
            stuck = 0
        else:
            stuck += 1

        # Softmax weights over peaks
        scaled = b / T
        m = scaled.max()

        w = w / w.sum()

        # Gradient: 2n * cross-correlation of w and p

        # Update
        upd = -eta * g
        upd = np.clip(upd, -4.0, 4.0)
        p = p * np.exp(upd)
        p = np.maximum(p, 1e-300)
        s = p.sum()
        if s < 1e-15:
            p = best_p.copy()
        else:
            p = p / s

        eta *= lr_decay
        T *= T_decay

        if stuck > 500:
            rng = np.random.default_rng(it * 17 + 3)
            p = best_p + rng.standard_normal(n) * 0.01
            p = np.maximum(p, 1e-15)
            p = p / p.sum()
            stuck = 0

        if verbose and it % 10000 == 0:
            el = time.monotonic() - t0
            print(f"    {label} iter={it} ({el:.0f}s): T={T:.4f} eta={eta:.5f} "
                  f"score={cur:.7f} best={best_score:.7f}")

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi = fi / ss
    return fi


def check_update(a, label, best_seq, best_score):
    """Evaluate candidate using evaluate_sequence and update best if improved."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    an = a / s
    score = float(evaluate_sequence(an.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = an.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(an)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Optimal subgradient MW with epsilon annealing (primary)
    Phase 2: Softmax-smoothed MW
    Phase 3: L-BFGS with temperature schedule
    Phase 4: Refined MW (tight active set)
    Phase 5: Low-temperature L-BFGS
    Phase 6: Alternative length exploration
    Phase 7: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 18.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default init: n={len(best_seq)}, score={best_score:.7f}")

    # === Phase 1: Optimal Subgradient MW (primary workhorse) ===
    print("\n=== Phase 1: Optimal Subgradient MW ===")
    configs_p1 = [
        # (eps_start, eps_end, eta_start, eta_end, time_s)
        (0.15, 0.01, 0.10, 0.005, 250),
        (0.05, 0.003, 0.06, 0.002, 250),
        (0.02, 0.001, 0.04, 0.001, 250),
        (0.01, 0.0005, 0.02, 0.0005, 200),
    ]
    for eps_s, eps_e, eta_s, eta_e, t_alloc in configs_p1:
        rem = deadline - time.monotonic() - 500
        if rem < 60:
            break
        ta = min(t_alloc, rem * 0.35)
        print(f"  MW-Opt: eps=[{eps_s},{eps_e}], eta=[{eta_s},{eta_e}], budget={ta:.0f}s")
        a, _ = multiplicative_weights_opt(best_seq, ta, eps_start=eps_s, eps_end=eps_e,
                                           eta_start=eta_s, eta_end=eta_e, label="MW1")
        best_seq, best_score = check_update(a, f"mw1_e{eps_s}", best_seq, best_score)

    # === Phase 2: Softmax-smoothed MW (different geometry) ===
    print(f"\n=== Phase 2: Softmax MW (best={best_score:.7f}) ===")
    for T_s, T_e, eta_s, t_alloc in [(1.0, 0.01, 0.05, 200), (0.3, 0.005, 0.03, 200)]:
        rem = deadline - time.monotonic() - 400
        if rem < 60:
            break
        ta = min(t_alloc, rem * 0.35)
        print(f"  SoftMW: T=[{T_s},{T_e}], eta={eta_s}, budget={ta:.0f}s")
        a, _ = soft_max_mw(best_seq, ta, T_start=T_s, T_end=T_e,
                           eta_start=eta_s, eta_end=eta_s * 0.02, label="SMW")
        best_seq, best_score = check_update(a, f"smw_T{T_s}", best_seq, best_score)

    # === Phase 3: L-BFGS with temperature schedule ===
    print(f"\n=== Phase 3: L-BFGS (best={best_score:.7f}) ===")
    for T in [80, 40, 20, 10, 5]:
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        ta = min(rem * 0.2, 70.0)
        print(f"  L-BFGS T={T}, budget={ta:.0f}s")
        a = lbgfs_smooth(best_seq, ta, T=T, maxiter=3000)
        best_seq, best_score = check_update(a, f"lbfgs_T{T}", best_seq, best_score)

    # === Phase 4: Refined MW with very tight active set ===
    print(f"\n=== Phase 4: Refined MW (best={best_score:.7f}) ===")
    for eps_s, eta_s, t_alloc in [(0.005, 0.02, 150), (0.001, 0.01, 150), (0.0003, 0.005, 120)]:
        rem = deadline - time.monotonic() - 200
        if rem < 40:
            break
        ta = min(t_alloc, rem * 0.4)
        print(f"  MW-RF: eps={eps_s}, eta={eta_s}, budget={ta:.0f}s")
        a, _ = multiplicative_weights_opt(best_seq, ta, eps_start=eps_s, eps_end=eps_s * 0.3,
                                           eta_start=eta_s, eta_end=eta_s * 0.1, label="MWRF")
        best_seq, best_score = check_update(a, f"mwrf_{eps_s}", best_seq, best_score)

    # === Phase 5: Low-temperature L-BFGS ===
    print(f"\n=== Phase 5: Low-T L-BFGS (best={best_score:.7f}) ===")
    for T in [3, 1.5, 1.0, 0.5]:
        rem = deadline - time.monotonic() - 120
        if rem < 20:
            break
        ta = min(rem * 0.3, 40.0)
        print(f"  L-BFGS T={T}, budget={ta:.0f}s")
        a = lbgfs_smooth(best_seq, ta, T=T, maxiter=5000)
        best_seq, best_score = check_update(a, f"lbfgs_low{T}", best_seq, best_score)

    # === Phase 6: Alternative length exploration ===
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.7f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [1.5, 2.0, 0.75, 1.33]:
            alt_n = int(n_best * factor)
            if 64 <= alt_n <= 8192 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:2]:
            rem = deadline - time.monotonic() - 80
            if rem < 60:
                break
            ta = min(rem * 0.45, 120.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            print(f"  Alt n={alt_n}: MW budget={ta*0.5:.0f}s")
            a, _ = multiplicative_weights_opt(init_alt, ta * 0.5, eps_start=0.05,
                                               eps_end=0.003, eta_start=0.06,
                                               eta_end=0.003, label=f"alt{alt_n}")
            best_alt, _ = check_update(a, f"alt_mw_{alt_n}", best_seq, best_score)
            # If this alt length is now best, polish it
            if best_alt is not None and len(best_alt) == alt_n:
                rem = deadline - time.monotonic() - 50
                if rem > 20:
                    a2 = lbgfs_smooth(best_alt, min(rem * 0.5, 40.0), T=15, maxiter=2000)
                    best_seq, best_score = check_update(a2, f"alt_lb{alt_n}", best_seq, best_score)

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.7f}) ===")
    if best_seq is not None:
        # Very tight MW
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            ta = min(rem * 0.4, 20.0)
            a, _ = multiplicative_weights_opt(best_seq, ta, eps_start=0.001, eps_end=0.0003,
                                               eta_start=0.008, eta_end=0.001, label="fin")
            best_seq, best_score = check_update(a, "final_mw", best_seq, best_score)

        # Final L-BFGS
        rem = deadline - time.monotonic() - 8
        if rem > 8:
            a = lbgfs_smooth(best_seq, min(rem * 0.6, 15.0), T=0.5, maxiter=5000)
            best_seq, best_score = check_update(a, "final_lbfgs", best_seq, best_score)

        # Final MW
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a, _ = multiplicative_weights_opt(best_seq, min(rem * 0.7, 8.0),
                                               eps_start=0.0005, eps_end=0.0001,
                                               eta_start=0.003, eta_end=0.001, label="fin2")
            best_seq, best_score = check_update(a, "final_mw2", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
