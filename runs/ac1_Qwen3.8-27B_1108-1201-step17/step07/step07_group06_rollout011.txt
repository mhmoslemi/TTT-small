def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

    """Cross-correlation of a and b via FFT, result length n."""
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:

    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def compute_grad_a(a, w, n, grad_fft_size):
    """
    Compute gradient of 2n * sum_k w_k * b_k w.r.t. a.
    grad_a[i] = 4n * sum_j w[i+j] * a[j]
    """
    W = rfft(w, grad_fft_size)
    A = rfft(a, grad_fft_size)
    corr = irfft(W * np.conj(A), grad_fft_size)
    return 4.0 * n * corr[:n]

        grad_z = a_curr * (grad_a - S)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - cur_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_from_z(z)
        score_new = score_norm(a_new)

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.008
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            cur_lr = max(0.001, cur_lr * 0.6)
        elif no_improve > max_no_improve // 3:
            cur_lr = max(0.0005, cur_lr * (1.0 - 0.002))

    return best_a, best_score


def lbgfs_polish(a0, time_limit, T=200, maxiter=3000):
    """L-BFGS on soft-max smooth approximation."""
    n = len(a0)
    z0 = z_from_a(a0)
    grad_fft_size = 1 << (3 * n - 2).bit_length()
    t0 = time.monotonic()
    stopped = [False]

    def obj(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        return 2.0 * n * T * (max_b / T + np.log(exp_b.sum() + 1e-300))

    def grad(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        ga = compute_grad_a(a, w, n, grad_fft_size)
        S = np.dot(ga, a)
        return a * (ga - S)

    try:
        res = minimize(fun=obj, x0=z0, jac=grad, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12,
                                'maxfun': maxiter * 30, 'disp': False})
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = a0.copy()
    return a_out


def peak_shave_advance(a0, time_limit, n_trials=50000):
    """
    Advanced peak shaving: identifies the autocorrelation peak, finds top
    contributing pairs, and applies mass-transfer moves (reducing contributors
    while boosting non-contributors) to flatten the peak.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find peak excluding boundaries
        b_search = b.copy()
        b_search[0] = -1e30
        b_search[-1] = -1e30
        peak_k = int(np.argmax(b_search))

        # Contributors to peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_count = min(10, len(idx))
        top_order = np.argsort(contribs)[::-1][:top_count]

        # Non-contributor mask (elements NOT in [js, je])
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je+1] = False
        nc_indices = np.where(non_contrib)[0]
        nc_sum = a[nc_indices].sum() if len(nc_indices) > 0 else 0.0

        improved = False

        # Strategy 1: Reduce top contributor pair
        for ci in top_order:
            i = int(idx[ci])
            j = int(peak_k - i)
            if i == j:
                # Self-contribution: reduce a[i]
                for factor in [0.92, 0.88, 0.83, 0.78, 0.72]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = a_trial / a_trial.sum()
                    sc = score_norm(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
                continue

            if a[j] < 1e-15 or a[i] < 1e-15:
                continue

            for f1 in [0.95, 0.92, 0.88, 0.83, 0.78]:
                for f2 in [1.0, 1.03, 1.06, 1.10]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] = max(0, a_trial[j] * f2)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    sc = score_norm(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        # Strategy 2: Mass transfer from peak contributor to non-contributor
        if not improved:
            for ci in top_order[:5]:
                i = int(idx[ci])
                if a[i] < 1e-14 or nc_sum < 1e-14:
                    continue
                for frac in [0.02, 0.05, 0.08, 0.12, 0.18]:
                    transfer = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= transfer
                    # Distribute to random non-contributors weighted by current value
                    if len(nc_indices) > 0:
                        weights = a_trial[nc_indices]
                        if weights.sum() > 1e-15:
                            weights = weights / weights.sum()
                            a_trial[nc_indices] += transfer * weights
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    sc = score_norm(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        # Strategy 3: Pairwise swap between a peak contributor and a random other
        if not improved and trial % 3 == 0:
            for ci in top_order[:3]:
                i = int(idx[ci])
                if a[i] < 1e-14:
                    continue
                # Pick a random target
                j = int(rng.integers(0, n))
                if j == i or a[j] < 1e-15:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    transfer = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= transfer
                    a_trial[j] += transfer
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    sc = score_norm(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        # Escape mechanism
        if not improved and trial > 500 and trial % 200 == 0:
            a = best_a + rng.standard_normal(n) * 0.008
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def coord_descent_fine(a0, time_limit, n_sweeps=1000):
    """Fine-grained coordinate descent with adaptive factor grid."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(4242)

    coarse_factors = [0.85, 0.9, 0.95, 1.05, 1.1, 1.15]
    fine_factors = [0.97, 0.98, 0.99, 1.01, 1.02, 1.03]

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            if a[i] < 1e-15:
                continue

            # Coarse search
            best_local_score = best_score
            best_local_val = a[i]
            for factor in coarse_factors:
                a_trial = a.copy()
                a_trial[i] *= factor
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                sc = score_norm(a_trial)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local_val = a[i] * factor

            if best_local_val != a[i]:
                a[i] = best_local_val
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True

        if not sweep_improved:
            # Try fine factors on a few elements
            fine_order = order[:min(50, n)]
            any_fine = False
            for i in fine_order:
                if a[i] < 1e-15:
                    continue
                for factor in fine_factors:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    sc = score_norm(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        any_fine = True
                        break
            if not any_fine:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a unit-sum sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def make_initializations(parent, n, rng):
    """Generate diverse initial sequences of length n."""
    candidates = []

    if parent is not None:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            p = p / p.sum()
            candidates.append(p.copy())
            for scale in [0.01, 0.02, 0.04, 0.06]:
                pert = p + rng.standard_normal(n) * scale
                pert = np.maximum(pert, 0)
                pert = pert / pert.sum()
                candidates.append(pert)
            # Cosine perturbations
            for freq in [2, 3, 5, 7]:
                t = np.linspace(0, 2 * np.pi * freq, n)
                pert = p + 0.02 * np.cos(t)
                pert = np.maximum(pert, 0)
                sp = pert.sum()
                if sp > 1e-10:
                    candidates.append(pert / sp)
        else:
            interp = interpolate_sequence(parent, n)
            candidates.append(interp)

    # Uniform
    candidates.append(np.ones(n) / n)

    # Raised cosine (Hann window)
    t = np.linspace(0, 1, n)
    hann = 0.5 * (1 + np.cos(2 * np.pi * t))
    hann = np.maximum(hann, 0.001)
    candidates.append(hann / hann.sum())

    # Triangular
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float) + 1
    candidates.append(tri / tri.sum())

    # Gaussian
    x = np.linspace(-3, 3, n)
    gau = np.exp(-x ** 2 / 3)
    candidates.append(gau / gau.sum())

    # Random
    rand = np.abs(rng.standard_normal(n)) + 0.01
    candidates.append(rand / rand.sum())

    return candidates


def evaluate_and_track(a, label, best_seq, best_score):
    """Evaluate via the official evaluator and track best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Adam log-space, advanced peak shaving, fine coordinate descent,
    and multi-length exploration within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception:
        pass

    if best_seq is None:
        n_default = 768
        best_seq = np.ones(n_default) / n_default
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default uniform: n={len(best_seq)}, score={best_score:.8f}")

    n0 = len(best_seq)
    print(f"Starting best: score={best_score:.8f}, n={n0}")

    # === Phase 1: Adam optimization at parent length (45% time) ===
    print(f"\n=== Phase 1: Adam (n={n0}) ===")
    t1 = 0.45 * (deadline - time.monotonic())
    inits = make_initializations(parent, n0, rng)
    # Sort by initial score
    scored_inits = []
    for init in inits:
        sc = score_norm(init)
        scored_inits.append((sc, init))
    scored_inits.sort(key=lambda x: x[0])

    n_starts = min(5, len(scored_inits))
    for idx in range(n_starts):
        rem = deadline - time.monotonic() - 300
        if rem < 60:
            break
        t_alloc = min(t1 / n_starts, rem * 0.85)
        if t_alloc < 30:
            break

        init_a = scored_inits[idx][1]
        print(f"  Adam start {idx+1}/{n_starts} (init score={scored_inits[idx][0]:.6f}), t={t_alloc:.0f}s")

        # Try 2 temperature schedules
        rem = deadline - time.monotonic() - 280
        if rem < 20:
            break
        t_sub = min(t_alloc * 0.55, rem * 0.7)
        a_res, _ = adam_logspace(init_a, t_sub, T_init=0.06, T_final=0.002,
                                 lr=0.008, cyclic=False)
        best_seq, best_score = evaluate_and_track(a_res, f"adam_{idx}_std", best_seq, best_score)

        rem = deadline - time.monotonic() - 260
        if rem > 30:
            t_sub2 = min(t_alloc * 0.4, rem * 0.6)
            a_res2, _ = adam_logspace(init_a, t_sub2, T_init=0.04, T_final=0.003,
                                      lr=0.006, cyclic=True, cycle_period=0.2)
            best_seq, best_score = evaluate_and_track(a_res2, f"adam_{idx}_cyc", best_seq, best_score)

        print(f"    After start {idx+1}: best={best_score:.8f}")

    # === Phase 2: Intensive local polishing (35% time) ===
    print(f"\n=== Phase 2: Local Polishing (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    t2 = 0.35 * (deadline - time.monotonic())
    t2_start = time.monotonic()
    a_work = best_seq.copy()

    # Alternate peak shaving and coordinate descent
    n_rounds = 4
    for r in range(n_rounds):
        rem = deadline - time.monotonic() - 120
        if rem < 30:
            break
        t_sub = min(t2 / n_rounds, rem * 0.45)
        if t_sub < 15:
            break

        # Peak shave
        a_work, _ = peak_shave_advance(a_work, t_sub, n_trials=80000)
        best_seq, best_score = evaluate_and_track(a_work, f"peak_r{r}", best_seq, best_score)

        # Coord descent
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_sub2 = min(rem * 0.3, 80.0)
            a_work, _ = coord_descent_fine(a_work, t_sub2, n_sweeps=500)
            best_seq, best_score = evaluate_and_track(a_work, f"coord_r{r}", best_seq, best_score)

        # L-BFGS polish
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_sub3 = min(rem * 0.2, 40.0)
            a_lb = lbgfs_polish(a_work, t_sub3, T=100, maxiter=2000)
            best_seq, best_score = evaluate_and_track(a_lb, f"lbfgs_r{r}", best_seq, best_score)
            a_work = best_seq.copy()

        print(f"    Round {r+1}: best={best_score:.8f}")

    # === Phase 3: Multi-length exploration (15% time) ===
    print(f"\n=== Phase 3: Multi-Length (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 80
    if rem > 60:
        alt_lengths = []
        for factor in [1.25, 1.5, 0.75, 2.0, 1.125, 0.875]:
            alt_n = int(n_best * factor)
            if 384 <= alt_n <= 4096 and alt_n != n_best:
                if alt_n not in alt_lengths:
                    alt_lengths.append(alt_n)

        t_per_len = min(rem * 0.25, 100.0)
        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic() - 50
            if rem < 40:
                break
            init_alt = interpolate_sequence(best_seq, alt_n)
            # Quick Adam
            t_sub = min(t_per_len * 0.6, rem * 0.5)
            a_alt, _ = adam_logspace(init_alt, t_sub, T_init=0.05, T_final=0.003,
                                     lr=0.007, cyclic=False)
            best_seq, best_score = evaluate_and_track(a_alt, f"alt_n{alt_n}_adam", best_seq, best_score)

            # Quick peak shave
            rem = deadline - time.monotonic() - 40
            if rem > 20:
                t_sub2 = min(t_per_len * 0.35, rem * 0.4)
                a_alt2, _ = peak_shave_advance(a_alt, t_sub2, n_trials=30000)
                best_seq, best_score = evaluate_and_track(a_alt2, f"alt_n{alt_n}_peak", best_seq, best_score)
            print(f"  n={alt_n}: best={best_score:.8f}")

    # === Phase 4: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 4: Final Polish (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 15

    if rem > 30:
        # Very low temperature Adam
        t_sub = min(rem * 0.3, 60.0)
        a_fin, _ = adam_logspace(best_seq, t_sub, T_init=0.015, T_final=0.0005,
                                 lr=0.003, cyclic=False)
        best_seq, best_score = evaluate_and_track(a_fin, "final_adam", best_seq, best_score)

    rem = deadline - time.monotonic() - 10
    if rem > 20:
        t_sub = min(rem * 0.4, 50.0)
        a_fin2, _ = peak_shave_advance(best_seq, t_sub, n_trials=50000)
        best_seq, best_score = evaluate_and_track(a_fin2, "final_peak", best_seq, best_score)

    rem = deadline - time.monotonic() - 5
    if rem > 15:
        t_sub = min(rem * 0.5, 30.0)
        a_fin3, _ = coord_descent_fine(best_seq, t_sub, n_sweeps=300)
        best_seq, best_score = evaluate_and_track(a_fin3, "final_coord", best_seq, best_score)

    # Final verification with official evaluator
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK: score={best_score:.8f}")

    return [float(x) for x in best_seq]
