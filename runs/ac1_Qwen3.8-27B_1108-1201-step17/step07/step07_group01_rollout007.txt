import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def softmax_from_z(z):
    """Convert z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def log_softmax_obj(z, T, n):
    """Soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    scaled = b / T
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return 2.0 * n * T * lse


def log_softmax_grad(z, T, n):
    """Gradient of soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    scaled = b / T
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()
    # Gradient of soft_max wrt a: (2/T) * cross_correlation(w_shifted, a)
    # conv[k] = sum_i a[i]*a[k-i], d(conv[k])/da[i] = 2*a[k-i]
    # d(soft_max)/da[i] = sum_k w[k] * 2*a[k-i]
    # = 2 * (w * reversed_a)[i] (with appropriate shifting)
    # Use FFT for efficiency
    n_w = len(w)
    fft_size = 1 << (n_w + n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # cross_corr[i] = sum_k w[k] * a[k-i] = (w * flip(a))[i]
    # Actually: sum_k w[k] * a[k-i] where k-i must be in [0,n-1]
    # This is the cross-correlation of w with a
    correl = irfft(np.conj(W) * A, fft_size)
    grad_a = 2.0 * correl[:n]
    # Project to tangent space of simplex: grad_z = a * (grad_a - sum(a*grad_a))
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)
    return grad_z


def lbgfs_logspace(a_init, time_limit, T, maxiter=5000, verbose=False, label=""):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    z0 = np.log(a)
    z0 = z0 - z0.mean()
    t0 = time.monotonic()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return log_softmax_obj(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        return log_softmax_grad(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        z_opt = res.x
        a_out = softmax_from_z(z_opt)
    except Exception:
        a_out = a_init.copy()
    return a_out

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad_a = 2.0 * correl[:n]

        # Multiplicative update
        log_update = -eta * grad_a
        log_update = np.clip(log_update, -1.0, 1.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        # Decay eta slightly
        eta *= 0.99995

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} MD iter={it} ({elapsed:.0f}s): "
                  f"score={current_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def evaluate_candidate(a, best_seq, best_score, label=""):
    """Evaluate a candidate sequence using the actual evaluate_sequence function."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimum auto-convolution peak:
    Phase 1: L-BFGS with temperature annealing (primary)
    Phase 2: Mirror descent for geometric exploration
    Phase 3: Active peak equalization
    Phase 4: Multiplicative peak shaving
    Phase 5: Alternative length exploration
    Phase 6: Final polish cycle
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        pass

    if parent is None:
        # Fallback: start with a structured initialization
        n0 = 768
        x = np.linspace(0, 1, n0)
        parent = np.exp(-3.0 * x) + 0.1 * np.cos(4 * np.pi * x)
        parent = np.maximum(parent, 0.01)
        parent = parent / parent.sum()
        p_score = float(evaluate_sequence(parent.tolist()))
        best_seq = parent.copy()
        best_score = p_score
        print(f"Generated initial: n={n0}, score={p_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: L-BFGS with temperature annealing ===
    print(f"\n=== Phase 1: L-BFGS Temperature Annealing (best={best_score:.8f}) ===")
    temperatures = [100, 50, 30, 20, 10, 7, 5, 3, 2, 1.5, 1]
    for T in temperatures:
        rem = deadline - time.monotonic() - 200
        if rem < 20:
            break
        t_alloc = min(rem * 0.3, 120.0)
        if T <= 5:
            t_alloc = min(rem * 0.25, 90.0)
        print(f"  L-BFGS T={T}, n={n_best}, budget={t_alloc:.0f}s")
        a_out = lbgfs_logspace(best_seq, t_alloc, T=T, maxiter=4000,
                               verbose=True, label=f"lbfs_T{T}")
        best_seq, best_score = evaluate_candidate(a_out, best_seq, best_score,
                                                    f"lbfs_T{T}")

    # === Phase 2: Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.8f}) ===")
    md_configs = [
        (10.0, 0.008, "MD_T10"),
        (5.0, 0.005, "MD_T5"),
        (3.0, 0.003, "MD_T3"),
        (2.0, 0.002, "MD_T2"),
    ]
    for T, eta, lbl in md_configs:
        rem = deadline - time.monotonic() - 250
        if rem < 30:
            break
        t_alloc = min(rem * 0.3, 100.0)
        print(f"  {lbl}: T={T}, eta={eta}, budget={t_alloc:.0f}s")
        a_md, _ = mirror_descent_short(best_seq, t_alloc, T=T, eta=eta, label=lbl)
        best_seq, best_score = evaluate_candidate(a_md, best_seq, best_score, lbl)

    # === Phase 3: Active Peak Equalization ===
    print(f"\n=== Phase 3: Active Peak Equalization (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_alloc = min(rem * 0.5, 120.0)
        print(f"  PE: budget={t_alloc:.0f}s")
        a_pe, _ = peak_equalization(best_seq, t_alloc, verbose=True, label="PE")
        best_seq, best_score = evaluate_candidate(a_pe, best_seq, best_score, "PE")

    # === Phase 4: Multiplicative Peak Shaving ===
    print(f"\n=== Phase 4: Multiplicative Peak Shaving (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 30:
        t_alloc = min(rem * 0.6, 150.0)
        print(f"  MS: budget={t_alloc:.0f}s")
        a_ms, _ = multiplicative_shave(best_seq, t_alloc, verbose=True, label="MS")
        best_seq, best_score = evaluate_candidate(a_ms, best_seq, best_score, "MS")

    # === Phase 5: Low-temperature L-BFGS + Peak Shaving cycle ===
    print(f"\n=== Phase 5: Final Polish Cycle (best={best_score:.8f}) ===")
    for cycle in range(3):
        rem = deadline - time.monotonic() - 50
        if rem < 30:
            break

        # Low-T L-BFGS
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            t_alloc = min(rem * 0.4, 60.0)
            T = max(1, 3 - cycle)
            a_out = lbgfs_logspace(best_seq, t_alloc, T=T, maxiter=3000,
                                   verbose=False, label=f"polish_lbfs_{cycle}")
            best_seq, best_score = evaluate_candidate(a_out, best_seq, best_score,
                                                        f"polish_lbfs_{cycle}")

        # Peak shaving
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_alloc = min(rem * 0.5, 50.0)
            a_ms, _ = multiplicative_shave(best_seq, t_alloc, verbose=False,
                                            label=f"polish_ms_{cycle}")
            best_seq, best_score = evaluate_candidate(a_ms, best_seq, best_score,
                                                        f"polish_ms_{cycle}")

        # Peak equalization
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem * 0.5, 30.0)
            a_pe, _ = peak_equalization(best_seq, t_alloc, verbose=False,
                                         label=f"polish_pe_{cycle}")
            best_seq, best_score = evaluate_candidate(a_pe, best_seq, best_score,
                                                        f"polish_pe_{cycle}")

    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    alt_lengths = []
    for factor in [1.1, 1.2, 0.9, 1.15, 0.85]:
        alt_n = int(n_best * factor)
        if 256 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
            alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic() - 40
        if rem < 30:
            break
        t_alloc = min(rem * 0.3, 60.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        # Quick L-BFGS
        a_alt = lbgfs_logspace(init_alt, t_alloc * 0.6, T=10, maxiter=2000,
                               verbose=False, label=f"alt_{alt_n}")
        best_seq, best_score = evaluate_candidate(a_alt, best_seq, best_score,
                                                    f"alt_{alt_n}")

        # Quick peak shave
        rem = deadline - time.monotonic() - 20
        if rem > 10 and best_seq is not None and len(best_seq) == alt_n:
            t_alloc2 = min(rem * 0.5, 30.0)
            a_ms_alt, _ = multiplicative_shave(best_seq, t_alloc2, verbose=False,
                                                 label=f"alt_ms_{alt_n}")
            best_seq, best_score = evaluate_candidate(a_ms_alt, best_seq, best_score,
                                                        f"alt_ms_{alt_n}")

    # === Phase 7: Final micro-polish ===
    print(f"\n=== Phase 7: Final Micro-Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 5
    if rem > 5:
        t_alloc = min(rem * 0.7, 20.0)
        a_ms_fin, _ = multiplicative_shave(best_seq, t_alloc, verbose=False,
                                            label="final_ms")
        best_seq, best_score = evaluate_candidate(a_ms_fin, best_seq, best_score,
                                                    "final_ms")

    rem = deadline - time.monotonic() - 3
    if rem > 3:
        a_pe_fin, _ = peak_equalization(best_seq, min(rem * 0.7, 10.0),
                                         verbose=False, label="final_pe")
        best_seq, best_score = evaluate_candidate(a_pe_fin, best_seq, best_score,
                                                    "final_pe")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
