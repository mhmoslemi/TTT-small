import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    """Compute score for normalized sequence (sum=1)."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()

def simplex_project(v):
    """Project vector onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        idx = np.nonzero(cond)[0][-1]
        theta = (cssv[idx] - 1.0) / (idx + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def build_conv_matrix(a):
    """Build matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():
    import cvxpy as cp
    global _lp_solver_cached
    if '_lp_solver_cached' in globals() and _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def fictitious_play_enhanced(n, time_limit, initial=None, solver=None):

def smooth_gradient_step(a, beta, lr):
    """One step of projected subgradient using log-sum-exp smoothing."""
    n = len(a)
    conv = fast_convolve(a)
    shifted = conv - conv.max()
    weights = np.exp(beta * shifted)
    weights /= weights.sum()

    # Gradient: grad[j] = 2n * sum_k weights[k] * 2 * a[k-j] (for valid k-j)
    # = 4n * (weights * a) cross-correlation
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-14:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * weights[k] * a[k - j]
    grad *= 2.0 * n

    # Projected gradient step
    a_new = a - lr * grad
    a_new = simplex_project(a_new)
    return a_new

def smooth_anneal(a0, time_limit, beta_start=5.0, beta_end=800.0, lr_init=0.005):
    """Anneal beta from small to large with projected subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    n_total_steps = 3000
    for step in range(n_total_steps):
        if time.monotonic() - t0 > time_limit:
            break
        progress = step / max(n_total_steps - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress

        a_new = smooth_gradient_step(a, beta, lr)
        sc = score_of(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new

        if no_improve > 100:
            lr *= 0.8
            no_improve = 0
        lr *= 0.9995

    return best_a, best_score

def coord_descent_pairwise(a0, time_limit):
    """Coordinate descent with single and pairwise moves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    factors_single = [0.5, 0.7, 0.85, 0.93, 1.07, 1.15, 1.3, 1.6]
    factors_pair = [0.7, 0.85, 1.15, 1.3]

    sweep = 0
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-12:
                continue

            # Single coordinate move
            for f in factors_single:
                a[i] = old_val * f
                sc = score_of(a)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a.copy()
                    old_val = a[i]
                    improved = True
                else:
                    a[i] = old_val

            # Pairwise move with a nearby coordinate
            if sweep % 3 == 0 and i < n - 1:
                j = i + 1 if rng.random() < 0.5 else (i - 1 if i > 0 else i + 1)
                if j == i or a[j] < 1e-12:
                    continue
                old_j = a[j]
                for f1 in factors_pair:
                    for f2 in factors_pair:
                        a[i] = old_val * f1
                        a[j] = old_j * f2
                        sc = score_of(a)
                        if sc < best_score - 1e-13:
                            best_score = sc
                            best_a = a.copy()
                            old_val = a[i]
                            old_j = a[j]
                            improved = True
                        else:
                            a[i] = old_val
                            a[j] = old_j

        if not improved and sweep > 10:
            break

    return best_a, best_score

def basin_hopping(a0, time_limit, n_restarts=20):
    """Basin hopping: random perturbation + local search."""
    n = len(a0)
    best_a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = best_a.sum()
    best_a = best_a / s
    best_score = score_of(best_a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        # Random perturbation
        sigma = 0.05 / (1 + restart * 0.3)
        perturbed = best_a + rng.standard_normal(n) * sigma
        perturbed = simplex_project(perturbed)

        # Quick local search
        a_local, s_local = coord_descent_pairwise(perturbed, min(5.0, (time_limit - (time.monotonic() - t0)) * 0.4))
        if s_local < best_score:
            best_score = s_local
            best_a = a_local.copy()

        # Also try from a random point
        if restart % 5 == 0:
            random_a = simplex_project(rng.standard_normal(n) * 0.3 + best_a)
            a_r, s_r = coord_descent_pairwise(random_a, min(3.0, (time_limit - (time.monotonic() - t0)) * 0.3))
            if s_r < best_score:
                best_score = s_r
                best_a = a_r.copy()

    return best_a, best_score

def peak_shave(a0, time_limit):

    n = len(a0)

                    best_a = a.copy()
                    improved = True
                    break

                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
            if not improved:

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_fourier(n, rng, n_harmonics=None):
    """Fourier-based initialization."""
    if n_harmonics is None:
        n_harmonics = int(rng.integers(3, 15))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

    print(f"\n=== Phase 9: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            a_f1, _ = peak_shave(best_seq, min(rem * 0.3, 30.0))
            best_seq, best_score = check_score(a_f1, best_seq, best_score, "final_ps")

        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_f2, _ = coord_descent_pairwise(best_seq, min(rem * 0.5, 20.0))
            best_seq, best_score = check_score(a_f2, best_seq, best_score, "final_cd")

        rem = deadline - time.monotonic() - 5
        if rem > 8:
            a_f3, _ = peak_shave(best_seq, min(rem * 0.6, 10.0))
            best_seq, best_score = check_score(a_f3, best_seq, best_score, "final_ps2")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
