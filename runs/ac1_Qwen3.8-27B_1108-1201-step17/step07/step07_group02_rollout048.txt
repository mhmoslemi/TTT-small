def spectral_init(n, rng):
    # Create random phases
    phases = rng.uniform(0, 2*np.pi, n)
    # Set magnitude to 1 for all frequencies
    A = np.exp(1j * phases)
    # Ensure A[0] = 1 (DC component)
    A[0] = 1.0
    # Ensure Hermitian symmetry (real time domain)
    # A[n-k] = conj(A[k])
    # Average: A[k] = (A[k] + conj(A[n-k]))/2
    for k in range(1, (n+1)//2):
        A[k] = (A[k] + np.conj(A[n-k])) / 2
    # For even n, A[n/2] should be real
    if n % 2 == 0:
        A[n//2] = np.real(A[n//2])
    # Inverse DFT
    a = np.real(np.fft.ifft(A).astype(np.complex128)) * n  # normalize
    # Actually: ifft(A) gives sum A[f] * exp(2πifk/n) / n
    # We want a[k] = sum A[f] * exp(2πifk/n) / n
    a = np.real(np.fft.ifft(A))
    a = a * n  # to get sum = A[0] = 1... actually let me just normalize
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:

    return a

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

# ============ Fast Convolution ============
def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_convolve_pair(a, b):
    """Compute conv(a, b) via FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

# ============ Subgradient ============
def compute_subgradient(a, active_indices):
    """Compute subgradient of max_k conv_k(a) at point a."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active_indices)
    return g

# ============ Soft-max objective and gradient (in log-space) ============
def soft_max_obj_grad_z(z, T, n):
    """Compute soft-max smoothed objective and gradient in log-space."""
    a = np.exp(z - z.max())
    s = a.sum()
    a = a / s
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = 2.0 * n * log_sum_exp / T
    
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def lbgfs_multi_temp(a0, time_limit, temps, maxiter_per_stage=3000):
    """L-BFGS with geometric temperature schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_val, T_val=T):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = np.exp(z_val - z_val.max())
            a = a / a.sum()
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            return 2.0 * n * (max_Tb + np.log(exp_Tb.sum())) / T_val
        
        def jac(z_val, T_val=T):
            if time.monotonic() - t0 > time_limit:
                return np.zeros(n)
            a = np.exp(z_val - z_val.max())
            a = a / a.sum()
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': maxiter_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        
        elapsed = time.monotonic() - t0
        a_cur = np.exp(z - z.max())
        a_cur = a_cur / a_cur.sum()
        sc = 2.0 * n * fast_convolve(a_cur).max()
        print(f"    L-BFGS T={T:.2f} ({elapsed:.0f}s): score={sc:.6f}")
    
    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out

# ============ Mirror Descent ============
def mirror_descent(a0, time_limit, lr_init=0.1, momentum=0.9):
    """Mirror descent with momentum on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    z = np.log(np.maximum(a, 1e-300))
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b
        
        # Active set: top peaks
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Subgradient
        g = compute_subgradient(a, active)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * g
        z = z - lr * v
        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            z = z + lr * v  # undo
            v = v * 0.5  # damp momentum
        
        if no_improve > 50:
            lr *= 0.8
            no_improve = 0
            momentum *= 0.95
        lr = max(lr, 1e-8)
        
        step += 1
        if step % 5000 == 0:
            print(f"    MirrorDesc {step} ({time.monotonic()-t0:.0f}s): best={best_score:.6f} lr={lr:.6f}")
    
    return best_a, best_score

# ============ Peak Shaving ============
def peak_shave(a0, time_limit, n_steps=10000):
    """Aggressive peak shaving."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find the peak (exclude boundaries which are naturally small)
        b_inner = b.copy()
        b_inner[0] = -np.inf
        b_inner[-1] = -np.inf
        peak_k = int(np.argmax(b_inner))
        peak_val = b[peak_k]
        
        # Find top contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        order = np.argsort(contribs)[::-1]
        
        improved = False
        # Try reducing top contributors
        for top_m in [3, 5, 8]:
            top_idx = idx[order[:top_m]]
            for scale in [0.9, 0.85, 0.8, 0.7]:
                a_trial = a.copy()
                total_redund = 0
                for i in top_idx:
                    total_redund += a[i] * (1 - scale)
                    a_trial[i] *= scale
                
                # Distribute the reduced mass to non-contributing positions
                mask = np.ones(n, dtype=bool)
                mask[top_idx] = False
                others = a_trial[mask]
                if others.sum() > 1e-15:
                    a_trial[mask] = others + total_redund * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        # Also try paired reduction
        if not improved and len(order) >= 2:
            for ci in range(min(5, len(order))):
                i = idx[order[ci]]
                j = peak_k - i
                if i == j or a[j] < 1e-14 or a[i] < 1e-14:
                    continue
                for f1 in [0.95, 0.9]:
                    for f2 in [0.95, 0.9]:
                        a_trial = a.copy()
                        diff = a_trial[i]*(1-f1) + a_trial[j]*(1-f2)
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                        mask[j] = False
                        others = a_trial[mask]
                        if others.sum() > 1e-15:
                            a_trial[mask] = others + diff * others / others.sum()
                        a_trial = a_trial / a_trial.sum()
                        sc = 2.0 * n * fast_convolve(a_trial).max()
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved:
            # Try random perturbation
            perturbation = np.random.randn(n) * 0.01
            a_trial = a * (1 + perturbation)
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score - 1e-15:
                a = a_trial
                best_score = sc
                best_a = a.copy()
            else:
                # No improvement, try a larger random step
                if np.random.random() < 0.1:
                    break  # Stuck, exit
    
    return best_a, best_score

# ============ Spectral initialization ============
def spectral_init(n, rng):
    """Initialize with flat-DFT magnitude sequence."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.exp(1j * phases)
    A[0] = 1.0
    # Enforce Hermitian symmetry
    for k in range(1, (n + 1) // 2):
        A[k] = (A[k] + np.conj(A[n - k])) / 2
    if n % 2 == 0:
        A[n // 2] = np.real(A[n // 2])
    a = np.real(np.fft.ifft(A))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

def gerchberg_saxton_init(n, rng, n_iters=100):
    """Gerchberg-Saxton / IFTM initialization for flat autocorrelation."""
    a = rng.random(n)
    a = a / a.sum()
    for _ in range(n_iters):
        A = np.fft.fft(a)
        # Set magnitude to constant (target: flat autocorrelation)
        target_mag = np.ones(n)
        # Keep phase, set magnitude
        phase = np.angle(A)
        A = target_mag * np.exp(1j * phase)
        # Ensure A[0] is real positive
        A[0] = np.abs(A[0])
        # Inverse FFT
        a_new = np.real(np.fft.ifft(A))
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a = a_new / s
        else:
            break
    return a

# ============ Interpolation ============
def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

# ============ Evaluate safely ============
def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
