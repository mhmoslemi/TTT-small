import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.GUROBI, cp.MOSEK, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """LP: given fixed a (sum=1), find b (sum=1, b>=0) minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def solve_qp_refine(a_target, a_current, M_bound, solver=None):
    """QP: minimize ||a - a_target||^2 s.t. conv(a_current, a) <= M_bound, sum(a)=1, a>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_current)
    a_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a_current)
    constraints = [
        T @ a_var <= M_bound,
        cp.sum(a_var) == 1
    ]
    obj = cp.sum_squares(a_var - a_target)
    prob = cp.Problem(cp.Minimize(obj), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    a_val = np.asarray(a_var.value, dtype=np.float64)
    a_val = np.maximum(a_val, 0)
    s = a_val.sum()
    if s < 1e-15:
        return None
    return a_val / s

            ema_b = b.copy()

            w = exp_b / exp_b.sum()

            # Compute gradient via FFT for speed
            # grad_a[j] = 2*n * sum_k w[k] * (a_local[k-j] + a_local[j-k]) for valid indices
            # = 2*n * 2 * (w * a_local)[correlation form]
            # Use: grad_a = 2*n * (w_rev * a_local + w * a_local_rev) ... 
            # Simpler: use the direct approach with FFT
            W = rfft(w, 2 * n)
            A = rfft(a_local, 2 * n)
            # conv(w, a) gives sum_k w[k]*a[j-k] for each j
            grad_a = irfft(W * A, 2 * n)[:n]
            # Also need conv(w, a_reversed) contribution
            a_rev = a_local[::-1]
            A_rev = rfft(a_rev, 2 * n)
            grad_a += irfft(W * A_rev, 2 * n)[-n:]
            
            # Actually let me just do it directly but efficiently
            # conv(a,a)[k] = sum_j a[j]*a[k-j]
            # d/d a[j] conv(a,a)[k] = a[k-j] if valid + a[k-j] if k-j != j and valid
            # Total: d/d a[j] max ≈ sum_k w[k] * (a[k-j] + [a[j-k] if j != k-j])
            # = sum_k w[k]*a[k-j] + sum_k w[k]*a[j-k]  (the two terms are the same by symmetry)
            # = 2 * sum_k w[k]*a[k-j]  (when we account for both contributions)
            # Actually for conv(a,a): d conv(a,a)[k]/d a[j] = a[k-j] (if 0<=k-j<n) 
            #   + a[k-j] (if 0<=k-j<n and k-j != j, i.e., the "other" occurrence)
            # For k != 2j: d conv(a,a)[k]/d a[j] = a[k-j] + a[j-k] if both valid
            # For k == 2j: d conv(a,a)[k]/d a[j] = 2*a[j]
            # This is complex. Let me use FFT approach.
            # 
            # The gradient of conv(a,a) wrt a is related to the circular correlation.
            # For linear convolution of length 2n-1, the gradient at a[j] is:
            # sum_k w[k] * (contribution of a[j] to conv[k])
            # a[j] contributes to conv[k] when k-j is in [0,n-1], giving a[k-j]
            # So grad[j] = sum_{k: 0<=k-j<n} w[k] * a[k-j] = sum_{m: 0<=m<n, 0<=j+m<2n-1} w[j+m] * a[m]
            # = (w_shifted * a)[j] where w is zero-padded
            
            # Let me just compute it directly using the matrix form for clarity
            # For n up to ~2048 this should be fine with vectorization
            grad_a = np.zeros(n)
            for j in range(n):
                # a[j] appears in conv[k] for k from j to j+n-1 (capped at 2n-2)
                k_start = j
                k_end = min(j + n, 2 * n - 1)
                if k_end <= k_start:
                    continue
                indices = np.arange(k_start, k_end)
                paired = indices - j  # these are the a[paired] values
                grad_a[j] = np.dot(w[k_start:k_end], a_local[paired])
            
            # Symmetry: also the "other" occurrence
            # conv[k] = sum_i a[i]*a[k-i], a[j] appears when i=j (gives a[k-j]) 
            # and when k-i=j i.e. i=k-j (gives a[k-j]=a[j]... no, gives a[k-(k-j)]=a[j])
            # Wait, when i = k-j: the term is a[k-j]*a[j], derivative wrt a[j] is a[k-j]
            # So total derivative is a[k-j] + a[k-j] = 2*a[k-j] when k != 2j
            # And 2*a[j] when k = 2j
            # So actually grad[j] = sum_k w[k] * (a[k-j] + a[k-j]) for valid k where k!=2j
            # = 2 * sum_k w[k] * a[|k-j|] ... no this isn't right either.
            
            # Let me just redo this properly.
            # conv(a,a)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * a[k-i]
            # d/d a[j]: 
            #   If j is in [max(0,k-n+1), min(n-1,k)]: term a[j]*a[k-j], derivative = a[k-j]
            #   If k-j is in [max(0,k-n+1), min(n-1,k)] and k-j != j: term a[k-j]*a[j], derivative = a[k-j]
            #   If k-j == j (k=2j): only one term a[j]^2, derivative = 2*a[j]
            #
            # So d conv[k]/d a[j] = a[k-j] * [I(0<=k-j<n) + I(0<=k-j<n, k-j!=j)]
            #                        = a[k-j] * [I(0<=k-j<n)] * 2 - a[j]*I(k=2j)
            # Hmm, this is getting too complex. Let me just use the simple FFT approach.
            
            break  # This direct approach is too slow, let me use a different method
        
        # Use a simpler gradient computation via the "T matrix" approach
        # Actually for L-BFGS the gradient computation is the bottleneck.
        # Let me use a finite-difference-free approach with FFT.
        
        def func2(z):
            if monotonic() - t0 > time_limit:
                stopped[0] = True
                return 1e15
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a_local = exp_z / exp_z.sum()
            b = fast_convolve(a_local)
            max_b = b.max()
            scaled_b = (b - max_b) / T
            exp_b = np.exp(scaled_b)
            lse = max_b + T * np.log(exp_b.sum())
            return 2.0 * n * lse

        def jac2(z):
            if monotonic() - t0 > time_limit:
                stopped[0] = True
                return np.zeros(n)
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a_local = exp_z / exp_z.sum()
            b = fast_convolve(a_local)
            max_b = b.max()
            scaled_b = (b - max_b) / T
            exp_b = np.exp(scaled_b)
            w = exp_b / exp_b.sum()
            
            # Gradient of sum_k w[k]*conv(a,a)[k] wrt a
            # conv(a,a)[k] = sum_i a[i]*a[k-i]
            # d/d a[j] = sum_k w[k] * a[k-j] * I(0<=k-j<n) + sum_k w[k] * a[k-j] * I(0<=k-j<n, k-j!=j)
            # = 2 * sum_k w[k] * a[k-j] * I(0<=k-j<n) - sum_k w[k] * a[j] * I(k=2j)
            # The second term: k=2j, so = w[2j]*a[j] if 2j < 2n-1
            # So grad[j] = 2 * sum_{k: 0<=k-j<2n-1} w[k] * a[k-j] - w[2j]*a[j]*I(2j<2n-1)
            # Let m = k-j, so m ranges from 0 to min(n-1, 2n-2-j)
            # grad[j] = 2 * sum_{m=0}^{min(n-1, 2n-2-j)} w[j+m] * a[m] - w[2j]*a[j]*I(2j<2n-1)
            # 
            # The sum part is a cross-correlation: (w * a)[j] where w is treated as a sequence
            # Let's compute via FFT: pad w and a to length 2n, compute circular cross-correlation
            
            # Actually, the sum_{m} w[j+m]*a[m] for m=0..n-1 is:
            # (flipped_w * a) circular conv at position (len-1-j)... 
            # Let me just use the direct matrix approach with numpy for speed
            # For n <= 2048, a (2n-1) x n matrix times w (2n-1) gives n results
            
            # Efficient computation:
            # sum_m w[j+m]*a[m] for m in valid range
            # This is the cross-correlation of w and a, evaluated at offset j
            # = (w ⊛ a)[j] where ⊛ is cross-correlation
            # Using FFT: ifft(fft(w padded) * conj(fft(a padded)))... 
            # Actually: cross_corr[j] = sum_m w[j+m] * a[m] = (flip(w) * a)[N-1-j]
            # where * is linear convolution and N is appropriate length
            
            w_padded = np.zeros(2 * n - 1)
            w_padded[:len(w)] = w
            # sum_m w[m+j]*a[m] for m=0..n-1 (with w having valid entries at 0..2n-2)
            # = (w_rev * a)[2n-2-j] where w_rev = w[::-1] and * is linear conv
            w_rev = w[::-1]
            W_rev = rfft(w_rev, 4 * n)
            A = rfft(a_local, 4 * n)
            cross = irfft(W_rev * A, 4 * n)
            # cross[k] = sum_i w_rev[i] * a[k-i] = sum_i w[2n-2-i] * a[k-i]
            # We want sum_m w[j+m]*a[m]. Let i = 2n-2-m, then m = 2n-2-i
            # sum_i w[j+2n-2-i]*a[2n-2-i]... this is getting confusing.
            
            # Let me just do it with a simple loop that's vectorized over j:
            # For each j, grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m] (capped at 2n-2)
            grad_a = np.zeros(n)
            for j in range(n):
                m_max = min(n - 1, 2 * n - 2 - j)
                if m_max < 0:
                    continue
                w_slice = w[j:j + m_max + 1]
                a_slice = a_local[:m_max + 1]
                grad_a[j] = 2.0 * np.dot(w_slice, a_slice)
                # Subtract the double-counted diagonal term
                if 2 * j < 2 * n - 1:
                    grad_a[j] -= w[2 * j] * a_local[j]
            
            # Transform to z-space: a = softmax(z) (up to normalization)
            # d a / d z = a * (I - a a^T) * ... 
            # Actually: a[j] = exp(z[j]) / sum(exp(z)), so da[j]/dz[l] = a[j](delta[jl] - a[l])
            # grad_z[l] = sum_j grad_a[j] * da[j]/dz[l] = sum_j grad_a[j]*a[j]*(delta[jl]-a[l])
            # = grad_a[l]*a[l] - a[l]*sum_j grad_a[j]*a[j]
            S = np.dot(grad_a, a_local)
            grad_z = a_local * (grad_a - a_local * S)
            return grad_z

        maxiter = max(100, int(rem / 0.05))
        try:
            res = minimize(
                fun=func2, x0=z0, jac=jac2,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                         'maxfun': maxiter * 10, 'disp': False}
            )
            if not stopped[0]:
                z_final = res.x
                z_shifted = z_final - z_final.max()
                a_out = np.exp(z_shifted)
                a_out = a_out / a_out.sum()
                a_out = np.maximum(a_out, 0)
                s = a_out.sum()
                if s > 1e-10:
                    a_out = a_out / s
                    score = 2.0 * n * fast_convolve(a_out).max()
                    if score < best_score:
                        best_score = score
                        best_a = a_out.copy()
                    a = a_out
        except Exception:
            pass

    return best_a, best_score


def pairwise_local_search(a0, time_limit, max_pairs=50):
    """Local search that tries pairwise transfers between elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    no_improve = 0

    while monotonic() - t0 < time_limit:
        if no_improve > 200:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try to reduce the peak by transferring mass from high contributors to low ones
        top_idx = idx[np.argsort(contribs)[::-1][:max_pairs]]
        bottom_candidates = np.argsort(a)[::-1][::-1]  # smallest elements first
        bottom_candidates = bottom_candidates[:max_pairs * 2]
        
        improved = False
        for i in top_idx[:20]:
            if a[i] < 1e-10:
                continue
            for j in bottom_candidates:
                if j == i:
                    continue
                # Try transferring a small amount from i to j
                for frac in [0.05, 0.1, 0.2]:
                    amount = a[i] * frac
                    if amount < 1e-12:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= amount
                    a_trial[j] += amount
                    if a_trial[i] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            # Try random perturbation
            a_trial = a + np.random.standard_normal(n) * 0.001
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a.copy()
                no_improve = 0

    return best_a, best_score


def coordinate_descent(a0, time_limit):
    """Coordinate descent: optimize one element at a time."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    
    order = np.random.permutation(n)
    it = 0
    while monotonic() - t0 < time_limit:
        j = order[it % n]
        it += 1
        
        # Optimize a[j] with all others fixed (re-normalize)
        # Try several values
        current = a[j]
        best_val = current
        best_s = best_score
        
        # The score as a function of a[j] when others are fixed:
        # We need a[j] >= 0 and sum = 1, so a[j] can range from 0 to 1
        # But we should try values near the current one
        candidates = [current * 0.5, current * 0.75, current * 1.25, current * 1.5,
                      current + 0.001, current - 0.001, 0.0]
        # Also try values based on the constraint
        for trial in candidates:
            if trial < 0:
                continue
            a_trial = a.copy()
            a_trial[j] = trial
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                continue
            a_trial = a_trial / s_trial
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_s - 1e-13:
                best_s = score
                best_val = trial
        
        if best_val != current:
            a[j] = best_val
            a = a / a.sum()
            if best_s < best_score:
                best_score = best_s
                best_a = a.copy()
    
    return best_a, best_score


def spectral_init(n):
    """Initialize with a spectrally-shaped sequence."""
    # Start with a raised cosine window and add some structure
    t = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    # Add a secondary modulation
    a += 0.1 * np.cos(2 * np.pi * 3 * t + 0.5)
    a += 0.05 * np.cos(2 * np.pi * 7 * t + 1.2)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def damped_qp_loop(a0, time_limit, solver=None, max_rounds=100):
    """Iterative QP refinement: alternate between LP best response and QP projection."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    
    for rnd in range(max_rounds):
        if monotonic() - t0 > time_limit:
            break
        
        # Get LP best response
        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            continue
        
        # Use QP to find a point close to b but with bounded convolution
        # Set M_bound slightly above current max
        current_max = fast_convolve(a).max()
        M_bound = current_max * 1.01  # Allow slight increase
        
        a_qp = solve_qp_refine(b, a, M_bound, solver)
        if a_qp is not None:
            score = 2.0 * n * fast_convolve(a_qp).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_qp.copy()
                a = a_qp
            else:
                # Try direct convex combination
                for alpha in [0.3, 0.5, 0.7]:
                    a_try = (1 - alpha) * a + alpha * b
                    a_try = np.maximum(a_try, 0)
                    a_try = a_try / a_try.sum()
                    score_try = 2.0 * n * fast_convolve(a_try).max()
                    if score_try < best_score - 1e-14:
                        best_score = score_try
                        best_a = a_try.copy()
                        a = a_try
                        break
        
        if (rnd + 1) % 30 == 0:
            print(f"    QP loop round={rnd+1} ({monotonic()-t0:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Initialize from parent or spectral init
    2. FP with exponential averaging
    3. Damped QP loop
    4. Multi-temperature L-BFGS cooling
    5. Pairwise local search
    6. Coordinate descent polish
    7. Final L-BFGS + local search
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    n_use = 768
    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
        n_use = len(parent)
    else:
        best_seq = spectral_init(n_use)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial spectral: n={n_use}, score={best_score:.6f}")

    # === Phase 1: Fictitious Play ===
    rem = deadline - monotonic() - 600
    if rem > 120:
        t_fp = min(rem, 500.0)
        print(f"\n=== Phase 1: FP (budget={t_fp:.0f}s) ===")
        a_fp, score_fp, iters = fictitious_play_v2(n_use, t_fp, initial=best_seq, solver=solver)
        best_seq, best_score = check_and_update(a_fp, f"FP_it{iters}", best_seq, best_score)

    # === Phase 2: Damped QP Loop ===
    rem = deadline - monotonic() - 500
    if rem > 60:
        t_qp = min(rem * 0.6, 300.0)
        print(f"\n=== Phase 2: QP Loop (budget={t_qp:.0f}s, best={best_score:.6f}) ===")
        a_qp, _ = damped_qp_loop(best_seq, t_qp, solver=solver)
        best_seq, best_score = check_and_update(a_qp, "qp_loop", best_seq, best_score)

    # === Phase 3: L-BFGS Cooling ===
    rem = deadline - monotonic() - 350
    if rem > 60:
        t_lb = min(rem * 0.7, 300.0)
        print(f"\n=== Phase 3: L-BFGS Cooling (budget={t_lb:.0f}s, best={best_score:.6f}) ===")
        a_lb, score_lb = lbgfs_cooling(best_seq, t_lb, T_start=300.0, T_end=3.0, n_rounds=6)
        best_seq, best_score = check_and_update(a_lb, "lbgfs_cool", best_seq, best_score)

    # === Phase 4: Pairwise Local Search ===
    rem = deadline - monotonic() - 200
    if rem > 30:
        t_pw = min(rem * 0.6, 120.0)
        print(f"\n=== Phase 4: Pairwise Search (budget={t_pw:.0f}s, best={best_score:.6f}) ===")
        a_pw, score_pw = pairwise_local_search(best_seq, t_pw, max_pairs=30)
        best_seq, best_score = check_and_update(a_pw, "pairwise", best_seq, best_score)

    # === Phase 5: Second L-BFGS with lower temperature ===
    rem = deadline - monotonic() - 150
    if rem > 30:
        t_lb2 = min(rem * 0.6, 100.0)
        print(f"\n=== Phase 5: L-BFGS Fine (budget={t_lb2:.0f}s, best={best_score:.6f}) ===")
        a_lb2, score_lb2 = lbgfs_cooling(best_seq, t_lb2, T_start=50.0, T_end=2.0, n_rounds=4)
        best_seq, best_score = check_and_update(a_lb2, "lbgfs_fine", best_seq, best_score)

    # === Phase 6: More pairwise search ===
    rem = deadline - monotonic() - 80
    if rem > 20:
        t_pw2 = min(rem * 0.7, 80.0)
        print(f"\n=== Phase 6: Pairwise 2 (budget={t_pw2:.0f}s, best={best_score:.6f}) ===")
        a_pw2, _ = pairwise_local_search(best_seq, t_pw2, max_pairs=20)
        best_seq, best_score = check_and_update(a_pw2, "pairwise2", best_seq, best_score)

    # === Phase 7: Another FP round ===
    rem = deadline - monotonic() - 60
    if rem > 40:
        t_fp2 = min(rem * 0.6, 100.0)
        print(f"\n=== Phase 7: FP Round 2 (budget={t_fp2:.0f}s) ===")
        a_fp2, _, _ = fictitious_play_v2(n_use, t_fp2, initial=best_seq, solver=solver, verbose=False)
        best_seq, best_score = check_and_update(a_fp2, "FP2", best_seq, best_score)

    # === Phase 8: Final L-BFGS polish ===
    rem = deadline - monotonic() - 20
    if rem > 15:
        t_final = min(rem * 0.7, 40.0)
        print(f"\n=== Phase 8: Final Polish (budget={t_final:.0f}s) ===")
        a_fin, _ = lbgfs_cooling(best_seq, t_final, T_start=20.0, T_end=1.0, n_rounds=3)
        best_seq, best_score = check_and_update(a_fin, "final", best_seq, best_score)

    # === Phase 9: Very short pairwise ===
    rem = deadline - monotonic() - 5
    if rem > 5:
        a_fin2, _ = pairwise_local_search(best_seq, rem * 0.8, max_pairs=15)
        best_seq, best_score = check_and_update(a_fin2, "final_pw", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
