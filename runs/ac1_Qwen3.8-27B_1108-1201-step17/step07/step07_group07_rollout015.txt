1. Smooth gradient descent with log-sum-exp approximation (adaptive beta, Nesterov momentum)
2. LP-based refinement for principled peak reduction

4. Multi-length exploration
5. Coordinate descent polish

Key innovation over previous approach: replaces non-smooth subgradient method with

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def smooth_gradient(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*conv(a,a)[k])).
    Returns gradient of length n.
    """
    n = len(a)
    b = fast_convolve(a)
    # Stable softmax
    b_max = b.max()
    exp_b = np.exp(beta * (b - b_max))
    p = exp_b / exp_b.sum()
    # Compute grad[i] = 2 * sum_j a[j] * p[i+j] via FFT
    n_fft = 1 << (3 * n - 1).bit_length()
    a_rev = a[::-1]
    A = rfft(a_rev, n_fft)
    P = rfft(p, n_fft)
    conv_full = irfft(A * P, n_fft)
    grad = 2.0 * conv_full[n - 1:2 * n - 1]
    return grad


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_max_b(a):
    """Return max of auto-convolution for normalized sequence (sum=1)."""
    b = fast_convolve(a)
    return float(b.max())


def smooth_gradient_descent(a0, time_limit, beta_init=30.0, beta_final=500.0,
                            lr_init=0.01, momentum=0.9, seed=42):
    """
    Smooth gradient descent on the simplex using log-sum-exp surrogate.
    Adaptive beta increases over time for sharper peak targeting.
    Nesterov momentum for acceleration.
    Returns (best_a, best_max_b).
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = eval_max_b(a)
    cur_lr = lr_init
    no_improve = 0
    it_count = 0
    max_no_improve = 2000

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        # Adaptive beta: exponential ramp
        beta = beta_init * (beta_final / beta_init) ** progress

        grad = smooth_gradient(a, beta)

        # Nesterov momentum
        v = momentum * v - cur_lr * grad
        a_new = project_simplex(a + v)

        obj_new = eval_max_b(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it_count += 1

        # Adaptive strategy
        if no_improve >= max_no_improve:
            # Restart with perturbation
            kick = 0.002 * (1.0 - 0.5 * progress)
            a = project_simplex(best_a + rng.standard_normal(n) * kick)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.05)
            no_improve = 0
        elif no_improve >= max_no_improve * 0.6:
            cur_lr *= 0.95
        elif no_improve >= max_no_improve * 0.3:
            cur_lr *= 0.98

    return best_a, best_obj


def lp_refine(f, max_iters=50, time_limit=60.0, verbose=False):
    """
    LP-based refinement: solve max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Blend with current f to reduce max convolution.
    Returns (best_a, best_max_b) where a is normalized to sum=1.
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = eval_max_b(a)

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = fast_convolve(a).max()
        M *= 1.0001  # tiny tolerance to avoid degeneracy

        # Build constraint matrix (banded structure)
        num_constraints = 2 * n - 1
        c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
        b_lp = np.full(num_constraints, M)

        # Efficient construction using the banded structure
        A_ub = np.zeros((num_constraints, n))
        for k in range(num_constraints):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A_ub[k, k - j] = a[j]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_lp,
                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                break

            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break

            g_norm = g / g_sum

            # Line search for blending
            best_t = 0.0
            best_blend_obj = best_obj
            t_candidates = [0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
            for t in t_candidates:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                ss = blend.sum()
                if ss < 1e-10:
                    continue
                blend = blend / ss
                obj_blend = eval_max_b(blend)
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t

            if best_t <= 0.0:
                for t in [0.005, 0.01, 0.015, 0.025]:
                    blend = (1.0 - t) * best_a + t * g_norm
                    blend = np.maximum(blend, 0.0)
                    blend = blend / blend.sum()
                    obj_blend = eval_max_b(blend)
                    if obj_blend < best_obj - 1e-14:
                        best_t = t
                        best_blend_obj = obj_blend
                        break

            if best_t > 0.0:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                obj = eval_max_b(a)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                    if verbose:
                        print(f"    LP iter {iteration}: t={best_t:.3f}, max_b={best_obj:.12f}")
                else:
                    a = best_a.copy()
            else:
                if verbose:
                    print(f"    LP iter {iteration}: no improvement, stopping")
                break

        except Exception as e:
            if verbose:
                print(f"    LP error: {e}")
            break

    return best_a, best_obj


def perturbation_search(a0, time_limit, num_perturb=5, seed=42):
    """
    Random perturbation-based local search.
    Randomly scales subsets of coordinates and accepts if objective improves.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    while time.monotonic() - t0 < time_limit:
        # Random subset perturbation
        subset_size = max(2, min(n // 4, rng.integers(3, n // 3)))
        indices = rng.choice(n, size=subset_size, replace=False)
        scales = rng.uniform(0.5, 2.0, size=subset_size)

        a_pert = a.copy()
        a_pert[indices] *= scales
        a_pert = np.maximum(a_pert, 0.0)
        ss = a_pert.sum()
        if ss < 1e-10:
            continue
        a_pert = a_pert / ss

        obj_pert = eval_max_b(a_pert)
        if obj_pert < best_obj - 1e-14:
            best_obj = obj_pert
            best_a = a_pert.copy()
            a = a_pert.copy()
        else:
            # Occasional random restart from best with small perturbation
            if rng.random() < 0.1:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)

    return best_a, best_obj


def coordinate_descent(a0, time_limit, multipliers=None, verbose=False):
    """
    Coordinate descent on the simplex: scale one element, rescale others.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0])

    improved_global = True
    pass_count = 0
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        pass_count += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 20.0:
                    continue

                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = eval_max_b(a_tr)
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True

    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1152) / 1152
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"No parent, using uniform n=1152, score={best_score:.6f}")

    n_parent = len(parent)
    print(f"Starting optimization. n={n_parent}, initial score={best_score:.8f}")

    # === Phase 1: Smooth gradient descent on parent (main workhorse) ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Smooth Gradient Descent (n={n_parent})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 500
    if rem > 60:
        t_sg = min(rem, 500.0)
        print(f"  Time: {t_sg:.0f}s")
        a1, obj1 = smooth_gradient_descent(
            parent, time_limit=t_sg,
            beta_init=20.0, beta_final=400.0,
            lr_init=0.008, momentum=0.91, seed=seed
        )
        update_best(a1, "sgd_phase1")

    # === Phase 2: LP refinement ===
    print(f"\n{'='*60}")
    print(f"Phase 2: LP Refinement (best={best_score:.8f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_lp = min(rem, 200.0)
            print(f"  Time: {t_lp:.0f}s")
            a2, obj2 = lp_refine(best_seq, max_iters=50, time_limit=t_lp, verbose=True)
            update_best(a2, "lp_phase2")

    # === Phase 3: More smooth gradient from LP result ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Second Smooth Gradient Pass (best={best_score:.8f})")
    print(f"{'='*60}")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 300
        if rem > 40:
            t_sg2 = min(rem, 250.0)
            print(f"  Time: {t_sg2:.0f}s, n={n_best}")
            a3, obj3 = smooth_gradient_descent(
                best_seq, time_limit=t_sg2,
                beta_init=100.0, beta_final=1000.0,
                lr_init=0.003, momentum=0.93, seed=seed + 1000
            )
            update_best(a3, "sgd_phase3")

    # === Phase 4: Perturbation search ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Perturbation Search (best={best_score:.8f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_pert = min(rem, 100.0)
            print(f"  Time: {t_pert:.0f}s")
            a4, obj4 = perturbation_search(best_seq, time_limit=t_pert, seed=seed + 2000)
            update_best(a4, "perturb")

    # === Phase 5: LP again ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Second LP Pass (best={best_score:.8f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_lp2 = min(rem, 80.0)
            print(f"  Time: {t_lp2:.0f}s")
            a5, obj5 = lp_refine(best_seq, max_iters=30, time_limit=t_lp2, verbose=True)
            update_best(a5, "lp_phase5")

    # === Phase 6: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Multi-length Exploration (best={best_score:.8f})")
    print(f"{'='*60}")
    if best_seq is not None:
        n_best = len(best_seq)
        target_lengths = [int(n_best * f) for f in [0.85, 1.1, 1.25, 0.75, 1.5]]
        target_lengths = sorted(set(max(256, min(4096, tl)) for tl in target_lengths))

        for target_n in target_lengths[:4]:
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.3, 40.0)
            if t_alloc < 10:
                continue
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_len, obj_len = smooth_gradient_descent(
                f_interp, time_limit=t_alloc * 0.7,
                beta_init=50.0, beta_final=300.0,
                lr_init=0.005, momentum=0.92, seed=seed + target_n
            )
            # Quick LP polish
            rem_sub = deadline - time.monotonic() - 80
            if rem_sub > 10:
                a_len, obj_len = lp_refine(a_len, max_iters=10, time_limit=min(rem_sub, 15.0))
            update_best(a_len, f"len_{target_n}")

    # === Phase 7: Diverse initializations (short runs) ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Diverse Inits (best={best_score:.8f})")
    print(f"{'='*60}")
    n_div = len(best_seq) if best_seq is not None else n_parent

    dinit_list = [
        ("ramp", make_ramp_init(n_div)),
        ("bump_0.3", make_bump_init(n_div, 0.3, 0.25)),
        ("bump_0.7", make_bump_init(n_div, 0.7, 0.25)),
        ("gauss_0.2", make_gaussian_init(n_div, 0.2)),
        ("rect_pert", make_rect_perturb_init(n_div, seed=7)),
    ]

    for name, init_a in dinit_list:
        rem = deadline - time.monotonic() - 80
        if rem < 15:
            break
        t_alloc = min(rem * 0.2, 30.0)
        if t_alloc < 8:
            continue
        print(f"  {name} (t={t_alloc:.0f}s)")
        a_div, obj_div = smooth_gradient_descent(
            init_a, time_limit=t_alloc,
            beta_init=20.0, beta_final=200.0,
            lr_init=0.006, momentum=0.9, seed=seed + hash(name) % 100000
        )
        update_best(a_div, f"div_{name}")

    # === Phase 8: Final deep optimization ===
    print(f"\n{'='*60}")
    print(f"Phase 8: Final Deep Optimization (best={best_score:.8f})")
    print(f"{'='*60}")
    if best_seq is not None:
        n_final = len(best_seq)
        rem = deadline - time.monotonic() - 50
        if rem > 30:
            t_deep = min(rem * 0.5, 100.0)
            print(f"  Deep SGD: t={t_deep:.0f}s")
            a_deep, obj_deep = smooth_gradient_descent(
                best_seq, time_limit=t_deep,
                beta_init=200.0, beta_final=2000.0,
                lr_init=0.001, momentum=0.95, seed=seed + 99999
            )
            update_best(a_deep, "deep_sgd")

            rem = deadline - time.monotonic() - 30
            if rem > 15:
                t_lp_final = min(rem, 40.0)
                a_lp_f, obj_lp_f = lp_refine(best_seq, max_iters=20, time_limit=t_lp_final, verbose=True)
                update_best(a_lp_f, "deep_lp")

    # === Phase 9: Coordinate descent polish ===
    print(f"\n{'='*60}")
    print(f"Phase 9: Coordinate Descent Polish (best={best_score:.8f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_cd = min(rem, 40.0)
            n_cd = len(best_seq)
            print(f"  CD: n={n_cd}, t={t_cd:.0f}s")
            a_cd, obj_cd = coordinate_descent(
                best_seq, time_limit=t_cd,
                multipliers=np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0])
            )
            update_best(a_cd, "cd")

    # === Phase 10: Tiny final SGD ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            t_micro = min(rem, 20.0)
            a_micro, obj_micro = smooth_gradient_descent(
                best_seq, time_limit=t_micro,
                beta_init=500.0, beta_final=5000.0,
                lr_init=0.0005, momentum=0.97, seed=seed + 777777
            )
            update_best(a_micro, "micro_final")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
