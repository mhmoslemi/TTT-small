import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Score for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_at(a, conv=None):
    """Compute subgradient of 2n*max(conv(a,a)) at a.
    Returns gradient vector and the peak positions."""
    n = len(a)
    if conv is None:
        conv = fast_convolve(a)
    max_val = conv.max()
    # Find all peaks that are "active" (within epsilon of max)
    threshold = max_val - 1e-10
    active = np.where(conv >= threshold)[0]
    
    # Subgradient: weighted average of gradients at active peaks
    # Gradient at peak k: 2*a_{k-j} for j where k-j is valid
    g = np.zeros(n)
    for k in active:
        k = int(k)
        # For position j, contribution is 2*a_{k-j} if 0 <= k-j < n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            g[js:je+1] += 2.0 * a[k - np.arange(js, je+1)]
    
    # Normalize by number of active peaks (average subgradient)
    g /= len(active)
    g *= 2.0 * n  # Scale by 2n for the actual objective
    
    return g, active, max_val

def nesterov_subgradient(a0, time_limit, eta0=0.01, decay=0.9999, 
                          momentum=0.9, avg_start=100):
    """Nesterov accelerated projected subgradient with running average."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    
    v = np.zeros(n)
    best_a = a.copy()
    best_score = compute_score(a)
    avg_a = np.zeros(n)
    avg_count = 0
    eta = eta0
    t0 = time.monotonic()
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Nesterov lookahead
        a_la = a - momentum * v
        a_la = project_simplex(a_la)
        
        # Compute subgradient at lookahead point
        conv_la = fast_convolve(a_la)
        max_la = conv_la.max()
        threshold = max_la * 0.99  # Use 99% of max as threshold for "active" peaks
        active = np.where(conv_la >= threshold)[0]
        
        g = np.zeros(n)
        for k in active:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                g[js:je+1] += 2.0 * a_la[k - np.arange(js, je+1)]
        g /= max(1, len(active))
        g *= 2.0 * n
        
        # Check score at current point
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Update
        v = momentum * v + g
        a_new = a - eta * v
        a_new = project_simplex(a_new)
        
        # Running average
        if iters >= avg_start:
            avg_a += a_new
            avg_count += 1
            avg_score = compute_score(avg_a / avg_count)
            if avg_score < best_score:
                best_score = avg_score
                best_a = (avg_a / avg_count).copy()
        
        a = a_new
        eta *= decay
    
    return best_a, best_score

def active_set_equalize(a0, time_limit, K=10, step_size=0.01):

        top_k = np.argpartition(conv, -K)[-K:]

        grads = []
        for k in top_k:
            k = int(k)
            g = np.zeros(n)

            je = min(n - 1, k)
            if js <= je:
                g[js:je+1] = 2.0 * a[k - np.arange(js, je+1)]
            grads.append(g)
        
        grads = np.array(grads)  # shape (K, n)
        
        # Find direction d that minimizes max_i grad_i^T d
        # subject to sum(d) = 0, ||d|| <= 1
        # Approximate: d = -normalize(mean(grads) - mean(grads)*ones/n)
        mean_grad = grads.mean(axis=0)
        # Project to sum=0
        d = mean_grad - np.mean(mean_grad)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            # Try random perturbation
            d = np.random.randn(n)
            d -= d.mean()
            d /= np.linalg.norm(d)
        else:
            d /= d_norm
        
        # Take step
        step = step_size
        a_new = a + step * (-d)  # Move opposite to gradient
        a_new = project_simplex(a_new)
        
        score_new = compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            # Increase K if improving
        else:
            # Try smaller step
            for shrink in [0.5, 0.25, 0.1]:
                a_try = a - step * shrink * d
                a_try = project_simplex(a_try)
                sc_try = compute_score(a_try)
                if sc_try < best_score:
                    best_score = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                # No improvement, perturb
                a = best_a + np.random.randn(n) * 0.001
                a = project_simplex(a)
        
        step *= 0.999
    
    return best_a, best_score

def coordinate_descent_fast(a0, time_limit, n_pairs=500):
    """Fast coordinate descent with pairwise transfers."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(a)
        peak = conv.max()
        peak_idx = int(np.argmax(conv))
        
        # Focus on coordinates that contribute to the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        contrib = a[js:je+1] * a[peak_idx - np.arange(js, je+1)]
        
        # Try to reduce top contributors
        top_contrib = np.argsort(contrib)[::-1][:20]
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Choose source from top contributors, target randomly
            if len(top_contrib) > 0 and rng.random() < 0.6:
                i = js + top_contrib[rng.integers(0, min(len(top_contrib), 10))]
            else:
                i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue
            
            # Try transferring
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                t = frac * a[i]
                if t >= a[i]:
                    t = a[i] * 0.5
                a_try = a.copy()
                a_try[i] -= t
                a_try[j] += t
                a_try = np.maximum(a_try, 0)
                ss = a_try.sum()
                if ss < 1e-15:
                    continue
                a_try /= ss
                sc = compute_score(a_try)
                if sc < best_score - 1e-12:
                    a = a_try
                    best_a = a.copy()
                    best_score = sc
                    improved = True
                    break
        
        if not improved:
            # Perturb
            a = best_a + rng.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a, best_score

def lbgfs_smoothed(a0, time_limit, T=500, maxiter=3000):
    """L-BFGS on smoothed objective in log-space."""
    from scipy.optimize import minimize
    
    n = len(a0)
    a0 = np.maximum(a0, 1e-30)
    a0 = a0 / a0.sum()
    z0 = np.log(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        zm = z - z.max()
        ez = np.exp(zm)
        a = ez / ez.sum()
        conv = fast_convolve(a)
        shifted = conv / T
        mx = shifted.max()
        lse = mx + np.log(np.exp(shifted - mx).sum())
        return (2.0 * n / T) * np.exp(T * lse)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        zm = z - z.max()
        ez = np.exp(zm)
        a = ez / ez.sum()
        conv = fast_convolve(a)
        shifted = conv / T
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w /= w.sum()
        
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return (2.0 * n / T) * grad_z
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        zm = res.x - res.x.max()
        a_out = np.exp(zm)
        a_out /= a_out.sum()
        sc = compute_score(a_out)
        if sc < best_score:
            best_a = a_out
            best_score = sc
    except:
        pass
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. Nesterov accelerated subgradient with running averages (fast global convergence)
2. Active-set peak equalization with proper subgradient directions

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score for normalized sequence (sum=1): 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_computation(a, conv, threshold_frac=0.98):
    """Compute averaged subgradient of 2n*max(conv) at a, using active peaks."""
    n = len(a)
    max_val = conv.max()
    threshold = max_val * threshold_frac
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])
    
    g = np.zeros(n)
    for k in active:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
    g /= len(active)
    g *= 2.0 * n
    return g, active, max_val


def nesterov_subgradient(a0, time_limit, eta0=0.003, decay=0.99997,
                         momentum=0.85, avg_start=50, threshold_frac=0.97):
    """Nesterov accelerated projected subgradient with running average."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    
    v = np.zeros(n)
    best_a = a.copy()
    best_score = compute_score(a)
    avg_a = np.zeros(n)
    avg_count = 0
    eta = eta0
    t0 = time.monotonic()
    iters = 0
    no_improve_count = 0
    improvement_threshold = 1e-10
    
    while time.monotonic() - t0 < time_limit:
        batch = 0
        while time.monotonic() - t0 < time_limit:
            batch += 1
            iters += 1
            
            # Nesterov lookahead
            a_la = a - momentum * v
            a_la = project_simplex(a_la)
            
            # Compute subgradient at lookahead
            conv_la = fast_convolve(a_la)
            g, active, max_val = subgradient_computation(a_la, conv_la, threshold_frac)
            
            # Check score periodically
            if iters % 10 == 0:
                score = compute_score(a)
                if score < best_score - improvement_threshold:
                    best_score = score
                    best_a = a.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1
                
                # Running average check
                if avg_count > 0 and iters % 50 == 0:
                    avg_score = compute_score(avg_a / avg_count)
                    if avg_score < best_score - improvement_threshold:
                        best_score = avg_score
                        best_a = (avg_a / avg_count).copy()
                        no_improve_count = 0
            
            # Update
            v = momentum * v + g
            a_new = a - eta * v
            a_new = project_simplex(a_new)
            a = a_new
            
            # Running average (start after warmup)
            if iters >= avg_start:
                avg_a += a_new
                avg_count += 1
            
            # Adaptive step size
            eta *= decay
            
            # If stuck, reduce momentum and restart from best
            if no_improve_count > 100:
                a = best_a + np.random.default_rng(iters).standard_normal(n) * 0.002
                a = project_simplex(a)
                v = np.zeros(n)
                eta = eta0 * 0.5
                no_improve_count = 0
    
    # Final check
    score = compute_score(a)
    if score < best_score:
        best_a = a.copy()
        best_score = score
    if avg_count > 0:
        avg_score = compute_score(avg_a / avg_count)
        if avg_score < best_score:
            best_a = (avg_a / avg_count).copy()
            best_score = avg_score
    
    return best_a, best_score


def active_set_peak_equalize(a0, time_limit, K=12, step_init=0.005, decay=0.9995):
    """Active-set peak equalization using gradient direction."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    step = step_init
    iters = 0
    rng = np.random.default_rng(12345)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Find top K peaks
        K_eff = min(K, len(conv))
        top_k = np.argpartition(conv, -K_eff)[-K_eff:]
        top_k = top_k[np.argsort(conv[top_k])[::-1]]
        
        # Compute gradient directions for each peak
        grads = np.zeros((len(top_k), n))
        for ii, k in enumerate(top_k):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx = np.arange(js, je + 1)
                grads[ii, idx] = 2.0 * a[k - idx]
        
        # Direction: negative mean gradient, projected to sum=0
        mean_grad = grads.mean(axis=0)
        d = -(mean_grad - np.mean(mean_grad))
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            d = rng.standard_normal(n)
            d -= d.mean()
            d /= np.linalg.norm(d)
        else:
            d /= d_norm
        
        # Try step
        a_new = a + step * d
        a_new = project_simplex(a_new)
        sc_new = compute_score(a_new)
        
        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            # Backtracking
            found = False
            for shrink in [0.5, 0.25, 0.1, 0.05]:
                a_try = a + step * shrink * d
                a_try = project_simplex(a_try)
                sc_try = compute_score(a_try)
                if sc_try < best_score - 1e-12:
                    best_score = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    no_improve = 0
                    found = True
                    break
            if not found:
                no_improve += 1
                if no_improve > 50:
                    # Perturb and reset
                    a = best_a + rng.standard_normal(n) * 0.003
                    a = project_simplex(a)
                    step = step_init * 0.3
                    no_improve = 0
        
        step *= decay
    
    return best_a, best_score

    no_improve = 0

        conv = fast_convolve(a)
        max_val = conv.max()
        threshold = max_val * threshold_frac
        active_peaks = np.where(conv >= threshold)[0]
        
        improved = False
        
        # For each active peak, try to reduce its top contributors
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick a random active peak
            if len(active_peaks) == 0:
                active_peaks = np.array([int(np.argmax(conv))])
            pk = int(active_peaks[rng.integers(0, len(active_peaks))])
            
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            
            # Pick a source from top contributors
            top_c = np.argsort(contribs)[::-1][:min(10, len(idx))]
            if len(top_c) == 0:
                continue
            ci = top_c[rng.integers(0, len(top_c))]
            src = idx[ci]
            
            # Pick target (random, or from a low-contribution area)
            tgt = rng.integers(0, n)
            if src == tgt or a[src] < 1e-14:
                continue
            
            # Try several transfer fractions
            best_frac = 0
            best_sc = best_score
            for frac in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                t = frac * a[src]
                if t >= a[src] - 1e-15:
                    t = a[src] * 0.4
                a_try = a.copy()
                a_try[src] -= t
                a_try[tgt] += t
                a_try = np.maximum(a_try, 0)
                ss = a_try.sum()
                if ss < 1e-15:
                    continue
                a_try /= ss
                sc = compute_score(a_try)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_frac = frac
                    best_a_try = a_try.copy()
            
            if best_frac > 0:
                a = best_a_try
                best_score = best_sc
                best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 30:
                a = best_a + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
    
    return best_a, best_score


def lbgfs_smoothed(a0, time_limit, T=500, maxiter=5000):
    """L-BFGS on soft-max smoothed objective in log-space."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-30)
    a0 = a0 / a0.sum()
    z0 = np.log(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        zm = z - z.max()
        ez = np.exp(zm)
        a = ez / ez.sum()
        conv = fast_convolve(a)
        shifted = conv / T
        mx = shifted.max()
        lse = mx + np.log(np.exp(shifted - mx).sum())
        return (2.0 * n / T) * np.exp(T * lse)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        zm = z - z.max()
        ez = np.exp(zm)
        a = ez / ez.sum()
        conv = fast_convolve(a)
        shifted = conv / T
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w /= w.sum()
        
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return (2.0 * n / T) * grad_z
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 60, 'maxls': 50, 'disp': False}
        )
        zm = res.x - res.x.max()
        a_out = np.exp(zm)
        a_out /= a_out.sum()
        sc = compute_score(a_out)
        if sc < best_score:
            best_a = a_out
            best_score = sc
    except Exception:
        pass
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def symmetrize(a):
    """Symmetrize: (a + reverse(a)) / 2, then renormalize."""
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    a_norm = a / s
    return (a_norm + a_norm[::-1]) / 2


def try_update(a, label, best_seq, best_score):
    """Try to update best with candidate a."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score
    a = a / s
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Uses Nesterov subgradient, active-set equalization, L-BFGS, coordinate descent,
    symmetrization, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 768
        t = np.linspace(0, np.pi, n_init)
        parent = 0.5 + 0.5 * np.cos(t)
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        best_score = compute_score(parent)
        best_seq = parent.copy()
        print(f"Init: n={n_init}, score={best_score:.8f}")
    
    n_parent = len(best_seq)
    
    # === Phase 1: Nesterov subgradient (global phase) ===
    print(f"\n=== Phase 1: Nesterov Subgradient (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(250.0, rem * 0.18)
    print(f"  Budget: {t1:.0f}s, n={n_parent}")
    a_ns, sc_ns = nesterov_subgradient(best_seq, t1, eta0=0.003, decay=0.99997,
                                        momentum=0.85, avg_start=50)
    best_seq, best_score = try_update(a_ns, "nesterov", best_seq, best_score)
    
    # Symmetrize
    a_sym = symmetrize(best_seq)
    best_seq, best_score = try_update(a_sym, "sym1", best_seq, best_score)
    
    # === Phase 2: Active-set equalization ===
    print(f"\n=== Phase 2: Active-Set Equalization (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(250.0, rem * 0.15)
    print(f"  Budget: {t2:.0f}s, K=15")
    a_as, sc_as = active_set_peak_equalize(best_seq, t2, K=15, step_init=0.004)
    best_seq, best_score = try_update(a_as, "active_set", best_seq, best_score)
    
    # === Phase 3: L-BFGS with multiple temperatures ===
    print(f"\n=== Phase 3: L-BFGS Multi-T (best={best_score:.8f}) ===")
    for T_val in [100, 300, 1000, 5000]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t3 = min(150.0, rem * 0.12)
        if t3 < 20:
            break
        print(f"  L-BFGS T={T_val}, budget={t3:.0f}s")
        a_lb, sc_lb = lbgfs_smoothed(best_seq, t3, T=T_val, maxiter=4000)
        best_seq, best_score = try_update(a_lb, f"lbfgs_T{T_val}", best_seq, best_score)
    
    # === Phase 4: Coordinate descent (peak-focused) ===
    print(f"\n=== Phase 4: Peak Coord Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t4 = min(180.0, rem * 0.12)
        print(f"  Budget: {t4:.0f}s")
        a_cd, sc_cd = coordinate_descent_peak(best_seq, t4, n_pairs=400)
        best_seq, best_score = try_update(a_cd, "cd_peak", best_seq, best_score)
    
    # === Phase 5: Second Nesterov round (finer) ===
    print(f"\n=== Phase 5: Nesterov v2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t5 = min(150.0, rem * 0.12)
        print(f"  Budget: {t5:.0f}s")
        a_ns2, _ = nesterov_subgradient(best_seq, t5, eta0=0.001, decay=0.99999,
                                         momentum=0.9, avg_start=100)
        best_seq, best_score = try_update(a_ns2, "nesterov2", best_seq, best_score)
    
    # === Phase 6: Active-set with more peaks ===
    print(f"\n=== Phase 6: Active-Set K=25 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t6 = min(120.0, rem * 0.1)
        print(f"  Budget: {t6:.0f}s")
        a_as2, _ = active_set_peak_equalize(best_seq, t6, K=25, step_init=0.003)
        best_seq, best_score = try_update(a_as2, "aset_K25", best_seq, best_score)
    
    # === Phase 7: Try different lengths ===
    print(f"\n=== Phase 7: Alternative Lengths (best={best_score:.8f}) ===")
    alt_lengths = sorted(set([
        n_parent - 2, n_parent - 1, n_parent + 1, n_parent + 2,
        n_parent - 4, n_parent + 4, n_parent + 8,
        n_parent // 2 if n_parent >= 128 else n_parent,
        n_parent * 2 if n_parent * 2 <= 3000 else n_parent,
    ]))
    for alt_n in alt_lengths[:6]:
        if alt_n < 64 or alt_n > 3000 or alt_n == n_parent:
            continue
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t7 = min(80.0, rem * 0.1)
        if t7 < 15:
            break
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  L-BFGS n={alt_n}, budget={t7:.0f}s")
        a_alt, _ = lbgfs_smoothed(init_alt, t7, T=500, maxiter=3000)
        best_seq, best_score = try_update(a_alt, f"n{alt_n}", best_seq, best_score)
    
    # === Phase 8: Final L-BFGS polish ===
    print(f"\n=== Phase 8: Final L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(80.0, rem * 0.25)
        print(f"  Final LBFGS budget={t8:.0f}s")
        a_f, _ = lbgfs_smoothed(best_seq, t8, T=10000, maxiter=8000)
        best_seq, best_score = try_update(a_f, "final_lbfgs", best_seq, best_score)
    
    # === Phase 9: Final CD ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t9 = min(40.0, rem * 0.3)
        a_cdf, _ = coordinate_descent_peak(best_seq, t9, n_pairs=600)
        best_seq, best_score = try_update(a_cdf, "final_cd", best_seq, best_score)
    
    # === Phase 10: Symmetrize final ===
    rem = deadline - time.monotonic()
    if rem > 10:
        a_sf = symmetrize(best_seq)
        best_seq, best_score = try_update(a_sf, "final_sym", best_seq, best_score)
        # Quick L-BFGS on symmetrized
        if rem > 30:
            a_fs, _ = lbgfs_smoothed(a_sf, min(30.0, rem * 0.4), T=5000, maxiter=3000)
            best_seq, best_score = try_update(a_fs, "final_sym_lbfgs", best_seq, best_score)
    
    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
    
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
