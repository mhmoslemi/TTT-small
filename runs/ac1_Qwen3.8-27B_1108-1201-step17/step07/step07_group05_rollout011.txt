"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative weight (mirror descent) updates with peak subgradients
2. L-BFGS polishing in log-space
3. Length exploration via interpolation
4. Peak equalization heuristics
Key innovation: direct multiplicative updates on the simplex (no softmax parameterization)
with carefully computed subgradients from the top convolution peaks.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score_normalized(a):
    """Score for a sequence normalized to sum=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def peak_subgradient(a, top_k=30, T_frac=0.05):
    """
    Compute a subgradient of max_k conv[k] w.r.t. a.
    Returns (gradient, convolution, peak_indices).
    Uses formula: d(conv[k])/da[m] = 2*a[k-m] if max(0,k-n+1) <= m <= min(n-1,k).
    """
    n = len(a)
    conv = fast_convolve(a)
    total_size = 2 * n - 1
    
    if total_size <= top_k:
        peak_indices = np.arange(total_size)
        weights = np.ones(total_size) / total_size
    else:
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_values = conv[peak_indices]
        max_val = peak_values.max()
        T = T_frac * max_val
        weights = np.exp((peak_values - max_val) / T)
        weights = weights / weights.sum()
    
    grad = np.zeros(n)
    
    for k_idx in range(len(peak_indices)):
        k = int(peak_indices[k_idx])
        w = weights[k_idx]
        
        # Valid m range: [max(0, k-n+1), min(n-1, k)]
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        
        if m_start > m_end:
            continue
        
        # d(conv[k])/da[m] = 2 * a[k-m] for m in [m_start, m_end]
        # a[k-m] for m in [m_start, m_end] is a[k-m_start] down to a[k-m_end]
        # = a[k-m_start : k-m_end+1] reversed
        # k - m_start to k - m_end
        vals = a[k - m_end : k - m_start + 1][::-1] * (2 * w)
        grad[m_start : m_end + 1] += vals
    
    return grad, conv, peak_indices

def mirror_descent(a0, time_limit, eta0=0.1, decay=0.9999, top_k=30, T_frac=0.05, verbose=False):
    """
    Mirror descent (multiplicative weights) on the simplex.
    a is normalized to sum=1.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    eta = eta0
    iters = 0
    patience = 0
    patience_limit = 500
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        grad, conv, peaks = peak_subgradient(a, top_k=top_k, T_frac=T_frac)
        
        # Multiplicative update: a <- a * exp(-eta * grad)
        # Clip grad to prevent numerical issues
        grad_clipped = np.clip(grad, -10, 10)
        a_new = a * np.exp(-eta * grad_clipped)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        score = compute_score_normalized(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        eta *= decay
        
        if patience > patience_limit:
            # Reset with perturbation
            a = best_a + np.random.default_rng(iters).standard_normal(n) * 0.01
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            eta = eta0
            patience = 0
        
        if verbose and iters % 100 == 0:
            el = time.monotonic() - t0
            print(f"    MD iter={iters}, t={el:.1f}s, eta={eta:.6f}, best={best_score:.8f}")
    
    return best_a, best_score

def lbgfs_logspace(a0, time_limit, T=500, maxiter=3000):
    """L-BFGS in log-space with softmax smoothing."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-15))
    t0 = time.monotonic()
    best = {'a': a0.copy(), 'score': compute_score_normalized(a0)}
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()
        conv = fast_convolve(a)
        shifted = conv / T
        max_s = shifted.max()
        lse = max_s + np.log(np.exp(shifted - max_s).sum())
        return (2.0 * n / T) * np.exp(T * lse)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()
        conv = fast_convolve(a)
        shifted = conv / T
        max_s = shifted.max()
        w = np.exp(shifted - max_s)
        w = w / w.sum()
        
        # Gradient of (2n/T)*exp(T*max(conv)) w.r.t. a
        # = 2n * (sum_k w_k * grad conv[k])
        # Use the same subgradient formula
        grad_a = np.zeros(n)
        for k in range(len(conv)):
            if w[k] < 1e-10:
                continue
            m_start = max(0, k - n + 1)
            m_end = min(n - 1, k)
            if m_start > m_end:
                continue
            vals = a[k - m_end : k - m_start + 1][::-1] * (2 * w[k])
            grad_a[m_start : m_end + 1] += vals
        
        # Project to log-space
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return (2.0 * n) * grad_z
    
    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxfun': maxiter * 40, 'disp': False})
        z_out = res.x
        z_max = z_out.max()
        exp_z = np.exp(z_out - z_max)
        a_out = exp_z / exp_z.sum()
        score = compute_score_normalized(a_out)
        if score < best['score']:
            best['a'] = a_out
            best['score'] = score
    except:
        pass
    
    best['time'] = time.monotonic() - t0
    return best['a'], best['score']

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_inter = np.interp(new_x, old_x, f_n)
    f_inter = np.maximum(f_inter, 0)
    ss = f_inter.sum()
    if ss > 1e-15:
        f_inter = f_inter / ss
    return f_inter

def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative weight (mirror descent) updates with analytically computed peak subgradients
2. L-BFGS in log-space with tempered subgradients
3. Length exploration via interpolation
4. Momentum-based gradient descent with adaptive steps
Key innovations:
- Exact subgradient formula: d(conv[k])/da[m] = 2*a[k-m] for max(0,k-n+1)<=m<=min(n-1,k)
- Direct multiplicative updates on the simplex (no softmax parameterization)
- Combined mirror descent + L-BFGS with multiple temperature settings
- Adaptive length search around the parent length
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score_normalized(a):
    """Score for a sequence normalized to sum=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def peak_subgradient(a, top_k=30, T_frac=0.05):
    """
    Compute a subgradient of max_k conv[k] w.r.t. a using tempered weights.
    Formula: d(conv[k])/da[m] = 2*a[k-m] for max(0,k-n+1) <= m <= min(n-1,k).
    Returns (gradient, convolution).
    """
    n = len(a)
    conv = fast_convolve(a)
    total_size = 2 * n - 1

    if total_size <= top_k:
        peak_indices = np.arange(total_size)
        weights = np.ones(total_size) / total_size
    else:
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_values = conv[peak_indices]
        max_val = peak_values.max()
        if max_val < 1e-15:
            weights = np.ones(top_k) / top_k
        else:
            T = T_frac * max_val
            weights = np.exp((peak_values - max_val) / T)
            weights = weights / weights.sum()

    grad = np.zeros(n)

    for k_idx in range(len(peak_indices)):
        k = int(peak_indices[k_idx])
        w = weights[k_idx]

        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start > m_end:
            continue

        # d(conv[k])/da[m] = 2*a[k-m] for m in [m_start, m_end]
        # a[k-m] as m goes from m_start to m_end is a[k-m_start], a[k-m_start-1], ..., a[k-m_end]
        # i.e., a[k-m_end : k-m_start+1] reversed
        segment = a[k - m_end : k - m_start + 1][::-1]
        grad[m_start : m_end + 1] += 2.0 * w * segment

    return grad, conv


def mirror_descent(a0, time_limit, eta0=0.1, decay=0.9999, top_k=30, T_frac=0.05,
                   verbose=False, momentum_beta=0.0):
    """
    Mirror descent (multiplicative weights) on the simplex with optional momentum.
    a is kept normalized to sum=1.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-15).astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    eta = eta0
    iters = 0
    patience = 0
    patience_limit = 300
    rng = np.random.default_rng(int(t0 * 1000) % 100000)
    velocity = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        iters += 1
        grad, conv = peak_subgradient(a, top_k=top_k, T_frac=T_frac)

        # Momentum on gradient
        velocity = momentum_beta * velocity + (1 - momentum_beta) * grad
        eff_grad = velocity if momentum_beta > 0 else grad

        # Multiplicative update with clipped gradient
        eff_grad_clipped = np.clip(eff_grad, -5.0 / max(eta, 1e-6), 5.0 / max(eta, 1e-6))
        log_update = -eta * eff_grad_clipped
        log_update = np.clip(log_update, -5, 5)  # prevent overflow
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        score = compute_score_normalized(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new
        eta *= decay

        if patience > patience_limit:
            # Perturb to escape
            noise_scale = 0.005 * (1 + patience / patience_limit)
            a = best_a + rng.standard_normal(n) * noise_scale
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            eta = eta0 * 0.5
            velocity = np.zeros(n)
            patience = 0

        if verbose and iters % 200 == 0:
            el = time.monotonic() - t0
            print(f"    MD iter={iters}, t={el:.1f}s, eta={eta:.6f}, best={best_score:.8f}")

    return best_a, best_score


def lbgfs_logspace(a0, time_limit, T=500, maxiter=3000):
    """L-BFGS in log-space with tempered subgradient smoothing."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-15))
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score_normalized(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()
        conv = fast_convolve(a)
        # Smoothed max: (1/T)*log(sum(exp(T*conv)))
        Tc = T * conv
        Tc_max = Tc.max()
        lse = Tc_max + np.log(np.exp(Tc - Tc_max).sum())
        return 2.0 * n * lse / T

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()
        conv = fast_convolve(a)

        # Tempered weights
        Tc = T * conv
        Tc_max = Tc.max()
        w = np.exp(Tc - Tc_max)
        w = w / w.sum()

        # grad_a = sum_k w_k * d(conv[k])/da
        grad_a = np.zeros(n)
        for k in range(len(conv)):
            if w[k] < 1e-12:
                continue
            m_start = max(0, k - n + 1)
            m_end = min(n - 1, k)
            if m_start > m_end:
                continue
            segment = a[k - m_end : k - m_start + 1][::-1]
            grad_a[m_start : m_end + 1] += 2.0 * w[k] * segment

        # Project to log-space: grad_z = a * (grad_a - mean_weighted)
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return 2.0 * n * grad_z / 1.0  # The 1/T is already absorbed in func scaling

    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxfun': maxiter * 50, 'disp': False})
        z_out = res.x
        z_max = z_out.max()
        exp_z = np.exp(z_out - z_max)
        a_out = exp_z / exp_z.sum()
        a_out = np.maximum(a_out, 0)
        s = a_out.sum()
        a_out = a_out / s
        score = compute_score_normalized(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_inter = np.interp(new_x, old_x, f_n)
    f_inter = np.maximum(f_inter, 0)
    ss = f_inter.sum()
    if ss > 1e-15:
        f_inter = f_inter / ss
    return f_inter

def coordinate_descent(a0, time_limit, n_pairs=200):
    """Coordinate descent with pairwise transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        improved = False
        for _ in range(n_pairs):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            base = min(a[i], a[j]) if a[j] > 1e-12 else a[i] * 0.1
            for frac in [0.1, 0.3, 0.5, 0.7]:
                t = frac * base
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 1e-15)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score_normalized(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
                # Try reverse
                a_trial2 = a.copy()
                t2 = frac * a[j]
                a_trial2[j] -= t2
                a_trial2[i] += t2
                a_trial2 = np.maximum(a_trial2, 1e-15)
                a_trial2 = a_trial2 / a_trial2.sum()
                sc2 = compute_score_normalized(a_trial2)
                if sc2 < best_score - 1e-13:
                    a = a_trial2
                    best_score = sc2
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            # Perturb
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

    return best_a, best_score

        t_alloc = min(rem * 0.15, 120.0)
        print(f"  MD[{idx}] eta0={cfg['eta0']}, k={cfg['top_k']}, T={cfg['T_frac']}, "
              f"mom={cfg['momentum_beta']}, budget={t_alloc:.0f}s")
        a_md, sc_md = mirror_descent(best_seq, t_alloc, verbose=False, **cfg)
        best_seq, best_score = check_and_update(a_md, f"MD{idx}_e{cfg['eta0']}", best_seq, best_score)

    # === Phase 2: L-BFGS with multiple temperatures ===
    print(f"\n=== Phase 2: L-BFGS Multi-T (best={best_score:.8f}) ===")
    for T_val in [100, 300, 1000, 5000]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.12, 100.0)
        print(f"  L-BFGS T={T_val}, budget={t_alloc:.0f}s")
        a_lb, sc_lb = lbgfs_logspace(best_seq, t_alloc, T=T_val, maxiter=4000)
        best_seq, best_score = check_and_update(a_lb, f"LBFGS_T{T_val}", best_seq, best_score)

    # === Phase 3: Mirror Descent Round 2 (from improved point, finer params) ===
    print(f"\n=== Phase 3: MD Round 2 (best={best_score:.8f}) ===")
    md_configs_2 = [
        {'eta0': 0.02, 'decay': 0.99999, 'top_k': 10, 'T_frac': 0.01, 'momentum_beta': 0.0},
        {'eta0': 0.04, 'decay': 0.99997, 'top_k': 15, 'T_frac': 0.02, 'momentum_beta': 0.2},
        {'eta0': 0.08, 'decay': 0.99993, 'top_k': 25, 'T_frac': 0.04, 'momentum_beta': 0.4},
    ]
    for idx, cfg in enumerate(md_configs_2):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.12, 100.0)
        print(f"  MD2[{idx}] eta0={cfg['eta0']}, k={cfg['top_k']}, budget={t_alloc:.0f}s")
        a_md, sc_md = mirror_descent(best_seq, t_alloc, verbose=False, **cfg)
        best_seq, best_score = check_and_update(a_md, f"MD2_{idx}", best_seq, best_score)

    # === Phase 4: Length exploration ===
    print(f"\n=== Phase 4: Length Exploration (best={best_score:.8f}) ===")
    candidate_lengths = []
    for delta in [1, -1, 2, -2, 4, -4, 8, -8, 12, -12, 16, -16, 32, -32]:
        nl = n_parent + delta
        if 64 <= nl <= 4096 and nl != n_parent and nl not in candidate_lengths:
            candidate_lengths.append(nl)
    # Add some structured lengths
    for nl in [512, 768, 1024, 1536, 2048]:
        if nl != n_parent and nl not in candidate_lengths:
            candidate_lengths.append(nl)

    for nl in candidate_lengths[:8]:
        rem = deadline - time.monotonic()
        if rem < 100:
            break
        t_alloc = min(rem * 0.1, 80.0)
        init_alt = interpolate_seq(best_seq, nl)
        # Quick MD
        a_md, _ = mirror_descent(init_alt, t_alloc * 0.4,
                                  eta0=0.08, decay=0.9999, top_k=30, T_frac=0.05)
        # Quick L-BFGS
        a_lb, sc_lb = lbgfs_logspace(a_md, t_alloc * 0.5, T=500, maxiter=2000)
        best_seq, best_score = check_and_update(a_lb, f"len{nl}", best_seq, best_score)

    # === Phase 5: Coordinate descent for local refinement ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(rem * 0.12, 80.0)
        print(f"  CD budget={t_alloc:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t_alloc, n_pairs=300)
        best_seq, best_score = check_and_update(a_cd, "CD", best_seq, best_score)

    # === Phase 6: Final L-BFGS polish ===
    print(f"\n=== Phase 6: Final L-BFGS Polish (best={best_score:.8f}) ===")
    for T_val in [200, 1000, 10000]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.15, 60.0)
        a_lb, sc_lb = lbgfs_logspace(best_seq, t_alloc, T=T_val, maxiter=5000)
        best_seq, best_score = check_and_update(a_lb, f"polish_T{T_val}", best_seq, best_score)

    # === Phase 7: Final fine mirror descent ===
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.3, 50.0)
        print(f"\n=== Phase 7: Final Fine MD (best={best_score:.8f}) ===")
        a_md, sc_md = mirror_descent(best_seq, t_alloc,
                                      eta0=0.01, decay=0.999995, top_k=10, T_frac=0.01,
                                      momentum_beta=0.0)
        best_seq, best_score = check_and_update(a_md, "final_MD", best_seq, best_score)

    # === Phase 8: Ultra-fine final polish ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(rem * 0.4, 30.0)
        a_lb, sc_lb = lbgfs_logspace(best_seq, t_alloc, T=50, maxiter=10000)
        best_seq, best_score = check_and_update(a_lb, "ultra_polish", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    return [float(x) for x in best_seq]
