import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_refine_cutting_plane(f, time_limit=60.0, K=100, max_iters=50):
    """
    LP-based refinement using only top-K binding constraints (cutting plane).
    Solves: max sum(g) s.t. conv(f,g)(k) <= M for top-K k, g >= 0
    Then blends with f.
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        
        # Select top-K most binding constraints
        if K < len(b):
            top_idx = np.argpartition(b, -K)[-K:]
        else:
            top_idx = np.arange(len(b))
        
        # Build constraint matrix for selected constraints
        A = np.zeros((len(top_idx), n))
        for row, k in enumerate(top_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # A[row, i] = f[k-i] for valid i
            for i in range(lo, hi + 1):
                A[row, i] = a[k - i]
        
        b_vec = np.full(len(top_idx), M)
        c = -np.ones(n)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_vec,
                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                break
            
            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            
            g_norm = g / g_sum
            
            # Line search for best blend
            best_t = 0.0
            best_blend_obj = best_obj
            for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                s = blend.sum()
                if s < 1e-10:
                    continue
                blend = blend / s
                obj = fast_convolve(blend).max()
                if obj < best_blend_obj:
                    best_blend_obj = obj
                    best_t = t
            
            if best_t > 0:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                obj = fast_convolve(a).max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                else:
                    a = best_a.copy()
            else:
                break
                
        except Exception:
            break
    
    return best_a, best_obj


def peak_shaving_cd(a0, time_limit=30.0):
    """
    Coordinate descent with peak-shaving heuristic.
    For each coordinate, find the optimal value that minimizes the max conv.
    Uses the insight that reducing high-contributing coordinates at peak lags
    and redistributing to low-contributing ones reduces the peak.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    # Identify the peak lags
    b = fast_convolve(a)
    peak_val = b.max()
    peak_mask = b >= peak_val * 0.95
    peak_lags = np.where(peak_mask)[0]
    
    def contribution(i):
        """Total contribution of coordinate i to peak lags."""
        c = 0.0
        for k in peak_lags:
            j = k - i
            if 0 <= j < n:
                c += a[j]
            j2 = i - k
            if 0 <= j2 < n:
                c += a[j2]
        return c
    
    improved = True
    passes = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        passes += 1
        
        # Sort coordinates by contribution to peaks (descending)
        # Higher contribution = more "blame" for the peak
        contribs = np.zeros(n)
        for i in range(n):
            contribs[i] = contribution(i)
        
        # Try reducing high-contribution coordinates
        order = np.argsort(contribs)[::-1]
        
        for idx in order[:n // 2]:  # Only try top n/2 coordinates
            if time.monotonic() - t0 > time_limit:
                break
            
            i = idx
            old_val = a[i]
            if old_val < 1e-15:
                continue
            
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            
            # Try reducing this coordinate
            best_trial_obj = best_obj
            best_trial_val = old_val
            
            for frac in [0.3, 0.5, 0.7, 0.8, 0.9]:
                new_val = old_val * frac
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                obj = fast_convolve(a_trial).max()
                if obj < best_trial_obj - 1e-14:
                    best_trial_obj = obj
                    best_trial_val = new_val
            
            if best_trial_val != old_val:
                scale = (1.0 - best_trial_val) / denom
                a[i] = best_trial_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_trial_obj < best_obj - 1e-14:
                    best_obj = best_trial_obj
                    best_a = a.copy()
                    improved = True
                    # Update peak info
                    b = fast_convolve(a)
                    peak_val = b.max()
                    peak_mask = b >= peak_val * 0.95
                    peak_lags = np.where(peak_mask)[0]
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Cutting-plane LP refinement (top-K binding constraints)
- Peak-shaving coordinate descent with targeted updates
- Softened subgradient descent with FFT-accelerated gradient
- Multi-scale coarse-to-fine optimization
- Diverse window-function and spectral initializations
- Simulated-annealing-style restarts for escaping local minima

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_fft(a, w):
    """
    Compute subgradient g[i] = sum_k w[k]*(a[k-i] + a[i-k]) with bounds.
    Uses FFT for efficiency.
    """
    n = len(a)
    # Pad to avoid circular convolution artifacts
    # We need length >= (2n-1) + (n-1) - 1 + 1 = 3n-1 for safety
    L = 1 << (3 * n).bit_length()
    
    A = rfft(a, L)
    W = rfft(w, L)
    
    # h1[i] = sum_k w[k] * a[k-i] = cross-correlation
    # h1 = ifft(conj(FFT(w)) * FFT(a))
    H1 = irfft(np.conj(W) * A, L)
    
    # h2[i] = sum_k w[k] * a[i-k] = convolution
    # h2 = ifft(FFT(w) * FFT(a))
    H2 = irfft(W * A, L)
    
    # h1[i] for i in [0, n): this is (w cross-correlate a)[i]
    # With our FFT convention: ifft(conj(W)*A)[i] = sum_j w[j+i] * a[j]
    # which is sum_m w[m+i]*a[m] = h1[i] ✓
    
    # h2[i] for i in [0, n): ifft(W*A)[i] = sum_j w[j]*a[i-j]
    # which is sum_k w[k]*a[i-k] = h2[i] ✓
    
    g = H1[:n] + H2[:n]
    return g


def lp_refine_topk(f, time_limit=60.0, K=80, max_iters=40, verbose=False):
    """
    LP-based refinement using only top-K binding constraints.
    Solves: max sum(g) s.t. conv(f,g)(k) <= M for top-K lags, g >= 0
    Blends solution with f to reduce peak.
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        
        # Select top-K most binding constraints
        if K < len(b):
            top_idx = np.argpartition(b, -K)[-K:]
        else:
            top_idx = np.arange(len(b))
        
        # Build constraint matrix: A[row, i] = f[k-row - i] for valid indices
        A = np.zeros((len(top_idx), n))
        for row, k in enumerate(top_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                A[row, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
        
        b_vec = np.full(len(top_idx), M)
        c = -np.ones(n)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_vec,
                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                break
            
            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            
            g_norm = g / g_sum
            
            # Line search for best blend
            best_t = 0.0
            best_blend_obj = best_obj
            for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.25, 0.35, 0.5, 0.7, 1.0]:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                s = blend.sum()
                if s < 1e-10:
                    continue
                blend = blend / s
                obj = fast_convolve(blend).max()
                if obj < best_blend_obj:
                    best_blend_obj = obj
                    best_t = t
            
            if best_t > 0:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                obj = fast_convolve(a).max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    if verbose:
                        print(f"      LP iter {it}: t={best_t:.3f}, max_b={best_obj:.10f}")
                else:
                    a = best_a.copy()
            else:
                break
                
        except Exception:
            break
    
    return best_a, best_obj


def peak_shaving_cd(a0, time_limit=30.0, verbose=False):
    """
    Coordinate descent with peak-shaving heuristic.
    Reduces coordinates that contribute most to peak lags,
    redistributing mass to low-contribution coordinates.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    def compute_contributions(a_cur, b_cur):
        """Compute contribution of each coordinate to the top peaks."""
        peak_val = b_cur.max()
        peak_mask = b_cur >= peak_val * 0.92
        peak_lags = np.where(peak_mask)[0]
        
        contribs = np.zeros(n)
        for k in peak_lags:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                contribs[i] += a_cur[k - i]
        return contribs
    
    improved_global = True
    passes = 0
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        passes += 1
        if passes > 20:
            break
        
        b = fast_convolve(a)
        contribs = compute_contributions(a, b)
        
        # Sort by contribution descending (most "blame" first)
        order = np.argsort(contribs)[::-1]
        
        for idx in order[:n // 2]:
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            
            i = int(idx)
            old_val = a[i]
            if old_val < 1e-14:
                continue
            
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            
            best_trial_obj = fast_convolve(a).max()
            best_trial_val = old_val
            
            for frac in [0.4, 0.6, 0.75, 0.85, 0.92]:
                new_val = old_val * frac
                scale = (1.0 - new_val) / denom
                if scale < 0 or scale > 10.0:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                obj = fast_convolve(a_trial).max()
                if obj < best_trial_obj - 1e-14:
                    best_trial_obj = obj
                    best_trial_val = new_val
            
            if best_trial_val != old_val:
                scale = (1.0 - best_trial_val) / denom
                a[i] = best_trial_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a[mask] *= scale
                a = a / a.sum()
                cur_obj = fast_convolve(a).max()
                if cur_obj < best_obj - 1e-14:
                    best_obj = cur_obj
                    best_a = a.copy()
                    improved_global = True
    
    return best_a, best_obj


def subgradient_optimize(a0, time_limit, lr_init=0.004, momentum=0.92,
                         n_top=50, seed=42, verbose=False):
    """
    Subgradient method with FFT-accelerated gradient and adaptive step size.
    Uses top-n_top peaks with softmax weighting.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    no_improve = 0
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        
        # Select active peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))
        
        # Use peaks within 5% of max for robustness
        threshold = b_max * 0.95
        active_mask_local = b[top_idx] >= threshold
        if not active_mask_local.any():
            active_mask_local = np.zeros(len(top_idx), dtype=bool)
            active_mask_local[np.argmax(b[top_idx])] = True
        
        active_indices = top_idx[active_mask_local]
        n_active = len(active_indices)
        
        # Softmax weights
        active_vals = b[active_indices]
        temp = max(b_max * 0.001, 1e-10)
        w_active = np.exp((active_vals - active_vals.max()) / temp)
        w_active = w_active / w_active.sum()
        
        w = np.zeros(len(b))
        w[active_indices] = w_active
        
        # Compute subgradient via FFT
        g = subgradient_fft(a, w)
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm
        
        # Nesterov momentum step
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)
        
        obj_new = fast_convolve(a_new).max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and step % 500 == 0:
                print(f"      SG step {step}: max_b={best_obj:.10f}, lr={cur_lr:.6f}")
        else:
            no_improve += 1
        
        a = a_new
        step += 1
        
        # Adaptive strategy
        if no_improve >= 500:
            # Restart with kick
            kick = 0.01 / (1 + step / 500)
            a = project_simplex(best_a + rng.standard_normal(n) * kick)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, lr_init * 0.02)
            no_improve = 0
        elif no_improve >= 200:
            cur_lr *= 0.7
        elif no_improve >= 100:
            cur_lr *= 0.85
    
    return best_a, best_obj


def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):

    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:

        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):

def make_flat_top_init(n, w=0.6):

def make_two_bump_init(n, sep=0.3):

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

    # === Phase 3: Diverse initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Diverse Initializations (best={best_score:.6f})")
    print(f"{'='*60}")
    n_base = len(best_seq) if best_seq is not None else n_parent
    
    inits = {}
    inits["flat_top_0.7"] = make_flat_top_init(n_base, 0.7)
    inits["flat_top_0.85"] = make_flat_top_init(n_base, 0.85)
    inits["tukey_0.3"] = make_tukey_init(n_base, 0.3)
    inits["tukey_0.6"] = make_tukey_init(n_base, 0.6)
    inits["cosine"] = make_cosine_init(n_base)
    inits["gauss_0.35"] = make_gaussian_init(n_base, 0.35)
    inits["two_bump"] = make_two_bump_init(n_base, 0.25)
    
    total_diverse = min(250.0, (deadline - time.monotonic() - 350.0) * 0.35)
    time_per_init = total_diverse / len(inits)
    
    for name, init_a in inits.items():
        rem = deadline - time.monotonic() - 300
        if rem < 15:
            break
        t_alloc = min(time_per_init, rem * 0.25)
        if t_alloc < 10:
            continue
        t_sg_i = t_alloc * 0.4
        t_lp_i = t_alloc * 0.35
        t_cd_i = t_alloc * 0.25
        print(f"  {name} (total={t_alloc:.0f}s)")
        a_div, obj_div = combined_optimize(
            init_a, t_sg_i, t_lp_i, t_cd_i, n_base,
            seed + hash(name) % 100000
        )
        update_best(a_div, f"div_{name}")
    
    # === Phase 4: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-length Exploration (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        target_lengths = [512, 768, 1024, 1280, 1536]
        target_lengths = [x for x in target_lengths if 256 <= x <= 3000]
        target_lengths.sort(key=lambda x: abs(x - n_base))
        
        for target_n in target_lengths[:4]:
            rem = deadline - time.monotonic() - 200
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.2, 50.0)
            if t_alloc < 12:
                continue
            t_sg_l = t_alloc * 0.4
            t_lp_l = t_alloc * 0.35
            t_cd_l = t_alloc * 0.25
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_len, obj_len = combined_optimize(
                f_interp, t_sg_l, t_lp_l, t_cd_l, target_n,
                seed + target_n
            )
            update_best(a_len, f"len_{target_n}")
    
    # === Phase 5: Deep optimization on best ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Deep Optimization (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 200
        if rem > 40:
            print(f"  Deep: n={n_best}, rem={rem:.0f}s")
            t_sg_d = min(rem * 0.35, 100.0)
            t_lp_d = min(rem * 0.3, 70.0)
            t_cd_d = min(rem * 0.2, 40.0)
            a_deep, obj_deep = combined_optimize(
                best_seq, t_sg_d, t_lp_d, t_cd_d, n_best,
                seed + 424242, verbose=True
            )
            update_best(a_deep, "deep1")
        
        # Extra LP with larger K
        rem = deadline - time.monotonic() - 120
        if rem > 15 and best_seq is not None:
            n_cur = len(best_seq)
            K_large = min(200, 2 * n_cur)
            print(f"  Extra LP: K={K_large}, t={min(rem,40.0):.0f}s")
            a_extra, obj_extra = lp_refine_topk(
                best_seq, time_limit=min(rem, 40.0),
                K=K_large, max_iters=30
            )
            update_best(a_extra, "extra_lp")
        
        # Final SG with very small step
        rem = deadline - time.monotonic() - 80
        if rem > 15 and best_seq is not None:
            n_cur = len(best_seq)
            t_fine = min(rem * 0.5, 30.0)
            print(f"  Fine SG: t={t_fine:.0f}s")
            a_fine, obj_fine = subgradient_optimize(
                best_seq, time_limit=t_fine,
                lr_init=0.0005, momentum=0.96,
                n_top=max(40, n_cur // 5), seed=seed + 888888
            )
            update_best(a_fine, "fine_sg")
    
    # === Phase 6: Final polish ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final Polish (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            # Quick peak shaving
            t_ps = min(rem * 0.4, 20.0)
            a_ps, obj_ps = peak_shaving_cd(best_seq, time_limit=t_ps)
            update_best(a_ps, "final_peaks")
            
            # Tiny LP
            rem = deadline - time.monotonic() - 15
            if rem > 8:
                n_f = len(best_seq)
                a_fl, obj_fl = lp_refine_topk(
                    best_seq, time_limit=min(rem, 15.0),
                    K=min(100, n_f), max_iters=15
                )
                update_best(a_fl, "final_lp")
            
            # Quick CD
            rem = deadline - time.monotonic() - 10
            if rem > 5:
                a_fc, obj_fc = peak_shaving_cd(best_seq, time_limit=min(rem, 8.0))
                update_best(a_fc, "final_cd")
    
    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")
    
    return [float(x) for x in best_seq]
