(1) Adam gradient descent on the probability simplex with adaptive temperature annealing,
(2) exact pairwise coordinate optimization with golden-section search,

    n = len(a)

    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]

    cond = u * idx > (cssv - 1)
    rho = np.nonzero(cond)[0]

    theta = (cssv[r] - 1.0) / (r + 1.0)

                if c < 1e-14:
                    continue
                best_t = a[i] / c
                best_trial_score = best_score

                # Coarse grid search
                for t_val in np.linspace(0.0, 1.0, grid_size):
                    a_trial = a.copy()
                    a_trial[i] = c * t_val
                    a_trial[j] = c * (1.0 - t_val)
                    score = get_score_normalized(a_trial)
                    if score < best_trial_score:
                        best_trial_score = score
                        best_t = t_val

                # Fine refinement around best_t
                if best_trial_score < best_score - improvement_threshold:
                    lo = max(0.0, best_t - 1.0 / grid_size)
                    hi = min(1.0, best_t + 1.0 / grid_size)
                    for _ in range(refine_iters):
                        mid1 = lo + (hi - lo) / 3.0
                        mid2 = hi - (hi - lo) / 3.0
                        s1 = get_score_normalized(_make_trial(a, i, j, c, mid1))
                        s2 = get_score_normalized(_make_trial(a, i, j, c, mid2))
                        if s1 < s2:
                            hi = mid2
                        else:
                            lo = mid1
                    best_t = (lo + hi) / 2.0
                    best_trial_score = get_score_normalized(_make_trial(a, i, j, c, best_t))

                if best_trial_score < best_score - improvement_threshold:
                    a[i] = c * best_t
                    a[j] = c * (1.0 - best_t)
                    a = proj_simplex(a)
                    best_score = get_score_normalized(a)
                    best_a = a.copy()
                    improved = True

        if verbose and sweep % 10 == 0:
            print(f"    Pairwise sweep {sweep}: score={best_score:.8f}")

        if not improved:
            break

    return best_a, best_score


def _make_trial(a, i, j, c, t):
    """Create a trial sequence with a[i]=c*t, a[j]=c*(1-t)."""
    a_trial = a.copy()
    a_trial[i] = c * t
    a_trial[j] = c * (1.0 - t)
    return a_trial


def coordinate_optimize(a0, time_limit, grid_size=30, refine_iters=8, verbose=True):
    """
    Coordinate-wise 1D optimization: for each element, find the best value
    (given all others fixed) via golden section search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = proj_simplex(a)
    best_score = get_score_normalized(a)
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0
    rng = np.random.default_rng(99)
    improvement_threshold = 1e-14

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            # Try a range of values for a[i]
            # Scale range based on current value
            lo = 0.0
            hi = a[i] * 3.0 + 0.01 / n

            # Coarse grid
            best_val = a[i]
            best_trial_score = best_score
            for val in np.linspace(lo, hi, grid_size):
                a_trial = a.copy()
                a_trial[i] = val
                a_trial = proj_simplex(a_trial)
                # Only accept if sum constraint is approximately maintained
                score = get_score_normalized(a_trial)
                if score < best_trial_score:
                    best_trial_score = score
                    best_val = val

            # Fine refinement
            if best_trial_score < best_score - improvement_threshold:
                idx_in_grid = int(round((best_val - lo) / (hi - lo) * (grid_size - 1)))
                idx_in_grid = max(1, min(grid_size - 2, idx_in_grid))
                refined_lo = lo + (hi - lo) * (idx_in_grid - 1) / (grid_size - 1)
                refined_hi = lo + (hi - lo) * (idx_in_grid + 1) / (grid_size - 1)
                for _ in range(refine_iters):
                    m1 = refined_lo + (refined_hi - refined_lo) / 3.0
                    m2 = refined_hi - (refined_hi - refined_lo) / 3.0
                    s1 = get_score_normalized(proj_simplex(_set_coord(a, i, m1)))
                    s2 = get_score_normalized(proj_simplex(_set_coord(a, i, m2)))
                    if s1 < s2:
                        refined_hi = m2
                    else:
                        refined_lo = m1
                best_val = (refined_lo + refined_hi) / 2.0
                a_new = proj_simplex(_set_coord(a, i, best_val))
                best_trial_score = get_score_normalized(a_new)

            if best_trial_score < best_score - improvement_threshold:
                a = proj_simplex(_set_coord(a, i, best_val))
                best_score = get_score_normalized(a)
                best_a = a.copy()
                sweep_improved = True

        if verbose and sweep % 5 == 0:
            print(f"    Coord sweep {sweep}: score={best_score:.8f}")

        if not sweep_improved:
            break

    return best_a, best_score


def _set_coord(a, i, val):
    """Return copy of a with coordinate i set to val."""
    a_new = a.copy()
    a_new[i] = val
    return a_new


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, maintaining sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def kkt_diagnostic(a, n_peaks=3):
    """
    Check KKT conditions: at optimum, for active peak k*,
    a[k*-i] should be ~constant for all active i.
    Returns a measure of how far from KKT we are.
    """
    n = len(a)
    b = fast_convolve(a)
    peak_indices = np.argsort(b)[::-1][:n_peaks]
    worst = 0.0
    for k_star in peak_indices:
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        vals = []
        for i in range(i_min, i_max + 1):
            if a[i] > 0.005:
                vals.append(a[k_star - i] if 0 <= k_star - i < n else 0.0)
        if len(vals) > 5:
            vals = np.array(vals)
            ratio = vals.max() / (vals.min() + 1e-15)
            worst = max(worst, ratio)
    return worst


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization to minimize the convolution peak score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_default = 768
        best_seq = np.ones(n_default) / n_default
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default uniform: n={n_default}, score={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting best: {best_score:.8f} (n={n_parent})")

    def track(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        try:
            sc = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a_norm)})")

    # === Phase 1: Adam refinement from parent (35% time) ===
    print(f"\n=== Phase 1: Adam Refinement (n={n_parent}) ===")
    t_p1 = (deadline - time.monotonic()) * 0.35
    t_p1 = max(60.0, min(t_p1, 700.0))

    # Multiple Adam runs with different schedules
    n_adam_runs = 4
    time_per_adam = t_p1 / n_adam_runs
    schedules = [
        ('hold_anneal', 0.04, 0.002, 0.008),
        ('cubic', 0.05, 0.001, 0.006),
        ('exp', 0.03, 0.002, 0.01),
        ('hold_anneal', 0.02, 0.001, 0.005),
    ]

    for idx in range(n_adam_runs):
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        t_alloc = min(time_per_adam, rem * 0.85)
        if t_alloc < 20:
            continue

        sched, T_i, T_f, lr_val = schedules[idx % len(schedules)]
        print(f"  Adam run {idx+1}/{n_adam_runs}: sched={sched}, T=[{T_i},{T_f}], lr={lr_val}, budget={t_alloc:.0f}s")

        a_res, score_res = adam_simplex(
            best_seq, t_alloc, T_init=T_i, T_final=T_f,
            lr=lr_val, schedule=sched, verbose=False
        )
        track(a_res, f"adam_{idx}_{sched}")

    print(f"  After Phase 1: best={best_score:.8f}")

    # === Phase 2: Pairwise optimization (25% time) ===
    print(f"\n=== Phase 2: Pairwise Optimization ===")
    rem = deadline - time.monotonic() - 300
    t_p2 = max(30.0, min(rem * 0.25, 450.0))
    print(f"  Budget: {t_p2:.0f}s")

    # Multiple passes with increasing aggressiveness
    for pass_idx in range(3):
        rem = deadline - time.monotonic() - 200
        if rem < 20:
            break
        t_sub = min(t_p2 / (3 - pass_idx), rem * 0.7)
        if t_sub < 10:
            break

        n_pk = 3 + pass_idx
        n_tp = 20 + pass_idx * 10
        print(f"  Pairwise pass {pass_idx+1}: peaks={n_pk}, top_pairs={n_tp}, budget={t_sub:.0f}s")

        a_res, score_res = pairwise_optimize(
            best_seq, t_sub, n_peaks=n_pk,
            top_pairs_per_peak=n_tp, grid_size=30, refine_iters=12
        )
        track(a_res, f"pairwise_pass{pass_idx}")

    print(f"  After Phase 2: best={best_score:.8f}")

    # === Phase 3: Coordinate optimization (20% time) ===
    print(f"\n=== Phase 3: Coordinate Optimization ===")
    rem = deadline - time.monotonic() - 200
    t_p3 = max(20.0, min(rem * 0.25, 350.0))
    print(f"  Budget: {t_p3:.0f}s")

    a_res, score_res = coordinate_optimize(
        best_seq, t_p3, grid_size=25, refine_iters=10, verbose=True
    )
    track(a_res, "coordinate")

    print(f"  After Phase 3: best={best_score:.8f}")

    # === Phase 4: Second round of Adam + pairwise (15% time) ===
    print(f"\n=== Phase 4: Second Round Refinement ===")
    rem = deadline - time.monotonic() - 150
    t_p4 = max(20.0, min(rem * 0.5, 300.0))

    # Adam with very low temperature
    t_sub = t_p4 * 0.5
    a_res, score_res = adam_simplex(
        best_seq, t_sub, T_init=0.01, T_final=0.0005,
        lr=0.003, schedule='exp', verbose=False
    )
    track(a_res, "adam_lowT")

    # Pairwise
    rem = deadline - time.monotonic() - 100
    if rem > 20:
        t_sub = min(rem * 0.6, 150.0)
        a_res, score_res = pairwise_optimize(
            best_seq, t_sub, n_peaks=5, top_pairs_per_peak=40,
            grid_size=35, refine_iters=15
        )
        track(a_res, "pairwise_final")

    print(f"  After Phase 4: best={best_score:.8f}")

    # === Phase 5: Multi-length exploration (remaining time) ===
    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    rem = deadline - time.monotonic() - 60

    if rem > 60:
        n_best = len(best_seq)
        alt_lengths = []
        for delta in [64, 128, 192, 256, 32, -64, -128, -32]:
            alt_n = n_best + delta
            if 256 <= alt_n <= 4096 and alt_n != n_best:
                if alt_n not in alt_lengths:
                    alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:5]:
            rem = deadline - time.monotonic() - 30
            if rem < 30:
                break
            t_sub = min(rem * 0.4, 80.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, score_alt = adam_simplex(
                init_alt, t_sub, T_init=0.03, T_final=0.002,
                lr=0.007, schedule='hold_anneal'
            )
            track(a_alt, f"alt_n{alt_n}")
            print(f"  n={alt_n}: score={get_score_normalized(a_alt):.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    rem = deadline - time.monotonic() - 10

    if rem > 20:
        # Quick pairwise
        t_sub = min(rem * 0.5, 60.0)
        a_res, _ = pairwise_optimize(
            best_seq, t_sub, n_peaks=3, top_pairs_per_peak=20,
            grid_size=20, refine_iters=8
        )
        track(a_res, "final_pairwise")

    rem = deadline - time.monotonic() - 5
    if rem > 10:
        # Very short Adam
        t_sub = min(rem * 0.7, 30.0)
        a_res, _ = adam_simplex(
            best_seq, t_sub, T_init=0.005, T_final=0.0005,
            lr=0.002, schedule='exp'
        )
        track(a_res, "final_adam")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.8f}")

    return [float(x) for x in best_seq]
