    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
(1) FFT subgradient with log-sum-exp smoothing and Nesterov momentum
(2) LP-based peak reduction with golden-section line search
(3) Coordinate descent polish
(4) Multi-length exploration via interpolation + optimization
(5) Deep final optimization on the best candidate
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def max_conv(a):
    """Maximum value of auto-convolution."""
    return float(fast_convolve(a).max())


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([min(max(v[0], 0.0), 1.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_step(a, b, temperature=0.01):
    """
    Compute smoothed subgradient using log-sum-exp weighting.
    Returns gradient vector g of length n.
    """
    n = len(a)
    b_len = len(b)
    # Log-sum-exp for numerical stability
    b_max = b.max()
    exp_b = np.exp((b - b_max) / temperature)
    weights = exp_b / exp_b.sum()
    
    # g[j] = sum_k weights[k] * a[k-j] (cross-correlation)
    fft_size = 1 << (n + b_len - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(weights, fft_size)
    g = irfft(np.conj(A) * W, fft_size)[:n]
    return g

    t0 = time.monotonic()

    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = max_conv(a)
    lr = lr_init
    no_improve = 0
    it = 0

        g = subgradient_step(a, b, temperature)

        v = momentum * v - lr * g
        a_new = project_simplex(a + v)
        
        obj = max_conv(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new

def interpolate_seq(f, new_n):
    """Interpolate to new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss > 1e-15:
        fi /= ss
    else:
        fi = np.ones(new_n) / new_n
    return fi

"""
Multi-phase optimizer for minimizing 2n*max(auto-convolution)/(sum)^2.
Phases: (1) SG descent, (2) LP peak reduction, (3) CD polish,
(4) multi-length exploration, (5) deep final optimization.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def max_conv(a):
    return float(fast_convolve(a).max())


def project_simplex(v):
    n = len(v)
    if n == 1:
        val = float(v[0])
        return np.array([min(max(val, 0.0), 1.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a, n):
    rows = 2 * n - 1
    A = np.zeros((rows, n))
    for k in range(rows):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = a[k - hi : k - lo + 1][::-1]
    return A

                temperature=0.02, seed=42):

    t0 = time.monotonic()
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = max_conv(a)
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        # Log-sum-exp weights
        bm = b.max()
        exp_b = np.exp((b - bm) / temperature)
        w = exp_b / exp_b.sum()
        
        # Subgradient via FFT
        fft_size = 1 << (n + len(b) - 1).bit_length()
        A_f = rfft(a, fft_size)
        W_f = rfft(w, fft_size)
        g = irfft(np.conj(A_f) * W_f, fft_size)[:n]
        
        v = momentum * v - lr * g
        a_new = project_simplex(a + v)
        obj = max_conv(a_new)
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        it += 1
        
        if no_improve > 3000:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            lr = max(lr * 0.5, lr_init * 0.05)
            no_improve = 0
        elif no_improve > 1500:
            lr *= 0.96
        elif no_improve > 800:
            temperature = max(temperature * 0.97, 0.001)
    
    return best_a, best_obj
