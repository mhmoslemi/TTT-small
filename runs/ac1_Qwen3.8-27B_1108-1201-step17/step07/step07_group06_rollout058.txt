    # We'll ignore it for now

"""
Novel optimization for step function score minimization.
Combines soft-max gradient descent with simulated annealing,
spectral domain shape control, and multi-scale exploration.
Key innovations: (1) frequency-domain peak suppression via 
iterative power spectrum shaping, (2) simulated annealing with 
structured moves to escape local minima, (3) adaptive multi-
temperature scheduling based on progress.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def get_score_fast(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def softmax_from_z(z):
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_grad_a(a, w, n, fft_size):
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * n * corr


def adam_softmax(a0, time_limit, T_init=0.05, T_final=0.002, lr=0.01,
                 beta1=0.9, beta2=0.999, eps=1e-8, schedule='exp'):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 5000
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        
        if schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        else:  # hold_anneal
            if progress < 0.3:
                T = T_init
            else:
                p2 = (progress - 0.3) / 0.7
                T = T_init * (T_final / T_init) ** p2
        
        a_curr = softmax_from_z(z)
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        
        grad_a = compute_grad_a(a_curr, w, n, fft_size)
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)
        
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.01
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            T_init = max(0.005, T_init * 0.5)
    
    return best_a, best_score


def simulated_annealing(a0, time_limit, T_init=0.001, T_final=0.00001, 
                        n_iter_max=None):
    """
    Simulated annealing with structured moves for the step function problem.
    Uses the exact score function (no soft-max approximation).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = get_score_fast(a)
    current_score = best_score
    
    t0 = time.monotonic()
    elapsed_total = time_limit
    if n_iter_max is None:
        n_iter_max = int(time_limit * 100)  # rough estimate
    
    # Estimate iterations per second
    # For n=768, each evaluation takes ~0.1ms, so ~10000 iter/s
    # But we also do conv computations in moves
    
    rng = np.random.default_rng(42)
    
    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, (time.monotonic() - t0) / elapsed_total)
        T = T_init * (T_final / T_init) ** progress
        
        # Choose move type
        move_type = rng.integers(0, 4)
        a_trial = a.copy()
        
        if move_type == 0:
            # Small Gaussian perturbation on random subset
            n_perturb = rng.integers(max(1, n // 50), max(2, n // 10))
            idx = rng.choice(n, size=n_perturb, replace=False)
            a_trial[idx] += rng.standard_normal(n_perturb) * T * 100
            a_trial = np.maximum(a_trial, 0)
            
        elif move_type == 1:
            # Swap two elements
            i, j = rng.choice(n, size=2, replace=False)
            a_trial[i], a_trial[j] = a_trial[j], a_trial[i]
            
        elif move_type == 2:
            # Mass redistribution: take from high, give to low
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a_trial[i] < 1e-10:
                continue
            amt = a_trial[i] * rng.uniform(0.01, 0.2)
            a_trial[i] -= amt
            a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            
        else:
            # "Smooth" move: blend with a smoothed version
            kernel_size = rng.integers(3, 7)
            kernel = np.ones(kernel_size) / kernel_size
            smoothed = np.convolve(a_trial, kernel, mode='same')
            blend = rng.uniform(0.05, 0.3)
            a_trial = (1 - blend) * a_trial + blend * smoothed
            a_trial = np.maximum(a_trial, 0)
        
        ss = a_trial.sum()
        if ss < 1e-15:
            continue
        a_trial = a_trial / ss
        
        trial_score = get_score_fast(a_trial)
        
        # Accept/reject
        if trial_score < current_score:
            a = a_trial
            current_score = trial_score
            if trial_score < best_score:
                best_score = trial_score
                best_a = a_trial.copy()
        else:
            # Metropolis acceptance
            delta = trial_score - current_score
            if delta < T:
                accept_prob = np.exp(-delta / max(T, 1e-15))
                if rng.random() < accept_prob:
                    a = a_trial
                    current_score = trial_score
    
    return best_a, best_score

def exact_peak_reduction(a0, time_limit, n_trials=100000):
    """
    Direct reduction of the maximum convolution peak.
    At each step, find the peak and reduce the contributing pairs.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = get_score_fast(a)
    t0 = time.monotonic()
    
    # Budget per trial
    time_per_trial = time_limit / n_trials
    max_per_trial = max(10, int(time_per_trial * 1000))  # in microseconds
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]
        target = best_score / (2.0 * n)
        
        if peak_val <= target * 1.0001:
            break
        
        # Find contributing pairs
        # conv[peak_k] = sum_{i=max(0,peak_k-n+1)}^{min(n-1,peak_k)} a[i] * a[peak_k-i]
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        pairs = [(i, peak_k - i) for i in idx]
        contribs = np.array([a[i] * a[j] for i, j in pairs])
        
        # Try reducing the top contributor
        order = np.argsort(contribs)[::-1]
        improved = False
        
        for oi in order[:10]:
            i, j = int(idx[order[oi]]), int(peak_k - idx[order[oi]])
            if i == j:
                # Can only reduce a[i]
                for factor in [0.99, 0.98, 0.95, 0.9]:
                    a_trial = a.copy()
                    removed = a_trial[i] * (1 - factor)
                    a_trial[i] *= factor
                    # Redistribute
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask]
                    os = others.sum()
                    if os < 1e-15:
                        continue
                    a_trial[mask] += removed * others / os
                    score = get_score_fast(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
            else:
                # Can reduce a[i] or a[j] or both
                for fi in [0.99, 0.95, 0.9]:
                    for fj in [1.0, 1.01, 1.02]:
                        a_trial = a.copy()
                        a_trial[i] *= fi
                        a_trial[j] *= fj
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        score = get_score_fast(a_trial)
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a_trial.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break
        
        if not improved:
            # Try a random perturbation
            if trial % 100 == 0:
                a = best_a.copy()
                idx_pert = np.random.choice(n, size=10, replace=False)
                a[idx_pert] *= (1 + np.random.standard_normal(10) * 0.01)
                a = np.maximum(a, 0)
                a = a / a.sum()
    
    return best_a, best_score

"""
Novel optimization for step function score minimization combining:
(1) Multi-peak soft-max gradient descent with adaptive temperature,
(2) Exact greedy peak-pair reduction for precise local polishing,
(3) Directional restart strategy using peak attribution vectors,
(4) p-norm smoothing (p=4,8,16) as alternative to log-sum-exp,
(5) Multi-scale exploration and coordinate refinement.
Key difference from prior approach: targets multiple peaks simultaneously
and uses a two-stage strategy (broad gradient search → exact greedy polish).
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def get_score_fast(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def softmax_from_z(z):
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_grad_a(a, w, n, fft_size):
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * n * corr


def pnorm_weights(b, p):
    """Compute weights for p-norm smoothing: w[k] = (b[k]/sum(b))^p * len(b)"""
    total = b.sum()
    if total < 1e-300:
        return np.ones_like(b) / len(b)
    probs = b / total
    w = probs ** p * len(b)
    w = w / w.sum()
    return w


def gradient_pnorm(a, b, p, n, fft_size):
    """Gradient of p-norm of convolution using weights."""
    total = b.sum()
    if total < 1e-300:
        return np.zeros(n)
    probs = b / total
    w = probs ** (p - 1) * n  # normalized
    w = w / w.sum()
    return compute_grad_a(a, w, n, fft_size)


def adam_multipeak(a0, time_limit, T_init=0.05, T_final=0.001, lr=0.01,
                   beta1=0.9, beta2=0.999, eps=1e-8, schedule='hold_anneal',
                   top_k=5, p_mode=0):
    """
    Adam optimizer targeting top-k peaks simultaneously.
    p_mode: 0=softmax, 1=p-norm(p=4), 2=p-norm(p=8)
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 8000
    current_T_init = T_init

    while time.monotonic() - t0 < time_limit:
        t += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)

        if schedule == 'exp':
            T = current_T_init * (T_final / current_T_init) ** progress
        elif schedule == 'linear':
            T = current_T_init + (T_final - current_T_init) * progress
        else:  # hold_anneal
            if progress < 0.25:
                T = current_T_init
            else:
                p2 = (progress - 0.25) / 0.75
                T = current_T_init * (T_final / current_T_init) ** p2

        a_curr = softmax_from_z(z)
        b = fast_convolve(a_curr)

        if p_mode == 0:
            # Soft-max with multi-peak weighting
            max_b = b.max()
            exp_b = np.exp(b / T - max_b / T)
            w = exp_b / exp_b.sum()
            # Boost top-k peaks
            if top_k > 1 and top_k < len(b):
                top_idx = np.argsort(b)[::-1][:top_k]
                boost = 3.0
                w_top = w[top_idx].copy()
                w[top_idx] = 0
                w[top_idx] += boost * w_top
                w = w / w.sum()
        elif p_mode == 1:
            w = pnorm_weights(b, 4.0)
        else:
            w = pnorm_weights(b, 8.0)

        grad_a = compute_grad_a(a_curr, w, n, fft_size)
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            # Directional restart: perturb toward reducing top peaks
            b_best = fast_convolve(best_a)
            top_idx = np.argsort(b_best)[::-1][:top_k]
            direction = np.zeros(n)
            for ki in top_idx:
                k = int(ki)
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                for i in range(js, je + 1):
                    j = k - i
                    if 0 <= j < n:
                        direction[i] -= best_a[j] * 0.01
                        if i != j:
                            direction[j] -= best_a[i] * 0.01
            z = z_from_a(best_a) - direction / best_a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            current_T_init = max(0.003, current_T_init * 0.6)

    return best_a, best_score


def exact_peak_reduction(a0, time_limit, n_trials=200000):
    """
    Greedy peak-pair reduction: directly targeting the maximum convolution
    entry by reducing the highest-contributing pairs.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = get_score_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    trial = 0
    while time.monotonic() - t0 < time_limit:
        trial += 1
        conv = fast_convolve(a)
        # Find all peaks above threshold
        peak_val = conv.max()
        target = best_score / (2.0 * n)
        if peak_val <= target * 1.00001:
            # Try to find any other peak above target
            above = conv > target * 1.001
            if not above.any():
                break
            peak_k = int(np.argmax(above * conv))
        else:
            peak_k = int(np.argmax(conv))

        # Contributing pairs for this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        contribs = np.array([a[i] * a[peak_k - i] for i in idx])
        order = np.argsort(contribs)[::-1]

        improved = False
        # Try top contributing pairs
        n_try = min(8, len(order))
        for oi in range(n_try):
            i = int(idx[order[oi]])
            j = int(peak_k - i)
            if a[i] < 1e-15 or a[j] < 1e-15:
                continue

            if i == j:
                for factor in [0.995, 0.99, 0.98, 0.95, 0.9]:
                    a_trial = a.copy()
                    removed = a_trial[i] * (1.0 - factor)
                    a_trial[i] *= factor
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask]
                    os = others.sum()
                    if os < 1e-15:
                        continue
                    a_trial[mask] += removed * others / os
                    score = get_score_fast(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
            else:
                # Reduce the larger of the pair
                if a[i] >= a[j]:
                    for factor in [0.995, 0.99, 0.98, 0.95]:
                        a_trial = a.copy()
                        removed = a_trial[i] * (1.0 - factor)
                        a_trial[i] *= factor
                        # Redistribute to non-contributing elements
                        mask = np.ones(n, dtype=bool)
                        # Try to redistribute to elements far from peak
                        for r in [i, j]:
                            if 0 <= r < n:
                                mask[r] = False
                        others = a_trial[mask]
                        os = others.sum()
                        if os < 1e-15:
                            continue
                        a_trial[mask] += removed * others / os
                        score = get_score_fast(a_trial)
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a_trial.copy()
                            improved = True
                            break
                else:
                    for factor in [0.995, 0.99, 0.98, 0.95]:
                        a_trial = a.copy()
                        removed = a_trial[j] * (1.0 - factor)
                        a_trial[j] *= factor
                        mask = np.ones(n, dtype=bool)
                        for r in [i, j]:
                            if 0 <= r < n:
                                mask[r] = False
                        others = a_trial[mask]
                        os = others.sum()
                        if os < 1e-15:
                            continue
                        a_trial[mask] += removed * others / os
                        score = get_score_fast(a_trial)
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a_trial.copy()
                            improved = True
                            break

            if improved:
                break

        if not improved:
            # Stuck: try broader perturbation
            if trial % 500 == 0:
                a = best_a.copy()
                n_pert = min(20, n // 10)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a[idx_pert] *= (1.0 + rng.standard_normal(n_pert) * 0.02)
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss > 1e-15:
                    a = a / ss
                    if get_score_fast(a) < best_score:
                        best_score = get_score_fast(a)
                        best_a = a.copy()

    return best_a, best_score


def lbgfs_polish(a0, time_limit, T=100, maxiter=3000):
    """L-BFGS polishing with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    stopped = [False]

    def soft_max_obj(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        return 2.0 * n * (max_b + T * np.log(exp_b.sum()))

    def soft_max_grad(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        grad_a = compute_grad_a(a, w, n, fft_size)
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    try:
        res = minimize(
            fun=soft_max_obj,
            x0=z0,
            jac=soft_max_grad,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 100, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = a0.copy()

    return a_out


def multi_peak_gradient_step(a, n_top=5, step_size=0.01):
    """
    Single gradient step targeting all top peaks simultaneously.
    Returns the updated sequence.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    b = fast_convolve(a)
    top_idx = np.argsort(b)[::-1][:n_top]
    # Weight peaks by how much they exceed the mean
    mean_b = b.mean()
    weights = np.zeros(len(b))
    for ki in top_idx:
        excess = max(0, b[ki] - mean_b)
        weights[ki] = excess
    total_w = weights.sum()
    if total_w < 1e-300:
        return a.copy()
    weights = weights / total_w

    grad_a = compute_grad_a(a, weights, n, fft_size)
    a_new = a - step_size * grad_a / (a.sum() * n)
    a_new = np.maximum(a_new, 0)
    ss = a_new.sum()
    if ss > 1e-15:
        a_new = a_new / ss
    return a_new


def coordinate_refine(a0, time_limit, n_sweeps=500):
    """
    Coordinate descent with adaptive step sizes.
    For each element, try multiple multiplicative factors and keep the best.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = get_score_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    factors = [0.999, 0.998, 0.995, 0.99, 0.98, 0.95,
               1.001, 1.002, 1.005, 1.01, 1.02, 1.05]

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-18:
                continue

            best_local_factor = 1.0
            best_local_score = best_score

            for f in factors:
                a_trial = a.copy()
                a_trial[i] *= f
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = get_score_fast(a_trial)
                if score < best_local_score - 1e-15:
                    best_local_score = score
                    best_local_factor = f

            if best_local_factor != 1.0:
                a[i] *= best_local_factor
                a = np.maximum(a, 0)
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True

        if not sweep_improved and sweep > 20:
            # Stuck: perturb
            n_pert = min(10, n // 20)
            idx = rng.choice(n, size=n_pert, replace=False)
            a[idx] *= (1.0 + rng.standard_normal(n_pert) * 0.01)
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_and_track(a, label, best_seq, best_score):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

                break
            t_sub = min(rem * 0.25, 50.0)
            a_lb = lbgfs_polish(best_seq, t_sub, T=T, maxiter=2000)
            best_seq, best_score = evaluate_and_track(
                a_lb, f"lbgfs_T{T}", best_seq, best_score
            )
            print(f"  L-BFGS T={T}: {best_score:.6f}")

    # Coordinate refinement
    rem = deadline - time.monotonic() - 120
    if rem > 20:
        t_sub = min(rem * 0.7, 120.0)
        a_cd, _ = coordinate_refine(best_seq, t_sub, n_sweeps=500)
        best_seq, best_score = evaluate_and_track(a_cd, "coord_refine", best_seq, best_score)
        print(f"  After coord refine: {best_score:.6f}")

    # More peak reduction
    rem = deadline - time.monotonic() - 100
    if rem > 15:
        t_sub = min(rem * 0.6, 60.0)
        a_pr2, _ = exact_peak_reduction(best_seq, t_sub, n_trials=100000)
        best_seq, best_score = evaluate_and_track(a_pr2, "peak_reduce_2", best_seq, best_score)

    # === Phase 4: Multi-length exploration (10% time) ===
    print(f"\n=== Phase 4: Multi-Length (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 60

    if rem > 60:
        alt_lengths = []
        for factor in [1.5, 1.25, 0.75, 2.0, 1.1]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 4096 and alt_n != n_best:
                if alt_n not in alt_lengths:
                    alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic() - 40
            if rem < 40:
                break
            t_sub = min(rem * 0.35, 60.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = adam_multipeak(
                init_alt, t_sub, T_init=0.04, T_final=0.002,
                lr=0.008, schedule='hold_anneal', top_k=5, p_mode=0
            )
            best_seq, best_score = evaluate_and_track(
                a_alt, f"alt_n{alt_n}", best_seq, best_score
            )
            print(f"  Alt n={alt_n}: {best_score:.6f}")

    # === Phase 5: Final intensive polish (10% time) ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 10

    if rem > 20:
        # Very low temperature Adam
        t_sub = min(rem * 0.4, 40.0)
        a_fin, _ = adam_multipeak(
            best_seq, t_sub, T_init=0.005, T_final=0.0002,
            lr=0.003, schedule='exp', top_k=3, p_mode=0
        )
        best_seq, best_score = evaluate_and_track(a_fin, "final_adam", best_seq, best_score)

    rem = deadline - time.monotonic() - 5
    if rem > 10:
        # Final exact peak reduction
        t_sub = min(rem * 0.8, 20.0)
        a_fin2, _ = exact_peak_reduction(best_seq, t_sub, n_trials=50000)
        best_seq, best_score = evaluate_and_track(a_fin2, "final_pr", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.6f}")

    return [float(x) for x in best_seq]
