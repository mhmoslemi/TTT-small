"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-peak subgradient descent
with active-set QP polishing, structured local search, and simulated annealing.
Key innovations: (1) direct peak-reduction via partner-pair adjustment,
(2) active-set identification for QP sub-problems, (3) frequency-domain
informed initialization, (4) multi-scale length exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft
from numpy.lib.stride_tricks import sliding_window_view


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def compute_gradient(a, w, n):
    """
    Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    grad[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    """
    # w has length 2n-1
    H = sliding_window_view(w, n)  # H[i,j] = w[i+j], shape (n, n)
    return H @ a


def softmax_obj_grad(z, n, T):
    """Objective and gradient for soft-max of conv in log-space."""
    a = np.exp(z - np.max(z))
    a = a / a.sum()
    b = fast_convolve(a)
    b_max = b.max()
    exp_b = np.exp(b / T - b_max / T)
    w = exp_b / exp_b.sum()
    
    # Soft-max objective (in normalized form)
    val = 2.0 * n * T * (b_max / T + np.log(exp_b.sum()) - np.log(exp_b.size))
    # Actually: soft_max = T * log(sum(exp(b/T)))
    # = T * (b_max/T + log(sum(exp((b-b_max)/T))))
    # = b_max + T * log(sum(exp((b-b_max)/T)))
    # We want 2n * soft_max / (sum a)^2 but sum a = 1 here
    # So objective = 2n * (b_max + T * log(sum(exp((b-b_max)/T))))
    # But we only need the part that depends on z, so:
    # obj = 2n * T * log(sum(exp(b/T)))  (the b_max/T part is constant in the log)
    # Actually let me just return the soft-max value and gradient
    
    sm_val = T * np.log(exp_b.sum()) + b_max
    obj = 2.0 * n * sm_val  # since sum(a) = 1
    
    # Gradient in a-space
    grad_a = compute_gradient(a, w, n)
    # Transform to z-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def multi_peak_subgrad(a0, time_limit, T_init=0.05, T_final=0.001, 
                        lr=0.005, momentum=0.9, K=3, verbose=False):
    """
    Multi-peak subgradient descent with momentum and temperature annealing.
    Operates in log-space (z = log(a)) with softmax normalization.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = np.log(a)
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        t_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_init * (T_final / T_init) ** t_frac
        
        a_curr = np.exp(z - np.max(z))
        a_curr = a_curr / a_curr.sum()
        b = fast_convolve(a_curr)
        b_max = b.max()
        
        # Soft-max weights
        exp_b = np.exp(b / T - b_max / T)
        w = exp_b / exp_b.sum()
        
        # Gradient
        grad_a = compute_gradient(a_curr, w, n)
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)
        
        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad_z
        z = z - lr * v
        
        # Evaluate
        a_new = np.exp(z - np.max(z))
        a_new = a_new / a_new.sum()
        score = 2.0 * n * fast_convolve(a_new).max()
        
        iter_count += 1
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive restart
        if no_improve > 2000:
            z = np.log(np.maximum(best_a, 1e-300)) + np.random.standard_normal(n) * 0.01
            v = np.zeros(n)
            no_improve = 0
            lr *= 0.8
        
        # Slow LR decay
        lr *= (1.0 - 0.00005)
        lr = max(lr, 0.0005)
        
        if verbose and iter_count % 5000 == 0:
            print(f"      Subgrad iter={iter_count} T={T:.5f} lr={lr:.6f} score={best_score:.6f}")
    
    return best_a, best_score


def peak_pair_reduction(a0, time_limit, n_trials=10000):
    """
    Peak-targeted pair reduction: identify the peak lag, find the element
    pair (i, k-i) with the largest product, and try to reduce it while
    maintaining the sum constraint.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find top peaks (excluding lag 0 and lag 2n-2 which are just a[0]^2 and a[n-1]^2)
        b_search = b.copy()
        b_search[0] = -1
        b_search[-1] = -1
        # Find top K peaks
        top_k = 3
        peak_indices = np.argsort(b_search)[::-1][:top_k]
        
        any_improved = False
        for peak_k in peak_indices:
            if b[peak_k] < b[peak_indices[0]] * 0.95:
                continue  # Only work on near-peak lags
            
            # Find contributing pairs
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx_range = np.arange(js, je + 1)
            partners = peak_k - idx_range
            products = a[idx_range] * a[partners]
            
            # Focus on top contributors
            top_contrib = np.argsort(products)[::-1][:8]
            
            for ci in top_contrib:
                i = int(idx_range[ci])
                j = int(partners[ci])
                if i == j:
                    # Single element contributes: reduce a[i]
                    for factor in [0.95, 0.9, 0.85]:
                        a_trial = a.copy()
                        a_trial[i] *= factor
                        s = a_trial.sum()
                        if s < 1e-15:
                            continue
                        # Redistribute removed mass to random other elements
                        removed = a[i] * (1 - factor)
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                        others = a_trial[mask]
                        if others.sum() < 1e-15:
                            continue
                        a_trial[mask] += removed * others / others.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a_trial.copy()
                            any_improved = True
                            break
                else:
                    # Pair (i, j): try asymmetric reductions
                    for f1 in [0.95, 0.9, 0.85]:
                        for f2 in [1.0, 1.01, 1.02]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            # Renormalize to maintain sum
                            s = a_trial.sum()
                            if s < 1e-15:
                                continue
                            a_trial = a_trial / s
                            score = 2.0 * n * fast_convolve(a_trial).max()
                            if score < best_score - 1e-14:
                                a = a_trial
                                best_score = score
                                best_a = a_trial.copy()
                                any_improved = True
                                break
                        if any_improved:
                            break
                
                if any_improved:
                    break
            if any_improved:
                break
        
        if not any_improved and trial > 50:
            # Random perturbation to escape
            if trial % 100 == 0:
                a = best_a + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0)
                a = a / a.sum()
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
    
    return best_a, best_score


def coordinate_peak_descent(a0, time_limit, n_sweeps=50):
    """
    Coordinate descent with peak-aware scoring.
    For each element, try adjusting it and keep the best.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue
            
            best_val = a[i]
            best_score_local = best_score
            
            # Try several multiplicative factors
            for factor in [0.85, 0.9, 0.95, 1.05, 1.1, 1.15]:
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                # Fast score: only need max of conv
                b_trial = fast_convolve(a_trial)
                score = 2.0 * n * b_trial.max()
                if score < best_score_local:
                    best_score_local = score
                    best_val = a[i] * factor
            
            if best_val != a[i]:
                a[i] = best_val
                a = a / a.sum()
                if best_score_local < best_score:
                    best_score = best_score_local
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_score


def active_set_qp(a0, time_limit, n_iter=100):
    """
    Active set method: identify the peak lags and try to reduce them
    by solving a local optimization problem.
    
    Uses a simple projected gradient approach on the active constraints.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for iteration in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        b_max = b.max()
        active_lags = np.where(b >= b_max * 0.99)[0]
        
        # Compute subgradient from active lags
        w = np.zeros(2 * n - 1)
        w[active_lags] = 1.0 / len(active_lags)
        
        grad_a = compute_gradient(a, w, n)
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        
        # Projected gradient step (in log-space)
        lr = 0.01 * (1.0 - iteration / n_iter)
        z = np.log(np.maximum(a, 1e-300))
        z = z - lr * grad_z
        a_new = np.exp(z - np.max(z))
        a_new = a_new / a_new.sum()
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            lr *= 0.5
            z = np.log(np.maximum(a, 1e-300))
            z = z - lr * grad_z
            a_new = np.exp(z - np.max(z))
            a_new = a_new / a_new.sum()
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score - 1e-15:
                best_score = score
                best_a = a_new.copy()
                a = a_new
            else:
                break
    
    return best_a, best_score


def simulated_annealing_step(a, T, n):
    """
    Generate a neighbor via simulated annealing-style perturbation.
    Focus perturbation on elements that contribute to peak lags.
    """
    b = fast_convolve(a)
    b_max = b.max()
    # Elements contributing to the peak
    peak_k = int(np.argmax(b))
    js = max(0, peak_k - n + 1)
    je = min(n - 1, peak_k)
    idx_range = np.arange(js, je + 1)
    partners = peak_k - idx_range
    products = a[idx_range] * a[partners]
    
    # Perturb the top contributors more
    n_perturb = min(10, len(idx_range))
    top_idx = np.argsort(products)[::-1][:n_perturb]
    
    a_new = a.copy()
    for ci in top_idx:
        i = int(idx_range[ci])
        # Random multiplicative perturbation
        factor = np.exp(np.random.standard_normal() * T)
        a_new[i] *= factor
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-peak subgradient descent
with safe mass redistribution, active-set refinement, and coordinate descent.
Key improvements: (1) targeted peak reduction via partner-pair analysis with
safe-location mass redistribution, (2) multi-peak tracking with adaptive
temperature, (3) fine-grained local search with peak-aware scoring.

import numpy as np
import time
from scipy.fft import rfft, irfft
from numpy.lib.stride_tricks import sliding_window_view


def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s*s)


def grad_from_weights(a, w, n):
    """grad[i] = sum_j a[j] * w[i+j] where w has length 2n-1."""
    H = sliding_window_view(w, n)
    return H @ a


def soft_peak_subgrad(a0, time_budget, T0, T1, lr0, mom):
    """Soft-peak subgradient descent in log-space with momentum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, float), 1e-300)
    a /= a.sum()
    z = np.log(a)
    v = np.zeros(n)
    t0 = time.monotonic()
    best = a.copy()
    best_s = 2*n*fast_convolve(a).max()
    stall = 0
    
    while time.monotonic() - t0 < time_budget:
        fr = min(1.0, (time.monotonic()-t0)/time_budget)
        T = T0*(T1/T0)**fr
        ac = np.exp(z-np.max(z)); ac/=ac.sum()
        b = fast_convolve(ac)
        bm = b.max()
        eb = np.exp(b/T - bm/T)
        w = eb/eb.sum()
        ga = grad_from_weights(ac, w, n)
        S = ga@ac
        gz = ac*(ga-S)
        v = mom*v + (1-mom)*gz
        z -= lr0*v
        an = np.exp(z-np.max(z)); an/=an.sum()
        sc = 2*n*fast_convolve(an).max()
        if sc < best_s - 1e-15:
            best_s = sc; best = an.copy(); stall = 0
        else:
            stall += 1
        if stall > 3000:
            z = np.log(np.maximum(best,1e-300)) + np.random.standard_normal(n)*0.008
            v[:] = 0; stall = 0; lr0 *= 0.75
        lr0 = max(0.0003, lr0*(1-0.00003))
    
    return best, best_s


def safe_redistribute(a, peak_lags, n):
    """
    For the given peak lags, find the best (source, target) pair to reduce
    the peak: reduce a[i] (contributing to peak) and increase a[j] (not
    contributing to any peak).
    
    Returns the best (i, j, factor) or None.
    """
    b = fast_convolve(a)
    b_max = b.max()
    
    # Find peak-contributing elements
    # For each peak lag k, the elements a[i] and a[k-i] contribute a[i]*a[k-i]
    contrib_pressure = np.zeros(n)
    for k in peak_lags:
        js = max(0, k-n+1)
        je = min(n-1, k)
        for i in range(js, je+1):
            j = k - i
            contrib_pressure[i] += a[j]
    
    # Find "safe" targets: elements with low pressure
    # But also: increasing a[j] should not create new peaks
    # Approximation: j that is far from all peak lags
    safe_score = np.zeros(n)
    for k in peak_lags:
        js = max(0, k-n+1)
        je = min(n-1, k)
        # Elements in [js, je] contribute to lag k
        safe_score[js:je+1] += 1
    
    # Best sources: high pressure
    # Best targets: low safe_score (not contributing to any peak lag)
    
    top_sources = np.argsort(contrib_pressure)[::-1][:10]
    top_targets = np.argsort(safe_score)[:10]
    
    best_move = None
    best_sc = b_max
    
    for i in top_sources:
        if a[i] < 1e-10:
            continue
        for j in top_targets:
            if i == j:
                continue
            for factor in [0.97, 0.95, 0.92, 0.9]:
                a_t = a.copy()
                a_t[i] *= factor
                removed = a[i]*(1-factor)
                a_t[j] += removed
                s = a_t.sum()
                if s < 1e-15:

def targeted_local_search(a0, time_budget, max_trials=5000):

def coord_descent(a0, time_budget, max_sweeps=100):
    """Coordinate descent with peak-aware scoring."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, float), 0.0)
    a /= a.sum()
    best = a.copy()
    best_s = 2*n*fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(31)
    factors = [0.9, 0.93, 0.96, 1.04, 1.07, 1.1]
    
    for sweep in range(max_sweeps):
        if time.monotonic()-t0 > time_budget:
            break
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic()-t0 > time_budget:
                break
            if a[i] < 1e-12:
                continue
            bv = a[i]; bs = best_s
            for f in factors:
                at = a.copy()
                at[i] = a[i]*f
                at /= at.sum()
                sc = 2*n*fast_convolve(at).max()
                if sc < bs:
                    bs = sc; bv = a[i]*f
            if bv != a[i]:
                a[i] = bv
                a /= a.sum()
                if bs < best_s:
                    best_s = bs; best = a.copy();
