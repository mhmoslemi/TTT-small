    n = len(v)
    if n == 1:
        return np.array([max(target_sum, 0.0)])
    # Scale to sum 1, then scale back
    v1 = v / max(v.sum(), 1e-30)
    # Project onto probability simplex
    u = np.sort(v1)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        result = np.ones(n) / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        result = np.maximum(v1 - theta, 0.0)
    return result * target_sum

"""
Combined LP + smooth-gradient optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(target_sum, 0.0)])
    v1 = v / max(abs(v).sum(), 1e-30)
    if v1.sum() < 0:
        v1 = np.abs(v1) / np.abs(v1).sum()
    u = np.sort(v1)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        result = np.ones(n) / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        result = np.maximum(v1 - theta, 0.0)
    return result * target_sum


def build_constraint_matrix(f):
    """Build the (2n-1) x n convolution constraint matrix A where A[k,i] = f[k-i]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[lo:hi + 1]
    return A


def lp_solve_full(f, solver=None):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for ALL k.
    Returns (g, M) or (None, M).
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    A = build_constraint_matrix(f)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solvers_to_try = []
    if solver is not None:
        solvers_to_try.append(solver)
    solvers_to_try.extend([cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS])

    for s in solvers_to_try:
        try:

            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                if np.all(np.isfinite(g_val)):

        except Exception:
            continue

    return None, M

def line_search(f, h, n_evals=8):
    """Find best step size t for f + t*(h - f) using multi-start + refinement."""
    f_norm = fast_convolve(f).max()
    best_t = 0.0
    best_obj = f_norm

    # Coarse search
    t_candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]
    for t in t_candidates:
        f_trial = (1.0 - t) * f + t * h
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial = f_trial / s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t

    if best_t == 0.0:
        return f, f_norm

    # Refine around best_t
    lo = max(0.0, best_t - 0.1)
    hi = min(1.0, best_t + 0.1)
    for _ in range(10):
        mid = (lo + hi) / 2.0
        t1 = lo + (mid - lo) * 0.618
        t2 = hi - (hi - mid) * 0.618
        f1 = np.maximum((1.0 - t1) * f + t1 * h, 0.0)
        s1 = f1.sum()
        if s1 > 1e-10:
            f1 /= s1
            o1 = fast_convolve(f1).max()
        else:
            o1 = np.inf
        f2 = np.maximum((1.0 - t2) * f + t2 * h, 0.0)
        s2 = f2.sum()
        if s2 > 1e-10:
            f2 /= s2
            o2 = fast_convolve(f2).max()
        else:
            o2 = np.inf
        if o1 < o2:
            hi = t2
        else:
            lo = t1

    t_best = (lo + hi) / 2.0
    f_best = np.maximum((1.0 - t_best) * f + t_best * h, 0.0)
    s_best = f_best.sum()
    if s_best > 1e-10:

def smooth_grad_step(f, tau, lr, momentum, v):
    """One step of smoothed-max gradient descent on the simplex."""
    n = len(f)
    b = fast_convolve(f)

    # Compute weights
    scaled = b / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum

    # Compute gradient: grad[i] = 2 * sum_k w[k] * a[k-i]
    # Use correlation: grad[i] = 2 * (a_rev * w)[n-1+i]
    a_rev = f[::-1]
    conv_res = np.convolve(a_rev, w, mode='full')
    grad = 2.0 * conv_res[n - 1: n - 1 + n]

    # Momentum update
    v = momentum * v - lr * grad
    f_new = project_simplex(f + v, target_sum=1.0)
    return f_new, v


def run_smooth_phase(f_init, time_limit, tau_init=0.005, tau_final=0.0001,

    """Smooth max gradient optimization with decreasing temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    lr = lr_init
    tau = tau_init
    no_improve = 0
    it = 0

    total_time = max(time_limit, 1.0)

    while time.monotonic() - t0 < total_time:
        it += 1
        progress = (time.monotonic() - t0) / total_time

        # Anneal temperature
        tau = tau_init * (tau_final / tau_init) ** progress
        tau = max(tau, tau_final)

        # Adaptive learning rate
        if it % 200 == 0 and no_improve < 5:
            lr = min(lr * 1.02, 0.01)
        elif it % 200 == 0:
            lr *= 0.9
            no_improve = 0

        f, v = smooth_grad_step(f, tau, lr, 0.9, v)

        if it % 50 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 30:
                # Restart with noise
                rng = np.random.default_rng(it)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001, 1.0)
                v = np.zeros(n)
                no_improve = 0

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} obj={best_obj:.12f} score={score:.6f} "
                  f"tau={tau:.6f} lr={lr:.6f} t={elapsed:.0f}s")

    return best_f, best_obj


def coordinate_polish(f_init, time_limit, n_sweeps_max=200):
    """Coordinate descent: try multiplying each element by various factors."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0]
    improved = True
    sweeps = 0

    while improved and sweeps < n_sweeps_max:
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        sweeps += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = fast_convolve(f).max()
            for mult in multipliers:
                f[j] = f[j] * mult
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_j_val
                    continue
                f_norm = f / s
                obj = fast_convolve(f_norm).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = f[j]
            f[j] = best_j_val
            if best_j_obj < best_obj - 1e-13:
                s = f.sum()
                f = f / s
                best_obj = best_j_obj
                improved = True

    return f, best_obj


def make_geometric(n, r=0.97):
    """Create a geometric sequence normalized to sum 1."""
    a = np.array([r ** i for i in range(n)])
    a = a / a.sum()
    return a


def make_triangular(n):
    """Create a triangular (tent) sequence normalized to sum 1."""
    mid = n // 2
    a = np.array([min(i + 1, n - i) for i in range(n)], dtype=float)
    a = a / a.sum()
    return a


def make_bump(n, sigma_frac=0.3):
    """Create a Gaussian bump sequence normalized to sum 1."""
    x = np.linspace(0, 1, n)
    center = 0.5
    sigma = sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = a / a.sum()
    return a


def make_one_sided(n, alpha=2.0):
    """Create a one-sided power-law sequence."""
    i = np.arange(n)
    a = (i + 1) ** (-alpha)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: multi-phase approach combining LP, smooth gradient,
    coordinate descent, and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    def update_best(f_new, label):
        nonlocal best_seq, best_score
        s = f_new.sum()
        if s < 1e-10:
            return
        f_new = f_new / s
        sc = float(evaluate_sequence(f_new.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_new.copy()
            print(f"  NEW BEST [{label}]: score={sc:.6f} n={len(f_new)}")
        else:
            print(f"  [{label}]: score={sc:.6f} (not better than {best_score:.6f})")

    n_parent = len(parent)

    # === Phase 1: LP on parent ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 1: LP main (n={n_parent}, budget={rem:.0f}s) ===")
    t1 = min(rem * 0.35, 600.0)
    if t1 > 30:
        f1, _ = run_lp_phase(parent, time_limit=t1, top_k_init=150,
                             top_k_max=2 * n_parent - 1, use_full=False,
                             label="lp_main")
        update_best(f1, "lp_main")

    # === Phase 2: LP with full constraints ===
    rem = deadline - time.monotonic()
    n_b = len(best_seq)
    print(f"\n=== Phase 2: LP full (n={n_b}, budget={rem:.0f}s) ===")
    t2 = min(rem * 0.2, 300.0)
    if t2 > 30:
        f2, _ = run_lp_phase(best_seq, time_limit=t2, top_k_init=2 * n_b - 1,
                             top_k_max=2 * n_b - 1, use_full=True,
                             label="lp_full")
        update_best(f2, "lp_full")

    # === Phase 3: Smooth gradient ===
    rem = deadline - time.monotonic()
    n_b = len(best_seq)
    print(f"\n=== Phase 3: Smooth gradient (n={n_b}, budget={rem:.0f}s) ===")
    t3 = min(rem * 0.2, 250.0)
    if t3 > 30:
        f3, _ = run_smooth_phase(best_seq, time_limit=t3, tau_init=0.01,
                                 tau_final=0.0005, lr_init=0.0005,
                                 label="smooth")
        update_best(f3, "smooth")

    # === Phase 4: Multi-length exploration ===
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    target_lengths = set()
    for factor in [0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
        tl = int(n_base * factor)
        tl = max(512, min(4000, tl))
        if tl != n_base:
            target_lengths.add(tl)

    # Also try structured initializations at parent length
    structured_inits = []
    if n_parent <= 3000:
        structured_inits.append(("geometric", make_geometric(n_parent, 0.97)))
        structured_inits.append(("triangular", make_triangular(n_parent)))
        structured_inits.append(("bump", make_bump(n_parent, 0.3)))
        structured_inits.append(("one_sided", make_one_sided(n_parent, 2.0)))

    for name, f_struct in structured_inits:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_struct = min(rem * 0.08, 60.0)
        if t_struct < 15:
            break
        print(f"  Structured init: {name}, budget={t_struct:.0f}s")
        f_s, _ = run_lp_phase(f_struct, time_limit=t_struct,
                              top_k_init=100, top_k_max=2 * n_parent - 1,
                              label=f"struct_{name}")
        update_best(f_s, f"struct_{name}")

    for target_n in sorted(target_lengths, key=lambda x: abs(x - n_base))[:4]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t4 = min(rem * 0.12, 120.0)
        if t4 < 20:
            break
        # Interpolate best_seq to new length
        old_x = np.linspace(0.0, 1.0, len(best_seq))
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  Multi-length: n={target_n}, budget={t4:.0f}s")
        f_t, _ = run_lp_phase(f_interp, time_limit=t4, top_k_init=100,
                              top_k_max=2 * target_n - 1, label=f"ml_n{target_n}")
        update_best(f_t, f"ml_n{target_n}")

    # === Phase 5: Coordinate descent ===
    rem = deadline - time.monotonic()
    n_b = len(best_seq)
    print(f"\n=== Phase 5: Coordinate descent (n={n_b}, budget={rem:.0f}s) ===")
    t5 = min(rem * 0.15, 120.0)
    if t5 > 10:
        f5, _ = coordinate_polish(best_seq, time_limit=t5)
        update_best(f5, "coord")

    # === Phase 6: Final LP burst ===
    rem = deadline - time.monotonic()
    n_b = len(best_seq)
    print(f"\n=== Phase 6: Final LP (n={n_b}, budget={rem:.0f}s) ===")
    t6 = min(rem * 0.4, 150.0)
    if t6 > 15:
        f6, _ = run_lp_phase(best_seq, time_limit=t6, top_k_init=2 * n_b - 1,
                             top_k_max=2 * n_b - 1, use_full=True,
                             label="lp_final")
        update_best(f6, "lp_final")

    # === Phase 7: Another smooth pass if time ===
    rem = deadline - time.monotonic()
    if rem > 40:
        n_b = len(best_seq)
        t7 = min(rem * 0.3, 80.0)
        print(f"\n=== Phase 7: Smooth polish (n={n_b}, budget={t7:.0f}s) ===")
        f7, _ = run_smooth_phase(best_seq, time_limit=t7, tau_init=0.003,
                                 tau_final=0.0002, lr_init=0.0003,
                                 label="smooth2")
        update_best(f7, "smooth2")

    # === Phase 8: Final coordinate + LP ===
    rem = deadline - time.monotonic()
    if rem > 20:
        n_b = len(best_seq)
        t8a = min(rem * 0.4, 40.0)
        t8b = rem - t8a - 5
        if t8a > 5:
            f8, _ = coordinate_polish(best_seq, time_limit=t8a)
            update_best(f8, "coord2")
            best_seq = f8
        if t8b > 10:
            f9, _ = run_lp_phase(best_seq, time_limit=t8b, top_k_init=2 * n_b - 1,
                                 top_k_max=2 * n_b - 1, use_full=True,
                                 label="lp_veryfinal")
            update_best(f9, "lp_veryfinal")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
