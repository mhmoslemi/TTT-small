def linear_auto_convolve(a):
    """Linear auto convolution: same as full mode with no circularity."""
    n = len(a)
    size = 2 * n - 1
    a_arr = np.array(a, dtype=float)
    padded = np.zeros(size, dtype=float)
    padded[:n] = a_arr
    conv = np.zeros(size, dtype=float)
    for i in range(size):
        for j in range(max(0, i - n + 1), min(i+1, n)):
            conv[i] += a_arr[j] * padded[i - j]
    return conv
