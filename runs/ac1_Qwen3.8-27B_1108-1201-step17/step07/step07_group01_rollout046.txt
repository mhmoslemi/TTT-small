BLISS-style optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative-weight LP approximations for the bottleneck reduction step,
combined with exact LP solves, L-BFGS polishing, and multi-start strategy.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_general(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def bliss_mw_direction(a, T_mw=200, eta=0.5):
    """
    Compute BLISS direction using multiplicative weights approximation.
    Returns b (the direction vector) with sum(b) = 1.
    """
    n = len(a)
    M = fast_convolve(a).max()
    b = np.ones(n)
    b = b / b.sum()
    
    for t in range(T_mw):
        # Compute (a*b) to find most violated constraint
        ab = fast_convolve_general(a, b)
        k_star = int(np.argmax(ab))
        
        # Update: b_i *= (1 - eta * a[k_star - i] / M)
        # a[k_star - i] is non-zero only when 0 <= k_star - i < n
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        if i_min <= i_max:
            idx = np.arange(i_min, i_max + 1)
            coeffs = a[k_star - idx]
            b[idx] *= (1.0 - eta * coeffs / M)
            b[idx] = np.maximum(b[idx], 1e-30)
    
    b = np.maximum(b, 0)
    s = b.sum()
    if s < 1e-15:
        return a.copy()
    return b / s

    from scipy.sparse import csr_matrix

    cols = []

        i_end = min(n - 1, k)

            rows.append(k)
            cols.append(i)
            data.append(a[k - i])

    c = -np.ones(n)
    A_ub = F.toarray()  # Convert to dense for linprog (might be slow for large n)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')

        s = b.sum()
        if s > 1e-15:

"""
BLISS-style optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines multiplicative-weight LP approximations for bottleneck direction finding,
analytical pairwise swaps targeting peak-contributing pairs, L-BFGS polishing in
log-space with adaptive temperature, and multi-start strategy over diverse
initializations and sequence lengths.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve2(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def bliss_mw_direction(a, T_mw=300, eta=0.3):
    """
    Compute BLISS direction using multiplicative weights approximation of the LP:
    max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
    Returns normalized direction b (sum=1).
    """
    n = len(a)
    b = fast_convolve(a)
    M = b.max()
    if M < 1e-15:
        return a.copy()

    d = np.ones(n)
    d = d / d.sum()

    for t in range(T_mw):
        # Find most violated constraint
        ad = fast_convolve2(a, d)
        k_star = int(np.argmax(ad))

        # Update: d_i *= (1 - eta * a[k_star - i] / M)
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        if i_min <= i_max:
            idx = np.arange(i_min, i_max + 1)
            coeffs = a[k_star - idx]
            multipliers = 1.0 - eta * coeffs / M
            multipliers = np.maximum(multipliers, 1e-10)
            d[idx] *= multipliers

    d = np.maximum(d, 0.0)
    s = d.sum()
    if s < 1e-15:
        return a.copy()
    return d / s


def bliss_optimize(a_init, time_limit, mw_steps=200, mw_eta=0.3,

def pairwise_swap_optimize(a_init, time_limit, verbose=True, label=""):

    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Find top peaks
        k_peaks = np.argsort(b)[::-1][:5]
        peak_improved = False

        for k in k_peaks:
            k = int(k)
            # For this peak, find the best (i, j) swap
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            idx = np.arange(i_min, i_max + 1)
            contributors = a[idx]

            # First-order change in peak k when moving mass from idx[ci] to j:
            # delta_peak = 2*d*(a[k-j] - a[k-idx[ci]])
            # We want this to be negative: a[k-j] < a[k-idx[ci]]
            # Also need to check other peaks don't increase too much.

            # Find the top contributor to this peak
            ci_best = int(np.argmax(contributors))
            i = int(idx[ci_best])

            # Find best j to move mass to: maximize (a[k-i] - a[k-j]) while
            # keeping other peaks from increasing
            # a[k-j] should be small, and we need j != i
            # Score for j: benefit = a[k-i] - a[k-j] (for this peak)
            # But also check: for other peak k2, change = 2*d*(a[k2-j] - a[k2-i])
            # We want net benefit across all peaks

            best_j = -1
            best_benefit = 0
            for j in range(n):
                if j == i:

                # Benefit for peak k

                bk_i = a[k - i] if 0 <= k - i < n else 0.0

                    bk2_i = a[k2 - i] if 0 <= k2 - i < n else 0.0
                    penalty += max(0, bk2_j - bk2_i)

                    best_benefit = net_benefit
                    best_j = j

                j = best_j
                # Try different step sizes

                    a_trial = a.copy()
                    a_trial[i] -= d
                    a_trial[j] += d
                    if a_trial.min() < 0:
                        continue
                    s = a_trial.sum()
                    a_trial = a_trial / s
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        peak_improved = True
                        break

                if peak_improved:
                    break

            no_improve += 1

        if verbose and no_improve % 100 == 0:

            print(f"      {label} PSWAP ({elapsed:.1f}s): score={best_score:.8f}, no_improve={no_improve}")

def softmax_from_z(z):
    """Convert z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def log_space_func(z, T, n):
    """Objective in log-space with soft-max smoothing."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    scaled = b / T
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return 2.0 * n * T * lse


def log_space_grad(z, T, n):
    """Gradient in log-space with soft-max smoothing."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    scaled = b / T
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()

    # Compute gradient: 4n * cross_corr(w, a)
    fft_size = 1 << (3 * n).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    correl = irfft(np.conj(W) * A, fft_size)
    grad_a = 4.0 * n * correl[:n]
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)
    return grad_z


def lbfgs_log_space(a_init, time_limit, T=50, maxiter=5000, verbose=False):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    z0 = np.log(a)
    z0 = z0 - z0.mean()
    t0 = time.monotonic()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return log_space_func(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        return log_space_grad(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        z_opt = res.x
        a_out = softmax_from_z(z_opt)
    except Exception:
        a_out = a_init.copy()

    return a_out


def multipeak_mirror_descent(a_init, time_limit, T_start=0.01, T_end=0.0005,
                             eta_start=0.02, eta_end=0.0005, max_iters=50000,
                             verbose=True, label=""):
    """Mirror descent with multi-peak aware weighting (top-K peaks get more weight)."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stuck = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (0.7 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Multi-peak aware weighting: top peaks get higher weight
        sorted_b = np.sort(b)[::-1]
        b_max = b.max()
        # Weight: peaks close to max get weight 1, others get less
        w = np.exp(-(b_max - b) / (T * 0.5))
        w = w / w.sum()

        # Gradient
        fft_size = 1 << (3 * n).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad_a = 4.0 * n * correl[:n]

        # Multi
        log_update = -eta * grad_a
        log_update = np.clip(log_update, -3.0, 3.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        if stuck > 500:
            rng = np.random.default_rng(it * 104729)
            noise = rng.standard_normal(n) * 0.03
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck = 0

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      {label} MD iter={it} ({elapsed:.1f}s): T={T:.5f} score={current_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using the actual evaluate_sequence function."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def hann_window(n):
    """Generate a Hann window normalized to sum=1."""
    t = np.arange(n)
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / (n - 1)))
    w = np.maximum(w, 1e-10)
    return w / w.sum()


def raised_cosine_variant(n, alpha=0.5):
    """Generate a raised cosine variant with tunable flatness."""
    t = np.linspace(0, 1, n)
    w = np.cos(np.pi * (2 * alpha * t - alpha)) ** 2
    w = np.maximum(w, 1e-10)
    return w / w.sum()


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Initialize from parent, Hann window, and other structures
    2. BLISS with MW (primary fast method)
    3. Multi-peak mirror descent
    4. L-BFGS polishing with adaptive temperature
    5. Pairwise swap local search
    6. Multi-length exploration
    7. Final refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score

    # === Phase 1: BLISS from parent ===
    print("\n=== Phase 1: BLISS from parent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 600
        if rem > 30:
            t_alloc = min(rem * 0.4, 300.0)
            print(f"  BLISS: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_bliss, _ = bliss_optimize(best_seq, t_alloc, mw_steps=200, mw_eta=0.3, label="P1")
            best_seq, best_score = check_and_update(a_bliss, "BLISS_P1", best_seq, best_score)

    # === Phase 2: Multi-peak mirror descent ===
    print(f"\n=== Phase 2: Multi-peak MD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t_alloc = min(rem * 0.4, 200.0)
            print(f"  MPMD: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_mpmd, _ = multipeak_mirror_descent(
                best_seq, t_alloc, T_start=0.02, T_end=0.0003,
                eta_start=0.02, eta_end=0.0003, label="MPMD"
            )
            best_seq, best_score = check_and_update(a_mpmd, "MPMD", best_seq, best_score)

    # === Phase 3: BLISS round 2 ===
    print(f"\n=== Phase 3: BLISS Round 2 (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem * 0.35, 180.0)
            print(f"  BLISS2: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_bliss2, _ = bliss_optimize(best_seq, t_alloc, mw_steps=300, mw_eta=0.2, label="P3")
            best_seq, best_score = check_and_update(a_bliss2, "BLISS_P3", best_seq, best_score)

    # === Phase 4: L-BFGS polishing ===
    print(f"\n=== Phase 4: L-BFGS (best={best_score:.8f}) ===")
    if best_seq is not None:
        for T in [100, 50, 20, 10, 5]:
            rem = deadline - time.monotonic() - 200
            if rem < 15:
                break
            t_alloc = min(rem * 0.2, 60.0)
            print(f"  L-BFGS T={T}, budget={t_alloc:.0f}s")
            a_lb = lbfgs_log_space(best_seq, t_alloc, T=T, maxiter=3000)
            best_seq, best_score = check_and_update(a_lb, f"lbfgs_T{T}", best_seq, best_score)

    # === Phase 5: Pairwise swaps ===
    print(f"\n=== Phase 5: Pairwise Swaps (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem * 0.4, 100.0)
            print(f"  PSWAP: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_psw, _ = pairwise_swap_optimize(best_seq, t_alloc, label="PSWAP")
            best_seq, best_score = check_and_update(a_psw, "pairwise_swap", best_seq, best_score)

    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [1.25, 1.5, 2.0, 0.75]:
            alt_n = int(n_best * factor)
            if 128 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic() - 100
            if rem < 60:
                break
            t_alloc = min(rem * 0.3, 80.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            print(f"  Alt n={alt_n}, budget={t_alloc:.0f}s")

            a_alt_bliss, _ = bliss_optimize(init_alt, t_alloc * 0.5, mw_steps=200, mw_eta=0.3, label=f"alt{alt_n}")
            best_seq, best_score = check_and_update(a_alt_bliss, f"alt_bliss_{alt_n}", best_seq, best_score)

            if len(best_seq) == alt_n:
                rem = deadline - time.monotonic() - 60
                if rem > 20:
                    a_alt_lb = lbfgs_log_space(best_seq, min(rem * 0.4, 40.0), T=30, maxiter=2000)
                    best_seq, best_score = check_and_update(a_alt_lb, f"alt_lb_{alt_n}", best_seq, best_score)

    # === Phase 7: Hann window start (diverse initialization) ===
    print(f"\n=== Phase 7: Diverse Initializations (best={best_score:.8f}) ===")
    for n_try in [768, 1024, 1536]:
        rem = deadline - time.monotonic() - 80
        if rem < 60:
            break
        t_alloc = min(rem * 0.25, 60.0)
        init_hann = hann_window(n_try)
        print(f"  Hann n={n_try}, budget={t_alloc:.0f}s")
        a_hann, _ = bliss_optimize(init_hann, t_alloc, mw_steps=150, mw_eta=0.4, label=f"hann{n_try}")
        best_seq, best_score = check_and_update(a_hann, f"hann_{n_try}", best_seq, best_score)

    # === Phase 8: Final refinement ===
    print(f"\n=== Phase 8: Final Refinement (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Low-T L-BFGS
        for T in [3, 1]:
            rem = deadline - time.monotonic() - 20
            if rem < 10:
                break
            t_alloc = min(rem * 0.4, 30.0)
            a_fin = lbfgs_log_space(best_seq, t_alloc, T=T, maxiter=5000)
            best_seq, best_score = check_and_update(a_fin, f"final_T{T}", best_seq, best_score)

        # Final BLISS
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_alloc = min(rem * 0.5, 30.0)
            a_fin_bliss, _ = bliss_optimize(best_seq, t_alloc, mw_steps=300, mw_eta=0.1, label="fin_bliss")
            best_seq, best_score = check_and_update(a_fin_bliss, "final_bliss", best_seq, best_score)

        # Final pairwise swaps
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            t_alloc = min(rem * 0.5, 15.0)
            a_fin_psw, _ = pairwise_swap_optimize(best_seq, t_alloc, label="fin_psw")
            best_seq, best_score = check_and_update(a_fin_psw, "final_pswap", best_seq, best_score)

    # Fallback
    if best_seq is None:
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
