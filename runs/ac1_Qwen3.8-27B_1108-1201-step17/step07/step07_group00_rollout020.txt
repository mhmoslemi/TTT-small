    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def grad_of_max_conv(f, temp=0.001, n_top=None):
    """Compute weighted gradient of max(conv(f,f)) w.r.t. f."""
    n = len(f)
    b = fast_convolve(f)

    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:

        w[b.argmax()] = 1.0
    else:

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient_fft(f, temp=0.001, n_top=None):
    """
    Compute the (weighted) gradient of max(conv(f,f)) w.r.t. f using FFT.
    Returns grad of shape (n,).
    """
    n = len(f)
    b = fast_convolve(f)
    
    # Softmax weights over the convolution values
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum
    
    if n_top is not None and n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx].copy()
        ws = w_sel.sum()
        if ws > 1e-30:
            w_sel /= ws
        else:
            w_sel = np.ones(len(top_idx)) / len(top_idx)
        w_new = np.zeros(len(b))
        w_new[top_idx] = w_sel
        w = w_new
    
    # Compute g[j] = 2 * sum_k w[k] * f[k-j] via FFT-based cross-correlation
    f_rev = f[::-1].copy()
    # Need N >= len(w) + len(f_rev) - 1 = (2n-1) + n - 1 = 3n-2
    N = 1 << (3 * n - 2 - 1).bit_length()
    W = rfft(w, N)
    F = rfft(f_rev, N)
    conv_result = irfft(W * F, N)
    g = 2.0 * conv_result[n - 1:2 * n - 1]
    
    return g

def softmax(x):
    x_max = x.max()
    e = np.exp(x - x_max)
    return e / e.sum()

def lp_solve_reduced(f, n, top_k, eps=1e-8):
    """Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for top_k peaks."""
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M
    
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)
    
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=eps)
    except Exception:
        return None, M
    
    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M

def run_adam_phase(f_init, time_limit, lr_init=0.005, temp_init=0.0003):
    """
    Adam gradient descent in log-space (softmax parameterization).
    Minimizes max(conv(f,f)) subject to sum(f)=1, f>=0.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    # Initialize log-space parameters
    f_clipped = np.maximum(f, 1e-30)
    f_norm = f_clipped / f_clipped.sum()
    x = np.log(f_norm / n)  # log(f[j] * n) - log(n) = log(f[j]) when sum(f)=1... 
    # Actually: f[j] = exp(x[j]) / sum(exp(x)), so x[j] = log(f[j]) works if sum(f)=1
    x = np.log(np.maximum(f_norm, 1e-30))
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr_init
    temp = temp_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute gradient in f-space
        df = compute_gradient_fft(f, temp=temp)
        
        # Convert to x-space gradient: dx[j] = f[j] * (df[j] - sum(f*df))
        center = np.sum(f * df)
        dx = f * (df - center)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * dx
        v = beta2 * v + (1 - beta2) * dx * dx
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        
        # Convert back to f-space
        f = softmax(x)
        
        # Evaluate
        if it % 10 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.1)
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.95
                    no_improve = 0
            
            # Periodically increase temperature (annealing in reverse)
            if it % 200 == 0:
                temp = min(temp * 1.3, 0.05)
            
            if it % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    adam iter={it} obj={best_obj:.10f} "
                      f"score={2.0*n*best_obj:.6f} lr={lr:.6f} temp={temp:.6f} t={elapsed:.0f}s")
            
            # Restart if stuck
            if no_improve > 30:
                # Add noise
                x = x + rng.standard_normal(n) * 0.01
                f = softmax(x)
                lr = lr_init
                temp = temp_init
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
    
    return best_f, best_obj

    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def compute_gradient_fft(f, temp=0.001, n_top=None):
    n = len(f)
    b = fast_convolve(f)
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum
    if n_top is not None and n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx].copy()
        ws = w_sel.sum()
        if ws > 1e-30:
            w_sel /= ws
        else:
            w_sel = np.ones(len(top_idx)) / len(top_idx)
        w_new = np.zeros(len(b))
        w_new[top_idx] = w_sel
        w = w_new
    f_rev = f[::-1].copy()
    N = 1 << (3 * n - 2 - 1).bit_length()
    W = rfft(w, N)
    F = rfft(f_rev, N)
    conv_result = irfft(W * F, N)
    g = 2.0 * conv_result[n - 1:2 * n - 1]
    return g


def run_adam_phase(f_init, time_limit, lr_init=0.003, temp_init=0.0005, seed=0):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f_norm = f / f.sum()
    x = np.log(np.maximum(f_norm, 1e-30))
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr_init
    temp = temp_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    restart_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        df = compute_gradient_fft(f, temp=temp)
        center = np.sum(f * df)
        dx = f * (df - center)
        m = beta1 * m + (1 - beta1) * dx
        v = beta2 * v + (1 - beta2) * dx * dx
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        f = softmax_fn(x)

        if it % 5 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                if it % 20 == 0:
                    lr = min(lr * 1.01, 0.05)
            else:
                no_improve += 1
                if no_improve > 3:
                    lr *= 0.97
                    no_improve = 0

            if it % 100 == 0:
                temp = min(temp * 1.15, 0.03)
            if it % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    adam iter={it} obj={best_obj:.10f} "
                      f"score={2.0 * n * best_obj:.6f} lr={lr:.6f} temp={temp:.6f} t={elapsed:.0f}s")

            if no_improve > 40:
                restart_count += 1
                if restart_count > 15:
                    break
                x = x + rng.standard_normal(n) * 0.005
                f = softmax_fn(x)
                lr = lr_init * (0.8 + 0.1 * rng.random())
                temp = temp_init * (0.5 + rng.random())
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0

    return best_f, best_obj


def run_adam_lp_hybrid(f_init, time_limit, seed=0):
    """Hybrid: alternate between Adam steps and LP-based direction finding."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f_norm = f / f.sum()
    x = np.log(np.maximum(f_norm, 1e-30))
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.003
    temp = 0.0005
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    lp_count = 0
    rng = np.random.default_rng(seed)
    top_k = 30

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Every 50 iterations, try an LP step
        if it % 50 == 0 and time.monotonic() - t0 < time_limit - 30:
            g, M = lp_solve_reduced(f, n, top_k)
            if g is not None:
                S_g = g.sum()
                if S_g > 1.0 + 1e-5:
                    h = g / S_g
                    # Line search
                    best_t = 0.05
                    best_t_obj = fast_convolve(f).max()
                    for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
                        f_trial = (1.0 - t) * f + t * h
                        f_trial = np.maximum(f_trial, 0.0)
                        s = f_trial.sum()
                        if s < 1e-10:
                            continue
                        f_trial /= s
                        obj = fast_convolve(f_trial).max()
                        if obj < best_t_obj:
                            best_t_obj = obj
                            best_t = t
                    f = (1.0 - best_t) * f + best_t * h
                    f = np.maximum(f, 0.0)
                    s = f.sum()
                    if s > 1e-10:
                        f /= s
                    x = np.log(np.maximum(f, 1e-30))
                    lp_count += 1
                    if lp_count % 5 == 0:
                        top_k = min(top_k + 10, 150)
                    cur_obj = fast_convolve(f).max()
                    if cur_obj < best_obj - 1e-13:
                        best_obj = cur_obj
                        best_f = f.copy()
                        no_improve = 0
                    continue

        # Adam step
        df = compute_gradient_fft(f, temp=temp)
        center = np.sum(f * df)
        dx = f * (df - center)
        m = beta1 * m + (1 - beta1) * dx
        v = beta2 * v + (1 - beta2) * dx * dx
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        f = softmax_fn(x)

        if it % 5 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.005, 0.03)
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.98
                    no_improve = 0

            if it % 200 == 0:
                temp = min(temp * 1.2, 0.02)
            if it % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    hybrid iter={it} obj={best_obj:.10f} "
                      f"score={2.0 * n * best_obj:.6f} lr={lr:.6f} lp={lp_count} t={elapsed:.0f}s")

            if no_improve > 50:
                x = x + rng.standard_normal(n) * 0.003
                f = softmax_fn(x)
                lr = 0.003
                temp = 0.0005
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
                if it > 50000:
                    break

    return best_f, best_obj


def lp_solve_reduced(f, n, top_k, eps=1e-8):
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=eps)
    except Exception:
        return None, M
    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M


def run_lp_phase(f_init, time_limit, top_k_init=30, top_k_max=150, label="lp"):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 50 == 0 and top_k < top_k_max:
            top_k = min(top_k + 10, top_k_max)
        g, M = lp_solve_reduced(f, n, top_k)
        if g is None:
            g, M = lp_solve_reduced(f, n, min(2 * top_k, 2 * n - 1))
            if g is None:
                break
        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 10:
                break
            top_k = min(top_k * 2, top_k_max)
            continue
        h = g / S_g
        best_t = 0.05
        best_t_obj = fast_convolve(f).max()
        for t in [0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 50:
                break
        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0 * n * best_obj:.6f} top_k={top_k} t={elapsed:.0f}s")
    return best_f, best_obj


def coordinate_polish(f_init, time_limit):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
    improved = True
    sweep = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        sweep += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val > 0.999 or new_val < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-12:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    improved = True
    return f, best_obj


def spectral_init(n, seed=0):
    rng = np.random.default_rng(seed)
    F = np.zeros(n, dtype=complex)
    if n % 2 == 0:
        F[0] = 1.0
        F[n // 2] = 1.0
        for k in range(1, n // 2):
            phase = rng.uniform(0, 2 * np.pi)
            F[k] = np.exp(1j * phase)
            F[n - k] = np.conj(F[k])
    else:
        F[0] = 1.0
        for k in range(1, (n + 1) // 2):
            phase = rng.uniform(0, 2 * np.pi)
            F[k] = np.exp(1j * phase)
            F[n - k] = np.conj(F[k])
    f = np.fft.ifft(F).real
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return f / s


def smooth_init(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 2 * np.pi, n)
    f = np.ones(n)
    for _ in range(3):
        freq = rng.integers(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        f = f + 0.3 * np.sin(freq * x + phase)
    f = np.maximum(f, 0.01)
    f /= f.sum()
    return f
