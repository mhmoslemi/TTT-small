```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Power-sum smoothed L-BFGS optimization (smooths max(conv) via Lp norm)
2. KKT-based peak equalization (solves local linear system to equalize top peaks)
3. Multi-length exploration with interpolation
4. High-precision coordinate descent with ternary search
5. Random perturbations to escape local minima

Key innovation over previous approach: The KKT peak equalization method
solves a small (K+1)x(K+1) linear system at each step to find the optimal
direction that equalizes the top K peaks, which is much more precise than
the previous heuristic peak equalization.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n*max(conv) for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def power_sum_obj(z, p, n):
    """Power-sum objective in log-space (softmax parameterization)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    conv = fast_convolve(a)
    return np.sum(conv ** p)


def power_sum_grad(z, p, n):
    """Gradient of power-sum objective in log-space."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    conv = fast_convolve(a)
    # weights: conv[k]^(p-1)
    w = conv ** (p - 1)
    # grad w.r.t. a[j] = 2p * sum_k w[k] * a[k-j] (cross-correlation)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * p * irfft(np.conj(W) * A, fft_size)[:n]
    # Project to log-space (variance-preserving)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return grad_z


def lbgfs_power(a0, time_limit, p=10, maxiter=5000):
    """L-BFGS optimization with power-sum smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    a_safe = a_safe / a_safe.sum()
    z0 = np.log(a_safe)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return power_sum_obj(z, p, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return power_sum_grad(z, p, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        z_max = res.x.max()
        exp_z = np.exp(res.x - z_max)
        a_out = exp_z / exp_z.sum()
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass

    return best_a, best_score


def kkt_peak_equalization(a0, time_limit, K=20, max_iter=10000):
    """
    KKT-based peak equalization.
    At each step, identify top K peaks, solve the KKT system to find
    the optimal direction that equalizes them, take a damped step.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iter_count = 0
    stall_count = 0
    best_stall_score = best_score

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(a)
        
        # Identify top K peaks
        if K >= len(conv):
            K = len(conv)
        top_k = np.argpartition(conv, -K)[-K:]
        
        # Compute current peak values
        peak_vals = conv[top_k]
        c = peak_vals.mean()  # target level
        
        # Residual: how much each peak exceeds c
        r = peak_vals - c
        
        # If peaks are already equalized, break
        if np.max(np.abs(r)) < 1e-14:
            break
        
        # Build the K x n Jacobian J where J[m, j] = d(conv[k_m])/d(a[j]) = 2*a[k_m - j] if valid
        # J J^T is K x K: (JJ^T)[m1, m2] = sum_j 4*a[k_m1 - j]*a[k_m2 - j] (with valid bounds)
        # This is 4 * conv[k_m1 - k_m2 + (n-1)] ... actually let me just compute it directly
        # for small K.
        
        # For efficiency with small K, compute JJ^T directly
        # (JJ^T)[m1, m2] = sum_j J[m1,j] * J[m2,j] = sum_j 4*a[k_m1-j]*a[k_m2-j] [valid]
        # = 4 * conv[k_m1 - k_m2] (where conv is the auto-cov/correlation)
        # Actually: sum_j a[k_m1-j]*a[k_m2-j] = sum_i a[i+k_m1-k_m2]*a[i] = conv[k_m1-k_m2] (shifted)
        # More precisely: sum_j a[j1]*a[j2] where j1 = k_m1-j, j2 = k_m2-j
        # Let i = j2, then j1 = i + (k_m1 - k_m2)
        # sum_i a[i + (k_m1-k_m2)] * a[i]
        # This is the cross-correlation at lag (k_m1 - k_m2), which is conv[k_m1-k_m2 + (n-1)]... 
        # Actually for auto-conv: conv[d] = sum_j a[j]*a[j+d] (with bounds)
        # So sum_i a[i+(k_m1-k_m2)]*a[i] = conv[k_m1-k_m2] if k_m1>=k_m2
        # or conv[k_m2-k_m1] if k_m2>=k_m1 (by symmetry of auto-conv)
        
        # Let's just compute JJ^T using the convolution
        # (JJ^T)[m1,m2] = 4 * corr(k_m1 - k_m2) where corr(d) = sum_j a[j]*a[j+|d|] (with proper bounds)
        # corr(d) = conv[n-1+d] if d >= 0, conv[n-1-d] if d < 0
        
        A_mat = np.zeros((K, K))
        for i in range(K):
            for j in range(K):
                d = top_k[i] - top_k[j]
                if d >= 0:
                    idx = n - 1 + d
                    if idx < len(conv):
                        A_mat[i, j] = 4.0 * conv[idx]
                    else:
                        A_mat[i, j] = 0.0
                else:
                    idx = n - 1 + d
                    if idx >= 0:
                        A_mat[i, j] = 4.0 * conv[idx]
                    else:
                        A_mat[i, j] = 0.0
        
        # b = J * ones (K-vector): b[m] = sum_j J[m,j] = sum_j 2*a[k_m - j] [valid]
        b_vec = np.zeros(K)
        for i in range(K):
            k_m = top_k[i]
            js = max(0, k_m - n + 1)
            je = min(n - 1, k_m)
            b_vec[i] = 2.0 * a[js:je+1].sum()
        
        # Solve (A - b*b^T/n) * lambda = r
        # Add small regularization for numerical stability
        M = A_mat - np.outer(b_vec, b_vec) / n
        M_reg = M + 1e-12 * np.eye(K)
        
        try:
            lambda_vec = np.linalg.solve(M_reg, r)
        except np.linalg.LinAlgError:
            lambda_vec = np.linalg.lstsq(M_reg, r, rcond=None)[0]
        
        mu = -np.dot(b_vec, lambda_vec) / n
        
        # da = -(J^T * lambda + mu * 1)
        # J^T * lambda: (J^T lambda)[j] = sum_m lambda[m] * J[m,j] = sum_m lambda[m] * 2*a[k_m - j] [valid]
        Jt_lambda = np.zeros(n)
        for m in range(K):
            if abs(lambda_vec[m]) < 1e-15:
                continue
            k_m = top_k[m]
            js = max(0, k_m - n + 1)
            je = min(n - 1, k_m)
            Jt_lambda[js:je+1] += lambda_vec[m] * 2.0 * a[k_m - js:k_m - js + (je - js + 1)]
        
        da = -(Jt_lambda + mu)
        
        # Ensure da sums to 0 (it should by construction, but clip for safety)
        da -= da.mean()
        
        # Line search: find alpha such that a + alpha*da is non-negative and score decreases
        # Max alpha before any component goes negative
        neg_mask = da < -1e-15
        if np.any(neg_mask):
            alpha_max = (a[neg_mask] / (-da[neg_mask])).min()
        else:
            alpha_max = 1.0
        alpha_max = min(alpha_max * 0.99, 1.0)
        
        # Try several step sizes
        best_alpha = 0.0
        best_trial_score = best_score
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            alpha = alpha_max * frac
            a_trial = a + alpha * da
            if a_trial.min() < -1e-14:
                continue
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            sc = compute_score(a_trial)
            if sc < best_trial_score - 1e-15:
                best_trial_score = sc
                best_alpha = alpha
        
        if best_alpha > 0:
            a = a + best_alpha * da
            a = np.maximum(a, 0)
            a = a / a.sum()
            if best_trial_score < best_score - 1e-14:
                best_score = best_trial_score
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
                # Still accept the step (it reduces the peaks towards equalization)
                if best_trial_score < best_stall_score + 1e-10:
                    stall_count += 1
        else:
            stall_count += 1
        
        if stall_count > 200:
            # Perturb to escape
            noise = rng.standard_normal(n) * 0.002
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall_count = 0
            best_stall_score = compute_score(a)
    
    return best_a, best_score


def coordinate_descent_precise(a0, time_limit, focus_on_peaks=True):
    """
    High-precision coordinate descent using ternary search.
    Focuses on coordinates that contribute to the top peaks.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    
    # Identify important coordinates (those contributing to top peaks)
    if focus_on_peaks:
        conv = fast_convolve(a)
        top_k = np.argpartition(conv, -20)[-20:]
        important = set()
        for ki in top_k:
            ki = int(ki)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            important.update(range(js, je + 1))
        important_list = np.array(list(important))
    else:
        important_list = np.arange(n)
    
    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        # Shuffle the important coordinates
        perm = rng.permutation(len(important_list))
        
        for idx in perm:
            if time.monotonic() - t0 > time_limit:
                    break
            i = important_list[idx]
            if a[i] < 1e-14:
                continue
            
            # For coordinate i, find optimal value
            # The score as a function of a[i] (with others fixed and renormalized)
            # is a piecewise quadratic (since max of quadratics)
            # Use ternary/golden section search on [0, 2*a[i]]
            
            # Quick scan first
            base_score = best_score
            best_val = a[i]
            best_local = base_score
            
            # Try a range of values
            for frac in np.logspace(-3, 0.3, 20):
                new_val = a[i] * frac
                if new_val < 0:
                    new_val = 0
                a_trial = a.copy()
                a_trial[i] = new_val
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_val = new_val
            
            if best_val != a[i] and best_local < best_score - 1e-14:
                a[i] = best_val
                ss = a.sum()
                a = a / ss
                best_score = best_local
                best_a = a.copy()
                improved = True
        
        if not improved and iters > 10:
            # Small random perturbation
            n_perturb = min(5, n)
            perturb_idx = rng.choice(n, size=n_perturb, replace=False)
            for pi in perturb_idx:
                a[pi] *= (1 + rng.standard_normal() * 0.01)
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def logsumexp_smoothed_obj(z, T, n):
    """Log-sum-exp smoothed max(conv) objective."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * lse)


def logsumexp_smoothed_grad(z, T, n):
    """Gradient of log-sum-exp smoothed objective."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    # Cross-correlation of w with a
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lbgfs_logsumexp(a0, time_limit, T, maxiter=5000):

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a_norm)})")
    return best_seq, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Power-sum smoothed L-BFGS (smooths max(conv) via Lp norm, efficient FFT gradients)
2. KKT-based peak equalization (solves (K+1)x(K+1) linear system to equalize top peaks)
3. Multi-length exploration with interpolation
4. High-precision coordinate descent focused on peak-contributing coordinates
5. Log-sum-exp smoothing with L-BFGS for additional refinement

Key innovation: The KKT peak equalization method solves a small linear system
at each step to find the exact optimal direction that equalizes the top K peaks
subject to the constraint sum(da)=0, which is far more precise than heuristic
approaches. The power-sum gradient is computed via a single FFT (cross-correlation
of a with conv^(p-1)), making each iteration very fast.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n*max(conv) for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def power_sum_grad(z, p, n):
    """Gradient of sum(conv^p) w.r.t. log-space z, via FFT cross-correlation."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    w = conv ** (p - 1)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * p * irfft(np.conj(W) * A, fft_size)[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return grad_z


def power_sum_obj(z, p, n):
    """Power-sum objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    return np.sum(conv ** p)


def lbgfs_power(a0, time_limit, p=10, maxiter=5000):
    """L-BFGS optimization with power-sum smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    a_safe = a_safe / a_safe.sum()
    z0 = np.log(a_safe)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return power_sum_obj(z, p, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return power_sum_grad(z, p, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass

    return best_a, best_score


def lse_smoothed_grad(z, T, n):
    """Gradient of log-sum-exp smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lse_smoothed_obj(z, T, n):
    """Log-sum-exp smoothed max(conv) objective."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * lse)


def lbgfs_lse(a0, time_limit, T, maxiter=5000):
    """L-BFGS with log-sum-exp smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    a_safe = a_safe / a_safe.sum()
    z0 = np.log(a_safe)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return lse_smoothed_obj(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return lse_smoothed_grad(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass

    return best_a, best_score


def kkt_peak_equalization(a0, time_limit, K=20, max_iter=50000):
    """
    KKT-based peak equalization.
    At each step: identify top K peaks, solve the KKT system for the
    optimal direction that equalizes them (min ||da||^2 s.t. J*da = -r, 1'da = 0),
    then take a damped step with line search.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iter_count = 0
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(a)

        # Identify top K peaks
        eff_K = min(K, len(conv))
        if eff_K < 2:
            break
        top_k = np.argpartition(conv, -eff_K)[-eff_K:]
        top_k_sorted = np.sort(top_k)

        # Current peak values and target
        peak_vals = conv[top_k_sorted]
        c = peak_vals.mean()
        r = peak_vals - c

        # If peaks are already well equalized, break
        if np.max(np.abs(r)) < 1e-15:
            break

        # Build A = J*J^T (K x K) and b = J*1 (K-vector)
        A_mat = np.zeros((eff_K, eff_K))
        b_vec = np.zeros(eff_K)

        for i in range(eff_K):
            k_i = int(top_k_sorted[i])
            js_i = max(0, k_i - n + 1)
            je_i = min(n - 1, k_i)
            b_vec[i] = 2.0 * a[js_i:je_i + 1].sum()

            for j in range(i, eff_K):
                k_j = int(top_k_sorted[j])
                js = max(js_i, k_j - n + 1, 0)
                je = min(je_i, k_j, n - 1)
                if js <= je:
                    idx1 = k_i - np.arange(js, je + 1)
                    idx2 = k_j - np.arange(js, je + 1)
                    val = 4.0 * np.dot(a[idx1], a[idx2])
                else:
                    val = 0.0
                A_mat[i, j] = val
                A_mat[j, i] = val

        # Solve (A - b*b^T/n) * lambda = r
        M = A_mat - np.outer(b_vec, b_vec) / n
        # Add regularization for numerical stability
        eig_min = np.abs(np.linalg.eigvalsh(M)).min() if eff_K > 1 else 1.0
        reg = max(1e-10, eig_min * 1e-6)
        M_reg = M + reg * np.eye(eff_K)

        try:
            lambda_vec = np.linalg.solve(M_reg, r)
        except np.linalg.LinAlgError:
            try:
                lambda_vec = np.linalg.lstsq(M_reg, r, rcond=None)[0]
            except Exception:
                lambda_vec = np.zeros(eff_K)

        mu = -np.dot(b_vec, lambda_vec) / n

        # Compute da = -(J^T * lambda + mu * 1)
        Jt_lambda = np.zeros(n)
        for m in range(eff_K):
            lm = lambda_vec[m]
            if abs(lm) < 1e-18:
                continue
            k_m = int(top_k_sorted[m])
            js = max(0, k_m - n + 1)
            je = min(n - 1, k_m)
            idx = k_m - np.arange(js, je + 1)
            Jt_lambda[js:je + 1] += lm * 2.0 * a[idx]

        da = -(Jt_lambda + mu)
        da -= da.mean()  # Ensure sum(da) = 0

        # Line search
        neg_mask = da < -1e-15
        if np.any(neg_mask):
            alpha_max = (a[neg_mask] / (-da[neg_mask])).min()
        else:
            alpha_max = 1.0
        alpha_max = min(alpha_max * 0.99, 1.0)

        best_alpha = 0.0
        best_trial_score = best_score
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            alpha = alpha_max * frac
            if alpha < 1e-15:
                continue
            a_trial = a + alpha * da
            if a_trial.min() < -1e-14:
                continue
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            sc = compute_score(a_trial)
            if sc < best_trial_score - 1e-16:
                best_trial_score = sc
                best_alpha = alpha

        if best_alpha > 0:
            a = a + best_alpha * da
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            if best_trial_score < best_score - 1e-15:
                best_score = best_trial_score
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
        else:
            stall_count += 1

        if stall_count > 300:
            # Perturb to escape local minimum
            noise_scale = 0.001 * (1 + stall_count / 1000.0)
            noise = rng.standard_normal(n) * noise_scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            stall_count = 0

    return best_a, best_score


def coordinate_descent_precise(a0, time_limit, focus_on_peaks=True):
    """
    High-precision coordinate descent.
    For each important coordinate, scan a range of values to find the best.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(77)

    # Identify important coordinates
    if focus_on_peaks:
        conv = fast_convolve(a)
        top_k = np.argpartition(conv, -25)[-25:]
        important = set()
        for ki in top_k:
            ki = int(ki)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            important.update(range(js, je + 1))
        important_list = np.array(sorted(important))
    else:
        important_list = np.arange(n)

    # Scan values: logarithmically spaced multipliers
    scan_fracs = np.concatenate([
        np.logspace(-4, -1, 10),
        np.logspace(-0.5, 1.5, 15),
        [1.0]
    ])

    iters = 0
    no_improve_iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        perm = rng.permutation(len(important_list))

        for idx in perm:
            if time.monotonic() - t0 > time_limit:
                break
            i = important_list[idx]
            if a[i] < 1e-15:
                continue

            best_val = a[i]
            best_local = best_score

            for frac in scan_fracs:
                new_val = a[i] * frac
                if new_val < 0:
                    new_val = 0.0
                a_trial = a.copy()
                a_trial[i] = new_val
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_val = new_val

            if best_val != a[i] and best_local < best_score - 1e-14:
                a[i] = best_val
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True

        if improved:
            no_improve_iters = 0
        else:
            no_improve_iters += 1
            if no_improve_iters > 5:
                # Small perturbation of a few coordinates
                n_perturb = min(10, len(important_list))
                pert_idx = rng.choice(len(important_list), size=n_perturb, replace=False)
                for pi in pert_idx:
                    i = important_list[pi]
                    a[i] *= (1.0 + rng.standard_normal() * 0.02)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve_iters = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate candidate via evaluate_sequence and update best."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization:
    1. Power-sum L-BFGS (multiple p) - fast smooth optimization
    2. Log-sum-exp L-BFGS (multiple T) - alternative smoothing
    3. KKT peak equalization (multiple K) - precise peak balancing
    4. Multi-length exploration
    5. Coordinate descent polish
    6. Final combined refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized uniform: n={n_init}, score={best_score:.10f}")

    n_parent = len(parent)
    a = best_seq.copy()

    # === Phase 1: Power-sum L-BFGS ===
    print(f"\n=== Phase 1: Power-sum L-BFGS (best={best_score:.10f}) ===")
    for p_val in [4, 6, 10, 20]:
        rem = deadline - time.monotonic() - 1100
        if rem < 30:
            break
        t_alloc = min(rem * 0.3, 100.0)
        print(f"  PowerSum p={p_val}, budget={t_alloc:.0f}s")
        a_p, sc_p = lbgfs_power(a, t_alloc, p=p_val, maxiter=4000)
        best_seq, best_score = check_and_update(a_p, f"ps_p{p_val}", best_seq, best_score)
        if a_p is not None and a_p.sum() > 0.01:
            a = a_p / a_p.sum()
        print(f"    current best: {best_score:.10f}")

    # === Phase 2: Log-sum-exp L-BFGS ===
    print(f"\n=== Phase 2: LogSumExp L-BFGS (best={best_score:.10f}) ===")
    for T_val in [30, 100, 500, 2000]:
        rem = deadline - time.monotonic() - 900
        if rem < 25:
            break
        t_alloc = min(rem * 0.25, 80.0)
        print(f"  LSE T={T_val}, budget={t_alloc:.0f}s")
        a_l, sc_l = lbgfs_lse(a, t_alloc, T=T_val, maxiter=4000)
        best_seq, best_score = check_and_update(a_l, f"lse_T{T_val}", best_seq, best_score)
        if a_l is not None and a_l.sum() > 0.01:
            a = a_l / a_l.sum()

    # === Phase 3: KKT Peak Equalization (main) ===
    print(f"\n=== Phase 3: KKT Peak Eq (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_alloc = min(rem * 0.45, 250.0)
        print(f"  KKT K=15, budget={t_alloc:.0f}s")
        a_k, sc_k = kkt_peak_equalization(a, t_alloc, K=15, max_iter=100000)
        best_seq, best_score = check_and_update(a_k, "kkt_K15", best_seq, best_score)
        if a_k is not None and a_k.sum() > 0.01:
            a = a_k / a_k.sum()
        print(f"    current best: {best_score:.10f}")

    # === Phase 4: KKT with different K values ===
    print(f"\n=== Phase 4: KKT Variants (best={best_score:.10f}) ===")
    for K_val in [8, 25, 40]:
        rem = deadline - time.monotonic() - 450
        if rem < 20:
            break
        t_alloc = min(rem * 0.3, 100.0)
        print(f"  KKT K={K_val}, budget={t_alloc:.0f}s")
        a_k2, sc_k2 = kkt_peak_equalization(a, t_alloc, K=K_val, max_iter=50000)
        best_seq, best_score = check_and_update(a_k2, f"kkt_K{K_val}", best_seq, best_score)
        if a_k2 is not None and a_k2.sum() > 0.01:
            a = a_k2 / a_k2.sum()

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.10f}) ===")
    alt_lengths = [n_parent - 2, n_parent + 2, n_parent - 1, n_parent + 1,
                   n_parent - 4, n_parent + 4, n_parent - 8, n_parent + 8,
                   n_parent // 2, n_parent * 2]
    for alt_n in alt_lengths:
        if alt_n < 32 or alt_n > 8192:
            continue
        rem = deadline - time.monotonic() - 300
        if rem < 25:
            break
        t_alloc = min(rem * 0.12, 50.0)
        init_alt = interpolate_sequence(a, alt_n)
        # Quick KKT then power-sum
        a_alt1, _ = kkt_peak_equalization(init_alt, t_alloc * 0.5, K=10, max_iter=20000)
        a_alt2, _ = lbgfs_power(a_alt1, t_alloc * 0.4, p=10, maxiter=2000)
        best_seq, best_score = check_and_update(a_alt2, f"len_{alt_n}", best_seq, best_score)

    # === Phase 6: Final combined refinement ===
    print(f"\n=== Phase 6: Final Refinement (best={best_score:.10f}) ===")
    a = best_seq.copy()

    rem = deadline - time.monotonic() - 250
    if rem > 30:
        t1 = min(rem * 0.3, 70.0)
        a_r1, _ = kkt_peak_equalization(a, t1, K=20, max_iter=50000)
        best_seq, best_score = check_and_update(a_r1, "fin_kkt20", best_seq, best_score)
        if a_r1 is not None and a_r1.sum() > 0.01:
            a = a_r1 / a_r1.sum()

    rem = deadline - time.monotonic() - 200
    if rem > 20:
        t2 = min(rem * 0.3, 50.0)
        a_r2, _ = lbgfs_power(a, t2, p=15, maxiter=3000)
        best_seq, best_score = check_and_update(a_r2, "fin_ps15", best_seq, best_score)
        if a_r2 is not None and a_r2.sum() > 0.01:
            a = a_r2 / a_r2.sum()

    rem = deadline - time.monotonic() - 150
    if rem > 20:
        t3 = min(rem * 0.35, 50.0)
        a_r3, _ = kkt_peak_equalization(a, t3, K=12, max_iter=50000)
        best_seq, best_score = check_and_update(a_r3, "fin_kkt12", best_seq, best_score)
        if a_r3 is not None and a_r3.sum() > 0.01:
            a = a_r3 / a_r3.sum()

    # === Phase 7: Coordinate descent polish ===
    print(f"\n=== Phase 7: Coord Descent (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic() - 80
    if rem > 20:
        t4 = min(rem * 0.5, 60.0)
        a_cd, _ = coordinate_descent_precise(a, t4, focus_on_peaks=True)
        best_seq, best_score = check_and_update(a_cd, "coord", best_seq, best_score)
        if a_cd is not None and a_cd.sum() > 0.01:
            a = a_cd / a_cd.sum()

    # === Phase 8: Very final KKT + LSE ===
    rem = deadline - time.monotonic() - 30
    if rem > 15:
        t5 = min(rem * 0.4, 25.0)
        a_vf, _ = kkt_peak_equalization(a, t5, K=10, max_iter=20000)
        best_seq, best_score = check_and_update(a_vf, "vf_kkt", best_seq, best_score)

    rem = deadline - time.monotonic() - 15
    if rem > 10:
        t6 = min(rem * 0.6, 15.0)
        a_vf2, _ = lbgfs_lse(a, t6, T=100, maxiter=2000)
        best_seq, best_score = check_and_update(a_vf2, "vf_lse", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
