Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding

import time

from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy import sparse

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_convolve(a, b):
    """Cross-convolution of a and b via FFT (a has length n, b has length m)."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())

    """Convert z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

    """Compute softmax weights over auto-correlation values."""
    scaled = b / T
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()
    return w

    """
    Solve the LP: max sum(b) s.t. b >= 0, conv(a,b) <= M
    where M = max(conv(a,a)).
    Returns the optimal b (normalized to sum=1) and the peak M.
    """
    n = len(a)
    aa = fast_convolve(a)
    M = float(aa.max())
    
    if M < 1e-30:
        return None, M
    
    # Build constraint matrix A such that (A @ b)[k] = conv(a,b)[k]
    # A[k, l] = a[k-l] if 0 <= k-l < n, else 0
    # k ranges from 0 to 2n-2, l ranges from 0 to n-1

    rows = []
    cols = []
    vals = []

                rows.append(k)
                cols.append(l)
                vals.append(a[idx])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # LP: max 1^T b s.t. A @ b <= M, b >= 0
    c = -np.ones(n)  # minimize -sum(b)
    b_ub = np.full(2 * n - 1, M)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs')
        if res.success:
            b_opt = res.x
            s = b_opt.sum()
            if s > 1e-15:
                return b_opt / s, M
            else:
                return None, M
        else:
            return None, M
    except Exception:
        return None, M

                stuck_count = 0

        new_score = score_normalized(a_new)
        if new_score < best_score - 1e-15:

            best_a = a_new.copy()
            a = a_new
            stuck_count = 0
        else:

            if t < 0.001:

                    stuck_count = 0

                new_score = score_normalized(a_new)
                if new_score < best_score - 1e-15:

                    best_a = a_new.copy()
                    a = a_new

                  f"score={new_score:.7f} best={best_score:.7f}")

        l_start = max(0, k - n + 1)
        l_end = min(n - 1, k)
        for l in range(l_start, l_end + 1):
            idx = k - l
            rows.append(k)
            cols.append(l)
            a_idxs.append(idx)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), np.array(a_idxs, dtype=np.int32)


def solve_lp_direction(a, lp_idx, max_time=5.0):
    """
    Solve LP: max sum(b) s.t. b>=0, conv(a,b) <= M.
    Returns (b_normalized, M) or (None, M) on failure.
    """
    n = len(a)
    aa = fast_convolve(a)
    M = float(aa.max())

    if M < 1e-30:
        return None, M

    rows, cols, a_idxs = lp_idx
    vals = a[a_idxs]

    # Filter out zero-value entries to speed up the LP
    mask = np.abs(vals) > 1e-15
    rows_f = rows[mask]
    cols_f = cols[mask]
    vals_f = vals[mask]

    if len(vals_f) == 0:
        return None, M

    A = csr_matrix((vals_f, (rows_f, cols_f)), shape=(2 * n - 1, n))

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    t0 = time.monotonic()
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': max_time, 'presolve': True})
        if res.success and res.x is not None:
            b_opt = np.maximum(res.x, 0.0)
            s = b_opt.sum()
            if s > 1e-15:
                return b_opt / s, M
    except Exception:
        pass
    return None, M


def lp_optimize(a_init, lp_idx, time_limit, verbose=True, label="LP"):
    """
    LP-based optimization with adaptive step size.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    iter_count = 0
    stuck_count = 0
    stuck_threshold = 150
    base_step = 0.15

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        b_dir, M = solve_lp_direction(a, lp_idx, max_time=3.0)

        if b_dir is None:
            if stuck_count > stuck_threshold:
                rng = np.random.default_rng(iter_count * 104729)
                noise = rng.standard_normal(n) * 0.01
                a = best_a + noise
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                stuck_count = 0
            continue

        # Adaptive step size: try decreasing steps
        found_better = False
        for t in [base_step, base_step * 0.5, base_step * 0.25, base_step * 0.1]:
            a_new = (1.0 - t) * a + t * b_dir
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            new_score = score_normalized(a_new)
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                found_better = True
                stuck_count = 0
                break
        if not found_better:
            stuck_count += 1
            if stuck_count > stuck_threshold:
                # More aggressive perturbation
                rng = np.random.default_rng(iter_count * 104729 + 7)
                noise = rng.standard_normal(n) * 0.03
                a = best_a + noise
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                stuck_count = 0
                # Occasionally try mixing with a random direction
                if iter_count % 5 == 0:
                    rand_dir = rng.dirichlet(np.ones(n))
                    a = 0.7 * a + 0.3 * rand_dir
                    a = a / a.sum()

        if verbose and iter_count % 30 == 0:
            elapsed = time.monotonic() - t0
            cur = score_normalized(a)
            print(f"    {label} iter={iter_count} ({elapsed:.0f}s): "
                  f"cur={cur:.7f} best={best_score:.7f} stuck={stuck_count}")

    return best_a, best_score


def compute_soft_max_weights(b, T):
    """Compute softmax weights over auto-correlation values."""
    scaled = b / T
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()
    return w


def mirror_descent_optimize(a_init, time_limit, T_start=0.01, T_end=0.0005,
                            eta_start=0.02, eta_end=0.0005, max_iters=50000,
                            verbose=True, label="MD"):
    """Adaptive mirror descent with temperature annealing."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stuck_count = 0

    def grad_a_from_w(a, w):
        n_a = len(a)
        fft_size = 1 << (3 * n_a - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        return 4.0 * n_a * correl[:n_a]

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (0.7 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        w = compute_soft_max_weights(b, T)
        grad_a = grad_a_from_w(a, w)

        log_update = -eta * grad_a
        log_update = np.clip(log_update, -3.0, 3.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        if stuck_count > 200:
            rng = np.random.default_rng(it * 7919)
            noise = rng.standard_normal(n) * 0.02
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck_count = 0

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} ({elapsed:.0f}s): "
                  f"T={T:.5f} score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def multiplicative_weights_exact(a_init, time_limit, eta_start=0.005,
                                 decay=0.9995, max_iters=50000, verbose=True, label="MW"):
    """Multiplicative weights using exact peak subgradient."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = eta_start

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()

        k_peak = int(np.argmax(b))

        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * a[k_peak - idx]

        grad = grad - np.dot(a, grad)

        log_update = -eta * grad
        log_update = np.clip(log_update, -2.0, 2.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        eta *= decay

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} ({elapsed:.0f}s): eta={eta:.6f} "
                  f"score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def coordinate_descent(a_init, time_limit, verbose=True, label="CD"):
    """Coordinate descent with proportional redistribution."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    factors = [0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5]
    consecutive_no_improve = 0
    max_no_improve = 300

    while time.monotonic() - t0 < time_limit and consecutive_no_improve < max_no_improve:
        consecutive_no_improve = 0
        improved_pass = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for f in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                new_val = old_val * f
                if new_val < 0:
                    continue
                a_trial[i] = new_val
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others_sum
                if a_trial.min() < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved_pass = True
                    break

        if not improved_pass:
            consecutive_no_improve += 1

    return best_a, best_score


def peak_shave(a_init, time_limit, verbose=True, label="PS"):
    """Targeted peak shaving: reduce contributions to the top peaks."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(20000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7, 0.5]:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * factor
                diff = old_val - a_trial[i]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            for ci in top_idx[:3]:
                i = int(idx[ci])
                j = peak_k - i
                if i == j or a[j] < 1e-14:
                    continue
                for f1 in [0.9, 0.8, 0.7]:
                    for f2 in [0.9, 0.8, 0.7]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using the actual evaluate_sequence function."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP-based direction finding,
    mirror descent, multiplicative weights, coordinate descent,
    peak shaving, and length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score

    # Build LP indices for the parent length
    if best_seq is not None:
        n_best = len(best_seq)
        lp_idx_best = build_lp_indices(n_best)
        print(f"LP indices built for n={n_best}")
    else:
        n_best = 768
        best_seq = np.ones(n_best) / n_best
        best_score = float(evaluate_sequence(best_seq.tolist()))
        lp_idx_best = build_lp_indices(n_best)
        print(f"No parent found, using uniform n={n_best}, score={best_score:.7f}")

    # === Phase 1: LP-based optimization (primary novel method) ===
    print(f"\n=== Phase 1: LP Direction Finding (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem * 0.5, 500.0)
        print(f"  LP: n={n_best}, budget={t_alloc:.0f}s")
        a_lp, _ = lp_optimize(best_seq, lp_idx_best, t_alloc, label="LP_main")
        best_seq, best_score = check_and_update(a_lp, "LP_main", best_seq, best_score)

    # === Phase 2: Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.7f}) ===")
    if best_seq is not None:
        n_cur = len(best_seq)
        lp_idx_cur = build_lp_indices(n_cur) if n_cur != n_best else lp_idx_best
        configs = [
            (0.02, 0.002, 0.03, 0.001, "MD1"),
            (0.01, 0.001, 0.02, 0.0005, "MD2"),
            (0.005, 0.0003, 0.01, 0.0002, "MD3"),
        ]
        for T_s, T_e, eta_s, eta_e, lbl in configs:
            rem = deadline - time.monotonic() - 400
            if rem < 30:
                break
            t_alloc = min(rem * 0.3, 150.0)
            print(f"  {lbl}: n={n_cur}, T=[{T_s},{T_e}], budget={t_alloc:.0f}s")
            a_md, _ = mirror_descent_optimize(
                best_seq, t_alloc, T_start=T_s, T_end=T_e,
                eta_start=eta_s, eta_end=eta_e, label=lbl
            )
            best_seq, best_score = check_and_update(a_md, lbl, best_seq, best_score)

    # === Phase 3: Multiplicative Weights ===
    print(f"\n=== Phase 3: Multiplicative Weights (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem * 0.4, 150.0)
            print(f"  MW: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_mw, _ = multiplicative_weights_exact(
                best_seq, t_alloc, eta_start=0.008, decay=0.9997, label="MW"
            )

    print(f"\n=== Phase 4: LP Round 2 (best={best_score:.7f}) ===")
    if best_seq is not None:

        rem = deadline - time.monotonic() - 250
        if rem > 40:

            t_alloc = min(rem * 0.5, 200.0)
            print(f"  LP2: n={n_cur}, budget={t_alloc:.0f}s")
            a_lp2, _ = lp_optimize(best_seq, lp_idx4, t_alloc, label="LP2")
            best_seq, best_score = check_and_update(a_lp2, "LP2", best_seq, best_score)

    # === Phase 5: Peak Shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem * 0.5, 100.0)
            print(f"  PS: budget={t_alloc:.0f}s")
            a_ps, _ = peak_shave(best_seq, t_alloc)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 6: Coordinate Descent ===
    print(f"\n=== Phase 6: Coord Descent (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem * 0.6, 80.0)
            print(f"  CD: budget={t_alloc:.0f}s")
            a_cd, _ = coordinate_descent(best_seq, t_alloc)
            best_seq, best_score = check_and_update(a_cd, "coord_descent", best_seq, best_score)

    # === Phase 7: Alternative Lengths ===
    print(f"\n=== Phase 7: Alt Lengths (best={best_score:.7f}) ===")
    if best_seq is not None:
        n_cur = len(best_seq)
        alt_lengths = []
        for factor in [1.25, 1.5, 0.8, 2.0, 0.667]:
            alt_n = int(n_cur * factor)
            if 128 <= alt_n <= 4096 and alt_n != n_cur and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:3]:

            # LP on new length
            t_lp = t_alloc * 0.5
            a_alt, _ = lp_optimize(init_alt, lp_idx_alt, t_lp, label=f"alt_n{alt_n}")

            # MD polish
            rem2 = min(deadline - time.monotonic() - 60, t_alloc * 0.5)
            if rem2 > 15:
                a_alt2, _ = mirror_descent_optimize(
                    a_alt, rem2, T_start=0.005, T_end=0.0005,
                    eta_start=0.01, eta_end=0.0005, label=f"md_n{alt_n}"
                )
                a_alt = a_alt2

            best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}", best_seq, best_score)

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.7f}) ===")
    if best_seq is not None:
        n_final = len(best_seq)
        lp_idx_final = build_lp_indices(n_final)

        # Short LP
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_alloc = min(rem * 0.4, 40.0)
            a_fin_lp, _ = lp_optimize(best_seq, lp_idx_final, t_alloc, label="LP_fin")
            best_seq, best_score = check_and_update(a_fin_lp, "LP_fin", best_seq, best_score)

        # Short peak shave
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_ps_fin, _ = peak_shave(best_seq, min(rem * 0.5, 20.0))
            best_seq, best_score = check_and_update(a_ps_fin, "PS_fin", best_seq, best_score)

        # Final coordinate descent
        rem = deadline - time.monotonic() - 8
        if rem > 8:
            a_cd_fin, _ = coordinate_descent(best_seq, min(rem * 0.6, 15.0))
            best_seq, best_score = check_and_update(a_cd_fin, "CD_fin", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
