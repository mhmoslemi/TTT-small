def fast_autoconv(a):
    """Fast auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute the objective score for a sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:

    b = fast_autoconv(a)
    return 2.0 * n * b.max() / (s * s)


def softmax_fn(z):
    """Stable softmax."""
    z_shift = z - z.max()
    e = np.exp(z_shift)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

    n = len(z)
    a = softmax_fn(z)
    b = fast_autoconv(a)
    Tb = T * b
    m = Tb.max()

    w = e / e.sum()

    # This is the cross-correlation of w and a at lag i
    # cross_corr[i] = sum_{j=0}^{n-1} a[j] * w[i+j]

    S = np.dot(grad_a, a)

Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines FFT-accelerated L-BFGS with geometric temperature annealing,
LP-based best-response, randomized restarts, and multi-length search.

import numpy as np
import time

import cvxpy as cp

# Global state for L-BFGS temperature (avoids closures/lambdas)
_GLOBAL_T = 100.0
_GLOBAL_N = 768


def fast_autoconv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (max(size, 2) - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = fast_autoconv(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_vec(z):
    z_shift = z - z.max()
    e = np.exp(z_shift)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def log_space(a):
    return np.log(np.maximum(a, 1e-300))


def _obj(z):
    T = _GLOBAL_T
    a = softmax_vec(z)
    b = fast_autoconv(a)
    Tb = T * b
    m = Tb.max()
    lse = m + np.log(np.exp(Tb - m).sum())
    return (2.0 * _GLOBAL_N / T) * lse


def _grad(z):
    T = _GLOBAL_T
    n = _GLOBAL_N
    a = softmax_vec(z)
    b = fast_autoconv(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()

    return a * (grad_a - S)

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_response(a_fixed):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    b_val /= s
                    return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_refine(a0, time_limit, n_max_iters=2000):
    """LP-based alternating refinement with Nesterov averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a /= a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_response(y)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y /= sy

        else:

            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated L-BFGS with geometric temperature annealing (T: 200->0.05),
LP-based Nesterov best-response refinement, fine peak shaving, randomized restarts,
and multi-length exploration to beat previous best of 1.504870.
"""

import numpy as np
import time
import math
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

# Global state for L-BFGS (avoids closures/lambdas)
_GLOBAL_T = 100.0
_GLOBAL_N = 768


def fast_autoconv(a):
    """Fast auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (max(size, 2) - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score = 2*n*max_b/(sum_a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = fast_autoconv(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_vec(z):
    """Stable softmax."""
    z_shift = z - z.max()
    e = np.exp(z_shift)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def log_space(a):
    """Probability vector to log-space."""
    return np.log(np.maximum(a, 1e-300))


def _softmax_obj(z):
    """Soft-max smoothed objective in log-space (uses globals)."""
    T = _GLOBAL_T
    n = _GLOBAL_N
    a = softmax_vec(z)
    b = fast_autoconv(a)
    Tb = T * b
    m = Tb.max()
    lse = m + np.log(np.exp(Tb - m).sum())
    return (2.0 * n / T) * lse


def _softmax_grad(z):
    """Gradient of soft-max objective w.r.t. log-space z. FFT-accelerated."""
    T = _GLOBAL_T
    n = _GLOBAL_N
    a = softmax_vec(z)
    b = fast_autoconv(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()

    # cross_corr[lag] = sum_{j=0}^{n-1} a[j] * w[lag+j]
    # = conv(a[::-1], w)[lag + n - 1]
    a_rev = a[::-1]
    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    AR = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv_result = irfft(AR * W, fft_size)[:conv_len]
    cross_corr = conv_result[n - 1:2 * n - 1]

    grad_a = (4.0 * n / T) * cross_corr
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbfgs_phase(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=10):
    """L-BFGS with geometric temperature annealing."""
    global _GLOBAL_T, _GLOBAL_N
    t0 = time.monotonic()
    _GLOBAL_N = len(a0)
    z = log_space(a0)
    temps = np.geomspace(T_start, T_end, n_stages)

    for i, T in enumerate(temps):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 3:
            break
        _GLOBAL_T = float(T)
        max_iter = max(50, int(remaining * 40))
        try:
            res = minimize(_softmax_obj, z, jac=_softmax_grad, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'ftol': 1e-18,
                                    'gtol': 1e-13, 'maxfun': max_iter * 300,
                                    'disp': False})
            z = res.x
        except Exception as ex:
            print(f"    L-BFGS stage {i} T={T:.3f} error: {ex}")
            break

    return softmax_vec(z)


def build_conv_matrix(a):
    """Build matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    Tv = np.zeros((2 * n - 1, n))
    for j in range(n):
        Tv[j:j + n, j] = a
    return Tv


def solve_lp_response(a_fixed):
    """LP: min M s.t. conv(a_fixed, b) <= M componentwise, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for solver in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    b_val /= s
                    return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_nesterov_refine(a0, time_limit, n_max_iters=3000):
    """Nesterov-averaged LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a /= a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_response(y)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y /= sy

        w_alpha = alpha * (it + 2)
        a = ((it + 1) * a + w_alpha * b_new) / (it + 1 + w_alpha)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a /= sa

        sc_a = compute_score(a)
        sc_y = compute_score(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            print(f"    LP iter={it+1} ({time.monotonic()-t0:.0f}s): best={best_score:.7f}")

    return best_a, best_score


def peak_shave_fine(a0, time_limit, n_steps=5000):
    """Fine-grained peak shaving with 1D and 2D moves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    a /= a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autoconv(a)
        b_inner = b.copy()
        b_inner[0] = -np.inf
        b_inner[-1] = -np.inf
        peak_k = int(np.argmax(b_inner))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top = np.argsort(contribs)[::-1][:10]

        improved = False
        # 1D moves
        for ci in top:
            i = int(idx[ci])
            if a[i] < 1e-14:
                continue
            j = peak_k - i
            if a[j] < 1e-14:
                continue
            for factor in [0.99, 0.97, 0.95, 0.92, 0.88, 0.82]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others.sum()
                a_trial /= a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # 2D moves
        if not improved and len(top) >= 2:
            for ci1 in top[:5]:
                for ci2 in top[:5]:
                    i1 = int(idx[ci1])
                    i2 = int(idx[ci2])
                    j1 = peak_k - i1
                    j2 = peak_k - i2
                    if a[i1] < 1e-14 or a[i2] < 1e-14:
                        continue
                    if i1 == i2 or (j1 == i2 and j2 == i1):
                        continue
                    for f1, f2 in [(0.98, 0.98), (0.96, 0.96),
                                   (0.94, 0.98), (0.98, 0.94), (0.95, 0.95)]:
                        a_trial = a.copy()
                        a_trial[i1] *= f1
                        a_trial[i2] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial /= a_trial.sum()
                        sc = compute_score(a_trial)
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            # Try reducing all top contributors slightly
            top3 = top[:4]
            a_trial = a.copy()
            for ci in top3:
                a_trial[int(idx[ci])] *= 0.97
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial /= s_t
                sc = compute_score(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                else:
                    break

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def random_perturb_seq(a, sigma, seed):
    """Add Gaussian noise and re-project to simplex."""
    rng = np.random.default_rng(seed)
    n = len(a)
    noise = rng.normal(0, sigma, n)
    a_pert = a + noise
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

    # Phase 6: Final peak shave + L-BFGS polish
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 15:
        t_f1 = min(remaining * 0.4, 40)
        a_new = lbfgs_phase(best_a, t_f1, T_start=1.0, T_end=0.01, n_stages=4)
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new

    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 10:
        a_ps, sc_ps = peak_shave_fine(best_a, min(remaining * 0.5, 30))
        if sc_ps < best_score:
            best_score = sc_ps
            best_a = a_ps

    return best_a, best_score


def try_alt_length(a_best, best_score, time_limit, alt_n):
    """Try optimizing at an alternative sequence length."""
    a_interp = interpolate_seq(a_best, alt_n)
    t0 = time.monotonic()
    a_new = lbfgs_phase(a_interp, min(time_limit * 0.5, 120),
                         T_start=50.0, T_end=0.1, n_stages=5)
    sc = compute_score(a_new)
    if sc < best_score:
        print(f"  Alt n={alt_n}: {sc:.7f}")
        # Continue with peak shave
        remaining = time_limit - (time.monotonic() - t0)
        if remaining > 10:
            a_ps, sc_ps = peak_shave_fine(a_new, min(remaining * 0.4, 40))
            if sc_ps < sc:
                a_new = a_ps
                sc = sc_ps
        return a_new, sc
    return a_best, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Search for the best sequence minimizing 2*n*max_b/(sum_a)^2.
    Multi-phase: FFT L-BFGS -> LP Nesterov -> low-T L-BFGS -> peak shave
    -> random restarts -> multi-length -> final polish.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    # Load parent
    best_a = None
    best_score = float('inf')
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_a = p / p.sum()
            best_score = compute_score(best_a)
            print(f"Parent: n={len(best_a)}, score={best_score:.7f}")
    except Exception:
        pass

    if best_a is None:
        best_a = np.ones(768) / 768
        best_score = compute_score(best_a)
        print(f"Default init: n=768, score={best_score:.7f}")

    n_best = len(best_a)
    print(f"Starting: n={n_best}, score={best_score:.7f}, budget={budget_s:.0f}s")

    # ===== Main optimization at best length =====
    remaining = deadline - time.monotonic()
    if remaining > 100:
        a_opt, sc_opt = multi_phase_optimize(best_a, min(remaining * 0.75, 1200))
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt
        print(f"After main optimization: score={best_score:.7f}, n={len(best_a)}")

    # ===== Try alternative lengths =====
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\nTrying alternative lengths (budget={remaining:.0f}s)")
        for alt_n in [int(n_best * 1.5), int(n_best * 0.75), int(n_best * 2)]:
            if alt_n < 256 or alt_n > 4096 or alt_n == n_best:
                continue
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            t_alloc = min(remaining * 0.3, 150)
            a_alt, sc_alt = try_alt_length(best_a, best_score, t_alloc, alt_n)
            if sc_alt < best_score:
                best_score = sc_alt
                best_a = a_alt
                n_best = len(best_a)
    print(f"After alt lengths: score={best_score:.7f}, n={len(best_a)}")

    # ===== Final refinement cycle =====
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\nFinal refinement cycle (budget={remaining:.0f}s)")
        a_final, sc_final = multi_phase_optimize(best_a, min(remaining * 0.8, remaining))
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final

    # ===== Verify with evaluate_sequence =====
    try:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_a]
        eval_score = evaluate_sequence(final_list)
        if np.isfinite(eval_score):
            print(f"Evaluated score: {eval_score:.7f}")
            if eval_score < best_score:
                best_score = eval_score
        else:
            print(f"Warning: evaluate_sequence returned {eval_score}")
    except Exception as ex:
        print(f"Verification error: {ex}")

    print(f"\nFINAL: score={best_score:.7f}, n={len(best_a)}")
    return [float(max(0.0, min(1000.0, x))) for x in best_a]
