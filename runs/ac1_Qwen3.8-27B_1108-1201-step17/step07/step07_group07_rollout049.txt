    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = float(fast_autoconv(a).max())
    no_improve = 0
    v = np.zeros(n) if use_momentum else None
    cur_lr = lr
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autoconv(a)
        b_max = b.max()

        # Select active peaks
        if n_peaks < len(b):
            # Get top n_peaks, but also include those within 0.5% of max
            top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
            threshold = b_max * 0.995
            all_candidates = np.where(b >= threshold)[0]
            active = np.unique(np.concatenate([top_idx, all_candidates]))
            active = active[:200]  # Cap at 200 active peaks
        else:
            active = np.where(b >= b_max * 0.995)[0]
            if len(active) < 1:
                active = np.array([np.argmax(b)])

        # Compute weights: proportional to peak heights (emphasize tallest)
        peak_vals = b[active]
        # Use exponential weighting to emphasize the tallest peaks
        w_max = peak_vals.max()
        weights = np.exp(3.0 * (peak_vals / w_max - 1.0))
        weights = weights / weights.sum()

        # Compute subgradient
        g = subgradient_from_peaks(a, active, weights)

        # Multiplicative update (mirror descent step)
        if use_momentum:
            v = momentum * v - cur_lr * g
            log_update = v
        else:
            log_update = -cur_lr * g

        # Clamp to prevent under/overflow
        log_update = np.clip(log_update, -10, 10)
        a_new = a * np.exp(log_update)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a
        else:
            a_new = a_new / s_new

        obj_new = float(fast_autoconv(a_new).max())

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Occasionally revert to best to prevent drift
            if no_improve % 100 == 0:
                a = best_a.copy()

        a = a_new
        iteration += 1

        # Adaptive learning rate
        if no_improve >= 2000:
            cur_lr *= 0.5
            no_improve = 0
            # Perturbation to escape
            a = a * (1.0 + rng.standard_normal(n) * 0.002)
            a = np.maximum(a, 0)
            a = a / a.sum()
        elif no_improve >= 1000:
            cur_lr *= 0.95

    return best_a, best_obj


def lp_refine_adaptive(f, max_iters=50, time_limit=120.0, verbose=False):
    """
    Adaptive LP refinement with:
    - Dual-guided peak selection
    - Adaptive M (trying to reduce M each iteration)
    - Golden section search for blend parameter
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = float(fast_autoconv(a).max())

    # Build constraint matrix once (it depends on a, which changes... but we rebuild periodically)
    def build_A(a_curr):
        """Build the convolution constraint matrix A where (A*g)[k] = (a*g)[k]."""
        num_rows = 2 * n - 1
        cols = a_curr.copy()  # length n
        A = np.zeros((num_rows, n))
        for k in range(num_rows):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[k, k - hi:k - lo + 1] = a_curr[lo:hi + 1][::-1]
        return A

    A = build_A(a)

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = float(fast_autoconv(a).max())

        # Try multiple M values to guide the LP
        # Start with current M, then try slightly relaxed versions
        m_values = [M, M * 1.001, M * 1.005, M * 1.01]
        
        best_g = None
        best_g_obj = best_obj

        for m_trial in m_values:
            if time.monotonic() - t0 > time_limit:
                break
            try:
                c = -np.ones(n)
                res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, m_trial),
                              bounds=[(0.0, None)] * n, method='hihogs' if False else 'highs',
                              options={'presolve': True})
                if not res.success or res.x is None:
                    continue
                g = res.x
                g_sum = g.sum()
                if g_sum < 1e-10:
                    continue

                g_norm = g / g_sum

                # Golden section search for blend parameter t
                best_t = 0.0
                best_blend_obj = best_obj
                # Try a range of t values first
                t_candidates = np.concatenate([
                    np.array([0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]),
                    np.array([0.005, 0.008, 0.03, 0.07, 0.12, 0.25, 0.35, 0.55, 0.65, 0.8])
                ])
                for t in sorted(t_candidates):
                    blend = (1.0 - t) * best_a + t * g_norm
                    blend = np.maximum(blend, 0.0)
                    ss = blend.sum()
                    if ss < 1e-10:
                        continue
                    blend = blend / ss
                    obj_blend = float(fast_autoconv(blend).max())
                    if obj_blend < best_blend_obj - 1e-15:
                        best_blend_obj = obj_blend
                        best_t = t

                if best_t > best_g_obj and best_g is None:
                    best_g = g_norm
                    best_g_obj = best_blend_obj
                    if best_t > 0:
                        new_a = (1.0 - best_t) * best_a + best_t * g_norm
                        new_a = np.maximum(new_a, 0.0)
                        new_a = new_a / new_a.sum()
                        new_obj = float(fast_autoconv(new_a).max())
                        if new_obj < best_obj:
                            best_obj = new_obj
                            best_a = new_a.copy()
                            a = new_a.copy()
                            # Rebuild A since a changed
                            A = build_A(a)
                            if verbose:
                                print(f"    LP iter {iteration}: M_trial={m_trial:.10f}, t={best_t:.3f}, obj={best_obj:.10f}")
                        else:
                            a = best_a.copy()
                    if best_obj < m_values[0] * (1 - 0.01):
                        break  # Good improvement, try next iteration

            except Exception:
                continue

        if best_g is None and not any(m > M for m in m_values):
            # No improvement found
            if iteration > 5:
                break

    return best_a, best_obj


def subgradient_momentum(a0, time_limit, lr_init=0.003, momentum=0.91,
                         n_top=30, seed=42, restart_every=1500):
    """
    Projected subgradient with Nesterov momentum and adaptive peak selection.
    Uses exponential peak weighting to focus on the tallest peaks.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = float(fast_autoconv(a).max())
    cur_lr = lr_init
    no_improve = 0
    kick_size = 0.003

    while time.monotonic() - t0 < time_limit:
        b = fast_autoconv(a)
        b_max = b.max()

        # Adaptive peak selection: top peaks + those within threshold
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))
        
        threshold = b_max * 0.998
        active = np.unique(np.concatenate([top_idx, np.where(b >= threshold)[0]]))
        active = active[:100]

        # Exponential weighting emphasizing tallest peaks
        peak_vals = b[active]
        w_scale = 5.0  # Higher = more focus on tallest
        weights = np.exp(w_scale * (peak_vals / peak_vals.max() - 1.0))
        weights = weights / weights.sum()

        # Compute subgradient
        g = subgradient_from_peaks(a, active, weights)

        # Nesterov step
        v = momentum * v - cur_lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        obj_new = float(fast_autoconv(a_new).max())

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve >= restart_every:
            a = best_a + rng.standard_normal(n) * kick_size
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.6, lr_init * 0.03)
            no_improve = 0
            kick_size *= 0.65
        elif no_improve >= restart_every // 2:
            cur_lr *= 0.95
        elif no_improve >= restart_every // 4:
            cur_lr *= 0.98

    return best_a, best_obj


def coordinate_descent_fast(a0, time_limit, multipliers=None):
    """
    Fast coordinate descent: for each coordinate, try a few scale factors.
    Uses FFT for evaluation.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = float(fast_autoconv(a).max())

    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0, 4.0])

    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0.01 or scale > 50.0:
                    continue

                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-12:
                    continue
                a_tr = a_tr / ss
                val = float(fast_autoconv(a_tr).max())
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True

    return best_a, best_obj


def spectral_init(n, seed=42):
    """
    Create initialization from a flat-spectrum design.
    In frequency domain, try to make |F| as flat as possible,
    then take inverse DFT and clip to non-negative.
    """
    rng = np.random.default_rng(seed)
    # Create a random phase in frequency domain
    fft_size = 1 << (2 * n - 1).bit_length()
    # Flat magnitude with random phase
    mag = np.ones(fft_size // 2 + 1)
    phase = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    # Make it symmetric for real inverse
    F = mag * np.exp(1j * phase)
    F[0] = mag[0]  # DC component
    if fft_size % 2 == 0:
        F[-1] = mag[-1]
    # Hermitian symmetry
    F_full = np.zeros(fft_size, dtype=complex)
    F_full[:fft_size//2+1] = F
    for i in range(1, fft_size//2 + 1):
        F_full[fft_size - i] = np.conj(F[i])

    f = irfft(F_full)[:n]
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return f / s


def diverse_init(n, seed=42, sigma=None):
    """Create diverse initializations."""
    rng = np.random.default_rng(seed)
    
    if sigma is not None:
        x = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    else:
        d = rng.choice(['gauss', 'cosine', 'tukey', 'triangle', 'poly', 'flat_with_holes', 'bimodal'])
        if d == 'gauss':
            sigma = rng.uniform(0.1, 0.5)
            x = np.linspace(0, 1, n)
            a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        elif d == 'cosine':
            x = np.linspace(0, 1, n)
            a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
        elif d == 'tukey':
            r = rng.uniform(0.2, 0.8)
            a = np.ones(n)
            taper = max(1, int(r * n // 2))
            if taper < n:
                t = np.linspace(0, 1, taper)
                a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
                a[-taper:] = a[:taper][::-1]
        elif d == 'triangle':
            half = n // 2
            a = np.concatenate([np.linspace(0.1, 1.0, half), np.linspace(1.0, 0.1, n - half)])
        elif d == 'poly':
            p = rng.uniform(0.5, 3.0)
            x = np.linspace(0, 1, n)
            a = x ** p * (1 - x) ** p
        elif d == 'flat_with_holes':
            a = np.ones(n) * 0.5
            n_holes = rng.integers(3, 10)
            for _ in range(n_holes):
                pos = rng.integers(0, n)
                width = rng.integers(2, max(3, n // 20))
                a[max(0, pos-width):min(n, pos+width)] *= 0.1
        elif d == 'bimodal':
            x = np.linspace(0, 1, n)
            s1 = rng.uniform(0.2, 0.4)
            s2 = rng.uniform(0.6, 0.8)
            sigma = rng.uniform(0.05, 0.15)
            a = np.exp(-0.5 * ((x - s1) / sigma) ** 2) + np.exp(-0.5 * ((x - s2) / sigma) ** 2)
    
    a = np.maximum(a, 1e-12)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def combined_optimize(a_init, time_budget, n, seed, verbose=False):
    """
    Combined optimization: alternating multiplicative descent, subgradient, and LP.
    """
    t0 = time.monotonic()
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = float(fast_autoconv(a).max())

    # Phase A: Multiplicative descent (good for early progress)
    t_a = min(time_budget * 0.3, 60.0)
    if t_a > 5:
        a_m, obj_m = multiplicative_descent(
            a, time_limit=t_a, lr=0.015, n_peaks=30,
            seed=seed, use_momentum=False
        )
        if obj_m < best_obj:
            best_obj = obj_m
            best_a = a_m.copy()
            if verbose:
                print(f"    MD: obj={best_obj:.10f}")

    # Phase B: Subgradient with momentum (good for mid-range)
    t_b = min(time_budget * 0.3, 80.0)
    rem = time_budget - (time.monotonic() - t0) - 10
    if t_b > 5 and rem > 10:
        t_b = min(t_b, rem)
        a_sg, obj_sg = subgradient_momentum(
            best_a, time_limit=t_b, lr_init=0.003, momentum=0.92,
            n_top=max(20, n // 15), seed=seed + 111
        )
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            if verbose:
                print(f"    SG: obj={best_obj:.10f}")

    # Phase C: LP refinement (good for fine tuning)
    rem = time_budget - (time.monotonic() - t0) - 5
    t_c = min(time_budget * 0.35, rem, 120.0)
    if t_c > 10:
        a_lp, obj_lp = lp_refine_adaptive(
            best_a, max_iters=min(40, int(t_c)), time_limit=t_c, verbose=verbose
        )
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            if verbose:
                print(f"    LP: obj={best_obj:.10f}")

    # Phase D: Final multiplicative polish
    rem = time_budget - (time.monotonic() - t0)
    if rem > 10:
        a_md2, obj_md2 = multiplicative_descent(
            best_a, time_limit=rem * 0.5, lr=0.003, n_peaks=50,
            seed=seed + 222, use_momentum=True, momentum=0.9
        )
        if obj_md2 < best_obj:
            best_obj = obj_md2
            best_a = a_md2.copy()

        # Phase E: Final LP
        rem = rem - (time.monotonic() - t0)
        # Actually rem is already computed, let me just check time
        rem2 = time_budget - (time.monotonic() - t0)
        if rem2 > 8:
            a_lp2, obj_lp2 = lp_refine_adaptive(
                best_a, max_iters=15, time_limit=min(rem2, 30.0)
            )
            if obj_lp2 < best_obj:
                best_obj = obj_lp2
                best_a = a_lp2.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Combines spectral init, multiplicative descent, subgradient, LP refinement,
    multi-length search, and coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)

    best_seq = None
    best_score = float('inf')

    def score_of(a):
        """Compute the actual evaluation score."""
        a_clip = np.clip(a, 0.0, 1000.0)
        s = a_clip.sum()
        if s < 0.01:
            return float('inf')
        n = len(a_clip)
        b = fast_autoconv(a_clip)
        return float(2.0 * n * b.max() / (s * s))

    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = score_of(a_norm)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = score_of(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_default = 1152
        parent = np.ones(n_default) / n_default
        best_seq = parent.copy()
        best_score = score_of(parent)
        print(f"No parent, using uniform n={n_default}, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Deep LP on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Deep LP + MD on parent (n={n_parent})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_alloc = min(rem * 0.4, 200.0)
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, _ = combined_optimize(parent, t_alloc, n_parent, seed, verbose=True)
        update_best(a_opt, "phase1_parent")

    # === Phase 2: Spectral + diverse initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Diverse Initializations (best={best_score:.6f})")
    print(f"{'='*60}")
    n_base = len(best_seq)

    # Generate diverse inits
    inits = []
    # Spectral inits
    for s in range(5):
        inits.append(('spectral_' + str(s), spectral_init(n_base, seed=1000 + s)))
    # Diverse shapes
    for s in range(8):
        inits.append(('diverse_' + str(s), diverse_init(n_base, seed=2000 + s)))
    # Specific shapes
    x = np.linspace(0, 1, n_base)
    inits.append(('gauss_0.2', np.maximum(np.exp(-0.5*((x-0.5)/0.2)**2), 1e-12)))
    inits.append(('gauss_0.3', np.maximum(np.exp(-0.5*((x-0.5)/0.3)**2), 1e-12)))
    inits.append(('cosine', 0.5*(1+np.cos(2*np.pi*x))))
    inits.append(('sqrt_cos', np.sqrt(np.maximum(0.5*(1+np.cos(2*np.pi*x)), 0))))
    inits.append(('bell_4', (x*(1-x))**2 + 0.01))

    for name, init_a in inits:
        init_a = np.maximum(init_a, 1e-12)
        init_a = init_a / init_a.sum()

    total_time_for_diverse = min(400.0, (deadline - time.monotonic() - 400.0) * 0.35)
    time_per_init = total_time_for_diverse / len(inits)

    for i, (name, init_a) in enumerate(inits):
        rem = deadline - time.monotonic() - 350
        if rem < 15:
            break
        t_alloc = min(time_per_init, rem * 0.25)
        if t_alloc < 8:
            continue
        print(f"  {name} (t={t_alloc:.0f}s)")
        a_div, _ = combined_optimize(init_a, t_alloc, n_base, seed + 5000 + i, verbose=False)
        update_best(a_div, f"div_{name}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-length Exploration (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        target_lengths = []
        for factor in [0.5, 0.75, 1.25, 1.5, 2.0, 0.6, 1.33, 0.8, 1.1]:
            tl = int(n_parent * factor)
            tl = max(128, min(4096, tl))
            # Round to power of 2-ish for FFT efficiency
            tl = 1 << (tl - 1).bit_length()
            if tl not in target_lengths:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_parent))

        for target_n in sorted(set(target_lengths))[:6]:
            rem = deadline - time.monotonic() - 250
            if rem < 25:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.2, 80.0)
            if t_alloc < 10:
                continue
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_len, _ = combined_optimize(f_interp, t_alloc, target_n, seed + target_n, verbose=False)
            update_best(a_len, f"len_n{target_n}")

    # === Phase 4: Deep optimization on best ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Deep Optimization (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 120
        if rem > 30:
            print(f"  Deep combined: n={n_best}, t={min(rem, 200.0):.0f}s")
            a_deep, _ = combined_optimize(best_seq, min(rem, 200.0), n_best, seed + 99999, verbose=True)
            update_best(a_deep, "deep")

    # === Phase 5: Aggressive LP on best ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Aggressive LP (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_lp = min(rem, 80.0)
            print(f"  LP: n={len(best_seq)}, t={t_lp:.0f}s")
            a_lp, obj_lp = lp_refine_adaptive(best_seq, max_iters=30, time_limit=t_lp, verbose=True)
            update_best(a_lp, "aggressive_lp")

    # === Phase 6: Coordinate descent polish ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Coordinate Descent (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_cd = min(rem, 60.0)
            print(f"  CD: n={len(best_seq)}, t={t_cd:.0f}s")
            a_cd, _ = coordinate_descent_fast(best_seq, time_limit=t_cd)
            update_best(a_cd, "cd")

    # === Phase 7: Final micro-polish ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Final Micro-polish (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        n_final = len(best_seq)
        rem = deadline - time.monotonic() - 15
        if rem > 8:
            # Multiplicative with very small lr
            a_micro, obj_micro = multiplicative_descent(
                best_seq, time_limit=min(rem * 0.4, 20.0),
                lr=0.001, n_peaks=50, seed=seed + 777777,
                use_momentum=True, momentum=0.93
            )
            update_best(a_micro, "micro_md")

            rem = deadline - time.monotonic() - 10
            if rem > 5:
                a_micro_cd, _ = coordinate_descent_fast(
                    best_seq, time_limit=min(rem, 15.0),
                    multipliers=np.array([0.8, 0.9, 0.95, 1.05, 1.1, 1.2])
                )
                update_best(a_micro_cd, "micro_cd")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = score_of(best_seq)

    # Final verification using the actual evaluate_sequence
    fs = float(evaluate_sequence([float(x) for x in best_seq]))
    if np.isfinite(fs):
        if fs < best_score:
            best_score = fs
        print(f"\nFinal score (evaluated): {fs:.8f}")
    else:
        print("WARNING: evaluation returned inf, falling back to parent")
        best_seq = parent.copy()
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))

    print(f"\n{'='*60}")
    print(f"FINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
