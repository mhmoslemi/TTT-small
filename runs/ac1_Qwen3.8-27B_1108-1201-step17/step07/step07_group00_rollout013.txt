Improved optimization for minimizing 2*n*max(auto_conv)/sum^2.

Key improvements over previous approach:

2. Active-set BFGS phase exploiting quadratic structure of selected peaks

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lp_matrix(f, n, constraint_indices):
    """Build the K x n constraint matrix for (f*g)[k] <= M."""
    K = len(constraint_indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = constraint_indices[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp(f, n, constraint_indices, M, warm_start=None):
    """Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in constraint_indices."""
    A = compute_lp_matrix(f, n, constraint_indices)
    g = cp.Variable(n)
    if warm_start is not None:
        try:
            g.value = warm_start
        except Exception:
            pass
    objective = cp.Maximize(cp.sum(g))
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=1e-7)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=300000, abort_rule=1)
        except Exception:
            return None, None
    if g.value is None:
        return None, None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, None
    return g_val, prob.value

                break
            alpha *= rho

        if best_f_trial is not None:
            f = best_f_trial
            cur_obj = best_obj_trial
            if cur_obj < best_obj * (1 - 1e-12):
                best_obj = cur_obj
                best_f = f.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            f = (1.0 - alpha) * f + alpha * h
            f = np.maximum(f, 0.0)
            s = f.sum()
            if s > 1e-10:
                f /= s
            stuck += 1

        if stuck > 30:
            # Perturbation to escape
            f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
            stuck = 0
            warm_g = None

        if it % 40 == 0:
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} K={len(constraint_indices)} "
                  f"thr={threshold_frac:.2f} t={elapsed:.0f}s")

    return best_f, best_obj


def gradient_active_set(a, active_indices, n):
    """
    Compute gradient of (1/|S|) * sum_{k in S} conv(a,a)[k] w.r.t. a.
    Using: grad = (2/|S|) * sum_{k in S} R_k a, where (R_k a)[j] = a[k-j] if valid.
    """
    S = len(active_indices)
    grad = np.zeros(n)
    for k in active_indices:
        # (R_k a)[j] = a[k-j] if 0 <= k-j <= n-1
        # So for j in [max(0, k-n+1), min(n-1, k)]: grad[j] += a[k-j]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += a[hi - lo + (k - hi):hi - lo + 1] if False else a[k - hi:k - lo + 1][::-1]
    grad *= 2.0 / S
    return grad


def gradient_active_set_v2(a, active_indices, n):
    """Vectorized gradient computation for active set objective."""
    S = len(active_indices)
    grad = np.zeros(n)
    for k in active_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # (R_k a)[j] = a[k-j] for j in [lo, hi]
            # a[k-j] for j=lo,lo+1,...,hi gives a[k-lo], a[k-lo-1], ..., a[k-hi]
            # which is a[k-hi : k-lo+1][::-1]
            grad[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    grad *= 2.0 / S
    return grad


def run_bfgs_phase(f_init, time_limit, label="bfgs"):
    """
    Active-set smoothing: minimize average of top-K convolution peaks using BFGS.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    K = max(20, n // 10)
    no_improve = 0
    it = 0
    lr = 0.001
    momentum = 0.9
    v = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        b = fast_convolve(f)

        # Update active set every 20 iterations
        if it % 20 == 1:
            active = np.argpartition(b, -K)[-K:]
            active = np.sort(active)

        # Compute gradient
        grad = gradient_active_set_v2(f, active, n)

        # Momentum update
        v = momentum * v - lr * grad
        f_new = f + v
        f_new = project_simplex(f_new)

        cur_obj = fast_convolve(f_new).max()
        f = f_new

        if it % 10 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj * (1 - 1e-10):
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.03, 0.005)
            else:
                no_improve += 1
                if no_improve > 10:
                    lr *= 0.9
                    no_improve = 0
                    if lr < 1e-5:
                        f = project_simplex(best_f)
                        v = np.zeros(n)
                        lr = 0.002
                        no_improve = 0

            if it % 100 == 0:
                print(f"    {label} iter={it} obj={best_obj:.10f} "
                      f"score={2.0*n*best_obj:.6f} lr={lr:.6f} t={elapsed:.0f}s")

            if no_improve > 40:
                break

    return best_f, best_obj


def coordinate_polish(f_init, time_limit, label="cd"):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    multipliers = [0.3, 0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.3, 1.5, 2.0]

    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        any_improved = False
        rng = np.random.default_rng(passes * 1000)
        # Randomize order each pass
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            if f[j] < 1e-10:
                # Try setting it to a small positive value
                for trial_val in [0.001 / n, 0.005 / n, 0.01 / n]:
                    f_trial = f.copy()
                    f_trial[j] = trial_val
                    s = f_trial.sum()
                    f_trial /= s
                    obj = fast_convolve(f_trial).max()
                    if obj < best_obj * (1 - 1e-12):
                        f = f_trial
                        best_obj = obj
                        best_f = f.copy()
                        any_improved = True
                        break
                continue

            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj * (1 - 1e-12):
                    f = f_trial
                    best_obj = obj
                    best_f = f.copy()
                    any_improved = True
                    break

        if not any_improved:
            # Try random coordinate moves
            for _ in range(50):
                if time.monotonic() - t0 > time_limit:
                    return best_f, best_obj
                j = rng.integers(0, n)
                f_trial = f.copy()
                f_trial[j] = max(0, f[j] * rng.uniform(0.3, 3.0))
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj * (1 - 1e-12):
                    f = f_trial
                    best_obj = obj
                    best_f = f.copy()

    return best_f, best_obj


def make_symmetric(a):
    """Make a sequence symmetric: a[i] = a[n-1-i]."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s > 1e-10:
        a_sym /= s
    return a_sym


def load_parent():
    """Load and normalize parent sequence."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p / p.sum()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization combining LP, active-set gradient,
    and coordinate descent with multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    parent = load_parent()

    if parent is not None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={best_score:.8f}")
    else:
        best_seq = np.ones(1024) / 1024.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"No parent, uniform n=1024, score={best_score:.8f}")

    n_base = len(best_seq)

    # ============================================================
    # Phase 1: LP optimization on base length (50% of budget)
    # ============================================================
    print(f"\n=== Phase 1: LP main (n={n_base}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.50, 900.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = run_lp_phase(best_seq, time_limit=t1, label="lp_main")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [lp_main]: {score1:.8f}")

    # ============================================================
    # Phase 2: Active-set gradient (20% of budget)
    # ============================================================
    print(f"\n=== Phase 2: Active-set grad (n={len(best_seq)}, score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 20:
        f2, obj2 = run_bfgs_phase(best_seq, time_limit=t2, label="bfgs")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [bfgs]: {score2:.8f}")

    # ============================================================
    # Phase 3: Coordinate descent polish (10% of budget)
    # ============================================================
    print(f"\n=== Phase 3: Coord descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.10, 120.0)
    if t3 > 15:
        f3, obj3 = coordinate_polish(best_seq, time_limit=t3, label="cd")
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [cd]: {score3:.8f}")

    # ============================================================
    # Phase 4: Shorter LP burst (10%)
    # ============================================================
    print(f"\n=== Phase 4: LP burst (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 120.0)
    if t4 > 15:
        f4, obj4 = run_lp_phase(best_seq, time_limit=t4, label="lp_burst")
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [lp_burst]: {score4:.8f}")

    # ============================================================
    # Phase 5: Multi-length exploration (10%)
    # ============================================================
    print(f"\n=== Phase 5: Multi-length (score={best_score:.8f}) ===")
    n_cur = len(best_seq)
    candidate_lengths = []
    for factor in [0.5, 0.75, 1.25, 1.5, 2.0]:
        tl = int(round(n_cur * factor))
        tl = max(256, min(4000, tl))
        if tl != n_cur and tl not in candidate_lengths:
            candidate_lengths.append(tl)
    candidate_lengths.sort(key=lambda x: abs(x - n_cur))

    for target_n in candidate_lengths[:2]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t5 = min(rem * 0.5, 120.0)
        if t5 < 20:
            break

        # Interpolate to new length
        old_x = np.linspace(0.0, 1.0, n_cur)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  Trying n={target_n}, budget={t5:.0f}s")
        f5, obj5 = run_lp_phase(f_interp, time_limit=t5, label=f"lp_n{target_n}")
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [n{target_n}]: {score5:.8f}")

    # ============================================================
    # Phase 6: Final coordinate polish on best
    # ============================================================
    rem = deadline - time.monotonic()
    if rem > 20:
        t6 = min(rem * 0.7, 60.0)
        print(f"\n=== Phase 6: Final CD polish (score={best_score:.8f}) ===")
        f6, obj6 = coordinate_polish(best_seq, time_limit=t6, label="cd_final")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [cd_final]: {score6:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent if parent is not None else np.ones(n_base) / n_base
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*55}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*55}")

    return [float(x) for x in best_seq]
