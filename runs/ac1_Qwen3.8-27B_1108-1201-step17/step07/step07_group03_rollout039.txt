import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_autoconv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute log-sum-exp smoothed objective and its gradient."""
    n = len(a)
    b = fast_autoconv(a)
    # Log-sum-exp for numerical stability

"""
Optimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. LP-based fictitious play for global convergence (main workhorse)
2. Nesterov-accelerated projected subgradient for fast local refinement

4. Multi-length exploration with interpolation
5. Coordinate descent polishing

import numpy as np
from scipy.fft import rfft, irfft
import scipy.sparse as sp
import time


def fast_autoconv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _build_conv_matrix_sparse(a):
    """Build sparse matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = j + n
        k_range = np.arange(start, end)
        idx = k_range - j
        rows.extend(k_range.tolist())
        cols.extend([j] * n)
        vals.extend(a[idx].tolist())
    return sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _build_conv_matrix_csr(a):
    """Build CSR matrix T such that T @ b = conv(a, b) - more efficient for LP."""
    n = len(a)
    # Each column j has nonzeros at rows j..j+n-1 with values a[0..n-1]
    indptr = np.arange(0, (n + 1) * n + 1, n, dtype=np.int32)
    indices = np.tile(np.arange(0, n, dtype=np.int32), n)
    data = np.tile(a, n)
    # Fix: column j has entries at rows j, j+1, ..., j+n-1
    # In CSR format, we need row-sorted
    # Let's use the indirect approach
    rows_all = []
    cols_all = []
    data_all = []
    for j in range(n):
        rows_all.extend(range(j, j + n))
        cols_all.extend([j] * n)
        data_all.extend(a.tolist())
    T = sp.csr_matrix((data_all, (rows_all, cols_all)), shape=(2 * n - 1, n))
    return T


def _solve_lp_response(a_fixed, T_precomputed=None):
    """Given fixed a (sum=1, a>=0), find b (sum=1, b>=0) minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    n = len(a_fixed)
    if T_precomputed is None:
        T_precomputed = _build_conv_matrix_csr(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_precomputed @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    last_err = None
    for solver in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver, **({'max_iter': 5000} if solver == cp.ECOS else {}))
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    b_val /= s
                else:
                    return None, M_var.value
                return b_val, float(M_var.value)
        except Exception as e:
            last_err = e
            continue
    return None, None


def lp_fictitious_play(n, time_limit, initial, verbose=False):
    """LP-based fictitious play: alternating LP best responses with averaging."""
    t0 = time.monotonic()
    a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    b = a.copy()
    avg_a = a.copy()
    avg_b = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autoconv(a).max()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        # LP best response
        b_new, _ = _solve_lp_response(a)
        if b_new is not None:
            b = b_new
        a_new, _ = _solve_lp_response(b)
        if a_new is not None:
            a = a_new

        avg_a = (iters * avg_a + a) / (iters + 1)
        avg_b = (iters * avg_b + b) / (iters + 1)

        # Evaluate candidates
        for cand in [a, b, 0.5 * (a + b), avg_a, avg_b, 0.7 * avg_a + 0.3 * a]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_n = cand / sc
            score = 2.0 * n * fast_autoconv(cand_n).max()
            if score < best_score:
                best_score = score
                best_a = cand_n.copy()

        if verbose and iters % 200 == 0:
            print(f"    LP-FP n={n} it={iters} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def nesterov_subgrad(a0, time_limit, top_k=30, lr_init=0.01, verbose=False):
    """Nesterov-accelerated projected subgradient on max(conv(a,a))."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autoconv(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lr = lr_init
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_k = (1 + np.sqrt(1 + 4 * (it - 1) ** 2)) / 2
        beta_k = (t_k - 1) / t_k

        # Compute subgradient at y
        b_y = fast_autoconv(y)
        sorted_idx = np.argsort(b_y)[::-1][:top_k]
        peak_vals = b_y[sorted_idx]
        # Softmax weights
        temp = 0.01 * max(peak_vals.max() - peak_vals.min(), 1e-10)
        w = np.exp((peak_vals - peak_vals.max()) / temp)
        w /= w.sum()

        grad = np.zeros(n)
        for idx in range(len(sorted_idx)):
            if w[idx] < 1e-14:
                continue
            k = int(sorted_idx[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            idx_arr = k - np.arange(i_start, i_end + 1)
            grad[i_start:i_end + 1] += 2.0 * w[idx] * y[idx_arr]

        # Momentum
        v = beta_k * v + grad
        v_norm = np.linalg.norm(v)
        if v_norm > 1e-15:
            v = v / v_norm

        # Step
        y_new = y - lr * v
        y_projected = simplex_project(y_new)

        score = 2.0 * n * fast_autoconv(y_projected).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = y_projected.copy()
            no_improve = 0
        else:
            no_improve += 1

        y = y_projected

        # Adapt learning rate
        if no_improve > 100:
            lr *= 0.9
            no_improve = 0
        if no_improve > 400:
            # Restart from best with noise
            noise = rng.standard_normal(n) * 0.005
            y = simplex_project(best_a + noise)
            v = np.zeros(n)
            lr = max(0.001, lr_init * 0.7)
            no_improve = 0

        lr *= 0.9998

        if verbose and it % 500 == 0:
            print(f"    Nesterov it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.6f} lr={lr:.5f}")

    return best_a, best_score

            break

        b = fast_autoconv(a)
        # Find top peaks
        peaks = np.argsort(b)[::-1][:n_peaks]
        
        # For each peak, find the contributing pairs and try to reduce
        improved = False
        for pk in peaks:
            k = int(pk)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            idx_arr = np.arange(i_start, i_end + 1)
            contribs = a[idx_arr] * a[k - idx_arr]
            top_contrib = np.argsort(contribs)[::-1][:4]
            
            for ci in top_contrib:
                i = int(idx_arr[ci])
                j = k - i
                old_val = a[i]
                if old_val < 1e-14:

                for factor in [0.85, 0.7, 0.55]:

                    if s_t < 1e-15:

                    a_trial = a_trial / s_t
                    score = 2.0 * n * fast_autoconv(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True

                if not improved and i != j and a[j] >= 1e-14:

                        a_trial[i] = a[i] * 0.9

                        if s_t < 1e-15:

                        score = 2.0 * n * fast_autoconv(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                if improved:
                    break

        if not improved:
            # Try LP-based direction: find b that minimizes max_k conv(a, b)_k
            # then move towards it
            T_mat = _build_conv_matrix_csr(a)
            b_dir, _ = _solve_lp_response(a, T_mat)

                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    score = 2.0 * n * fast_autoconv(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
            if not improved:

    return best_a, best_score


def coord_descent(a0, time_limit, n_sweeps=100):
    """Coordinate descent: scale individual coordinates to reduce peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autoconv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.7, 0.85, 0.92, 1.08, 1.15, 1.4]:
                a[i] = old_val * factor
                a_pos = np.maximum(a, 0)
                s_t = a_pos.sum()
                if s_t < 1e-15:
                    a[i] = old_val
                    continue
                a_trial = a_pos / s_t
                score = 2.0 * n * fast_autoconv(a_trial).max()
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a_trial.copy()
                    a = a_trial
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()
        if not improved and sweep > 5:
            break

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng):
    """Initialize with random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.3
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = rng.choice(np.arange(1, 30), size=n_harmonics, replace=False)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def _check_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
