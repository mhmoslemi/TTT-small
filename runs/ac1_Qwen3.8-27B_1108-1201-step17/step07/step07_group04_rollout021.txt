import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring PDLP for speed."""
    if hasattr(get_lp_solver, '_result'):
        return get_lp_solver._result
    for solver_name in [cp.PDLP, cp.GUROBI, cp.MOSEK, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                get_lp_solver._result = solver_name
                return solver_name
        except Exception:
            continue
    get_lp_solver._result = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b (non-negative, sum=1) minimizing max_k conv(a,b)[k]."""

    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)

        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fictitious_play(n, time_limit, initial, solver, verbose=True):
    """Run fictitious play for auto-correlation minimization."""
    t0 = monotonic()
    if initial is None:
        a = np.ones(n) / n
        t_ar = np.linspace(0, 2 * np.pi, n)
        a = a + 0.005 * np.cos(3 * t_ar)
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    iters = 0
    stagnation = 0

    while monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new

        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        candidates = [a, b, 0.5 * (a + b)]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand_norm = cand / s
            max_conv = fast_convolve(cand_norm).max()
            score = 2.0 * n * max_conv
            if score < best_score - 1e-12:
                best_score = score
                best_a = cand_norm.copy()
                stagnation = 0
            else:
                stagnation += 1

        if stagnation > 150:
            noise_scale = 0.003 * max(1.0, 1.0 / (1.0 + 0.01 * iters))
            a = a + np.random.standard_normal(n) * noise_scale
            a = np.maximum(a, 0)
            a = a / a.sum()
            b = b + np.random.standard_normal(n) * noise_scale
            b = np.maximum(b, 0)
            b = b / b.sum()
            stagnation = 0

        if verbose and iters % 20 == 0:
            elapsed = monotonic() - t0
            ma_str = f"{M_a:.6f}" if M_a is not None else "N/A"
            mb_str = f"{M_b:.6f}" if M_b is not None else "N/A"
            print(f"  FP n={n} iter={iters} ({elapsed:.0f}s): M_a={ma_str} M_b={mb_str} best={best_score:.6f}")

    return best_a, best_score, iters


def linearized_peak_shave(a0, time_limit, solver, verbose=True):
    """
    Iterative linearized peak shaving via LP.
    At each step, linearize the auto-convolution around current point,
    and solve an LP to find a small perturbation that reduces the peak.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    reduction = 0.008
    no_improve = 0
    step_count = 0

    while monotonic() - t0 < time_limit:
        current_conv = fast_convolve(a)
        M = current_conv.max()
        M_target = M * (1.0 - reduction)
        T = build_conv_matrix(a)

        delta = cp.Variable(n)
        constraints = [
            delta >= -a,
            cp.sum(delta) == 0,
            current_conv + 2.0 * (T @ delta) <= M_target,
        ]
        prob = cp.Problem(cp.Minimize(cp.norm1(delta)), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            reduction = max(0.001, reduction * 0.5)
            no_improve += 1
            continue

        if prob.status in ('optimal', 'optimal_inaccurate'):
            delta_val = np.asarray(delta.value, dtype=np.float64)
            a_candidate = np.maximum(a + delta_val, 0)
            cs = a_candidate.sum()
            if cs > 1e-10:
                a_candidate = a_candidate / cs
                score = 2.0 * n * fast_convolve(a_candidate).max()
                step_count += 1
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a_candidate.copy()
                    a = a_candidate
                    no_improve = 0
                    reduction = min(0.03, reduction * 1.05)
                else:
                    no_improve += 1
                    reduction = max(0.001, reduction * 0.6)
        else:
            reduction = max(0.001, reduction * 0.5)
            no_improve += 1

        if no_improve > 80:
            # Try a larger random perturbation to escape
            a = best_a + np.random.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            reduction = 0.005
            no_improve = 0

        if verbose and step_count % 20 == 0 and step_count > 0:
            elapsed = monotonic() - t0
            print(f"    LinShave {elapsed:.0f}s: M={M:.8f} red={reduction:.5f} best={best_score:.6f} steps={step_count}")

    return best_a, best_score


def damped_lp_refine(a0, time_limit, solver):
    """Damped alternating LP refinement with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()

    for it in range(10000):
        if monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        alpha = min(1.0, 2.0 / (it + 2))
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def lbgfs_smooth(a0, time_limit, T=80, maxiter=5000):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse / T

    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()

        # Gradient: 2*n*sum_k w[k] * (conv(a,a))[k] grad wrt a
        # (conv(a,a))[k] = sum_j a[j]*a[k-j], gradient wrt a[i] = sum_k w[k]*(a[k-i] + a[i-k] adjusted)
        # More precisely: d/di [sum_j a[j]*a[k-j]] = a[k-i] if 0<=k-i<n, plus a[i-(k-n+1+...)] 
        # Actually: d/di(a*a)[k] = a[k-i] (if valid) + a[i-k+n-1] ... 
        # Let me use the direct formula: (conv(a,a))[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j]*a[k-j]
        # d/di = a[k-i] if 0<=k-i<n else 0, plus a[i-k] if 0<=i-k<n else 0... 
        # Actually it's symmetric: d/di [(a*a)[k]] = a[k-i] (if 0<=k-i<n) + a[i-(k-n+1)] ... 
        # Let me just do it directly:
        # d/di sum_j a[j]*a[k-j] = sum_j (delta(j,i)*a[k-j] + a[j]*delta(k-j,i))
        # = a[k-i] (if 0<=k-i<n) + a[i-k+n-1]... 
        # Wait: if k-j = i then j = k-i, so a[k-i] if 0<=k-i<n
        # And if j = i, then a[k-i]... no wait.
        # d/di [a[j]*a[k-j]] = delta(j,i)*a[k-j] + a[j]*delta(k-j,i)
        # Sum over j: a[k-i] (from first term, j=i) + a[i] ... no.
        # First term: j=i gives a[k-i] (if k-i in [0,n-1])
        # Second term: k-j=i gives j=k-i, gives a[k-i] (if k-i in [0,n-1])... wait no, gives a[j]=a[k-i]
        # So total: 2*a[k-i] if 0<=k-i<n, else a[k-i] if only one is valid... 
        # Actually for auto-convolution, it's symmetric so it's 2*a[k-i] when both indices are valid.
        # But at the edges it might be just one. Let me handle it carefully.
        
        grad_a = np.zeros(n)
        # For each k, (a*a)[k] = sum_j a[j]*a[k-j] where j ranges over valid indices
        # d/di = sum_j [delta(j,i)*a[k-j] + a[j]*delta(k-j,i)]
        # = a[k-i] (if 0<=k-i<n, from first term with j=i) 
        #   + a[k-i] (if 0<=k-i<n, from second term with j=k-i)  [wait, second term: k-j=i means j=k-i, so a[j]=a[k-i]]
        # So d/di (a*a)[k] = 2*a[k-i] (if 0<=k-i<n) ... but this counts the same product twice when i=k-i, i.e., k=2i.
        # Hmm, actually let me reconsider. (a*a)[k] = sum_j a[j]*a[k-j].
        # If we set j=i: contributes a[i]*a[k-i]
        # If we set k-j=i, i.e., j=k-i: contributes a[k-i]*a[i] = same thing
        # But these are the same term! So d/di of sum_j a[j]*a[k-j] = a[k-i] + a[k-i] = 2*a[k-i] only if i != k-i (i.e., 2i != k).
        # If 2i == k, then there's only one term a[i]*a[i] = a[i]^2, and d/di = 2*a[i] = 2*a[k-i].
        # So in all cases: d/di (a*a)[k] = 2*a[k-i] if 0 <= k-i < n, else 0.
        # Wait that's not right either. Let me be more careful.
        # (a*a)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
        # d/di = sum_{j: 0<=j<n, 0<=k-j<n} [delta(j,i)*a[k-j] + a[j]*delta(k-j,i)]
        # = a[k-i] (if 0<=i<n and 0<=k-i<n) + a[k-i] (if 0<=k-i<n and 0<=i<n)
        # = 2*a[k-i] (if 0<=k-i<n)
        # Hmm, but when i = k-i (i.e., i = k/2), the two terms are the same product a[i]*a[i].
        # d/di [a[i]*a[i]] = 2*a[i] (by product rule). And 2*a[k-i] = 2*a[i]. So it's consistent!
        # Great, so d/di (a*a)[k] = 2*a[k-i] if 0<=k-i<n, else 0.
        
        # So: grad_a[i] = 2 * sum_k w[k] * a[k-i] (if 0<=k-i<n)
        # = 2 * sum_k w[k] * a[k-i] * (0<=k-i<n)
        # This is 2 * (w_reversed * a)[i] ... let me think.
        # sum_k w[k]*a[k-i] where k-i >= 0 means k >= i, and k-i < n means k < i+n.
        # Since k ranges from 0 to 2n-2, and i from 0 to n-1:
        # sum_{k=max(0,i)}^{min(2n-2, i+n-1)} w[k]*a[k-i]
        # Let m = k-i, then m ranges from max(0,0) to min(2n-2-i, n-1) = 0 to n-1
        # So it's sum_{m=0}^{n-1} w[i+m]*a[m] (if i+m <= 2n-2)
        # = (w * a)[i] in the correlation sense
        
        # Use FFT for efficiency
        # corr[i] = sum_m w[i+m] * a[m] for m=0..n-1
        # This is the cross-correlation of w and a
        # = sum_m a[m] * conj(w[i+m])
        # In FFT terms: if we pad w to length 2n and a to length 2n:
        # FFT(w)*conj(FFT(a)) gives the circular cross-correlation
        # We need the linear cross-correlation at lags 0 to n-1
        
        w_full = np.zeros(2 * n - 1)
        w_full[:len(w)] = w
        # cross-correlation: r[i] = sum_m a[m]*w[i+m] = (a * w_flipped)[i + n - 1]... 
        # Actually let me just use direct computation for clarity, or a simpler FFT trick.
        # r[i] = sum_{m=0}^{n-1} a[m] * w[i+m]
        # This is the convolution of a with w, evaluated at position (n-1) + i... 
        # conv(a,w)[k] = sum_j a[j]*w[k-j]
        # r[i] = sum_m a[m]*w[i+m] ... this is not a standard convolution.
        # r[i] = sum_m a[m]*w[m+i] = cross-correlation
        
        # Let's use: flip w, convolve with a
        w_flip = w_full[::-1]
        ft_size = 1 << (4 * n - 2).bit_length()
        A_ft = rfft(a, ft_size)
        W_ft = rfft(w_flip, ft_size)
        conv_result = irfft(A_ft * W_ft, ft_size)[:2 * n - 1]
        # conv(a, w_flip)[k] = sum_j a[j] * w_flip[k-j] = sum_j a[j] * w_full[2n-2-(k-j)]
        # = sum_j a[j] * w_full[2n-2-k+j]
        # We want r[i] = sum_m a[m]*w_full[m+i]
        # Set 2n-2-k+j = m+i, and j=m: 2n-2-k+m = m+i => k = 2n-2-i
        # So r[i] = conv(a, w_flip)[2n-2-i]
        
        r = conv_result[2 * n - 2 - np.arange(n)]
        grad_a = 2.0 * 2.0 * n * r  # factor 2 from derivative, factor 2n from objective
        
        # Transform to z-space: a = softmax(z), so da_i/dz_i = a_i(1-a_i), da_i/dz_j = -a_i*a_j
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0
    except Exception:
        a_out = a0

    return a_out


def subgradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.92, top_k=50):
    """Subgradient method with momentum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    lr = lr_init
    t0 = monotonic()
    no_improve = 0
    step_count = 0

    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        temp = 0.001 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for ii in range(len(peaks)):
            if w[ii] < 1e-12:
                continue
            k = int(peaks[ii])
            m = k  # d/di (a*a)[k] = 2*a[k-i] if 0<=k-i<n
            # sg[i] += 2*w[ii]*a[k-i] for valid i
            # a[k-i] is valid when 0<=k-i<n, i.e., k-n+1 <= i <= k
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += 2.0 * w[ii] * a[k - i_start:i_end + 1][::-1]

        v = momentum * v - lr * sg
        a_candidate = a + v

        # Project onto simplex
        u = np.sort(a_candidate)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_candidate - theta, 0.0)
        else:
            a_new = np.ones(n) / n

        a_new = a_new / a_new.sum()
        max_conv = fast_convolve(a_new).max()
        score = 2.0 * n * max_conv
        step_count += 1

        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        if no_improve > 200:
            a = best_a + np.random.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 3, 0.004)
            no_improve = 0

        if step_count % 5000 == 0:
            elapsed = monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial: n=768, score={best_score:.6f}")

    print(f"\n=== Phase 1: Fictitious Play n={n_parent} ===")

    print(f"\n=== Phase 2: Linearized Peak Shaving (best={best_score:.6f}) ===")

            best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T}", best_seq, best_score)

    # === Phase 4: Damped LP on best ===
    print(f"\n=== Phase 4: Damped LP (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 250
        if rem > 30:
            t_dl = min(rem * 0.5, 150.0)
            print(f"  Budget: {t_dl:.0f}s")
            a_dl, _ = damped_lp_refine(best_seq, t_dl, solver=solver)
            best_seq, best_score = check_and_update(a_dl, "damped_lp", best_seq, best_score)

    # === Phase 5: Try a different length ===
    print(f"\n=== Phase 5: Alternative Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Try a longer length
        alt_n = int(n_best * 1.3)
        if alt_n > n_best and alt_n <= 3072:
            rem = deadline - monotonic() - 200
            if rem > 80:
                t_alloc = min(rem * 0.7, 200.0)
                t_fp_alt = t_alloc * 0.6
                t_ls_alt = t_alloc * 0.4
                print(f"  Trying n={alt_n} (FP {t_fp_alt:.0f}s + LS {t_ls_alt:.0f}s)")
                init_alt = interpolate_sequence(best_seq, alt_n)
                a_alt, _, _ = fictitious_play(alt_n, t_fp_alt, initial=init_alt, solver=solver, verbose=False)
                best_seq, best_score = check_and_update(a_alt, f"FP_alt_n{alt_n}", best_seq, best_score)
                if best_seq is not None and len(best_seq) == alt_n and t_ls_alt > 20:
                    a_alt2, _ = linearized_peak_shave(best_seq, t_ls_alt, solver=solver, verbose=False)
                    best_seq, best_score = check_and_update(a_alt2, f"LS_alt_n{alt_n}", best_seq, best_score)

    # === Phase 6: Subgradient on best ===
    print(f"\n=== Phase 6: Subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 120
        if rem > 30:
            t_sg = min(rem * 0.5, 120.0)
            n_best = len(best_seq)
            top_k = max(30, n_best // 8)
            a_sg, _ = subgradient_optimize(best_seq, t_sg, lr_init=0.004, momentum=0.92, top_k=top_k)
            best_seq, best_score = check_and_update(a_sg, "subgrad", best_seq, best_score)

    # === Phase 7: Final linearized shaving ===
    print(f"\n=== Phase 7: Final Linearized Shaving (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 60
        if rem > 20:
            t_final_ls = min(rem, 80.0)
            a_fs, _ = linearized_peak_shave(best_seq, t_final_ls, solver=solver, verbose=False)
            best_seq, best_score = check_and_update(a_fs, "final_ls", best_seq, best_score)

    # === Phase 8: Final L-BFGS ===
    print(f"\n=== Phase 8: Final L-BFGS (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 20
        if rem > 15:
            t_final = min(rem * 0.6, 40.0)
            a_final = lbgfs_smooth(best_seq, t_final, T=60, maxiter=3000)
            best_seq, best_score = check_and_update(a_final, "final_lbgfs", best_seq, best_score)

    # === Final: One more quick linearized shave ===
    if best_seq is not None:
        rem = deadline - monotonic() - 10
        if rem > 15:
            a_last, _ = linearized_peak_shave(best_seq, min(rem, 20.0), solver=solver, verbose=False)
            best_seq, best_score = check_and_update(a_last, "last_ls", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
