Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def spectral_flatten(a, time_limit=120.0):
    """Iteratively flatten the power spectrum to equalize autocorrelation."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    it = 0
    while monotonic() - t0 < time_limit:
        it += 1
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        
        # Target: blend between flat and current spectrum
        # DC component is fixed (A[0] = 1 after normalization)
        if len(P) > 1:
            P_non_dc = P[1:]
            P_mean = P_non_dc.mean()
            # Smoothing parameter: start with aggressive flattening
            progress = min(1.0, it / 3000.0)
            alpha = 0.05 + 0.95 * progress  # 0.05 = almost flat, 1.0 = no change
            P_target = P.copy()
            P_target[1:] = P_mean + alpha * (P_non_dc - P_mean)
        else:
            P_target = P.copy()
        
        P_target = np.maximum(P_target, 1e-30)
        
        # Update: preserve phase, adjust magnitude
        abs_A = np.abs(A)
        A_new = np.where(abs_A > 1e-30, A * (np.sqrt(P_target) / abs_A), A)
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b (non-negative, sum=1) minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def multi_peak_equalize(a0, time_limit=120.0, n_iters=10000, threshold_frac=0.85):
    """Equalize multiple peaks of the autocorrelation simultaneously."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    lr = 0.01
    
    for it in range(n_iters):
        if monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        max_c = c.max()
        
        # Find significant peaks
        threshold = threshold_frac * max_c
        peak_mask = c >= threshold
        peak_indices = np.where(peak_mask)[0]
        
        if len(peak_indices) == 0:
            break
        
        # Compute target level (mean of significant peaks)
        target = c[peak_indices].mean()
        
        # Compute gradient: for each peak k, the gradient of c[k] wrt a is 2*a[k-i] for each i
        # We want to reduce peaks above target
        grad = np.zeros(n)
        for k in peak_indices:
            excess = max(0, c[k] - target)
            if excess < 1e-15:
                continue
            weight = excess / max(target, 1e-15)  # Relative excess
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * weight * a[k - idx]
        
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm
        
        # Gradient descent with adaptive step
        a_trial = a - lr * grad
        a_trial = np.maximum(a_trial, 0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            lr *= 0.5
            continue
        a_trial = a_trial / s_trial
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            lr = min(lr * 1.05, 0.1)  # Increase step slightly
        else:
            lr *= 0.8
            # If lr gets too small, try perturbation
            if lr < 1e-6:
                a = best_a + np.random.standard_normal(n) * 0.001
                a = np.maximum(a, 0)
                a = a / a.sum()
                lr = 0.01
        
        if (it + 1) % 500 == 0:
            elapsed = monotonic() - t0
            print(f"    PeakEq {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")
    
    return best_a, best_score


def peak_shave(a0, time_limit=60.0, n_steps=5000):

                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial

                    improved = True

            for ci in top_contrib:
                i = int(idx[ci])
                j = int(peak_k - i)
                if i == j or a[j] < 1e-14:
                    continue
                for f1 in [0.9, 0.85, 0.8]:
                    for f2 in [0.9, 0.85, 0.8]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        # Strategy 3: Small random perturbation
        if not improved:
            for attempt in range(10):
                a_trial = a + np.random.standard_normal(n) * 0.0003
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            failed += 1
            if failed > 100:
                break
        else:
            failed = 0
    
    return best_a, best_score


def lbgfs_smooth(a0, time_limit=30.0, T=50, maxiter=3000):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]
    
    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse / T
    
    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()
        
        # Gradient via FFT: grad_a[i] = 2 * sum_k w[k] * a[k-i]
        # This is 2 * conv(w, a_flipped)[i] but let's use direct via FFT
        # Actually, let's use the matrix form
        # grad_a = 2 * T^T @ w where T @ a = conv(a, a)... 
        # Wait, that's not right. Let me compute directly.
        # c[k] = sum_i a[i]*a[k-i]
        # d(c[k])/da[j] = 2*a[k-j] (for k-j valid)
        # So grad_a[j] = 2 * sum_k w[k] * a[k-j]
        # = 2 * (w * a)[j] where * is cross-correlation
        
        # Cross-correlation of w and a: (w ⋆ a)[j] = sum_k w[k] * a[k-j]
        # This can be computed as: conv(w, flip(a))[j + offset]
        w_padded = np.zeros(2 * n - 1)
        w_padded[:len(w)] = w
        a_flipped = a[::-1]
        corr = np.convolve(w_padded, a_flipped, mode='full')
        # corr[k] = sum_i w_padded[i] * a_flipped[k-i]
        # We want sum_k w[k] * a[k-j] for j = 0,...,n-1
        # = (w ⋆ a)[j]
        # np.correlate(w, a, 'full') gives sum_k w[k] * a[k+j] ... 
        # Let me just do it directly for clarity
        grad_a = np.zeros(n)
        for j in range(n):
            i_min = max(0, j)  # k-j >= 0 => k >= j
            i_max = min(len(w) - 1, 2 * n - 2 - j)  # k-j < n => k < n+j
            # Actually k ranges from 0 to 2n-2, and k-j ranges from 0 to n-1
            # So k ranges from j to j+n-1
            k_min = j
            k_max = min(len(w) - 1, j + n - 1)
            if k_min <= k_max:
                ks = np.arange(k_min, k_max + 1)
                grad_a[j] = 2.0 * np.dot(w[ks], a[ks - j])
        
        grad_a *= 1.0  # Already have the 2x factor
        
        # Transform to z-space
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 10, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0
    except Exception:
        a_out = a0
    
    return a_out
