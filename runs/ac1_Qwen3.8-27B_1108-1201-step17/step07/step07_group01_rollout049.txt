```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding 
with cutting-plane constraints, line-search step sizes, active-set Newton 
refinement, and multi-start diversification.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(f, g):
    """Linear cross-convolution via FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def score_normalized(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())

def lp_direction(f, M, max_rounds=30, verbose=False):
    """
    Find LP direction: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Uses cutting-plane: start with no constraints, add most violated.
    """
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(g))
    constraints = []
    
    g_val = np.ones(n) / n  # initial guess
    
    for rnd in range(max_rounds):
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=20000)
        except Exception:
            break
        if g.value is None:
            break
        g_candidate = np.maximum(g.value, 0)
        if g_candidate.sum() < 1e-10:
            break
        
        # Check all constraints
        cross = fast_cross_convolve(f, g_candidate)
        tol = 1e-6
        violated = np.where(cross > M + tol)[0]
        if len(violated) == 0:
            g_val = g_candidate
            break
        
        # Add the most violated constraint
        k_v = int(violated[np.argmax(cross[violated])])
        i_start = max(0, k_v - n + 1)
        i_end = min(n - 1, k_v)
        coeffs = f[i_start:i_end + 1]
        j_start = k_v - i_end
        j_end = k_v - i_start
        constraints.append(coeffs @ g[j_start:j_end + 1] <= M)
        
        if verbose:
            print(f"    LP round {rnd}: added constraint k={k_v}, violated by {cross[k_v]-M:.6f}")
    
    g_val = np.maximum(g.value, 0) if g.value is not None else np.ones(n) / n
    return g_val

def line_search(f, g, t_max=1.0, n_points=20):
    """Find best t in [0, t_max] for f_t = (1-t)*f + t*g."""
    best_t = 0.0
    best_score = score_normalized(f)
    ts = np.linspace(0, t_max, n_points)
    for t in ts:
        f_t = (1 - t) * f + t * g
        f_t = np.maximum(f_t, 0)
        s = f_t.sum()
        if s < 1e-15:

        f_t = f_t / s
        sc = score_normalized(f_t)
        if sc < best_score:
            best_score = sc
            best_t = t
    return best_t, best_score

        #   j-i=p, i=j-p: a[j-p]*a[p] (derivative w.r.t. a[p] is a[j-p])

        # If i = j-i, the term a[i]^2 appears once.
        # Therefore: db[j]/da[p] = a[j-p] if j-p is in [0,n-1], else 0.
        # (The derivative of a[p]*a[j-p] w.r.t. a[p] is a[j-p].)
        # If p = j-p, i.e., j = 2p, then the term is a[p]^2, and the derivative is 2*a[p].
        
        # So: db[j]/da[p] = a[j-p] + (a[j-p] if j == 2p else 0) = a[j-p] * (1 + [j==2p])
        # where [j==2p] is 1 if j==2p else 0.
        # Equivalently: db[j]/da[p] = 2*a[j-p] if j==2p, else a[j-p] (when j-p in [0,n-1])
        
        # For the loss L = sum_{k in active} (b[k] - target)^2:
        # dL/da[p] = sum_{k in active} 2*(b[k]-target) * db[k]/da[p]
        
        grad = np.zeros(n)
        for k in active:
            diff = b[k] - target
            # db[k]/da[p] for valid p

                else:
                    grad[p] += 2.0 * diff * a[q]  # 2*(b[k]-target) * a[q]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < tol:
            break
        
        # Gradient step with projection
        lr = 0.01 / (gnorm + 1e-10)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

            best_score = sc
            best_a = a_new.copy()
            a = a_new

            print(f"    Newton iter={it} ({elapsed:.0f}s): score={score_normalized(a):.7f} best={best_score:.7f}")
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding with
cutting-plane constraints, adaptive line search, multi-phase refinement, and
coordinate descent polishing. Key ideas: (1) LP gives optimal direction at each
step, (2) cutting-plane keeps LP small, (3) careful line search prevents
overshooting, (4) final polish with targeted coordinate updates.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Linear cross-convolution via FFT."""
    nf = len(f)
    ng = len(g)
    size = nf + ng - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def lp_direction_cutting_plane(f, M, max_rounds=40, solver=None):
    """
    Find LP direction: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Uses cutting-plane: iteratively add most-violated constraint.
    Returns the optimal g vector (unnormalized).
    """
    n = len(f)
    g_var = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(g_var))
    constraints = []

    for rnd in range(max_rounds):
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=solver, max_iter=50000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iter=50000, verbose=False)
            except Exception:
                break

        if g_var.value is None:
            break

        g_val = np.maximum(g_var.value, 0.0)
        if g_val.sum() < 1e-12:
            break

        # Check all constraints
        cross = fast_cross_convolve(f, g_val)
        tol = 1e-7
        violated = np.where(cross > M + tol)[0]

        if len(violated) == 0:
            return g_val

        # Add the most violated constraint
        k_v = int(violated[np.argmax(cross[violated])])
        i_start = max(0, k_v - n + 1)
        i_end = min(n - 1, k_v)
        coeffs = f[i_start:i_end + 1]
        j_start = k_v - i_end
        j_end = k_v - i_start
        constraints.append(coeffs @ g_var[j_start:j_end + 1] <= M)

    # If we used up all rounds, return current best
    return np.maximum(g_var.value, 0.0) if g_var.value is not None else np.ones(n) / n


def line_search(f, g, t_values=None):
    """Find best t in [0,1] for f_t = (1-t)*f + t*g."""
    if t_values is None:
        t_values = np.concatenate([
            np.linspace(0.01, 0.1, 5),
            np.linspace(0.1, 0.3, 5),
            np.linspace(0.3, 0.6, 4),
            np.linspace(0.6, 1.0, 4)
        ])

    best_t = 0.0
    best_score = score_normalized(f)

    for t in t_values:
        f_t = (1.0 - t) * f + t * g
        f_t = np.maximum(f_t, 0.0)
        s = f_t.sum()
        if s < 1e-15:
            continue
        f_t = f_t / s
        sc = score_normalized(f_t)
        if sc < best_score - 1e-14:
            best_score = sc
            best_t = t

    return best_t, best_score


def lp_optimize(f_init, time_budget, max_rounds=35, verbose=True, label="LP"):
    """
    Main LP-based optimization loop.
    At each step: find LP direction, do line search, update.
    """
    n = len(f_init)
    f = np.maximum(f_init, 0.0).astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_normalized(f)
    t0 = time.monotonic()
    no_improve_count = 0
    max_no_improve = 50

    for it in range(100000):
        if time.monotonic() - t0 > time_budget:
            break

        b = fast_convolve(f)
        M = float(b.max())
        current_score = 2.0 * n * M

        if current_score < best_score - 1e-13:
            best_score = current_score
            best_f = f.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count > max_no_improve:
            break

        # Find LP direction
        g = lp_direction_cutting_plane(f, M, max_rounds=max_rounds)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Check if g is meaningfully different from f
        if np.allclose(g, f, atol=1e-6):
            no_improve_count += 10
            continue

        # Line search
        t_opt, sc = line_search(f, g)

        if t_opt > 0 and sc < best_score - 1e-13:
            f_new = (1.0 - t_opt) * f + t_opt * g
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            f = f_new
            no_improve_count = 0
        elif t_opt > 0:
            # Try smaller step
            t_small = min(t_opt, 0.05)
            f_new = (1.0 - t_small) * f + t_small * g
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            sc_small = score_normalized(f_new)
            if sc_small < score_normalized(f) - 1e-14:
                f = f_new
            else:
                no_improve_count += 5

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} ({elapsed:.0f}s): "
                  f"score={current_score:.7f} best={best_score:.7f} t={t_opt:.4f}")

    return best_f, best_score


def peak_equalize(f_init, time_budget, verbose=True, label="PEQ"):
    """
    Peak equalization: try to make all near-peak values equal.
    Uses iterative proportional reduction of contributing elements.
    """
    n = len(f_init)
    f = np.maximum(f_init, 1e-15).astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_normalized(f)
    t0 = time.monotonic()

    for it in range(5000):
        if time.monotonic() - t0 > time_budget:
            break

        b = fast_convolve(f)
        M = b.max()
        # Active peaks: within 0.5% of max
        active_threshold = M * 0.995
        active = np.where(b >= active_threshold)[0]

        if len(active) <= 1:
            break

        # For each active peak, find top contributors
        contributions = {}
        for k in active:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                if 0 <= j < n:
                    c = f[i] * f[j]
                    if i not in contributions:
                        contributions[i] = 0.0
                    contributions[i] += c

        # Reduce elements that contribute to multiple high peaks
        # and add mass to elements that contribute to few peaks
        n_active = len(active)
        peak_count = np.zeros(n)
        for k in active:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                if 0 <= j < n:
                    peak_count[i] += 1

        # Target: reduce elements with high contribution to active peaks
        # and increase elements with low contribution
        contrib_array = np.array([contributions.get(i, 0.0) for i in range(n)])
        
        # Find elements to reduce (high contribution to peaks)
        sorted_idx = np.argsort(contrib_array)[::-1]
        n_reduce = max(1, n // 20)
        reduce_idx = sorted_idx[:n_reduce]
        
        # Find safe elements to increase (low contribution)
        safe_idx = sorted_idx[n_reduce:]
        
        if len(safe_idx) == 0:
            break

        # Try reducing top contributors and adding to safe elements
        improved = False
        for frac in [0.05, 0.03, 0.02, 0.01]:
            f_trial = f.copy()
            total_reduce = 0.0
            for i in reduce_idx:
                reduction = f_trial[i] * frac
                f_trial[i] -= reduction
                total_reduce += reduction
            f_trial = np.maximum(f_trial, 0.0)
            
            # Distribute to safe elements proportionally
            safe_sum = f_trial[safe_idx].sum()
            if safe_sum < 1e-15:
                continue
            f_trial[safe_idx] += total_reduce * f_trial[safe_idx] / safe_sum
            f_trial = f_trial / f_trial.sum()
            
            sc = score_normalized(f_trial)
            if sc < best_score - 1e-13:
                best_score = sc
                f = f_trial
                best_f = f_trial.copy()
                improved = True
                break

        if not improved:
            # Try paired reduction
            if len(reduce_idx) >= 2:
                for frac in [0.03, 0.02]:
                    f_trial = f.copy()
                    total_reduce = 0.0
                    for i in reduce_idx[:max(2, n_reduce // 2)]:
                        reduction = f_trial[i] * frac
                        f_trial[i] -= reduction
                        total_reduce += reduction
                    f_trial = np.maximum(f_trial, 0.0)
                    safe_sum = f_trial[safe_idx].sum()
                    if safe_sum < 1e-15:
                        continue
                    f_trial[safe_idx] += total_reduce * f_trial[safe_idx] / safe_sum
                    f_trial = f_trial / f_trial.sum()
                    sc = score_normalized(f_trial)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        f = f_trial
                        best_f = f_trial.copy()
                        improved = True
                        break
            if not improved:
                break

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} ({elapsed:.0f}s): score={score_normalized(f):.7f} best={best_score:.7f}")

    return best_f, best_score


def coordinate_descent_polish(f_init, time_budget, verbose=True, label="CD"):
    """
    Coordinate descent: try scaling individual elements and redistributing.
    Also tries paired swaps.
    """
    n = len(f_init)
    f = np.maximum(f_init, 0.0).astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_normalized(f)
    t0 = time.monotonic()

    factors = [0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.3]
    passes = 0
    no_improve_passes = 0

    while time.monotonic() - t0 < time_budget and no_improve_passes < 10:
        no_improve_passes += 1
        passes += 1
        improved_this_pass = False
        order = np.random.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_budget:
                break

            for fac in factors:
                f_trial = f.copy()
                old_val = f_trial[i]
                new_val = old_val * fac
                f_trial[i] = new_val
                diff = f_trial.sum() - 1.0
                if abs(diff) < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = f_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                f_trial[mask] -= diff * f_trial[mask] / others_sum
                if f_trial.min() < -1e-12:
                    continue
                f_trial = np.maximum(f_trial, 0.0)
                f_trial = f_trial / f_trial.sum()
                sc = score_normalized(f_trial)
                if sc < best_score - 1e-13:
                    best_score = sc
                    f = f_trial
                    best_f = f_trial.copy()
                    improved_this_pass = True
                    break

        if improved_this_pass:
            no_improve_passes = 0

        if verbose and passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} pass={passes} ({elapsed:.0f}s): best={best_score:.7f}")

    return best_f, best_score


def mirror_descent_polish(f_init, time_budget, T=0.005, eta=0.01, verbose=True, label="MD"):
    """Quick mirror descent polish with soft-max smoothing."""
    n = len(f_init)
    f = np.maximum(f_init, 1e-15).astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_normalized(f)
    t0 = time.monotonic()

    for it in range(50000):
        if time.monotonic() - t0 > time_budget:
            break

        b = fast_convolve(f)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-13:
            best_score = current_score
            best_f = f.copy()

        # Soft-max weights
        scaled = b / T
        m = scaled.max()
        exp_scaled = np.exp(scaled - m)
        w = exp_scaled / exp_scaled.sum()

        # Gradient via cross-correlation
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(f, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad = 4.0 * n * correl[:n]

        # Mirror update
        log_update = -eta * grad
        log_update = np.clip(log_update, -2.0, 2.0)
        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-300)
        f = f / f.sum()

        # Decay eta slightly
        eta *= 0.9999

    return best_f, best_score


def check_candidate(a, label, best_seq, best_score):
    """Evaluate using the actual evaluate_sequence function."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase LP-based optimization with cutting-plane, line search,
    peak equalization, coordinate descent, and mirror descent polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score

    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback uniform: score={best_score:.7f}")

    n = len(best_seq)
    print(f"\nStarting optimization with n={n}, score={best_score:.7f}")

    # === Phase 1: LP Optimization (main workhorse) ===
    print("\n=== Phase 1: LP Optimization ===")
    rem = deadline - time.monotonic() - 300

        t_alloc = min(rem * 0.45, 500.0)
        print(f"  LP main: budget={t_alloc:.0f}s")
        a_lp, _ = lp_optimize(best_seq, t_alloc, max_rounds=35,

        t_alloc = min(rem * 0.4, 400.0)
        print(f"  LP round2: budget={t_alloc:.0f}s")

        best_seq, best_score = check_candidate(a_lp2, "LP_r2", best_seq, best_score)

    # === Phase 3: Peak Equalization ===
    print(f"\n=== Phase 3: Peak Equalization (best={best_score:.7f}) ===")

        t_alloc = min(rem * 0.45, 200.0)
        print(f"  LP round3: budget={t_alloc:.0f}s")
        a_lp3, _ = lp_optimize(best_seq, t_alloc, max_rounds=50,
                               verbose=True, label="LP_r3")
        best_seq, best_score = check_candidate(a_lp3, "LP_r3", best_seq, best_score)

    # === Phase 5: Mirror Descent Polish ===
    print(f"\n=== Phase 5: Mirror Descent (best={best_score:.7f}) ===")

        t_alloc = min(rem * 0.5, 80.0)

                                        verbose=True, label="MD")

    print(f"\n=== Phase 6: LP Round 4 (best={best_score:.7f}) ===")

        a_lp4, _ = lp_optimize(best_seq, t_alloc, max_rounds=50,

        best_seq, best_score = check_candidate(a_lp4, "LP_r4", best_seq, best_score)

    # Fallback
    if best_seq is None:

        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
