```python
def compute_gradient_fast(a, weights):
    """
    Compute gradient of sum_k weights[k] * conv(a,a)[k] w.r.t. a.
    Uses FFT: grad_i = 2 * (flip(a) * weights)[i + n - 1]
    """
    n = len(a)
    a_flipped = a[::-1]
    # Linear convolution of a_flipped (length n) and weights (length 2n-1)
    # Result has length n + (2n-1) - 1 = 3n - 2
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a_flipped, fft_size)
    W = rfft(weights, fft_size)
    conv = irfft(A * W, fft_size)
    # We want indices n-1 to 2n-2 (inclusive), i.e., conv[n-1 : 2n-1]
    return 2.0 * conv[n-1 : 2*n-1]
```

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_gradient_fft(a, weights):
    """
    Compute gradient of sum_k weights[k]*conv(a,a)[k] w.r.t. a.
    Uses identity: grad_i = 2 * (flip(a) * weights)[i + n - 1]
    where * is linear convolution, computed via FFT.
    """
    n = len(a)
    a_flipped = a[::-1]
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a_flipped, fft_size)
    W = rfft(weights, fft_size)
    conv_result = irfft(A * W, fft_size)
    return 2.0 * conv_result[n - 1: 2 * n - 1]

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng, n_harmonics=None):
    """Initialize using random Fourier superposition."""
    if n_harmonics is None:
        n_harmonics = int(rng.integers(5, 25))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_trapezoid(n, rng, width=0.6):

    a = np.maximum(1.0 - np.abs(t - center) / half_w, 0.0)

    a = a / a.sum()
    return a

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    n_parent = len(parent) if parent is not None else 768

    # === Phase 1: Diverse initialization + quick Adam ===
    print("\n=== Phase 1: Diverse Init + Quick Adam ===")
    t1_start = time.monotonic()
    t_phase1 = 200.0

    n_values = [n_parent]
    for alt in [640, 896, 1024, 512]:
        if alt not in n_values and 256 <= alt <= 2048:
            n_values.append(alt)

    candidates = []
    per_init = t_phase1 / (len(n_values) * 4)

    for n in n_values:
        if time.monotonic() - t1_start > t_phase1:
            break
        inits = []
        if parent is not None:
            inits.append(interpolate_seq(parent, n))
        inits.append(init_fourier(n, rng))
        inits.append(init_trapezoid(n, rng, width=rng.uniform(0.4, 0.9)))
        inits.append(init_u_shape(n, rng, depth=rng.uniform(0.2, 0.7)))

        for init_a in inits:
            rem = t_phase1 - (time.monotonic() - t1_start)
            if rem < 5:
                break
            t_alloc = min(per_init, rem * 0.8)
            a_opt, s_opt = adam_logsumexp_optimize(
                init_a, t_alloc, t_init=0.05, lr=0.01, verbose=False, label=f"p1_n{n}"
            )
            candidates.append((s_opt, a_opt, n))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1 done: {len(candidates)} candidates, best={candidates[0][0]:.6f} (n={candidates[0][2]})")

    for s_c, a_c, n_c in candidates[:4]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}")

    # === Phase 2: Extended Adam on best candidates ===
    print(f"\n=== Phase 2: Extended Adam (best={best_score:.6f}) ===")
    t2_start = time.monotonic()
    t_phase2 = 500.0

    # Focus on the best 2-3 candidates with longer optimization
    top_cands = candidates[:2]
    if best_seq is not None and best_seq not in [c[1] for c in top_cands]:
        top_cands.append((best_score, best_seq, len(best_seq)))

    time_per = t_phase2 / len(top_cands)

    for s_c, a_c, n_c in top_cands:
        rem = t_phase2 - (time.monotonic() - t2_start)
        if rem < 30:
            break
        t_alloc = min(time_per * 1.5, rem * 0.85)
        print(f"  Extended Adam n={n_c}, start_score={s_c:.6f}, budget={t_alloc:.0f}s")
        a_opt, s_opt = adam_logsumexp_optimize(
            a_c, t_alloc, t_init=0.02, lr=0.006, verbose=True, label=f"p2_n{n_c}"
        )
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"p2_n{n_c}")

        # Follow with fine coordinate descent
        rem2 = deadline - time.monotonic() - 600
        if rem2 > 30 and best_seq is not None:
            t_cd = min(rem2 * 0.3, 80.0)
            a_cd, s_cd = fine_coord_descent(best_seq, t_cd)
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"p2_cd_n{len(best_seq)}")

    # === Phase 3: Pair coordinate descent ===
    print(f"\n=== Phase 3: Pair Coord Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t_pc = min(rem * 0.5, 200.0)
            print(f"  PairCD n={len(best_seq)}, budget={t_pc:.0f}s")
            a_pc, s_pc = pair_coord_descent(best_seq, t_pc, verbose=True)
            best_seq, best_score = check_and_update(a_pc, best_seq, best_score, f"p3_pc_n{len(best_seq)}")

    # === Phase 4: Try different lengths ===
    print(f"\n=== Phase 4: Alternative Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [0.75, 0.85, 1.15, 1.25, 1.5]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 3000 and alt_n != n_best:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic() - 300
            if rem < 40:
                break
            t_alt = min(rem * 0.4, 150.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            # Quick Adam on the interpolated sequence
            a_alt, s_alt = adam_logsumexp_optimize(
                init_alt, t_alt, t_init=0.03, lr=0.008, verbose=False, label=f"alt_n{alt_n}"
            )
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

            # Quick fine CD
            rem2 = deadline - time.monotonic() - 200
            if rem2 > 20:
                a_cd, s_cd = fine_coord_descent(
                    best_seq, min(rem2 * 0.3, 50.0)
                )
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"alt_cd_n{len(best_seq)}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Adam with very low temperature
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_final = min(rem * 0.4, 120.0)
            print(f"  Final Adam, budget={t_final:.0f}s")
            a_fin, s_fin = adam_logsumexp_optimize(
                best_seq, t_final, t_init=0.01, lr=0.003, verbose=True, label="final"
            )
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_adam")

        # Fine coordinate descent
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_fcd = min(rem * 0.5, 80.0)
            print(f"  Final FineCD, budget={t_fcd:.0f}s")
            a_fcd, s_fcd = fine_coord_descent(best_seq, t_fcd)
            best_seq, best_score = check_and_update(a_fcd, best_seq, best_score, "final_cd")

        # One more Adam round
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_fin2, s_fin2 = adam_logsumexp_optimize(
                best_seq, min(rem * 0.7, 40.0), t_init=0.008, lr=0.002,
                verbose=False, label="last"
            )
            best_seq, best_score = check_and_update(a_fin2, best_seq, best_score, "last_adam")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
