Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using projected gradient descent

step sizes. Combines multiple restarts with different initializations and
temperatures, followed by coordinate descent polishing.

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_gradient_fft(a, w, fft_size):
    """
    Compute g_m = 2 * sum_j a[j] * w[m+j] for m = 0,...,n-1
    using the FFT identity: IFFT(conj(FFT(a)) * FFT(w)) gives the cross-correlation.
    """
    n = len(a)
    A = np.fft.fft(a, fft_size)
    W = np.fft.fft(w, fft_size)
    # IFFT(conj(A) * W)[m] = sum_j a[j] * w[j+m] (for real a, w)
    cross = np.fft.ifft(np.conj(A) * W)[:n]
    return 2.0 * cross.real

    """Compute softmax weights over the auto-convolution values."""
    c_shifted = c - c.max()
    w = np.exp(c_shifted / temperature)
    s = w.sum()
    if s < 1e-300:

        w = np.zeros(len(c))
        w[c == c.max()] = 1.0
        w /= w.sum()
    else:
        w /= s
    return w

    c = fast_convolve(a)
    w = softmax_weights(c, temperature)
    g = compute_gradient_fft(a, w, fft_size)
    a_new = simplex_project(a - lr * g)
    return a_new, c.max()

    if s < 1e-15:

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_grad_fft(a, w, fft_size):
    """Compute g[m] = 2 * sum_j a[j] * w[m+j] via FFT cross-correlation."""
    n = len(a)
    A = np.fft.fft(a, fft_size)
    W = np.fft.fft(w, fft_size)
    cross = np.fft.ifft(np.conj(A) * W)[:n]
    return 2.0 * cross.real

def optimize_pgd(a0, time_limit, name="pgd"):
    """
    Projected gradient descent with softmax smoothing, adaptive temperature,
    and backtracking line search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    # FFT size for gradient computation (need a and w to fit, w has length 2n-1)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    
    # Parameters
    T = 0.05 * best_score / n  # Initial temperature (scaled)
    T_min = T * 1e-4
    lr = 0.5 / max(fast_convolve(a).max() * 2 * n, 1e-10)  # Initial LR
    lr_min = 1e-6
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute gradient
        c = fast_convolve(a)
        c_max = c.max()
        if c_max < 1e-15:
            break
        
        c_norm = c / c_max  # Normalize for numerical stability
        w = np.exp(c_norm / T - 1.0 / T)  # exp((c - c_max)/T)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(c))
            w[c == c_max] = 1.0
            w /= w.sum()
        else:
            w /= w_sum
        
        g = compute_grad_fft(a, w, fft_size)
        
        # Backtracking line search
        f_curr = c_max
        alpha = lr
        improved = False
        for bt in range(10):
            a_trial = simplex_project(a - alpha * g)
            c_trial = fast_convolve(a_trial)
            f_trial = c_trial.max()
            # Armijo condition
            if f_trial <= f_curr - 1e-4 * alpha * np.dot(g, a - a_trial):
                a_new = a_trial
                improved = True
                break
            alpha *= 0.5
            if alpha < lr_min:
                break
        
        if not improved:
            # Try a smaller step anyway
            alpha = max(alpha, 1e-8)
            a_new = simplex_project(a - alpha * g)
        
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adapt temperature and step size
        if no_improve > 0 and no_improve % 30 == 0:
            T = max(T * 0.9, T_min)
        
        if no_improve > 100:
            lr *= 0.7
            no_improve = 0
            if lr < lr_min:
                # Reset with perturbation
                rng = np.random.default_rng(it * 31 + 7)
                noise = rng.standard_normal(n) * 0.01
                a = simplex_project(best_a + noise)
                lr = 0.5 / max(fast_convolve(a).max() * 2 * n, 1e-10)
                T = 0.05 * fast_convolve(a).max() / n
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PGD[{name}] it={it} ({elapsed:.0f}s): score={best_score:.6f} T={T:.6f}")
    
    return best_a, best_score

"""

Phase B: Softmax-smoothed max minimization via FISTA with geometric annealing

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_complex(a, N):
    """Compute auto-convolution using N-point complex FFT. Returns length 2n-1."""
    n = len(a)
    A = np.fft.fft(a, N)
    C = np.fft.ifft(A * A, N)
    return C.real[:2 * n - 1]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def compute_softmax_grad(a, c, T, fft_size):
    """
    Compute gradient of T*log(sum exp(c/T)) w.r.t. a.
    grad[m] = 2 * sum_k w[k] * a[k-m] for valid k-m in [0, n-1].
    Uses FFT cross-correlation: IFFT(conj(FFT(a)) * FFT(w))[m] = sum_j a[j]*w[j+m].
    """
    n = len(a)
    c_shifted = c - c.max()
    w = np.exp(c_shifted / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[c == c.max()] = 1.0
        w /= w.sum()
    else:
        w /= w_sum
    A = np.fft.fft(a, fft_size)
    W = np.fft.fft(w, fft_size)
    cross = np.fft.ifft(np.conj(A) * W)[:n]
    return 2.0 * cross.real


def compute_l2_grad(a, N):
    """
    Compute gradient of sum_k c_k^2 (L2 norm squared of auto-convolution).
    Equivalent to (8/N) * Re(FFT(|A|^2 * conj(A)))[:n] where A = FFT(a, N).
    """
    n = len(a)
    A = np.fft.fft(a, N)
    B = np.abs(A) ** 2 * np.conj(A)
    grad = (8.0 / N) * np.real(np.fft.fft(B))[:n]
    return grad


def l2_norm_of_conv(a, N):
    """Compute sum_k c_k^2 using FFT (Parseval)."""
    A = np.fft.fft(a, N)
    return np.sum(np.abs(A) ** 4) / N


def softmax_obj(a, c, T):
    """Compute T * log(sum exp(c/T))."""
    c_shifted = c - c.max()
    val = T * (np.log(np.sum(np.exp(c_shifted / T))) + c.max() / T)
    return val


def objective_max(a, n):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def fista_l2(a0, time_limit, name="l2"):
    """
    FISTA (Nesterov-accelerated proximal gradient) to minimize L2 norm of auto-convolution
    on the probability simplex.
    """
    n = len(a0)

        a = a / s

    best_l2 = l2_norm_of_conv(a, N_fft)

    # Initial step size estimate

    g0_norm = np.linalg.norm(g0)
    if g0_norm < 1e-15:
        lr = 0.01
    else:
        lr = 0.5 / g0_norm

    a_prev = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it - 1) ** 2)) / 2.0
        beta = (t_k - 1.0) / t_k

        y = np.maximum(y, 0)
        y = y / y.sum()

        a_trial = simplex_project(y - lr * g)

        # Backtracking

        l2_trial = l2_norm_of_conv(a_trial, N_fft)
        bt = 0
        while l2_trial > l2_curr + 1e-6 * lr * np.dot(g, y - a_trial) and bt < 15:
            lr *= 0.5
            a_trial = simplex_project(y - lr * g)
            l2_trial = l2_norm_of_conv(a_trial, N_fft)
            bt += 1

        a = a_trial

                best_a = a.copy()

def fista_softmax(a0, time_limit, T_init=None, T_min=None, name="softmax"):
    """
    FISTA to minimize softmax-smoothed max on the simplex with geometric annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    fft_size = 1 << (3 * n - 2 - 1).bit_length()

    c_init = fast_convolve(a)
    c_max_init = c_init.max()
    if T_init is None:
        T_init = c_max_init / 5.0
    if T_min is None:
        T_min = T_init * 1e-5

    best_a = a.copy()
    best_score = 2.0 * n * c_max_init

    # Initial step size
    T_curr = T_init
    g0 = compute_softmax_grad(a, c_init, T_curr, fft_size)
    g0_norm = np.linalg.norm(g0)
    if g0_norm < 1e-15:
        lr = 0.01
    else:
        lr = 1.0 / g0_norm

    a_prev = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    n_total_est = int(time_limit / 0.002)  # estimate total iterations
    gamma = (T_min / T_init) ** (1.0 / max(n_total_est, 100))

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it - 1) ** 2)) / 2.0
        beta = (t_k - 1.0) / t_k

        y = a + beta * (a - a_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = best_a.copy()
        else:
            y = y / sy

        c_y = fast_convolve(y)
        c_max_y = c_y.max()
        g = compute_softmax_grad(y, c_y, T_curr, fft_size)

        # Backtracking line search
        a_trial = simplex_project(y - lr * g)
        f_trial = softmax_obj(a_trial, fast_convolve(a_trial), T_curr)
        f_curr = softmax_obj(y, c_y, T_curr)
        bt = 0
        while f_trial > f_curr + 1e-4 * lr * np.dot(g, y - a_trial) and bt < 12:
            lr *= 0.5
            a_trial = simplex_project(y - lr * g)
            f_trial = softmax_obj(a_trial, fast_convolve(a_trial), T_curr)
            bt += 1

        a_prev = a
        a = a_trial

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Anneal temperature
        T_curr = max(T_curr * gamma, T_min)

        # Adaptive restart
        if no_improve > 200:
            lr = min(lr * 0.5, 1.0 / max(np.linalg.norm(compute_softmax_grad(a, fast_convolve(a), T_curr, fft_size)) + 1e-15, 1e-15))
            no_improve = 0
            if no_improve > 500:
                break

        if it % 500 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTASoftmax[{name}] it={it} ({elapsed:.0f}s): score={best_score:.6f} T={T_curr:.6f}")

        if no_improve > 500:
            break

    return best_a, best_score


def multiplicative_weights_step(a, g, lr):
    """Multiplicative weights (mirror descent) step on the simplex."""
    exp_g = np.exp(-lr * g)
    a_new = a * exp_g
    s = a_new.sum()
    if s < 1e-300:
        a_new = a * (1.0 - lr * g)
        s = a_new.sum()
        if s < 1e-300:
            return a.copy()
    a_new = a_new / s
    return a_new


def mw_optimize(a0, time_limit, T_init=None, T_min=None, name="mw"):
    """
    Mirror descent with multiplicative weights update for softmax-smoothed max.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    fft_size = 1 << (3 * n - 2 - 1).bit_length()

    c_init = fast_convolve(a)
    c_max_init = c_init.max()
    if T_init is None:
        T_init = c_max_init / 5.0
    if T_min is None:
        T_min = T_init * 1e-4

    best_a = a.copy()
    best_score = 2.0 * n * c_max_init

    T_curr = T_init
    lr = 0.5 / max(np.linalg.norm(compute_softmax_grad(a, c_init, T_curr, fft_size)) + 1e-15, 0.01)

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    n_total_est = int(time_limit / 0.003)
    gamma = (T_min / T_init) ** (1.0 / max(n_total_est, 100))

    while time.monotonic() - t0 < time_limit:
        it += 1

        c_y = fast_convolve(a)
        g = compute_softmax_grad(a, c_y, T_curr, fft_size)

        a_new = multiplicative_weights_step(a, g, lr)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        T_curr = max(T_curr * gamma, T_min)

        if no_improve > 100:
            lr *= 0.8
            no_improve = 0

        if no_improve > 400:
            rng = np.random.default_rng(it * 17 + 3)
            noise = rng.standard_normal(n) * 0.008
            a = simplex_project(best_a + noise)
            lr = 0.5
            T_curr = max(T_curr * 2, min(T_init * 0.1, c_max_init / 3.0))
            no_improve = 0

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW[{name}] it={it} ({elapsed:.0f}s): score={best_score:.6f} T={T_curr:.6f}")

        if no_improve > 400:
            break

    return best_a, best_score


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.CBC, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.ECOS, cp.SCS, cp.CBC, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def fictitious_play(a0, time_limit, solver=None, verbose=False, name="fp"):
    """Alternating LP best responses with damped averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new

        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        a_avg = (iters * a_avg + a) / (iters + 1)

        candidates = [a, b, 0.5 * (a + b), a_avg, 0.3 * a + 0.7 * a_avg]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s_c = cand.sum()
            if s_c < 1e-10:
                continue
            cand_norm = cand / s_c
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand_norm.copy()

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP[{name}] it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def coord_descent_polish(a0, time_limit, n_sweeps=500, name="cd"):
    """Full coordinate descent with adaptive scaling."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-14:
                # Try setting to small positive
                for new_v in [0.001 / n, 0.01 / n]:
                    a[i] = new_v
                    a = a / a.sum()
                    score = 2.0 * n * fast_convolve(a).max()
                    if score < best_score - 1e-13:
                        best_score = score
                        best_a = a.copy()
                        improved = True
                    else:
                        a[i] = 0.0
                        a = a / a.sum()
                continue

            # Try several scaling factors (log-spaced around 1)
            for factor in [0.5, 0.7, 0.85, 0.92, 0.97, 1.03, 1.08, 1.15, 1.3, 1.6, 2.0]:
                new_val = old_val * factor
                a[i] = new_val
                a = np.maximum(a, 0)
                s_t = a.sum()
                if s_t < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s_t
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a.copy()
                    old_val = a[i]
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()

            # Also try setting to zero
            if old_val > 1e-6:
                a[i] = 0.0
                a = a / a.sum()
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()

        if not improved and sweep > 10:
            # Try smaller perturbations
            any_small = False
            for i in rng.choice(n, size=min(n, 200), replace=False):
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = a[i]
                if old_val < 1e-14:
                    continue
                for factor in [0.99, 0.995, 1.005, 1.01]:
                    a[i] = old_val * factor
                    a = a / a.sum()
                    score = 2.0 * n * fast_convolve(a).max()
                    if score < best_score - 1e-14:
                        best_score = score
                        best_a = a.copy()
                        old_val = a[i]
                        improved = True
                        any_small = True
                    else:
                        a[i] = old_val
                        a = a / a.sum()
                if not any_small and sweep > 20:
                    break

            if not any_small and sweep > 30:
                break

    return best_a, best_score


def peak_shave(a0, time_limit, name="ps"):
    """Focus on reducing the highest peaks by adjusting contributing coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        top_k = min(7, len(b))
        peaks = np.argpartition(b, -top_k)[-top_k:]

        improved = False
        for peak_k in peaks:
            peak_k = int(peak_k)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]

            top_contrib = np.argsort(contribs)[::-1][:6]
            for ci in top_contrib:
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(idx[ci])
                j = peak_k - i
                if a[i] < 1e-14:
                    continue

                # Try reducing a[i]
                for factor in [0.95, 0.9, 0.85, 0.8, 0.7]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break

                # Try reducing both a[i] and a[j] if i != j
                if not improved and i != j and a[j] >= 1e-14:
                    for f1 in [0.9, 0.85]:
                        for f2 in [0.9, 0.85]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            a_trial = np.maximum(a_trial, 0)
                            s_t = a_trial.sum()
                            if s_t < 1e-15:
                                continue
                            a_trial = a_trial / s_t
                            score = 2.0 * n * fast_convolve(a_trial).max()
                            if score < best_score - 1e-13:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                if improved:
                    break
            if improved:
                break

        if not improved:
            break

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng, n_harmonics=None):
    """Initialize using random Fourier superposition."""
    if n_harmonics is None:
        n_harmonics = int(rng.integers(5, 20))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_tukey(n, rng):
    """Initialize with a Tukey (raised cosine) window shape."""
    alpha = rng.uniform(0.3, 0.9)
    t = np.arange(n) / n
    a = np.ones(n)
    half = alpha / 2.0
    for i in range(n):
        if t[i] < half:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * t[i] / alpha - 1)))
        elif t[i] > 1 - half:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * t[i] / alpha - 2 + 1)))
    a += rng.standard_normal(n) * 0.02
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

        # One more short MW run
        t_e4 = min(20.0, deadline - time.monotonic() - 10.0)
        if t_e4 > 3:
            print(f"  E4: MW final, budget={t_e4:.0f}s")
            a_f4, s_f4 = mw_optimize(best_seq, t_e4, name="mw_final")
            best_seq, best_score = check_and_update(a_f4, best_seq, best_score, "mw_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        if fs < best_score:
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print(f"\nWARNING: final evaluation returned {fs}, using backup")
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Backup FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")

    return [float(x) for x in best_seq]
