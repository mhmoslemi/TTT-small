import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax(x, T=1.0):
    x = x / T
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def compute_lse_gradient(a, T):
    """Compute gradient of 2*n*LSE_T(conv(a,a)) at point a."""
    n = len(a)
    b = fast_convolve(a)  # shape (2n-1,)
    w = softmax(b, T)  # shape (2n-1,)
    # g_i = 4*n * sum_j a_j * w_{i+j} for i = 0..n-1
    # This is cross-correlation of w and a
    g = 4.0 * n * np.correlate(w, a, mode='valid')
    return g, b, w

def mirror_descent_lse(a0, time_limit, T_init=1.0, T_min=0.01, 
                        eta_init=0.01, eta_decay=0.9999, 
                        T_anneal_rate=0.99995, verbose=False):
    """Mirror descent with log-sum-exp loss."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    T = T_init
    eta = eta_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        g, b, w = compute_lse_gradient(a, T)
        
        # Mirror descent update (multiplicative)
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - eta * g / (4.0 * n)  # normalized gradient
        log_a_new = log_a_new - log_a_new.max()  # for numerical stability
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        T = max(T_min, T * T_anneal_rate)
        eta *= eta_decay
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.0f}s): best={best_score:.6f} T={T:.4f} eta={eta:.6f}")
    
    return best_a, best_score

Actually, let me think about this differently. The key issue is that the log-sum-exp gradient is a weighted average of the "peak gradients" (weighted by softmax weights). When T is large, the weights are nearly uniform, so the gradient is nearly the average of all peak gradients, which pushes toward making the convolution flat. When T is small, the weights concentrate on the actual peaks, and the gradient focuses on reducing those peaks.

This is exactly what we want! Start with large T to spread out the convolution, then decrease T to focus on the remaining peaks.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp mirror descent with adaptive temperature annealing
2. LP-based fictitious play (alternating best responses)
3. Coordinate descent with adaptive step sizes
4. Multi-phase optimization with progress tracking

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_lse_gradient(a, T):
    n = len(a)
    b = fast_convolve(a)
    # Softmax with temperature T
    scaled = b / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    # g_i = 4*n * sum_j a_j * w_{i+j}
    g = 4.0 * n * np.correlate(w, a, mode='valid')
    return g, b, w

def mirror_descent(a0, time_limit, T_init=1.0, T_min=0.02, eta=0.05, 
                   momentum=0.9, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    T = T_init
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    total_iters = max(1, int(time_limit / 0.0005))  # rough estimate
    T_decay = T_min / T_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        g, b, w = compute_lse_gradient(a, T)
        
        # Momentum
        v = momentum * v + (1 - momentum) * g
        
        # Normalize step
        g_norm = np.abs(v).max()
        if g_norm < 1e-15:
            break
        v_norm = v / g_norm
        
        # Mirror descent update
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - eta * v_norm
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new /= a_new.sum()
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        # Anneal temperature
        frac = min(1.0, it / total_iters)
        T = T_init * (T_decay ** frac)
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.1f}s): best={best_score:.6f} T={T:.4f}")
    
    return best_a, best_score

def build_conv_matrix(a):
    n = len(a)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    return T_mat

def get_lp_solver():
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.CBC, cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [
        T_mat @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def fictitious_play(a0, time_limit, solver=None, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    b = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b_new = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
        
        # Check candidate
        for cand in [a, b, 0.5*(a+b)]:
            cand = np.maximum(cand, 0)
            s_c = cand.sum()
            if s_c < 1e-10:
                continue
            cand = cand / s_c
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score:
                best_score = score
                best_a = cand.copy()
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP it={it} ({elapsed:.1f}s): best={best_score:.6f}")
    
    return best_a, best_score

def peak_shave(a0, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Find the peak
        k_peak = int(np.argmax(b))
        
        # Find contributing indices
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_peak - idx]
        
        if contribs.max() < 1e-15:
            break
        
        # Try reducing the top contributors
        top_n = min(8, len(idx))
        top_idx = idx[np.argsort(contribs)[::-1][:top_n]]
        
        improved = False
        for i in top_idx:
            for factor in [0.85, 0.75, 0.65, 0.5]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try redistributing: reduce peak contributor, add to non-peak
            i = int(top_idx[0])
            j_best = None
            best_j_score = best_score
            # Find position with lowest contribution to peak
            low_idx = idx[np.argsort(contribs)[:min(5, len(idx))]]
            for j in low_idx:
                if j == i:
                    continue
                # Move some mass from i to j
                for frac in [0.1, 0.2, 0.3]:
                    amount = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amount
                    a_trial[j] += amount
                    if a_trial[i] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_j_score - 1e-14:
                        best_j_score = score
                        j_best = (j, frac)
            
            if j_best is not None:
                j, frac = j_best
                amount = a[i] * frac
                a[i] -= amount
                a[j] += amount
                a = a / a.sum()
                best_score = best_j_score
                best_a = a.copy()
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
