    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s
    
    N = 1 << (4*n - 1).bit_length()  # FFT size, power of 2, >= 4n-1
    
    best_a = a.copy()
    best_score = 2*n*fast_convolve(a).max()
    
    t0 = monotonic()
    T = T0

        if monotonic() - t0 > time_limit:
            break
        
        # Compute convolution
        c = fast_convolve(a)  # length 2n-1
        
        # Compute softmax weights
        c_scaled = (c - c.max()) / T
        exp_c = np.exp(c_scaled)
        rho = exp_c / exp_c.sum()

        g1 = irfft(A * np.conj(R), N)[:n]  # cross-correlation
        g2 = irfft(A * R, N)[:n]  # convolution

        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Evaluate
        max_c = fast_convolve(a_new).max()
        score = 2*n*max_c
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        # Decrease temperature
        T = max(T_min, T * 0.999)
    
    return best_a, best_score

Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based smoothed gradient descent, direct subgradient methods,
L-BFGS refinement, and peak shaving.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum x = s}."""
    v = np.maximum(v, 0)
    ss = v.sum()
    if ss < 1e-15:
        return np.ones(len(v)) / len(v) * s
    return v * s / ss

def compute_grad_fft(a, rho, N):

    n = len(a)
    A = rfft(a, N)
    R = rfft(rho, N)
    T1 = irfft(np.conj(A) * R, N)[:n]
    T2 = irfft(R * A, N)[:n]
    return T1 + T2
