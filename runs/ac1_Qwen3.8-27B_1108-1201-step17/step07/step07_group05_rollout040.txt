import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score_normalized(a):
    """Score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def project_simplex(v):
    """Project v onto the probability simplex {x: sum x = 1, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_method(a0, time_limit, dt=0.01, verbose=False):
    """Projected subgradient method for minimizing max of convolution."""
    n = len(a0)
    a = a0.copy()
    a = project_simplex(a)
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    
    # Polyak-style step size
    f0 = best_score
    f_opt_lower = 2.0  # theoretical lower bound is 2... wait, no, we're below 2
    
    # Actually let me use a decreasing step size
    eta = dt
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient
        g = np.zeros(n)
        for j in range(n):
            idx1 = k_star - j
            idx2 = j - k_star
            if 0 <= idx1 < n:
                g[j] += a[idx1]
            if 0 <= idx2 < n:
                g[j] += a[idx2]
        
        # Projected update
        a_new = project_simplex(a - eta * g)
        score = compute_score_normalized(a_new)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            # Still update a (for averaging)
            a = a_new
        
        # Decrease step size
        if no_improve > 50:
            eta *= 0.95
            no_improve = 0
        
        if verbose and iters % 1000 == 0:
            print(f"    Subgrad iter={iters}, eta={eta:.6f}, best={best_score:.8f}")
    
    return best_a, best_score

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute score for a normalized (sum=1) sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project v onto the probability simplex {x: sum x = 1, x >= 0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_idx = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def get_subgradient_at_lag(a, k):
    """Compute subgradient of conv(a,a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    j_min = max(0, k - n + 1)
    j_max = min(n - 1, k)
    if j_min <= j_max:

            for frac in [0.05, 0.1, 0.2, 0.3]:
                delta = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= delta
                # Redistribute to partner or random position
                if a_trial[p] >= 0:
                    a_trial[p] += delta * 0.5
                    # Rest to random
                    ridx = int(rng.integers(0, n))
                    a_trial[ridx] += delta * 0.5
                else:
                    a_trial[p] += delta
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-14:
                    continue
                a_trial = a_trial / ss
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a_trial.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            # Perturb around peak positions
            all_peak_pos = set()
        for pi in peak_indices[:5]:
            pi = int(pi)
            si = max(0, pi - n + 1)
            ei = min(n - 1, pi)
            all_peak_pos.update(range(si, ei + 1))
            if all_peak_pos:
                pos_list = list(all_peak_pos)
                for _ in range(15):
                    p_pos = pos_list[int(rng.integers(0, len(pos_list)))]
                    a_trial = a.copy()
                    delta = a[p_pos] * 0.1 * rng.standard_normal()
                    a_trial[p_pos] -= delta
                    if delta > 0:
                        add_to = int(rng.integers(0, n))
                        a_trial[add_to] += delta
                    else:
                        a_trial[p_pos] = 0  # already negative, clip
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss > 1e-14:
                        a_trial = a_trial / ss
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a_trial.copy()
                            break
        if verbose and iters % 50 == 0:
            print(f"    PE iter={iters}, spread={spread:.2e}, best={best_score:.8f}")
    return best_a, best_score


def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0)
    ss = a_new.sum()
    if ss > 1e-15:
        a_new = a_new / ss
    return a_new


def softmax_from_z(z):
    """Convert log-domain to probability vector."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lbgfs_smoothed(a0, time_limit, T, maxiter=2000):
    """L-BFGS on smoothed objective in log-space."""
    from scipy.optimize import minimize
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-30))
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e10
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        m = shifted.max()
        lse = m + np.log(np.exp(shifted - m).sum())
        return (2.0 * n / T) * np.exp(T * lse)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        m = shifted.max()
        w = np.exp(shifted - m)
        w = w / w.sum()
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return (2.0 * n / T) * grad_z

    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'gtol': 1e-10, 'disp': False})
        a_out = softmax_from_z(res.x)
        sc = score_normalized(a_out)
        if sc < best_score:
            best_a = a_out
            best_score = sc
    except Exception:
        pass
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining:
    - Active-set subgradient descent
    - LP-based direction finding
    - Coordinate descent
    - Peak balancing
    - L-BFGS with soft-max smoothing
    - Multiple sequence lengths
    
    Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = score_normalized(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        t = np.linspace(0, 2 * np.pi * 2, n_init)
        parent = parent + 0.1 * np.cos(t)
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        best_seq = parent.copy()
        best_score = score_normalized(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    def check_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0).astype(np.float64)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = score_normalized(a_norm)
        if sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a_norm)})")
        # Verify with evaluate_sequence
        try:
            eval_sc = float(evaluate_sequence(a_norm.tolist()))
            if np.isfinite(eval_sc) and eval_sc < best_score:
                best_score = eval_sc
                best_seq = a_norm.copy()
                print(f"  NEW BEST [verify {label}]: {eval_sc:.8f}")
        except Exception:
            pass

    # === Phase 1: L-BFGS smoothed (quick warm start) ===
    print(f"\n=== Phase 1: L-BFGS Warm Start (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(120.0, rem * 0.08)
        print(f"  L-BFGS T=500, n={n_parent}, alloc={t_alloc:.0f}s")
        a_res, sc_res = lbgfs_smoothed(best_seq, t_alloc, T=500, maxiter=2000)
        check_update(a_res, "lbgfs_warm")

    # === Phase 2: Active-set subgradient descent ===
    print(f"\n=== Phase 2: Active-Set Subgradient (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(300.0, rem * 0.18)
        print(f"  SG n={n_parent}, alloc={t_alloc:.0f}s")
        a_sg, sc_sg = subgradient_descent(best_seq, t_alloc, n_active=7, verbose=True)
        check_update(a_sg, "subgrad")

    # === Phase 3: LP direction + line search iterations ===
    print(f"\n=== Phase 3: LP Direction Finding (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(200.0, rem * 0.12)
        print(f"  LP+LS n={n_parent}, alloc={t_alloc:.0f}s")
        a_cur = best_seq.copy()
        t_lp0 = time.monotonic()
        lp_iters = 0
        while time.monotonic() - t_lp0 < t_alloc:
            lp_iters += 1
            conv_cur = fast_convolve(a_cur)
            max_cur = conv_cur.max()
            threshold = max_cur * 0.998
            active = np.where(conv_cur >= threshold)[0]
            if len(active) > 10:
                active = active[np.argsort(conv_cur[active])[::-1][:10]]
            if len(active) == 0:
                active = [int(np.argmax(conv_cur))]
            a_lp = lp_direction(a_cur, [int(k) for k in active])
            if a_lp is None:
                break
            # Line search between a_cur and a_lp
            best_alpha = 0.0
            best_sc_lp = score_normalized(a_cur)
            for alpha in [0.3, 0.5, 0.7, 0.9, 1.0, 0.1, 0.2]:
                a_try = (1 - alpha) * a_cur + alpha * a_lp
                a_try = np.maximum(a_try, 0)
                ss = a_try.sum()
                if ss < 1e-14:
                    continue
                a_try = a_try / ss
                sc_try = score_normalized(a_try)
                if sc_try < best_sc_lp - 1e-14:
                    best_sc_lp = sc_try
                    best_alpha = alpha
            if best_alpha > 0:
                a_cur = (1 - best_alpha) * a_cur + best_alpha * a_lp
                a_cur = np.maximum(a_cur, 0)
                a_cur = a_cur / a_cur.sum()
                check_update(a_cur, f"lp_iter{lp_iters}")
            else:
                # Try small step in LP direction
                a_try = a_cur + 0.05 * (a_lp - a_cur)
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                sc_try = score_normalized(a_try)
                if sc_try < score_normalized(a_cur) - 1e-14:
                    a_cur = a_try
                    check_update(a_cur, f"lp_small{lp_iters}")
                else:
                    break
        print(f"  LP iterations: {lp_iters}")

    # === Phase 4: Peak balancing ===
    print(f"\n=== Phase 4: Peak Balancing (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_alloc = min(150.0, rem * 0.1)
        print(f"  PE n={n_parent}, alloc={t_alloc:.0f}s")
        a_pe, sc_pe = peak_balancing(best_seq, t_alloc, n_peaks=12, verbose=True)
        check_update(a_pe, "peak_bal")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_alloc = min(200.0, rem * 0.12)
        print(f"  CD n={n_parent}, alloc={t_alloc:.0f}s")
        a_cd, sc_cd = coordinate_descent_fine(best_seq, t_alloc, n_attempts_per_iter=300, verbose=True)
        check_update(a_cd, "coord_desc")

    # === Phase 6: Second round of subgradient on best ===
    print(f"\n=== Phase 6: Subgradient Round 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_alloc = min(150.0, rem * 0.08)
        print(f"  SG2 n={n_parent}, alloc={t_alloc:.0f}s")
        a_sg2, _ = subgradient_descent(best_seq, t_alloc, n_active=10, verbose=False)
        check_update(a_sg2, "subgrad2")

    # === Phase 7: Try nearby lengths ===
    print(f"\n=== Phase 7: Nearby Lengths (best={best_score:.8f}) ===")
    nearby_ns = [n_parent - 2, n_parent - 1, n_parent + 1, n_parent + 2,
                 n_parent + 4, n_parent - 4]
    for alt_n in nearby_ns:
        if alt_n < 128 or alt_n > 2048:
            continue
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(100.0, rem * 0.08)
        init_alt = interpolate_to_length(best_seq, alt_n)
        print(f"  Subgrad n={alt_n}, alloc={t_alloc:.0f}s")
        a_alt, sc_alt = subgradient_descent(init_alt, t_alloc, n_active=5, verbose=False)
        check_update(a_alt, f"alt_n{alt_n}")
        # Quick CD on the best of alt
        rem2 = deadline - time.monotonic()
        if rem2 > 20:
            t_alloc2 = min(40.0, rem2 * 0.05)
            a_alt2, _ = coordinate_descent_fine(a_alt, t_alloc2, n_attempts_per_iter=150)
            check_update(a_alt2, f"alt_cd_n{alt_n}")

    # === Phase 8: Final L-BFGS polish ===
    print(f"\n=== Phase 8: Final L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(60.0, rem * 0.05)
        print(f"  L-BFGS final, alloc={t_alloc:.0f}s")
        a_fin, sc_fin = lbgfs_smoothed(best_seq, t_alloc, T=2000, maxiter=1500)
        check_update(a_fin, "lbgfs_final")

    # === Phase 9: Final coordinate descent polish ===
    print(f"\n=== Phase 9: Final CD Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(60.0, rem * 0.08)
        print(f"  Final CD, alloc={t_alloc:.0f}s")
        a_fin2, _ = coordinate_descent_fine(best_seq, t_alloc, n_attempts_per_iter=500)
        check_update(a_fin2, "final_cd")

    # === Phase 10: One more peak balancing round ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_alloc = min(30.0, rem * 0.5)
        a_fin3, _ = peak_balancing(best_seq, t_alloc, n_peaks=15)
        check_update(a_fin3, "final_pe")

    # Final verification
    if best_seq is not None:
        final_score = score_normalized(best_seq)
        try:
            eval_final = float(evaluate_sequence(best_seq.tolist()))
            if np.isfinite(eval_final):
                final_score = min(final_score, eval_final)
        except Exception:
            pass
        print(f"\nFINAL SCORE: {final_score:.8f} (n={len(best_seq)})")

    if best_seq is None:
        best_seq = np.ones(768) / 768

    return [float(x) for x in best_seq]
