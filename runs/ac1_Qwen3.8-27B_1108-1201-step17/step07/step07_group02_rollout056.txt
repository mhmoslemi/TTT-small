    """Compute H @ a where H[i,j] = lambda_vec[i+j]."""
    n = len(a)
    # H @ a has entries (H@a)[i] = sum_j lambda[i+j]*a[j] for i=0..n-1
    # This is a cross-correlation
    # lambda has indices 0..2n-2, a has indices 0..n-1
    # (H@a)[i] = sum_{j=0}^{n-1} lambda[i+j] * a[j]
    # = cross-correlation of lambda[i:i+n] with a at lag 0
    # Better: use FFT

    # Then (H@a)[i] = sum_j lambda[i+j] * a[j] = sum_j lambda[i+j] * ra[n-1-j]

    # = sum_k lambda[i+n-1-k] * ra[k]

        a = solve_inner_fista(lambda_vec, n, a_init=a, n_iters=n_fista_iters, L=2.0)

        # Compute convolution and score
        b_new = fast_convolve(a)
        max_b = b_new.max()
        score = 2.0 * n * max_b

        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()

        # Update lambda via exponentiated gradient
        eta = np.sqrt(np.log(K) / t)
        # log-domain update for stability
        log_lam = np.log(np.maximum(lambda_vec, 1e-300)) + b_new / eta
        log_lam -= log_lam.max()
        lambda_vec = np.exp(log_lam)
        lambda_vec /= lambda_vec.sum()

        it_count += 1
        if (it_count + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DD iter={it_count+1} ({elapsed:.0f}s): score={best_score:.6f}, max_b={max_b:.8f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit):
    """Subgradient method with diminishing steps for final polishing."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for t in range(1, 1000000):
        if time.monotonic() > t0 + time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b

        # Find active (near-max) indices
        threshold = max_b - 0.01 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute subgradient: average of gradient contributions from active lags
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Subgradient step with diminishing step size
        step = 0.5 / np.sqrt(t + 1)
        a_new = a - step * g
        a = simplex_project(a_new)

        sc = 2.0 * n * fast_convolve(a).max()
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a.copy()

        if (t + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={t+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate using the provided evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]

    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. L-BFGS log-sum-exp for coarse optimization
    2. Dual decomposition (exponentiated gradient + FISTA inner QP) for main optimization
    3. Subgradient polishing
    4. Final L-BFGS at very low temperature
    5. Try alternative sequence lengths if time permits
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(seed)

    # Get initial sequence from parent
    a = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            a = p / p.sum()
    except Exception:
        pass

    if a is None:
        a = np.ones(768) / 768

    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    print(f"Initial: n={n}, score={best_score:.6f}")

    # Phase 1: L-BFGS for coarse optimization (300s)
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 300.0)
    if t_alloc > 30:
        print(f"\nPhase 1: L-BFGS coarse (budget={t_alloc:.0f}s)")
        t0 = time.monotonic()
        a_lb = lbgfs_smooth(best_a, t_alloc, T_start=100.0, T_end=1.0, maxiter=5000)
        sc = compute_score(a_lb)
        if sc < best_score:
            best_score = sc
            best_a = a_lb.copy()
            print(f"  L-BFGS coarse: {best_score:.6f}")
        else:
            print(f"  L-BFGS coarse: no improvement ({sc:.6f} vs {best_score:.6f})")
        print(f"  Elapsed: {time.monotonic()-t0:.0f}s")

    # Phase 2: Dual decomposition (main optimization, ~1000s)
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.55, 1000.0)
    if t_alloc > 30:
        print(f"\nPhase 2: Dual Decomposition (budget={t_alloc:.0f}s)")
        t0 = time.monotonic()
        a_dd, sc_dd = dual_decompose(best_a, t_alloc, n_fista_iters=250)
        if sc_dd < best_score:
            best_score = sc_dd
            best_a = a_dd.copy()
            print(f"  Dual Decomp: improved to {best_score:.6f}")
        else:
            print(f"  Dual Decomp: no improvement ({sc_dd:.6f} vs {best_score:.6f})")
        print(f"  Elapsed: {time.monotonic()-t0:.0f}s")

    # Phase 3: Subgradient polishing (200s)
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.1, 200.0)
    if t_alloc > 20:
        print(f"\nPhase 3: Subgradient Polish (budget={t_alloc:.0f}s)")
        t0 = time.monotonic()
        a_sg, sc_sg = subgradient_polish(best_a, t_alloc)
        if sc_sg < best_score:
            best_score = sc_sg
            best_a = a_sg.copy()
            print(f"  SubGrad: improved to {best_score:.6f}")
        else:
            print(f"  SubGrad: no improvement ({sc_sg:.6f})")
        print(f"  Elapsed: {time.monotonic()-t0:.0f}s")

    # Phase 4: Final L-BFGS at very low temperature (300s)
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 300.0)
    if t_alloc > 20:
        print(f"\nPhase 4: L-BFGS fine (budget={t_alloc:.0f}s)")
        t0 = time.monotonic()
        a_fine = lbgfs_smooth(best_a, t_alloc, T_start=5.0, T_end=0.05, maxiter=8000)
        sc = compute_score(a_fine)
        if sc < best_score:
            best_score = sc
            best_a = a_fine.copy()
            print(f"  L-BFGS fine: improved to {best_score:.6f}")
        else:
            print(f"  L-BFGS fine: no improvement ({sc:.6f})")
        print(f"  Elapsed: {time.monotonic()-t0:.0f}s")

    # Phase 5: Second dual decomposition round with more iterations
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(rem * 0.4, 400.0)
        print(f"\nPhase 5: Dual Decomp round 2 (budget={t_alloc:.0f}s)")
        t0 = time.monotonic()
        a_dd2, sc_dd2 = dual_decompose(best_a, t_alloc, n_fista_iters=400)
        if sc_dd2 < best_score:
            best_score = sc_dd2
            best_a = a_dd2.copy()
            print(f"  DD2: improved to {best_score:.6f}")
        else:
            print(f"  DD2: no improvement ({sc_dd2:.6f})")
        print(f"  Elapsed: {time.monotonic()-t0:.0f}s")

    # Phase 6: Try alternative lengths if time permits
    rem = deadline - time.monotonic()
    if rem > 120:
        print(f"\nPhase 6: Alternative lengths (budget={rem:.0f}s)")
        alt_ns = [int(n * 1.25), int(n * 0.75), int(n * 1.5)]
        alt_ns = [x for x in alt_ns if 256 <= x <= 4096 and x != n]
        for alt_n in alt_ns[:2]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(rem * 0.4, 100.0)
            if t_alloc < 30:
                break
            print(f"  Trying n={alt_n} (budget={t_alloc:.0f}s)")
            init_alt = interpolate_sequence(best_a, alt_n)
            a_alt = lbgfs_smooth(init_alt, t_alloc * 0.5, T_start=50.0, T_end=1.0, maxiter=2000)
            sc_alt = compute_score(a_alt)
            if sc_alt < best_score:
                best_score = sc_alt
                best_a = a_alt.copy()
                print(f"    n={alt_n}: improved to {best_score:.6f}")
                # Also try dual decomp at this length
                rem2 = deadline - time.monotonic()
                if rem2 > 40:
                    a_dd_alt, sc_dd_alt = dual_decompose(best_a, min(rem2 * 0.5, 60.0), n_fista_iters=150)
                    if sc_dd_alt < best_score:
                        best_score = sc_dd_alt
                        best_a = a_dd_alt.copy()
                        print(f"    n={alt_n} DD: improved to {best_score:.6f}")
            else:
                print(f"    n={alt_n}: no improvement ({sc_alt:.6f})")

    # Final: subgradient micro-polish with remaining time
    rem = deadline - time.monotonic()
    if rem > 15:
        t_alloc = min(rem * 0.6, 60.0)
        print(f"\nFinal Polish: SubGrad (budget={t_alloc:.0f}s)")
        a_final, sc_final = subgradient_polish(best_a, t_alloc)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()

    # Final verification
    final_score = compute_score(best_a)
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {final_score:.6f} (n={len(best_a)})")
    print(f"Target: 1.5027, Achieved: {'YES' if final_score < 1.5027 else 'NO'}")
    print(f"{'='*50}")

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    return result
