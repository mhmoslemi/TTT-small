import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)≈1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def convolve_xy(a, b):
    """Cross-correlation a * b (full linear convolution)."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:n]

    it = 0
    no_improve = 0

            no_improve += 1

        k_star = int(np.argmax(b))

def lp_descent_step(a, time_limit=None):
    """
    Solve LP to find a descent direction.
    Maximize sum(b_j) s.t. (a*b)_k <= M, b >= 0.
    Returns the normalized direction b/sum(b).
    """
    n = len(a)
    b_conv = fast_convolve(a)
    M = b_conv.max()
    
    # LP: max sum(b_j) s.t. A_ub @ b <= M, b >= 0
    # A_ub has shape (2n-1, n), A_ub[k,j] = a[j] if 0 <= k-j < n
    c = -np.ones(n)
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = a[i]
    
    b_ub = np.full(2*n-1, M)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, 
                      method='highs', options={'time_limit': 30})
        if res.success:
            b_dir = res.x
            s = b_dir.sum()
            if s > 1e-10:
                return normalize(b_dir)
    except Exception:
        pass
    
    return None


def pair_redistribute(a, time_limit, verbose=False, label=""):
    """
    Targeted pair redistribution: find balanced pairs at the peak and unbalance them.
    """
    n = len(a)
    a = normalize(a)
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        current_score = 2.0 * n * b[k_star]
        
        # Find all pairs (i, k_star-i) and their products
        pairs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n and i <= j:  # avoid double counting
                product = a[i] * a[j]
                if i != j:
                    pairs.append((product, i, j))
                else:
                    pairs.append((product, i, i))
        
        # Sort by product (most balanced first - these contribute most)
        pairs.sort(reverse=True)
        
        improved = False
        for product, i, j in pairs[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            if i == j:
                # Single element: try reducing it
                for factor in [0.9, 0.8, 0.7]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = normalize(a_trial)
                    sc = score_from_a(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
            else:
                # Pair: try making them more unequal
                # Move mass from the larger to the smaller (or vice versa)
                if a[i] > a[j]:
                    large, small = i, j
                else:
                    large, small = j, i
                
                for frac in [0.3, 0.5, 0.7]:
                    a_trial = a.copy()
                    move = a_trial[large] * frac
                    a_trial[large] -= move
                    a_trial[small] += move
                    a_trial = normalize(a_trial)
                    sc = score_from_a(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved:
            break
    
    return best_a, best_score


def momentum_subgrad(a_init, time_limit, lr=0.01, momentum=0.9, verbose=False, label=""):
    """Momentum-based subgradient descent on the simplex."""
    n = len(a_init)
    a = normalize(a_init)
    best_a = a.copy()
    best_score = score_from_a(a)
    velocity = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and it < 20000:
        it += 1
        
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        
        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Subgradient (from active peak)
        k_star = int(np.argmax(b))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Project to tangent space
        grad = grad - np.mean(grad)
        
        # Momentum update
        velocity = momentum * velocity - lr * grad
        velocity = velocity - np.mean(velocity)
        
        a_new = a + velocity
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check if it improved
        trial_score = score_from_a(a_new)
        if trial_score < current_score:
            a = a_new
        else:
            # Accept with probability (simulated annealing)
            T = 0.01 * max(0, 1 - it/5000)
            if T > 0 and np.random.random() < np.exp(-(trial_score - current_score) / T):
                a = a_new
        
        # Decay learning rate
        lr *= 0.9995
        
        if no_improve > 200:
            break
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} SubGrad it={it} ({elapsed:.0f}s): "
                  f"score={current_score:.7f} best={best_score:.7f} lr={lr:.6f}")
    
    return best_a, best_score


def check_and_update(a, label, best_seq, best_score):
    """Evaluate and update best."""
    if a is None:
        return best_seq, best_score
    a = normalize(a)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a)})")
    return best_seq, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining Newton, LP, coordinate descent,
    and momentum subgradient methods with diversity.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass
    
    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    
    if best_seq is None:
        # Fallback: uniform initialization
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Uniform init: n={len(best_seq)}, score={best_score:.7f}")
    
    n = len(best_seq)
    
    # === Phase 1: Newton Optimization (primary) ===
    print(f"\n=== Phase 1: Newton Optimization (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.3, 400.0)
    print(f"  Budget: {t_alloc:.0f}s, n={n}")
    a_newton, _ = newton_optimize(best_seq, t_alloc, max_iters=10000,
                                   verbose=True, label="N1")
    best_seq, best_score = check_and_update(a_newton, "Newton1", best_seq, best_score)
    
    # === Phase 2: LP-guided steps ===
    print(f"\n=== Phase 2: LP-Guided (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(rem * 0.2, 200.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a = best_seq.copy()
        t0 = time.monotonic()
        no_lp_improve = 0
        while time.monotonic() - t0 < t_alloc and no_lp_improve < 3:
            # Solve LP for direction
            b_dir = lp_descent_step(a)
            if b_dir is not None:

                    sc = score_from_a(a_trial)

                        a = a_trial
                        no_lp_improve = 0
                        break
                else:
                    no_lp_improve += 1
            else:
                no_lp_improve += 1
            
            # Newton polish after LP step
            rem_inner = t_alloc - (time.monotonic() - t0)
            if rem_inner > 30:
                a, _ = newton_optimize(a, min(rem_inner, 60.0), max_iters=2000,
                                       verbose=False, label="N_lp")
            
            best_seq, best_score = check_and_update(a, "LP", best_seq, best_score)
    
    # === Phase 3: Newton (second round) ===
    print(f"\n=== Phase 3: Newton Round 2 (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()

        print(f"  Budget: {t_alloc:.0f}s")

                                        verbose=True, label="N2")
