Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based best-response iterations
with adaptive step sizes, combined with gradient methods and coordinate descent polishing.
The LP approach solves an exact best-response problem at each iteration, providing
a principled path toward the optimal solution. Multi-phase strategy with length exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csr_matrix
import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())

def build_conv_matrix(f):
    """Build the convolution matrix T where (T @ b)[k] = conv(f,b)[k]."""
    n = len(f)
    T = lil_matrix((2*n-1, n))
    for i in range(n):
        for k in range(i, i + n):
            T[k, i] = f[k-i]
    return csr_matrix(T)

def lp_best_response(f, M, solver=cp.ECOS):
    """Solve the LP to find the best response to f."""
    n = len(f)
    b_var = cp.Variable(n, non_neg=True)
    T = build_conv_matrix(f)
    constraints = [T @ b_var <= M * np.ones(2*n-1)]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver)
    except:
        try:
            prob.solve(solver=cp.SCS)
        except:
            return None
    if b_var.value is None:
        return None
    b_val = np.maximum(b_var.value, 0)
    return b_val

def lp_iteration(f, step_size, solver=cp.ECOS):
    """One LP-based best response iteration."""
    b = fast_convolve(f)
    M = b.max()
    b_response = lp_best_response(f, M, solver=solver)
    if b_response is None:
        return f, M
    s = b_response.sum()
    if s < 1e-15:
        return f, M
    b_norm = b_response / s
    f_new = (1 - step_size) * f + step_size * b_norm
    f_new = np.maximum(f_new, 0)
    f_new = f_new / f_new.sum()
    return f_new, M

def lp_optimize(f_init, time_limit, verbose=True, label=""):
    """LP-based best response optimization with adaptive step size."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 1e-15).astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_from_a(f)
    t0 = time.monotonic()
    step_size = 0.5
    no_improve = 0
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        f_new, M = lp_iteration(f, step_size)
        
        sc = score_from_a(f_new)
        if sc < best_score - 1e-15:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            # Increase step size slightly
            step_size = min(step_size * 1.02, 0.9)
        else:
            no_improve += 1
            # Decrease step size
            step_size = max(step_size * 0.95, 0.05)
            f = f_new  # Still use the new point, just don't update best
        
        if verbose and iteration % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} LP iter={iteration} ({elapsed:.0f}s): "
                  f"step={step_size:.3f} score={sc:.7f} best={best_score:.7f}")
        
        if no_improve > 50:
            # Try a random perturbation
            rng = np.random.default_rng(iteration * 31)
            noise = rng.standard_normal(n) * 0.01
            f = best_f + noise
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            no_improve = 0
            step_size = 0.3
    
    return best_f, best_score

4. Coordinate descent polishing (for final refinement)
5. Length exploration (to find the optimal n)

import numpy as np

from scipy.sparse import lil_matrix, csr_matrix

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())

def build_conv_matrix(f):
    """Build sparse matrix T where (T @ b)[k] = conv(f,b)[k]."""
    n = len(f)
    T = lil_matrix((2*n-1, n))
    for i in range(n):
        for k in range(i, i + n):
            T[k, i] = f[k-i]
    return csr_matrix(T)

def lp_best_response(f, M, topK=None, solver='ECOS'):
    """Solve LP to find best response to f with peak bound M."""
    n = len(f)
    b_var = cp.Variable(n, non_neg=True)
    T = build_conv_matrix(f)
    
    if topK is not None:
        b_conv = fast_convolve(f)
        top_indices = np.argsort(b_conv)[::-1][:topK]
        constraints = [T[k, :] @ b_var <= M for k in top_indices]
    else:
        constraints = [T @ b_var <= M * np.ones(2*n-1)]
    
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except:
        try:
            prob.solve(solver='SCS', verbose=False)
        except:
            return None
    if b_var.value is None:
        return None
    return np.maximum(b_var.value, 0)

def lp_optimize(f_init, time_limit, verbose=True, label="LP", topK=None):
    """LP-based best response optimization."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 1e-15).astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_from_a(f)
    t0 = time.monotonic()
    step_size = 0.3
    no_improve = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        b = fast_convolve(f)
        M = b.max()
        
        g = lp_best_response(f, M, topK=topK)
        if g is None:
            break
        
        s = g.sum()
        if s < 1e-15:
            break
        g = g / s
        
        f_new = (1 - step_size) * f + step_size * g
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        sc = score_from_a(f_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_f = f_new.copy()
            no_improve = 0
            step_size = min(step_size * 1.01, 0.8)
        else:
            no_improve += 1
            step_size = max(step_size * 0.97, 0.05)
        
        f = f_new
        
        if no_improve > 100:
            rng = np.random.default_rng(iteration * 17)
            noise = rng.standard_normal(n) * 0.02
            f = best_f + noise
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            no_improve = 0
            step_size = 0.3
        
        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={iteration} ({elapsed:.0f}s): "
                  f"step={step_size:.3f} score={sc:.7f} best={best_score:.7f}")
    
    return best_f, best_score

def exact_subgradient_step(f, eta, k=None):
    """One multiplicative weights step using exact peak subgradient."""
    n = len(f)
    b = fast_convolve(f)
    if k is None:
        k = int(np.argmax(b))
    
    # Subgradient: for each j, grad[j] = a[k-j] + a[j-k] (with zero padding)
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= k - j < n:
            val += f[k - j]
        if 0 <= j - k < n:
            val += f[j - k]
        grad[j] = val
    
    f_new = f * np.exp(-eta * grad)
    f_new = np.maximum(f_new, 1e-300)
    s = f_new.sum()
    if s < 1e-15:
        return f, k
    f_new = f_new / s
    return f_new, k

def multi_peak_subgradient_step(f, eta, K=5):
    """Step using weighted combination of top K peaks."""
    n = len(f)
    b = fast_convolve(f)
    b_max = b.max()
    
    # Get top K peaks
    top_k = np.argsort(b)[::-1][:K]
    # Weight by how close to max
    weights = b[top_k] / b_max
    weights = weights / weights.sum()
    
    f_new = f.copy()
    for k, w in zip(top_k, weights):
        grad = np.zeros(n)
        for j in range(n):
            val = 0.0
            if 0 <= k - j < n:
                val += f[k - j]
            if 0 <= j - k < n:
                val += f[j - k]
            grad[j] = val
        f_new = f_new * np.exp(-eta * w * grad)
    
    f_new = np.maximum(f_new, 1e-300)
    s = f_new.sum()
    if s < 1e-15:
        return f
    return f_new / s

def mirror_descent_optimize(f_init, time_limit, T_start=0.01, T_end=0.0005,
                            eta_start=0.02, eta_end=0.0005, max_iters=20000,
                            verbose=True, label="MD"):
    """Mirror descent with temperature annealing."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 1e-15).astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_from_a(f)
    t0 = time.monotonic()
    stuck_count = 0
    
    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, it / (0.7 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress
        
        b = fast_convolve(f)
        current_score = 2.0 * n * b.max()
        
        if current_score < best_score - 1e-15:
            best_score = current_score
            best_f = f.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        # Softmax weights
        scaled = b / T
        m = scaled.max()
        exp_scaled = np.exp(scaled - m)
        w = exp_scaled / exp_scaled.sum()
        
        # Gradient via FFT
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        F = rfft(f, fft_size)
        correl = irfft(np.conj(W) * F, fft_size)[:n]
        grad = 4.0 * n * correl
        grad = grad - np.dot(f, grad)  # Project to tangent space
        
        # Multiplicative update
        log_update = -eta * grad
        log_update = np.clip(log_update, -3.0, 3.0)
        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-300)
        s = f.sum()
        if s < 1e-15:
            f = best_f.copy()
        else:
            f = f / s
        
        if stuck_count > 300:
            rng = np.random.default_rng(it * 7919)
            noise = rng.standard_normal(n) * 0.02
            f = best_f + noise
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            stuck_count = 0
            eta = min(eta * 3.0, eta_start * 0.5)
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} ({elapsed:.0f}s): "
                  f"T={T:.5f} score={current_score:.7f} best={best_score:.7f}")
    
    return best_f, best_score

def mw_exact_optimize(f_init, time_limit, eta_start=0.005, decay=0.9998,
                      max_iters=50000, verbose=True, label="MW"):
    """Multiplicative weights with exact peak subgradient."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 1e-15).astype(np.float64)

    best_f = f.copy()

    t0 = time.monotonic()

        if time.monotonic() - t0 > time_limit:

        f, k = exact_subgradient_step(f, eta)
        
        sc = score_from_a(f)
        if sc < best_score - 1e-15:
            best_score = sc
            best_f = f.copy()
        
        eta *= decay
    
    return best_f, best_score

def coordinate_descent_polish(f_init, time_limit, verbose=True, label="CD"):
    """Coordinate descent with proportional redistribution."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 1e-15).astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_from_a(f)
    t0 = time.monotonic()
    
    factors = [0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.3]
    no_improve = 0
    max_no_improve = 100
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve = 0
        improved = False
        order = np.random.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for fac in factors:
                f_trial = f.copy()
                old_val = f_trial[i]
                new_val = old_val * fac
                f_trial[i] = new_val
                diff = f_trial.sum() - 1.0
                if abs(diff) < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = f_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                f_trial[mask] = others - diff * others / others_sum
                if f_trial.min() < -1e-12:
                    continue
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
                sc = score_from_a(f_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    f = f_trial.copy()
                    best_f = f.copy()
                    improved = True
                    break
        
        if not improved:
            no_improve += 1
        if verbose and no_improve % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} ({elapsed:.0f}s): best={best_score:.7f} no_improve={no_improve}")
    
    return best_f, best_score

def peak_shave(f_init, time_limit, verbose=True, label="PS"):
    """Targeted peak shaving."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 1e-15).astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_from_a(f)
    t0 = time.monotonic()
    
    for step in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(f)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        
        if b[peak_k] <= best_score / (2.0 * n) + 1e-16:
            break
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = f[idx] * f[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:5]
        
        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            j = peak_k - i
            if f[i] < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7]:
                f_trial = f.copy()
                old_val = f_trial[i]
                f_trial[i] = old_val * factor
                diff = old_val - f_trial[i]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = f_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                f_trial[mask] = others + diff * others / others_sum
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
                sc = score_from_a(f_trial)
                if sc < best_score - 1e-15:
                    f = f_trial
                    best_score = sc
                    best_f = f.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try reducing both i and j
            for ci in top_idx[:3]:
                i = int(idx[ci])
                j = peak_k - i
                if i == j or f[j] < 1e-14:
                    continue
                for f1 in [0.9, 0.8]:
                    for f2 in [0.9, 0.8]:
                        f_trial = f.copy()
                        f_trial[i] *= f1
                        f_trial[j] *= f2
                        f_trial = np.maximum(f_trial, 0)
                        f_trial = f_trial / f_trial.sum()
                        sc = score_from_a(f_trial)

                            f = f_trial
                            best_score = sc
                            best_f = f.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break
    
    return best_f, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    s = a.sum()

        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. LP-based best response iterations (principled, convergent)
2. Mirror descent with temperature-annealed softmax (fast gradient method)
3. Multiplicative weights with exact peak subgradient (robust local optimization)
4. Peak shaving and coordinate descent (fine-grained polishing)
5. Length exploration via interpolation (escape local minima at different n)
Key novelty: combines exact LP best-response with smooth gradient methods,
using adaptive step sizes and top-K constraint relaxation for speed.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csr_matrix
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def compute_peak_gradient(f, k):
    """Compute subgradient of max(conv(f,f)) at peak position k.
    grad[j] = 2*f[k-j] if 0 <= k-j < n, else 0."""
    n = len(f)
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    idx = np.arange(js, je + 1)
    grad[idx] = 2.0 * f[k - idx]
    return grad


def compute_smooth_gradient(f, b, T):
    """Compute smoothed gradient using softmax weights over b."""
    n = len(f)
    scaled = b / T
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()
    # Gradient: 4n * cross_correlation(w, f)
    fft_size = 1 << (3 * n - 1).bit_length()
    W = rfft(w, fft_size)
    F = rfft(f, fft_size)
    correl = irfft(np.conj(W) * F, fft_size)[:n]
    return 4.0 * n * correl


def lp_best_response(f, M, topK=50, solver='ECOS'):
    """Solve LP to find best response to f with peak bound M on top-K peaks."""
    n = len(f)
    b_conv = fast_convolve(f)
    top_indices = np.argsort(b_conv)[::-1][:topK]

    b_var = cp.Variable(n, non_neg=True)

    # Build constraint matrix for top K rows only
    T_rows = np.zeros((topK, n))
    for idx_i, k in enumerate(top_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        i_range = np.arange(js, je + 1)
        T_rows[idx_i, i_range] = f[k - i_range]

    constraints = [T_rows @ b_var <= M * np.ones(topK)]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='SCS', verbose=False)
        except Exception:
            return None
    if b_var.value is None:
        return None
    return np.maximum(b_var.value, 0)


def lp_optimize(f_init, time_limit, topK=50, verbose=True, label="LP"):
    """LP-based best response optimization with adaptive step size."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 1e-15).astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_from_a(f)
    t0 = time.monotonic()
    step_size = 0.4
    no_improve = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        b = fast_convolve(f)
        M = b.max()

        g = lp_best_response(f, M, topK=topK)
        if g is None:
            break

        s = g.sum()
        if s < 1e-15:
            break
        g = g / s

        f_new = (1.0 - step_size) * f + step_size * g
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        sc = score_from_a(f_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_f = f_new.copy()
            no_improve = 0
            step_size = min(step_size * 1.01, 0.85)
        else:
            no_improve += 1
            step_size = max(step_size * 0.96, 0.05)

        f = f_new

        if no_improve > 80:
            rng = np.random.default_rng(iteration * 31 + 7)
            noise = rng.standard_normal(n) * 0.015
            f = best_f + noise
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            no_improve = 0
            step_size = 0.3

        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={iteration} ({elapsed:.0f}s): "
                  f"step={step_size:.3f} score={sc:.7f} best={best_score:.7f}")

    return best_f, best_score


def mirror_descent_optimize(f_init, time_limit, T_start=0.01, T_end=0.0005,
                            eta_start=0.02, eta_end=0.0005, max_iters=30000,
                            verbose=True, label="MD"):
    """Mirror descent with temperature annealing on the simplex."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 1e-15).astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_from_a(f)
    t0 = time.monotonic()
    stuck_count = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (0.7 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress

        b = fast_convolve(f)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_f = f.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        grad = compute_smooth_gradient(f, b, T)
        grad = grad - np.dot(f, grad)

        log_update = -eta * grad
        log_update = np.clip(log_update, -3.0, 3.0)
        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-300)
        s = f.sum()
        if s < 1e-15:
            f = best_f.copy()
        else:
            f = f / s

        if stuck_count > 200:
            rng = np.random.default_rng(it * 7919)
            noise = rng.standard_normal(n) * 0.02
            f = best_f + noise
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            stuck_count = 0
            eta = min(eta * 3.0, eta_start * 0.5)

        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} ({elapsed:.0f}s): "
                  f"T={T:.5f} score={current_score:.7f} best={best_score:.7f}")

    return best_f, best_score
