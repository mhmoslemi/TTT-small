import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():

        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):

def compute_smooth_gradient(a, top_k=50, temp_scale=0.01):
    """Compute softmax-weighted gradient of max(conv(a,a))."""
    n = len(a)
    C = fast_convolve(a)
    k_actual = min(top_k, len(C))
    # Get top peaks
    peak_indices = np.argpartition(C, -k_actual)[-k_actual:]
    peak_vals = C[peak_indices]
    # Adaptive temperature
    T = max(temp_scale * peak_vals.max(), 1e-15)
    # Softmax weights
    w = np.exp((peak_vals - peak_vals.max()) / T)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(k_actual)
        w[np.argmax(peak_vals)] = 1.0
    else:
        w = w / w_sum
    
    # Compute gradient
    grad = np.zeros(n)
    for idx in range(k_actual):
        if w[idx] < 1e-14:
            continue
        k = int(peak_indices[idx])
        # d C[k] / d a[i] = 2*a[k-i] if 0 <= k-i <= n-1
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[idx] * a[k - j]
    return grad, C.max()

        grad, current_max = compute_smooth_gradient(a, top_k=30, temp_scale=temp_scale)

        log_update = -eta * grad
        # Clip to prevent numerical issues

        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

                # Reshuffle

                a = a + noise * a

                a = a / a.sum()
        
        a = a_new
        
        # Adaptive temperature
        temp_scale *= 0.9995
        
        if verbose and it % 100 == 0:

def coord_descent_with_transfers(a0, time_limit, max_sweeps=200):
    """Coordinate descent with both scaling and pairwise mass transfers."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[i]
            if old_val < 1e-12:
                continue
            
            # Strategy 1: Scale coordinate
            for factor in [0.8, 0.9, 0.95, 1.05, 1.1, 1.2]:
                a[i] = old_val * factor
                a = np.maximum(a, 0)
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s_new
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    old_val = a[i]
                    break
                else:
                    a[i] = old_val
                    a = a / a.sum()
            
            # Strategy 2: Pairwise transfer (move mass from i to j)
            if not improved and old_val > 1e-10:
                # Choose j that's "far" from i (to reduce central peak)
                j_candidates = [(i + n//2) % n, (i + n//3) % n, (i + 2*n//3) % n]
                for j in j_candidates:
                    if j == i or a[j] < 0:
                        continue
                    for frac in [0.1, 0.3, 0.5]:
                        transfer = old_val * frac
                        a_trial = a.copy()
                        a_trial[i] = old_val - transfer
                        a_trial[j] = a[j] + transfer
                        if a_trial[i] < 0:
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            
            if improved:
                a = best_a.copy() if not improved else a
        
        if not improved and sweep > 10:
            break
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def build_conv_matrix(a_fixed):
    """Build matrix T such that T @ b = conv(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
