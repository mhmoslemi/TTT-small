"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play
(cutset game) with multiplicative updates, followed by local refinement.
The key idea: formulate as a zero-sum game where Player A chooses the sequence
and Player B chooses the "attack" direction. The Nash equilibrium gives the optimum.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Compute 2n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(a):
    """Build the convolution matrix C where (C @ b)[k] = (a*b)[k]."""
    n = len(a)
    C = np.zeros((2*n-1, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C


def solve_dual_lp(a, M):
    """Solve: max sum(b) s.t. b>=0, (a*b)[k] <= M for all k."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    C = build_conv_matrix(a)
    constraints = [C @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if b.value is not None:
            return np.maximum(np.asarray(b.value), 0)
    except Exception:
        pass
    
    try:
        prob.solve(solver=cp.SCS, verbose=False)

        pass
    
    return None


def fictitious_play(a0, time_limit, n_iter_max=5000, t_init=0.5, t_decay=0.9995):
    """LP-based fictitious play for the cutset game."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    a = a * S / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a / S)
    
    t0 = time.monotonic()
    t = t_init
    
    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute M
        conv_a = fast_convolve(a)
        M = conv_a.max()
        
        if M < 1e-15:
            break
        
        # Solve dual LP
        b = solve_dual_lp(a, M)
        if b is None or b.sum() < 1e-10:
            break
        
        # Rescale b to have sum S
        g = b * S / b.sum()
        
        # Update
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new * S / s
        
        # Check improvement
        score = compute_score(a_new / S)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        t *= t_decay
    
    return best_a

OK the formulation looks correct.

Now, one important optimization: instead of rebuilding the constraint matrix C every iteration, I can reuse the matrix structure and just update the values. But since C depends on `a` which changes every iteration, I do need to rebuild it. For n=768, building a 1535 x 768 matrix takes about 1.2 million assignments, which is fast in numpy.

Actually, I can build C more efficiently:
```python
def build_conv_matrix_fast(a):
    n = len(a)
    # C[k, j] = a[k-j] if 0 <= k-j < n
    # This is a banded matrix
    C = np.zeros((2*n-1, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C
```

For n=768, this is 768 array assignments of length 768, which is fast.

Alternatively, I can use a more vectorized approach:
```python

    # k-j = offset, so value = a[offset]

```

Let me just use the simple loop version, it's fast enough.

Now let me think about the overall algorithm more carefully.

**Key insight for improvement**: The previous approach achieved 1.504883. To get below 1.5027, I need a ~0.1% improvement. This is a small but meaningful gap.

The LP-based fictitious play is theoretically guaranteed to converge to the optimum (of the game, which equals the optimum of our problem). The question is whether it converges fast enough in the given time.

With HiGHS solving each LP in ~1ms, and the overhead of building the matrix and doing the update being maybe another 1-2ms, we're looking at about 3-5ms per iteration. In 600 seconds, that's 120,000-200,000 iterations. This should be more than enough for convergence.

Let me also think about the step size. In fictitious play, the step size affects convergence speed:
- Too large: oscillation
- Too small: slow convergence
- Optimal: depends on the problem

A common choice is `t_k = 1/(k+1)` (harmonic) or `t_k = c/sqrt(k)`.

For the first few iterations, a larger step helps make quick progress. Then we switch to smaller steps for fine-tuning.

Let me use a piecewise step size: `t_k = min(0.5, 1/(k+1))` for the first 100 iterations, then `t_k = 0.01 / (1 + 0.001*k)` afterwards.

Actually, let me think about this differently. The fictitious play update is:
`a_{k+1} = (1-t_k) * a_k + t_k * g_k`

where `g_k` is the LP solution (best response of the dual player).

If we think of this as a convex combination, the "average" of the iterates converges to the equilibrium. So we should track the time-average:
`a_avg = (1/(K+1)) * sum_{k=0}^{K} a_k`

And evaluate the score on `a_avg` rather than on the latest iterate.

This is a standard technique in fictitious play / no-regret algorithms.

Let me also consider: should I normalize `a` after each update? The LP gives `g` with some sum, and I rescale it to have sum = S = sqrt(2n). Then the convex combination `(1-t)*a + t*g` automatically has sum = S (since both a and g have sum S). So no re-normalization needed (beyond the non-negativity clip).

Wait, after clipping to non-negative, the sum might decrease. So I do need to renormalize.

Let me now write the final, clean implementation:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play
with HiGHS solver, followed by local refinement.
The cutset game formulation ensures convergence to the global optimum.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    na = len(a)
    nb = len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Compute 2n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(a):
    """Build matrix C where (C @ b)[k] = (a*b)[k]."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def solve_lp(a, M):
    """Solve: max sum(b) s.t. b>=0, C@a_b <= M."""
    n = len(a)
    c = -np.ones(n)
    C = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None


def fictitious_play_lp(a_init, time_limit, verbose=True):
    """
    LP-based fictitious play for the cutset game.
    Returns the best solution found.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.maximum(a_init.astype(np.float64), 0).copy()
    a = a * S / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a / S)
    
    # Running average (for no-regret convergence)
    a_avg = a.copy().astype(np.float64)
    
    t0 = time.monotonic()
    it = 0
    last_print = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute max convolution
        conv_a = fast_convolve(a)
        M = conv_a.max()
        
        if M < 1e-15:
            break
        
        # Solve LP for best response
        b = solve_lp(a, M)
        if b is None or b.sum() < 1e-10:
            break
        
        # Rescale to sum S
        g = b * S / b.sum()
        
        # Step size: harmonic decay
        t_step = 1.0 / (1 + it * 0.1)
        if t_step > 0.5:
            t_step = 0.5
        
        # Update
        a_new = (1 - t_step) * a + t_step * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new * S / s_new
        
        # Update running average
        a_avg = (it * a_avg + a_new) / (it + 1)
        
        # Check score
        score = compute_score(a_new / S)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        # Also check the average
        if it % 10 == 0:
            score_avg = compute_score(a_avg / S)
            if score_avg < best_score:
                best_score = score_avg
                best_a = a_avg.copy()
        
        a = a_new
        
        # Progress print
        if verbose and time.monotonic() - last_print > 10:
            print(f"    FP iter={it}, score={best_score:.8f}, M={M:.6f}")
            last_print = time.monotonic()
    
    if verbose:
        print(f"    FP done: {it} iters in {time.monotonic()-t0:.1f}s, best={best_score:.8f}")
    
    return best_a


def gradient_refine(a_init, time_limit, T=1000, lr_init=0.5):
    """
    Gradient descent with momentum in log-space using softmax parameterization.
    Uses a smoothed (log-sum-exp) surrogate for the max operation.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    
    def softmax(z):
        z_max = z.max()
        e = np.exp(z - z_max)
        return e / e.sum()
    
    def log_softmax_scores(z):
        a = softmax(z)
        conv = fast_convolve(a)
        return conv
    
    z = np.log(np.maximum(a_init, 1e-30))
    v = np.zeros(n)
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        a = softmax(z)
        conv = fast_convolve(a)
        
        # Smoothed gradient (log-sum-exp based)
        shifted = conv / T
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        w_sum = w.sum()
        w = w / w_sum
        
        # Gradient of (2n * LSE(conv/T)) w.r.t. z
        # dLSE/dz_i = sum_k w_k * d(conv_k)/dz_i
        # d(conv_k)/da_j = 2 * sum_i a[i] * delta(k, i+j) ... 
        # Actually d(conv_k)/da_j = a[k-j] + a[j-(k-n+1)] ... 
        # More precisely: conv_k = sum_i a_i * a_{k-i}
        # d(conv_k)/da_j = a_{k-j} + a_{j} (if k-j is valid) ... 
        # Let me be more careful:
        # conv_k = sum_{i: 0<=i<n, 0<=k-i<n} a_i * a_{k-i}
        # d(conv_k)/da_j = a_{k-j} (if 0<=k-j<n) + a_{k-j} (if j appears as k-i, i.e., i=k-j, 0<=k-j<n)
        # Wait: if j = i, then the term is a_j * a_{k-j}, derivative w.r.t. a_j is a_{k-j}
        # If j = k-i (i.e., i = k-j), then the term is a_{k-j} * a_j, derivative w.r.t. a_j is a_{k-j}
        # So d(conv_k)/da_j = a_{k-j} if 0<=k-j<n (counting twice if j = k-j, i.e., k=2j)
        # Actually: if j ≠ k-j (i.e., j ≠ k/2), then d(conv_k)/da_j = a_{k-j}
        # If j = k/2 (i.e., k = 2j), then d(conv_k)/da_j = 2*a_j
        # More compactly: d(conv_k)/da_j = a_{k-j} * [0<=k-j<n] + a_{k-j} * [0<=j<n and k-j=j] 
        # Hmm, let me just use: d(conv_k)/da_j = a_{k-j} if 0 <= k-j < n
        # This is correct because conv_k = sum_i a_i * a_{k-i}, and d/da_j of this is:
        # - If j appears as i: a_{k-j} (one term)
        # - If j appears as k-i (i.e., i=k-j): a_{k-j} (one term)
        # - If j = i = k-i (i.e., k=2j): 2*a_j
        # So total: a_{k-j} + a_{k-j} = 2*a_{k-j} if j = k-j, else a_{k-j}
        # Which is: a_{k-j} * (1 + [k=2j])
        # But for the purposes of the gradient via FFT, we can just compute:
        # grad_a_j = 2 * sum_k w_k * a_{k-j} (the factor of 2 accounts for both appearances)
        # This is 2 * (w * a reversed)[j] ... actually it's 2 * cross_correlation(w, a)[j]
        
        # Using FFT: cross_correlation(w, a)[j] = sum_k w[k] * a[k-j] (with appropriate bounds)
        # = ifft(conj(FFT(w)) * FFT(a))[j] (for the right range)
        
        fft_size = 1 << (3 * n - 2 - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        
        # Project to log-space (Riemannian gradient for softmax)
        S_proj = np.dot(grad_a, a)
        grad_z = a * (grad_a - S_proj)
        
        # Scale by 2n/T for the smoothed objective
        grad_z = (2.0 * n / T) * grad_z
        
        # Momentum update
        v = 0.9 * v + lr * grad_z
        z_new = z - v
        z_new = np.clip(z_new, -30, 30)
        
        a_new = softmax(z_new)
        score = compute_score(a_new)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        
        z = z_new
        
        # Adaptive learning rate
        if it % 200 == 0:
            lr *= 0.95
    
    return best_a


def peak_refine(a_init, time_limit, n_peaks=20, verbose=False):
    """Peak equalization: iteratively reduce the maximum peak."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.maximum(a_init.astype(np.float64), 0).copy()
    a = a * S / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a / S)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    it = 0
    last_print = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        if M < 1e-15:
            break
        
        # Find top peaks
        top_k = min(n_peaks, len(conv))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_values = conv[peak_indices]
        spread = peak_values.max() - peak_values.min()
        
        if spread < 1e-8 * M:
            if verbose:
                print(f"    PE: peaks equalized, breaking")
            break
        
        # Try to reduce the max peak
        max_idx = int(np.argmax(conv))
        js = max(0, max_idx - n + 1)
        je = min(n - 1, max_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_idx - idx]
        
        # Top contributors
        n_top = min(8, len(contribs))
        top_contrib = np.argsort(contribs)[::-1][:n_top]
        
        improved = False
        for ci in top_contrib:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            
            for frac in [0.03, 0.07, 0.12, 0.2]:
                delta = a[i] * frac
                a_trial = a.copy()
                a_trial[i] = max(0, a_trial[i] - delta)
                s_trial = a_trial.sum()
                if s_trial < 1e-10:
                    continue
                a_trial = a_trial * S / s_trial
                sc = compute_score(a_trial / S)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try random perturbation at peak-contributing positions
            peak_positions = set()
            for pi in peak_indices[:5]:
                si = max(0, int(pi) - n + 1)
                ei = min(n - 1, int(pi))
                for p in range(si, ei + 1):
                    peak_positions.add(p)
            
            if peak_positions:
                pos_list = np.array(list(peak_positions))
                p = pos_list[rng.integers(0, len(pos_list))]
                a_trial = a.copy()
                delta = a[p] * rng.standard_normal() * 0.05
                a_trial[p] = max(0, a_trial[p] - delta)
                s_trial = a_trial.sum()
                if s_trial > 1e-10:
                    a_trial = a_trial * S / s_trial
                    sc = compute_score(a_trial / S)
                    # Accept if not much worse (simulated annealing style)
                    if sc < best_score * 1.001:
                        a = a_trial
        
        if verbose and time.monotonic() - last_print > 10:
            print(f"    PE iter={it}, best={best_score:.8f}")
            last_print = time.monotonic()
    
    return best_a


def coordinate_refine(a_init, time_limit, n_pairs=500, verbose=False):
    """Coordinate descent with pairwise transfers."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.maximum(a_init.astype(np.float64), 0).copy()
    a = a * S / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a / S)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    it = 0
    last_print = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue
            
            base = min(a[i], max(a[j], a[i] * 0.3))
            for frac in [0.1, 0.3, 0.5, 0.8]:
                t_transfer = frac * base
                if t_transfer >= a[i] * 0.95:
                    t_transfer = a[i] * 0.5
                
                a_trial = a.copy()
                a_trial[i] = max(0, a_trial[i] - t_transfer)
                a_trial[j] += t_transfer
                s_trial = a_trial.sum()
                if s_trial < 1e-10:
                    continue
                a_trial = a_trial * S / s_trial
                sc = compute_score(a_trial / S)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved and it > 100:
            # Perturb
            noise = rng.standard_normal(n) * 0.002 * S / n
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert * S / a_pert.sum()
            a = a_pert
        
        if verbose and time.monotonic() - last_print > 10:
            print(f"    CD iter={it}, best={best_score:.8f}")
            last_print = time.monotonic()
    
    return best_a


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
