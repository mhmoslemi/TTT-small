Gradient-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses softmax-weighted subgradient with momentum, multiple structured
initializations, adaptive temperature, and LP polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import math

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(p, w):
    """Compute weighted subgradient. w has length 2n-1."""
    n = len(p)
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = np.dot(p, w[j:j+n])
    return grad

def run_gradient_opt(p_init, time_limit, tau_init=0.001, tau_rate=1.05, 
                     lr=0.5, momentum=0.9, label="grad", seed=42):
    """Run softmax-weighted subgradient optimization with momentum."""
    t0 = time.monotonic()
    n = len(p_init)
    p = p_init.copy()
    v = np.zeros(n)
    tau = tau_init
    best_p = p.copy()
    best_obj = fast_convolve(p).max()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(p)
        cur_obj = c.max()
        
        # Update best
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute softmax weights
        c_shifted = c - c.max()
        w = np.exp(c_shifted / tau)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[c.argmax()] = 1.0
        else:
            w /= w_sum
        
        # Compute subgradient
        grad = compute_subgradient(p, w)
        
        # Scale gradient
        g_max = np.max(np.abs(grad))
        if g_max > 1e-15:
            grad = grad / g_max
        
        # Momentum update
        v = momentum * v - lr * grad
        p = project_simplex(p + v)
        
        # Adaptive tau
        if it % 100 == 0:
            tau *= tau_rate
            if no_improve > 50:
                # Restart from best with perturbation
                p = project_simplex(best_p + rng.standard_normal(n) * 0.005)
                v = np.zeros(n)
                no_improve = 0
                tau = tau_init
        
        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} iter={it} obj={best_obj:.10f} score={score:.6f} "
                  f"tau={tau:.6f} t={elapsed:.0f}s")
    
    return best_p, best_obj

    elif kind == "sine":

    elif kind == "two_hump":

    elif kind == "random_cascade":

        p = np.ones(n)
    
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s > 1e-10:
        p /= s
    else:
        p = np.ones(n) / n

"""
Hybrid gradient + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp smoothed gradient with FFT-based correlation for fast subgradient
computation, multiple structured initializations, adaptive temperature annealing,
random restarts, and LP polishing for final improvement.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.signal import correlate as sig_correlate
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(p, w):
    """
    Compute grad[j] = sum_{m=0}^{n-1} p[m] * w[j+m] using FFT correlation.
    p has length n, w has length 2n-1. Result has length n.
    """
    n = len(p)
    grad = sig_correlate(w, p, mode='valid')
    return grad


def logsumexp_objective(c, tau):
    """Compute tau * log(sum(exp(c/tau))) as a smooth approximation to max(c)."""
    c_shifted = c - c.max()
    return c.max() + tau * np.log(np.exp(c_shifted / tau).sum())


def run_gradient_opt(p_init, time_limit, tau_init=0.0005, tau_end=0.005,
                     lr=0.4, momentum=0.9, label="grad", seed=42,
                     restart_every=2000, restart_perturb=0.01):
    """
    Run smoothed gradient descent on log-sum-exp approximation to max(conv(p,p)).
    Uses FFT-accelerated gradient computation, momentum, and periodic restarts.
    """
    t0 = time.monotonic()
    n = len(p_init)
    p = p_init.copy()
    v = np.zeros(n)
    tau = tau_init
    best_p = p.copy()
    best_obj = fast_convolve(p).max()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(seed)
    total_iters_target = max(1, int(time_limit * 150))  # estimate
    tau_factor = (tau_end / tau_init) ** (1.0 / max(1, total_iters_target))

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(p)
        cur_obj = c.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute softmax weights (gradient of log-sum-exp)
        c_shifted = c - c.max()
        w = np.exp(c_shifted / tau)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[c.argmax()] = 1.0
        else:
            w /= w_sum

        # Compute gradient via FFT correlation
        grad = compute_gradient_fft(p, w)

        # Normalize gradient for stable stepping
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        # Momentum update
        v = momentum * v - lr * grad
        p = project_simplex(p + v)

        # Animate temperature
        if it < total_iters_target:
            tau *= tau_factor
        else:
            tau = tau_end

        # Periodic restart to escape local minima
        if no_improve > restart_every or (it % (restart_every * 3) == 0 and no_improve > 500):
            p = project_simplex(best_p + rng.standard_normal(n) * restart_perturb)
            v = np.zeros(n)
            no_improve = 0
            tau = tau_init * 0.5

        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} iter={it} obj={best_obj:.10f} score={score:.6f} "
                  f"tau={tau:.6f} t={elapsed:.0f}s")

    return best_p, best_obj


def run_lp_phase(f_init, time_limit, top_k_init=40, top_k_max=200, label="lp"):
    """LP-based direction finding with line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 40 == 0 and top_k < top_k_max:
            top_k = min(top_k + 15, top_k_max)

        b = fast_convolve(f)
        M = b.max()
        if M < 1e-15:
            break

        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        top_idx = np.sort(top_idx)
        K = len(top_idx)

        A = np.zeros((K, n))
        for row_i in range(K):
            k = top_idx[row_i]
            lo = max(0, int(k) - n + 1)
            hi = min(n - 1, int(k))
            A[row_i, lo:hi + 1] = f[lo:hi + 1]

        g = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-8)
        except Exception:
            break

        if g.value is None:
            break
        g_val = np.maximum(g.value, 0.0)
        if np.any(np.isnan(g_val)):
            break

        S_g = float(g_val.sum())
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 15:
                break
            top_k = min(top_k * 2, top_k_max)
            continue

        h = g_val / S_g
        best_t = 0.1
        best_t_obj = fast_convolve(f).max()

        for t in [0.02, 0.04, 0.06, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.6]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial = f_trial / s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 60:
                break

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0 * n * best_obj:.6f} top_k={top_k} t={elapsed:.0f}s")

    return best_f, best_obj


def make_init(n, kind, rng):
    """Create a structured initialization of length n."""
    x = np.arange(n, dtype=np.float64)
    if kind == "uniform":
        p = np.ones(n)
    elif kind == "cosine_bell":
        p = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    elif kind == "anti_cosine":
        p = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / max(1, n - 1)))
    elif kind == "sine":
        p = np.sin(np.pi * x / max(1, n - 1)) ** 2
    elif kind == "two_hump":
        p = np.sin(2.0 * np.pi * x / max(1, n - 1)) ** 2
    elif kind == "three_hump":
        p = np.sin(3.0 * np.pi * x / max(1, n - 1)) ** 2
    elif kind == "geometric":
        r = rng.uniform(0.6, 0.95)
        p = r ** x
    elif kind == "random_wave":
        p = np.ones(n)
        for _ in range(rng.integers(3, 7)):
            freq = rng.uniform(0.05, 6.0)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.2, 1.2)
            p = p + amp * np.sin(2 * np.pi * freq * x / n + phase)
        p = np.maximum(p, 0.001)
    elif kind == "peaked":
        center = n / 2.0
        width = n / 4.0
        p = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    else:
        p = np.ones(n)

    p = np.maximum(p, 0.0)
    s = p.sum()
    if s > 1e-10:
        p /= s
    else:
        p = np.ones(n) / n
    return p


def mirror_swap_perturb(p, rng, num_swaps=20):
    """Perform random mirror swaps to perturb while preserving structure."""
    n = len(p)
    p_new = p.copy()
    for _ in range(num_swaps):
        j = rng.integers(0, n)
        k = rng.integers(0, n)
        t = rng.uniform(0.1, 0.5)
        p_new[j] = p[j] * (1 - t) + p[k] * t
        p_new[k] = p[k] * (1 - t) + p[j] * t
    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s > 1e-10:
        p_new /= s
    return p_new

        if t_alloc < 10:
            break

        # Interpolate best to new length
        old_x = np.linspace(0.0, 1.0, len(best_seq))
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  n={target_n}, budget={t_alloc:.0f}s")
        f_opt, obj_opt = run_gradient_opt(
            f_interp, time_limit=t_alloc,
            tau_init=0.0008, tau_end=0.004,
            lr=0.35, momentum=0.9,
            label=f"len_{target_n}", seed=seed + target_n
        )
        score_opt = float(evaluate_sequence(f_opt.tolist()))
        if np.isfinite(score_opt) and score_opt < best_score:
            best_score = score_opt
            best_seq = f_opt.copy()
            print(f"  NEW BEST [len_{target_n}]: {score_opt:.6f}")
        else:
            print(f"  [len_{target_n}] best={score_opt:.6f}")

    # === Phase 4: Deeper gradient on best with mirror perturbation ===
    print(f"\n=== Phase 4: Deep gradient + perturb (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.25, budget_s * 0.15)
    if t4 > 15:
        print(f"  Budget: {t4:.0f}s, n={len(best_seq)}")

        # First: direct deep optimization
        t4a = t4 * 0.5
        f4, obj4 = run_gradient_opt(
            best_seq, time_limit=t4a,
            tau_init=0.0001, tau_end=0.002,
            lr=0.25, momentum=0.92,
            label="grad_deep", seed=seed + 500
        )
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [grad_deep]: {score4:.6f}")

        # Then: mirror swap perturb and re-optimize
        t4b = t4 * 0.3
        if t4b > 10:
            p_pert = mirror_swap_perturb(best_seq, rng, num_swaps=30)
            f4b, obj4b = run_gradient_opt(
                p_pert, time_limit=t4b,
                tau_init=0.0003, tau_end=0.003,
                lr=0.3, momentum=0.9,
                label="grad_pert", seed=seed + 600
            )
            score4b = float(evaluate_sequence(f4b.tolist()))
            if np.isfinite(score4b) and score4b < best_score:
                best_score = score4b
                best_seq = f4b.copy()
                print(f"  NEW BEST [grad_pert]: {score4b:.6f}")

    # === Phase 5: LP polishing ===
    print(f"\n=== Phase 5: LP polishing (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.30, budget_s * 0.20)
    if t5 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t5:.0f}s, n={n_b}, top_k=50->200")
        f5, obj5 = run_lp_phase(
            best_seq, time_limit=t5,
            top_k_init=50, top_k_max=200,
            label="lp_polish"
        )
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [lp_polish]: {score5:.6f}")
        else:
            print(f"  [lp_polish] best={score5:.6f}")

    # === Phase 6: Final gradient burst on polished result ===
    print(f"\n=== Phase 6: Final gradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.5, budget_s * 0.05)
    if t6 > 5:
        print(f"  Budget: {t6:.0f}s")
        f6, obj6 = run_gradient_opt(
            best_seq, time_limit=t6,
            tau_init=0.0001, tau_end=0.001,
            lr=0.15, momentum=0.93,
            label="grad_final", seed=seed + 900
        )
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [grad_final]: {score6:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'=' * 50}")

    return [float(x) for x in best_seq]
