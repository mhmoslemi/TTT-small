import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, beta):
    """Compute gradient of smoothed objective (1/beta)*log(sum(exp(beta*b)))."""
    b = fast_convolve(a)
    # Numerical stability
    b_shifted = b - b.max()
    exp_b = np.exp(beta * b_shifted)
    w = exp_b / exp_b.sum()

    # = (2/beta) * correlate(w, a, 'full')[:n]

    # Wait, I need to be careful with the convention.

    b = fast_convolve(a)
    b_max = b.max()
    return (1.0/beta) * np.log(np.sum(np.exp(beta * (b - b_max)))) + b_max / beta

                         lr_init=0.01, momentum=0.9, label="smooth"):
    """Projected Nesterov gradient descent with adaptive temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    y = f.copy()  # Nesterov momentum variable
    v = np.zeros(n)
    beta = beta_init
    lr = lr_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)
    
    total_iters = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        total_iters += 1
        
        # Nesterov step
        y = f + momentum * v
        grad, w, b = compute_gradient(y, beta)
        
        # Gradient step
        v = momentum * v - lr * grad
        f_new = project_simplex(y + v)
        
        # Update
        f = f_new
        v = f - y  # This is the standard Nesterov update: v = f_new - y_old
        
        # Check improvement (every 10 iterations to save time)
        if it % 10 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 20:
                    # Cool down and restart
                    beta = min(beta * 1.5, beta_final)
                    lr *= 0.9
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                    v = np.zeros(n)
                    no_improve = 0
        
        # Adaptive beta: increase slowly
        if it % 50 == 0:
            beta = min(beta * 1.1, beta_final)
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} beta={beta:.1f} lr={lr:.5f} t={elapsed:.0f}s")
    
    return best_f, best_obj

"""
Optimization of step functions to minimize 2*n*max(conv(a,a))/(sum(a))^2.

Combines three approaches:
1. Smoothed gradient descent with log-sum-exp and adaptive temperature
2. LP-based direction finding with soft peak weighting and warm-starting
3. Multi-length exploration and coordinate polish

Key innovations over previous approach:
- Smoothed objective with FFT-based gradient for stable optimization
- Soft (exponential) peak weighting instead of hard top-K in LP
- Nesterov momentum for accelerated convergence
- Adaptive temperature schedule for smooth-to-nonsmooth transition
- Better line search with proper backtracking
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(a, beta):
    """
    Compute gradient of f(a) = (1/beta)*log(sum_k exp(beta * b[k]))
    where b = convolve(a, a).
    Returns (gradient, b, max_b).
    """
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    # Numerically stable log-sum-exp
    shifted = beta * (b - b_max)
    # Clip for numerical safety
    shifted = np.clip(shifted, -700, 700)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    # grad[j] = (2/beta) * sum_{m=0}^{n-1} a[m] * w[j+m]
    # Using FFT: convolve(a[::-1], w)[j+n-1] = sum_m a[m] * w[j+m]
    total_len = 3 * n - 2
    N = 1 << (total_len - 1).bit_length()
    a_rev = a[::-1]
    A_fft = rfft(a_rev, N)
    W_fft = rfft(w, N)
    conv_full = irfft(A_fft * W_fft, N)[:total_len]
    # c[j] = conv_full[j + n - 1] for j = 0, ..., 2n-2
    c = conv_full[n - 1: n - 1 + 2 * n - 1]
    grad = (2.0 / beta) * c[:n]
    return grad, b, b_max

def lp_direction(f, n, peak_weights, M):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for weighted peaks.
    Returns g or None.
    """
    b = fast_convolve(f)

    # Select peaks based on weights
    num_peaks = min(len(b), 2 * n - 1)
    K = min(num_peaks, 120)  # Cap for speed

    # Get top weighted peaks
    if peak_weights is not None:
        scores = peak_weights[:num_peaks]
    else:
        scores = b[:num_peaks]

    top_idx = np.argpartition(scores, -K)[-K:]
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    # Build constraint matrix
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-7)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=100000, abstol=1e-7, reltol=1e-7)
        except Exception:
            return None, M

    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M

def coordinate_polish(f_init, time_limit, label="cd"):
    """Coordinate descent on individual elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()

                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val > 0.5:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-12:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:

                s = f.sum()
                if s > 1e-10:
                    f /= s
                if f[j] < 1e-15:
                    f[j] = 0.0
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    improved = True
        if not improved:
            break
    return f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if new_n == old_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t4 = min(rem * 0.25, 200.0)
        if t4 < 25:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        print(f"  LP n={target_n}, budget={t4:.0f}s")

        # First smoothed, then LP
        t4a = t4 * 0.3
        t4b = t4 * 0.7
        f4s, _ = run_smoothed_phase(f_interp, time_limit=t4a, label=f"sm_n{target_n}")
        score4s = float(evaluate_sequence(f4s.tolist()))
        if np.isfinite(score4s) and score4s < best_score:
            best_score = score4s
            best_seq = f4s.copy()
            print(f"  NEW BEST [smooth_n{target_n}]: {score4s:.6f}")

        f4, _ = run_lp_phase(f4s, time_limit=t4b,
                             top_k_init=20, top_k_max=80, label=f"lp_n{target_n}")
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [lp_n{target_n}]: {score4:.6f}")

    # === Phase 5: Final LP burst (5% time) ===
    print(f"\n=== Phase 5: Final LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.3, 150.0)
    if t5 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t5:.0f}s, n={n_b}, K=60->120")
        f5, _ = run_lp_phase(best_seq, time_limit=t5,
                             top_k_init=60, top_k_max=120, label="lp_final")
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [lp_final]: {score5:.6f}")

    # === Phase 6: Coordinate polish (remaining time) ===
    print(f"\n=== Phase 6: Coordinate polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.5, 100.0)
    if t6 > 10:
        print(f"  Budget: {t6:.0f}s")
        f6, _ = coordinate_polish(best_seq, time_limit=t6, label="cd")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [cd]: {score6:.6f}")

    # === Final verification ===
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'=' * 50}")

    return [float(x) for x in best_seq]
