```python
import numpy as np
import time

def simplex_projection(v, s=1.0):
    """Project v onto the simplex {x : sum(x) = s, x >= 0}."""
    n = len(v)
    mu = np.zeros(n)
    sv = np.sort(v)[::-1]
    cv = np.cumsum(sv)
    r = np.arange(1, n+1)
    cond = sv - (cv - s) / r > 0
    r = np.max(r[cond])
    mu = (sv[r-1] - s / r) / 1.0  # This is the water level
    return np.maximum(v - mu, 0)

def subgradient_step(f, alpha):
    """One subgradient step to minimize max(convolve(f,f)) subject to sum(f)=1, f>=0."""
    n = len(f)
    b = np.convolve(f, f)
    k_star = np.argmax(b)
    s = np.zeros(n)
    # s_j = f_{k_star - j} if 0 <= k_star - j < n
    valid = (k_star - np.arange(n) >= 0) & (k_star - np.arange(n) < n)
    idx = k_star - np.arange(n)
    s[valid] = f[idx[valid]]
    # Project gradient onto the simplex tangent
    g = s - np.mean(s)
    f_new = f - alpha * g
    # Project onto simplex {sum=1, f>=0}
    f_new = simplex_projection(f_new, 1.0)
    return f_new, k_star, b[k_star]
```

```python
def simplex_projection(v, s=1.0):
    """Project v onto the simplex {x : sum(x) = s, x >= 0}."""
    n = len(v)
    sorted_v = np.sort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    rho_candidates = np.arange(1, n + 1)
    cond = sorted_v - (cumsum - s) / rho_candidates > 0
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.max(rho_candidates[cond])
    theta = (cumsum[rho - 1] - s) / rho
    return np.maximum(v - theta, 0)

def compute_objective(f):
    """Compute the objective value for a non-negative sequence f."""
    n = len(f)
    s = np.sum(f)
    if s < 0.01:
        return np.inf
    b = np.convolve(f, f)
    m = np.max(b)
    return 2.0 * n * m / (s * s)

def subgradient_step(f, alpha):
    """One subgradient step to minimize max(conv(f,f)) given sum(f)=1, f>=0."""
    n = len(f)
    b = np.convolve(f, f)
    k_star = np.argmax(b)
    # Subgradient: s_j = f_{k_star - j} if 0 <= k_star - j < n
    j = np.arange(n)
    idx = k_star - j
    s = np.zeros(n)
    valid = (idx >= 0) & (idx < n)
    s[valid] = f[idx[valid]]
    # Project gradient onto tangent of simplex
    g = s - np.mean(s)
    f_new = f - alpha * g
    f_new = simplex_projection(f_new, 1.0)
    return f_new, k_star, b[k_star]

def lp_direction(f, M):
    """Solve LP to find direction g that keeps f*g <= M and maximizes sum(g)."""
    n = len(f)
    # Build constraint matrix A (2n-1 x n), A[k,j] = f_{k-j} if valid
    # This is a Toeplitz matrix
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # For row k, non-zero at columns j where max(0, k-n+1) <= j <= min(n-1, k)
    
    row_indices = []
    col_indices = []
    data = []
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            row_indices.extend([k] * len(js))
            col_indices.extend(js.tolist())
            data.extend(f[k - js].tolist())
    
    A = sparse.csc_matrix((data, (row_indices, col_indices)), shape=(2*n-1, n))
    
    # Maximize sum(g) subject to A @ g <= M, g >= 0
    c = -np.ones(n)  # minimize -sum(g)
    
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    return None

    n = len(f)
    b = np.convolve(f, f)
    M = np.max(b)

    if g is None:
        return f, M

    s_g = np.sum(g)
    if s_g < 1e-10:

    # Update

    # Re-normalize to sum = 1
    s_new = np.sum(f_new)
    if s_new > 1e-10:
        f_new = f_new / s_new
    return f_new, M
```

```python
"""
LP-based local search to minimize 2*n*max(conv(f,f))/(sum(f))^2.
Uses the SOTA LP formulation (max sum(g) s.t. conv(f,g) <= peak, g >= 0)
with pre-computed sparse structure, subgradient pre-pass, and multiple
initializations including Gaussian windows and random starts.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def precompute_convolve_structure(n):
    """Pre-compute row/col indices for the convolve constraint matrix."""
    rows_list = []
    cols_list = []
    for d in range(n):
        js = np.arange(0, n - d)
        rows_list.append(js + d)
        cols_list.append(js)
    for d in range(1, n):
        ks = np.arange(0, n - d)
        rows_list.append(ks)
        cols_list.append(ks + d)
    return np.concatenate(rows_list), np.concatenate(cols_list)


def make_constraint_matrix(f, rows, cols):
    """Build sparse constraint matrix A[k,j]=f[k-j] using pre-computed indices."""
    n = len(f)
    lags = rows - cols
    vals = f[lags]
    return sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def simplex_project(v, target_sum):
    """Project v onto {x : sum(x) = target_sum, x >= 0}."""
    n = len(v)
    if n == 0:
        return v.copy()
    sv = np.sort(v)[::-1]
    cv = np.cumsum(sv)
    r = np.arange(1, n + 1)
    cond = (sv - (cv - target_sum) / r) > 0
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = int(np.max(r[cond]))
    theta = (cv[rho - 1] - target_sum) / rho
    return np.maximum(v - theta, 0.0)


def subgradient_iter(f, alpha, n_steps):
    """Run subgradient iterations to reduce peak. f normalized to sum=1."""
    n = len(f)
    f = f.copy()
    j = np.arange(n)
    for _ in range(n_steps):
        b = np.convolve(f, f)
        k_star = int(np.argmax(b))
        idx = k_star - j
        s = np.zeros(n)
        valid = (idx >= 0) & (idx < n)
        s[valid] = f[idx[valid]]
        g = s - np.mean(s)
        f = f - alpha * g
        f = simplex_project(f, 1.0)
    return f


def lp_solve(f, M, rows, cols):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g."""
    n = len(f)
    A = make_constraint_matrix(f, rows, cols)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        s = np.sum(res.x)
        if s > 1e-15:
            return res.x / s
    return None


def run_lp_search(f, rows, cols, deadline, verbose=True):
    """LP-based local search. f is normalized to sum=1, returns best f and M."""
    n = len(f)
    f = f.copy()
    b = np.convolve(f, f)
    M = float(np.max(b))
    best_f = f.copy()
    best_M = M
    t = 0.5
    stall = 0
    iters = 0
    t0 = time.monotonic()

    while time.monotonic() < deadline:
        g = lp_solve(f, M, rows, cols)
        if g is None:
            break

        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s = np.sum(f_new)
        if s < 1e-15:
            break
        f_new = f_new / s

        b_new = np.convolve(f_new, f_new)
        M_new = float(np.max(b_new))
        iters += 1

        if M_new < M * (1.0 - 1e-12):
            f = f_new
            M = M_new
            stall = 0
            if M < best_M:
                best_M = M
                best_f = f.copy()
        else:
            stall += 1
            t = max(0.05, t * 0.75)

        if stall > 80:
            break

        if verbose and iters % 20 == 0:
            print(f"    iter {iters}: M={M:.12f}, t={t:.3f}, stall={stall}")

    if verbose:
        print(f"  LP done: {iters} iters in {time.monotonic()-t0:.1f}s, "

    return best_f, best_M
```
