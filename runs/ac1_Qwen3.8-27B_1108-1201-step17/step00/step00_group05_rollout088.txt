```
def softmax_weights(c, kappa):
    L=len(c)
    maxc=float(np.max(c))
    z=c-maxc

       return np.full(L,1.0/L)

    return ez/s
```

```python
"""
Search algorithm for nonnegative step-function heights minimizing
    2*n*max(convolve(a,a)) / sum(a)^2.

The method uses:
  1. Fast FFT-based evaluation and smooth log-sum-exp gradients for the
     convolution maximum.
  2. Several polynomial-parameterized starts and a parameter-space Adam
     optimizer to find a good macroscopic shape.
  3. Full-sequence log-space gradient refinement with stochastic restarts.
  4. Final validation using the provided evaluate_sequence function.
"""

import numpy as np
import time


def next_power_of_two(n):
    n = int(n)
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def fft_conv(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    if a.size == 0 or b.size == 0:
        return np.empty(0, dtype=np.float64)
    out_len = a.size + b.size - 1
    n_fft = next_power_of_two(out_len)
    fa = np.fft.rfft(a, n_fft)
    fb = np.fft.rfft(b, n_fft)
    c = np.fft.irfft(fa * fb, n_fft)
    return c[:out_len]


def sequence_score_fast(p):
    p = np.asarray(p, dtype=np.float64)
    if p.size == 0:
        return float("inf")
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 0.0:
        return float("inf")
    c = fft_conv(p, p)
    c = np.maximum(c, 0.0)
    if c.size == 0:
        return float("inf")
    return float(2.0 * p.size * np.max(c) / (s * s))


def softmax_weights(c, kappa):
    c = np.asarray(c, dtype=np.float64)
    m = c.size
    if m == 0:
        return np.empty(0, dtype=np.float64)
    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        w = np.zeros(m, dtype=np.float64)
        w[0] = 1.0
        return w
    if kappa <= 1e-8:
        return np.full(m, 1.0 / m, dtype=np.float64)
    z = c - maxc
    ez = np.exp(kappa * z)
    s = float(np.sum(ez))
    if not np.isfinite(s) or s <= 0.0:
        idx = np.where(c == maxc)[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(c))], dtype=int)
        w = np.zeros(m, dtype=np.float64)
        w[idx] = 1.0 / float(idx.size)
        return w
    return ez / s


def make_basis(n, d):
    y = np.linspace(-1.0, 1.0, max(1, int(n)))
    d = int(max(1, d))
    B = np.zeros((y.size, d), dtype=np.float64)
    B[:, 0] = y
    if d > 1:
        B[:, 1] = y * y - 1.0 / 3.0
    if d > 2:
        B[:, 2] = y * y * y
    if d > 3:
        B[:, 3] = y ** 4 - 1.0 / 5.0
    if d > 4:
        B[:, 4] = y ** 5
    if d > 5:
        B[:, 5] = y ** 6 - 1.0 / 7.0
    if d > 6:
        B[:, 6] = (y * y - 1.0 / 3.0) ** 2 - 1.0 / 5.0
    if d > 7:
        B[:, 7] = y ** 8 - 1.0 / 9.0
    return B


def normalize_positive(f):
    f = np.asarray(f, dtype=np.float64)
    if f.size == 0:
        return None
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.clip(f, 0.0, 1000.0)
    s = float(np.sum(f))
    if not np.isfinite(s) or s <= 0.0:
        return None
    f = f / s
    if np.max(f) > 1000.0:
        return None
    return f


def candidate_sort_key(item):
    return item[0]


def add_candidate(candidates, p, score=None, keep=60):
    if p is None:
        return candidates
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0 or not np.all(np.isfinite(p)):
        return candidates
    p = np.clip(p, 0.0, 1000.0)
    if score is None or not np.isfinite(float(score)):
        score = sequence_score_fast(p)
    score = float(score)
    if not np.isfinite(score):
        return candidates
    candidates.append((score, p))
    candidates.sort(key=candidate_sort_key)
    if len(candidates) > keep:
        candidates = candidates[:keep]
    return candidates


def update_best(best, p, score=None):
    if p is None:
        return best
    p = np.asarray(p, dtype=np.float64)
    if p.size == 0 or not np.all(np.isfinite(p)):
        return best
    if score is None or not np.isfinite(float(score)):
        score = sequence_score_fast(p)
    score = float(score)
    if np.isfinite(score) and (best is None or len(best) != 2 or best[1] == float("inf") or score < best[1]):
        best = [p.copy(), score]
    return best

def theta_to_sequence(n, d, theta):
    n = int(n)
    if n < 2:
        return np.array([1.0 / max(1, int(n))], dtype=np.float64)
    B = make_basis(n, d)
    theta = np.asarray(theta, dtype=np.float64)
    if theta.size < d:
        theta = np.concatenate([theta, np.zeros(d - theta.size, dtype=np.float64)])
    theta = theta[:d]
    z = B.dot(theta)
    z = z - float(np.max(z))
    p = np.exp(z)
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 0.0:
        p = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        p = p / s
    return p


def param_search(n, d, seed, time_budget, init_theta=None):
    n = int(n)
    d = int(d)
    if n < 2 or time_budget <= 0.0:
        return None, float("inf"), None

    start = time.monotonic()
    deadline = start + max(0.0, float(time_budget))
    B = make_basis(n, d)
    rng = np.random.default_rng(int(seed))

    presets = [
        np.array([0.55, -0.25, 0.00, -0.10, 0.00, 0.00, 0.00, 0.00], dtype=np.float64),
        np.array([0.75, -0.40, 0.08, -0.20, 0.02, -0.03, 0.00, 0.00], dtype=np.float64),
        np.array([0.90, -0.50, 0.16, -0.28, 0.05, -0.06, 0.00, 0.00], dtype=np.float64),
        np.array([1.05, -0.58, 0.22, -0.34, 0.07, -0.08, -0.02, 0.03], dtype=np.float64),
        np.array([0.80, -0.20, 0.30, -0.12, 0.10, 0.00, 0.02, 0.00], dtype=np.float64),
        np.array([0.60, -0.35, -0.12, -0.18, -0.03, 0.02, 0.05, 0.00], dtype=np.float64),
    ]
    starts = []
    if init_theta is not None:
        t = np.asarray(init_theta, dtype=np.float64)
        if t.size < d:
            t = np.concatenate([t, np.zeros(d - t.size, dtype=np.float64)])
        starts.append(t[:d].copy())
    for pr in presets:
        if pr.size < d:
            pr = np.concatenate([pr, np.zeros(d - pr.size, dtype=np.float64)])
        starts.append(pr[:d].copy())

    scale = np.array([0.7, 0.5, 0.4, 0.35, 0.3, 0.25, 0.2, 0.2], dtype=np.float64)
    if scale.size < d:
        scale = np.concatenate([scale, np.zeros(d - scale.size, dtype=np.float64)])
    scale = scale[:d]
    for _ in range(5):
        starts.append(rng.normal(0.0, 1.0, size=d) * scale)

    best_p = None
    best_score = float("inf")
    best_theta = None

    n_starts = len(starts)
    per_start = max(5.0, (deadline - start) / float(max(1, n_starts)))

    for idx in range(n_starts):
        if time.monotonic() >= deadline:
            break
        local_deadline = min(deadline, time.monotonic() + per_start)

        th = starts[idx].copy()
        th = np.clip(th, -8.0, 8.0)
        m = np.zeros(d, dtype=np.float64)
        v = np.zeros(d, dtype=np.float64)
        t = 0
        lr = 0.07
        beta1 = 0.90
        beta2 = 0.999
        kappa = 20.0
        no_imp = 0

        z = B.dot(th)
        z = z - float(np.max(z))
        p = np.exp(z)
        p = p / float(np.sum(p))
        c = fft_conv(p, p)
        c = np.maximum(c, 0.0)
        score = float(2.0 * n * np.max(c))
        local_best_score = score
        local_best_p = p.copy()
        local_best_th = th.copy()

        while time.monotonic() < local_deadline:
            if t > 0:
                z = B.dot(th)
                z = z - float(np.max(z))
                p = np.exp(z)
                s = float(np.sum(p))
                if not np.isfinite(s) or s <= 0.0:
                    break
                p = p / s

            c = fft_conv(p, p)
            if not np.all(np.isfinite(c)):
                break
            c = np.maximum(c, 0.0)
            maxc = float(np.max(c))
            score = float(2.0 * n * maxc)

            if np.isfinite(score) and score < local_best_score:
                local_best_score = score
                local_best_p = p.copy()
                local_best_th = th.copy()
                no_imp = 0
            else:
                no_imp += 1

            w = softmax_weights(c, kappa)
            if not np.all(np.isfinite(w)):
                break
            w_rev = w[::-1]
            h = fft_conv(p, w_rev)
            if h.size != 3 * n - 2:
                break
            r = h[2 * n - 2:n - 2:-1]
            f_grad = 4.0 * n * r
            if not np.all(np.isfinite(f_grad)):
                break
            e = f_grad - float(np.dot(p, f_grad))
            grad_theta = B.T.dot(p * e)
            if not np.all(np.isfinite(grad_theta)):
                break

            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_theta
            v = beta2 * v + (1.0 - beta2) * (grad_theta * grad_theta)
            mhat = m / (1.0 - beta1 ** t)
            vhat = v / (1.0 - beta2 ** t)
            th = th - lr * mhat / (np.sqrt(vhat) + 1e-8)
            th = np.clip(th, -8.0, 8.0)

            if t % 80 == 0 and no_imp < 250:
                kappa = min(8000.0, kappa * 1.08)

            if no_imp > 220:
                th = local_best_th + 0.25 * rng.normal(0.0, 1.0, size=d)
                th = np.clip(th, -8.0, 8.0)
                m = np.zeros(d, dtype=np.float64)
                v = np.zeros(d, dtype=np.float64)
                t = 0
                kappa = max(20.0, kappa * 0.6)
                lr = max(0.02, lr * 0.75)
                no_imp = 0
                continue

            if no_imp > 1000:
                break

        if local_best_score < best_score:
            best_score = local_best_score
            best_p = local_best_p.copy()
            best_theta = local_best_th.copy()

    return best_p, best_score, best_theta


def full_refine(p0, time_budget, seed=0, kappa_start=30.0, lr_start=0.18):
    p = normalize_positive(p0)
    if p is None:
        if p0 is None:
            return None, float("inf")
        p = np.asarray(p0, dtype=np.float64)
        s = float(np.sum(p))
        if not np.isfinite(s) or s <= 0.0:
            return None, float("inf")
        p = p / s

    n = int(p.size)
    if n < 2:
        return p, sequence_score_fast(p)

    start = time.monotonic()
    deadline = start + max(0.0, float(time_budget))
    rng = np.random.default_rng(int(seed))

    curr_p = p.copy()
    curr_c = fft_conv(curr_p, curr_p)
    curr_c = np.maximum(curr_c, 0.0)
    curr_score = float(2.0 * n * np.max(curr_c))

    best_p = curr_p.copy()
    best_score = curr_score

    y = np.log(np.maximum(curr_p, 1e-15))
    y = y - float(np.mean(y))

    n_r = next_power_of_two(3 * n - 2)
    p_fft_r = np.fft.rfft(curr_p, n_r)

    kappa = float(kappa_start)
    lr = float(lr_start)
    no_imp = 0
    step = 0

    while time.monotonic() < deadline:
        maxc = float(np.max(curr_c))
        curr_score = float(2.0 * n * maxc)
        if not np.isfinite(curr_score):
            break
        if curr_score < best_score:
            best_score = curr_score
            best_p = curr_p.copy()
            no_imp = 0

        w = softmax_weights(curr_c, kappa)
        if not np.all(np.isfinite(w)):
            kappa = max(20.0, kappa * 0.7)
            no_imp += 1
            continue

        w_rev = w[::-1]
        fw = np.fft.rfft(w_rev, n_r)
        h = np.fft.irfft(p_fft_r * fw, n_r)
        h = h[:3 * n - 2]
        r = h[2 * n - 2:n - 2:-1]
        f_grad = 4.0 * n * r
        if not np.all(np.isfinite(f_grad)):
            break

        e = f_grad - float(np.mean(f_grad))
        en = float(np.sqrt(np.mean(e * e)))
        if not np.isfinite(en) or en <= 1e-14:
            e = rng.normal(0.0, 1.0, size=n)
        else:
            e = e / en
        e = np.clip(e, -3.0, 3.0)

        if rng.random() < 0.08:
            e = e + 0.25 * rng.normal(0.0, 1.0, size=n)

        y_try = y - lr * e
        y_try = y_try - float(np.mean(y_try))
        y_try = np.clip(y_try, -30.0, 30.0)
        p_try = np.exp(y_try)
        s_try = float(np.sum(p_try))
        if not np.isfinite(s_try) or s_try <= 0.0:
            y_try = y - 0.5 * lr * e
            y_try = np.clip(y_try, -30.0, 30.0)
            p_try = np.exp(y_try)
            s_try = float(np.sum(p_try))
            if not np.isfinite(s_try) or s_try <= 0.0:
                p_try = curr_p

        p_try = p_try / s_try
        c_try = fft_conv(p_try, p_try)
        if not np.all(np.isfinite(c_try)):
            break
        c_try = np.maximum(c_try, 0.0)
        score_try = float(2.0 * n * np.max(c_try))

        accept = False
        if np.isfinite(score_try):
            accept = score_try < curr_score
            if not accept and no_imp > 70:
                if score_try < best_score * 1.03 and rng.random() < 0.12:
                    accept = True

        if accept:
            y = y_try.copy()
            curr_p = p_try.copy()
            curr_c = c_try.copy()
            curr_score = score_try
            p_fft_r = np.fft.rfft(curr_p, n_r)
            if score_try < best_score:
                best_score = score_try
                best_p = p_try.copy()
                no_imp = 0
                lr = min(0.75, lr * 1.07)
            else:
                no_imp += 1
        else:
            no_imp += 1
            lr = max(0.02, lr * 0.97)

        step += 1
        if step % 120 == 0 and no_imp < 220:
            kappa = min(9000.0, kappa * 1.07)

        if no_imp > 380:
            y = np.log(np.maximum(best_p, 1e-15))
            y = y - float(np.mean(y))
            y = y + 0.25 * rng.normal(0.0, 1.0, size=n)
            y = np.clip(y, -30.0, 30.0)
            curr_p = np.exp(y)
            s_curr = float(np.sum(curr_p))
            if not np.isfinite(s_curr) or s_curr <= 0.0:
                break
            curr_p = curr_p / s_curr
            curr_c = fft_conv(curr_p, curr_p)
            curr_c = np.maximum(curr_c, 0.0)
            curr_score = float(2.0 * n * np.max(curr_c))
            p_fft_r = np.fft.rfft(curr_p, n_r)
            kappa = max(20.0, kappa * 0.6)
            lr = max(0.05, lr * 0.75)
            no_imp = 0

    return best_p, best_score


def clean_sequence(p):
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return np.array([1.0], dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    p[p < 1e-14] = 0.0
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones(p.size, dtype=np.float64) / float(p.size)
    else:
        p = p / s
    return p


def propose_candidate(seed=42, budget_s=1800):
    if budget_s is None:
        budget_s = 1800
    budget_s = float(max(60.0, budget_s))
    big_seed = int(seed) if np.isfinite(float(seed)) else 42

    start = time.monotonic()
    final_reserve = 60.0 if budget_s > 600.0 else 12.0
    hard_deadline = start + max(0.0, budget_s - final_reserve)

    best = [None, float("inf")]
    candidates = []

    global_n = None
    try:
        hs = height_sequence_1
        L = len(hs)
        if 512 <= L <= 8192:
            global_n = int(L)
            arr = np.asarray(hs, dtype=np.float64)
            if arr.ndim == 1 and arr.size == L and np.all(np.isfinite(arr)):
                p = normalize_positive(arr)
                if p is not None:
                    candidates = add_candidate(candidates, p, None)
                    best = update_best(best, p, None)
    except Exception:
        global_n = None

    if global_n is not None:
        main_n = global_n
    else:
        main_n = 2048

    search_lens = []
    for n in (main_n, 1024, 2048, 4096):
        n = int(n)
        if 256 <= n <= 8192 and n not in search_lens:
            search_lens.append(n)

    if budget_s >= 400.0:
        direct_lens = search_lens[:3]
        direct_count = 260 if main_n <= 2500 else 200
    else:
        direct_lens = search_lens[:1]
        direct_count = 120

    print("main_n", main_n, "budget", budget_s, flush=True)

    for n in direct_lens:
        if time.monotonic() >= hard_deadline:
            break
        items = direct_candidates(n, count=direct_count, seed=big_seed + n)
        for sc, p in items:
            candidates = add_candidate(candidates, p, sc)
            best = update_best(best, p, sc)
        print("after direct n=", n, "best=", best[1], "time=", time.monotonic() - start, flush=True)

    D = 8
    if time.monotonic() < hard_deadline - 30.0:
        param_time = max(20.0, min(450.0, (hard_deadline - time.monotonic()) * 0.30))
        theta_best, param_p, param_sc = (None, None, None, None)[0:4] if False else (None, float("inf"), None)
        p, sc, theta = param_search(main_n, D, big_seed + 17, param_time, init_theta=None)
        if p is not None:
            candidates = add_candidate(candidates, p, sc)
            best = update_best(best, p, sc)
            if theta is not None:
                theta_best = theta
        print("after param best=", best[1], "time=", time.monotonic() - start, flush=True)

    if theta_best is not None:
        for n in (main_n, 2048, 4096, 1024):
            if time.monotonic() >= hard_deadline - 20.0:
                break
            if n < 256 or n > 8192:
                continue
            p = theta_to_sequence(n, D, theta_best)
            candidates = add_candidate(candidates, p, None)
            best = update_best(best, p, None)
        print("after theta lifts best=", best[1], "time=", time.monotonic() - start, flush=True)

    candidates.sort(key=candidate_sort_key)
    if len(candidates) == 0:
        p = np.full(max(256, main_n), 1.0 / max(256, main_n), dtype=np.float64)
        candidates = add_candidate(candidates, p, None)
        best = update_best(best, p, None)

    weights = [0.32, 0.24, 0.18, 0.12, 0.08, 0.06]
    max_rounds = min(6, len(candidates))

    for i in range(max_rounds):
        now = time.monotonic()
        remaining = hard_deadline - now
        if remaining < 25.0:
            break
        sc_i, p_i = candidates[i]
        alloc = remaining * weights[i] / float(len(weights))
        alloc = min(alloc, remaining * 0.48)
        alloc = max(20.0, alloc)
        alloc = min(alloc, 900.0)
        print("refine top", i, "score=", sc_i, "alloc=", alloc, "time=", now - start, flush=True)
        ref_p, ref_sc = full_refine(p_i, alloc, seed=big_seed + 101 + 13 * i)
        if ref_p is not None:
            candidates = add_candidate(candidates, ref_p, ref_sc)
            best = update_best(best, ref_p, ref_sc)
            print("after refine", i, "best=", best[1], "time=", time.monotonic() - start, flush=True)
        candidates.sort(key=candidate_sort_key)

    if time.monotonic() < hard_deadline - 30.0:
        second_weights = [0.55, 0.30, 0.15]
        second_count = min(3, len(candidates))
        candidates.sort(key=candidate_sort_key)
        for i in range(second_count):
            now = time.monotonic()
            remaining = hard_deadline - now
            if remaining < 25.0:
                break
            sc_i, p_i = candidates[i]
            alloc = remaining * second_weights[i] / float(len(second_weights))
            alloc = min(alloc, remaining * 0.45)
            alloc = max(20.0, alloc)
            alloc = min(alloc, 700.0)
            print("second refine", i, "score=", sc_i, "alloc=", alloc, flush=True)
            ref_p, ref_sc = full_refine(p_i, alloc, seed=big_seed + 500 + 17 * i)
            if ref_p is not None:
                candidates = add_candidate(candidates, ref_p, ref_sc)
                best = update_best(best, ref_p, ref_sc)
            candidates.sort(key=candidate_sort_key)

    final_candidates = []
    if best is not None and best[0] is not None:
        final_candidates.append((best[1], clean_sequence(best[0])))
    for sc, p in candidates[:8]:
        final_candidates.append((sc, clean_sequence(p)))

    final_seq = final_candidates[0][1] if final_candidates else None
    final_score = float("inf")

    for _, p in final_candidates:
        try:
            s = float(evaluate_sequence([float(x) for x in p]))
        except Exception:
            s = float("inf")
        if np.isfinite(s) and s < final_score:
            final_score = s
            final_seq = p.copy()

    if final_seq is None:
        final_seq = clean_sequence(np.full(1024, 1.0 / 1024, dtype=np.float64))
        try:
            final_score = float(evaluate_sequence([float(x) for x in final_seq]))
        except Exception:
            final_score = float("inf")

    print("final score evaluate_sequence:", final_score, flush=True)
    return [float(x) for x in final_seq]
```
