import numpy as np
import time

def compute_conv(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(fa * fa, fft_size)[:2*n-1]
    return b

def softmax_obj(a, beta):
    """Compute the soft-max smoothed objective and its gradient."""
    n = len(a)
    b = compute_conv(a)
    # Soft-max
    max_b = np.max(b)
    exp_b = np.exp(beta * (b - max_b))
    sum_exp = np.sum(exp_b)
    s = exp_b / sum_exp  # softmax weights
    log_sum_exp = np.log(sum_exp) + beta * max_b
    sum_a = np.sum(a)
    obj = 2 * n * log_sum_exp / (sum_a ** 2)
    # Gradient
    # cross[i] = sum_{j=0}^{n-1} s[i+j] * a[j] for i = 0, ..., n-1
    # = conv(s, a[::-1])[n-1 : 2*n-1]
    cross = np.convolve(s, a[::-1], mode='full')[n-1 : 2*n-1]
    # Actually, let me verify the length. s has length 2n-1, a[::-1] has length n.
    # conv has length (2n-1) + n - 1 = 3n-2.

    grad = 4 * n * beta * cross / (sum_a ** 2) - 2 * obj * a / sum_a
    return obj, grad, b, sum_a
