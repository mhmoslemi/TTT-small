import numpy as np
import time
from scipy.signal import fftconvolve

def _fft_convolve(x, y=None, mode='full'):
    """FFT-based convolution."""
    if y is None:
        y = x
    n = len(x) + len(y) - 1
    size = 1
    while size < n:
        size *= 2
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    return np.fft.irfft(X * Y, size)[:n]

def _fast_score(x, n):
    """Compute 2*n*max(conv(x,x))/(sum(x))^2."""
    s = x.sum()
    if s < 1e-10:
        return np.inf
    b = _fft_convolve(x)
    return 2.0 * n * b.max() / (s * s)

def _project(x):
    """Project onto x >= 0, sum(x) = 1."""
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.ones(len(x)) / len(x)
    return x / s

def _solve_lp_mw(f, M, n, n_iters=30, lr=1.0):
    """
    Solve max sum(g) s.t. conv(f,g) <= M, g >= 0 using multiplicative weights.
    Returns g normalized to sum 1.
    """
    g = np.ones(n) / n
    for _ in range(n_iters):
        # Forward pass: r = conv(f, g)
        r = _fft_convolve(f, g)
        # Violations (clipped at 0)
        viol = np.maximum(r - M, 0.0)
        if viol.max() < 1e-12:
            break
        # Score for each g[j]: sum of violations of constraints involving g[j]
        # g[j] appears in constraint k iff j <= k <= j+n-1
        # score[j] = sum_{k=j}^{j+n-1} viol[k]
        # This is a sliding window sum
        # viol has length 2n-1
        # Use cumulative sum
        cs = np.zeros(2 * n)
        cs[1:] = np.cumsum(viol)
        # score[j] = cs[j+n] - cs[j] for j = 0, ..., n-1
        scores = cs[n:2*n] - cs[:n]
        # Multiplicative weights update
        g = g * np.exp(-lr * scores / M)
        g = _project(g)
    return g

def _lp_update(f, n, n_lp_iters=30, lr_lp=1.0):

    M = b.max()

def _gradient_step(f, n, lr=0.1):

"""
Minimizes 2*n*max(conv(x,x))/(sum(x))^2 using LP-guided direction (multiplicative weights),
projected gradient polish, diverse multi-start initialization, and symmetrization.
Uses FFT for all convolution/correlation operations for O(n log n) per step.
"""

import numpy as np
import time
from scipy.signal import fftconvolve


def _fft_convolve(x, y=None):
    """Full linear convolution via FFT."""
    if y is None:
        y = x
    n = len(x) + len(y) - 1
    size = 1
    while size < n:
        size <<= 1
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    return np.fft.irfft(X * Y, size)[:n]


def _cross_corr(a, b):
    """Cross-correlation: out[j] = sum_m a[m]*b[j+m], j=0..len(a)-1. Uses FFT."""
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    return (np.fft.irfft(A * np.conj(B), size))[:na]


def _fast_score(x, n):
    """Compute 2*n*max(conv(x,x))/(sum(x))^2 quickly."""
    s = x.sum()
    if s < 1e-10:
        return np.inf
    b = _fft_convolve(x)
    return 2.0 * n * b.max() / (s * s)


def _project(x):
    """Project onto x >= 0, sum(x) = 1."""
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.ones(len(x)) / len(x)
    return x / s


def _solve_lp_mw(f, M, n, n_iters=25, lr=1.0):
    """
    Solve: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0.
    Using multiplicative weights (exponentiated gradient) method.
    Returns g normalized to sum 1, and the LP objective value (sum before normalize).
    """
    g = np.ones(n) / n
    fft_size = 1
    while fft_size < 3 * n:
        fft_size <<= 1

    for _ in range(n_iters):
        # Forward: r = conv(f, g)
        r = _fft_convolve(f, g)
        # Violations
        viol = np.maximum(r - M, 0.0)
        if viol.max() < 1e-14:
            break
        # Score: score[j] = sum_m f[m]*viol[j+m] = cross_corr(f, viol)[j]
        scores = _cross_corr(f, viol)
        # Multiplicative update
        g = g * np.exp(-lr * scores / M)
        # Project
        g = _project(g)

    return g, g.sum()


def _lp_step(f, n, n_lp_iters=25, lr_lp=1.0):
    """
    One LP-guided step: solve LP, find direction, take step with backtracking.
    Returns (new_f, improved_this_step).
    """
    b = _fft_convolve(f)
    M = b.max()
    current_score = 2.0 * n * M

    g, g_sum = _solve_lp_mw(f, M, n, n_iters=n_lp_iters, lr=lr_lp)

    if g_sum < 1.0 + 1e-5:
        return f, False

    # Backtracking step
    best_candidate = f
    best_sc = current_score
    for t in [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02]:
        candidate = _project((1.0 - t) * f + t * g)
        sc = _fast_score(candidate, n)
        if sc < best_sc:
            best_sc = sc
            best_candidate = candidate

    improved = best_sc < current_score
    return best_candidate, improved


def _gradient_step(f, n, lr=0.05):
    """One projected gradient step using subgradient of max(conv)."""
    b = _fft_convolve(f)
    M = b.max()
    k_star = int(np.argmax(b))

    # Subgradient: grad[j] = 2*f[k_star - j] for valid j
    j_lo = max(0, k_star - n + 1)
    j_hi = min(n - 1, k_star)
    grad = np.zeros(n)
    if j_hi >= j_lo:
        grad[j_lo:j_hi + 1] = 2.0 * f[k_star - j_hi:k_star - j_lo + 1][::-1]

    candidate = _project(f - lr * grad)
    return candidate

    print(f"Phase 1: {len(inits)} inits, ~{time_per:.0f}s each")

    for i, init in enumerate(inits):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        t_lim = min(time.monotonic() + time_per, deadline - 30)
        print(f"  Init {i + 1}/{len(inits)}: ", end="", flush=True)
        result, score = _optimize(init, n, t_lim, rng, verbose=False)
        status = ""
        if score < best_score:
            best = result.copy()
            best_score = score
            status = " *** NEW BEST ***"
        print(f"{score:.6f}{status}")

    # Phase 2: Polish the best with remaining time
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"Phase 2: polishing ({remaining:.0f}s)")
        # Try a few perturbations of the best
        attempts = 0
        polish_deadline = deadline - 5
        while time.monotonic() < polish_deadline and attempts < 10:
            attempts += 1
            # Create a perturbed version
            if attempts == 1:
                start = best.copy()
            elif attempts == 2:
                start = _project(best + rng.normal(0, 0.05, n))
            elif attempts == 3:
                start = _project(0.7 * best + 0.3 * inits[rng.integers(len(inits))])
            else:
                start = _project(best + rng.normal(0, 0.03 * attempts / 3.0, n))

            t_lim = min(time.monotonic() + max(10.0, (polish_deadline - time.monotonic()) / (10 - attempts + 1)),
                        polish_deadline)
            result, score = _optimize(start, n, t_lim, rng, verbose=False)
            if score < best_score:
                best = result.copy()
                best_score = score
                print(f"  polish {attempts}: {best_score:.6f} *** NEW BEST ***")
            else:
                print(f"  polish {attempts}: {score:.6f}")

    # Final symmetrization check
    sym = _project(0.5 * (best + best[::-1]))
    sym_sc = _fast_score(sym, n)
    if sym_sc < best_score:
        best = sym
        best_score = sym_sc

    # Final evaluation
    final_score = float(evaluate_sequence(best.tolist()))
    print(f"Final score: {final_score:.6f}")

    return [float(v) for v in best]
