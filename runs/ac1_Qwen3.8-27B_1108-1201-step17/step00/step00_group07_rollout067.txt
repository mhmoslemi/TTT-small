"""
Peak-autocorrelation minimiser for a non‑negative step function.

The objective  f(a) = 2n * max(conv(a,a)) / (sum a)^2  is scale invariant,
so we work on a normalised shape p (sum p = 1, p >= 0) and minimise
G(p) = 2n * max_k (p*p)[k].

Search = (1) structured seeds (the SOTA sequence, the 2‑periodic [1,2] family
  which is known to give 16/9 ~= 1.778, plus a handful of periods and random
  shapes), and (2) an active‑set projected sub‑gradient step on the maximum
  member of the convolution (peak reduction), interleaved with randomised block
  perturbations to escape local minima.  The authoritative evaluator is called
  periodically to record the best valid sequence; the internal objective is used
  for fast step acceptance.  A hard deadline guarantees a return of the best
  sequence found.
"""

import time
import numpy as np


# --------------------------------------------------------------------------- #
# low level helpers (all top level, no closures / lambdas)
# --------------------------------------------------------------------------- #
def _clip_nonneg(seq):
    """Return a finite, non‑negative float64 array."""
    a = np.asarray(seq, dtype=np.float64)
    a = a.ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    return a


def _normalise(p):
    """Normalise so that sum(p) == 1 (returns p unchanged-ish on failure)."""
    s = p.sum()
    if s <= 1e-12:
        return np.full_like(p, 1.0 / p.size)
    return p / s


def _internal_objective(p):
    """Fast proxy for the true score on a unit‑sum shape: 2n * max(conv)."""
    if p.size == 0:
        return np.inf
    c = np.convolve(p, p)
    return 2.0 * p.size * float(c.max())


def _active_set_max_lags(c, frac=1e-4):
    """Indices of convolution entries within `frac` of the running max."""
    m = c.max()
    thr = m - frac * m
    return np.nonzero(c >= thr)[0]


def _subgrad_direction(p, lags):
    """Sum of gradients of the active convolution terms w.r.t. a unit‑sum shape.
    For C(k)=sum_i p_i p_{k-i},  dC(k)/dp_m = p_{k-m} + p_m  (in range)."""
    n = p.size
    grad = np.zeros(n, dtype=np.float64)
    for k in lags:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if hi < lo:
            continue
        idx = np.arange(lo, hi + 1)
        comp = (k - idx).astype(np.int64)
        grad[idx] += p[idx] + p[comp]
    return grad


def _peak_reduction_step(p, lr, active_frac=1e-4, max_lags=256):
    """One projected sub‑gradient step that lowers the active convolution peaks."""
    c = np.convolve(p, p)
    lags = _active_set_max_lags(c, active_frac)
    if lags.size > max_lags:                       # cap work, keep largest
        order = np.argsort(c[lags])[::-1][:max_lags]
        lags = lags[order]
    g = _subgrad_direction(p, lags)
    p_new = p - lr * g
    # restore the unit‑sum first‑order (Lagrange) term, then project
    p_new = p_new - (p_new.sum() - 1.0) / p.size
    p_new = np.clip(p_new, 0.0, None)
    return _normalise(p_new)


def _block_perturbation(p, rng, sigma=0.10):
    """Randomised multiplicative stretch on a random contiguous block."""
    n = p.size
    w_lo = max(1, n // 400)
    w_hi = max(2, n // 16)
    width = int(rng.integers(w_lo, w_hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    q = p.copy()
    q[start:stop] *= float(np.exp(rng.normal(0.0, sigma)))
    if rng.random() < 0.25:                         # occasional mirror
        q = 0.5 * (q + q[::-1])
    q = np.clip(q, 0.0, None)
    return _normalise(q)


def _add_peak(p, rng, strength=0.35):
    """Inject a small local hump (helps exploration of sharper shapes)."""
    n = p.size
    width = int(rng.integers(2, max(3, n // 64) + 1))
    start = int(rng.integers(0, max(1, n - width)))
    q = p.copy()
    off = np.arange(width)
    bump = np.exp(-(off - 0.5 * (width - 1)) ** 2 / (0.5 * width))
    q[start:start + width] += strength * bump
    return _normalise(np.clip(q, 0.0, None))

# seed constructors

def _seed_periodic(length, period):
    perm = np.asarray(period, dtype=np.float64)
    rep = int(np.ceil(length / perm.size))
    a = np.tile(perm, rep)[:length]
    return a


def _make_seed(rng, name):

        return _normalise(a)
    a = np.ones(1024, dtype=np.float64)

def _optimise_seed(seed, rng, deadline):
    """Run peak‑reduction + perturbations from one seed until the deadline."""
    p = seed.copy()
    best_p = p.copy()
    best_val = _internal_objective(p)
    lr = 0.5 / (p.size + 1.0)         # scale‑free-ish learning rate
    no_improve = 0
    last_eval = 0.0

    def _maybe_record():
        nonlocal best_val, best_p
        # periodic authoritative evaluation to keep the reported value honest
        if time.time() - last_eval > 6.0:
            try:
                sc = float(evaluate_sequence([float(x) for x in p]))   # noqa: F821
            except Exception:
                sc = np.inf
            last_eval = time.time()
            _maybe_record.last_score = sc
        if best_val > 0:
            # keep the internal best shape
            return

    while time.time() < deadline:
        # ---- accepted peak reduction ------------------------------------ #
        cand = _peak_reduction_step(p, lr, active_frac=1e-4, max_lags=256)
        val = _internal_objective(cand)
        if val < best_val - 1e-12:
            best_p = cand.copy()
            best_val = val
            no_improve = 0
            lr = min(0.9, lr * 1.05)      # reward progress
        else:
            p = cand
            no_improve += 1
            lr = max(0.02, lr * 0.85)     # cool down

        # ---- keep current point on track with best ---------------------- #

            p = best_p.copy()

        if r < 0.10:

        elif r < 0.18:

        else:

        if val < best_val - 1e-12:
            best_p = cand.copy()
            best_val = val
            p = cand.copy()
            no_improve = 0
            lr = 0.5 / (p.size + 1.0)

        _maybe_record()

    return best_p, best_val

def _authoritative(seq):
    try:
        return float(evaluate_sequence([float(x) for x in seq]))   # noqa: F821
    except Exception:
        return np.inf


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for the best non‑negative coefficient sequence.

    Returns a Python list of floats.  Always returns the best sequence found
    strictly before ``budget_s`` seconds (a safety margin is reserved).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + max(30.0, float(budget_s) - 30.0)

    # ---- establish an initial best (fast, authoritative) ---------------- #
    name_list = ["p12", "sota", "p3", "p4", "rand", "rand", "rand"]
    global_best = None
    global_score = np.inf

    for nm in name_list:
        s = _make_seed(rng, nm)
        sc = _authoritative(s)
        if np.isfinite(sc) and sc < global_score:
            global_score = sc
            global_best = s

    # ---- per‑seed optimisation with a time slice each ------------------- #
    # each seed gets a slice of the remaining time; we refresh global best
    for i, nm in enumerate(name_list):
        if time.time() >= deadline:
            break
        s = _make_seed(rng, nm)
        # rough per‑seed budget: spread what's left, at least a little each
        remaining = max(1.0, deadline - time.time())
        per = remaining / max(1, (len(name_list) - i))
        seed_deadline = min(deadline - 1.0, time.time() + per)
        if seed_deadline <= time.time() + 0.5:
            continue
        try:
            bp, bv = _optimise_seed(s, rng, seed_deadline)
        except Exception:
            continue
        sc = _authoritative(bp)
        if np.isfinite(sc) and sc < global_score:
            global_score = sc
            global_best = bp

    # ---- final top‑up: refine the current global best if time remains ---- #
    if global_best is not None and time.time() < deadline:
        try:
            bp, _ = _optimise_seed(global_best, rng, deadline)
            sc = _authoritative(bp)
            if np.isfinite(sc) and sc < global_score:
                global_score = sc
                global_best = bp
        except Exception:
            pass

    if global_best is None:
        global_best = _normalise(_seed_periodic(1024, [1.0, 2.0]))

    print("best score = %.6f" % global_score)          # noqa
    return [float(x) for x in global_best]
