def compute_conv(a):
    """Compute full convolution of a with itself."""
    return np.convolve(a, a)

    """Find the index of the maximum."""
    return int(np.argmax(conv))

    """Compute the mirror of a at lag k_star."""
    n = len(a)
    m = np.zeros(n)
    left = max(0, k_star - (n - 1))
    right = min(n - 1, k_star)
    if left <= right:
        idx = k_star - np.arange(left, right + 1)
        m[left:right+1] = a[idx]
    return m
