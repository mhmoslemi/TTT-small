def project_to_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_conv_and_grad(p, argmax_indices):
    """Compute convolution and subgradient for given p and argmax indices."""
    n = len(p)
    conv = np.convolve(p, p)
    # Subgradient: for each i, grad[i] = 2 * mean_{k in argmax} p[k-i]
    grad = np.zeros(n)
    for k in argmax_indices:
        # p[k-i] for i in range(n), with zero extension
        # j = k - i, valid when 0 <= j < n, i.e., k-n+1 <= i <= k
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)

            j_end = k - i_start
            grad[i_start:i_end+1] += 2.0 * p[j_start:j_end+1][::-1]
    grad /= len(argmax_indices)
    return conv, grad

def subgradient_descent(p, max_iters, lr_init, lr_decay):

def fast_convolve(a, b):
    """FFT-based convolution."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:

    fa = np.fft.rfft(a, size)

**Smoothing / Mirror Descent:**
Instead of using the hard max, use a smooth approximation like the log-sum-exp:
`f_τ(p) = τ * log(sum_k exp((p*p)[k] / τ))`

This is smooth and its gradient is well-defined. As τ → 0, it approaches the hard max. We can use a decreasing τ schedule.

The gradient of `f_τ(p)` with respect to `p[i]`:
`∂f_τ/∂p[i] = sum_k softmax_k * ∂(p*p)[k]/∂p[i] = sum_k softmax_k * 2 * p[k-i]`

where `softmax_k = exp((p*p)[k] / τ) / sum_j exp((p*p)[j] / τ)`.

This is a weighted average of the subgradients, which is smoother and better-behaved.

Let me verify with a small example. n=3, p=[a,b,c], q=[c,b,a].
w has length 5 (2n-1=5). Say w=[w0,w1,w2,w3,w4].

(w*q)[m] = sum_k w[k] * q[m-k]

(w*q)[2] = w[0]*q[2] + w[1]*q[1] + w[2]*q[0] = w0*a + w1*b + w2*c
(w*q)[3] = w[1]*q[2] + w[2]*q[1] + w[3]*q[0] = w1*a + w2*b + w3*c
(w*q)[4] = w[2]*q[2] + w[3]*q[1] + w[4]*q[0] = w2*a + w3*b + w4*c

grad[0]/2 = sum_k w[k]*p[k-0] = w[0]*p[0] + w[1]*p[1] + w[2]*p[2] = w0*a + w1*b + w2*c = (w*q)[2] ✓ (n-1+0 = 2)
grad[1]/2 = sum_k w[k]*p[k-1] = w[1]*p[0] + w[2]*p[1] + w[3]*p[2] = w1*a + w2*b + w3*c = (w*q)[3] ✓ (n-1+1 = 3)
grad[2]/2 = sum_k w[k]*p[k-2] = w[2]*p[0] + w[3]*p[1] + w[4]*p[2] = w2*a + w3*b + w4*c = (w*q)[4] ✓ (n-1+2 = 4)

Great! So `grad = 2 * (w * p[::-1])[n-1 : 2*n-1]` (the last n elements of the convolution of w and reversed p).

This can be computed via FFT in O(n log n).

Now the full algorithm per iteration:
1. Compute conv = p * p (FFT, O(n log n))
2. Compute weights = softmax(conv / tau) (O(n))
3. Compute grad = 2 * (weights * p[::-1])[n-1:2n-1] (FFT, O(n log n))
4. Update p = project_simplex(p - lr * grad) (O(n log n))

Total: O(n log n) per iteration, with a small constant (2 FFTs of size ~4n).

For n = 4498, FFT size ~16384, each FFT takes maybe 50-200 μs. So each iteration is maybe 200-800 μs. In 1800 seconds, we can do 2M-9M iterations. That's plenty.

import numpy as np
import time
from scipy.fft import fft, ifft

def fast_convolve(a, b):
    """FFT-based full convolution."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()  # next power of 2
    fa = fft(a, size)
    fb = fft(b, size)
    result = ifft(fa * fb)
    return result[:n].real

def fast_convolve_self(a):
    """FFT-based auto-convolution (a * a)."""
    n = 2 * len(a) - 1
    size = 1 << (n - 1).bit_length()
    fa = fft(a, size)
    result = ifft(fa * fa)
    return result[:n].real

def project_to_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return np.array([])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_optimize(p0, max_iters, tau_start=0.01, tau_end=0.001, lr_start=0.05, lr_end=0.001):
    """Smooth (log-sum-exp) gradient descent on the simplex."""
    n = len(p0)
    p = p0.copy()
    s = p.sum()
    if s > 0:
        p = p / s
    else:
        p = np.ones(n) / n
        s = 1.0
    
    best_p = p.copy()
    best_max = np.inf
    
    # Precompute FFT size
    conv_len = 2 * n - 1
    fsize = 1 << (conv_len - 1).bit_length()
    p_rev = p[::-1].copy()
    
    for it in range(max_iters):
        # Interpolate parameters
        frac = it / max(1, max_iters - 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        # Compute convolution
        conv = fast_convolve_self(p)
        conv_max = conv.max()
        
        if conv_max < best_max:
            best_max = conv_max
            best_p = p.copy()
        
        # Compute softmax weights
        logits = (conv - conv_max) / max(tau, 1e-10)
        # Clip for numerical stability
        logits = np.clip(logits, -50, 50)
        weights = np.exp(logits)
        wsum = weights.sum()
        if wsum > 0:
            weights /= wsum
        else:
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0
        
        # Compute gradient via convolution: grad = 2 * (weights * p_rev)[n-1 : 2n-1]
        # But p changes, so we need to update p_rev
        p_rev = p[::-1]
        grad_conv = fast_convolve(weights, p_rev)
        grad = 2.0 * grad_conv[n - 1: n - 1 + n]
        
        # Gradient step
        p = p - lr * grad
        
        # Project onto simplex
        p = project_to_simplex(p)
    
    return best_p
