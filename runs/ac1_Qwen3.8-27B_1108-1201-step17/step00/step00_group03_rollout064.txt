"""
LP-based refinement with multiple restarts to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses scipy linprog for the LP subproblem and line search for step size.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix


def compute_objective(a):
    """Fast computation of the objective value."""
    n = len(a)
    b = np.convolve(a, a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * np.max(b) / (sum_a ** 2))


def lp_refine_step(f, M, max_iter_lp=100):
    """
    Solve the LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns the solution g (or None if infeasible/unbounded).
    """
    n = len(f)
    num_constraints = 2 * n - 1

    # Build constraint matrix (sparse)
    # A[k, j] = f[k-j] if 0 <= k-j < n else 0
    # This is a banded matrix
    A_rows = []
    A_cols = []
    A_vals = []
    for k in range(num_constraints):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A_rows.append(k)
            A_cols.append(j)
            A_vals.append(f[k - j])

    A = lil_matrix((num_constraints, n))
    for i in range(len(A_rows)):
        A[A_rows[i], A_cols[i]] = A_vals[i]
    A = csr_matrix(A)

    # Objective: minimize -sum(g) (to maximize sum(g))
    c = -np.ones(n)

    # Constraints: A @ g <= M
    b_ub = M * np.ones(num_constraints)

    # Bounds: g >= 0
    bounds = [(0, None)] * n

    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        g = result.x
        if np.any(g > 0):
            return g
    return None


def line_search(f, g, M_current, max_trials=20):
    """
    Find best step size t in [0,1] such that peak of blended sequence <= M_current.
    Returns the best t and the blended sequence.
    """
    sum_f = np.sum(f)
    sum_g = np.sum(g)
    if sum_g <= 0:
        return 0.0, f.copy()

    # Rescale g to have same sum as f
    g_rescaled = g * (sum_f / sum_g)

    best_t = 0.0
    best_obj = M_current
    best_seq = f.copy()

    # Try a range of t values
    t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]

    for t in t_candidates:
        candidate = (1 - t) * f + t * g_rescaled
        candidate = np.maximum(candidate, 0)
        b = np.convolve(candidate, candidate)
        peak = np.max(b)
        if peak <= best_obj:
            best_obj = peak
            best_t = t
            best_seq = candidate

    return best_t, best_seq


def run_lp_refinement(f0, time_budget):

    start_time = time.monotonic()

def random_start(n, rng):
    """Generate a random non-negative sequence."""
    # Mix of different structures
    choice = rng.integers(0, 5)
    if choice == 0:
        a = np.ones(n)
    elif choice == 1:
        a = np.random.default_rng(rng.integers(0, 2**31)).exponential(1.0, n)
    elif choice == 2:
        # Sinusoidal perturbation
        a = 1.0 + 0.5 * np.sin(2 * np.pi * rng.uniform(0.5, 5) * np.arange(n) / n)
        a = np.maximum(a, 0)
    elif choice == 3:
        # Two-phase
        a = np.ones(n)
        a[:n//2] = rng.uniform(0.5, 2.0)
        a[n//2:] = rng.uniform(0.5, 2.0)
    else:
        # Bump
        a = np.ones(n)
        center = rng.integers(0, n)
        width = rng.integers(n//4, n//2)
        for i in range(n):
            d = abs(i - center)
            if d < width:
                a[i] *= rng.uniform(1.5, 3.0)
            else:
                a[i] *= rng.uniform(0.3, 1.0)
    s = np.sum(a)
    if s < 0.01:
        a = np.ones(n)
    else:
        a = a / s
    return a


def block_perturb(f, rng, strength=0.3):
    """Apply a random block perturbation."""
    n = len(f)
    candidate = f.copy()
    num_blocks = rng.integers(1, 5)
    for _ in range(num_blocks):
        width = rng.integers(max(2, n // 100), max(3, n // 10))
        start = rng.integers(0, max(1, n - width))
        scale = np.exp(rng.normal(0, strength))
        candidate[start:start+width] *= scale
    candidate = np.maximum(candidate, 0)
    s = np.sum(candidate)
    if s > 0.01:
        candidate = candidate / s
    return candidate
