def _convolve(f):
    """Linear autocorrelation (convolution of f with itself)."""
    n = len(f)
    size = 2 * n - 1
    F = np.fft.rfft(f, n=size)
    return np.fft.irfft(F * F, n=size)

def _softmax_weights(conv, tau):
    """Compute softmax weights of convolution values."""
    # Numerically stable softmax
    shift = np.max(conv)
    exp_vals = np.exp(tau * (conv - shift))
    total = np.sum(exp_vals)
    weights = exp_vals / total
    A = (1.0 / tau) * (np.log(total) + tau * shift)

    while time.monotonic() < deadline:
        conv = _fft_autocorr(f)
        max_b = np.max(conv)
        score = 2.0 * n * max_b

        if score < best_score - 1e-12:
            best_score = score
            best_f = f.copy()
            no_improve = 0
            if verbose and iters % 5000 == 0:
                elapsed = time.monotonic() - (_optimize._t0 if hasattr(_optimize, '_t0') else time.monotonic())
                print(f"      iter={iters}, score={score:.8f}, tau={tau:.1f}")
        else:
            no_improve += 1

        if no_improve > best_no_improve_limit:
            tau = min(tau * 1.5, 20000.0)
            lr *= 0.6
            mu = min(mu + 0.03, 0.96)
            best_no_improve_limit = max(1000, int(best_no_improve_limit * 0.7))
            if lr < 1e-14:
                break
            no_improve = 0

        # Softmax weights
        shift = max_b
        ex = np.exp(tau * (conv - shift))
        tot = ex.sum()
        if tot < 1e-300:
            break
        weights = ex / tot
        A = (np.log(tot) + tau * shift) / tau

        # Gradient: (w*f)[j] - mean_active
        wf = _fft_grad_term(weights, f)
        active = f > 1e-14
        if active.any():
            lam = wf[active].mean()
        else:
            lam = wf.mean()
        grad = wf - lam

        # Momentum update
        vel = mu * vel - lr * grad
        f_new = f + vel
        f_new = np.clip(f_new, 0, None)
        S_new = f_new.sum()
        if S_new < 1e-15:
            f = best_f.copy()
            vel[:] = 0
            lr *= 0.5
            break
        f = f_new / S_new

        iters += 1

    return best_f, best_score


def _local_search(f, rng, deadline, max_steps=20000):
    """Block perturbation local search on normalized sequence."""
    n = len(f)
    best_f = f.copy()
    best_score = 2.0 * n * np.max(_fft_autocorr(best_f))

    step_size = 0.1
    width = max(1, n // 200)
    accepted = 0
    steps = 0

    while time.monotonic() < deadline and steps < max_steps:
        start = rng.integers(0, max(1, n - width))
        end = min(n, start + width)
        idx = np.arange(start, end)

        cand = best_f.copy()
        cand[idx] += rng.normal(0, step_size, size=len(idx))
        cand = np.clip(cand, 0, None)
        S = cand.sum()
        if S < 1e-10:
            steps += 1
            continue
        cand /= S

        conv = _fft_autocorr(cand)
        sc = 2.0 * n * np.max(conv)
        steps += 1

        if sc < best_score:
            best_score = sc
            best_f = cand
            accepted += 1

        if steps % 1000 == 0:
            if accepted / 1000.0 > 0.4:
                step_size *= 1.3
            elif accepted / 1000.0 < 0.05:
                step_size *= 0.6
            else:
                step_size *= 0.9
            accepted = 0

    return best_f, best_score


def _try_parent(deadline):
    """Try to optimize the parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.clip(parent, 0, 1000.0)
            S = parent.sum()
            if S > 0:
                return parent / S
    except (NameError, Exception):
        pass
    return None


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(f*f)/(sum(f))^2.
    Uses soft-max gradient descent with momentum, multiple initializations
    and lengths, followed by local search refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_deadline = t_start + budget_s - 20.0

    best_seq = None
    best_score = float('inf')

    def _report(score, n):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST: score={score:.8f}, n={n}, t={elapsed:.1f}s")

    # Phase 1: Optimize parent
    parent_f = _try_parent(total_deadline)
    if parent_f is not None:
        n_p = len(parent_f)
        print(f"Phase 1: Parent n={n_p}")
        phase1_end = min(total_deadline, t_start + 350)
        f_opt, score = _optimize(parent_f, rng, phase1_end, verbose=True)
        if best_seq is None:
            best_seq = f_opt.copy()
            best_score = score
            print(f"  Parent optimized: {score:.8f}")
        _report(score, n_p)

        # Local search on result
        ls_end = min(total_deadline, t_start + 450)
        if time.monotonic() < ls_end:
            f_ls, score_ls = _local_search(f_opt, rng, ls_end)
            if best_seq is None or score_ls < best_score:
                best_seq = f_ls.copy()
                best_score = score_ls
            _report(score_ls, n_p)

    # Phase 2: Multiple lengths and initializations
    lengths = [4096, 2048, 8192, 3072, 6144, 5120, 16384, 2560, 1024, 12288]
    # Sort by expected value: shorter lengths converge faster
    time_remaining = max(60, (total_deadline - time.monotonic()))
    time_per_config = time_remaining / (len(lengths) * 2)

    print(f"Phase 2: Multi-length search, ~{time_per_config:.0f}s per config")

    for n in lengths:
        if time.monotonic() > total_deadline - 60:
            break

        for ii in range(2):
            init_end = time.monotonic() + time_per_config
            if time.monotonic() > init_end:
                continue

            if ii == 0:
                # Random
                f = rng.random(n) + 0.01
            else:
                # Block-structured
                bs = max(1, n // 60)
                f = np.zeros(n)
                nb = (n + bs - 1) // bs
                for b in range(nb):
                    s = b * bs
                    e = min(n, s + bs)
                    if rng.random() < 0.6:
                        f[s:e] = rng.uniform(0.3, 1.0)
                    else:
                        f[s:e] = rng.uniform(0.0, 0.3)
                f += 0.01

            f /= f.sum()

            opt_end = time.monotonic() + time_per_config * 0.75
            f_opt, score = _optimize(f, rng, opt_end)
            if best_seq is None or score < best_score:
                if best_seq is None:
                    best_seq = f_opt.copy()
                _report(score, n)

            # Quick local search
            ls_end = min(init_end, time.monotonic() + time_per_config * 0.2)
            if time.monotonic() < ls_end:
                f_ls, score_ls = _local_search(f_opt, rng, ls_end)
                _report(score_ls, n)

    # Phase 3: Final refinement of best
    if best_seq is not None:
        print(f"Phase 3: Refining best (n={len(best_seq)}, score={best_score:.8f})")
        fin_end = total_deadline
        f_ref, score_ref = _optimize(best_seq, rng, fin_end, verbose=True)
        _report(score_ref, len(best_seq))

        if time.monotonic() < fin_end - 5:
            f_ls, score_ls = _local_search(f_ref, rng, fin_end, max_steps=50000)
            _report(score_ls, len(f_ls))

    if best_seq is None:
        best_seq = np.ones(1024)
        best_score = 2.0

    print(f"\nFINAL: score={best_score:.8f}, length={len(best_seq)}, total_time={time.monotonic()-t_start:.1f}s")

    # Optionally verify with evaluator
    try:
        actual = evaluate_sequence([float(x) for x in best_seq])
        print(f"Evaluator score: {actual:.8f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
