def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""

import numpy as np
import time


def compute_gradient(a, epsilon):
    """Compute gradient of smoothed max of auto-correlation of a via FFT."""
    n = len(a)
    size = 2 * n - 1

    # b = a * a (full linear convolution)
    fftsz = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fftsz)
    b = np.fft.irfft(fa * fa, fftsz)[:size]

    # Softmax weights
    bs = b / epsilon
    bs -= np.max(bs)
    p = np.exp(bs)
    p /= np.sum(p)

    # grad_j = 2 * sum_m p_{j+m} * a_m = 2 * (p star a)_j
    # (p star a)_j = (p^R * a)_{(2n-2)-j}
    pR = p[::-1]
    conv_len = 3 * n - 2
    fftsz2 = 1 << (conv_len - 1).bit_length()
    fpR = np.fft.rfft(pR, fftsz2)
    fa2 = np.fft.rfft(a, fftsz2)
    conv = np.fft.irfft(fpR * fa2, fftsz2)[:conv_len]

    g = conv[n - 1:2 * n - 1][::-1]
    return 2.0 * g, b


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

    n = len(a_init)

                best_a = a_new.copy()
                stagnant = 0
            else:
                stagnant += 1

            # Decay

            epsilon = max(0.005 / n, epsilon * 0.99)

            # Kick if stagnant

                noise = rng.normal(0, 1.0 / n, n)
                a_new = project_simplex(a_new + noise)
                stagnant = 0

        a = a_new

    return best_a, best_score
