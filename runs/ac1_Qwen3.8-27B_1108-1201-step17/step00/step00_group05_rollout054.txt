def solve_lp(a, M, active_indices, n):

    # A is (len(active_indices), n)
    A_ub = np.zeros((len(active_indices), n))
    for i, k in enumerate(active_indices):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A_ub[i, j] = a[k-j]
    
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    res = linprog(c_obj, A_ub=A_ub, b_ub=np.full(len(active_indices), M), 
                  bounds=[(0, None)]*n, method='highs')
    return res.x

import time
from scipy.optimize import linprog

def compute_convolution(a):
    """Compute full auto-convolution."""
    return np.convolve(a, a, mode='full')

def get_active_set(c, threshold=None):
    """Get indices where c is close to the maximum."""
    if threshold is None:
        threshold = 0.99 * c.max()
    return np.where(c >= threshold)[0]

def solve_lp_active(a, M, active_indices, n):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active, b >= 0."""
    m = len(active_indices)
    if m == 0:
        return np.ones(n) / n
    
    # Build constraint matrix A (m x n)
    # A[i, j] = a[k_i - j] if 0 <= k_i - j < n, else 0
    A = np.zeros((m, n))
    for i, k in enumerate(active_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[i, j_min:j_max+1] = a[k - np.arange(j_min, j_max+1)]
    
    # Objective: minimize -sum(b)
    c_obj = -np.ones(n)
    b_ub = np.full(m, M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, 
                     bounds=[(0, None)]*n, method='highs')
        if res.success:
            return res.x
    except:
        pass

def build_convolution_matrix(a, n):
    """Build the (2n-1) x n convolution matrix."""
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)

        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

```python
def solve_lp_iterative(a, M, n, max_rounds=5):
    """Solve LP with constraint generation."""
    c = np.convolve(a, a, 'full')
    active = np.where(c >= 0.95 * M)[0]
    
    for _ in range(max_rounds):
        # Build constraint matrix for active constraints
        m = len(active)
        A = np.zeros((m, n))
        for i, k in enumerate(active):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_min <= j_max:
                A[i, j_min:j_max+1] = a[k - np.arange(j_min, j_max+1)]
        
        c_obj = -np.ones(n)
        b_ub = np.full(m, M)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        
        if not res.success:
            return None
        
        b = res.x
        
        # Check for violations
        ab = np.convolve(a, b, 'full')
        violated = np.where(ab > M * 1.01)[0]
        
        if len(violated) == 0:
            return b
        
        # Add violated constraints to active set
        active = np.unique(np.concatenate([active, violated]))
    
    # Final solve with all active constraints
    m = len(active)
    A = np.zeros((m, n))
    for i, k in enumerate(active):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[i, j_min:j_max+1] = a[k - np.arange(j_min, j_max+1)]
    
    c_obj = -np.ones(n)
    b_ub = np.full(m, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None
```

1. The first LP (with only peak constraints) is very fast.
