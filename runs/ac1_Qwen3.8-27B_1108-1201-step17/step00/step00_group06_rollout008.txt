import time

def smooth_max(c, temperature=0.01):
    """Compute a smooth approximation of max(c)."""
    scaled = c / temperature
    max_val = np.max(scaled)
    return temperature * (max_val + np.log(np.sum(np.exp(scaled - max_val))))

def smooth_max_grad(c, temperature=0.01):
    """Compute gradient of smooth max."""
    scaled = c / temperature
    max_val = np.max(scaled)
    exp_vals = np.exp(scaled - max_val)
    weights = exp_vals / np.sum(exp_vals)
    return weights / temperature  # This is the subgradient direction
