import time

def compute_grad(a):
    """Compute the gradient of F(a) = 2*n*max(conv(a,a)) / (sum(a))^2"""
    n = len(a)
    S = np.sum(a)
    c = np.convolve(a, a)
    max_c = np.max(c)
    
    # Find lags where c is near max (within tolerance)
    tol = 0.01 * max_c
    near_max = np.where(c >= max_c - tol)[0]
    
    # Average gradient over near-max lags
    grad = np.zeros(n)
    for k in near_max:
        m = np.arange(n)
        idx = k - m
        valid = (idx >= 0) & (idx < n)
        g = np.zeros(n)
        g[valid] = 2.0 * a[idx[valid]]
        grad += g
    
    grad /= len(near_max)
    grad = (2.0 * n / S**2) * (grad - max_c / S)
    
    return grad, max_c, S

    c = np.convolve(a, a)
    max_c = float(np.max(c))
    score = 2.0 * n * max_c / (S * S)
    
    # Find near-max lags
    tol = max(1e-6, 0.005 * max_c)
    near_max = np.where(c >= max_c - tol)[0]
    
    # Compute gradient
    grad = np.zeros(n)
    for k in near_max:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    grad /= len(near_max)
    grad = (2.0 * n / (S * S)) * (grad - max_c / S)
    
    return grad, score

def gradient_descent(a, rng, deadline, initial_lr=0.01):
    """Run gradient descent on sequence a until deadline."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    
    # Compute initial score
    _, best_score = compute_grad_score(a)
    if not np.isfinite(best_score):
        return a, np.inf
    
    lr = initial_lr
    momentum = 0.9
    v = np.zeros(n)
    stale = 0
    patience = 100
    
    while time.monotonic() < deadline:
        grad, score = compute_grad_score(a)
        if grad is None:
            break
        
        if score < best_score:
            best_a = a.copy()
            best_score = score
            stale = 0
        else:
            stale += 1
        
        # Gradient step with momentum
        v = momentum * v - lr * grad
        a = a + v
        a = np.clip(a, 0, 1000)
        
        # Adaptive learning rate
        if stale > patience:
            lr *= 0.5
            stale = 0
            v = np.zeros(n)
            a = best_a + rng.normal(0, 0.01, n)
            a = np.clip(a, 0, 1000)
            if lr < 1e-6:
                break
    
    return best_a, best_score


def random_perturbation(a, rng):
    """Apply a random perturbation to escape local minima."""
    n = len(a)
    a = a.copy()
    n_pert = rng.integers(1, max(2, n // 10))
    indices = rng.choice(n, size=n_pert, replace=False)
    a[indices] *= np.exp(rng.normal(0, 0.1, n_pert))
    return np.clip(a, 0, 1000)
