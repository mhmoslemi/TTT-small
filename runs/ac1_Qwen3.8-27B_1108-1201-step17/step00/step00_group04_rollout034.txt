    4. Use gradient-based fine-tuning between LP iterations

            raise ValueError("invalid starting sequence")

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial score: {best_score:.6f}")

        f = f / f.sum()

"""LP-based iterative optimization for minimum-peak autocorrelation sequences.
Uses alternating LP (to find spread-out direction) and blending to reduce
the peak of the autocorrelation below the uniform-sequence baseline of 2.0.
"""
import numpy as np
import time
from scipy.optimize import linprog


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix T where T[k,j] = f[k-j]."""
    cols = np.arange(n)[None, :]
    rows = np.arange(2*n-1)[:, None]
    idx = rows - cols
    T = np.zeros((2*n-1, n), dtype=np.float64)
    valid = (idx >= 0) & (idx < n)
    T[valid] = f[idx[valid]]
    return T


def gradient_step(f, eta, n):
    """One projected subgradient step to reduce max autocorrelation."""
    ff = np.convolve(f, f)
    k_star = np.argmax(ff)
    # Subgradient: grad_i = 2 * f[k_star - i] if 0 <= k_star - i < n
    grad = np.zeros(n)
    valid_start = max(0, k_star - n + 1)
    valid_end = min(n, k_star + 1)
    for i in range(valid_start, valid_end):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * f[j]
    
    # Projected gradient step on simplex
    new_f = f - eta * grad
    # Project onto simplex {a >= 0, sum(a) = 1}
    new_f = np.maximum(new_f, 0)
    s = new_f.sum()
    if s > 1e-10:
        new_f = new_f / s
    else:
        new_f = np.ones(n) / n
    return new_f
