Multi-start projected gradient descent with momentum on the probability simplex

Uses smoothed-max objective with FFT-accelerated gradient computation and

def project_onto_simplex(v, target_sum):
    """Project vector v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = int(np.max(np.nonzero(cond)[0]))
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_fft_size(n):
    """Return next power of 2 >= 3n-2 (needed for cross-correlation)."""
    N = 1
    while N < 3 * n - 2:
        N *= 2
    return N


def fft_reverse_relation(f_fft, n, N):
    """Compute rfft of reversed f from rfft of f using phase relation."""
    k = np.arange(len(f_fft))
    phase = np.exp(-2j * np.pi * (n - 1) * k / N)
    return phase * np.conj(f_fft)
