import numpy as np
import time

def project_simplex(y):
    """Project y onto {x >= 0, sum(x) = 1}."""
    n = len(y)
    # Water-filling: find t such that sum(max(y - t, 0)) = 1
    # Binary search on t
    y_sorted = np.sort(y)[::-1]
    # Find the largest k such that y_sorted[k] > (sum(y_sorted[:k]) - 1) / k
    # This is the standard simplex projection algorithm
    cv = np.cumsum(y_sorted)
    rho = np.arange(1, n+1)
    cond = y_sorted - (cv - 1) / rho > 0
    if not np.any(cond):
        # All equal, return uniform
        return np.ones(n) / n
    k = np.max(np.where(cond)[0]) + 1
    t = (cv[k-1] - 1) / k
    return np.maximum(y - t, 0)

def project_to_simplex(y):
    """Project y onto {x >= 0, sum(x) = 1}."""
    n = len(y)
    u = np.sort(y)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    k = k[-1]
    theta = (cssv[k] - 1.0) / (k + 1)
    return np.maximum(y - theta, 0.0)
