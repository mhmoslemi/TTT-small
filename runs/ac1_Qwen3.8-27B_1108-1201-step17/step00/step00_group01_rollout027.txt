import numpy as np
import time

def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_gradient_smoothed(y, weights):
    """Compute the smooth gradient of max(a*a) using softmax weights."""
    n = len(y)
    N = 2 * n
    W = np.fft.rfft(weights, N)
    Y = np.fft.rfft(y, N)
    g = 2.0 * np.real(np.fft.irfft(W * np.conj(Y), N))[:n]
    return g

    """
    Minimize 2n*max(a*a)/(sum(a))^2 using smoothed gradient descent with FISTA.

    and minimize max(a*a). We use a softmax approximation of the max to get smooth
    gradients, and FISTA (Nesterov acceleration) for faster convergence.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size

        parent = np.ones(256, dtype=np.float64)
        n = parent.size
    
    a = parent / np.sum(parent)  # Normalize to sum 1
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial: n={n}, score={best_score:.6f}")
    
    # FISTA parameters
    tau = 0.005  # Temperature for softmax (smaller = closer to true max)
    tau_min = 0.0005
    eta = 0.5  # Step size (for the smoothed gradient)
    k_fista = 1.0
    
    y = a.copy()  # FISTA extrapolation variable
    
    # We'll use a decreasing temperature schedule
    n_total_iters = 200000
    tau_decay = (tau_min / tau) ** (1.0 / n_total_iters)
    
    for iteration in range(n_total_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution of y with itself
        b = np.convolve(y, y, mode='full')
        b_max = np.max(b)
        
        # Softmax weights
        exp_b = np.exp((b - b_max) / tau)
        weights = exp_b / np.sum(exp_b)
        
        # Compute smooth gradient
        g = compute_gradient_smoothed(y, weights)
        
        # Gradient step
        a_new = y - eta * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * k_fista * k_fista)) / 2.0
        y = a_new + ((k_fista - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)  # Keep non-negative
        s = np.sum(y)
        if s > 0:
            y = y / s  # Renormalize (approximately in simplex)
        else:
            y = a_new
        a = a_new
        k_fista = t_new
        
        # Update temperature
        tau = max(tau_min, tau * tau_decay)
        
        # Periodic evaluation
        if iteration % 500 == 0:
            score = float(evaluate_sequence(a_new.tolist()))
            if score < best_score:
                best = a_new.copy()
                best_score = score
                print(f"Iter {iteration}: score={best_score:.6f}, tau={tau:.6f}")

    if score < best_score:

        best_score = score
    
    print(f"Final: score={best_score:.6f}")
