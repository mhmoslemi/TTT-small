import time

def _smooth_objective_grad(a, beta):
    """Compute smooth objective (logsumexp) and gradient."""
    n = len(a)
    c = np.convolve(a, a)
    s = np.sum(a)
    if s < 1e-10:
        return 1e10, np.zeros_like(a)
    
    max_c = np.max(c)
    shifted = beta * (c - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    lse = max_c + np.log(sum_exp) / beta
    
    obj = 2.0 * n * lse / (s * s)

    w = exp_shifted / sum_exp
