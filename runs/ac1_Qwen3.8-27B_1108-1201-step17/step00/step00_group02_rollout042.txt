This is a weighted average of the per-lag gradients, with weights concentrated on the maximum lags.

However, computing the softmax over 2n-1 values and then the weighted sum is O(n) per iteration (after the O(n) convolution). This is fine.

Actually, for numerical stability, I should subtract the max before the exponential:
`p_k = exp(c[k] - c_max) / sum_j exp(c[j] - c_max)`

And the gradient is:
`df/da[m] = 2 * sum_k p_k * a[k-m]`

This is a "blended" version of a, where each position m gets contributions from a[k-m] weighted by p_k.

Let me code this up properly:

```python
import numpy as np
import time

def soft_gradient(a, c, tau):
    """Compute gradient of soft-max of convolution values."""
    n = len(a)
    # Compute softmax weights
    c_shifted = c - c.max()
    exp_c = np.exp(c_shifted)
    p = exp_c / exp_c.sum()
    
    # Compute weighted gradient
    # g[m] = 2 * sum_k p[k] * a[k-m] (for valid k-m)
    g = np.zeros(n)
    
    # For efficiency, we can compute this as a weighted sum of shifted a's
    # But for n ~ 7000, a direct loop over k (with p[k] > threshold) is fine
    # Only the top few p[k] values are significant
    threshold = 1e-6
    significant = np.where(p > threshold)[0]
    
    for k in significant:
        # g[m] += 2 * p[k] * a[k-m] for valid m
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            indices = k - np.arange(lo, hi + 1)
            g[lo:hi+1] += 2.0 * p[k] * a[indices]
    
    return g
```

The number of "significant" k values depends on tau. For small tau, only a few lags have significant weight. For larger tau, more lags contribute.

For n = 6962, the convolution has 13923 values. If tau is small enough that only ~10-50 values have significant weight, the loop is fast.

import numpy as np
import time

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes J(a) = 2n*max(convolve(a,a))/(sum(a))^2 using soft-gradient descent.
    Starts from a linear ramp and iteratively reduces the peak autocorrelation
    by optimizing the sequence shape.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0
    
    n = 6962
    
    # Initialize with a linear ramp (gives J ≈ 4/3)
    j = np.arange(1, n+1, dtype=np.float64)
    a = j / j[-1]
    
    # Normalize to sum = 1 (J is scale-invariant)
    a = a / a.sum()
    
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    print(f"Initial: n={n}, J={best_score:.6f}")
    
    # Hyperparameters
    tau = 0.01  # Softmax temperature
    lr = 0.1
    lr_min = 1e-8
    
    for iteration in range(50000):
        if time.monotonic() > deadline:
            break
        
        c = np.convolve(a, a, mode='full')
        S = a.sum()
        max_c = c.max()
        current_J = 2.0 * n * max_c / (S * S)
        
        if current_J < best_score:
            best_score = current_J
            best = a.copy()
        
        if iteration % 100 == 0:
            print(f"Iter {iteration}: J={current_J:.6f}, best={best_score:.6f}, lr={lr:.6f}, tau={tau:.4f}")
            # Slowly increase tau to consider more lags
            tau = min(0.1, tau * 1.01)
        
        # Compute soft gradient
        c_shifted = c - c.max()
        exp_c = np.exp(c_shifted)
        p = exp_c / exp_c.sum()
        
        # Compute weighted gradient g[m] = 2 * sum_k p[k] * a[k-m]
        g = np.zeros(n, dtype=np.float64)
        threshold = 1e-8
        significant = np.where(p > threshold)[0]
        
        for k in significant:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                indices = k - np.arange(lo, hi + 1)
                g[lo:hi+1] += 2.0 * p[k] * a[indices]
        
        # Update: minimize sum of top c[k] values
        # We want to reduce c[k] for high-p[k] lags
        # dc[k]/da[m] = 2*a[k-m], so to reduce c[k], decrease a[m] where a[k-m] is large
        # But we also need to maintain sum = 1 (or at least keep it positive)
        # Simple approach: a -= lr * g, then clip and renormalize
        a = a - lr * g
        a = np.maximum(a, 0.0)
        
        # Renormalize
        S_new = a.sum()
        if S_new > 1e-10:
            a = a / S_new
        else:
            # Reset if all zeros
            a = j / j[-1]
            a = a / a.sum()
            break
        
        # Decay learning rate
        lr *= 0.999
        if lr < lr_min:
            lr = lr_min
    
    # Final evaluation
    final_score = evaluate_sequence(best.tolist())
    print(f"Final: J={final_score:.6f}")
    
    return [float(x) for x in best]
