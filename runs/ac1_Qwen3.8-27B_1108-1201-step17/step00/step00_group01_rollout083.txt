```
   if len(lst) >= max_items: return
   if a is None: return
   arr=np.asarray(a,dtype=np.float64)
   arr=_sanitize_normalize(arr)
   for b in lst:
      if b.size == arr.size:
         diff=float(np.linalg.norm(arr - b))
         if diff < 1e-10:
            return
   lst.append(arr)
```

```python
"""
Search for a non-negative step function minimizing
    2 * n * max(convolve(a, a)) / (sum(a) ** 2).

The main strategy is:
1. Try a few sequence lengths.
2. Optimize a low-dimensional Fourier/polynomial parameterization using
   L-BFGS-B on a smooth surrogate of the max-convolution objective.
3. Polish the full non-negative sequence with smooth local perturbations
   and adaptive gradient-like multiplicative steps.
4. Verify the final answer with the supplied exact evaluator.
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import minimize as _scipy_minimize
except Exception:
    _scipy_minimize = None

_TWO_PI = 2.0 * math.pi

# Global state for the low-dimensional Fourier objective callback.
_FOUR_N = 0
_FOUR_ONES = None
_FOUR_BASIS = None
_FOUR_TEMP_ABS = 0.0


def _fft_convolution(x):
    """Full linear convolution x*x using FFT."""
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    X = np.fft.rfft(x, N)
    Y = np.fft.irfft(X * X, N)
    return Y[:L]


def _fast_objective(x):
    """Fast scale-invariant objective for positive/fixed-sum search sequences."""
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    if n == 0:
        return np.inf
    if not np.all(np.isfinite(x)):
        return np.inf
    s = float(np.sum(x))
    if s < 1e-12 or not math.isfinite(s):
        return np.inf
    c = _fft_convolution(x)
    if c.size == 0:
        return np.inf
    mc = float(c.max())
    if not math.isfinite(mc):
        return np.inf
    val = 2.0 * n * mc / (s * s)
    if not math.isfinite(val):
        return np.inf
    return val


def _normalize(x):
    """Clamp to nonnegative, remove non-finite values, normalize sum to 1."""
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, 0.0, 1000.0)
    s = float(np.sum(x))
    n = x.size if x.size > 0 else 1
    if s <= 1e-12:
        return np.ones(n, dtype=np.float64) / float(n)
    return x / s


def _true_objective_normalized(a):
    """Objective assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    c = _fft_convolution(a)
    if c.size == 0:
        return np.inf
    mc = float(c.max())
    if not math.isfinite(mc):
        return np.inf
    return float(2.0 * n * mc)


def _softmax_weights(c, temperature):
    """Softmax weights over convolution values, emphasizing the peak."""
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    if L == 0:
        return c
    if temperature <= 0.0 or not math.isfinite(temperature):
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    x = c / temperature
    x = x - float(np.max(x))
    e = np.exp(x)
    s = float(np.sum(e))
    if not math.isfinite(s) or s <= 0.0:
        w = np.zeros(L, dtype=np.float64)
        k = max(1, min(8, L))
        idx = np.argpartition(c, -k)[-k:]
        w[idx] = 1.0 / float(k)
        return w
    return e / s


def _smooth_F_grad(a, temp_div=10.0, temp_abs=0.0):
    """
    Return (F, gradient_a, c), where F = sum_k softmax(c/temp)_k c_k.
    gradient_a is the gradient of F with respect to a.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()

    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:L]

    if temp_abs is not None and temp_abs > 0.0:
        temp = temp_abs
    else:
        peak = float(c.max()) if c.size else 0.0
        temp = max(peak / float(temp_div), 1e-12)

    w = _softmax_weights(c, temp)
    F = float(np.dot(w, c))

    W = np.fft.rfft(w, N)
    g = 2.0 * np.fft.irfft(np.conj(W) * A, N)[:n]

    if not np.all(np.isfinite(g)):
        g = np.zeros(n, dtype=np.float64)
    if not math.isfinite(F):
        F = 1e100
    return F, g, c


def _simple_sine_a(n, alpha=0.6):
    """Fallback smooth non-negative sequence."""
    m = max(1, n - 1)
    t = np.arange(n, dtype=np.float64) / float(m)
    f = 1.0 + alpha * np.sin(_TWO_PI * t)
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s <= 1e-12:
        return np.ones(n, dtype=np.float64) / float(n)
    return f / s


def _fourier_basis(n, M):
    """
    Build basis matrix with M cosine/sine harmonics plus two
    low-order polynomial terms to allow boundary asymmetry.
    """
    total = 2 * M + 2
    basis = np.empty((n, total), dtype=np.float64)
    m = max(1, n - 1)
    t = np.arange(n, dtype=np.float64) / float(m)
    for q in range(1, M + 1):
        ang = _TWO_PI * float(q) * t
        basis[:, 2 * q - 2] = np.cos(ang)
        basis[:, 2 * q - 1] = np.sin(ang)
    basis[:, 2 * M] = t - 0.5
    basis[:, 2 * M + 1] = t * t - (1.0 / 3.0)
    return basis


def _fourier_bounds(M):
    """Bound Fourier and polynomial coefficients to improve numerical stability."""
    limits = [0.80, 0.32, 0.24, 0.18, 0.13, 0.10, 0.08, 0.06, 0.05, 0.04]
    bounds = []
    for m in range(1, M + 1):
        lim = limits[m - 1] if m - 1 < len(limits) else 0.03
        bounds.append((-lim, lim))
        bounds.append((-lim, lim))
    bounds.append((-0.35, 0.35))
    bounds.append((-0.25, 0.25))
    return bounds


def _fourier_a(basis, ones, x):
    """Reconstruct normalized non-negative sequence from coefficient vector x."""
    if x is None or len(x) != basis.shape[1]:
        return None
    x = np.asarray(x, dtype=np.float64)
    if not np.all(np.isfinite(x)):
        return None
    f = ones + basis.dot(x)
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s <= 1e-12:
        return None
    return f / s


def _fourier_start_list(M, seed):
    """Generate low-dimensional starting points."""
    total = 2 * M + 2
    rng = np.random.default_rng(seed)
    starts = []

    z = np.zeros(total, dtype=np.float64)
    first_amps = (0.25, 0.45, 0.60, 0.75, 0.90, 1.00)
    for a in first_amps:
        x = z.copy()
        x[1] = float(a)
        starts.append(x)

    x = z.copy()
    x[1] = -0.60
    starts.append(x)

    x = z.copy()
    x[1] = 0.60
    x[3] = 0.20
    starts.append(x)

    x = z.copy()
    x[1] = 0.60
    x[3] = -0.20
    starts.append(x)

    # Random starts.
    for _ in range(8):
        x = np.zeros(total, dtype=np.float64)
        for q in range(1, M + 1):
            if q == 1:
                amp = float(rng.uniform(0.15, 0.95))
            elif q == 2:
                amp = float(rng.uniform(0.0, 0.32))
            elif q == 3:
                amp = float(rng.uniform(0.0, 0.24))
            elif q == 4:
                amp = float(rng.uniform(0.0, 0.18))
            else:
                amp = float(rng.uniform(0.0, 0.10))
            ph = float(rng.uniform(0.0, 2.0 * math.pi))
            x[2 * q - 2] = amp * math.sin(ph)
            x[2 * q - 1] = amp * math.cos(ph)
        x[2 * M] = float(rng.normal(0.0, 0.08))
        x[2 * M + 1] = float(rng.normal(0.0, 0.06))
        starts.append(x)

    return starts


def _fourier_objective(x):
    """
    Smooth surrogate objective and gradient for the low-dimensional
    Fourier parameterization. Uses the global basis state.
    """
    global _FOUR_N, _FOUR_ONES, _FOUR_BASIS, _FOUR_TEMP_ABS
    x = np.asarray(x, dtype=np.float64)
    n = _FOUR_N
    if n <= 0 or _FOUR_BASIS is None:
        return 1e100, np.zeros_like(x)

    f_raw = _FOUR_ONES + _FOUR_BASIS.dot(x)
    f = np.maximum(f_raw, 0.0)
    s = float(np.sum(f))
    if s <= 1e-12:
        return 1e100, np.zeros_like(x)
    a = f / s

    F, g_a, _ = _smooth_F_grad(a, temp_div=10.0, temp_abs=_FOUR_TEMP_ABS)

    mean = float(np.dot(a, g_a))
    g_f = (g_a - a * mean) / s

    if np.any(f_raw <= 0.0):
        g_f[f_raw <= 0.0] = 0.0

    g_x = _FOUR_BASIS.T.dot(g_f)
    if not np.all(np.isfinite(g_x)):
        g_x = np.zeros_like(x)

    obj = float(2.0 * n * F)
    if not math.isfinite(obj):
        obj = 1e100
    return obj, g_x


def _fourier_random_refine(basis, ones, bounds, x, deadline, seed):
    """Simple random coordinate refinement if scipy is unavailable."""
    x = np.asarray(x, dtype=np.float64).copy()
    best_x = x.copy()
    best_a = _fourier_a(basis, ones, best_x)
    if best_a is None:
        return None, np.inf
    best_f = _fast_objective(best_a)

    rng = np.random.default_rng(seed)
    amps = np.zeros(len(bounds), dtype=np.float64)
    for i, (lo, hi) in enumerate(bounds):
        amps[i] = max(1e-3, 0.04 * (hi - lo))

    no_improve = 0
    while time.monotonic() < deadline:
        x = best_x.copy()
        j = int(rng.integers(0, len(bounds)))
        x[j] += float(rng.normal(0.0, float(amps[j])))
        lo, hi = bounds[j]
        x[j] = min(float(hi), max(float(lo), x[j]))

        a = _fourier_a(basis, ones, x)
        if a is None:
            continue
        f = _fast_objective(a)
        if f < best_f * (1.0 - 1e-12):
            best_f = f
            best_x = x.copy()
            best_a = a
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 300:
                amps *= 0.92
                if np.any(amps < 1e-4):
                    amps = np.maximum(amps, 1e-3)
                no_improve = 0

    return best_a, best_f


def _fourier_refine(n, deadline, seed):
    """
    Low-dimensional Fourier optimization. Returns (best_a, best_fast_score).
    """
    global _FOUR_N, _FOUR_ONES, _FOUR_BASIS, _FOUR_TEMP_ABS

    M = 10 if n <= 16384 else 8
    ones = np.ones(n, dtype=np.float64)
    basis = _fourier_basis(n, M)
    bounds = _fourier_bounds(M)

    _FOUR_N = n
    _FOUR_ONES = ones
    _FOUR_BASIS = basis
    _FOUR_TEMP_ABS = 1.0 / (30.0 * float(n))

    starts = _fourier_start_list(M, seed + 17)
    best_a = None
    best_f = np.inf
    best_x = starts[0]
    scored = []

    for idx, x in enumerate(starts):
        a = _fourier_a(basis, ones, x)
        if a is None:
            continue
        f = _fast_objective(a)
        if math.isfinite(f) and f < best_f:
            best_f = f
            best_a = a
            best_x = x.copy()
        scored.append((f, idx))

    scored.sort()

    if _scipy_minimize is not None:
        for rank, (score0, idx) in enumerate(scored):
            if time.monotonic() >= deadline:
                break
            x0 = starts[idx]
            if rank < 3:
                maxiter = 35
                maxfun = 320
            else:
                maxiter = 18
                maxfun = 130

            try:
                res = _scipy_minimize(
                    _fourier_objective,
                    x0,
                    method="L-BFGS-B",
                    jac=True,
                    bounds=bounds,
                    options={
                        "maxiter": maxiter,
                        "maxfun": maxfun,
                        "ftol": 1e-14,
                        "gtol": 1e-8,
                        "maxls": 10,
                        "disp": False,
                    },
                )
            except Exception:
                continue

            if getattr(res, "x", None) is not None:
                x = np.asarray(res.x, dtype=np.float64)
                if np.all(np.isfinite(x)):
                    a = _fourier_a(basis, ones, x)
                    if a is not None:
                        f = _fast_objective(a)
                        if math.isfinite(f) and f < best_f:
                            best_f = f
                            best_a = a
                            best_x = x.copy()

        # Sharpen the smooth surrogate and run from the best point if time remains.
        if time.monotonic() < deadline - 10.0:
            for _ in range(2):
                if time.monotonic() >= deadline:
                    break
                if _FOUR_TEMP_ABS > 1.0 / (200.0 * float(n)):
                    _FOUR_TEMP_ABS *= 0.5
                try:
                    res = _scipy_minimize(
                        _fourier_objective,
                        best_x,
                        method="L-BFGS-B",
                        jac=True,
                        bounds=bounds,
                        options={
                            "maxiter": 30,
                            "maxfun": 250,
                            "ftol": 1e-14,
                            "gtol": 1e-8,
                            "maxls": 10,
                            "disp": False,
                        },
                    )
                except Exception:
                    continue
                if getattr(res, "x", None) is not None:
                    x = np.asarray(res.x, dtype=np.float64)
                    if np.all(np.isfinite(x)):
                        a = _fourier_a(basis, ones, x)
                        if a is not None:
                            f = _fast_objective(a)
                            if math.isfinite(f) and f < best_f:
                                best_f = f
                                best_a = a
                                best_x = x.copy()

    if _scipy_minimize is None or time.monotonic() < deadline:
        if time.monotonic() + 5.0 < deadline:
            ra, rf = _fourier_random_refine(basis, ones, bounds, best_x, deadline, seed + 991)
            if ra is not None and rf < best_f:
                best_f = rf
                best_a = ra

    return best_a, best_f


def _smooth_perturb(a, rng, amp, t=None, use_block=True):
    """Generate a smooth nonnegative multiplicative perturbation."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if t is None:
        m = max(1, n - 1)
        t = np.arange(n, dtype=np.float64) / float(m)

    factor = np.ones(n, dtype=np.float64)
    for _ in range(2):
        freq = float(rng.uniform(0.5, 4.0))
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        fa = amp * float(rng.uniform(0.4, 1.0))
        factor += fa * np.sin(_TWO_PI * freq * t + ph)

    if use_block and n > 16:
        low = max(2, n // 1200)
        high = max(low + 1, n // 25)
        if high > n:
            high = n
        if high > low:
            width = int(rng.integers(low, high))
            max_start = max(1, n - width)
            if max_start > 0:
                start = int(rng.integers(0, max_start))
                stop = min(n, start + width)
                if stop > start:
                    factor[start:stop] *= np.exp(rng.normal(0.0, float(amp), size=stop - start))

    factor = np.clip(factor, 0.05, 5.0)
    cand = a * factor
    return _normalize(cand)


def _smooth_local_search(a, deadline, seed):
    """Full-sequence smooth local search."""
    a = _normalize(a)
    n = a.size
    m = max(1, n - 1)
    t = np.arange(n, dtype=np.float64) / float(m)

    rng = np.random.default_rng(seed)
    current = a.copy()
    c = _fft_convolution(current)
    current_f = float(2.0 * n * c.max()) if c.size else np.inf

    best = current.copy()
    best_f = current_f
    amp = 0.045
    no_improve = 0

    while time.monotonic() < deadline:
        cand = _smooth_perturb(current, rng, amp, t, True)
        c = _fft_convolution(cand)
        f = float(2.0 * n * c.max()) if c.size else np.inf

        if math.isfinite(f) and f < current_f * (1.0 - 1e-12):
            current = cand
            current_f = f
            if f < best_f * (1.0 - 1e-12):
                best_f = f
                best = cand.copy()
                no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 90:
                amp *= 0.95
                if amp < 1e-3:
                    amp = 0.04
                no_improve = 0
                if current_f > best_f * (1.0 + 1e-10):
                    current = best.copy()
                    current_f = best_f

    return best, best_f


def _gradient_polish(a, deadline, seed):
    """Multiplicative gradient-like polishing on the full sequence."""
    a = _normalize(a)
    n = a.size
    m = max(1, n - 1)
    t = np.arange(n, dtype=np.float64) / float(m)

    rng = np.random.default_rng(seed)
    current = a.copy()

    c = _fft_convolution(current)
    current_f = float(2.0 * n * c.max()) if c.size else np.inf
    best = current.copy()
    best_f = current_f

    lr = 0.07
    temp_div = 6.0
    no_improve = 0

    while time.monotonic() < deadline:
        F, g_a, c = _smooth_F_grad(current, temp_div=temp_div)
        peak = float(c.max()) if c.size else 0.0
        if peak <= 0.0 or not math.isfinite(peak):
            break
        current_f = float(2.0 * n * peak)

        if current_f < best_f * (1.0 - 1e-12):
            best_f = current_f
            best = current.copy()

        g = float(n) * g_a
        mean = float(np.dot(current, g))
        delta = -lr * (g - mean)
        delta = np.clip(delta, -2.0, 2.0)

        cand = current * np.exp(delta)
        cand = _normalize(cand)

        cc = _fft_convolution(cand)
        f = float(2.0 * n * cc.max()) if cc.size else np.inf

        if math.isfinite(f) and f < current_f * (1.0 - 1e-12):
            current = cand
            current_f = f
            if f < best_f * (1.0 - 1e-12):
                best_f = f
                best = cand.copy()
                no_improve = 0
        else:
            no_improve += 1
            lr *= 0.82
            if no_improve >= 14:
                no_improve = 0
                lr = 0.12
                temp_div = max(3.0, temp_div * 0.85)
                if rng.random() < 0.25:
                    current = _smooth_perturb(current, rng, 0.015, t, True)
                    c2 = _fft_convolution(current)
                    current_f = float(2.0 * n * c2.max()) if c2.size else np.inf
                if current_f > best_f * (1.0 + 1e-9):
                    current = best.copy()
                    current_f = best_f

    return best, best_f


def _chosen_lengths(base_len, remaining_time):
    """Select a small set of lengths to try."""
    lengths = []
    if base_len is not None:
        b = int(base_len)
        if 512 <= b <= 16384:
            lengths.append(b)

    preferred = (6121, 8192, 12288, 4097)
    for L in preferred:
        if L not in lengths and 256 <= L <= 16384:
            lengths.append(L)

    if remaining_time < 260.0:
        lengths = lengths[:2]
    else:
        lengths = lengths[:4]

    out = []
    for L in lengths:
        L = int(L)
        if 128 <= L <= 16384 and L not in out:
            out.append(L)
    if not out:
        out = [6121]
    return out


def _add_unique_candidate(candidates, a, max_items=4):
    """Add a normalized candidate if not nearly identical to existing ones."""
    if len(candidates) >= max_items or a is None:
        return
    arr = _normalize(a)
    for b in candidates:
        if b.size == arr.size:
            diff = float(np.linalg.norm(arr - b))
            if diff < 1e-10:
                return
    candidates.append(arr)


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Returns a list of non-negative floats whose
    evaluation score is as low as found within the time budget.
    """
    start = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1800.0
    if not math.isfinite(budget) or budget <= 0.0:
        budget = 1800.0

    hard = start + max(1.0, budget - 30.0)

    base_len = None
    base_seq = None
    try:
        base = height_sequence_1  # noqa: F821
        if base is not None and len(base) > 0:
            base_seq = np.asarray(base, dtype=np.float64)
            if base_seq.ndim == 1 and np.all(np.isfinite(base_seq)):
                base_len = int(base_seq.size)
            else:
                base_seq = None
                base_len = None
    except Exception:
        base_seq = None
        base_len = None

    lengths = _chosen_lengths(base_len, max(0.0, hard - time.monotonic()))

    best_a = None
    best_fast = np.inf

    now = time.monotonic()
    fourier_limit = max(now + 5.0, hard - 70.0)
    share = max(2.0, min(70.0, max(0.0, fourier_limit - now) / float(len(lengths))))

    for idx, L in enumerate(lengths):
        dl = min(fourier_limit, time.monotonic() + share)
        if dl <= time.monotonic():
            continue
        a, f = _fourier_refine(L, dl, int(seed) + idx * 977)
        if a is not None and math.isfinite(f):
            if f < best_fast:
                best_fast = f
                best_a = a
            print(f"fourier n={L} fast={f:.8f}")

    if best_a is None:
        best_a = _simple_sine_a(6121, 0.60)
        best_fast = _fast_objective(best_a)
        print(f"fallback simple sine fast={best_fast:.8f}")

    now = time.monotonic()
    remaining = max(0.0, hard - now - 12.0)

    if remaining > 12.0:
        local_dl = min(hard - 10.0, now + 0.45 * remaining)
        a2, f2 = _smooth_local_search(best_a, local_dl, int(seed) + 31415)
        if a2 is not None and f2 < best_fast:
            best_fast = f2
            best_a = a2
        print(f"local fast={best_fast:.8f}")

    now = time.monotonic()
    remaining = max(0.0, hard - now - 6.0)
    if remaining > 8.0:
        grad_dl = min(hard - 3.0, now + remaining)
        a3, f3 = _gradient_polish(best_a, grad_dl, int(seed) + 27182)
        if a3 is not None and f3 < best_fast:
            best_fast = f3
            best_a = a3
        print(f"gradient fast={best_fast:.8f}")

    candidates = []
    _add_unique_candidate(candidates, best_a, max_items=4)
    _add_unique_candidate(candidates, _simple_sine_a(best_a.size, 0.60), max_items=4)
    if base_seq is not None and base_seq.size == best_a.size:
        _add_unique_candidate(candidates, base_seq, max_items=4)

    max_final_evals = 2 if best_a.size > 10000 else 3
    candidates = candidates[:max_final_evals]

    best_list = None
    best_exact = np.inf

    for a in candidates:
        if time.monotonic() > hard - 2.0:
            break
        arr = list(np.asarray(a, dtype=np.float64).flat)
        try:
            sc = evaluate_sequence(arr)  # noqa: F821
            sc = float(sc)
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < best_exact:
            best_exact = sc
            best_list = arr
            if sc < 1.25:
                break

    if best_list is None:
        arr = list(np.asarray(best_a, dtype=np.float64).flat)
        try:
            sc = float(evaluate_sequence(arr))  # noqa: F821
        except Exception:
            sc = np.inf
        if np.isfinite(sc):
            best_list = arr
            best_exact = sc

    if best_list is None:
        n = max(1, int(best_a.size))
        arr = [1.0 / float(n) for _ in range(n)]
        try:
            sc = float(evaluate_sequence(arr))  # noqa: F821
        except Exception:
            sc = np.inf
        best_list = arr
        best_exact = sc

    print(f"FINAL n={len(best_list)} exact={best_exact:.8f}")
    return best_list
```
