"""
Minimize  f(a) = 2*n*max(a*a) / (sum a)^2  over non-negative sequences a.

Key insight used here: the uniform sequence is a SADDLE point, not a local
minimum.  Antisymmetric (about the centre) perturbations lower the central
autocorrelation peak.  So we (1) start from skew / smooth profiles that break
the symmetry, (2) run a differentiable search on the log-sum-exp surrogate of
max(a*a) (a smooth upper bound on the peak whose gradient is explicit), and
(3) mix in random block perturbations + light annealing to escape local minima.

The objective is homogeneous of degree 0, so we keep sum(a)=1 throughout and
minimise 2*n*max(a*a).  The convolution is done with an FFT so long sequences
(used for extra degrees of freedom) stay fast.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------------
# Fast primitives (identical to evaluate_sequence but vectorised / FFT based)
# ----------------------------------------------------------------------------

def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def autcorr_full(a):
    """Aperiodic auto-convolution a*a of length 2n-1 via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    size = 2 * n - 1
    L = _next_pow2(size)
    A = np.fft.rfft(a, L)
    C = np.fft.irfft(A * A, L)
    return C[:size]


def objective(a):
    """Return 2*n*max(b)/(sum a)^2  (same value as evaluate_sequence for valid a)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = a.sum()
    if s <= 0:
        return np.inf
    b = autcorr_full(a)
    return float(2.0 * n * b.max() / (s * s))


def normalise(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 0:
        return np.ones_like(a) / a.size
    return a / s


# ----------------------------------------------------------------------------
# Initialiser family

def make_ramp(n, s):
    """Uniform plus an antisymmetric linear ramp (breaks the saddle)."""
    x = np.arange(n, dtype=np.float64)
    u = (x - 0.5 * (n - 1)) / (0.5 * (n - 1))  # in [-1, 1], antisymmetric
    a = 1.0 + s * u
    return normalise(np.clip(a, 1e-9, None))


def make_gauss(n, alpha):
    x = np.arange(n, dtype=np.float64)
    c = 0.5 * (n - 1)
    a = np.exp(-((x - c) ** 2) / (2.0 * (alpha * n) ** 2))
    return normalise(a)


def make_skew_gauss(n, alpha, s):
    x = np.arange(n, dtype=np.float64)
    c = 0.5 * (n - 1) + s * 0.25 * n
    a = np.exp(-((x - c) ** 2) / (2.0 * (alpha * n) ** 2))
    return normalise(np.clip(a, 1e-9, None))


def make_hann(n):
    x = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n - 1)))
    return normalise(np.clip(a, 1e-9, None))


def build_inits(n):
    inits = []
    inits.append(np.ones(n) / n)
    for s in (0.3, 0.6, 0.8):
        inits.append(make_ramp(n, s))
        inits.append(make_ramp(n, -s))
    for al in (0.18, 0.28, 0.4):
        inits.append(make_gauss(n, al))
        inits.append(make_skew_gauss(n, al, 1.0))
        inits.append(make_skew_gauss(n, al, -1.0))
    inits.append(make_hann(n))
    return inits

def grad_logsumexp(a, b, beta):
    """Gradient of (1/beta)*log sum_k exp(beta*b_k) w.r.t a (length n)."""
    n = a.size
    peak = b.max()
    w = np.exp(beta * (b - peak))
    w /= w.sum()
    # grad_i = 2 * sum_j a_j w_{i+j}  =  2 * corr(w, a)[i]
    g = np.correlate(w, a, mode="full")[:n]
    return 2.0 * g


def proj_grad_step(a, b, lr, beta):
    g = grad_logsumexp(a, b, lr, )  # placeholder, replaced below
    return a


def do_grad_step(a, lr):
    b = autcorr_full(a)
    peak = max(b.max(), 1e-12)
    beta = 5.0 / peak
    g = grad_logsumexp(a, b, beta)
    g = g - g.mean()  # project to keep the sum fixed
    a_new = a - lr * g
    a_new = 0.5 * a + 0.5 * a_new            # damping
    a_new = np.clip(a_new, 0.0, 1000.0)
    return normalise(a_new)


def random_block_move(a, rng):
    n = a.size
    out = a.copy()
    mode = rng.integers(0, 3)
    if mode == 0:  # multiplicative smooth bump / dip
        w = int(rng.integers(max(3, n // 64), max(4, n // 8)))
        st = int(rng.integers(0, max(1, n - w)))
        out[st:st + w] *= float(math.exp(rng.normal(0.0, 0.25)))
    elif mode == 1:  # move mass between two blocks
        w1 = int(rng.integers(max(3, n // 64), max(4, n // 12)))
        w2 = int(rng.integers(max(3, n // 64), max(4, n // 12)))
        s1 = int(rng.integers(0, max(1, n - w1)))
        s2 = int(rng.integers(0, max(1, n - w2)))
        m = 0.1 * a[s1].sum()
        out[s1:s1 + w1] -= m / max(1.0, w1)
        out[s2:s2 + w2] += m / max(1.0, w2)
    else:  # mirrored multiplicative wiggle
        w = int(rng.integers(max(3, n // 128), max(4, n // 32)))
        st = int(rng.integers(0, max(1, n - w)))
        e = float(math.exp(rng.normal(0.0, 0.35)))
        out[st:st + w] *= e
        out[max(0, n - 1 - st - w): n - st] *= 1.0 / e
    return normalise(np.clip(out, 0.0, 1000.0))

def run_local_search(a, n, deadline, rng):

    cur_score = objective(cur)
    best = cur.copy()
    best_score = cur_score
    lr = 0.05
    step = 0
    last_improve = step
    progress = 0.0
