"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses a combination of projected gradient descent (on the peak autocorrelation),
soft-max smoothing, and LP-based refinement (min peak autocorrelation approach).
Multiple starting points of varying lengths are explored within the time budget.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    """Normalize sequence to have a specific sum (default sqrt(2n))."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)


def _grad_one_step(a, lr, use_softmax=False, tau=50.0):
    """One projected gradient step minimizing peak autocorrelation (normalized)."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = _normalize(a, S_t)

    c = np.convolve(a, a)

    if use_softmax:
        # Soft-max weighted gradient
        c_shifted = c - c.max()
        w = np.exp(tau * c_shifted)
        w /= w.sum()
        # grad[j] = 2 * sum_k a[k] * w[j+k] for j+k < 2n-1
        grad = np.zeros(n)
        for j in range(n):
            k_max = min(n, 2 * n - 1 - j)
            if k_max > 0:
                grad[j] = 2.0 * np.dot(a[:k_max], w[j:j + k_max])
    else:
        # Hard max gradient
        m_star = int(c.argmax())
        grad = np.zeros(n)
        lo = max(0, m_star - n + 1)
        hi = min(n - 1, m_star)
        for j in range(lo, hi + 1):
            grad[j] = 2.0 * a[m_star - j]

    # Project out the mean (fixed-sum constraint)
    grad -= grad.mean()

    a = a - lr * grad
    a = np.maximum(a, 0.0)
    a = _normalize(a, S_t)
    return a


def _lp_solve(f, M):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M (elementwise), g >= 0."""
    n = len(f)
    if n > 2048:
        return None

    c_obj = -np.ones(n)

    # Build constraint matrix: A[m,j] = f[m-j] if 0<=m-j<n, else 0
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]

    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _optimize_sequence(a0, deadline, max_grad_iters=3000, max_lp_iters=200):
    """Optimize a single starting sequence. Returns (best_seq, best_score)."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = _normalize(a)
    score = _compute_obj(a)
    best = a.copy()
    best_score = score
    S_t = np.sqrt(2.0 * n)

    # Phase 1: Gradient descent with cosine-annealed learning rate
    lr_init = 0.15
    for it in range(max_grad_iters):
        if time.monotonic() > deadline - 30.0:
            break
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * it / max_grad_iters))
        lr = max(lr, 0.002)

        a = _grad_one_step(a, lr)

        if (it + 1) % 100 == 0:
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()

    # Phase 2: LP-based refinement
    if n <= 1024 and time.monotonic() < deadline - 15.0:
        c = np.convolve(a, a)
        M = float(c.max())

        for it in range(max_lp_iters):
            if time.monotonic() > deadline - 5.0:
                break

            g = _lp_solve(a, M)
            if g is None:
                break

            # Try blending with decreasing t
            improved = False
            for t in [0.8, 0.5, 0.3, 0.1]:
                a_new = (1.0 - t) * a + t * g
                a_new = np.maximum(a_new, 0.0)
                a_new = _normalize(a_new, S_t)
                s_new = _compute_obj(a_new)
                if s_new < best_score:
                    a = a_new
                    best = a.copy()
                    best_score = s_new
                    improved = True
                    break

            if not improved:
                # Try a small gradient step to escape
                a = _grad_one_step(a, 0.02)
                a = _normalize(a, S_t)
                s = _compute_obj(a)
                if s < best_score:
                    best = a.copy()
                    best_score = s
                    improved = True

            # Update M for next LP
            c = np.convolve(a, a)
            M = float(c.max())

    return best, best_score


def _optimize_sequence_soft(a0, deadline, max_iters=2000):
    """Optimize using soft-max gradient (smoother, may find different minima)."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = _normalize(a)
    score = _compute_obj(a)
    best = a.copy()
    best_score = score
    S_t = np.sqrt(2.0 * n)

    tau = 20.0
    lr = 0.1

    for it in range(max_iters):
        if time.monotonic() > deadline - 30.0:
            break

        # Anneal tau (increase to focus on peak) and decay learning rate
        tau = 20.0 + 30.0 * (it / max_iters)
        lr = 0.1 * (1.0 + np.cos(np.pi * it / max_iters)) * 0.5
        lr = max(lr, 0.003)

        a = _grad_one_step(a, lr, use_softmax=True, tau=tau)

        if (it + 1) % 150 == 0:
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()

    return best, best_score
