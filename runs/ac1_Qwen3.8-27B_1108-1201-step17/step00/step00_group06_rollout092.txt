   n=len(v)

   tmp=x.cumsum()
   rho=np.nonzero(x > (tmp-s)/np.arange(1,n+1))[0]

def top_bottom_indices(arr, k=30):
   n=arr.size

   if k <=0: k=1

   top_idx=np.argpartition(arr, n-k)[-k:]

   low_idx=np.argpartition(arr, k-1)[:k]

def fft_info(n):
    info = _FFT_CACHE.get(n)
    if info is not None:
        return info
    need = max(1, 3 * n - 2)
    N = 1
    while N < need:
        N <<= 1
    info = (N, 2 * n - 1)
    _FFT_CACHE[n] = info
    return info


def prepare_vector(a):
    if a is None:
        return None
    try:
        arr = np.asarray(a, dtype=float).reshape(-1)
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None
    arr = arr / s
    return arr


def binomial_sequence(n):
    if n <= 0:
        return np.array([], dtype=float)
    if n == 1:
        return np.array([1.0], dtype=float)
    k = np.arange(n - 1, dtype=float)
    ratios = (2.0 * k + 1.0) / (2.0 * k + 2.0)
    b = np.concatenate(([1.0], np.cumprod(ratios)))
    s = float(b.sum())
    if s <= 0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return b / s


def interpolate_sequence(a_small, n):
    a_small = prepare_vector(a_small)
    if a_small is None or n <= 0:
        return None
    if a_small.size == n:
        return a_small.copy()
    x_old = np.linspace(0.0, 1.0, a_small.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a_small)
    return prepare_vector(out)


def compute_y(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=float)
    N, m = fft_info(n)
    A = np.fft.rfft(a, N)
    C = np.fft.irfft(A * A, N)[:m]
    if C.size:
        C[C < 0.0] = 0.0
    return m * C


def internal_score(a):

def grad_from_weights(w, a, A=None):
    n = a.size
    N, m = fft_info(n)
    if A is None:
        A = np.fft.rfft(a, N)
    W = np.fft.rfft(w[::-1], N)
    H = np.fft.irfft(W * A, N)
    h = H[2 * n - 2:n - 2:-1]
    g = 2.0 * h
    if not np.all(np.isfinite(g)):
        g = np.zeros(n, dtype=float)
    return g


def smooth_eval_y(a, tau, lam, need_grad=True):
    n = a.size
    if n == 0:
        return 1e100, 0.0, None, np.array([], dtype=float)
    if not np.all(np.isfinite(a)):
        return 1e100, 0.0, None, np.zeros(n, dtype=float)

    N, m = fft_info(n)
    A = np.fft.rfft(a, N)
    C = np.fft.irfft(A * A, N)[:m]
    C[C < 0.0] = 0.0
    y = m * C

    if not np.all(np.isfinite(y)):
        return 1e100, 0.0, None, y

    M = float(np.max(y))
    tau = max(float(tau), 1e-3)
    u = (y - M) / tau
    u = np.minimum(u, 0.0)
    expu = np.exp(u)
    sumexp = float(expu.sum())

    if sumexp <= 0.0:
        S = M
        w_sm = np.zeros(m, dtype=float)
    else:
        S = M + tau * math.log(sumexp)
        w_sm = expu / sumexp

    V = float(np.mean((y - 1.0) ** 2))
    J = S + float(lam) * V

    if not np.isfinite(J):
        return 1e100, M, None, y

    if not need_grad:
        return J, M, None, y

    w = m * w_sm + float(lam) * 2.0 * (y - 1.0)
    g = grad_from_weights(w, a, A)
    return J, M, g, y


def project_simplex(v, s=1.0):
    v = np.asarray(v, dtype=float).reshape(-1)
    n = v.size
    if n == 0:
        return np.array([], dtype=float)
    if n == 1:
        return np.array([s], dtype=float)
    v = np.nan_to_num(v, nan=0.0, posinf=1e3, neginf=-1e3)

    x = np.sort(v)[::-1]
    tmp = x.cumsum()
    denom = np.arange(1, n + 1, dtype=float)
    cond = x > (tmp - s) / denom
    if not np.any(cond):
        out = np.full(n, s / n, dtype=float)
    else:
        rho = int(np.max(np.nonzero(cond)[0]))
        theta = (tmp[rho] - s) / float(rho + 1)
        out = np.maximum(v - theta, 0.0)
        tot = float(out.sum())
        if tot <= 0 or not np.isfinite(tot):
            out = np.full(n, s / n, dtype=float)
        else:
            out *= s / tot
    return out


def softmax_vector(x, n=None):
    x = np.asarray(x, dtype=float).reshape(-1)
    if n is None:
        n = x.size
    if n <= 0:
        return np.array([], dtype=float)
    if x.size != n:
        x = np.resize(x, n)
    x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
    shift = float(np.max(x))
    e = np.exp(x - shift)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n, dtype=float)
    a = e / s
    return prepare_vector(a)


def lb_objective(x):

    if x.size != n:
        x = np.resize(x, n)
    x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)

    a = softmax_vector(x, n)
    J, M, g, y = smooth_eval_y(a, float(_LB['tau']), float(_LB['lam']), True)

        return 1e100, np.zeros(n, dtype=float)

    score = (2.0 * n / (2 * n - 1)) * M

    grad_u = a * (1.0 - a) * g
    grad_u -= float(np.mean(grad_u))
    grad_u = np.nan_to_num(grad_u, nan=0.0, posinf=0.0, neginf=0.0)

def fista_optimize(a, tau, lam, deadline):
    arr = prepare_vector(a)
    if arr is None or time.monotonic() > deadline:
        return

    n = arr.size
    m = 2 * n - 1
    scale = (2.0 * n) / m

    x = arr.copy()
    ypt = arr.copy()
    xprev = arr.copy()
    k = 1.0

    J_y, M_y, g_y, _ = smooth_eval_y(ypt, tau, lam, True)
    if not np.isfinite(J_y) or g_y is None:
        return

    maxL = max(1000.0, 20.0 * (2.0 * n) / max(float(tau), 0.1) + 100.0 * n)
    L = max(10.0, 0.1 * (2.0 * n) / max(float(tau), 0.1) + 2.0 * n)

    iter_count = 0
    while time.monotonic() < deadline and iter_count < 200000:
        fails = 0
        xc = x
        Jx = J_y
        Mx = M_y

        while fails < 40:
            step = 1.0 / L
            xc = project_simplex(ypt - step * g_y)
            Jx, Mx, _, _ = smooth_eval_y(xc, tau, lam, False)

            if np.isfinite(Jx):
                diff = xc - ypt
                val = J_y + float(np.dot(g_y, diff)) + 0.5 * L * float(np.dot(diff, diff))
                if Jx <= val + 1e-8:
                    break

            L = min(maxL, L * 2.0)
            fails += 1

        if not np.isfinite(Jx):
            break

        score = scale * Mx
        if np.isfinite(score) and score < _BEST['score'] - 1e-12:
            update_best(xc)

        k_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * k * k))
        y_new = xc + ((k - 1.0) / k_new) * (xc - xprev)
        xprev = x
        x = xc

        yp = prepare_vector(y_new)
        if yp is None:
            yp = project_simplex(y_new)
        ypt = yp

        J_y, M_y, g_y, _ = smooth_eval_y(ypt, tau, lam, True)
        if not np.isfinite(J_y) or g_y is None:
            break

        if fails == 0:
            L = min(maxL, L * 1.1)

        iter_count += 1


def top_bottom_indices(arr, k=30):
    n = arr.size
    if n == 0:
        return (np.array([], dtype=int), np.array([], dtype=int))
    if k <= 0:
        k = 1
    if n <= 2 * k:
        order = np.argsort(arr)[::-1]
        kk = min(k, n)
        return order[:kk].copy(), order[::-1][:kk].copy()

    top_idx = np.argpartition(arr, n - k)[-k:]
    top_idx = top_idx[np.argsort(arr[top_idx])[::-1]]

    low_idx = np.argpartition(arr, k - 1)[:k]
    low_idx = low_idx[np.argsort(arr[low_idx])]

    return top_idx, low_idx


def eval_delta_y(y, Lin, R, delta, scale):
    if delta == 0.0:
        mx = float(np.max(y)) if y.size else 0.0
    else:
        val = y + delta * Lin + delta * delta * R
        mx = float(np.max(val)) if val.size else 0.0
    if not np.isfinite(mx):
        return 1e100
    if mx < 0.0:
        mx = 0.0
    return scale * mx


def pairwise_polish(a, deadline):
    arr = prepare_vector(a)
    if arr is None:
        return

    n = arr.size
    m = 2 * n - 1
    scale = (2.0 * n) / m

    update_best(a)

    pair_count = 0
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:

        if not np.isfinite(J) or g is None:
            break

        if np.isfinite(score) and score < _BEST['score'] - 1e-12:
            update_best(a)

        lower_bound = 1.0 + 1.0 / float(2 * n)
        if score < lower_bound + 1e-5:
            break

        top, low = top_bottom_indices(g, 24)
        if top.size == 0 or low.size == 0:
            break

        improved = False

        for trial in range(12):
            pc = pair_count + trial

            p = None
            for idx in range(top.size):
                cand = int(top[(pc + idx) % top.size])
                if a[cand] > 1e-12:
                    p = cand
                    break
            if p is None:
                p = int(np.argmax(a))

            q = None
            for idx in range(low.size):
                cand = int(low[(pc + idx) % low.size])
                if cand != p:
                    q = cand
                    break
            if q is None:
                q = int(np.argmax(np.where(np.arange(n) == p, -1.0, a)))

            ap = float(a[p])
            if ap <= 1e-14:
                continue

            Aarr = np.zeros(m, dtype=float)
            Aarr[q:q + n] = a

            Barr = np.zeros(m, dtype=float)
            Barr[p:p + n] = a

            Lin = m * 2.0 * (Aarr - Barr)

            R = np.zeros(m, dtype=float)
            if 2 * p < m:
                R[2 * p] += m
            if 2 * q < m:
                R[2 * q] += m
            if p + q < m:
                R[p + q] -= 2.0 * m

            lo = 0.0
            hi = ap
            best_d = 0.0
            best_s = score

            s_hi = eval_delta_y(y, Lin, R, hi, scale)
            if s_hi < best_s:
                best_s = s_hi
                best_d = hi

            for _ in range(14):
                d1 = hi - (hi - lo) / 3.0
                d2 = lo + (hi - lo) / 3.0
                s1 = eval_delta_y(y, Lin, R, d1, scale)
                s2 = eval_delta_y(y, Lin, R, d2, scale)

                if s1 < s2:
                    hi = d2
                    if s1 < best_s:
                        best_s = s1
                        best_d = d1
                else:
                    lo = d1
                    if s2 < best_s:
                        best_s = s2
                        best_d = d2

            dm = 0.5 * (lo + hi)
            sm = eval_delta_y(y, Lin, R, dm, scale)
            if sm < best_s:
                best_s = sm
                best_d = dm

            if best_d > 1e-15 and best_s < score - 1e-9:
                a[p] = ap - best_d
                a[q] = a[q] + best_d
                a = np.clip(a, 0.0, None)
                ssum = float(a.sum())
                if ssum > 0.0:
                    a *= 1.0 / ssum
                else:
                    break

                y = compute_y(a)
                M = float(np.max(y))
                score = scale * M
                update_best(a)

                no_improve = 0
                improved = True
                tau = max(0.1, tau * 0.9)
                break

        if improved:
            pair_count += 12
        else:
            no_improve += 1
            pair_count += 12
            if no_improve % 100 == 0:
                tau = min(1.5, tau * 1.1)
            if no_improve % 2000 == 0:
                a = project_simplex(a + 0.01 * _RNG.normal(size=n))
                y = compute_y(a)
                M = float(np.max(y))
                score = scale * M
                update_best(a)
            if no_improve > 20000:
                break


def try_cvxpy(n):
    global _CVX_SMALL
    try:
        import cvxpy as cp
        import scipy.sparse as sps
    except Exception:
        return None

    try:
        m = 2 * n - 1
        a = cp.Variable(n, nonneg=True)
        T = cp.Variable()

        cons = [
            cp.sum(a) == 1.0,
            T >= 1.0,
        ]

        for k in range(m):
            rows = []
            cols = []
            vals = []

            i0 = max(0, k - (n - 1))
            i1 = min(n - 1, k)

            for i in range(i0, i1 + 1):
                j = k - i
                if i == j:
                    rows.append(i)
                    cols.append(i)
                    vals.append(1.0)
                elif i < j:
                    rows.append(i)
                    cols.append(j)
                    vals.append(0.5)
                    rows.append(j)
                    cols.append(i)
                    vals.append(0.5)

            Q = sps.coo_matrix((vals, (rows, cols)), shape=(n, n))

            try:
                cons.append(cp.quad_form(a, Q, assume_slack=False) <= T / m + 1e-9)
            except TypeError:
                cons.append(cp.quad_form(a, Q) <= T / m + 1e-9)

        prob = cp.Problem(cp.Minimize(T), cons)

        ok = False
        for name in ['CLARABEL', 'SCS', 'ECOS', 'OSQP']:
            solver = getattr(cp, name, None)
            if solver is None:
                continue
            try:
                prob.solve(solver=solver, verbose=False)
                ok = True
                break
            except Exception:
                continue

        if ok and prob.status in ('optimal', 'optimal_inaccurate', 'optimal_inaccurate_solution'):
            av = a.value
            av = prepare_vector(av)
            if av is not None:
                update_best(av)
                if n >= 100:
                    _CVX_SMALL = av.copy()
                return av
    except Exception:
        return None

    return None


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search nonnegative step-function coefficients by combining:
    1) analytic binomial initialization,
    2) optional small CVXPY convex moment/convolution warm starts,
    3) L-BFGS-B on softmax parameterization of a smooth-max + variance objective,
    4) FISTA projected gradient polishing,
    5) exact line-search pairwise coordinate swaps on the true max-convolution objective.
    """
    global _BEST, _CVX_SMALL, _RNG

    _BEST = {'a': None, 'score': np.inf, 'n': 0}
    _CVX_SMALL = None
    _FFT_CACHE.clear()
    _RNG = np.random.default_rng(seed)

    deadline = time.monotonic() + max(0.0, float(budget_s)) - 5.0

    h = globals().get('height_sequence_1', None)
    update_best(h)

    update_best(binomial_sequence(4096))
    update_best(binomial_sequence(5582))

    if time.monotonic() < deadline - 1500:
        try_cvxpy(128)
        if _CVX_SMALL is not None and time.monotonic() < deadline - 1650:
            try_cvxpy(256)

    if time.monotonic() < deadline - 1200:
        allow = min(90.0, deadline - time.monotonic() - 1200)
        if allow > 5.0:
            run_lbfgs(1024, 0.8, 1.0, allow, choose_large_init(1024))

    if time.monotonic() < deadline - 1000:
        allow = min(300.0, deadline - time.monotonic() - 1000)
        if allow > 10.0:
            run_lbfgs(4096, 0.5, 1.0, allow, choose_large_init(4096))

    if time.monotonic() < deadline - 700:
        allow = min(300.0, deadline - time.monotonic() - 700)
        if allow > 10.0:
            run_lbfgs(5582, 0.4, 1.0, allow, choose_large_init(5582))

    if _BEST.get('a') is not None and time.monotonic() < deadline - 120:
        rem = deadline - time.monotonic()
        fdeadline = time.monotonic() + min(600.0, 0.45 * rem)
        fdeadline = min(fdeadline, deadline - 80.0)
        fista_optimize(_BEST['a'].copy(), 0.5, 1.0, fdeadline)

    if _BEST.get('a') is not None and time.monotonic() < deadline - 2.0:
        pairwise_polish(_BEST['a'].copy(), deadline - 2.0)

    best = _BEST.get('a')
    if best is None:
        best = binomial_sequence(1024)
    best = prepare_vector(best)

    final_a = best
    final_score = safe_evaluate(best)

    if not np.isfinite(final_score):
        fallback = binomial_sequence(4096)
        fs = safe_evaluate(fallback)
        if np.isfinite(fs):
            final_a = fallback
            final_score = fs

    if final_a is None or not np.isfinite(final_score):
        final_a = binomial_sequence(1024)

    return [float(x) for x in final_a]
