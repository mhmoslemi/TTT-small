```python
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import fft, ifft, next_fast_len

def _conv_square(a):
    n = len(a)
    size = next_fast_len(2*n - 1)
    A = fft(a, size)
    return np.real(ifft(A*A, size)[:2*n-1])

def _softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def _lsmax_grad(theta, tau):
    """Smoothed objective 2n*logsumexp(b/tau) and gradient."""
    a = _softmax(theta)
    n = len(a)
    b = _conv_square(a)
    
    # logsumexp(b/tau)
    bm = np.max(b)
    s = (b - bm) / tau
    es = np.exp(s)
    se = np.sum(es)
    lse = bm + tau * np.log(se)
    f = 2.0 * n * lse
    
    # sw = softmax(b/tau)
    sw = es / se
    sw_dot_b = np.dot(sw, b)
    
    # c[j] = sum_m a[m] * sw[j+m]
    size = next_fast_len(3*n)
    A = fft(a, size)
    SWr = fft(sw[::-1], size)
    conv = np.real(ifft(A * SWr, size))
    c = conv[n-1:2*n-1][::-1]
    
    grad = 4.0 * n * a * (c - sw_dot_b)
    return f, grad
```
