def build_convolution_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (A@g)[k] = (f*g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            i = k - j
            if f[i] != 0:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
    A = sp.coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
    return A

def solve_lp_for_g(f, S, n):
    """Solve the LP: minimize t s.t. (f*g)[k] <= t, sum(g) = S, g >= 0."""
    A = build_convolution_matrix(f, n)
    
    # Variables: [g_0, ..., g_{n-1}, t]
    # Minimize t => c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
