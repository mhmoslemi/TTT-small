import numpy as np
import time

def compute_convolution(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fc = np.fft.irfft(fa * fa, size)
    return fc[:2*n-1]

def objective(a):
    """Compute 2n * max(conv) / sum²."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    conv = compute_convolution(a)
