```python
        # The term a[i] appears when j=i (giving a[i]*a[k-i]) and when k-j=i i.e. j=k-i (giving a[k-i]*a[i])
        # So d(conv[k])/d(a[i]) = a[k-i] + a[k-i] = 2*a[k-i] (if i != k-i, i.e., 2i != k)
        # If 2i == k: d(conv[k])/d(a[i]) = 2*a[i]
        
        grad = np.zeros(n)
        # For each i in [0, n-1], valid j = k*-i must be in [0, n-1]
        # i.e., 0 <= k*-i <= n-1, so k*-n+1 <= i <= k*
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        for i in range(i_min, i_max + 1):
            j = k_star - i
            if i == j:
                grad[i] = 2.0 * a[i]
            else:
                grad[i] = 2.0 * a[j]
        
        # dJ/da[i] = 2n/S^2 * grad[i] - 4nM/S^3
        J_grad = (2.0 * n / (S*S)) * grad - (4.0 * n * M) / (S*S*S)
        
        # Line search or fixed step
        a_new = a - eta * J_grad
        a_new = np.maximum(a_new, 0.0)
        
        J_new = compute_J(a_new)
        if J_new < J:
            a = a_new
            eta = min(eta * 1.05, 1.0)  # increase step size slightly
        else:
            eta *= 0.8  # decrease step size
            # Occasionally try a random perturbation
            if rng.random() < 0.1:
                pert = rng.normal(0, 0.01 * a.mean(), size=n)
                a = np.maximum(a + pert, 0.0)
        
        # Periodically try new starting points
        if it % 5000 == 0 and it > 0:
            # Try a different power law
            alpha = rng.uniform(0.3, 0.7)
            a_alt = np.array([(i+1)**(-alpha) for i in range(n)])
            J_alt = compute_J(a_alt)
            if J_alt < best_J:
                best = a_alt.copy()
                best_J = J_alt
                a = a_alt.copy()
                print(f"New best from random start: J={J_alt:.6f}")
    
    return [float(x) for x in best]
```

For a practical approach, I'll use the following strategy:
1. Compute the gradient with respect to the current max lag k*
2. Take a step
3. Check if the max lag has changed
4. If so, recompute the gradient

This is a "tracking" approach that works well in practice.

def fast_autocorr(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 1 << (2*n - 2).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

    n = len(a)
    conv = fast_autocorr(a)
    M = conv.max()
    S = a.sum()
    if S < 0.01:
        return np.inf, conv, 0
    return 2.0 * n * M / (S * S), conv, int(conv.argmax())
