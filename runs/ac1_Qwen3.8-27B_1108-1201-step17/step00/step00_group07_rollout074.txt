"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Strategy:
  * Work with sequences normalized to sum = sqrt(2*n), where the objective
    reduces to exactly max of the full self-convolution, so the inner loop
    can use a fast FFT-based convolution and avoid calling evaluate_sequence
    on every step.
  * Use several global initializations (the provided construction, random,
    smooth/sinusoidal) and several lengths to avoid a single local minimum.
  * For each initialization run three complementary first-order methods:
      - Adam with Euclidean projection onto the non-negative simplex
      - Mirror descent (multiplicative weights) which respects non-negativity
      - Frank-Wolfe (conditional gradient) which moves mass to the best vertex
  * Keep a global best and spend the remaining budget polishing that best
    sequence with a finer learning-rate schedule and a larger top-k subgradient.
  * Periodically verify with the authoritative evaluate_sequence and return the
    best valid sequence before the deadline.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, n, top_indices, c):
    """Weighted subgradient of max(c) using the top-k active lags.

    dc[k]/da[m] = 2*a[k-m] for 0 <= k-m < n, else 0, so the gradient at a
    single lag is a reversed (possibly truncated) copy of a.  We average
    (with weights proportional to c[k]) over the top active lags.
    """
    grad = np.zeros(n, dtype=np.float64)
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total <= 1e-15:
        return grad
    weights = top_vals / total
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)


def run_adam(a_init, n, S, max_time, rng, lr_init, top_k):
    """Adam optimizer with simplex projection; returns (best_a, best_maxc)."""
    a = safe_normalize(a_init, n)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = lr_init
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        top_indices = np.argsort(c)[::-1][:top_k]
        grad = compute_subgradient(a, n, top_indices, c)
        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-12:
            break
        grad = grad / gnorm

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)

        lr = lr_init * (0.5 ** (float(t) / 400.0))
        if t % 2500 == 0 and rng.random() < 0.1:
            a = project_simplex(a + rng.normal(0.0, 0.01 * S / n, n), S)

    return best_local, best_score


def run_mirror(a_init, n, S, max_time, rng, eta_init, top_k):
    """Mirror descent / multiplicative weights; respects x >= 0 exactly."""
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    eta = eta_init
    start = time.monotonic()
    t = 0

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        top_indices = np.argsort(c)[::-1][:top_k]
        grad = compute_subgradient(a, n, top_indices, c)
        gmax = np.abs(grad).max()
        if gmax <= 1e-12:
            break
        grad = grad / gmax

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 1e-15:
            a = np.full(n, S / n, dtype=np.float64)
        else:
            a = a_new * (S / s)

        eta = eta_init * max(0.01, 1.0 / (1.0 + 0.004 * t))

    return best_local, best_score


def run_frank_wolfe(a_init, n, S, max_time, top_k):
    """Frank-Wolfe (conditional gradient) on the non-negative simplex."""
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    t = 0

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        top_indices = np.argsort(c)[::-1][:top_k]
        grad = compute_subgradient(a, n, top_indices, c)
        m_star = int(np.argmin(grad))
        gamma = 2.0 / (t + 2.0)
        a_dir = np.zeros(n, dtype=np.float64)
        a_dir[m_star] = S
        a = (1.0 - gamma) * a + gamma * a_dir

    return best_local, best_score

def polish(best_a, budget_left, rng):
    """Spend remaining time refining the globally best sequence."""
    if best_a is None or budget_left < 3.0:
        return best_a
    a = safe_normalize(best_a, len(best_a))
    n = len(a)
    S = np.sqrt(2.0 * float(n))
    mid = budget_left / 2.0
    if mid > 1.0:
        a, s1 = run_adam(a, n, S, mid, rng, lr_init=0.5 * S / n, top_k=8)
        print("polish adam  maxc=%.6f" % s1)
    if time.monotonic() > 0 and budget_left - mid + 1.0 > 1.0:
        a2, s2 = run_mirror(a, n, S, max(0.0, budget_left - mid - 1.0),
                            rng, eta_init=0.3, top_k=8)
        print("polish mirror maxc=%.6f" % s2)
        if s2 < s1:
            a = a2
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 8.0)

    inits = collect_initializations(rng)
    usable = max(1.0, (deadline - time.monotonic()))
    explore_budget = 0.72 * usable
    polish_budget = usable - explore_budget

    global_best = np.inf
    global_a = None

    n_inits = len(inits)
    per_init = explore_budget / max(1, n_inits)

    for name, arr in inits:
        if time.monotonic() > deadline - polish_budget - 2.0:
            break
        n = len(arr)
        if n < 16:
            continue
        S = np.sqrt(2.0 * float(n))
        t0 = time.monotonic()
        left = per_init - (t0 - t0)
        seg = max(1.0, left / 3.0)

        a_adam, s_adam = run_adam(arr, n, S, seg, rng,
                                  lr_init=0.5 * S / n, top_k=5)
        if s_adam < global_best:
            global_best, global_a = s_adam, a_adam.copy()

        if time.monotonic() < deadline - polish_budget:
            a_mir, s_mir = run_mirror(a_adam, n, S, seg, rng,
                                      eta_init=0.5, top_k=5)
            if s_mir < global_best:
                global_best, global_a = s_mir, a_mir.copy()

        if time.monotonic() < deadline - polish_budget:
            a_fw, s_fw = run_frank_wolfe(a_adam, n, S, seg, top_k=5)
            if s_fw < global_best:
                global_best, global_a = s_fw, a_fw.copy()

        # Authoritative check of the best we have for this length.
        ver = evaluate_sequence(a_adam.tolist())
        print("%s  adam=%.6f mirror? fw?  verify_adam=%s  global=%.6f"
              % (name, s_adam, "%.6f" % ver if np.isfinite(ver) else "inf",
                 global_best))

    # Polish the globally best sequence with the remaining budget.
    left = deadline - time.monotonic()
    if left > 2.0 and global_a is not None:
        global_a = polish(global_a, left, rng)
        ver = evaluate_sequence(np.clip(global_a, 0.0, 1000.0).tolist())
        print("after polish verify=%.6f" % ver if np.isfinite(ver)
              else "after polish verify=inf")
        global_best = ver if np.isfinite(ver) else np.inf

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    # Avoid the sum < 0.01 guard.
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)
    return [float(x) for x in global_a]
