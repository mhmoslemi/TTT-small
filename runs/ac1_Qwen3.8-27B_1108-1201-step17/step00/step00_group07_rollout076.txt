def _next_pow2(x):
    return 1 << (int(x) - 1).bit_length()

def _fft_conv(a, b):
    n = len(a) + len(b) - 1
    N = _next_pow2(n)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(A * B, N)[:n]

def _compute_c(a):
    return _fft_conv(a, a)

def _compute_grad(w, a, n):
    # grad[j] = 2 * sum_m a[m] * w[m+j] for j=0,...,n-1
    # = 2 * _fft_conv(w, a[::-1])[n-1:2*n-1]

    W = np.fft.rfft(w, N)
    Ar = np.fft.rfft(a[::-1], N)
    conv = np.fft.irfft(W * Ar, N)
    return 2.0 * conv[n - 1:2 * n - 1]

def _lse_grad(c, a, n, tau):
    cmax = np.max(c)
    s = tau * (c - cmax)
    s -= np.max(s)
    es = np.exp(s)
    tot = np.sum(es)
    w = es / tot
    se = cmax + np.log(tot) / tau
    grad = _compute_grad(w, a, n)
    return se, grad

    a_new = np.maximum(a_new, 0.0)

    return a_new / s
