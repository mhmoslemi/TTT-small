        # Shift c for numerical stability
        c_shifted = c - c.max()
        exp_c = np.exp(beta * c_shifted)
        p = exp_c / exp_c.sum()
        M_beta = c.max() + np.log(exp_c.sum()) / beta
        
        # Compute correlation

        grad = 4.0 * n / (S**2) * (corr - M_beta / S)

        # Take step
        candidate = best - lr * grad
        candidate = np.clip(candidate, 0, 1000)
        
        # Evaluate
        score = evaluate_sequence(candidate.tolist())
        if score < best_score:
            best = candidate
            best_score = score
