```python
import numpy as np
import time
from scipy.optimize import minimize
from operator import itemgetter

USE_FFT_CORR = False


def _next_pow2(m):
    """Smallest power of two >= m."""
    m = int(max(1, m))
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def corr_fft_valid(w, a):
    """Valid-mode cross-correlation: c[i] = sum_j w[i+j] a[j].

    Assumes len(w) >= len(a).
    """
    w = np.asarray(w, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    m = w.size
    n = a.size
    if n == 0 or m == 0:
        return np.zeros(0, dtype=np.float64)
    if m < n:
        # Not expected in this task, but fall back safely.
        return np.correlate(a, w, mode="valid")[::-1]
    N = _next_pow2(m + n - 1)
    fa = np.fft.fft(a, N)
    fw = np.fft.fft(w, N)
    cc = np.fft.ifft(np.conj(fa) * fw)
    return np.ascontiguousarray(cc.real[:n])


def correlate_valid(w, a):
    """Valid-mode cross-correlation c[i] = sum_j w[i+j] a[j]."""
    if USE_FFT_CORR:
        return corr_fft_valid(w, a)
    return np.ascontiguousarray(np.correlate(w, a, mode="valid"))


def choose_corr_method(n):
    """Pick the faster correlate implementation for this length."""
    if n < 128:
        return False
    rng = np.random.default_rng(7)
    w = rng.random(2 * n - 1)
    a = rng.random(n)

    t0 = time.monotonic()
    for _ in range(3):
        np.correlate(w, a, mode="valid")
    td = time.monotonic() - t0

    if n > 409:
        try:
            t0 = time.monotonic()
            for _ in range(3):
                corr_fft_valid(w, a)
            tf = time.monotonic() - t0
            if tf < td * 0.90:
                return True
        except Exception:
            pass
    return False


def conv_self(a):
    """Linear convolution of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    need = 2 * n - 1
    L = _next_pow2(need)
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)
    return np.ascontiguousarray(c[:need])


def stable_softmax(x):
    """Stable softmax for arbitrary finite arrays."""
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    if n == 0:
        return None
    if not np.isfinite(x).all():
        x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
    mx = float(np.max(x))
    mn = float(np.min(x))
    off = 0.5 * (mx + mn)
    y = x - off
    ym = float(np.max(y))
    if not np.isfinite(ym):
        return np.ones(n, dtype=np.float64) / float(n)
    y = y - ym
    p = np.exp(y)
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    return p / s


def normalize_to_sum(a, S):
    """Clip to [0, 1000] and rescale so that sum(a) == S."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-12:
        return np.full(a.shape, S / max(1, a.size), dtype=np.float64)
    return a * (S / s)


def score_a(a, n):
    """Score a nonnegative array of length n using the objective formula."""
    a = np.asarray(a, dtype=np.float64)
    if a.size != n:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = conv_self(a)
    if not np.isfinite(c).all():
        return np.inf
    m = float(np.max(c))
    return float((2.0 * n * m) / (s * s))


def smooth_obj_full(x, n, S, tau):
    """Smooth-max objective on the full simplex-x parameterization."""
    x = np.asarray(x, dtype=np.float64)
    if x.size != n or not np.isfinite(x).all():
        return np.inf
    p = stable_softmax(x)
    a = S * p
    c = conv_self(a)
    if not np.isfinite(c).all():
        return np.inf
    cs = c / float(tau)
    z = float(np.max(cs))
    e = np.exp(cs - z)
    es = float(np.sum(e))
    if not np.isfinite(es) or es <= 0.0:
        return np.inf
    return float(tau * (z + np.log(es)))


def smooth_grad_full(x, n, S, tau):
    """Gradient of smooth_obj_full w.r.t. x, where a = S softmax(x)."""
    x = np.asarray(x, dtype=np.float64)
    if x.size != n:
        return np.zeros(x.size, dtype=np.float64)
    if not np.isfinite(x).all():
        return np.zeros(n, dtype=np.float64)

    p = stable_softmax(x)
    a = S * p
    c = conv_self(a)
    if not np.isfinite(c).all():
        return np.zeros(n, dtype=np.float64)

    cs = c / float(tau)
    z = float(np.max(cs))
    e = np.exp(cs - z)
    es = float(np.sum(e))
    if not np.isfilees if not np.isfinite(es) or es <= 0.0:
        return np.zeros(n, dtype=np.float64)

    w = e / es
    ga = 2.0 * correlate_valid(w, a)
    if not np.isfinite(ga).all():
        return np.zeros(n, dtype=np.float64)

    G = float(np.dot(a, ga))
    gx = a * ga - a * (G / S)
    gx = gx - float(np.mean(gx))
    if not np.isfinite(gx).all():
        gx = np.zeros(n, dtype=np.float64)
    return gx


def low_obj(theta, n, S, tau, B):
    """Smooth-max objective for low-dimensional basis parameters theta."""
    theta = np.asarray(theta, dtype=np.float64)
    if theta.size != B.shape[1] or not np.isfinite(theta).all():
        return np.inf
    x = theta @ B.T
    return smooth_obj_full(x, n, S, tau)


def low_grad(theta, n, S, tau, B):
    """Gradient of low_obj w.r.t. theta."""
    theta = np.asarray(theta, dtype=np.float64)
    d = theta.size
    if theta.size != B.shape[1]:
        return np.zeros(theta.size, dtype=np.float64)
    if not np.isfinite(theta).all():
        return np.zeros(d, dtype=np.float64)

    x = theta @ B.T
    gx = smooth_grad_full(x, n, S, tau)
    gtheta = B.T @ gx
    if not np.isfinite(gtheta).all():
        gtheta = np.zeros(d, dtype=np.float64)
    return gtheta


def make_basis(n, max_freq=5):
    """A compact real basis of smooth profiles on [0, 1)."""
    n = int(n)
    i = np.arange(n, dtype=np.float64)
    if n > 1:
        t = 2.0 * i / float(n - 1) - 1.0
    else:
        t = np.zeros(1, dtype=np.float64)

    cols = [
        t,
        t * t,
        t * t * t,
        t * t * t * t,
    ]

    ang = 2.0 * np.pi * i / float(max(1, n))
    for f in range(1, max_freq + 1):
        cols.append(np.cos(f * ang))
        cols.append(np.sin(f * ang))

    B = np.column_stack(cols).astype(np.float64, copy=False)
    for j in range(B.shape[1]):
        col = B[:, j]
        col = col - float(np.mean(col))
        sd = float(np.std(col))
        if sd > 1e-12:
            B[:, j] = col / sd
        else:
            B[:, j] = 0.0
    return B


def x_to_a(x, n, S):
    """Convert softmax parameters x to the normalized nonnegative array."""
    p = stable_softmax(np.asarray(x, dtype=np.float64))
    a = S * p
    return np.clip(a, 0.0, 1000.0)


def score_from_x(x, n, S):
    """Score a softmax-parameter candidate."""
    x = np.asarray(x, dtype=np.float64)
    if x.size != n:
        return np.inf
    a = x_to_a(x, n, S)
    return score_a(a, n)


def add_candidate_from_a(a, x, top, limit, n, S):
    """Score, normalize, and keep only the best candidates."""
    a = normalize_to_sum(a, S)
    sc = score_a(a, n)
    if not np.isfinite(sc):
        return
    if x is None:
        x = np.log(np.maximum(a, 1e-12))
    x = np.asarray(x, dtype=np.float64)
    if x.size != n or not np.isfinite(x).all():
        x = np.zeros(n, dtype=np.float64)
    x = x - float(np.mean(x))
    top.append((float(sc), np.ascontiguousarray(x)))
    top.sort(key=itemgetter(0))
    if len(top) > limit:
        del top[limit:]


def add_candidate_from_x(x, top, limit, n, S):
    """Score and add a softmax-parameter candidate."""
    x = np.asarray(x, dtype=np.float64)
    if x.size != n or not np.isfinite(x).all():
        x = np.zeros(n, dtype=np.float64)
    x = x - float(np.mean(x))
    a = x_to_a(x, n, S)
    sc = score_a(a, n)
    if not np.isfinite(sc):
        return
    top.append((float(sc), np.ascontiguousarray(x)))
    top.sort(key=itemgetter(0))
    if len(top) > limit:
        del top[limit:]


def structured_candidates(top, limit, n, S, rng):
    """Add a broad set of inexpensive analytic/random candidates."""
    if n <= 0:
        return

    add_candidate_from_x(np.zeros(n), top, limit, n, S)

    i = np.arange(n, dtype=np.float64)
    if n > 1:
        t = 2.0 * i / float(n - 1) - 1.0
    else:
        t = np.zeros(n, dtype=np.float64)
    ang = 2.0 * np.pi * i / float(max(1, n))

    # Monotone/exponential one-sided profiles.
    for beta in (0.2, 0.5, 1.0, 2.0, 3.0, 5.0, 8.0, 12.0):
        add_candidate_from_x(beta * t, top, limit, n, S)
        add_candidate_from_x(-beta * t, top, limit, n, S)

    # Symmetric U- and cap-shaped profiles.
    for beta in (0.3, 0.7, 1.5, 3.0, 6.0):
        add_candidate_from_x(beta * (t * t), top, limit, n, S)
        add_candidate_from_x(-beta * (t * t), top, limit, n, S)

    # Simple asymmetric polynomial combinations.
    for b1 in (0.3, 0.8, 1.5):
        for b2 in (0.3, 0.8, 1.5):
            add_candidate_from_x(b1 * t + b2 * t * t, top, limit, n, S)
            add_candidate_from_x(b1 * t - b2 * t * t, top, limit, n, S)
            add_candidate_from_x(-b1 * t + b2 * t * t, top, limit, n, S)
            add_candidate_from_x(-b1 * t - b2 * t * t, top, limit, n, S)

    # Random finite-Fourier profiles.
    for _ in range(45):
        nf = int(rng.integers(1, 5))
        x = np.zeros(n, dtype=np.float64)
        for f in range(1, nf + 1):
            amp = float(rng.uniform(0.05, 1.6))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            x += amp * np.cos(f * ang + ph)
            if rng.random() < 0.75:
                x += amp * np.sin(f * ang + ph)
        add_candidate_from_x(x, top, limit, n, S)

    # Low-frequency smooth random profiles.
    for _ in range(25):
        nf = int(rng.integers(2, 8))
        freqs = np.arange(1, nf + 1, dtype=np.float64)
        amps = 1.0 / (0.5 + freqs ** 1.4)
        amps *= float(rng.uniform(0.5, 2.0))
        ph = rng.uniform(0.0, 2.0 * np.pi, nf)
        x = np.zeros(n, dtype=np.float64)
        for k in range(nf):
            x += amps[k] * np.cos(freqs[k] * ang + ph[k])
            if rng.random() < 0.5:
                x += amps[k] * np.sin(freqs[k] * ang + ph[k])
        add_candidate_from_x(x, top, limit, n, S)

    # High-dimensional random lognormal starting points.
    for sigma in (0.5, 1.0, 2.0, 3.0, 4.0):
        x = rng.normal(0.0, sigma, n)
        add_candidate_from_x(x, top, limit, n, S)


def run_low_phase(top, limit, n, S, rng, low_deadline):
    """Run many low-dimensional L-BFGS restarts to find promising shapes."""
    if n <= 0:
        return
    if n >= 2500:
        max_freq = 5
    elif n >= 256:
        max_freq = 4
    else:
        max_freq = 2

    B = make_basis(n, max_freq)
    d = B.shape[1]
    tau = 0.12
    opts_base = {"ftol": 1e-12, "gtol": 1e-8, "maxls": 40, "disp": False}

    it = 0
    while time.monotonic() < low_deadline:
        # A deterministic cycle of start types, with randomness inside.
        if it % 20 == 0:
            theta = np.zeros(d, dtype=np.float64)
        elif it % 5 == 0:
            theta = np.zeros(d, dtype=np.float64)
            j = int(rng.integers(0, d))
            theta[j] = float(rng.choice((-2.0, -1.0, 1.0, 2.0)))
        else:
            sigma = float(rng.choice((0.5, 1.0, 2.0, 3.0)))
            theta = rng.normal(0.0, sigma, d)

        # Time one gradient call to budget iterations safely.
        t0 = time.monotonic()
        g = low_grad(theta, n, S, tau, B)
        dtg = time.monotonic() - t0
        if not np.isfinite(g).all():
            it += 1
            continue

        remaining = max(0.0, low_deadline - time.monotonic())
        stage_budget = min(5.0, max(1.0, remaining / 6.0))
        if stage_budget < 0.3:
            break

        maxfun = int(min(260, max(35, stage_budget / max(dtg * 1.35, 1e-4))))
        maxiter = int(min(110, max(15, maxfun // 2)))
        opts = dict(opts_base)
        opts["maxfun"] = maxfun
        opts["maxiter"] = maxiter

        try:
            res = minimize(
                low_obj,
                theta,
                args=(n, S, tau, B),
                method="L-BFGS-B",
                jac=low_grad,
                options=opts,
            )
        except Exception:
            res = None

        if res is not None and res.x is not None:
            x = np.asarray(res.x, dtype=np.float64)
            if x.size == d and np.isfinite(x).all():
                x = x - float(np.mean(x))
                add_candidate_from_x(x, top, limit, n, S)

        it += 1
        if it % 12 == 0:
            top.sort(key=itemgetter(0))
            best = top[0][0] if top else np.inf
            print(f"low-phase it={it}, best={best:.6f}")


def polish_x(x0, n, S, branch_deadline, rng):
    """Polish a full-dimensional candidate by annealing the smooth-max tau."""
    x = np.asarray(x0, dtype=np.float64)
    if x.size != n or not np.isfinite(x).all():
        x = np.zeros(n, dtype=np.float64)
    x = x - float(np.mean(x))

    best_x = x.copy()
    best_score = score_from_x(x, n, S)
    if not np.isfinite(best_score):
        best_x = np.zeros(n, dtype=np.float64)
        best_score = score_from_x(best_x, n, S)

    tau_steps = (0.50, 0.25, 0.12, 0.06, 0.025, 0.010)
    m = len(tau_steps)
    opts_base = {"ftol": 1e-12, "gtol": 1e-8, "maxls": 40, "disp": False}

    for idx, tau in enumerate(tau_steps):
        now = time.monotonic()
        if now > branch_deadline - 2.0:
            break

        remaining = max(0.0, branch_deadline - now)
        rem = max(1, m - idx)
        stage_budget = min(120.0, max(2.0, remaining / float(rem)))

        t0 = time.monotonic()
        g = smooth_grad_full(x, n, S, tau)
        dtg = time.monotonic() - t0
        if not np.isfinite(g).all():
            break

        maxfun = int(min(900, max(45, stage_budget / max(dtg * 1.45, 1e-4))))
        maxiter = int(min(420, max(20, maxfun // 2)))
        opts = dict(opts_base)
        opts["maxfun"] = maxfun
        opts["maxiter"] = maxiter

        try:
            res = minimize(
                smooth_obj_full,
                x,
                args=(n, S, tau),
                method="L-BFGS-B",
                jac=smooth_grad_full,
                options=opts,
            )
        except Exception:
            res = None

        cand_x = x
        cand_score = np.inf
        if res is not None and res.x is not None:
            cand = np.asarray(res.x, dtype=np.float64)
            if cand.size == n and np.isfinite(cand).all():
                cand_x = cand - float(np.mean(cand))
                cand_score = score_from_x(cand_x, n, S)

        if np.isfinite(cand_score) and cand_score < best_score:
            best_score = cand_score
            best_x = cand_x

        # Continue from the candidate if it is not clearly worse; otherwise
        # revert to the best actual-score point.
        if np.isfinite(cand_score):
            if cand_score <= best_score * 1.08:
                x = cand_x
            else:
                x = best_x

        # One modest perturbation in the middle of annealing can help escape
        # very flat local minima.
        if idx == max(0, m // 2) and time.monotonic() < branch_deadline - 6.0:
            xp = x + rng.normal(0.0, 0.35, n)
            sp = score_from_x(xp, n, S)
            if np.isfinite(sp) and sp < best_score:
                best_score = sp
                best_x = xp
                x = xp

    return best_x, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a nonnegative sequence minimizing the sequence evaluator.

    Algorithm:
    - Normalize all candidates to sum S = sqrt(2n), where the objective equals
      max(convolve(a, a)).
    - Generate cheap analytic and random candidates.
    - Run many low-dimensional L-BFGS restarts using a smooth basis to find a
      strong global shape.
    - Polish the best full-dimensional candidates with tau-annealed smooth-max
      L-BFGS in softmax-parameterized simplex space.
    - Return the best actual score found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)

    # Prefer the supplied construction length when safe.
    n = 3809
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size >= 16 and np.isfinite(arr).all():
            if arr.size <= 20000:
                n = int(arr.size)
                parent = arr
    except Exception:
        pass

    if n < 16:
        n = 3809
    S = float(np.sqrt(2.0 * n))

    global USE_FFT_CORR
    USE_FFT_CORR = choose_corr_method(n)

    top = []
    limit = 32

    if parent is not None:
        add_candidate_from_a(parent, None, top, limit, n, S)
    add_candidate_from_a(np.ones(n, dtype=np.float64), None, top, limit, n, S)
    add_candidate_from_x(np.zeros(n, dtype=np.float64), top, limit, n, S)

    structured_candidates(top, limit, n, S, rng)
    top.sort(key=itemgetter(0))
    if len(top) > limit:
        del top[limit:]

    best_after_struct = top[0][0] if top else np.inf
    print(f"setup n={n}, corr_fft={USE_FFT_CORR}, best_structured={best_after_struct:.6f}")

    # Low-dimensional global exploration.  Keep this well below total budget.
    low_budget = max(45.0, 0.17 * float(budget_s))
    low_deadline = min(deadline, start + low_budget)
    if time.monotonic() < low_deadline:
        run_low_phase(top, limit, n, S, rng, low_deadline)
        top.sort(key=itemgetter(0))
        if len(top) > limit:
            del top[limit:]
    best_after_low = top[0][0] if top else np.inf
    print(f"after low-phase best={best_after_low:.6f}, elapsed={time.monotonic()-start:.1f}s")

    # Full-dimensional polishing of the most promising candidates.
    k = min(8, len(top))
    for idx in range(k):
        now = time.monotonic()
        if now > deadline - 25.0:
            break

        rem_candidates = k - idx
        total_rem = max(0.0, deadline - now)
        share = total_rem / float(max(1, rem_candidates))
        branch_budget = max(15.0, min(360.0, share * 1.25))
        branch_deadline = min(deadline, now + branch_budget)

        x0 = top[idx][1].copy()
        x1, s1 = polish_x(x0, n, S, branch_deadline, rng)
        add_candidate_from_x(x1, top, limit, n, S)
        top.sort(key=itemgetter(0))
        if len(top) > limit:
            del top[limit:]

        print(
            f"polish candidate {idx + 1}/{k}, "
            f"score={s1:.6f}, global_best={top[0][0]:.6f}, "
            f"elapsed={time.monotonic()-start:.1f}s"
        )

        if time.monotonic() > deadline - 30.0:
            break

    # Final deep polish of the current best if enough time remains.
    if top and time.monotonic() < deadline - 25.0:
        x_best = top[0][1].copy()
        x_fin, s_fin = polish_x(x_best, n, S, deadline, rng)
        add_candidate_from_x(x_fin, top, limit, n, S)
        top.sort(key=itemgetter(0))

    if top:
        best_x = top[0][1]
    else:
        best_x = np.zeros(n, dtype=np.float64)

    best_a = normalize_to_sum(x_to_a(best_x, n, S), S)

    final_internal = score_a(best_a, n)
    try:
        final_eval = evaluate_sequence([float(v) for v in best_a])
    except Exception:
        final_eval = np.inf

    if not np.isfinite(final_eval):
        # Safe fallback: normalized uniform sequence.
        best_a = np.full(n, S / float(n), dtype=np.float64)
        final_internal = score_a(best_a, n)
        try:
            final_eval = evaluate_sequence([float(v) for v in best_a])
        except Exception:
            final_eval = np.inf

    print(f"final n={n}, internal={final_internal:.6f}, evaluate={final_eval:.6f}")
    return [float(v) for v in best_a]
```
