import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _build_convolution_matrix(a):
    """Build the matrix A where A[k,j] = a[j] if 0 <= k-j < len(a), else 0.
    This is the matrix such that A @ g = convolve(a, g)."""
    n = len(a)
    # Column j has a[j] in rows j through j+n-1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        if a[j] > 0:
            rows.extend(range(j, j+n))
            cols.extend([j]*n)
            vals.extend([a[j]]*n)
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def _lp_step(a, M):
    """Solve the LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns normalized g (sum=1) or None if LP fails."""
    n = len(a)
    A = _build_convolution_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)]*n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 10})
    
    if result.success:
        g = np.maximum(result.x, 0)
        s = g.sum()
        if s > 1e-10:
            return g / s
    return None

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses projected gradient descent, LP-based iterative refinement (complementary sequence
construction), and simulated annealing with periodic LP restarts.
"""

import numpy as np

def _build_conv_matrix(a):
    """Build sparse matrix A where A[k,j] = a[j] if 0<=k-j<len(a), else 0.
    So that A @ g = convolve(a, g)."""
    n = len(a)
    rows_list = []
    cols_list = []
    vals_list = []
    for j in range(n):
        v = a[j]
        if v > 0:
            rows_list.append(np.arange(j, j + n))
            cols_list.append(np.full(n, j))
            vals_list.append(np.full(n, v))
    if not rows_list:
        return csr_matrix((2*n-1, n))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def _lp_refine_step(a, M, max_time=5.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns normalized g (sum=1) or None."""
    n = len(a)
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'time_limit': max_time})
        if result.success:
            g = np.maximum(result.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None
