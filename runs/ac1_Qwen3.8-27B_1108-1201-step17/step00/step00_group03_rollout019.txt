from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

"""
LP-based iterative optimization to minimize 2n*||f*f||_inf / (sum f)^2.
Uses a linear program at each step to find the optimal direction,
combined with line search and multiple initializations.
"""

def build_A_indices(n):
    """Precompute row/col indices for the (2n-1) x n LP constraint matrix."""
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    idx = 0
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        count = j_end - j_start + 1
        rows[idx:idx + count] = k
        cols[idx:idx + count] = np.arange(j_start, j_end + 1)
        idx += count
    return rows, cols

def make_A(f, rows, cols):
    """Build sparse constraint matrix with current f values."""
    n = len(f)
    data = np.empty(len(rows), dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1)
        count = len(js)
        data[idx:idx + count] = f[k - js]
        idx += count
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

def eval_score(f):
    """Compute 2n * max(f*f) / (sum f)^2."""
    n = len(f)
    if n == 0:
        return np.inf
    conv = np.convolve(f, f)
    M = np.max(conv)
    S = np.sum(f)
    if S < 0.01 or not np.isfinite(S) or not np.isfinite(M):
        return np.inf
    return 2.0 * n * M / (S * S)

def solve_lp(f, rows, cols):
    """Solve: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    conv = np.convolve(f, f)
    M = np.max(conv)
    if M <= 0 or not np.isfinite(M):
        return None

                return np.clip(g, 0.0, None)
