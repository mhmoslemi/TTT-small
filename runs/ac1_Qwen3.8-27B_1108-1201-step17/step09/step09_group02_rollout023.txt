import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def golden_section_search(f, lo, hi, max_iter=30):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def coord_descent_fast(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_convolve(a).copy()
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        no_improve_streak = 0

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            v = a[i]
            if v < 1e-15:
                continue

            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            partners = np.arange(k_lo, k_hi + 1) - i
            a_p = a[partners]

            # Unaffected max
            unaff_max = 0.0
            if k_lo > 0:
                um = float(conv[:k_lo].max())
                if um > unaff_max:
                    unaff_max = um
            if k_hi < 2 * n - 2:
                um = float(conv[k_hi + 1:].max())
                if um > unaff_max:
                    unaff_max = um

            # Affected constants
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            c_aff = conv[k_lo:k_hi + 1].copy()
            # c_aff[k] - 2*v*a[partner] for k != 2i

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def eval_score(x, c_base, a_p, unaff_max, has_quad, qi, cur_s, v, n):
    """Score function for golden section search (top-level, no closure)."""
    ns = cur_s + (x - v)
    if ns < 0.01:
        return 1e10
    nv = c_base + 2.0 * x * a_p
    if has_quad:
        nv[qi] = c_base[qi] + x * x
    nm = float(nv.max())
    if unaff_max > nm:
        nm = unaff_max
    return 2.0 * n * nm / (ns * ns)

def golden_section_coord(c_base, a_p, unaff_max, has_quad, qi, cur_s, v, n, max_iter=28):
    """Golden section search for optimal value of coordinate."""
    hi = max(3.0 * v, 5.0 / n)
    lo = 0.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = eval_score(c, c_base, a_p, unaff_max, has_quad, qi, cur_s, v, n)
    fd = eval_score(d, c_base, a_p, unaff_max, has_quad, qi, cur_s, v, n)
    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = eval_score(c, c_base, a_p, unaff_max, has_quad, qi, cur_s, v, n)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = eval_score(d, c_base, a_p, unaff_max, has_quad, qi, cur_s, v, n)
    return (lo + hi) * 0.5

def coord_descent(a0, time_limit, order="sequential", verbose=False, label=""):

    Coordinate descent with golden section search.
    order: "sequential" or "adaptive" (by peak contribution)
    """

    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_convolve(a).copy()
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Determine order
        if order == "adaptive" and sweeps > 1:
            # Sort by contribution to top peaks
            top_k = min(10, 2 * n - 1)
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            importance = np.zeros(n)
            for k in peaks:
                k = int(k)
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                j = np.arange(i_start, i_end + 1)
                importance[j] += a[j] * a[k - j]
            order_indices = np.argsort(-importance)
        else:
            order_indices = np.arange(n)

        for i in order_indices:
            if time.monotonic() - t0 > time_limit:
                break
            v = a[i]
            if v < 1e-15:
                continue

            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            a_p = a[partners]

            unaff_max = 0.0
            if k_lo > 0:
                um = float(conv[:k_lo].max())
                if um > unaff_max:
                    unaff_max = um
            if k_hi < 2 * n - 2:
                um = float(conv[k_hi + 1:].max())
                if um > unaff_max:
                    unaff_max = um

            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            c_aff = conv[k_lo:k_hi + 1]
            c_base = c_aff - 2.0 * v * a_p
            if has_quad:
                qi = 2 * i - k_lo
                c_base = c_base.copy()
                c_base[qi] = c_aff[qi] - v * v
            else:
                qi = -1

            best_x = golden_section_coord(c_base, a_p, unaff_max, has_quad, qi,
                                          s, v, n)

            if abs(best_x - v) > 1e-13:
                delta = best_x - v

                    changes[qi] = delta * (2.0 * v + delta)

def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_chirp(n, rng):
    """Chirp-based initialization (quadratic phase)."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    # Quadratic phase: exp(i * alpha * k^2)
    alpha = rng.uniform(0.1, 0.5) * 2.0 * np.pi
    k = np.arange(n_freq)
    phase = alpha * k * k / n_freq
    amps = np.ones(n_freq)
    # Add slight randomization
    amps *= rng.uniform(0.9, 1.1, n_freq)
    phase += rng.uniform(-0.1, 0.1, n_freq)
    spectrum = amps * np.exp(1j * phase)

    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)

    if s < 1e-10:

def init_cosine(n, rng):

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def get_lp_solver():
    import cvxpy as cp
    for solver_name in [cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver):
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def basin_hopping(a0, time_limit, rng, cd_budget=60, n_perturbs=10, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break

        # Different perturbation strategies
        strategy = p % 3
        if strategy == 0:
            # Gaussian noise
            sigma = 0.02 + 0.04 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif strategy == 1:
            # Peak-breaking: reduce coordinates near peaks
            conv_p = fast_convolve(a)
            top_k = min(5, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            perturbed = a.copy()

                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)

                n_pick = max(2, len(idx) // 4)
                picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                perturbed[picks] *= (1.0 + rng.standard_normal(len(picks)) * 0.08)
            perturbed += rng.standard_normal(n) * 0.01
        else:

            perturbed = a.copy()

                i2 = rng.integers(0, n)
                if i1 != i2:
                    perturbed[[i1, i2]] = perturbed[[i2, i1]]
            perturbed += rng.standard_normal(n) * 0.005

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        t_alloc = min(rem * 0.7, cd_budget)
        a_opt, s_opt, _ = coord_descent(perturbed, t_alloc, order="sequential")
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p} (strat={strategy}): {best_score:.8f}")

    return best_a, best_score

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

    # Fallback
    if best_seq is None:

    if np.isfinite(fs):

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
