```python
    lo = 0.0
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-length exploration with FFT-based evaluation
2. Nesterov-accelerated LP alternating for global optimization
3. L-BFGS with multi-stage log-sum-exp annealing for local refinement
4. Coordinate descent with gradient-based 1D steps
5. Adaptive perturbation and restart strategies

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute the objective score."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def norm_seq(a):
    """Normalize to sum 1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_response(a_fixed, solver=None):
    """Given fixed a, find b minimizing max_k conv(a,b)[k] subject to b>=0, sum(b)=1."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def nesterov_lp(a0, time_limit, solver=None, max_iters=50000):
    """Nesterov dual averaging LP refinement."""
    n = len(a0)
    a = norm_seq(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new = solve_lp_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        sc = score_of(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            stall = 0
        if it % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 8:
                rng = np.random.default_rng(999 + it)
                y = y + 0.05 * rng.random(n)
                y = norm_seq(y)
                stall = 0
    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=10, maxiter_per_stage=2000):
    """L-BFGS with log-sum-exp smoothing."""
    n = len(a0)
    a = norm_seq(a0)
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()

    def obj_lse(z_in, T_val):
        a_in = softmax_from_z(z_in)
        b = fast_convolve(a_in)
        Tb = T_val * b
        m = Tb.max()
        return (2.0 * n / T_val) * (m + np.log(np.exp(Tb - m).sum()))

    def grad_lse(z_in, T_val):
        a_in = softmax_from_z(z_in)
        b = fast_convolve(a_in)
        Tb = T_val * b
        m = Tb.max()
        w = np.exp(Tb - m)
        w = w / w.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a_in[k - idx]
        S = np.dot(grad_a, a_in)
        return a_in * (grad_a - S)

    temps = np.geomspace(T_start, T_end, n_stages)
    for T_val in temps:
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 5:
            break
        try:
            res = minimize(
                fun=lambda z_in: obj_lse(z_in, T_val),
                x0=z,
                jac=lambda z_in: grad_lse(z_in, T_val),
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxfun': maxiter_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)


def softmax_from_z(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def coordinate_descent(a0, time_limit, sweeps=100):
    """Coordinate descent with gradient-based 1D optimization."""
    n = len(a0)
    a = norm_seq(a0).copy()
```

```python
    # Using: cross_corr = ifft(conj(FFT(A)) * FFT(W))
    # where FFT is of size >= n + 2n-1 - 1 = 3n-2 to avoid circular aliasing
    # Actually, for cross-correlation of length-n and length-(2n-1) signals,
    # we need FFT size >= n + (2n-1) - 1 = 3n-2
    
    # Hmm, this is getting complicated. Let me just use a simpler approach:
    # cross_corr[i] = sum_j a[j] * w[i+j]
    # For i from 0 to n-1, this is a sum of n terms (or fewer near the boundaries)
    # We can compute this as: cross_corr = np.array([sum(a[j]*w[i+j] for j in range(n) if i+j < len(w)) for i in range(n)])
    # But that's O(n^2). With FFT it should be O(n log n).
    
    # Let's use the convolution approach:
    # cross_corr[i] = sum_j a[j] * w[i+j]
    # Define A_rev[j] = a[n-1-j] (reversed a)
    # Then cross_corr[i] = sum_j A_rev[n-1-j] * w[i+j]
    # = (A_rev convolved with w)[n-1+i]
    # So we compute the convolution of A_rev and w, and take indices n-1, n, ..., 2n-2
    
    size2 = 1 << (n + 2*n - 2 - 1).bit_length()  # size >= 3n-2
    A_rev = np.zeros(size2)
    A_rev[:n] = a[::-1]
    W_padded = np.zeros(size2)
    W_padded[:len(w)] = w
    conv_result = irfft(rfft(A_rev, size2) * rfft(W_padded, size2), size2)
    cross_corr = conv_result[n-1 : 2*n-1]
    
    return 2.0 * cross_corr
```

Hmm, this is getting complex. Let me simplify and just use the direct approach for the gradient, but vectorize it differently.

Actually, for n=960, the direct O(n^2) approach for the gradient is 960^2 = 921,600 operations, which is very fast (microseconds in numpy). The L-BFGS overhead is in the solver itself, not the gradient computation. So the gradient computation is not the bottleneck.

Let me focus on what's actually going to help:

1. **Try more n values** - this is likely the biggest lever
2. **Better local optimization** - lower temperature L-BFGS
3. **More LP iterations** - the saddle point might not be fully reached

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-length exploration (n=768 to 1280)
2. Nesterov-accelerated LP alternating for global structure
3. L-BFGS with aggressive log-sum-exp annealing for local refinement
4. Greedy coordinate descent for final polish
5. Adaptive time allocation
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)

def norm_seq(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=s)
            if p.status in ('optimal', 'optimal_inaccurate'):
                return s
        except:
            pass
    return cp.ECOS

def solve_lp(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    prob = cp.Problem(cp.Minimize(M_var), [T @ b_var <= M_var, cp.sum(b_var) == 1])
    try:
        prob.solve(solver=solver)
    except:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    bv = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
    s = bv.sum()
    if s < 1e-15:
        return None
    return bv / s

def nesterov_lp(a0, t_lim, solver, max_it=100000):
    n = len(a0)
    a = norm_seq(a0)
    y = a.copy()
    best_a = a.copy()
    best_sc = score_of(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_sc
    for it in range(max_it):
        if time.monotonic() - t0 > t_lim:
            break
        b = solve_lp(y, solver)
        if b is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y /= sy
        denom = (it + 1) + alpha * (it + 2)
        a = ((it + 1) * a + alpha * (it + 2) * b) / denom
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a /= sa
        sc = score_of(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            stall = 0
        if it % 300 == 0:
            if best_sc < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_sc
            if stall >= 10:
                rng = np.random.default_rng(777 + it)
                y = y * (1 + 0.1 * rng.standard_normal(n))
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y /= sy
                stall = 0
    return best_a, best_sc

def softmax_z(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def lse_refine(a0, t_lim, T_start=100.0, T_end=0.05, n_stages=15, iters_per=1000):
    n = len(a0)
    a = norm_seq(a0)
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    for T in temps:
        if time.monotonic() - t0 > t_lim - 3:
            break
        def func(zi):
            ai = softmax_z(zi)
            b = fast_convolve(ai)
            Tb = T * b
            m = Tb.max()
            return (2.0 * n / T) * (m + np.log(np.exp(Tb - m).sum()))
        def jac(zi):
            ai = softmax_z(zi)
            b = fast_convolve(ai)
            Tb = T * b
            m = Tb.max()
            w = np.exp(Tb - m)
            w /= w.sum()
            ga = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                ga[js:je+1] += 2.0 * w[k] * ai[k - np.arange(js, je+1)]
            return ai * (ga - np.dot(ga, ai))
        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters_per, 'ftol': 1e-20, 'gtol': 1e-16,
                                  'maxfun': iters_per*100, 'disp': False})
            z = res.x
        except:
            pass
    return softmax_z(z)

def coord_descent(a0, t_lim):
    n = len(a0)
    a = norm_seq(a0).copy()
    best_a = a.copy()
    best_sc = score_of(a)
    t0 = time.monotonic()
    no_improve = 0
    for sweep in range(200):
        if time.monotonic() - t0 > t_lim:
            break
        improved = False
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        # Try adjusting top contributors
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je+1)
        contribs = a[idx] * a[peak_k - idx]
        top = idx[np.argsort(contribs)[::-1][:10]]
        for i in top:
            if time.monotonic() - t0 > t_lim:
                break
            old = a[i]
            if old < 1e-14:
                continue
            for f in [0.7, 0.8, 0.9, 1.1, 1.3]:
                a[i] = old * f
                a = np.maximum(a, 0)
                sc = score_of(a)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    break
                a[i] = old
        if not improved:
            # Try non-peak coordinates
            low_idx = np.argsort(a)[:5]
            for i in low_idx:
                if time.monotonic() - t0 > t_lim:
                    break
                for val in [0.001, 0.005, 0.01]:
                    a[i] = val
                    sc = score_of(a)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        best_a = a.copy()
                        improved = True
                        break
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
    return best_a, best_sc
