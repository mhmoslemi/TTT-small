        # Wait, that's the same formula: 2*a_{k-i} = 2*a_i when k=2i.
        # Actually let me redo this. c_k = sum_j a_j * a_{k-j}
        # d(c_k)/d(a_i) = d/d(a_i) [sum_j a_j * a_{k-j}]
        # = (d/d(a_i) a_i) * a_{k-i} + a_i * (d/d(a_i) a_{k-i})
        #   [from j=i term]         [from j=k-i term, if k-i ≠ i]
        # = a_{k-i} + a_i * (1 if k-i = i else 0) ... no wait
        # 
        # Let me be more careful. c_k = sum_j a_j * a_{k-j}.
        # d(c_k)/d(a_i) = sum_j [delta(j,i) * a_{k-j} + a_j * delta(k-j, i)]
        # = a_{k-i} + a_{k-i}  (if k-i ≠ i, i.e., k ≠ 2i)
        # = a_{k-i} + a_i      (if k = 2i, then k-i = i, so both terms are a_i)
        # = 2 * a_{k-i}        (in both cases, since when k=2i, k-i=i so 2*a_{k-i}=2*a_i)
        # 
        # Wait no. When k=2i: the sum_j delta(j,i)*a_{k-j} gives a_{k-i} = a_i.
        # And sum_j a_j * delta(k-j,i) = sum_j a_j * delta(j, k-i) = a_{k-i} = a_i.
        # So total = 2*a_i. Which equals 2*a_{k-i}. So the formula is always 2*a_{k-i}
        # (with the understanding that a_{k-i} = 0 if k-i is outside [0,n-1]).
        
        # But actually for the objective 2n * max(c) / (sum a)^2, the gradient is more complex
        # due to the (sum a)^2 in the denominator. Let me just use the gradient of the
        # p-norm for the update direction, but use the exact max for the step acceptance.
        
        # Simpler approach: use the p-norm gradient as direction
        p = 100  # fixed for mirror descent direction
        g = pnorm_grad(a, p)
        # Adjust for the (sum a)^2 denominator:
        # d/d(a_i) [2n * maxc / s^2] ≈ (2n/s^2) * g_i - (2n * maxc * 2/s^3) * 1 = ...
        # For simplicity, since we normalize after each step, just use g directly.
        
        # Multiplicative update
        a_new = a * np.exp(-lr * g / (np.max(np.abs(g)) + 1e-300))
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a * np.exp(-lr * 0.1 * g / (np.max(np.abs(g)) + 1e-300))
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 100:
                    lr *= 0.5
                    stall = 0
        
        lr *= lr_decay
        if lr < 1e-8:
            break
        
        if verbose and it % 5000 == 0 and it > 0:
            print(f"        MW it={it} ({time.monotonic()-t0:.1f}s): best={best_score:.6f}, lr={lr:.2e}")
    
    return best_a


def peak_shaving(a0, time_limit, verbose=True):
    """Iteratively shave the largest peaks in the convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for step in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        # Find the top peak (excluding edges which are just a_0^2 and a_{n-1}^2)
        c_work = c.copy()
        if len(c_work) > 2:
            c_work[0] = -1
            c_work[-1] = -1
        
        peak_k = np.argmax(c_work)
        peak_val = c[peak_k]
        
        # Find contributors to this peak
        # c[peak_k] = sum_j a[j] * a[peak_k - j] for valid j
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing the top contributors
        top_i = np.argsort(contribs)[::-1][:10]
        improved = False
        
        for ci in top_i:
            i = idx[ci]
            j = peak_k - i
            if i == j:
                # Reducing a[i] reduces the i*i term
                for factor in [0.99, 0.98, 0.97, 0.95, 0.90]:
                    a_trial = a.copy()
                    diff = a_trial[i] * (1 - factor)
                    a_trial[i] = a[i] * factor
                    # Redistribute
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask].sum()
                    if others < 1e-15:
                        break
                    a_trial[mask] += diff * a_trial[mask] / others
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_a = a.copy()
                        best_score = sc
                        improved = True
                        break
            else:
                # Reducing a[i] or a[j] (they're different)
                if j < 0 or j >= n:
                    continue
                for factor in [0.99, 0.98, 0.97, 0.95, 0.90]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    si = a_trial.sum()
                    a_trial = a_trial / si
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_a = a.copy()
                        best_score = sc
                        improved = True
                        break
            if improved:
                break
        
        # Also try increasing non-contributors (to spread mass)
        if not improved:
            # Find positions that don't contribute to the peak
            non_contrib = np.setdiff1d(np.arange(n), idx)
            if len(non_contrib) > 0:
                # Redistribute some mass from peak contributors to non-contributors
                for ci in top_i[:3]:
                    i = idx[ci]
                    for j_target in non_contrib[:5]:
                        a_trial = a.copy()
                        transfer = a[i] * 0.05
                        a_trial[i] -= transfer
                        a_trial[j_target] += transfer
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial = a_trial / st
                        sc = compute_score(a_trial)
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_a = a.copy()
                            best_score = sc
                            improved = True
                            break
                    if improved:
                        break
        
        if not improved and step > 100:
            # Add small noise to escape
            if step % 50 == 0:
                rng = np.random.default_rng(step)
                a_trial = a * (1 + 0.01 * rng.standard_normal(n))
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial = a_trial / st
                    sc = compute_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_a = a.copy()
                        best_score = sc
        
        if not improved and step % 500 == 499:
            # Check if we're stuck
            pass
    
    return best_a


def spectral_flat_init(n, rng, variant=0):
    """Initialize using spectral design principles."""
    k = np.arange(n)
    if variant == 0:
        # Flat magnitude with random phases (then make real and non-negative)
        phases = rng.standard_normal(n)
        # Enforce Hermitian symmetry for real sequence
        for i in range(1, n // 2 + 1):
            if n - i < n:
                phases[n - i] = -phases[i]
        if n % 2 == 0:
            phases[n // 2] = 0
        # DFT^(-1) of exp(i*phases) gives the sequence
        spectrum = np.exp(1j * phases)
        a = np.real(np.fft.ifft(spectrum))
    elif variant == 1:
        # Cosine bell (Hamming-like)
        x = np.linspace(0, 1, n)
        a = 0.54 + 0.46 * np.cos(2 * np.pi * x)
    elif variant == 2:
        # Blackman window
        x = np.linspace(0, 1, n)
        a = 0.42 + 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    elif variant == 3:
        # Flat-top window
        x = np.linspace(0, 1, n)
        a = (0.21557895 - 0.41663158 * np.cos(2*np.pi*x) + 0.277263158 * np.cos(4*np.pi*x)
             - 0.083578947 * np.cos(6*np.pi*x) + 0.006947368 * np.cos(8*np.pi*x))
    elif variant == 4:
        # Gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-x**2 * 2)
    elif variant == 5:
        # Papoulis window
        x = np.linspace(0, 1, n)
        a = 0.54 + 0.46 * np.cos(2*np.pi*x) + 0.08 * np.cos(4*np.pi*x)
    elif variant == 6:
        # Nuttall window
        x = np.linspace(0, 1, n)
        a = (0.3635819 - 0.4891775 * np.cos(2*np.pi*x) + 0.1365995 * np.cos(4*np.pi*x)
             - 0.0106411 * np.cos(6*np.pi*x) + 0.0006740 * np.cos(8*np.pi*x))
    else:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x**2 * 3)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_symmetric(a):
    """Make a sequence palindromic (symmetric)."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    return a_sym


def optimize_symmetric_pnorm(a0, time_limit, p_start=10.0, p_end=300.0, n_stages=8):
    """Optimize a symmetric sequence using p-norm L-BFGS on half the parameters."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = make_symmetric(a)
    s = a.sum()
    a = a / s
    
    # Work with half the sequence
    half = (n + 1) // 2
    a_half = a[:half].copy()
    
    best_half = a_half.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Log-domain for the half sequence
    z_half = np.log(np.maximum(a_half, 1e-300))
    z_half = z_half - z_half.max()
    
    temps = np.geomspace(p_start, p_end, n_stages)
    
    def full_from_half(z_h):
        """Reconstruct full symmetric sequence from log-domain half."""
        zh_max = z_h.max()
        ah = np.exp(z_h - zh_max)
        ah = ah / ah.sum()
        if n % 2 == 0:
            full = np.concatenate([ah, ah[::-1]])
        else:
            full = np.concatenate([ah, ah[-2::-1]])
        full = full / full.sum()
        return full
    
    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        if time_limit - (time.monotonic() - t0) < 5:
            break
        
        iters = max(30, int(4000 / n_stages))
        
        def func(z_h):
            full = full_from_half(z_h)
            return 2.0 * n * pnorm_value(full, stage_p)
        
        def jac(z_h):
            full = full_from_half(z_h)
            g_full = pnorm_grad(full, stage_p)
            # Map gradient back to half
            if n % 2 == 0:
                g_half = g_full[:half] + g_full[half-1::-1]
            else:
                g_half = g_full[:half].copy()
                g_half[-1] = g_full[half-1] + g_full[half]  # center element
                # Actually for odd n, the center is a[half-1] = a[half] (same position)
                # The gradient at the center gets double contribution
                # Let me just use the first half gradient
                g_half = g_full[:half]
            # Adjust for the normalization (sum = 1)
            # The Jacobian of full_from_half is complex, so let's use finite differences
            # Actually, let's just use the gradient directly (approximate)
            return g_half
        
        try:
            res = minimize(
                fun=func, x0=z_half, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': iters * 300, 'disp': False}
            )
            if res.fun < func(z_half):
                z_half = res.x
        except Exception:
            pass
        
        # Check score
        full = full_from_half(z_half)
        sc = compute_score(full)
        if sc < best_score:
            best_score = sc
            best_half = full[:half].copy()
    
    full = full_from_half(z_half)
    sc_final = compute_score(full)
    if sc_final < best_score:
        best_score = sc_final
        best_half = full[:half].copy()
    
    # Reconstruct from best half
    if n % 2 == 0:
        best_full = np.concatenate([best_half, best_half[::-1]])
    else:
        best_full = np.concatenate([best_half, best_half[-2::-1]])
    best_full = best_full / best_full.sum()
    return best_full


def nesterov_alternating(a0, time_limit, n_max_iters=30000, verbose=True):
    """
    Nesterov-type alternating optimization.
    Minimize max_k (T @ b)[k] where T is the convolution matrix with a fixed.
    Use a simple subgradient method for the outer problem.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # We'll use a "tug-of-war" approach:
    # At each iteration, find the peak of conv(a,a), then adjust a to reduce it.
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        max_c = c.max()
        
        # Find all peaks within 5% of max
        threshold = max_c * 0.95
        peak_indices = np.where(c >= threshold)[0]
        
        if len(peak_indices) == 0:
            break
        
        # For each peak, compute the "pressure" on each coordinate
        pressure = np.zeros(n)
        for pk in peak_indices:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            # Contribution of a[i] to c[pk] is a[i] * a[pk-i] + a[pk-i] * a[i] = 2*a[i]*a[pk-i]
            # (simplified; the actual gradient is 2*a[pk-i] for each i in range)
            contribs = 2.0 * a[idx] * a[pk - idx]
            pressure[idx] += contribs / (max_c + 1e-300)
        
        # Normalize pressure
        p_max = pressure.max()
        if p_max < 1e-300:
            break
        pressure = pressure / p_max
        
        # Step: reduce high-pressure coordinates, increase low-pressure ones
        # Alpha decays over time
        alpha = 0.1 / (1 + it * 0.001)
        
        a_new = a * (1 - alpha * pressure)
        a_new = np.maximum(a_new, 1e-300)
        # Add small uniform to prevent zeros
        a_new += 1e-10
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Revert and try smaller step
            a_new2 = a * (1 - alpha * 0.1 * pressure)
            a_new2 = np.maximum(a_new2, 1e-300)
            a_new2 += 1e-10
            a_new2 = a_new2 / a_new2.sum()
            sc2 = compute_score(a_new2)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_new2.copy()
                a = a_new2
            elif it % 1000 == 0:
                # Periodic noise
                rng = np.random.default_rng(it)
                a_noisy = a * (1 + 0.02 * rng.standard_normal(n))
                a_noisy = np.maximum(a_noisy, 0)
                a_noisy = a_noisy / a_noisy.sum()
                sc3 = compute_score(a_noisy)
                if sc3 < best_score:
                    best_score = sc3
                    best_a = a_noisy.copy()
                    a = a_noisy
        
        if verbose and it % 3000 == 0 and it > 0:
            print(f"          NA iter={it} ({time.monotonic()-t0:.1f}s): best={best_score:.6f}")
    
    return best_a


def lp_best_response(a_fixed):
    """Given fixed a (nonneg, sum=1), find b minimizing max conv(a,b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    # Build convolution matrix as sparse
    # T[k, j] = a_fixed[k-j] if 0 <= k-j < n, else 0
    # conv(a_fixed, b)[k] = sum_j a_fixed[k-j] * b[j]
    # For fixed a_fixed, this is linear in b
    # So we solve: min M s.t. T @ b <= M, b >= 0, sum(b) = 1
    
    # For efficiency, use the fact that T is a Toeplitz matrix
    # Let's just build it densely for n <= 2048
    if n > 2048:
        return None, None
    
    T = np.zeros((size, n))
    for j in range(n):
        a_idx = np.arange(max(0, -j), min(n, size - j))
        for ai in a_idx:
            kval = ai + j  # k such that k-j = ai
            T[kval, j] = a_fixed[ai]
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, n_max_iters=20000, verbose=True):
    """Nesterov LP-based alternating optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    y = a.copy()  # Nesterov momentum point
    stall = 0
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = lp_best_response(y)
        if b_new is None:
            break
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = compute_score(a)
        score_y = compute_score(y)
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 200:
            # Perturb
            rng = np.random.default_rng(it + 100)
            y = y + 0.03 * rng.standard_normal(n)
            y = np.maximum(y, 0)
            y = y / y.sum()
            stall = 0
        
        if verbose and (it + 1) % 2000 == 0:
            print(f"          NL iter={it+1} ({time.monotonic()-t0:.1f}s): best={best_score:.6f}")
    
    return best_a


def evaluate_safe(a):
    """Safely evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_state, a, label):
    """Try to update best state."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False


def propose_candidate(seed=42, budget_s=1800.0):
    """

    1. p-norm L-BFGS from parent
    2. Nesterov LP
    3. Peak shaving
    4. Symmetric optimization
    5. Spectral init + optimization
    6. Multi-length exploration
    7. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_state = {'score': float('inf'), 'seq': None}
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim
