Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Corrected Nesterov-accelerated projected gradient with log-sum-exp smoothing
2. LP-based best response refinement (fictitious play)
3. Targeted peak shaving via coordinate descent
4. Multi-length, multi-start exploration with diverse initializations
Key innovation: correct gradient for the bilinear auto-convolution objective.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:size]


def score_simplex(a):
    """Objective on simplex: 2*n*max(conv(a,a))."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum x = 1}. O(n log n)."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, T):
    """
    Correct gradient of (1/T)*log(sum_k exp(T*conv(a,a)_k)) w.r.t. a.
    
    Uses the identity: grad[p] = 2*S[p] - a[p]*w[2p]
    where S[p] = sum_j a[j]*w[j+p], w = softmax(T*conv(a,a)).
    """
    n = len(a)
    c = fast_convolve(a)  # length 2n-1
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    w = exp_Tc / exp_Tc.sum()  # length 2n-1, sums to 1
    
    # Compute S[p] = sum_j a[j] * w[j+p] for p = 0, ..., n-1
    # S = (a_rev * w)[n-1 : 2n-1]
    N = 1 << (3 * n - 2 - 1).bit_length()
    a_rev = a[::-1]
    A = rfft(a_rev, N)
    W = rfft(w, N)
    conv_result = irfft(A * W, N)
    S = conv_result[n - 1: n - 1 + n]
    
    # Correction: a[p] * w[2p] for p = 0, ..., n-1
    correction = a * w[0::2]
    
    return 2.0 * S - correction


def nesterov_pg(a0, time_limit, T_start=5.0, T_end=500.0,
                 lr_scale=1.0, n_stages=10, verbose=True):
    """
    Nesterov accelerated projected gradient with temperature annealing.
    Optimizes F_T(a) = (1/T)*log(sum exp(T*conv(a,a)_k)) on the simplex.
    """
    n = len(a0)
    x = project_simplex(np.maximum(a0, 0).astype(np.float64))
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_simplex(x)
    t0 = time.monotonic()
    
    # Precompute temperature schedule
    stages = n_stages
    steps_per_stage = max(100, int(time_limit / stages / 0.001))  # rough estimate
    # Actually, let's just run until time limit with fixed steps per stage
    steps_per_stage = 2000
    
    t_nesterov = 1.0
    step_count = 0
    total_steps = stages * steps_per_stage
    
    for stage in range(stages):
        # Temperature for this stage
        frac = stage / max(1, stages - 1)
        T = T_start * (T_end / T_start) ** frac
        
        # Learning rate: decrease with temperature and step
        lr = lr_scale / (T * np.sqrt(n))
        
        for step in range(steps_per_stage):
            if time.monotonic() - t0 >= time_limit:
                break
            
            # Nesterov extrapolation
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nesterov * t_nesterov))
            beta = (t_nesterov - 1.0) / t_new
            t_nesterov = t_new
            
            y = x + beta * (x - x_prev)
            y = np.maximum(y, 0)  # quick clip (will be projected anyway)
            
            # Gradient step
            grad = compute_gradient(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1
            
            # Track best (periodically to save time)
            if step_count % 100 == 0:
                sc = score_simplex(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
                    if verbose and step_count % 1000 == 0:
                        elapsed = time.monotonic() - t0
                        print(f"      NG step={step_count} T={T:.1f} score={best_score:.6f} ({elapsed:.0f}s)")
    
    # Final check
    sc = score_simplex(x)
    if sc < best_score:
        best_score = sc
        best_a = x.copy()
    
    return best_a, best_score


def build_conv_matrix_scaled(a):
    """Build matrix T such that T @ b = conv(a, b) / (for LP)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint: each row of T @ b <= M
    # T @ b = conv(a_fixed, b)
    T = build_conv_matrix_scaled(a_fixed)
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    if solver is None:
        solver = get_lp_solver()
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def lp_refine(a0, time_limit, solver=None, max_iters=5000):
    """LP-based best response refinement with averaging."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    x_avg = a.copy()
    best_a = a.copy()
    best_score = score_simplex(a)
    t0 = time.monotonic()
    iteration = 0
    
    # Use discounted best response
    discount = 0.95
    
    y = a.copy()  # "fictitious play" state
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_start = time.monotonic()
        b_best, M_val = solve_lp_best_response(y, solver)
        sol_time = time.monotonic() - t_start
        
        if b_best is None:
            continue
        
        # Update fictitious play state
        y = (1 - 1.0/(it + 1)) * y + (1.0/(it + 1)) * b_best
        y = project_simplex(y)
        
        # Also maintain a "current" iterate
        a = project_simplex(a * discount + b_best * (1 - discount))
        
        # Track best
        for candidate in [a, y, b_best]:
            sc = score_simplex(candidate)
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
        
        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it+1} M={M_val:.6f} score={best_score:.6f} ({elapsed:.0f}s, sol_time={sol_time:.2f}s)")
        
        iteration += 1
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Corrected Nesterov-accelerated projected gradient (fixes gradient bug in prior code)
2. LP-based fictitious play refinement
3. Targeted peak shaving via coordinate descent
4. Multi-length, multi-start exploration
Key fix: correct gradient accounts for the bilinear structure of auto-convolution,
grad[p] = 2*sum_j a[j]*w[j+p] - a[p]*w[2p], where w = softmax(T*conv(a,a)).
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:size]


def score_simplex(a):
    """Objective on simplex (sum=1): 2*n*max(conv(a,a))."""
    b = fast_convolve(a)
    return 2.0 * len(a) * float(b.max())


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, T):
    """
    Correct gradient of (1/T)*log(sum_k exp(T*conv(a,a)_k)) w.r.t. a.
    Uses: grad[p] = 2*S[p] - a[p]*w[2p], S[p] = sum_j a[j]*w[j+p].
    """
    n = len(a)
    c = fast_convolve(a)
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    w = exp_Tc / exp_Tc.sum()

    # S[p] = sum_j a[j] * w[j+p] = (a_rev * w)[n-1+p]
    N = 1 << (3 * n - 3).bit_length()
    a_rev = a[::-1]
    A = rfft(a_rev, N)
    W = rfft(w, N)
    conv_result = irfft(A * W, N)
    S = conv_result[n - 1: n - 1 + n]

    correction = a * w[0::2]
    return 2.0 * S - correction


def nesterov_pg(a0, time_limit, T_start=5.0, T_end=800.0,
                 lr_base=1.0, n_stages=12, verbose=True):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(np.maximum(a0, 0).astype(np.float64))
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_simplex(x)
    t0 = time.monotonic()
    steps_per_stage = 3000
    t_nest = 1.0
    step_count = 0

    for stage in range(n_stages):
        frac = stage / max(1, n_stages - 1)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_base / (T * np.sqrt(n))

        for step in range(steps_per_stage):
            if time.monotonic() - t0 >= time_limit:
                break

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            beta = (t_nest - 1.0) / t_new
            t_nest = t_new

            y = x + beta * (x - x_prev)
            grad = compute_gradient(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1

            if step_count % 200 == 0:
                sc = score_simplex(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
                    if verbose and step_count % 2000 == 0:
                        elapsed = time.monotonic() - t0
                        print(f"      NG step={step_count} T={T:.1f} score={best_score:.6f} ({elapsed:.0f}s)")

    sc = score_simplex(x)
    if sc < best_score:
        best_score = sc
        best_a = x.copy()
    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def peak_shave(a0, time_limit, n_steps=20000):
    """Targeted peak reduction via coordinate descent."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    best_a = a.copy()
    best_score = score_simplex(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -1e30
            b_work[-1] = -1e30
        peak_k = int(np.argmax(b_work))
        peak_val = b[peak_k]

        # Find top contributors to peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top = np.argsort(contribs)[::-1][:8]

        improved = False
        for ci in top:
            i = idx[ci]
            j = peak_k - i
            if i >= n or j >= n:
                continue
            for f in [0.95, 0.90, 0.85, 0.80]:
                a_t = a.copy()
                a_t[i] *= f
                if i != j:
                    a_t[j] *= 0.97
                a_t = project_simplex(a_t)
                sc = score_simplex(a_t)
                if sc < best_score - 1e-12:
                    a = a_t
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            # Try single coordinate
            for i in top[:5]:
                for f in [0.97, 0.93, 0.88]:
                    a_t = a.copy()
                    a_t[i] *= f
                    a_t = project_simplex(a_t)
                    sc = score_simplex(a_t)
                    if sc < best_score - 1e-12:
                        a = a_t
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 50:
                break

    return best_a, best_score
