This often works well in practice.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. High-efficiency coordinate descent with incremental convolution and smart 1D line search
2. Log-sum-exp smoothed gradient with mirror descent for escaping local minima
3. Multi-start with diverse initializations (parent, periodic, random, sparse)
4. Multiple sequence lengths explored
5. FFT-based fast convolution throughout
6. Active-set subgradient with Polyak step for final squeezing

import numpy as np
from scipy.fft import rfft, irfft
import time

TOTAL_BUDGET = 1770.0


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution a*b via FFT."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return v / s


def compute_local_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 locally (no clamping)."""
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (s * s)


def safe_evaluate(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        s = sum(a_list)
        if s < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def coord_descent(a0, time_limit, verbose=True, label=""):
    """
    Coordinate descent with incremental convolution updates.
    Works in unnormalized space (tracks sum separately).
    For each coordinate, tries multiple multipliers and picks the best.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = 1.0

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M / (S * S)
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    # Precompute multipliers to try
    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 0.1, 2.0, 0.3, 3.0])

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Randomize sweep order for diversity
        order = np.arange(n)
        if sweeps % 3 == 0:
            np.random.shuffle(order)

        for i in order:
            old_ai = a[i]
            if old_ai < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            aff_len = k_end - k_start + 1

            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end + 1 < 2 * n - 1:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            # a[j-i] for j in [k_start, k_end]
            # j-i ranges from k_start-i=0 to k_end-i <= n-1
            aff_a = a[0:aff_len]  # a[j-i] where j-i goes 0 to aff_len-1
            aff_conv = conv[k_start:k_end + 1]

            has_2i = (2 * i >= k_start) and (2 * i <= k_end)
            idx_2i = 2 * i - k_start if has_2i else -1

            base_score_denom = S * S
            best_new_score = 2.0 * n * M / base_score_denom
            best_new_val = old_ai
            best_new_M = M
            best_new_S = S

            # Try different multipliers
            for mult in multipliers:
                new_val = old_ai * mult
                delta = new_val - old_ai
                new_S = S + delta
                if new_S < 1e-15:
                    continue

                # Compute new affected max
                new_aff_val = aff_conv + 2.0 * delta * aff_a
                new_M = max(unaff_max, float(new_aff_val.max()))
                if has_2i:
                    quadratic_val = aff_conv[idx_2i] + 2.0 * delta * old_ai + delta * delta
                    if quadratic_val > new_M:
                        new_M = quadratic_val

                new_score = 2.0 * n * new_M / (new_S * new_S)
                if new_score < best_new_score:
                    best_new_score = new_score
                    best_new_val = new_val
                    best_new_M = new_M
                    best_new_S = new_S

            if best_new_val != old_ai and best_new_score < best_score * (1.0 - 1e-12):
                delta = best_new_val - old_ai
                # Apply update
                a[i] = best_new_val
                S = best_new_S

                # Incremental conv update
                conv[k_start:k_end + 1] += 2.0 * delta * aff_a
                if has_2i:
                    conv[2 * i] += delta * delta

                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M / (S * S)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): score={best_score:.6f} M={M:.10f} S={S:.6f}")

    return best_a, best_score

def smoothed_gradient_descent(a0, time_limit, beta_init=50.0, beta_end=500.0, verbose=True):
    """
    Mirror descent with log-sum-exp smoothed objective.
    f_beta(a) = (1/beta) * log(sum_j exp(beta * (a*a)[j]))
    Gradient computed via FFT cross-correlation.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    a = a / s  # normalize to sum=1

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Anneal beta
        progress = min(1.0, it / max(1, int(time_limit / 0.5)))
        beta = beta_init * (beta_end / beta_init) ** progress

        # Compute convolution
        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            break

        # Compute smoothed gradient: grad[i] = 2 * sum_j w[j] * a[j-i]
        # where w[j] = exp(beta * conv[j]) / sum_k exp(beta * conv[k])
        # This is 2 * cross_corr(w, a) where cross_corr[w,a][i] = sum_j w[j] * a[j-i]

        # Compute weights
        conv_shifted = conv - conv.max()  # numerical stability
        log_w = beta * conv_shifted
        log_w -= log_w.max()
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(conv)) / len(conv)
        else:
            w = w / w_sum

        # Compute gradient via FFT cross-correlation
        # grad[i] = 2 * sum_j w[j] * a[j-i] for valid j-i
        # = 2 * (w convolve a)[i] ... let's think about this
        # Actually: sum_j w[j] * a[j-i] where j ranges over [0, 2n-2] and j-i over [0, n-1]
        # Let k = j-i, so j = k+i: sum_k w[k+i] * a[k]
        # This is a cross-correlation of w and a

        # Use FFT: cross_corr(w, a) can be computed as ifft(conj(rfft(a)) * rfft(w))
        # But we need to be careful about the definition
        # sum_k w[k+i] * a[k] = (w ⊛ a)(i) (cross-correlation)
        # = ifft(rfft(w) * conj(rfft(a)))
        # But w has length 2n-1 and a has length n, so we need to pad

        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        # cross_corr: sum_k w[k+i] * a[k]
        # = ifft(W * conj(A)) shifted appropriately
        cc = irfft(W * np.conj(A), fft_size)
        # cc[i] should give sum_k w[k+i] * a[k] for i in [0, n-1]
        # Need to verify the shift... Let's just use it in [0, n-1]
        grad = 2.0 * cc[0:n]

        # Mirror descent step
        lr = 1.0 / np.sqrt(it + 1.0)
        lr = min(lr, 1.0)

        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        a = a_new

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD it={it} ({elapsed:.0f}s): best={best_score:.6f} beta={beta:.1f}")

    return best_a, best_score


def subgradient_active_set(a0, time_limit, tol=1e-6, verbose=True):
    """
    Subgradient method using full active set (all j within tol of max).
    Polyak step size.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    conv = fast_convolve(a)
    M = conv.max()
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        # Active set: all j where conv[j] >= M - tol*M
        active = conv >= M * (1.0 - tol)
        n_active = active.sum()
        if n_active == 0:
            active[np.argmax(conv)] = True
            n_active = 1

        # Subgradient: average of grad_j over active set
        # grad_j[i] = 2*a[j-i] for valid j-i
        sg = np.zeros(n)
        active_indices = np.where(active)[0]
        for j in active_indices:
            i_start = max(0, j - n + 1)
            i_end = min(n - 1, j)
            sg[i_start:i_end + 1] += 2.0 * a[j - i_end:j - i_start + 1][::-1]
        sg /= n_active

        # Polyak step
        gap = M - best_score / (2.0 * n)
        sg_norm_sq = (sg * sg).sum()
        if sg_norm_sq < 1e-30:
            break
        eta = max(gap / sg_norm_sq, 0.0)
        eta = min(eta, 1.0)

        # Projected gradient step
        a_new = a - eta * sg
        a_new = simplex_project(a_new)

        a = a_new

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.6f} n_act={n_active}")

    return best_a, best_score


def periodic_init(n, period_range=(2, 12), rng=None):
    """Initialize with a periodic pattern."""
    if rng is None:
        rng = np.random.default_rng()
    period = int(rng.integers(period_range[0], period_range[1]))
    a = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        a[i] = 0.5 + 0.4 * np.cos(2 * np.pi * phase)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def random_init(n, rng=None):
    """Random non-negative initialization."""
    if rng is None:
        rng = np.random.default_rng()
    a = rng.random(n)
    a = a / a.sum()
    return a


def sparse_init(n, sparsity=0.3, rng=None):
    """Initialize with many zeros."""
    if rng is None:
        rng = np.random.default_rng()
    mask = rng.random(n) < sparsity
    a = np.zeros(n)
    a[mask] = rng.random(mask.sum())
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    return a


def cosine_init(n, power=1.0, rng=None):
    """Cosine window initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    a = (0.5 * (1 + np.cos(np.pi * x))) ** power
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def two_hump_init(n, rng=None):
    """Two-peak initialization."""
    if rng is None:
        rng = np.random.default_rng()
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.2, 0.4)
    c2 = n * rng.uniform(0.6, 0.8)
    w = n * rng.uniform(0.08, 0.2)
    a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def geometric_init(n, ratio_range=(0.8, 0.99), rng=None):
    """Geometric sequence initialization."""
    if rng is None:
        rng = np.random.default_rng()
    r = rng.uniform(ratio_range[0], ratio_range[1])
    a = r ** np.arange(n, dtype=np.float64)
    a = a / a.sum()
    return a

def run_pipeline(a0, time_budget):
    """Run the full optimization pipeline on a single sequence."""
    best_a = a0.copy()
    best_score = compute_local_score(a0)
    t_start = time.monotonic()

    # Phase 1: Coarse coordinate descent
    t1 = time.monotonic()
    t_cd_coarse = min(time_budget * 0.45, time_budget - 10)
    if t_cd_coarse > 5:
        a1, s1 = coord_descent(a0, t_cd_coarse, verbose=True, label="coarse")
        if s1 < best_score:
            best_score = s1
            best_a = a1.copy()

    # Phase 2: Smoothed gradient (escape local minima)
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed - 5
    if remaining > 10:
        t_sgd = min(remaining * 0.25, 100.0)
        a2, s2 = smoothed_gradient_descent(best_a, t_sgd, beta_init=30.0, beta_end=200.0, verbose=True)
        if s2 < best_score:
            best_score = s2
            best_a = a2.copy()
        # Also try CD on the SGD result
        elapsed = time.monotonic() - t_start
        remaining = time_budget - elapsed - 5
        if remaining > 10:
            t_cd2 = min(remaining * 0.3, 60.0)
            a3, s3 = coord_descent(a2, t_cd2, verbose=False, label="post_sgd")
            if s3 < best_score:
                best_score = s3
                best_a = a3.copy()

    # Phase 3: Fine coordinate descent
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed - 5
    if remaining > 5:
        t_fine = min(remaining * 0.4, 80.0)
        a4, s4 = coord_descent_fine(best_a, t_fine, verbose=True, label="fine")
        if s4 < best_score:
            best_score = s4
            best_a = a4.copy()

    # Phase 4: Subgradient active set
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed - 3
    if remaining > 10:
        t_sg = min(remaining * 0.5, 40.0)
        a5, s5 = subgradient_active_set(best_a, t_sg, tol=1e-5, verbose=False)
        if s5 < best_score:
            best_score = s5
            best_a = a5.copy()
        # Quick CD after subgradient
        elapsed = time.monotonic() - t_start
        remaining = time_budget - elapsed - 3
        if remaining > 5:
            a6, s6 = coord_descent_fine(a5, min(remaining * 0.6, 20.0), verbose=False, label="post_sg")
            if s6 < best_score:
                best_score = s6
                best_a = a6.copy()

    # Phase 5: Final fine CD
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed - 2
    if remaining > 3:
        a7, s7 = coord_descent_fine(best_a, remaining * 0.8, verbose=False, label="final")
        if s7 < best_score:
            best_score = s7
            best_a = a7.copy()

    return best_a, best_score


def propose_candidate():
    """
    Main entry point. Multi-phase optimization with:
    - Parent-based refinement (primary)
    - Diverse initializations (periodic, sparse, geometric, two-hump)
    - Multiple sequence lengths
    - Coordinate descent + smoothed gradient + subgradient
    """
    global_start = time.monotonic()
    deadline = global_start + TOTAL_BUDGET
    rng = np.random.default_rng(12345)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        local_score = compute_local_score(a_norm)
        if local_score < best_score:
            # Verify with actual evaluator
            verified = safe_evaluate(a_norm)
            if np.isfinite(verified) and verified < best_score:
                best_score = verified
                best_seq = a_norm.copy()
                elapsed = time.monotonic() - global_start
                print(f"  *** NEW BEST [{label}]: {best_score:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
            elif np.isfinite(verified) and verified < local_score + 0.01:
                best_score = min(best_score, verified)
                if local_score < best_score + 0.001:
                    best_seq = a_norm.copy()

    # Get parent
    parent = get_parent()
    if parent is not None:
        print(f"Parent: n={len(parent)}, score={safe_evaluate(parent):.6f}")
        try_update(parent, "parent")
    else:
        print("No parent found, using random init")
        parent = random_init(1024, rng)
        try_update(parent, "random_init")

    n_parent = len(parent)

    # === Strategy 1: Refine parent with full pipeline ===
    print(f"\n=== Strategy 1: Refine parent (n={n_parent}) ===")
    t_alloc = min((deadline - time.monotonic()) * 0.40, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_best, s_best = run_pipeline(parent, t_alloc)
        try_update(a_best, "parent_pipeline")

    # === Strategy 2: Perturbed parent ===
    print(f"\n=== Strategy 2: Perturbed parent ===")
    elapsed = time.monotonic() - global_start
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t_alloc = min(remaining * 0.20, 200.0)
        # Add noise and re-optimize
        noisy = parent + rng.standard_normal(n_parent) * 0.02
        noisy = np.maximum(noisy, 0)
        noisy = noisy / noisy.sum()
        a_best2, s_best2 = run_pipeline(noisy, t_alloc)
        try_update(a_best2, "perturbed")

    # === Strategy 3: Parent with different noise levels ===
    elapsed = time.monotonic() - global_start
    remaining = deadline - time.monotonic()
    if remaining > 40:
        t_alloc = min(remaining * 0.12, 100.0)
        noisy2 = parent + rng.standard_normal(n_parent) * 0.08
        noisy2 = np.maximum(noisy2, 0)
        noisy2 = noisy2 / noisy2.sum()
        a_best3, s_best3 = run_pipeline(noisy2, t_alloc)
        try_update(a_best3, "noisy")

    # === Strategy 4: Different initializations at parent length ===
    print(f"\n=== Strategy 4: Diverse inits (n={n_parent}) ===")
    elapsed = time.monotonic() - global_start
    remaining = deadline - time.monotonic()
    if remaining > 30:
        inits = [
            ("periodic", periodic_init(n_parent, (3, 9), rng)),
            ("cosine", cosine_init(n_parent, 1.5, rng)),
            ("two_hump", two_hump_init(n_parent, rng)),
            ("sparse", sparse_init(n_parent, 0.4, rng)),
            ("geometric", geometric_init(n_parent, (0.85, 0.99), rng)),
        ]
        n_inits = len(inits)
        per_init_time = min(remaining * 0.8 / n_inits, 40.0)
        for name, init_a in inits:
            if time.monotonic() > deadline - 20:
                break
            print(f"  Trying {name} init, budget={per_init_time:.0f}s")
            a_r, s_r = run_pipeline(init_a, per_init_time)
            try_update(a_r, f"init_{name}")

    # === Strategy 5: Different sequence lengths ===
    print(f"\n=== Strategy 5: Different lengths ===")
    elapsed = time.monotonic() - global_start
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for n_try in [512, 2048]:
            if time.monotonic() > deadline - 20:
                break
            t_alloc = min(remaining * 0.25, 60.0)
            if t_alloc < 10:
                break
            # Resample parent to new length or use fresh init
            if n_try != n_parent:
                # Interpolate parent to new length
                x_old = np.linspace(0, 1, n_parent)
                x_new = np.linspace(0, 1, n_try)
                init_n = np.interp(x_new, x_old, parent)
                init_n = np.maximum(init_n, 0)
                init_n = init_n / init_n.sum()
            else:
                init_n = parent.copy()

            print(f"  n={n_try}, budget={t_alloc:.0f}s")
            a_r, s_r = run_pipeline(init_n, t_alloc)
            try_update(a_r, f"n_{n_try}")

    # === Strategy 6: Final polishing of best ===
    print(f"\n=== Strategy 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        t_final = min((deadline - time.monotonic()) * 0.5, 60.0)
        if t_final > 5:
            print(f"  Final polish budget: {t_final:.0f}s")
            a_fin, s_fin = coord_descent_fine(best_seq, t_final, verbose=True, label="FINAL")
            try_update(a_fin, "final_polish")

    # === Strategy 7: One more CD pass ===
    if best_seq is not None and time.monotonic() < deadline - 5:
        t_last = min((deadline - time.monotonic()) * 0.7, 30.0)
        if t_last > 3:
            a_last, s_last = coord_descent(best_seq, t_last, verbose=False, label="last")
            try_update(a_last, "last_cd")

    # Final fallback
    if best_seq is None:
        best_seq = parent if parent is not None else np.ones(1024) / 1024
        best_score = safe_evaluate(best_seq)

    # Final verification
    final = safe_evaluate(best_seq)
    if np.isfinite(final):
        best_score = final
    else:
        # Shouldn't happen, but fallback
        best_seq = np.ones(1024) / 1024
        best_score = safe_evaluate(best_seq)

    print(f"\n{'='*60}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}")
    print(f"Total time: {time.monotonic() - global_start:.1f}s")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
