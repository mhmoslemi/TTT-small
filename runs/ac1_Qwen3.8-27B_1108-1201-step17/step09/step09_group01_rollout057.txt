"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:

import numpy as np
from scipy.fft import rfft, irfft
import time

# --- Core utilities ---

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# --- LP solver ---

def get_lp_solver():
    import cvxpy as cp
    if hasattr(get_lp_solver, '_cached'):
        return get_lp_solver._cached
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                get_lp_solver._cached = solver_name
                return solver_name
        except:
            continue
    get_lp_solver._cached = cp.ECOS
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    # conv(a, b)[k] = sum_i a[i] * b[k-i] for valid i, k-i
    # = sum_j a[k-j] * b[j] for valid k-j, j
    # This is: (T @ b)[k] where T[k,j] = a[k-j] if 0<=k-j<n else 0
    # Build T as a Toeplitz-like matrix
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

# --- Exponentiated Gradient (Mirror Descent) ---

def expo_gradient(a, beta, conv_cache=None):
    """
    Exponentiated gradient step for minimizing max_k conv(a,a)[k] on the simplex.
    Returns (a_new, score).
    """
    n = len(a)
    if conv_cache is None:
        conv_a = fast_convolve(a)
    else:
        conv_a = conv_cache
    
    max_conv = conv_a.max()
    scaled = beta * (conv_a - max_conv)
    scaled = np.clip(scaled, -100, 0)
    weights = np.exp(scaled)
    Z = weights.sum()
    if Z < 1e-30:
        # Fallback: uniform over argmax
        idx = np.where(conv_a == max_conv)[0]
        weights = np.zeros_like(conv_a)
        weights[idx] = 1.0 / len(idx)
        Z = 1.0
    weights = weights / Z
    
    # Gradient: grad[j] = 2 * sum_i a[i] * weights[j+i]
    # = 2 * np.correlate(weights, a, mode='valid')
    # But weights has length 2n-1 and a has length n
    # np.correlate(weights, a, 'valid') has length (2n-1) - n + 1 = n ✓
    # [j] = sum_i weights[i+j] * a[i]
    grad = 2.0 * np.correlate(weights, a, mode='valid')
    
    return grad, weights, conv_a

def expo_gradient_descent(a0, time_limit, lr_init=0.01, beta_init=20.0, 
                          momentum_init=0.5, verbose=False):
    """
    Nesterov-accelerated exponentiated gradient (mirror descent) 
    with adaptive beta for the smooth max.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    
    # Nesterov state
    y = a.copy()  # lookahead
    a_prev = a.copy()
    
    best_a = a.copy()
    conv_a = fast_convolve(a)
    best_score = 2.0 * n * conv_a.max()
    
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    beta = beta_init
    momentum = momentum_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        if it == 1:
            y = a.copy()
        else:
            mu = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
            y = (mu * a - momentum * y)
            # Project to simplex
            y = simplex_project(y)
        
        # Compute gradient at y
        conv_y = fast_convolve(y)
        grad, weights, _ = expo_gradient(y, beta, conv_y)
        
        # Mirror descent update: a_new = a * exp(-lr * grad)
        # Normalize to simplex
        log_a = np.log(a + 1e-30) - lr * grad
        # Shift for numerical stability
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * conv_new.max()
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a_prev = a
        a = a_new
        
        # Adaptive parameters
        if no_improve > 200:
            lr *= 0.8
            beta *= 1.3
            no_improve = 0
            if lr < 1e-10:
                break
        elif it % 100 == 0:
            beta *= 1.01
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    EGD it={it} ({elapsed:.0f}s): score={best_score:.6f} lr={lr:.6f} beta={beta:.1f}")
    
    return best_a, best_score

def coord_descent(a0, time_limit, n_steps_per_coord=8, verbose=False, label=""):
    """
    Fast coordinate descent with multiple step sizes.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    sweep = 0
    no_improve = 0
    
    step_factors = [0.97, 0.99, 0.995, 1.005, 1.01, 1.03, 0.9, 1.1]
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))
            
            affected = conv[k_start:k_end + 1].copy()
            
            best_delta = 0.0
            best_new_M = M
            
            for f in step_factors:
                new_val = a[i] * f
                delta = new_val - a[i]
                if new_val < -1e-15:
                    continue
                
                # Update affected conv values
                new_affected = affected + 2.0 * delta * a[:n]
                # Handle the quadratic term at position 2*i
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-16:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                
                # Update conv
                changes = 2.0 * best_delta * a[:n].copy()
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                
                score = 2.0 * n * M  # sum is 1
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0
    
    return best_a, best_score

# --- Interpolation and initialization ---

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def get_lp_solver():
    """Find a working LP solver."""
    if hasattr(get_lp_solver, '_cached'):
        return get_lp_solver._cached
    import cvxpy as cp
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                get_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    get_lp_solver._cached = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)

            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def compute_smooth_grad(a, beta):
    """
    Compute gradient of smooth max approximation (1/beta)*log(sum exp(beta*q_k))
    of max_k conv(a,a)[k] at point a.
    Returns (gradient, conv_value, max_conv).
    """
    n = len(a)
    conv_a = fast_convolve(a)
    max_conv = conv_a.max()
    scaled = beta * (conv_a - max_conv)
    scaled = np.clip(scaled, -100, 0)
    weights = np.exp(scaled)
    Z = weights.sum()
    if Z < 1e-30:
        idx = np.where(np.isclose(conv_a, max_conv, atol=1e-10))[0]
        weights = np.zeros_like(conv_a)
        weights[idx] = 1.0 / len(idx)
        Z = 1.0
    weights = weights / Z
    # grad[j] = 2 * sum_i a[i] * weights[j+i] = 2 * np.correlate(weights, a, 'valid')[j]
    grad = 2.0 * np.correlate(weights, a, mode='valid')
    return grad, conv_a, max_conv

        a_prev = a.copy()
        momentum = 1.0

        else:
            y = a

    b_seq = b.copy()  # Nesterov sequent for b

                best_a = cand.copy()

        # Periodic perturbation to escape local minima
        if perturb_every > 0 and it % perturb_every == 0:
            noise = rng.standard_normal(n) * 0.02
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            b = b + noise
            b = np.maximum(b, 0)
            b = b / b.sum()

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP[{label}] it={it} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Fast coordinate descent with multiple step sizes for polishing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    sweep = 0
    no_improve = 0

    step_factors = [0.98, 0.99, 0.995, 1.005, 1.01, 1.02, 0.95, 1.05]

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            best_delta = 0.0
            best_new_M = M

            for f in step_factors:
                new_val = a[i] * f
                delta = new_val - a[i]
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-16:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a.copy()
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] = conv[k_start:k_end + 1] + changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                score = 2.0 * n * M
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweep % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweep} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate via evaluate_sequence and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization combining:

    2. LP fictitious play with Nesterov averaging and periodic perturbations
    3. Coordinate descent polishing
    4. Multi-length exploration

    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(42)

    best_seq = None

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initialized uniform: n=1024, score={best_score:.6f}")

    n_best = len(best_seq)

    # === Phase 1: Quick Exponentiated Gradient (warm-up) ===
    print(f"\n=== Phase 1: EGD warm-up (score={best_score:.6f}) ===")
    t1_budget = 60.0

        n_best = len(best_seq)

    # === Phase 4: Second round of EGD + FP ===
    print(f"\n=== Phase 4: Second round (score={best_score:.6f}) ===")
    for rnd in range(3):

        if rem < 60:
            break
        # EGD

    return [float(x) for x in best_seq]
