        a_new = irfft(A_new, n_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a  # revert
        else:
            a_new = a_new / s
        a = a_new
    return a

def gs_warmup(a, n_fft, n_iters=20):
    """GS warmup with uniform target amplitude."""
    n = len(a)
    # Target: uniform amplitude
    # The DC component should be sum(a) = 1
    # Other components: 1/sqrt(n) or similar
    target = np.ones(n_fft // 2 + 1)
    target[0] = 1.0  # DC
    target[1:] = 1.0 / np.sqrt(n)
    return gs_iteration(a, n_fft, target, n_iters)

def golden_section_search(f, lo, hi, max_iter=40):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if fc <= fd:
            hi = d; d = c; fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c; c = d; fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent with golden section 1D search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))
            
            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            
            c_vals = np.empty(k_hi - k_lo + 1)
            for idx in range(k_hi - k_lo + 1):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            
            if has_quad:
                quad_idx = 2 * i - k_lo
            
            def compute_score(x):
                ns = current_sum + (x - v)
                if ns < 0.01:
                    return 1e10
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                nm = max(unaff_max, float(nv.max()))
                return 2.0 * n * nm / (ns * ns)
            
            upper = max(3.0 * v, 5.0 / n)
            best_x = golden_section_search(compute_score, 0.0, upper, max_iter=35)
            
            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes
                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

    return best_a, best_score

def adam_gradient_descent(a0, time_limit, beta_init=500.0, lr_init=0.05,
                          beta_growth=1.005, verbose=False, label=""):
    """
    Adam gradient descent on the softmax-smoothed max objective.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    lr = lr_init
    beta = beta_init
    
    t0 = time.monotonic()
    no_improve = 0
    patience = 200
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        t += 1
        # Compute convolution
        conv = fast_convolve(a)
        
        # Compute softmax weights
        w = compute_smoothing_weights(conv, beta)
        
        # Compute gradient
        h = compute_gradient_fft(a, w)
        grad = 4.0 * n * h
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Gradient step
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_cand = a - step
        a_cand = np.maximum(a_cand, 0.0)
        cs = a_cand.sum()
        if cs < 1e-10:
            a_cand = a
        else:
            a_cand = a_cand / cs
        
        # Evaluate
        conv_cand = fast_convolve(a_cand)
        score_cand = 2.0 * n * float(conv_cand.max())
        
        if score_cand < best_score - 1e-15:
            best_score = score_cand
            best_a = a_cand.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_cand
        
        # Adaptive adjustments
        if no_improve > patience:
            lr *= 0.7
            no_improve = 0
            if lr < 1e-6:
                lr = lr_init
                # Perturb to escape
                a = a + np.random.default_rng(t * 12345).standard_normal(n) * 0.01
                a = np.maximum(a, 0)
                a = a / a.sum()
        
        # Grow beta (temperature) over time
        beta *= beta_growth
        
        # Periodically reduce beta to explore
        if t % 5000 == 0:
            beta = beta_init * 10  # reset to a moderate value
        
        if verbose and t % 500 == 0:
            print(f"    AdamGD {label} t={t} ({elapsed:.0f}s): "
                  f"score={best_score:.8f}, beta={beta:.0f}, lr={lr:.6f}")
    
    return best_a, best_score
