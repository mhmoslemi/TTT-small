    return irfft(A * A, fft_size)[:size]


def compute_lse_gradient(a, T):
    """
    Compute gradient of T*log(sum exp(conv(a,a)/T)) with respect to a.
    Returns gradient vector of same length as a.
    Uses FFT for efficiency.

    b = fast_convolve(a)
    # Softmax weights
    b_shifted = b / T - (b / T).max()
    exp_b = np.exp(b_shifted)
    w = exp_b / exp_b.sum()

    # This is 2 * (w cross-correlate a)[i]
