def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def softmax_from_z(z):

def smart_coordinate_descent(a0, time_limit, n_rounds=20000):
    """
    Smart coordinate descent that identifies the most impactful coordinates
    to modify and makes targeted moves.
    """
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    
    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        
        # Find top contributors to the peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_star - idx]
        
        # Also consider secondary peaks
        b_sorted = np.argsort(b)[::-1]
        top_peaks = b_sorted[:3]
        
        # Compute "cost" of each coordinate (how much it contributes to all top peaks)
        cost = np.zeros(n)
        for kp in top_peaks:
            jp = max(0, kp - n + 1)
            jp_end = min(n - 1, kp)
            idp = np.arange(jp, jp_end + 1)
            cost[idp] += b[kp] * a[idp] * a[kp - idp]
        
        # Try to reduce the most costly coordinates
        top_cost_idx = np.argsort(cost)[::-1][:10]
        
        improved = False
        for i in top_cost_idx:
            if a[i] < 1e-14:
                continue
            # Try reducing coordinate i and redistributing
            for factor in [0.95, 0.90, 0.85, 0.80, 0.70]:
                diff = a[i] * (1 - factor)
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] *= factor
                # Redistribute to low-cost coordinates
                low_cost_mask = cost <= np.mean(cost)
                low_cost_mask[i] = False
                if low_cost_mask.sum() > 0:
                    dist = a_trial[low_cost_mask] + 1e-10
                    a_trial[low_cost_mask] = dist + diff * dist / dist.sum()
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
            if improved:
                break
        
        if not improved:
            # Try pair moves
            top_contrib = np.argsort(contribs)[::-1][:5]
            found = False
            for ci in top_contrib:
                i = idx[ci]
                j = k_star - i
                if i == j or a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.90, 0.85]:
                    for f2 in [0.95, 0.90, 0.85]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = norm_sequence(a_trial)
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            found = True
                            improved = True
                            break
                    if found:
                        break
                if found:
                    break
            
            if not improved:
                no_improve += 1
                if no_improve > 100:
                    # Small perturbation to escape
                    rng_local = np.random.default_rng(rnd * 7919)
                    noise = rng_local.standard_normal(n) * 0.005
                    a = a + noise
                    a = norm_sequence(a)
                    no_improve = 0
        else:
            no_improve = 0
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:

                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

    t0 = time.monotonic()

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()

        y = y / sy
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

    return best_a, best_score


def mirror_descent_refine(a0, time_limit, n_steps=200000):
    """
    Mirror descent (multiplicative weights) with multi-peak subgradient
    and adaptive step size. Uses momentum for better convergence.
    """
    n = len(a0)
    a = norm_sequence(a0)
    z = z_from_a(a)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Momentum buffer
    momentum = np.zeros(n)
    beta = 0.7  # momentum coefficient
    base_lr = 0.08
    decay_rate = 0.9998
    M_peaks = 5
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)

        # Compute multi-peak weighted subgradient
        peak_indices = np.argsort(b)[::-1][:M_peaks]
        peak_vals = b[peak_indices]
        # Weight by proximity to max (softmax-like)
        w_temp = 50.0
        w_shifted = peak_vals - peak_vals[0]
        w_exp = np.exp(w_shifted * w_temp)
        weights = w_exp / w_exp.sum()

        g = np.zeros(n)
        for wi, ki in zip(weights, peak_indices):
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * wi * a[ki - idx]

        # Momentum update
        momentum = beta * momentum + (1 - beta) * g

        # Adaptive learning rate
        lr = base_lr * (decay_rate ** step)

        # Mirror descent update in log space
        z_trial = z - lr * momentum
        a_trial = softmax_from_z(z_trial)
        score_trial = 2.0 * n * fast_convolve(a_trial).max()

        # Try multiple step sizes if primary doesn't improve
        accepted = False
        if score_trial < best_score:
            z = z_trial
            a = a_trial
            best_score = score_trial
            best_a = a.copy()
            no_improve = 0
            accepted = True
        else:
            # Try smaller steps
            for mult in [0.5, 0.25, 0.1]:
                z_trial = z - lr * mult * momentum
                a_trial = softmax_from_z(z_trial)
                score_trial = 2.0 * n * fast_convolve(a_trial).max()
                if score_trial < best_score:
                    z = z_trial
                    a = a_trial
                    best_score = score_trial
                    best_a = a.copy()
                    no_improve = 0
                    accepted = True
                    break

        if not accepted:
            no_improve += 1
            if no_improve > 300:
                # Perturbation to escape
                rng_local = np.random.default_rng(step * 104729 + 7)
                noise = rng_local.standard_normal(n) * 0.01
                z = z + noise
                a = softmax_from_z(z)
                no_improve = 0

    return best_a, best_score


def smart_coord_descent(a0, time_limit, n_rounds=50000):
    """
    Smart coordinate descent with targeted moves on peak-contributing coordinates.
    Uses multi-peak cost analysis and pair moves.
    """
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    N_PEAKS = 4

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        top_peaks = np.argsort(b)[::-1][:N_PEAKS]

        # Compute total cost per coordinate
        cost = np.zeros(n)
        for kp in top_peaks:
            js = max(0, kp - n + 1)
            je = min(n - 1, kp)
            idx = np.arange(js, je + 1)
            cost[idx] += b[kp] * a[idx] * a[kp - idx]

        k_star = int(top_peaks[0])
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx_peak = np.arange(js, je + 1)
        contribs = a[idx_peak] * a[k_star - idx_peak]

        improved = False

        # Strategy 1: Reduce high-cost coordinates, redistribute to low-cost
        top_cost_idx = np.argsort(cost)[::-1][:12]
        for i in top_cost_idx:
            if a[i] < 1e-14:
                continue
            for factor in [0.96, 0.92, 0.88, 0.82, 0.75]:
                diff = a[i] * (1.0 - factor)
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] *= factor
                # Redistribute proportionally to low-cost positions
                low_mask = cost < np.percentile(cost, 40)
                low_mask[i] = False
                if low_mask.sum() > 0:
                    dist = a_trial[low_mask] + 1e-12
                    a_trial[low_mask] = dist + diff * dist / dist.sum()
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
            if improved:
                break

        # Strategy 2: Pair moves on peak-contributing pairs
        if not improved:
            top_contrib = np.argsort(contribs)[::-1][:6]
            for ci in top_contrib:
                i = idx_peak[ci]
                j = k_star - i
                if i == j or i >= n or j >= n or a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.94, 0.89, 0.84]:
                    for f2 in [0.94, 0.89, 0.84]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = norm_sequence(a_trial)
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve > 80:
                rng_local = np.random.default_rng(rnd * 7919 + 3)
                noise = rng_local.standard_normal(n) * 0.004
                a = a + noise
                a = norm_sequence(a)
                no_improve = 0
        else:
            no_improve = 0

    return best_a, best_score


def lbgfs_func(z_in, n, T_val):
    """L-BFGS objective: smoothed max via log-sum-exp."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T_val) * log_sum_exp


def lbgfs_jac(z_in, n, T_val):
    """L-BFGS gradient for smoothed objective."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_refine(a0, time_limit, T_start=20.0, T_end=0.05, maxiter=10000, n_stages=10):
    """L-BFGS with log-sum-exp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=lbgfs_func,
                x0=z,
                jac=lbgfs_jac,
                args=(n, stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def frank_wolfe_refine(a0, time_limit, n_steps=30000):
    """Frank-Wolfe with toward and away steps."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_star = int(np.argmax(b))

        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        g[idx] = 2.0 * a[k_star - idx]

        i_toward = int(np.argmin(g))
        best_trial = None
        best_trial_score = best_score

        for t_val in [0.4, 0.25, 0.15, 0.08, 0.04]:
            a_trial = a.copy()
            a_trial[i_toward] += t_val * (1.0 - a[i_toward])
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            score_trial = 2.0 * n * fast_convolve(a_trial).max()
            if score_trial < best_trial_score:
                best_trial_score = score_trial
                best_trial = a_trial.copy()

        # Away step
        mask = a > 1e-10
        if mask.sum() > 2:
            g_masked = g.copy()
            g_masked[~mask] = -np.inf
            i_away = int(np.argmax(g_masked))
            for gamma in [0.2, 0.1, 0.05]:
                a_trial = a.copy()
                a_trial[i_away] *= (1.0 - gamma)
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                score_trial = 2.0 * n * fast_convolve(a_trial).max()
                if score_trial < best_trial_score:
                    best_trial_score = score_trial
                    best_trial = a_trial.copy()

        if best_trial is not None:
            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 150:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
