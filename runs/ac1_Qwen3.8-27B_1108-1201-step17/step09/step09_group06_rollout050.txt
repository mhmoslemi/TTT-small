def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    """Project vector onto probability simplex (non-negative, sum=1)."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    if solver is None:

    b_val = np.asarray(b_var.value, dtype=np.float64)

def compute_subgradient(a, n_active_frac=0.1):
    """Compute subgradient of max_k (a*a)_k using active set."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b

def lbgfs_logsumexp(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=12):
    """L-BFGS with LogSumExp approximation, multi-stage annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def lse_objective(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        if max_Tb > 700:
            max_Tb = 700
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp

    def lse_jacobian(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        if max_Tb > 700:
            max_Tb = 700
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, 100 // n_stages * 3)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(z_in):
            return lse_objective(z_in, stage_T)

        def jac(z_in):
            return lse_jacobian(z_in, stage_T)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def simulated_annealing(a0, time_limit, T_start=0.1, T_end=0.001, rng=None):
    """Simulated annealing with structured moves for non-convex problem."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    current_score = best_score
    t0 = time.monotonic()
    if rng is None:
        rng = np.random.default_rng(999)

    n_total = max(1000, int(time_limit * 50))
    runs = np.geomspace(T_start, T_end, n_total)

    for it in range(n_total):
        if time.monotonic() - t0 > time_limit:
            break
        T = runs[it]

        move_type = rng.integers(0, 4)

        if move_type == 0:
            # Gaussian perturbation on random subset
            k = rng.integers(max(2, n // 10), n // 2)
            idx = rng.choice(n, k, replace=False)
            a_trial = a.copy()
            a_trial[idx] += rng.normal(0, T * 0.1, k)
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()

        elif move_type == 1:
            # Swap: move mass from high to low contributing position
            b = fast_convolve(a)
            peak_k = int(np.argmax(b))
            contrib = np.zeros(n)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contrib[idx] = a[idx] * a[peak_k - idx]
            src = np.argmax(contrib)
            dst = rng.integers(n)
            if dst == src:
                dst = (dst + 1) % n
            frac = rng.uniform(0.01, 0.1)
            a_trial = a.copy()
            amt = a[src] * frac
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()

        elif move_type == 2:
            # Segment flip (mirror a sub-segment)
            i = rng.integers(0, n - 2)
            j = rng.integers(i + 1, n)
            a_trial = a.copy()
            a_trial[i:j+1] = a_trial[i:j+1][::-1]
            a_trial = a_trial / a_trial.sum()

        else:
            # Scale random segment
            i = rng.integers(0, n - 1)
            j = rng.integers(i + 1, n)
            factor = rng.uniform(0.5, 1.5)
            a_trial = a.copy()
            a_trial[i:j+1] *= factor
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()

        score_trial = 2.0 * n * fast_convolve(a_trial).max()
        delta = score_trial - current_score

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_trial
            current_score = score_trial
            if score_trial < best_score:
                best_score = score_trial
                best_a = a.copy()

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def optimize_at_length(a_init, n, time_budget, solver, label=""):
    """Run multi-phase optimization at a fixed length."""
    best_a = np.maximum(a_init.copy(), 0)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
    best_a = best_a / best_a.sum()
    best_score = evaluate_safe(best_a)
    t0 = time.monotonic()

    def try_update(a, src_label):
        nonlocal best_a, best_score
        a = np.maximum(a, 0).astype(np.float64)
        ss = a.sum()
        if ss < 0.01:
            return
        a = a / ss
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"  NEW BEST [{src_label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase A: Quick mirror descent
    rem = time_limit_global - time.monotonic()
    t_a = min(rem * 0.15, 120.0)
    if t_a > 10:
        a_md, _ = mirror_descent_bt(best_a, t_a, n_active_frac=0.05)
        try_update(a_md, f"md_{label}")

    # Phase B: LP Nesterov (main workhorse)
    rem = time_limit_global - time.monotonic()
    t_b = min(rem * 0.45, 500.0)
    if t_b > 20:
        a_lp, _ = nesterov_lp(best_a, t_b, solver=solver, n_max_iters=200000)
        try_update(a_lp, f"lp_{label}")

    # Phase C: LogSumExp L-BFGS
    rem = time_limit_global - time.monotonic()
    t_c = min(rem * 0.20, 200.0)
    if t_c > 10:
        a_lb = lbgfs_logsumexp(best_a, t_c, T_start=100.0, T_end=0.05, n_stages=12)
        try_update(a_lb, f"lse_{label}")

    # Phase D: Simulated annealing
    rem = time_limit_global - time.monotonic()
    t_d = min(rem * 0.10, 80.0)
    if t_d > 5:
        rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
        a_sa, _ = simulated_annealing(best_a, t_d, T_start=0.05, T_end=0.001, rng=rng)
        try_update(a_sa, f"sa_{label}")

    # Phase E: Peak equalization
    rem = time_limit_global - time.monotonic()
    t_e = min(rem * 0.10, 60.0)
    if t_e > 5:
        a_pe, _ = peak_equalization(best_a, t_e, n_steps=30000)
        try_update(a_pe, f"pe_{label}")

    return best_a, best_score


# Global time limit for internal tracking
time_limit_global = 1750.0


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    global time_limit_global
    rng = np.random.default_rng(seed)
    time_limit_global = time.monotonic() + max(1.0, float(budget_s) - 50.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 960

    if parent is None:
        n_base = 960
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  GLOBAL NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # Phase 1: Optimize at base length
    print(f"\n=== Phase 1: Optimize at n={n_base} (best={best_score:.6f}) ===")
    rem = time_limit_global - time.monotonic()
    t1 = min(rem * 0.40, 600.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_out, _ = optimize_at_length(best_seq, n_base, t1, solver, label=f"n{n_base}")
        try_update(a_out, f"opt_n{n_base}")

    # Phase 2: Try longer lengths
    print(f"\n=== Phase 2: Longer lengths (best={best_score:.6f}) ===")
    longer_lengths = []
    for mult in [1.25, 1.5, 2.0]:
        ln = int(n_base * mult)
        if ln > n_base and ln <= 4000:
            longer_lengths.append(ln)
    longer_lengths = sorted(set(longer_lengths))[:3]

    for ln in longer_lengths:
        rem = time_limit_global - time.monotonic()
        if rem < 120:
            break
        t_alloc = min(rem * 0.25, 300.0)
        if t_alloc < 60:
            break
        print(f"  Trying n={ln}, budget={t_alloc:.0f}s")

        # Initialize by interpolating best_seq
        init_long = interpolate_sequence(best_seq, ln)
        sc_init = evaluate_safe(init_long)
        cur = init_long
        if sc_init < best_score:
            best_score = sc_init
            best_seq = init_long.copy()

        a_out, _ = optimize_at_length(cur, ln, t_alloc, solver, label=f"n{ln}")
        try_update(a_out, f"opt_long_n{ln}")

        # If this length is better, keep it as the new best
        if a_out is not None and evaluate_safe(a_out) <= best_score:
            # Continue optimizing this length
            rem2 = time_limit_global - time.monotonic()
            if rem2 > 60:
                t2 = min(rem2 * 0.2, 150.0)
                if t2 > 30:
                    a_out2, _ = nesterov_lp(a_out, t2, solver=solver, n_max_iters=100000)
                    try_update(a_out2, f"lp_refine_n{ln}")

    # Phase 3: Try shorter lengths
    print(f"\n=== Phase 3: Shorter lengths (best={best_score:.6f}) ===")
    shorter_lengths = []
    for mult in [0.7, 0.5]:
        sn = int(n_base * mult)
        if sn < n_base and sn >= 200:
            shorter_lengths.append(sn)
    shorter_lengths = sorted(set(shorter_lengths))[:2]

    for sn in shorter_lengths:
        rem = time_limit_global - time.monotonic()
        if rem < 90:
            break
        t_alloc = min(rem * 0.15, 120.0)
        if t_alloc < 30:
            break
        print(f"  Trying n={sn}, budget={t_alloc:.0f}s")
        init_short = interpolate_sequence(best_seq, sn)
        a_out, _ = optimize_at_length(init_short, sn, t_alloc, solver, label=f"n{sn}")
        try_update(a_out, f"opt_short_n{sn}")

    # Phase 4: Final polish on the best
    print(f"\n=== Phase 4: Final Polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = time_limit_global - time.monotonic()
    if rem > 30:
        # More LP iterations
        t_lp = min(rem * 0.35, 120.0)
        if t_lp > 10:
            a_fin, _ = nesterov_lp(best_seq, t_lp, solver=solver, n_max_iters=50000)
            try_update(a_fin, "final_lp")

        # LogSumExp fine-tune
        rem = time_limit_global - time.monotonic()
        if rem > 20:
            t_lse = min(rem * 0.3, 80.0)
            if t_lse > 10:
                a_fin2 = lbgfs_logsumexp(best_seq, t_lse, T_start=5.0, T_end=0.02, n_stages=10)
                try_update(a_fin2, "final_lse")

        # Peak equalization
        rem = time_limit_global - time.monotonic()
        if rem > 15:
            t_pe = min(rem * 0.25, 50.0)
            if t_pe > 5:
                a_fin3, _ = peak_equalization(best_seq, t_pe, n_steps=20000)
                try_update(a_fin3, "final_pe")

        # One more LP pass
        rem = time_limit_global - time.monotonic()
        if rem > 10:
            t_lp2 = min(rem * 0.4, 30.0)
            if t_lp2 > 5:
                a_fin4, _ = nesterov_lp(best_seq, t_lp2, solver=solver, n_max_iters=20000)
                try_update(a_fin4, "final_lp2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
