2. LP-based block coordinate descent with peak-directed active sets

import numpy as np
import cvxpy as cp
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 for non-negative a."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    an = a / s
    return 2.0 * len(an) * fast_convolve(an).max()


def p_norm_value_and_grad_z(z, p):
    """Compute value and gradient of 2n*(sum c^p)^(1/p) w.r.t. z (softmax param)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    val = 2.0 * n * S ** (1.0 / p)
    if S < 1e-300:
        return val, np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 2.0 * factor * corr
    g_z = a * (g_a - np.dot(g_a, a))
    g_z *= 2.0 * n
    return val, g_z


def p_norm_lbfgs(a0, time_limit, p_values, iters_per_p=200):
    """L-BFGS with schedule of p values (increasing)."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_of(a0)

    for p in p_values:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_val):
            v, _ = p_norm_value_and_grad_z(z_val, p)
            return v

        def jac(z_val):
            _, g = p_norm_value_and_grad_z(z_val, p)
            return g

        max_it = min(iters_per_p, int(remaining * 50))
        max_it = max(max_it, 50)
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_it, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': max_it * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = score_of(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
        elapsed = time.monotonic() - t0
        print(f"    p-norm p={p:.0f} ({elapsed:.0f}s): score={score_of(a_cur):.6f}, best={best_score:.6f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    sc = score_of(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


_lp_solver_name = None


def get_lp_solver():
    global _lp_solver_name
    if _lp_solver_name is not None:
        return _lp_solver_name
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_name = solver_name
                return _lp_solver_name
        except Exception:
            continue
    _lp_solver_name = cp.ECOS
    return _lp_solver_name


def build_conv_matrix(a):
    """Build convolution matrix T where T @ b = convolve(a, b). Returns (2n-1, n)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_stall_score = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = score_of(a)
        score_y = score_of(y)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 300 == 0:
            if best_score < prev_stall_score - 1e-8:
                stall_count = 0
                prev_stall_score = best_score
            else:
                stall_count += 1
            if stall_count >= 3:
                rng = np.random.default_rng(it + 123)
                noise_scale = 0.02 + 0.01 * stall_count
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0
                prev_stall_score = best_score

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def local_refinement_v2(a0, time_limit, n_steps=50000):
    """Improved local refinement with line search and multi-peak targeting."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf

        peak_val = b_work.max()
        if peak_val <= 0:
            break
        near_threshold = peak_val - 0.03 * peak_val
        near_peaks = np.where(b_work >= near_threshold)[0][:10]

        # Compute contribution scores for each coordinate
        contrib_scores = np.zeros(n)
        for k in near_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) > 0:
                contribs = a[idx] * a[k - idx]
                contrib_scores[idx] += contribs

        top_coords = np.argsort(contrib_scores)[::-1][:20]
        improved = False

        # Strategy 1: Single coordinate reduction with line search
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            # Binary search for best reduction factor
            lo, hi = 0.70, 1.0
            best_f = 1.0
            best_sc = best_score
            for _ in range(8):
                mid = (lo + hi) / 2
                a_trial = a.copy()
                diff = a[i] * (1 - mid)
                a_trial[i] = a[i] * mid
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] += diff * a_trial[mask] / others_sum
                sc = fast_convolve(a_trial / a_trial.sum()).max() * 2 * n
                if sc < best_sc:
                    best_sc = sc
                    best_f = mid
                    lo = mid
                else:
                    hi = mid
            if best_f < 1.0:
                diff = a[i] * (1 - best_f)
                a[i] = a[i] * best_f
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a[mask].sum()
                if others_sum > 1e-15:
                    a[mask] += diff * a[mask] / others_sum
                a = a / a.sum()
                best_score = best_sc
                best_a = a.copy()
                improved = True
                break

        # Strategy 2: Pair adjustment at top peak
        if not improved:
            peak_k = int(near_peaks[0])
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_pairs = np.argsort(contribs)[::-1][:8]
            for ci in top_pairs:
                i = int(idx[ci])
                j = peak_k - i
                if i == j or j >= n or j < 0:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                # Try reducing both
                for f1 in [0.93, 0.88, 0.82, 0.75]:
                    for f2 in [0.93, 0.88, 0.82, 0.75]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        st = a_trial.sum()
                        if st < 0.01:
                            continue
                        a_trial = a_trial / st
                        sc = 2.0 * n * fast_convolve(a_trial).max()
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Strategy 3: Redistribute mass from high-contribution to low-contribution
        if not improved and step % 50 == 0:
            src_i = top_coords[0] if len(top_coords) > 0 else None
            if src_i is not None and a[src_i] > 1e-10:
                # Find low-contribution coordinate to receive mass
                low_coords = np.argsort(contrib_scores)[:10]
                for dst_i in low_coords:
                    if dst_i == src_i:
                        continue
                    for frac in [0.01, 0.03, 0.05, 0.10]:
                        amount = a[src_i] * frac
                        a_trial = a.copy()
                        a_trial[src_i] -= amount
                        a_trial[dst_i] += amount
                        if a_trial[src_i] < 0:
                            continue
                        st = a_trial.sum()
                        if st < 0.01:
                            continue
                        a_trial = a_trial / st
                        sc = 2.0 * n * fast_convolve(a_trial).max()
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break

        # Strategy 4: Small random perturbation
        if not improved and step % 200 == 0:
            rng = np.random.default_rng(step + 31)
            n_pert = int(rng.integers(2, 6))
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for idx in indices:
                a_trial[idx] *= (1.0 + rng.uniform(-0.04, 0.04))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved and step > 100:
            # Try larger perturbation to escape
            if step % 500 == 0:
                rng = np.random.default_rng(step + 77)
                a_trial = a + 0.05 * rng.standard_normal(n)
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 0.01:
                    a_trial = a_trial / st
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score + 0.01:
                        a = a_trial
                        improved = True
            elif step > 500:
                break

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    LocalRefine step={step} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def symmetric_local_refine(a0, time_limit):
    """Local refinement exploiting symmetry: only modify first half, mirror to second."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    half = (n + 1) // 2

    # Work with the "effective" first half
    # If n is odd: indices 0..(n-1)//2 are free, rest are mirrors
    # If n is even: indices 0..n//2-1 are free, index n//2 is fixed (center)

    for step in range(20000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        if b[peak_k] <= 0:
            break

        # Find which "free" indices contribute to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Only consider indices in the first half (or their mirrors)
        free_idx = []
        for i in idx:
            mirror = n - 1 - i
            rep = min(i, mirror)
            if rep < half:
                free_idx.append(rep)
        if len(free_idx) == 0:
            break

        free_idx = np.array(free_idx)
        # Sum contributions for each free index
        free_contrib = np.zeros(half)
        for pos, fi in enumerate(free_idx):
            orig_i = idx[pos]
            c = a[orig_i] * a[peak_k - orig_i]
            # Also count mirror contribution
            mirror = n - 1 - orig_i
            c2 = a[mirror] * a[peak_k - mirror]
            free_contrib[fi] += c + c2

        top_free = np.argsort(free_contrib)[::-1][:5]
        improved = False

        for fi in top_free:
            if a[fi] < 1e-14:
                continue
            for f in [0.92, 0.85, 0.78, 0.70]:
                a_trial = a.copy()
                mirror = n - 1 - fi
                diff_fi = a_trial[fi] * (1 - f)
                a_trial[fi] *= f
                if mirror != fi:
                    diff_m = a_trial[mirror] * (1 - f)
                    a_trial[mirror] *= f
                    diff_fi += diff_m
                mask = np.ones(n, dtype=bool)
                mask[fi] = False
                if mirror != fi:
                    mask[mirror] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff_fi * a_trial[mask] / others_sum
                sc = 2.0 * n * fast_convolve(a_trial / a_trial.sum()).max()
                if sc < best_score - 1e-15:
                    a = a_trial / a_trial.sum()
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try asymmetric adjustment
            if step % 100 == 0:
                rng = np.random.default_rng(step + 5)
                fi = int(rng.integers(0, half))
                a_trial = a.copy()
                a_trial[fi] *= (1.0 + rng.uniform(-0.05, 0.02))
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 0.01:
                    a_trial = a_trial / st
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True

            if not improved and step > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_state, a, label):
    """Try to update best state."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False


def block_lp_refine(a0, time_limit, block_size=64, solver=None):
    """Block coordinate descent with LP: optimize one block at a time."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    n_blocks = int(np.ceil(n / block_size))
    rng = np.random.default_rng(42)

    iteration = 0
    while time.monotonic() - t0 < time_limit:
        # Random block order
        block_order = rng.permutation(n_blocks)
        improved_this_pass = False

        for bi in block_order:
            if time.monotonic() - t0 > time_limit:
                break
            start = bi * block_size
            end = min(start + block_size, n)
            if start >= n:
                continue

            # Fix all coordinates except the block
            # Solve LP: min M s.t. T_b @ b_block + T_rest @ a_rest <= M, sum(b_block) = 1 - sum(a_rest), b_block >= 0
            a_rest = a.copy()
            a_rest[start:end] = 0
            rest_sum = a_rest.sum()
            block_sum = 1.0 - rest_sum
            if block_sum < 1e-10:
                continue

            b_var = cp.Variable(end - start, nonneg=True)
            M_var = cp.Variable()
            T_block = build_conv_matrix(np.ones(end - start))  # placeholder
            # Actually we need the correct convolution matrix for the block interaction
            # conv(a, b) where a is fixed except for the block
            # This is complex; let's just use the full LP for simplicity
            # Instead, do a projected gradient step on the block

            # Simpler approach: use the gradient to update the block
            b_full = fast_convolve(a)
            c = np.maximum(b_full, 1e-300)
            p = 50.0
            w = c ** (p - 1)
            S = np.sum(c ** p)
            factor = S ** (1.0 / p - 1.0)
            corr = np.correlate(w, a, mode='valid')
            g = 2.0 * factor * corr

            # Line search on the block
            a_trial = a.copy()
            for lr in [0.01, 0.02, 0.05, 0.1]:
                a_trial = a.copy()
                block_grad = g[start:end]
                a_trial[start:end] -= lr * block_grad
                a_trial[start:end] = np.maximum(a_trial[start:end], 0)
                st = a_trial.sum()
                if st < 0.01:
                    continue
                a_trial = a_trial / st
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved_this_pass = True
                    break

        iteration += 1
        if (iteration + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BlockLP iter={iteration} ({elapsed:.0f}s): best={best_score:.6f}")

        if not improved_this_pass and iteration > 5:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Load parent, quick p-norm LBFGS with high p values
    2. LP-based Nesterov refinement
    3. Local refinement v2
    4. Symmetric refinement
    5. Length exploration
    6. Final high-p polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_state = {'score': float('inf'), 'seq': None}

    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        x = np.linspace(-1, 1, n_init)
        parent = np.exp(-x ** 2 * 3)
        parent = parent / parent.sum()
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.6f}")

    # Phase 1: p-norm L-BFGS with aggressive schedule
    print(f"\n=== Phase 1: p-norm LBFGS (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        p_schedule = [20, 40, 80, 150, 300, 600, 1200, 2500, 5000]
        iters = max(100, int(t_alloc / len(p_schedule) * 30))
        a_pn, sc = p_norm_lbfgs(best_state['seq'], t_alloc, p_schedule, iters_per_p=iters)
        try_update(best_state, a_pn, "pn_main")

    # Phase 2: Nesterov LP
    print(f"\n=== Phase 2: Nesterov LP (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc = nesterov_lp_refine(best_state['seq'], t_alloc, solver=solver, n_max_iters=100000)
        try_update(best_state, a_lp, "nesterov_lp")

    # Phase 3: Local refinement v2
    print(f"\n=== Phase 3: Local Refine v2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lr, sc = local_refinement_v2(best_state['seq'], t_alloc, n_steps=100000)
        try_update(best_state, a_lr, "local_v2")

    # Phase 4: Symmetric refinement
    print(f"\n=== Phase 4: Symmetric Refine (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sym, sc = symmetric_local_refine(best_state['seq'], t_alloc)
        try_update(best_state, a_sym, "symmetric")

    # Phase 5: p-norm L-BFGS round 2 (very high p)
    print(f"\n=== Phase 5: High-p LBFGS (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        p_schedule2 = [500, 1000, 2000, 4000, 8000]
        iters2 = max(80, int(t_alloc / len(p_schedule2) * 20))
        a_pn2, sc = p_norm_lbfgs(best_state['seq'], t_alloc, p_schedule2, iters_per_p=iters2)
        try_update(best_state, a_pn2, "pn_high")

    # Phase 6: Nesterov LP round 2
    print(f"\n=== Phase 6: Nesterov LP r2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, sc = nesterov_lp_refine(best_state['seq'], t_alloc, solver=solver, n_max_iters=60000)
        try_update(best_state, a_lp2, "nesterov_lp2")

    # Phase 7: Length exploration
    print(f"\n=== Phase 7: Length Exploration (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    n_best = len(best_state['seq'])
    candidate_lengths = set()
    for delta in [-48, -32, -16, 16, 32, 48, 64, 96]:
        alt_n = n_best + delta
        if 512 <= alt_n <= 2048:
            candidate_lengths.add(alt_n)

    candidate_lengths = sorted(candidate_lengths, key=lambda x: abs(x - n_best))
    n_explored = 0
    for alt_n in candidate_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 100 or n_explored >= 2:
            break
        n_explored += 1
        t_total = min(rem * 0.20, 200.0)
        if t_total < 50:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_state['seq'], alt_n)
        p_sched = [50, 100, 200, 500, 1000]
        iters_alt = max(60, int(t_total * 0.5 / len(p_sched) * 20))
        a_alt, sc = p_norm_lbfgs(init_alt, t_total * 0.5, p_sched, iters_per_p=iters_alt)
        improved = try_update(best_state, a_alt, f"len_{alt_n}")

        if improved and rem > 30:
            t_lp = min(rem * 0.15, 100.0)
            a_alt2, _ = nesterov_lp_refine(best_state['seq'], t_lp, solver=solver, n_max_iters=20000)
            try_update(best_state, a_alt2, f"lp_{alt_n}")

    # Phase 8: Perturbation + re-optimization
    print(f"\n=== Phase 8: Perturbation (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for noise_scale in [0.02, 0.05, 0.10]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            n_cur = len(best_state['seq'])
            perturbed = best_state['seq'] + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            t_sub = min(rem * 0.25, 80.0)
            p_sched_p = [100, 300, 800, 2000]
            a_pert, sc = p_norm_lbfgs(perturbed, t_sub, p_sched_p, iters_per_p=100)
            try_update(best_state, a_pert, f"pert_{noise_scale}")

    # Phase 9: Final polish
    print(f"\n=== Phase 9: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.30, 80.0)
        a_fin, sc = local_refinement_v2(best_state['seq'], t_fin, n_steps=50000)
        try_update(best_state, a_fin, "final_local")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_p = min(rem * 0.40, 60.0)
        p_final = [2000, 5000, 10000]
        a_pf, sc = p_norm_lbfgs(best_state['seq'], t_p, p_final, iters_per_p=100)
        try_update(best_state, a_pf, "final_pnorm")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_lp_fin = min(rem * 0.50, 40.0)
        a_lpf, _ = nesterov_lp_refine(best_state['seq'], t_lp_fin, solver=solver, n_max_iters=10000)
        try_update(best_state, a_lpf, "final_lp")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    print(f"\nFINAL: score={best_state['score']:.6f}, n={len(best_state['seq'])}")
    return [float(x) for x in best_state['seq']]
