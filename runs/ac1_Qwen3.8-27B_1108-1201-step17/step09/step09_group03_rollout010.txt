"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Alternating LP (exact best-response iteration) for global optimization
2. Fast coordinate descent with adaptive steps for local polishing
3. Multi-peak mirror descent for smooth refinement

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(b):
    """Build matrix T such that T @ a = conv(b, a)."""
    n = len(b)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = b
    return T


def scale_max_M(a, b):
    c = convolve_ab(a, b)
    return float(c.max())


def lp_best_response(b, solver=None):
    """
    Solve: min t s.t. (a*b)_k <= t for all k, a >= 0, sum(a) = 1.
    Returns (a_opt, t_opt).
    """
    n = len(b)
    a = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    T = build_conv_matrix(b)
    constraints = [
        T @ a <= t,
        cp.sum(a) == 1
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=solver)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    a_val = np.asarray(a.value, dtype=np.float64)
    a_val = np.maximum(a_val, 0)
    s = a_val.sum()
    if s < 1e-15:
        return None, None
    a_val = a_val / s
    t_val = max(float(t.value), float(fast_convolve(a_val).max() * b.sum() / b.sum()))
    # Actually t_val should be the max of (a*b), let's just compute it
    t_val = float(convolve_ab(a_val, b).max())
    return a_val, t_val


def alternating_lp(a_init, time_limit, solver=None, verbose=True):
    """
    Alternating LP: solve for a given b, then for b given a, repeat.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    b = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    
    history = [a.copy()]
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP for a given b
        a_new, _ = lp_best_response(b, solver=solver)
        if a_new is None:
            # Fallback: use FW step
            break
        a = a_new
        history.append(a.copy())
        
        # Check score
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Also check the average of recent iterates
        if len(history) >= 3 and it % 3 == 0:
            avg = np.mean(history[-5:], axis=0)
            avg = np.maximum(avg, 0)
            sa = avg.sum()
            if sa > 1e-15:
                avg = avg / sa
                score_avg = 2.0 * n * fast_convolve(avg).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg.copy()
        
        if time.monotonic() - t0 >= time_limit:
            break
        
        # Solve LP for b given a
        b_new, _ = lp_best_response(a, solver=solver)
        if b_new is None:
            break
        b = b_new
        history.append(b.copy())
        
        # Check score
        score_b = 2.0 * n * fast_convolve(b).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b.copy()
        
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AltLP it={it} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

        a = a / s
    return a

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Alternating LP via scipy HiGHS (exact best-response iteration) - main workhorse
2. Averaging of iterates for convergence improvement
3. Random restarts to escape local minima

5. Fine-grained coordinate descent for last-mile improvements
6. Diverse initialization strategies

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(b):
    """Build matrix T such that T @ a = conv(b, a) for a of same length."""
    n = len(b)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = b
    return T


def lp_best_response(b, method='highs'):
    """
    Solve: min t s.t. (a*b)_k <= t for all k, a >= 0, sum(a) = 1.
    Uses scipy linprog with HiGHS.
    Returns a_opt or None.
    """
    n = len(b)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    T = build_conv_matrix(b)
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0

    a_val = np.maximum(a_val, 0)
    s = a_val.sum()
    if s < 1e-12:
        return None
    a_val = a_val / s

        # Periodic restart: add noise to escape local minima
        if it > 20 and it % restart_interval == 0:
            noise = np.random.standard_normal(n) * noise_level * a.mean()
            a_noisy = a + noise
            a_noisy = np.maximum(a_noisy, 0)
            sn = a_noisy.sum()
            if sn > 1e-12:
                a_noisy = a_noisy / sn
                score_noisy = 2.0 * n * fast_convolve(a_noisy).max()
                if score_noisy < best_score * 1.05:  # Only restart if not too bad
                    b = a_noisy.copy()
        
        # Solve LP for a given b
        a_new = lp_best_response(b)
        if a_new is None:
            break
        a = a_new
        
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        if time.monotonic() - t0 >= time_limit:
            break
        
        # Solve LP for b given a
        b_new = lp_best_response(a)
        if b_new is None:
            break
        b = b_new
        
        score_b = 2.0 * n * fast_convolve(b).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b.copy()
        
        # Also try the average
        if it >= 4:
            avg = 0.5 * (a + b)
            avg = np.maximum(avg, 0)
            sa = avg.sum()
            if sa > 1e-12:
                avg = avg / sa
                score_avg = 2.0 * n * fast_convolve(avg).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg.copy()
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AltLP it={it} ({elapsed:.0f}s): best={best_score:.6f} a_score={score_a:.6f} b_score={score_b:.6f}")
    
    return best_a, best_score


def coord_descent_v2(a0, time_limit, verbose=True, label=""):
    """Coordinate descent with prefix/suffix max for O(n) per coordinate."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        # Compute prefix max and suffix max
        c_len = 2 * n - 1
        prefix_max = np.full(c_len, -np.inf)
        for i in range(1, c_len):
            prefix_max[i] = max(prefix_max[i-1], conv[i-1])
        suffix_max = np.full(c_len, -np.inf)
        for i in range(c_len - 2, -1, -1):
            suffix_max[i] = max(suffix_max[i+1], conv[i+1])
        
        for i in range(n):
            if a[i] < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            # Unaffected max
            unaff_max = -np.inf
            if k_start > 0:
                unaff_max = max(unaff_max, prefix_max[k_start])
            if k_end + 1 < c_len:
                unaff_max = max(unaff_max, suffix_max[k_end + 1])
            
            a_shifted = a[k_start - i : k_end - i + 1]
            seg = conv[k_start:k_end + 1]
            
            best_delta = 0.0
            best_new_M = M
            
            base = a[i]
            # Multiple step sizes
            for factor in [-1.0, -0.7, -0.4, -0.2, -0.1, 0.05, 0.1, 0.2, 0.4, 0.7, 1.0]:
                delta = base * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                # Compute new affected segment
                new_seg = seg + 2.0 * delta * a_shifted
                # Fix quadratic term
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_seg = new_seg.copy()
                    new_seg[idx] = conv[2*i] + 2.0 * delta * a[i] + delta * delta
                
                new_M = max(unaff_max, float(new_seg.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end+1] += changes
                M = max(unaff_max, float(conv[k_start:k_end+1].max()))
                
                # Update prefix/suffix (approximate)
                # We'll just recompute at start of next sweep
                for k in range(k_start, k_end + 1):
                    if k > 0:
                        prefix_max[k] = max(prefix_max[k-1], conv[k-1])
                    if k < c_len - 1:
                        suffix_max[k] = max(suffix_max[k+1], conv[k+1])
                
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score
