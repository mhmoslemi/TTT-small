def pnorm_obj(z, p, n):
    """2*n*(sum c_k^p)^(1/p) where a=softmax(z), c=conv(a,a)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * n * val


def pnorm_grad(z, p, n):
    """Gradient of pnorm_obj w.r.t. z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))

def subgradient_polish(a0, time_limit, n_max_iter=1000000):
    """Subgradient method with heavy-ball momentum for the max-conv objective."""
    n = len(a0)
    a = normalize(a0)
    vel = np.zeros(n)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    momentum = 0.0

    for t in range(1, n_max_iter + 1):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        k_star = np.argmax(c)
        M_val = c[k_star]

        # Subgradient: d/d(a_j) conv(a,a)[k_star]
        g = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:
                g[j] = 2.0 * a[kj]
            # Handle the symmetric case where j = k_star - j
            # Already counted if kj == j, that's fine (it's 2*a[j] which is correct for a[j]^2)

        # Heavy ball momentum
        momentum = 0.9 * momentum + 0.1 * (1.0 / np.sqrt(t))
        step = 0.5 / np.sqrt(t + 1)
        a_new = a - step * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        else:
            a_new = a.copy()

        # Momentum update
        vel = momentum * vel + (1 - momentum) * (a_new - a)
        a = np.maximum(a + vel, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s

        sc = compute_score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()

        if t % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Subgrad t={t} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def active_set_newton(a0, time_limit, tol=1e-10):
    """Identify active constraints and solve the KKT system via Newton's method."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for newton_iter in range(50):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        M = c.max()
        # Active set: peaks within relative tolerance of max
        rel_tol = 1e-6
        active = np.where(c >= M * (1 - rel_tol))[0]
        # Also include near-absolute peaks
        abs_tol = 0.02 * M
        active = np.where(c >= M - abs_tol)[0]

        if len(active) == 0:
            break

        # Build the nonlinear system: F(a, M) = 0
        # F_k = conv(a,a)[k] - M = 0 for k in active
        # F_sum = sum(a) - 1 = 0
        # Unknowns: a (n vars) + M (1 var) = n+1 unknowns
        # Equations: len(active) + 1

        m = len(active)
        if m + 1 > n + 1:
            # Too many active constraints, reduce
            active = active[:n]
            m = len(active)

        # Jacobian: J[i, j] for the conv constraints
        # d(conv(a,a)[k])/d(a_j) = 2*a[k-j] if 0<=k-j<n (and j!=k-j)
        #                            = a[k-j] if j==k-j (i.e., k=2j)
        #                             = 0 otherwise
        # d(conv(a,a)[k])/d(M) = -1
        # d(sum(a))/d(a_j) = 1
        # d(sum(a))/d(M) = 0

        # Solve using least squares (underdetermined if m < n)
        for inner_iter in range(20):
            if time.monotonic() - t0 > time_limit:
                break

            c = fast_convolve(a)
            M = c.max()
            residuals = c[active] - M
            residuals = np.append(residuals, a.sum() - 1.0)

            if np.max(np.abs(residuals)) < tol:
                break

            # Build Jacobian (m+1) x (n+1)
            J = np.zeros((m + 1, n + 1))
            for ii, k in enumerate(active):
                for j in range(n):
                    kj = k - j
                    if 0 <= kj < n:
                        if j == kj:
                            J[ii, j] = a[kj]  # Wait, d(a_j*a_j)/da_j = 2*a_j
                            # Actually d/d(a_j)(a_j * a_{k-j}) when j=k-j: d(a_j^2)/da_j = 2*a_j
                            # But in the sum sum_i a_i * a_{k-i}, the term with i=j is a_j*a_{k-j}=a_j^2
                            # AND the term with i=k-j=j is the same term. So we count it once.
                            # The derivative is 2*a_j.
                            J[ii, j] = 2.0 * a[j]
                        else:
                            J[ii, j] = 2.0 * a[kj]
                J[ii, n] = -1.0  # d/dM of (conv[k] - M) = -1
            # Last row: sum(a) - 1 = 0
            J[m, :n] = 1.0
            J[m, n] = 0.0

            # Newton step: solve J @ [da; dM] = -residuals
            # Underdetermined: use minimum norm solution
            try:
                delta, _, _, _ = np.linalg.lstsq(J, -residuals, rcond=None)
            except Exception:
                break

            da = delta[:n]
            dM = delta[n]

            # Damped step
            for lam in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
                a_trial = a + lam * da
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                sc_trial = compute_score(aTrial)
                if sc_trial <= best_score * (1 + 1e-10):
                    a = a_trial
                    break
            else:
                break

        sc = compute_score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()

        if time.monotonic() - t0 > time_limit:
            break

    return best_a, best_score


def flat_spectrum_init(n, rng):
    """Initialize using flat spectrum (random phase) sequence."""
    fft_size = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0, 2 * np.pi, fft_size)
    # Make symmetric for real result
    for k in range(1, fft_size // 2):
        phases[fft_size - k] = -phases[k]
    spectrum = np.exp(1j * phases)
    if fft_size % 2 == 0:
        spectrum[fft_size // 2] = 1.0
    spectrum[0] = 1.0
    a = np.real(irfft(spectrum, fft_size))[:n]
    a = np.maximum(a, 0)
    return normalize(a)


def structured_init(n, variant=0, rng=None):
    """Create structured initial sequence."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = (1 + np.cos(2 * np.pi * x)) / 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def coordinate_descent_polish(a0, time_limit, n_steps=50000):
    """Coordinate descent: reduce peaks by shrinking contributing coordinates."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        M = c.max()
        # Find top few peaks
        peak_thresh = M * 0.95
        peak_indices = np.where(c >= peak_thresh)[0]

        if len(peak_indices) == 0:
            break

        improved = False

        for k in peak_indices[:5]:
            # Find coordinates contributing to this peak
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            if len(idx) == 0:
                continue
            contribs = a[idx] * a[k - idx]
            top_i = idx[np.argmax(contribs)]

            # Try reducing this coordinate
            for factor in [0.95, 0.90, 0.85, 0.80, 0.70]:
                a_trial = a.copy()
                a_trial[top_i] *= factor
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved and step % 500 == 0:
            # Small random perturbation to escape
            rng = np.random.default_rng(step)
            n_pert = rng.integers(2, 6)
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in indices:
                a_trial[i] *= (1.0 + rng.uniform(-0.03, 0.03))
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial = a_trial / s
                sc = compute_score(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()

        if not improved and step % 2000 == 1999:
            # Check if we should stop (no progress in many steps)
            if step > 5000:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining Nesterov LP game solving, p-norm L-BFGS,
    subgradient polishing, active set Newton, coordinate descent, and
    multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)
    t_start = time.monotonic()

    best_state = {'score': float('inf'), 'seq': None}

    solver = get_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = structured_init(n_init, variant=3, rng=rng)
        best_state['seq'] = parent.copy()
        best_state['score'] = compute_score(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.6f}")

    def try_update(a, label):
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return False
        a = a / s
        sc = eval_safe(a)
        if np.isfinite(sc) and sc < best_state['score']:
            best_state['score'] = sc
            best_state['seq'] = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
            return True
        return False

    n_best = len(best_state['seq'])

    # === Phase 1: p-norm L-BFGS (main workhorse) ===
    print(f"\n=== Phase 1: p-norm L-BFGS n={n_best} (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pn = pnorm_lbfgs(best_state['seq'], t_alloc, p_start=15.0, p_end=400.0, n_stages=10)
        try_update(a_pn, "pnorm_1")

    # === Phase 2: Nesterov LP game ===
    print(f"\n=== Phase 2: Nesterov LP (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 540.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_np, _ = nesterov_game(best_state['seq'], t_alloc, solver, n_iters_max=500000)
        try_update(a_np, "nesterov_1")

    # === Phase 3: p-norm L-BFGS round 2 (higher p) ===
    print(f"\n=== Phase 3: p-norm L-BFGS r2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 270.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pn2 = pnorm_lbfgs(best_state['seq'], t_alloc, p_start=100.0, p_end=800.0, n_stages=6)
        try_update(a_pn2, "pnorm_2")

    # === Phase 4: Subgradient polish ===
    print(f"\n=== Phase 4: Subgradient polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 180.0)
    if t_alloc > 10:
        a_sg, _ = subgradient_polish(best_state['seq'], t_alloc)
        try_update(a_sg, "subgrad")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: Coord descent (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 180.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_descent_polish(best_state['seq'], t_alloc)
        try_update(a_cd, "coord_descent")

    # === Phase 6: Nesterov LP round 2 (from improved start) ===
    print(f"\n=== Phase 6: Nesterov LP r2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 216.0)
    if t_alloc > 20:
        a_np2, _ = nesterov_game(best_state['seq'], t_alloc, solver, n_iters_max=300000)
        try_update(a_np2, "nesterov_2")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Length exploration (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        candidate_lengths = set()
        base = n_best
        for delta in [-128, -64, -32, 32, 64, 128, 192]:
            alt_n = base + delta
            if 512 <= alt_n <= 2048:
                candidate_lengths.add(alt_n)
        for rn in [768, 832, 896, 960, 1024, 1088, 1152, 1216, 1280]:
            if 512 <= rn <= 2048:
                candidate_lengths.add(rn)

        candidate_lengths = sorted(candidate_lengths, key=lambda x: abs(x - base))
        n_explored = 0
        for alt_n in candidate_lengths[:5]:
            rem = deadline - time.monotonic()
            if rem < 60 or n_explored >= 4:
                break
            n_explored += 1
            t_total = min(rem * 0.18, 150.0)
            if t_total < 30:
                break
            print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

            init_alt = interpolate_sequence(best_state['seq'], alt_n)

            t_pn = t_total * 0.6
            a_alt = pnorm_lbfgs(init_alt, t_pn, p_start=30.0, p_end=300.0, n_stages=5)
            improved = try_update(a_alt, f"pn_n{alt_n}")

            if improved and rem > 30:
                rem = deadline - time.monotonic()
                t_lp = min(rem * 0.2, 80.0)
                a_alt2, _ = nesterov_game(best_state['seq'], t_lp, solver, n_iters_max=50000)
                try_update(a_alt2, f"lp_n{alt_n}")

    # === Phase 8: Perturbation + re-optimization ===
    print(f"\n=== Phase 8: Perturbation (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_state['seq'])
        for noise_scale in [0.02, 0.05, 0.10]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            perturbed = best_state['seq'] + noise_scale * rng.standard_normal(n_cur)
            perturbed = normalize(perturbed)
            t_sub = min(rem * 0.3, 60.0)
            a_pert = pnorm_lbfgs(perturbed, t_sub, p_start=50.0, p_end=400.0, n_stages=5)
            try_update(a_pert, f"pert_{noise_scale}")

    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.30, 50.0)
        a_fin = pnorm_lbfgs(best_state['seq'], t_fin, p_start=300.0, p_end=1000.0, n_stages=4)
        try_update(a_fin, "final_pnorm")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_cd = min(rem * 0.30, 30.0)
        a_cd2, _ = coordinate_descent_polish(best_state['seq'], t_cd)
        try_update(a_cd2, "final_cd")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_nd = min(rem * 0.40, 20.0)
        a_nd, _ = subgradient_polish(best_state['seq'], t_nd)
        try_update(a_nd, "final_subgrad")

    # Final verification
    if best_state['seq'] is not None:
        fs = eval_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    elapsed_total = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])}, time={elapsed_total:.0f}s)")

    if best_state['seq'] is not None:
        return [float(x) for x in best_state['seq']]
    return [1.0]
