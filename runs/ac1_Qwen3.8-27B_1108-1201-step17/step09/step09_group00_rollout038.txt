def fast_conv(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr(a, y):
    """Compute (a *circ* y)[j] = sum_m a[m] * y[j+m] for j=0..n-1.
    This is the cross-correlation needed for the dual LP evaluation."""
    n = len(a)
    a_rev = a[::-1]
    # (a_rev * y)[n-1+j] = sum_m a_rev[m] * y[n-1+j-m] = sum_m a[n-1-m] * y[n-1+j-m]
    # Let l = n-1-m, then m = n-1-l: sum_l a[l] * y[j+l]
    # So (a_rev * y)[n-1+j] = sum_l a[l] * y[j+l] = (a *circ* y)[j]
    full = np.convolve(a_rev, y, mode='full')
    return full[n - 1:n - 1 + len(a)]


def build_conv_rows(a, active_k):
    """Build rows of convolution matrix for given active k positions.
    Returns matrix C where C[i, j] = a[k_i - j] if 0 <= k_i - j < n, else 0."""
    n = len(a)
    C = np.zeros((len(active_k), n))
    for i, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        C[i, idx] = a[k - idx]
    return C


def mwu_best_response(a, n_iter=200, eta=0.1):
    """Multiplicative Weight Update approximation to the best response LP.
    Given fixed a (sum=1, a>=0), approximate b that minimizes max_k conv(a,b)[k].
    Uses MWU on the dual: max_y min_j (a *circ* y)_j"""
    n = len(a)
    n_y = 2 * n - 1
    w = np.ones(n_y)
    
    for it in range(n_iter):
        y = w / w.sum()
        # Compute (a *circ* y)_j for all j
        ay = cross_corr(a, y)
        # Best response of row player
        j_star = int(np.argmin(ay))
        # Update column player weights
        # P[j*, k] = a[k - j*] if 0 <= k - j* < n
        diff = np.arange(n_y) - j_star
        valid = (diff >= 0) & (diff < n)
        payoff = np.zeros(n_y)
        payoff[valid] = a[diff[valid]]
        w *= np.exp(eta * payoff)
        # Normalize to prevent overflow
        w_max = w.max()
        if w_max > 1e10 or w_max < 1e-10:
            w = w / w_max
    
    y = w / w.sum()
    # Convert dual to primal: b_j proportional to exp(-beta * (a*circ*y)_j)
    ay = cross_corr(a, y)
    beta = 50.0
    b = np.exp(-beta * (ay - ay.min()))
    b = b / b.sum()
    return b

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_conv(a).max()
        score_y = 2.0 * n * fast_conv(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-9:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            
            if stall >= 10:
                noise = rng.standard_normal(n) * 0.02 / np.sqrt(stall)
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                stall = 0
            
            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def kkt_peak_equalize(a0, time_limit, n_steps=50000, active_frac=0.01):
    """KKT peak equalization: directly reduce the spread of top peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    lr = 0.1
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        # Active set: positions within active_frac of max
        threshold = max_b * (1.0 - active_frac)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Compute gradient of peak spread
        # For each active k, gradient of b[k] w.r.t. a is 2*a shifted
        grad = np.zeros(n)
        b_active = b[active]
        b_mean = b_active.mean()
        
        for k in active:
            weight = b[k] - b_mean
            # grad_a b[k] at position m = 2*a[k-m] (if valid)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += weight * 2.0 * a[k - idx]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 1
            if no_improve > 500:
                break
            continue
        
        # Take step
        a_trial = a - lr * grad / grad_norm
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        score = 2.0 * n * fast_conv(a_trial).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr = min(lr * 1.05, 1.0)
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.85
            if no_improve > 2000:
                break
        
        # Adaptively increase active set
        if step % 5000 == 0:
            active_frac = min(active_frac * 1.5, 0.05)
        
        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKT {elapsed:.0f}s: score={best_score:.8f} |S|={len(active)} lr={lr:.4f}")
    
    return best_a, best_score


def kkt_newton(a0, time_limit, n_steps=200):
    """Newton method on the linearized KKT peak-equalization system."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        # Get top peaks
        n_peaks = min(8, 2 * n - 1)
        sorted_idx = np.argsort(b)[::-1]
        # Only include peaks that are close to max
        active = sorted_idx[b[sorted_idx] >= max_b * 0.999][:n_peaks]
        
        if len(active) == 0:
            break
        
        # Build linearized system
        # 2*C @ delta = M*1 - b_active, subject to sum(delta) = 0
        C = build_conv_rows(a, active)
        b_act = b[active]
        M = b_act.max()
        rhs = M - b_act  # What we need to add to each peak
        
        # Solve: [2*C; 1^T] @ delta = [rhs; 0] (least squares)
        A_sys = np.vstack([2.0 * C, np.ones((1, n))])
        b_sys = np.concatenate([rhs, [0.0]])
        
        try:
            delta, _, _, _ = np.linalg.lstsq(A_sys, b_sys, rcond=None)
        except Exception:
            break
        
        # Line search
        best_t = 0.0
        best_sc = best_score
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = a + t * delta
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            sc = 2.0 * n * fast_conv(a_trial).max()
            if sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_t > 0:
            a = a + best_t * delta
            a = np.maximum(a, 0)
            a = a / a.sum()
            if best_sc < best_score:
                best_score = best_sc
                best_a = a.copy()
        
        if step % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKT-N {elapsed:.0f}s: score={best_score:.8f} |S|={len(active)}")
    
    return best_a, best_score


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_and_grad(z, n, T):
    """Compute LogSumExp objective and gradient in log-domain."""
    a = softmax_from_z(z)
    b = fast_conv(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    W = exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(W)
    
    # Gradient: 4*n * sum_k (w_k/W) * a[k-m] for each m
    w = exp_Tb / W
    grad_a = np.zeros(n)
    for k in range(len(b)):
        wk = w[k]
        if wk < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += wk * a[k - idx]
    grad_a *= 4.0 * n / T
    
    # Convert to log-domain gradient
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return (2.0 * n / T) * log_sum_exp, grad_z


def lbgfs_logsumexp_refine(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS on LogSumExp with decreasing temperature."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02]
    
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = logsumexp_obj_and_grad(z, n, stage_T)
            return obj
        
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            _, g = logsumexp_obj_and_grad(z, n, stage_T)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def subgradient_refine(a0, time_limit, n_steps=50000):
    """Exact subgradient method on the max of convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    lr = 0.05
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        # Use top few peaks for subgradient
        n_peaks = min(5, 2 * n - 1)
        sorted_idx = np.argsort(b)[::-1][:n_peaks]
        
        grad = np.zeros(n)
        for k in sorted_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= n_peaks
        
        a_trial = a - lr * grad
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        score = 2.0 * n * fast_conv(a_trial).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.9
        lr = max(lr, 1e-7)
        
        if no_improve > 3000:
            break
        
        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Fast local evaluation (same formula as evaluate_sequence)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    a = np.minimum(a, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining Nesterov LP alternating, KKT Newton,
    L-BFGS LogSumExp, and subgradient methods with multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        # Create a good initial sequence
        x = np.linspace(0, 1, n_init)
        # Start with a slightly skewed distribution
        best_seq = 0.7 + 0.3 * np.cos(np.pi * x)
        best_seq = np.maximum(best_seq, 0.01)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    n_best = len(best_seq)
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        a = np.minimum(a, 1000.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST [{label}] ({elapsed:.0f}s): {sc:.8f} (n={len(a)})")
    
    # === Phase 1: Nesterov LP (long) ===
    elapsed = time.monotonic() - t_start
    t_alloc = min((deadline - time.monotonic()) * 0.45, 700.0)
    if t_alloc > 90:
        print(f"\n=== Phase 1: Nesterov LP n={n_best} (score={best_score:.8f}) budget={t_alloc:.0f}s ===")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, max_iters=500000)
        try_update(a_lp, "nesterov_lp")
    
    # === Phase 2: KKT Newton peak equalization ===
    elapsed = time.monotonic() - t_start
    t_alloc = min((deadline - time.monotonic()) * 0.15, 250.0)
    if t_alloc > 30:
        print(f"\n=== Phase 2: KKT Newton (score={best_score:.8f}) budget={t_alloc:.0f}s ===")
        a_kkt, _ = kkt_newton(best_seq, t_alloc)
        try_update(a_kkt, "kkt_newton")
    
    # === Phase 3: Peak equalization (subgradient on active set) ===
    t_alloc = min((deadline - time.monotonic()) * 0.12, 200.0)
    if t_alloc > 20:
        print(f"\n=== Phase 3: Peak Equalize (score={best_score:.8f}) budget={t_alloc:.0f}s ===")
        a_pe, _ = kkt_peak_equalize(best_seq, t_alloc, n_steps=100000, active_frac=0.005)
        try_update(a_pe, "peak_eq")
    
    # === Phase 4: L-BFGS LogSumExp (medium temp) ===
    t_alloc = min((deadline - time.monotonic()) * 0.15, 250.0)
    if t_alloc > 30:
        print(f"\n=== Phase 4: L-BFGS LogSumExp (score={best_score:.8f}) budget={t_alloc:.0f}s ===")
        T_stages = [15.0, 8.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.15, 0.08, 0.04]
        a_lb = lbgfs_logsumexp_refine(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=6000)
        try_update(a_lb, "lbgfs_1")
    
    # === Phase 5: Subgradient refinement ===
    t_alloc = min((deadline - time.monotonic()) * 0.08, 120.0)
    if t_alloc > 15:
        print(f"\n=== Phase 5: Subgradient (score={best_score:.8f}) budget={t_alloc:.0f}s ===")
        a_sg, _ = subgradient_refine(best_seq, t_alloc, n_steps=80000)
        try_update(a_sg, "subgrad")
    
    # === Phase 6: Multi-length exploration ===
    t_rem = deadline - time.monotonic()
    if t_rem > 120:
        print(f"\n=== Phase 6: Multi-n exploration (score={best_score:.8f}) ===")
        alt_ns = [int(n_best * 0.85), int(n_best * 1.15), int(n_best * 1.3), 
                  int(n_best * 0.7), int(n_best * 1.5)]
        # Filter valid
        alt_ns = [n for n in alt_ns if 200 <= n <= 3000 and n != n_best]
        # Remove duplicates
        alt_ns = list(dict.fromkeys(alt_ns))
        
        t_per_alt = min(t_rem / max(len(alt_ns), 1) * 0.8, 150.0)
        for alt_n in alt_ns[:4]:
            if time.monotonic() > deadline - 30:
                break
            if t_per_alt < 20:
                break
            print(f"  Trying n={alt_n} (budget={t_per_alt:.0f}s)")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = nesterov_lp_refine(init_alt, t_per_alt * 0.5, solver=solver, max_iters=50000)
            try_update(a_alt, f"lp_n{alt_n}")
            t_rem2 = deadline - time.monotonic()
            if t_rem2 > 20 and best_seq is not None:
                a_alt2 = lbgfs_logsumexp_refine(best_seq, min(t_rem2 * 0.3, 40.0),
                                                 T_stages=[5.0, 2.0, 1.0, 0.5, 0.2],
                                                 maxiter_per_stage=2000)
                try_update(a_alt2, f"lbgfs_n{alt_n}")
    
    # === Phase 7: Final ultra-precise polishing ===
    t_rem = deadline - time.monotonic()
    if t_rem > 30:
        print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
        t_fin = min(t_rem * 0.5, 60.0)
        if t_fin > 15:
            a_fin = lbgfs_logsumexp_refine(best_seq, t_fin,
                                           T_stages=[3.0, 1.5, 1.0, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01],
                                           maxiter_per_stage=5000)
            try_update(a_fin, "final_lbgfs")
        
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            a_kkt2, _ = kkt_newton(best_seq, min(t_rem * 0.4, 30.0))
            try_update(a_kkt2, "final_kkt")
        
        t_rem = deadline - time.monotonic()
        if t_rem > 15:
            a_sg2, _ = subgradient_refine(best_seq, min(t_rem * 0.4, 25.0), n_steps=30000)
            try_update(a_sg2, "final_sg")
        
        # One more ultra-low T pass
        t_rem = deadline - time.monotonic()
        if t_rem > 10:
            a_ultra = lbgfs_logsumexp_refine(best_seq, min(t_rem * 0.5, 15.0),
                                             T_stages=[0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01],
                                             maxiter_per_stage=3000)
            try_update(a_ultra, "ultra_lbgfs")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL: score={best_score:.8f} (n={len(best_seq)}, time={elapsed:.0f}s)")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
