```python
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s
    
    # Zero-pad to at least 2n-1
    fft_size = 1 << (2*n-1).bit_length()
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2*n*max(fast_convolve(a))
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute DFT
        A = np.fft.fft(a, fft_size)
        power = np.abs(A)**2
        
        # Target: flat power spectrum
        # Use a soft constraint: push power toward the average
        avg_power = power.sum() / fft_size
        # New power: geometric mean of current and target
        target_ratio = avg_power / np.maximum(power, 1e-30)
        new_power = power * target_ratio**0.5  # partial update
        
        # Take IFFT to get new a (real part)
        # Need |A_new|^2 = new_power, so A_new = sqrt(new_power) * exp(i*phase)
        phase = np.angle(A)
        A_new = np.sqrt(np.maximum(new_power, 0)) * np.exp(1j*phase)
        a_new = np.fft.ifft(A_new).real[:n]
        
        # Project onto simplex (a >= 0, sum = 1)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        score = 2*n*max(fast_convolve(a_new))
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score
```

```python
def proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_with(a_fixed, b):
    n = len(a_fixed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a_fixed, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_soft_subgradient(a, n_active_frac=0.1, temperature=0.5):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    # Soft weighting: use exp((b - max_b) / (temperature * max_b)) for weights
    weights = np.exp((b - max_b) / (temperature * max_b))
    # Also apply a hard top-k mask for efficiency
    threshold = max_b - n_active_frac * max_b
    mask = (b >= threshold).astype(float)
    weights = weights * mask
    total_w = weights.sum()
    if total_w < 1e-300:
        active = np.array([int(np.argmax(b))])
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        return g, max_b
    weights = weights / total_w
    g = np.zeros(n)
    for k in np.where(mask > 0)[0]:
        if weights[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += weights[k] * a[k - idx]
    return g, max_b

    b = fast_convolve(a)

    best_score = 2.0 * n * max_b
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    lr = 0.3
    no_improve = 0
    total_steps = 300000

    for step in range(total_steps):
        if time.monotonic() - t0 > time_limit:
            break
        progress = min(step / total_steps, 1.0)
        temp = temp_init + (temp_end - temp_init) * progress

        g, max_b = compute_soft_subgradient(a, n_active_frac, temp)
        g_norm_inf = np.abs(g).max()
        if g_norm_inf < 1e-15:
            break

        gap = best_score - f_lb
        if gap < 1e-10:
            break

        # Adaptive step: Polyak-inspired
        polyak_lr = gap / (2.0 * g_norm_inf * best_score + 1e-15)
        alpha = min(lr, polyak_lr)
        alpha = max(alpha, 0.005)

        z_new = z - alpha * g
        a_new = softmax_from_z(z_new)
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()

            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.03, 1.5)
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.8
                no_improve = 0
            lr = max(lr, 0.005)

        if step % 50000 == 0 and step > 0:
            print(f"    MD-soft {time.monotonic()-t0:.0f}s: score={best_score:.6f} lr={lr:.4f} temp={temp:.3f}")

def qp_frank_wolfe(a0, time_limit, n_fw_inner=50, n_outer=20000):
    """
    Alternating:
    - Given c (distribution over 2n-1 lags), solve QP: min a^T T(c) a s.t. a in simplex
      using projected gradient (Frank-Wolfe style)
    - Update c based on current a*a peaks
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    size = 2 * n - 1
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    # Initialize c: uniform over peaks of a*a
    b = fast_convolve(a)
    c = np.ones(size) / size

    f_lb = 2.0 * n / (2.0 * n - 1.0)
    stal = 0

    for outer in range(n_outer):
        if time.monotonic() - t0 > time_limit:
            break

        # Build T(c) product: (T(c) @ a) [j] = sum_k c[|k-j|] * a[k]
        # This is a convolution-like operation
        # (T(c) @ a)[j] = sum_k c_{|k-j|} a_k = (reverse(c) * a)[j] in the appropriate range
        # Actually: (T(c) a)_j = sum_{k=0}^{n-1} c_{|j-k|} a_k
        # Let's compute via correlation
        c_pad = np.zeros(size)
        c_pad[:len(c)] = c
        # Toeplitz matrix-vector product via FFT
        # T(c) is symmetric Toeplitz with first column = [c_0, c_1, ..., c_{n-1}]
        # Actually first row = [c_0, c_1, ..., c_{n-1}] and it's symmetric
        # So (T(c) a)_j = sum_k c_{|j-k|} a_k
        # This is the aperiodic cross-correlation of c (padded) and a
        # = (c_full * flip(a))[j] where c_full is c extended symmetrically

        # More directly: (T(c)a)_j = sum_k c_{|j-k|} a_k
        # Let's use the fact that this equals the j-th element of the convolution
        # of the "two-sided" c with a
        c_two_sided = np.zeros(2 * size)
        c_two_sided[n - 1:n - 1 + size] = c  # center c
        # Now (T(c)a)_j = sum_k c_two_sided[j-k+n-1] * a_k for k=0..n-1, j=0..n-1
        # = conv(c_two_sided, a)[n-1 : n-1+n]
        fft_sz = 1 << (2 * size - 1).bit_length()
        C = rfft(c_two_sided, fft_sz)
        A = rfft(a, fft_sz)
        Ta = irfft(C * A, fft_sz)[:len(c_two_sided) + n - 1]
        Ta = Ta[n - 1:n - 1 + n]

        # Projected gradient for QP: min a^T T(c) a s.t. a in simplex
        # = min 0.5 * a^T 2T(c) a (the 0.5 is convenient)
        # Gradient = 2 * T(c) * a
        for fw_it in range(n_fw_inner):
            grad = 2.0 * Ta
            # Backtracking line search
            step_lr = 1.0 / (2.0 * np.abs(grad).max() + 1e-15)
            for _ in range(10):
                a_trial = proj_simplex(a - step_lr * grad)
                # Check sufficient decrease
                Ta_trial = None  # Skip exact check for speed, use Armijo
                a = a_trial
                # Recompute Ta
                C2 = rfft(c_two_sided, fft_sz)
                A2 = rfft(a, fft_sz)
                Ta = irfft(C2 * A2, fft_sz)[:len(c_two_sided) + n - 1]
                Ta = Ta[n - 1:n - 1 + n]
                break  # Single step per FW iteration for speed

        # Evaluate
        b = fast_convolve(a)
        score = 2.0 * n * b.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stal = 0
        else:
            stal += 1

        # Update c: concentrate on peaks of a*a
        # Use softmax with adaptive temperature
        temp_c = 0.1 * b.max()
        c = np.exp((b - b.max()) / temp_c)
        c = c / c.sum()

        # Nesterov-like momentum on c
        if outer > 0:
            c_prev = c  # Simplified: just use current
            pass

        if stal > 200:
            # Perturb a
            rng = np.random.default_rng(outer * 31 + 7)
            noise = rng.random(n) * 0.01
            a = a + noise
            a = proj_simplex(a)
            stal = 0

        if outer % 2000 == 0 and outer > 0:
            print(f"    QP-FW {time.monotonic()-t0:.0f}s: outer={outer} score={best_score:.6f}")

    return best_a, best_score


def spectral_equalize(a0, time_limit, n_iters=50000):
    """
    Iteratively flatten the power spectrum of a to reduce peak autocorrelation.
    a*a has DFT = |DFT(a)|^2. Flattening |DFT(a)|^2 reduces max(a*a).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Progress-based damping
        progress = min(it / n_iters, 1.0)
        damping = 0.3 + 0.5 * progress

        A = np.fft.fft(a, fft_size)
        power = np.abs(A) ** 2

        # Target: geometric mean with flat spectrum
        avg_power = power.sum() / fft_size
        # Move power toward average by partial replacement
        new_power = power + damping * (avg_power - power)
        new_power = np.maximum(new_power, 1e-30)

        # Preserve phase
        phase = np.angle(A)
        A_new = np.sqrt(new_power) * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]

        # Project to simplex
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0 and it > 0:
            print(f"    Spectral {time.monotonic()-t0:.0f}s: iter={it} score={best_score:.6f}")

    return best_a, best_score


def lbgfs_entropic(a0, time_limit, T_start=300.0, T_end=0.1, n_stages=10):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    # Geometric temperature schedule with more stages at low T
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, 8000 // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False,
                         'maxls': 50, 'ml': 30}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def lp_best_response_fast(a_fixed, n, solver):
    """Solve LP: min M s.t. (a*b)(k) <= M for all k, b >= 0, sum(b) = 1"""
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2 * n - 1

    # Build constraint matrix implicitly using FFT for checking,
    # but cvxpy needs explicit matrix. For n=960, this is 1919x960.
    # Use a pre-built approach: the constraint is that the Toeplitz product <= M
    # T @ b <= M * ones where T has rows corresponding to lags

    # For efficiency, build T as a scipy sparse-like structure
    # Actually for cvxpy, let's just build it directly
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=50000):

    for it in range(n_max_iters):

def local_refine(a0, time_limit, n_steps=100000):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

    a = a / s

    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 5000:
            break

        b = fast_convolve(a)
        peak_val = b.max()
        # Find positions contributing most to peaks
        near_peaks = np.where(b >= 0.95 * peak_val)[0]

        contrib = np.zeros(n)
        for k in near_peaks[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]

        # Source: high contribution; Dest: low contribution
        src_candidates = np.argsort(contrib)[::-1][:10]
        dst_candidates = np.argsort(contrib)[:10]

        improved = False
        for si in src_candidates:
            if a[si] < 1e-12:
                continue
            for di in dst_candidates:
                if si == di:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    amount = a[si] * frac
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            # Random perturbation occasionally
            if step % 100 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-12:
                    amount = a[i] * 0.05

                    a_trial = np.maximum(a_trial, 0)

                        sc = 2.0 * n * fast_convolve(a_trial).max()
                        if sc < best_score:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2 * x) + 0.2 * np.cos(4 * x)
    elif choice == 6:
        # Raised cosine window
        x = np.linspace(-1, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * x)) for _ in [0]
        a = 0.5 * (1 + np.cos(np.pi * np.linspace(-1, 1, n)))
    else:
        # Product structure
        half = n // 2
        x1 = np.linspace(0, 1, half)
        a1 = np.exp(-3 * x1 * (1 - x1))
        a = np.zeros(n)
        a[:half] = a1
        a[half:] = a1[::-1]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def get_solver():
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    solver = get_solver()
    print(f"Using solver: {solver}")

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except (NameError, Exception):
        pass

    n_base = len(best_seq) if best_seq is not None else 960

    if parent is None:
        n_base = 960
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Init: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  ** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Main optimization pipeline for base length ===
    print(f"\n=== Phase 1: Mirror Descent Soft (n={n_base}, best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.20, 300.0)
    if t1 > 20:
        a_md, _ = mirror_descent_soft(best_seq, t1, n_active_frac=0.12, temp_init=1.5, temp_end=0.3)
        try_update(a_md, "md_soft")

    print(f"\n=== Phase 2: QP Frank-Wolfe (n={n_base}, best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 200.0)
    if t2 > 20:
        a_qp, _ = qp_frank_wolfe(best_seq, t2, n_fw_inner=30, n_outer=15000)
        try_update(a_qp, "qp_fw")

    print(f"\n=== Phase 3: Spectral Equalization (n={n_base}, best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.10, 120.0)
    if t3 > 10:
        a_sp, _ = spectral_equalize(best_seq, t3, n_iters=30000)
        try_update(a_sp, "spectral")

    print(f"\n=== Phase 4: Nesterov LP (n={n_base}, best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.20, 300.0)
    if t4 > 30:
        a_lp, _ = nesterov_lp_refine(best_seq, t4, solver=solver, n_max_iters=30000)
        try_update(a_lp, "nesterov_lp")

    print(f"\n=== Phase 5: L-BFGS Entropic (n={n_base}, best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 200.0)
    if t5 > 20:
        a_lb = lbgfs_entropic(best_seq, t5, T_start=300.0, T_end=0.1, n_stages=10)
        try_update(a_lb, "lbgfs_ent")

    print(f"\n=== Phase 6: Local Refine (n={n_base}, best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.08, 80.0)
    if t6 > 10:
        a_lr, _ = local_refine(best_seq, t6, n_steps=50000)
        try_update(a_lr, "local")

    # === Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length (best={best_score:.6f}) ===")
    alt_lengths = set()
    for mult in [1.25, 1.5, 0.75, 2.0, 1.1]:
        alt_n = int(n_base * mult)
        if 512 <= alt_n <= 3000 and alt_n != n_base:
            alt_lengths.add(alt_n)
    alt_lengths = sorted(alt_lengths)[:3]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_total = min(rem * 0.30, 250.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()
            print(f"  Interpolated improvement: {sc_alt:.6f}")

        # MD on alt length
        t_md = t_total * 0.4
        if t_md > 10:
            a_alt, _ = mirror_descent_soft(init_alt, t_md, n_active_frac=0.10, temp_init=1.0, temp_end=0.3)
            try_update(a_alt, f"md_n{alt_n}")

        # Spectral on alt length
        rem = deadline - time.monotonic()
        if rem > 20:
            t_sp = min(rem * 0.25, 60.0)
            if t_sp > 5:
                a_alt2, _ = spectral_equalize(best_seq if len(best_seq) == alt_n else init_alt, t_sp, n_iters=20000)
                try_update(a_alt2, f"spec_n{alt_n}")

        # LP on alt length (only if fast enough)
        rem = deadline - time.monotonic()
        if rem > 40 and alt_n <= 1500:
            t_lp = min(rem * 0.3, 80.0)
            if t_lp > 10:
                a_alt3, _ = nesterov_lp_refine(best_seq if len(best_seq) == alt_n else init_alt, t_lp, solver=solver, n_max_iters=10000)
                try_update(a_alt3, f"lp_n{alt_n}")

    # === Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.3, 60.0)
        a_fin = lbgfs_entropic(best_seq, t_fin, T_start=20.0, T_end=0.05, n_stages=6)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lr = min(rem * 0.25, 40.0)
        a_lr2, _ = local_refine(best_seq, t_lr, n_steps=30000)
        try_update(a_lr2, "final_local")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_md_fin = min(rem * 0.3, 30.0)
        a_md_fin, _ = mirror_descent_soft(best_seq, t_md_fin, n_active_frac=0.05, temp_init=0.5, temp_end=0.1)
        try_update(a_md_fin, "final_md")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
```
