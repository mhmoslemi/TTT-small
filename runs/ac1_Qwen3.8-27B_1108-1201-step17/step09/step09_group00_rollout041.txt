"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated L-BFGS
with LogSumExp smoothing, LP-based initialization, and adaptive multi-length

Python-loop gradient, leading to significantly better convergence.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_cross_corr(w, a):
    """
    Compute cross-correlation (w ⋆ a)[j] = sum_k w[k] * a[k-j] for j=0,...,n-1.
    w has length 2n-1, a has length n. Uses FFT for O(n log n) computation.
    """
    n = len(a)
    # We need (w ⋆ a)[j] = sum_k w[k+j] * a[k] for j = 0, ..., n-1
    # This is ifft(FFT(w, L) * conj(FFT(a, L)))[0:n] where L >= 3n-2
    L = 1
    while L < 3 * n - 2:
        L *= 2
    W = rfft(w.astype(np.float64), L)
    A = rfft(a.astype(np.float64), L)
    # Cross-correlation: ifft(FFT(w) * conj(FFT(a)))
    # But we need real-valued result
    W_full = np.fft.fft(w.astype(np.float64), L)
    A_full = np.fft.fft(a.astype(np.float64), L)
    result = np.fft.ifft(W_full * np.conj(A_full)).real[:n]
    return result


def compute_obj_grad_logdomain(z, T):
    """
    Compute the LogSumExp smoothed objective and gradient in log-domain.
    z = log(a), a = softmax(z).
    Returns (objective, gradient w.r.t. z).
    """
    n = len(z)
    # a = softmax(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    # b = conv(a, a)
    b = fast_convolve(a)
    
    # LogSumExp of T * b
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    # Weights
    w = exp_Tb / exp_Tb.sum()
    
    # grad_a = 2 * (w ⋆ a) [cross-correlation]
    grad_a = 2.0 * compute_cross_corr(w, a)
    
    # Gradient in log-domain: a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def lbgfs_fft_optimize(a0, time_limit, T_stages=None, maxiter_per_stage=50000):
    """L-BFGS with FFT-based LogSumExp gradient."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
    
    best_z = z.copy()
    best_obj = compute_obj_grad_logdomain(z, T_stages[0])[0]
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        iters_left = int(maxiter_per_stage * remaining / time_limit)
        iters_left = max(50, min(iters_left, maxiter_per_stage))
        
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e30
            return compute_obj_grad_logdomain(z, stage_T)[0]
        
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            return compute_obj_grad_logdomain(z, stage_T)[1]
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_left, 'ftol': 1e-18,
                         'gtol': 1e-16, 'maxfun': iters_left * 500,
                         'disp': False, 'maxls': 50}
            )
            z = res.x
            # Compute actual objective at this T
            obj_val = compute_obj_grad_logdomain(z, stage_T)[0]
            if obj_val < best_obj:
                best_z = z.copy()
                best_obj = obj_val
        except Exception as e:
            print(f"  L-BFGS error at T={stage_T}: {e}")
        
        elapsed = time.monotonic() - t0
        print(f"    L-BFGS T={stage_T:.2f} ({elapsed:.0f}s/el={time_limit:.0f}s): "
              f"raw_obj={obj_val:.10f}")
    
    # Convert back to probability vector
    z_shifted = best_z - best_z.max()
    a = np.exp(z_shifted) / np.exp(z_shifted).sum()
    return a


def fast_convolve_matrix_vec(a_fixed, b):
    """Compute conv(a_fixed, b) efficiently."""
    return np.convolve(a_fixed, b)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def solve_lp_fast(a_fixed):
    """
    Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Uses scipy.optimize.linprog for speed.
    """
    n = len(a_fixed)
    T = build_conv_matrix(a_fixed)
    
    # minimize M
    # s.t. T @ b - M <= 0 (2n-1 constraints)
    #      sum(b) = 1
    #      b >= 0
    #      M free (but will be positive)
    
    # Decision variables: [b_0, ..., b_{n-1}, M]
    # Objective: minimize M → c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: T @ b - M <= 0
    # A_ub @ x <= b_ub
    A_ub = np.hstack([T, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b_j >= 0, M >= 0 (M will be positive at optimum)
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            b_val = res.x[:n]
            M_val = res.x[n]
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val /= s
            return b_val, M_val
    except Exception:
        pass
    
    # Fallback: use cvxpy
    try:
        import cvxpy as cp
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1, M_var >= 0]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        prob.solve(solver=cp.ECOS)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val /= s
            return b_val, float(M_var.value)
    except Exception:
        pass
    
    return None, None
