import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def normalize(a):
    """Normalize a to sum to 1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lp_best_response(a, solver):
    """
    Solve LP: minimize M s.t. T@b <= M*1, sum(b)=1, b>=0
    where T[k,i] = a[k-i] if 0<=k-i<n and 0<=i<n.
    Returns the optimal b.
    """
    import cvxpy as cp
    n = len(a)
    
    # Build T efficiently
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

        it += 1

def mirror_descent(a_init, time_limit, lr_init=0.1, top_k=50, verbose=True):
    """Multi-peak subgradient mirror descent."""
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

            lr *= 0.7
            no_improve = 0
            if lr < 1e-10:
                break

        k_act = min(top_k, 2 * n - 1)
        if k_act < 2 * n - 1:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
        else:
            peaks = np.arange(2 * n - 1)
        
        pv = conv[peaks]
        temp = 0.01 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws
        
        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end+1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]
        
        # Mirror descent step
        a_new = a * np.exp(-lr * sg)
        a_new = normalize(a_new)
        a = a_new
        conv = fast_convolve(a)

            print(f"    MD it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    n = len(a_init)

    best_a = a.copy()

        improved = False
        
        for i in range(n):
            if a[i] < 1e-15:

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = conv[:k_start].max()
            else:
                unaff_max = 0.0
            if k_end < 2 * n - 2:
                right_max = conv[k_end+1:].max()
                if right_max > unaff_max:
                    unaff_max = right_max
            
            a_shifted = a[k_start - i:k_end - i + 1]

            best_new_M = M
            
            for factor in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:

                new_val = a[i] + delta
                if new_val < -1e-15:

                if 2 * i >= k_start and 2 * i <= k_end:

                    best_delta = delta

    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def normalize(a):
    """Normalize a to be non-negative with sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _find_solver():
    """Find the best available LP solver."""
    import cvxpy as cp
    global _cached_solver_name
    if '_cached_solver_name' in dir() and _cached_solver_name is not None:
        return _cached_solver_name
    _cached_solver_name = None
    for name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK, cp.CBC]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _cached_solver_name = name
                return name
        except Exception:
            continue
    _cached_solver_name = cp.ECOS
    return cp.ECOS


def lp_best_response(a, solver):
    """
    Solve LP: minimize M s.t. T@b <= M*1, sum(b)=1, b>=0
    where T[k,i] = a[k-i] if 0<=k-i<n and 0<=i<n.
    Returns the optimal b (the best response to a).
    """
    import cvxpy as cp
    n = len(a)

    # Build T efficiently using stride tricks or loop
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver, verbose=False)

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def lp_alternating(a_init, time_limit, solver=None, verbose=True, label=""):
    """
    LP-based alternating best-response method.
    At each step, solve LP to find best b for current a, then update a toward b.
    """
    if solver is None:
        solver = _find_solver()

    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(7)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP for best response b
        b = lp_best_response(a, solver)
        if b is None:
            break

        # Evaluate b as a direct candidate
        conv_b = fast_convolve(b)
        score_b = 2.0 * n * conv_b.max()
        if score_b < best_score:
            best_score = score_b
            best_a = b.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Damped update with decaying step size
        alpha = 3.0 / (it + 1.0)
        a_new = (1.0 - alpha) * a + alpha * b
        a_new = normalize(a_new)

        # Evaluate the damped update
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0

        # Every 50 iterations, try a more aggressive jump
        if it % 50 == 0:
            a_jump = b.copy()
            score_jump = 2.0 * n * fast_convolve(a_jump).max()
            if score_jump < best_score:
                best_score = score_jump
                best_a = a_jump.copy()
            # Also try 50/50 blend
            a_blend = 0.5 * (a + b)
            a_blend = normalize(a_blend)
            score_blend = 2.0 * n * fast_convolve(a_blend).max()
            if score_blend < best_score:
                best_score = score_blend
                best_a = a_blend.copy()

        a = a_new

        # Convergence check
        if no_improve > 300:
            # Perturb and restart
            a = best_a + rng.standard_normal(n) * 0.02
            a = normalize(a)
            no_improve = 0

        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-alt [{label}] it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def mirror_descent(a_init, time_limit, lr_init=0.08, top_k=40, verbose=True, label=""):
    """Multi-peak subgradient mirror descent on the simplex."""
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            lr *= 0.75
            no_improve = 0
            if lr < 1e-10:
                break

        # Multi-peak subgradient with softmax weighting
        k_act = min(top_k, 2 * n - 1)
        if k_act < 2 * n - 1:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
        else:
            peaks = np.arange(2 * n - 1)

        pv = conv[peaks]
        temp = 0.01 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Compute weighted subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Mirror descent (multiplicative) step
        a_new = a * np.exp(-lr * sg)
        a_new = normalize(a_new)
        a = a_new
        conv = fast_convolve(a)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD [{label}] it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def coord_descent(a_init, time_limit, verbose=True, label=""):
    """Coordinate descent with adaptive steps for local optimization."""
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        improved = False

        for i in range(n):
            if a[i] < 1e-14:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            length_affected = k_end - k_start + 1

            # Unaffected maximum
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            a_shifted = a[k_start - i:k_end - i + 1]

            best_delta = 0.0
            best_new_M = M

            # Try several multiplicative factors
            factors = [0.3, 0.6, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.5, 2.0]
            for factor in factors:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0:
                old_val = a[i]
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * old_val + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 15:
                break
        else:
            no_improve_sweeps = 0

        if verbose:
            elapsed = time.monotonic() - t0
            if no_improve_sweeps == 1:
                print(f"    CD [{label}] converged at {elapsed:.0f}s: best={best_score:.6f}")
            elif elapsed > 60:
                print(f"    CD [{label}] ({elapsed:.0f}s): best={best_score:.6f} improved={improved}")

    return best_a, best_score


def fine_coord_descent(a_init, time_limit, verbose=True, label=""):

    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))

    M = float(conv.max())

            a_shifted = a[k_start - i:k_end - i + 1]

            best_delta = 0.0
            best_new_M = M

            # Very fine steps around current value
            for factor in [0.97, 0.99, 1.01, 1.03]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:

                new_M = max(unaff_max, float(new_affected.max()))

            if best_delta != 0.0:

                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1

        else:

        if verbose:

            if no_improve == 1:
                print(f"    FineCD [{label}] done at {elapsed:.0f}s: best={best_score:.6f}")

    return best_a, best_score


def upsample(a, factor):
    """Upsample sequence by inserting values (nearest neighbor style)."""
    n = len(a)
    new_n = n * factor
    new_a = np.zeros(new_n)
    for i in range(n):
        for j in range(factor):
            new_a[i * factor + j] = a[i] / factor
    return new_a


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    return normalize(a)


def random_smooth_init(n, rng, num_peaks=3):
    """Random smooth initialization with several Gaussian bumps."""
    a = np.zeros(n)
    for _ in range(num_peaks):
        center = rng.uniform(0, n)
        width = rng.uniform(0.05, 0.25) * n
        amplitude = rng.uniform(0.5, 1.5)
        t = np.arange(n, dtype=np.float64)
        a += amplitude * np.exp(-0.5 * ((t - center) / width) ** 2)
    return normalize(np.maximum(a, 0))


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate using evaluate_sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  *** NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
