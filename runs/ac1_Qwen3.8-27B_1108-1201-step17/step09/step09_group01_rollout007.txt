import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size[:size])[:size] if size < len(irfft(A * A, fft_size)) else irfft(A * A, fft_size)[:size]


def fast_convolve2(a):
    """Linear auto-convolution via FFT (safer version)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)
    return result[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, maintaining sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def soft_gradient_descent(a0, time_limit, beta_start=30.0, beta_end=200.0,
                          lr_start=0.01, lr_end=0.001, verbose=True):
    """
    Soft gradient descent using logsumexp approximation of max.
    Optimizes over the simplex using Adam optimizer on log-parameters.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Use log-parameters: a = exp(z) / sum(exp(z))
    z = np.log(a + 1e-30) - np.log(a + 1e-30).max()

    # Adam optimizer state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t_step = 0

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve2(a).max()
    t0 = time.monotonic()
    total_steps = max(1, int(time_limit / 0.05))  # estimate
    batch_size = max(5, n // 50)

    # Precompute index pairs for gradient computation
    # grad of conv[k] w.r.t. a[i]: d(conv[k])/da[i] = a[k-i] + a[k-i] (symmetric) = 2*a[k-i] if k-i in range
    # For logsumexp: weight_k = exp(beta * conv[k]) / sum(exp(beta * conv))
    # grad f / da[i] = 2 * sum_k weight_k * a[k-i] (for k where k-i in [0,n-1])
    #               + product rule for 1/s^2

    step_count = 0
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # Compute current a from z
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()

        # Compute convolution and soft max gradient
        conv = fast_convolve2(a)
        max_conv = conv.max()
        scaled = beta * (conv - max_conv)
        scaled = np.clip(scaled, -100, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        weights = weights / Z

        # Gradient of soft_max w.r.t. a
        # d(soft_max)/da[i] = sum_k weights[k] * d(conv[k])/da[i]
        # d(conv[k])/da[i] = 2 * a[k-i] if 0 <= k-i < n
        #                   (plus delta^2 correction not needed for gradient)
        grad_a = np.zeros(n)
        for k in range(len(conv)):
            if weights[k] < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad_a[j] += 2.0 * weights[k] * a[k - j]

        # Full gradient of score = 2*n*soft_max / s^2
        # d(score)/da[i] = (2*n*grad_soft[i]/s^2) - (4*n*soft_max*a[i]/s^3)
        # Since s=1 (we're on simplex), simplifies to:
        # d(score)/da[i] = 2*n*grad_a[i] - 4*n*soft_max*a[i]
        score_grad = 2.0 * n * grad_a - 4.0 * n * max_conv * a

        # Convert to z-gradient: da/dz = diag(a) - outer(a, a)
        # dz[i] = (score_grad . (da/dz[:,i])) = score_grad[i]*a[i] - a[i]*(score_grad*a).sum()
        # Actually: dL/dz[i] = sum_j (dL/da[j]) * (da[j]/dz[i])
        #           = sum_j score_grad[j] * (a[j]*delta[ji] - a[j]*a[i])
        #           = score_grad[i]*a[i] - a[i]*sum_j score_grad[j]*a[j]
        dot = np.dot(score_grad, a)
        grad_z = a * (score_grad - dot)

        # Normalize gradient for stability
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm > 1e-10:
            grad_z = grad_z / grad_norm

        # Adam update
        t_step += 1
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Clip z for numerical stability
        z = np.clip(z, -30, 30)

        # Evaluate
        if step_count % 50 == 0:
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a_eval = exp_z / exp_z.sum()
            score = 2.0 * n * fast_convolve2(a_eval).max()
            if score < best_score:
                best_score = score
                best_a = a_eval.copy()
                if verbose:
                    elapsed = time.monotonic() - t0
                    print(f"    SGD step={step_count} ({elapsed:.0f}s): score={score:.6f} beta={beta:.1f}")

        step_count += 1

    return best_a, best_score

                if new_s < 1e-10:
                    continue
                new_score = 2.0 * n * new_M / (new_s * new_s)

                if new_score < best_new_score - 1e-15:
                    best_new_score = new_score
                    best_delta = delta

            # Golden section refinement around best_delta
            if best_delta != 0.0:
                # Search in [best_delta*0.5, best_delta*1.5] or a window around it
                if best_delta > 0:
                    lo = max(-a[i] + 1e-10, best_delta * 0.3 - 0.01 * a[i])
                    hi = min(best_delta * 2.0 + 0.01 * a[i], 5.0 * a[i] + 0.1)
                else:
                    lo = max(-a[i] + 1e-10, best_delta * 2.0 - 0.01 * a[i])
                    hi = min(best_delta * 0.3 + 0.01 * a[i], 0.0)

                # Golden section search (6 iterations)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                for _ in range(8):
                    if hi - lo < 1e-12:
                        break
                    m1 = lo + gr * (hi - lo)
                    m2 = hi - gr * (hi - lo)

                    def eval_delta(d):
                        nv = a[i] + d
                        if nv < -1e-15:
                            return 1e30
                        na = affected + 2.0 * d * a_shift
                        if 2 * i >= k_start and 2 * i <= k_end:
                            idx2i = 2 * i - k_start
                            na = na.copy()
                            na[idx2i] += d * d
                        nM = max(M_unaff, float(na.max()))
                        ns = s + d
                        if ns < 1e-10:
                            return 1e30
                        return 2.0 * n * nM / (ns * ns)

                    f1 = eval_delta(m1)
                    f2 = eval_delta(m2)
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1

                opt_delta = (lo + hi) / 2.0
                if eval_delta(opt_delta) < best_new_score - 1e-15:
                    best_new_score = eval_delta(opt_delta)
                    best_delta = opt_delta

            # Apply best delta
            if best_delta != 0.0 and best_new_score < 2.0 * n * M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                s += best_delta

                # Update conv
                changes = 2.0 * best_delta * a_shift
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(M_unaff, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (s * s)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                sweep_improved = True

        if not sweep_improved:
            no_improve += 1
            if no_improve >= patience:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score, sweeps


def active_set_perturb(a0, rng, strength=0.05, verbose=False):
    """
    Perturbation that specifically targets active (peak) convolution positions.
    Reduces contributions to peaks and adds mass to non-peak-contributing positions.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    a = a / s

    conv = fast_convolve2(a)
    M = conv.max()

    # Find active positions (at 95% of max)
    threshold = M * 0.95
    active_positions = np.where(conv >= threshold)[0]

    # Compute pressure on each coordinate
    pressure = np.zeros(n)
    for k in active_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if j < n:
                product = a[i] * a[j]
                pressure[i] += product
                if j != i:
                    pressure[j] += product * 0.3

    p_max = pressure.max()
    if p_max < 1e-20:
        return a.copy()
    pressure = pressure / p_max

    # Apply: reduce high-pressure, add noise
    noise = rng.standard_normal(n) * 0.5
    a_new = a * (1.0 - strength * pressure + strength * 0.3 * noise)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


def swap_perturb(a0, rng, n_swaps=5, fraction=0.05):
    """Swap mass between random pairs of coordinates."""
    a = a0.copy()
    n = len(a)
    for _ in range(n_swaps):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        amt = min(a[i], a[j]) * fraction * rng.uniform(0.5, 1.5)
        a[i] -= amt
        a[j] += amt
        a[i] = max(a[i], 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def fourier_perturb(a0, rng, n_modes=5, strength=0.03):
    """Add low-frequency Fourier noise to the sequence."""
    a = a0.copy()
    n = len(a)
    noise = np.zeros(n)
    for m in range(1, n_modes + 1):
        amp = strength / m
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n)
        noise += amp * np.cos(2 * np.pi * m * t / n + phase)
    a_new = a * (1.0 + noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def zero_and_redistribute(a0, rng, frac=0.1):
    """Zero out a fraction of smallest elements and redistribute."""
    a = a0.copy()
    n = len(a)
    n_zero = max(1, int(n * frac))
    # Zero out the smallest elements
    idx = np.argsort(a)[:n_zero]
    mass = a[idx].sum()
    a[idx] = 0
    # Redistribute proportionally to remaining
    remaining_mask = a > 0
    if remaining_mask.any():
        a[remaining_mask] += mass * a[remaining_mask] / a[remaining_mask].sum()
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def multi_start_optimize(initial_pool, time_limit, verbose=True):
    """
    Main optimization loop: iterate over starting points, apply
    soft GD -> exact CD -> perturb -> repeat.
    """
    deadline = time.monotonic() + time_limit
    rng = np.random.default_rng(12345)
    best_seq = None
    best_score = float('inf')

    # Initialize pool
    pool = []
    for init in initial_pool:
        s = init.sum()
        if s > 1e-15:
            pool.append(init / s)

    if not pool:
        pool.append(np.ones(1024) / 1024)

    # Track which ones we've already tried
    tried = set()
    cycle = 0

    while time.monotonic() < deadline - 60:
        cycle += 1
        # Pick starting point (round-robin with some randomness)
        start_idx = min(cycle, len(pool) - 1) % len(pool)
        start = pool[start_idx].copy()
        n = len(start)

        # Check remaining time
        rem = deadline - time.monotonic()
        if rem < 30:
            break

        # Phase 1: Soft GD (if time allows)
        t_sgd = min(rem * 0.15, 120.0)
        if t_sgd > 10:
            a_soft, s_soft = soft_gradient_descent(start, t_sgd,
                                                   beta_start=30.0, beta_end=150.0,
                                                   lr_start=0.005, lr_end=0.0005,
                                                   verbose=False)
            if s_soft < best_score:
                best_score = s_soft
                best_seq = a_soft.copy()
                print(f"  Cycle {cycle}: SGD improved to {s_soft:.8f}")

        rem = deadline - time.monotonic()
        if rem < 20:
            break

        # Phase 2: Exact coordinate descent
        a_cd_in = best_seq.copy() if best_seq is not None else start
        t_cd = min(rem * 0.4, 300.0)
        if t_cd > 5:
            a_cd, s_cd, sw = exact_coord_descent(a_cd_in, t_cd, patience=3,
                                                 verbose=False, label=f"c{cycle}")
            if s_cd < best_score:
                best_score = s_cd
                best_seq = a_cd.copy()
                print(f"  Cycle {cycle}: ECD improved to {s_cd:.8f} ({sw} sweeps)")

        rem = deadline - time.monotonic()
        if rem < 20:
            break

        # Phase 3: Perturbation and generate new starting points
        if best_seq is not None:
            # Generate several perturbed versions
            perturbations = []
            perturbations.append(active_set_perturb(best_seq, rng, strength=0.03 + 0.01 * (cycle % 5)))
            perturbations.append(swap_perturb(best_seq, rng, n_swaps=10, fraction=0.03))
            perturbations.append(fourier_perturb(best_seq, rng, n_modes=3, strength=0.02))
            perturbations.append(zero_and_redistribute(best_seq, rng, frac=0.05))

            # Stronger perturbation every few cycles
            if cycle % 4 == 0:
                perturbations.append(swap_perturb(best_seq, rng, n_swaps=30, fraction=0.1))
                perturbations.append(fourier_perturb(best_seq, rng, n_modes=8, strength=0.05))

            # Try different length
            if cycle % 3 == 0 and n > 512:
                new_n = n + rng.choice([-128, 128, 256])
                if 512 <= new_n <= 2048:
                    perturbations.append(interpolate_seq(best_seq, new_n))

            for p in perturbations:
                ps = p.sum()
                if ps > 1e-15:
                    pool.append(p / ps)

            # Keep pool manageable
            if len(pool) > 15:
                # Keep best and most recent
                pool = pool[-10:] + [best_seq.copy()]

        # Update best_seq in pool
        if best_seq is not None:
            pool[0] = best_seq.copy()

    return best_seq, best_score


def fine_polish(a0, time_limit, verbose=False):
    """Very fine coordinate descent with tiny steps for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_convolve2(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    # Very fine steps
    step_factors = [0.999, 0.998, 0.995, 0.99, 0.98, 0.97,
                    1.001, 1.002, 1.005, 1.01, 1.02, 1.03, 1.05]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            length_aff = k_end - k_start + 1

            if k_start > 0:
                unaff_l = float(conv[:k_start].max())
            else:
                unaff_l = -1e30
            if k_end < 2 * n - 2:
                unaff_r = float(conv[k_end + 1:].max())
            else:
                unaff_r = -1e30
            M_unaff = max(unaff_l, unaff_r)

            affected = conv[k_start:k_end + 1]
            a_shift = a[:length_aff]

            best_delta = 0.0
            best_new_score = 2.0 * n * max(M_unaff, float(affected.max())) / (s * s)

            for factor in step_factors:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                na = affected + 2.0 * delta * a_shift
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx2i = 2 * i - k_start
                    na = na.copy()
                    na[idx2i] += delta * delta
                nM = max(M_unaff, float(na.max()))
                ns = s + delta
                if ns < 1e-10:
                    continue
                ns_score = 2.0 * n * nM / (ns * ns)
                if ns_score < best_new_score - 1e-16:
                    best_new_score = ns_score
                    best_delta = delta

            if best_delta != 0.0:
                a[i] = max(a[i] + best_delta, 0.0)
                s += best_delta
                changes = 2.0 * best_delta * a_shift

                    idx2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx2i] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(M_unaff, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M / (s * s)
                if cur_score < best_score - 1e-15:
                    best_score
