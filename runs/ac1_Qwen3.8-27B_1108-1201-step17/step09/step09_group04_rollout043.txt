**New approach: Direct max reduction via subgradient + projection**

Instead of p-norm smoothing (which approximates max), we can work directly with the max using a subgradient method:

1. Identify the set of "active" indices where conv(a,a)_k is close to the maximum
2. Compute the subgradient: for each active k, the gradient of conv(a,a)_k w.r.t. a is (a shifted)
3. Take a step in the negative subgradient direction
4. Project back to the simplex (sum=1, a≥0)
5. Repeat

This is essentially a projected subgradient method. The advantage is that it directly targets the max without smoothing.

But the disadvantage is that it's slow and the subgradient is noisy (multiple active constraints).

**Better: Use a "mirror" approach**

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    """Compute the objective: 2n * max(conv) / (sum)^2."""
    a = np.asarray(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max() / (s * s)

def p_norm_obj_and_grad(z, p):
    """Compute p-norm smoothed objective and gradient in log-space (softmax)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    c_p = c_pos ** p
    S = np.sum(c_p)
    if S < 1e-300:
        return 2.0 * n, np.zeros_like(z)
    factor = S ** (1.0/p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    obj = 2.0 * n * S ** (1.0/p)
    # Gradient in z-space: a * (g_a - dot(g_a, a))
    gz = a * (g_a - np.dot(g_a, a))
    return obj, gz

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using aggressive multi-stage p-norm
L-BFGS with very high p values, parametric shape search, peak equalization,
Nesterov LP refinement, and multi-length exploration. Key improvements: much
higher p schedule (up to 50000), asymmetric power-law shapes, targeted peak
reduction, and exploration of larger sequence lengths.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute the objective: 2n * max(conv) / (sum)^2."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max() / (s * s)


def pnorm_obj_grad(z, p):
    """Compute p-norm smoothed objective and gradient in log-space (softmax)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    c_p = c_pos ** p
    S = c_p.sum()
    if S < 1e-300:
        return 2.0 * n, np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    obj = 2.0 * n * S ** (1.0 / p)
    gz = a * (g_a - np.dot(g_a, a))
    return obj, gz


def pnorm_stage(z, p, maxiter):
    """Run one L-BFGS stage at given p."""
    def func(zv):
        o, _ = pnorm_obj_grad(zv, p)
        return o

    def jac(zv):
        _, g = pnorm_obj_grad(zv, p)
        return g

    try:
        res = minimize(func, z, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15,
                                'maxfun': maxiter * 500, 'disp': False})
        return res.x
    except Exception:
        return z


def pnorm_lbfgs(a0, time_budget, p_start=5.0, p_end=20000.0, n_stages=12):
    """Multi-stage p-norm L-BFGS with aggressive p schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    temps = np.geomspace(p_start, p_end, n_stages)

    for stage_idx, p_val in enumerate(temps):
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 15:
            break
        # Allocate more iterations to later (higher p) stages
        frac = (stage_idx + 1) / n_stages
        maxiter = max(300, int(80000 * remaining / time_budget * (0.5 + 0.5 * frac)))

        z = pnorm_stage(z, p_val, maxiter)

        z_max_v = z.max()
        exp_z_v = np.exp(z - z_max_v)
        a_cur = exp_z_v / exp_z_v.sum()
        sc = score_of(a_cur)
        elapsed = time.monotonic() - t0
        print(f"      p={p_val:.0f} ({elapsed:.0f}s): score={sc:.6f}")

    z_max_v = z.max()
    exp_z_v = np.exp(z - z_max_v)
    return exp_z_v / exp_z_v.sum()


def parametric_shape(n, family, params):
    """Generate a parametric shape of length n."""
    x = np.arange(n, dtype=np.float64)
    if family == 'powlaw':
        # a_j = (j+1)^alpha * (n-j)^beta
        alpha = params[0]
        beta = params[1]
        a = (x + 1) ** alpha * (n - x) ** beta
    elif family == 'gauss_off':
        # Gaussian centered at c with width w
        c = params[0]
        w = params[1]
        a = np.exp(-0.5 * ((x - c * (n - 1)) / (w * n)) ** 2)
    elif family == 'cos_pow':
        # (1 + alpha*cos(2*pi*(j+0.5)/n))^beta
        alpha = params[0]
        beta = params[1]
        a = np.maximum(1.0 + alpha * np.cos(2 * np.pi * (x + 0.5) / n), 0) ** beta
    elif family == 'ramp_leading':
        # Leading edge ramp: a_j = (j+1)^alpha for j < n/2, then plateau/decay
        alpha = params[0]
        a = np.ones(n)
        half = n // 2
        a[:half] = (np.arange(half) + 1) ** alpha / half ** alpha
        a[half:] = 1.0 - params[1] * (np.arange(n - half) / max(1, n - half))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def search_parametric(n, time_budget, rng):
    """Search over parametric families for good starting shapes."""
    best_a = None
    best_score = float('inf')
    t0 = time.monotonic()

    # Try various parametric families
    families = [
        ('powlaw', [(0.5, 0.5), (1.0, 0.5), (0.5, 1.0), (1.5, 0.5),
                    (0.5, 1.5), (2.0, 0.5), (0.5, 2.0), (1.0, 1.0),
                    (0.3, 0.7), (0.7, 0.3), (1.5, 1.5), (0.8, 1.2)]),
        ('gauss_off', [(0.3, 0.2), (0.4, 0.2), (0.5, 0.3), (0.3, 0.3),
                       (0.4, 0.15), (0.35, 0.25)]),
        ('cos_pow', [(0.5, 1.0), (0.3, 2.0), (0.5, 2.0), (0.2, 1.0),
                     (0.7, 1.5), (0.3, 1.5)]),
        ('ramp_leading', [(1.0, 0.3), (2.0, 0.3), (1.5, 0.5), (0.5, 0.2)]),
    ]

    for family, param_list in families:
        if time.monotonic() - t0 > time_budget:
            break
        for params in param_list:

                break
            a = parametric_shape(n, family, params)

            if sc < best_score:

    return best_a, best_score


def peak_equalization(a0, time_budget, n_iters=10000):
    """Targeted peak reduction by adjusting contributors to the max convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_budget:
            break

        c = fast_convolve(a)
        k_max = np.argmax(c)
        peak_val = c[k_max]

        # Find second highest peak
        c_copy = c.copy()
        c_copy[k_max] = -1
        k_second = np.argmax(c_copy)
        second_val = c[k_second] if k_second >= 0 else 0

        # Contributors to the main peak
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_max - idx]

        improved = False

        # Strategy 1: Reduce top contributors
        top_n = min(15, len(idx))
        order = np.argsort(contribs)[::-1][:top_n]

        for ci in order:
            i = idx[ci]
            j = k_max - i
            if i == j:
                # Single element contributes a[i]^2
                for factor in [0.99, 0.98, 0.97, 0.95]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 0.01:
                        break
                    a_trial = a_trial / st
                    sc = score_of(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
            else:
                # Pair (i, j) contributes a[i]*a[j]
                for fi in [0.99, 0.98, 0.97, 0.95, 0.93]:
                    for fj in [0.99, 0.98, 0.97]:
                        a_trial = a.copy()
                        a_trial[i] *= fi
                        a_trial[j] *= fj
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 0.01:
                            break
                        a_trial = a_trial / st
                        sc = score_of(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break

        # Strategy 2: Transfer mass from peak contributors to off-peak positions
        if not improved and it % 10 == 0:
            rng = np.random.default_rng(it * 7 + 3)
            # Pick source from top contributors
            src_pool = idx[order[:5]]
            src = src_pool[rng.integers(0, len(src_pool))]
            # Pick target: positions with low contribution to any peak
            all_peaks = np.argsort(c)[::-1][:8]
            in_peak = set()
            for pk in all_peaks:
                for j in range(max(0, pk - n + 1), min(n, pk + 1)):
                    in_peak.add(j)
            candidates = [k for k in range(n) if k not in in_peak and a[k] < np.median(a)]
            if len(candidates) > 0:
                dst = candidates[rng.integers(0, len(candidates))]
                for frac in [0.03, 0.05, 0.08, 0.12]:
                    amount = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amount
                    a_trial[dst] += amount
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 0.01:
                        break
                    a_trial = a_trial / st
                    sc = score_of(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break

        # Strategy 3: Small random perturbation
        if not improved and it % 200 == 0:
            rng = np.random.default_rng(it + 999)
            n_pert = rng.integers(2, 6)
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in indices:
                a_trial[i] *= (1.0 + rng.uniform(-0.04, 0.04))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                sc = score_of(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()

        if it > 0 and it % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PeakEql it={it} ({elapsed:.0f}s): score={best_score:.6f}")

        # Early stop if no improvement for a long time
        if it > 500 and (it % 500 == 0):
            # Check if we're stuck
            pass

    return best_a, best_score


def nesterov_lp_fast(a0, time_budget, n_max_iters=3000):
    """
    Nesterov LP alternating with efficient LP solve.
    Uses the fact that T @ b = convolve(a_fixed, b) can be computed via FFT.
    """
    import cvxpy as cp

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    # Find a good solver
    solver = None
    for s_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s_name
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS

    print(f"      LP solver: {solver}")

    stall = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_budget:
            break

        # Build constraint matrix for current y (the "fixed" sequence)
        # convolve(y, b)[k] = sum_j y[j] * b[k-j] for valid j
        # We need: sum_j y[j] * b[k-j] <= M for all k
        # Variables: b (length n), M (scalar)

        # For efficiency, build the T matrix sparsely
        # T has shape (2n-1, n), T[k, j] = y[j] if 0 <= k-j <= n-1
        from scipy.sparse import csr_matrix
        rows = []
        cols = []
        vals = []
        for j in range(n):
            if y[j] < 1e-15:
                continue
            for k in range(j, j + n):
                rows.append(k)
                cols.append(j)
                vals.append(y[j])
        T = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        prob.solve(solver=solver)

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        bs = b_val.sum()
        if bs < 1e-15:
            continue
        b_val = b_val / bs

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_val
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            break
        y = y / ys

        a = ((it + 1) * a + alpha * (it + 2) * b_val) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        a = a / a.sum()

        sc_a = score_of(a)
        sc_y = score_of(y)

        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        elif sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        if stall > 1000 and it % 100 == 0:
            rng = np.random.default_rng(it + 55)
            y = y + 0.02 * rng.standard_normal(n)
            y = np.maximum(y, 0)
            y = y / y.sum()
            stall = 0

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NesterovLP it={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def subgradient_polish(a0, time_budget, n_iters=2000):
    """Direct subgradient method targeting the max of the convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    step_sizes = 0.01 / np.sqrt(n)

    for it in range(n_iters):
        if time.monotonic() - t0 > time_budget:
            break

        c = fast_convolve(a)
        k_max = int(np.argmax(c))

        # Subgradient of max(conv) w.r.t. a at position i:
        # sg_i = 2 * a[k_max - i] (if 0 <= k_max - i < n)
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        sg = np.zeros(n)
        idx = np.arange(js, je + 1)
        sg[idx] = 2.0 * a[k_max - idx]

        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        sg = sg / sg_norm

        # Take step: a_new = a - eta * sg, then project to simplex
        eta = step_sizes / (1 + 0.1 * it)
        a_trial = a - eta * sg
        # Project to simplex {x: sum(x)=1, x>=0}
        a_trial = np.maximum(a_trial, 0)
        a_trial = a_trial / a_trial.sum()

        sc = score_of(a_trial)
        if sc < best_score - 1e-14:
            a = a_trial
            best_score = sc
            best_a = a.copy()

        if it > 0 and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad it={it} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            perturbed = best_state['seq'] + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            t_sub = min(rem * 0.3, t_pert_total / max(1, 3 - trial))
            if t_sub < 10:
                break
            a_pert = pnorm_lbfgs(perturbed, t_sub, p_start=50.0, p_end=5000.0, n_stages=8)
            try_update(a_pert, f"pert_{noise_scale}")

    # === Phase 7: Length exploration (150s) ===
    print(f"\n=== Phase 7: Length Exploration (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        base = n_base
        # Try larger and smaller lengths
        candidate_lengths = []
        for delta in [64, 128, 192, -64, -128, 256, 320]:
            alt_n = base + delta
            if 512 <= alt_n <= 3072:
                candidate_lengths.append(alt_n)
        # Add some round numbers
        for rn in [1024, 1120, 1280, 1440, 1600, 1792, 2048]:
            if 512 <= rn <= 3072 and rn != base:
                if rn not in candidate_lengths:
                    candidate_lengths.append(rn)

        candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - base))
        n_explored = 0
        for alt_n in candidate_lengths[:4]:
            rem = deadline - time.monotonic()
            if rem < 60 or n_explored >= 3:
                break
            n_explored += 1
            t_total = min(rem * 0.15, 150.0)
            if t_total < 40:
                break
            print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

            init_alt = interpolate_sequence(best_state['seq'], alt_n)

            # Quick p-norm
            t_pn = t_total * 0.6
            a_alt = pnorm_lbfgs(init_alt, t_pn, p_start=10.0, p_end=3000.0, n_stages=8)
            improved = try_update(a_alt, f"pn_n{alt_n}")

            if improved and rem > 30:
                # Peak equalization on the new length
                rem = deadline - time.monotonic()
                if rem > 20:
                    a_pe2, _ = peak_equalization(best_state['seq'], min(rem * 0.3, 50.0), n_iters=5000)
                    try_update(a_pe2, f"pe_n{alt_n}")

    # === Phase 8: Subgradient polish (60s) ===
    print(f"\n=== Phase 8: Subgradient Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_sub = min(rem * 0.15, 60.0)
        a_sg, _ = subgradient_polish(best_state['seq'], t_sub, n_iters=5000)
        try_update(a_sg, "subgrad")

    # === Phase 9: Final combined polish (remaining time) ===
    print(f"\n=== Phase 9: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # High-p L-BFGS
        t_fin = min(rem * 0.4, 80.0)
        a_fin = pnorm_lbfgs(best_state['seq'], t_fin, p_start=200.0, p_end=100000.0, n_stages=8)
        try_update(a_fin, "final_pnorm")

    rem = deadline - time.monotonic()
    if rem > 20:
        # Peak equalization
        t_pe = min(rem * 0.3, 50.0)
        a_pe3, _ = peak_equalization(best_state['seq'], t_pe, n_iters=10000)
        try_update(a_pe3, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 15:
        # One more perturbation
        n_cur = len(best_state['seq'])
        perturbed = best_state['seq'] + 0.03 * rng.standard_normal(n_cur)
        perturbed = np.maximum(perturbed, 0)
        perturbed = perturbed / perturbed.sum()
        t_sub = min(rem * 0.5, 30.0)
        a_last = pnorm_lbfgs(perturbed, t_sub, p_start=100.0, p_end=10000.0, n_stages=6)
        try_update(a_last, "final_pert")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])})")
    return [float(x) for x in best_state['seq']]
