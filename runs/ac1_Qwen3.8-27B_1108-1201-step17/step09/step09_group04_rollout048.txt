import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_correlate(w, a):
    """Compute correlate(w, a, 'valid') via FFT for efficiency."""
    n = len(a)
    m = len(w)
    # correlate(w, a, 'valid')[i] = sum_j w[i+j] * a[j] for j=0..n-1
    # This equals convolution of w with flip(a), taking the right part
    fft_size = 1 << (m + n - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a[::-1], fft_size)
    conv = irfft(W * A, fft_size)
    # conv[k] = sum_j w[k-j]*flip(a)[j] = sum_j w[k-j]*a[n-1-j]
    # We want result[i] = sum_j w[i+j]*a[j]
    # Let m2 = m-1, then result[i] = sum_j w[i+j]*a[j]
    # conv has length m + n - 1
    # conv[n-1+i] = sum_j w[n-1+i-j]*a[n-1-j] = sum_j' w[n-1+i-(n-1-j')]*a[j'] = sum_j' w[i+j']*a[j']
    # where j' = n-1-j
    return conv[n - 1: n - 1 + (m - n + 1)]


def subgradient_max_conv(a):
    """Compute subgradient of max(convolve(a,a)) w.r.t. a.
    Returns sg of same length as a."""
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    # Tolerance for considering ties
    tol = 1e-10 * max_c + 1e-15
    S = np.where(c >= max_c - tol)[0]
    p_vec = np.zeros(2 * n - 1)
    p_vec[S] = 1.0 / len(S)
    # sg_i = 2 * sum_k p_vec[k] * a[k-i] for 0 <= k-i < n
    # = 2 * correlate(p_vec, a, 'valid')[i]
    corr = np.correlate(p_vec, a, mode='valid')
    return 2.0 * corr, max_c


def p_norm_objective(a, p):
    """Compute 2n * (sum c^p)^(1/p) for c = convolve(a,a), assuming sum(a)=1."""
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    return 2.0 * len(a) * (np.sum(c_pos ** p)) ** (1.0 / p)


def p_norm_grad_z(z, p):
    """Gradient of 2n * (sum c^p)^(1/p) w.r.t. z where a = softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def p_norm_lbfgs_refine(a0, time_limit, p_schedule=None, maxiter_per_stage=200):
    """L-BFGS with adaptive p-norm smoothing schedule in log-domain."""
    if p_schedule is None:

    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    for stage_p in p_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        def func(z_val):
            return p_norm_objective(softmax_safe(z_val), stage_p)

        def jac(z_val):
            return p_norm_grad_z(z_val, stage_p)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': maxiter_per_stage * 500, 'disp': False}

            z = res.x

        a_cur = softmax_safe(z)
        score = 2.0 * n * fast_convolve(a_cur).max()
        elapsed = time.monotonic() - t0

    a = softmax_safe(z)
    return a


def softmax_safe(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

            break

        # Compute subgradient at y
        sg, max_val = subgradient_max_conv(y)
        
        # Nesterov dual averaging: alpha = 2/(it+2)
        alpha = 2.0 / (it + 2)
        
        # Gradient step in y
        y = y - alpha * sg
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # x update (averaged)
        a = ((it + 1) * a + (alpha * (it + 2)) * y) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Evaluate both a and y
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        # Check for stall periodically
        if (it + 1) % 2000 == 0:
            if best_score < prev_best - 1e-8:
                prev_best = best_score
                stall_count = 0
            else:
                stall_count += 1

            if stall_count >= 3 and noise_restarts < 20:
                # Perturb y to escape local structure
                rng = np.random.default_rng(it + noise_restarts * 1000)
                scale = 0.01 + 0.01 * noise_restarts
                noise = rng.standard_normal(n) * scale
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                noise_restarts += 1
                stall_count = 0
                prev_best = best_score

            if (it + 1) % 10000 == 0:
                elapsed = time.monotonic() - t0
                print(f"        SubGrad iter={it+1} ({elapsed:.0f}s): score={best_score:.7f}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build matrix T where T @ b = convolve(a, b). Shape (2n-1, n)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, a>=0), find b maximizing sum(b) s.t. conv(a,b) <= min_val."""
    """Actually: min_b max_k conv(a,b)[k] s.t. sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_stall_score = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if (it + 1) % 500 == 0:
            if best_score < prev_stall_score - 1e-8:
                stall_count = 0
                prev_stall_score = best_score
            else:
                stall_count += 1
            if stall_count >= 3:
                rng = np.random.default_rng(it + 42)
                noise = rng.standard_normal(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0
                prev_stall_score = best_score

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.7f}")

    return best_a, best_score


def local_peak_shaving(a0, time_limit):
    """Aggressive local peak shaving: reduce contributions to the maximum lag."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        b = fast_convolve(a)
        # Find the peak lag(s)
        max_val = b.max()
        peak_lags = np.where(b >= max_val - 1e-12)[0]
        
        improved = False
        
        for k in peak_lags:
            # Contributors to b[k]: pairs (i, k-i) where 0<=i<n and 0<=k-i<n
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            
            # Try reducing the largest contributor
            max_idx = np.argmax(contribs)
            i = idx[max_idx]
            j = k - i
            
            # Try different reduction factors
            for f in [0.99, 0.97, 0.95, 0.90, 0.80]:
                a_trial = a.copy()
                a_trial[i] *= f
                st = a_trial.sum()
                if st < 0.01:
                    continue
                a_trial /= st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                break
        
        if not improved:
            # Try pair reduction
            for k in peak_lags[:3]:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[k - idx]
                top = np.argsort(contribs)[::-1][:5]
                for ci in top:
                    i = idx[ci]
                    j = k - i
                    if i == j or j < 0 or j >= n or a[i] < 1e-14 or a[j] < 1e-14:
                        continue
                    for f1 in [0.98, 0.95, 0.90]:
                        for f2 in [0.98, 0.95, 0.90]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            a_trial = np.maximum(a_trial, 0)
                            st = a_trial.sum()
                            if st < 0.01:
                                continue
                            a_trial /= st
                            score = 2.0 * n * fast_convolve(a_trial).max()
                            if score < best_score - 1e-15:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved:
            # Random small perturbation
            if step % 50 == 0:
                rng = np.random.default_rng(step)
                a_trial = a.copy()
                a_trial += 0.005 * rng.standard_normal(n)
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 0.01:
                    a_trial /= st
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
            
            if not improved and step > 200:
                break
        
        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        PeakShave step={step} ({elapsed:.0f}s): score={best_score:.7f}")

    return best_a, best_score


def structured_init(n, variant=0):
    """Create structured initial sequence of length n."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x) ** 2
    elif variant == 4:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif variant == 5:
        # Wide Gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 1.5)
    elif variant == 6:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1.0 - 2.0 * np.abs(x - 0.5)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_state, a, label):
    """Try to update the best state."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.7f} (n={len(a)})")
        return True
    return False


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Wide multi-length subgradient scan
    Phase 2: Deep p-norm L-BFGS on top candidates
    Phase 3: Nesterov subgradient refinement
    Phase 4: LP best-response
    Phase 5: Length fine-tuning
    Phase 6: Peak shaving + final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_state = {'score': float('inf'), 'seq': None}
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = structured_init(n_init, variant=4)
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.7f}")

    # === Phase 1: Quick multi-length subgradient scan ===
    print(f"\n=== Phase 1: Multi-length scan (best={best_state['score']:.7f}) ===")
    n_best = len(best_state['seq'])
    
    # Lengths to try: spread around the parent length + some unusual ones
    scan_lengths = set()
    for delta in range(-192, 193, 32):
        alt = n_best + delta
        if 256 <= alt <= 2560:
            scan_lengths.add(alt)
    # Add some specific interesting lengths
    for L in [384, 512, 640, 768, 896, 1024, 1280, 1536, 1920, 2048]:
        if 256 <= L <= 2560:
            scan_lengths.add(L)
    
    scan_lengths = sorted(scan_lengths)
    # Prioritize: close to current best, then spread out
    scan_lengths.sort(key=lambda x: (abs(x - n_best), x))
    
    print(f"  Scanning {min(len(scan_lengths), 12)} lengths...")
    candidates = []
    t0_scan = time.monotonic()
    
    for alt_n in scan_lengths[:12]:
        if time.monotonic() > deadline:
            break
        # Initialize: interpolate parent if close, else use structured
        if abs(alt_n - n_best) <= 64:
            init = interpolate_sequence(best_state['seq'], alt_n)
        else:
            variant = (alt_n // 32) % 7
            init = structured_init(alt_n, variant)
        
        # Quick subgradient: 2000 iterations
        a_sc, sc = nesterov_subgrad_refine(init, time_limit=8.0, n_max_iters=2000)
        candidates.append((sc, alt_n, a_sc))
        
        sc_eval = evaluate_safe(a_sc)
        if np.isfinite(sc_eval):
            candidates[-1] = (sc_eval, alt_n, a_sc)
    
    candidates.sort(key=lambda x: x[0])
    print(f"  Top 3 from scan: {[(f'{c[0]:.5f}', c[1]) for c in candidates[:3]]}")
    
    # Update best from scan
    for sc, alt_n, a_sc in candidates[:3]:
        try_update(best_state, a_sc, f"scan_n{alt_n}")

    # === Phase 2: Deep p-norm L-BFGS on best ===
    print(f"\n=== Phase 2: p-norm L-BFGS deep (best={best_state['score']:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        p_sched = [5.0, 20.0, 80.0, 300.0, 1000.0, 3000.0, 8000.0]
        a_pn = p_norm_lbfgs_refine(best_state['seq'], t_alloc, p_schedule=p_sched, maxiter_per_stage=300)
        try_update(best_state, a_pn, "pnorm_deep")

    # === Phase 3: Nesterov subgradient on best ===
    print(f"\n=== Phase 3: Nesterov SubGrad (best={best_state['score']:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sg, sc_sg = nesterov_subgrad_refine(best_state['seq'], t_alloc, n_max_iters=1000000)
        try_update(best_state, a_sg, "subgrad_nesterov")

    # === Phase 4: LP best-response ===
    print(f"\n=== Phase 4: Nesterov LP (best={best_state['score']:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc_lp = nesterov_lp_refine(best_state['seq'], t_alloc, solver=solver, n_max_iters=80000)
        try_update(best_state, a_lp, "nesterov_lp")

    # === Phase 5: Try the best candidate length from Phase 1 if different ===
    print(f"\n=== Phase 5: Alt length deep (best={best_state['score']:.7f}) ===")
    rem = deadline - time.monotonic()
    n_cur = len(best_state['seq'])
    
    tried_lengths = {n_cur}
    for sc_c, alt_n, a_c in candidates:
        if alt_n in tried_lengths:
            continue
        if abs(alt_n - n_cur) < 32:
            continue
        if rem < 100:
            break
        tried_lengths.add(alt_n)
        print(f"  Trying n={alt_n} (score={sc_c:.5f}), budget={min(rem*0.15, 120):.0f}s")
        
        # Interpolate current best to this length
        init_alt = interpolate_sequence(best_state['seq'], alt_n)
        
        t_sub = min(rem * 0.15, 120.0)
        a_alt = p_norm_lbfgs_refine(init_alt, t_sub, p_schedule=[20, 100, 500, 2000], maxiter_per_stage=200)
        improved = try_update(best_state, a_alt, f"alt_n{alt_n}")
        
        if improved and rem > 60:
            a_alt2, _ = nesterov_subgrad_refine(best_state['seq'], min(rem * 0.10, 60.0), n_max_iters=200000)
            try_update(best_state, a_alt2, f"alt_n{alt_n}_sg")
        
        rem = deadline - time.monotonic()
        if rem < 80:
            break

    # === Phase 6: Length fine-tuning around best ===
    print(f"\n=== Phase 6: Length fine-tune (best={best_state['score']:.7f}) ===")
    rem = deadline - time.monotonic()
    n_cur = len(best_state['seq'])
    
    if rem > 150:
        for delta in [-64, -32, 32, 64]:
            alt_n = n_cur + delta
            if alt_n < 256 or alt_n > 3000:
                continue
            if rem < 60:
                break
            rem = deadline - time.monotonic()
            init_ft = interpolate_sequence(best_state['seq'], alt_n)
            t_sub = min(rem * 0.12, 80.0)
            a_ft = p_norm_lbfgs_refine(init_ft, t_sub, p_schedule=[30, 150, 600, 2000], maxiter_per_stage=150)
            try_update(best_state, a_ft, f"finetune_n{alt_n}")
            rem = deadline - time.monotonic()

    # === Phase 7: Another round of subgradient + LP on current best ===
    print(f"\n=== Phase 7: Second round (best={best_state['score']:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_sub = min(rem * 0.20, 200.0)
        a_sg2, _ = nesterov_subgrad_refine(best_state['seq'], t_sub, n_max_iters=500000)
        try_update(best_state, a_sg2, "subgrad_r2")
        
        rem = deadline - time.monotonic()
        if rem > 60:
            t_lp = min(rem * 0.15, 150.0)
            a_lp2, _ = nesterov_lp_refine(best_state['seq'], t_lp, solver=solver, n_max_iters=40000)
            try_update(best_state, a_lp2, "lp_r2")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final polish (best={best_state['score']:.7f}) ===")
    rem = deadline - time.monotonic()
    
    # High-p L-BFGS
    if rem > 40:
        t_fin = min(rem * 0.20, 80.0)
        a_fin = p_norm_lbfgs_refine(best_state['seq'], t_fin, 
                                     p_schedule=[500, 2000, 8000, 20000], maxiter_per_stage=200)
        try_update(best_state, a_fin, "final_pnorm")

    # Peak shaving
    rem = deadline - time.monotonic()
    if rem > 30:
        t_ps = min(rem * 0.25, 100.0)
        a_ps, sc_ps = local_peak_shaving(best_state['seq'], t_ps)
        try_update(best_state, a_ps, "peak_shave")

    # One more subgradient round
    rem = deadline - time.monotonic()
    if rem > 30:
        t_sg = min(rem * 0.30, 80.0)
        a_sg3, _ = nesterov_subgrad_refine(best_state['seq'], t_sg, n_max_iters=300000)
        try_update(best_state, a_sg3, "final_subgrad")

    # Final LP
    rem = deadline - time.monotonic()
    if rem > 20:
        t_lp_fin = min(rem * 0.40, 60.0)
        a_lp_fin, _ = nesterov_lp_refine(best_state['seq'], t_lp_fin, solver=solver, n_max_iters=20000)
        try_update(best_state, a_lp_fin, "final_lp")

    # === Final verification ===
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs
        print(f"\nFINAL: score={best_state['score']:.7f}, n={len(best_state['seq'])}")
    else:
        # Fallback
        best_state['seq'] = parent
        best_state['score'] = evaluate_safe(parent)
        print(f"\nFALLBACK: score={best_state['score']:.7f}")

    return [float(x) for x in best_state['seq']]
