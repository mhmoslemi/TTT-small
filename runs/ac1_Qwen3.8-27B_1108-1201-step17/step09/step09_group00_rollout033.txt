def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_pair(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)

    T = np.zeros((size, n))
    for j in range(n):

    return T

def project_simplex(v):
    """Project vector v onto the probability simplex (sum=1, v>=0)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def lp_best_response(a_fixed, solver=None):

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solved = False
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True
    except Exception:
        pass

        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue

                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue

        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=5000):

            # Gradient of logsumexp(b/T) w.r.t. a is sum_k w[k] * d(b[k])/da

def structured_init(n, rng):

        # Three-level step

    elif choice == 2:
        # Smooth step (sigmoid-like)
        a = 1.0 / (1.0 + np.exp(-8 * (x - 0.5)))
    elif choice == 3:
        # Linear ramp
        a = 0.2 + 0.8 * x

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)

    if s < 0.01:

    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    # === Phase 1: Multi-start initialization ===
    print(f"\n=== Phase 1: Initialization (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_init = min(rem * 0.05, 30.0)
        # Try a few structured inits at same n
        for i in range(4):
            a_init = structured_init(n_best, rng)
            # Quick active-set optimization
            a_quick, _ = active_set_lbgfs(a_init, t_init / 4.0, threshold=0.95, maxiters=2000)
            try_update(a_quick, f"init_{i}")
    
    # === Phase 2: Active-set L-BFGS (main optimization) ===
    print(f"\n=== Phase 2: Active-Set L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 550.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_as, _ = active_set_lbgfs(best_seq, t_alloc, threshold=0.97, maxiters=30000)
        try_update(a_as, "active_set_1")
        
        # Second pass with tighter threshold
        rem = deadline - time.monotonic()
        if rem > 30:
            a_as2, _ = active_set_lbgfs(best_seq, min(rem * 0.3, 150.0), threshold=0.99, maxiters=20000)
            try_update(a_as2, "active_set_2")
    
    # === Phase 3: Nesterov LP refinement ===
    print(f"\n=== Phase 3: Nesterov LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_lp, "nesterov_lp")
    
    # === Phase 4: LogSumExp L-BFGS polishing ===
    print(f"\n=== Phase 4: LogSumExp L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_ls = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=6000)
        try_update(a_ls, "logsumexp")
    
    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: Coordinate Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        a_cd, _ = coordinate_descent_polish(best_seq, t_alloc, n_steps=50000)
        try_update(a_cd, "coord_desc")
    
    # === Phase 6: Second active-set pass with different threshold ===
    print(f"\n=== Phase 6: Active-Set Pass 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 20:
        a_as3, _ = active_set_lbgfs(best_seq, t_alloc, threshold=0.995, maxiters=15000)
        try_update(a_as3, "active_set_3")
    
    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        # Try n * 1.25
        alt_n1 = int(n_best * 1.25)
        t_alt = min(rem * 0.25, 200.0)
        print(f"  Trying n={alt_n1}, budget={t_alt:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n1)
        a_alt, _ = active_set_lbgfs(init_alt, t_alt * 0.4, threshold=0.95, maxiters=10000)
        try_update(a_alt, f"as_n{alt_n1}")
        
        rem = deadline - time.monotonic()
        if rem > 30:
            a_alt2, _ = nesterov_lp_refine(best_seq, min(rem * 0.4, 100.0), solver=solver, n_max_iters=30000)
            try_update(a_alt2, f"lp_n{alt_n1}")
        
        # Try n * 0.8
        rem = deadline - time.monotonic()
        if rem > 80:
            alt_n2 = int(n_best * 0.8)
            t_alt2 = min(rem * 0.2, 120.0)
            print(f"  Trying n={alt_n2}, budget={t_alt2:.0f}s")
            init_alt2 = interpolate_sequence(best_seq, alt_n2)
            a_alt3, _ = active_set_lbgfs(init_alt2, t_alt2 * 0.5, threshold=0.95, maxiters=8000)
            try_update(a_alt3, f"as_n{alt_n2}")
    
    # === Phase 8: Perturbation escape ===
    print(f"\n=== Phase 8: Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_pert = min(rem * 0.25, 150.0)
        n_cur = len(best_seq)
        x = np.linspace(0, 2 * np.pi, n_cur)
        
        # Try different perturbation strategies
        for freq in [1, 2, 3, 5]:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            perturbation = 0.015 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
            perturbed = best_seq + perturbation
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps < 1e-15:
                continue
            perturbed = perturbed / ps
            a_pert, _ = active_set_lbgfs(perturbed, t_pert / 5.0, threshold=0.95, maxiters=5000)
            try_update(a_pert, f"pert_f{freq}")
    
    # === Phase 9: Final comprehensive polish ===
    print(f"\n=== Phase 9: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Active set with very tight threshold
        t_fin = min(rem * 0.3, 60.0)
        if t_fin > 10:
            a_fin, _ = active_set_lbgfs(best_seq, t_fin, threshold=0.998, maxiters=10000)
            try_update(a_fin, "final_as")
        
        rem = deadline - time.monotonic()
        if rem > 15:
            # Quick LP
            t_lp = min(rem * 0.4, 40.0)
            a_fin2, _ = nesterov_lp_refine(best_seq, t_lp, solver=solver, n_max_iters=10000)
            try_update(a_fin2, "final_lp")
        
        rem = deadline - time.monotonic()
        if rem > 10:
            # Very low temp LogSumExp
            t_ls = min(rem * 0.5, 30.0)
            a_fin3 = logsumexp_lbgfs(best_seq, t_ls,
                                      T_stages=[1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01],
                                      maxiter_per_stage=3000)
            try_update(a_fin3, "final_ls")
        
        rem = deadline - time.monotonic()
        if rem > 5:
            # Final coordinate descent
            t_cd = min(rem * 0.6, 20.0)
            a_fin4, _ = coordinate_descent_polish(best_seq, t_cd, n_steps=20000)
            try_update(a_fin4, "final_cd")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
