    while time.monotonic() - t0 < time_limit:
        it += 1
        frac = (time.monotonic() - t0) / total_time
        # Adaptive temperature: start smooth, end sharp
        epsilon = epsilon_start * (epsilon_end / epsilon_start) ** frac

        # Nesterov step
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        y_new = a + momentum * (a - best_a)  # simplified momentum
        y_new = simplex_project(y_new)

        # Compute gradient at y
        conv_y = fast_convolve(y_new)
        max_cv = np.max(conv_y)
        shifted = conv_y / max(epsilon, 1e-10) - max_cv / max(epsilon, 1e-10)
        w = np.exp(shifted)
        w = w / w.sum()
        grad = compute_smoothed_grad(y_new, conv_y, w)

        # Gradient step with projection
        a_new = simplex_project(y_new - lr * grad)

        # Line search: try a few steps
        best_local_score = 2.0 * n * float(fast_convolve(a_new).max())
        for alpha in [0.5, 0.25, 0.1]:
            trial = simplex_project(y_new - lr * alpha * grad)
            ts = 2.0 * n * float(fast_convolve(trial).max())
            if ts < best_local_score:
                best_local_score = ts
                a_new = trial

        a = a_new
        cur_score = best_local_score

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.8
                no_improve = 0

        # Adapt learning rate
        if it % 100 == 0:
            lr = max(0.001, min(0.05, lr * (1.0 + 0.01 * np.sign(best_score - cur_score))))

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSGD {label} it={it} ({elapsed:.0f}s) eps={epsilon:.4f}: best={best_score:.8f}")

    return best_a, best_score


def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent with golden section 1D search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            if has_quad:
                quad_idx = 2 * i - k_lo

            # Golden section search
            best_x = _golden_search_coord(c_vals, a_partners, unaff_max, has_quad, quad_idx, current_sum, v, n, 0.0, max(3.0 * v, 5.0 / n))

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes
                a[i] = best_x
                current_sum += delta

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _golden_search_coord(c_vals, a_partners, unaff_max, has_quad, quad_idx, current_sum, v, n, lo, hi):
    """Golden section search for optimal value of a single coordinate."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _coord_score(c, c_vals, a_partners, unaff_max, has_quad, quad_idx, current_sum, v, n)
    fd = _coord_score(d, c_vals, a_partners, unaff_max, has_quad, quad_idx, current_sum, v, n)
    for _ in range(35):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = _coord_score(c, c_vals, a_partners, unaff_max, has_quad, quad_idx, current_sum, v, n)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = _coord_score(d, c_vals, a_partners, unaff_max, has_quad, quad_idx, current_sum, v, n)
    return (lo + hi) * 0.5


def _coord_score(x, c_vals, a_partners, unaff_max, has_quad, quad_idx, current_sum, v, n):
    """Score for a single coordinate value."""
    ns = current_sum + (x - v)
    if ns < 0.01:
        return 1e10
    nv = c_vals + 2.0 * x * a_partners
    if has_quad:
        nv[quad_idx] = c_vals[quad_idx] + x * x
    nm = max(unaff_max, float(nv.max()))
    return 2.0 * n * nm / (ns * ns)


def peak_shaving(a0, time_limit, verbose=False, label=""):
    """Peak-targeted mass redistribution: directly attack the maximum lag."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    conv = fast_convolve(a).copy()
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        max_lag = int(np.argmax(conv))
        max_val = float(conv[max_lag])

        # Find top 3 lags
        top_lags = np.argsort(conv)[-3:]

        # Compute pressure on each coordinate
        pressure = np.zeros(n)
        for k in top_lags:
            weight = (conv[k] - max_val + 0.01) / 0.01  # higher for hotter lags
            weight = max(weight, 1.0)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                pressure[i] += weight * a[j]

        # Identify source (high pressure) and target (low pressure) coordinates
        pressure_sorted = np.argsort(pressure)
        n_move = max(1, n // 16)
        sources = pressure_sorted[-n_move:]  # high pressure
        targets = pressure_sorted[:n_move]    # low pressure

        # Move mass
        total_src_mass = a[sources].sum()
        if total_src_mass < 1e-12:
            break
        frac = 0.02 + 0.03 * (it % 5) / 5.0  # adaptive fraction
        move_amount = min(frac * total_src_mass, 0.1 / n_move)

        # Distribute proportionally
        src_weights = a[sources] / total_src_mass
        tgt_weights = np.ones(n_move)
        tgt_weights = tgt_weights / tgt_weights.sum()

        for idx in range(n_move):
            src_i = int(sources[idx])
            tgt_i = int(targets[idx])
            amt = move_amount * src_weights[idx]
            amt = min(amt, a[src_i] * 0.5)
            a[src_i] -= amt
            a[tgt_i] += amt

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        cur_score = 2.0 * n * float(fast_convolve(a).max())
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PS {label} it={it} ({elapsed:.0f}s): best={best_score:.8f}, max_lag={max_lag}")

    return best_a, best_score


def block_coord_descent(a0, time_limit, block_size=8, verbose=False, label=""):
    """Block coordinate descent: update blocks of coordinates simultaneously."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n

    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for start in range(0, n, block_size):
            if time.monotonic() - t0 > time_limit:
                break
            end = min(start + block_size, n)
            indices = np.arange(start, end)
            n_block = end - start

            # For each coordinate in block, try improving
            for i in indices:
                if a[i] < 1e-14:
                    continue
                v = a[i]
                k_lo = i
                k_hi = min(i + n - 1, 2 * n - 2)

                unaff_max = -1e30
                if k_lo > 0:
                    unaff_max = max(unaff_max, float(conv[:k_lo].max()))
                if k_hi < 2 * n - 2:
                    unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

                partners = np.arange(k_lo, k_hi + 1) - i
                a_partners = a[partners]
                has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

                c_vals = np.empty(k_hi - k_lo + 1)
                for idx in range(len(c_vals)):
                    k = k_lo + idx
                    if k == 2 * i:
                        c_vals[idx] = conv[k] - v * v
                    else:
                        c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

                if has_quad:

                best_x = _golden_search_coord(c_vals, a_partners, unaff_max, has_quad, quad_idx, current_sum, v, n, 0.0, max(3.0 * v, 5.0 / n))

                if abs(best_x - v) > 1e-13:
                    delta = best_x - v
                    changes = 2.0 * delta * a_partners
                    if has_quad:
                        changes[quad_idx] = delta * (2.0 * v + delta)
                    conv[k_lo:k_hi + 1] += changes
                    a[i] = best_x
                    current_sum += delta

        cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)

            print(f"    BCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.65)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_interpolate(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def basin_hopping(a0, time_limit, cd_func, rng, n_perturbs=8, verbose=False):
    """Basin hopping: perturb and re-optimize."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break
        t_alloc = min(rem * 0.5, 50.0)

        # Various perturbation strategies
        strategy = p % 3
        if strategy == 0:
            # Gaussian noise
            sigma = 0.015 + 0.02 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif strategy == 1:
            # Peak-breaking: reduce top conv contributions
            conv_p = fast_convolve(a)
            top_k = np.argsort(conv_p)[-5:]
            perturbed = a.copy()
            for k in top_k:
                i_start = max(0, int(k) - n + 1)
                i_end = min(n - 1, int(k))
                idx = np.arange(i_start, i_end + 1)
                picks = rng.choice(idx, size=max(1, len(idx) // 4), replace=False)
                for ii in picks:
                    perturbed[ii] *= (1.0 + rng.standard_normal() * 0.04)
        else:
            # Random rescaling of random subset
            perturbed = a.copy()
            subset = rng.choice(n, size=n // 4, replace=False)
            scales = rng.uniform(0.7, 1.4, len(subset))
            perturbed[subset] *= scales

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = cd_func(perturbed, t_alloc)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p} strat={strategy}: improved to {best_score:.8f}")

    return best_a, best_score


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def fine_polish(a0, time_limit, label=""):
    """Fine coordinate descent with very small multiplicative steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            if has_quad:
                quad_idx = 2 * i - k_lo

            # Try very small multiplicative steps
            best_local_max = float(nv_max(c_vals, a_partners, v, has_quad, quad_idx, unaff_max))
            best_delta = 0.0

            for factor in [0.999, 0.998, 0.997, 0.995, 0.99, 1.005, 1.002, 1.001]:
                x = v * factor
                if x < -1e-15:
                    continue
                local_max = nv_max(c_vals, a_partners, x, has_quad, quad_idx, unaff_max)
                if local_max < best_local_max - 1e-16:
                    best_local_max = local_max
                    best_delta = x - v

            if best_delta != 0.0:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                current_sum += best_delta

                changes = 2.0 * best_delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = best_delta * (2.0 * v + best_delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

    return best_a, best_score


def nv_max(c_vals, a_partners, x, has_quad, quad_idx, unaff_max):
    """Compute max of affected convolution values for given x."""
    nv = c_vals + 2.0 * x * a_partners
    if has_quad:
        nv = nv.copy()
        nv[quad_idx] = c_vals[quad_idx] + x * x
    return max(unaff_max, float(nv.max()))

            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ecd_r{rnd}")

            if best_score > 1.5035:
                # Perturb to escape local optimum
                sigma = 0.02 + 0.01 * rnd
                perturbed = a_opt + rng.standard_normal(n_best) * sigma
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
                current = perturbed
            else:
                current = a_opt.copy()

    # === Phase 3: Peak shaving (novel) ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_ps = min(rem * 0.5, 150.0)
            print(f"  Peak shaving, budget={t_ps:.0f}s")
            a_ps, s_ps = peak_shaving(best_seq, t_ps, verbose=True, label="ph3")
            best_seq, best_score = check_and_update(a_ps, best_seq, best_score, "peak_shave")

            # Follow up with ECD
            rem = deadline - time.monotonic() - 250
            if rem > 20:
                t_cd = min(rem * 0.4, 80.0)
                a_cd, s_cd, _ = exact_coord_descent(a_ps, t_cd, label="post_ps")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "ecd_after_ps")

    # === Phase 4: Try other lengths ===
    print(f"\n=== Phase 4: Multi-length ECD (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Try a few other lengths
        alt_lengths = [n_best]
        for alt in [1024, 1280, 1536, 2048]:
            if alt != n_best and 512 <= alt <= 2560:
                alt_lengths.append(alt)
        alt_lengths = list(set(alt_lengths))[:3]

        for n_alt in alt_lengths[1:]:  # skip the current best length
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            t_alloc = min(rem * 0.3, 80.0)
            print(f"  Trying n={n_alt}, budget={t_alloc:.0f}s")

            init_alt = init_interpolate(best_seq, n_alt)
            a_opt, s_opt, _ = exact_coord_descent(init_alt, t_alloc, label=f"alt_n{n_alt}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"alt_n{n_alt}")

    # === Phase 5: Basin hopping ===
    print(f"\n=== Phase 5: Basin Hopping (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_bh = min(rem * 0.6, 120.0)
            print(f"  Basin hopping, budget={t_bh:.0f}s")
            a_bh, s_bh = basin_hopping(best_seq, t_bh, exact_coord_descent, rng,
                                       n_perturbs=6, verbose=True)
            best_seq, best_score = check_and_update(a_bh, best_seq, best_score, "basin_hop")

            # ECD after basin hopping
            rem = deadline - time.monotonic() - 120
            if rem > 20:
                a_cd2, s_cd2, _ = exact_coord_descent(a_bh, min(rem * 0.5, 60.0), label="post_bh")
                best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "ecd_post_bh")

    # === Phase 6: Final intensive ECD ===
    print(f"\n=== Phase 6: Final ECD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_final = min(rem * 0.6, 100.0)
            print(f"  Final ECD, budget={t_final:.0f}s")
            a_f, s_f, _ = exact_coord_descent(best_seq, t_final, verbose=True, label="final")
            best_seq, best_score = check_and_update(a_f, best_seq, best_score, "final_ecd")

    # === Phase 7: Fine polishing ===
    print(f"\n=== Phase 7: Fine Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_fine = min(rem * 0.7, 40.0)
            print(f"  Fine polish, budget={t_fine:.0f}s")
            a_fin, s_fin = fine_polish(best_seq, t_fine, label="fin")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "fine")

            # One more quick ECD
            rem2 = deadline - time.monotonic() - 5
            if rem2 > 10:
                a_last, s_last, _ = exact_coord_descent(a_fin, rem2 * 0.8, label="last")
                best_seq, best_score = check_and_update(a_last, best_seq, best_score, "last")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
