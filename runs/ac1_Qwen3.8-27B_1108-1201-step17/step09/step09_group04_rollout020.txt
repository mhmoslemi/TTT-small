"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive p-norm smoothing with L-BFGS in log-domain
2. Sequential Convex Programming (SCP) with LP solving
3. Multiplicative weight subgradient method
4. Multi-start with diverse initializations
5. Local refinement via coordinate descent
6. Length exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def normalize(a, target_sum=1.0):
    """Normalize a to have the target sum, keeping non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_sum / len(a)
    return a * (target_sum / s)


def p_norm_val(a, p):
    """Compute (sum max(c,0)^p)^(1/p) for c = convolve(a,a)."""
    c = fast_convolve(a)
    c_pos = np.maximum(c, 0.0)
    return (np.sum(c_pos ** p)) ** (1.0 / p)


def p_norm_grad_logdomain(z, p):
    """Gradient of 2n * (sum c_k^p)^(1/p) w.r.t. log-domain z (softmax parameterization)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = np.power(c_pos, p)
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = np.power(c_pos, p - 1.0)
    # Gradient w.r.t. a: 2 * correlate(w, a, 'valid')
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    # Gradient w.r.t. z via softmax: a * (g_a - dot(g_a, a))
    return a * (g_a - np.dot(g_a, a))


def p_norm_lbfgs(a0, time_limit, p_start=10.0, p_end=500.0, n_stages=8):
    """L-BFGS with adaptive p-norm smoothing in log-domain."""
    n = len(a0)
    a = normalize(a0)
    z = np.log(np.maximum(a, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    temps = np.geomspace(p_start, p_end, n_stages)
    iters_per_stage = max(50, 5000 // n_stages)

    for stage_i, stage_p in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_val):
            zm = z_val.max()
            ez = np.exp(z_val - zm)
            aa = ez / ez.sum()
            c = fast_convolve(aa)
            cp_ = np.maximum(c, 1e-300)
            return 2.0 * len(aa) * (np.sum(cp_ ** stage_p)) ** (1.0 / stage_p)

        def jac(z_val):
            return p_norm_grad_logdomain(z_val, stage_p)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-15,
                         'gtol': 1e-10, 'maxfun': iters_per_stage * 100}
            )
            z = res.x
        except Exception:
            pass

        # Evaluate
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        s = score_normalized(a_cur)
        elapsed = time.monotonic() - t0
        print(f"    p-norm L-BFGS stage {stage_i+1}/{n_stages} p={stage_p:.1f} ({elapsed:.1f}s): score={s:.6f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


def build_conv_matrix(a):
    """Build matrix T where T @ b = convolve(a, b) (full)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def solve_scp_lp(a_fixed):
    """
    Solve the SCP LP:
    min M
    s.t. 2*(T_a @ b)[k] - c_a[k] <= M for all k
         sum(b) = 1, b >= 0
    where T_a is the conv matrix of a_fixed and c_a = convolve(a_fixed, a_fixed).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    c_a = fast_convolve(a_fixed)
    # Constraint: 2*(T @ b) - c_a <= M  =>  2*(T @ b) - M <= c_a
    # Rearranged: T @ b - M/2 <= c_a / 2
    # Or: 2*(T @ b) - M <= c_a
    constraints = [2.0 * (T @ b_var) - M_var <= c_a, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            prob.solve(solver=cp.SCS)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def scp_optimize(a0, time_limit, damping=0.5, max_iters=200):
    """Sequential Convex Programming with LP."""
    a = normalize(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    n = len(a)
    t0 = time.monotonic()
    # Precompute T once (it changes each iteration, but we rebuild)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        t_start = time.monotonic()
        
        b_new = solve_scp_lp(a)
        if b_new is None:
            break
        
        # Damped update
        a_new = (1 - damping) * a + damping * b_new
        a_new = normalize(a_new)
        
        s_new = score_normalized(a_new)
        if s_new < best_score:
            best_score = s_new
            best_a = a_new.copy()
        
        a = a_new
        
        if (it + 1) % 20 == 0:
            elapsed = time.monotonic() - t0
            t_iter = time.monotonic() - t_start
            print(f"    SCP iter {it+1} ({elapsed:.1f}s, {t_iter:.2f}s/iter): score={best_score:.6f}")
        
        # Check for convergence
        if it > 5 and abs(score_normalized(a) - best_score) < 1e-10:
            # Try smaller damping
            damping = max(0.1, damping * 0.5)
    
    return best_a


def multiplicative_weight_optimize(a0, time_limit, n_iters=5000):
    """Multiplicative weight updates targeting the convolution peaks."""
    a = normalize(a0)
    n = len(a)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    eta = 1.0 / n
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        k_max = np.argmax(c)
        
        # Gradient of c[k_max] w.r.t. a: 2 * a[k_max - i] (shifted/reversed)
        grad = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Multiplicative weight update
        a = a * np.exp(-eta * grad)
        a = normalize(a)
        
        s = score_normalized(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()
        
        # Adaptive step size
        if it % 100 == 0 and it > 0:
            eta = eta * 0.95  # slow decay
            # Or adaptive: increase if improving, decrease if not
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW iter {it+1} ({elapsed:.1f}s): score={best_score:.6f}")
    
    return best_a


def local_refinement(a0, time_limit, max_steps=50000):
    """Coordinate descent + peak shaving local refinement."""
    a = normalize(a0)
    n = len(a)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        # Find the peak (excluding endpoints which are just a[0]^2 and a[n-1]^2)
        if len(c) > 2:
            c_inner = c[1:-1].copy()
            peak_idx = np.argmax(c_inner) + 1
        else:
            peak_idx = np.argmax(c)
        peak_val = c[peak_idx]
        
        improved = False
        
        # Strategy 1: Reduce the largest contributors to the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_idx - idx]
        
        # Find top contributors
        top = np.argsort(contribs)[::-1][:10]
        
        for ci in top:
            i = idx[ci]
            j = peak_idx - i
            if j < 0 or j >= n or j == i:
                continue
            if a[i] < 1e-14 or a[j] < 1e-14:
                continue
            
            # Try reducing one of the pair
            for factor in [0.95, 0.90, 0.85, 0.80]:
                a_trial = a.copy()
                a_trial[i] *= factor
                diff = a[i] * (1 - factor)
                # Redistribute to others
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask].sum()
                if others < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others
                s = score_normalized(a_trial)
                if s < best_score - 1e-12:
                    a = a_trial
                    best_score = s
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        # Strategy 2: Spread mass from peak contributors to non-contributors
        if not improved and step % 10 == 0:
            # Find indices NOT contributing to the peak
            non_contrib = np.setdiff1d(np.arange(n), idx)
            if len(non_contrib) > 0:
                # Move a small amount from the largest contributor to a random non-contributor
                ci = top[0]
                i = idx[ci]
                j_dest = non_contrib[np.random.randint(len(non_contrib))]
                for frac in [0.01, 0.02, 0.05]:
                    amount = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amount
                    a_trial[j_dest] += amount
                    a_trial = np.maximum(a_trial, 0)
                    s_trial = score_normalized(a_trial)
                    if s_trial < best_score - 1e-12:
                        a = a_trial
                        best_score = s_trial
                        best_a = a.copy()
                        improved = True
                        break
        
        # Strategy 3: Random perturbation
        if not improved and step % 50 == 0:
            n_pert = np.random.randint(1, 5)
            indices = np.random.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in indices:
                a_trial[i] *= (1.0 + np.random.uniform(-0.05, 0.05))
            a_trial = normalize(a_trial)
            s_trial = score_normalized(a_trial)
            if s_trial < best_score - 1e-12:
                a = a_trial
                best_score = s_trial
                best_a = a.copy()
                improved = True
        
        if not improved and step > 100:
            # Try a few more steps before giving up
            if step % 200 == 0 and not improved:
                break
    
    return best_a


def structured_init(n, variant=0, rng=None):
    """Create structured initial sequence of length n, normalized to sum=1."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        a = np.exp(-4 * x * (1 - x))
    elif variant == 2:
        x2 = (x - 0.5) * 2
        a = np.exp(-x2 ** 2 * 2)
    elif variant == 3:
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))  # sigmoid
    elif variant == 4:
        a = np.ones(n)
    else:
        a = rng.random(n)
    a = np.maximum(a, 0)
    return normalize(a)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(f_interp)


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_score, best_seq, a, label, verbose=True):
    """Try to update best with candidate a. Returns (improved, new_best_score, new_best_seq)."""
    a = normalize(a)
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        if verbose:
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True, sc, a.copy()
    return False, best_score, best_seq

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive p-norm smoothing with L-BFGS in log-domain (softmax parameterization)
2. Sequential Convex Programming (SCP) with LP subproblems
3. Multiplicative weight (mirror descent) subgradient method
4. Adam optimizer in log-domain for robustness
5. Multi-start perturbation for escaping local optima
6. Local refinement via coordinate descent and peak shaving
7. Length exploration around the best found dimension
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Full linear auto-convolution via FFT. O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def normalize(a, target_sum=1.0):
    """Normalize a to have target_sum, keeping non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_sum / len(a)
    return a * (target_sum / s)


def p_norm_grad_logdomain(z, p):
    """Gradient of 2n * (sum c_k^p)^(1/p) w.r.t. log-domain z (softmax parameterization)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = np.power(c_pos, p)
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = np.power(c_pos, p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def p_norm_lbfgs(a0, time_limit, p_start=10.0, p_end=500.0, n_stages=8):
    """L-BFGS with adaptive p-norm smoothing in log-domain (softmax)."""
    n = len(a0)
    a = normalize(a0)
    z = np.log(np.maximum(a, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    temps = np.geomspace(p_start, p_end, n_stages)
    iters_per_stage = max(80, 6000 // n_stages)

    for stage_i, stage_p in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_val):
            zm = z_val.max()
            ez = np.exp(z_val - zm)
            aa = ez / ez.sum()
            c = fast_convolve(aa)
            cp_ = np.maximum(c, 1e-300)
            return 2.0 * len(aa) * (np.sum(np.power(cp_, stage_p))) ** (1.0 / stage_p)

        def jac(z_val):
            return p_norm_grad_logdomain(z_val, stage_p)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-15,
                         'gtol': 1e-10, 'maxfun': iters_per_stage * 150}
            )
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        s = score_normalized(a_cur)
        elapsed = time.monotonic() - t0
        print(f"    p-LBFGS stage {stage_i+1}/{n_stages} p={stage_p:.1f} ({elapsed:.1f}s): score={s:.6f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


def p_norm_adam(a0, time_limit, p=100.0, lr=0.05, n_iters=3000):
    """Adam optimizer in log-domain for p-norm objective."""
    n = len(a0)
    a = normalize(a0)
    z = np.log(np.maximum(a, 1e-300))
    z = z - z.max()
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    t0 = time.monotonic()
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_s = score_normalized(a)
    lr_schedule = [lr, lr * 0.5, lr * 0.25, lr * 0.1]
    lr_switch = n_iters // 4

    while t < n_iters:
        if time.monotonic() - t0 > time_limit:
            break
        t += 1
        current_lr = lr_schedule[min(t // lr_switch, 3)]
        g = p_norm_grad_logdomain(z, p)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - current_lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - z.max()

        if t % 200 == 0:
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            s = score_normalized(a_cur)
            if s < best_s:
                best_s = s
                best_a = a_cur.copy()

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a_final = exp_z / exp_z.sum()
    # Return the better of final and best tracked
    if score_normalized(a_final) < best_s:
        return a_final
    return best_a


def build_conv_matrix(a):
    """Build matrix T (2n-1 x n) where T @ b = convolve(a, b) full."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_chosen = None


def get_lp_solver():

    if _lp_solver_chosen is not None:
        return _lp_solver_chosen
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_chosen = solver

def solve_scp_lp(a_fixed):
    """
    Solve SCP LP:
    min M
    s.t. 2*(T_a @ b) - c_a[k] <= M for all k
         sum(b) = 1, b >= 0
    """
    solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    c_a = fast_convolve(a_fixed)
    constraints = [2.0 * (T @ b_var) - M_var <= c_a, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def solve_best_response_lp(a_fixed):
    """
    Solve: min M s.t. (T_a @ b)[k] <= M for all k, sum(b)=1, b>=0
    (The standard best-response LP)
    """
    solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def scp_optimize(a0, time_limit, damping=0.6, max_iters=150):
    """Sequential Convex Programming with LP subproblems."""
    a = normalize(a0)
    n = len(a)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    prev_score = best_score
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        t_start = time.monotonic()

        b_new = solve_scp_lp(a)
        if b_new is None:
            break

        a_new = (1 - damping) * a + damping * b_new
        a_new = normalize(a_new)

        s_new = score_normalized(a_new)
        if s_new < best_score - 1e-12:
            best_score = s_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new

        if stall > 20:
            damping = max(0.2, damping * 0.7)
            stall = 0

        if (it + 1) % 15 == 0:
            elapsed = time.monotonic() - t0
            t_iter = time.monotonic() - t_start
            print(f"    SCP iter {it+1}/{max_iters} ({elapsed:.1f}s, {t_iter:.2f}s/iter): best={best_score:.6f}")

    return best_a

def multiplicative_weight_optimize(a0, time_limit, max_iters=10000):
    """Multiplicative weight (mirror descent) targeting convolution peaks."""
    a = normalize(a0)
    n = len(a)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 1.0 / (2.0 * n)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        # Focus on interior peaks (avoid trivial endpoint peaks)
        if n > 20:
            search_start = n // 4
            search_end = 3 * n // 4 + n // 2
            c_search = c[search_start:min(search_end, len(c))]
            k_max = np.argmax(c_search) + search_start
        else:
            k_max = np.argmax(c)

        # Gradient of c[k_max] w.r.t. a
        grad = np.zeros(n)
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * a[k_max - idx]

        # Multiplicative weight update
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        s = score_normalized(a)
        if s < best_score - 1e-12:
            best_score = s
            best_a = a.copy()

        # Adaptive eta: decrease slowly
        if (it + 1) % 500 == 0:
            eta *= 0.9
            elapsed = time.monotonic() - t0
            print(f"    MW iter {it+1} ({elapsed:.1f}s): score={best_score:.6f}")

    return best_a

                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * factor
                diff = old_val - a_trial[i]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                s = score_normalized(a_trial)
                if s < best_score - 1e-12:
                    a = a_trial
                    best_score = s
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
            if improved:
                break

        # Strategy 2: Pair modification at peak
        if not improved:
            for ci in top_contribs[:6]:
                i = idx[ci]
                j = peak_idx - i
                if i == j or j < 0 or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.93, 0.88]:
                    for f2 in [0.93, 0.88]:
                        a_trial = a.copy()
                        old_i = a_trial[i]
                        old_j = a_trial[j]
                        a_trial[i] = old_i * f1
                        a_trial[j] = old_j * f2
                        diff = (old_i - a_trial[i]) + (old_j - a_trial[j])
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                        mask[j] = False
                        others_sum = a_trial[mask].sum()
                        if others_sum < 1e-15:
                            continue
                        a_trial[mask] += diff * a_trial[mask] / others_sum
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial = a_trial / st
                        s = score_normalized(a_trial)
                        if s < best_score - 1e-12:
                            a = a_trial
                            best_score = s
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
                    if improved:
                        break
                if improved:
                    break

        # Strategy 3: Move mass from contributors to non-contributors
        if not improved and step % 20 == 0:
            non_contrib = np.setdiff1d(np.arange(n), idx)
            if len(non_contrib) > 2:
                ci = top_contribs[0]
                i = idx[ci]
                if a[i] > 1e-6:
                    # Pick a random non-contributor
                    rng = np.random.default_rng(step * 7 + 1)
                    j_dest = non_contrib[rng.integers(len(non_contrib))]
                    for frac in [0.005, 0.01, 0.02, 0.04]:
                        amount = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= amount
                        a_trial[j_dest] += amount
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial = a_trial / st
                        s = score_normalized(a_trial)
                        if s < best_score - 1e-12:
                            a = a_trial
                            best_score = s
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break

        # Strategy 4: Random perturbation
        if not improved and step % 100 == 0:
            rng = np.random.default_rng(step * 3 + 42)
            n_pert = rng.integers(2, max(3, n // 50) + 1)
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in indices:
                a_trial[i] *= (1.0 + rng.uniform(-0.03, 0.03))
            a_trial = normalize(a_trial)
            s = score_normalized(a_trial)
            if s < best_score - 1e-12:
                a = a_trial
                best_score = s
                best_a = a.copy()
                improved = True
                no_improve = 0

        if not improved:
            no_improve += 1
            if no_improve > 500:
                break

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    LocalRefine step={step} ({elapsed:.1f}s): score={best_score:.6f}")

    return best_a


def structured_init(n, variant=0, rng=None):
    """Create structured initial sequence, normalized to sum=1."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        a = np.exp(-4.0 * x * (1.0 - x))
    elif variant == 2:
        x2 = (x - 0.5) * 2.0
        a = np.exp(-x2 ** 2 * 2.0)
    elif variant == 3:
        a = 1.0 / (1.0 + np.exp(-12.0 * (x - 0.5)))
    elif variant == 4:
        a = np.ones(n)
    elif variant == 5:
        a = np.cos(np.pi * x / 2.0) ** 2
    else:
        a = rng.random(n)
    a = np.maximum(a, 0)
    return normalize(a)
