import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_convolve_ab(a, b):
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def sqp_step(a, c, M, k_peaks, a_min=0.0):
    """
    One SQP step: solve the linearized system and do a line search.
    
    Parameters:
    - a: current point (probability vector)
    - c: auto-convolution of a
    - M: max of c
    - k_peaks: indices of the active peaks
    - a_min: minimum value constraint (0 for non-negativity)
    
    Returns:
    - a_new: updated point
    - M_new: new max of auto-convolution
    - c_new: new auto-convolution
    """
    n = len(a)
    K = len(k_peaks)
    
    # Build the linear system
    # A is (K+1) x n, b is (K+1)
    A = np.zeros((K + 1, n))
    b = np.zeros(K + 1)
    
    for s, k in enumerate(k_peaks):
        # Gradient of (a*a)_k w.r.t. a: 2 * shift(a, k)
        # shift(a, k)_i = a[k-i] if 0 <= k-i <= n-1, else 0
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        A[s, i_start:i_end+1] = 2.0 * a[k - i_end:k - i_start + 1][::-1]
        b[s] = M - c[k]
    
    A[K, :] = 1.0  # normalization constraint
    b[K] = 0.0
    
    # Solve minimum-norm least squares: d = A^T (A A^T)^{-1} b
    AT = A.T  # n x (K+1)
    AAT = A @ AT  # (K+1) x (K+1)
    # Regularize for numerical stability
    AAT += 1e-12 * np.eye(K + 1)
    try:
        y = np.linalg.solve(AAT, b)
    except np.linalg.LinAlgError:
        y = np.linalg.lstsq(AAT, b, rcond=None)[0]
    d = AT @ y  # n-vector: the step direction
    
    # Line search
    best_alpha = 0.0
    best_M = M
    best_a = a.copy()
    
    # Find maximum alpha such that a + alpha*d >= 0
    if d.min() < -1e-15:
        alpha_max = -a[d < -1e-15].min() / (-d[d < -1e-15].min())
        alpha_max = min(alpha_max, 1.0)
    else:
        alpha_max = 1.0
    
    # Try various alphas
    for alpha in [1.0, 0.5, 0.75, 0.25, 0.6, 0.4, 0.8, 0.9, 0.3, 0.15]:
        if alpha > alpha_max:
            continue
        a_try = a + alpha * d
        if a_try.min() < -1e-15:
            continue
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        # We don't need to renormalize since the constraint ensures sum(d) = 0
        c_try = fast_convolve(a_try)
        M_try = c_try.max()
        if M_try < best_M:
            best_M = M_try
            best_alpha = alpha
            best_a = a_try.copy()
            best_c = c_try.copy()
    
    if best_alpha == 0.0:
        # Try smaller alphas
        for alpha in [0.1, 0.05, 0.02, 0.01, 0.005, 0.001]:
            if alpha > alpha_max:
                continue
            a_try = a + alpha * d
            if a_try.min() < -1e-15:
                continue
            a_try = np.maximum(a_try, 0.0)
            c_try = fast_convolve(a_try)
            M_try = c_try.max()
            if M_try < best_M:
                best_M = M_try
                best_alpha = alpha
                best_a = a_try.copy()
                best_c = c_try.copy()
    
    return best_a, best_M, best_c
