For the minimization over M to be bounded below, we need 1 - sum(nu) >= 0, i.e., sum(nu) <= 1.
At the optimum, sum(nu) = 1 (unless M = 0, which doesn't happen here).

import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_with(a_fixed, b):
    n = len(a_fixed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a_fixed, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def compute_weighted_subgradient(a, weight_power=2.0, min_weight=0.01):
    """Compute subgradient weighted by peak excess."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    mean_b = b.mean()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    excess = np.maximum(b - mean_b, 0)
    total_excess = excess.sum()
    if total_excess < 1e-30:
        weights = np.ones(len(b)) / len(b)
    else:
        weights = (excess / total_excess) ** weight_power
        weights = np.maximum(weights, min_weight / len(b))
        weights /= weights.sum()
    g = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for k in active:
        w = weights[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * w * a[k - idx]
    return g, max_b


def nesterov_subgradient_simplex(a0, time_limit, weight_power=2.0, momentum_decay=0.95):
    """Nesterov-accelerated subgradient method on the probability simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    v = np.zeros(n)
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    it = 0
    stall = 0
    prev_best = best_score

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        step_size = 1.0 / np.sqrt(1.0 + it)
        g, max_b = compute_weighted_subgradient(y, weight_power)

        v = momentum_decay * v + (1 - momentum_decay) * g
        y_new = y - step_size * v
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            break
        y_new = y_new / sy

        alpha = 2.0 / (it + 2)
        a_new = (1 - alpha) * a + alpha * y_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new / sa

        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        y = y_new

        if it % 2000 == 0:
            if best_score < prev_best - 1e-8:
                stall = 0
            prev_best = best_score
            elapsed = time.monotonic() - t0
            if elapsed > 5:
                print(f"    NSG iter={it}: score={best_score:.6f} lb={f_lb:.6f} t={elapsed:.0f}s")
            if stall >= 30:
                rng = np.random.default_rng(it * 31 + 7)
                noise = rng.random(n) * 0.01
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0

    return best_a, best_score


def lbgfs_refine_improved(a0, time_limit, T_start=500.0, T_end=0.05, n_stages=15):
    """L-BFGS with log-sum-exp, improved multi-stage annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    elapsed_total = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        frac = 1.0 - (si + 1) / n_stages
        iters = max(20, int(2000 * (0.3 + 0.7 * frac)))
        if si == 0:
            iters = min(iters, 300)
        elif si == n_stages - 1:
            iters = max(iters, 500)

        print(f"    L-BFGS stage {si+1}/{n_stages} T={stage_T:.2f} iters={iters}")

        def make_func(T_val):
            def func(z_in):
                if time.monotonic() - t0 > time_limit:
                    return 1e15
                a_tmp = softmax_from_z(z_in)
                b_tmp = fast_convolve(a_tmp)
                Tb = T_val * b_tmp
                max_Tb = Tb.max()
                exp_Tb = np.exp(Tb - max_Tb)
                log_sum_exp = max_Tb + np.log(exp_Tb.sum())
                return (2.0 * n / T_val) * log_sum_exp
            return func

        def make_jac(T_val):
            def jac(z_in):
                if time.monotonic() - t0 > time_limit:
                    return np.zeros_like(z_in)
                a_tmp = softmax_from_z(z_in)
                b_tmp = fast_convolve(a_tmp)
                Tb = T_val * b_tmp
                max_Tb = Tb.max()
                exp_Tb = np.exp(Tb - max_Tb)
                w = exp_Tb / exp_Tb.sum()
                grad_a = np.zeros(n)
                for k in range(len(b_tmp)):
                    if w[k] < 1e-15:
                        continue
                    js = max(0, k - n + 1)
                    je = min(n - 1, k)
                    idx = np.arange(js, je + 1)
                    grad_a[idx] += 2.0 * w[k] * a_tmp[k - idx]
                S = np.dot(grad_a, a_tmp)
                return a_tmp * (grad_a - S)
            return jac

        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def mass_transfer_search(a0, time_limit, n_steps=50000):
    """Local search: transfer mass from high-contributing to low-contributing positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 5000:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        threshold = peak_val - 0.03 * peak_val
        near_peaks = np.where(b >= threshold)[0]
        near_peaks = near_peaks[:5]

        if len(near_peaks) == 0:
            near_peaks = np.array([peak_k])

        contrib = np.zeros(n)
        for k in near_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]

        top_idx = np.argsort(contrib)[::-1][:8]
        low_idx = np.argsort(contrib)[:8]

        improved = False
        for si in top_idx:
            if a[si] < 1e-12:
                continue
            for di in low_idx:
                if si == di:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.08, 0.12]:
                    amount = a[si] * frac
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve % 500 == 0:
                for _ in range(5):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    amount = a[i] * rng.uniform(0.01, 0.05)
                    a_trial = a.copy()
                    a_trial[i] -= amount
                    a_trial[j] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=100000):
    """Nesterov-accelerated LP alternating method."""
    if solver is None:

        s = 1.0

    t0 = time.monotonic()
    stall_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            stall_count += 1
            if stall_count > 20:
                break
            continue
        stall_count = 0

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:

            a = a / sa

        if score_a < best_score:

            best_a = a.copy()

            if best_score < prev_score - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
            if stall_count >= 15:

                noise = rng.random(n) * 0.015

                y = np.maximum(y, 0)
                y = y / y.sum()

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 1.5
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.abs(np.sin(4 * np.pi * x))
    else:
        a = rng.random(n)
        a = a ** 0.5
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        sc = evaluate_sequence(a_list)

    except Exception:
        return float('inf')

        t_frac = 0.55
        a_alt = lbgfs_refine_improved(cur_best, t_total * t_frac, T_start=200.0, T_end=0.1, n_stages=10)
        try_update(a_alt, f"lbgfs_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 30:
            a_alt2 = nesterov_lp_refine(best_seq, min(rem * 0.3, 80.0), solver=solver, n_max_iters=20000)
            try_update(a_alt2, f"lp_n{alt_n}")

    # Phase 7: Final polish
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.35, 100.0)
        a_fin = lbgfs_refine_improved(best_seq, t_fin, T_start=20.0, T_end=0.02, n_stages=8)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 25:
        t_ls = min(rem * 0.3, 60.0)
        a_ls = mass_transfer_search(best_seq, t_ls, n_steps=30000)
        try_update(a_ls, "final_mt")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_nsg = min(rem * 0.3, 50.0)
        a_nsg2 = nesterov_subgradient_simplex(best_seq, t_nsg, weight_power=3.0)
        try_update(a_nsg2, "final_nsg")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
