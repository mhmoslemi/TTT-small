"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based minimax alternating
with Nesterov acceleration, subgradient polish, and multi-length exploration.
Key: efficient LP solves, adaptive step sizes, symmetry exploitation.

import numpy as np
import time
from scipy.fft import rfft, irfft

_lp_solver = None

def _get_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    import cvxpy as cp
    for name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(2, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=name)
            if p.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = name
                return name
        except:
            continue
    _lp_solver = cp.SCS
    return _lp_solver

def fast_convolve(a):
    n = len(a)
    sz = 2 * n - 1
    fs = 1 << (sz - 1).bit_length()
    A = rfft(a, fs)
    return irfft(A * A, fs)[:sz]

def eval_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * len(a) * fast_convolve(a).max() / (s * s)

def build_T(a):
    """Build (2n-1) x n matrix for convolution with a."""
    n = len(a)
    sz = 2 * n - 1
    T = np.zeros((sz, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_best_response(a, solver=None):
    """Solve LP: min t s.t. T@b <= t, sum(b)=1, b>=0."""
    import cvxpy as cp
    if solver is None:
        solver = _get_solver()
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    T = build_T(a)
    cons = [T @ b <= t, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(t), cons)
    for sol in [solver] + [s for s in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP] if s != solver]:
        try:
            prob.solve(solver=sol)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                bv = np.maximum(np.asarray(b.value, dtype=np.float64), 0)
                s = bv.sum()
                if s > 1e-15:
                    bv /= s
                    return bv, float(t.value)
        except:
            continue
    return None, None

def nesterov_lp(a0, time_limit, solver=None, max_iters=200000):
    """LP alternating with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()

    best_a = a.copy()
    best_sc = eval_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M = solve_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        y /= y.sum()
        
        a = ((it+1) * a + alpha * (it+2) * b_new) / (it+1 + alpha*(it+2))
        a = np.maximum(a, 0)
        a /= a.sum()
        
        sc_a = eval_local(a)
        sc_y = eval_local(y)
        if sc_a < best_sc:
            best_sc = sc_a
            best_a = a.copy()
        if sc_y < best_sc:
            best_sc = sc_y

                y /= y.sum()
                stall = 0
        
        if (it+1) % 2000 == 0:
            print(f"  NLP iter={it+1} t={time.monotonic()-t0:.0f}s best={best_sc:.8f}")
    
    return best_a

def subgradient_refine(a0, time_limit, n_steps=50000):
    """Subgradient method on the max of auto-convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a /= a.sum()
    best_a = a.copy()
    best_sc = eval_local(a)
    t0 = time.monotonic()
    lr = 0.01
    no_impr = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:

        b = fast_convolve(a)
        km = int(np.argmax(b))
        # Subgradient of conv(a,a)[km] w.r.t. a
        g = np.zeros(n)
        for j in range(n):
            k2 = km - j
            if 0 <= k2 < n:
                g[j] += a[k2]
                g[k2] += a[j] if k2 != j else 0
        g /= 2  # since conv is quadratic
        # Actually, d/d a[j] of sum_i a[i]*a[km-i] = a[km-j] + a[j]*delta... 
        # Let me just use the simple version
        g = np.zeros(n)
        for j in range(n):
            k2 = km - j
            if 0 <= k2 < n:
                g[j] += a[k2]
        # Projected gradient step
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        a_trial /= a_trial.sum()
        sc = eval_local(a_trial)

        else:

            if no_impr > 50:

                no_impr = 0

    return best_a

def lbgfs_logsumexp(a0, time_limit, T_stages=None):
    """L-BFGS with LogSumExp smoothing."""
    from scipy.optimize import minimize
    n = len(a0)
    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
    
    # Work in log-domain for the probability vector
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = np.log(a)
    t0 = time.monotonic()
    n_stages = len(T_stages)
    
    for stage_i, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

            a_val = np.exp(z_val - z_val.max())
            a_val /= a_val.sum()
            b = fast_convolve(a_val)
            lb = b.max()
            if lb < 1e-300:
                return 1e20
            return (2.0 * n / T) * (lb + np.log(np.sum(np.exp(b - lb))))

            a_val = np.exp(z_val - z_val.max())
            a_val /= a_val.sum()
            b = fast_convolve(a_val)
            lb = b.max()
            w = np.exp(b - lb)
            w /= w.sum()
            g = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                for j in range(js, je + 1):
                    g[j] += w[k] * a_val[k - j]
            S = g @ a_val
            return 2.0 / T * a_val * (g - S)
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_it, 'ftol': 1e-16, 'gtol': 1e-12, 'disp': False})
            z = res.x
        except:
            pass
    
    a_out = np.exp(z - z.max())
    a_out /= a_out.sum()
    return a_out

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based minimax alternating
with Nesterov acceleration, subgradient refinement, and multi-length search.

import numpy as np

from scipy.fft import rfft, irfft

# Top-level helper functions (no closures)

def fast_conv(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    sz = 2 * n - 1
    fs = 1 << (sz - 1).bit_length()
    A = rfft(a, fs)
    return irfft(A * A, fs)[:sz]

def score_of(a):
    """Compute the evaluation score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return 2.0 * n * b.max() / (s * s)

def norm1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    sz = 2 * n - 1
    T = np.zeros((sz, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T
