def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fourier_equalize(a0, time_limit, alpha_start=0.3, alpha_end=0.05, verbose=False):
    """
    Fourier-domain magnitude equalization for global exploration.
    Iteratively flattens |FFT(a)| while projecting back to non-negative.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    converged_iters = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / max(1, int(time_limit / 2.0)))
        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        # FFT of current a
        A = np.fft.fft(a)
        mag = np.abs(A)
        phase = np.angle(A)

        # Log-magnitude equalization
        log_mag = np.log(mag + 1e-30)
        # Target: blend toward the median (robust to outliers)
        target_log = np.median(log_mag)
        # Also try mean as alternative target
        target_log_mean = np.mean(log_mag)
        # Use a blend
        target = (1.0 - 0.3) * target_log + 0.3 * target_log_mean

        # New log-magnitude: move toward target
        new_log_mag = (1.0 - alpha) * log_mag + alpha * target
        new_mag = np.exp(new_log_mag)

        # Reconstruct with original phase
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            # Fallback: just project
            a_new = np.maximum(a, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            converged_iters = 0
        else:
            converged_iters += 1
            if converged_iters > 500:
                # Add perturbation to escape
                rng = np.random.default_rng(hash((n, it)) & 0xFFFFFFFF)
                a_new = a_new * (1.0 + 0.05 * rng.standard_normal(n))
                a_new = np.maximum(a_new, 0)
                ss = a_new.sum()
                if ss > 1e-15:
                    a_new = a_new / ss
                converged_iters = 0

        a = a_new

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FE it={it} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def pressure_coord_descent(a0, time_limit, verbose=False, label=""):

    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    sweeps = 0
    stall = 0

    peak_frac = 0.95

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Identify active peaks
        threshold = M * peak_frac
        active_peaks = np.where(conv >= threshold)[0]

        # Compute pressure on each coordinate
        pressure = np.zeros(n)
        for k in active_peaks:
            w = conv[k] - threshold
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                pressure[i] += w * a[j]
                if j != i:
                    pressure[j] += 0.5 * w * a[i]

        p_max = pressure.max()
        if p_max > 1e-20:
            pressure = pressure / p_max

        # Process coordinates in order of pressure (highest first)
        order = np.argsort(pressure)[::-1]

        for i in order:
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:

            affected = conv[k_start:k_end + 1].copy()
            a_shifted = a.copy()

            best_delta = 0.0
            best_new_max = M

            # Scale step size by pressure
            max_frac = 0.5 if pressure[i] > 0.5 else 0.3

            for f in [0.01, 0.02, 0.05, 0.1, 0.2, 0.35, 0.5, max_frac]:
                for sign in [-1, 1]:
                    delta = sign * f * a[i]
                    new_val = a[i] + delta
                    if new_val < -1e-15 or new_val > 1.0:
                        continue

                    new_affected = affected + 2.0 * delta * a_shifted
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx_2i = 2 * i - k_start
                        new_affected = new_affected.copy()
                        new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                    new_max = max(unaff_max, float(new_affected.max()))
                    if new_max < best_new_max - 1e-16:
                        best_new_max = new_max
                        best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a = a.sum()

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:

        else:
            stall = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

            best_delta = 0.0
            best_new_max = M

            fracs = [0.01, 0.03, 0.05, 0.1, 0.2, 0.35, 0.5]
            for f in fracs:
                for sign in [-1, 1]:
                    delta = sign * f * a[i]
                    new_val = a[i] + delta
                    if new_val < -1e-15 or new_val > 1.0:
                        continue

                    new_affected = affected + 2.0 * delta * a_shifted
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx_2i = 2 * i - k_start
                        new_affected = new_affected.copy()
                        new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                    new_max = max(unaff_max, float(new_affected.max()))
                    if new_max < best_new_max - 1e-16:
                        best_new_max = new_max
                        best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a = a.sum()

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_count += 1
            if no_improve_count > 80:
                break
        else:
            no_improve_count = 0

        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps


def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with very small steps for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            a_shifted = a[0:n]

            best_delta = 0.0
            best_new_max = M

            for factor in [0.995, 0.99, 0.985, 0.98, 0.97, 1.005, 1.01, 1.015, 1.02, 1.03]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected.copy() + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-17:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-16:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a += best_delta

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 30:
            break

    return best_a, best_score, sweeps


def simulated_annealing(a0, time_limit, rng, verbose=False, label=""):
    """
    Simulated annealing with structured moves targeting convolution peaks.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    it = 0

    # Temperature schedule
    T_start = 0.02
    T_end = 0.001
    total_iters = int(time_limit / 0.005)  # estimate

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / max(1, total_iters))
        T = T_start * (T_end / T_start) ** progress

        # Generate a structured move
        move_type = rng.integers(0, 4)

        if move_type == 0:
            # Peak-targeted: reduce a coordinate that contributes to a peak
            conv = fast_convolve(a)
            M_cur = conv.max()
            peak_k = int(np.argmax(conv))
            i_start = max(0, peak_k - n + 1)
            i_end = min(n - 1, peak_k)
            candidates = list(range(i_start, i_end + 1))
            # Weight by contribution
            contribs = np.array([a[i] * a[peak_k - i] for i in candidates])
            total_c = contribs.sum()
            if total_c > 1e-30:
                probs = contribs / total_c
                idx = rng.choice(len(candidates), p=probs)
                i = candidates[idx]
            else:
                i = int(rng.integers(0, n))
            delta = -rng.uniform(0.01, 0.08) * a[i]
            a_new = a.copy()
            a_new[i] = max(0, a[i] + delta)
            # Redistribute mass to random coordinates
            s_new = a_new.sum()
            deficit = 1.0 - s_new
            if deficit > 1e-15:
                n_add = int(rng.integers(1, min(5, n) + 1))
                add_idx = rng.choice(n, size=n_add, replace=False)
                a_new[add_idx] += deficit / n_add
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss > 1e-15:
                a_new = a_new / ss

        elif move_type == 1:
            # Random pair swap
            idx = rng.choice(n, size=2, replace=False)
            i, j = int(idx[0]), int(idx[1])
            transfer = rng.uniform(0.01, 0.1) * abs(a[i] - a[j])
            if transfer > min(a[i], a[j]):
                transfer = 0.5 * min(a[i], a[j])
            direction = 1 if a[i] > a[j] else -1
            a_new = a.copy()
            a_new[i] = max(0, a[i] + direction * transfer)
            a_new[j] = max(0, a[j] - direction * transfer)
            ss = a_new.sum()
            if ss > 1e-15:
                a_new = a_new / ss
            else:
                a_new = a.copy()

        elif move_type == 2:
            # Gaussian perturbation
            noise_scale = T * 0.5
            a_new = a * (1.0 + noise_scale * rng.standard_normal(n))
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss > 1e-15:
                a_new = a_new / ss
            else:
                a_new = a.copy()

        else:
            # Edge modification: modify a few edge coordinates
            edge_len = max(5, n // 20)
            a_new = a.copy()
            for _ in range(3):
                side = rng.integers(0, 2)
                i = int(rng.integers(0, edge_len))
                if side == 1:
                    i = n - 1 - i
                a_new[i] = max(0, a[i] * rng.uniform(0.5, 1.5))
            ss = a_new.sum()
            if ss > 1e-15:
                a_new = a_new / ss
            else:
                a_new = a.copy()

        # Evaluate
        conv_new = fast_convolve(a_new)
        M_new = float(conv_new.max())
        score_new = 2.0 * n * M_new

        # Acceptance criterion
        delta_score = score_new - best_score
        if delta_score < 0:
            a = a_new
            best_a = a_new.copy()
            best_score = score_new
            M = M_new
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-delta_score / max(T, 1e-10)):
                a = a_new
                M = M_new

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA {label} it={it} ({elapsed:.0f}s): best={best_score:.6f} T={T:.5f}")

    return best_a, best_score


def nesterov_fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """Fictitious play with Nesterov-style averaging."""
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_sequent = a.copy()
    b_sequent = b.copy()

    best_a = a.copy()
    best_score = float('inf')
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        gamma_it = 2.0 / (iters + 2.0)
        a_sequent = (1.0 - gamma_it) * a_sequent + gamma_it * a
        b_sequent = (1.0 - gamma_it) * b_sequent + gamma_it * b

        for cand in [a, b, a_sequent, b_sequent, 0.5 * (a_sequent + b_sequent)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NFP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def damped_lp_refine(a0, time_limit, solver=None, verbose=False):
    """Damped LP refinement with adaptive step size."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        alpha = min(1.0, 5.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DLP it={it} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_symmetric(n, rng):
    """Create a symmetric initialization."""
    half = n // 2
    t = np.linspace(0, 1, half)
    n_comp = int(rng.integers(2, 8))
    profile = np.ones(half)
    for j in range(n_comp):
        freq = rng.uniform(0.5, 4.0)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0)
    if n % 2 == 0:
        a = np.concatenate([profile, profile[::-1]])
    else:
        mid = max(profile[-1], profile[0]) * rng.uniform(0.5, 1.5)
        a = np.concatenate([profile, [mid], profile[::-1]])
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_bell(n, rng):
    """Initialize with a bell-shaped profile (small edges, large middle)."""
    center = n / 2.0
    width = n * rng.uniform(0.25, 0.45)
    t = np.arange(n)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = a + rng.uniform(0.05, 0.15) * np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_flat_ridges(n, rng):
    """Flat base with periodic ridges."""
    period = int(n / rng.integers(3, 8))
    t = np.arange(n)
    a = np.ones(n) * 0.5
    ripple_amp = rng.uniform(0.1, 0.3)
    a += ripple_amp * np.cos(2 * np.pi * t / period)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

    if best_seq is not None:
        # Fine CD
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_fine = min(rem * 0.5, 60.0)
            print(f"  FineCD budget={t_fine:.0f}s")
            a_fine, s_fine, _ = fine_coord_descent(best_seq, t_fine, label="fine1")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd_1")

        # Pressure CD
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_pc = min(rem * 0.5, 40.0)
            a_pc, s_pc = pressure_coord_descent(best_seq, t_pc, label="p_final")
            best_seq, best_score = check_and_update(a_pc, best_seq, best_score, "p_final")

        # Another fine pass
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_f2, s_f2, _ = fine_coord_descent(best_seq, min(rem * 0.6, 25.0), label="fine2")
            best_seq, best_score = check_and_update(a_f2, best_seq, best_score, "fine_cd_2")

    # === Phase 7: Try alternative lengths (remaining time) ===
    print(f"\n=== Phase 7: Alt Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 10
        if rem > 30:
            for alt_n in [n_best + 128, n_best - 128, n_best + 256]:
                if alt_n <= 256 or alt_n > 2048:
                    continue
                if rem < 20:
                    break
                t_alt = min(rem * 0.4, 40.0)
                rem = deadline - time.monotonic() - 10
                if rem < 15:
                    break
                init_alt = interpolate_seq(best_seq, alt_n)
                a_alt, s_alt, _ = fast_coord_descent(init_alt, t_alt, n_trials=10, label=f"alt{alt_n}")
                best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")
                # Quick fine polish
                rem2 = deadline - time.monotonic() - 10
                if rem2 > 10:
                    a_alt2, s_alt2, _ = fine_coord_descent(a_alt, min(rem2 * 0.5, 15.0), label=f"fine_alt{alt_n}")
                    best_seq, best_score = check_and_update(a_alt2, best_seq, best_score, f"fine_alt_n{alt_n}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
