    amps = np.ones(n_freq)

        freq_idx = np.arange(n_freq).astype(np.float64)
        amps = amps * np.exp(-smooth * (freq_idx / n_freq) ** 2)
    amps *= rng.uniform(0.85, 1.15, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_super_gaussian(n, rng):
    """Initialize with a super-Gaussian shape."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.3, 0.5)
    p = rng.uniform(1.5, 3.0)
    a = np.exp(-np.abs((t - center) / width) ** p)
    a += rng.uniform(0, 0.005, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def golden_section_search(f, lo, hi, max_iter=40):
    """Golden section search for minimum of f on [lo, hi]. Returns optimal x."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def get_peak_lags(conv, n_top=5, rel_tol=0.01):
    """Get the top peak lags and their values."""
    max_val = float(conv.max())
    threshold = max_val * (1.0 - rel_tol)
    peaks = np.where(conv >= threshold)[0]
    # Sort by value descending
    peak_vals = conv[peaks]
    order = np.argsort(peak_vals)[::-1]
    return peaks[order[:n_top]], peak_vals[order[:n_top]]

def get_peak_contributors(a, k, n):
    """Get coordinates that contribute to conv[k]. Returns list of (i, partner, product)."""
    contribs = []
    if k == 0:
        for i in range(n):
            contribs.append((i, i, a[i] * a[i]))
    else:
        max_i = n - k
        for i in range(max_i):
            contribs.append((i, i + k, a[i] * a[i + k]))
    return contribs

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Exact coordinate descent with golden section 1D search.
    Uses peak-oriented ordering: process coordinates that most affect the peak first.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Peak-oriented ordering: sort coordinates by contribution to current peak
        peak_lags, _ = get_peak_lags(conv, n_top=3, rel_tol=0.005)
        # For each peak lag, compute contribution weight for each coordinate
        coord_priority = np.zeros(n)
        for k in peak_lags:
            k = int(k)
            if k == 0:
                coord_priority += 2.0 * a
            else:
                coord_priority[:n - k] += a[k:]
                coord_priority[k:] += a[:n - k]

        order = np.argsort(coord_priority)[::-1]

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            # Unaffected max
            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            if has_quad:
                quad_idx = 2 * i - k_lo

            # Precompute for fast evaluation
            c_vals_copy = c_vals.copy()
            ap_copy = a_partners.copy()
            qm = unaff_max
            hq = has_quad
            qi = quad_idx if has_quad else -1
            cs = current_sum
            vv = v
            nn = n

            def compute_score(x):
                ns = cs + (x - vv)
                if ns < 0.01:
                    return 1e10
                nv = c_vals_copy + 2.0 * x * ap_copy
                if hq:
                    nv[qi] = c_vals_copy[qi] + x * x
                nm = max(qm, float(nv.max()))
                return 2.0 * nn * nm / (ns * ns)

            # Search range: use current value as center
            upper = max(3.0 * v, 4.0 / n)
            best_x = golden_section_search(compute_score, 0.0, upper, max_iter=35)

            if abs(best_x - v) > 1e-12:
                delta = best_x - v

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                a[i] = best_x
                current_sum += delta

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def fine_coord_descent(a0, time_limit, verbose=False):
    """Fine coordinate descent with very small multiplicative steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweeps = 0

    factors = [0.999, 0.998, 0.997, 0.995, 0.99, 0.985,
               1.001, 1.002, 1.003, 1.005, 1.01, 1.015]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            if has_quad:
                quad_idx = 2 * i - k_lo

            best_delta = 0.0
            best_local_max = float('inf')

            for factor in factors:
                x = v * factor
                if x < -1e-15:
                    continue
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                local_max = max(unaff_max, float(nv.max()))
                if local_max < best_local_max:
                    best_local_max = local_max
                    best_delta = x - v

            if best_delta != 0.0:
                a[i] = max(a[i] + best_delta, 0.0)
                current_sum += best_delta

                changes = 2.0 * best_delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = best_delta * (2.0 * v + best_delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def coordinated_peak_breaking(a, conv, rng, strength=0.05):
    """
    Perturb coordinates that contribute to peak lags in a coordinated way.
    For each peak lag k, find the pair with largest product and anti-correlate them.
    """
    n = len(a)
    a_new = a.copy()
    peak_lags, peak_vals = get_peak_lags(conv, n_top=3, rel_tol=0.005)

    for k in peak_lags:
        k = int(k)
        if k == 0:
            # For lag 0, reduce the largest elements slightly and increase smaller ones
            idx_max = np.argmax(a_new)
            idx_min_candidates = np.where(a_new < np.median(a_new))[0]
            if len(idx_min_candidates) > 0:
                idx_min = rng.choice(idx_min_candidates)
                delta = a_new[idx_max] * strength * rng.uniform(0.3, 0.8)
                a_new[idx_max] -= delta
                a_new[idx_min] += delta
        else:
            # For lag k, find the pair (i, i+k) with largest product
            max_i = n - k
            if max_i <= 0:
                continue
            products = a_new[:max_i] * a_new[k:k + max_i]
            top_pairs = np.argpartition(products, -min(5, max_i))[-min(5, max_i):]
            for pidx in top_pairs:
                # Anti-correlate: decrease the larger, increase the smaller
                i1 = pidx
                i2 = pidx + k
                if a_new[i1] >= a_new[i2]:
                    delta = a_new[i1] * strength * rng.uniform(0.2, 0.6)
                    a_new[i1] -= delta
                    a_new[i2] += delta
                else:
                    delta = a_new[i2] * strength * rng.uniform(0.2, 0.6)
                    a_new[i2] -= delta
                    a_new[i1] += delta

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def nesterov_subgradient(a0, time_limit, verbose=False):
    """Nesterov-accelerated subgradient descent with adaptive peak weighting."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    lr0 = 0.15
    lr_decay = 0.9995
    mu = 0.85
    n_peaks = 15
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        lr = lr0 * (lr_decay ** it)

        conv_y = fast_convolve(y)
        max_y = float(conv_y.max())
        threshold = max_y * 0.97
        active_peaks = np.where(conv_y >= threshold)[0]
        if len(active_peaks) == 0:
            active_peaks = np.array([np.argmax(conv_y)])
        active_peaks = active_peaks[:n_peaks]

        pv = conv_y[active_peaks]
        temp = 0.005 * max_y
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(active_peaks)) / len(active_peaks)
        else:
            w = w / ws

        grad = np.zeros(n)
        for idx in range(len(active_peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(active_peaks[idx])
            if k == 0:
                grad += 2.0 * w[idx] * y
            else:
                grad[:n - k] += w[idx] * y[k:]
                grad[k:] += w[idx] * y[:n - k]

        # Project to tangent space
        grad = grad - grad.mean()

        # Nesterov update
        v = mu * v - lr * grad
        y = a + v
        a_new = simplex_project(y)

        conv_new = fast_convolve(a_new)
        score = 2.0 * n * float(conv_new.max())
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 300:
            break
        if verbose and it % 50 == 0:
            print(f"    NesterovSG it={it}: score={best_score:.8f}, lr={lr:.6f}")

    return best_a, best_score

def pairwise_cd_step(a, conv, k_peak, i1, i2, current_sum):
    """
    Try a pairwise update of coordinates i1 and i2 that both contribute to peak lag k_peak.
    Returns (delta1, delta2, improvement) or (0, 0, -inf) if no improvement.
    """
    n = len(a)
    v1 = a[i1]
    v2 = a[i2]

    def score_at(d1, d2):
        x1 = max(v1 + d1, 0)
        x2 = max(v2 + d2, 0)
        if x1 + x2 - v1 - v2 + current_sum - (v1 + v2) < 0.01:
            return 1e10
        ns = current_sum + d1 + d2
        if ns < 0.01:
            return 1e10
        # Estimate new max: focus on affected lags
        # For simplicity, just evaluate the full score
        a_mod = a.copy()
        a_mod[i1] = x1
        a_mod[i2] = x2
        # Quick check: only compute affected conv lags
        # This is expensive, so we approximate
        # For now, return a proxy score
        c0_change = (x1 * x1 - v1 * v1) + (x2 * x2 - v2 * v2) + 2.0 * (x1 * x2 - v1 * v2)
        # Just use the sum of squares as a proxy for the peak at lag 0
        return 2.0 * n * (float(conv.max()) + max(0, c0_change)) / (ns * ns)

    # Try a small set of (d1, d2) pairs
    best_s = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
    best_d1, best_d2 = 0.0, 0.0
    scale = max(v1, v2) * 0.05
    for f1 in [-1, -0.5, 0.5, 1]:
        for f2 in [-1, -0.5, 0.5, 1]:
            d1 = f1 * scale
            d2 = f2 * scale
            s = score_at(d1, d2)
            if s < best_s:
                best_s = s
                best_d1, best_d2 = d1, d2

    return best_d1, best_d2

def basin_hopping_structured(a0, time_limit, rng, verbose=False):
    """
    Basin hopping with structured peak-breaking perturbations.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    attempt = 0

    while time.monotonic() - t0 < time_limit:
        attempt += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break
        t_alloc = min(rem * 0.4, 60.0)

        conv_a = fast_convolve(a)

        # Choose perturbation strategy based on attempt number
        strategy = attempt % 3

        if strategy == 0:
            # Coordinated peak breaking
            perturbed = coordinated_peak_breaking(a, conv_a, rng, strength=0.03 + 0.04 * rng.random())
        elif strategy == 1:
            # Gaussian noise scaled by local structure
            noise_scale = 0.02 + 0.03 * rng.random()
            perturbed = a + rng.standard_normal(n) * noise_scale
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps < 1e-10:
                continue
            perturbed = perturbed / ps
        else:
            # Peak-specific: reduce top contributors to peak, add to bottom
            peak_lags, _ = get_peak_lags(conv_a, n_top=2, rel_tol=0.003)
            contrib_weights = np.zeros(n)
            for k in peak_lags:
                k = int(k)
                if k == 0:
                    contrib_weights += 2.0 * a
                else:
                    contrib_weights[:n - k] += a[k:]
                    contrib_weights[k:] += a[:n - k]
            # Reduce high-contributing coords, increase low-contributing
            top_mask = contrib_weights > np.percentile(contrib_weights, 80)
            bot_mask = contrib_weights < np.percentile(contrib_weights, 20)
            perturbed = a.copy()
            perturbed[top_mask] *= (1.0 - 0.05 * rng.random(top_mask.sum().astype(int)))
            perturbed[bot_mask] *= (1.0 + 0.1 * rng.random(bot_mask.sum().astype(int)))
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps < 1e-10:
                continue
            perturbed = perturbed / ps

        # Optimize the perturbed sequence
        a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc, label=f"bh{attempt}")
        if s_opt < best_score - 1e-14:
            best_score = s_opt
            best_a = a_opt.copy()
            a = a_opt.copy()
            if verbose:
                print(f"    BH attempt={attempt} (strat={strategy}): improved to {best_score:.8f}")

    return best_a, best_score

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

            a_opt, s_opt, sweeps = exact_coord_descent(current, t_alloc,
                                                       verbose=True, label=f"ph2_r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ph2_ecd_r{rnd}")

            if best_score > 1.5035:
                # Perturb using peak breaking
                conv_cur = fast_convolve(a_opt)
                perturbed = coordinated_peak_breaking(a_opt, conv_cur, rng, strength=0.04)
                current = perturbed
            else:
                current = a_opt.copy()

    # === Phase 3: Structured basin hopping ===
    print(f"\n=== Phase 3: Basin Hopping (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = _DEADLINE - time.monotonic() - 400
        if rem > 60:
            t_bh = min(rem * 0.7, 300.0)
            print(f"  Basin hopping, budget={t_bh:.0f}s")
            a_bh, s_bh = basin_hopping_structured(best_seq, t_bh, rng, verbose=True)
            best_seq, best_score = check_and_update(a_bh, best_seq, best_score, "basin_hop")

            # Follow up with CD
            rem = _DEADLINE - time.monotonic() - 300
            if rem > 20:
                t_cd = min(rem * 0.5, 100.0)
                a_cd, s_cd, _ = exact_coord_descent(a_bh, t_cd, label="post_bh")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "ecd_after_bh")

    # === Phase 4: Nesterov subgradient ===
    print(f"\n=== Phase 4: Nesterov Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = _DEADLINE - time.monotonic() - 250
        if rem > 30:
            t_sg = min(rem * 0.6, 150.0)
            print(f"  Nesterov SG, budget={t_sg:.0f}s")
            a_sg, s_sg = nesterov_subgradient(best_seq, t_sg, verbose=True)
            best_seq, best_score = check_and_update(a_sg, best_seq, best_score, "nesterov_sg")

            # Polish the subgradient result
            rem = _DEADLINE - time.monotonic() - 200
            if rem > 15:
                t_cd = min(rem * 0.5, 60.0)
                a_cd, s_cd, _ = exact_coord_descent(a_sg, t_cd, label="after_sg")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "ecd_after_sg")

    # === Phase 5: Ultra-fine polishing ===
    print(f"\n=== Phase 5: Fine Polishing (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Fine coordinate descent
        rem = _DEADLINE - time.monotonic() - 80
        if rem > 20:
            t_fine = min(rem * 0.6, 120.0)
            print(f"  FineCD, budget={t_fine:.0f}s")
            a_fine, s_fine, _ = fine_coord_descent(best_seq, t_fine, verbose=True)
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

        # Another exact CD round
        rem = _DEADLINE - time.monotonic() - 50
        if rem > 15:
            t_cd2 = min(rem * 0.6, 80.0)
            print(f"  ECD-polish, budget={t_cd2:.0f}s")
            a_cd2, s_cd2, _ = exact_coord_descent(best_seq, t_cd2, label="polish")
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "ecd_polish")

        # Final fine pass
        rem = _DEADLINE - time.monotonic() - 20
        if rem > 10:
            a_fin, s_fin, _ = fine_coord_descent(best_seq, min(rem * 0.7, 40.0), label="last")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "last_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
