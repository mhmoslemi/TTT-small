def fast_convolve(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr(w, a):
    """Cross-correlation: result[j] = sum_k w[k] * a[k-j] (valid indices only)."""
    n = len(a)
    # We want for each j: sum over k where 0<=k-j<n of w[k]*a[k-j]
    # This equals: for each j, sum over t=0..n-1 of w[t+j]*a[t]
    # = (w cross-correlate a)[j]
    # Using FFT: cross_corr = ifft(conj(fft(w)) * fft(a))
    N = 1 << (len(w) + n - 1 - 1).bit_length()
    W = rfft(np.asarray(w, dtype=np.float64), N)
    A = rfft(np.asarray(a, dtype=np.float64), N)
    # cross_corr[j] = ifft(conj(W) * A)[j] for j = 0, 1, ..., n-1
    cc = irfft(np.conj(W) * A, N)
    return cc[:n]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_score_and_grad(z, T):
    """Compute log-sum-exp score and its gradient w.r.t. z."""
    n = len(z)
    a = softmax_from_z(z)
    c = fast_convolve(a)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    Z = exp_Tc.sum()
    score = (2.0 * n / T) * (max_Tc + np.log(Z))
    
    # Gradient
    w = exp_Tc / Z
    # grad_a[j] = 2n * sum_k w[k] * 2 * a[k-j] (for valid k-j)
    # = 4n * cross_corr(w, a)[j]

    # d score / da_j = 2n * sum_k w_k * dc_k/da_j = 2n * sum_k w_k * 2*a_{k-j} = 4n * cross_corr(w,a)[j]
    grad_a = 4.0 * n * cross_corr(w, a)

    grad_z = a * (grad_a - S)
    
    return score, grad_z


def ramp_init(n):
    """Linear ramp initialization."""
    a = np.arange(1, n+1, dtype=np.float64)
    return a / a.sum()


def power_ramp_init(n, p):
    """Power ramp: a_j = (j+1)^p."""
    a = np.arange(1, n+1, dtype=np.float64) ** p
    return a / a.sum()
