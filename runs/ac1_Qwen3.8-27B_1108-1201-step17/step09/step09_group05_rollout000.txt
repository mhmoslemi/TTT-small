import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    return 2.0 * len(a) * fast_convolve(a).max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_objective_factory(T, n):
    """Create log-sum-exp smoothed objective."""
    def obj(z_in):
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T) * log_sum_exp
    return obj


def lse_gradient_factory(T, n):
    """Create log-sum-exp smoothed gradient."""
    def grad(z_in):
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)
    return grad

                j = pk - i
                if i == j or a[i] < 1e-12:
                    continue

                # Try reducing a[i] and redistributing
                for factor in [0.95, 0.90, 0.85, 0.78]:
                    a_trial = a.copy()
                    old_val = a_trial[i]
                    a_trial[i] = old_val * factor
                    diff = old_val - a_trial[i]
                    if diff < 1e-15:
                        continue
                    # Redistribute to coordinate with lowest contribution to pk
                    contrib_all = a * a[pk - np.arange(n)]
                    contrib_all = np.where(np.arange(n) >= 0, contrib_all, 1e10)
                    # Find coordinate where adding mass hurts least
                    low_mask = np.ones(n, dtype=bool)
                    low_mask[i] = False
                    if j < n:
                        low_mask[j] = False
                    candidates = np.where(low_mask)[0]
                    if len(candidates) == 0:
                        continue
                    # Score each candidate by how much it would increase pk
                    gains = a[candidates] * a[pk - candidates]
                    gains[(pk - candidates) < 0] = 1e10
                    gains[(pk - candidates) >= n] = 1e10
                    best_target = candidates[np.argmin(gains)]
                    a_trial[best_target] += diff
                    a_trial = normalize(a_trial)
                    sc = exact_score(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, n_steps=30000):
    """Pairwise coordinate descent: swap mass between pairs of coordinates."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = exact_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 200:
            break

        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_star - idx]
        top_contrib = np.argsort(contribs)[::-1][:5]

        # Pick a high-contrib coordinate and a low-contrib target
        for ci in top_contrib:
            i = idx[ci]
            if a[i] < 1e-10:
                continue
            # Random target
            for _ in range(3):
                j = int(rng.integers(n))
                if j == i or a[k_star - j] < 1e-10 if 0 <= k_star - j < n else True:
                    if j == i:
                        continue
                    for frac in [0.1, 0.2, 0.35, 0.5]:
                        a_trial = a.copy()
                        move = a_trial[i] * frac
                        a_trial[i] -= move
                        a_trial[j] += move
                        a_trial = normalize(a_trial)
                        sc = exact_score(a_trial)
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            no_improve = 0
                            break
                        else:
                            no_improve += 1
        if no_improve > 200:
            break

    return best_a, best_score


def structured_init(n, mode=0):
    """Create structured initial sequence."""
    if mode == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2.5)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a ** 2
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * x * (1 - x)) * (1 + 0.2 * np.cos(2 * np.pi * 3 * x))
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 0.7 * (0.5 + 0.5 * np.cos(np.pi * x))
    else:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(-20 * (x - 0.5)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        s = sum(a_list)
        if s < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Multi-start exploration with quick L-BFGS
    Phase 2: Deep L-BFGS optimization with aggressive cooling
    Phase 3: Nesterov subgradient for periodic restarts
    Phase 4: Targeted peak shaving and coordinate descent
    Phase 5: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = normalize(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Load parent if available ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        best_seq = structured_init(n_init, mode=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"\nStarting: n={n_best}, score={best_score:.8f}")

    # === Phase 1: Multi-start exploration ===
    print(f"\n=== Phase 1: Multi-Start Exploration ===")
    rem = deadline - time.monotonic()
    t_phase1 = min(rem * 0.20, 300.0)
    if t_phase1 > 30:
        candidates = []
        # Parent-based candidates
        candidates.append(('parent', best_seq.copy()))

        # Different lengths from parent
        for mult in [1.25, 1.5, 2.0, 0.75]:
            alt_n = int(n_best * mult)
            if 256 <= alt_n <= 3000:
                candidates.append((f'interp_{alt_n}', interpolate_sequence(best_seq, alt_n)))

        # Structured inits at the best length
        for mode in [1, 2, 3, 4]:
            candidates.append((f'struct_{mode}_n{n_best}', structured_init(n_best, mode)))

        # Random init
        rand_a = rng.random(n_best)
        candidates.append(('random', normalize(rand_a)))

        # Quick L-BFGS on each candidate
        t_per = t_phase1 / len(candidates)
        for label, cand in candidates:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_alloc = min(t_per, rem * 0.9)
            if t_alloc < 10:
                break
            a_opt = lbgfs_refine(cand, t_alloc, T_start=30.0, T_end=0.5, n_stages=6)
            try_update(a_opt, f'p1_{label}')

    # === Phase 2: Deep L-BFGS on best ===
    print(f"\n=== Phase 2: Deep L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase2 = min(rem * 0.30, 480.0)
    if t_phase2 > 30:
        print(f"  Budget: {t_phase2:.0f}s, n={len(best_seq)}")
        a_deep = lbgfs_refine(best_seq, t_phase2, T_start=50.0, T_end=0.03, n_stages=16)
        try_update(a_deep, 'deep_lbgfs')

    # === Phase 3: Nesterov subgradient + L-BFGS interleaved ===
    print(f"\n=== Phase 3: Nesterov + L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase3 = min(rem * 0.20, 300.0)
    if t_phase3 > 20:
        # Run Nesterov for a while
        t_nest = t_phase3 * 0.4
        a_nest, _ = subgradient_nesterov(best_seq, t_nest, lr_init=0.015, n_max_iters=100000)
        try_update(a_nest, 'nesterov')

        # Then L-BFGS on the result
        rem = deadline - time.monotonic()
        if rem > 20:
            t_lb3 = min(t_phase3 * 0.6, rem * 0.9)
            a_lb3 = lbgfs_refine(best_seq, t_lb3, T_start=20.0, T_end=0.05, n_stages=10)
            try_update(a_lb3, 'p3_lbgfs')

    # === Phase 4: Peak shaving + coordinate descent ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase4 = min(rem * 0.15, 200.0)
    if t_phase4 > 15:
        t_ps = t_phase4 * 0.5
        a_ps, _ = peak_shave_direct(best_seq, t_ps, n_steps=30000)
        try_update(a_ps, 'peak_shave')

        rem = deadline - time.monotonic()
        if rem > 10:
            t_cd = min(t_phase4 * 0.5, rem * 0.9)
            a_cd, _ = coordinate_descent_refine(best_seq, t_cd, n_steps=20000)
            try_update(a_cd, 'coord_desc')

    # === Phase 5: Final aggressive L-BFGS polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.4, 120.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=10.0, T_end=0.02, n_stages=10)
        try_update(a_fin, 'final_lbgfs')

    # === Phase 6: Another round of peak shaving ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps2 = min(rem * 0.3, 60.0)
        a_ps2, _ = peak_shave_direct(best_seq, t_ps2, n_steps=20000)
        try_update(a_ps2, 'peak_shave_2')

    # === Phase 7: Coordinate descent final ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_cd2 = min(rem * 0.3, 60.0)
        a_cd2, _ = coordinate_descent_refine(best_seq, t_cd2, n_steps=15000)
        try_update(a_cd2, 'coord_desc_2')

    # === Phase 8: Final L-BFGS at ultra-low temperature ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t_ult = min(rem * 0.5, 40.0)
        a_ult = lbgfs_refine(best_seq, t_ult, T_start=3.0, T_end=0.01, n_stages=8)
        try_update(a_ult, 'ultra_low')

    # === Verification ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
