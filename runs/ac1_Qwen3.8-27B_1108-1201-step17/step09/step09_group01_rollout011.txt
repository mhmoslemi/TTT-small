import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

            if k_lo > 0:
                unaff_l = float(conv[:k_lo].max())
            else:
                unaff_l = -1e30
            if k_hi < 2 * n - 2:
                unaff_r = float(conv[k_hi + 1:].max())
            else:
                unaff_r = -1e30
            M_unaff = max(unaff_l, unaff_r)

            affected = conv[k_lo:k_hi + 1]
            offsets = np.arange(k_lo, k_hi + 1) - i
            lin_coef = a[offsets]
            q_pos = 2 * i - k_lo
            has_quad = (0 <= q_pos < len(affected))

            best_d = 0.0
            best_nm = M

            for f in factors:
                d = ai * (f - 1.0)
                new_ai = ai + d
                if new_ai < -1e-14:
                    continue

                new_aff = affected + 2.0 * d * lin_coef
                if has_quad:
                    new_aff = new_aff.copy()
                    new_aff[q_pos] = affected[q_pos] + 2.0 * d * ai + d * d

                nm = max(M_unaff, float(new_aff.max()))
                if nm < best_nm - 1e-16:
                    best_nm = nm
                    best_d = d

            if best_d != 0.0 and best_nm < M - 1e-15:
                a[i] = max(a[i] + best_d, 0.0)
                conv[k_lo:k_hi + 1] += 2.0 * best_d * lin_coef
                if has_quad:
                    conv[2 * i] += best_d * best_d
                M = max(M_unaff, float(conv[k_lo:k_hi + 1].max()))

                cs = 2.0 * n * M
                if cs < best_score - 1e-14:
                    best_score = cs
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 20:
            break

    return best_a, best_score, sweeps


def peak_pressure_perturb(a0, rng, strength=0.05):
    """Perturb by applying pressure to high-contributing coordinates."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()

    # Pressure: how much each coordinate contributes to the top peaks
    threshold = M * 0.9
    peak_pos = np.where(conv >= threshold)[0]

    pressure = np.zeros(n)
    for k in peak_pos:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            w = a[i] * a[j]
            pressure[i] += w
            if j != i:
                pressure[j] += 0.5 * w

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure /= p_max

    noise = rng.standard_normal(n) * 0.2
    a_new = a * (1.0 - strength * pressure + strength * 0.3 * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def random_block_perturb(a0, rng, block_frac=0.2):
    """Randomly zero out a random block and redistribute."""
    n = len(a0)
    a = a0.copy()
    block_size = max(4, int(n * block_frac))
    start = rng.integers(0, n - block_size)
    a[start:start + block_size] = 0
    # Add small random values in the zeroed block
    a[start:start + block_size] = rng.random(block_size) * a.max() * 0.3
    # Add noise everywhere
    a = a * (1.0 + rng.standard_normal(n) * 0.05)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def init_diverse(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []

    if parent is not None:
        # Interpolate parent
        old_n = len(parent)
        old_x = np.linspace(0, 1, old_n)
        new_x = np.linspace(0, 1, n)
        p_interp = np.interp(new_x, old_x, parent / parent.sum())
        p_interp = np.maximum(p_interp, 0)
        ss = p_interp.sum()
        if ss > 1e-15:
            inits.append(p_interp / ss)

    # Symmetric cosine superposition
    t = np.linspace(0, 1, n)
    for trial in range(3):
        profile = np.ones(n) * 0.5
        n_comp = rng.integers(3, 8)
        for j in range(n_comp):
            freq = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            phase = rng.uniform(0, np.pi)
            profile += amp * np.cos(np.pi * freq * t + phase)
        profile = np.maximum(profile, 0)
        ss = profile.sum()
        if ss > 1e-15:
            inits.append(profile / ss)

    # Bimodal
    for trial in range(2):
        c1 = n * rng.uniform(0.15, 0.35)
        c2 = n * rng.uniform(0.65, 0.85)
        w1 = n * rng.uniform(0.05, 0.15)
        w2 = n * rng.uniform(0.05, 0.15)
        idx = np.arange(n)
        a = np.exp(-0.5 * ((idx - c1) / w1) ** 2) + np.exp(-0.5 * ((idx - c2) / w2) ** 2)
        ss = a.sum()
        if ss > 1e-15:
            inits.append(a / ss)

    # Anti-symmetric (a[i] * a[n-1-i] should be small)
    half = n // 2
    t_half = np.linspace(0, 1, half)
    left = np.exp(-((t_half - 0.3) ** 2) / 0.05) + 0.3
    right = np.exp(-((t_half - 0.7) ** 2) / 0.05) + 0.3
    a_asym = np.concatenate([left, right[::-1]])
    if n % 2 == 1:
        a_asym = np.insert(a_asym, n // 2, 0.2)
    ss = a_asym.sum()
    if ss > 1e-15:
        inits.append(a_asym[:n] / a_asym[:n].sum())

    # Uniform
    inits.append(np.ones(n) / n)

    return inits


def tempered_restart_loop(current, time_limit, rng, verbose=False):
    """
    Main optimization loop with tempered restarts.
    Alternates between exploitation (exact CD) and exploration (perturbation).
    """
    n = len(current)
    a = current.copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    # Temperature schedule
    T_init = 0.15
    T_min = 0.005
    n_restarts = 0

    phase = "exploit"
    exploit_time = 0.0
    explore_time = 0.0

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        if phase == "exploit":
            # Do exact coordinate descent
            t_alloc = min(rem * 0.6, 120.0)
            a_opt, s_opt, sweeps = exact_coord_descent(a, t_alloc, top_k=6,
                                                       verbose=verbose, label="main")
            exploit_time += time.monotonic() - t0
            if s_opt < best_score - 1e-12:
                best_score = s_opt
                best_a = a_opt.copy()
            a = a_opt.copy()

            # Decide whether to explore
            n_restarts += 1
            # Explore more frequently as we go on (to find new basins)
            if n_restarts % 3 == 0 or best_score > 1.51:
                phase = "explore"
            else:
                phase = "exploit"
        else:
            # Explore: perturb and try
            t_alloc_explore = min(rem * 0.3, 60.0)
            t_alloc_reopt = rem - t_alloc_explore - 10

            # Generate perturbed version
            r = rng.random()
            if r < 0.4:
                a_pert = peak_pressure_perturb(best_a, rng, strength=0.03 + 0.02 * n_restarts * 0.1)
            elif r < 0.7:
                a_pert = random_block_perturb(best_a, rng, block_frac=0.05 + 0.03 * n_restarts * 0.1)
            else:
                # Mix with random
                a_rand = np.abs(rng.standard_normal(n))
                a_rand = a_rand / a_rand.sum()
                alpha = rng.uniform(0.3, 0.7)
                a_pert = (1 - alpha) * best_a + alpha * a_rand
                a_pert = a_pert / a_pert.sum()

            # Quick optimize the perturbed version
            a_opt, s_opt, _ = exact_coord_descent(a_pert, t_alloc_reopt, top_k=5,
                                                  verbose=False, label="perturb")
            if s_opt < best_score - 1e-12:
                print(f"      Explorer found improvement: {s_opt:.8f} (restart {n_restarts})")
                best_score = s_opt
                best_a = a_opt.copy()
                a = a_opt.copy()
                # Reset exploit phase
                phase = "exploit"
            else:
                # Stay with current best
                a = best_a.copy()
                phase = "exploit"

        if verbose and n_restarts % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TL: restart={n_restarts} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def multi_length_search(n_best, a_best, time_limit, rng, verbose=False):
    """Try optimizing at nearby lengths and interpolate back."""
    improvements = []
    for delta_n in [64, 128, 256, -64, -128]:
        n_try = n_best + delta_n
        if n_try < 256 or n_try > 4096:
            continue
        rem = time_limit - (time.monotonic() - (time_limit - time_limit))  # just check time
        # Interpolate
        old_x = np.linspace(0, 1, n_best)
        new_x = np.linspace(0, 1, n_try)
        a_interp = np.interp(new_x, old_x, a_best)
        a_interp = np.maximum(a_interp, 0)
        ss = a_interp.sum()
        if ss < 1e-15:
            continue
        a_interp /= ss

        score_interp = score_of(a_interp)
        improvements.append((score_interp, a_interp, n_try))

    return improvements


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate with the actual evaluate_sequence and update best."""
    try:
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return best_seq, best_score
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    except Exception as e:
        pass
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization + quick exact CD
    Phase 2: Tempered restart loop (main workhorse)
    Phase 3: Fine polishing
    Phase 4: Alternative length exploration
    Phase 5: Final verification
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(12345)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Diverse init + quick optimization ===
    print("\n=== Phase 1: Diverse Init ===")
    t1_end = time.monotonic() + 120.0

    n_list = [1024, 768, 1280, 960, 1152]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_list:
            n_list.insert(0, np_)

    candidates = []
    for n in n_list:
        if time.monotonic() > t1_end - 10:
            break
        inits = init_diverse(n, rng, parent)
        for idx, init in enumerate(inits):
            if time.monotonic() > t1_end - 5:
                break
            t_alloc = min(10.0, (t1_end - time.monotonic()) / max(1, len(inits) - idx))
            a_opt, s_opt, _ = exact_coord_descent(init, t_alloc, top_k=5, label=f"p1_n{n}_{idx}")
            candidates.append((s_opt, a_opt, n))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, top: " +
          ", ".join([f"{c[0]:.6f}(n={c[2]})" for c in candidates[:5]]))

    for s_c, a_c, n_c in candidates[:6]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}")

    # Use the best from phase 1 as starting point
    if best_seq is not None:
        start_a = best_seq
    else:
        start_a = candidates[0][1]

    # === Phase 2: Tempered restart loop (main) ===
    print(f"\n=== Phase 2: Tempered Restart (score={best_score:.8f}) ===")
    t2_end = time.monotonic() + min(1000.0, (deadline - time.monotonic()) * 0.6)
    t2_alloc = t2_end - time.monotonic()
    if t2_alloc > 30:
        a_opt, s_opt = tempered_restart_loop(start_a, t2_alloc, rng, verbose=True)
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, "phase2")

    # === Phase 3: Fine polishing ===
    print(f"\n=== Phase 3: Fine Polish (score={best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        t3_alloc = min(200.0, (deadline - time.monotonic()) * 0.35)
        a_fine, s_fine, _ = fine_coord_descent(best_seq, t3_alloc, label="fine")
        best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

        # Another exact CD round
        if time.monotonic() < deadline - 60:
            t3b = min(100.0, (deadline - time.monotonic()) * 0.3)
            a_e2, s_e2, _ = exact_coord_descent(best_seq, t3b, top_k=10, label="ecd_polish")
            best_seq, best_score = check_and_update(a_e2, best_seq, best_score, "ecd_polish")

    # === Phase 4: Try different lengths ===
    print(f"\n=== Phase 4: Alt Lengths (score={best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        for alt_n in [n_best + 128, n_best - 128, n_best + 256, 1024]:
            if time.monotonic() > deadline - 30:
                break
            if alt_n <= 0 or alt_n > 3000 or alt_n == n_best:
                continue
            t_alt = min(40.0, (deadline - time.monotonic()) * 0.3)
            if t_alt < 5:
                break
            # Interpolate
            old_x = np.linspace(0, 1, n_best)
            new_x = np.linspace(0, 1, alt_n)
            a_interp = np.interp(new_x, old_x, best_seq)
            a_interp = np.maximum(a_interp, 0)
            ss = a_interp.sum()
            if ss > 1e-15:
                a_interp /= ss
                a_opt, s_opt, _ = exact_coord_descent(a_interp, t_alt, top_k=6, label=f"alt{alt_n}")
                best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"alt_n{alt_n}")

    # === Phase 5: Final verification ===
    if best_seq is None:
        best_seq = np.ones(1024) / 1024

    final_seq = np.maximum(best_seq, 0.0)
    fs = final_seq.sum()
    if fs > 0.01:
        final_seq = final_seq / fs

    final_score = float(evaluate_sequence(final_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.8f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
