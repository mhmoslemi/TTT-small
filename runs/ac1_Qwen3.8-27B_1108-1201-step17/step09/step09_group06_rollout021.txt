def fast_convolve(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Compute convolution a*b using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix for fixed a."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):

    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def compute_subgradient(a, n_active_frac=0.1):
    """Compute subgradient of max(conv(a,a)) wrt a."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]

def solve_lp_scipy(a_fixed):
    """Solve the LP best-response using scipy linprog (HiGHS)."""
    n = len(a_fixed)
    m = 2 * n - 1
    
    # Build A_ub: (2n-1) x (n+1)
    # A_ub[i, j] = a_fixed[j] if j <= i <= j+n-1, else 0
    # A_ub[i, n] = -1 (for the -M term)
    A_ub = np.zeros((m, n + 1))
    for j in range(n):
        A_ub[j:j+n, j] = a_fixed[j]
    A_ub[:, n] = -1
    
    b_ub = np.zeros(m)
    
    # A_eq: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[n] = 1  # minimize M
    
    bounds = [(0, None)] * (n + 1)
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    b_val = res.x[:n]
    M_val = res.x[n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    
    return b_val, M_val

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def solve_lp_highe(a_fixed):
    """Solve LP best-response using scipy HiGHS for speed."""
    n = len(a_fixed)
    m = 2 * n - 1
    # A_ub: (2n-1) x (n+1), sparse
    A_ub = lil_matrix((m, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = a_fixed[j]
    A_ub[:, n] = -1.0
    A_ub = A_ub.tocsr()
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = [(0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 2.0})
    except Exception:
        return None, None
    if res.status != 0 or res.x is None:
        return None, None
    b_val = np.maximum(res.x[:n], 0)
    M_val = float(res.x[n])
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, M_val


def nesterov_lp(a0, time_limit, max_iters=500000, diversify_interval=300):
    """Nesterov-accelerated LP alternation with adaptive diversification."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    b_curr = fast_convolve(a)
    best_score = 2.0 * n * b_curr.max()
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(42)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_new = solve_lp_highe(y)
        if b_new is None:
            stall += 1
            if stall > 50:
                break
            continue
        
        alpha = 2.0 / (it + 2.0)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-9:
                prev_best = best_score
            else:
                stall += 1
        
        if stall >= diversify_interval and it > 500:
            # Diversify: perturb y and restart counter
            noise_scale = 0.02 + 0.01 * min(stall // diversify_interval, 5)
            noise = rng.random(n) * noise_scale
            y = y + noise
            y = np.maximum(y, 0)
            y = y / max(y.sum(), 1e-300)
            stall = 0
            prev_best = best_score
            if (it + 1) % 500 == 0:
                print(f"    NLP iter={it+1} diversify: score={best_score:.8f}")
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.1):
    """L-BFGS on log-sum-exp smoothed objective with multi-stage annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    def make_func(T_val, t0_ref):
        def func(z_in):
            if time.monotonic() - t0_ref > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            if max_Tb > 700:
                return (2.0 * n / T_val) * 700
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func
    
    def make_jac(T_val, t0_ref):
        def jac(z_in):
            if time.monotonic() - t0_ref > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            if max_Tb > 700:
                return np.zeros_like(z_in)
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-18:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac
    
    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    for stage_idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        iter_budget = max(20, int(remaining * 0.7 / (n_stages - stage_idx)))
        try:
            res = minimize(
                fun=make_func(stage_T, t0),
                x0=z,
                jac=make_jac(stage_T, t0),
                method='L-BFGS-B',
                options={'maxiter': iter_budget, 'ftol': 1e-15, 'gtol': 1e-12,
                         'maxfun': iter_budget * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        if (stage_idx + 1) % 3 == 0:
            a_check = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_check).max()
            elapsed = time.monotonic() - t0
            print(f"    LBBF stage {stage_idx+1}/{n_stages} T={stage_T:.2f} ({elapsed:.0f}s): raw_score={sc:.8f}")
    
    return softmax_from_z(z)


def local_coordinate_search(a0, time_limit, n_steps=100000):
    """Local search: move mass from high-contributing to low-contributing positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / max(a.sum(), 1e-300)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        threshold = peak_val * 0.95
        near_peaks = np.where(b >= threshold)[0]
        
        contrib = np.zeros(n)
        for k in near_peaks[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]
        
        # Find top contributors and lowest positions
        top_src = np.argsort(contrib)[::-1][:8]
        low_dst = np.argsort(contrib)[:8]
        
        improved = False
        for si in top_src:
            for di in low_dst:
                if si == di or a[si] < 1e-14:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amount = a[si] * frac
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                amount = a[i] * rng.uniform(0.01, 0.08)
                a_trial = a.copy()
                a_trial[i] -= amount
                a_trial[j] += amount
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
        
        if no_improve > 5000:
            break
        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    LocSearch step={step} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
