import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(convolve(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max() / (s * s)


def plateau_gradient(a, fraction=0.05, top_k=20):
    """
    Compute gradient of max(c) using only the plateau positions.
    c[k] = sum_i a[i]*a[k-i]
    dc[k]/da[j] = 2*a[k-j] if j in S_k, else 0
    """
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    
    # Identify plateau: positions where c[k] >= max_c * (1 - fraction)
    threshold = max_c * (1.0 - fraction)
    plateau = np.where(c >= threshold)[0]
    
    # Also include top_k positions for robustness
    top_idx = np.argsort(c)[::-1][:top_k]
    plateau = np.unique(np.concatenate([plateau, top_idx]))
    
    # Compute gradient: average of dc[k]/da for k in plateau
    grad = np.zeros(n)
    for k in plateau:
        # dc[k]/da[j] = 2*a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            # a[k-j] for j in [js, je] means a[k-je] to a[k-js]
            # which is a[k-je : k-js+1]
            start = k - je
            end = k - js + 1
            grad[js:je+1] += 2.0 * a[start:end]
    
    grad /= len(plateau)
    return grad, max_c, plateau


def exponentiated_gradient_step(a, grad, eta):
    """Multiplicative update: a_new = a * exp(-eta * grad) / sum."""
    log_a = np.log(np.maximum(a, 1e-300))
    log_a_new = log_a - eta * grad
    log_a_new -= log_a_new.max()  # for numerical stability
    exp_vals = np.exp(log_a_new)
    s = exp_vals.sum()
    if s < 1e-15:
        return a.copy()
    return exp_vals / s

def build_conv_matrix(a):
    """Build convolution matrix T where T @ b = convolve(a, b) (full)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        prob.solve(solver=cp.ECOS, verbose=False)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def nesterov_lp(a0, time_limit, solver, max_iters=50000):

    for it in range(max_iters):

            break
        
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            b_new = y.copy()  # fallback
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()

            stall = 0

def pnorm_objective(z, p):
    """2n * (sum c_k^p)^(1/p) in log-domain."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def pnorm_gradient(z, p):
    """Gradient of pnorm_objective w.r.t. z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def pn orm_lbfgs(a0, time_limit, p_start=50.0, p_end=500.0, n_stages=6):
    """L-BFGS with p-norm smoothing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    temps = np.geomspace(p_start, p_end, n_stages)
    iters_per_stage = max(50, 6000 // n_stages)
    
    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_val):
            return pnorm_objective(z_val, stage_p)
        
        def jac(z_val):
            return pnorm_gradient(z_val, stage_p)
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                                   'disp': False})
            z = res.x
        except Exception:
            pass
    
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, a_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
