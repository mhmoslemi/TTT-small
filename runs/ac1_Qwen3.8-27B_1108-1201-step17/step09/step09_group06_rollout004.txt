def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_with(a_fixed, b):
    n = len(a_fixed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a_fixed, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:

    except Exception:

    if prob.status not in ('optimal', 'optimal_inaccurate'):

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()

        return None, None

def compute_subgradient_multi(a, n_active_frac=0.05):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b

def lbgfs_refine(a0, time_limit, T_start=200.0, T_end=0.15, maxiter=12000):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_local = softmax_from_z(z_in)
            b_local = fast_convolve(a_local)
            Tb = T_val * b_local
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a_local = softmax_from_z(z_in)
            b_local = fast_convolve(a_local)
            Tb = T_val * b_local
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b_local)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_local[k - idx]
            S = np.dot(grad_a, a_local)
            return a_local * (grad_a - S)
        return jac

    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 150, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

def plateau_init(n, plateau_frac=0.4, taper_power=1.5, rng=None):
    p = int(n * plateau_frac)
    a = np.zeros(n)
    a[:p+1] = 1.0
    if p + 1 < n:
        x = np.linspace(0, 1, n - p - 1)
        a[p+1:] = (1 - x) ** taper_power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                    a_trial = a_trial / ss

                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved and step > 5000:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_best = len(best_seq) if best_seq is not None else 960

    if parent is None:
        n_init = 960
        best_seq = plateau_init(n_init, 0.4, 1.5, rng)
        best_score = evaluate_safe(best_seq)
        n_best = n_init
        print(f"Default plateau init: n={n_init}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # Phase 0: Try plateau initializations to see if any beat parent
    if parent is not None:
        frac = deadline - time.monotonic()
        if frac > 60:
            for pf in [0.3, 0.35, 0.45, 0.55]:
                for tp in [1.0, 1.5, 2.0, 3.0]:
                    a_try = plateau_init(n_best, pf, tp, rng)
                    sc = evaluate_safe(a_try)
                    if sc < best_score:
                        best_score = sc
                        best_seq = a_try.copy()
                        print(f"  Plateau init ({pf},{tp}): {sc:.6f}")

    # Phase 1: Mirror descent (first round, wider active set)
    print(f"\n=== Phase 1: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent_adaptive(best_seq, t_alloc, n_active_frac=0.05)
        try_update(a_md, "mirror_desc_1")

    # Phase 2: Nesterov LP (main refinement)
    print(f"\n=== Phase 2: Nesterov LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.28, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_lp, "nesterov_lp_1")

    # Phase 3: L-BFGS LogSumExp
    print(f"\n=== Phase 3: L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=300.0, T_end=0.1, maxiter=12000)
        try_update(a_lb, "lbgfs_1")

    # Phase 4: Second mirror descent (tighter active set)
    print(f"\n=== Phase 4: Mirror Descent r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        a_md2, _ = mirror_descent_adaptive(best_seq, t_alloc, n_active_frac=0.02)
        try_update(a_md2, "mirror_desc_2")

    # Phase 5: Peak-directed local search
    print(f"\n=== Phase 5: Peak-Directed Search (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        a_pd, _ = peak_directed_refine(best_seq, t_alloc, n_steps=40000)
        try_update(a_pd, "peak_directed")

    # Phase 6: Local swap search
    print(f"\n=== Phase 6: Local Swap (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_ls, _ = local_swap_search(best_seq, t_alloc, n_steps=60000)
        try_update(a_ls, "local_swap")

    # Phase 7: Alternative lengths
    print(f"\n=== Phase 7: Alt Lengths (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.1, 1.25, 1.5, 0.85, 0.75, 2.0]:
        alt_n = int(n_best * mult)
        if 300 <= alt_n <= 4000 and alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:3]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_total = min(rem * 0.35, 250.0)
        if t_total < 40:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        cur_best_alt = init_alt.copy()
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()

        t_lp = t_total * 0.55
        a_alt, _ = nesterov_lp_refine(cur_best_alt, t_lp, solver=solver, n_max_iters=30000)
        try_update(a_alt, f"lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None and len(best_seq) == alt_n:
            t_md = min(rem * 0.3, 80.0)
            if t_md > 10:
                a_alt2, _ = mirror_descent_adaptive(best_seq, t_md, n_active_frac=0.04)
                try_update(a_alt2, f"md_n{alt_n}")

    # Phase 8: Final deep LP polish
    print(f"\n=== Phase 8: Final LP Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.35, 100.0)
        a_fin, _ = nesterov_lp_refine(best_seq, t_fin, solver=solver, n_max_iters=20000)
        try_update(a_fin, "final_lp")

    # Phase 9: Final L-BFGS polish
    rem = deadline - time.monotonic()
    if rem > 25:
        t_lb = min(rem * 0.3, 60.0)
        a_lb2 = lbgfs_refine(best_seq, t_lb, T_start=20.0, T_end=0.05, maxiter=6000)
        try_update(a_lb2, "final_lbgfs")

    # Phase 10: Final local search
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ls = min(rem * 0.4, 40.0)
        a_ls2, _ = local_swap_search(best_seq, t_ls, n_steps=30000)
        try_update(a_ls2, "final_swap")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
