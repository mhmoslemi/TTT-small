"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

3. Spectral-aware gradient descent with line search

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_objective(a):
    """Compute 2*n*max_conv/sum^2 for normalized a (sum=1)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def lbgfs_adaptive(a0, time_limit, T_start=50.0, T_end=0.05, maxiter=20000):
    """L-BFGS with adaptive temperature based on progress."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    best_local = compute_objective(softmax_from_z(z))
    best_z_local = z.copy()
    
    # Adaptive temperature: start high, adapt based on progress
    T_current = T_start
    n_total = 0
    stall_count = 0
    prev_best = best_local
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        # Decide how many iterations for this stage
        iters = max(20, min(500, int(remaining * 5)))
        
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_current * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_current) * log_sum_exp
        
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_current * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            break
        
        n_total += iters
        a_new = softmax_from_z(z)
        score = compute_objective(a_new)
        
        if score < best_local - 1e-10:
            best_local = score
            best_z_local = z.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Adaptive temperature update
        if stall_count >= 3:
            # Stuck - increase temperature to escape
            T_current = min(T_current * 2.0, 100.0)
            # Add small perturbation
            z = z + 0.01 * np.random.standard_normal(n)
            stall_count = 0
        elif T_current > T_end:
            # Making progress - slowly decrease temperature
            T_current = max(T_current * 0.95, T_end)
    
    return softmax_from_z(best_z_local)


def spectral_gradient_descent(a0, time_limit, max_iter=50000):
    """Gradient descent with spectral awareness and line search."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()
    
    lr = 0.1
    momentum = np.zeros(n)
    beta_m = 0.9
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        
        # Subgradient of max conv
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            g[i] = a[k_star - i]
        
        # Also add contribution from top-5 peaks for smoother gradient
        b_sorted = np.sort(b)[::-1]
        peak_threshold = b_sorted[min(4, len(b_sorted)-1)]
        near_peaks = np.where(b >= peak_threshold * 0.99)[0]
        
        g_smooth = np.zeros(n)
        for k in near_peaks:

            je2 = min(n - 1, k)
            w_k = b[k] / b[max(near_peaks)]  # weight by peak value
            for i in range(js2, je2 + 1):
                g_smooth[i] += w_k * a[k - i]
        
        # Use smoothed gradient
        g = 0.5 * g + 0.5 * g_smooth
        
        # Normalize gradient
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        
        # Momentum update
        momentum = beta_m * momentum - lr * g
        a_new = a + momentum
        
        # Project to simplex (a >= 0, sum = 1)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Line search (backtracking)
        score_new = compute_objective(a_new)
        if score_new > best_score:
            # Reduce step
            a_new = 0.5 * a + 0.5 * a_new
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            score_new = compute_objective(a_new)
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            # Occasionally increase lr
            if it % 1000 == 0 and it > 0:
                lr = min(lr * 1.1, 0.5)
        else:
            # Reduce learning rate
            lr = max(lr * 0.9, 0.001)
    
    return best_a, best_score


def peak_targeted_descent(a0, time_limit, max_iter=30000):
    """Targeted descent: find which coordinates to reduce based on peak analysis."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        
        # Find top peaks (excluding edges which are harder to fix)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        top_k_vals = np.argsort(b_work)[::-1][:5]
        
        # Compute "danger" score for each coordinate
        # A coordinate i is "dangerous" if it appears in high-value products for top peaks
        danger = np.zeros(n)
        for k in top_k_vals:
            if b[k] < b[top_k_vals[0]] * 0.95:
                break
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                j = k - i
                danger[i] += a[i] * a[j] * b[k]
        
        # Find safe coordinates (low danger, currently have mass)
        safe_mask = (danger < np.median(danger)) & (a > 1e-8)
        danger_coords = np.argsort(danger)[::-1][:10]  # most dangerous
        safe_coords = np.where(safe_mask)[0][:10]  # safest with mass
        
        improved = False
        for i in danger_coords:
            if a[i] < 1e-10:
                continue
            # Try reducing coordinate i
            for frac in [0.95, 0.90, 0.85, 0.80, 0.70]:
                a_trial = a.copy()
                diff = a_trial[i] * (1 - frac)
                a_trial[i] *= frac
                # Redistribute to safe coords
                if len(safe_coords) > 0:
                    safe_weights = a_trial[safe_coords] + 1e-10
                    a_trial[safe_coords] += diff * safe_weights / safe_weights.sum()
                else:
                    # Redistribute uniformly to non-zero coords
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    a_trial[mask] += diff * a_trial[mask] / a_trial[mask].sum()
                
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = compute_objective(a_trial)
                
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            # Try pair adjustment
            k_peak = int(np.argmax(b_work))
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k_peak - idx]
            top_ci = np.argsort(contribs)[::-1][:5]
            
            found_pair = False
            for ci in top_ci:
                i = idx[ci]
                j = k_peak - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-10 or a[j] < 1e-10:
                    continue
                for f1 in [0.93, 0.88, 0.82, 0.75]:
                    for f2 in [0.93, 0.88, 0.82, 0.75]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = norm_sequence(a_trial)
                        score = compute_objective(a_trial)
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            found_pair = True
                            improved = True
                            break
                    if found_pair:
                        break
                if found_pair:
                    break
        
        if not improved:
            no_improve += 1
            if no_improve > 500:
                break
        else:
            no_improve = 0
    
    return best_a, best_score


def parametric_init(n, mode=0):
    """Create parametric initial sequence."""
    if mode == 0:  # Gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif mode == 1:  # Flat-top
        x = np.linspace(-1, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif mode == 2:  # Raised cosine
        x = np.linspace(-1, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * x))
        a = a ** 1.5
    elif mode == 3:  # Super-Gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-np.abs(x) ** 3 * 2)
    elif mode == 4:  # Bimodal
        x = np.linspace(-1, 1, n)
        a = np.exp(-(x-0.5)**2 * 8) + np.exp(-(x+0.5)**2 * 8)
    elif mode == 5:  # Wide Gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 0.5)
    else:
        x = np.linspace(-1, 1, n)
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                    best_t = -gamma  # negative means away step

        # Actually let me redo this more carefully

        best_cand_score = best_score
        
        # Toward steps

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive-temperature L-BFGS with log-sum-exp smoothing (main workhorse)
2. Spectral gradient descent with momentum and line search
3. Peak-targeted coordinate descent with danger scoring
4. Frank-Wolfe with proper line search and away steps
5. Multi-restart with structured (sinusoidal) perturbations
6. Length adjustment via interpolation
7. Final multi-algorithm polish
Key innovations: adaptive temperature that responds to progress, spectral-aware
gradient computation, danger-score-based coordinate selection, and structured
perturbations for exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

# Global mutable holder for L-BFGS temperature (avoids closures)
_T_holder = [50.0]
_n_holder = [0]
_deadline_holder = [0.0]


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def compute_objective(a):
    """Compute 2*n*max_conv/sum^2 for any non-negative a."""
    n = len(a)
    sa = a.sum()
    if sa < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (sa * sa)


def evaluate_safe(a):
    """Evaluate sequence safely via the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def _lse_func(z_in):
    """L-BFGS objective using current T from holder."""
    T = _T_holder[0]
    n = _n_holder[0]
    if time.monotonic() > _deadline_holder[0]:
        return 1e15
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    s = exp_Tb.sum()
    if s < 1e-300:
        return 0.0
    log_sum_exp = max_Tb + np.log(s)
    return (2.0 * n / T) * log_sum_exp


def _lse_jac(z_in):
    """L-BFGS gradient using current T from holder."""
    T = _T_holder[0]
    n = _n_holder[0]
    if time.monotonic() > _deadline_holder[0]:
        return np.zeros_like(z_in)
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    s = exp_Tb.sum()
    if s < 1e-300:
        return np.zeros_like(z_in)
    w = exp_Tb / s
    # grad_a[i] = sum_k 2*w[k]*a[k-i] (for valid i, k)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        wk = w[k]
        if wk < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for i in range(js, je + 1):
            grad_a[i] += 2.0 * wk * a[k - i]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_adaptive(a0, time_limit, T_start=50.0, T_end=0.05):
    """L-BFGS with adaptive temperature based on optimization progress."""
    global _T_holder, _n_holder, _deadline_holder
    n = len(a0)
    _n_holder[0] = n
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    total_deadline = t0 + time_limit
    _deadline_holder[0] = total_deadline

    best_local = compute_objective(softmax_from_z(z))
    best_z_local = z.copy()
    T_current = T_start
    stall_count = 0

    while time.monotonic() < total_deadline:
        remaining = total_deadline - time.monotonic()
        if remaining < 3:
            break

        _T_holder[0] = T_current
        iters = max(20, min(300, int(remaining * 3)))

        try:
            res = minimize(
                fun=_lse_func, x0=z, jac=_lse_jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            break

        a_new = softmax_from_z(z)
        score = compute_objective(a_new)

        if score < best_local - 1e-10:
            best_local = score
            best_z_local = z.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive temperature
        if stall_count >= 3:
            T_current = min(T_current * 2.5, 80.0)
            rng_local = np.random.default_rng(int(time.monotonic() * 1000))
            z = z + 0.02 * rng_local.standard_normal(n)
            stall_count = 0
        elif T_current > T_end:
            T_current = max(T_current * 0.93, T_end)

    return softmax_from_z(best_z_local)


def spectral_gradient_descent(a0, time_limit, max_iter=50000):
    """Gradient descent with spectral awareness, momentum, and backtracking."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()

    lr = 0.05
    momentum = np.zeros(n)
    beta_m = 0.85

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)

        # Multi-peak weighted gradient (top 3 peaks)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        top3 = np.argsort(b_work)[::-1][:3]
        peak_vals = b[top3]
        wsum = peak_vals.sum()
        if wsum < 1e-30:
            break

        g = np.zeros(n)
        for k in top3:
            wk = peak_vals[np.where(top3 == k)[0][0]] / wsum
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                g[i] += 2.0 * wk * a[k - i]

        # Normalize
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn

        # Momentum
        momentum = beta_m * momentum - lr * g
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        # Backtracking line search
        score_new = compute_objective(a_new)
        attempts = 0
        while score_new > best_score and attempts < 5:
            a_new = 0.5 * a + 0.5 * a_new
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            score_new = compute_objective(a_new)
            attempts += 1

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            if it % 2000 == 0 and it > 0:
                lr = min(lr * 1.05, 0.3)
        else:
            lr = max(lr * 0.92, 0.002)

    return best_a, best_score


def peak_targeted_descent(a0, time_limit, max_iter=30000):
    """Targeted descent using danger scores from peak analysis."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        top_k = np.argsort(b_work)[::-1][:4]

        # Compute danger score
        danger = np.zeros(n)
        max_bk = b[top_k[0]]
        for k in top_k:
            if b[k] < max_bk * 0.93:
                break
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                j = k - i
                danger[i] += a[i] * a[j] * (b[k] / max_bk)

        # Find most dangerous coordinates with mass
        danger_weighted = danger * (a > 1e-10).astype(float) * a
        top_danger = np.argsort(danger_weighted)[::-1][:12]

        # Find safe redistribution targets
        safe_mask = (danger < np.percentile(danger, 30)) & (a > 1e-10)
        safe_coords = np.where(safe_mask)[0]
        if len(safe_coords) == 0:
            safe_coords = np.argsort(danger)[:n // 4]

        improved = False
        for i in top_danger:
            if a[i] < 1e-10:
                continue
            for frac in [0.94, 0.90, 0.85, 0.78, 0.70]:
                a_trial = a.copy()
                diff = a_trial[i] * (1.0 - frac)
                a_trial[i] *= frac
                if len(safe_coords) > 0:
                    sw = a_trial[safe_coords] + 1e-12
                    a_trial[safe_coords] += diff * sw / sw.sum()
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = compute_objective(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Pair adjustment at the main peak
            k_peak = int(np.argmax(b_work))
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k_peak - idx]
            top_ci = np.argsort(contribs)[::-1][:6]

            found = False
            for ci in top_ci:
                i = idx[ci]
                j = k_peak - i
                if i == j or i >= n or j >= n or i < 0 or j < 0:
                    continue
                if a[i] < 1e-10 or a[j] < 1e-10:
                    continue
                for f1 in [0.93, 0.88, 0.82]:
                    for f2 in [0.93, 0.88, 0.82]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = norm_sequence(a_trial)
                        score = compute_objective(a_trial)
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            found = True
                            improved = True
                            break
                    if found:
                        break
                if found:
                    break

        if not improved:
            no_improve += 1
            if no_improve > 800:
                break
        else:
            no_improve = 0

    return best_a, best_score


def frank_wolfe_refine(a0, time_limit, max_steps=25000):
    """Frank-Wolfe with line search and away steps."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_star = int(np.argmax(b))

        # Subgradient
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            g[i] = a[k_star - i]

        i_min = int(np.argmin(g))
        best_cand = None
        best_cand_score = best_score

        # Toward steps
        for t in [2.0 / (step + 2), 0.4, 0.25, 0.15, 0.08, 0.04, 0.02]:
            a_trial = (1 - t) * a + t * np.eye(n)[i_min]
            score = compute_objective(a_trial)
            if score < best_cand_score:
                best_cand_score = score
                best_cand = a_trial

        # Away steps
        mask = a > 1e-10
        if mask.sum() > 3:
            g_masked = g.copy()
            g_masked[~mask] = -np.inf
            i_away = int(np.argmax(g_masked))
            for gamma in [0.25, 0.15, 0.10, 0.05, 0.02]:
                a_trial = a.copy()
                a_trial[i_away] *= (1.0 - gamma)
                a_trial = norm_sequence(a_trial)
                score = compute_objective(a_trial)
                if score < best_cand_score:
                    best_cand_score = score
                    best_cand = a_trial

        if best_cand is not None:
            a = best_cand
            best_score = best_cand_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 600:
                break

    return best_a, best_score


def parametric_init(n, mode=0):
    """Create parametric initial sequence."""
    if mode == 0:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2.0)
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.cos(np.pi * x / 2.0) ** 2
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = (0.5 * (1 + np.cos(np.pi * x))) ** 1.5
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = np.exp(-np.abs(x) ** 3 * 2.0)
    elif mode == 4:
        x = np.linspace(-1, 1, n)
        a = np.exp(-(x - 0.4) ** 2 * 8) + np.exp(-(x + 0.4) ** 2 * 8) + 0.3
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    rem = deadline - time.monotonic()
    if rem > 10:
        t11 = min(rem * 0.5, 50.0)
        a11 = lbgfs_adaptive(best_seq, t11, T_start=3.0, T_end=0.02)
        try_update(a11, "ultra_low_T")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
