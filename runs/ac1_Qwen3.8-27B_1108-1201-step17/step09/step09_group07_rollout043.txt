"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated Frank-Wolfe with
top-k soft subgradients, Nesterov projected gradient on log-sum-exp,
LP-based refinement, and adaptive peak shaving.
Key innovations: FW method with direct O(n*k) subgradient, spectral initialization,
hybrid multi-algorithm search with diverse restarts.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def topk_subgradient_direct(a, k_top=10):
    """Compute top-k soft subgradient of max_k conv(a,a)[k] in O(n*k_top)."""
    n = len(a)
    b = fast_convolve(a)
    size = len(b)

    top_k = min(k_top, size)
    part = np.argpartition(b, -top_k)[-top_k:]
    top_vals = b[part]
    top_max = top_vals.max()
    w = np.exp(top_vals - top_max)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(top_k) / top_k
    else:
        w /= w_sum

    g = np.zeros(n)
    for idx in range(top_k):
        k = part[idx]
        wi = w[idx]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # g[i] += 2*wi*a[k-i] for i in [i_start, i_end]
        # a[k-i] for i=i_start..i_end is a[k-i_start], a[k-i_start-1], ..., a[k-i_end]
        # = a[k-i_end : k-i_start+1][::-1]
        g[i_start:i_end + 1] += 2.0 * wi * a[k - i_end:k - i_start + 1][::-1]

    return g


def grad_smooth_max(a, T):
    """Gradient of (1/T)*log(sum_k exp(T*conv_k)) w.r.t. a via FFT."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()

    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def accelerated_fw(a0, time_limit, k_top=10, verbose=True, label=""):
    """Accelerated Frank-Wolfe with top-k subgradient."""
    n = len(a0)
    x = project_simplex(np.maximum(a0, 0).copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    t_count = 0
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        t_count += 1
        # Nesterov momentum
        if t_count == 1:
            beta = 0.0
        else:
            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * t_count * t_count) / 2.0
            t_prev = 1.0 + np.sqrt(1.0 + 4.0 * (t_count - 1) ** 2) / 2.0
            beta = (t_prev - 1.0) / t_curr

        y = x + beta * (x - x_prev)
        y = project_simplex(y)

        # Subgradient at y
        g = topk_subgradient_direct(y, k_top)

        # Linear minimization: pick min coordinate
        j_star = int(np.argmin(g))

        # Step size with occasional restart
        gamma = 2.0 / (t_count + 2.0)

        x_prev = x.copy()
        x[j_star] += gamma
        # Renormalize to simplex (faster than full projection for small updates)
        s = x.sum()
        if s > 0:
            x = x / s
        else:
            x = np.ones(n) / n

        # Evaluate periodically

                    best_score = sc_pert
                    best_a = x.copy()
                else:
                    x = y_pert
                stall = 0

                elapsed = time.monotonic() - t0

def nesterov_pg(a0, time_limit, T_start=2.0, T_end=300.0, lr_init=0.08,
                 n_stages=10, steps_per_stage=3000, verbose=True, label=""):
    """Nesterov accelerated projected gradient with log-sum-exp smoothing."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            global_step = stage * steps_per_stage + step
            lr = lr_init / (1.0 + 0.005 * global_step)

            if step_count == 0:
                beta = 0.0
            else:
                t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
                t_prev = 1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) ** 2) / 2.0
                beta = (t_prev - 1.0) / t_curr

            y = x + beta * (x - x_prev)
            y = project_simplex(y)

            grad = grad_smooth_max(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1

            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
                if verbose and step_count % 2000 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"      {label} PG step={step_count} ({elapsed:.0f}s): score={best_score:.6f} T={T:.1f}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_refine(a0, time_limit, solver=None, n_max_iters=50000):

    n = len(a0)

    best_a = a.copy()

    stall_count = 0

    rng = np.random.default_rng(777)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)

        score_a = score_normalized(a)
        score_y = score_normalized(y)
        if score_a < best_score:

            stall_count = 0

            stall_count = 0

        if it % 500 == 0:

                if a[i] > 1e-14:
                    a_trial[i] *= 0.97
            a_trial = project_simplex(a_trial)
            score = score_normalized(a_trial)
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:
                break

    return best_a, best_score


def spectral_init(n, rng):
    """Initialize from random-phase flat spectrum, clipped to non-negative."""
    fft_len = 1 << (n - 1).bit_length()
    phase = rng.standard_normal(fft_len // 2 + 1)
    # Ensure conjugate symmetry for real output
    spectrum = np.exp(1j * phase)
    full_spectrum = np.zeros(fft_len, dtype=complex)
    full_spectrum[:fft_len // 2 + 1] = spectrum
    for i in range(1, fft_len // 2 + 1):
        full_spectrum[fft_len - i] = np.conj(full_spectrum[i])
    a = np.real(irfft(full_spectrum, fft_len)[:n])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s
    return a


def geometric_init(n, rate, rng):
    """Geometric (exponential decay) initialization with random start position."""
    a = np.zeros(n)
    start = int(rng.integers(0, n // 2))
    for j in range(n):
        dist = abs(j - start)
        a[j] = rate ** dist
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s
    return a


def two_bump_init(n, frac, rng):
    """Two-bump initialization: mass concentrated in two regions."""
    a = np.zeros(n)
    w = max(2, int(n * frac))
    pos1 = int(rng.integers(0, n // 3))
    pos2 = int(rng.integers(2 * n // 3, n - w))
    width1 = max(1, int(w * rng.uniform(0.5, 1.5)))
    width2 = max(1, int(w * rng.uniform(0.5, 1.5)))
    a[pos1:pos1 + width1] = 1.0
    a[pos2:pos2 + width2] = 1.0
    # Add small noise to break symmetry
    a += 0.01 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s
    return a


def mixed_init(n, rng):
    """Mix of parent-like structure with random perturbation."""
    x = np.linspace(0, 1, n)
    mode = rng.integers(0, 4)
    alpha = rng.uniform(0.3, 1.0)
    if mode == 0:
        a = 1.0 + alpha * np.sin(2 * np.pi * x + rng.uniform(0, 2 * np.pi))
    elif mode == 1:
        a = 1.0 + alpha * (2 * x - 1)
    elif mode == 2:
        a = np.exp(-2.0 * (x - rng.uniform(0.2, 0.5)) ** 2 / 0.05) + \
            0.5 * np.exp(-2.0 * (x - rng.uniform(0.5, 0.8)) ** 2 / 0.05)
    else:
        a = 1.0 + alpha * np.sin(2 * np.pi * rng.uniform(1, 5) * x)
    a = np.maximum(a, 1e-10)
    a += 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-algorithm optimization:
    1. Diverse initializations (parent, spectral, geometric, two-bump, mixed)
    2. Accelerated Frank-Wolfe with top-k subgradients (main workhorse)
    3. Nesterov projected gradient with log-sum-exp (escape mechanism)
    4. LP-based best response refinement (precision polish)
    5. Peak shaving (local refinement)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    n_default = 960
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        best_seq = mixed_init(n_default, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_default}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Accelerated Frank-Wolfe (main optimizer) ===
    print(f"\n=== Phase 1: Accelerated FW n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 360.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_fw, _ = accelerated_fw(best_seq, t_alloc, k_top=8, label="main")
        try_update(a_fw, "fw_main")

    # === Phase 2: Nesterov PG (different smoothing) ===
    print(f"\n=== Phase 2: Nesterov PG (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ng, _ = nesterov_pg(best_seq, t_alloc, T_start=5.0, T_end=500.0,
                              lr_init=0.05, n_stages=8, steps_per_stage=2000, label="pg1")
        try_update(a_ng, "pg_1")

    # === Phase 3: Second FW with different k_top ===
    print(f"\n=== Phase 3: FW k=20 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_fw2, _ = accelerated_fw(best_seq, t_alloc, k_top=20, label="fw20")
        try_update(a_fw2, "fw_k20")

    # === Phase 4: LP Refinement ===
    print(f"\n=== Phase 4: LP Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=60000)
        try_update(a_lp, "lp_main")

    # === Phase 5: Diverse multi-start exploration ===
    print(f"\n=== Phase 5: Multi-Start Exploration (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    inits = []

    # Spectral init
    if rem > 120:
        inits.append(("spectral", spectral_init(n_best, rng)))

    # Geometric inits with different rates
    for rate in [0.97, 0.94]:
        if rem > 100:
            inits.append((f"geom_{rate}", geometric_init(n_best, rate, rng)))

    # Two-bump inits
    for frac in [0.1, 0.2]:
        if rem > 100:
            inits.append((f"bump_{frac}", two_bump_init(n_best, frac, rng)))

    # Mixed inits
    inits.append(("mixed1", mixed_init(n_best, rng)))
    inits.append(("mixed2", mixed_init(n_best, rng)))

    for name, a_init in inits:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_sub = min(rem * 0.15, 120.0)
        if t_sub < 30:
            break
        print(f"  {name}: budget={t_sub:.0f}s")
        # Quick FW
        t_fw = t_sub * 0.5
        a_sub, _ = accelerated_fw(a_init, t_fw, k_top=10, label=name)
        try_update(a_sub, f"{name}_fw")
        # Quick PG if time remains
        rem = deadline - time.monotonic()
        if rem > 30 and t_sub * 0.3 > 15:
            a_sub2, _ = nesterov_pg(a_sub, t_sub * 0.3, T_start=3.0, T_end=200.0,
                                     lr_init=0.06, n_stages=5, steps_per_stage=1000, label=name)
            try_update(a_sub2, f"{name}_pg")

    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = []
    for delta in [-100, -50, -20, 20, 50, 100]:
        alt_n = n_best + delta
        if 256 <= alt_n <= 2560:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_total = min(rem * 0.15, 100.0)
        if t_total < 25:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = accelerated_fw(init_alt, t_total * 0.6, k_top=10, label=f"n{alt_n}")
        try_update(a_alt, f"fw_n{alt_n}")
        rem = deadline - time.monotonic()
        if rem > 20:
            a_alt2, _ = lp_refine(a_alt, min(rem * 0.3, 30.0), solver=solver, n_max_iters=3000)
            try_update(a_alt2, f"lp_n{alt_n}")

    # === Phase 7: Final LP polish on best ===
    print(f"\n=== Phase 7: Final LP Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.25, 120.0)
        a_fin, _ = lp_refine(best_seq, t_fin, solver=solver, n_max_iters=80000)
        try_update(a_fin, "final_lp")

    # === Phase 8: Final peak shaving ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.20, 40.0)
        a_ps, _ = peak_shave_local(best_seq, t_ps, n_steps=30000)
        try_update(a_ps, "final_ps")

    # === Phase 9: One more FW pass with very small k (aggressive) ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fw_final = min(rem * 0.3, 60.0)
        a_fwf, _ = accelerated_fw(best_seq, t_fw_final, k_top=3, label="fw_final")
        try_update(a_fwf, "fw_final_k3")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")

    return [float(x) for x in best_seq]
