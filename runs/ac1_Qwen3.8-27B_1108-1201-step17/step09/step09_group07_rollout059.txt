def fast_convolve(a):

    return irfft(A * A, fft_size)[:size]

def convolve_grad(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    Result: 2 * (w cross-correlate a) truncated to length n."""
    n = len(a)

    A = rfft(a, fft_size)

def lp_best_response(a_fixed, solver='ECOS'):
    """Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint matrix
    size = 2 * n - 1
    # conv(a,b)[k] = sum_i a[i] * b[k-i]
    # We need: for all k, sum_i (a[i] if 0<=k-i<n else 0) * b[k-i] <= M
    # Let j = k-i, so i = k-j, and we need 0 <= k-j < n, i.e., k-n+1 <= j <= k
    # conv[k] = sum_j a[k-j] * b[j] where 0<=k-j<n and 0<=j<n
    
    # Build the matrix T of shape (2n-1, n) where T[k,j] = a[k-j] if valid
    T = np.zeros((size, n))
    for j in range(n):
        for k in range(j, j + n):
            T[k, j] = a_fixed[k - j]
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        prob.solve(solver='SCS')
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Strongly asymmetric initializations (ramp, sigmoid, poly)
2. Nesterov accelerated projected gradient with log-sum-exp smoothing and adaptive lr
3. LP best-response refinement with Nesterov dual averaging
4. Multi-length exploration
5. High-temperature final polishing
Key innovation: fan-shaped asymmetric inits exploit the anti-palindromic structure
needed to minimize the center of the convolution, combined with aggressive T annealing.

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_grad(a, w):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a (length n)."""
    n = len(a)
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def lse_obj_grad(a, T):
    """Log-sum-exp of conv(a,a) scaled by 1/T, and its gradient."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    se = e.sum()
    w = e / se
    obj = (m + np.log(se)) / T
    grad = convolve_grad(a, w)
    return obj, grad


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def score_of(a):
    """Score for normalized (sum=1) non-negative sequence."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def make_fan_init(n, beta, center_frac, rng):
    """Fan-shaped (asymmetric bell) initialization centered at center_frac."""
    x = np.arange(n) / n
    a = np.exp(-beta * (x - center_frac) ** 2)
    a = a / a.sum()
    return a


def make_ramp_init(n, beta, rng):
    """Exponential ramp initialization."""
    x = np.arange(n) / n
    a = np.exp(beta * x)
    a = a / a.sum()
    return a


def make_poly_init(n, p, rng):
    """Polynomial ramp initialization."""
    x = np.arange(n) / max(n - 1, 1)
    a = (x + 0.05) ** p
    a = a / a.sum()
    return a


def make_sigmoid_init(n, sharpness, center, rng):
    """Sigmoid-like (step) initialization."""
    x = np.arange(n) / n
    a = 1.0 / (1.0 + np.exp(-sharpness * (x - center)))
    a = a / a.sum()
    return a


def make_two_bump_init(n, rng):
    """Two-bump asymmetric initialization."""
    c1 = rng.uniform(0.1, 0.4)
    c2 = rng.uniform(0.5, 0.9)
    w1 = rng.uniform(0.05, 0.15)
    w2 = rng.uniform(0.05, 0.15)
    amp2 = rng.uniform(0.3, 0.8)
    x = np.arange(n) / n
    a = np.exp(-((x - c1) / w1) ** 2) + amp2 * np.exp(-((x - c2) / w2) ** 2)
    a = a / a.sum()
    return a


def nesterov_projected(a0, time_limit, T_start=2.0, T_end=500.0, lr_init=0.05,
                       n_stages=15, steps_per_stage=3000):
    """Nesterov accelerated projected gradient with T annealing."""
    n = len(a0)
    x = project_simplex(np.maximum(a0, 0))
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_of(x)
    t0 = time.monotonic()
    step = 0
    temps = np.geomspace(T_start, T_end, n_stages)

    for stage in range(n_stages):
        T = temps[stage]
        for _ in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step * step) / 2.0
            if step > 0:
                t_prev_v = 1.0 + np.sqrt(1.0 + 4.0 * (step - 1) ** 2) / 2.0
                beta = (t_prev_v - 1.0) / t_curr
            else:
                beta = 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)

            _, grad = lse_obj_grad(y, T)
            lr = lr_init / (1.0 + 0.005 * step)
            x_new = project_simplex(y - lr * grad)

            x_prev = x.copy()
            x = x_new
            step += 1

            if step % 30 == 0:
                sc = score_of(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
                    if step % 300 == 0:
                        el = time.monotonic() - t0
                        print(f"      NG s={step} ({el:.0f}s) T={T:.1f} score={best_score:.6f}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a_fixed, solver='ECOS'):
    """Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver='SCS')
        except Exception:
            return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in ['SCS', 'CLARABEL', 'ECOS']:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_refine_nesterov(a0, time_limit, solver='ECOS', n_max_iters=10000):
    """LP best-response with Nesterov dual averaging outer loop."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, _ = lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = project_simplex(y)

        a = ((it + 1) * a + (it + 2) * alpha * b_new) / (it + 1 + (it + 2) * alpha)
        a = project_simplex(a)

        if (it + 1) % 50 == 0:
            sc_a = score_of(a)
            sc_y = score_of(y)
            sc = min(sc_a, sc_y)
            if sc < best_score:
                best_score = sc
                best_a = a.copy() if sc_a <= sc_y else y.copy()
                stall = 0
            else:
                if sc < prev_best - 1e-8:
                    stall = 0
                else:
                    stall += 1
            prev_best = best_score

            if stall >= 6:
                rng = np.random.default_rng(it * 7 + 3)
                y = y + 0.03 * rng.standard_normal(n)
                y = project_simplex(y)
                stall = 0

            if (it + 1) % 500 == 0:
                el = time.monotonic() - t0
                print(f"      LP it={it+1} ({el:.0f}s) score={best_score:.6f}")

    return best_a, best_score


def lbfgs_optimize(a0, time_limit, T_start=5.0, T_end=500.0):
    """L-BFGS-B optimization with softmax reparametrization."""
    from scipy.optimize import minimize
    n = len(a0)
    a_init = np.maximum(a0, 1e-15)
    a_init = a_init / a_init.sum()
    x = np.log(a_init)
    best_a = a_init.copy()
    best_score = score_of(a_init)
    t0 = time.monotonic()
    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)

    for stage, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break

        def fun(xv):
            ax = xv - xv.max()
            e = np.exp(ax)
            ax = e / e.sum()
            val, _ = lse_obj_grad(ax, T)
            return val

        def jac(xv):
            ax = xv - xv.max()
            e = np.exp(ax)
            ax = e / e.sum()
            _, grad_a = lse_obj_grad(ax, T)
            dot = np.dot(grad_a, ax)
            return ax * (grad_a - dot)

        iters = max(30, min(300, 3000 // n_stages))
        try:
            res = minimize(fun, x, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-12, 'gtol': 1e-8})
            x = res.x
        except Exception:
            break

        ax = x - x.max()
        e = np.exp(ax)
        ax = e / e.sum()
        sc = score_of(ax)
        if sc < best_score:
            best_score = sc
            best_a = ax.copy()

    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=20000):
    """Local peak shaving via coordinate descent."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]
        threshold = peak_val - 0.03 * peak_val
        near_peaks = np.where(b_work >= threshold)[0]

        contrib_count = np.zeros(n)
        for k in near_peaks[:4]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            top = np.argsort(contribs)[::-1][:6]
            for ci in top:
                contrib_count[idx[ci]] += 1

        top_coords = np.argsort(contrib_count)[::-1][:8]
        improved = False

        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.96, 0.93, 0.89, 0.85]:
                a_t = a.copy()
                a_t[i] *= factor
                a_t = project_simplex(a_t)
                sc = score_of(a_t)
                if sc < best_score - 1e-15:
                    a = a_t
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_c = np.argsort(contribs)[::-1][:4]
            for ci in top_c:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.94, 0.90]:
                    for f2 in [0.94, 0.90]:
                        a_t = a.copy()
                        a_t[i] *= f1
                        a_t[j] *= f2
                        a_t = project_simplex(a_t)
                        sc = score_of(a_t)
                        if sc < best_score - 1e-15:
                            a = a_t
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            break

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi = fi / ss
    return fi

            a_lb, _ = lbfgs_optimize(a_opt, t_total * 0.3, T_start=10.0, T_end=400.0)
            try_update(a_lb, f"lbfgs_n{alt_n}")

    # === Phase 7: Final high-T Nesterov polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.25, 90.0)
        a_fin, _ = nesterov_projected(best_seq, t_fin, T_start=100.0, T_end=3000.0,
                                      lr_init=0.01, n_stages=8, steps_per_stage=2000)
        try_update(a_fin, "polish_highT")

    # === Phase 8: Final LP if time ===
    rem = deadline - time.monotonic()
    if rem > 40:
        t_final = min(rem * 0.5, 120.0)
        print(f"\n=== Phase 8: Final LP (best={best_score:.6f}) budget={t_final:.0f}s ===")
        a_fl, _ = lp_refine_nesterov(best_seq, t_final, solver=solver, n_max_iters=10000)
        try_update(a_fl, "lp_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f} n={len(best_seq)}")

    return [float(x) for x in best_seq]
