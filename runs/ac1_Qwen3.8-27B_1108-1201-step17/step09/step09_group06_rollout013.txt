import numpy as np

import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgradient_full(a):
    """Compute the full subgradient of max(conv(a,a)) at a."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    peak_positions = np.where(b >= max_b - 1e-12)[0]
    g = np.zeros(n)
    for k in peak_positions:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            g[j] += a[k - j]
    g /= len(peak_positions)
    return g, max_b


def compute_subgradient_active(a, frac=0.05):
    """Compute subgradient from active (near-peak) convolution positions."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b - frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum velocity
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    
    lr = 0.1
    no_improve = 0
    step = 0

        y = a + momentum * v
        y = project_simplex(y)
        
        # Compute subgradient at y

        # Update velocity

        a_new = a + v
        a_new = project_simplex(a_new)
        
        # Evaluate
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()

            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt learning rate

            if no_improve > 500:

            print(f"    PGD {elapsed:.0f}s step={step}: score={best_score:.6f} lr={lr:.4f}")
            break  # Safety
    
    return best_a, best_score

    M_var = cp.Variable()
    
    # Build constraints using convolution structure
    # conv(a_fixed, b)_k = sum_j a_fixed[j] * b[k-j] for valid k,j
    constraints = []
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            # sum_{j=js}^{je} a_fixed[j] * b[k-j] <= M
            expr = a_fixed[js:je+1] @ b_var[k-je:k-js+1][::-1]
            constraints.append(expr <= M_var)
    
    constraints.append(cp.sum(b_var) == 1)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def solve_lp_dual(a_fixed, solver=None):
    """Solve the dual LP to get lambda (weights on convolution positions).
    Dual: max mu s.t. (T^T lambda)_j >= mu, sum(lambda)=1, lambda>=0
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    size = 2 * n - 1
    
    lambda_var = cp.Variable(size, nonneg=True)
    mu_var = cp.Variable()
    
    constraints = [cp.sum(lambda_var) == 1]
    
    # (T^T lambda)_j >= mu for all j
    for j in range(n):
        # (T^T lambda)_j = sum_k lambda_k * T[k,j] = sum_k lambda_k * a_fixed[k-j] (valid k)
        # T[k,j] = a_fixed[k-j] if 0 <= k-j < n
        ks = j  # k-j >= 0 → k >= j
        ke = j + n - 1  # k-j < n → k < j+n
        ks = max(0, ks)
        ke = min(size - 1, ke)
        if ks <= ke:
            weights = a_fixed[0:ke-ks+1]  # a_fixed[k-j] for k from ks to ke
            expr = weights @ lambda_var[ks:ke+1]
            constraints.append(expr >= mu_var)
    
    prob = cp.Problem(cp.Maximize(mu_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    lambda_val = np.asarray(lambda_var.value, dtype=np.float64)
    lambda_val = np.maximum(lambda_val, 0)
    s = lambda_val.sum()
    if s < 1e-15:
        return None, None
    lambda_val = lambda_val / s
    return lambda_val, float(mu_var.value)


def lp_alternating_adaptive(a0, time_limit, solver=None, n_max_iters=500000):
    """LP alternating with adaptive Polyak step size."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Nesterov variables
    y = a.copy()
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    
    # Lower bound on optimal value: 2n/(2n-1) (all convolution terms equal)
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP best response for y
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        # Polyak step size
        gap = best_score - f_lb
        if gap < 1e-10:
            break
        # M_b is the max convolution value when a=y, b=b_new
        # The "gradient" magnitude is related to M_b
        polyak_step = gap / (2.0 * M_b * best_score + 1e-15)
        alpha = min(2.0 / (it + 2.0), max(0.01, polyak_step))
        alpha = min(alpha, 0.5)  # Cap for stability
        
        # Nesterov update
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        # Evaluate
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            # Stall-breaking
            if stall_count >= 6:
                rng = np.random.default_rng(hash((it, n)) % (2**31))
                noise_scale = 0.01 + 0.02 * stall_count
                noise = rng.random(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
            
            print(f"    LP-Alt {elapsed:.0f}s iter={it+1}: score={best_score:.8f} alpha={alpha:.4f}")
    
    return best_a, best_score


def lbgfs_smooth(a0, time_limit, T_start=50.0, T_end=0.05, maxiter=20000):
    """L-BFGS on smooth LogSumExp approximation with aggressive annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(np.maximum(a, 1e-300)).copy()
    
    t0 = time.monotonic()
    
    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_in = softmax_from_z(z_in)
            b = fast_convolve(a_in)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func
    
    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a_in = softmax_from_z(z_in)
            b = fast_convolve(a_in)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            # Gradient of sum_k w_k * b_k with respect to a
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_in[k - idx]
            S = np.dot(grad_a, a_in)
            return a_in * (grad_a - S)
        return jac
    
    n_stages = 12
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def coordinate_local_search(a0, time_limit, n_rounds=50):
    """Coordinate-wise local search: for each coordinate, find optimal value."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    b = fast_convolve(a)
    best_max_b = b.max()
    best_score = 2.0 * n * best_max_b
    t0 = time.monotonic()
    
    for round_num in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        
        # Process coordinates in order of their contribution to the peak
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # Contribution of coordinate j to the peak
        contrib = np.zeros(n)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        for j in range(js, je + 1):
            contrib[j] = a[j] * a[peak_k - j]
        
        order = np.argsort(contrib)[::-1]
        
        for idx in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[idx] < 1e-15:
                continue
            
            # Try reducing a[idx] and redistributing
            # The optimal a[idx] minimizes max_k conv(a,a)_k
            # For a fixed a[idx], the rest is fixed, so we can do a 1D search
            old_val = a[idx]
            
            # Try a few candidate values
            for frac in [0.5, 0.7, 0.85, 0.9, 1.15, 1.3]:
                a_trial = a.copy()
                delta = (frac - 1.0) * old_val
                a_trial[idx] = old_val + delta
                if a_trial[idx] < 0:
                    continue
                # Redistribute delta to coordinates with low contribution
                if abs(delta) > 1e-10:
                    # Add to the coordinate with lowest contribution
                    low_idx = int(np.argmin(contrib))
                    if low_idx != idx:
                        a_trial[low_idx] += delta
                        if a_trial[low_idx] < 0:
                            a_trial[low_idx] = 0
                            a_trial[idx] = old_val  # revert
                            continue
                
                a_trial = np.maximum(a_trial, 0)
                s_trial = a_trial.sum()
                if s_trial < 1e-15:
                    continue
                a_trial = a_trial / s_trial
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    best_max_b = score / (2.0 * n)
                    improved = True
                    break
        
        if not improved and round_num > 5:
            break
    
    return best_a, best_score


def pairwise_swap_search(a0, time_limit, n_rounds=100000):
    """Pairwise swap local search: move mass from one coordinate to another."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    no_improve = 0
    
    for step in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_val = b.max()
        threshold = peak_val - 0.03 * peak_val
        near_peaks = np.where(b >= threshold)[0]
        
        # Compute contribution to peaks
        contrib = np.zeros(n)
        for k in near_peaks[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]
        
        # Try moving mass from high-contribution to low-contribution
        top_src = np.argsort(contrib)[::-1][:10]
        low_dst = np.argsort(contrib)[:10]
        
        improved = False
        for si in top_src:
            if improved:
                break
            for di in low_dst:
                if si == di or a[si] < 1e-14:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amount = a[si] * frac
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved:
            no_improve += 1
            # Random perturbation
            if no_improve % 50 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-14:
                        continue
                    amount = a[i] * 0.02
                    a_trial = a.copy()
                    a_trial[i] -= amount
                    a_trial[j] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        no_improve = 0
                        break
        else:
            no_improve = 0
        
        if no_improve > 500:
            break
    
    return best_a, best_score


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 1.5
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    elif choice == 6:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1 - 2 * np.abs(x - 0.5)
    else:
        # Uniform with slight noise
        a = np.ones(n) + 0.1 * rng.standard_normal(n)
        a = np.maximum(a, 0)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
