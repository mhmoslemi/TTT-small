"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Mirror descent (multiplicative weights) with top-K peak subgradient and backtracking
2. Periodic L-BFGS LogSumExp polishing integrated into the main loop

5. Multi-length exploration

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgradient_topk(a, K=3):
    """Compute subgradient using top-K peaks of the autocorrelation."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    
    # Get top K peak positions
    # Use a threshold to include near-peaks
    threshold = max_b - 1e-10
    peak_indices = np.where(b >= threshold)[0]
    
    # If too many, take top K
    if len(peak_indices) > K:
        # Sort by value and take top K
        vals = b[peak_indices]
        top_k_local = np.argsort(vals)[::-1][:K]
        peak_indices = peak_indices[top_k_local]
    
    # Compute subgradient: g[m] = 2 * sum_{k in peaks} a[k-m] / len(peaks)
    # (using uniform weights for simplicity)
    g = np.zeros(n)
    for k in peak_indices:
        # a[k-m] is valid when 0 <= k-m < n, i.e., k-n+1 <= m <= k
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_end >= m_start:
            # g[m] += 2 * a[k-m] for m in [m_start, m_end]
            # a[k-m] for m in [m_start, m_end] is a[k-m_start], a[k-m_start-1], ...
            # i.e., a[m_end], a[m_end-1], ..., a[m_start]
            # So g[m_start:m_end+1] += 2 * a[m_end:m_start-1:-1]
            g[m_start:m_end+1] += 2.0 * a[m_end:m_start-1:-1]
    
    g /= len(peak_indices)
    return g, max_b, b


def mirror_descent_backward(a0, time_limit, K=3, initial_lr=1.0, max_inner=5, polish_every=5000, polish_time=5.0):
    """Mirror descent with backtracking line search and periodic L-BFGS polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    f_lb = 2.0 * n / (2.0 * n - 1)
    
    t0 = time.monotonic()
    lr = initial_lr
    no_improve = 0
    
    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Periodic L-BFGS polish
        if step > 0 and step % polish_every == 0:
            remaining_polish = time_limit - (time.monotonic() - t0)
            if remaining_polish > polish_time + 10:
                a_polished = _quick_lbfgs(a, polish_time)
                if a_polished is not None:
                    sc = 2.0 * n * fast_convolve(a_polished).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a_polished.copy()
                        a = a_polished.copy()
                        no_improve = 0
                    else:
                        # Revert
                        a = best_a.copy()
        
        # Compute subgradient
        g, max_b, b = compute_subgradient_topk(a, K)
        g_norm_inf = np.abs(g).max()
        if g_norm_inf < 1e-15:
            break
        
        gap = best_score - f_lb
        if gap < 1e-10:
            break
        
        # Backtracking line search
        eta = min(lr, gap / (2.0 * g_norm_inf * best_score + 1e-15))
        eta = max(eta, 1e-6)
        
        improved = False
        for inner in range(max_inner):
            a_new = a * np.exp(-eta * g)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                eta *= 0.5
                continue
            a_new = a_new / s_new
            
            b_new = fast_convolve(a_new)
            score_new = 2.0 * n * b_new.max()
            
            if score_new < best_score - 1e-12:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new
                no_improve = 0
                improved = True
                # Increase lr slightly
                lr = min(lr * 1.02, 5.0)
                break
            else:
                eta *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve >= 50:
                lr *= 0.8
                no_improve = 0
                # Try a perturbation
                if no_improve % 3 == 0:
                    rng = np.random.default_rng(step)
                    noise = rng.random(n) * 0.01
                    a = a + noise
                    a = np.maximum(a, 0)
                    a = a / a.sum()
            lr = max(lr, 0.01)
        
        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD step={step} ({elapsed:.0f}s): score={best_score:.6f} lr={lr:.4f} K={K}")
    
    return best_a, best_score


def _quick_lbfgs(a0, time_limit, T=5.0):
    """Quick L-BFGS polish on log-sum-exp objective."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30))
    t0 = time.monotonic()
    
    def func(z_in):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = _softmax(z_in)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        return (2.0 * n / T) * (max_Tb + np.log(np.exp(Tb - max_Tb).sum()))
    
    def jac(z_in):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = _softmax(z_in)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        # grad wrt a: 2 * sum_k w[k] * a[k-m] for each m
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            m_start = max(0, k - n + 1)
            m_end = min(n - 1, k)
            if m_end >= m_start:
                grad_a[m_start:m_end+1] += 2.0 * w[k] * a[m_end:m_start-1:-1]
        # Chain rule: d/ dz = a * (grad_a - sum(grad_a * a))
        S = np.dot(grad_a, a)
        return a * (grad_a - S)
    
    try:
        res = minimize(
            fun=func, x0=z, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': 50, 'ftol': 1e-15, 'gtol': 1e-12, 'disp': False}
        )
        return _softmax(res.x)
    except Exception:
        return None


def _softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lbfgs_multistage(a0, time_limit, T_start=100.0, T_end=0.1, n_stages=10):
    """Multi-stage L-BFGS with temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30))
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = _softmax(z_in)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            return (2.0 * n / stage_T) * (max_Tb + np.log(np.exp(Tb - max_Tb).sum()))
        
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = _softmax(z_in)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                m_start = max(0, k - n + 1)
                m_end = min(n - 1, k)
                if m_end >= m_start:
                    grad_a[m_start:m_end+1] += 2.0 * w[k] * a[m_end:m_start-1:-1]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        maxiter = max(20, int(remaining / 0.05))  # rough estimate
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14,
                         'maxfun': maxiter * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return _softmax(z)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=100000):
    """Nesterov-accelerated LP best-response refinement."""
    if solver is None:
        solver = _get_lp_solver()
    
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    # Precompute the convolution matrix structure (as a function, not explicit matrix for large n)
    t0 = time.monotonic()
    stall_count = 0
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP: min M s.t. (y * b)_k <= M for all k, b >= 0, sum(b) = 1
        b_new, _ = _solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        if it % 1000 == 0:
            if stall_count >= 10:
                rng = np.random.default_rng(it)
                noise = rng.random(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
            stall_count += 1
            
            if it % 5000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP it={it} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def _get_lp_solver():

    _get_lp_solver.solver = cp.ECOS
    return cp.ECOS


def _solve_lp_best_response(a_fixed, solver):
    """Solve min M s.t. T @ b <= M, b >= 0, sum(b) = 1."""
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build T matrix: T[k, j] = a_fixed[k-j] if 0 <= k-j < n, else 0
    # T has shape (2n-1, n)
    # For efficiency with large n, we could use a sparse representation,
    # but for n ~ 1000, a dense matrix is 2000*1000 = 2M entries, which is fine.
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def structured_init(n, rng):
    """Generate structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 1.5
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    elif choice == 6:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2 * x - 1)
    else:
        # Flat with sinusoidal perturbation
        a = 1.0 + 0.3 * np.cos(2 * np.pi * np.linspace(0, 1, n))
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def local_swap_search(a0, time_limit, n_steps=50000):
    """Local search by moving mass from high-contribution to low-contribution positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        
        # Find positions that contribute most to the peak
        m_start = max(0, peak_k - n + 1)
        m_end = min(n - 1, peak_k)
        contrib = np.zeros(n)
        contrib[m_start:m_end+1] = a[m_start:m_end+1] * a[peak_k - m_start:peak_k - m_end - 1:-1] if m_end > m_start else a[m_start] * a[peak_k - m_start]
        
        # Actually, contribution of a[m] to b[peak_k] is 2*a[m]*a[peak_k - m]
        # Let me just compute it directly
        contrib = np.zeros(n)
        for m in range(m_start, m_end + 1):
            contrib[m] = 2.0 * a[m] * a[peak_k - m]
        
        # Move mass from high-contribution to low-contribution positions
        top_src = np.argsort(contrib)[::-1][:3]
        low_dst = np.argsort(contrib)[:3]
        
        improved = False
        for si in top_src:
            for di in low_dst:
                if si == di or contrib[si] < 1e-14:
                    continue
                for frac in [0.01, 0.03, 0.05]:
                    amount = a[si] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
            if improved:
                break
        
        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-14:
                amount = a[i] * 0.02
                a_trial = a.copy()
                a_trial[i] -= amount
                a_trial[j] += amount
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
        
        if step > 10000 and not improved:
            # Check if we've been stuck
            if step % 5000 == 0:
                break
    
    return best_a, best_score
