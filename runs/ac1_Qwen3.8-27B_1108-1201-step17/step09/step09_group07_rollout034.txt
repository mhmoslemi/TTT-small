```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov accelerated projected gradient with log-sum-exp smoothing (warm-up)
2. LP-based best response with Nesterov dual averaging (main optimization)
3. Subgradient polishing on the true max objective
4. Multi-length exploration and symmetry-constrained optimization
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv(a,a))."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_projected_gradient(a0, time_limit, T_start=2.0, T_end=300.0,
                                 lr_init=0.05, n_stages=10, steps_per_stage=3000):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            lr = lr_init / (1.0 + 0.01 * step_count)
            
            # Nesterov momentum
            t_curr = (1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count)) / 2.0
            t_prev_calc = (1.0 + np.sqrt(1.0 + 4.0 * max(0, step_count - 1) ** 2)) / 2.0 if step_count > 0 else 1.0
            beta = (t_prev_calc - 1.0) / t_curr if step_count > 0 else 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)
            
            # Gradient of log-sum-exp smoothed max
            b = fast_convolve(y)
            Tb = T * b
            m = Tb.max()
            e = np.exp(Tb - m)
            w = e / e.sum()
            
            # grad[i] = 2 * sum_j a[j] * w[j+i] = 2 * (a correlate w)[i]
            fft_size = 1 << (n + len(w) - 1).bit_length()
            A = rfft(y, fft_size)
            W = rfft(w, fft_size)
            corr = irfft(np.conj(W) * A, fft_size)[:n]
            grad = 2.0 * corr
            
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1
            
            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=100000):
    """Nesterov dual averaging LP refinement with adaptive step size."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        # Adaptive step size: smaller as we progress
        alpha = 2.0 / (it + 2)
        # Reduce step size in later iterations for finer convergence
        if it > n_max_iters * 0.3:
            alpha *= 0.5
        if it > n_max_iters * 0.6:
            alpha *= 0.5
            
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)
        
        # Nesterov-type update for a
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)

        score_a = score_normalized(a)
        score_y = score_normalized(y)
        
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 100 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 6:
                # Perturb to escape local minimum
                rng = np.random.default_rng(it * 7919)
                y = y + 0.01 * rng.standard_normal(n)
                y = project_simplex(y)
                stall_count = 0

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, n_steps=50000, n_peaks=10):
    """Subgradient polishing on the true max objective."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    lr = 0.01
    decay = 0.9999
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        # Find peaks
        threshold = max_b - 0.02 * max_b
        peak_idx = np.where(b >= threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b)])
        if len(peak_idx) > n_peaks:
            # Keep only the top n_peaks
            peak_vals = b[peak_idx]
            top = np.argsort(peak_vals)[::-1][:n_peaks]
            peak_idx = peak_idx[top]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in peak_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1)
            grad[i_range] += a[k - i_range]
        grad *= 2.0 / len(peak_idx)
        
        # Take step
        a_trial = a - lr * grad
        a_trial = project_simplex(a_trial)
        score = score_normalized(a_trial)
        
        if score < best_score:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
        else:
            # Reduce learning rate
            lr *= 0.95
        
        # Periodic annealing of lr
        if step % 1000 == 999:
            lr = max(0.001, lr * decay)

    return best_a, best_score


def symmetry_constrained_optimize(a0, time_limit, solver=None):
    """Optimize with symmetry constraint a[i] = a[n-1-i]."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    # Make symmetric
    a_sym = 0.5 * (a + a[::-1])
    a_sym = project_simplex(a_sym)
    
    best_a = a_sym.copy()
    best_score = score_normalized(a_sym)
    t0 = time.monotonic()

    half_n = (n + 1) // 2

    stall_count = 0
    prev_best = best_score
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Reconstruct full symmetric sequence from half
        if n % 2 == 0:

        else:
            a_full = np.concatenate([y_half, y_half[-2::-1]])
        a_full = project_simplex(a_full)
        
        # Solve LP with symmetric constraint
        b_var = cp.Variable(half_n, nonneg=True)
        M_var = cp.Variable()

            b_full = np.concatenate([b_var, b_var[-2::-1]])

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
```

```python
import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv(a,a))."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def nesterov_pg(a0, time_limit, T_start=5.0, T_end=300.0,
                lr_init=0.05, n_stages=8, steps_per_stage=2000):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            lr = lr_init / (1.0 + 0.01 * step_count)
            t_curr = (1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count)) / 2.0
            t_prev_calc = (1.0 + np.sqrt(1.0 + 4.0 * max(0, step_count - 1) ** 2)) / 2.0 if step_count > 0 else 1.0
            beta = (t_prev_calc - 1.0) / t_curr if step_count > 0 else 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)
            b = fast_convolve(y)
            Tb = T * b
            m = Tb.max()
            e = np.exp(Tb - m)
            w = e / e.sum()
            fft_size = 1 << (n + len(w) - 1).bit_length()
            A = rfft(y, fft_size)
            W = rfft(w, fft_size)
            corr = irfft(np.conj(W) * A, fft_size)[:n]
            grad = 2.0 * corr
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1
            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()

    return best_a, best_score

def subgradient_polish(a0, time_limit, n_steps=100000, n_peaks=5):
    """Subgradient method on the true max objective with adaptive step size."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    lr = 0.005
    best_lr = lr
    consecutive_fails = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b - 0.01 * max_b
        peak_idx = np.where(b >= threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b)])
        if len(peak_idx) > n_peaks:
            peak_vals = b[peak_idx]
            top = np.argsort(peak_vals)[::-1][:n_peaks]
            peak_idx = peak_idx[top]

        grad = np.zeros(n)
        for k in peak_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1)
            grad[i_range] += a[k - i_range]
        grad *= 2.0 / len(peak_idx)

        a_trial = project_simplex(a - lr * grad)
        score = score_normalized(a_trial)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            consecutive_fails = 0
            if score < best_score - 1e-12:
                best_lr = lr
        else:
            consecutive_fails += 1
            if consecutive_fails > 10:
                lr *= 0.8
                consecutive_fails = 0
            elif step % 50 == 0:
                lr *= 0.95

        if step % 2000 == 0 and step > 0:
            lr = max(0.0005, best_lr * 0.5)

    return best_a, best_score

    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b - 0.15 * max_b
        near_peaks = np.where(b >= threshold)[0]
        improved = False

        for k in near_peaks[:4]:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            contribs = a[idx] * a[k - idx]
            top_idx = np.argsort(contribs)[::-1][:4]
            for ci in top_idx:
                i = idx[ci]
                if a[i] < 1e-14:
                    continue
                for factor in [0.96, 0.92, 0.88, 0.82]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = project_simplex(a_trial)
                    score = score_normalized(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            k = int(np.argmax(b))
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            contribs = a[idx] * a[k - idx]
            top_contrib = np.argsort(contribs)[::-1][:5]
            for ci in top_contrib:
                i = idx[ci]
                j = k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.94, 0.90]:
                    for f2 in [0.94, 0.90]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = project_simplex(a_trial)
                        score = score_normalized(a_trial)
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


class State:
    """Mutable state container to avoid closures."""
    def __init__(self):
        self.best_seq = None
        self.best_score = float('inf')


def try_update(state, a, label):
    """Try to update best sequence if score improves."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < state.best_score:
        state.best_score = sc
        state.best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Nesterov PG warm-up
    2. Reduced active-set LP refinement (many fast iterations)
    3. Full LP refinement (fewer but more accurate iterations)
    4. Subgradient polishing
    5. Peak shaving
    6. Multi-length exploration
    7. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    state = State()
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            state.best_seq = parent.copy()
            state.best_score = p_score
    except NameError:
        pass
    except Exception as e:
        print(f"Error loading parent: {e}")

    if state.best_seq is None:
        n_init = 960
        x = np.linspace(0, 1, n_init)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 1e-10)
        state.best_seq = a / a.sum()
        state.best_score = evaluate_safe(state.best_seq)
        print(f"Default init: n={n_init}, score={state.best_score:.8f}")

    n_best = len(state.best_seq)

    # === Phase 1: Quick Nesterov PG warm-up ===
    print(f"\n=== Phase 1: NG Warm-up n={n_best} (best={state.best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 90.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ng, _ = nesterov_pg(state.best_seq, t_alloc, T_start=5.0, T_end=300.0,
                              lr_init=0.05, n_stages=8, steps_per_stage=3000)
        try_update(state, a_ng, "ng_warmup")

    # === Phase 2: Reduced active-set LP refinement (main workhorse) ===
    print(f"\n=== Phase 2: Reduced LP Refinement (best={state.best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, n_peaks=25")
        a_lp1, _ = lp_refine_reduced(state.best_seq, t_alloc, solver=solver,
                                     n_peaks=25, n_max_iters=500000)
        try_update(state, a_lp1, "lp_reduced_1")

    # === Phase 3: Second reduced LP with more peaks ===
    print(f"\n=== Phase 3: LP Reduced 2 peaks (best={state.best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, n_peaks=50")
        a_lp2, _ = lp_refine_reduced(state.best_seq, t_alloc, solver=solver,
                                     n_peaks=50, n_max_iters=200000)
        try_update(state, a_lp2, "lp_reduced_2")

    # === Phase 4: Full LP refinement ===
    print(f"\n=== Phase 4: Full LP Refinement (best={state.best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_full, _ = lp_refine_full(state.best_seq, t_alloc, solver=solver,
                                   n_max_iters=50000)
        try_update(state, a_full, "lp_full")

    # === Phase 5: Subgradient polishing ===
    print(f"\n=== Phase 5: Subgradient Polish (best={state.best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 80.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sg, _ = subgradient_polish(state.best_seq, t_alloc, n_steps=200000, n_peaks=5)
        try_update(state, a_sg, "subgrad")

    # === Phase 6: Peak shaving ===
    print(f"\n=== Phase 6: Peak Shaving (best={state.best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.06, 60.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ps, _ = peak_shave(state.best_seq, t_alloc, n_steps=100000)
        try_update(state, a_ps, "peak_shave")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length (best={state.best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        alt_lengths = sorted(set([n_best + d for d in [-20, -10, 10, 20, 40, -40]
                                  if 512 <= n_best + d <= 2048]))
        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_total = min(rem * 0.12, 90.0)
            if t_total < 25:
                break
            print(f"  n={alt_n}, budget={t_total:.0f}s")
            init_alt = interpolate_sequence(state.best_seq, alt_n)
            a_alt, _ = lp_refine_reduced(init_alt, t_total * 0.7, solver=solver,
                                         n_peaks=25, n_max_iters=100000)
            try_update(state, a_alt, f"alt_n{alt_n}")
            rem = deadline - time.monotonic()
            if rem > 15:
                a_alt2, _ = peak_shave(a_alt, t_total * 0.3, n_steps=20000)
                try_update(state, a_alt2, f"alt_ps_n{alt_n}")

    # === Phase 8: Final LP polish ===
    print(f"\n=== Phase 8: Final Polish (best={state.best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        t_fin = min(rem * 0.5, 80.0)
        a_fin, _ = lp_refine_full(state.best_seq, t_fin, solver=solver, n_max_iters=50000)
        try_update(state, a_fin, "final_lp")

    # === Phase 9: Final peak shave ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t_ps = min(rem * 0.6, 30.0)
        a_ps2, _ = peak_shave(state.best_seq, t_ps, n_steps=30000)
        try_update(state, a_ps2, "final_ps")

    # Final verification
    if state.best_seq is not None:
        fs = evaluate_safe(state.best_seq)
        if np.isfinite(fs):
            state.best_score = fs

    print(f"\nFINAL SCORE: {state.best_score:.8f} (n={len(state.best_seq)})")
    return [float(x) for x in state.best_seq]
```
