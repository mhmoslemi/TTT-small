def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ipv = np.arange(1, n + 1)
    cond = u * ipv > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_smoothed_grad(a, T):
    """Compute gradient of (2n/T) * logsumexp(T * conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    # grad = (2n/T) * sum_k w_k * grad conv_k
    # grad conv_k[i] = a[k-i] + a[i-k] (where valid)
    grad = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        grad[idx] += w[k] * a[k - idx]
        i_start2 = max(0, k)
        i_end2 = min(n - 1, k + n - 1)
        if i_start2 <= i_end2:
            idx2 = np.arange(i_start2, i_end2 + 1)
            grad[idx2] += w[k] * a[idx2 - k]
    
    # Scale by 2n/T
    grad *= (2.0 * n) / T
    return grad, w

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ipv = np.arange(1, n + 1)
    cond = u * ipv > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def softmax_from_y(y):
    """Convert dual variable to probability vector."""
    y_shifted = y - y.max()
    exp_y = np.exp(y_shifted)
    s = exp_y.sum()
    if s < 1e-300:
        return np.ones(len(y)) / len(y)
    return exp_y / s


def smooth_obj_func(z, T, n):
    """Smoothed objective: (2n/T) * logsumexp(T * conv(softmax(z)))."""
    a = softmax_from_y(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    return (2.0 * n / T) * (max_Tb + np.log(exp_Tb.sum()))


def smooth_grad_func(z, T, n):
    """Gradient of smoothed objective w.r.t. z (log-domain)."""
    a = softmax_from_y(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            grad_a[idx] += w[k] * a[k - idx]
        i_start2 = max(0, k)
        i_end2 = min(n - 1, k + n - 1)
        if i_start2 <= i_end2:
            idx2 = np.arange(i_start2, i_end2 + 1)
            grad_a[idx2] += w[k] * a[idx2 - k]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)

def lse_fista(a0, time_limit, T_start=30.0, T_end=300.0, n_stages=12, max_iter_per_stage=100000):
    """FISTA on smoothed objective with geometric temperature annealing."""
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_score = true_score(a)
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for stage_idx, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        # Estimate Lipschitz constant
        max_a = a.max()
        L = T * 6.0 * max(0.05, max_a) + 1.0

        x = a.copy()
        y_fista = a.copy()
        t_param = 1.0
        iters_done = 0

        for it in range(max_iter_per_stage):
            if time.monotonic() - t0 > time_limit - 5:
                break

            grad = smooth_grad_func(y_fista, T, n)
            x_new = project_simplex(y_fista - (1.0 / L) * grad)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
            y_fista = x_new + ((t_param - 1.0) / t_new) * (x_new - x)
            x = x_new
            t_param = t_new
            iters_done += 1

            # Periodically check best

                sc = true_score(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()

        a = project_simplex(x)
        sc = true_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.05, maxiter=8000, n_stages=14):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = true_score(a0)

            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break
        try:
            res = minimize(
                fun=smooth_obj_func,
                x0=z,
                jac=smooth_grad_func,
                args=(T, n),

                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
            a_new = softmax_from_y(z)
            sc = true_score(a_new)
            if sc < best_score:
                best_score = sc

        except Exception:
            pass

def mirror_descent_subgrad(a0, time_limit, f_lower=1.2):
    """Nesterov mirror descent with Polyak step on the true (non-smooth) objective."""
    n = len(a0)
    y = np.log(np.maximum(a0, 1e-300)).copy()
    best_a = project_simplex(a0.copy())
    best_score = true_score(best_a)
    t0 = time.monotonic()

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit - 2:
            break

        a_curr = softmax_from_y(y)
        b = fast_convolve(a_curr)
        k_star = int(np.argmax(b))

        # Subgradient of max_conv
        g = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            g[idx] += a_curr[k_star - idx]
        i_start2 = max(0, k_star)
        i_end2 = min(n - 1, k_star + n - 1)
        if i_start2 <= i_end2:
            idx2 = np.arange(i_start2, i_end2 + 1)
            g[idx2] += a_curr[idx2 - k_star]
        g *= 2.0 * n

        # Polyak step
        f_curr = 2.0 * n * b.max()
        gap = f_curr - f_lower
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            eta = 1e-8
        else:
            eta = gap / (2.0 * g_norm_sq)
            eta = min(eta, 1.0)

        y = y - eta * g
        a_new = softmax_from_y(y)

        if t % 50 == 0:
            sc = true_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        if best_score < f_lower + 0.3:
            f_lower = max(1.0, best_score - 0.15)

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_shave_coord(a0, time_limit):
    """Coordinate descent peak shaving: reduce elements contributing to peak, redistribute to others."""
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_score = true_score(a)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit - 2:
        b = fast_convolve(a)
        # Find interior peak (avoid boundary effects)
        b_work = b.copy()
        if len(b_work) > 4:
            b_work[0] = -np.inf
            b_work[1] = -np.inf
            b_work[-1] = -np.inf
            b_work[-2] = -np.inf
        k_star = int(np.argmax(b_work))

        # Find top contributors to this peak
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1)
        contribs = a[idx] * a[k_star - idx]
        total_contrib = contribs.sum()
        if total_contrib < 1e-30:
            break

        # Sort by contribution (highest first)
        top_order = np.argsort(contribs)[::-1]
        improved = False

        for ci in top_order[:20]:
            i = idx[ci]
            j = k_star - i
            if i == j:
                # Self-contribution: just reduce a[i] and redistribute
                for factor in [0.95, 0.90, 0.85, 0.80]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    diff = a[i] - a_trial[i]
                    if diff < 1e-15:
                        continue
                    # Redistribute to low-contribution positions
                    low_mask = np.ones(n, dtype=bool)
                    low_mask[i] = False
                    low_contrib = np.zeros(n)
                    low_contrib[idx] = contribs
                    reorder = np.argsort(low_contrib)[:n // 3]
                    low_mask[reorder] = False
                    if low_mask.sum() > 0:
                        a_trial[low_mask] += diff * a_trial[low_mask] / a_trial[low_mask].sum()
                    a_trial = project_simplex(a_trial)
                    sc = true_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
            else:
                # Pair (i, j): reduce one or both
                if j < n and a[j] > 1e-14:
                    for f1 in [0.97, 0.93, 0.88]:
                        for f2 in [0.97, 0.93, 0.88, 1.0]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            if f2 < 1.0:
                                a_trial[j] *= f2
                            a_trial = project_simplex(a_trial)
                            sc = true_score(a_trial)
                            if sc < best_score - 1e-14:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve > 100:
                break
        else:
            no_improve = 0

    return best_a, best_score


def multi_peak_shave(a0, time_limit):
    """Shave multiple peaks simultaneously using weighted subgradient."""
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_score = true_score(a)
    t0 = time.monotonic()

    for outer in range(200):
        if time.monotonic() - t0 > time_limit - 2:
            break

        b = fast_convolve(a)
        # Find top 3 peaks (excluding boundary)
        b_work = b.copy()
        if len(b_work) > 6:
            b_work[0:2] = -np.inf
            b_work[-2:] = -np.inf
        sorted_peaks = np.argsort(b_work)[::-1][:3]
        peak_vals = b[sorted_peaks]
        max_peak = peak_vals[0]
        if max_peak < 1e-15:
            break

        # Build combined subgradient from top peaks (weighted by proximity to max)
        g = np.zeros(n)
        for pk in sorted_peaks:
            weight = 0.5 ** (int(list(sorted_peaks).index(pk)))
            i_start = max(0, pk - n + 1)
            i_end = min(n - 1, pk)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                g[idx] += weight * a[pk - idx]
            i_start2 = max(0, pk)
            i_end2 = min(n - 1, pk + n - 1)
            if i_start2 <= i_end2:
                idx2 = np.arange(i_start2, i_end2 + 1)
                g[idx2] += weight * a[idx2 - pk]

        # Try reducing along this gradient direction
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-30:
            break

        for step_frac in [0.02, 0.01, 0.005, 0.002]:
            a_trial = a - step_frac * g
            a_trial = project_simplex(a_trial)
            sc = true_score(a_trial)
            if sc < best_score - 1e-14:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                break
        else:
            # No improvement, try perturbation
            if outer > 50:
                break

    return best_a, best_score

    print("\n=== Phase 7: Multi-Peak Shave (best={:.8f}) ===".format(best_score))
    rem = deadline - time.monotonic()
    t_mps = min(rem * 0.08, 100.0)
    if t_mps > 10:
        print("  Budget: {:.0f}s".format(t_mps))
        a_mps, _ = multi_peak_shave(best_seq, t_mps)
        try_update(a_mps, "multi_peak")

    # === Phase 8: Coordinate peak shaving ===
    print("\n=== Phase 8: Coord Peak Shave (best={:.8f}) ===".format(best_score))
    rem = deadline - time.monotonic()
    t_ps = min(rem * 0.08, 100.0)
    if t_ps > 10:
        print("  Budget: {:.0f}s".format(t_ps))
        a_ps, _ = peak_shave_coord(best_seq, t_ps)
        try_update(a_ps, "coord_shave")

    # === Phase 9: Final combined polish ===
    print("\n=== Phase 9: Final Polish (best={:.8f}) ===".format(best_score))
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin_lb = min(rem * 0.3, 60.0)
        if t_fin_lb > 10:
            a_fin, _ = lbgfs_refine(best_seq, t_fin_lb, T_start=10.0, T_end=0.02,
                                     maxiter=5000, n_stages=8)
            try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin_fps = min(rem * 0.3, 40.0)
        if t_fin_fps > 10:
            a_fps, _ = multi_peak_shave(best_seq, t_fin_fps)
            try_update(a_fps, "final_mps")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin_ps = min(rem * 0.3, 30.0)
        if t_fin_ps > 5:
            a_fps2, _ = peak_shave_coord(best_seq, t_fin_ps)
            try_update(a_fps2, "final_coord")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin_f = min(rem * 0.4, 30.0)
        if t_fin_f > 5:
            a_fin_f, _ = lse_fista(best_seq, t_fin_f, T_start=100.0, T_end=500.0,
                                    n_stages=6, max_iter_per_stage=50000)
            try_update(a_fin_f, "final_fista")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print("\nFINAL: score={:.8f}, n={}".format(best_score, len(best_seq)))
    return [float(x) for x in best_seq]
