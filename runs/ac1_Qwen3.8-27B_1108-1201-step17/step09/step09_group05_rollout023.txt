"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-initialization with diverse structured sequences
2. Aggressive L-BFGS with log-sum-exp smoothing and multi-stage geometric cooling
3. Targeted coordinate descent (peak shaving) for final local optimization
4. Multi-length exploration with warm-start interpolation
Key: Focus on L-BFGS effectiveness with very low final temperatures,
and use coordinate descent to break through the smooth approximation barrier.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_score(a):
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * len(a) * fast_convolve(a).max() / (s * s)


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_objective(z, T):
    a = softmax_from_z(z)
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp


def lse_gradient(z, T):
    a = softmax_from_z(z)
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_anneal(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=15):
    """L-BFGS with log-sum-exp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(norm_sequence(a0)).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(50, 20000 // n_stages)

    for stage_idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=lse_objective,
                x0=z,
                jac=lse_gradient,
                args=(stage_T,),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
            a_cur = softmax_from_z(z)
            sc = compute_score(a_cur)
            if stage_idx % 3 == 0:
                print(f"    L-BFGS stage {stage_idx}/{n_stages} (T={stage_T:.2f}): score={sc:.6f}")
        except Exception as e:
            print(f"    L-BFGS stage {stage_idx} error: {e}")
            break

    return softmax_from_z(z)


def coordinate_descent(a0, time_limit, n_steps=50000):
    """Targeted coordinate descent: reduce peak-contributing coords, add to low-contrib."""
    n = len(a0)
    a = norm_sequence(a0).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Focus on interior peaks (exclude edges)
        b_work = b.copy()
        if len(b_work) > 4:
            b_work[0] = -np.inf
            b_work[1] = -np.inf
            b_work[-2] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b[peak_k]

        # Find top contributors to peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Total contribution
        total_contrib = contribs.sum()
        
        # Top 10 contributors
        n_top = min(10, len(idx))
        top_local = np.argsort(contribs)[::-1][:n_top]
        top_coords = idx[top_local]
        
        # Bottom 10/4 contributors (least influential)
        n_bottom = min(max(n // 10, 5), len(idx))
        bottom_local = np.argsort(contribs)[:n_bottom]
        bottom_coords = idx[bottom_local]
        
        improved = False
        # Try reducing each top contributor
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.90, 0.85, 0.80, 0.70]:
                red_amt = a[i] * (1.0 - factor)
                if red_amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] *= factor
                # Redistribute to bottom coords
                if len(bottom_coords) > 0:
                    dist_weights = a[bottom_coords] + 1e-12
                    a_trial[bottom_coords] += red_amt * dist_weights / dist_weights.sum()
                a_trial = norm_sequence(a_trial)
                sc = compute_score(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair adjustments
            for ci1 in top_local[:5]:
                i = idx[ci1]
                j = peak_k - i
                if i >= n or j >= n or i == j:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.93, 0.87, 0.80]:
                    for f2 in [0.93, 0.87, 0.80]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = norm_sequence(a_trial)
                        sc = compute_score(a_trial)
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve > 100:
                break
        else:
            no_improve = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, mode, rng):
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif mode == 1:
        a = np.exp(-3 * (x - 0.5) ** 2)
    elif mode == 2:
        a = (1 + 0.5 * np.cos(4 * np.pi * x)) ** 0.5
    elif mode == 3:
        a = np.abs(np.sin(np.pi * x)) ** 0.7 + 0.05
    elif mode == 4:
        a = 0.3 + 0.7 * np.cos(2 * np.pi * (x + 0.25)) ** 2
    elif mode == 5:
        a = np.exp(-2 * x * (1 - x)) * (1 + 0.3 * np.cos(6 * np.pi * x))
    else:
        a = 1 + 0.2 * np.sin(2 * np.pi * rng.random() * 5 * x + rng.random() * 2 * np.pi)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):

        s = sum(a_list)
        if s < 0.01:

        return float('inf')

Let me also think about whether there's a smarter approach than L-BFGS. The problem is essentially:

minimize max_k (a*a)[k] subject to a >= 0, sum(a) = 1

This is a convex problem! (The max of linear functions is convex, and the constraint set is convex.) Wait, is it? (a*a)[k] = sum_i a_i * a_{k-i} is NOT linear in a (it's quadratic). So the problem is NOT convex.

However, if we fix one variable (say, we're optimizing over b given a), then (a*b)[k] IS linear in b. This is the basis of the LP approach in the parent code.

For the joint optimization, it's non-convex. L-BFGS on the smoothed version is a reasonable approach.

Another idea: use the fact that the problem has a "game" structure. If we alternate between:
1. Fix a, find b that minimizes max_k (a*b)[k] (LP)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-initialization with diverse structured sequences + parent warm-start
2. Aggressive L-BFGS with log-sum-exp smoothing and multi-stage geometric cooling
3. Targeted coordinate descent (peak shaving) for local optimization
4. Multi-length exploration with warm-start interpolation
5. Iterative deepening at very low temperatures for final polish
Key improvements over parent: more L-BFGS stages at lower temperatures,
faster coordinate descent with smarter redistribution, and better time allocation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_score(a):
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * len(a) * fast_convolve(a).max() / (s * s)


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_objective(z, T):
    a = softmax_from_z(z)
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp


def lse_gradient(z, T):
    a = softmax_from_z(z)
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_anneal(a0, time_limit, T_start, T_end, n_stages):
    n = len(a0)
    z = z_from_a(norm_sequence(a0)).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(50, 25000 // n_stages)

    for stage_idx, stage_T in enumerate(temps):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=lse_objective,
                x0=z,
                jac=lse_gradient,
                args=(stage_T,),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_cur = softmax_from_z(z)
            sc = compute_score(a_cur)
            if stage_idx % 4 == 0 or stage_idx == n_stages - 1:
                print(f"    L-BFGS stage {stage_idx}/{n_stages} (T={stage_T:.3f}): score={sc:.6f}")
        except Exception as e:
            print(f"    L-BFGS stage {stage_idx} error: {e}")
            break

    return softmax_from_z(z)


def coordinate_descent(a0, time_limit, n_steps):
    n = len(a0)
    a = norm_sequence(a0).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 4:
            b_work[0] = -np.inf
            b_work[1] = -np.inf
            b_work[-2] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        n_top = min(12, len(idx))
        top_local = np.argsort(contribs)[::-1][:n_top]
        top_coords = idx[top_local]

        n_bottom = min(max(n // 8, 5), len(idx))
        bottom_local = np.argsort(contribs)[:n_bottom]
        bottom_coords = idx[bottom_local]

        improved = False
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.90, 0.85, 0.78, 0.70]:
                red_amt = a[i] * (1.0 - factor)
                if red_amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] *= factor
                if len(bottom_coords) > 0:
                    dist_weights = a[bottom_coords] + 1e-12
                    a_trial[bottom_coords] += red_amt * dist_weights / dist_weights.sum()
                a_trial = norm_sequence(a_trial)
                sc = compute_score(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            for ci1 in top_local[:6]:
                i = idx[ci1]
                j = peak_k - i
                if i >= n or j >= n or i == j:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.93, 0.87, 0.80, 0.70]:
                    for f2 in [0.93, 0.87, 0.80, 0.70]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = norm_sequence(a_trial)
                        sc = compute_score(a_trial)
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve > 150:
                break
        else:
            no_improve = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, mode, rng):
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.cos(np.pi * x / 2) ** 2 + 0.01
    elif mode == 1:
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif mode == 2:
        a = (1.0 + 0.5 * np.cos(4.0 * np.pi * x)) ** 0.5
    elif mode == 3:
        a = np.abs(np.sin(np.pi * x)) ** 0.7 + 0.05
    elif mode == 4:
        a = 0.3 + 0.7 * np.cos(2.0 * np.pi * (x + 0.25)) ** 2
    elif mode == 5:
        a = np.exp(-2.0 * x * (1.0 - x)) * (1.0 + 0.3 * np.cos(6.0 * np.pi * x))
    elif mode == 6:
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * x) * np.cos(2.0 * np.pi * 5.0 * x)
    else:
        freqs = rng.random(3) * 8 + 1
        phases = rng.random(3) * 2 * np.pi
        a = 1.0
        for f_val, p_val in zip(freqs, phases):
            a += 0.15 * np.cos(2.0 * np.pi * f_val * x + p_val)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        s = sum(a_list)
        if s < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
