def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solved = False
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True
    except Exception:
        pass
    if not solved:
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def compute_lse_obj(z, n, T):
    """Compute smoothed objective in log-domain."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    log_sum_exp = max_b + T * np.log(exp_b.sum())
    return (2.0 * n) * log_sum_exp


def compute_lse_grad(z, n, T):
    """Compute gradient of smoothed objective in log-domain."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()
    grad_a = np.zeros(n)
    # Only compute for significant weights
    threshold = 1e-12
    for k in range(len(b)):
        if w[k] < threshold:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS with multi-stage temperature LogSumExp."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]

    n_stages = len(T_stages)
    iters_per_stage = max(30, maxiter_per_stage // n_stages)
    best_a = a0.copy()
    best_sc = 2.0 * n * fast_convolve(a0).max()

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_inner):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            return compute_lse_obj(z_inner, n, stage_T)

        def jac(z_inner):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_inner)
            return compute_lse_grad(z_inner, n, stage_T)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        # Check improvement
        a_trial = softmax_from_z(z)
        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_sc:
            best_sc = sc
            best_a = a_trial.copy()

    return best_a

        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()

    t0 = time.monotonic()
    prev_best = best_score
    stall_rounds = 0
    rng = np.random.default_rng(99999)

    for it in range(n_max_iters):

            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        if sy < 1e-15:
            y = a.copy()
            sy = 1.0

def peak_gradient_refine(a0, time_limit, n_steps=100000):
    """Direct peak-gradient descent with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.05
    no_improve = 0
    best_lr = lr

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find all positions within 0.1% of max
        threshold = max_b * 0.9999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute averaged subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Backtracking line search
        best_trial_score = best_score
        best_trial_a = a
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            a_trial = a - trial_lr * g
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_trial_score - 1e-14:
                best_trial_score = sc
                best_trial_a = a_trial
                best_lr = trial_lr
                break

        if best_trial_a is not a:
            a = best_trial_a
            if best_score > best_trial_score:
                best_score = best_trial_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Adapt learning rate
        if no_improve >= 5:
            lr *= 0.7
            no_improve = 0
        lr = max(1e-8, min(lr, 0.1))

        if step % 20000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def shave_peak(a0, time_limit, n_steps=50000):
    """Coordinate descent: reduce top contributors to peak, redistribute."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(777)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Find top contributors to peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(12, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

        improved = False
        # Try reducing individual top contributors
        for pos in top_positions:
            if a[pos] < 1e-14:
                continue
            for factor in [0.999, 0.998, 0.997, 0.995, 0.99, 0.98]:
                a_trial = a.copy()
                old_val = a_trial[pos]
                a_trial[pos] = old_val * factor
                diff = old_val - a_trial[pos]
                if diff < 1e-16:
                    continue
                # Redistribute to random subset of others
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                other_idx = np.where(mask)[0]
                # Use weighted redistribution (proportional to current values)
                weights = a_trial[other_idx]
                w_sum = weights.sum()
                if w_sum < 1e-15:
                    break
                a_trial[other_idx] = a_trial[other_idx] + diff * weights / w_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try simultaneous reduction
            a_trial = a.copy()
            for pos in top_positions[:5]:
                if a[pos] > 1e-14:
                    a_trial[pos] *= 0.9995
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 500:
                    # Random perturbation to escape
                    pert = rng.standard_normal(n) * 0.01
                    a = a + pert
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    stall = 0
        else:
            stall = 0

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    Shave {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 5 * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.25) ** 2) + 0.6 * np.exp(-3 * (x - 0.75) ** 2)
    elif choice == 6:
        # Asymmetric ramp with oscillation
        a = (1 - x) * (1 + 0.2 * np.sin(20 * np.pi * x))
        a = np.maximum(a, 0)
    else:
        # Spline-like
        a = np.maximum(0, 1 - 2 * np.abs(x - 0.5)) + 0.1 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    n_best = len(best_seq)

    # === Phase 1: LP alternating on parent ===
    print(f"\n=== Phase 1: LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=500000)
        try_update(a_lp, "lp_parent")

    # === Phase 2: Peak gradient refinement ===
    print(f"\n=== Phase 2: PeakGrad (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        a_pg, _ = peak_gradient_refine(best_seq, t_alloc, n_steps=100000)
        try_update(a_pg, "peak_grad_1")

    # === Phase 3: Ultra-low-T LogSumExp L-BFGS ===
    print(f"\n=== Phase 3: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        print(f"  T_stages: {T_stages}, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=10000)
        try_update(a_lb, "lse_1")

    # === Phase 4: Shave peak ===
    print(f"\n=== Phase 4: Shave Peak (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_sh = shave_peak(best_seq, t_alloc, n_steps=50000)
        try_update(a_sh, "shave_1")

    # === Phase 5: Try n = n_best * 1.25 ===
    print(f"\n=== Phase 5: Longer n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 200:
        alt_n = int(n_best * 1.25)
        t_alt = min(rem * 0.20, 250.0)
        print(f"  Trying n={alt_n}, budget={t_alt:.0f}s")
        alt_seq = interpolate_sequence(best_seq, alt_n)
        # LP on new length
        t_lp = t_alt * 0.5
        a_alt_lp, _ = nesterov_lp_refine(alt_seq, t_lp, solver=solver, n_max_iters=100000)
        try_update(a_alt_lp, f"lp_n{alt_n}")
        # LSE polish
        rem = deadline - time.monotonic()
        if rem > 40 and best_seq is not None:
            a_alt_lb = lbgfs_refine(best_seq, min(rem * 0.2, 120.0),
                                    T_stages=[10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05],
                                    maxiter_per_stage=5000)
            try_update(a_alt_lb, f"lse_n{alt_n}")

    # === Phase 6: Try n = n_best * 1.5 ===
    print(f"\n=== Phase 6: Even longer n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        alt_n2 = int(n_best * 1.5)
        t_alt2 = min(rem * 0.15, 180.0)
        print(f"  Trying n={alt_n2}, budget={t_alt2:.0f}s")
        alt_seq2 = interpolate_sequence(best_seq, alt_n2)
        a_alt2_lp, _ = nesterov_lp_refine(alt_seq2, t_alt2 * 0.6, solver=solver, n_max_iters=50000)
        try_update(a_alt2_lp, f"lp_n{alt_n2}")
        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None:
            a_alt2_lb = lbgfs_refine(best_seq, min(rem * 0.3, 80.0),
                                     T_stages=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05],
                                     maxiter_per_stage=3000)
            try_update(a_alt2_lb, f"lse_n{alt_n2}")

    # === Phase 7: Perturbation + re-optimize on best ===
    print(f"\n=== Phase 7: Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_pert = min(rem * 0.25, 200.0)
        n_cur = len(best_seq)
        x = np.linspace(0, 2 * np.pi, n_cur)
        n_pert = 3
        t_each = t_pert / n_pert
        for freq in [2, 4, 7]:
            if time.monotonic() > deadline:
                break
            perturbation = 0.03 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
            perturbed = best_seq + perturbation
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps < 1e-15:
                continue
            perturbed = perturbed / ps
            a_pert, _ = nesterov_lp_refine(perturbed, t_each * 0.7, solver=solver, n_max_iters=50000)
            try_update(a_pert, f"pert_f{freq}_lp")
            # Quick LSE polish
            rem_inner = deadline - time.monotonic()
            if rem_inner > 20:
                a_pert2 = lbgfs_refine(best_seq, min(rem_inner * 0.3, 40.0),
                                       T_stages=[2.0, 1.0, 0.5, 0.2, 0.1],
                                       maxiter_per_stage=2000)
                try_update(a_pert2, f"pert_f{freq}_lse")

    # === Phase 8: Random perturbation escape ===
    print(f"\n=== Phase 8: Random Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_esc = min(rem * 0.15, 100.0)
        n_cur = len(best_seq)
        for trial in range(3):
            if time.monotonic() > deadline:
                break
            noise_scale = 0.02
            noise = rng.standard_normal(n_cur) * noise_scale
            perturbed = best_seq + noise
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps < 1e-15:
                continue
            perturbed = perturbed / ps
            a_esc, _ = nesterov_lp_refine(perturbed, t_esc / 3.0, solver=solver, n_max_iters=30000)
            try_update(a_esc, f"esc_{trial}")

    # === Phase 9: Final ultra-precise polish ===
    print(f"\n=== Phase 9: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.35, 100.0)
        if t_fin > 10:
            a_fin = lbgfs_refine(best_seq, t_fin,
                                 T_stages=[1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005],
                                 maxiter_per_stage=8000)
            try_update(a_fin, "final_lse")

    # === Phase 10: Final shave + peak gradient ===
    rem = deadline - time.monotonic()
    if rem > 30:
        t_ps = min(rem * 0.3, 50.0)
        if t_ps > 10:
            a_ps = shave_peak(best_seq, t_ps, n_steps=20000)
            try_update(a_ps, "final_shave")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_pg = min(rem * 0.4, 30.0)
        if t_pg > 8:
            a_pg2, _ = peak_gradient_refine(best_seq, t_pg, n_steps=30000)
            try_update(a_pg2, "final_peakgrad")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
