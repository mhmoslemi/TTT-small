import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_c / (sum_a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def compute_max_c(a):
    """Compute max of convolve(a,a) assuming sum(a)=1."""
    c = fast_convolve(a)
    return c.max()


def normalize_a(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def subgradient_at_peak(a, k):
    """Compute gradient of c[k] w.r.t. a, where c = convolve(a,a)."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    for i in range(i_start, i_end + 1):
        j = k - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def subgradient_top_peaks(a, n_peaks=5):
    """Compute subgradient (average of top peak gradients)."""
    c = fast_convolve(a)
    n = len(a)
    # Find top peaks, but weight by value
    peak_indices = np.argsort(c)[::-1][:n_peaks]
    peak_values = c[peak_indices]
    
    # Weighted average of gradients
    total_weight = peak_values.sum()
    if total_weight < 1e-30:
        return np.zeros(n), c.max()
    
    grad = np.zeros(n)
    for idx, k in enumerate(peak_indices):
        weights = peak_values[idx] / total_weight
        grad += weights * subgradient_at_peak(a, k)
    
    return grad, c.max()


def projected_subgradient_descent(a0, time_limit=60.0, n_peaks=5, 
                                    step_init=0.1, step_min=0.001,
                                    step_decay=0.95, patience=50):
    """Projected subgradient descent with adaptive step size."""
    n = len(a0)
    a = normalize_a(a0.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    step = step_init
    no_improve = 0
    prev_score = best_score
    
    while time.monotonic() - t0 < time_limit:
        grad, max_c = subgradient_top_peaks(a, n_peaks)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Take step
        a_new = a - step * grad
        a_new = normalize_a(a_new)
        
        # Evaluate
        score = compute_score(a_new)
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Try to increase step size slightly
            step = min(step * 1.02, step_init)
        else:
            no_improve += 1
            step *= step_decay
            if step < step_min:
                # Restart with perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                a = best_a + 0.02 * rng.standard_normal(n)
                a = normalize_a(a)
                step = step_init * 0.5
                no_improve = 0
        
        if no_improve > patience:
            break
    
    return best_a, best_score

def p_norm_smoothed_objective(z, p, n):
    """Objective: 2n * (sum c^p)^(1/p) where a = softmax(z) normalized."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = np.sum(c_pos ** p) ** (1.0 / p)
    return 2.0 * n * val


def p_norm_smoothed_gradient(z, p, n):
    """Gradient of p-norm smoothed objective w.r.t. z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    c_pow = c_pos ** p
    S = np.sum(c_pow)
    if S < 1e-300:
        return np.zeros_like(z)
    
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    
    # Gradient w.r.t. a: g_a[i] = 2 * factor * sum_k w[k] * a[k-i]
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    
    # Gradient w.r.t. z (softmax parameterization)
    return a * (g_a - np.dot(g_a, a))


def p_norm_lbfgs(a0, time_limit=60.0, p_start=50.0, p_end=2000.0, n_stages=6):
    """L-BFGS with p-norm smoothing, operating in log-domain."""
    n = len(a0)
    a = normalize_a(a0.copy())
    z = np.log(np.maximum(a, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    
    temps = np.geomspace(p_start, p_end, n_stages)
    
    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        iters = max(50, int(remaining * 8))
        
        res = minimize(
            fun=lambda z_val: p_norm_smoothed_objective(z_val, stage_p, n),
            x0=z,
            jac=lambda z_val: p_norm_smoothed_gradient(z_val, stage_p, n),
            method='L-BFGS-B',
            options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-16,
                     'maxfun': iters * 200, 'disp': False}
        )
        z = res.x
    
    z_max = z.max()
    exp_z = np.exp(z - z_max)

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_c / (sum_a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_a(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def subgradient_top_peaks(a, n_peaks):
    """Compute weighted subgradient from top peaks of autocorrelation."""
    c = fast_convolve(a)
    n = len(a)
    peak_indices = np.argsort(c)[::-1][:n_peaks]
    peak_values = c[peak_indices]
    total_weight = peak_values.sum()
    if total_weight < 1e-30:
        return np.zeros(n), c.max()
    grad = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = peak_indices[idx]
        w = peak_values[idx] / total_weight
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                grad[i] += w * 2.0 * a[j]
    return grad, c.max()


def subgradient_top_peaks_fast(a, n_peaks):
    """Faster subgradient computation using slicing."""
    c = fast_convolve(a)
    n = len(a)
    peak_indices = np.argsort(c)[::-1][:n_peaks]
    peak_values = c[peak_indices]
    total_weight = peak_values.sum()
    if total_weight < 1e-30:
        return np.zeros(n), c.max()
    grad = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = peak_indices[idx]
        w = peak_values[idx] / total_weight
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        grad[i_range] += w * 2.0 * a[j_range]
    return grad, c.max()

    """Projected subgradient descent with adaptive step and optional momentum."""
    n = len(a0)

    no_improve = 0

    beta = 0.9

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if use_momentum:
            v_mom = beta * v_mom + (1 - beta) * grad_normed

def p_norm_smoothed_objective(z, p, n):
    """Objective: 2n * (sum c^p)^(1/p) where a = softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = np.sum(c_pos ** p) ** (1.0 / p)
    return 2.0 * n * val


def p_norm_smoothed_gradient(z, p, n):
    """Gradient of p-norm smoothed objective w.r.t. z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    c_pow = c_pos ** p
    S = np.sum(c_pow)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))

    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)

            break

            z = res.x
        except Exception:
            pass

    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a

        n_stages = 5
        a_pn = p_norm_lbfgs(a, rem, p_start, p_end, n_stages)
        sc_pn = compute_score(a_pn)
        if sc_pn < best_score:
            best_score = sc_pn
            best_a = a_pn.copy()
            print(f"    [{label}] pnorm: {best_score:.8f}")
        a = best_a
    
    # Stage 4: Local coordinate descent (fine polish)
    rem = min(time_budget * 0.15, deadline - time.monotonic())
    if rem > 5 and not check_deadline():
        a_cd, sc_cd = local_coord_descent(a, rem, 8000)
        if sc_cd < best_score:
            best_score = sc_cd
            best_a = a_cd.copy()
            print(f"    [{label}] coord: {best_score:.8f}")
    
    # Stage 5: Second round of subgradient from best
    rem = min(time_budget * 0.1, deadline - time.monotonic())
    if rem > 5 and not check_deadline():
        a_sg2, sc_sg2 = projected_subgradient_descent(best_a, rem, 8, 0.04, True)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_a = a_sg2.copy()
            print(f"    [{label}] subgrad2: {best_score:.8f}")
    
    # Stage 6: Final p-norm with very high p
    rem = deadline - time.monotonic()
    if rem > 15:
        a_final = p_norm_lbfgs(best_a, min(rem, 30.0), 500.0, 5000.0, 3)
        sc_final = compute_score(a_final)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
            print(f"    [{label}] pnorm_final: {best_score:.8f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: subgradient + spectral shaping + p-norm L-BFGS
    + coordinate descent, with multi-length exploration and diverse starts.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(seed)
    
    best_state = {'score': float('inf'), 'seq': None}
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = parent_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1024
        parent = structured_init(n_init, 0, seed)
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={len(parent)}, score={best_state['score']:.8f}")
    
    def try_update(a, label):
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return False
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_state['score']:
            best_state['score'] = sc
            best_state['seq'] = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
            return True
        return False
    
    # === Phase 1: Optimize parent sequence ===
    print(f"\n=== Phase 1: Optimize parent n={len(best_state['seq'])} (score={best_state['score']:.8f}) ===")
    t_alloc = min((deadline - time.monotonic()) * 0.35, 600.0)
    if t_alloc > 30:
        a_opt, _ = optimize_sequence(best_state['seq'], t_alloc, "parent", deadline)
        try_update(a_opt, "parent_opt")
    
    # === Phase 2: Spectral shaping from parent ===
    print(f"\n=== Phase 2: Spectral shaping (best={best_state['score']:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.1, 120.0)
    if t_alloc > 10:
        n_cur = len(best_state['seq'])
        for gamma in [1.8, 2.5, 3.5]:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            a_init = best_state['seq'] + 0.02 * rng.standard_normal(n_cur)
            a_init = normalize_a(a_init)
            a_sh, _ = spectral_shaping(a_init, min(rem * 0.3, 40.0), gamma)
            if compute_score(a_sh) < compute_score(best_state['seq']):
                a_sh = p_norm_lbfgs(a_sh, min(rem * 0.2, 30.0), 100.0, 1000.0, 3)
            try_update(a_sh, f"spectral_g{gamma}")
    
    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Length exploration (best={best_state['score']:.8f}) ===")
    n_best = len(best_state['seq'])
    candidate_lengths = []
    for delta in [-128, -96, -64, -32, 32, 64, 96, 128, 192]:
        alt_n = n_best + delta
        if 384 <= alt_n <= 2560:
            candidate_lengths.append(alt_n)
    # Add round numbers
    for rn in [512, 640, 768, 896, 1024, 1152, 1280, 1408, 1536]:
        if 384 <= rn <= 2560 and abs(rn - n_best) > 32:
            candidate_lengths.append(rn)
    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_best))
    
    n_explored = 0
    for alt_n in candidate_lengths[:6]:
        rem = deadline - time.monotonic()
        if rem < 100 or n_explored >= 5:
            break
        n_explored += 1
        t_total = min(rem * 0.18, 200.0)
        if t_total < 40:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")
        
        init_alt = interpolate_sequence(best_state['seq'], alt_n)
        a_alt, _ = optimize_sequence(init_alt, t_total, f"n{alt_n}", deadline)
        try_update(a_alt, f"len_{alt_n}")
    
    # === Phase 4: Diverse initializations at best length ===
    print(f"\n=== Phase 4: Diverse inits n={n_best} (best={best_state['score']:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_per = min(rem * 0.12, 80.0)
        for variant in [2, 3, 5, 8, 9]:
            rem = deadline - time.monotonic()
            if rem < 30 or t_per < 15:
                break
            a_init = structured_init(n_best, variant, seed + variant * 100)
            # Quick optimize
            a_opt, _ = optimize_sequence(a_init, t_per, f"v{variant}", deadline)
            try_update(a_opt, f"init_v{variant}")
    
    # === Phase 5: Aggressive subgradient with many peaks ===
    print(f"\n=== Phase 5: Aggressive subgradient (best={best_state['score']:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.15, 150.0)
        for n_pk in [10, 15]:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            a_sg, _ = projected_subgradient_descent(best_state['seq'], min(rem * 0.5, t_alloc), n_pk, 0.05, True)
            if compute_score(a_sg) < best_state['score'] * 0.999:
                a_sg = p_norm_lbfgs(a_sg, min(rem * 0.3, 30.0), 200.0, 2000.0, 3)
            try_update(a_sg, f"subgrad_pk{n_pk}")
    
    # === Phase 6: Final polish rounds ===
    print(f"\n=== Phase 6: Final polish (best={best_state['score']:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Very high p L-BFGS
        t_alloc = min(rem * 0.3, 60.0)
        a_fin = p_norm_lbfgs(best_state['seq'], t_alloc, 500.0, 5000.0, 4)
        try_update(a_fin, "final_pnorm_high")
    
    rem = deadline - time.monotonic()
    if rem > 15:
        t_alloc = min(rem * 0.25, 40.0)
        a_cd, _ = local_coord_descent(best_state['seq'], t_alloc, 20000)
        try_update(a_cd, "final_coord")
    
    rem = deadline - time.monotonic()
    if rem > 10:
        a_sg_f, _ = projected_subgradient_descent(best_state['seq'], min(rem, 30.0), 5, 0.03, True)
        try_update(a_sg_f, "final_subgrad")
    
    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs
    
    print(f"\nFINAL SCORE: {best_state['score']:.8f} (n={len(best_state['seq'])}, "
          f"time_remaining={deadline - time.monotonic():.1f}s)")
    return [float(x) for x in best_state['seq']]
