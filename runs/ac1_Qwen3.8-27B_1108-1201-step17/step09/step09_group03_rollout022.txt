def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact-optimal coordinate descent (binary search on 1D linear subproblems)
2. Pairwise mass transfer optimization between coordinates
3. Multi-peak mirror descent with adaptive temperature
4. Frank-Wolfe with away steps for global moves
5. Random restarts with re-optimization
6. Fine polishing with adaptive step refinement
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def opt_delta_1d(c_aff, coeff, unaff_max, d_min, d_max):
    """
    Find delta in [d_min, d_max] minimizing max(c_aff + delta*coeff, unaff_max).
    Returns (delta, optimal_max_value).
    """
    if len(c_aff) == 0:
        return 0.0, unaff_max

    cur_max = float(np.max(c_aff))
    T_lo = unaff_max
    T_hi = max(cur_max, unaff_max)

    pos = coeff > 1e-18
    neg = coeff < -1e-18
    zero = ~(pos | neg)

    if zero.any():
        zm = float(np.max(c_aff[zero]))
        if zm > T_hi:
            T_hi = zm
            T_lo = max(T_lo, zm)

    for _ in range(45):
        T_mid = (T_lo + T_hi) / 2.0
        if T_mid < unaff_max:
            T_mid = unaff_max

        dl = d_min
        dh = d_max

        if pos.any():
            ub = (T_mid - c_aff[pos]) / coeff[pos]
            dh = min(dh, float(np.min(ub)))

        if neg.any():
            lb = (T_mid - c_aff[neg]) / coeff[neg]
            dl = max(dl, float(np.max(lb)))

        if dl <= dh:
            T_hi = T_mid
        else:
            T_lo = T_mid

    T_opt = T_hi
    dl = d_min
    dh = d_max
    if pos.any():
        ub = (T_opt - c_aff[pos]) / coeff[pos]
        dh = min(dh, float(np.min(ub)))
    if neg.any():
        lb = (T_opt - c_aff[neg]) / coeff[neg]
        dl = max(dl, float(np.max(lb)))

    delta = (dl + dh) / 2.0
    delta = max(delta, d_min)
    delta = min(delta, d_max)
    return delta, T_opt


def exact_cd_phase(a, c, n_sweeps, time_limit, label=""):
    """Phase of exact-optimal coordinate descent with multiple sweeps."""
    n = len(a)
    t0 = time.monotonic()
    M = float(c.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        
        for i in range(n):
            if a[i] < 1e-18:
                # Can only increase
                if M <= (2 * n - 1 - i) * 1e-18:
                    continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            if k_start > 0:
                unaff_l = float(np.max(c[:k_start]))
            else:
                unaff_l = -1e30

def pairwise_optimize(a, c, time_limit, n_trials_per_iter=30):

    n = len(a)

        sg[i_start:i_end + 1] = 2.0 * a[k_star - i_end:k_star - i_start + 1][::-1]
        
        bad_scores = a * sg
        top_bad = np.argsort(bad_scores)[::-1][:8]
        
        good_mask = sg < 0.05 * np.max(sg)
        good_coords = np.where(good_mask)[0]
        if len(good_coords) < 10:
            good_coords = np.argsort(sg)[:20]
        
        for _ in range(n_trials_per_iter):
            if time.monotonic() - t0 > time_limit:
                break
            
            i = int(top_bad[rng.integers(0, len(top_bad))])
            j = int(good_coords[rng.integers(0, len(good_coords))])
            if i == j:
                continue
            if a[j] < 1e-15 and a[i] < 1e-15:
                continue
            
            k_start = min(i, j)
            k_end = max(i + n - 1, j + n - 1)

                unaff_l = float(np.max(c[:k_start]))
            else:
                unaff_l = -1e30

            else:
                unaff_r = -1e30
            unaff_max = max(unaff_l, unaff_r)

            c_aff = c[k_start:k_end + 1]
            
            coeff = np.zeros(m)
            for ki in range(m):
                k = k_start + ki
                val = 0.0
                if i <= k < i + n:
                    val += 2.0 * a[k - i]
                if j <= k < j + n:
                    val -= 2.0 * a[k - j]
                coeff[ki] = val

                continue
            
            # Transfer t from j to i: a[i]+=t, a[j]-=t
            # We want t in [-a[i], a[j]]
            d_min = max(-a[i], -a[j])  # t >= -a[i] means a[i]+t >= 0; t <= a[j] means a[j]-t >= 0
            d_max = min(a[j], a[i])
            
            # Actually: constraint is a[i]+t >= 0 => t >= -a[i], and a[j]-t >= 0 => t <= a[j]
            d_min = -a[i]
            d_max = a[j]
            
            delta, T_new = opt_delta_1d(c_aff, coeff, unaff_max, d_min, d_max)
            
            if abs(delta) > 1e-14 and T_new < M - 1e-13:
                a[i] = max(a[i] + delta, 0.0)
                a[j] = max(a[j] - delta, 0.0)
                c[k_start:k_end + 1] += delta * coeff
                M = max(unaff_max, float(np.max(c[k_start:k_end + 1])))

def multi_peak_mirror_descent(a0, time_limit, lr_init=0.03, top_k=30, temp_scale=0.005):
    """Multi-peak mirror descent with adaptive parameters."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    temp = temp_scale
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = float(conv.max())
        score = 2.0 * n * M
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            lr *= 0.85
            temp *= 0.9
            no_improve = 0
            if lr < 1e-9:
                break
        
        k_act = min(top_k, 2 * n - 1)
        if k_act <= 1:
            peaks = np.array([int(np.argmax(conv))])
        else:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
        
        pv = conv[peaks]
        w = np.exp((pv - pv.max()) / max(temp * M, 1e-15))
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws
        
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]
        
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn

        if it % 200 == 0:

        n_init = 1024
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    n = len(best_seq)
    print(f"Starting: n={n}, score={best_score:.6f}")
    
    # === Phase 1: Exact Coordinate Descent ===
    remaining = deadline - time.monotonic() - 600
    t_cd = max(100, min(remaining * 0.25, 250))
    print(f"\n=== Phase 1: Exact CD (budget={t_cd:.0f}s) ===")
    
    a_work = best_seq.copy()
    c_work = fast_convolve(a_work)
    a_cd, s_cd = exact_cd_phase(a_work, c_work, n_sweeps=200, time_limit=t_cd, label="exact_cd")
    best_seq, best_score = check_candidate(a_cd, best_seq, best_score, "exact_cd")
    
    # === Phase 2: Pairwise Optimization ===
    remaining = deadline - time.monotonic() - 450
    t_pw = max(60, min(remaining * 0.2, 150))
    print(f"\n=== Phase 2: Pairwise (budget={t_pw:.0f}s) ===")
    
    a_work = best_seq.copy()
    c_work = fast_convolve(a_work)
    a_pw, s_pw = pairwise_optimize(a_work, c_work, time_limit=t_pw, n_trials_per_iter=40)
    best_seq, best_score = check_candidate(a_pw, best_seq, best_score, "pairwise")
    
    # === Phase 3: More Exact CD ===
    remaining = deadline - time.monotonic() - 350
    t_cd2 = max(60, min(remaining * 0.25, 200))
    print(f"\n=== Phase 3: Exact CD round 2 (budget={t_cd2:.0f}s) ===")
    
    a_work = best_seq.copy()
    c_work = fast_convolve(a_work)
    a_cd2, s_cd2 = exact_cd_phase(a_work, c_work, n_sweeps=200, time_limit=t_cd2, label="exact_cd2")
    best_seq, best_score = check_candidate(a_cd2, best_seq, best_score, "exact_cd2")
    
    # === Phase 4: Multi-peak Mirror Descent ===
    remaining = deadline - time.monotonic() - 250
    t_md = max(60, min(remaining * 0.3, 200))
    print(f"\n=== Phase 4: Mirror Descent (budget={t_md:.0f}s) ===")
    
    a_md, s_md = multi_peak_mirror_descent(best_seq, t_md, lr_init=0.02, top_k=40, temp_scale=0.003)
    best_seq, best_score = check_candidate(a_md, best_seq, best_score, "mirror_descent")
    
    # === Phase 5: FW with Away Steps ===
    remaining = deadline - time.monotonic() - 200
    t_fw = max(60, min(remaining * 0.35, 180))
    print(f"\n=== Phase 5: FW with Away (budget={t_fw:.0f}s) ===")
    
    a_fw, s_fw = fw_with_away(best_seq, t_fw, steps_per_round=80, n_rounds_max=100)
    best_seq, best_score = check_candidate(a_fw, best_seq, best_score, "fw_away")
    
    # === Phase 6: Random Restarts ===
    remaining = deadline - time.monotonic() - 150
    t_restart = max(40, min(remaining * 0.3, 120))
    print(f"\n=== Phase 6: Random Restarts (budget={t_restart:.0f}s) ===")
    
    n_restarts = max(3, min(10, int(t_restart / 12)))
    for r in range(n_restarts):
        if deadline - time.monotonic() < 30:
            break
        t_per = min(20, (deadline - time.monotonic() - 10) / max(1, n_restarts - r))
        
        noise_scale = 0.03 * (1.0 + 0.5 * r)
        noisy = best_seq + rng.standard_normal(n) * noise_scale
        noisy = np.maximum(noisy, 0)
        sn = noisy.sum()
        if sn < 1e-15:
            continue
        noisy = noisy / sn
        
        a_r, s_r = exact_cd_phase(noisy, fast_convolve(noisy), n_sweeps=50, time_limit=t_per, label=f"restart_{r}")
        best_seq, best_score = check_candidate(a_r, best_seq, best_score, f"restart_{r}")
    
    # === Phase 7: Final Fine Polishing ===
    remaining = deadline - time.monotonic() - 30
    t_final = max(20, min(remaining * 0.8, 150))
    print(f"\n=== Phase 7: Final Polish (budget={t_final:.0f}s) ===")
    
    a_work = best_seq.copy()
    c_work = fast_convolve(a_work)
    a_fin, s_fin = exact_cd_phase(a_work, c_work, n_sweeps=500, time_limit=t_final, label="final_cd")
    best_seq, best_score = check_candidate(a_fin, best_seq, best_score, "final_cd")
    
    # === Phase 8: Pairwise polish ===
    remaining = deadline - time.monotonic() - 15
    if remaining > 20:
        t_pw2 = min(remaining * 0.7, 60)
        print(f"\n=== Phase 8: Pairwise Polish (budget={t_pw2:.0f}s) ===")
        a_work = best_seq.copy()
        c_work = fast_convolve(a_work)
        a_pw2, s_pw2 = pairwise_optimize(a_work, c_work, time_limit=t_pw2, n_trials_per_iter=50)
        best_seq, best_score = check_candidate(a_pw2, best_seq, best_score, "pairwise_polish")
    
    # Final verification
    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
