import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Compute a * a (full linear convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_with(a_fixed, b):
    """Compute a_fixed * b using FFT."""
    n = len(a_fixed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a_fixed, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    """Convert logits to probabilities via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probabilities to logits via log."""
    return np.log(np.maximum(a, 1e-300))


def compute_subgradient_fft(a, active_indices):
    """
    Compute subgradient of max_k c[k] using FFT.
    g[m] = (1/|A|) * sum_{k in A} dc[k]/da[m]
    where dc[k]/da[m] = 2*a[k-m] (approximately, for interior points)
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Build weight vector w where w[k] = 1/|A| for k in active_indices
    w = np.zeros(size)
    w[active_indices] = 1.0 / len(active_indices)
    
    # g[m] = sum_k w[k] * 2 * a[k-m] (for valid k-m)
    # This is 2 * (w * a)[m] where * is cross-correlation
    # w * a at position m = sum_k w[k] * a[k-m] = sum_k w[k] * a[m-k] (reversed)
    # Actually: sum_k w[k] * a[k-m] = (w_full_corr a)[m]
    # = sum_k w[k-m] * a[k] (cross-correlation)
    
    # Let's use: g[m] = 2/|A| * sum_{k in A} a[k-m] (for 0 <= k-m < n)
    # = 2 * sum_k w[k] * a[k-m]
    # The sequence a_rev[m] = a[n-1-m], then sum_k w[k] * a[k-m] = conv(w, a_rev)[n-1-m + something]
    
    # Simpler: just use direct computation for small |A|, FFT for large |A|
    if len(active_indices) <= 50:
        g = np.zeros(n)
        for k in active_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                g[j] += 2.0 * a[k - j]
        g /= len(active_indices)
        return g
    else:
        # Use FFT
        # g[m] = 2/|A| * sum_{k in A} a[k-m] * I(0 <= k-m < n)
        # Define v[k] = a[k] for 0 <= k < n, else 0 (already is a)
        # Then sum_{k in A} a[k-m] = sum_k w[k] * a[k-m] (where w is indicator/|A|)
        # = sum_k w[k-m] * a[k] (by change of variables, with boundary issues)
        # 
        # Let's define: h[m] = sum_{k} w[k] * a[k-m] for 0 <= k-m < n
        # This is the cross-correlation of w and a
        # corr[w,a][m] = sum_k w[k] * conj(a[k+m]) -- not quite what we want
        #
        # Actually: sum_k w[k] * a[k-m] = (w * a_flipped)[m + n - 1]
        # where a_flipped[j] = a[n-1-j]
        #
        # Hmm, let me just do it directly with FFT:
        # g[m] = 2 * sum_k w[k] * a[k-m]
        # Let a_ext[k] = a[k] for 0 <= k < n, 0 otherwise (length 2n-1)
        # Then g[m] = 2 * sum_k w[k] * a_ext[k-m] = 2 * (w * a_ext)[m] (cross-correlation)
        # cross-correlation = conv(w, flip(a_ext))
        
        a_ext = np.zeros(size)
        a_ext[:n] = a
        w_full = np.zeros(size)
        w_full[active_indices] = 1.0 / len(active_indices)
        
        # cross-correlation: corr[m] = sum_k w_full[k] * a_ext[k-m]
        # = conv(w_full, flip(a_ext))[m + size - 1 - (size-1)] ... 
        # Let me just use the direct FFT approach:
        # corr[m] = sum_k w_full[k] * a_ext[k-m]
        # = IFFT(FFT(w_full) * conj(FFT(a_ext)))
        
        W = rfft(w_full, fft_size)
        A_ext = rfft(a_ext, fft_size)
        corr = irfft(W * np.conj(A_ext), fft_size)[:size]
        
        g = 2.0 * corr[:n]
        # Boundary correction: the above gives the "full" cross-correlation
        # but we only want terms where 0 <= k-m < n
        # For the full convolution, this is automatically handled
        # since a_ext is zero outside [0, n-1]
        
        return g


def solve_lp_best_response(a_fixed, M_warm=None, solver=None):
    """Solve the LP to find best response b given fixed a."""
    global _lp_solver
    if solver is None:
        solver = get_lp_solver()
    
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    if M_warm is not None:
        constraints.append(M_var <= M_warm * 1.5)  # Warm start bound
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            prob = cp.Problem(cp.Minimize(M_var), [T @ b_var <= M_var, cp.sum(b_var) == 1])
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def mirror_descent_fft(a0, time_limit, n_active_frac=0.08, use_fft_sg=True):
    """Mirror descent with FFT-accelerated subgradient and Polyak step."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    z = z_from_a(a).copy()
    
    b = fast_convolve(a)
    max_b = b.max()
    best_score = 2.0 * n * max_b
    best_a = a.copy()
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    
    t0 = time.monotonic()
    lr = 0.3
    no_improve = 0
    
    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient
        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-300:
            break
        
        threshold = max_b - n_active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        if use_fft_sg and len(active) > 20:
            g = compute_subgradient_fft(a, active)
        else:
            g = np.zeros(n)
            for k in active:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                g[idx] += a[k - idx]
            g /= len(active)
            g *= 2.0  # factor of 2 from the derivative
        
        g_norm = np.max(np.abs(g))
        if g_norm < 1e-15:
            break
        
        # Polyak step size
        gap = best_score - f_lb
        if gap < 1e-12:
            break
        polyak_lr = gap / (2.0 * g_norm * best_score + 1e-15)
        alpha = min(lr, polyak_lr)
        alpha = max(alpha, 0.005)
        
        # Update
        z_new = z - alpha * g
        a_new = softmax_from_z(z_new)
        
        b_new = fast_convolve(a_new)
        score_new = 2.0 * b_new.max()  # with sum(a)=1, it's 2n*max_b
        
        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.03, 1.5)
        else:
            no_improve += 1
            a = a_new  # still move (for exploration)
            z = z_new
            if no_improve >= 15:
                lr *= 0.75
                no_improve = 0
            lr = max(lr, 0.005)
        
        if step % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD-FFT {elapsed:.0f}s: score={best_score:.6f} lr={lr:.4f} active={len(active)}")
    
    return best_a, best_score


def nesterov_lp(a0, time_limit, solver=None, n_max_iters=50000):
    """LP alternating with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    M_warm = fast_convolve(a).max()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_best_response(y, M_warm=M_warm, solver=solver)
        if b_new is None:
            continue
        M_warm = M_b
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        # Nesterov update for a
        t_k = (1 + np.sqrt(1 + 4 * (it + 1) ** 2)) / 2
        beta = (t_k - 1) / (it + 2)
        a = (1 - beta) * y + beta * a
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        if (it + 1) % 500 == 0:
            prev_best = best_score
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            if stall_count >= 6:
                rng = np.random.default_rng(it * 31 + 7)
                noise = rng.random(n) * 0.03
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                stall_count = 0
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def lbgfs_smooth_refine(a0, time_limit, T_start=100.0, T_end=0.1, maxiter=10000):
    """L-BFGS on log-sum-exp smoothed objective with multi-stage annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    # Multi-stage temperature schedule
    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=_make_lse_func(n, stage_T),
                x0=z,
                jac=_make_lse_jac(n, stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            if res.success or res.fun < 1e15:
                z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Diverse initializations including asymmetric (monotone) sequences exploiting
   the rearrangement inequality (asymmetric sequences can achieve lower peaks)
2. FFT-accelerated mirror descent with Polyak step size
3. LP alternating method with Nesterov acceleration and warm starts
4. L-BFGS on log-sum-exp smoothed objective with multi-stage annealing
5. Local coordinate/swap search
6. Multi-length exploration

Key innovations over previous approach:
- Asymmetric initializations (linear/exponential ramps) that break the symmetry
  barrier, allowing the optimizer to find lower-peak solutions
- FFT-based subgradient computation for large active sets
- Nesterov momentum with proper t_k schedule (not simple 2/(t+2))
- Adaptive time allocation based on early convergence signals
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_pair(a_fixed, b):
    n = len(a_fixed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a_fixed, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_subgradient_direct(a, active):
    n = len(a)
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * a[k - idx]
    g /= len(active)
    return g


def compute_subgradient_fft(a, active):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    w_full = np.zeros(size)
    w_full[active] = 1.0 / len(active)
    a_ext = np.zeros(size)
    a_ext[:n] = a
    W = rfft(w_full, fft_size)
    A_ext = rfft(a_ext, fft_size)
    corr = irfft(W * np.conj(A_ext), fft_size)[:size]
    return 2.0 * corr[:n]


def solve_lp_best_response(a_fixed, M_warm=None, solver=None):
    global _lp_solver
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            prob = cp.Problem(cp.Minimize(M_var), [T @ b_var <= M_var, cp.sum(b_var) == 1])
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def mirror_descent(a0, time_limit, n_active_frac=0.08, use_fft=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    b = fast_convolve(a)
    max_b = b.max()
    best_score = 2.0 * n * max_b
    best_a = a.copy()
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    lr = 0.4
    no_improve = 0

    for step in range(300000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-300:
            break

        threshold = max_b - n_active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        if use_fft and len(active) > 30:
            g = compute_subgradient_fft(a, active)
        else:
            g = compute_subgradient_direct(a, active)

        g_norm = np.max(np.abs(g))
        if g_norm < 1e-15:
            break

        gap = best_score - f_lb
        if gap < 1e-12:
            break
        polyak_lr = gap / (2.0 * g_norm * best_score + 1e-15)
        alpha = min(lr, polyak_lr)
        alpha = max(alpha, 0.003)

        z_new = z - alpha * g
        a_new = softmax_from_z(z_new)
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.04, 2.0)
        else:
            no_improve += 1
            a = a_new
            z = z_new
            if no_improve >= 12:
                lr *= 0.7
                no_improve = 0
            lr = max(lr, 0.003)

        if step % 30000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {elapsed:.0f}s: score={best_score:.6f} lr={lr:.4f} |A|={len(active)}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver=solver)
        if b_new is None:
            continue

        # Nesterov momentum: t_k = (1 + sqrt(1 + 4k^2)) / 2
        k = it + 1
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * k * k)) / 2.0
        beta = (t_k - 1.0) / (it + 2.0)

        y = (1.0 - beta) * y + beta * a
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        a = b_new

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if (it + 1) % 300 == 0:
            stall_count += 1
            if stall_count >= 8:
                rng = np.random.default_rng(it * 17 + 3)
                noise = rng.random(n) * 0.04
                y = np.maximum(y + noise, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0

        if (it + 1) % 1500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def _lse_func(z, n, T_val):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T_val) * log_sum_exp


def _lse_jac(z, n, T_val):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    active = w > 1e-12
    for k in np.where(active)[0]:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.05, maxiter=12000):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=_lse_func,
                x0=z,
                jac=_lse_jac,
                args=(n, stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            if res.fun < 1e15:
                z = res.x
        except Exception:
            pass
    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        print(f"Default init: n={n_init}, score={best_score:.6f}")

    # Generate diverse initializations
    inits = []
    labels = []

    # Parent (already best)
    inits.append(best_seq)
    labels.append("parent")

    # Asymmetric inits at the best length
    for k in range(5):
        inits.append(init_asymmetric(n_best, k, rng))
        labels.append(f"asym_{k}_n{n_best}")

    # Symmetric inits at the best length
    for k in range(3):
        inits.append(init_symmetric(n_best, k, rng))
        labels.append(f"sym_{k}_n{n_best}")

    # Different lengths
    for mult in [0.75, 1.25, 1.5]:
        alt_n = int(n_best * mult)
        if 384 <= alt_n <= 2500:
            inits.append(interpolate_sequence(best_seq, alt_n))
            labels.append(f"interp_n{alt_n}")
            inits.append(init_asymmetric(alt_n, 1, rng))
            labels.append(f"asym1_n{alt_n}")

    # Shuffle to get variety
    order = rng.permutation(len(inits))
    inits = [inits[i] for i in order]
    labels = [labels[i] for i in order]

    print(f"\nTotal initializations to try: {len(inits)}")

    # Time allocation: give more time to promising inits
    # First pass: quick evaluation of all inits
    print("\n=== Phase 0: Quick evaluation ===")
    scored_inits = []
    for i, (init_a, lab) in enumerate(zip(inits, labels)):
        if time.monotonic() > deadline:
            break
        sc = evaluate_safe(init_a)
        scored_inits.append((sc, init_a, lab))
        if sc < best_score:
            best_score = sc
            best_seq = init_a.copy()
            print(f"  New best from init [{lab}]: {sc:.6f}")

    scored_inits.sort(key=lambda x: x[0])
    print(f"  Best init score: {scored_inits[0][0]:.6f} [{scored_inits[0][2]}]")

    # Second pass: full optimization on top candidates
    # Give more time to the top candidates
    n_cands = min(len(scored_inits), 6)
    total_time = deadline - time.monotonic()
    # Allocate time proportionally: top candidate gets most time
    weights = []
    for i in range(n_cands):
        if i == 0:
            weights.append(3.0)
        elif i == 1:
            weights.append(2.5)
        elif i == 2:
            weights.append(2.0)
        else:
            weights.append(1.0)
    total_w = sum(weights)

    print(f"\n=== Phase 1: Full optimization ({n_cands} candidates) ===")
    for i in range(n_cands):
        if time.monotonic() > deadline:
            break
        t_budget = (total_time * weights[i]) / total_w
        t_budget = max(30.0, t_budget)
        # Don't exceed remaining time
        t_budget = min(t_budget, deadline - time.monotonic() - 20.0)
        if t_budget < 20:
            break

        init_a, lab = scored_inits[i][1], scored_inits[i][2]
        print(f"\n  Optimizing [{lab}] (init score={scored_inits[i][0]:.6f}, budget={t_budget:.0f}s)")
        result_a, result_sc = optimize_single(init_a, t_budget, solver, label=lab)
        if result_a is not None and result_sc < best_score:
            best_score = result_sc
            best_seq = result_a.copy()
            print(f"  *** NEW BEST: {best_score:.6f} [{lab}] ***")
        elif result_a is not None:
            print(f"  [{lab}] final: {result_sc:.6f} (no improvement)")

    # Phase 2: Try a few more creative asymmetric inits if time remains
    rem = deadline - time.monotonic()
    if rem > 120 and best_seq is not None:
        print(f"\n=== Phase 2: Creative exploration ({rem:.0f}s remaining) ===")
        # Try very asymmetric: power-law with different exponents
        for exp_val in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0]:
            if time.monotonic() > deadline:
                break
            n_try = n_best
            x = np.linspace(0, 1, n_try)
            a_try = (x + 0.01) ** exp_val
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            sc_try = evaluate_safe(a_try)
            # Quick mirror descent
            t_quick = min(rem * 0.15, 100.0)
            if t_quick > 10:
                a_opt, _ = mirror_descent(a_try, t_quick, n_active_frac=0.06)
                sc_opt = evaluate_safe(a_opt)
                if sc_opt < best_score:
                    best_score = sc_opt
                    best_seq = a_opt.copy()
                    print(f"  Power({exp_val}): {sc_opt:.6f} *** NEW BEST ***")

    # Phase 3: Final polish on the best solution
    rem = deadline - time.monotonic()
    if rem > 30 and best_seq is not None:
        print(f"\n=== Phase 3: Final polish ({rem:.0f}s) ===")
        # L-BFGS with low temperature
        t_fin = min(rem * 0.4, 100.0)
        if t_fin > 10:
            a_fin = lbgfs_refine(best_seq, t_fin, T_start=20.0, T_end=0.05, maxiter=6000)
            sc_fin = evaluate_safe(a_fin)
            if sc_fin < best_score:
                best_score = sc_fin
                best_seq = a_fin.copy()
                print(f"  Final L-BFGS: {sc_fin:.6f} *** NEW BEST ***")

        rem = deadline - time.monotonic()
        # Local search
        t_ls = min(rem * 0.3, 60.0)
        if t_ls > 8:
            a_ls, _ = local_search(best_seq, t_ls, n_steps=30000)
            sc_ls = evaluate_safe(a_ls)
            if sc_ls < best_score:
                best_score = sc_ls
                best_seq = a_ls.copy()
                print(f"  Final LS: {sc_ls:.6f} *** NEW BEST ***")

        rem = deadline - time.monotonic()
        # One more LP round
        t_lp = min(rem * 0.3, 40.0)
        if t_lp > 10:
            a_lp, _ = nesterov_lp_refine(best_seq, t_lp, solver=solver, n_max_iters=5000)
            sc_lp = evaluate_safe(a_lp)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: {sc_lp:.6f} *** NEW BEST ***")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
