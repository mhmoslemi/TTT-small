Now let me think about the simplex projection. Given a vector v, project onto {a >= 0, sum(a) = 1}:
1. Find tau such that sum(max(v[i] - tau, 0)) = 1
2. This is a 1D convex problem that can be solved by bisection or by sorting

The standard algorithm (Duchi et al. 2008):
1. Sort v in descending order: v_(1) >= v_(2) >= ... >= v_(n)
2. Find rho = max{j: v_(j) - (v_(1) + ... + v_(j) - 1)/j > 0}
3. tau = (v_(1) + ... + v_(rho) - 1) / rho
4. a[i] = max(v[i] - tau, 0)

This is O(n log n) due to sorting.

Alternatively, for large n, we can use bisection on tau, which is O(n * log(precision)) without sorting.

For n ~ 1000, sorting is fast enough.

def fast_convolve(a):
    """Linear auto-convolution via FFT (zero-padded)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Return 2*n*max(conv(a,a)) for a with sum=1."""
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max()


def evaluate_safe(a):
    """Call the global evaluate_sequence with safety."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]

    except Exception:
        return float('inf')


def p_norm_obj_z(z, p):
    """2n * (sum max(c,0)^p)^(1/p) in log-softmax domain."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def p_norm_grad_z(z, p):
    """Gradient of p_norm_obj_z w.r.t. z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 2.0 * factor * corr
    return a * (4.0 * n * g_a - np.dot(4.0 * n * g_a, a))


def p_norm_lbfgs(a0, time_limit, p_start=15.0, p_end=500.0, n_stages=8):
    """L-BFGS on log-softmax with adaptive p-norm schedule."""
    n = len(a0)

    z = np.log(a0)
    z = z - z.max()
    t0 = time.monotonic()

        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

            return p_norm_obj_z(zv, stage_p)

        def jac(zv):
            return p_norm_grad_z(zv, stage_p)

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',

            z = res.x

            pass

    z_max = z.max()

def build_conv_matrix(a):
    """Build T where T @ b = convolve(a, b). Shape (2n-1, n)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_response(a_fixed, solver):
    """Given fixed a (sum=1), find b minimizing max conv(a,b)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    cond = (u - (css - 1.0) / rho_candidates) > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(rho_candidates[cond])
    theta = (css[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def subgradient_nesterov(a0, time_limit, max_iters=50000):
    """Subgradient method with Nesterov momentum on the nonsmooth objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

        c = fast_convolve(y)
        cmax = c.max()
        S = np.where(c >= cmax - 1e-12)[0]
        w = np.zeros_like(c)
        w[S] = 1.0 / len(S)
        corr = np.correlate(w, y, mode='valid')
        g = 2.0 * corr
        step = 1.0 / (np.sqrt(t) * 10.0)
        a_new = y - step * g
        a_new = simplex_project(a_new)
        momentum = (t - 1) / (t + 2.0)

        y = simplex_project(y)
        a_prev = a_new

            best_score = sc
            best_a = y.copy()

        if t % 10000 == 0:

def coord_descent(a0, time_limit, max_steps=50000):
    """Greedy coordinate descent with analytical step on true max objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()

        if cmax < 1e-15:
            break
        threshold = cmax * 0.95
        peak_set = np.where(c >= threshold)[0]
        contrib = np.zeros(n)
        for k in peak_set:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]
        order = np.argsort(contrib)[::-1]
        improved = False
        for i in order[:20]:
            if a[i] < 1e-14:
                continue
            # Find optimal delta by bisection
            lo, hi = 0.0, a[i] * 0.5

                mid = (lo + hi) / 2.0
                a_try = a.copy()
                a_try[i] -= mid
                if a_try[i] < 0:
                    a_try[i] = 0
                    mid = a[i]
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_try[mask].sum()
                if others < 1e-15:
                    break
                a_try[mask] += mid * a_try[mask] / others
                sc = compute_objective(a_try)
                if sc < best_score:
                    lo = mid
                    best_score = sc
                    best_a = a_try.copy()
                    improved = True
                else:
                    hi = mid
            if improved:
                a = best_a.copy()
                break
        if not improved:

def structured_init(n, variant=0):
    """Create structured initial sequence."""
    if variant == 0:

        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2.5 * x ** 2)
    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

    return a / s


def interpolate_seq(f, new_n):
    """Cubic-like interpolation to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi = fi / ss
    return fi

    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pn2 = p_norm_lbfgs(best_state['seq'], t_alloc, p_start=50.0, p_end=800.0, n_stages=7)
        try_update(a_pn2, "pn2")

    # === Phase 6: Nesterov LP from perturbed ===
    print(f"\n=== Phase 6: Nesterov LP perturbed (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.14, 200.0)
    if t_alloc > 30:
        n_cur = len(best_state['seq'])
        perturbed = best_state['seq'] + 0.05 * rng.standard_normal(n_cur)
        perturbed = np.maximum(perturbed, 0)
        perturbed = perturbed / perturbed.sum()
        print(f"  Budget: {t_alloc:.0f}s, perturbation=0.05")
        a_lp2, _ = nesterov_lp(perturbed, t_alloc, solver, max_iters=100000, step_mult=2.5)
        try_update(a_lp2, "lp2_pert")

    # Also try from a different perturbation on the LP result
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc2 = min(rem * 0.08, 100.0)
        if t_alloc2 > 30:
            n_cur = len(best_state['seq'])
            perturbed2 = best_state['seq'] + 0.1 * rng.standard_normal(n_cur)
            perturbed2 = np.maximum(perturbed2, 0)
            perturbed2 = perturbed2 / perturbed2.sum()
            a_lp3, _ = nesterov_lp(perturbed2, t_alloc2, solver, max_iters=60000, step_mult=2.0)
            try_update(a_lp3, "lp3_pert")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length (best={best_state['score']:.6f}) ===")
    candidate_lengths = []
    base = n_best
    for delta in [-128, -64, -32, 32, 64, 128, 192]:
        alt_n = base + delta
        if 384 <= alt_n <= 2560:
            candidate_lengths.append(alt_n)
    for rn in [512, 640, 768, 1024, 1152, 1280, 1536]:
        if 384 <= rn <= 2560 and abs(rn - base) > 64 and rn not in candidate_lengths:
            candidate_lengths.append(rn)
    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - base))

    n_explored = 0
    for alt_n in candidate_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 60 or n_explored >= 4:
            break
        n_explored += 1
        t_total = min(rem * 0.12, 120.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_seq(best_state['seq'], alt_n)

        # Quick p-norm
        t_pn = t_total * 0.5
        a_alt = p_norm_lbfgs(init_alt, t_pn, p_start=20.0, p_end=200.0, n_stages=5)
        ok = try_update(a_alt, f"pn_n{alt_n}")

        # If improved, do quick LP
        if ok or True:
            rem = deadline - time.monotonic()
            if rem > 30:
                t_lp = min(rem * 0.08, 60.0)
                if t_lp > 15:
                    a_alt2, _ = nesterov_lp(best_state['seq'], t_lp, solver, max_iters=30000, step_mult=2.0)
                    try_update(a_alt2, f"lp_n{alt_n}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t1 = min(rem * 0.3, 80.0)
        a_cd2, _ = coord_descent(best_state['seq'], t1, max_steps=20000)
        try_update(a_cd2, "cd_final")

    rem = deadline - time.monotonic()
    if rem > 30:
        t2 = min(rem * 0.3, 60.0)
        a_pn3 = p_norm_lbfgs(best_state['seq'], t2, p_start=200.0, p_end=1000.0, n_stages=4)
        try_update(a_pn3, "pn_final")

    rem = deadline - time.monotonic()
    if rem > 20:
        t3 = min(rem * 0.4, 40.0)
        a_cd3, _ = coord_descent(best_state['seq'], t3, max_steps=15000)
        try_update(a_cd3, "cd_final2")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    print(f"\nFINAL: score={best_state['score']:.6f}, n={len(best_state['seq'])}")
    return [float(x) for x in best_state['seq']]
