import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def norm_sequence(a):
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:

    return cp.ECOS


def sparse_lp_best_response(a_fixed, solver, n_top=50):
    """LP best response using only top-K convolution constraints for speed."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)

    # Get top n_top + some margin

    top_idx = np.argsort(importance)[::-1][:n_cons]

    try:

    except Exception:

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS]:
            if alt == solver:

            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):

                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)

    s = b_val.sum()
    if s < 1e-15:

    b_val = b_val / s
    return b_val, float(M_var.value)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full LP alternating with Nesterov acceleration (global structure)
2. Multi-restart Adam on log-sum-exp (local refinement)
3. Peak-directed greedy coordinate descent (fine tuning)
4. Multi-length exploration (1024, 768)
Key innovations: Adam optimizer with proper bias correction, aggressive multi-restart
at different temperatures, and peak-aware coordinate moves.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def norm_sequence(a):
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    for sn in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=sn)
            if p.status in ('optimal', 'optimal_inaccurate'):
                return sn
        except Exception:
            continue
    return cp.ECOS


def solve_lp(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def nesterov_lp(a0, time_limit, solver, n_max_iters=100000):
    """Nesterov LP alternating."""
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = solve_lp(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        y = y / max(y.sum(), 1e-300)
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        a = a / max(a.sum(), 1e-300)
        
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def adam_lsexp(a0, time_limit, T_schedule=None, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on log-sum-exp smoothing of the max."""
    n = len(a0)
    z = z_from_a(a0).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    best_z = z.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    
    if T_schedule is None:
        T_schedule = [50.0, 30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.2, 0.1]
    
    iters_per_T = max(50, int(time_limit / len(T_schedule) / 0.05))
    
    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        for _ in range(iters_per_T):
            if time.monotonic() - t0 > time_limit:
                break
            t += 1
            
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            
            # Gradient of (2n/T) * log(sum exp(T*b[k])) with respect to z
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            a_check = softmax_from_z(z)
            score = 2.0 * n * fast_convolve(a_check).max()
            if score < best_score:
                best_score = score
                best_z = z.copy()
    
    return softmax_from_z(best_z)


def peak_coordinate_descent(a0, time_limit, n_steps=100000):
    """Greedy coordinate descent focused on peak reduction."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 500:
            break
        
        b = fast_convolve(a)
        # Find top peaks (excluding boundary effects)
        b_inner = b.copy()
        b_inner[:3] = -np.inf
        b_inner[-3:] = -np.inf
        peak_k = int(np.argmax(b_inner))
        
        # Find top contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_contrib = np.argsort(contribs)[::-1][:10]
        
        improved = False
        for ci in top_contrib:
            i = idx[ci]
            j = peak_k - i
            if i == j:
                # Single coordinate: reduce a[i] and redistribute
                for frac in [0.95, 0.90, 0.85, 0.80]:
                    a_trial = a.copy()
                    reduce_amt = a[i] * (1 - frac)
                    a_trial[i] *= frac
                    # Redistribute to low-contribution positions
                    low_mask = np.ones(n, dtype=bool)
                    low_mask[idx] = False  # Don't redistribute to peak contributors
                    if a_trial[low_mask].sum() > 1e-15:
                        weights = a_trial[low_mask] + 1e-10
                        a_trial[low_mask] += reduce_amt * weights / weights.sum()
                    a_trial = norm_sequence(a_trial)
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
            else:
                # Pair: reduce a[i] and a[j]
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.90, 0.85]:
                    for f2 in [0.95, 0.90, 0.85]:
                        a_trial = a.copy()
                        red1 = a_trial[i] * (1 - f1)
                        red2 = a_trial[j] * (1 - f2)
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        total_red = red1 + red2
                        # Redistribute to low positions
                        low_mask = np.ones(n, dtype=bool)
                        low_mask[idx] = False
                        if a_trial[low_mask].sum() > 1e-15:
                            weights = a_trial[low_mask] + 1e-10
                            a_trial[low_mask] += total_red * weights / weights.sum()
                        a_trial = norm_sequence(a_trial)
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            
            if improved:
                break
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
    
    return best_a, best_score

    """L-BFGS with log-sum-exp smoothing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max
