def cross_correlate(w, a):
    """Compute (w star a)_i = sum_k w_k * a_{k-i} for i in [0, n-1].
    w has length 2n-1, a has length n."""
    n = len(a)
    total_len = len(w) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = rfft(w, fft_size)
    A_flip = rfft(a[::-1], fft_size)
    corr = irfft(W * A_flip, fft_size)
    return corr[n-1 : 2*n-1]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    import cvxpy as cp
    # Try fastest solvers first
    for solver_name in [cp.GUROBI, cp.GLOP, cp.MOSEK, cp.CBC, cp.GLPK, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 1])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                print(f"  LP solver: {solver_name} (test time {time.monotonic()-t0:.2f}s)")
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.GUROBI, cp.GLOP, cp.MOSEK, cp.CBC, cp.GLPK, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def golden_section_search(f, lo, hi, max_iter=40):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent with golden section 1D search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            if has_quad:
                quad_idx = 2 * i - k_lo

            def _score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                       _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                       _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(3.0 * v, 5.0 / n)
            best_x = golden_section_search(_score, 0.0, upper, max_iter=35)

            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with small multiplicative steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            if has_quad:
                quad_idx = 2 * i - k_lo

            best_delta = 0.0
            best_new_max = float(conv[k_lo:k_hi+1].max())

            for factor in [0.998, 0.995, 0.99, 0.985, 0.98, 0.97, 0.95,
                          1.002, 1.005, 1.01, 1.015, 1.02, 1.03, 1.05]:
                delta = v * (factor - 1.0)
                x = v + delta
                if x < -1e-15:
                    continue
                nv = c_vals.copy() + 2.0 * x * a_partners
                if has_quad:
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                local_max = float(nv.max())
                if local_max < best_new_max - 1e-16:
                    best_new_max = local_max
                    best_delta = delta

            if best_delta != 0.0:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                current_sum += best_delta

                changes = 2.0 * best_delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = best_delta * (2.0 * v + best_delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def smoothed_gradient_descent(a0, time_limit, verbose=False, label=""):
    """
    Projected gradient descent with log-sum-exp smoothed max.
    Uses adaptive beta and backtracking line search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    
    # Adaptive beta schedule
    betas = [5, 10, 20, 40, 60, 80, 100, 150, 200, 300, 500]
    beta_idx = 0
    
    # Initial learning rate (scaled by problem size)
    lr = 0.1 / n
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient
    
    it = 0
    no_improve = 0
    plateau_count = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        beta = betas[min(beta_idx, len(betas) - 1)]
        
        # Compute conv and gradient
        conv = fast_convolve(a)
        max_conv = conv.max()
        
        # Softmax weights
        shifted = beta * (conv - max_conv)
        # Clip for numerical stability
        shifted = np.clip(shifted, -500, 500)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()
        
        # Gradient of LSE(β*conv) w.r.t. a: 2β * (w star a)
        grad_conv = 2.0 * beta * cross_correlate(w, a)
        
        # Smoothed score for tracking
        lse_val = max_conv + np.log(exp_shifted.sum()) / beta
        smooth_score = 2.0 * n * lse_val
        actual_score = 2.0 * n * float(conv.max())
        
        if actual_score < best_score - 1e-12:
            best_score = actual_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Gradient step with momentum
        momentum = mu * momentum + (1 - mu) * grad_conv
        a_new = a - lr * momentum
        a_new = simplex_project(a_new)
        
        # Check if this improved the actual score
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * float(conv_new.max())
        
        if score_new < actual_score + 1e-10:
            a = a_new
            # Increase learning rate slightly
            lr *= 1.01
        else:
            # Backtrack
            lr *= 0.7
            no_improve += 1
        
        # Adapt beta: increase when making progress
        if no_improve % 50 == 0 and no_improve > 0:
            plateau_count += 1
            if plateau_count >= 3 and beta_idx < len(betas) - 1:
                beta_idx += 1
                plateau_count = 0
                lr = 0.1 / n  # Reset lr
                momentum = np.zeros(n)
        
        if no_improve >= 200:
            # Stuck, try to escape
            if beta_idx < len(betas) - 1:
                beta_idx += 1
            lr = 0.05 / n
            momentum = np.zeros(n)
            no_improve = 0
            plateau_count = 0
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD {label} it={it} beta={beta} ({elapsed:.0f}s): actual={actual_score:.8f} best={best_score:.8f}")
    
    return best_a, best_score

def mirror_descent(a0, time_limit, verbose=False, label=""):
    """
    Mirror descent (multiplicative weight updates) for the smoothed max.
    Works in log-space to naturally maintain positivity.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10).copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    
    betas = [5, 10, 20, 40, 80, 120, 200]
    beta_idx = 0
    lr = 0.5 / n
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        beta = betas[min(beta_idx, len(betas) - 1)]
        
        conv = fast_convolve(a)
        max_conv = conv.max()
        shifted = np.clip(beta * (conv - max_conv), -500, 500)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()
        
        # Gradient direction
        grad = 2.0 * beta * cross_correlate(w, a)
        
        # Mirror descent update: a_new_i = a_i * exp(-lr * grad_i) / sum
        log_update = -lr * grad
        log_update = np.clip(log_update, -10, 10)
        a_new = a * np.exp(log_update)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * float(conv_new.max())
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if score_new < 2.0 * n * float(conv.max()) + 1e-10:
            a = a_new
            lr *= 1.005
        else:
            lr *= 0.8
        
        if no_improve >= 150:
            if beta_idx < len(betas) - 1:
                beta_idx += 1
            lr = 0.5 / n
            no_improve = 0
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {label} it={it} beta={beta} ({elapsed:.0f}s): actual={2.0*n*float(conv_new.max()):.8f} best={best_score:.8f}")
    
    return best_a, best_score

def fictitious_play_with_linesearch(a0, time_limit, solver=None, verbose=False):
    """LP fictitious play with adaptive step size and time-average tracking."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    iters = 0
    a_avg = a.copy()
    
    # Adaptive step: start large, decay
    base_lr = 1.0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 45:
            break

        # Solve LP best response
        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            continue

        # Adaptive step size
        alpha = base_lr / (iters ** 0.5)
        alpha = min(alpha, 1.0)
        
        # Update
        a_new = (1.0 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0)
        cs = a_new.sum()
        if cs < 1e-10:
            continue
        a_new = a_new / cs
        a = a_new
        
        # Time average
        a_avg = (iters * a_avg + a) / (iters + 1)
        
        # Check candidates: current, average, and best response
        for cand, cname in [(a, 'cur'), (a_avg, 'avg'), (b, 'br')]:
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = cand.copy()

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.8f} avg_M={M_val:.6f}")

    return best_a, best_score, iters

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.5, 1.5, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.2, 0.7)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

def pair_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Pair coordinate descent: optimize pairs of coordinates simultaneously.
    This can escape local minima that single-coordinate descent gets stuck in.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        # Try pairs of close coordinates
        for i in range(0, n - 1, 2):
            if time.monotonic() - t0 > time_limit:
                break
            # Also try (i, i+1) and (i, i+n//4 mod n)
            for j in [i + 1, (i + n // 3) % n]:
                if j == i or a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                
                vi = a[i]
                vj = a[j]
                
                # Try transferring mass between i and j
                for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9, 0.95]:
                    # Transfer frac of vi to vj (or vice versa)
                    di = -vi * frac * 0.5
                    dj = vi * frac * 0.5
                    a[i] = max(vi + di, 0)
                    a[j] = max(vj + dj, 0)
                    new_sum = current_sum + di + dj
                    
                    if new_sum < 0.01:
                        a[i] = vi
                        a[j] = vj
                        continue
                    
                    conv_new = fast_convolve(a)
                    sc = 2.0 * n * float(conv_new.max()) / (new_sum * new_sum)
                    
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a.copy()
                        conv = conv_new
                        current_sum = new_sum
                        improved = True
                    else:
                        a[i] = vi
                        a[j] = vj
                        conv = fast_convolve(a) if not np.array_equal(conv, conv_new) else conv_new
                        # Reconvolve to be safe
                        conv = fast_convolve(a)

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0
            if verbose and sweeps % 10 == 0:
                elapsed = time.monotonic() - t0
                print(f"    PCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def propose_candidate():
    """
    Multi-phase optimization combining:
    - Smoothed gradient descent (NEW) for steady progress
    - Mirror descent (NEW) as alternative
    - LP fictitious play (improved with adaptive step)
    - Exact coordinate descent for polishing
    - Pair coordinate descent (NEW) for escaping local minima
    - Multi-length exploration
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(42)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: score={best_score:.8f}")

    # === Phase 1: Quick multi-length exploration ===
    print(f"\n=== Phase 1: Multi-length exploration (budget=120s) ===")
    t1_start = time.monotonic()
    t1_budget = 120.0

    n_values = [len(parent)]
    for alt in [768, 1024, 1280, 1536, 640]:
        if alt not in n_values and 256 <= alt <= 2048:
            n_values.append(alt)
    n_values = n_values[:5]

    candidates = []
    for n in n_values:
        elapsed = time.monotonic() - t1_start
        if elapsed > t1_budget - 10:
            break
        t_alloc = max(5.0, min(20.0, (t1_budget - elapsed) / 2.0))
        
        inits = [
            ('parent', interpolate_seq(parent, n)),
            ('flat', init_flat_spectrum(n, rng)),
            ('cos', init_cosine(n, rng)),
        ]
        for itype, cand in inits:
            if time.monotonic() - t1_start > t1_budget - 5:
                break
            t_sub = max(2.0, min(8.0, (t1_budget - (time.monotonic() - t1_start)) / 3.0))
            a_opt, s_opt, _ = exact_coord_descent(cand, t_sub, label=f"p1_{n}_{itype}")
            candidates.append((s_opt, a_opt, n, itype))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, best: {candidates[0][0]:.8f}")
    for s_c, a_c, n_c, it_c in candidates[:3]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{n_c}_{it_c}")

    # === Phase 2: Smoothed gradient descent (NEW) ===
    print(f"\n=== Phase 2: Smoothed Gradient Descent (budget=300s) ===")
    if best_seq is not None:
        t2_start = time.monotonic()
        t2_budget = 300.0
        n_best = len(best_seq)
        
        # Run SGD for multiple beta stages
        a_sgd = best_seq.copy()
        for stage, t_alloc in [(1, t2_budget * 0.4), (2, t2_budget * 0.35), (3, t2_budget * 0.25)]:
            rem = t2_budget - (time.monotonic() - t2_start)
            if rem < 10:
                break
            t_use = min(t_alloc, rem)
            print(f"  SGD stage {stage}, n={n_best}, budget={t_use:.0f}s")
            a_opt, s_opt = smoothed_gradient_descent(a_sgd, t_use, verbose=True, label=f"sgd_s{stage}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"sgd_s{stage}")
            a_sgd = a_opt.copy()
        
        # Mirror descent
        rem = t2_budget - (time.monotonic() - t2_start)
        if rem > 20:
            print(f"  Mirror descent, budget={min(rem, 60):.0f}s")
            a_md, s_md = mirror_descent(best_seq, min(rem, 60.0), verbose=True, label="md")
            best_seq, best_score = check_and_update(a_md, best_seq, best_score, "mirror")

    # === Phase 3: LP Fictitious Play ===
    print(f"\n=== Phase 3: LP Fictitious Play (budget=400s) ===")
    if best_seq is not None:
        t3_start = time.monotonic()
        t3_budget = 400.0
        n_best = len(best_seq)
        
        # Run on best length first, then try another
        fp_lengths = [n_best]
        for alt in [1024, 768, 1280]:
            if alt != n_best and 256 <= alt <= 2048:
                fp_lengths.append(alt)
        fp_lengths = list(set(fp_lengths))[:2]

        for n_fp in fp_lengths:
            rem = t3_budget - (time.monotonic() - t3_start)
            if rem < 60:
                break
            t_alloc = min(rem * 0.5, 180.0)
            print(f"  FP n={n_fp}, budget={t_alloc:.0f}s")

            initial = interpolate_seq(best_seq, n_fp)
            a_fp, score_fp, iters = fictitious_play_with_linesearch(
                initial, t_alloc, solver=solver, verbose=True
            )
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"fp_{n_fp}")

            # Follow up with coordinate descent
            rem = t3_budget - (time.monotonic() - t3_start)
            if rem > 30:
                t_cd = min(rem * 0.4, 80.0)
                a_cd, s_cd, _ = exact_coord_descent(a_fp, t_cd, label=f"post_fp_{n_fp}")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"ecd_fp_{n_fp}")

    # === Phase 4: Intensive coordinate descent ===
    print(f"\n=== Phase 4: Intensive ECD (budget=350s) ===")
    if best_seq is not None:
        t4_start = time.monotonic()
        t4_budget = 350.0
        n_best = len(best_seq)
        current = best_seq.copy()

        for rnd in range(5):
            rem = t4_budget - (time.monotonic() - t4_start)
            if rem < 20:
                break
            t_alloc = min(rem * 0.4, 100.0)
            print(f"  ECD round {rnd+1}, n={n_best}, budget={t_alloc:.0f}s")
            a_opt, s_opt, sweeps = exact_coord_descent(current, t_alloc,
                                                       verbose=True, label=f"ecd_r{rnd+1}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ecd_r{rnd+1}")

            if best_score > 1.5035:
                # Perturb to escape local minimum
                perturbed = a_opt + rng.standard_normal(n_best) * 0.02
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
                current = perturbed
            else:
                current = a_opt.copy()

    # === Phase 5: Pair coordinate descent + basin hopping ===
    print(f"\n=== Phase 5: Pair CD + Basin Hopping (budget=200s) ===")
    if best_seq is not None:
        t5_start = time.monotonic()
        t5_budget = 200.0
        n_best = len(best_seq)
        
        # Pair coordinate descent
        rem = t5_budget - (time.monotonic() - t5_start)
        if rem > 20:
            t_pcd = min(rem * 0.3, 60.0)
            print(f"  Pair CD, budget={t_pcd:.0f}s")
            a_pcd, s_pcd, _ = pair_coord_descent(best_seq, t_pcd, verbose=True, label="pair")
            best_seq, best_score = check_and_update(a_pcd, best_seq, best_score, "pair_cd")

        # Basin hopping with SGD as local optimizer
        rem = t5_budget - (time.monotonic() - t5_start)
        if rem > 30:
            t_bh = min(rem * 0.5, 80.0)
            print(f"  Basin hopping, budget={t_bh:.0f}s")
            n_perturbs = 5
            for p in range(n_perturbs):
                if time.monotonic() - t5_start > t5_budget - 10:
                    break
                t_sub = max(10.0, (t5_budget - (time.monotonic() - t5_start)) / (n_perturbs - p))
                
                # Different perturbation strategies
                if p % 2 == 0:
                    sigma = 0.02 + 0.02 * rng.random()
                    perturbed = best_seq + rng.standard_normal(n_best) * sigma
                else:
                    # Peak-breaking perturbation
                    conv_p = fast_convolve(best_seq)
                    peaks = np.argpartition(conv_p, -5)[-5:]
                    perturbed = best_seq.copy()
                    for k in peaks:
                        i_start = max(0, int(k) - n_best + 1)
                        i_end = min(n_best - 1, int(k))
                        idx = np.arange(i_start, i_end + 1)
                        picks = rng.choice(idx, size=max(1, len(idx) // 4), replace=False)
                        for ii in picks:
                            perturbed[ii] *= (1.0 + rng.standard_normal() * 0.1)
                
                perturbed = np.maximum(perturbed, 0)
                ps = perturbed.sum()
                if ps < 1e-10:
                    continue
                perturbed = perturbed / ps
                
                # Quick SGD polish
                a_opt, s_opt = smoothed_gradient_descent(perturbed, t_sub * 0.5, label=f"bh{p}")
                best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"bh_{p}")

    # === Phase 6: Final polishing ===
    print(f"\n=== Phase 6: Final Polish (budget=120s) ===")
    if best_seq is not None:
        t6_start = time.monotonic()
        t6_budget = 120.0

        # Fine coordinate descent
        rem = t6_budget - (time.monotonic() - t6_start)
        if rem > 15:
            t_fine = min(rem * 0.5, 50.0)
            print(f"  FineCD, budget={t_fine:.0f}s")
            a_fine, s_fine, _ = fine_coord_descent(best_seq, t_fine,
                                                   verbose=True, label="final_fine")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

        # Another exact CD round
        rem = t6_budget - (time.monotonic() - t6_start)
        if rem > 15:
            t_cd2 = min(rem * 0.5, 40.0)
            print(f"  ECD-polish, budget={t_cd2:.0f}s")
            a_cd2, s_cd2, _ = exact_coord_descent(best_seq, t_cd2, label="polish")
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "ecd_polish")

        # Final fine pass
        rem = t6_budget - (time.monotonic() - t6_start)
        if rem > 10:
            a_fin, s_fin, _ = fine_coord_descent(best_seq, min(rem * 0.6, 20.0), label="last")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "last_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
