import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_ab(a, b):
    """Linear cross-convolution a*b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_improved_lp(a_fixed, solver=None):
    """
    Improved LP using successive linearization:
    minimize M
    subject to 2*(T_a @ b)_k - conv(a,a)_k <= M for all k
    sum(b) = 1, b >= 0
    This focuses on reducing relative peaks.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    aa_conv = fast_convolve(a_fixed)
    # Constraint: 2*(T_a @ b)_k - aa_conv_k <= M  =>  (T_a @ b)_k <= (M + aa_conv_k) / 2
    # Equivalently: 2*(T_a @ b)_k - M <= aa_conv_k
    constraints = [2 * (T @ b_var) - M_var <= aa_conv, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solved = False
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True
    except Exception:
        pass
    if not solved:
        for alt in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def compute_lse_obj(z, n, T_temp):
    """Compute LogSumExp smoothed objective."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b - max_b)
    lse = max_b + np.log(exp_b.sum())
    return (2.0 * n / T_temp) * lse


def compute_lse_grad(z, n, T_temp):
    """Compute gradient of LogSumExp smoothed objective."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b - max_b)
    w = exp_b / exp_b.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    # grad_a now has factor 2 from symmetry, accounts for both terms
    grad_a *= 2.0
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_polish(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS with very low temperature LogSumExp for precise optimization."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]

    n_stages = len(T_stages)
    iters_per_stage = max(30, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(z, _T=stage_T):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            return compute_lse_obj(z, n, _T)

        def jac(z, _T=stage_T):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            return compute_lse_grad(z, n, _T)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

def evaluate_local(a):
    """Evaluate sequence locally (no function call overhead)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def spectral_init(n, rng):
    """Initialize with flat-power-spectrum structure."""
    # Random phases to create a signal with near-flat autocorrelation
    phases = rng.uniform(0, 2 * np.pi, size=n)
    # Use cosine to keep it real
    a = np.cos(phases)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        return a
    return a / s


def raised_cosine_init(n):
    """Raised cosine window initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def multi_peak_init(n, rng):
    """Multi-Gaussian peak initialization."""
    n_peaks = rng.integers(2, 5)
    a = np.zeros(n)
    for _ in range(n_peaks):
        center = rng.uniform(0.2, 0.8) * (n - 1)
        width = rng.uniform(0.05, 0.2) * n
        a += np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbation_polish(a0, time_limit, n_trials=50000):

def subgradient_polish(a0, time_limit, n_steps=100000):
    """Subgradient method on the exact max function."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

    a = a / s
    best_a = a.copy()

    no_improve = 0

    for step in range(n_steps):

            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9995
        active = np.where(b >= threshold)[0]

        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        a_trial = a_trial / a_trial.sum()

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 20:
            lr *= 0.92
            no_improve = 0
        lr = max(1e-7, lr)

        if step % 20000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.10f} lr={lr:.8f}")

    return best_a, best_score

    t_alloc = min(rem * 0.35, 550.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, use_improved=True, n_max_iters=500000)
        try_update(a_lp, "lp_improved")

    # === Phase 2: L-BFGS LogSumExp polish (ultra-low T) ===
    print(f"\n=== Phase 2: L-BFGS polish, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_lb = lbgfs_polish(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=6000)
        try_update(a_lb, "lbgfs_1")

    # === Phase 3: Subgradient refinement ===
    print(f"\n=== Phase 3: Subgradient, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        a_sg, _ = subgradient_polish(best_seq, t_alloc, n_steps=200000)
        try_update(a_sg, "subgrad")

    # === Phase 4: Perturbation polish ===
    print(f"\n=== Phase 4: Perturbation, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        a_pert, _ = perturbation_polish(best_seq, t_alloc, n_trials=80000)
        try_update(a_pert, "perturb_1")

    # === Phase 5: Second LP round (standard formulation for diversity) ===
    print(f"\n=== Phase 5: Nesterov LP (standard) score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        # Slightly perturb before LP to escape potential fixed point
        noise = rng.normal(0, 0.005, size=len(best_seq))
        perturbed = best_seq + noise
        perturbed = np.maximum(perturbed, 0)
        perturbed = perturbed / perturbed.sum()
        a_lp2, _ = nesterov_lp_refine(perturbed, t_alloc * 0.7, solver=solver, use_improved=False, n_max_iters=200000)
        try_update(a_lp2, "lp_standard")
        rem = deadline - time.monotonic()
        if rem > 30:
            a_lb2 = lbgfs_polish(best_seq, rem * 0.5, T_stages=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02], maxiter_per_stage=4000)
            try_update(a_lb2, "lbgfs_2")

    # === Phase 6: Try alternative length ===
    print(f"\n=== Phase 6: Alt length, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        alt_n = n_init + 400
        print(f"  Trying n={alt_n}")
        t_alt = min(rem * 0.25, 250.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = nesterov_lp_refine(init_alt, t_alt * 0.5, solver=solver, use_improved=True, n_max_iters=50000)
        try_update(a_alt, f"lp_n{alt_n}")
        rem = deadline - time.monotonomic() if False else deadline - time.monotonic()
        if rem > 30:
            a_alt2 = lbgfs_polish(best_seq, rem * 0.4, T_stages=[10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1], maxiter_per_stage=3000)
            try_update(a_alt2, f"lbgfs_n{alt_n}")

        # Also try slightly shorter
        rem = deadline - time.monotonic()
        if rem > 100:
            alt_n2 = n_init - 200
            if alt_n2 > 400:
                print(f"  Trying n={alt_n2}")
                t_alt2 = min(rem * 0.2, 150.0)
                init_alt2 = interpolate_sequence(best_seq, alt_n2)
                a_alt3, _ = nesterov_lp_refine(init_alt2, t_alt2 * 0.5, solver=solver, use_improved=True, n_max_iters=30000)
                try_update(a_alt3, f"lp_n{alt_n2}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final polish, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.35, 80.0)
        a_fin = lbgfs_polish(best_seq, t_fin,
                             T_stages=[2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005],
                             maxiter_per_stage=5000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ps = min(rem * 0.3, 40.0)
        a_ps, _ = perturbation_polish(best_seq, t_ps, n_trials=30000)
        try_update(a_ps, "final_perturb")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_sg = min(rem * 0.4, 25.0)
        a_sg2, _ = subgradient_polish(best_seq, t_sg, n_steps=50000)
        try_update(a_sg2, "final_subgrad")

    # === Phase 8: One more LP round with best found so far ===
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\n=== Phase 8: Final LP round, score={best_score:.10f} ===")
        t_final = min(rem * 0.5, 100.0)
        a_fin2, _ = nesterov_lp_refine(best_seq, t_final, solver=solver, use_improved=True, n_max_iters=100000)
        try_update(a_fin2, "final_lp")
        rem = deadline - time.monotonic()
        if rem > 15:
            a_fin3 = lbgfs_polish(best_seq, rem * 0.6,
                                  T_stages=[1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01],
                                  maxiter_per_stage=3000)
            try_update(a_fin3, "final_lbgfs2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
