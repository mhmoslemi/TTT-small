def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

    theta = (cssv[rho] - 1) / (rho + 1.0)

```python
def solve_best_response_lpb(a, method='highs'):
    """Solve the LP: min M s.t. T@b <= M, sum(b)=1, b>=0, M>=0."""
    from scipy.optimize import linprog
    n = len(a)
    # Build T matrix: (T@b)_k = sum_j a_{k-j} * b_j (where 0<=k-j<n)
    # T has shape (2n-1, n)
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    T = np.zeros((2*n-1, n))
    for j in range(n):
        js = max(0, 0)  # k-j >= 0 => k >= j
        je = min(2*n-2, j+n-1)  # k-j < n => k < j+n
        idx = np.arange(j, je+1)
        T[idx, j] = a[:len(idx)]
    
    # LP: min M s.t. T@b - M <= 0, sum(b) = 1, b >= 0, M >= 0
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.hstack([T, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method=method)
    
    if res.status != 0:
        return None, None
    
    b_val = res.x[:n]
    M_val = res.x[-1]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, M_val
```
