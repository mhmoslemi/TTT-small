def fast_convolve(a):

    n = len(a)

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def golden_section_search(f, lo, hi, max_iter=40):
    """Golden section search for minimum of f on [lo, hi]. Returns optimal x."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    # Slight random amplitude variation to avoid degeneracies
    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.65)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)

def init_fourier_mix(n, rng):
    """Mix of Fourier modes with random coefficients."""
    n_harmonics = rng.integers(3, 15)
    amps = rng.random(n_harmonics) * 0.3
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = rng.choice(np.arange(1, n // 2 + 1), size=n_harmonics, replace=False)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amps[j] * np.cos(2.0 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def subgradient_step(a, conv, lr, n_peaks=10):
    """One subgradient step using active peak weighting."""
    n = len(a)
    # Identify top peaks
    k_act = min(n_peaks, 2 * n - 1)
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks]
    # Soft weighting (more weight to larger peaks)
    temp = 0.005 * max(pv.max(), 1e-10)
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        return a
    w = w / ws

    # Compute subgradient
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        grad[j] += 2.0 * w[idx] * a[k - j]

    # Subgradient descent with simplex projection
    a_new = simplex_project(a - lr * grad)
    return a_new

def fictitious_play_with_linesearch(a0, time_limit, solver=None, verbose=False):
    """LP fictitious play with line search along the best response direction."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    iters = 0
    a_avg = a.copy()
    decay = 2.0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break

        # Solve LP best response
        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            continue

        # Line search: a_new = (1-t)*a + t*b for t in [0, 1]
        best_t = 0.0
        best_ls_score = best_score
        for trial_t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_ls_score:
                best_ls_score = sc
                best_t = trial_t

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * b
            a_new = np.maximum(a_new, 0)
            cs = a_new.sum()
            if cs < 1e-10:
                continue
            a_new = a_new / cs
            score_new = 2.0 * n * float(fast_convolve(a_new).max())
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
            a = a_new
        else:
            # Fallback: use averaging
            alpha = decay / (iters + decay)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                continue
            a = a / cs

        a_avg = (iters * a_avg + a) / (iters + 1)

        # Also check the average
        for cand in [a, a_avg, b]:
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, iters

def basin_hopping(a0, time_limit, cd_func, rng, n_perturbs=10, verbose=False):
    """Basin hopping: perturb the sequence and re-optimize."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break
        t_alloc = min(rem * 0.6, 60.0)

        # Perturb: add noise scaled by current variance
        sigma = 0.02 + 0.03 * rng.random()
        perturbed = a + rng.standard_normal(n) * sigma
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        # Also try a "peak breaking" perturbation
        conv_p = fast_convolve(a)
        top_k = min(5, 2 * n - 1)
        peaks = np.argpartition(conv_p, -top_k)[-top_k:]
        for k in peaks:
            i_start = max(0, int(k) - n + 1)
            i_end = min(n - 1, int(k))
            idx = np.arange(i_start, i_end + 1)
            n_pick = max(1, len(idx) // 5)
            picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
            for ii in picks:
                perturbed[ii] *= (1.0 + rng.standard_normal() * 0.05)
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = cd_func(perturbed, t_alloc)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p}: improved to {best_score:.8f}")

    return best_a, best_score

def check_and_update(a, best_seq, best_score, label=""):
