"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (accelerated projected gradient) with adaptive logsumexp smoothing
2. Analytical coordinate descent with optimal delta computation
3. Pairwise mass-transfer moves for peak reduction
4. Structured gradient-based perturbations for exploration
5. Multi-phase time allocation with adaptive refinement

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1] if cond.any() else n - 1
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_conv_gradient(conv, a, n):
    """Compute gradient of max(conv) w.r.t. a using uniform weights on argmax."""
    M = conv.max()
    # Positions where conv is close to max (for numerical stability)
    threshold = M - 1e-10 * max(1, M)
    peak_mask = conv >= threshold
    n_peaks = peak_mask.sum()
    if n_peaks == 0:
        peak_mask = conv == M
    
    grad = np.zeros(n)
    for k in np.where(peak_mask)[0]:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            grad[i] += 2.0 * a[k - i]
    grad /= n_peaks
    return grad


def compute_smooth_gradient(conv, a, n, beta):
    """Compute gradient of smooth max (logsumexp) w.r.t. a."""
    M = conv.max()
    scaled = beta * (conv - M)
    scaled = np.clip(scaled, -50, 0)
    w = np.exp(scaled)
    w = w / w.sum()
    
    grad = np.zeros(n)
    significant = w > 1e-12
    for k in np.where(significant)[0]:
        wk = w[k]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        grad[j] += 2.0 * wk * a[k - j]
    return grad


def fista_simplex(a0, time_limit, beta_init=20.0, beta_max=200.0, verbose=False, label=""):
    """
    FISTA on the simplex with adaptive logsumexp smoothing.
    Minimizes f_beta(a) = (1/beta)*log(sum exp(beta*conv[k])) subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    z = a.copy()  # FISTA extrapolation point
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    k = 0  # FISTA iteration counter
    t_fista = 1.0
    beta = beta_init
    step = 0.01
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # FISTA extrapolation
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista))
        z = a + ((t_fista - 1.0) / t_new) * (a - z)
        z = np.maximum(z, 0)
        sz = z.sum()
        if sz < 1e-15:
            z = a.copy()
        else:
            z = z / sz  # Keep z on simplex for stability
        
        # Compute gradient at z
        conv_z = fast_convolve(z)
        grad = compute_smooth_gradient(conv_z, z, n, beta)
        
        # Backtracking linesearch
        step_new = step
        for _ in range(20):
            a_trial = simplex_project(z - step_new * grad)
            # Check sufficient decrease (Armijo condition, simplified)
            # For simplicity, just check that the objective decreased
            if step_new < 1e-10:
                break
            # Accept if step is small enough or objective improved
            conv_trial = fast_convolve(a_trial)
            f_trial = conv_trial.max()
            f_current = conv_z.max()
            if f_trial <= f_current + 1e-12:
                break
            step_new *= 0.5
        
        a = simplex_project(z - step_new * grad)
        t_fista = t_new
        k += 1
        
        # Evaluate and update best
        conv_a = fast_convolve(a)
        score = 2.0 * n * conv_a.max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive beta: increase when stuck
        if no_improve > 50:
            beta = min(beta * 1.5, beta_max)
            no_improve = 0
            step = max(step * 0.8, 1e-6)
            if beta >= beta_max and step < 1e-6:
                break
        
        # Gradually increase beta
        if k % 20 == 0:
            beta = min(beta * 1.05, beta_max)
        
        if verbose and k % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA {label} k={k} ({elapsed:.0f}s): score={best_score:.8f} beta={beta:.1f}")
    
    return best_a, best_score

    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        any_improve = False

def pairwise_swap_cd(a0, time_limit, verbose=False, label=""):
    """
    Coordinate descent using pairwise mass-transfer moves.
    Transfer mass from coordinate i to coordinate j (preserving sum).
    """
    n = len(a0)

        peak_pos = int(conv.argmax())

            break

        # to low-contribution positions

                        continue

                    # conv changes: for each k, delta_conv[k] = delta*(a[k-i_to] - a[k-i_from])
                    # (plus quadratic corrections at 2*i_from, 2*i_to, i_from+i_to)
                    
                    # Quick check: does this reduce the peak?

                        # This is getting too complex, just recompute

def gradient_perturb(a0, rng, strength=0.02):
    """
    Apply a gradient-based perturbation that moves mass away from peak-contributing
    coordinates and towards low-contributing ones.
    """
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    
    # Compute "pressure" on each coordinate (how much it contributes to peaks)
    pressure = np.zeros(n)
    # Use top 20% of conv positions
    threshold = M * 0.8
    peak_mask = conv >= threshold
    n_peaks = peak_mask.sum()
    
    for k in np.where(peak_mask)[0]:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            # Coordinate i contributes a[i]*a[j] to conv[k]
            # "Pressure" on i is proportional to a[j] (how much reducing a[i] helps)
            pressure[i] += a[j]
    
    # Normalize
    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max
    
    # Move mass: reduce high-pressure, increase low-pressure
    # Use a flow: from high pressure to low pressure
    high_mask = pressure > 0.7
    low_mask = pressure < 0.3
    n_high = high_mask.sum()
    n_low = low_mask.sum()
    
    if n_high > 0 and n_low > 0:
        # Total mass to transfer
        high_mass = a[high_mask].sum()
        transfer = strength * high_mass
        a[high_mask] -= transfer * (pressure[high_mask] / pressure[high_mask].sum())
        a[low_mask] += transfer * (a[low_mask] / max(a[low_mask].sum(), 1e-20))
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA on simplex with adaptive logsumexp smoothing (O(1/k^2) convergence)
2. Fast coordinate descent with vectorized optimal-delta search
3. Simulated annealing with CD refinement for escaping local optima
4. Multi-scale structured perturbation for global exploration
5. Adaptive time allocation across phases

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1] if cond.any() else n - 1
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(conv, a, n, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*conv[k])) w.r.t. a."""
    M = conv.max()
    scaled = beta * (conv - M)
    scaled = np.clip(scaled, -60, 0)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[conv.argmax()] = 1.0
    else:
        w = w / w_sum

    grad = np.zeros(n)
    significant = w > 1e-12
    peak_indices = np.where(significant)[0]
    for k in peak_indices:
        wk = w[k]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        grad[j] += 2.0 * wk * a[k - j]
    return grad


def fista_simplex(a0, time_limit, beta_init=30.0, beta_max=500.0, verbose=False, label=""):
    """
    FISTA on the simplex with adaptive logsumexp smoothing.
    Minimizes smooth surrogate of max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    z = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    t0 = time.monotonic()
    k = 0
    t_fista = 1.0
    beta = beta_init
    step = 0.05
    no_improve = 0
    patience = 80

    while time.monotonic() - t0 < time_limit:
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista))
        z_new = a + ((t_fista - 1.0) / t_new) * (a - z)
        z_new = np.maximum(z_new, 0.0)
        sz = z_new.sum()
        if sz < 1e-15:
            z_new = a.copy()
        else:
            z_new = z_new / sz

        conv_z = fast_convolve(z_new)
        grad = compute_smooth_gradient(conv_z, z_new, n, beta)

        # Backtracking linesearch
        step_local = step
        for _ in range(15):
            a_trial = simplex_project(z_new - step_local * grad)
            if step_local < 1e-12:
                break
            # Simple acceptance: always accept (FISTA is robust)
            break

        a_new = simplex_project(z_new - step_local * grad)

        # Evaluate
        conv_a = fast_convolve(a_new)
        score = 2.0 * n * conv_a.max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        z = a
        a = a_new
        t_fista = t_new
        k += 1

        # Adaptive parameter tuning
        if no_improve > patience:
            beta = min(beta * 2.0, beta_max)
            step = max(step * 0.5, 1e-8)
            no_improve = 0
            patience = max(patience * 0.9, 30)
            if beta >= beta_max and step < 1e-7:
                break
        elif k % 30 == 0:
            beta = min(beta * 1.03, beta_max)

        if verbose and k % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA {label} k={k} ({elapsed:.0f}s): score={best_score:.8f} beta={beta:.1f} step={step:.2e}")

    return best_a, best_score


def fast_cd_optimal(a0, time_limit, verbose=False, label=""):
    """
    Fast coordinate descent with optimized delta selection.
    For each coordinate, tries a range of deltas and picks the one that
    most reduces the convolution peak.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    # Precompute delta fractions (log-spaced for better coverage)
    pos_fracs = np.array([0.001, 0.003, 0.01, 0.03, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9])
    neg_fracs = np.array([0.001, 0.003, 0.01, 0.03, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9, 0.99])

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-14:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            if unaff_max >= M - 1e-14:
                continue

            # Coefficients for linear part
            k_range = np.arange(k_start, k_end + 1)
            valid_k_minus_i = np.clip(k_range - i, 0, n - 1)
            coeffs = 2.0 * a[valid_k_minus_i]

            idx_2i = 2 * i - k_start if (k_start <= 2 * i <= k_end) else -1
            rest_2i = conv[2 * i] - ai * ai if idx_2i >= 0 else 0.0

            affected = conv[k_start:k_end + 1].copy()

            best_delta = 0.0
            best_new_M = M

            # Try positive deltas
            for f in pos_fracs:
                d = f * ai
                nv = ai + d
                if nv > 1.0:
                    continue
                new_aff = affected + coeffs * d
                if idx_2i >= 0:
                    new_aff = new_aff.copy()
                    new_aff[idx_2i] = nv * nv + rest_2i
                nm = max(unaff_max, float(new_aff.max()))
                if nm < best_new_M - 1e-15:
                    best_new_M = nm
                    best_delta = d

            # Try negative deltas
            for f in neg_fracs:
                d = -f * ai
                nv = ai + d
                if nv < -1e-15:
                    continue
                new_aff = affected + coeffs * d
                if idx_2i >= 0:
                    new_aff = new_aff.copy()
                    new_aff[idx_2i] = max(nv, 0.0) ** 2 + rest_2i
                nm = max(unaff_max, float(new_aff.max()))
                if nm < best_new_M - 1e-15:
                    best_new_M = nm
                    best_delta = d

            if best_delta != 0.0:
                new_ai = max(ai + best_delta, 0.0)
                # Update conv in-place
                conv[k_start:k_end + 1] = affected + coeffs * best_delta
                if idx_2i >= 0:
                    conv[2 * i] = new_ai * new_ai + rest_2i
                a[i] = new_ai

                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCDO {label} sweep={sweeps} ({elapsed:.0f}s): score={best_score:.8f} M={M:.10f}")

    return best_a, best_score

def simulated_annealing_cd(a0, time_limit, rng, verbose=False, label=""):
    """
    Simulated annealing: periodically perturb the solution, then do local CD.
    Accept worse perturbations with temperature-based probability.
    """
    n = len(a0)
    current = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    cs = current.sum()
    if cs < 1e-15:
        current = np.ones(n) / n
    else:
        current = current / cs

    best_a = current.copy()
    best_score = 2.0 * n * fast_convolve(current).max()

    t0 = time.monotonic()
    temperature = 1.0
    cooling = 0.97
    iterations = 0

    while time.monotonic() - t0 < time_limit:
        iterations += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        # Generate perturbation
        perturb_mag = temperature * 0.05
        noise = rng.standard_normal(n) * perturb_mag
        perturbed = current + noise
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-15:
            perturb_mag *= 0.5
            continue
        perturbed = perturbed / ps

        # Quick evaluation
        p_score = 2.0 * n * fast_convolve(perturbed).max()

        # Accept/reject
        if p_score < best_score or rng.random() < np.exp(-(p_score - best_score) / max(temperature * 0.001, 1e-10)):
            current = perturbed

        # Do a few rounds of CD to refine
        cd_time = min(rem * 0.3, 30.0)
        a_cd, s_cd = fast_cd_optimal(current, cd_time, verbose=False, label=f"sa{iterations}")
        if s_cd < best_score:
            best_score = s_cd
            best_a = a_cd.copy()
        current = a_cd

        temperature *= cooling

            print(f"    SA {label} it={iterations} ({elapsed:.0f}s): best={best_score:.8f} T={temperature:.4f}")

    return best_a, best_score


def gradient_flow_perturb(a0, rng, strength=0.03):
    """
    Perturb along the negative subgradient direction (out of the peaks).
    """
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()

    # Subgradient of max(conv): average of gradients at argmax positions
    threshold = M - 1e-8 * max(1.0, M)
    peak_mask = conv >= threshold
    peak_indices = np.where(peak_mask)[0]
    n_peaks = len(peak_indices)

    if n_peaks == 0:
        return a

    grad = np.zeros(n)
    for k in peak_indices:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            grad[i] += 2.0 * a[k - i]
    grad /= n_peaks

    # Move opposite to gradient (reduce peak-contributing coords)
    a_new = a - strength * grad
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    a_new = a_new / sn

    # Add small random component for diversity
    a_new += rng.standard_normal(n) * strength * 0.1
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


def block_perturb(a0, rng, block_size=32):
    """
    Perturb a random block of coordinates by redistributing mass.
    """
    n = len(a0)
    a = a0.copy()
    start = int(rng.integers(0, max(1, n - block_size)))
    end = min(start + block_size, n)
    block = a[start:end].copy()
    bsum = block.sum()
    if bsum < 1e-15:
        return a
    # Redistribute within block
    new_block = rng.dirichlet(np.ones(end - start) * 2.0) * bsum
    a[start:end] = new_block
    return a


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
