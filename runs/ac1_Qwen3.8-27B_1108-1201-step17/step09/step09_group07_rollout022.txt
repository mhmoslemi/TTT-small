import time

def fast_convolve(a):

def project_simplex(v):

def ftrl_lp_refine(a0, time_limit, solver, n_max_iters=100000):
    """FTRL outer loop with LP best response."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    # FTRL state
    cumulative_loss = np.zeros(n)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    eta0 = 0.1
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_best_response(a, solver)
        if b_new is None:
            continue
        
        # Compute "loss" for FTRL: the gradient of the smoothed max
        # Use the convolution with b_new as a proxy
        conv_ab = np.convolve(a, b_new)
        M_curr = conv_ab.max()
        
        # FTRL update: a_new[i] proportional to exp(-eta * cumulative_loss[i])
        # But we need to add the new loss first
        # The "loss" for coordinate i is the contribution to the max
        # Approximate: loss[i] = sum_k lambda[k] * a[k-i] where lambda is dual
        # Simplification: use the gradient of max(conv(a, b)) w.r.t. a
        # grad[i] = b[max_idx - i] + b[i - max_idx] (approximately)
        
        # Actually, let's use a simpler update: Nesterov-style averaging
        # with adaptive step size
        alpha = 2.0 / (it + 2)
        a_trial = (1 - alpha) * a + alpha * b_new
        a_trial = project_simplex(a_trial)
        
        # Backtracking: try to reduce alpha if score increases
        score_trial = score_normalized(a_trial)
        if score_trial > best_score * 1.001:
            alpha = alpha * 0.5
            a_trial = (1 - alpha) * a + alpha * b_new
            a_trial = project_simplex(a_trial)
            score_trial = score_normalized(a_trial)
        
        a = a_trial
        
        if score_trial < best_score:
            best_score = score_trial
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if it % 200 == 0:
            if best_score < prev_best - 1e-8:
                stall = 0
                prev_best = best_score
            if stall >= 8:
                # Perturb
                rng = np.random.default_rng(it * 7 + 13)
                a = a + 0.01 * rng.standard_normal(n)
                a = project_simplex(a)
                stall = 0
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FTRL iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} M_b={M_b:.8f}")

def kl_mirror_lp_refine(a0, time_limit, solver, n_max_iters=50000):
    """KL mirror descent with LP best response."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_best_response(a, solver)
        if b_new is None:
            continue
        
        # KL mirror descent step
        # min_b [KL(b || a) + eta * <L, b>] where L is the "loss"
        # Solution: b_new[i] = a[i] * exp(-eta * L[i]) / Z
        # Approximate L from the dual: L[i] ~ gradient of max conv
        # Use: L[i] = sum_k (conv(a,b_new)[k] / M_b) * b_new[max_idx - i]
        # Simplification: just mix with geometric rate
        eta = 0.5 / (1.0 + 0.1 * it)
        
        # Compute gradient of max(conv(a, b_new)) w.r.t. a
        conv_ab = np.convolve(a, b_new)
        peak_idx = int(np.argmax(conv_ab))
        
        # grad[i] = b_new[peak_idx - i] (if valid) + b_new[i - peak_idx] (if valid and different)
        grad = np.zeros(n)
        for i in range(n):
            j1 = peak_idx - i
            j2 = i - peak_idx
            if 0 <= j1 < n:
                grad[i] += b_new[j1]
            if 0 <= j2 < n and j2 != j1:
                grad[i] += b_new[j2]
        
        # KL update
        log_a = np.log(a)
        new_log_a = log_a - eta * grad
        new_log_a -= new_log_a.max()
        a_new = np.exp(new_log_a)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        score_new = score_normalized(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        a = a_new
        
        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      KL iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score
