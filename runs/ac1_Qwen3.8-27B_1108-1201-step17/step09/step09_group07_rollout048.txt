import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def grad_smooth_max(a, T):
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr

def nesterov_pg(a0, time_limit, T_start=2.0, T_end=200.0, lr_init=0.05, n_stages=12, steps_per_stage=2000):
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score
            lr = lr_init / (1.0 + 0.01 * step_count)
            t_curr = (1.0 + np.sqrt(1.0 + 4.0 * step_count)) / 2.0
            t_prev = (1.0 + np.sqrt(1.0 + 4.0 * max(0, step_count - 1))) / 2.0
            beta = (t_prev - 1.0) / t_curr
            y = x + beta * (x - x_prev)
            y = project_simplex(y)
            grad = grad_smooth_max(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1
            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
    return best_a, best_score

    T = np.zeros((size, n))

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv)."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def grad_smooth_max(a, T):
    """Gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr

    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_chirp_init(n, alpha, freq_scale=1.0):
    """Chirp-based initialization for frequency-domain flatness."""
    x = np.arange(n)
    phase = 2.0 * np.pi * freq_scale * x * x / (4.0 * n)
    a = 1.0 + alpha * np.cos(phase)
    a = np.maximum(a, 1e-8)
    a = a / a.sum()
    return a


def make_init(n, rng, mode=0):
    """Various initialization modes."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        k = rng.integers(1, 5)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * k * x)
    elif mode == 2:
        a = 1.0 + 0.4 * rng.standard_normal(n)
    elif mode == 3:
        c1 = rng.uniform(0.2, 0.4)
        c2 = rng.uniform(0.6, 0.8)
        a = np.exp(-4 * (x - c1) ** 2) + np.exp(-4 * (x - c2) ** 2) + 0.1
    elif mode == 4:
        a = x ** rng.uniform(0.5, 3.0)
    elif mode == 5:
        k = rng.integers(3, 10)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * k * x)
    elif mode == 6:
        a = rng.random(n) + 0.1
    elif mode == 7:
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.sin(6 * np.pi * x)
    elif mode == 8:
        # Chirp
        a = make_chirp_init(n, 0.4, rng.uniform(0.5, 2.0))
    elif mode == 9:
        # Anti-symmetric emphasis
        a = 1.0 + 0.3 * np.abs(np.sin(np.pi * x)) ** 2
    else:
        k = rng.choice([1, 2, 3, 4])
        a = 1.0 + rng.uniform(0.3, 0.6) * np.sin(2 * np.pi * k * x + rng.uniform(0, np.pi))

    a = np.maximum(a, 1e-8)
    a = a / a.sum()
    return a


def evaluate_safe(a):
    """Evaluate sequence with the external evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Quick multi-n sweep
    2. Focused Nesterov PG
    3. Mirror descent
    4. FW-LP refinement
    5. Multi-start diverse initializations
    6. Peak shaving
    7. Final LP polish
    """
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s))
    deadline = time.monotonic() + total_budget - 25.0

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {total_budget:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except (NameError, Exception) as e:
        print(f"No parent available: {e}")

    if parent is None:
        n_init = 960
        best_seq = make_chirp_init(n_init, 0.4, 1.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default chirp init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            improvement = best_score - sc
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)}, imp={improvement:.6f})")

    # === Phase 1: Quick multi-n sweep ===
    print(f"\n=== Phase 1: Multi-n Sweep (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    candidate_ns = []
    for delta in [-100, -64, -32, -16, 0, 16, 32, 64, 100, 128, 200]:
        cn = n_best + delta
        if 256 <= cn <= 3000:
            candidate_ns.append(cn)
    # Add some absolute values
    for cn in [400, 512, 640, 768, 896, 1024, 1280, 1600, 2000]:
        if cn not in candidate_ns and 256 <= cn <= 3000:
            candidate_ns.append(cn)
    candidate_ns = sorted(set(candidate_ns))

    t_per_n = min(rem * 0.18 / len(candidate_ns), 25.0)
    for cn in candidate_ns:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        if cn == n_best:
            init_cn = best_seq.copy()
        else:
            init_cn = interpolate_sequence(best_seq, cn)

        a_sub, _ = nesterov_pg(init_cn, t_per_n, T_start=5.0, T_end=200.0,
                               lr_init=0.08, n_stages=6, steps_per_stage=2000)
        try_update(a_sub, f"sweep_n{cn}")

    n_best = len(best_seq)
    print(f"  After sweep: best n={n_best}, score={best_score:.6f}")

    # === Phase 2: Focused Nesterov PG ===
    print(f"\n=== Phase 2: Focused NG (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ng, _ = nesterov_pg(best_seq, t_alloc, T_start=3.0, T_end=400.0,
                              lr_init=0.05, n_stages=12, steps_per_stage=3000)
        try_update(a_ng, "ng_focus")

    # === Phase 3: Mirror descent ===
    print(f"\n=== Phase 3: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 20:
        a_md, _ = mirror_descent(best_seq, t_alloc, lr_init=0.06, decay=0.99995, T_smooth=40.0)
        try_update(a_md, "mirror")

    # === Phase 4: FW-LP Refinement (main workhorse) ===
    print(f"\n=== Phase 4: FW-LP Refine (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_fw, _ = fp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_fw, "fw_lp_main")

    # === Phase 5: Multi-start with diverse inits ===
    print(f"\n=== Phase 5: Multi-Start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_seq)
        modes = [8, 1, 3, 5, 7]  # chirp, sin, two-bump, high-freq, cosine mix
        for trial, mode in enumerate(modes):
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_sub = min(rem * 0.22, 100.0)
            if t_sub < 25:
                break
            a_init = make_init(n_cur, rng, mode=mode)
            print(f"  mode={mode} (t={t_sub:.0f}s)")
            a_sub, _ = nesterov_pg(a_init, t_sub * 0.6, T_start=2.0, T_end=300.0,
                                   lr_init=0.08, n_stages=8, steps_per_stage=2000)
            try_update(a_sub, f"ms_m{mode}")
            rem = deadline - time.monotonic()
            if rem > 25:
                a_sub2, _ = fp_refine(a_sub, min(rem * 0.35, 40.0), solver=solver, n_max_iters=8000)
                try_update(a_sub2, f"ms_m{mode}_lp")

    # === Phase 6: Second FW pass with slightly perturbed start ===
    print(f"\n=== Phase 6: FW Pass 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(rem * 0.20, 200.0)
        # Perturb slightly to escape potential local minimum
        a_pert = best_seq + 0.01 * rng.standard_normal(len(best_seq))
        a_pert = project_simplex(a_pert)
        a_fw2, _ = fp_refine(a_pert, t_alloc, solver=solver, n_max_iters=100000)
        try_update(a_fw2, "fw_pass2")

    # === Phase 7: Peak shaving ===
    print(f"\n=== Phase 7: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 80.0)
    if t_alloc > 10:
        a_ps, _ = peak_shave(best_seq, t_alloc, n_steps=80000)
        try_update(a_ps, "peak_shave")

    # === Phase 8: Final short LP polish ===
    print(f"\n=== Phase 8: Final LP Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.25, 60.0)
        a_fin, _ = fp_refine(best_seq, t_fin, solver=solver, n_max_iters=50000)
        try_update(a_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            if fs < best_score:
                best_score = fs
            print(f"\nFinal verification: {fs:.6f}")

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
