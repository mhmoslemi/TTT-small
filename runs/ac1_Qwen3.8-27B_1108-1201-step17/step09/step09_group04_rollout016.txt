```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using soft-max smoothed L-BFGS
with adaptive temperature scheduling, active-set subgradient polishing,
and multi-length exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def get_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def softmax_obj(z, t):
    """Soft-max smoothed objective in log-domain."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_max = c.max()
    exp_tc = np.exp(t * (c - c_max))
    val = c_max + np.log(exp_tc.sum()) / t
    return 2.0 * len(a) * val


def softmax_grad(z, t):
    """Gradient of soft-max objective w.r.t. z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_tc = np.exp(t * (c - c_max))
    p = exp_tc / exp_tc.sum()
    g_a = 2.0 * np.correlate(p, a, mode='valid')
    return a * (g_a - np.dot(g_a, a))


def lbfgs_softmax(a0, time_limit, t_start=20.0, t_end=5000.0, n_stages=10):
    """L-BFGS with adaptive soft-max temperature schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.mean()  # center for stability
    t0 = time.monotonic()
    
    temps = np.geomspace(t_start, t_end, n_stages)
    
    for stage_t in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        max_iter = max(50, int(remaining * 50))  # ~50 iterations per second
        
        result = minimize(
            fun=softmax_obj,
            x0=z,
            args=(stage_t,),
            jac=softmax_grad,
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'gtol': 1e-12, 'ftol': 1e-15, 'disp': False}
        )
        z = result.x
        z = z - z.mean()
        
        # Evaluate
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        s = get_score(a_cur)
        elapsed = time.monotonic() - t0
        print(f"    t={stage_t:.0f} ({elapsed:.0f}s): score={s:.6f}")
    
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


def active_set_polish(a0, time_limit, n_max_iters=50000):
    """Active-set subgradient polishing for the max-norm."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = get_score(a)
    t0 = time.monotonic()
    
    # Use a combination of subgradient and Nesterov averaging
    y = a.copy()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(y)
        k_max = np.argmax(c)
        
        # Subgradient of max(c) at a
        i_start = max(0, k_max - n + 1)
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        g = np.zeros(n)
        g[idx] = 2.0 * a[k_max - idx]
        
        # Nesterov step
        alpha = 2.0 / (it + 2)
        
        # Projected gradient step
        y_new = y - alpha * 0.5 * g
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y
        else:
            y_new = y_new / s_y
        
        # Nesterov averaging
        a_new = ((it + 1) * a + alpha * (it + 2) * y_new) / (it + 1 + alpha * (it + 2))
        a_new = np.maximum(a_new, 0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new
        else:
            a_new = a_new / s_a
        
        s_new = get_score(a_new)
        if s_new < best_score:
            best_score = s_new
            best_a = a_new.copy()
        
        s_y_new = get_score(y_new)
        if s_y_new < best_score:
            best_score = s_y_new
            best_a = y_new.copy()
        
        a = a_new
        y = y_new
    
    return best_a
