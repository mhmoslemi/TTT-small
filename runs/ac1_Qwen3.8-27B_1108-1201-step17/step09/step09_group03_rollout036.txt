        a_new = np.exp(x_shifted)
        a_new = a_new / a_new.sum()
        a = a_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimization on log-sum-exp smooth approximation (log-domain softmax parameterization)
2. LP-based exact best response alternating for verification
3. Multi-peak mirror descent as complementary approach
4. Coordinate descent for final polishing
5. Diverse multi-start with temperature annealing
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_cross_corr(a, w):
    """
    Compute g[i] = sum_j a[j] * w[i+j] for i in [0, n-1].
    a has length n, w has length 2n-1.
    Returns array of length n.
    """
    n = len(a)
    w_flipped = w[::-1]
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    WF = rfft(w_flipped, fft_size)
    conv_res = irfft(A * WF, fft_size)[:size]
    g = conv_res[2 * n - 2 : n - 2 : -1]
    return g


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def score_normalized(a):
    """Compute the score for a normalized (sum=1) non-negative sequence."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def adam_smooth_optimize(a_init, time_limit, beta_start=30.0, beta_end=300.0,
                         lr=0.005, verbose=True, label=""):
    """
    Adam optimization on log-sum-exp smooth approximation.
    Uses softmax (log-domain) parameterization.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Log-domain: x = log(a), a = softmax(x)
    a_safe = np.maximum(a, 1e-30)
    x = np.log(a_safe)
    x = x - x.mean()  # center for stability

    # Adam hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    no_improve = 0
    patience = 2000

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(t / (time_limit * 80.0), 1.0)  # estimate ~80 iters/sec

        # Adaptive temperature: start smooth, end sharp
        beta = beta_start + (beta_end - beta_start) * min(progress * 1.5, 1.0)
        beta = min(beta, 500.0)

        # Compute autocorrelation
        c = fast_convolve(a)
        M = float(c.max())
        cur_score = 2.0 * n * M

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            # Increase beta more aggressively or break
            if beta >= 250:
                break
            beta = min(beta * 1.5, 500.0)
            no_improve = 0
            patience = int(patience * 0.7)

        # Compute softmax weights (numerically stable)
        # w_k = exp(beta * (c_k - M) / M) / sum(...)
        scaled = beta * (c - M) / max(M, 1e-30)
        # scaled is <= 0, max is 0
        exp_scaled = np.exp(scaled)
        Z = float(exp_scaled.sum())
        if Z < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = exp_scaled / Z

        # Compute gradient: grad_a[i] = (4n/Z_raw) * g[i]
        # where g[i] = sum_j a[j] * w[i+j]
        # Note: we use normalized weights w, so the (4n/Z) factor is absorbed
        # The direction is what matters for Adam
        g = compute_cross_corr(a, w)
        grad_a = g  # direction (scale doesn't matter for Adam)

        # Project to tangent space: grad_x = a * (grad_a - dot(a, grad_a))
        mean_grad = float(np.dot(a, grad_a))
        grad_x = a * (grad_a - mean_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * grad_x * grad_x
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Update a from x (softmax)
        x_shifted = x - x.max()
        a_new = np.exp(x_shifted)
        s_new = a_new.sum()
        if s_new < 1e-30:
            a = np.ones(n) / n
        else:
            a = a_new / s_new

        if verbose and t % 2000 == 0:
            print(f"    Adam {label} t={t} ({elapsed:.0f}s): best={best_score:.8f} beta={beta:.1f}")

    return best_a, best_score


def mirror_descent_optimize(a_init, time_limit, lr_init=0.02, top_k=50, verbose=True, label=""):
    """
    Mirror descent (multiplicative weights) with multi-peak subgradient.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = float(conv.max())
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 1000:
            lr *= 0.85
            no_improve = 0
            if lr < 1e-10:
                break

        # Multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        if k_act == 2 * n - 1:
            peaks = np.arange(2 * n - 1)
            pv = conv
        else:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
            pv = conv[peaks]

        temp = 0.01 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Mirror descent step
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a = np.ones(n) / n
        else:
            a = a_new / sn

        conv = fast_convolve(a)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {label} it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def lp_alternating(a_init, time_limit, verbose=True, label=""):
    """
    LP-based alternating: solve exact best response LP, iterate.
    min M s.t. T @ b <= M * 1, sum(b) = 1, b >= 0
    """
    try:
        import cvxpy as cp
    except ImportError:
        return a_init, score_normalized(a_init)

    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Build T matrix: T[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    # T has shape (2n-1, n)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [
            T @ b_var <= M_var,
            cp.sum(b_var) == 1
        ]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve()
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                try:
                    prob.solve(solver=cp.SCS)
                except Exception:
                    break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s_b = b_val.sum()
        if s_b < 1e-15:
            break
        b_val = b_val / s_b

        score_b = score_normalized(b_val)
        if score_b < best_score:
            best_score = score_b
            best_a = b_val.copy()

        # Damped update
        alpha = 2.0 / (it + 2.0)
        a_new = (1.0 - alpha) * a + alpha * b_val
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score_new = score_normalized(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new

        # Rebuild T
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = a

        if verbose and it % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP {label} it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coord_descent_polish(a_init, time_limit, verbose=True, label=""):
    """
    Coordinate descent with adaptive steps for final polishing.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]

            best_delta = 0.0
            best_new_M = M

            factors = [0.95, 0.9, 0.8, 0.7, 0.5, 1.1, 1.2, 1.3, 1.5]
            for factor in factors:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv[k_start:k_end + 1].copy() + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-16:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-16:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 5:
            break
        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

    return best_a, best_score


def zadoff_chu_init(n, rng):
    """Initialize inspired by Zadoff-Chu sequences (low autocorrelation)."""
    # Use a chirp-like sequence
    m = rng.integers(1, n // 2)
    k = np.arange(n)
    phase = np.pi * m * k * (k + 1) / n
    # Take real part and make non-negative
    a = (1.0 + np.cos(phase)) / 2.0
    a = np.maximum(a, 0.0)
    # Add some randomization
    a += rng.uniform(0, 0.1, n)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def fourier_init(n, rng):
    """Random Fourier-based initialization."""
    # Random spectrum with low-pass characteristic
    num_freqs = n // 2 + 1
    spectrum = rng.normal(0, 1, num_freqs)
    # Low-pass filter
    cutoff = rng.uniform(0.2, 0.8)
    freqs = np.linspace(0, 1, num_freqs)
    mask = np.exp(-freqs / cutoff)
    spectrum *= mask
    # Make real and non-negative
    a = irfft(spectrum, n)
    a = (a - a.min())  # shift to non-negative
    a = np.maximum(a, 0.0)
    # Add small random component
    a += rng.uniform(0, 0.05, n)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def pulse_train_init(n, rng):
    """Regularly spaced pulse train with Gaussian envelopes."""
    num_pulses = rng.integers(3, 15)
    positions = np.linspace(0, n, num_pulses + 2)[1:-1]
    width = rng.uniform(0.02, 0.1) * n
    a = np.zeros(n)
    t = np.arange(n)
    for pos in positions:
        a += np.exp(-0.5 * ((t - pos) / width) ** 2)
    a += rng.uniform(0.01, 0.05, n)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    1. Adam on log-sum-exp smooth approximation (primary)
    2. Mirror descent (complementary)
    3. LP alternating (exact best response)
    4. Coordinate descent polish
    5. Fine polishing
    """
    deadline = time.monotonic() + 1750.0
    rng = np.random.default_rng(12345)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    best_seq = None
    best_score = float('inf')

    if parent is not None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n_init = 1024
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Adam smooth optimization from multiple inits ===
    print(f"\n=== Phase 1: Adam Smooth Optimization (best={best_score:.8f}) ===")
    t1_start = time.monotonic()

    # Build diverse initializations
    inits = [best_seq.copy()]

    # Perturbed parent
    for noise_level in [0.02, 0.05, 0.1]:
        noisy = best_seq + rng.standard_normal(n_best) * noise_level
        noisy = np.maximum(noisy, 0)
        s = noisy.sum()
        if s > 1e-10:
            inits.append(noisy / s)

    # Diverse structured inits
    inits.append(zadoff_chu_init(n_best, rng))
    inits.append(fourier_init(n_best, rng))
    inits.append(pulse_train_init(n_best, rng))

    # Scalloped (alternating high/low)
    period = rng.integers(3, 7)
    scalp = 0.5 + 0.3 * np.cos(2 * np.pi * np.arange(n_best) / period)
    scalp = np.maximum(scalp, 0.01)
    scalp = scalp / scalp.sum()
    inits.append(scalp)

    # Two-scale: mix of broadband and narrowband
    broadband = np.ones(n_best) / n_best
    narrow = np.zeros(n_best)
    center = n_best // 2
    width = n_best // 8
    t_arr = np.arange(n_best)
    narrow = np.exp(-0.5 * ((t_arr - center) / width) ** 2)
    mix = 0.3 * broadband + 0.7 * narrow
    mix = mix / mix.sum()
    inits.append(mix)

    best_adam_score = best_score
    best_adam_seq = best_seq.copy()

    for idx, init in enumerate(inits):
        remaining = deadline - time.monotonic() - 600
        if remaining < 60:
            break

        if idx == 0:
            t_alloc = min(remaining * 0.3, 400.0)
            beta_s, beta_e = 50.0, 350.0
            lr = 0.005
        elif idx <= 3:
            t_alloc = min(remaining * 0.15, 150.0)
            beta_s, beta_e = 30.0, 200.0
            lr = 0.008
        else:
            t_alloc = min(remaining * 0.1, 100.0)
            beta_s, beta_e = 20.0, 150.0
            lr = 0.01

        print(f"  Adam init {idx}, budget={t_alloc:.0f}s, beta=[{beta_s},{beta_e}], lr={lr}")
        a_adam, s_adam = adam_smooth_optimize(init, t_alloc, beta_start=beta_s,
                                              beta_end=beta_e, lr=lr, verbose=True,
                                              label=f"init{idx}")
        if s_adam < best_adam_score:
            best_adam_score = s_adam
            best_adam_seq = a_adam.copy()
        best_seq, best_score = check_and_update(a_adam, best_seq, best_score, f"adam_{idx}")

    print(f"  Phase 1 done: best={best_score:.8f} ({time.monotonic()-t1_start:.0f}s)")

    # === Phase 2: Refine best with high-beta Adam ===
    print(f"\n=== Phase 2: High-Beta Refinement (best={best_score:.8f}) ===")
    remaining = deadline - time.monotonic() - 400
    if remaining > 60:
        t_refine = min(remaining * 0.5, 300.0)
        print(f"  Refine budget={t_refine:.0f}s")
        a_ref, s_ref = adam_smooth_optimize(best_seq, t_refine,
                                            beta_start=100.0, beta_end=500.0,
                                            lr=0.002, verbose=True, label="refine")
        best_seq, best_score = check_and_update(a_ref, best_seq, best_score, "refine")

    # === Phase 3: Mirror Descent ===
    print(f"\n=== Phase 3: Mirror Descent (best={best_score:.8f}) ===")
    remaining = deadline - time.monotonic() - 300
    if remaining > 60:
        t_md = min(remaining * 0.4, 200.0)
        print(f"  MD budget={t_md:.0f}s")
        a_md, s_md = mirror_descent_optimize(best_seq, t_md, lr_init=0.01,
                                             top_k=50, verbose=True, label="main")
        best_seq, best_score = check_and_update(a_md, best_seq, best_score, "mirror_descent")

        # Also try from adam result if different
        if best_adam_score < best_score + 0.005 and best_adam_seq is not None:
            remaining = deadline - time.monotonic() - 250
            if remaining > 30:
                a_md2, s_md2 = mirror_descent_optimize(best_adam_seq, min(remaining * 0.3, 80.0),
                                                       lr_init=0.01, top_k=30, verbose=False, label="adam")
                best_seq, best_score = check_and_update(a_md2, best_seq, best_score, "md_adam")

    # === Phase 4: LP Alternating ===
    print(f"\n=== Phase 4: LP Alternating (best={best_score:.8f}) ===")
    remaining = deadline - time.monotonic() - 200
    if remaining > 60:
        t_lp = min(remaining * 0.5, 120.0)
        print(f"  LP budget={t_lp:.0f}s")
        try:
            a_lp, s_lp = lp_alternating(best_seq, t_lp, verbose=True, label="main")
            best_seq, best_score = check_and_update(a_lp, best_seq, best_score, "lp_main")
        except Exception as e:
            print(f"  LP failed: {e}")

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    remaining = deadline - time.monotonic() - 100
    if remaining > 60:
        t_cd = min(remaining * 0.5, 150.0)
        print(f"  CD budget={t_cd:.0f}s")
        a_cd, s_cd = coord_descent_polish(best_seq, t_cd, verbose=True, label="main")
        best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "cd_main")

    # === Phase 6: Fine Polish ===
    print(f"\n=== Phase 6: Fine Polish (best={best_score:.8f}) ===")
    remaining = deadline - time.monotonic() - 30
    if remaining > 20:
        t_fine = min(remaining * 0.7, 80.0)
        print(f"  FineCD budget={t_fine:.0f}s")
        a_fine, s_fine = fine_cd_polish(best_seq, t_fine, verbose=True, label="fine")
        best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

    # === Phase 7: Final Adam pass on the polished result ===
    remaining = deadline - time.monotonic() - 10
    if remaining > 20:
        t_final = min(remaining * 0.8, 40.0)
        a_fin, s_fin = adam_smooth_optimize(best_seq, t_final,
                                            beta_start=200.0, beta_end=500.0,
                                            lr=0.001, verbose=False, label="final")
        best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_adam")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification with evaluate_sequence
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        print(f"\nVERIFIED FINAL SCORE: {final_score:.8f} (n={len(best_seq)})")
    else:
        print(f"\nWARNING: Final verification failed, using computed score: {best_score:.8f}")

    return [float(x) for x in best_seq]
