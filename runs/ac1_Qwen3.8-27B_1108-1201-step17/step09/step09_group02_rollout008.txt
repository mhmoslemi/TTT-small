def fast_convolve(a):

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def golden_section_search(f, lo, hi, max_iter=40):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)

def golden_section_search(f, lo, hi, max_iter=35):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def init_flat_spectrum(n, rng):

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def spectral_shape(a, rng, n_flatten=20):
    """
    Frequency-domain spectral shaping:
    - Identify the largest magnitude frequency bins
    - Reduce their magnitude to flatten the spectrum
    - Inverse FFT and project to non-negative cone
    This can escape local minima by making coordinated changes.
    """
    n = len(a)
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)

    # Reduce top n_flatten magnitudes (excluding DC)
    if len(mag) > n_flatten + 1:
        top_idx = np.argsort(mag[1:])[::-1][:n_flatten] + 1
        for idx in top_idx:
            mag[idx] *= 0.7

    # Reconstruct
    A_new = mag * np.exp(1j * phase)
    a_new = irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

        # Partner indices for this lag
        partners_idx = ki - idx
        # Contribution pairs: a[j] * a[k-j]
        # Reduce the larger, increase the smaller
        n_perturb = max(2, len(idx) // 4)
        if len(idx) < 4:
            n_perturb = max(1, len(idx) // 2)
        picks = rng.choice(idx, size=min(n_perturb, len(idx)), replace=False)
        for ii in picks:
            partner = ki - ii
            if 0 <= partner < n:
                # Anti-correlated perturbation
                if a_new[ii] > a_new[partner]:
                    a_new[ii] *= (1.0 - strength * rng.random())
                    a_new[partner] *= (1.0 + strength * rng.random())
                else:
                    a_new[partner] *= (1.0 - strength * rng.random())
                    a_new[ii] *= (1.0 + strength * rng.random())

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def lp_guided_cd(a0, time_limit, solver, verbose=False, label=""):
    """
    Hybrid LP-guided coordinate descent:
    Alternates between coordinate descent passes and LP direction finding.
    After each CD pass, solve the LP to get a global direction,
    then do a line search along that direction before continuing CD.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    cd_passes = 0

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break

        # CD pass
        cd_budget = min(rem * 0.6, 120.0)
        cd_budget = max(cd_budget, 10.0)
        a_cd, s_cd, _ = exact_coord_descent(a, cd_budget, label=f"lp_cd_{label}")
        if s_cd < best_score:
            best_score = s_cd
            best_a = a_cd.copy()
        a = a_cd.copy()
        cd_passes += 1

        # LP direction
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break
        t_budget = min(rem * 0.3, 60.0)

        b_lp, M_val = solve_lp_best_response(a, solver)
        if b_lp is None:
            continue

        # Line search: a_new = (1-t)*a + t*b_lp
        best_t = 0.0
        best_ls = best_score
        for trial_t in [0.15, 0.25, 0.4, 0.55, 0.7, 0.85, 1.0]:
            cand = (1.0 - trial_t) * a + trial_t * b_lp
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_ls:
                best_ls = sc
                best_t = trial_t

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * b_lp
            a_new = np.maximum(a_new, 0)
            cs = a_new.sum()
            if cs < 1e-10:
                continue
            a_new = a_new / cs
            score_new = 2.0 * n * float(fast_convolve(a_new).max())
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
            a = a_new
        else:
            # Use averaging as fallback
            alpha = 2.0 / (cd_passes + 2.0)
            a = (1.0 - alpha) * a + alpha * b_lp
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                continue
            a = a / cs

        if verbose and cd_passes % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP+CD {label} pass={cd_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def basin_hopping_advanced(a0, time_limit, rng, solver, verbose=False):
    """
    Advanced basin hopping with multiple perturbation strategies:
    1. Random Gaussian noise
    2. Anti-peak targeted perturbation
    3. Spectral shaping
    4. LP-guided perturbation
    After each perturbation, do quick CD to converge.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    p = 0

    while time.monotonic() - t0 < time_limit:
        p += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 25:
            break
        t_alloc = min(rem * 0.5, 45.0)
        t_alloc = max(t_alloc, 8.0)

        # Choose perturbation strategy based on iteration
        strategy = p % 4

        if strategy == 0:
            # Random Gaussian
            sigma = 0.02 + 0.04 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif strategy == 1:
            # Anti-peak
            conv_p = fast_convolve(a)
            perturbed = anti_peak_perturb(a, conv_p, rng, strength=0.04 + 0.04 * rng.random())
        elif strategy == 2:
            # Spectral shaping
            perturbed = spectral_shape(a, rng, n_flatten=int(n * 0.05) + 5)
            # Mix with original to avoid too-large jumps
            alpha = 0.5 + 0.3 * rng.random()
            perturbed = alpha * a + (1.0 - alpha) * perturbed
        else:
            # LP-guided
            b_lp, _ = solve_lp_best_response(a, solver)
            if b_lp is None:
                perturbed = a + rng.standard_normal(n) * 0.03
            else:
                alpha = rng.uniform(0.3, 0.8)
                perturbed = alpha * a + (1.0 - alpha) * b_lp

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        # Quick CD on perturbed
        a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc, label=f"bh{p}")
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p} (strat={strategy}): improved to {best_score:.8f}")
            a = a_opt.copy()
        else:
            # Occasionally update a to explore even if not improved
            if rng.random() < 0.3:
                a = perturbed.copy()

    return best_a, best_score

def subgradient_descent(a0, time_limit, verbose=False):
    """Subgradient descent with adaptive peak weighting."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s

    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    lr = 0.01
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Identify active peaks
        peak_val = float(conv.max())
        n_peaks = min(20, 2 * n - 1)
        peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
        pv = conv[peaks]

        # Softmax weighting (hotter = more weight)
        temp = 0.01 * max(peak_val, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            break
        w = w / ws

        # Compute subgradient
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[idx] * a[k - j]

        # Step with projection
        a_new = simplex_project(a - lr * grad)
        conv = fast_convolve(a_new)
        sc = 2.0 * n * float(conv.max())
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.8
                if lr < 1e-6:
                    break
        lr *= 0.998

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG it={it} ({elapsed:.0f}s): best={best_score:.8f}, lr={lr:.6f}")

    return best_a, best_score

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

def propose_candidate():
    """
    Multi-phase optimization combining:
    - Peak-directed coordinate descent
    - LP-guided hybrid optimization
    - Advanced basin hopping (anti-peak, spectral, LP perturbations)
    - Subgradient descent
    - Final fine polishing
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(12345)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Quick exploration with diverse initializations ===
    print("\n=== Phase 1: Diverse Init + Quick ECD ===")
    t1_start = time.monotonic()

    n_values = [1024, 768, 1280, 1536]
    if parent is not None:
        np_ = len(parent)
        n_values.insert(0, np_)
        for delta in [-128, 128, -256, 256]:
            alt = np_ + delta
            if 256 <= alt <= 2048 and alt not in n_values:
                n_values.append(alt)

    candidates = []

    for n in n_values[:5]:
        elapsed = time.monotonic() - t1_start
        if elapsed > 150:
            break

        inits = []
        if parent is not None:
            inits.append(('parent', interpolate_seq(parent, n)))
        inits.append(('flat', init_flat_spectrum(n, rng)))
        inits.append(('asym', init_asymmetric_bell(n, rng)))
        inits.append(('2bump', init_two_bump(n, rng)))
        inits.append(('fourier', init_fourier_mix(n, rng)))
        inits.append(('flat2', init_flat_spectrum(n, rng)))

        for itype, cand in inits[:5]:
            t_left = 150.0 - (time.monotonic() - t1_start)
            if t_left < 4:
                break
            t_alloc = max(3.0, min(15.0, t_left / 7.0))
            a_opt, s_opt, _ = exact_coord_descent(cand, t_alloc, label=f"p1_n{n}_{itype}")
            candidates.append((s_opt, a_opt, n, itype))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, best: {candidates[0][0]:.8f}")

    for s_c, a_c, n_c, it_c in candidates[:4]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}_{it_c}")

    # === Phase 2: Intensive optimization of best candidate ===
    print(f"\n=== Phase 2: Intensive ECD (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        current = best_seq.copy()

        for rnd in range(3):
            rem = deadline - time.monotonic() - 600
            if rem < 30:
                break
            t_alloc = min(rem * 0.35, 180.0)
            print(f"  ECD round {rnd+1}, n={n_best}, budget={t_alloc:.0f}s")
            a_opt, s_opt, sweeps = exact_coord_descent(current, t_alloc,
                                                       verbose=True, label=f"ph2_r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ph2_ecd_r{rnd}")

            if best_score > 1.5040:
                # Perturb to escape
                perturbed = a_opt + rng.standard_normal(n_best) * 0.03
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
                current = perturbed
            else:
                current = a_opt.copy()

    # === Phase 3: LP-guided hybrid optimization ===
    print(f"\n=== Phase 3: LP+CD Hybrid (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 60:
            t_lp = min(rem * 0.5, 250.0)
            print(f"  LP+CD n={len(best_seq)}, budget={t_lp:.0f}s")
            a_lp, s_lp, _ = lp_guided_cd(best_seq, t_lp, solver, verbose=True, label="main")
            best_seq, best_score = check_and_update(a_lp, best_seq, best_score, "lp_cd")

            # Follow up with CD
            rem = deadline - time.monotonic() - 250
            if rem > 20:
                t_cd = min(rem * 0.5, 80.0)
                a_cd, s_cd, _ = exact_coord_descent(a_lp, t_cd, label="post_lp")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "ecd_after_lp")

    # === Phase 4: Advanced Basin Hopping ===
    print(f"\n=== Phase 4: Basin Hopping (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 40:
            t_bh = min(rem * 0.65, 250.0)
            print(f"  BH n={len(best_seq)}, budget={t_bh:.0f}s")
            a_bh, s_bh = basin_hopping_advanced(best_seq, t_bh, rng, solver, verbose=True)
            best_seq, best_score = check_and_update(a_bh, best_seq, best_score, "basin_hop")

    # === Phase 5: Subgradient refinement ===
    print(f"\n=== Phase 5: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_sg = min(rem * 0.6, 100.0)
            print(f"  Subgradient, budget={t_sg:.0f}s")
            a_sg, s_sg = subgradient_descent(best_seq, t_sg, verbose=True)
            best_seq, best_score = check_and_update(a_sg, best_seq, best_score, "subgrad")

            # CD after subgradient
            rem = deadline - time.monotonic() - 80
            if rem > 15:
                a_cd, s_cd, _ = exact_coord_descent(a_sg, min(rem * 0.6, 50.0), label="post_sg")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "ecd_post_sg")

    # === Phase 6: Try different length ===
    print(f"\n=== Phase 6: Length exploration (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            for alt_n in [len(best_seq) + 128, len(best_seq) - 128]:
                if rem < 20:
                    break
                if 256 <= alt_n <= 2048:
                    init = interpolate_seq(best_seq, alt_n)
                    t_alloc = min(rem * 0.4, 40.0)
                    a_alt, s_alt, _ = exact_coord_descent(init, t_alloc, label=f"len{alt_n}")
                    best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"len_{alt_n}")
                    rem = deadline - time.monotonic() - 100

    # === Phase 7: Final polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Fine coordinate descent
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_fine = min(rem * 0.6, 80.0)
            print(f"  FineCD, budget={t_fine:.0f}s")
            a_fine, s_fine, _ = fine_coord_descent(best_seq, t_fine,
                                                   verbose=True, label="final")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

        # Another exact CD round
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_cd2, s_cd2, _ = exact_coord_descent(best_seq, min(rem * 0.7, 40.0), label="last_cd")
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "last_cd")

        # Final fine pass
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            a_fin, s_fin, _ = fine_coord_descent(best_seq, min(rem * 0.8, 20.0), label="very_last")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "very_last")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
