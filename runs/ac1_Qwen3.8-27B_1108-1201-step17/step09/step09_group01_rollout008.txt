def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def cross_correlate(a, v):
    """Compute (a star v)[j] = sum_i a[i] * v[i+j] for j = 0,...,n-1."""
    n = len(a)
    nv = len(v)
    # cross-correlation: result[j] = sum_i a[i] * v[i+j]
    # This is like a flipped cross-correlation
    a_flip = a[::-1]
    fft_size = 1 << (n + nv - 1 - 1).bit_length()
    A = rfft(a_flip, fft_size)
    V = rfft(v, fft_size)
    result = irfft(A * V, fft_size)
    # result[k] = sum_i a_flip[i] * v[k-i] = sum_i a[n-1-i] * v[k-i]
    # Let j = n-1-i, so i = n-1-j: result[k] = sum_j a[j] * v[k-n+1+j]
    # We want c[j] = sum_i a[i] * v[i+j] = sum_i a[i] * v[(i+j)]
    # From result: result[k] = sum_i a[n-1-i] * v[k-i], let m = n-1-i, i = n-1-m
    # result[k] = sum_m a[m] * v[k-n+1+m]
    # So c[j] = sum_i a[i] * v[i+j] = result[j + n - 1]
    start = n - 1
    end = start + n
    if end > len(result):
        c = np.zeros(n)
        c[:len(result)-start] = result[start:len(result)]
        return c
    return result[start:end]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if '_lp_solver_cached' in globals():
        try:

                return _lp_solver_cached
        except Exception:

    _lp_solver_cached = None

    for solver_name in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP, cp.GLOP]:
        try:

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None, max_constraints=500):
    """
    Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Uses reduced constraints for speed.
    """
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)

    # Use all constraints (needed for correctness)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def dual_subgradient_best_response(a, n_iters=2000, step_init=0.1):
    """
    Use dual subgradient method to approximate the best response b to a.
    Faster than solving the full LP, gives approximate solution.
    """
    n = len(a)
    # Dual: max_v min_j (a star v)[j] s.t. v >= 0, sum(v) = 1
    # v is a probability distribution over 2n-1 positions
    v = np.ones(2 * n - 1) / (2 * n - 1)
    step = step_init / (n ** 0.5)

    for it in range(n_iters):
        # Compute cross-correlation c[j] = sum_i a[i] * v[i+j] for j = 0,...,n-1
        c = cross_correlate(a, v)
        j_star = int(np.argmin(c))

        # Subgradient: shift(a, j_star) -> g[k] = a[k - j_star] if valid
        g = np.zeros(2 * n - 1)
        k_start = j_star
        k_end = min(j_star + n, 2 * n - 1)
        g[k_start:k_end] = a[:k_end - k_start]

        # Update v (maximizing concave function)
        step_t = step / (np.sqrt(it + 1))
        v = simplex_project(v + step_t * g)

    # From dual solution v, extract primal solution
    # The primal b is proportional to: for each j, b[j] is "active" where c[j] is at minimum
    # Actually, use complementary slackness: b[j] > 0 only if c[j] = lambda*
    c_final = cross_correlate(a, v)
    lambda_star = float(c_final.min())
    tol = 0.01 * (c_final.max() - c_final.min() + 1e-15)
    active = c_final >= lambda_star - tol
    if active.sum() == 0:
        active = np.ones(n, dtype=bool)
    b = np.zeros(n)
    b[active] = 1.0
    b = b / b.sum()
    return b


def fast_coord_descent(a0, time_limit, n_trials=10, verbose=False, label=""):
    """
    Fast coordinate descent with vectorized peak tracking.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_count = 0

    # Precompute factors for step sizes
    factors = np.array([0.99, 0.98, 0.97, 0.96, 0.95,
                        1.01, 1.02, 1.03, 1.04, 1.05,
                        0.9, 0.85, 1.1, 1.2, 0.7, 1.5])

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
                if unaff_right > unaff_max:
                    unaff_max = unaff_right

            affected = conv[k_start:k_end + 1].copy()
            a_shifted = a[0:n]
            current_2i_val = conv[2 * i]
            in_affected = (2 * i >= k_start) and (2 * i <= k_end)
            idx_2i = 2 * i - k_start if in_affected else -1

            best_delta = 0.0
            best_new_max = M

            for factor in factors:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                if new_val > 1.0:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if in_affected:
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = current_2i_val + 2.0 * delta * a[i] + delta * delta
                else:
                    new_affected = new_affected.copy()

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a += best_delta

                changes = 2.0 * best_delta * a_shifted.copy()
                if in_affected:
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_count += 1
            if no_improve_count > 50:
                break
        else:
            no_improve_count = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f} M={M:.8f}")

    return best_a, best_score, sweeps

def peak_targeted_perturb(a0, rng, strength=0.03):
    """Structured perturbation targeting peak regions."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a0)
    M = conv.max()

    peak_threshold = M * 0.92
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            product = a0[i] * a0[j]
            pressure[i] += product
            if j != i:
                pressure[j] += product * 0.5

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.3 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_symmetric(n, rng):
    """Create a symmetric initialization."""
    half = n // 2
    t = np.linspace(0, 1, half)
    n_comp = int(rng.integers(2, 8))
    profile = np.ones(half)
    for j in range(n_comp):
        freq = rng.uniform(0.5, 4.0)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0)

    if n % 2 == 0:
        a = np.concatenate([profile, profile[::-1]])
    else:
        mid = max(profile[-1], profile[0]) * rng.uniform(0.5, 1.5)
        a = np.concatenate([profile, [mid], profile[::-1]])

    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_bimodal(n, rng):
    """Initialize with bimodal distribution."""
    center1 = n * rng.uniform(0.2, 0.4)
    center2 = n * rng.uniform(0.6, 0.8)
    width1 = n * rng.uniform(0.05, 0.15)
    width2 = n * rng.uniform(0.05, 0.15)
    t = np.arange(n)
    a = np.exp(-0.5 * ((t - center1) / width1) ** 2) + np.exp(-0.5 * ((t - center2) / width2) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_random_sparse(n, rng, frac=0.5):

    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def lp_fictitious_play(n, time_limit, initial=None, solver=None, verbose=True):
    """
    LP-based fictitious play with Nesterov-style averaging.
    This is the main optimization workhorse.
    """
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()

    # Nesterov/Averaged approximation: track running averages
    a_avg = a.copy()
    b_avg = b.copy()

    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    iters = 0

    # Track all iterates for potential use
    all_iterates = []

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # Solve LP: find best response b to a
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new

        # Solve LP: find best response a to b
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        # Nesterov-style averaging with step size 2/(t+2)
        gamma_it = 2.0 / (iters + 2.0)
        a_avg = (1.0 - gamma_it) * a_avg + gamma_it * a
        b_avg = (1.0 - gamma_it) * b_avg + gamma_it * b

        # Also try simple running average
        simple_a_avg = a_avg.copy()

        # Evaluate candidates periodically
        if iters % 3 == 0 or iters <= 10:
            candidates = [a, b, a_avg, b_avg, 0.5 * (a_avg + b_avg)]
            for cand in candidates:
                cand = np.maximum(cand, 0)
                sc = cand.sum()
                if sc < 1e-10:
                    continue
                cand_norm = cand / sc
                score = 2.0 * n * fast_convolve(cand_norm).max()
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = cand_norm.copy()

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f} M_a={M_a:.6f} M_b={M_b:.6f}")

    # Final evaluation
    final_score = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    return best_a, best_score, iters


def nesterov_mirror_descent(a0, time_limit, lr_init=0.005, beta_init=50.0):
    """
    Nesterov-accelerated mirror descent with smooth max approximation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    y_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    beta = beta_init
    no_improve = 0
    gamma = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov extrapolation
        if it == 1:
            y = a.copy()
            y_prev = a.copy()
        else:
            y = (1.0 + gamma) * a - gamma * y_prev
            y_prev = y

        y_proj = simplex_project(y)

        # Compute smooth gradient
        conv_y = fast_convolve(y_proj)
        max_conv = conv_y.max()
        scaled = beta * (conv_y - max_conv)
        scaled = np.clip(scaled, -50, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[conv_y.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z

        # Gradient
        grad = np.zeros(n)
        peak_indices = np.where(weights > 1e-10)[0]
        for k in peak_indices:
            w = weights[k]
            if w < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w * y_proj[k - j]

        # Mirror descent update
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        a_new = np.exp(log_a - log_a.max())
        a = simplex_project(a_new)

        # Update best
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive params
        if no_improve > 50:
            lr *= 0.9
            beta *= 1.5
            no_improve = 0
            if lr < 1e-9:
                break
        elif it % 30 == 0:
            beta *= 1.03

        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization combining LP fictitious play,
    mirror descent, and coordinate descent for minimizing
    the auto-convolution peak ratio.
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(12345)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Diverse initialization + Fast CD (60s) ===
    print("\n=== Phase 1: Diverse Init + Fast CD ===")
    t1_start = time.monotonic()

    n_target = 1024
    if parent is not None:
        n_target = len(parent)

    n_values = [n_target]
    for alt in [896, 1024, 1152, 1280]:
        if alt != n_target and 512 <= alt <= 1536:
            n_values.append(alt)

    candidates = []

    for n in n_values[:4]:
        elapsed = time.monotonic() - t1_start
        if elapsed > 50:
            break

        init_cands = []
        if parent is not None:
            init_cands.append(("parent", interpolate_seq(parent, n)))
        init_cands.append(("symmetric", init_symmetric(n, rng)))
        init_cands.append(("bimodal", init_bimodal(n, rng)))
        init_cands.append(("random", init_random_sparse(n, rng, 0.6)))
        init_cands.append(("uniform", np.ones(n) / n))

        for name, cand in init_cands:
            t_left = 55.0 - (time.monotonic() - t1_start)
            if t_left < 2:
                break
            t_alloc = max(1.0, min(4.0, t_left / 2.0))
            a_opt, s_opt, _ = fast_coord_descent(cand, t_alloc, n_trials=8, label=f"p1_{name}_{n}")
            candidates.append((s_opt, a_opt, n, name))

    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates, top-3: " +
          ", ".join([f"{c[0]:.6f}(n={c[2]},{c[3]})" for c in candidates[:3]]))

    for s_c, a_c, n_c, it_c in candidates[:6]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{it_c}_n{n_c}")

    print(f"  After Phase 1: best={best_score:.6f}")

    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.6f}) ===")
    t2_start = time.monotonic()

    if best_seq is not None:
        n_fp = len(best_seq)
