```python
import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def compute_gradient(a, conv, beta):
    """Compute gradient of soft-max objective."""
    n = len(a)
    # Softmax weights
    conv_clipped = np.clip(conv, -1e10, 1e10)
    log_w = conv_clipped * beta - np.max(conv_clipped * beta)
    p = np.exp(log_w)
    p /= p.sum()
    
    # Gradient: grad[i] = sum_k p[k] * 2 * a[k-i] (valid range)
    grad = np.zeros(n)
    for k in range(len(conv)):
        if p[k] < 1e-14:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # a[k-i] for i in [i_start, i_end]
        # k-i goes from k-i_start down to k-i_end
        grad[i_start:i_end+1] += 2.0 * p[k] * a[k-i_end:k-i_start+1][::-1]
    return grad, p

def smooth_optimize(a0, time_budget, beta_start=10.0, beta_end=500.0, 
                    lr_init=0.1, verbose=True):
    """Smooth optimization with annealed soft-max and momentum."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient
    lr = lr_init
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        it += 1
        # Anneal beta
        progress = min(1.0, it / (time_budget / 0.01))  # rough progress estimate
        # Actually, let's use time-based annealing
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute gradient
        grad, p = compute_gradient(a, conv, beta)
        
        # Momentum update
        momentum = mu * momentum + grad
        # Use effective gradient (divide by 1-mu for heavy ball)
        eff_grad = momentum / (1.0 - mu)
        
        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.9
            no_improve = 0
            mu *= 0.99  # slightly reduce momentum
            if lr < 1e-10:
                break
        
        # Gradient step
        a_new = a - lr * eff_grad
        
        # Project to simplex
        a_new = project_simplex(a_new)
        
        # Line search (simple: accept if improvement, otherwise halve step)
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * conv_new.max()
        if score_new > score * (1 + 1e-6):
            lr *= 0.5
            a_new = a - lr * eff_grad
            a_new = project_simplex(a_new)
        
        a = a_new
        
        if verbose and it % 500 == 0:
            print(f"    Smooth opt it={it} ({elapsed:.1f}s): score={score:.6f} beta={beta:.1f} lr={lr:.6f}")
    
    return best_a, best_score
```

"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """FFT-based full auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve_fft(a, b):
    """FFT-based linear convolution of a and b."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def compute_soft_max_gradient(a, conv, beta):
    """
    Compute gradient of h_beta(a) = (1/beta)*log(sum_k exp(beta*conv[k])).
    Uses FFT for O(n log n) computation.
    
    grad[i] = 2 * sum_j a[j] * p[i+j] where p = softmax(beta * conv)
    This equals 2 * (a_rev * p)[n-1+i] where a_rev = a reversed.
    """
    n = len(a)
    bc = conv * beta
    bc_shifted = bc - bc.max()
    p = np.exp(bc_shifted)
    p /= p.sum()
    
    # grad[i] = 2 * (a_rev * p)[n-1+i]
    a_rev = a[::-1]
    # Convolution of a_rev (len n) and p (len 2n-1) has length 3n-2
    size = n + (2 * n - 1) - 1
    fft_size = 1 << (size - 1).bit_length()
    A_rev = rfft(a_rev, fft_size)
    P = rfft(p, fft_size)
    conv_ap = irfft(A_rev * P, fft_size)[:size]
    
    grad = 2.0 * conv_ap[n - 1:n - 1 + n]
    return grad, p


def smooth_optimize(a0, time_budget, beta_start=0.1, beta_end=200.0,
                    lr_init=0.5, mu_init=0.85, verbose=True, label=""):
    """
    Smooth optimization with annealed soft-max, Nesterov momentum,
    and adaptive step size. Operates on the probability simplex.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    velocity = np.zeros(n)
    mu = mu_init
    lr = lr_init
    it = 0
    no_improve = 0
    best_beta_score = best_score
    patience = 0

    while time.monotonic() - t0 < time_budget:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_budget)
        # Anneal beta: start low (smooth), end high (sharp)
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.7)

        conv = fast_convolve(a)
        M = float(conv.max())
        score = 2.0 * n * M

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            patience = 0
        else:
            no_improve += 1
            patience += 1

        # Adaptive learning rate
        if no_improve > 500:
            lr *= 0.8
            no_improve = 0
        if patience > 3000:
            break
        if lr < 1e-12:
            break

        # Nesterov lookahead
        a_lookahead = a + mu * velocity
        conv_la = fast_convolve(a_lookahead)
        grad, p = compute_soft_max_gradient(a_lookahead, conv_la, beta)

        # Velocity update
        velocity = mu * velocity - lr * grad

        # Step
        a_new = a + velocity

        # Project to simplex
        a_new = project_simplex(a_new)

        # Accept/reject (simple backtracking)
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * conv_new.max()
        if score_new <= score * 1.01 + 1e-12:
            a = a_new
        else:
            # Reject step, reset velocity
            velocity *= 0.5
            lr *= 0.9
            a_new = a + velocity
            a_new = project_simplex(a_new)
            a = a_new

        if verbose and it % 2000 == 0:
            print(f"    Smooth[{label}] it={it} ({elapsed:.0f}s): score={score:.6f} beta={beta:.2f} lr={lr:.6f} mu={mu:.3f}")

    return best_a, best_score

    no_improve = 0

    while time.monotonic() - t0 < time_budget:
        sweeps += 1
        improved = False

        factors = [0.995, 0.99, 0.985, 0.98, 1.005, 1.01, 1.015, 1.02]

        for i in range(n):
            if a[i] < 1e-14:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            a_shifted = a[k_start - i:k_end - i + 1]
            best_M = M
            best_t = a[i]

            for f in factors:
                t_new = a[i] * f
                d = t_new - a[i]
                new_sum = s + d
                if new_sum < 1e-10:
                    continue
                scale_sq = 1.0 / (new_sum * new_sum)

                new_aff = conv[k_start:k_end + 1] + 2.0 * d * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] += d * d

                M_test = max(float(new_aff.max()) * scale_sq, unaff_max * scale_sq)
                if M_test < best_M - 1e-15:
                    best_M = M_test
                    best_t = t_new

            if best_t != a[i] and best_M < M - 1e-15:
                a[i] = best_t
                conv = fast_convolve(a)
                M = float(conv.max())
                s = a.sum()
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD[{label}] sweep={sweeps} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def shifted_cosine_init(n, center_frac, width_frac, power, rng):
    """Non-palindromic cosine window initialization."""
    t = np.arange(n, dtype=np.float64)
    center = n * center_frac
    width = n * width_frac
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = a ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def periodic_asymmetric_init(n, rng):
    """Periodic with asymmetric modulation."""
    period = int(rng.integers(3, 15))
    t = np.arange(n, dtype=np.float64)
    # Base periodic
    a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    # Asymmetric envelope
    env = np.exp(-((t / n - 0.35) ** 2) / (2 * 0.15 ** 2)) + 0.3
    a = a * env
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def two_peak_asymmetric_init(n, rng):
    """Two peaks at asymmetric positions."""
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.2, 0.35)
    c2 = n * rng.uniform(0.55, 0.75)
    w1 = n * rng.uniform(0.08, 0.2)
    w2 = n * rng.uniform(0.08, 0.2)
    amp1 = rng.uniform(0.5, 1.0)
    amp2 = rng.uniform(0.3, 0.8)
    a = amp1 * np.exp(-0.5 * ((t - c1) / w1) ** 2) + amp2 * np.exp(-0.5 * ((t - c2) / w2) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def spectrum_flatting_init(n, rng):
    """Initialize by trying to flatten the power spectrum."""
    # Start with random and do a few iterations of spectrum flattening
    a = rng.random(n)
    a = np.maximum(a, 0.1)  # keep all positive
    a = a / a.sum()
    
    for _ in range(50):
        conv = fast_convolve(a)
        M = conv.max()
        # Reduce values at peak locations
        peak_indices = np.argsort(conv)[-5:]
        for k in peak_indices:
            # Reduce the coordinates that contribute most to conv[k]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            contributors = a[i_start:i_end + 1]
            top_c = np.argsort(contributors)[-3:]
            for idx in top_c:
                a[i_start + idx] *= 0.95
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a /= s
    
    return a


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate using the official evaluator and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def fw_alternate(a_init, b_init, time_budget, verbose=False):
    """
    Frank-Wolfe alternating minimization.
    Alternates between: given a, find b that minimizes max(conv(a,b)).
    Uses the LP relaxation solved approximately via subgradient method.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    b = np.maximum(b_init.astype(np.float64), 0.0)
    s = b.sum()
    b = b / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Precompute T matrix columns (shifts of a)
    # T[:, j] = shift(a, j) as a vector of length 2n-1
    # conv(a, b)[k] = sum_j b[j] * a[k-j] (for valid k-j)
    # = sum_j b[j] * (T[:, j])[k]

    for round_num in range(50):
        if time.monotonic() - t0 > time_budget:
            break

        # Subgradient descent on b: minimize max_k (T @ b)[k]
        ab_conv = cross_convolve_fft(a, b)
        for step in range(100):
            k_star = int(np.argmax(ab_conv))
            # Gradient w.r.t. b: grad_b[j] = a[k_star - j] if 0 <= k_star-j <= n-1
            j_start = max(0, k_star - n + 1)
            j_end = min(n - 1, k_star)
            grad_b = np.zeros(n)
            grad_b[j_start:j_end + 1] = a[k_star - j_end:k_star - j_start + 1][::-1]

            # FW step: b_new = (1-t)*b + t*e_{j_min}
            j_star = j_start + int(np.argmin(grad_b[j_start:j_end + 1]))
            t_fw = 2.0 / (step + 2.0)
            b = (1.0 - t_fw) * b
            b[j_star] += t_fw

        # Evaluate
        a_score = 2.0 * n * fast_convolve(a).max()
        if a_score < best_score:
            best_score = a_score
            best_a = a.copy()

        # Update a: set a to b (they should converge to the same fixed point)
        # Actually, alternate: given b, optimize a.
        ba_conv = cross_convolve_fft(a, b)
        for step in range(100):
            k_star = int(np.argmax(ba_conv))
            j_start = max(0, k_star - n + 1)
            j_end = min(n - 1, k_star)
            grad_a = np.zeros(n)
            grad_a[j_start:j_end + 1] = b[k_star - j_end:k_star - j_start + 1][::-1]

            j_star = j_start + int(np.argmin(grad_a[j_start:j_end + 1]))
            t_fw = 2.0 / (step + 2.0)
            a = (1.0 - t_fw) * a
            a[j_star] += t_fw

        a_score = 2.0 * n * fast_convolve(a).max()
        if a_score < best_score:
            best_score = a_score
            best_a = a.copy()

        if verbose and round_num % 5 == 0:
            print(f"    FW round={round_num} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")

    return best_a, best_score
