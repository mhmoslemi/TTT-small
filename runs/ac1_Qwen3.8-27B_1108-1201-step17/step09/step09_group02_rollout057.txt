import numpy as np
from scipy.fft import rfft, irfft
import time

_lp_solver_cached = None


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    import cvxpy as cp
    for solver_name in [cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return cp.ECOS


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.PDLP, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def golden_section_search(f, lo, hi, max_iter=30):
    """Golden section search for minimum of f on [lo, hi]."""
    if hi - lo < 1e-15:
        return (lo + hi) * 0.5
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5


def analytical_coord_opt(i, a, conv, n, s):
    """
    Find optimal x for coordinate i using piecewise analytical approach.
    Returns (best_x, best_score_for_this_coord).
    """
    v = a[i]
    k_lo = i
    k_hi = min(i + n - 1, 2 * n - 2)

    unaff_max = 0.0
    if k_lo > 0:
        um1 = float(conv[:k_lo].max())
        if um1 > unaff_max:
            unaff_max = um1
    if k_hi < 2 * n - 2:
        um2 = float(conv[k_hi + 1:].max())
        if um2 > unaff_max:
            unaff_max = um2

    # Build piece descriptions: (kind, c, m)
    # Linear: val = c + m*x
    # Quadratic: val = c + x^2
    # Constant: val = unaff_max
    pieces = []
    has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
    quad_idx_in_pieces = -1

    for idx in range(k_hi - k_lo + 1):
        k = k_lo + idx
        partner = k - i
        if k == 2 * i:
            c = conv[k] - v * v
            pieces.append(('Q', c, 0.0))
            quad_idx_in_pieces = idx
        else:
            c = conv[k] - 2.0 * v * a[partner]
            m = 2.0 * a[partner]
            pieces.append(('L', c, m))

    n_pieces = len(pieces)

    def score_at(x, _unaff=unaff_max, _pieces=pieces, _n=n, _s=s, _v=v):
        ns = _s + x - _v
        if ns < 0.01:
            return 1e10
        mx = _unaff
        for kind, c, m in _pieces:
            if kind == 'L':
                val = c + m * x
            else:
                val = c + x * x
            if val > mx:
                mx = val
        return 2.0 * _n * mx / (ns * ns)

    # Start from current value
    x_current = v
    best_x = v
    best_sc = score_at(v)

    for _iter in range(8):
        # Find active piece at x_current
        active_kind = 'C'
        active_c = unaff_max
        active_m = 0.0
        active_val = unaff_max

        for idx_p in range(n_pieces):
            kind, c, m = pieces[idx_p]
            if kind == 'L':
                val = c + m * x_current
            else:
                val = c + x_current * x_current
            if val > active_val + 1e-18:
                active_val = val
                active_kind = kind
                active_c = c
                active_m = m

        # Compute candidate optimal x
        candidates = [0.0, x_current]
        if active_kind == 'L' and active_m > 1e-15:
            x_opt = (s - v) - 2.0 * active_c / active_m
            if 0.0 <= x_opt <= max(2.0 * v, 5.0 / n):
                candidates.append(x_opt)
            # Also add nearby points
            candidates.append(max(0.0, x_opt * 0.5))
            candidates.append(x_opt * 1.5 if x_opt > 0 else 0.0)
        elif active_kind == 'Q':
            disc = (s - v) ** 2 + active_c
            if disc > 0:
                x_opt = -(s - v) + np.sqrt(disc)
                if 0.0 <= x_opt <= max(2.0 * v, 5.0 / n):
                    candidates.append(x_opt)
                candidates.append(max(0.0, x_opt * 0.7))
                candidates.append(max(0.0, x_opt * 1.3))
        # Also try x = 0 and a scaled version of current
        candidates.append(0.0)
        if v > 0:
            candidates.append(min(v * 0.5, 2.0 / n))
            candidates.append(min(v * 1.5, 4.0 / n))

        # Evaluate all candidates
        local_best_x = best_x
        local_best_sc = best_sc
        for xc in candidates:
            if xc < 0 or xc > 10.0 / n * 10:
                continue
            sc = score_at(xc)
            if sc < local_best_sc - 1e-16:
                local_best_sc = sc
                local_best_x = xc

        if abs(local_best_x - x_current) < 1e-13 * max(1.0, v):
            best_x = local_best_x
            best_sc = local_best_sc
            break
        x_current = local_best_x
        best_x = local_best_x
        best_sc = local_best_sc

    # Final golden section refinement around best_x
    bracket = max(1e-4, best_x * 0.3, 0.5 / n)
    lo = max(0.0, best_x - bracket)
    hi = best_x + bracket
    if lo < 0:
        lo = 0.0
    if hi - lo > 1e-15:
        x_gs = golden_section_search(score_at, lo, hi, max_iter=12)
        sc_gs = score_at(x_gs)
        if sc_gs < best_sc:
            best_x = x_gs
            best_sc = sc_gs

    return best_x, best_sc


def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent using analytical optimization per coordinate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue

            best_x, _ = analytical_coord_opt(i, a, conv, n, current_sum)

            if abs(best_x - a[i]) > 1e-13:
                v = a[i]
                delta = best_x - v
                k_lo = i
                k_hi = min(i + n - 1, 2 * n - 2)
                partners = np.arange(k_lo, k_hi + 1) - i
                a_partners = a[partners]
                has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    qi = 2 * i - k_lo
                    changes = changes.copy()
                    changes[qi] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score, sweeps

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.6)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_fourier_mix(n, rng):
    """Mix of Fourier modes."""
    n_harmonics = rng.integers(3, 12)
    amps = rng.random(n_harmonics) * 0.3
    phases = rng.uniform(0, 2.0 * np.pi, n_harmonics)
    freqs = rng.choice(np.arange(1, n // 2 + 1), size=n_harmonics, replace=False)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amps[j] * np.cos(2.0 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_zadoff_chu_based(n, rng):
    """Zadoff-Chu inspired initialization (magnitude of complex sequence)."""
    root = rng.integers(1, max(2, n // 4))
    k = rng.integers(1, max(2, n // 4))
    t = np.arange(n, dtype=np.float64)
    # Zadoff-Chu sequence
    zc = np.exp(-1j * np.pi * root * t * (t + 1) / n)
    # Mix with random phase
    random_phase = np.exp(1j * rng.uniform(0, 2 * np.pi, n))
    mix = 0.7 * zc + 0.3 * random_phase
    a = np.abs(mix)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def fictitious_play_with_linesearch(a0, time_limit, solver=None, verbose=False):
    """LP fictitious play with line search."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break

        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            continue

        best_t = 0.0
        best_ls_score = best_score
        for trial_t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_ls_score:
                best_ls_score = sc
                best_t = trial_t

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * b
            a_new = np.maximum(a_new, 0)
            cs = a_new.sum()
            if cs < 1e-10:
                continue
            a_new = a_new / cs
            score_new = 2.0 * n * float(fast_convolve(a_new).max())
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
            a = a_new
        else:
            alpha = 2.0 / (iters + 2.0)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                continue
            a = a / cs

        # Check candidates
        for cand in [a, b]:
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        if verbose and iters % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score, iters


def freq_domain_perturb(a, rng, sigma=0.05):
    """Add structured noise in the frequency domain."""
    n = len(a)
    n_fft = 1 << (n - 1).bit_length()
    A = rfft(a, n_fft)
    # Add noise to phases
    magnitude = np.abs(A)
    phase = np.angle(A)
    noise_phase = rng.standard_normal(len(phase)) * sigma
    new_A = magnitude * np.exp(1j * (phase + noise_phase))
    a_new = irfft(new_A, n_fft)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def peak_targeted_perturb(a, conv, rng, n_peaks=8):
    """Perturb coordinates that contribute most to the top peaks."""
    n = len(a)
    a_new = a.copy()
    top_k = min(n_peaks, 2 * n - 1)
    peaks = np.argpartition(conv, -top_k)[-top_k:]

    for k in peaks:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        if len(idx) == 0:
            continue
        # Find the top contributors to this peak
        contributions = np.array([a[j] * a[k - j] for j in idx])
        n_reduce = max(1, len(idx) // 4)
        top_contrib = np.argsort(contributions)[::-1][:n_reduce]
        for tc in top_contrib:
            j = idx[tc]
            reduction = 1.0 + rng.standard_normal() * 0.1
            a_new[j] *= max(0.85, reduction * 0.95)

    # Renormalize
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def basin_hopping(a0, time_limit, cd_func, rng, verbose=False):
    """Structured basin hopping with diverse perturbation strategies."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    p = 0

    while time.monotonic() - t0 < time_limit:
        p += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 25:
            break
        t_alloc = min(rem * 0.5, 80.0)

        # Choose perturbation strategy
        strategy = p % 4
        conv_a = fast_convolve(a)

        if strategy == 0:
            # Gaussian noise
            sigma = 0.01 + 0.02 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif strategy == 1:
            # Frequency domain perturbation
            sigma = 0.02 + 0.04 * rng.random()
            perturbed = freq_domain_perturb(a, rng, sigma)
        elif strategy == 2:
            # Peak-targeted perturbation
            perturbed = peak_targeted_perturb(a, conv_a, rng, n_peaks=6 + p % 6)
        else:
            # Combination: partial random + partial freq
            perturbed1 = a + rng.standard_normal(n) * 0.015
            perturbed2 = freq_domain_perturb(a, rng, 0.03)
            perturbed = 0.5 * perturbed1 + 0.5 * perturbed2

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = cd_func(perturbed, t_alloc)
        if s_opt < best_score - 1e-14:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p} strat={strategy}: improved to {best_score:.10f}")
        # Also try the perturbed directly with a short optimization
        elif rem > 40:
            t_short = min(rem * 0.1, 15.0)
            a_opt2, s_opt2, _ = cd_func(perturbed, t_short)
            if s_opt2 < best_score - 1e-14:
                best_score = s_opt2
                best_a = a_opt2.copy()
                if verbose:
                    print(f"    BH p={p} (short): improved to {best_score:.10f}")

    return best_a, best_score


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a_norm)})")
    return best_seq, best_score


def subgradient_step(a, conv, lr, n_peaks=12):
    """One subgradient step using active peak weighting."""
    n = len(a)
    k_act = min(n_peaks, 2 * n - 1)
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks]
    temp = 0.003 * max(pv.max(), 1e-10)
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        return a
    w = w / ws

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        grad[j] += 2.0 * w[idx] * a[k - j]

    a_new = simplex_project(a - lr * grad)
    return a_new


def subgradient_descent(a0, time_limit, verbose=False):
    """Subgradient descent with adaptive learning rate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    lr = 0.003
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        a = subgradient_step(a, conv, lr, n_peaks=10 + (it % 5))
        conv = fast_convolve(a)
        sc = 2.0 * n * float(conv.max())
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.85
                no_improve = 0
        lr *= 0.999

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG it={it} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    1. Load parent + diverse initialization + quick coordinate descent
    2. Intensive analytical coordinate descent
    3. LP fictitious play refinement
    4. Structured basin hopping
    5. Subgradient refinement
    6. Final fine polishing
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(12345)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.10f}")
    except Exception as e:
        print(f"No parent available: {e}")

    n_best = len(best_seq) if best_seq is not None else 1024

    # === Phase 1: Quick exploration (120s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t1_start = time.monotonic()
    t1_budget = 120.0

    n_values = [n_best]
    for alt in [n_best - 64, n_best + 64, 512, 768, 1280]:
        if 256 <= alt <= 2048 and alt not in n_values:
            n_values.append(alt)

    candidates = []

    for n in n_values[:5]:
        if time.monotonic() - t1_start > t1_budget:
            break

        inits = []
        if parent is not None:
            inits.append(('parent', interpolate_seq(parent, n)))
        inits.append(('flat1', init_flat_spectrum(n, rng)))
        inits.append(('flat2', init_flat_spectrum(n, rng)))
        inits.append(('cos', init_cosine(n, rng)))
        inits.append(('fourier', init_fourier_mix(n, rng)))
        inits.append(('zc', init_zadoff_chu_based(n, rng)))

        for itype, cand in inits[:6]:
            t_left = t1_budget - (time.monotonic() - t1_start)
            if t_left < 8:
                break
            t_alloc = max(5.0, min(15.0, t_left / 8.0))
            a_opt, s_opt, _ = exact_coord_descent(cand, t_alloc, label=f"p1_{n}_{itype}")
            candidates.append((s_opt, a_opt, n, itype))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, best: {candidates[0][0]:.10f}")

    for s_c, a_c, n_c, it_c in candidates[:4]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{n_c}_{it_c}")

    n_best = len(best_seq) if best_seq is not None else 1024

    # === Phase 2: Intensive coordinate descent (500s) ===
    print(f"\n=== Phase 2: Intensive ECD (best={best_score:.10f}) ===")
    if best_seq is not None:
        t2_start = time.monotonic()
        t2_budget = 500.0

        current = best_seq.copy()
        for rnd in range(5):
            rem = t2_budget - (time.monotonic() - t2_start)
            if rem < 30:
                break
            t_alloc = min(rem * 0.4, 150.0)
            print(f"  ECD round {rnd + 1}, n={n_best}, budget={t_alloc:.0f}s")
            a_opt, s_opt, sweeps = exact_coord_descent(current, t_alloc,
                                                       verbose=True, label=f"ph2_r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ph2_r{rnd}")

            if best_score > 1.5035:
                # Try to escape via perturbation
                perturbed = a_opt + rng.standard_normal(n_best) * 0.02
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
                current = perturbed
            else:
                current = a_opt.copy()

    # === Phase 3: LP Fictitious Play (300s) ===
    print(f"\n=== Phase 3: LP Fictitious Play (best={best_score:.10f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        t3_start = time.monotonic()
        t3_budget = 300.0

        # Try LP at the current best length and one alternative
        fp_lengths = [n_best]
        if n_best != 1024:
            fp_lengths.append(1024)

        for n_fp in fp_lengths[:2]:
            rem = t3_budget - (time.monotonic() - t3_start)

            t_alloc = min(rem * 0.5, 180.0)

            rem = t3_budget - (time.monotonic() - t3_start)
            if rem > 20 and best_seq is not None:
                t_cd = min(rem * 0.4, 60.0)
                a_cd, s_cd, _ = exact_coord_descent(a_fp, t_cd, label=f"post_fp_{n_fp}")

    print(f"\n=== Phase 4: Basin Hopping (best={best_score:.10f}) ===")
    if best_seq is not None:
        t4_start = time.monotonic()
        t4_budget = 450.0

        a_bh, s_bh = basin_hopping(best_seq, t4_budget, exact_coord_descent, rng, verbose=True)
        best_seq, best_score = check_and_update(a_bh, best_seq, best_score, "bh")

    # === Phase 5: Subgradient refinement (150s) ===
    print(f"\n=== Phase 5: Subgradient (best={best_score:.10f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_sg = min(rem * 0.5, 150.0)
            print(f"  Subgradient, budget={t_sg:.0f}s")
            a_sg, s_sg = subgradient_descent(best_seq, t_sg, verbose=True)
            best_seq, best_score = check_and_update(a_sg, best_seq, best_score, "subgrad")
            # Follow with short CD
            rem2 = deadline - time.monotonic() - 80
            if rem2 > 15:
                a_cd, s_cd, _ = exact_coord_descent(a_sg, min(rem2 * 0.5, 40.0), label="post_sg")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "ecd_sg")

    # === Phase 6: Final polishing (remaining time) ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.10f}) ===")
    if best_seq is not None:
        # Fine coordinate descent
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_fine = min(rem * 0.5, 100.0)
            print(f"  FineCD, budget={t_fine:.0f}s")
            a_fine, s_fine, _ = fine_coord_descent(best_seq, t_fine,
                                                   verbose=True, label="final")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

        # Another exact CD pass
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_cd2 = min(rem * 0.6, 60.0)
            print(f"  ECD-polish, budget={t_cd2:.0f}s")
            a_cd2, s_cd2, _ = exact_coord_descent(best_seq, t_cd2, label="polish")
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "ecd_polish")

        # Final fine pass
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_fin, s_fin, _ = fine_coord_descent(best_seq, min(rem * 0.7, 30.0), label="last")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "last")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Should not happen but handle gracefully
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
