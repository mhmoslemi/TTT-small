"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using adaptive p-norm L-BFGS,
efficient FFT-based gradients, multi-start with diverse initializations,
and aggressive local refinement with peak shaving.
Key approach: p-norm smoothing of the max-convolution objective, solved via
L-BFGS in log-space (softmax parameterization), followed by sequential
local refinement and length exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the actual score for sequence a (assumed non-negative)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return float('inf')
    sum_a = a.sum()
    if sum_a < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    max_b = b.max()
    return 2.0 * n * max_b / (sum_a ** 2)


def p_norm_obj_and_grad_z(z, p):
    """
    Objective and gradient in log-space (softmax parameterization).
    a = softmax(z), objective = 2n * (sum c_k^p)^(1/p) where c = conv(a,a).
    """
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        val = 2.0 * n * c_pos.max()
        grad = np.zeros_like(z)
        return val, grad
    obj = 2.0 * n * S ** (1.0 / p)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    grad = a * (g_a - np.dot(g_a, a))
    return obj, grad


def p_norm_lbfgs(a0, time_limit, p_start=10.0, p_end=500.0, n_stages=8):
    """L-BFGS with adaptive p-norm smoothing schedule in log-space."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-300)
    z = np.log(a0)
    z = z - z.mean()
    t0 = time.monotonic()
    temps = np.geomspace(p_start, p_end, n_stages)

    for si, stage_p in enumerate(temps):
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        iters = max(50, int(remaining * 80 // n_stages))

        def func(z_val):
            return p_norm_obj_and_grad_z(z_val, stage_p)[0]

        def jac(z_val):
            return p_norm_obj_and_grad_z(z_val, stage_p)[1]

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = compute_score(a_cur)
        print(f"      p={stage_p:.0f} ({time.monotonic()-t0:.0f}s): score={sc:.6f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()

def structured_init(n, variant=0):
    """Create structured initial sequence of length n, normalized to sum=1."""
    x = np.linspace(0, 1, n)

    elif variant == 1:
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        a = np.exp(-3.0 * x * (1 - x))
    elif variant == 3:
        a = np.ones(n)
    elif variant == 4:
        xc = (x - 0.5) * 2
        a = np.exp(-xc ** 2 * 2.0)
    elif variant == 5:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * x * 3)
    elif variant == 6:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * x * 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_seq(best_seq, alt_n)
        t_pn = t_total * 0.6
        a_alt = p_norm_lbfgs(init_alt, t_pn, p_start=20.0, p_end=300.0, n_stages=5)
        improved = try_update(a_alt, f"pn_n{alt_n}")
        if improved and rem > 50:
            t_lr = min(rem * 0.10, 60.0)
            a_alt2, _ = local_refine(best_seq, t_lr, max_steps=10000)
            try_update(a_alt2, f"lr_n{alt_n}")
        rem = deadline - time.monotonic()

    # === Phase 6: Perturbation + re-optimization ===
    print(f"\n=== Phase 6: Perturbation ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_seq)
        for noise_scale in [0.02, 0.05, 0.10]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            perturbed = best_seq + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps < 1e-15:
                continue
            perturbed /= ps
            t_sub = min(rem * 0.30, 80.0)
            a_pert = p_norm_lbfgs(perturbed, t_sub, p_start=30.0, p_end=300.0, n_stages=5)
            try_update(a_pert, f"pert_{noise_scale}")
            rem = deadline - time.monotonic()

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.40, 100.0)
        a_fin = p_norm_lbfgs(best_seq, t_fin, p_start=100.0, p_end=5000.0, n_stages=4)
        try_update(a_fin, "final_pnorm")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_lr = min(rem * 0.40, 60.0)
        a_lr, _ = local_refine(best_seq, t_lr, max_steps=15000)
        try_update(a_lr, "final_local")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")

    # Fallback
    if best_seq is None or not np.isfinite(best_score):
        n_fb = 960
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: n={n_fb}, score={best_score:.6f}")

    total_time = time.monotonic() - start_time
    print(f"Total time: {total_time:.0f}s")
    return [float(x) for x in best_seq]
