def fast_convolve(a):
    """Compute a*a (full convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Compute a*b (cross-correlation/convolution) using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

    """Build the convolution matrix T where (T @ b)[k] = (conv(a_fixed, b))[k].
    Returns T as a list of column vectors for memory efficiency."""
    n = len(a_fixed)
    size = 2 * n - 1
    # T is size x n, where T[k, j] = a_fixed[k-j] if 0 <= k-j < n else 0

    return a_fixed


def compute_subgradient(a, active_frac=0.1):
    """Compute subgradient of max_k (a*a)_k with respect to a."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b


def get_lp_solver():
    """Find the best available LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: min_b max_k (conv(a_fixed, b))_k s.t. sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraints: (T @ b)_k <= M for all k
    # (T @ b)_k = sum_j a_fixed[k-j] * b_j (where valid)
    T_a = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_a[j:j+n, j] = a_fixed
    
    constraints = [T_a @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_alternation(a0, time_limit, solver=None, max_iters=2000, verbose=True):
    """
    LP alternation with adaptive mixing.
    Alternates between:
      - Given a, solve LP to get best response b
      - Given b, solve LP to get best response a
    With Nesterov-style acceleration.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    y = a.copy()  # Nesterov momentum variable
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if verbose:
        print(f"  LP Alt start: n={n}, score={best_score:.8f}")
    
    stall_count = 0
    prev_best = best_score
    
    for it in range(max_iters):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        # Solve LP for best response to y
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        # Nesterov update
        alpha = 2.0 / (it + 2.0)
        y_new = (1 - alpha) * y + alpha * b_new
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = b_new.copy()
            sy = 1.0
        y_new = y_new / sy
        
        # Update a
        a = ((it + 1.0) * a + (alpha * (it + 2.0)) * b_new) / (it + 1.0 + alpha * (it + 2.0))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y_new.copy()
            sa = 1.0
        a = a / sa
        
        y = y_new
        
        # Evaluate
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        # Periodic check and restart
        if it % 100 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 10:
                # Add noise to escape
                rng = np.random.default_rng(it * 31 + 7)
                noise = rng.random(n) * 0.01
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
            
            if verbose and it % 200 == 0:
                print(f"    LP Alt iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.1, maxiter=10000):

    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

        b = fast_convolve(a)

        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        lse = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_cur[0]) * lse
    
    def jac(z_in):
        a = np.exp(z_in - z_in.max())
        a = a / a.sum()
        b = fast_convolve(a)
        T_val = T_cur[0]
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)
    
    T_cur = [T_start]
    
    # Redefine functions to use T_cur
    def make_func(T_val):
        def f(z_in):
            a = np.exp(z_in - z_in.max())
            a = a / a.sum()
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            lse = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * lse
        return f
    
    def make_jac(T_val):
        def j(z_in):
            a = np.exp(z_in - z_in.max())
            a = a / a.sum()
            b = fast_convolve(a)
            T_val2 = T_val
            Tb = T_val2 * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

    n_stages = 6

    iters_per_stage = max(50, maxiter // n_stages)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:

                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_result = np.exp(z - z.max())
    a_result = a_result / a_result.sum()
    return a_result


def mirror_descent(a0, time_limit, active_frac=0.05):
    """Mirror descent with adaptive learning rate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = np.log(np.maximum(a, 1e-300))
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.3
    
    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        g, max_b = compute_subgradient(a, active_frac)
        g_norm = np.abs(g).max()
        if g_norm < 1e-15:
            break
        
        polyak_lr = (best_score - 2.0 * n / (2.0 * n - 1.0)) / (2.0 * g_norm * best_score + 1e-15)
        alpha = max(0.01, min(lr, polyak_lr))
        
        z_new = z - alpha * g
        a_new = np.exp(z_new - z_new.max())
        a_new = a_new / a_new.sum()
        score_new = 2.0 * n * fast_convolve(a_new).max()
        
        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            lr = min(lr * 1.03, 1.5)
        else:
            lr = max(lr * 0.95, 0.01)
    
    return best_a, best_score


def local_perturbation(a0, time_limit, n_trials=50000):
    """Local search with random perturbations."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    no_improve = 0
    
    for step in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute which positions to perturb
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        # Find positions contributing to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib = np.zeros(n)
        idx = np.arange(js, je + 1)
        contrib[idx] = a[idx] * a[peak_k - idx]
        
        # Try moving mass from high-contrib to low-contrib
        top = np.argsort(contrib)[::-1][:3]
        bottom = np.argsort(contrib)[:3]
        
        improved = False
        for si in top:
            for di in bottom:
                if si == di or a[si] < 1e-14:
                    continue
                frac = 0.05
                amount = a[si] * frac
                a_trial = a.copy()
                a_trial[si] -= amount
                a_trial[di] += amount
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    no_improve = 0
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-14:
                amount = a[i] * rng.random() * 0.05
                a_trial = a.copy()
                a_trial[i] -= amount
                a_trial[j] += amount
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        no_improve = 0
        
        if no_improve > 5000:
            break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 0.5
    else:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 10.0 * x * (1.0 - x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
