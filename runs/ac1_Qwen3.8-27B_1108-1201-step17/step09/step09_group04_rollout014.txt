"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. P-norm L-BFGS (coarse phase)
2. Multiplicative Weights / Mirror Descent with exact subgradient of max (fine phase)
3. Nesterov-type averaging
4. Local refinement (peak shaving)
5. Multi-length exploration

Key innovation: Direct use of the exact subgradient of max(convolve(a,a)) 
with multiplicative weights updates, which is O(n log n) per iteration 
via FFT. This avoids the approximation error of p-norm methods while 
being fast enough for many iterations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def get_subgradient_of_max(a):
    """
    Compute the (sub)gradient of max_k c_k(a) where c = convolve(a,a).
    Returns the gradient vector g such that g[m] = 2*a[k*-m] if 0<=k*-m<n else 0,
    where k* is an argmax of c.
    If multiple k are near the max, uses a weighted average.
    """
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    # Find all k within a small tolerance of the max
    tol = 1e-10 * max_c
    active = np.where(c >= max_c - tol)[0]
    if len(active) == 1:
        k_star = active[0]
        # Gradient: g[m] = 2*a[k_star - m] if 0 <= k_star - m < n else 0
        g = np.zeros(n)
        # m ranges 0..n-1, k_star-m ranges 0..n-1 => m ranges k_star-(n-1) .. k_star
        m_lo = max(0, k_star - (n - 1))
        m_hi = min(n - 1, k_star)
        if m_lo <= m_hi:
            # g[m] = 2*a[k_star - m] for m in [m_lo, m_hi]
            # a[k_star - m] for m = m_lo gives a[k_star - m_lo] = a[m_hi]
            # a[k_star - m] for m = m_hi gives a[k_star - m_hi] = a[m_lo]
            g[m_lo:m_hi+1] = 2.0 * a[m_hi:m_lo-1:-1]
        return g, c
    else:
        # Weighted average of gradients for active k's
        # Weight proportional to (c_k - max_c + tol) / (len(active) * tol) or just uniform
        g = np.zeros(n)
        for k in active:
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                g[m_lo:m_hi+1] += 2.0 * a[m_hi:m_lo-1:-1]
        g /= len(active)
        return g, c


def mirror_descent_refine(a0, time_limit, n_max_iters=200000, eta_init=0.1):
    """
    Mirror descent (multiplicative weights) with exact subgradient of max(convolve(a,a)).
    Uses Nesterov-type averaging for the primal iterate.
    
    The update: a_new[m] = a[m] * exp(-eta * g[m]) / Z
    where g = subgradient of max(c) at current a.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Nesterov averaging
    avg_a = a.copy()
    y = a.copy()  # extrapolated point
    
    eta = eta_init
    stall_count = 0
    prev_best = best_score
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient at y (extrapolated point) or a
        # For simplicity, use a directly (no extrapolation for now)
        g, c = get_subgradient_of_max(a)
        
        # Adaptive step size
        # Theoretical: eta = sqrt(2 * log(n) / (T * L)) where L is Lipschitz constant
        # But we don't know L. Use a heuristic: start with eta_init, decrease if oscillating
        # Simple approach: fixed eta with periodic halving
        if it % 5000 == 0 and it > 0:
            eta *= 0.5
        
        # Multiplicative weights update
        # a_new = a * exp(-eta * g) / Z
        # For numerical stability, subtract max of a*eta*g
        exponent = -eta * g
        exponent -= exponent.max()  # shift for stability
        weights = a * np.exp(exponent)
        Z = weights.sum()
        if Z < 1e-300:
            break
        a_new = weights / Z
        
        # Check if improvement
        score_new = 2.0 * n * fast_convolve(a_new).max()
        
        # Also try a "damped" step: a_damped = (1-t)*a + t*a_new
        # But multiplicative weights already gives a valid point on the simplex
        # So we just use a_new directly
        a = a_new
        
        # Nesterov averaging (standard convex averaging)
        # avg_a = (1/(it+1)) * sum of all iterates
        # Efficient: avg_a = ((it)*avg_a + a) / (it+1)
        avg_a = ((it) * avg_a + a) / (it + 1)
        
        # Track best
        if score_new < best_score:
            best_score = score_new
            best_a = a.copy()
            stall_count = 0
        else:
            # Check avg_a too
            score_avg = 2.0 * n * fast_convolve(avg_a).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_a.copy()
                stall_count = 0
        
        if it % 1000 == 0:
            if best_score < prev_best - 1e-8:
                prev_best = best_score
                stall_count = 0
            else:
                stall_count += 1
            if stall_count >= 3:
                # Reduce eta more aggressively
                eta *= 0.3
                stall_count = 0
                prev_best = best_score
            
            elapsed = time.monotonic() - t0
            if it % 5000 == 0:
                print(f"      MirrorDescent iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}, eta={eta:.2e}")
    
    return best_a, best_score


def p_norm_objective_and_grad(a, p):
    """Compute p-norm of positive part of convolve and its gradient."""
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return 0, np.zeros(n)
    factor = S ** (1.0/p - 1.0)
    w = c_pos ** (p - 1.0)
    # Gradient: g_a[m] = 2 * factor * sum_k w[k] * a[k-m] (if valid)
    # This is 2 * factor * correlate(w, a, 'valid')
    corr = np.correlate(w, a, mode='valid')
    g_a = 2.0 * factor * corr
    val = 2.0 * n * S ** (1.0/p)
    return val, g_a


def p_norm_lbfgs_refine(a0, time_limit, p_start=20.0, p_end=500.0, n_stages=8):
    """L-BFGS with adaptive p-norm smoothing schedule in log domain."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    
    temps = np.geomspace(p_start, p_end, n_stages)
    iters_per_stage = max(50, 5000 // n_stages)
    
    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_val):
            z_max = z_val.max()
            exp_z = np.exp(z_val - z_max)
            a = exp_z / exp_z.sum()
            c = fast_convolve(a)
            c_pos = np.maximum(c, 1e-300)
            val = 2.0 * n * (np.sum(c_pos ** stage_p)) ** (1.0/stage_p)
            return val
        
        def jac(z_val):
            z_max = z_val.max()
            exp_z = np.exp(z_val - z_max)
            a = exp_z / exp_z.sum()
            c = fast_convolve(a)
            c_pos = np.maximum(c, 1e-300)
            p_val = c_pos ** stage_p
            S = np.sum(p_val)
            if S < 1e-300:
                return np.zeros_like(z_val)
            factor = S ** (1.0/stage_p - 1.0)
            w = c_pos ** (stage_p - 1.0)
            corr = np.correlate(w, a, mode='valid')
            g_a = 2.0 * n * factor * corr
            # Gradient w.r.t. z
            return a * (g_a - np.dot(g_a, a))
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        score = 2.0 * n * fast_convolve(a_cur).max()
        elapsed = time.monotonic() - t0
        print(f"      p-norm p={stage_p:.1f} ({elapsed:.0f}s): score={score:.6f}")
    
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


def local_refinement(a0, time_limit, n_steps=50000):
    """Local refinement: peak shaving via coordinate descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 500:
            break
        
        c = fast_convolve(a)
        # Find the peak
        k_peak = np.argmax(c)
        peak_val = c[k_peak]
        
        # Find elements that contribute to this peak
        # c[k_peak] = sum_i a[i] * a[k_peak - i] for valid i
        m_lo = max(0, k_peak - (n - 1))
        m_hi = min(n - 1, k_peak)
        idx = np.arange(m_lo, m_hi + 1)
        if len(idx) == 0:
            break
        contribs = a[idx] * a[k_peak - idx]
        
        # Try reducing the top contributor
        top_i = idx[np.argmax(contribs)]
        if a[top_i] < 1e-14:
            no_improve += 1
            continue
        
        improved = False
        for factor in [0.95, 0.90, 0.85, 0.80, 0.70, 0.50]:
            a_trial = a.copy()
            diff = a_trial[top_i] * (1 - factor)
            a_trial[top_i] -= diff
            if a_trial[top_i] < 0:
                a_trial[top_i] = 0
                diff = a[top_i]
            # Redistribute to minimize the peak (add to elements that don't contribute to this peak)
            # Simple: add uniformly to all other elements
            mask = np.ones(n, dtype=bool)
            mask[top_i] = False
            others = a_trial[mask]
            if others.sum() < 1e-15:
                continue
            a_trial[mask] += diff * others / others.sum()
            
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
            # Try random perturbation
            if step % 100 == 0 and no_improve > 100:
                rng = np.random.default_rng(step)
                n_pert = rng.integers(2, 6)
                indices = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for i in indices:
                    a_trial[i] *= (1.0 + rng.uniform(-0.1, 0.1))
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 0.01:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        no_improve = 0
    
    return best_a, best_score


def structured_init(n, variant=0):
    """Create structured initial sequence."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif variant == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif variant == 3:
        # Asymmetric: peak off-center
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2)
    elif variant == 4:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 5:
        # Parabolic
        x = np.linspace(-1, 1, n)
        a = np.maximum(1 - x ** 2, 0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Evaluate safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if sum(a_list) < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_state, a, label):
    """Try to update best state."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Log-Sum-Exp (LSE) smoothed mirror descent with multiplicative weights - primary workhorse
2. P-norm L-BFGS in log-domain for coarse optimization and escaping local optima
3. Local refinement (peak shaving via coordinate descent)
4. Multi-length exploration with interpolation
5. Perturbation-based restarts

Key innovation: Uses LSE smoothing of max(convolve(a,a)) with the exact gradient
structure (convex combination of per-lag gradients), combined with multiplicative
weights (mirror descent on the simplex) for O(n log n) per iteration via FFT.
The LSE parameter tau is annealed from large to small, providing a smooth path
from a convex-like problem to the true non-smooth objective.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_objective_and_grad(a, tau):
    """
    Compute 2n * LSE(c, tau) and its gradient w.r.t. a,
    where c = convolve(a,a), LSE(c,tau) = tau*log(sum(exp((c-cmax)/tau))) + cmax.
    """
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    scaled = (c - c_max) / tau
    # Clip for numerical stability
    scaled = np.maximum(scaled, -50.0)
    exp_c = np.exp(scaled)
    sum_exp = exp_c.sum()
    if sum_exp < 1e-300:
        return 0.0, np.zeros(n)
    obj_val = c_max + tau * np.log(sum_exp)
    # Weights: softmax of c/tau
    weights = exp_c / sum_exp
    # Gradient of LSE w.r.t. a: sum_k weights[k] * grad_c_k
    # grad_c_k[m] = 2*a[k-m] if 0<=k-m<n else 0
    # So grad[m] = 2*sum_k weights[k]*a[k-m]*1[0<=k-m<n]
    # This is 2 * correlate(weights, a, 'valid')
    grad_inner = 2.0 * np.correlate(weights, a, mode='valid')
    obj_val *= 2.0 * n
    grad_inner *= 2.0 * n
    return obj_val, grad_inner


def lse_mirror_descent(a0, time_limit, tau_start=0.5, tau_end=0.001, n_max_iters=300000):
    """
    Mirror descent (multiplicative weights) with LSE-smoothed subgradient.
    Anneals tau from tau_start to tau_end.
    Uses Nesterov-style averaging for the returned iterate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Geometric schedule for tau
    n_tau_steps = 20
    tau_schedule = np.geomspace(tau_start, tau_end, n_tau_steps)
    iters_per_tau = max(10, n_max_iters // n_tau_steps)

    # Running average (for Nesterov-type convergence)
    avg_a = a.copy()
    count = 0

    # Adaptive step size
    eta = 0.1
    prev_obj = lse_objective_and_grad(a, tau_schedule[0])[0]

    for tau_idx, tau in enumerate(tau_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        for it in range(iters_per_tau):
            if time.monotonic() - t0 > time_limit:
                break

            obj_val, g = lse_objective_and_grad(a, tau)

            # Adaptive step size: if objective increased, reduce eta
            if obj_val > prev_obj * 1.01:
                eta *= 0.5
            elif obj_val < prev_obj * 0.99:
                eta = min(eta * 1.1, 0.5)
            prev_obj = obj_val

            # Multiplicative weights update: a_new = a * exp(-eta * g / (2n)) / Z
            # Scale gradient to get reasonable step
            g_scaled = g / (2.0 * n * max(1.0, np.abs(g).max() + 1e-300))
            exponent = -eta * g_scaled
            exponent -= exponent.max()
            weights = a * np.exp(exponent)
            Z = weights.sum()
            if Z < 1e-300:
                break
            a_new = weights / Z

            # Check score
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()

            # Update
            a = a_new
            count += 1
            avg_a = ((count - 1) * avg_a + a) / count

            # Periodic check of avg_a
            if it % 500 == 0:
                score_avg = 2.0 * n * fast_convolve(avg_a).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_a.copy()

        elapsed = time.monotonic() - t0
        print(f"      LSE-MD tau={tau:.4f} ({elapsed:.0f}s): score={best_score:.6f}, eta={eta:.4f}")

    return best_a, best_score


def subgradient_mirror_descent(a0, time_limit, n_max_iters=300000):
    """
    Mirror descent with exact subgradient of max(c).
    Uses the active set (all k within tolerance of max) for the subgradient.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    avg_a = a.copy()
    count = 0
    eta = 0.05
    stall = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        max_c = c.max()

        # Find active set (k's within relative tolerance of max)
        tol = 1e-6 * max_c
        active = np.where(c >= max_c - tol)[0]

        # Compute subgradient: average of grad_c_k over active set
        g = np.zeros(n)
        for k in active:
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                g[m_lo:m_hi+1] += 2.0 * a[m_hi:m_lo-1:-1]
        g /= len(active)

        # Scale for step size
        g_norm = np.abs(g).max()
        if g_norm < 1e-300:
            break
        g_scaled = g / g_norm

        # Multiplicative weights update
        exponent = -eta * g_scaled
        exponent -= exponent.max()
        weights = a * np.exp(exponent)
        Z = weights.sum()
        if Z < 1e-300:
            break
        a_new = weights / Z

        score = 2.0 * n * fast_convolve(a_new).max()
        a = a_new
        count += 1
        avg_a = ((count - 1) * avg_a + a) / count

        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive eta
        if it % 2000 == 0:
            if best_score < prev_best - 1e-9:
                prev_best = best_score
                stall = 0
            else:
                stall += 1
            if stall >= 3:
                eta *= 0.4
                stall = 0
                prev_best = best_score

            # Also check avg
            score_avg = 2.0 * n * fast_convolve(avg_a).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_a.copy()

            if it % 10000 == 0:
                elapsed = time.monotonic() - t0
                print(f"      SubGrad-MD iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}, eta={eta:.2e}")

    return best_a, best_score


def pnorm_obj_grad_z(z_val, p, n):
    """P-norm objective and gradient in log domain (z = log(a))."""
    z_max = z_val.max()
    exp_z = np.exp(z_val - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return 0.0, np.zeros_like(z_val)
    obj_val = 2.0 * n * S ** (1.0 / p)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 2.0 * n * factor * corr
    g_z = a * (g_a - np.dot(g_a, a))
    return obj_val, g_z


def pnorm_lbfgs_refine(a0, time_limit, p_start=20.0, p_end=500.0, n_stages=7):
    """L-BFGS with p-norm smoothing in log domain."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    temps = np.geomspace(p_start, p_end, n_stages)
    iters_per_stage = max(50, 4000 // n_stages)

    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_v):
            return pnorm_obj_grad_z(z_v, stage_p, n)[0]

        def jac(z_v):
            return pnorm_obj_grad_z(z_v, stage_p, n)[1]

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        score = 2.0 * n * fast_convolve(a_cur).max()
        elapsed = time.monotonic() - t0
        print(f"      p-norm p={stage_p:.1f} ({elapsed:.0f}s): score={score:.6f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z
