"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated gradient computation for log-sum-exp smoothing
2. Multi-stage L-BFGS with high-temperature annealing (T: 500 -> 0.5)
3. Multi-length exploration with warm-start interpolation
4. Subgradient + momentum for final polish
5. Peak-shaving coordinate descent
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_objective(z, T, n):
    """Compute log-sum-exp objective value."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    return (2.0 * n / T) * (max_Tb + np.log(exp_Tb.sum()))


def lse_gradient(z, T, n):
    """Compute gradient of log-sum-exp objective w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    # grad_a[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation)
    # Using FFT: grad_a = 2 * IFFT(FFT(w) * conj(FFT(a)))
    # But we need to be careful about the correlation direction.
    
    # Actually, let's use the direct formula with FFT:
    # grad_a[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    # This is the cross-correlation of w and a at lag j.
    
    # For cross-correlation via FFT:
    # corr(w, a)[j] = sum_m w[m+j] * a[m] = IFFT(FFT(w) * conj(FFT(a)))[j]
    # (with appropriate zero-padding)
    
    size = (2 * n - 1) + n - 1  # = 3n - 2
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    grad_a = 2.0 * corr
    
    # Chain rule: d/dz_j = sum_i (da_i/dz_j) * grad_a[i]
    # da_i/dz_j = a_i * (delta_ij - a_j)
    # d/dz_j = a_j * (grad_a[j] - sum_i a_i * grad_a[i])
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_refine(a0, time_limit, T_start=500.0, T_end=0.5, n_stages=15, maxiter_per_stage=None):
    """L-BFGS with log-sum-exp, multi-stage cooling."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    if maxiter_per_stage is None:
        maxiter_per_stage = max(50, int(3000 / n_stages))
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        if stage == 0:
            T = T_start
        else:
            T = T_start * (T_end / T_start) ** (stage / (n_stages - 1))
        
        # For early stages (high T), use fewer iterations
        # For later stages (low T), use more iterations
        iters = max(20, int(maxiter_per_stage * (1.0 + stage / n_stages)))
        
        def func(z_in):
            return lse_objective(z_in, T, n)
        
        def jac(z_in):
            return lse_gradient(z_in, T, n)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
            if stage % 3 == 0:
                a_cur = softmax_from_z(z)
                sc = 2.0 * n * fast_convolve(a_cur).max()
                print(f"    Stage {stage}/{n_stages}, T={T:.1f}, score={sc:.6f}")
        except Exception as e:
            print(f"    Stage {stage} error: {e}")
    
    return softmax_from_z(z)


def subgradient_polish(a0, time_limit, n_iters=100000):
    """Subgradient descent with momentum on the simplex for final polish."""
    n = len(a0)
    a = norm_sequence(a0)
    v = np.zeros(n)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.05
    mu = 0.9
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        
        # Subgradient of max_k b[k] at k_star
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            if 2 * i == k_star:
                g[i] += a[k_star - i]
            else:
                g[i] += 2.0 * a[k_star - i]
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Polyak step size or fixed with decay
        eta_it = eta / (1.0 + 0.001 * it)
        
        v = mu * v - eta_it * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            eta = max(0.001, eta * 0.99)  # slow decay
        else:
            eta = max(0.001, eta * 0.95)  # faster decay
        a = a_new
    
    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_rounds=5000):
    """Coordinate descent that reduces the peak."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    
    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        peak_val = b[k_star]
        
        # Find coordinates that contribute most to the peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_star - idx]
        
        # Sort by contribution
        order = np.argsort(contribs)[::-1]
        
        improved = False
        for ci in order[:20]:
            i = idx[ci]
            if a[i] < 1e-12:
                continue
            
            # Try reducing a[i] and redistributing
            for factor in [0.98, 0.95, 0.90, 0.85]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-15:
                    continue
                
                a_trial = a.copy()
                a_trial[i] = new_val
                
                # Redistribute to low-contribution positions
                low_order = order[order != ci]
                # Pick positions far from the peak
                redistribute_to = []
                for ci2 in low_order[:10]:
                    j2 = idx[ci2]
                    if j2 != i:
                        redistribute_to.append(j2)
                
                if redistribute_to:
                    rt = np.array(redistribute_to)
                    weights = a_trial[rt] + 1e-10
                    a_trial[rt] = weights + diff * weights / weights.sum()
                
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 100:
                break
        else:
            no_improve = 0
    
    return best_a, best_score
