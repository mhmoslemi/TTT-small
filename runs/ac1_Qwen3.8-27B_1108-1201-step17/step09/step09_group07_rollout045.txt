```python
def mwu_iteration(a, eta):
    """One step of multiplicative weights update."""
    # Compute subgradient of max(conv(a,a)) w.r.t. a
    b = convolve(a, a)
    peak_k = argmax(b)
    # Subgradient: for the peak at k, grad[i] = a[k-i] (for valid i)
    grad = zeros(n)
    for i in range(n):
        j = peak_k - i
        if 0 <= j < n:
            grad[i] = a[j]
    # Update
    a_new = a * exp(-eta * grad)
    a_new = a_new / sum(a_new)
    return a_new
```

def mwu_multi_peak(a, eta, threshold_frac=0.9):
    b = convolve(a, a)
    max_b = b.max()
    threshold = threshold_frac * max_b
    active_peaks = where(b >= threshold)[0]
    grad = zeros(n)
    for k in active_peaks:
        for i in range(max(0, k-n+1), min(n, k+1)):
            j = k - i
            if 0 <= j < n:
                grad[i] += a[j] / len(active_peaks)
    a_new = a * exp(-eta * grad)
    a_new = a_new / sum(a_new)

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv(a,a))."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_name = None


def get_lp_solver():
    global _lp_solver_name
    if _lp_solver_name is not None:
        return _lp_solver_name
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_name = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_name = cp.ECOS
    return _lp_solver_name


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def grad_smooth_max(a, T_temp):
    """Gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T_temp * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def nesterov_pg(a0, time_limit, T_start=2.0, T_end=400.0,
                 lr_init=0.06, n_stages=10, steps_per_stage=3000):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            lr = lr_init / (1.0 + 0.003 * step_count)
            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            t_prev_calc = 1.0 + np.sqrt(1.0 + 4.0 * max(0, step_count - 1) ** 2) / 2.0
            beta = (t_prev_calc - 1.0) / t_curr if step_count > 0 else 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)
            grad = grad_smooth_max(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1
            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()

    return best_a, best_score


def geometric_mean_mix(a, b, alpha=0.5):
    """Geometric mean mixing: a_new = a^(1-alpha) * b^alpha, projected to simplex."""
    a_safe = np.maximum(a, 1e-30)
    b_safe = np.maximum(b, 1e-30)
    mixed = a_safe ** (1.0 - alpha) * b_safe ** alpha
    s = mixed.sum()
    if s < 1e-15:
        return project_simplex(a)
    return project_simplex(mixed / s)


def freq_domain_smoothing(a, sigma=0.5):
    """Apply Gaussian smoothing in frequency domain to encourage flatter spectrum."""
    n = len(a)
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a, fft_size)
    k = np.arange(fft_size // 2 + 1)
    # Gaussian window centered at 0 frequency (DC)
    # Actually, we want to flatten the spectrum, so we want to REDUCE the variance
    # of |A|^2. A simple approach: blend A with a uniform-magnitude version.
    mag = np.abs(A)
    phase = np.angle(A)
    avg_mag = mag.mean()
    # Soft flatten: pull magnitudes toward the mean
    smoothed_mag = mag + sigma * (avg_mag - mag)
    smoothed_mag = np.maximum(smoothed_mag, 0)
    A_smooth = smoothed_mag * np.exp(1j * phase)
    a_smooth = irfft(A_smooth, fft_size)[:n]
    a_smooth = np.maximum(a_smooth, 0)
    s = a_smooth.sum()
    if s < 1e-15:
        return project_simplex(a)
    return project_simplex(a_smooth / s)


def lp_iteration(a0, time_limit, solver=None, n_max_iters=100000,
                 use_geometric=True, freq_smooth_every=200, freq_sigma=0.1,
                 perturb_every=500, perturb_size=0.01, rng=None):
    """LP-based best response iteration with geometric mean mixing and regularization."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            stall_count += 1
            if stall_count > 50:
                break
            continue

        # Adaptive mixing
        if use_geometric:
            alpha = 2.0 / (it + 2)
            a_new = geometric_mean_mix(a, b_new, alpha)
        else:
            alpha = 2.0 / (it + 2)
            a_new = project_simplex((1 - alpha) * a + alpha * b_new)

        # Periodic frequency-domain smoothing
        if freq_smooth_every > 0 and it % freq_smooth_every == 0 and it > 0:
            a_new = freq_domain_smoothing(a_new, freq_sigma)

        a = a_new

        # Track best
        score_a = score_normalized(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1

        # Periodic perturbation to escape local minima
        if perturb_every > 0 and stall_count >= perturb_every:
            a = a + perturb_size * rng.standard_normal(n)
            a = project_simplex(a)
            stall_count = 0

        prev_best = best_score
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it + 1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                                   T_start=1.0, T_end=200.0,
                                   lr_init=0.1, n_stages=6, steps_per_stage=2000)
            try_update(a_sub, f"ms_mode{mode}")
            # Then short LP
            rem = deadline - time.monotonic()
            if rem > 15:
                a_sub2, _ = lp_iteration(a_sub, min(rem * 0.08, 30.0), solver=solver,
                                         use_geometric=True, freq_smooth_every=100,
                                         freq_sigma=0.08, perturb_every=200,
                                         perturb_size=0.01, rng=rng)
                try_update(a_sub2, f"ms_mode{mode}_lp")

    # === Phase 6: Final LP polish (5% time) ===
    print(f"\n=== Phase 6: Final LP Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.25, 90.0)
        a_fin, _ = lp_iteration(best_seq, t_fin, solver=solver,
                                use_geometric=True, freq_smooth_every=500,
                                freq_sigma=0.02, perturb_every=800,
                                perturb_size=0.005, rng=rng)
        try_update(a_fin, "final_lp")

    # === Phase 7: Peak shaving polish (last few seconds) ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t_ps = min(rem * 0.30, 40.0)
        a_ps, _ = peak_shave(best_seq, t_ps, n_steps=20000)
        try_update(a_ps, "peak_shave")

    # === Final: one more LP iteration on the very best ===
    rem = deadline - time.monotonic()
    if rem > 15:
        a_last, _ = lp_iteration(best_seq, min(rem * 0.5, 30.0), solver=solver,
                                 use_geometric=True, freq_smooth_every=0,
                                 freq_sigma=0.0, perturb_every=1000,
                                 perturb_size=0.003, rng=rng)
        try_update(a_last, "last_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
