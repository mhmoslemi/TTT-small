"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses p-norm L-BFGS smoothing, active-set subgradient methods,
targeted local refinement, and multi-length exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2 using raw values (matching evaluate_sequence)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fast_convolve(a)
    return float(2 * n * c.max() / (s * s))

def score_normalized(a):
    """Assumes a is already normalized to sum=1, returns 2n*max(conv)."""
    c = fast_convolve(a)
    return float(2 * len(a) * c.max())

# p-norm based objective in log-domain (softmax parameterization)
def pnorm_obj_z(z, p):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-30)
    return 2.0 * len(a) * (np.sum(c_pos ** p)) ** (1.0/p)

def pnorm_grad_z(z, p):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-30)
    cp = c_pos ** p
    S = cp.sum()
    if S < 1e-30:
        return np.zeros_like(z)
    factor = S ** (1.0/p - 1.0)
    w = c_pos ** (p - 1.0)
    # Gradient of (sum c^p)^(1/p) w.r.t. a[i] = factor * 2 * sum_k w[k] * a[k-i] (valid range)
    corr = np.correlate(w, a, mode='valid')
    g_a = 2.0 * factor * corr
    # Chain rule through softmax: grad_z = a * (g_a - a @ g_a)
    # But objective is 2n * pnorm, so multiply by 2n
    g_a = 2.0 * n * g_a
    return a * (g_a - np.dot(g_a, a))

def pnorm_lbfgs(a0, time_limit, p_start=20, p_end=500, n_stages=8):
    """L-BFGS with p-norm smoothing schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30))
    z = z - z.max()
    t0 = time.monotonic()
    
    for stage_i in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        p_val = p_start * (p_end / p_start) ** (stage_i / max(1, n_stages - 1))
        
        def func(zv):
            return pnorm_obj_z(zv, p_val)
        
        def jac(zv):
            return pnorm_grad_z(zv, p_val)
        
        iters = max(50, int(6000 / n_stages))
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-15,
                                   'maxfun': iters * 500, 'disp': False})
            z = res.x
        except Exception as e:
            print(f"    L-BFGS stage {stage_i} failed: {e}")
        
        # Evaluate
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = score_normalized(a_cur)
        elapsed = time.monotonic() - t0
        print(f"    p-norm p={p_val:.1f} ({elapsed:.0f}s): score={sc:.6f}")
    
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        # c[peak_k] = sum_{i in idx} a[i] * a[peak_k - i]
        contribs = np.array([a[i] * a[peak_k - i] for i in idx])
        
        improved = False
        
        # Strategy 1: Reduce the largest contributor and redistribute
        for top_frac in [0.95, 0.90, 0.85, 0.80]:
            ci = np.argmax(contribs)
            i = idx[ci]
            j = peak_k - i
            if i == j:
                # Reduce a[i]
                reduction = a[i] * (1 - top_frac)
                if reduction < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[i] -= reduction
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask].sum()
                if others < 1e-15:
                    continue
                a_trial[mask] += reduction * a_trial[mask] / others
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            else:
                # Can reduce either a[i] or a[j]
                for target in [i, j]:
                    reduction = a[target] * (1 - top_frac)
                    if reduction < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[target] -= reduction
                    mask = np.ones(n, dtype=bool)
                    mask[target] = False
                    others = a_trial[mask].sum()
                    if others < 1e-15:
                        continue
                    a_trial[mask] += reduction * a_trial[mask] / others
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        # Strategy 2: Move mass from peak contributor to a "quiet" region
        if not improved:
            ci = np.argmax(contribs)
            i = idx[ci]
            j = peak_k - i
            if i != j:
                target = i if a[i] > a[j] else j
                # Find quiet positions (low contribution to all peaks)
                # Approximate: positions far from peak_k
                dist = np.abs(np.arange(n) - target)
                quiet_mask = dist > n // 4
                quiet_positions = np.where(quiet_mask)[0]
                if len(quiet_positions) > 0:
                    # Move a small fraction to a random quiet position
                    rng = np.random.default_rng(step)
                    dest = rng.choice(quiet_positions)
                    for frac in [0.01, 0.02, 0.05]:
                        amount = a[target] * frac
                        if amount < 1e-18:
                            continue
                        a_trial = a.copy()
                        a_trial[target] -= amount
                        a_trial[dest] += amount
                        a_trial = np.maximum(a_trial, 0)
                        s = a_trial.sum()
                        if s > 0:
                            a_trial = a_trial / s
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
        
        # Strategy 3: Pair of coordinates
        if not improved:
            # Reduce two large coordinates simultaneously
            ci1 = np.argmax(contribs)
            contribs_ci1 = contribs[ci1]
            contribs[ci1] = -1
            ci2 = np.argmax(contribs)
            i1, j1 = idx[ci1], peak_k - idx[ci1]
            i2, j2 = idx[ci2], peak_k - idx[ci2]
            
            # Try reducing the pair (i1, i2) or (j1, j2) etc.
            for pair in [(i1, i2), (j1, j2), (i1, j2), (j1, i2)]:
                p1, p2 = pair
                if p1 == p2 or p1 < 0 or p1 >= n or p2 < 0 or p2 >= n:
                    continue
                if a[p1] < 1e-14 or a[p2] < 1e-14:
                    continue
                for f in [0.95, 0.90, 0.85]:
                    a_trial = a.copy()
                    a_trial[p1] *= f
                    a_trial[p2] *= f
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s > 0:
                        a_trial = a_trial / s
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            # Stuck: try a random perturbation and a few more steps
            if step % 200 == 0:
                rng = np.random.default_rng(step * 31 + 7)
                n_pert = rng.integers(3, 10)
                indices = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for i in indices:
                    a_trial[i] *= (1.0 + rng.uniform(-0.1, 0.1))
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s > 0:
                    a_trial = a_trial / s
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            if not improved and step > 100:
                break
        
        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    LocalRefine step {step} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def nesterov_lp_like(a0, time_limit, n_iters=100000):
    """
    Alternating method: Given a, find b that minimizes max conv(a,b) via
    a simple iterative water-filling approach (avoiding full LP for speed).
    Then update a towards b.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    y = a.copy()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Given y, find b minimizing max_k conv(y, b)[k] subject to b>=0, sum(b)=1
        # This is: minimize t s.t. conv(y,b)[k] <= t for all k, b>=0, sum(b)=1
        # Use iterative water-filling / multiplicative weights
        # b_new[i] = b[i] * (1 - eta * gradient_i) where gradient_i = sum_{k: active} dy Conv...
        
        # Simpler: use the fact that the optimal b puts mass where y's
        # contribution to peaks is lowest
        # conv(y, b)[k] = sum_i y[k-i] * b[i]
        # We want to minimize max_k of this.
        # Heuristic: b[i] proportional to 1 / (max over k of y[k-i])
        # i.e., put more mass where y's "spread" is lowest
        
        # Compute for each b position i, the maximum contribution to any conv value
        # conv(y,b)[k] = sum_i y[k-i] * b[i]
        # If b is uniform, conv(y,b)[k] = sum_i y[k-i] / n
        # To reduce peaks, put less b where y has high values
        
        # Compute "exposure" of each position
        exposure = np.zeros(n)
        # For b[i], it contributes y[k-i] to conv[k] for all valid k
        # The "peak exposure" is max_k y[k-i]
        # This is just the max of y (since k ranges over a window of n values)
        # Actually max_k y[k-i] = max(y) for all i (since the window covers all of y)
        # So this heuristic doesn't differentiate.
        
        # Better: compute the actual conv with uniform b, find which i contribute
        # most to peaks, and reduce those.
        b_candidate = np.ones(n) / n
        conv_yb = fast_convolve-like... 
        
        # Actually, let me just use a simple approach:
        # Compute T = convolution matrix (T @ b = conv(y, b))
        # The LP is: min t s.t. T @ b <= t * 1, sum(b) = 1, b >= 0
        # Dual: max sum(y...) ... this gets complex.
        
        # Instead, use multiplicative weight updates:
        # b[i] *= (1 - eta * subgrad_i) where subgrad comes from active constraints
        
        c_yb = np.zeros(2 * n - 1)
        # conv(y, b_candidate)
        # Use FFT
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Y = rfft(y, fft_size)
        B = rfft(b_candidate, fft_size)
        c_yb = irfft(Y * B, fft_size)[:size]
        
        c_max = c_yb.max()
        # Find which b positions contribute most to the peak
        peak_k = c_yb.argmax()
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        # c_yb[peak_k] = sum_{i=js}^{je} y[peak_k - i] * b_candidate[i]
        # Contribution of b_candidate[i] is y[peak_k - i] * b_candidate[i]
        
        # Reduce b at positions with high contribution to peak
        eta = 0.1
        for k in [peak_k]:  # Just the main peak for now
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                # Reduce b[i] proportional to y[k-i] * b[i]
                pass  # Too slow for large n
        
        # This is getting too slow. Let me use a vectorized approach.
        # For the peak at k*, the gradient w.r.t. b[i] is y[k*-i] (if valid)
        # subgrad_b[i] = sum_{k: c_yb[k] near max} y[k-i]
        
        # Vectorized: for each i, subgrad_b[i] = sum over active k of y[k-i]
        # This is a correlation!
        active_mask = c_yb >= 0.99 * c_max
        active_indices = np.where(active_mask)[0]
        
        if len(active_indices) > 0:
            # subgrad_b[i] = sum_{k in active} y[k-i] if 0 <= k-i < n
            # = (y_reversed * active_indicator) correlated with something...
            # Actually: subgrad_b[i] = sum_k active[k] * y[k-i]
            # This is the cross-correlation of 'active' (as a signal) with y, evaluated at i
            # Let me compute it via FFT
            act_ext = np.zeros(fft_size)
            act_ext[active_indices] = 1.0
            Y_act = rfft(act_ext, fft_size)
            # We want sum_k act_ext[k] * y[k-i] for i = 0..n-1
            # This is the cross-correlation: (act_ext * y_reversed) in freq domain...
            # Actually: sum_k A[k] * Y[k-i] = correlation
            # In FFT: if we define the DFT, correlation is A_conj * Y in freq domain
            # But we want a "valid" correlation. Let me just use direct computation for small n
            # or FFT for large n
            
            if n <= 2000:
                # Direct: for each i, sum over k in active of y[k-i]
                sb = np.zeros(n)
                for k in active_indices[:50]:  # Limit to top 50 active
                    js = max(0, k - n + 1)
                    je = min(n - 1, k)
                    # sb[i] += y[k-i] for i in [js, je]
                    # y[k-i] for i in [js, je] is y[k-js : k-je+1][::-1] = y[je:js-1:-1]
                    # Actually: when i=js, k-i = k-js; when i=je, k-i = k-je
                    # So y[k-i] = y[je : js-1 : -1] = y[k-je : k-js-1 : -1]
                    # Length = je - js + 1
                    y_slice = y[peak_k - je + 1 : peak_k - js + 1][::-1] if k == peak_k else y[k-je+1:k-js+1][::-1]
                    # Hmm this is getting complicated. Let me just use a simple approach.
                    # sb[i] += y[k-i] for valid i
                    valid_i = np.arange(js, je+1)
                    sb[valid_i] += y[k - valid_i]
                b_candidate *= (1 - eta * sb / (sb.max() + 1e-30))
            else:
                # For large n, just reduce b at positions near the peak
                b_candidate[peak_k-je+1:peak_k-js+1] *= (1 - eta)
        
        b_candidate = np.maximum(b_candidate, 0)
        s_b = b_candidate.sum()
        if s_b > 0:
            b_candidate = b_candidate / s_b
        
        # Update y and a (Nesterov-like)
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_candidate
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 0:
            y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_candidate) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa > 0:
            a = a / sa
        
        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovAlt iter {it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def try_update(best_state, a, label):
    """Try to update best state."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        old = best_state['score']
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)}, was {old:.6f})")
        return True
    return False
