def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

    n = len(a)

    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

    s = b_val.sum()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based alternating minimization

Key improvements over previous approach:
- LP exact best-response per alternating step (faster convergence than FW subgradient)
- Vectorized convolution matrix construction
- Multi-phase time allocation with adaptive budget
- Random perturbation restarts to escape local minima
- Fine-grained coordinate descent with incremental O(n) convolution updates

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a):
    """
    Given fixed a (sum=1), solve LP:
      minimize M
      subject to: conv(a,b)[k] <= M for all k
                  sum(b) = 1, b >= 0
    Returns optimal b or None on failure.
    """
    import cvxpy as cp
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve()
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val /= s
    return b_val


def fw_best_response(a, b_init, n_steps=500):
    """Frank-Wolfe best response (fallback when LP unavailable)."""
    n = len(a)
    b = b_init.astype(np.float64).copy()
    s = b.sum()
    if s < 1e-15:
        b = np.ones(n) / n
    else:
        b /= s
    b_conv = convolve_ab(a, b)
    for step in range(n_steps):
        k_star = int(np.argmax(b_conv))
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        grad_slice = a[k_star - i_end:k_star - i_start + 1][::-1]
        j_star = i_start + int(np.argmin(grad_slice))
        t = 2.0 / (step + 2.0)
        b_conv = (1.0 - t) * b_conv
        ks_s = j_star
        ks_e = min(ks_s + n - 1, 2 * n - 2)
        b_conv[ks_s:ks_e + 1] += t * a
        b = (1.0 - t) * b
        b[j_star] += t
    return b
