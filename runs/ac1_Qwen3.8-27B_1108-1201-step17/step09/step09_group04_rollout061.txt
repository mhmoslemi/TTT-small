import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def conv_AB(a, b):
    """Linear convolution of a and b via FFT."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def get_lp_solver():
    # Try solvers in order of preference
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_best_response(a_fixed, solver):
    """Given fixed a, find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint: conv(a_fixed, b_var)[k] <= M for all k
    # conv(a, b)[k] = sum_i a[i] * b[k-i]
    # We need to build this as a linear expression in b_var
    # For each k: sum_{i:max(0,k-n+1)}^{min(n-1,k)} a[i] * b[k-i] <= M
    
    # Efficient way: build the toeplitz-like matrix
    # Row k of the matrix has a[i] at column (k-i) for valid i
    rows = []
    for k in range(2 * n - 1):
        # b[j] appears in conv[k] with coefficient a[k-j] where 0<=k-j<n, i.e., k-n+1<=j<=k
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # So sum_{j=js}^{je} a[k-j] * b[j] <= M
        coeff = np.zeros(n)
        for j in range(js, je + 1):
            coeff[j] = a_fixed[k - j]
        rows.append(coeff)
    
    T = np.array(rows)  # shape (2n-1, n)
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

    """Solve: min M s.t. T@b <= M, sum(b)=1, b>=0"""
    n = len(a)
    # Variables: [b_0, ..., b_{n-1}, M]
    # Objective: minimize M → c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # A_ub @ [b, M] <= b_ub
    A_ub = np.hstack([T, -np.ones((2*n-1, 1))])

    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b_i >= 0, M >= 0 (M is naturally >= 0)
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    M_val = res.x[n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_val)

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using accelerated alternating LP
(with HiGHS), spectral-mode diversification, multi-length search, p-norm
smoothing refinement, adaptive Nesterov with smart restarts, and structured
peak shaving. Key innovations: faster LP via scipy HiGHS, frequency-domain
perturbations for escape, length trimming, and aggressive convergence detection.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return float(2.0 * len(a) * c.max() / (s * s))


def evaluate_safe(a):
    """Evaluate using the external evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def build_T(a):
    """Build matrix T such that T @ b = convolve(a, b). Shape: (2n-1, n)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for k in range(n):
        T[k, :k + 1] = a[k:-1:-1]
    for k in range(n, 2 * n - 1):
        T[k, k - n + 1:] = a[n - 1:k - n:-1]
    return T


def solve_br_lp(T, n):
    """Solve best-response LP: min M s.t. T@b <= M, sum(b)=1, b>=0.
    Returns (b_val, M_val) or (None, None)."""
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([T, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None, None
    if res.status != 0:
        return None, None
    b_val = res.x[:n].copy()
    M_val = float(res.x[n])
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val /= s
    return b_val, M_val


def nesterov_lp_optimize(a0, time_limit, restart_interval=400, max_iters=500000):
    """Nesterov dual-averaging LP with periodic restarts and perturbation."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a /= a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    t = 0
    stall_count = 0
    prev_best = best_score

    while time.monotonic() - t0 < time_limit and t < max_iters:
        T = build_T(y)
        b_new, M_val = solve_br_lp(T, n)
        if b_new is None:
            break

        alpha = 2.0 / (t + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y /= sy

        a = ((t + 1) * a + alpha * (t + 2) * b_new) / (t + 1 + alpha * (t + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a /= sa

        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall_count = 0
        else:
            if t % restart_interval == 0 and score > prev_best + 1e-10:
                stall_count += 1
            else:
                stall_count = 0

        if t % restart_interval == 0:
            if best_score < prev_best - 1e-8:
                prev_best = best_score
                stall_count = 0
            else:
                stall_count += 1

                A = rfft(y, 2 * n)
                A_angle = np.angle(A) + rng.standard_normal(len(A)) * 0.05 * scale
                A_mag = np.abs(A) * (1 + rng.standard_normal(len(A)) * 0.03 * scale)
                A_pert = A_mag * np.exp(1j * A_angle)

def p_norm_score_z(z, p):
    """Compute 2n * (sum c_k^p)^(1/p) where a = softmax(z), c = conv(a,a)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-30)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def p_norm_grad_z(z, p):
    """Gradient of p_norm_score_z w.r.t. z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-30)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def p_norm_lbfgs(a0, time_limit, p_start=20.0, p_end=500.0, n_stages=8):
    """L-BFGS with adaptive p-norm smoothing in log-domain."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z = np.log(a_safe / a_safe.max())
    t0 = time.monotonic()
    temps = np.geomspace(p_start, p_end, n_stages)
    iters_per_stage = max(80, 6000 // n_stages)

    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_val):
            return p_norm_score_z(z_val, stage_p)

        def jac(z_val):
            return p_norm_grad_z(z_val, stage_p)

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': iters_per_stage * 100,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        score = compute_score(a_cur)
        elapsed = time.monotonic() - t0
        if stage_p >= p_end / 2 or n_stages <= 3:
            print(f"      p-norm p={stage_p:.1f} ({elapsed:.0f}s): score={score:.6f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


def peak_shaving(a0, time_limit, n_steps=50000):
    """Local refinement: reduce top peaks via coordinate adjustments."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a /= s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_work = c.copy()
        if len(c_work) > 2:
            c_work[0] = -np.inf
            c_work[-1] = -np.inf

        peak_idx = np.argmax(c_work)
        peak_val = c_work[peak_idx]
        threshold = peak_val * 0.92
        near_peaks = np.where(c_work >= threshold)[0][:6]

        # Compute contribution scores
        contrib = np.zeros(n)
        for k in near_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                contrib[js:je + 1] += a[js:je + 1] * a[k - je:k - js]

        top_i = int(np.argmax(contrib))
        improved = False

        # Try reducing top contributor
        for factor in [0.96, 0.93, 0.90, 0.86, 0.82, 0.78]:
            diff = a[top_i] * (1 - factor)
            if diff < 1e-16:
                continue
            a_trial = a.copy()
            a_trial[top_i] = max(0, a_trial[top_i] - diff)
            mask = np.ones(n, dtype=bool)
            mask[top_i] = False
            others = a_trial[mask].sum()
            if others < 1e-15:
                continue
            a_trial[mask] += diff * a_trial[mask] / others
            sc = compute_score(a_trial)
            if sc < best_score - 1e-15:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True
                break

        if not improved:
            # Try pair adjustment at peak
            k = peak_idx
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js < je:
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[k - idx]
                top_ci = int(np.argmax(contribs))
                i = idx[top_ci]
                j = k - i
                if 0 <= j < n and j != i:
                    for f1 in [0.94, 0.90, 0.86]:
                        for f2 in [0.94, 0.90, 0.86]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            st = a_trial.sum()
                            if st < 0.01:
                                continue
                            a_trial /= st
                            sc = compute_score(a_trial)
                            if sc < best_score - 1e-15:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break

        if not improved:
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def trim_sequence(a, max_trim=200):
    """Try trimming small values from ends to reduce n (improves score)."""
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    for trim in range(1, max_trim + 1):
        if trim * 2 >= n:
            break
        a_trimmed = a[trim:n - trim]
        s = a_trimmed.sum()
        if s < 0.01:
            break
        a_norm = a_trimmed / s
        sc = compute_score(a_norm)
        if sc < best_score:
            best_score = sc
            best_a = a_norm.copy()
        else:
            if sc > best_score * 1.02:
                break
    return best_a, best_score


def structured_init(n, variant=0):

    if variant == 0:

    elif variant == 2:

        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 3:
        x = np.linspace(0, 1, n)

    else:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2.5) * (0.7 + 0.3 * np.cos(2 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a

        return False

    # === Phase 1: Nesterov LP (main workhorse) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 550.0)
    if t_alloc > 30:
        print(f"  Budget: {talloc:.0f}s" if False else f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_optimize(best_state['seq'], t_alloc, restart_interval=500)
        try_update(a_lp, "nesterov_lp_1")

    # === Phase 2: p-norm L-BFGS refinement ===
    print(f"\n=== Phase 2: p-norm L-BFGS (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        a_pn = p_norm_lbfgs(best_state['seq'], t_alloc, p_start=30.0, p_end=400.0, n_stages=7)
        try_update(a_pn, "pnorm_1")

    # === Phase 3: Nesterov LP round 2 ===
    print(f"\n=== Phase 3: Nesterov LP r2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        a_lp2, _ = nesterov_lp_optimize(best_state['seq'], t_alloc, restart_interval=600)
        try_update(a_lp2, "nesterov_lp_2")

    # === Phase 4: Length exploration ===
    print(f"\n=== Phase 4: Length Exploration (best={best_state['score']:.6f}) ===")
    candidate_lengths = set()
    base = len(best_state['seq'])
    for delta in [-64, -48, -32, -16, 16, 32, 48, 64]:
        alt_n = base + delta
        if 400 <= alt_n <= 2500:
            candidate_lengths.add(alt_n)
    for rn in [768, 896, 1024, 1152, 1280, 1536]:
        if 400 <= rn <= 2500:
            candidate_lengths.add(rn)
    candidate_lengths = sorted(candidate_lengths, key=lambda x: abs(x - base))

    n_explored = 0
    for alt_n in candidate_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 100 or n_explored >= 3:
            break
        n_explored += 1
        t_total = min(rem * 0.15, 150.0)
        if t_total < 40:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_state['seq'], alt_n)
        # Quick Nesterov LP
        t_lp = t_total * 0.7
        a_alt, _ = nesterov_lp_optimize(init_alt, t_lp, restart_interval=300)
        improved = try_update(a_alt, f"lp_n{alt_n}")
        if improved:
            rem = deadline - time.monotonic()
            if rem > 30:
                t_pn = min(rem * 0.10, 60.0)
                a_alt2 = p_norm_lbfgs(best_state['seq'], t_pn, p_start=50.0, p_end=500.0, n_stages=4)
                try_update(a_alt2, f"pn_n{alt_n}")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        a_ps, _ = peak_shaving(best_state['seq'], t_alloc, n_steps=30000)
        try_update(a_ps, "peak_shave_1")

    # === Phase 6: Perturbation + re-optimization ===
    print(f"\n=== Phase 6: Perturbation (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_state['seq'])
        for noise_scale in [0.04, 0.08, 0.15]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            # Spectral perturbation
            A = rfft(best_state['seq'], 2 * n_cur)
            A_pert = A * (1 + noise_scale * rng.standard_normal(len(A)))
            perturbed = irfft(A_pert, 2 * n_cur)[:n_cur]
            perturbed = np.maximum(perturbed, 0)
            sp = perturbed.sum()
            if sp > 0.01:
                perturbed /= sp
                t_sub = min(rem * 0.3, 80.0)
                a_pert, _ = nesterov_lp_optimize(perturbed, t_sub, restart_interval=300)
                try_update(a_pert, f"pert_{noise_scale}")

    # === Phase 7: Final Nesterov LP + p-norm ===
    print(f"\n=== Phase 7: Final Optimization (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_lp_fin = min(rem * 0.30, 100.0)
        a_fin, _ = nesterov_lp_optimize(best_state['seq'], t_lp_fin, restart_interval=400)
        try_update(a_fin, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_pn_fin = min(rem * 0.30, 80.0)
        a_pn_fin = p_norm_lbfgs(best_state['seq'], t_pn_fin, p_start=100.0, p_end=1000.0, n_stages=4)
        try_update(a_pn_fin, "final_pn")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_ps2, _ = peak_shaving(best_state['seq'], min(rem * 0.4, 40.0), n_steps=10000)
        try_update(a_ps2, "final_ps")

    # === Phase 8: Length trimming ===
    print(f"\n=== Phase 8: Length Trimming (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        a_trimmed, _ = trim_sequence(best_state['seq'], max_trim=100)
        try_update(a_trimmed, "trimmed")

    # Final verification with evaluate_sequence
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs
        print(f"\nFINAL: score={best_state['score']:.6f}, n={len(best_state['seq'])}")

    return [float(x) for x in best_state['seq']]
