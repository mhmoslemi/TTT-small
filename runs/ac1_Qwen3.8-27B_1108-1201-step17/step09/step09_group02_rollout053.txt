    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = 2*n*float(fast_convolve(a).max())
    
    t0 = time.monotonic()
    
    # Annealing schedule
    n_levels = 20
    betas = np.geomspace(beta_start, beta_end, n_levels)
    
    for beta in betas:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Run several iterations at this beta
        n_iters = max(50, n * 5)
        lr = 0.1 / n  # initial learning rate
        
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            b = fast_convolve(a)
            # Smooth max gradient
            b_shifted = b - b.max()
            exp_b = np.exp(beta * b_shifted)
            p = exp_b / exp_b.sum()
            
            # Gradient = 2 * cross_correlate(p, a)
            grad = 2 * cross_correlate(p, a)[:n]
            
            # Gradient step with projection
            lr = 1.0 / (1.0 + lr)  # simple decay
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:

            a_new = a_new / s
            
            # Check improvement

            score = 2*n*float(b_new.max())
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            
            a = a_new

            elapsed = time.monotonic() - t0
            print(f"    SGD beta={beta:.0f} ({elapsed:.0f}s): best={best_score:.8f}")

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with annealed softmax (main innovation)
2. Exact coordinate descent with golden section search
3. LP-based fictitious play
4. Basin hopping with structured perturbations
5. Multi-length exploration

Key insight: The gradient of smooth_max(conv(a,a), beta) w.r.t. a_i is
2 * (p cross-correlate a)_i where p = softmax(beta * conv(a,a)).
This enables O(n log n) gradient steps via FFT, enabling many more
iterations than coordinate descent in the same time.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def next_power_of_two(x):
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = next_power_of_two(size)
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_correlate_fft(p, a):
    """Compute c_i = sum_k p[k] * a[k-i] for i = 0,...,len(a)-1 using FFT."""
    n = len(a)
    N = next_power_of_two(len(p) + n - 1)
    P = rfft(p, N)
    A = rfft(a, N)
    c = irfft(P * np.conj(A), N)
    return c[:n]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def init_flat_spectrum(n, rng):
    n_fft = next_power_of_two(n)
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrales = amps * np.exp(1j * phases)
    a = irfft(spectrales, n_fft)
    a = np.abs(a[:n])
    return normalize(a)

def init_cosine(n, rng):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.3, 0.6)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    return normalize(a)

def init_fourier_mix(n, rng):
    n_harmonics = rng.integers(5, 20)
    amps = rng.random(n_harmonics) * 0.3
    phases = rng.uniform(0, 2.0 * np.pi, n_harmonics)
    freqs = rng.choice(np.arange(1, n // 2 + 1), size=min(n_harmonics, n//2), replace=False)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(len(freqs)):
        a += amps[j] * np.cos(2.0 * np.pi * freqs[j] * t / n + phases[j])
    return normalize(a)

def smooth_gradient_descent(a0, time_limit, beta_start=1.0, beta_end=5000.0, 
                             n_levels=25, verbose=False, label=""):
    """Smooth gradient descent with annealed softmax."""
    n = len(a0)
    a = normalize(a0.copy())
    
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    
    t0 = time.monotonic()
    betas = np.geomspace(beta_start, beta_end, n_levels)
    
    for level, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit:
            break
        
        rem = time_limit - (time.monotonic() - t0)
        # Allocate time across levels (more time for higher beta)
        time_this_level = rem / (n_levels - level)
        
        # Learning rate schedule for this level
        lr = 0.05 / np.sqrt(n)
        n_max_iters = int(time_this_level / 0.005) + 50  # rough estimate
        
        stalling = 0
        for it in range(n_max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            b = fast_convolve(a)
            b_max = b.max()
            
            # Softmax (numerically stable)
            b_shifted = b - b_max
            exp_b = np.exp(beta * b_shifted)
            exp_sum = exp_b.sum()
            if exp_sum < 1e-300:
                p = np.full(len(b), 1.0 / len(b))
            else:
                p = exp_b / exp_sum
            
            # Gradient via cross-correlation
            grad = 2.0 * cross_correlate_fft(p, a)
            
            # Gradient step
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            
            # Evaluate
            b_new = fast_convolve(a_new)
            score = 2.0 * n * float(b_new.max())
            
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_new.copy()
                stalling = 0
            else:
                stalling += 1
            
            a = a_new
            
            # Adapt learning rate
            if stalling > 20:
                lr *= 0.8
                stalling = 0
            
            if it % 100 == 99:
                lr *= 0.95
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    SGD {label} beta={beta:.0f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def golden_section_search(f, lo, hi, max_iter=40):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent with golden section search."""
    n = len(a0)
    a = normalize(a0.copy())
    
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            if has_quad:
                quad_idx = 2 * i - k_lo

            def compute_score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                              _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                              _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(2.0 * v, 3.0 / n)
            best_x = golden_section_search(compute_score, 0.0, upper, max_iter=35)

            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

    return best_a, best_score

def subgradient_step(a, conv, lr, n_peaks=15):
    n = len(a)
    k_act = min(n_peaks, 2 * n - 1)
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks]
    temp = 0.005 * max(pv.max(), 1e-10)
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        return a
    w = w / ws
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        grad[j] += 2.0 * w[idx] * a[k - j]
    a_new = simplex_project(a - lr * grad)
    return a_new

def basin_hopping(a0, time_limit, cd_func, sgd_func, rng, n_perturbs=10, verbose=False):
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break
        t_alloc = min(rem * 0.5, 120.0)
        t_sgd = t_alloc * 0.5
        t_cd = t_alloc * 0.5

        # Perturb
        sigma = 0.02 + 0.04 * rng.random()
        perturbed = a + rng.standard_normal(n) * sigma
        perturbed = normalize(perturbed)

        # Quick SGD
        a_sgd, s_sgd = sgd_func(perturbed, t_sgd, beta_start=100.0, beta_end=5000.0, n_levels=10, label=f"bh{p}_sgd")
        # CD polishing
        a_cd, s_cd = cd_func(a_sgd, t_cd, label=f"bh{p}_cd")
        
        if s_cd < best_score:
            best_score = s_cd
            best_a = a_cd.copy()
            if verbose:
                print(f"    BH p={p}: improved to {best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)

def check_and_update(a, best_seq, best_score, label=""):
    a = normalize(a)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return best_seq, best_score
