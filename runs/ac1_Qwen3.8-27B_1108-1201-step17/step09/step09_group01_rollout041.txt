    n = len(v)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_ab(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = int(np.nonzero(cond)[0][-1])
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def init_cosine(n, alpha=1.0):
    """Cosine-shaped: high in middle, zero at edges."""
    t = np.linspace(0, 1, n)
    a = np.cos(np.pi * (t - 0.5)) ** (2.0 * alpha)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def init_gaussian(n, sigma_factor=0.3):
    """Gaussian centered at middle."""
    t = np.arange(n, dtype=np.float64)
    center = n / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s

    """Power-law shape: (1 - |2t-1|^p)^(1/p)."""
    t = np.linspace(0, 1, n)

    s = a.sum()
    if s < 1e-30:

    return a / s

    # For real sequences, we use a Chirp-like initialization

                a[i] = max(ai + best_delta, 0.0)
                # Update conv
                conv[k_start:k_end + 1] += 2.0 * best_delta * a
                if 2 * i <= 2 * n - 2:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_M
                score = 2.0 * n * M
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a.copy()
                improved = True

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweep} ({elapsed:.1f}s): best={best_score:.8f} M={M:.10f}")

        if not improved:
            no_improve += 1
            if no_improve >= 50:
                break
        else:
            no_improve = 0

    return best_a, best_score, sweep


def coord_descent_fine(a0, time_limit, verbose=False, label=""):
    """Very fine coordinate descent with tiny multiplicative steps for polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = i + n - 1

            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected_conv = conv[k_start:k_end + 1]
            w = 2.0 * a

            best_delta = 0.0
            best_new_M = M

            # Fine multiplicative steps
            for factor in [0.999, 0.998, 0.997, 0.995, 0.993, 0.99, 0.985, 0.98,
                           1.001, 1.002, 1.003, 1.005, 1.007, 1.01, 1.015, 1.02]:
                d = ai * (factor - 1.0)
                new_val = ai + d
                if new_val < -1e-15 or new_val > 1.0:
                    continue

                new_affected = affected_conv + 2.0 * d * w
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * d * ai + d * d

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_M - 1e-16:
                    best_new_M = new_max
                    best_delta = d

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += 2.0 * best_delta * a
                if 2 * i <= 2 * n - 2:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_M
                score = 2.0 * n * M
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a.copy()
                improved = True

        if not improved and sweep > 20:
            break

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD {label} sweep={sweep} ({elapsed:.1f}s): best={best_score:.8f}")

    return best_a, best_score, sweep


def smooth_mirror_descent(a0, time_limit, beta_start=20.0, beta_end=300.0,
                          lr_start=0.005, lr_end=0.0005, verbose=False, label=""):
    """
    Mirror descent with smooth max (logsumexp) and Nesterov acceleration.
    Uses FFT for efficient gradient computation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    y = a.copy()  # Nesterov lookahead
    prev_a = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    momentum = 0.9

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = (time.monotonic() - t0) / time_limit

        # Anneal parameters
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # Nesterov: y = a + momentum * (a - prev_a)
        y = a + momentum * (a - prev_a)
        prev_a = a.copy()

        # Project y to simplex
        y_proj = simplex_project(np.maximum(y, 0))

        # Compute conv and smooth weights
        conv_y = fast_convolve(y_proj)
        max_conv = float(conv_y.max())

        # Logsumexp weights
        scaled = beta * (conv_y - max_conv)
        scaled = np.clip(scaled, -50.0, 0.0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[np.argmax(conv_y)] = 1.0
            Z = 1.0
        weights = weights / Z

        # Gradient via FFT cross-correlation
        # grad[i] = 2 * sum_j y_proj[j] * weights[i+j]
        # = 2 * np.correlate(weights, y_proj, 'full')[:n]
        # Use FFT for speed
        fft_size = 1 << ((2 * n - 1) - 1).bit_length()
        W = rfft(weights, fft_size)
        Y = rfft(y_proj, fft_size)
        # Cross-correlation: sum_j weights[i+j] * y_proj[j]
        # = IFFT(FFT(weights) * conj(FFT(y_proj))) shifted
        # Actually, let's use the direct formula:
        # corr[i] = sum_j weights[i+j] * y_proj[j]
        # This is the convolution of weights with reversed y_proj, shifted
        # corr = irfft(W * conj(Y), fft_size) gives sum_j weights[j] * y_proj[j-i]... 
        # Let me just use np.correlate for correctness (it's fast enough for n<=4096)
        corr = np.correlate(weights, y_proj, mode='full')[:n]
        grad = 2.0 * corr

        # Mirror descent update
        log_a = np.log(a + 1e-30) - lr * grad
        a_new = np.exp(log_a - log_a.max())
        a = simplex_project(a_new)

        # Update best
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and it % 50 == 0:
            print(f"    SMD {label} it={it} beta={beta:.1f} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_pressure_update(a, strength=0.03, n_peaks=20):
    """
    Reduce coordinates that contribute most to peak convolution values.
    This is a structured perturbation targeting the peaks.
    """
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()

    # Get top peak positions
    peak_threshold = M * 0.95
    peak_positions = np.where(conv >= peak_threshold)[0]
    if len(peak_positions) > n_peaks:
        peak_positions = peak_positions[np.argsort(conv[peak_positions])[-n_peaks:]]

    # Compute pressure on each coordinate
    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            product = a[i] * a[j]
            pressure[i] += product
            if j != i:
                pressure[j] += product * 0.5

    p_max = pressure.max()
    if p_max < 1e-20:
        return a.copy()
    pressure = pressure / p_max

    # Apply: reduce high-pressure, boost low-pressure
    a_new = a * (1.0 - strength * pressure + 0.5 * strength * (1.0 - pressure))
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def random_kick(a, rng, strength=0.1):
    """Random perturbation to escape local optima."""
    n = len(a)
    a_new = a.copy()
    mask = rng.random(n) < 0.15
    a_new[mask] *= (1.0 + rng.standard_normal(mask.sum()) * strength)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate via evaluate_sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    
    Phase 1: Diverse multi-length exploration
    Phase 2: Intensive coordinate descent with periodic kicks
    Phase 3: Smooth mirror descent refinement
    Phase 4: Fine polishing
    Phase 5: Final verification and alternative lengths
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(12345)

    best_seq = None
    best_score = float('inf')

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Diverse multi-length exploration ===
    print("\n=== Phase 1: Diverse Exploration ===")
    t1_start = time.monotonic()
    p1_budget = 180.0

    n_values = [768, 1024, 1536, 2048]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    candidates = []

    for n in n_values:
        elapsed = time.monotonic() - t1_start
        if elapsed > p1_budget:
            break

        init_cands = []
        if parent is not None:
            init_cands.append(("parent", interpolate_seq(parent, n)))
        init_cands.append(("cos1", init_cosine(n, alpha=1.0)))
        init_cands.append(("cos2", init_cosine(n, alpha=2.0)))
        init_cands.append(("cos3", init_cosine(n, alpha=0.5)))
        init_cands.append(("gauss", init_gaussian(n, sigma_factor=0.25)))
        init_cands.append(("gauss_w", init_gaussian(n, sigma_factor=0.4)))
        init_cands.append(("power2", init_power(n, p=2.0)))
        init_cands.append(("power4", init_power(n, p=4.0)))
        init_cands.append(("superflat", init_superflat(n, rng)))
        init_cands.append(("smooth_rand", init_random_smooth(n, rng, n_harmonics=6)))
        init_cands.append(("uniform", np.ones(n) / n))

        for name, cand in init_cands:
            rem = p1_budget - (time.monotonic() - t1_start)
            if rem < 3:
                break
            t_alloc = max(3.0, min(8.0, rem / 4.0))
            try:
                a_opt, s_opt, _ = coord_descent_optimal(cand, t_alloc, label=f"p1_{name}_n{n}")
                candidates.append((s_opt, a_opt, n, name))
            except Exception as e:
                print(f"    Error in {name}_n{n}: {e}")

    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates")
    for c in candidates[:5]:
        print(f"    score={c[0]:.8f} n={c[2]} init={c[3]}")

    for s_c, a_c, n_c, it_c in candidates[:10]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{it_c}_n{n_c}")

    # === Phase 2: Intensive CD with periodic kicks ===
    print(f"\n=== Phase 2: Intensive CD + Kicks (best={best_score:.6f}) ===")
    t2_start = time.monotonic()
    p2_budget = 700.0

    if best_seq is not None:
        n_best = len(best_seq)
        current = best_seq.copy()
        kick_strength = 0.08

        round_idx = 0
        while time.monotonic() - t2_start < p2_budget:
            round_idx += 1
            rem = p2_budget - (time.monotonic() - t2_start)
            if rem < 15:
                break
            t_alloc = min(rem * 0.4, 80.0)
            print(f"  CD round {round_idx}, n={n_best}, budget={t_alloc:.0f}s, score={best_score:.6f}")

            a_opt, s_opt, sweeps = coord_descent_optimal(
                current, t_alloc, verbose=(round_idx % 3 == 0), label=f"ph2r{round_idx}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ph2_cd_r{round_idx}")

            # Check if we should kick
            if s_opt >= best_score * 0.9999 and round_idx % 2 == 0:
                # Stagnation detected, apply kick
                current = random_kick(best_seq, rng, strength=kick_strength)
                kick_strength *= 0.7
                if kick_strength < 0.005:
                    current = peak_pressure_update(best_seq, strength=0.02)
            else:
                current = best_seq.copy()

            # Also try mirror descent occasionally
            if round_idx % 3 == 0 and rem > 30:
                md_budget = min(rem * 0.15, 40.0)
                a_md, s_md = smooth_mirror_descent(
                    best_seq, md_budget,
                    beta_start=15.0, beta_end=100.0,
                    lr_start=0.003, lr_end=0.0003,
                    label=f"ph2_md_r{round_idx}")
                best_seq, best_score = check_and_update(a_md, best_seq, best_score, f"ph2_md_r{round_idx}")

    # === Phase 3: Focused mirror descent ===
    print(f"\n=== Phase 3: Mirror Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            md_budget = min(rem * 0.5, 200.0)
            print(f"  SMD n={len(best_seq)}, budget={md_budget:.0f}s")
            a_md, s_md = smooth_mirror_descent(
                best_seq, md_budget,
                beta_start=10.0, beta_end=200.0,
                lr_start=0.004, lr_end=0.0002,
                verbose=True, label="ph3")
            best_seq, best_score = check_and_update(a_md, best_seq, best_score, "ph3_smd")

            # Second MD with different params
            rem = deadline - time.monotonic() - 250
            if rem > 20:
                md2_budget = min(rem * 0.4, 100.0)
                a_md2, s_md2 = smooth_mirror_descent(
                    best_seq, md2_budget,
                    beta_start=50.0, beta_end=500.0,
                    lr_start=0.002, lr_end=0.0001,
                    label="ph3b")
                best_seq, best_score = check_and_update(a_md2, best_seq, best_score, "ph3b_smd")

    # === Phase 4: Intensive fine polishing ===
    print(f"\n=== Phase 4: Fine Polishing (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            fine_budget = min(rem * 0.5, 150.0)
            print(f"  FineCD n={len(best_seq)}, budget={fine_budget:.0f}s")
            a_fine, s_fine, _ = coord_descent_fine(
                best_seq, fine_budget, verbose=True, label="ph4_fine")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "ph4_fine1")

        rem = deadline - time.monotonic() - 100
        if rem > 15:
            cd_budget = min(rem * 0.5, 80.0)
            a_cd, s_cd, _ = coord_descent_optimal(
                best_seq, cd_budget, label="ph4_cd")
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "ph4_cd")

    # === Phase 5: Try alternative lengths ===
    print(f"\n=== Phase 5: Alt Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = sorted(set([n_best - 128, n_best, n_best + 128, n_best + 256]),
                             key=lambda x: abs(x - n_best))
        for alt_n in alt_lengths:
            if alt_n < 256 or alt_n > 4096:
                continue
            rem = deadline - time.monotonic() - 50
            if rem < 15:
                break
            t_alt = min(rem * 0.3, 40.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, s_alt, _ = coord_descent_optimal(init_alt, t_alt, label=f"alt{alt_n}")
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

    # === Phase 6: Final peak pressure refinement ===
    print(f"\n=== Phase 6: Final Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            # Apply peak pressure then fine CD
            a_pp = peak_pressure_update(best_seq, strength=0.01)
            t_ff = min(rem * 0.4, 30.0)
            a_ff, s_ff, _ = coord_descent_fine(a_pp, t_ff, label="ph6")
            best_seq, best_score = check_and_update(a_ff, best_seq, best_score, "ph6_pp_fine")

        # Final fine polish on best
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_final, s_final, _ = coord_descent_fine(
                best_seq, min(rem * 0.7, 25.0), label="ph6_final")
            best_seq, best_score = check_and_update(a_final, best_seq, best_score, "ph6_final")

    # Fallback
    if best_seq is None:
        best_seq = init_cosine(1024, alpha=1.0)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
