def get_lp_solver():

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return 1e10
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        idx = np.nonzero(cond)[0][-1]
        theta = (cssv[idx] - 1.0) / (idx + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def build_conv_matrix_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    return rows, cols, vals

def solve_lp_best_response(a_fixed, solver=None):

    rows, cols, vals = build_conv_matrix_sparse(a_fixed)
    import scipy.sparse as sp
    T_mat = sp.coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    constraints = [
        T_mat @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for sol in [solver, cp.GLPK, cp.CBC, cp.ECOS, cp.PDLP, cp.SCS]:

    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

def golden_section_search(f, lo, hi, max_iter=45):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            quad_idx = 0
            if has_quad:
                quad_idx = 2 * i - k_lo

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            upper = max(3.0 * v, 4.0 / n)

            def _score_fn(x):
                ns = current_sum + (x - v)
                if ns < 0.01:
                    return 1e10
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                nm = max(unaff_max, float(nv.max()))
                return 2.0 * n * nm / (ns * ns)

            best_x = golden_section_search(_score_fn, 0.0, upper, max_iter=40)

            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes
                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                return 1e10
            cand = cand / cs
            return compute_score(cand)

        best_t = golden_section_search(_lp_score_fn, 0.0, 1.0, max_iter=30)
        if best_t < 1e-6:

            best_a = a_new.copy()
        a = a_new

        # Running average
        a_avg = (iters * a_avg + a) / (iters + 1)
        avg_score = compute_score(a_avg)
        if avg_score < best_score:
            best_score = avg_score
            best_a = a_avg.copy()

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_val:.6f}")

    return best_a, best_score, iters

    no_improve = 0
    conv = fast_convolve(a)

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Find active peaks (top-k lags)
        k_peaks = min(20, 2 * n - 1)
        peaks = np.argpartition(conv, -k_peaks)[-k_peaks:]
        pv = conv[peaks]
        temp = 0.003 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            break
        w = w / ws

        # Compute subgradient
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            partners_idx = k - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi + 1] += 2.0 * w[idx] * a[partners_idx]

        a_new = simplex_project(a - lr * grad)
        conv = fast_convolve(a_new)
        sc = 2.0 * n * float(conv.max())
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.8
        a = a_new
        lr *= 0.9998

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG it={it} ({elapsed:.0f}s): best={best_score:.8f}, lr={lr:.6f}")

    return best_a, best_score

def basin_hopping(a0, time_limit, rng, n_perturbs=6, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break
        t_alloc = min(rem * 0.5, 50.0)

        # Multiple perturbation strategies
        strategy = p % 3
        if strategy == 0:
            # Random noise
            sigma = 0.02 + 0.04 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif strategy == 1:
            # Peak-directed: reduce peaks, increase troughs in a
            conv_p = fast_convolve(a)
            mid = n - 1
            # Scale a to reduce center correlation
            center_weight = np.ones(n)
            center_weight[np.abs(np.arange(n) - mid) < n // 4] = 0.7
            perturbed = a * center_weight + rng.standard_normal(n) * 0.02
        else:
            # Mirror perturbation: flip a portion
            flip_frac = rng.uniform(0.2, 0.5)
            flip_n = int(n * flip_frac)
            start = rng.integers(0, n - flip_n)
            perturbed = a.copy()
            perturbed[start:start + flip_n] = a[start + flip_n - 1:start - 1:-1] if start + flip_n <= n else a[start + flip_n - 1:start - 1:-1]
            perturbed += rng.standard_normal(n) * 0.01

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc, label=f"bh{p}")
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p} strat={strategy}: improved to {best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_ramp(n, rng):
    t = np.arange(n, dtype=np.float64)
    alpha = rng.uniform(-0.8, 0.8)
    a = 1.0 + alpha * (2.0 * t / (n - 1) - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_cosine_asym(n, rng):
    t = np.arange(n, dtype=np.float64)
    center = n * rng.uniform(0.3, 0.7)
    width = n * rng.uniform(0.2, 0.5)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

def propose_candidate():

    return [float(x) for x in best_seq]
