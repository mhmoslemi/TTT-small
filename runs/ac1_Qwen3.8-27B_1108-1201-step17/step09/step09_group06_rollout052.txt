16. **A new idea - using the eigenstructure of the convolution operator**:

The convolution with $a$ is a Toeplitz operator. Let $T_a$ be the Toeplitz operator with symbol $\hat{a}(\omega)$. Then $(a*a)_k$ is related to the diagonal of $T_a^2$ (or $T_a T_a^*$).

If we could diagonalize $T_a$, we could directly control its eigenvalues. But for a general Toeplitz matrix, this is computationally expensive.

However, for large $n$, the Toeplitz operator is approximately a multiplication operator in the Fourier basis. So the eigenvalues of $T_a$ are approximately $|\hat{a}(\omega_m)|$ for $m = 0, \ldots, n-1$.

This means that if $|\hat{a}(\omega_m)|$ is approximately constant (say $C$), then the eigenvalues of $T_a$ are approximately $C$, and the eigenvalues of $T_a^2$ are approximately $C^2$. The diagonal entries of $T_a^2$ (which are the $(a*a)_k$ values) would then be approximately $C^2$ on average.

Since $\sum_k (a*a)_k = 1$, we have $C^2 \approx 1/(2n-1)$, so $C \approx 1/\sqrt{2n-1}$.

This confirms that the GS approach (flattening the spectrum) is the right direction.

    """GS with over-relaxation and adaptive target"""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    
    # Use target based on current max
    over_relax = 1.0
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        F = np.fft.fft(a)
        mag = np.abs(F)
        phase = F / (mag + 1e-300)
        
        # Target: push towards flat, but use a softer target
        target_mag = mag.mean()
        
        # Adaptively adjust over-relaxation
        if it % 5000 == 0 and it > 0:
            b_curr = fast_convolve(a)
            curr_score = 2.0 * n * b_curr.max()
            if curr_score > best_score * 1.01:
                over_relax = max(0.5, over_relax * 0.9)
            else:
                over_relax = min(1.5, over_relax * 1.05)
        
        new_mag = mag + over_relax * (target_mag - mag)
        new_mag = np.maximum(new_mag, 0)
        F_new = new_mag * phase
        a_new = np.real(np.fft.ifft(F_new))
        
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if it % 10000 == 0:
            print(f"    GS2 iter={it} ({time.monotonic()-t0:.0f}s): score={best_score:.6f} orel={over_relax:.3f}")
    
    return best_a, best_score

def compute_subgradient(a, n_active_frac=0.1):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b

def mirror_descent_adaptive(a0, time_limit, n_active_frac=0.08):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    b = fast_convolve(a)
    max_b = b.max()
    best_score = 2.0 * n * max_b
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    lr = 0.5
    no_improve = 0

    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        g, max_b = compute_subgradient(a, n_active_frac)
        g_norm_inf = np.abs(g).max()
        if g_norm_inf < 1e-15:
            break

        gap = best_score - f_lb
        if gap < 1e-10:
            break

        polyak_lr = gap / (2.0 * g_norm_inf * best_score + 1e-15)
        alpha = min(lr, polyak_lr)
        alpha = max(alpha, 0.01)

        z_new = z - alpha * g
        a_new = softmax_from_z(z_new)
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.05, 2.0)
        else:
            no_improve += 1
            if no_improve >= 20:
                lr *= 0.7
                no_improve = 0
            lr = max(lr, 0.01)

        if step % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc {elapsed:.0f}s: score={best_score:.6f} lr={lr:.4f}")

    return best_a, best_score

def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=100000):
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_score = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 500 == 0:
            if best_score < prev_score - 1e-7:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            if stall_count >= 8:
                rng = np.random.default_rng(it * 7 + 13)
                noise = rng.random(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score

def lbgfs_refine(a0, time_limit, T_start=150.0, T_end=0.3, maxiter=10000):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-17, 'gtol': 1e-13,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

def local_swap_search(a0, time_limit, n_steps=30000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        threshold = peak_val - 0.05 * peak_val
        near_peaks = np.where(b >= threshold)[0]

        contrib = np.zeros(n)
        for k in near_peaks[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]

        top_src = np.argsort(contrib)[::-1][:8]
        low_dst = np.argsort(contrib)[:8]

        improved = False
        for si in top_src:
            for di in low_dst:
                if si == di or a[si] < 1e-14:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amount = a[si] * frac
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-14:
                    continue
                amount = a[i] * 0.03
                a_trial = a.copy()
                a_trial[i] -= amount
                a_trial[j] += amount
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and step > 10000:
            break

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 1.5
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    elif choice == 6:
        # Multi-scale Fourier
        x = np.linspace(0, 2 * np.pi, n)
        a = 1.0
        for k in range(1, 6):
            a += (0.5 / k) * np.cos(k * x + k * 0.7)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def evaluate_safe(a):

        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

_lp_solver_name = None

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def get_lp_solver():
    global _lp_solver_name
    if _lp_solver_name is not None:
        return _lp_solver_name
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_name = solver_name

def solve_lp_best_response(a_fixed, solver=None):
