def exact_subgradient_step(a, n_active_frac=0.05, lr=0.01):
    """Take a subgradient step using the exact max."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    
    # Subgradient of max(a*a) at active lags
    # d max / da[m] = average over active lags of 2*a[k-m]
    grad = np.zeros(n)
    for k in active:
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        grad[m_start:m_end+1] += 2.0 * a[k - np.arange(m_start, m_end + 1)]
    grad /= len(active)
    
    # Projected subgradient step (maintain sum=1, a>=0)
    # Work in log-domain for non-negativity
    z = z_from_a(a)
    S = np.dot(grad, a)
    grad_z = a * (grad - S)
    z_new = z - lr * grad_z
    a_new = softmax_from_z(z_new)
    
    return a_new

def coordinate_descent(a0, time_limit=30, n_trials=50000):

            delta = a[idx] * delta_frac

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.001)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

Now, let me also think about a more principled approach. The problem can be formulated as:

Minimize `t` subject to:
- `(a*a)[k] <= t` for all `k = 0, ..., 2n-2`
- `sum(a) = sqrt(2n * t_target)` (or just `sum(a) = 1` and minimize `2n * max_b`)
- `a >= 0`

This is a non-convex QP. But we can solve it approximately using:
1. **Semidefinite relaxation**: Relax the rank-1 constraint on the outer product `a * a^T`.
2. **Convex relaxation via lifting**: Introduce variables `X[k,m] = a[k] * a[m]` and relax `X = a * a^T` to `X >= 0, X symmetric, X[0,0] = sum a^2`.
3. **Alternating minimization**: Fix the "shape" of `a` and optimize the "amplitude" at each point, then vice versa.

Actually, approach 3 is interesting. If I fix the "shape" (the relative proportions of `a[j]`), then the problem becomes trivial (just scale). So that doesn't help.

What about: fix the "envelope" (a smooth function) and optimize the "modulation" (small perturbations)?

Or: use a "water-filling" approach. The constraint `(a*a)[k] <= t` for all k means that at each lag k, the "energy" of the product `a[j]*a[k-j]` must be bounded. This is like a "simultaneous" constraint on all lags.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer in log-domain with multi-stage LogSumExp annealing
2. Diverse initializations including flat-autocorrelation types
3. Multi-restart with perturbation-based escaping
4. Exact subgradient polishing with adaptive active set
5. Multi-length exploration
6. Gradient-guided local search
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def cross_corr_a_w(a, w):
    """Compute c[m] = sum_j a[j] * w[m+j] for m=0..n-1 using FFT."""
    n = len(a)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(A) * W, fft_size)
    return corr[:n]


def lse_val_and_grad_z(z, T):
    """Compute (2n/T)*logsumexp(T*conv(a,a)) and gradient w.r.t. z (log-domain)."""
    n = len(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    lse = max_Tb + np.log(exp_Tb.sum())
    val = (2.0 * n / T) * lse
    w = exp_Tb / exp_Tb.sum()
    c = cross_corr_a_w(a, w)
    grad_a = 4.0 * n * c
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return val, grad_z


def adam_log_domain(z0, T_list, lr_base, time_limit, max_iters_total):
    """Adam in log-domain with sequence of T values."""
    n = len(z0)
    z = z0.copy()
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_z = z.copy()
    best_score = 2.0 * n * fast_convolve(softmax_from_z(z)).max()
    t0 = time.monotonic()
    iters_per_T = max(10, max_iters_total // len(T_list))

    for T_val in T_list:
        if time.monotonic() - t0 > time_limit:
            break
        for it in range(iters_per_T):
            if time.monotonic() - t0 > time_limit:
                break
            t += 1
            val, grad_z = lse_val_and_grad_z(z, T_val)
            m_adam = beta1 * m_adam + (1 - beta1) * grad_z
            v_adam = beta2 * v_adam + (1 - beta2) * grad_z ** 2
            m_hat = m_adam / (1 - beta1 ** t)
            v_hat = v_adam / (1.0 - beta2 ** t)
            lr = lr_base / np.sqrt(1.0 + 0.01 * t)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            if it % 100 == 0:
                a_chk = softmax_from_z(z)
                sc = 2.0 * n * fast_convolve(a_chk).max()
                if sc < best_score:
                    best_score = sc
                    best_z = z.copy()
        a_chk = softmax_from_z(z)
        sc = 2.0 * n * fast_convolve(a_chk).max()
        if sc < best_score:
            best_score = sc
            best_z = z.copy()

    return best_z, best_score


def subgrad_polish(a0, time_limit, n_active_frac, lr_init):
    """Polish with exact subgradient in log-domain."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    z = z_from_a(a)
    best_z = z.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        a_cur = softmax_from_z(z)
        b = fast_convolve(a_cur)
        max_b = b.max()
        if max_b < 1e-300:
            break
        threshold = max_b - n_active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        grad_a = np.zeros(n)
        for k in active:
            m_start = max(0, k - n + 1)
            m_end = min(n - 1, k)
            grad_a[m_start:m_end + 1] += 2.0 * a_cur[k - np.arange(m_start, m_end + 1)]
        grad_a /= len(active)
        S = np.dot(grad_a, a_cur)
        grad_z = a_cur * (grad_a - S)
        z_new = z - lr * grad_z
        a_new = softmax_from_z(z_new)
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_z = z_new.copy()
            z = z_new
            no_improve = 0
            lr = min(lr * 1.01, 0.15)
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.7
                no_improve = 0
                if lr < 1e-7:
                    break
    return softmax_from_z(best_z), best_score


def gradient_local_search(a0, time_limit, n_trials):
    """Gradient-guided local search: move mass from high-contribution to low-contribution positions."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for it in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        threshold = peak_val * 0.95
        near_peaks = np.where(b >= threshold)[0][:5]
        contrib = np.zeros(n)
        for k in near_peaks:
            m_start = max(0, k - n + 1)
            m_end = min(n - 1, k)
            idx = np.arange(m_start, m_end + 1)
            contrib[idx] += a[idx] * a[k - idx]
        top_sources = np.argsort(contrib)[::-1][:8]
        low_targets = np.argsort(contrib)[:8]
        improved = False
        for si in top_sources:
            if a[si] < 1e-12:
                continue
            for di in low_targets:
                if si == di:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.08]:
                    amount = a[si] * frac
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-10:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            for _ in range(2):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-12:
                    continue
                amount = a[i] * rng.random() * 0.04
                a_trial = a.copy()
                a_trial[i] -= amount
                a_trial[j] += amount
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        if not improved and it > 3000:
            break
    return best_a, best_score


def init_flat_ac(n, rng):
    """Initialize with a sequence having approximately flat circular autocorrelation."""
    choice = int(rng.integers(0, 4))
    j = np.arange(n)
    if choice == 0:
        c = float(rng.choice([0.2, 0.3, 0.4, 0.5]))
        a = 1.0 + c * np.cos(2.0 * np.pi * j / n)
    elif choice == 1:
        c1 = float(rng.choice([0.2, 0.3, 0.4]))
        c2 = float(rng.choice([-0.1, -0.15, -0.2, 0.05, 0.1]))
        a = 1.0 + c1 * np.cos(2.0 * np.pi * j / n) + c2 * np.cos(4.0 * np.pi * j / n)
    elif choice == 2:
        c = float(rng.choice([0.3, 0.5, 0.6]))
        p = float(rng.choice([1.0, 1.5, 2.0]))
        a = (1.0 + c * np.cos(2.0 * np.pi * j / n)) ** p
    else:
        c = float(rng.choice([0.2, 0.35, 0.5]))
        a = 1.0 + c * np.cos(2.0 * np.pi * j / n) - 0.1 * np.cos(4.0 * np.pi * j / n)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_smooth(n, rng, parent=None):
    """Diverse smooth initialization."""
    if parent is not None and rng.random() < 0.25:
        a = parent + rng.normal(0, 0.03, n)
    else:
        choice = int(rng.integers(0, 6))
        x = np.linspace(0, 1, n)
        if choice == 0:
            a = np.cos(np.pi * x / 2) ** 2
        elif choice == 1:
            a = np.exp(-2.0 * x * (1 - x))
        elif choice == 2:
            a = np.sin(np.pi * x) ** 2
        elif choice == 3:
            a = 1.0 - 0.4 * x ** 2
        elif choice == 4:
            a = 0.5 + 0.5 * np.cos(2.0 * np.pi * x)
            a = a ** 1.5
        else:
            a = np.exp(-4.0 * (x - 0.5) ** 2)
            a = a / a.max()
    a = np.maximum(a, 0.005)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.001)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def full_optimize(a0, time_budget, rng):
    """Complete optimization pipeline for one starting point, returns (best_a, best_score)."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    def try_update(a_cand, label):
        nonlocal best_a, best_score
        ac = np.maximum(a_cand, 0).astype(np.float64)
        sc = ac.sum()
        if sc < 1e-10:
            return
        ac = ac / sc
        score = 2.0 * n * fast_convolve(ac).max()
        if score < best_score:
            best_score = score
            best_a = ac.copy()
            elapsed = time.monotonic() - t0
            print(f"        [{label}] {score:.6f} (t={elapsed:.0f}s)")

    z = z_from_a(a)

    # Stage A: Adam with low-mid T
    rem = time_budget - (time.monotonic() - t0)
    if rem > 10:
        tA = min(rem * 0.30, 80.0)
        z, _ = adam_log_domain(z, [25.0, 40.0, 60.0], 0.08, tA, 30000)
        try_update(softmax_from_z(z), "adam_A")

    # Stage B: Adam with mid T
    rem = time_budget - (time.monotonic() - t0)
    if rem > 10:
        tB = min(rem * 0.25, 60.0)
        z, _ = adam_log_domain(z, [100.0, 150.0, 200.0], 0.03, tB, 30000)
        try_update(softmax_from_z(z), "adam_B")

    # Stage C: Adam with high T
    rem = time_budget - (time.monotonic() - t0)
    if rem > 8:
        tC = min(rem * 0.20, 50.0)
        z, _ = adam_log_domain(z, [300.0, 400.0, 500.0], 0.01, tC, 30000)
        try_update(softmax_from_z(z), "adam_C")

    # Stage D: Subgradient polish (broad active set)
    rem = time_budget - (time.monotonic() - t0)
    if rem > 5:
        tD = min(rem * 0.20, 40.0)
        a_d, _ = subgrad_polish(best_a, tD, 0.08, 0.05)
        try_update(a_d, "subgrad_broad")

    # Stage E: Gradient local search

        tE = min(rem * 0.15, 30.0)
        a_e, _ = gradient_local_search(best_a, tE, 20000)

        a_f, _ = subgrad_polish(best_a, min(rem * 0.3, 20.0), 0.03, 0.03)
        try_update(a_f, "subgrad_tight")

    return best_a, best_score
