import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv)."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n, dtype=np.float64) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

    return 2.0 * corr


def nesterov_pg(a0, time_limit, T_start=5.0, T_end=300.0,
                 lr_init=0.08, n_stages=12, steps_per_stage=2000,
                 use_box=False, box_K=20, jump_interval=500):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            global_step = stage * steps_per_stage + step
            lr = lr_init / (1.0 + 0.003 * global_step)

            # Nesterov momentum
            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            if step_count > 0:
                t_prev = 1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) ** 2) / 2.0
                beta = (t_prev - 1.0) / t_curr
            else:
                beta = 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)

            # Gradient
            if use_box and T > T_end * 0.3:
                grad = grad_box_max(y, box_K)
            else:
                grad = grad_smooth_max(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1

            # Periodic jump to escape local minima
            if jump_interval > 0 and step_count % jump_interval == 0:
                rng = np.random.default_rng(step_count)
                x = x + 0.01 * rng.standard_normal(n)
                x = project_simplex(x)

            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
                if step_count % 500 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    NG step={step_count} ({elapsed:.0f}s): score={best_score:.6f} T={T:.1f}")

    return best_a, best_score


def peak_equalize(a0, time_limit, K=30, max_iters=5000):
    """
    Active-set peak equalization.
    Finds directions that decrease all top-K peaks simultaneously.
    Uses min-norm solution for equal decrease.
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    prev_score = best_score
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        
        # Find active peaks (within 5% of max)
        threshold = M * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        # Limit to top K
        if len(active) > K:
            active_idx = np.argsort(b[active])[::-1][:K]
            active = active[active_idx]
        
        ka = len(active)
        
        # Build the active gradient matrix A (ka x n)
        # A[k, i] = 2 * a[active[k] - i] if 0 <= active[k] - i < n
        A_mat = np.zeros((ka, n))
        for idx, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_mat[idx, j] = 2.0 * a[k - j]
        
        # We want: A_mat @ d = -1 (all peaks decrease equally by 1)
        #         1^T d = 0 (maintain sum)
        # Minimize ||d||^2
        # Solution: d = A^T @ (A@A^T + alpha*1*1^T)^{-1} @ [-1; 0]
        # Or use the KKT system
        
        sim = A_mat @ A_mat.T  # ka x ka
        A_mat_1 = A_mat @ np.ones(n)  # ka vector
        # Augmented system: [sim + beta, A_mat_1; A_mat_1^T, 0] [lambda; mu] = [-1; 0]
        # But this might be ill-conditioned. Use regularization.
        
        beta = 0.001
        # System: [sim + beta*I, A_mat_1] [lam; mu] = [-1; 0]
        #         [A_mat_1^T,    0     ]
        aug = np.zeros((ka + 1, ka + 1))
        aug[:ka, :ka] = sim + beta * np.eye(ka)
        aug[:ka, ka] = A_mat_1
        aug[ka, :ka] = A_mat_1
        rhs = np.zeros(ka + 1)
        rhs[:ka] = -1.0
        
        try:
            sol = np.linalg.solve(aug, rhs)
        except np.linalg.LinAlgError:
            sol = np.linalg.lstsq(aug, rhs, rcond=None)[0]
        
        lam = sol[:ka]
        # d = A^T @ lam
        d = A_mat.T @ lam
        
        # Project d onto sum=0 (should already be close due to constraint)
        d = d - np.mean(d)
        
        # Find max step size
        # a + step*d >= 0
        # step * d[i] >= -a[i] for all i where d[i] < 0
        neg_mask = d < -1e-12
        if neg_mask.any():
            max_step = np.min(-a[neg_mask] / d[neg_mask]) * 0.9
        else:
            max_step = 1.0
        max_step = min(max_step, 0.1)
        
        if max_step < 1e-10:
            stall += 1
            if stall > 20:
                break
            # Try a random direction
            rng = np.random.default_rng(it)
            d = rng.standard_normal(n)
            d = d - np.mean(d)
            d = d / (np.linalg.norm(d) + 1e-15)
            max_step = 0.01
        
        a_new = project_simplex(a + max_step * d)
        sc = score_normalized(a_new)
        
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                # Add noise and reset
                rng = np.random.default_rng(it + 1000)
                a = a + 0.02 * rng.standard_normal(n)
                a = project_simplex(a)
                stall = 0
        
        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PEAK_EQ it={it+1} ({elapsed:.1f}s): score={best_score:.6f} K={ka}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solar=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    if solar is None:
        solar = _find_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solar)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solar:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def lp_refine(a0, time_limit, solar=None, n_max_iters=20000):
    """LP-based best response iteration with Nesterov dual averaging."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y, solar)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)

        score_a = score_normalized(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        score_y = score_normalized(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-8:
                stall = 0
                prev_best = best_score
            else:
                stall += 1
            if stall >= 5:
                rng = np.random.default_rng(it)
                y = y + 0.015 * rng.standard_normal(n)
                y = project_simplex(y)
                stall = 0
            elapsed = time.monotonic() - t0
            print(f"    LP it={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def spectral_init(n, M=None, rng=None):
    """Create initialization using incommensurate sinusoidal components."""
    if rng is None:
        rng = np.random.default_rng(0)
    if M is None:
        M = int(np.sqrt(2 * n))
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    # Use golden ratio based frequencies
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    for m in range(M):
        f = ((m + 1) * golden) % n
        if f < 1:
            f += 1
        amp = 0.5 / (1.0 + m * 0.3)
        phi = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * f * i / n + phi)
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a


def diverse_init(n, rng):
    """Create a diverse random initialization."""
    choice = rng.integers(0, 5)
    if choice == 0:
        # Smooth random
        a = np.abs(rng.standard_normal(n))
        a = np.convolve(a, np.ones(10) / 10, mode='same')
    elif choice == 1:
        # Multi-scale
        a = np.zeros(n)
        for scale in [2, 5, 13, 37]:
            f = rng.integers(1, n // scale)
            phi = rng.uniform(0, 2 * np.pi)
            a += np.cos(2 * np.pi * f * np.arange(n) / n + phi) / scale
        a = np.abs(a)
    elif choice == 2:
        # Two humps
        c1, c2 = sorted(rng.uniform(0.2, 0.8, 2))
        a = np.exp(-50 * (np.arange(n) / n - c1) ** 2) + np.exp(-50 * (np.arange(n) / n - c2) ** 2)
    elif choice == 3:
        # Spectral
        a = spectral_init(n, rng=rng)
    else:
        # Geometric decay with oscillation
        x = np.arange(n) / n
        a = np.exp(-3 * x) * (1 + 0.5 * np.sin(20 * np.pi * x))
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

A[j, k] = 2 * a[active[j] - k] for valid k (0 <= active[j]-k < n)

(A @ A^T)[i, j] = sum_k A[i,k] * A[j,k]

The valid k range for row i is [max(0, active[i]-n+1), min(n-1, active[i])]
The valid k range for row j is [max(0, active[j]-n+1), min(n-1, active[j])]
The intersection is [max(0, active[i]-n+1, active[j]-n+1), min(n-1, active[i], active[j])]

And for k in this range:
A[i,k] * A[j,k] = 4 * a[active[i]-k] * a[active[j]-k]

Let m = active[i] - active[j]. Then active[i]-k = active[j]-k + m.
So the sum is 4 * sum_k a[active[j]-k+m] * a[active[j]-k] for k in the valid range.

This is 4 * (a auto-correlation at lag m) evaluated over the valid range.

If the valid range covers the full overlap, this is 4 * corr_a[m] where corr_a[m] = sum_k a[k] * a[k+m].

But the valid range might be truncated at the boundaries. For most pairs (i,j) where |active[i] - active[j]| is small and both are in the interior, the full correlation applies.

For simplicity, let me just use the full auto-correlation (ignoring boundary effects) as an approximation:
(A @ A^T)[i, j] ≈ 4 * auto_corr[active[i] - active[j]]

where auto_corr[m] = sum_k a[k] * a[k+m] (the full linear auto-correlation at lag m).

Similarly, A @ 1 (the sum of each row) = sum_k 2*a[active[i]-k] for valid k = 2 * sum of a over the valid range. For interior peaks, this is approximately 2 * sum(a) = 2.

So the augmented system becomes:
[4*auto_corr_matrix + beta*I, 2*ones] [lam; mu] = [-1; 0]

This is much more efficient! The auto-correlation matrix is ka x ka and can be computed in O(ka^2) after computing the auto-correlation in O(n log n).

Let me use this approach:
