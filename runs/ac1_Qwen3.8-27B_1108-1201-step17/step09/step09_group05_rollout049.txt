import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def fast_convolve(a):

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def score_of(a):

def norm_seq(a):

    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()

def build_T(a):

    n = len(a)
    size = 2 * n - 1

def solve_lp_best_response(a_fixed):

    n = len(a_fixed)
    T = build_T(a_fixed)
    
    # LP: min t s.t. T@b - t <= 0, sum(b) = 1, b >= 0
    c = np.zeros(n + 1)

def nesterov_refine(a0, time_limit, n_max_iters=100000):
    """Nesterov dual averaging for the bilinear auto-convolution problem."""
    n = len(a0)
    a = norm_seq(a0).copy()
    y = a.copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, _ = solve_lp_best_response(y)
        if b_new is None:
            break
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        sc = score_of(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            stall = 0
        # Check stall every 200 iterations
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-6:
                prev_best = best_score
                stall = 0
            else:
                stall += 1
            if stall >= 8:
                # Perturb to escape
                rng = np.random.default_rng(it * 7919 + 42)
                y = y + 0.05 * rng.random(n)
                y = norm_seq(y)
                stall = 0
    
    return best_a, best_score

def mirror_descent_refine(a0, time_limit, n_steps=50000):
    """Mirror descent (multiplicative weights) for the auto-convolution problem."""
    n = len(a0)
    a = norm_seq(a0).copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    # Learning rate schedule: eta_t = c / sqrt(t)
    # c chosen based on problem scale
    c = 0.5
    
    for t in range(1, n_steps + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        
        # Subgradient: g[i] = 2 * a[k_star - i] if 0 <= k_star - i < n
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            g[i] = 2.0 * a[k_star - i]
        
        eta = c / np.sqrt(t)
        
        # Multiplicative update with KL projection
        # a_new[i] ∝ a[i] * exp(-eta * g[i])
        log_a = np.log(np.maximum(a, 1e-300))
        log_new = log_a - eta * g
        log_new -= log_new.max()  # for numerical stability
        new_exp = np.exp(log_new)
        a_new = new_exp / new_exp.sum()
        
        sc = score_of(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=10, maxiter=10000):
    """L-BFGS with log-sum-exp smoothing."""
    n = len(a0)
    a = norm_seq(a0)
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    
    def make_func(T_val):
        def func(z_in):
            a_local = np.exp(z_in - z_in.max())
            s = a_local.sum()
            if s < 1e-300:
                return 1e15
            a_local = a_local / s
            b = fast_convolve(a_local)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            lse = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * lse
        return func
    
    def make_jac(T_val):
        def jac(z_in):
            a_local = np.exp(z_in - z_in.max())
            s = a_local.sum()
            if s < 1e-300:
                return np.zeros_like(z_in)
            a_local = a_local / s
            b = fast_convolve(a_local)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a_local)
            return a_local * (grad_a - S)
        return jac
    
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
            if time.monotonic() - t0 > time_limit:
                break
        except Exception:
            break
    
    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out

def peak_shave(a0, time_limit, n_steps=20000):
    """Coordinate descent focusing on peak reduction."""
    n = len(a0)
    a = norm_seq(a0).copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    no_improve = 0

        b = fast_convolve(a)

        b_work = b.copy()

            b_work[:2] = -np.inf
            b_work[-2:] = -np.inf
        peak_k = int(np.argmax(b_work))
        
        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing the top contributor
        top_idx = np.argsort(contribs)[::-1][:10]
        improved = False
        
        for ci in top_idx:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.90, 0.85, 0.80, 0.70]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = norm_seq(a_trial)

                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 500:
                break
    
    return best_a, best_score

def structured_init(n, mode=0):

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
