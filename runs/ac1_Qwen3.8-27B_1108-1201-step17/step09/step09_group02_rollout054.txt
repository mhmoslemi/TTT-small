"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_correlation(a, w):
    """
    Compute c[i] = sum_{j=0}^{n-1} a[j] * w[i+j] for i=0..n-1.
    w has length 2n-1. Uses FFT-based convolution.
    """
    n = len(a)
    # c[i] = sum_j a[j] * w[i+j]
    # This equals (a * flip(w))[i + (2n-2)]
    # where * is linear convolution
    flip_w = w[::-1]  # length 2n-1
    conv_size = n + (2 * n - 1) - 1  # 3n-3
    fft_size = 1 << (conv_size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(flip_w, fft_size)
    result = irfft(A * B, fft_size)[:conv_size]
    # grad[i] = result[2n-2-i]
    return result[2 * n - 2 - np.arange(n)]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.60)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_fourier_mix(n, rng):
    """Mix of Fourier modes with random coefficients."""
    n_harmonics = rng.integers(5, 20)
    freqs = rng.choice(np.arange(1, n // 2 + 1), size=min(n_harmonics, n // 2), replace=False)
    amps = rng.random(len(freqs)) * 0.25
    phases = rng.uniform(0, 2.0 * np.pi, len(freqs))
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(len(freqs)):
        a += amps[j] * np.cos(2.0 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def compute_score(a):
    """Compute the objective 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)


def softmax_objective_and_grad(z, T):
    """
    Compute soft-max objective and gradient w.r.t. z.
    a = softmax(z) ensures a >= 0, sum(a) = 1.
    f = 2*n*T*log(sum exp(conv/T))
    """
    a = np.exp(z - z.max())
    a = a / a.sum()
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = (conv - max_c) / T
    log_sum_exp = max_c / T + np.log(np.sum(np.exp(shifted)))
    f = 2.0 * n * T * log_sum_exp

    # Weights
    w = np.exp(shifted)
    w /= w.sum()

    # Gradient of f w.r.t. a: df/da_i = 2*n * sum_k w_k * d conv[k]/da_i
    # d conv[k]/da_i = 2*a[k-i] if 0<=k-i<n else 0
    # So df/da_i = 4*n * sum_j a[j] * w[i+j] = 4*n * c_i
    c = cross_correlation(a, w)
    g_a = 4.0 * n * c

    # Gradient w.r.t. z: df/dz_j = a_j * (g_a_j - sum_i a_i * g_a_i)
    C = np.dot(a, g_a)
    grad_z = a * (g_a - C)

    return f, grad_z

    Optimize using L-BFGS on the softmax-parameterized soft-max objective.
    Anneal T from T_start to T_end.
    """
    n = len(a_init)
    # Initial z: log of a (with small epsilon to avoid -inf)
    z = np.log(a_init + 1e-30)
    z -= z.mean()  # center for numerical stability

    best_a = a_init.copy()
    best_score = compute_score(a_init)
    t0 = time.monotonic()

    # Annealing schedule: geometric
    n_anneal_steps = 12
    temperatures = T_start * (T_end / T_start) ** (np.arange(n_anneal_steps + 1) / n_anneal_steps)

    for T in temperatures:
        elapsed = time.monotonic() - t0
        rem = time_limit - elapsed
        if rem < 5:
            break

        # Max iterations for this temperature
        max_iter = int(min(5000, max(100, rem * 30)))

        def obj_grad(z_vec):
            f, g = softmax_objective_and_grad(z_vec, T)
            return f, g

        try:
            res = minimize(obj_grad, z, method='L-BFGS-B', jac=True,
                           options={'maxiter': max_iter, 'ftol': 1e-15,
                                    'gtol': 1e-12, 'iprint': 0,
                                    'maxls': 50})
            z = res.x
        except Exception:
            pass

        # Extract a and check score
        a = np.exp(z - z.max())
        a = a / a.sum()
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if T < T_end * 2:
            break

    return best_a, best_score


def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Exact coordinate descent with golden section search.
    For each coordinate, find the exact best value via 1D golden section.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            if has_quad:
                quad_idx = 2 * i - k_lo

            def score_fn(x):
                ns = current_sum + (x - v)
                if ns < 0.01:
                    return 1e10
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                nm = max(unaff_max, float(nv.max()))
                return 2.0 * n * nm / (ns * ns)

            # Golden section search
            upper = max(3.0 * v, 4.0 / n)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo_g = 0.0
            hi_g = upper
            c_g = hi_g - gr * (hi_g - lo_g)
            d_g = lo_g + gr * (hi_g - lo_g)
            fc = score_fn(c_g)
            fd = score_fn(d_g)
            for _ in range(30):
                if hi_g - lo_g < 1e-14:
                    break
                if fc <= fd:
                    hi_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = hi_g - gr * (hi_g - lo_g)
                    fc = score_fn(c_g)
                else:
                    lo_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = lo_g + gr * (hi_g - lo_g)
                    fd = score_fn(d_g)
            best_x = (lo_g + hi_g) * 0.5

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_pressure_descent(a0, time_limit, verbose=False, label=""):
    """
    Peak pressure descent: iteratively push down coordinates that contribute
    most to the current peaks in the convolution.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = 0.05
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        # Identify peaks above threshold
        threshold = conv.mean() + 0.3 * conv.std()
        peak_mask = conv > threshold
        n_peaks = int(peak_mask.sum())
        if n_peaks == 0:
            peak_mask = conv > conv.mean()

        # Compute pressure on each coordinate
        pressure = np.zeros(n)
        peak_indices = np.where(peak_mask)[0]
        for k in peak_indices:
            i_start = max(0, int(k) - n + 1)
            i_end = min(n - 1, int(k))
            for i in range(i_start, i_end + 1):
                j = int(k) - i
                pressure[i] += conv[int(k)] * a[j]

        # Normalize pressure
        p_max = pressure.max()
        if p_max > 1e-15:
            pressure /= p_max

        # Update: reduce high-pressure coordinates, increase low-pressure
        a_new = a * (1.0 - lr * pressure)
        # Also add small uniform to prevent zeros
        a_new += lr * 0.1 / n
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-10:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.8
            if no_improve > 200:
                break
        a = a_new

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PP {label} it={iters} ({elapsed:.0f}s): best={best_score:.8f}, lr={lr:.4f}")

    return best_a, best_score


def golden_section_search(f, lo, hi, max_iter=40):
    """Golden section search for minimum."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization + quick scoring
    Phase 2: L-BFGS with soft-max annealing (main workhorse)
    Phase 3: Peak pressure descent
    Phase 4: Exact coordinate descent polishing
    Phase 5: Basin hopping + final refinement
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(12345)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Generate diverse candidates ===
    print("\n=== Phase 1: Diverse Initialization ===")
    t1_start = time.monotonic()
    t1_budget = 200.0

    n_values = [768, 1024, 1280, 1536]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    candidates = []

    for n in n_values:
        if time.monotonic() - t1_start > t1_budget:
            break

        inits = []
        if parent is not None:
            inits.append(('parent', interpolate_seq(parent, n)))
        inits.append(('flat1', init_flat_spectrum(n, rng)))
        inits.append(('flat2', init_flat_spectrum(n, rng)))
        inits.append(('cosine', init_cosine(n, rng)))
        inits.append(('fourier', init_fourier_mix(n, rng)))
        inits.append(('flat3', init_flat_spectrum(n, rng)))

        for itype, cand in inits:
            if time.monotonic() - t1_start > t1_budget:
                break
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            score = compute_score(cand)
            candidates.append((score, cand, n, itype))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, best: {candidates[0][0]:.8f}")

    for s_c, a_c, n_c, it_c in candidates[:5]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}_{it_c}")

    # === Phase 2: L-BFGS with soft-max annealing ===
    print(f"\n=== Phase 2: L-BFGS Soft-Max (best={best_score:.8f}) ===")
    t2_start = time.monotonic()
    t2_budget = 700.0

    # Pick top 3 candidates for L-BFGS
    top_candidates = candidates[:3]
    # Deduplicate by n
    seen_n = set()
    filtered = []
    for s_c, a_c, n_c, it_c in top_candidates:
        if n_c not in seen_n:
            seen_n.add(n_c)
            filtered.append((s_c, a_c, n_c, it_c))
    for s_c, a_c, n_c, it_c in candidates[3:]:
        if len(filtered) >= 3:
            break
        if n_c not in seen_n:
            seen_n.add(n_c)
            filtered.append((s_c, a_c, n_c, it_c))

    for idx, (s_c, a_c, n_c, it_c) in enumerate(filtered):
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        t_alloc = min(rem * 0.4, t2_budget / max(1, len(filtered) - idx))
        print(f"  L-BFGS n={n_c}, init={it_c}, score={s_c:.8f}, budget={t_alloc:.0f}s")

        a_opt, s_opt = lbfgs_softmax_optimize(a_c, t_alloc, T_start=3.0, T_end=0.02)
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"lbfgs_n{n_c}_{it_c}")

        # Also try with different starting T
        rem = deadline - time.monotonic() - 350
        if rem > 30 and s_opt < best_score + 0.01:
            a_opt2, s_opt2 = lbfgs_softmax_optimize(a_opt, min(rem * 0.3, 100), T_start=0.5, T_end=0.01)
            best_seq, best_score = check_and_update(a_opt2, best_seq, best_score, f"lbfgs2_n{n_c}")

    # === Phase 3: Peak pressure descent ===
    print(f"\n=== Phase 3: Peak Pressure (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 30:
            t_pp = min(rem * 0.3, 150.0)
            print(f"  Peak pressure, budget={t_pp:.0f}s")
            a_pp, s_pp = peak_pressure_descent(best_seq, t_pp, verbose=True, label="pp")
            best_seq, best_score = check_and_update(a_pp, best_seq, best_score, "peak_press")
            # Follow with ECD
            rem = deadline - time.monotonic() - 250
            if rem > 20:
                a_cd, s_cd = exact_coord_descent(a_pp, min(rem * 0.4, 60), label="after_pp")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "ecd_after_pp")

    # === Phase 4: Intensive exact coordinate descent ===
    print(f"\n=== Phase 4: Intensive ECD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            current = best_seq.copy()
            for rnd in range(3):
                rem = deadline - time.monotonic() - 150
                if rem < 20:
                    break
                t_alloc = min(rem * 0.35, 200.0)
                print(f"  ECD round {rnd+1}, n={len(current)}, budget={t_alloc:.0f}s")
                a_opt, s_opt = exact_coord_descent(current, t_alloc,
                                                   verbose=True, label=f"ecd_r{rnd}")
                best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ecd_r{rnd}")
                if best_score > 1.5035:
                    current = a_opt + rng.standard_normal(len(a_opt)) * 0.02
                    current = np.maximum(current, 0)
                    current = current / current.sum()
                else:
                    current = a_opt.copy()

    # === Phase 5: Basin hopping ===
    print(f"\n=== Phase 5: Basin Hopping (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 40:
            n_perturbs = min(6, int(rem / 30))
            for p in range(n_perturbs):
                rem = deadline - time.monotonic() - 80
                if rem < 20:
                    break
                t_alloc = min(rem * 0.5, 50.0)

                # Multiple perturbation strategies
                strategy = p % 3
                if strategy == 0:
                    # Gaussian noise
                    perturbed = best_seq + rng.standard_normal(len(best_seq)) * 0.02
                elif strategy == 1:
                    # Peak-breaking: reduce values at positions contributing to peaks
                    conv_p = fast_convolve(best_seq)
                    top_k = min(5, 2 * len(best_seq) - 1)
                    peaks = np.argpartition(conv_p, -top_k)[-top_k:]
                    perturbed = best_seq.copy()
                    for k in peaks:
                        i_start = max(0, int(k) - len(best_seq) + 1)
                        i_end = min(len(best_seq) - 1, int(k))
                        idx = np.arange(i_start, i_end + 1)
                        n_pick = max(1, len(idx) // 5)
                        picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                        for ii in picks:
                            perturbed[ii] *= (1.0 + rng.standard_normal() * 0.05)
                else:
                    # Spectral perturbation
                    n_s = len(best_seq)
                    n_fft = 1 << (n_s - 1).bit_length()
                    A = rfft(perturbed if False else best_seq, n_fft)
                    A += rng.standard_normal(n_fft // 2 + 1) * 0.01 + 1j * rng.standard_normal(n_fft // 2 + 1) * 0.01
                    perturbed = irfft(A, n_fft)[:n_s]

                perturbed = np.maximum(perturbed, 0)
                ps = perturbed.sum()
                if ps < 1e-10:
                    continue
                perturbed = perturbed / ps

                a_opt, s_opt = exact_coord_descent(perturbed, t_alloc, label=f"bh_p{p}")
                best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"bh_p{p}")

    # === Phase 6: Final polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Another L-BFGS with very low T
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem * 0.4, 60.0)
            print(f"  Final L-BFGS, budget={t_alloc:.0f}s")
            a_fin, s_fin = lbfgs_softmax_optimize(best_seq, t_alloc, T_start=0.1, T_end=0.005)
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_lbfgs")

        # Final ECD
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_cd = min(rem * 0.5, 50.0)
            print(f"  Final ECD, budget={t_cd:.0f}s")
            a_cd, s_cd = exact_coord_descent(best_seq, t_cd, label="final_ecd")
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "final_ecd")

        # Last quick ECD
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            a_last, s_last = exact_coord_descent(best_seq, min(rem * 0.7, 20), label="last")
            best_seq, best_score = check_and_update(a_last, best_seq, best_score, "last_ecd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
