def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # Security checks
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = np.max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a ** 2))

def simplex_project(v):
    """
    Project vector v onto the probability simplex.
    """
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u > (cssv - u[-1] - 1) / (len(u) - 1)
    if cond.any():
        b = cond[-1] - 1  # b is the last index with condition true
        theta = (cssv[b] - u[-1] - 1) / (b + 1 - 1)
        return np.maximum(v - theta, 0)
    else:
        return np.ones(n) / n

def build_conv_matrix(a):
    """
    Build the (2n-1)xn matrix T such that T@b = conv(a,b).
    """
    n = len(a)
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

        if status is None or status not in ['optimal', 'optimal_inaccurate']:
            return None, None
            
        b_val = b_var.value
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-07:
            b_val = np.ones(n) / n
        return b_val, float(M_var.value) if 'M_var' in locals() else None
            
    except Exception as e:
        print(f"LP solve failed: {e}")
        # Try fallback methods
        try:
            # Fallback to simplex projection if LP fails
            b_val = simplex_project(a_fixed_arr)
            return b_val, float(np.max(np.convolve(a_fixed_arr, b_val))) if not np.inf else np.inf
