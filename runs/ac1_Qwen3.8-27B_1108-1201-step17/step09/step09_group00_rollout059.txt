import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def compute_v(w, a):
    """Compute v[j] = sum_k w[k] * a[k-j] using FFT.
    w has length 2n-1, a has length n.
    """
    n = len(a)
    a_ext = np.zeros(2 * n - 1)
    a_ext[:n] = a
    N = 1 << (4 * n - 3).bit_length()
    W = rfft(w, N)
    A_ext = rfft(a_ext, N)
    v_full = irfft(W * np.conj(A_ext), N)
    return v_full[:n]

def smoothed_obj(z, T, n):
    """Compute g_T in log-domain."""
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    scaled = T * b
    max_s = scaled.max()
    log_sum = max_s + np.log(np.sum(np.exp(scaled - max_s)))
    return (2.0 * n / T) * log_sum

def smoothed_grad(z, T, n):
    """Compute gradient of g_T in log-domain."""
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    scaled = T * b
    max_s = scaled.max()
    w = np.exp(scaled - max_s)
    w = w / w.sum()
    
    v = compute_v(w, a)
    S = np.dot(a, v)
    grad_z = 4.0 * n * a * (v - S * a)
    return grad_z

    choice = rng.integers(0, 5)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)

    else:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)

    s = a.sum()
    if s < 1e-15:

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Direct convex optimization via adaptive log-sum-exp smoothing + L-BFGS (primary)
2. Exact subgradient refinement at T=infinity with adaptive step
3. Coordinate descent polish on peak-contributing elements
4. Multi-length exploration with interpolation
Key insight: the problem is convex in a (minimize max of convex quadratic forms),
so L-BFGS on the smoothed objective should converge to the global optimum.
"""

import numpy as np
import time
from scipy.fft import fft, ifft, rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation (fast, no function call overhead)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_v(w, a):
    """Compute v[j] = sum_k w[k] * a[k-j] using FFT cross-correlation.
    w has length 2n-1, a has length n.
    """
    n = len(a)
    a_ext = np.zeros(2 * n - 1)
    a_ext[:n] = a
    N = 1 << (4 * n - 3).bit_length()
    W = fft(w, N)
    A = fft(a_ext, N)
    v_full = ifft(W * np.conj(A), N).real
    return v_full[:n]


def smoothed_obj(z, T, n):
    """Compute g_T(a) in log-domain where a = softmax(z)."""
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    scaled = T * b
    max_s = scaled.max()
    log_sum = max_s + np.log(np.sum(np.exp(scaled - max_s)))
    return (2.0 * n / T) * log_sum


def smoothed_grad(z, T, n):
    """Compute gradient of g_T in log-domain."""
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    scaled = T * b
    max_s = scaled.max()
    w = np.exp(scaled - max_s)
    w = w / w.sum()

    v = compute_v(w, a)
    S = np.dot(a, v)
    grad_z = 4.0 * n * a * (v - S * a)
    return grad_z


def lbgfs_adaptive(a0, time_limit, T_list):
    """L-BFGS with explicit T schedule. Returns (best_a, best_score)."""
    n = len(a0)
    a = np.maximum(a0, 1e-300).copy()
    a = a / a.sum()
    z = np.log(a).copy()

    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    for T in T_list:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        elapsed = time.monotonic() - t0
        print(f"    T={T:.3f}, elapsed={elapsed:.0f}s, remaining={remaining:.0f}s")

        # Number of L-BFGS iterations proportional to remaining time
        maxiter = max(50, min(2000, int(remaining * 50)))

        try:
            res = minimize(
                fun=smoothed_obj, x0=z, args=(T, n),
                jac=smoothed_grad, method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-13,
                         'maxfun': maxiter * 50, 'disp': False}
            )

            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            score = evaluate_local(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                print(f"      Improved to {score:.8f}")
        except Exception as e:

    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=100000):
    """Exact subgradient method with Nesterov momentum for the unsmoothed objective."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    # Nesterov momentum
    y = a.copy()
    beta = 0.9

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()

        # Active set: positions within 0.01% of max
        threshold = max_b * (1 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Uniform weight over active set
        p = np.zeros(len(b))
        p[active] = 1.0 / len(active)

        # Subgradient: v[j] = sum_k p[k] * a[k-j]
        v = compute_v(p, y)
        grad_a = 2.0 * v  # grad of max(b) w.r.t. a

        # Projected gradient step with adaptive lr
        lr = 1.0 / (1.0 + 0.1 * step)
        a_trial = y - lr * grad_a
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = y.copy()
        else:
            a_trial = a_trial / st

        # Nesterov update
        beta = 2.0 / (step + 2)
        y_new = a_trial + beta * (a_trial - a)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_trial.copy()
        else:
            y_new = y_new / sy

        a = a_trial
        y = y_new

        score = evaluate_local(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        score_y = evaluate_local(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if step % 20000 == 0:
            print(f"    SubGrad step={step}: score={best_score:.8f}")

    return best_a, best_score

    a = a / a.sum()

    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break

        contribs = a[idx] * a[peak_k - idx]
        top_m = min(6, len(idx))
        top_pos = idx[np.argsort(contribs)[::-1][:top_m]]

        improved = False
        for pos in top_pos:
            if a[pos] < 1e-12:
                continue
            for factor in [0.99, 0.98, 0.97, 0.95, 0.93]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-16:
                    continue
                # Redistribute to others proportionally
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try simultaneous reduction of top 3
            a_trial = a.copy()
            for pos in top_pos[:3]:
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.995
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                break
            a_trial = a_trial / st
            score = evaluate_local(a_trial)
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 150:
                    break
        else:
            stall = 0

        if step % 5000 == 0:
            print(f"    CoordDesc step={step}: score={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    else:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def try_update_state(best_seq, best_score, candidate, label):
    """Try to update best sequence/score. Returns (new_best_seq, new_best_score)."""
    a = np.maximum(np.asarray(candidate, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_local(a)
    if np.isfinite(sc) and sc < best_score:
        best_seq = a.copy()
        best_score = sc
        print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Primary method: L-BFGS on log-sum-exp smoothed convex objective.
    Secondary: exact subgradient with Nesterov momentum, coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    # === Phase 1: Primary L-BFGS with high T (coarse to fine) ===
    print(f"\n=== Phase 1: L-BFGS Coarse-to-Fine (n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 600.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        T_coarse = [500.0, 300.0, 200.0, 150.0, 100.0, 75.0, 50.0, 35.0, 25.0]
        a1, _ = lbgfs_adaptive(best_seq, t_alloc, T_list=T_coarse)
        best_seq, best_score = try_update_state(best_seq, best_score, a1, "phase1")

    # === Phase 2: L-BFGS with low T (fine optimization) ===
    print(f"\n=== Phase 2: L-BFGS Fine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_fine = [20.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 2.0, 1.5, 1.2, 1.0, 0.8, 0.6, 0.5, 0.4]
        a2, _ = lbgfs_adaptive(best_seq, t_alloc, T_list=T_fine)
        best_seq, best_score = try_update_state(best_seq, best_score, a2, "phase2")

    # === Phase 3: Even lower T for maximum precision ===
    print(f"\n=== Phase 3: L-BFGS Ultra-Fine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        T_ultra = [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.12, 0.1, 0.08, 0.06, 0.05, 0.04, 0.03, 0.02, 0.01]
        a3, _ = lbgfs_adaptive(best_seq, t_alloc, T_list=T_ultra)
        best_seq, best_score = try_update_state(best_seq, best_score, a3, "phase3")

    # === Phase 4: Exact subgradient with Nesterov ===
    print(f"\n=== Phase 4: Exact Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        a4, _ = subgradient_refine(best_seq, t_alloc, n_steps=100000)
        best_seq, best_score = try_update_state(best_seq, best_score, a4, "phase4")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a5, _ = coordinate_descent_polish(best_seq, t_alloc, n_steps=50000)
        best_seq, best_score = try_update_state(best_seq, best_score, a5, "phase5")

    # === Phase 6: Try alternative lengths ===
    print(f"\n=== Phase 6: Alt Lengths (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for alt_n in [int(n_best * 0.9), int(n_best * 1.1), int(n_best * 1.25)]:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            if alt_n < 200 or alt_n > 3000:
                continue
            t_alt = min(rem * 0.25, 150.0)
            print(f"  Trying n={alt_n}, budget={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)

    print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.3, 60.0)
    if t_alloc > 10:
        T_final = [0.3, 0.2, 0.15, 0.1, 0.08, 0.06, 0.05, 0.04, 0.03]
        a_fin, _ = lbgfs_adaptive(best_seq, t_alloc, T_list=T_final)

    # === Phase 8: Final coordinate descent ===
    rem = deadline - time.monotonic()
    if rem > 15:

        if t_cd > 8:
            a_cd, _ = coordinate_descent_polish(best_seq, t_cd, n_steps=20000)
            best_seq, best_score = try_update_state(best_seq, best_score, a_cd, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = np.ones(1200) / 1200
        best_score = evaluate_local(best_seq)

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
