The first term is `2·(p ⊛ a)[i]` (cross-correlation, computable via FFT).
The second term is a simple pointwise correction.

from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()

def compute_gradient_smoothed(a, conv, tau):
    """
    Compute gradient of 2*n*LSE(tau*conv) w.r.t. a.
    Returns g = ∂f/∂a (not yet projected/placed on simplex).
    """
    n = len(a)
    # Softmax weights on conv
    c_scaled = conv / tau
    c_max = c_scaled.max()
    p = np.exp(c_scaled - c_max)
    p /= p.sum()
    
    # Cross-correlation: (p ⊛ a)[i] = sum_k p[k+i] * a[k] for valid k
    # Using FFT: ifft(fft(p) * conj(fft(a)))
    fft_size = 1 << (2 * n - 1).bit_length()
    P = rfft(p, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(P * np.conj(A), fft_size)[:n]
    
    g = 2.0 * cross
    
    # Correction for k=2i terms
    for i in range(n):
        k = 2 * i
        if k < 2 * n - 1:
            g[i] -= p[k] * a[i]
    
    return 2.0 * n * g  # scale by 2n

    """Objective and gradient in softmax parameterization."""
    a = softmax(x)
    conv = fast_convolve(a)
    
    # LSE smoothed objective

    # Gradient w.r.t. a
    g_a = compute_gradient_smoothed(a, conv, tau)
    
    # Chain rule through softmax: ∂f/∂x_j = a_j * (g_a[j] - sum(g_a * a))
    a_dot_g = np.dot(g_a, a)
    gx = a * (g_a - a_dot_g)
    
    return val, gx

    Subgradient descent with adaptive step size and top-k peak weighting.

    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    
    # Parameters
    k_peaks = 20
    lr = 0.01
    no_improve = 0
    
    it = 0
    # Momentum
    v = np.zeros(n)
    beta = 0.9
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Find top-k peaks
        if k_peaks < 2 * n - 1:
            peak_idx = np.argpartition(conv, -k_peaks)[-k_peaks:]
        else:
            peak_idx = np.arange(2 * n - 1)
        
        pk = conv[peak_idx]
        # Soft weights
        temp = 0.01 * max(pk.max() - pk.min(), 1e-10)
        w = np.exp((pk - pk.max()) / temp)
        w /= w.sum()
        
        # Compute gradient as weighted sum of subgradients
        g = np.zeros(n)
        for idx in range(len(peak_idx)):
            if w[idx] < 1e-10:
                continue
            k = int(peak_idx[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                partner = k - i
                if k == 2 * i:
                    g[i] += w[idx] * a[i]
                else:
                    g[i] += w[idx] * 2.0 * a[partner]
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g /= g_norm
        
        # Update with momentum
        v = beta * v - lr * g
        a_new = simplex_project(a + v)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * float(conv_new.max())
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.9
            if no_improve > 500:
                break
        
        # Adaptive lr
        lr *= 0.9999
        
        if verbose and it % 500 == 0:
            print(f"    SubGrad it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.8f} lr={lr:.6e}")
    
    return best_a, best_score

        a = a / s

    conv = fast_convolve(a).copy()
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()

    k_peaks = min(30, 2 * n - 1)
    lr = 0.005
    momentum = 0.85
    v = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Find top-k peaks
        if k_peaks < 2 * n - 1:
            peak_idx = np.argpartition(conv, -k_peaks)[-k_peaks:]
        else:
            peak_idx = np.arange(2 * n - 1)

        pk = conv[peak_idx]
        pk_range = pk.max() - pk.min()
        if pk_range < 1e-12:
            pk_range = 1e-12
        temp = 0.05 * pk_range
        w = np.exp((pk - pk.max()) / temp)
        w /= w.sum()

        # Build weight vector p over all conv positions
        p = np.zeros(2 * n - 1)
        p[peak_idx] = w

        # Compute gradient via FFT
        g = compute_fft_subgradient(a, p)

        # Normalize
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g /= g_norm

        # Momentum update
        v = momentum * v - lr * g
        a_new = simplex_project(a + v)

        # Evaluate
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * float(conv_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 200 == 0:
                lr *= 0.85
                momentum *= 0.98
            if no_improve > 1000:
                break

        lr *= 0.9997

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.2e}")

    return best_a, best_score


def golden_section_search(f, lo, hi, max_iter=35):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5


def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent with golden section 1D search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            quad_idx = -1
            if has_quad:
                quad_idx = 2 * i - k_lo

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            _c = c_vals
            _ap = a_partners
            _qm = unaff_max
            _hq = has_quad
            _qi = quad_idx
            _cs = current_sum
            _vv = v
            _nn = n

            def compute_score(x):
                ns = _cs + (x - _vv)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _nn * nm / (ns * ns)

            upper = max(3.0 * v, 5.0 / n)
            best_x = golden_section_search(compute_score, 0.0, upper, max_iter=30)

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.5, 1.5, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.2, 0.6)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_fourier_mix(n, rng):
    """Mix of Fourier modes."""
    n_harmonics = rng.integers(5, 20)
    amps = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2.0 * np.pi, n_harmonics)
    freqs = rng.choice(np.arange(1, n // 2 + 1), size=n_harmonics, replace=False)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amps[j] * np.cos(2.0 * np.pi * float(freqs[j]) * t / n + phases[j])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_random_smooth(n, rng, smoothness=3):
    """Random sequence with spatial smoothing."""
    a = rng.random(n)
    for _ in range(smoothness):
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def basin_hopping(a0, time_limit, rng, verbose=False):
    """Basin hopping with diverse perturbations."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break

        # Choose perturbation type
        ptype = rng.integers(0, 4)

        if ptype == 0:
            # Gaussian noise
            sigma = rng.uniform(0.01, 0.05)
            perturbed = a + rng.standard_normal(n) * sigma
        elif ptype == 1:
            # Peak-breaking: identify peaks and reduce contributing elements
            conv_p = fast_convolve(a)
            top_k = min(10, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            perturbed = a.copy()
            for k in peaks:
                i_start = max(0, int(k) - n + 1)
                i_end = min(n - 1, int(k))
                idx = np.arange(i_start, i_end + 1)
                n_pick = max(1, len(idx) // 8)
                picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                perturbed[picks] *= (1.0 - rng.uniform(0.1, 0.4, len(picks)))
        elif ptype == 2:
            # Swap perturbations: move mass between elements
            perturbed = a.copy()
            n_swaps = rng.integers(5, 20)
            for _ in range(n_swaps):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i != j:
                    amt = perturbed[i] * rng.uniform(0.1, 0.3)
                    perturbed[i] -= amt
                    perturbed[j] += amt
        else:
            # Rescale peaks of the sequence itself
            perturbed = a.copy()
            n_shifts = rng.integers(3, 10)
            for _ in range(n_shifts):
                i = rng.integers(0, n)
                factor = 1.0 + rng.standard_normal() * 0.15
                perturbed[i] = max(0, perturbed[i] * factor)

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        # Quick optimization on perturbed
        t_alloc = min(rem * 0.4, 60.0)
        a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc)

        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH iter={iteration} ptype={ptype}: improved to {best_score:.8f}")

    return best_a, best_score


def evaluate_local(a):
    """Fast local evaluation (assumes a is normalized, non-negative)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    n = len(a_norm)
    return 2.0 * n * float(conv.max())


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate via evaluate_sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    1. L-BFGS on smoothed objective with diverse initializations
    2. FFT-accelerated subgradient refinement
    3. Exact coordinate descent
    4. Multi-length exploration
    5. Basin hopping
    6. Final polishing
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(777)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # === Phase 1: L-BFGS smoothed optimization with diverse inits ===
    print("\n=== Phase 1: L-BFGS Smoothed Optimization ===")
    t1_start = time.monotonic()

    n_main = 1024
    if parent is not None:
        np_parent = len(parent)
        if np_parent >= 512:
            n_main = np_parent

    # Try several initializations for the main length
    inits = []
    if parent is not None:
        inits.append(('parent', interpolate_seq(parent, n_main)))
    inits.append(('flat1', init_flat_spectrum(n_main, rng)))
    inits.append(('flat2', init_flat_spectrum(n_main, rng)))
    inits.append(('flat3', init_flat_spectrum(n_main, rng)))
    inits.append(('cos', init_cosine(n_main, rng)))
    inits.append(('fourier', init_fourier_mix(n_main, rng)))
    inits.append(('smooth', init_random_smooth(n_main, rng)))
    inits.append(('smooth2', init_random_smooth(n_main, rng, smoothness=5)))

    candidates = []
    for itype, a_init in inits:
        rem = 300.0 - (time.monotonic() - t1_start)
        if rem < 20:
            break
        t_alloc = min(rem * 0.7, 80.0)
        print(f"  L-BFGS n={n_main} init={itype} budget={t_alloc:.0f}s")
        a_opt = lbfgs_optimize(a_init, t_alloc, verbose=True)
        local_score = evaluate_local(a_opt)
        candidates.append((local_score, a_opt, itype))
        if verbose:
            pass

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1 done: {len(candidates)} candidates, best local={candidates[0][0]:.8f}")

    for s_c, a_c, it_c in candidates[:5]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"lbfgs_{it_c}")

    # === Phase 2: Subgradient refinement ===
    print(f"\n=== Phase 2: Subgradient Refinement (best={best_score:.8f}) ===")
    if best_seq is not None:
        t2_start = time.monotonic()
        rem2 = 200.0
        a_sg, s_sg = subgradient_optimize(best_seq, min(rem2, deadline - time.monotonic() - 600),
                                          verbose=True)
        best_seq, best_score = check_and_update(a_sg, best_seq, best_score, "subgrad")

    # === Phase 3: Exact coordinate descent ===
    print(f"\n=== Phase 3: Exact Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for rnd in range(3):
            rem = deadline - time.monotonic() - 400
            if rem < 30:
                break
            t_alloc = min(rem * 0.35, 200.0)
            print(f"  ECD round {rnd+1}, n={n_best}, budget={t_alloc:.0f}s")
            a_opt, s_opt, sweeps = exact_coord_descent(best_seq, t_alloc,
                                                       verbose=True, label=f"ph3_r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ecd_r{rnd}")

    # === Phase 4: Try other lengths ===
    print(f"\n=== Phase 4: Multi-Length Exploration (best={best_score:.8f}) ===")
    if best_seq is not None:
        alt_lengths = []
        for alt_n in [768, 1536, 2048]:
            if alt_n != len(best_seq) and alt_n >= 512 and alt_n <= 3072:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:2]:
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            t_alloc = min(rem * 0.4, 100.0)
            print(f"  Trying n={alt_n}, budget={t_alloc:.0f}s")

            a_alt = interpolate_seq(best_seq, alt_n)
            a_opt = lbfgs_optimize(a_alt, t_alloc * 0.5)
            a_opt2, s_opt2, _ = exact_coord_descent(a_opt, t_alloc * 0.5)
            best_seq, best_score = check_and_update(a_opt2, best_seq, best_score, f"alt_n{alt_n}")

    # === Phase 5: Basin hopping ===
    print(f"\n=== Phase 5: Basin Hopping (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 40:
            t_bh = min(rem * 0.6, 200.0)
            print(f"  Basin hopping budget={t_bh:.0f}s")
            a_bh, s_bh = basin_hopping(best_seq, t_bh, rng, verbose=True)
            best_seq, best_score = check_and_update(a_bh, best_seq, best_score, "basin_hop")

    # === Phase 6: Final intensive polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # ECD polish
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_cd = min(rem * 0.6, 80.0)
            a_cd, s_cd, _ = exact_coord_descent(best_seq, t_cd, label="final_cd")
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "final_cd")

        # One more subgradient pass
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            a_sg2, s_sg2 = subgradient_optimize(best_seq, min(rem * 0.7, 40.0))
            best_seq, best_score = check_and_update(a_sg2, best_seq, best_score, "final_sg")

        # Final ECD
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_fin, s_fin, _ = exact_coord_descent(best_seq, min(rem * 0.8, 30.0), label="last")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "last_ecd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
