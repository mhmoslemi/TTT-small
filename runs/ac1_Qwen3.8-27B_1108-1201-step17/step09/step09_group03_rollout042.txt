Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS optimization with smooth log-sum-exp approximation of max
2. Adaptive temperature cooling schedule
3. Remez-like active-set refinement (linear system solve)
4. Fine coordinate descent polishing
5. Multi-start with diverse initializations

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time

# Global state for L-BFGS (to avoid closures)
_g_t = 1.0
_g_n = 0
_g_fft_size_conv = 0
_g_fft_size_xcorr = 0

def _fft_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:size]

def _xcorr(p, a):
    """
    Linear cross-correlation: r[j] = sum_m p[j+m] * a[m], for j = 0,...,n-1.
    p has length 2n-1, a has length n.
    """
    n = len(a)
    N = 1 << (3 * n - 2 - 1).bit_length()
    P = fft(p, N)
    A = fft(a, N)
    R = ifft(P * np.conj(A))
    return R.real[:n]

def _smooth_obj_grad(x):
    """
    Objective and gradient for L-BFGS.
    x is the parameter vector; a = softmax(x).
    Returns (objective, gradient).
    """
    global _g_t
    # Softmax
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    a = exp_x / exp_x.sum()
    
    # Compute convolution
    c = _fft_conv(a)
    
    # Log-sum-exp with temperature
    t = _g_t
    c_t = c / t
    c_t_max = c_t.max()
    exp_ct = np.exp(c_t - c_t_max)
    p_sum = exp_ct.sum()
    p = exp_ct / p_sum
    logsumexp_val = c_t_max + np.log(p_sum)
    
    n = len(a)
    obj = 2.0 * n * t * logsumexp_val
    
    # Gradient
    r = _xcorr(p, a)
    grad_a = (2.0 / t) * r
    grad_x = a * (grad_a - np.dot(a, grad_a))
    
    return obj, grad_x

def _lbfgs_optimize(x0, t_schedule, max_iter=500):
    """
    Run L-BFGS with a schedule of temperature values.
    x0: initial parameter vector
    t_schedule: list of (temperature, max_iterations) pairs
    Returns the best x found.
    """
    global _g_t, _g_n
    n = len(x0)
    _g_n = n
    x = x0.copy()
    best_x = x0.copy()
    best_f = np.inf
    
    for t, iters in t_schedule:
        _g_t = t
        res = minimize(_smooth_obj_grad, x, method='L-BFGS-B',
                       jac=True, options={'maxiter': iters, 'gtol': 1e-8, 'ftol': 1e-12})
        x = res.x
        f = res.fun
        if f < best_f:
            best_f = f
            best_x = x.copy()
    
    return best_x, best_f

def _softmax(x):
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    return exp_x / exp_x.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS optimization with smooth log-sum-exp approximation of max
2. Adaptive temperature cooling schedule for annealing
3. Remez-like active-set linear system solver for exact stationarity
4. Fine coordinate descent polishing
5. Multi-start with diverse initializations (Gaussian, cosine, parent)
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time

# Global state for L-BFGS callback (to avoid closures)
_g_t = 1.0


def _fft_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:size]


def _xcorr(p, a):
    """
    Linear cross-correlation: r[j] = sum_m p[j+m] * a[m], for j = 0,...,n-1.
    p has length 2n-1, a has length n.
    """
    n = len(a)
    N = 1 << (3 * n - 2 - 1).bit_length()
    P = fft(p, N)
    A = fft(a, N)
    R = ifft(P * np.conj(A))
    return R.real[:n]


def _softmax(x):
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s


def _smooth_obj_grad(x):
    """
    Objective and gradient for L-BFGS.
    x is parameter vector; a = softmax(x).
    Objective: 2*n*t*logsumexp(conv(a,a)/t)
    """
    global _g_t
    a = _softmax(x)
    c = _fft_conv(a)
    t = _g_t
    c_t = c / t
    c_t_max = c_t.max()
    exp_ct = np.exp(c_t - c_t_max)
    p_sum = exp_ct.sum()
    p = exp_ct / p_sum
    logsumexp_val = c_t_max + np.log(p_sum)
    n = len(a)
    obj = 2.0 * n * t * logsumexp_val
    r = _xcorr(p, a)
    grad_a = (2.0 / t) * r
    grad_x = a * (grad_a - np.dot(a, grad_a))
    return obj, grad_x


def _lbfgs_optimize(x0, t_schedule, max_iter_per_t=300):
    """Run L-BFGS with a sequence of temperatures."""
    global _g_t
    x = x0.copy()
    best_x = x0.copy()
    best_f = np.inf
    for t in t_schedule:
        _g_t = t
        try:
            res = minimize(_smooth_obj_grad, x, method='L-BFGS-B',
                           jac=True, options={'maxiter': max_iter_per_t,
                                              'gtol': 1e-9, 'ftol': 1e-14})
            x = res.x
            f_val = res.fun
            if f_val < best_f:
                best_f = f_val
                best_x = x.copy()
        except Exception:
            pass
    return best_x, best_f


def _remez_solve(active, n, max_iter=200, tol=1e-12):
    """
    Remez-like: solve stationarity linear system for given active set.
    Returns optimal a.
    """
    if len(active) == 0:
        return np.ones(n) / n
    mu = np.ones(len(active)) / len(active)
    a = np.ones(n) / n
    for it in range(max_iter):
        A_mat = np.zeros((n + 1, n + 1))
        for idx, k in enumerate(active):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            js = np.arange(j_min, j_max + 1)
            is_ = k - js
            A_mat[js, is_] += mu[idx]
        A_mat[:n, n] = -0.5
        A_mat[n, :n] = 1.0
        rhs = np.zeros(n + 1)
        rhs[n] = 1.0
        try:
            sol = np.linalg.solve(A_mat, rhs)
        except np.linalg.LinAlgError:
            sol = np.linalg.lstsq(A_mat, rhs, rcond=None)[0]
        a_new = sol[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        if np.max(np.abs(a_new - a)) < tol:
            a = a_new
            break
        a = a_new
    return a


def _remez_refine(a0, max_rounds=50):
    """Iterative Remez: alternate active set update and linear solve."""
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    for rnd in range(max_rounds):
        c = _fft_conv(a)
        M = c.max()
        if M < 1e-15:
            break
        threshold = M * 0.999
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        # Cap active set size to avoid singularity
        if len(active) > min(20, n // 2):
            top_k = min(20, n // 2)
            active = np.argsort(c[active])[::-1][:top_k] + active[np.argsort(c[active])[::-1][:top_k]]
            # Actually just take top peaks
            active = np.array(sorted(active))
        a_new = _remez_solve(active, n)
        if np.max(np.abs(a_new - a)) < 1e-10:
            a = a_new
            break
        # Blend to avoid instability
        alpha = 0.5
        a_blend = (1 - alpha) * a + alpha * a_new
        a_blend = np.maximum(a_blend, 0)
        s_b = a_blend.sum()
        if s_b < 1e-15:
            a = a_new
        else:
            a = a_blend / s_b
    return a


def _fine_cd(a0, time_limit, verbose=False):
    """Fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = _fft_conv(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 5:
        improved = False
        for i in range(n):
            if a[i] < 1e-15:
                continue
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)
            a_shifted = a[k_start - i:k_end - i + 1]
            best_delta = 0.0
            best_new_M = M
            for factor in [0.97, 0.985, 0.995, 1.005, 1.015, 1.03]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                new_affected = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta
            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        if verbose:
            print(f"    CD: best={best_score:.8f}")
    return best_a, best_score


def _gaussian_init(n, sigma_frac):
    """Gaussian-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    s = a.sum()
    return a / s


def _cosine_init(n, power):
    """Raised cosine window."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    a = (0.5 * (1 + np.cos(np.pi * x))) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
