"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None
            b_val = b_val / s
            return b_val
    except Exception:
        pass
    return None


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def cos_init(n, rng):
    """Cosine-squared initialization."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x / 2) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def uniform_init(n):
    """Uniform initialization."""
    return np.ones(n) / n


def gaussian_init(n, rng):
    """Gaussian-shaped initialization."""
    x = np.linspace(0, 1, n)
    centers = rng.uniform(0.2, 0.8, size=2)
    widths = rng.uniform(0.1, 0.3, size=2)
    a = np.zeros(n)
    for c, w in zip(centers, widths):
        a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def periodic_init(n, rng):

    p = rng.integers(3, min(20, n // 3))
    pattern = np.abs(np.cos(np.pi * np.arange(p) / p)) + 0.1

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

    return a / s

def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=200000):
    """Nesterov dual averaging LP refinement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_rounds = 0
    rng = np.random.default_rng(99999)
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        # Nesterov acceleration
        outer_prod = (it + 1) / (it + 1 + 2.0)
        a = outer_prod * a + (1.0 - outer_prod) * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-9:
                stall_rounds = 0
            else:
                stall_rounds += 1
            prev_best = best_score
            
            if stall_rounds >= 10:
                noise_scale = 0.05 / np.sqrt(stall_rounds)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_rounds = 0
    
    return best_a, best_score


def logsumexp_obj_and_grad(z, n, T_temp):
    """Compute LogSumExp-smoothed objective and gradient."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    b = fast_convolve(a)
    # For normalized a (sum=1), conv(a,a) has the autocorrelation
    # Objective: 2n * max(conv(a,a))
    # Smoothed: 2n * (1/T) * log(sum(exp(T * conv(a,a))))
    
    # Actually, let me use a different formulation.
    # The gradient of max_k b_k w.r.t. a is the column of the conv matrix at argmax.
    # For LogSumExp: the "weights" are softmax(T * b)
    
    max_b = b.max()
    Tb = T_temp * (b - max_b)
    exp_Tb = np.exp(Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_b + np.log(exp_Tb.sum()) / T_temp
    
    # Gradient of LogSumExp w.r.t. a
    # d/da_j [sum_k w_k * conv(a,a)_k] where w_k = softmax(T*b)_k
    # = sum_k w_k * (a_{k-j} + a_{k-j} for the symmetric part)
    # Actually conv(a,a)_k = sum_j a_j * a_{k-j}
    # d/da_j conv(a,a)_k = a_{k-j} + a_{k-j} (by symmetry, when both indices are valid)
    # More precisely: d/da_j sum_k w_k * sum_i a_i * a_{k-i}
    # = sum_k w_k * (a_{k-j} + a_{k-j}) if k-j is in range
    # = 2 * sum_k w_k * a_{k-j} (with appropriate bounds)
    
    grad_a = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        # d/da_j conv(a,a)_k = a_{k-j} for j where 0 <= k-j < n
        # So for a fixed j, it's a_{k-j} if 0 <= k-j < n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        # a_{k - idx} for idx in range js..je
        grad_a[idx] += w[k] * a[k - idx]
    
    # The objective is 2n * LSE / (sum_a)^2 but since sum_a = 1:
    # grad of 2n * LSE = 2n * grad_a (times 2 because of symmetry in conv)
    # Wait, let me be more careful.
    # conv(a,a)_k = sum_i a_i * a_{k-i}
    # d/da_j [conv(a,a)_k] = a_{k-j} + a_{k-j} = 2*a_{k-j} (when valid)
    # Hmm, actually no. Let's be explicit:
    # conv(a,a)_k = sum_i a_i * a_{k-i}
    # d/da_j = a_{k-j} (from the first factor being a_j) + a_{k-j} (from the second factor being a_j when i=k-j)
    # Wait no. Let me index more carefully.
    # conv(a,a)_k = sum_i a_i * a_{k-i} (sum over valid i)
    # If we differentiate w.r.t. a_j:
    # Term 1: a_j appears as a_i when i=j, giving a_{k-j}
    # Term 2: a_j appears as a_{k-i} when k-i=j, i.e., i=k-j, giving a_{k-j}
    # So d/da_j conv(a,a)_k = a_{k-j} + a_{k-j} = 2*a_{k-j} (when 0 <= k-j < n)
    # But wait, when j = k-j (i.e., 2j = k), both terms are the same, so it's still 2*a_{k-j}
    # Actually no, it's always 2*a_{k-j} for valid k-j.
    
    # So the gradient of sum_k w_k * conv(a,a)_k w.r.t. a_j is:
    # sum_k w_k * 2 * a_{k-j} (for valid k-j)
    
    grad_a = 2.0 * grad_a  # Account for symmetry
    
    # Convert to gradient w.r.t. z (log-domain)
    # a = softmax(z), so d a / d z_j = a_j (delta_ji - a_i)
    # grad_z = a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return 2.0 * n * log_sum_exp, grad_z


def lbgfs_logsumexp_refine(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z):
            val, _ = logsumexp_obj_and_grad(z, n, stage_T)
            return val
        
        def jac(z):
            _, g = logsumexp_obj_and_grad(z, n, stage_T)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': maxiter_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def peak_shaving(a0, time_limit, n_steps=50000):
    """Targeted coordinate descent reducing peak contributors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(7777)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find all near-max positions
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        
        improved = False
        
        for peak_k in active[:5]:  # Try top 5 peaks
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            
            # Find the element with highest contribution
            top_m = min(5, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
            
            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.99, 0.98, 0.97, 0.95]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    # Redistribute proportionally to other elements
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    a_trial = a_trial / a_trial.sum()
                    
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 500:
                # Add small random perturbation
                noise = rng.standard_normal(n) * 0.01
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
        else:
            stall = 0
    
    return best_a, best_score

    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9999
        active = np.where(b >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Average subgradient over active peaks
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)
        
        # Projected gradient step
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 50:
            lr *= 0.95
            no_improve = 0
        lr = max(1e-6, lr)
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines diverse multi-start initialization, Nesterov LP alternating,
adaptive-temperature LogSumExp L-BFGS, subgradient descent, peak-shaving
coordinate descent, and length exploration. Key improvements over baseline:
- More aggressive diversity in Phase 1 with 7 distinct initialization types
- Smarter Nesterov LP with dual-informed restarts

- KKT-based stopping to avoid unnecessary computation

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            return b_val, float(M_var.value)
    except Exception:
        pass
    # Fallback solvers
    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None, None
                b_val = b_val / s
                return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def cos_init(n, rng):
    """Cosine-squared initialization."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x / 2) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def uniform_init(n):
    """Uniform initialization."""
    return np.ones(n) / n


def gaussian_init(n, rng):
    """Gaussian-shaped initialization."""
    x = np.linspace(0, 1, n)
    centers = rng.uniform(0.25, 0.75, size=2)
    widths = rng.uniform(0.12, 0.28, size=2)
    a = np.zeros(n)
    for c, w in zip(centers, widths):
        a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def periodic_init(n, rng):
    """Periodic initialization."""
    p = int(rng.integers(3, min(20, max(4, n // 3))))
    pattern = np.abs(np.cos(np.pi * np.arange(p) / p)) + 0.1
    a = np.tile(pattern, n // p + 1)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def random_smooth_init(n, rng):
    """Smooth random initialization (low-pass filtered noise)."""
    raw = rng.random(n)
    # Apply simple moving average for smoothness
    kernel_size = max(3, n // 50)
    kernel = np.ones(kernel_size) / kernel_size
    if kernel_size > 1:
        raw = np.convolve(raw, kernel, mode='same')
    a = raw + 0.01
    a = np.maximum(a, 0)
    return a / a.sum()


def bump_init(n, rng):
    """Multiple bump initialization."""
    k = int(rng.integers(2, 6))
    a = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(k):
        c = rng.uniform(0.05, 0.95)
        w = rng.uniform(0.05, 0.2)
        a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def arch_init(n, rng):
    """Arch-shaped with modulation."""
    x = np.linspace(0, 1, n)
    freq = rng.integers(1, 5)
    a = (1 - 4 * (x - 0.5) ** 2) + 0.15 * np.cos(2 * np.pi * freq * x)
    a = np.maximum(a, 0.001)
    return a / a.sum()


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=300000):
    """Nesterov dual averaging LP refinement with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_rounds = 0
    rng = np.random.default_rng(42)
    last_restart = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Nesterov accelerated combination
        a = a + alpha * (b_new - a)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_rounds = 0
            else:
                stall_rounds += 1
            prev_best = best_score

            if stall_rounds >= 6:
                # Adaptive perturbation: scale increases with stall
                noise_scale = min(0.1, 0.02 * stall_rounds / 6)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_rounds = 0
                last_restart = it

            # Very long stall: reset y to a mix
            if stall_rounds >= 15 and it - last_restart > 2000:
                y = 0.5 * best_a + 0.5 * rng.dirichlet(np.ones(n))
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_rounds = 0
                last_restart = it

    return best_a, best_score


def lse_objective(z, n, temp):
    """Compute LogSumExp-smoothed objective value."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    max_b = b.max()
    Tb = temp * (b - max_b)
    exp_Tb = np.exp(Tb)
    log_sum_exp = max_b + np.log(exp_Tb.sum()) / temp
    return 2.0 * n * log_sum_exp


def lse_gradient(z, n, temp):
    """Compute gradient of LogSumExp-smoothed objective w.r.t. z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    max_b = b.max()
    Tb = temp * (b - max_b)
    exp_Tb = np.exp(Tb)
    w = exp_Tb / exp_Tb.sum()

    # Gradient of LSE w.r.t. a: sum_k w_k * d(conv(a,a)_k)/da
    # d(conv(a,a)_k)/da_j = 2 * a_{k-j} (for valid k-j)
    grad_a = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    grad_a *= 2.0  # symmetry factor

    # Convert to z-gradient: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    return 2.0 * n * a * (grad_a - S)


def lbgfs_logsumexp_refine(a0, time_limit, T_stages=None, maxiter_per_stage=4000):
    """L-BFGS with LogSumExp smoothing, multi-temperature schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [30.0, 15.0, 7.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_val):
            return lse_objective(z_val, n, stage_T)

        def jac(z_val):
            return lse_gradient(z_val, n, stage_T)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def peak_shaving(a0, time_limit, n_steps=50000):
    """Targeted coordinate descent reducing peak contributors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(1234)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 1e-6)
        active = np.where(b >= threshold)[0]

        improved = False

        for peak_k in active[:8]:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]

            top_m = min(6, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.995, 0.99, 0.985, 0.98, 0.97]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    a_trial = a_trial / a_trial.sum()

                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 300:
                # Gentle random perturbation
                noise = rng.standard_normal(n) * 0.005
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
        else:
            stall = 0

    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=40000):
    """Subgradient method on the exact objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.015
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 1e-5)
        active = np.where(b >= threshold)[0]

        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            lr *= 0.93
            no_improve = 0
        lr = max(1e-7, lr)

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multi_round_lp(a0, time_limit, solver, n_rounds=3):
    """Multiple LP rounds with re-initialization between rounds."""
    best_a = a0.copy()
    best_score = 2.0 * len(a0) * fast_convolve(a0).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for round_idx in range(n_rounds):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break
        t_round = rem / (n_rounds - round_idx)

        a_ref, _ = nesterov_lp_refine(best_a, t_round, solver=solver, n_max_iters=100000)
        sc = 2.0 * len(a_ref) * fast_convolve(a_ref).max()
        if sc < best_score:
            best_score = sc
            best_a = a_ref.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer combining diverse initialization, LP alternating,
    LogSumExp L-BFGS, subgradient, peak-shaving, and length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}", flush=True)

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}", flush=True)
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 1200

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01 or a.size == 0:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})", flush=True)

    # === Phase 1: Diverse multi-start exploration ===
    print(f"\n=== Phase 1: Multi-start (n={n_base}) ===", flush=True)
    rem = deadline - time.monotonic()

    inits = []
    if parent is not None:
        inits.append(("parent", parent.copy()))
    inits.append(("cos", cos_init(n_base, rng)))
    inits.append(("gauss", gaussian_init(n_base, rng)))
    inits.append(("periodic", periodic_init(n_base, rng)))
    inits.append(("smooth_rand", random_smooth_init(n_base, rng)))
    inits.append(("bump", bump_init(n_base, rng)))
    inits.append(("arch", arch_init(n_base, rng)))

    n_starts = min(len(inits), max(3, int(rem // 35)))
    time_per_start = min(rem / (n_starts + 2), 35.0)

    for i, (name, init_a) in enumerate(inits[:n_starts]):
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_alloc = min(time_per_start, rem - 50)
        if t_alloc < 5:
            break
        print(f"  Start {i+1}/{n_starts} [{name}], budget={t_alloc:.0f}s", flush=True)
        a_ref, _ = nesterov_lp_refine(init_a, t_alloc, solver=solver, n_max_iters=30000)
        try_update(a_ref, f"lp_{name}")

    # === Phase 2: Deep LP optimization on the best ===
    print(f"\n=== Phase 2: Deep LP (score={best_score:.10f}) ===", flush=True)
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 550.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s", flush=True)
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=500000)
        try_update(a_lp, "lp_deep")

    # === Phase 3: LogSumExp L-BFGS (main) ===
    print(f"\n=== Phase 3: LogSumExp L-BFGS (score={best_score:.10f}) ===", flush=True)
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
        print(f"  T_stages: {T_stages}, budget={t_alloc:.0f}s", flush=True)
        a_lb = lbgfs_logsumexp_refine(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=3000)
        try_update(a_lb, "lbgfs_main")

    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient (score={best_score:.10f}) ===", flush=True)
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 15:
        a_sg, _ = subgradient_refine(best_seq, t_alloc, n_steps=50000)
        try_update(a_sg, "subgrad")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving (score={best_score:.10f}) ===", flush=True)
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 15:
        a_ps, _ = peak_shaving(best_seq, t_alloc, n_steps=40000)
        try_update(a_ps, "peak_shave")

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length exploration (score={best_score:.10f}) ===", flush=True)
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_seq)
        alt_lengths = []
        for factor in [0.8, 1.2, 1.5]:
            alt_n = int(n_cur * factor)
            if 400 <= alt_n <= 3000 and alt_n != n_cur:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:2]:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_alt = min(rem * 0.3, 100.0)
            print(f"  Trying n={alt_n}, budget={t_alt:.0f}s", flush=True)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = nesterov_lp_refine(init_alt, t_alt * 0.65, solver=solver, n_max_iters=20000)
            try_update(a_alt, f"lp_n{alt_n}")
            rem = deadline - time.monotonic()
            if rem > 25:
                a_alt2 = lbgfs_logsumexp_refine(
                    best_seq, min(rem * 0.3, 30.0),
                    T_stages=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1],
                    maxiter_per_stage=2000)
                try_update(a_alt2, f"lse_n{alt_n}")

    # === Phase 7: Perturbation escape ===
    print(f"\n=== Phase 7: Perturbation (score={best_score:.10f}) ===", flush=True)
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        t_pert_total = min(rem * 0.25, 120.0)

        perturbation_configs = [
            ('gauss_noise', lambda: best_seq + rng.standard_normal(n_cur) * 0.02),
            ('cos_mod', lambda: best_seq + 0.04 * np.cos(2 * np.pi * 2 * np.arange(n_cur) / n_cur)),
            ('sparse_scale', None),
        ]

        for pname, pfunc in perturbation_configs:
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            t_single = min(t_pert_total / 3.0, rem - 15)
            if t_single < 10:
                continue

            if pname == 'sparse_scale':
                perturbed = best_seq.copy()
                n_pert = max(5, n_cur // 15)
                positions = rng.choice(n_cur, size=n_pert, replace=False)
                for pos in positions:
                    perturbed[pos] *= (0.5 + rng.random())
            else:
                perturbed = pfunc()

            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps < 1e-15:
                continue
            perturbed = perturbed / ps

            a_pert, _ = nesterov_lp_refine(perturbed, t_single, solver=solver, n_max_iters=15000)
            try_update(a_pert, f"pert_{pname}")

    # === Phase 8: Final precise polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.10f}) ===", flush=True)
    rem = deadline - time.monotonic()
    if rem > 25:
        t_fin = min(rem * 0.35, 60.0)
        if t_fin > 10:
            a_fin = lbgfs_logsumexp_refine(
                best_seq, t_fin,
                T_stages=[2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005],
                maxiter_per_stage=4000)
            try_update(a_fin, "final_lse")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.3, 25.0)
        if t_ps > 8:
            a_ps2, _ = peak_shaving(best_seq, t_ps, n_steps=15000)
            try_update(a_ps2, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_sg = min(rem * 0.4, 20.0)
        if t_sg > 5:
            a_sg2, _ = subgradient_refine(best_seq, t_sg, n_steps=10000)
            try_update(a_sg2, "final_sg")

    # One more LP round if time permits
    rem = deadline - time.monotonic()
    if rem > 40:
        t_final_lp = min(rem * 0.5, 60.0)
        if t_final_lp > 20:
            print(f"\n=== Final LP round ({t_final_lp:.0f}s) ===", flush=True)
            a_final_lp, _ = nesterov_lp_refine(best_seq, t_final_lp, solver=solver, n_max_iters=30000)
            try_update(a_final_lp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})", flush=True)

    if best_seq is None:
        return [1.0 / 1200] * 1200

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
