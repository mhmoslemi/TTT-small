    r = b - _toepmat_vec(p, x)
    pv = r.copy()
    r0 = r.dot(r)
    for _ in range(max_iters):
        Mp = _toepmat_vec(p, pv)
        pMp = pv.dot(Mp)
        if abs(pMp) < 1e-18:
            break
        alpha = r0 / pMp
        x += alpha * pv
        r -= alpha * Mp
        r_new = r.dot(r)
        if np.sqrt(r_new) < tol * np.sqrt(r0) + 1e-15:
            break
        beta_cg = r_new / r0
        pv = r + beta_cg * pv
        r0 = r_new
    return x


def _qp_solve(p, a_init, n_inner=300, lr=0.04, seed=0):
    """Solve min a^T M a s.t. simplex via infeasible IPM + PG fallback."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    t0ip = time.monotonic()

    mu = 0.5
    z = 1e-3 * np.ones(n)
    a = _project_simplex(a_init.copy())

    for ip_iter in range(5):
        if time.monotonic() - t0ip > 30.0:
            break
        Ba = _toepmat_vec(p, a)
        grad = 2.0 * Ba - mu * z / a
        d = _m_inv_ones(p, n, max_iters=40, tol=1e-12)
        alpha = 1.0
        for _ in range(30):
            a_t = np.maximum(a + alpha * d, 1e-12)
            z_t = z - alpha * 2.0 * np.maximum(Ba + d, 0.0)
            z_t = np.maximum(z_t, 1e-12)
            if np.all((a_t * z_t) > 1e-14):
                break
            alpha *= 0.5
        a = a + alpha * d
        a = np.maximum(a, 1e-12)
        z = z - alpha * 2.0 * np.maximum(Ba + d, 0.0)
        z = np.maximum(z, 1e-12)
        mu *= 0.5

    a = _project_simplex(a)
    s = a.sum()
    if s > 1e-15:
        a = a / s

    x_prev = a.copy()
    for inner in range(n_inner):
        grad = 2.0 * _toepmat_vec(p, a)
        t_c = 1.0 + np.sqrt(1.0 + 2.0 * inner)
        t_p = 1.0 + np.sqrt(1.0 + 2.0 * max(0, inner - 1)) if inner > 0 else 1.0
        beta = (t_p - 1.0) / t_c
        y = a + beta * (a - x_prev)
        y = _project_simplex(y)
        gy = 2.0 * _toepmat_vec(p, y)
        lr_y = lr / (1.0 + 0.001 * inner)
        a_new = _project_simplex(y - lr_y * gy)
        x_prev = a.copy()
        a = a_new

    return a


def _minmax_alt(n, time_budget, T_start=50.0, T_end=0.3, n_inner=100, seed=42, init=None):
    """Min-max alternating with temperature annealing and IPM inner solver."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    if init is not None:
        a = _project_simplex(np.maximum(init.astype(np.float64), 0))
    else:
        a = np.ones(n) / n
    p = np.ones(2 * n - 1) / (2 * n - 1)
    best_a = a.copy()
    best_score = _score_norm(a)
    n_outer = 5000
    T_sched = np.geomspace(T_start, T_end, min(n_outer, 800))

    for outer in range(n_outer):
        if time.monotonic() - t0 > time_budget:
            break
        T = T_sched[min(outer, len(T_sched) - 1)]
        r = _autocorr(a)
        r = np.maximum(r, 1e-15)
        log_w = r / T
        log_w -= log_w.max()
        w = np.exp(log_w)
        p = p * w
        p = p / p.sum()
        a = _qp_solve(p, a, n_inner=int(n_inner * (0.5 + 0.5 * outer / 100.0)), seed=outer)
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if outer % 300 == 0 and outer > 0:
            print(f"      MM o={outer} T={T:.2f} best={best_score:.6f} "
                  f"({time.monotonic()-t0:.0f}s)")
    return best_a, best_score


def _lse_polish(a0, time_budget, T_start=3.0, T_end=400.0, lr_init=0.06,
                n_stages=10, n_per_stage=2000, seed=42):
    """Log-sum-exp smoothed objective with Nesterov projected gradient."""
    n = len(a0)
    t0 = time.monotonic()
    a = _project_simplex(np.maximum(a0.astype(np.float64), 0))
    x_prev = a.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    T_sched = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = T_sched[stage]
        for step in range(n_per_stage):
            if time.monotonic() - t0 > time_budget:
                break
            t_c = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            t_p = (1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) ** 2) / 2.0) if step_count > 0 else 1.0
            beta = (t_p - 1.0) / t_c
            y = a + beta * (a - x_prev)
            y = _project_simplex(y)
            r = _autocorr(y)
            Tr = T * r - T * r.max()
            w = np.exp(Tr)
            w /= w.sum()
            grad = 2.0 * _toepmat_vec(w, y)
            lr = lr_init / (1.0 + 0.003 * step_count)
            x_prev = a.copy()
            a = _project_simplex(y - lr * grad)
            step_count += 1
            if step_count % 200 == 0:
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def _peak_shave(a0, time_budget, n_steps=30000, seed=42):
    """Local peak-shaving via coordinate and pair adjustments."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = _project_simplex(np.maximum(a0.astype(np.float64), 0))
    best_a = a.copy()
    best_score = _score_norm(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_budget:
            break
        r = _autocorr(a)
        r_work = r.copy()
        if len(r_work) > 2:
            r_work[0] = -np.inf
            r_work[-1] = -np.inf
        pk = int(np.argmax(r_work))
        thr = r_work[pk] * 0.95
        near = np.where(r_work >= thr)[0][:4]
        hit = np.zeros(n)
        for k in near:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            vals = a[idx] * a[k - idx]
            top5 = idx[np.argsort(vals)[::-1][:5]]
            hit[top5] += 1
        tc = np.argsort(hit)[::-1][:8]
        improved = False
        for i in tc:
            if a[i] < 1e-14:
                continue
            for fac in [0.96, 0.92, 0.85]:
                at = a.copy()
                at[i] *= fac
                at = _project_simplex(at)
                sc = _score_norm(at)
                if sc < best_score - 1e-14:
                    a = at
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            vals = a[idx] * a[pk - idx]
            tc4 = idx[np.argsort(vals)[::-1][:4]]
            for ci in tc4:
                i = ci
                j = pk - i
                if i == j or j < 0 or j >= n or i < 0 or i >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.94, 0.90]:
                    for f2 in [0.94, 0.90]:
                        at = a.copy()
                        at[i] *= f1
                        at[j] *= f2
                        at = _project_simplex(at)
                        sc = _score_norm(at)
                        if sc < best_score - 1e-14:
                            a = at
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        if not improved:
            at = a.copy()
            for i in tc[:3]:
                if a[i] > 1e-14:
                    at[i] *= 0.97
            at = _project_simplex(at)
            sc = _score_norm(at)
            if sc < best_score - 1e-14:
                a = at
                best_score = sc
                best_a = a.copy()
            else:
                break
    return best_a, best_score


def _get_lp_solver():
    global _LP_SOLVER
    if _LP_SOLVER is not None:
        return _LP_SOLVER
    import cvxpy as cp
    for sn in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=sn)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _LP_SOLVER = sn
                return sn
        except Exception:
            continue
    _LP_SOLVER = cp.ECOS
    return _LP_SOLVER


def _lp_best_response(a_fixed, solver, max_iters=1):
    """LP: given a (sum=1, nonneg), find b minimizing max_k (a*b)_k."""
    import cvxpy as cp
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    bv = np.asarray(b_var.value, dtype=np.float64)
    bv = np.maximum(bv, 0)
    s = bv.sum()
    if s < 1e-15:
        return None
    return bv / s


def _lp_refine(a0, time_budget, solver, n_max=20000, seed=42):
    """Alternating LP refinement with Nesterov dual averaging."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = _project_simplex(np.maximum(a0.astype(np.float64), 0))
    y = a.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score

    for it in range(n_max):
        if time.monotonic() - t0 > time_budget:
            break
        b = _lp_best_response(y, solver)
        if b is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b
        y = _project_simplex(y)
        a = ((it + 1) * a + alpha * (it + 2) * b) / (it + 1 + alpha * (it + 2))
        a = _project_simplex(a)
        sc_a = _score_norm(a)
        sc_y = _score_norm(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall = 0
        if it % 300 == 0:
            if best_score < prev_best - 1e-7:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 3:
                y = y + 0.02 * rng.standard_normal(n)
                y = _project_simplex(y)
                stall = 0
        if (it + 1) % 2000 == 0:
            print(f"      LP it={it+1} ({time.monotonic()-t0:.0f}s): "
                  f"score={best_score:.6f}")
    return best_a, best_score


def _interp_seq(f, new_n):
    """Interpolate to new length, preserve sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    ox = np.linspace(0, 1, old_n)
    nx = np.linspace(0, 1, new_n)
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi = fi / ss
    return fi


def _safe_eval(a):
    try:
        al = [float(max(0.0, min(1000.0, x))) for x in a]
        if sum(al) < 0.01:
            return float('inf')
        return float(evaluate_sequence(al))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase search: min-max alternating, LSE polish, peak shaving,
    LP refinement, multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')
    solver = _get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent
    n_best = 960
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        parent_raw = np.clip(parent_raw, 0, 1000)
        if parent_raw.ndim == 1 and parent_raw.size > 0 and \
                np.isfinite(parent_raw).all() and parent_raw.sum() > 1e-10:
            parent = parent_raw / parent_raw.sum()
            ps = _safe_eval(parent)
            print(f"Parent: n={len(parent)}, score={ps:.6f}")
            best_seq = parent.copy()
            best_score = ps
            n_best = len(parent)
    except (NameError, Exception):
        best_seq = np.ones(n_best) / n_best
        best_score = _safe_eval(best_seq)
        print(f"Default uniform init: n={n_best}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a.astype(np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _safe_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Min-max alternating (main) ===
    print(f"\n=== Phase 1: Min-Max Alt n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 540.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_mm, _ = _minmax_alt(n_best, t_alloc, T_start=80.0, T_end=0.2,
                              n_inner=120, seed=seed, init=best_seq)
        try_update(a_mm, "mm_main")

    # === Phase 2: LSE polish ===
    print(f"\n=== Phase 2: LSE Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lse, _ = _lse_polish(best_seq, t_alloc, T_start=3.0, T_end=500.0,
                               lr_init=0.06, n_stages=10, n_per_stage=2500, seed=seed)
        try_update(a_lse, "lse_1")

    # === Phase 3: LP refinement ===
    print(f"\n=== Phase 3: LP Refine (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = _lp_refine(best_seq, t_alloc, solver, n_max=50000, seed=seed)
        try_update(a_lp, "lp_1")

    # === Phase 4: Second min-max round ===
    print(f"\n=== Phase 4: MM Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        a_mm2, _ = _minmax_alt(n_best, t_alloc, T_start=20.0, T_end=0.1,
                               n_inner=150, seed=seed + 1, init=best_seq)
        try_update(a_mm2, "mm_2")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shave (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 10:
        a_ps, _ = _peak_shave(best_seq, t_alloc, n_steps=50000, seed=seed)
        try_update(a_ps, "peak_1")

    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.6f}) ===")
    alt_ns = []
    for d in [-100, -50, -20, 20, 50, 100, 150, 200]:
        an = n_best + d
        if 256 <= an <= 4096:
            alt_ns.append(an)
    alt_ns = sorted(set(alt_ns))[:4]
    for an in alt_ns:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_tot = min(rem * 0.15, 150.0)
        if t_tot < 30:
            break
        print(f"  n={an}, budget={t_tot:.0f}s")
        init_alt = _interp_seq(best_seq, an)
        a_alt, _ = _minmax_alt(an, t_tot * 0.5, T_start=30.0, T_end=0.3,
                               n_inner=80, seed=seed + an, init=init_alt)
        try_update(a_alt, f"mm_n{an}")
        rem = deadline - time.monotonic()
        if rem > 20:
            a_alt2, _ = _lse_polish(a_alt, min(rem * 0.3, 60.0),
                                    T_start=2.0, T_end=200.0, lr_init=0.05,
                                    n_stages=5, n_per_stage=1500, seed=seed + an)
            try_update(a_alt2, f"lse_n{an}")

    # === Phase 7: Final combined polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_f = min(rem * 0.25, 90.0)
        a_f, _ = _lse_polish(best_seq, t_f, T_start=5.0, T_end=600.0,
                             lr_init=0.03, n_stages=6, n_per_stage=2000, seed=seed)
        try_update(a_f, "fin_lse")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.15, 40.0)
        a_ps2, _ = _peak_shave(best_seq, t_ps, n_steps=20000, seed=seed + 99)
        try_update(a_ps2, "fin_ps")

    # Final verification
    if best_seq is not None:
        fs = _safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
    print(f"\nFINAL: {best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")
    if best_seq is None:
        best_seq = np.ones(n_best) / n_best
    return [float(x) for x in best_seq]
