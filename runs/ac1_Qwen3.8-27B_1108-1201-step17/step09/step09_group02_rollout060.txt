def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def compute_score(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max())

def init_gs_asymmetric(n, rng, n_iters=200, asymmetry=0.7):
    """
    Gerchberg-Saxton style initialization with asymmetry.
    Iteratively:
    1. FFT of current signal
    2. Flatten the magnitude spectrum
    3. IFFT
    4. Clip negatives and apply asymmetric window
    """
    n_fft = 1 << (n - 1).bit_length()
    
    # Start with random non-negative signal with asymmetry
    t = np.arange(n, dtype=np.float64)
    # Asymmetric envelope: peaks off-center
    center = n * asymmetry  # Could be 0.3n or 0.7n
    width = n * 0.3
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    for it in range(n_iters):
        # FFT
        A = rfft(a, n_fft)
        
        # Flatten magnitude (soft version: pull magnitudes toward their mean)
        mags = np.abs(A)
        mean_mag = mags.mean()
        # Blend: 50% flat + 50% current
        new_mags = 0.5 * mean_mag + 0.5 * mags
        phases = np.angle(A)
        A = new_mags * np.exp(1j * phases)
        
        # IFFT
        a = irfft(A, n_fft)[:n]
        
        # Enforce non-negativity
        a = np.maximum(a, 0)
        
        # Apply light asymmetric re-weighting
        # This helps break symmetry
        weight = 1.0 + 0.1 * np.sin(np.pi * t / n)  # Slight asymmetry
        a *= weight
        a = np.maximum(a, 0)
        
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            break
        a = a / s
    
    return a

def init_flat_spec_asymmetric(n, rng, asym_shift=0.0):
    """Flat spectrum with asymmetric time-domain window."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    a = irfft(np.exp(1j * phases), n_fft)[:n]
    
    # Apply asymmetric window
    t = np.arange(n, dtype=np.float64)
    if asym_shift == 0:
        asym_shift = rng.uniform(0.25, 0.75)
    center = n * asym_shift
    width = n * 0.35
    window = np.exp(-0.3 * ((t - center) / width) ** 2)
    a *= window
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_multi_gaussian(n, rng):
    """Sum of multiple Gaussians at random positions."""
    a = np.zeros(n)
    t = np.arange(n, dtype=np.float64)
    n_gauss = rng.integers(2, 6)
    for _ in range(n_gauss):
        center = rng.uniform(0.1, 0.9) * n
        width = rng.uniform(0.05, 0.2) * n
        amp = rng.uniform(0.5, 1.5)
        a += amp * np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def coord_descent_fast(a0, time_limit, max_iter_gs=25, verbose=False, label=""):
    """
    Fast coordinate descent with golden section search.
    Optimized for speed: uses precomputation and avoids unnecessary copies.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            v = a[i]
            if v < 1e-15:
                continue

            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            # Unaffected max
            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]

            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            quad_idx = 2 * i - k_lo if has_quad else -1

            # c_vals[k_lo + idx] = conv[k_lo+idx] - contribution of a[i]
            c_vals = np.empty(n_aff)
            c_vals[:] = conv[k_lo:k_hi+1] - 2.0 * v * a_partners
            if has_quad:
                c_vals[quad_idx] += v * v  # Add back v^2, since we subtracted 2*v*v but should only subtract 2*v*a[i] = 2*v^2, and the quad term is v^2 not 2*v^2

            # Wait, let me redo this.
            # conv[k] = sum_j a[j] * a[k-j]
            # When we change a[i] from v to x, the change in conv[k] is:
            # For k = i + j (where j is a partner): change = (x - v) * a[j] + a[i] * change_in_a[k-i]
            # Since we only change a[i]:
            # For k where both i and k-i are valid and k-i != i:
            #   new_conv[k] = old_conv[k] - v*a[k-i] - a[i]*v + x*a[k-i] + a[i]*x
            #   = old_conv[k] - 2*v*a[k-i] + 2*x*a[k-i]  (if k-i != i)
            #   Wait, that's not right either.
            
            # Let me be very careful.
            # conv[k] = sum_j a[j] * a[k-j] (summing over valid j)
            # a[i] appears in conv[k] when:
            #   j = i (so k-i must be valid: 0 <= k-i < n) → term is a[i]*a[k-i]
            #   k-j = i (so j = k-i must be valid: 0 <= k-i < n) → term is a[k-i]*a[i]
            # If i != k-i (i.e., k != 2i): both terms are present, total contribution = 2*a[i]*a[k-i]
            # If i == k-i (i.e., k == 2i): only one term, contribution = a[i]^2

            # When we change a[i] from v to x:
            # If k != 2i: new contribution = 2*x*a[k-i], old = 2*v*a[k-i]
            #   change = 2*(x-v)*a[k-i]
            #   new_conv[k] = conv[k] + 2*(x-v)*a[k-i]
            # If k == 2i: new contribution = x^2, old = v^2
            #   change = x^2 - v^2
            #   new_conv[k] = conv[k] + x^2 - v^2

            # So: new_conv[k] = conv[k] - 2*v*a[partner] + 2*x*a[partner] (for k != 2i)
            #                             = conv[k] - v^2 + x^2 (for k == 2i)

            # Let me redo c_vals:
            # For k != 2i: new_conv[k] = (conv[k] - 2*v*a[partner]) + 2*x*a[partner]
            # For k == 2i: new_conv[k] = (conv[k] - v^2) + x^2

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            def score_func(x):
                ns = current_sum + (x - v)
                if ns < 0.01:
                    return 1e10
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                nm = max(unaff_max, float(nv.max()))
                return 2.0 * n * nm / (ns * ns)

            # Golden section search
            upper = max(2.0 * v + 0.5/n, 3.0/n)
            lo, hi = 0.0, upper
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = score_func(c)
            fd = score_func(d)
            for _ in range(max_iter_gs):
                if hi - lo < 1e-14 * max(1.0, v):
                    break
                if fc <= fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = score_func(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = score_func(d)
            best_x = (lo + hi) * 0.5

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def peak_directed_perturb(a, rng, strength=0.05):
    """
    Perturb the sequence by reducing values that contribute to peaks.
    This is more targeted than random perturbations.
    """
    n = len(a)
    conv = fast_convolve(a)
    peak_val = conv.max()
    
    # Find all lags where conv is close to the peak
    threshold = peak_val * 0.95
    peak_lags = np.where(conv >= threshold)[0]
    
    a_new = a.copy()
    for k in peak_lags[:5]:  # Limit to top 5 peaks
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        if len(idx) == 0:
            continue
        # Reduce some of the contributing values
        n_reduce = max(1, len(idx) // 4)
        # Pick the largest contributors
        contributions = a[idx] * a[np.maximum(0, k - idx)]  # approximate contribution
        picks = idx[np.argsort(contributions)[-n_reduce:]]
        for ii in picks:
            a_new[ii] *= (1.0 - strength * rng.uniform(0.5, 1.5))
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def random_perturb(a, rng, sigma=0.03):
    """Simple random perturbation."""
    a_new = a + rng.standard_normal(len(a)) * sigma
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def spectral_flatten(a, n_iters=50, strength=0.3):
    """
    Apply spectral flattening: iteratively make the PSF (power spectrum) flatter.
    This helps reduce peaks in the convolution.
    """
    n = len(a)
    n_fft = 1 << (n - 1).bit_length()
    
    for _ in range(n_iters):
        A = rfft(a, n_fft)
        mags = np.abs(A)
        # Soft flatten: pull magnitudes toward geometric mean
        log_mags = np.log(mags + 1e-15)
        mean_log = log_mags.mean()
        new_log_mags = (1 - strength) * log_mags + strength * mean_log
        new_mags = np.exp(new_log_mags)
        phases = np.angle(A)
        A = new_mags * np.exp(1j * phases)
        a = irfft(A, n_fft)[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
    return a

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

But this creates a new array `nv` every call, which is expensive. For the golden section search, we're calling this ~30 times per coordinate, and we have n coordinates, so that's 30*n function evaluations per sweep, each creating an array of size ~n. That's O(n^2) memory allocations per sweep, which is slow.

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def compute_score_normalized(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max())

def init_gs_asymmetric(n, rng, n_iters=150, asymmetry=0.65, flatten_strength=0.4):
    n_fft = 1 << (n - 1).bit_length()
    t = np.arange(n, dtype=np.float64)
    center = n * asymmetry
    width = n * 0.3
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s
    for it in range(n_iters):
        A = rfft(a, n_fft)
        mags = np.abs(A)
        log_mags = np.log(mags + 1e-15)
        mean_log = log_mags.mean()
        new_log_mags = (1.0 - flatten_strength) * log_mags + flatten_strength * mean_log
        new_mags = np.exp(new_log_mags)
        phases = np.angle(A)
        A = new_mags * np.exp(1j * phases)
        a = irfft(A, n_fft)[:n]
        a = np.maximum(a, 0)
        weight = 1.0 + 0.05 * np.sin(np.pi * t / n)
        a = a * weight
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            break
        a = a / s
    return a

def init_flat_spec_asymmetric(n, rng, asym_shift=0.0):

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def spectral_flatten(a, n_iters=50, strength=0.35):
    n = len(a)
    n_fft = 1 << (n - 1).bit_length()
    for _ in range(n_iters):
        A = rfft(a, n_fft)
        mags = np.abs(A)
        log_mags = np.log(mags + 1e-15)
        mean_log = log_mags.mean()
        new_log_mags = (1.0 - strength) * log_mags + strength * mean_log
        new_mags = np.exp(new_log_mags)
        phases = np.angle(A)
        A = new_mags * np.exp(1j * phases)
        a = irfft(A, n_fft)[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            return a
        a = a / s
    return a

def _golden_section(c_vals, a_partners, unaff_max, has_quad, quad_idx,
                    current_sum, v, n, lo, hi, max_iter):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    ns_c = current_sum + (c - v)
    nv_c = c_vals + 2.0 * c * a_partners
    if has_quad:
        nv_c[quad_idx] = c_vals[quad_idx] + c * c
    fc = 2.0 * n * max(unaff_max, float(nv_c.max())) / (ns_c * ns_c) if ns_c > 0.01 else 1e10

    ns_d = current_sum + (d - v)
    nv_d = c_vals + 2.0 * d * a_partners
    if has_quad:
        nv_d[quad_idx] = c_vals[quad_idx] + d * d
    fd = 2.0 * n * max(unaff_max, float(nv_d.max())) / (ns_d * ns_d) if ns_d > 0.01 else 1e10

    for _ in range(max_iter):
        if hi - lo < 1e-14 * max(1.0, v):
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            ns_c = current_sum + (c - v)
            if ns_c < 0.01:
                fc = 1e10
            else:
                nv_c = c_vals + 2.0 * c * a_partners
                if has_quad:
                    nv_c[quad_idx] = c_vals[quad_idx] + c * c
                fc = 2.0 * n * max(unaff_max, float(nv_c.max())) / (ns_c * ns_c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            ns_d = current_sum + (d - v)
            if ns_d < 0.01:
                fd = 1e10
            else:
                nv_d = c_vals + 2.0 * d * a_partners
                if has_quad:
                    nv_d[quad_idx] = c_vals[quad_idx] + d * d
                fd = 2.0 * n * max(unaff_max, float(nv_d.max())) / (ns_d * ns_d)
    return (lo + hi) * 0.5

def coord_descent(a0, time_limit, max_iter_gs=30, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            v = a[i]
            if v < 1e-15:
                continue
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1
            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))
            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            quad_idx = 2 * i - k_lo if has_quad else -1
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            upper = max(2.0 * v + 0.5 / n, 4.0 / n)
            best_x = _golden_section(c_vals, a_partners, unaff_max, has_quad,
                                     quad_idx, current_sum, v, n, 0.0, upper, max_iter_gs)
            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes
                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score, sweeps

def peak_directed_perturb(a, rng, strength=0.06):

        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)

            continue
        n_reduce = max(1, len(idx) // 4)
