def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a_fixed, solver):
    """Find b that minimizes max conv(a_fixed, b) subject to b >= 0, sum(b) = 1."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver, max_iters=50000):
    """Nesterov dual averaging with LP best response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, _ = lp_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        sc_a = score_of(a)
        sc_y = score_of(y)
        
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        if (it + 1) % 2000 == 0:
            print(f"    NLP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def structured_init(n, variant):
    """Create structured initial sequence."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2 * x - 1)  # Triangle
    elif variant == 4:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif variant == 5:
        # Uniform
        a = np.ones(n)
    elif variant == 6:
        # Bimodal
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:

"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines SPSA global search, multi-scale p-norm L-BFGS, Nesterov LP alternating,
aggressive local refinement, and length exploration to beat previous best of 1.5047.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute objective: 2n * max(conv(a,a)) / (sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max() / (s * s)


def evaluate_safe(a):
    """Evaluate using actual evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def update_best(best_state, a, label):
    """Try to update best_state with candidate a."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False


def pnorm_obj(z, p):
    """p-norm smoothed objective in log-domain."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def pnorm_grad(z, p):
    """Gradient of p-norm smoothed objective w.r.t. z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def pnorm_lbfgs(a0, time_limit, p_values):
    """L-BFGS with p-norm smoothing over multiple stages."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    total_stages = len(p_values)
    iters_per_stage = max(80, 8000 // total_stages)

    for p in p_values:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=pnorm_obj, x0=z, args=(p,),
                jac=pnorm_grad, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = 2.0 * n * fast_convolve(a_cur).max()
        print(f"    p={p}: score={sc:.6f} ({time.monotonic() - t0:.0f}s)")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


def spsa_optimize(a0, time_limit, c0, alpha, gamma, mu):
    """SPSA derivative-free optimizer with adaptive gains."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        a_k = c0 / ((it + 10) ** alpha)
        c_k = mu / ((it + 10) ** gamma)

        delta = np.ones(n)
        mask = np.random.random(n) < 0.5
        delta[mask] = -1

        a_plus = np.maximum(a + c_k * delta, 0)
        sp = a_plus.sum()
        if sp < 1e-15:
            continue
        a_plus = a_plus / sp

        a_minus = np.maximum(a - c_k * delta, 0)
        sm = a_minus.sum()
        if sm < 1e-15:
            continue
        a_minus = a_minus / sm

        f_plus = score_of(a_plus)
        f_minus = score_of(a_minus)

        g = (f_plus - f_minus) / (2.0 * c_k)
        a_new = a - a_k * g * delta
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn > 1e-15:
            a_new = a_new / sn
            sc = score_of(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new

        if it % 50000 == 0:
            print(f"    SPSA it={it} ({time.monotonic() - t0:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def build_conv_matrix(a):
    """Matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a_fixed, solver):
    """Find b minimizing max conv(a_fixed, b) s.t. b>=0, sum(b)=1."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def nesterov_lp(a0, time_limit, solver, max_iters=50000):
    """Nesterov dual averaging with LP best response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new = lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        y = y / y.sum()
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        a = a / a.sum()
        sc_a = score_of(a)
        sc_y = score_of(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        if (it + 1) % 5000 == 0:
            print(f"    NLP it={it + 1} ({time.monotonic() - t0:.0f}s): best={best_score:.6f}")

    return best_a, best_score

    # === Phase 1: p-norm L-BFGS (main workhorse) ===
    print(f"\n=== P1: p-norm L-BFGS (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.22, 400.0)
    if t1 > 30:
        a1 = pnorm_lbfgs(best_state['seq'], t1,
                         p_values=[4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096, 8192])
        update_best(best_state, a1, "pnorm_1")

    # === Phase 2: SPSA global search ===
    print(f"\n=== P2: SPSA (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 250.0)
    if t2 > 30:
        a2, _ = spsa_optimize(best_state['seq'], t2, c0=0.03, alpha=0.6, gamma=0.6, mu=0.03)
        update_best(best_state, a2, "spsa_1")

    # SPSA from perturbed start
    rem = deadline - time.monotonic()
    if rem > 40:
        n_cur = len(best_state['seq'])
        pert = best_state['seq'] * (1.0 + 0.15 * np.random.standard_normal(n_cur))
        pert = np.maximum(pert, 0)
        pert = pert / pert.sum()
        t2b = min(rem * 0.3, 100.0)
        if t2b > 20:
            a2b, _ = spsa_optimize(pert, t2b, c0=0.05, alpha=0.55, gamma=0.55, mu=0.05)
            update_best(best_state, a2b, "spsa_pert")

    # === Phase 3: Nesterov LP ===
    print(f"\n=== P3: Nesterov LP (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.18, 300.0)
    if t3 > 30:
        a3, _ = nesterov_lp(best_state['seq'], t3, solver, max_iters=50000)
        update_best(best_state, a3, "nesterov_1")

    # === Phase 4: Local refinement ===
    print(f"\n=== P4: Local Refine (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 150.0)
    if t4 > 10:
        a4, _ = local_refine(best_state['seq'], t4)
        update_best(best_state, a4, "local_1")

    # === Phase 5: Length exploration ===
    print(f"\n=== P5: Length Exploration (best={best_state['score']:.6f}) ===")
    n_best = len(best_state['seq'])
    lengths_to_try = []
    for dl in [-128, -64, -32, 32, 64, 128, 192, 256, 384, 512, 768]:
        alt = n_best + dl
        if 256 <= alt <= 4096:
            lengths_to_try.append(alt)
    for rn in [384, 512, 640, 768, 896, 1024, 1152, 1280, 1536, 1920, 2048, 2560, 3072]:
        if abs(rn - n_best) > 64 and 256 <= rn <= 4096:
            lengths_to_try.append(rn)
    lengths_to_try = sorted(set(lengths_to_try))
    lengths_to_try.sort(key=lambda x: abs(x - n_best))

    n_explored = 0
    for alt_n in lengths_to_try[:10]:
        rem = deadline - time.monotonic()
        if rem < 50 or n_explored >= 7:
            break
        n_explored += 1
        t_total = min(rem * 0.12, 100.0)
        if t_total < 25:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_seq(best_state['seq'], alt_n)
        a_alt = pnorm_lbfgs(init_alt, t_total * 0.7, p_values=[16, 64, 256, 1024, 4096])
        improved = update_best(best_state, a_alt, f"pn_n{alt_n}")
        if improved:
            rem = deadline - time.monotonic()
            if rem > 25:
                t_lp = min(rem * 0.1, 60.0)
                if t_lp > 15:
                    a_alt2, _ = nesterov_lp(best_state['seq'], t_lp, solver, max_iters=15000)
                    update_best(best_state, a_alt2, f"lp_n{alt_n}")

    # === Phase 6: Diverse SPSA restarts ===
    print(f"\n=== P6: Diverse SPSA (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_state['seq'])
        for noise_scale in [0.05, 0.1, 0.2, 0.35]:
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            pert = best_state['seq'] * (1.0 + noise_scale * np.random.standard_normal(n_cur))
            pert = np.maximum(pert, 0)
            pert = pert / pert.sum()
            t_sub = min(rem * 0.25, 80.0)
            if t_sub < 15:
                break
            a_p, _ = spsa_optimize(pert, t_sub, c0=noise_scale * 0.4,
                                   alpha=0.55, gamma=0.55, mu=noise_scale * 0.4)
            update_best(best_state, a_p, f"spsa_ns{noise_scale}")

    # === Phase 7: Final p-norm + local + LP polish ===
    print(f"\n=== P7: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.25, 80.0)
        a_fin = pnorm_lbfgs(best_state['seq'], t_fin,
                            p_values=[128, 512, 2048, 8192, 16384, 32768])
        update_best(best_state, a_fin, "final_pn")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_lr = min(rem * 0.30, 60.0)
        a_lr, _ = local_refine(best_state['seq'], t_lr)
        update_best(best_state, a_lr, "final_local")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lp = min(rem * 0.40, 40.0)
        a_lp, _ = nesterov_lp(best_state['seq'], t_lp, solver, max_iters=8000)
        update_best(best_state, a_lp, "final_lp")

    # === Phase 8: One more p-norm round if time allows ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t_extra = min(rem * 0.5, 50.0)
        a_x = pnorm_lbfgs(best_state['seq'], t_extra,
                          p_values=[512, 2048, 8192, 32768, 65536])
        update_best(best_state, a_x, "extra_pn")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs
        print(f"\nFINAL: score={best_state['score']:.6f}, n={len(best_state['seq'])}")

    return [float(x) for x in best_state['seq']]
