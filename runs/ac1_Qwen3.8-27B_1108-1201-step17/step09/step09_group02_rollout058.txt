Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FFT-based fast convolution
2. Exact coordinate descent with golden section search
3. Simulated annealing with structured perturbations
4. Multi-length exploration
5. Final polishing with fine coordinate descent

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Compute score for a sequence assumed to sum to 1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max())

def golden_section_search(f, lo, hi, max_iter=50):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd

def golden_section_search_1d(f, lo, hi, max_iter=50):
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    d = lo + gr * (hi - lo)

    fd = f(d)

            fd = fc

            c = d
            fc = fd
