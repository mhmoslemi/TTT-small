"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast active-set projected gradient
with FFT-based correlation, adaptive peak weighting, Nesterov momentum,
population-based multi-length search, and final LogSumExp L-BFGS polishing.
Key ideas: (1) gradient of sum-of-squares of active peaks computed via cross-correlation in O(n log n),
(2) adaptive learning rate with backtracking, (3) multi-resolution exploration,
(4) direct peak-shaving coordinate moves for final polish.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def next_power_of_2(x):
    return 1 << (x - 1).bit_length()


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = next_power_of_2(size)
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def project_simplex_fast(v):
    """Project onto probability simplex using sort-based method."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1) / rho > 0
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0]) + 1
    theta = (cssv[rho - 1] - 1) / rho
    return np.maximum(v - theta, 0)


def compute_gradient_active(p, c, active_mask, fft_size):
    """Compute gradient of sum_{k in active} c_k^2 efficiently via FFT cross-correlation."""
    d = np.where(active_mask, c, 0.0)
    D = rfft(d, fft_size)
    P = rfft(p, fft_size)
    g = irfft(D * np.conj(P), fft_size)[:len(p)]
    return 4.0 * g


def active_set_gradient_descent(p0, time_budget, n_max_iters=2000000, verbosity=0):
    """
    Fast gradient descent on the probability simplex for minimizing sum of squares
    of active autocorrelation peaks. Uses FFT for O(n log n) per iteration.
    """
    p = np.asarray(p0, dtype=np.float64).copy()
    n = len(p)
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    best_p = p.copy()
    best_score = eval_local(p)
    fft_size = next_power_of_2(2 * n - 1)

    lr = 0.5
    momentum = np.zeros(n)
    beta = 0.9
    no_improve = 0
    t0 = time.monotonic()
    stall_count = 0
    rng = np.random.default_rng(777)
    active_power = 3.0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_budget:
            break

        c = fast_conv(p)
        M = c.max()
        if M < 1e-15:
            break

        # Soft active set: weight by (c/M)^power
        ratio = c / M
        weights = np.where(ratio > 0.01, ratio ** active_power, 0.0)
        active_mask = weights > 1e-8

        # Compute gradient via FFT
        g = compute_gradient_active(p, c, active_mask, fft_size)

        # Nesterov momentum
        momentum = beta * momentum + (1 - beta) * g
        p_trial = p - lr * momentum
        p_trial = np.maximum(p_trial, 0)
        st = p_trial.sum()
        if st < 1e-15:
            p_trial = p.copy()
        else:
            p_trial = p_trial / st

        new_score = 2.0 * n * fast_conv(p_trial).max()

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_p = p_trial.copy()
            p = p_trial
            no_improve = 0
            lr = min(lr * 1.005, 2.0)
            stall_count = 0
        else:
            no_improve += 1
            lr *= 0.97
            if no_improve > 200:
                # Perturb and restart
                noise = rng.standard_normal(n) * 0.01
                p = best_p + noise
                p = project_simplex_fast(p)
                momentum = np.zeros(n)
                lr = 0.3
                no_improve = 0
                stall_count += 1
                if stall_count > 5:
                    break

            if no_improve > 50:
                lr = max(lr * 0.95, 0.01)

        # Slowly increase active_power to focus more on peaks
        if it % 5000 == 0 and it > 0:
            active_power = min(8.0, active_power + 0.2)

        if verbosity > 0 and it % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AGD iter={it} ({elapsed:.0f}s): score={best_score:.8f} lr={lr:.4f} power={active_power:.1f}")

    return best_p, best_score


def peak_shave_descent(p0, time_budget, n_max_iters=50000):
    """Directly shave coordinates that contribute most to the peak."""
    p = np.asarray(p0, dtype=np.float64).copy()
    n = len(p)
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    best_p = p.copy()
    best_score = eval_local(p)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(999)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_budget:
            break

        c = fast_conv(p)
        peak_k = int(np.argmax(c))

        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = p[idx] * p[peak_k - idx]
        sorted_idx = idx[np.argsort(contribs)[::-1]]

        improved = False
        top_m = min(5, len(sorted_idx))
        for pos in sorted_idx[:top_m]:
            if p[pos] < 1e-10:
                continue
            for factor in [0.98, 0.96, 0.94, 0.90, 0.85]:
                p_trial = p.copy()
                p_trial[pos] *= factor
                diff = p[pos] - p_trial[pos]
                if diff < 1e-14:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others_sum = p_trial[mask].sum()
                if others_sum < 1e-15:
                    break
                p_trial[mask] += diff * p_trial[mask] / others_sum
                p_trial = p_trial / p_trial.sum()

                sc = 2.0 * n * fast_conv(p_trial).max()
                if sc < best_score - 1e-13:
                    p = p_trial
                    best_score = sc
                    best_p = p.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 100:
                # Random restart
                noise = rng.standard_normal(n) * 0.005
                p = best_p + noise
                p = project_simplex_fast(p)
                stall = 0
                if stall > 10:
                    break
        else:
            stall = 0

    return best_p, best_score


def logsumexp_lbgfs(p0, time_budget, T_stages=None, maxiter=3000):
    """LogSumExp smoothed optimization with L-BFGS-B."""
    n = len(p0)
    p = np.asarray(p0, dtype=np.float64).copy()
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    z = np.log(np.maximum(p, 1e-300))
    best_p = p.copy()
    best_score = eval_local(p)
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]

    iters_per_stage = max(20, maxiter // len(T_stages))

    for T in T_stages:
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 2:
            break

        def lse_obj(z_val, T_val=T):
            p_val = softmax_from_z(z_val)
            c = fast_conv(p_val)
            max_c = c.max()
            if max_c < 1e-300:
                return 0.0
            shifted = c / T_val - max_c / T_val
            return 2.0 * n * (max_c + T_val * np.log(np.exp(shifted).sum()) / (2 * n - 1))

        def lse_grad(z_val, T_val=T):
            p_val = softmax_from_z(z_val)
            c = fast_conv(p_val)
            max_c = c.max()
            shifted = c / T_val - max_c / T_val
            w = np.exp(shifted)
            w = w / w.sum()
            # Gradient wrt p: 2n/(2n-1) * sum_k w[k] * d(c_k)/dp
            # d(c_k)/dp[i] = 2*p[k-i] (for valid k-i)
            grad_p = np.zeros(n)
            for k in range(len(c)):
                if w[k] < 1e-10:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_p[idx] += 2.0 * w[k] * p_val[k - idx]
            grad_p *= 2.0 * n / (2 * n - 1)
            # Convert to z-space: grad_z = p * (grad_p - p dot grad_p)
            S = np.dot(p_val, grad_p)
            return p_val * (grad_p - S)

        try:
            res = minimize(
                fun=lse_obj, x0=z, jac=lse_grad, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-15,
                         'gtol': 1e-10, 'maxfun': iters_per_stage * 50, 'disp': False}
            )
            z = res.x
            p_new = softmax_from_z(z)
            sc = eval_local(p_new)
            if sc < best_score:
                best_score = sc
                best_p = p_new.copy()
                z = np.log(np.maximum(p_new, 1e-300))
        except Exception:
            pass

    return best_p, best_score


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def structured_init(n, seed=0):
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        # Two-lobe: high at edges, low in middle
        a = np.exp(-4 * (x - 0.15) ** 2) + np.exp(-4 * (x - 0.85) ** 2) + 0.1
    else:
        a = 1.0 / (1 + np.exp(-10 * (0.5 - x))) + 0.05

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0)
    ss = result.sum()
    if ss > 1e-15:
        result = result / ss
    return result


def nesterov_dual_averaging(p0, time_budget, n_max_iters=50000):
    """
    Dual averaging approach: maintain dual variables for the active peak constraints
    and use them to guide the primal updates.
    """
    p = np.asarray(p0, dtype=np.float64).copy()
    n = len(p)
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    best_p = p.copy()
    best_score = eval_local(p)
    t0 = time.monotonic()
    fft_size = next_power_of_2(2 * n - 1)

    # Dual variable: weights for each lag
    lam = np.ones(2 * n - 1) / (2 * n - 1)
    lr_dual = 0.1
    lr_primal = 0.1
    no_improve = 0
    rng = np.random.default_rng(42)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_budget:
            break

        # Compute correlation
        c = fast_conv(p)
        M = c.max()

        # Update dual weights (emphasize peaks)
        ratio = c / max(M, 1e-15)
        lam_new = 0.7 * lam + 0.3 * (ratio ** 2)
        lam_new = lam_new / lam_new.sum()
        lam = lam_new

        # Compute weighted gradient: sum_k lam[k] * dc_k/dp
        # dc_k/dp[i] = 2*p[k-i] for valid k-i
        # Weighted grad: 2 * sum_k lam[k] * p[k-i]
        d_lam = lam
        D = rfft(d_lam, fft_size)
        P = rfft(p, fft_size)
        grad = 2.0 * irfft(D * np.conj(P), fft_size)[:n]

        # Primal update
        p_trial = p - lr_primal * grad
        p_trial = np.maximum(p_trial, 0)
        st = p_trial.sum()
        if st < 1e-15:
            p_trial = p.copy()
        else:
            p_trial = p_trial / st

        sc = 2.0 * n * fast_conv(p_trial).max()
        if sc < best_score - 1e-13:
            best_score = sc
            best_p = p_trial.copy()
            p = p_trial
            no_improve = 0
            lr_primal = min(lr_primal * 1.01, 1.0)
        else:
            no_improve += 1
            lr_primal *= 0.98
            if no_improve > 200:
                p = best_p + rng.standard_normal(n) * 0.008
                p = project_simplex_fast(p)
                lr_primal = 0.2
                no_improve = 0

    return best_p, best_score

    rem = deadline - time.monotonic()
    if rem > 40:
        # AGD again
        t6a = min(rem * 0.35, 150.0)
        if t6a > 15:
            a_opt, _ = active_set_gradient_descent(best_seq, t6a, n_max_iters=2000000)
            try_update(a_opt, "grassd_final")

        rem = deadline - time.monotonic()
        if rem > 20:
            t6b = min(rem * 0.35, 100.0)
            if t6b > 10:
                a_opt, _ = nesterov_dual_averaging(best_seq, t6b, n_max_iters=50000)
                try_update(a_opt, "dual_final")

        rem = deadline - time.monotonic()
        if rem > 15:
            t6c = min(rem * 0.5, 60.0)
            if t6c > 10:
                T_final = [3.0, 1.5, 0.8, 0.4, 0.2, 0.1, 0.05, 0.02, 0.01]
                a_opt = logsumexp_lbgfs(best_seq, t6c, T_stages=T_final, maxiter=5000)
                try_update(a_opt, "lse_final")

        rem = deadline - time.monotonic()
        if rem > 10:
            t6d = min(rem * 0.5, 30.0)
            if t6d > 5:
                a_opt, _ = peak_shave_descent(best_seq, t6d, n_max_iters=20000)
                try_update(a_opt, "shave_final")

    # === Phase 7: Try a slightly different length ===
    rem = deadline - time.monotonic()
    if rem > 60 and best_seq is not None:
        n_cur = len(best_seq)
        # Try n + 100 or n - 100
        for n_try in [n_cur + 100, n_cur - 100, n_cur + 200]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            if n_try < 50:
                continue
            init_alt = interpolate_seq(best_seq, n_try)
            t_alt = min(rem * 0.3, 50.0)
            if t_alt < 10:
                continue
            a_alt, _ = active_set_gradient_descent(init_alt, t_alt, n_max_iters=200000)
            sc = try_update(a_alt, f"alt_n{n_try}")
            rem = deadline - time.monotonic()
            if sc < best_score + 0.01 and rem > 20:
                t_fin = min(rem * 0.3, 30.0)
                if t_fin > 10:
                    a_fin = logsumexp_lbgfs(best_seq, t_fin,
                                            T_stages=[2.0, 1.0, 0.5, 0.2, 0.1, 0.05],
                                            maxiter=3000)
                    try_update(a_fin, f"lse_alt_n{n_try}")

    # Final verification
    if best_seq is not None:
        fs = eval_local(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
        # Sort results for reporting
        results.sort()
        for sc, n_val, _ in results[:5]:
            print(f"  Top result: score={sc:.8f}, n={n_val}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
