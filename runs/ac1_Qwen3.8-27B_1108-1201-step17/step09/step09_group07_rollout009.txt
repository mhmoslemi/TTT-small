"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced LP game-theoretic approach:
1. Nesterov accelerated projected gradient on log-sum-exp smoothed objective
2. LP-based alternating minimization (game equilibrium) with Nesterov dual averaging
3. Exact subgradient refinement targeting the true max
4. Multi-length exploration and local coordinate refinement
Key improvements: efficient LP via scipy HiGHS, adaptive step sizes, Polyak averaging,
multi-start with diverse initializations, and aggressive local search.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv(a,a))."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def grad_smooth_max(a, T):
    """Gradient of smoothed max convolution."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def nesterov_projected_gradient(a0, time_limit, T_start=2.0, T_end=400.0,
                                 lr_init=0.06, n_stages=10, steps_per_stage=3000):
    """Nesterov accelerated projected gradient with temperature annealing and Polyak step."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            global_step = stage * steps_per_stage + step
            # Adaptive learning rate
            lr = lr_init / (1.0 + 0.003 * global_step)

            # Nesterov momentum
            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            t_prev_calc = 1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) * (step_count - 1)) / 2.0 if step_count > 0 else 1.0
            beta = (t_prev_calc - 1.0) / t_curr if step_count > 0 else 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)

            grad = grad_smooth_max(y, T)
            x_prev = x.copy()
            x_new = project_simplex(y - lr * grad)

            # Occasional Polyak-like step using best score
            if step_count > 100 and step_count % 200 == 0:
                current_smooth = (1.0 / T) * np.log(np.exp(T * fast_convolve(x)).sum())
                gap = current_smooth - best_score / (2.0 * n)
                if gap > 1e-8:
                    grad_norm_sq = np.sum(grad * grad)
                    if grad_norm_sq > 1e-12:
                        polyak_lr = min(lr * 2.0, gap / grad_norm_sq)
                        x_polyak = project_simplex(y - polyak_lr * grad)
                        sc_polyak = score_normalized(x_polyak)
                        if sc_polyak < best_score:
                            x = x_polyak
                        else:
                            x = x_new
                    else:
                        x = x_new
                else:
                    x = x_new
            else:
                x = x_new
            step_count += 1

            if step_count % 40 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()

    return best_a, best_score


def build_lp_matrices(a_fixed):
    """Build A_ub and bounds for the LP: min M s.t. T@b - M*1 <= 0, sum(b)=1, b>=0."""
    n = len(a_fixed)
    size = 2 * n - 1
    # A_ub has (size) rows for T@b - M <= 0, plus we handle sum(b)=1 as equality
    # Variables: [b_0, ..., b_{n-1}, M]
    A_ub = np.zeros((size, n + 1))
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[k, j] = a_fixed[j]
        A_ub[k, n] = -1.0  # -M
    b_ub = np.zeros(size)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    bounds = [(0, None)] * n + [(0, None)]
    return A_ub, b_ub, A_eq, b_eq, bounds


def solve_lp_best_response(a_fixed):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b) via LP."""
    n = len(a_fixed)
    A_ub, b_ub, A_eq, b_eq, bounds = build_lp_matrices(a_fixed)
    try:
        res = linprog(c=np.zeros(n + 1) + np.array([0.0]*n + [1.0]),
                      A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            b_val = res.x[:n]
            M_val = res.x[n]
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val = b_val / s
                return b_val, float(M_val)
    except Exception:
        pass
    return None, None


def nesterov_lp_refine(a0, time_limit, n_max_iters=100000):
    """Nesterov dual averaging LP refinement with convergence tracking."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()  # dual variable (mixed strategy for player 2)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)

        # Nesterov dual averaging for primal
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)

        score_a = score_normalized(a)
        score_y = score_normalized(y)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 300 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 5:
                rng = np.random.default_rng(it + 42)
                perturbation = 0.01 * rng.standard_normal(n)
                y = project_simplex(y + perturbation)
                a = project_simplex(a + perturbation * 0.5)
                stall_count = 0

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it+1} ({elapsed:.0f}s): best_score={best_score:.8f} M_b={M_b:.6f}")

    return best_a, best_score


def subgradient_refine(a0, time_limit, n_max_iters=50000):
    """Direct subgradient method on the exact (unsmoothed) max objective."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Maintain an average of subgradients for convergence
    avg_grad = np.zeros(n)
    counter = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_val = b.max()
        # Find all indices achieving (near) the max
        threshold = max_val - 1e-10 * max_val
        peak_indices = np.where(b >= threshold)[0]

        # Compute subgradient: average of gradients at peak indices
        # grad_k[i] = 2 * a_{k-i} for valid indices
        subgrad = np.zeros(n)
        for k in peak_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                subgrad[j] += 2.0 * a[k - j]
        subgrad /= len(peak_indices)

        # Update
        counter += 1
        avg_grad = (counter * avg_grad + subgrad) / (counter + 1)

        # Step size: diminishing
        lr = 0.1 / np.sqrt(1.0 + it)

        # Try both raw subgradient and averaged
        a_trial1 = project_simplex(a - lr * subgrad)
        a_trial2 = project_simplex(a - lr * avg_grad)

        score1 = score_normalized(a_trial1)
        score2 = score_normalized(a_trial2)

        if score1 < score2:
            a = a_trial1
        else:
            a = a_trial2

        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Periodic Nesterov-like extrapolation
        if it % 10 == 0 and it > 0:
            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * it * it) / 2.0
            t_prev = 1.0 + np.sqrt(1.0 + 4.0 * (it - 1) * (it - 1)) / 2.0
            beta = (t_prev - 1.0) / t_curr
            a_extrap = project_simplex(a + beta * (a - best_a + lr * subgrad))
            sc_extrap = score_normalized(a_extrap)
            if sc_extrap < best_score:
                a = a_extrap
                best_score = sc_extrap
                best_a = a.copy()

        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SG iter={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def local_coordinate_search(a0, time_limit, n_max_iters=100000):

    n = len(a0)
    a = project_simplex(a0.copy())

        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]

        # Identify top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_n = min(8, len(idx))
        top_idx = np.argsort(contribs)[::-1][:top_n]

        improved = False
        # Try reducing top contributors
        for ci in top_idx:

                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:

            for _ in range(3):

                if i1 == i2:

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_diverse_init(n, rng, variant=0):
    """Create diverse initial sequences."""
    x = np.linspace(0, 1, n)

        a_alt, _ = nesterov_projected_gradient(init_alt, t_total * 0.5,
                                               T_start=3.0, T_end=300.0,
                                               lr_init=0.06, n_stages=6, steps_per_stage=2000)
        try_update(a_alt, f"alt_n{alt_n}_ng")
        rem = deadline - time.monotonic()
        if rem > 15:
            a_alt2, _ = nesterov_lp_refine(a_alt, min(rem * 0.4, 30.0), n_max_iters=5000)
            try_update(a_alt2, f"alt_n{alt_n}_lp")

    # === Phase 7: Final LP polish on best ===
    print(f"\n=== Phase 7: Final LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.30, 60.0)
        a_fin, _ = nesterov_lp_refine(best_seq, t_fin, n_max_iters=50000)
        try_update(a_fin, "final_lp")

    # === Phase 8: Final local search ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t_ps = min(rem * 0.30, 30.0)
        a_ps, _ = local_coordinate_search(best_seq, t_ps, n_max_iters=100000)
        try_update(a_ps, "final_local")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
