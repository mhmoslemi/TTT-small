import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_from_conv(conv, n):
    """Compute score from precomputed convolution and length."""
    M = float(conv.max())
    return 2.0 * n * M


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lp_best_response(a, max_time=5.0):
    """
    Solve LP: min t s.t. conv(a,b)[k] <= t for all k, sum(b)=1, b>=0.
    Returns optimal b.
    """
    n = len(a)
    # Build constraint matrix: T[k,j] = a[k-j] if 0<=k-j<n else 0
    # conv(a,b)[k] = sum_j T[k,j] * b[j]
    # We need: T @ b - t <= 0  (2n-1 constraints)
    # sum(b) = 1 (equality)
    # b >= 0, t free (but t >= 0 implicitly from constraints)
    
    # Build T efficiently using the structure
    T_rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n + 1)
        # T[k,j] = a[k-j] if 0 <= k-j < n
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # row[j] = a[k-j] for j in [j_start, j_end]
            # k-j ranges from k-j_start down to k-j_end
            row[j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
        row[n] = -1.0  # coefficient of t
        T_rows.append(row)
    
    A_ub = np.array(T_rows)  # shape (2n-1, n+1)
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize t (last variable)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Bounds: b_i >= 0, t >= -inf
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    try:
        t0 = time.monotonic()
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if time.monotonic() - t0 > max_time:
            return None
        if res.status == 0:
            b_opt = res.x[:n]
            b_opt = np.maximum(b_opt, 0)
            s = b_opt.sum()
            if s > 1e-15:
                b_opt = b_opt / s
            return b_opt
    except Exception:
        pass
    return None


def fw_step_multi_peak(a, top_k=10, temperature=0.01):
    """
    One FW step with multi-peak gradient.
    Returns the vertex j* to move toward and the step direction info.
    """
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    
    # Get top-k peaks
    k_act = min(top_k, 2 * n - 1)
    if k_act < (2 * n - 1):
        peaks = np.argpartition(conv, -k_act)[-k_act:]
    else:
        peaks = np.arange(2 * n - 1)
    
    pv = conv[peaks]
    temp = temperature * max(M, 1e-10)
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    w = w / ws
    
    # Compute weighted gradient: grad[i] = sum_k w[k] * 2 * contribution of a[i] to conv[k]
    # conv[k] = sum_j a[j]*a[k-j], so d(conv[k])/d(a[i]) = a[k-i] + a[i-k] (with valid range)
    # For the linearized FW: we use the gradient of the active constraint
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # d conv[k] / d a[i] = a[k-i] (for i in valid range, counting both terms)
        # Actually: conv[k] = sum_j a[j] a[k-j]
        # d/d a[i]: if i <= k-i <= n-1, term a[i]*a[k-i] contributes a[k-i]
        #           if k-i <= i <= n-1 and k-i >= 0, term a[k-i]*a[i] contributes a[k-i]
        # So d conv[k]/d a[i] = a[k-i] + a[i-(k-i)]... 
        # Simpler: d conv[k]/d a[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n, i.e., i>=k)
        # But since k < 2n-1 and i < n:
        #   k-i in [0,n) iff i in [k-n+1, k]
        #   i-k in [0,n) iff i in [k, k+n-1], but i < n, so i in [k, n-1] (only if k < n)
        # Wait, let me just use: d conv[k]/d a[i] = a[k-i] if 0<=k-i<n, else 0
        # (this accounts for both terms since a[j]*a[k-j] with j=i gives a[k-i],
        #  and a[j]*a[k-j] with k-j=i gives j=k-i, contributing a[k-i])
        # Actually for i != k-i, we get TWO contributions: d/d a[i] of a[i]*a[k-i] + a[k-i]*a[i]
        # = a[k-i] + a[k-i] = 2*a[k-i]
        # For i == k-i (k=2i), we get d/d a[i] of a[i]^2 = 2*a[i]
        # So: d conv[k]/d a[i] = 2*a[k-i] if 0<=k-i<n and k!=2i, or 2*a[i] if k=2i
        # Which simplifies to: 2*a[k-i] if 0 <= k-i < n (covers both cases)

        seg = a[k - i_end:k - i_start + 1][::-1]
        grad[i_start:i_end + 1] += w[idx] * 2.0 * seg
    
    # FW: move toward e_{j*} where j* = argmin grad (since we want to decrease)
    j_star = int(np.argmin(grad))
    
    return j_star, grad, conv, M


def fw_optimize(a_init, time_limit, top_k=15, temperature=0.005, verbose=True, label=""):
    """
    Frank-Wolfe with multi-peak gradient and away-steps.
    """
    n = len(a_init)
    a = normalize(a_init.astype(np.float64).copy())
    conv = fast_convolve(a)
    best_a = a.copy()
    best_M = float(conv.max())
    best_score = 2.0 * n * best_M
    t0 = time.monotonic()
    step_count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        step_count += 1
        
        # Recompute conv periodically (or every step for accuracy)
        if step_count % 10 == 0:
            conv = fast_convolve(a)
        
        M = float(conv.max())
        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            # Restart with small perturbation
            noise = np.random.standard_normal(n) * 0.01
            a_new = np.maximum(a + noise, 0)
            a = normalize(a_new)
            conv = fast_convolve(a)
            no_improve = 0
            continue
        
        # Multi-peak FW step
        j_star, grad, conv, M = fw_step_multi_peak(a, top_k=top_k, temperature=temperature)
        
        # Away step
        step = 2.0 / (step_count + 2.0)
        
        # Check for away step
        if step_count > 50:
            away_scores = a * (grad + M * 0.1)  # heuristic away score
            i_away = int(np.argmax(away_scores))
            if a[i_away] > 1e-8 and grad[i_away] > 0:
                t_away = min(step * 0.5, a[i_away] * 0.9)
                # Away step: reduce a[i_away]
                a = a - t_away * np.eye(n)[i_away]
                a = np.maximum(a, 0)
                # Then FW step toward j_star
                a = a + step * (np.eye(n)[j_star])
                a = normalize(a)
                continue
        
        # Standard FW step
        e_j = np.zeros(n)
        e_j[j_star] = 1.0
        a = (1 - step) * a + step * e_j
        a = normalize(a)
    
    return best_a, best_score


def alternating_lp(a_init, time_limit, alpha_schedule='adams', verbose=True, label=""):
    """
    Alternating best-response using LP.
    At each step: solve LP to find best b given a, then update a.
    """
    n = len(a_init)
    a = normalize(a_init.astype(np.float64).copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP: find b that minimizes max_k conv(a,b)[k] s.t. sum(b)=1, b>=0
        b_opt = lp_best_response(a, max_time=10.0)
        if b_opt is None:
            break
        
        # Evaluate b_opt directly
        b_score = 2.0 * n * fast_convolve(b_opt).max()
        if b_score < best_score:
            best_score = b_score
            best_a = b_opt.copy()
        
        # Update a
        if alpha_schedule == 'adams':
            alpha = 2.0 / (it + 2.0)
        else:
            alpha = 0.3
        
        a_new = (1 - alpha) * a + alpha * b_opt
        a_new = normalize(a_new)
        
        a_new_score = 2.0 * n * fast_convolve(a_new).max()
        if a_new_score < best_score:
            best_score = a_new_score
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ALP {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    conv_len = 2 * n - 1
    prefix_max = np.zeros(conv_len)
    prefix_max[0] = conv[0]
    for k in range(1, conv_len):
        prefix_max[k] = max(prefix_max[k-1], conv[k])
    suffix_max = np.zeros(conv_len)
    suffix_max[conv_len - 1] = conv[conv_len - 1]
    for k in range(conv_len - 2, -1, -1):
        suffix_max[k] = max(suffix_max[k+1], conv[k])
    
    # Fine step factors
    fine_factors = [0.995, 0.99, 0.985, 0.98, 1.005, 1.01, 1.015, 1.02]
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-12:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            if k_start > 0:
                unaff_max = prefix_max[k_start - 1]
            else:
                unaff_max = 0.0
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, suffix_max[k_end + 1])
            
            affected = conv[k_start:k_end + 1].copy()
            a_shifted = a[k_start - i:k_end - i + 1]
            has_center = (2 * i >= k_start and 2 * i <= k_end)
            center_idx = 2 * i - k_start if has_center else -1
            
            best_delta = 0.0
            best_new_M = M
            
            for factor in fine_factors:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-14:
                    continue
                
                new_aff = affected + 2.0 * delta * a_shifted
                if has_center:
                    new_aff[center_idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                
                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_M < M - 1e-13:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if has_center:
                    old_ai = a[i] - best_delta
                    changes[center_idx] = 2.0 * best_delta * old_ai + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                improved = True
                
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def spectral_init(n, rng):
    """
    Initialize using spectral methods: create a sequence whose power spectrum
    is close to flat.
    """
    # Start with random complex spectrum with flat magnitude
    fft_size = 1 << (n - 1).bit_length()
    # Create a "target" flat spectrum
    target = np.ones(fft_size // 2 + 1)
    # Add some randomness
    target *= (1.0 + 0.1 * rng.standard_normal(fft_size // 2 + 1))
    target = np.maximum(target, 0.1)
    
    # Inverse FFT to get time domain
    # We need the IFFT of |X|^2 to give us a "real" signal
    # Actually, let's just create a random non-negative sequence and refine
    # Better: use the iterative projection approach
    
    # Project random vector to non-negative with unit sum
    a = np.abs(rng.standard_normal(n))
    a = normalize(a)
    
    # Do a few FW steps to flatten the autocorrelation
    for _ in range(100):
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        # Gradient at k_star
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        grad = np.zeros(n)
        grad[i_start:i_end + 1] = a[k_star - i_end:k_star - i_start + 1][::-1]
        j_star = int(np.argmin(grad))
        t = 2.0 / (101)
        e = np.zeros(n)
        e[j_star] = 1
        a = (1 - t) * a + t * e
        a = normalize(a)
    
    return a


def cosine_window(n, power=1.5):
    """Cosine window initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    a = (0.5 * (1 + np.cos(np.pi * x))) ** power
    a = np.maximum(a, 0)
    return normalize(a)


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
