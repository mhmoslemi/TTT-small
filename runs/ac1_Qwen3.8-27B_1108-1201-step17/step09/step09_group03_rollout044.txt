"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated multi-peak subgradient mirror descent with adaptive step sizes
2. Fast coordinate descent with O(1) range-max queries and smart line search
3. LP-based best-response alternating minimization
4. Structured initialization exploiting antisymmetric components
5. Perturbation-based restarts for escaping local minima

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_autoconv(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_ab(a, b):
    """Full convolution of a and b via FFT."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    return irfft(rfft(a, fft_size) * rfft(b, fft_size), fft_size)[:size]


def compute_score(a):
    """Compute objective: 2*n*max(conv(a,a)) for normalized a."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    an = a / s
    return 2.0 * n * fast_autoconv(an).max()


def build_prefix_suffix_max(conv):
    """Build prefix and suffix max arrays for O(1) range max outside [lo, hi]."""
    m = len(conv)
    prefix = np.empty(m, dtype=np.float64)
    suffix = np.empty(m, dtype=np.float64)
    prefix[0] = conv[0]
    for i in range(1, m):
        prefix[i] = prefix[i-1] if prefix[i-1] > conv[i] else conv[i]
    suffix[m-1] = conv[m-1]
    for i in range(m-2, -1, -1):
        suffix[i] = suffix[i+1] if suffix[i+1] > conv[i] else conv[i]
    return prefix, suffix


def unaffected_max(prefix, suffix, lo, hi):
    """Max of conv outside [lo, hi] inclusive."""
    m = len(prefix)
    result = -np.inf
    if lo > 0:
        result = prefix[lo - 1]
    if hi < m - 1:
        right = suffix[hi + 1]
        if right > result:
            result = right
    return result


def mirror_descent(a0, time_limit, lr_init=0.1, top_k=40, verbose=True):
    """
    Multi-peak subgradient mirror descent (multiplicative weight update).
    Works on the simplex (sum=1, a>=0).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_autoconv(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    # Running average for convergence guarantee
    avg_a = a.copy().astype(np.float64)

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = float(conv.max())
        cur_score = 2.0 * n * M

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            lr *= 0.7
            no_improve = 0
            if lr < 1e-10:
                break

        # Multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        if k_act < 2 * n - 1:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
        else:
            peaks = np.arange(2 * n - 1)

        pv = conv[peaks]
        temp = 0.003 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Compute subgradient vectorized in chunks
        sg = np.zeros(n, dtype=np.float64)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # sg[i] += w[idx] * a[k-i] for i in [i_start, i_end]
            # a[k-i] for i in [i_start, i_end] = a[k-i_end : k-i_start+1][::-1]
            seg = a[k - i_end:k - i_start + 1][::-1]
            sg[i_start:i_end + 1] += w[idx] * seg
            # Add the quadratic correction: if k = 2*i, add extra a[i]
            mid = k // 2
            if 2 * mid == k and 0 <= mid < n:
                sg[mid] += w[idx] * a[mid]

        # Mirror descent step
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 1e-30)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn

        # Running average
        avg_a = (it * avg_a + a_new) / (it + 1)

        a = a_new
        conv = fast_autoconv(a)

        if it % 200 == 0:
            avg_score = 2.0 * n * fast_autoconv(avg_a).max()
            if avg_score < best_score:
                best_score = avg_score
                best_a = avg_a.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    MD it={it} ({elapsed:.0f}s): best={best_score:.7f} lr={lr:.6f} M={M:.8f}")

    return best_a, best_score


def coord_descent(a0, time_limit, verbose=True, label=""):
    """
    Fast coordinate descent with smart line search.
    Works on raw (unnormalized) non-negative vectors.
    Minimizes max(conv) directly, then normalizes at the end.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    if a.sum() < 1e-15:
        a = np.ones(n) / n

    conv = fast_autoconv(a)
    M = float(conv.max())
    prefix, suffix = build_prefix_suffix_max(conv)
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            umax = unaffected_max(prefix, suffix, k_start, k_end)

            if M <= umax + 1e-15:
                # Peak is in unaffected region; can't improve by decreasing a[i]
                # Try increasing a[i] slightly? Usually not helpful.
                continue

            # Peak is in affected region. Try to decrease a[i].
            # affected[j] = conv[i+j] for j in [0, n-1] (within bounds)
            a_shift = a[max(0, k_start - i):min(n, k_end - i + 1)]
            affected = conv[k_start:k_end + 1].copy()

            # Current max in affected region
            aff_max = float(affected.max())
            aff_jmax = int(affected.argmax())  # index into affected array

            # Optimal delta: reduce affected_max to umax
            # affected[aff_jmax] + 2*delta*a_shift[aff_jmax] = umax
            if a_shift[aff_jmax] > 1e-15:
                delta_opt = (umax - affected[aff_jmax]) / (2.0 * a_shift[aff_jmax])
            else:
                delta_opt = -a[i]

            # Clamp: delta must be >= -a[i] (non-negativity)
            if delta_opt < -a[i]:
                delta_opt = -a[i]
            if delta_opt > 0:
                delta_opt = 0.0

            # Try delta_opt and a few alternatives
            candidates = [delta_opt, delta_opt * 0.5, delta_opt * 0.9, -a[i] * 0.5, -a[i] * 0.99]
            # Remove duplicates and invalid
            best_delta = 0.0
            best_new_M = M

            for delta in candidates:
                if delta >= -1e-15:
                    continue
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                # Compute new affected max
                new_affected = affected + 2.0 * delta * a_shift
                # Quadratic correction at position where k=2i
                pos_2i = 2 * i - k_start
                if 0 <= pos_2i < len(new_affected):
                    new_affected = new_affected.copy()
                    new_affected[pos_2i] += delta * delta
                new_M = max(umax, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta < -1e-15 and best_new_M < M - 1e-14:
                # Apply update
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shift
                pos_2i = 2 * i - k_start
                if 0 <= pos_2i < len(changes):
                    changes = changes.copy()
                    changes[pos_2i] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(umax, float(conv[k_start:k_end + 1].max()))

                # Update prefix/suffix (expensive but necessary for correctness)
                # Only rebuild if M changed significantly
                prefix, suffix = build_prefix_suffix_max(conv)

                if M < best_M:
                    best_M = M
                    best_a = a.copy()
                improved = True

        if not improved:
            break

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            cur_score = 2.0 * n * best_M
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best_score={cur_score:.7f} M={best_M:.8f}")

    # Normalize
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
    else:
        best_a = best_a / s
    best_score = 2.0 * n * fast_autoconv(best_a).max()
    return best_a, best_score

def lp_alternate(a0, time_limit, max_iters=20, verbose=True):
    """
    LP-based best response alternating minimization.
    Given a, find b = argmin max_k (a*b)[k] s.t. b>=0, sum(b)=1.
    Then update a toward b.
    """
    try:
        import cvxpy as cp
    except ImportError:
        if verbose:
            print("    cvxpy not available, skipping LP")
        return a0, compute_score(a0)

    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autoconv(a).max()
    t0 = time.monotonic()

    # Build T matrix: T @ b = conv(a, b), shape (2n-1, n)
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        T[j:j + n, j] = a

    it = 0
    alpha = 1.0

    while it < max_iters and time.monotonic() - t0 < time_limit:
        it += 1
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [
            T @ b_var <= M_var,
            cp.sum(b_var) == 1
        ]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        solved = False
        for solver in [cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver, max_iter=2000)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            break

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        sb = b_val.sum()
        if sb < 1e-15:
            break
        b_val = b_val / sb

        # Evaluate b
        score_b = 2.0 * n * fast_autoconv(b_val).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_val.copy()

        # Damped update
        a_new = (1 - alpha) * a + alpha * b_val
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            break
        a_new = a_new / sa

        score_new = 2.0 * n * fast_autoconv(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new
        alpha *= 0.85

        # Rebuild T
        T = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            T[j:j + n, j] = a

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score


def antisymmetric_init(n, rng):
    """
    Initialize with a sequence that has significant antisymmetric component.
    a[i] = base + amplitude * sin(pi * (i/n - 0.5))
    This exploits the insight that antisymmetric components reduce the center peak.
    """
    base = rng.uniform(0.3, 0.7)
    amplitude = rng.uniform(0.1, 0.4)
    t = np.arange(n, dtype=np.float64) / n
    a = base + amplitude * np.sin(np.pi * (t - 0.5))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def two_peak_asymmetric_init(n, rng):
    """Initialize with two asymmetric peaks."""
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.15, 0.35)
    c2 = n * rng.uniform(0.65, 0.85)
    w1 = n * rng.uniform(0.08, 0.2)
    w2 = n * rng.uniform(0.08, 0.2)
    h1 = rng.uniform(0.5, 1.0)
    h2 = rng.uniform(0.5, 1.0)
    a = h1 * np.exp(-0.5 * ((t - c1) / w1) ** 2) + h2 * np.exp(-0.5 * ((t - c2) / w2) ** 2)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def cosine_power_init(n, rng):
    """Cosine raised to a power."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    power = rng.uniform(0.3, 2.0)
    a = (0.5 * (1 + np.cos(np.pi * x))) ** power
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def noisy_perturb(a, rng, scale=0.1):
    """Add multiplicative noise and re-normalize."""
    noise = rng.lognormal(0, scale, size=len(a))
    a_new = a * noise
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def fine_cd(a0, time_limit, n_steps_per_coord=4, verbose=True, label="fine"):
    """
    Very fine coordinate descent for last-mile improvement.
    Uses small adaptive steps.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_autoconv(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        M = float(conv.max())
        prefix, suffix = build_prefix_suffix_max(conv)

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            umax = unaffected_max(prefix, suffix, k_start, k_end)

            if M <= umax + 1e-15:
                continue

            a_shift = a[max(0, k_start - i):min(n, k_end - i + 1)]
            affected = conv[k_start:k_end + 1]

            # Try small multiplicative changes
            for factor in [0.999, 0.998, 0.997, 0.995, 0.99]:
                delta = a[i] * (factor - 1.0)
                if abs(delta) < 1e-18:
                    continue
                new_affected = affected.copy() + 2.0 * delta * a_shift
                pos_2i = 2 * i - k_start
                if 0 <= pos_2i < len(new_affected):
                    new_affected[pos_2i] += delta * delta
                new_M = max(umax, float(new_affected.max()))
                if new_M < best_score_fast(n, new_M) - 1e-15:
                    # Apply
                    a[i] = max(a[i] + delta, 0.0)
                    changes = 2.0 * delta * a_shift
                    if 0 <= pos_2i < len(changes):
                        changes = changes.copy()
                        changes[pos_2i] += delta * delta
                    conv[k_start:k_end + 1] += changes
                    M = new_M
                    improved = True

                    # Update prefix/suffix periodically
                    if i % 16 == 0:
                        prefix, suffix = build_prefix_suffix_max(conv)
                    break  # Only one change per coordinate per sweep

        # Check improvement
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()

        if not improved and sweeps > 5:
            break

        if verbose and sweeps % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score


def best_score_fast(n, M):
    return 2.0 * n * M


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization (exploit antisymmetric structure)
    Phase 2: Subgradient mirror descent (global optimization)
    Phase 3: Coordinate descent (local optimization)
    Phase 4: LP-based alternation (structured refinement)
    Phase 5: Fine coordinate descent (polishing)
    Phase 6: Perturbation + re-optimization (escape local minima)
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(12345)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    n = 1024
    if parent is not None and len(parent) == n:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    elif parent is not None:
        # Resample to length n
        from numpy import interp
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        best_seq = np.interp(x_new, x_old, parent)
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"No parent, starting uniform: n={n}, score={best_score:.7f}")

    n = len(best_seq)
    print(f"Starting: n={n}, score={best_score:.7f}")

    def try_update(seq, label):
        nonlocal best_seq, best_score
        if seq is None:
            return
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s < 0.01:
            return
        seq = seq / s
        try:
            sc = float(evaluate_sequence(seq.tolist()))
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.7f}")

    # === Phase 1: Multi-start with diverse initializations ===
    print(f"\n=== Phase 1: Multi-start (budget check, best={best_score:.7f}) ===")
    remaining = deadline - time.monotonic() - 1200
    if remaining > 30:
        inits = [
            best_seq.copy(),
            antisymmetric_init(n, rng),
            two_peak_asymmetric_init(n, rng),
            cosine_power_init(n, rng),
            noisy_perturb(best_seq, rng, 0.15),
            noisy_perturb(best_seq, rng, 0.3),
        ]
        # Quick evaluation of all inits
        for idx, init in enumerate(inits):
            sc = compute_score(init)
            if sc < best_score:
                try_update(init, f"init_{idx}")
            if idx == 0:
                print(f"  init_0 score: {sc:.7f}")
            elif idx % 2 == 0:
                print(f"  init_{idx} score: {sc:.7f}")

    # === Phase 2: Subgradient Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.7f}) ===")
    remaining = deadline - time.monotonic() - 900
    if remaining > 60:
        t_md = min(remaining * 0.6, 500.0)
        print(f"  MD budget={t_md:.0f}s")
        a_md, s_md = mirror_descent(best_seq, t_md, lr_init=0.08, top_k=50, verbose=True)
        try_update(a_md, "mirror_descent")

    # === Phase 3: Coordinate Descent ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.7f}) ===")
    remaining = deadline - time.monotonic() - 600
    if remaining > 60:
        t_cd = min(remaining * 0.7, 400.0)
        print(f"  CD budget={t_cd:.0f}s")
        a_cd, s_cd = coord_descent(best_seq, t_cd, verbose=True, label="main")
        try_update(a_cd, "coord_descent")

        # Run CD on MD result too
        remaining = deadline - time.monotonic() - 400
        if remaining > 60:
            t_cd2 = min(remaining * 0.5, 200.0)
            a_cd2, s_cd2 = coord_descent(a_md, t_cd2, verbose=True, label="md_cd")
            try_update(a_cd2, "cd_md")

    # === Phase 4: LP Alternation ===
    print(f"\n=== Phase 4: LP Alternation (best={best_score:.7f}) ===")
    remaining = deadline - time.monotonic() - 300
    if remaining > 60:
        try:
            t_lp = min(remaining * 0.5, 150.0)
            a_lp, s_lp = lp_alternate(best_seq, t_lp, max_iters=15, verbose=True)
            try_update(a_lp, "lp_alt")
        except Exception as e:
            print(f"  LP failed: {e}")

    # === Phase 5: Fine Coordinate Descent ===
    print(f"\n=== Phase 5: Fine CD (best={best_score:.7f}) ===")
    remaining = deadline - time.monotonic() - 100
    if remaining > 30:
        t_fine = min(remaining * 0.6, 120.0)
        a_fine, s_fine = fine_cd(best_seq, t_fine, verbose=True, label="polish")
        try_update(a_fine, "fine_cd")

    # === Phase 6: Perturbation and Re-optimization ===
    print(f"\n=== Phase 6: Perturbation + Re-opt (best={best_score:.7f}) ===")
    remaining = deadline - time.monotonic() - 40
    if remaining > 60:
        # Try a few perturbations
        for pert_idx in range(3):
            remaining = deadline - time.monotonic() - 20
            if remaining < 30:
                break
            pert = noisy_perturb(best_seq, rng, scale=0.05 * (pert_idx + 1))
            # Quick MD
            t_pert = min(remaining * 0.3, 30.0)
            a_p, s_p = mirror_descent(pert, t_pert, lr_init=0.05, top_k=40, verbose=False)
            try_update(a_p, f"pert_md_{pert_idx}")
            # Quick CD
            remaining = deadline - time.monotonic() - 10
            if remaining > 20:
                t_pcd = min(remaining * 0.5, 30.0)
                a_pc, s_pc = coord_descent(a_p, t_pcd, verbose=False, label="pert_cd")
                try_update(a_pc, f"pert_cd_{pert_idx}")

    # === Final: One more CD pass on the best ===
    print(f"\n=== Final Polish (best={best_score:.7f}) ===")
    remaining = deadline - time.monotonic() - 5
    if remaining > 10:
        a_fin, s_fin = fine_cd(best_seq, remaining * 0.7, verbose=False, label="final")
        try_update(a_fin, "final_polish")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
