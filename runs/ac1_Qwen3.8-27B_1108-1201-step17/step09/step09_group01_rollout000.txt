import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build convolution matrix T such that T @ b = conv(a_fixed, b)
    # T[k, i] = a_fixed[k-i] if 0 <= k-i < n, else 0
    T = np.zeros((2 * n - 1, n))
    for i in range(n):
        T[i:i+n, i] = a_fixed
    
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.GLPK, cp.PDLP, cp.GLOP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def fast_coord_descent(a0, time_limit, verbose=False, label=""):

    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = float(conv.max())
        
        # Compute softmax weights over the convolution
        scaled = beta * (conv - M)
        scaled = np.clip(scaled, -50, 0)
        w = np.exp(scaled)
        w /= w.sum()
        
        # Gradient: d/d a_i [sum_k w_k * conv[k]]
        # conv[k] = sum_i a[i] * a[k-i]
        # d conv[k] / d a_i = a[k-i] + a[i-k] (with valid index checks)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                grad[i] += wk * a[j]
                if j != i:
                    grad[j] += wk * a[i]
        
        # Mirror descent update (multiplicative)
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        a_new = np.exp(log_a)
        a_new /= a_new.sum()
        
        score = 2.0 * n * float(fast_convolve(a_new).max())
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Don't update a if it didn't improve
        
        if no_improve > 50:
            lr *= 0.9
            beta *= 1.3
            no_improve = 0
            if lr < 1e-10:
                break
        elif it % 20 == 0:
            beta *= 1.05

    return best_a, best_score

def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """Fictitious play with Nesterov averaging."""
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    a_avg = a.copy()
    b_avg = a.copy()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is None:
            b_new = a.copy()
        a_new, _ = solve_lp_best_response(b_new, solver)
        if a_new is None:
            a_new = b_new.copy()

        # Nesterov averaging
        alpha = 2.0 / (iters + 2.0)
        a_avg = (1 - alpha) * a_avg + alpha * a_new
        b_avg = (1 - alpha) * b_avg + alpha * b_new

        # Update a and b
        a = a_new.copy()
        b_cur = b_new.copy()

        # Evaluate all candidates
        for cand in [a_new, b_new, a_avg, b_avg, 0.5*(a_avg + b_avg)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand = cand / sc
            score = 2.0 * n * float(fast_convolve(cand).max())
            if score < best_score:
                best_score = score
                best_a = cand.copy()

        if verbose and iters % 5 == 0:
            print(f"    FP n={n} it={iters}: best={best_score:.6f}")

    return best_a, best_score

def gradient_guided_perturb(a0, rng, strength=0.05):
    """Perturb in the direction that reduces the peak."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = float(conv.max())
    
    # Compute subgradient of max
    peak_positions = np.where(conv >= M * 0.99)[0]
    
    # Pressure on each coordinate
    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a[j]
            if j != i:
                pressure[j] += 0.5 * a[i]
    
    p_max = pressure.max()
    if p_max > 1e-20:
        pressure /= p_max
    
    # Perturb: reduce high-pressure, add noise
    noise = rng.standard_normal(n) * 0.3
    a_new = a * (1.0 - strength * pressure + strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def random_restructure(a0, rng, frac=0.2):
    """Randomly zero out fractions and redistribute."""
    n = len(a0)
    a = a0.copy()
    keep = rng.random(n) > frac
    if keep.sum() < n // 5:
        keep = np.ones(n, dtype=bool)
    a[~keep] = 0
    a[keep] *= (1.0 + rng.standard_normal(int(keep.sum())) * 0.15)
    a = np.maximum(a, 0)
    
    zero_mask = (a < 1e-15)
    if zero_mask.any():
        n_add = max(1, int(zero_mask.sum() * 0.1))
        add_idx = rng.choice(np.where(zero_mask)[0], size=min(n_add, int(zero_mask.sum())), replace=False)
        a[add_idx] = rng.random(n_add) * a.max() * 0.3
    
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def init_spectral(n, rng):
    """Initialize using spectral method: try to make power spectrum flat."""
    # Start with a random phase in frequency domain
    # A[0] = 1 (DC)
    # For other frequencies, try to make |A| roughly constant
    A = np.zeros(n, dtype=complex)
    A[0] = 1.0
    # Target |A[k]| for k != 0
    # If we want flat autocorrelation, we want |A[k]|^2 constant
    # But A[0] = 1 is fixed, so |A[0]|^2 = 1
    # For the autocorrelation at lag 0: sum(a_i^2) = (1/n)*sum|A|^2 = (1 + (n-1)*c^2)/n
    # For lag m != 0: = (c^2 * (n-1) - 1)/(n) ... this doesn't work out to be constant
    # So let's just try a few different target spectra
    for attempt in range(5):
        # Random target spectrum with |A[k]| roughly constant
        targets = np.ones(n) * rng.uniform(0.3, 0.8)
        phases = rng.uniform(0, 2*np.pi, n)
        A = targets * np.exp(1j * phases)
        A[0] = 1.0  # Fix DC
        # Enforce real symmetry: A[k] = conj(A[n-k])
        for k in range(1, n//2 + 1):
            avg_real = (A[k].real + A[n-k].real) / 2
            avg_imag = (A[k].imag - A[n-k].imag) / 2
            A[k] = avg_real + 1j * avg_imag
            A[n-k] = avg_real - 1j * avg_imag
    
    a = irfft(A, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def init_bimodal(n, rng):
    """Bimodal initialization."""
    c1 = n * rng.uniform(0.15, 0.35)
    c2 = n * rng.uniform(0.65, 0.85)
    w1 = n * rng.uniform(0.05, 0.2)
    w2 = n * rng.uniform(0.05, 0.2)
    t = np.arange(n)
    a = np.exp(-0.5*((t-c1)/w1)**2) + np.exp(-0.5*((t-c2)/w2)**2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def init_cosine(n, rng):
    """Cosine-modulated initialization."""
    t = np.arange(n)
    freq = rng.uniform(1, 4)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t / n + rng.uniform(0, 2*np.pi))
    a = np.maximum(a, 0) + 0.01  # Ensure non-negative
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
