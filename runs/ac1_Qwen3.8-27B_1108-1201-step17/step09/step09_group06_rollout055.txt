        grad = np.zeros(n)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast multiplicative-weight best response (replacing LP) for speed
2. Heavy-ball alternating with adaptive momentum
3. Peak-directed local search
4. Multi-length exploration (interpolation to larger n)

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_with(a_fixed, b):
    n = len(a_fixed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a_fixed, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def best_response_mw(a, eta=0.5, n_iter=200, threshold=0.85):
    """
    Find b >= 0, sum(b)=1 that approximately minimizes max_k (a*b)(k)
    using multiplicative weight updates.
    """
    n = len(a)
    b = np.ones(n) / n
    best_b = b.copy()
    best_max = convolve_with(a, b).max()
    
    for it in range(n_iter):
        conv = convolve_with(a, b)
        max_conv = conv.max()
        
        if max_conv < best_max:
            best_max = max_conv
            best_b = b.copy()
        
        # Active set: constraints near the peak
        active_mask = conv >= threshold * max_conv
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(conv))])
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        
        # Multiplicative weight update
        log_b = np.log(np.maximum(b, 1e-30))
        log_b -= eta * grad
        log_b -= log_b.max()  # for numerical stability
        b_new = np.exp(log_b)
        b_new = np.maximum(b_new, 1e-30)
        b_new = b_new / b_new.sum()
        
        # Check convergence
        if np.max(np.abs(b_new - b)) < 1e-12:
            b = b_new
            break
        b = b_new
    
    # Final check
    conv = convolve_with(a, b)
    if conv.max() < best_max:
        best_max = conv.max()
        best_b = b.copy()
    
    return best_b, best_max


def heavy_ball_alternate(a0, time_limit, n_mw_iters=100, eta_mw=0.5):
    """
    Alternating minimization with heavy-ball momentum.
    Alternates between optimizing a given b and optimizing b given a.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    b = a.copy()  # Initialize b = a
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    v_a = np.zeros(n)  # Momentum for a
    v_b = np.zeros(n)  # Momentum for b
    mu = 0.5  # Initial momentum
    
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Step 1: Best response for b given a
        b_new, M_b = best_response_mw(a, eta=eta_mw, n_iter=n_mw_iters)
        
        # Heavy ball update for b
        v_b = mu * v_b + (b_new - b)
        b = b + v_b
        b = np.maximum(b, 0)
        sb = b.sum()
        if sb < 1e-15:
            b = b_new
        else:
            b = b / sb
        
        # Step 2: Best response for a given b (by symmetry)
        a_new, M_a = best_response_mw(b, eta=eta_mw, n_iter=n_mw_iters)
        
        # Heavy ball update for a
        v_a = mu * v_a + (a_new - a)
        a = a + v_a
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = a_new
        else:
            a = a / sa
        
        # Evaluate
        score_a = 2.0 * n * fast_convolve(a).max()
        score_b = 2.0 * n * fast_convolve(b).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_b < best_score:
            best_score = score_b
            best_a = b.copy()
            stall = 0
        
        if it % 500 == 0:
            if best_score < prev_best - 1e-8:
                stall = 0
                prev_best = best_score
            else:
                stall += 1
                if stall >= 10:
                    # Increase momentum or restart
                    mu = min(mu * 1.1, 0.9)
                    if mu > 0.85:
                        # Reset with perturbation
                        rng = np.random.default_rng(it)
                        pert = rng.random(n) * 0.05
                        a = np.maximum(a + pert, 0)
                        a = a / a.sum()
                        b = np.maximum(b + rng.random(n) * 0.05, 0)
                        b = b / b.sum()
                        v_a = np.zeros(n)
                        v_b = np.zeros(n)
                        mu = 0.5
                        stall = 0
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    HB-Alt iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} mu={mu:.3f} stall={stall}")
    
    return best_a, best_score


def peak_reduction_search(a0, time_limit, n_steps=50000):
    """
    Targeted local search that specifically attacks the peaks of a*a.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_val = b.max()
        peak_k = int(np.argmax(b))
        
        # Find all significant peaks
        threshold = 0.92 * peak_val
        peaks = np.where(b >= threshold)[0]
        
        # For each peak, find which a_j values contribute most
        # and try to redistribute them
        improved = False
        
        for k in peaks[:5]:
            # Contribution to peak k: a_j * a[k-j] for valid j
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            
            if total_contrib < 1e-15:
                continue
            
            # Find the largest contributor and try to move it
            best_src = idx[np.argmax(contrib)]
            
            # Try moving a small amount from src to random destinations
            for _ in range(20):
                dst = rng.integers(n)
                if dst == best_src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amount = a[best_src] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[best_src] -= amount
                    a_trial[dst] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-10:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation
            no_improve += 1
            if no_improve > 200:
                # Add structured noise
                noise_scale = 0.02 / (1 + no_improve * 0.01)
                noise = rng.standard_t(df=3, size=n) * noise_scale
                a = np.maximum(a + noise, 0)
                a = a / a.sum()
                no_improve = 0
            elif step > 5000 and no_improve > 5000:
                break
        else:
            no_improve = 0
        
        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakRed step={step+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def lbgfs_smooth_refine(a0, time_limit, T_schedule=None):
    """
    L-BFGS refinement using LogSumExp smooth approximation of the max.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    
    if T_schedule is None:
        T_schedule = [100, 50, 20, 10, 5, 2, 1, 0.5, 0.2, 0.1]
    
    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b_in = fast_convolve(a_in)
            max_b = T * b_in
            return (2.0 * n / T) * (max_b.max() + np.log(np.exp(max_b - max_b.max()).sum()))
        
        def jac(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b_in = fast_convolve(a_in)
            max_b = T * b_in
            exp_shifted = np.exp(max_b - max_b.max())
            w = exp_shifted / exp_shifted.sum()
            
            grad_a = np.zeros(n)
            for k in range(len(b_in)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_in[k - idx]
            
            S = np.dot(grad_a, a_in)
            return a_in * (grad_a - S)
        
        iters = max(20, min(2000, int(remaining / max(T * 0.1, 0.1))))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def subgradient_descent_direct(a0, time_limit, n_active=0.05):
    """
    Direct subgradient descent on the simplex (no softmax parameterization).
    Uses Polyak step size.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    f_lb = 2.0 * n / (2.0 * n - 1.0)  # Lower bound
    
    t0 = time.monotonic()
    lr = 0.1
    
    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        current_score = 2.0 * n * max_b
        
        # Active set
        threshold = max_b * (1 - n_active)
        active = np.where(b >= threshold)[0]
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Polyak step size
        gap = current_score - f_lb

def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast multiplicative-weight and projected-gradient best response (no LP solver)
2. Heavy-ball alternating with fictitious-play averaging for convergence
3. Adaptive active-set subgradient descent

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_with(a_fixed, b):
    n = len(a_fixed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a_fixed, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def best_response_mw(a, eta=0.5, n_iter=100, threshold_frac=0.88):
    """
    Approximate best response: find b >= 0, sum(b)=1 minimizing max_k (a*b)_k
    using multiplicative weight updates.
    """
    n = len(a)
    b = np.ones(n) / n
    best_b = b.copy()
    best_val = convolve_with(a, b).max()

    for it in range(n_iter):
        conv = convolve_with(a, b)
        max_conv = conv.max()

        if max_conv < best_val:
            best_val = max_conv
            best_b = b.copy()

        active_mask = conv >= threshold_frac * max_conv
        active_idx = np.where(active_mask)[0]
        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(conv))])

        grad = np.zeros(n)
        for k in active_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]

        log_b = np.log(np.maximum(b, 1e-30))
        log_b -= eta * grad
        log_b -= log_b.max()
        b_new = np.exp(log_b)
        b_new = np.maximum(b_new, 1e-30)
        s = b_new.sum()
        if s < 1e-30:
            break
        b_new = b_new / s
        b = b_new

    conv = convolve_with(a, b)
    if conv.max() < best_val:
        best_val = conv.max()
        best_b = b.copy()

    return best_b, best_val

        conv = convolve_with(a, b)
        max_conv = conv.max()

        if max_conv < best_val:
            best_val = max_conv

        active_idx = np.where(active_mask)[0]
        if len(active_idx) == 0:

        grad = np.zeros(n)
        for k in active_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(active_idx)

        b_new = project_simplex(b_new)

        if (it + 1) % 50 == 0:
            lr *= 0.7

    conv = convolve_with(a, b)
    if conv.max() < best_val:
        best_val = conv.max()
        best_b = b.copy()

    return best_b, best_val


def heavy_ball_alternate_fp(a0, time_limit, use_mw=True, mw_iters=80, eta_mw=0.4,
                            pg_iters=100, pg_lr=0.01):
    """
    Alternating minimization with heavy-ball momentum and fictitious play.
    Tracks running averages of a and b for the FP trick.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    b = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    # Fictitious play averages
    fp_a = a.copy()
    fp_b = b.copy()
    fp_count = 1

    v_a = np.zeros(n)
    v_b = np.zeros(n)
    mu = 0.3

    t0 = time.monotonic()
    stall = 0
    prev_best = best_score

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Best response for b given a
        if use_mw:
            b_new, _ = best_response_mw(a, eta=eta_mw, n_iter=mw_iters)
        else:
            b_new, _ = best_response_pg(a, lr=pg_lr, n_iter=pg_iters)

        # Heavy ball for b
        v_b = mu * v_b + (b_new - b)
        b = np.maximum(b + v_b, 0)
        sb = b.sum()
        if sb < 1e-15:
            b = b_new
        else:
            b = b / sb

        # Best response for a given b
        if use_mw:
            a_new, _ = best_response_mw(b, eta=eta_mw, n_iter=mw_iters)
        else:
            a_new, _ = best_response_pg(b, lr=pg_lr, n_iter=pg_iters)

        # Heavy ball for a
        v_a = mu * v_a + (a_new - a)
        a = np.maximum(a + v_a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = a_new
        else:
            a = a / sa

        # Update FP averages
        fp_a = fp_a + (a - fp_a) / (fp_count + 1)
        fp_b = fp_b + (b - fp_b) / (fp_count + 1)
        fp_count += 1

        # Evaluate all candidates
        score_a = 2.0 * n * fast_convolve(a).max()
        score_b = 2.0 * n * fast_convolve(b).max()
        score_fp_a = 2.0 * n * fast_convolve(fp_a).max()
        score_fp_b = 2.0 * n * fast_convolve(fp_b).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_b < best_score:
            best_score = score_b
            best_a = b.copy()
            stall = 0
        if score_fp_a < best_score:
            best_score = score_fp_a
            best_a = fp_a.copy()
            stall = 0
        if score_fp_b < best_score:
            best_score = score_fp_b
            best_a = fp_b.copy()
            stall = 0

        if it % 200 == 0:
            if best_score < prev_best - 1e-9:
                stall = 0
                prev_best = best_score
                # Increase momentum when improving
                mu = min(mu + 0.01, 0.7)
            else:
                stall += 1
                if stall >= 20:
                    mu = max(mu - 0.05, 0.2)
                    if stall >= 50:
                        # Perturb and reset momentum
                        rng = np.random.default_rng(it * 31 + 7)
                        noise = rng.random(n) * 0.03
                        a = np.maximum(a + noise, 0)
                        a = a / a.sum()
                        b = np.maximum(b + rng.random(n) * 0.03, 0)
                        b = b / b.sum()
                        v_a = np.zeros(n)
                        v_b = np.zeros(n)
                        mu = 0.3
                        stall = 0

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    HB-FP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} mu={mu:.3f}")

    # Also check FP averages one more time
    score_fp_a = 2.0 * n * fast_convolve(fp_a).max()
    score_fp_b = 2.0 * n * fast_convolve(fp_b).max()
    if score_fp_a < best_score:
        best_score = score_fp_a
        best_a = fp_a.copy()
    if score_fp_b < best_score:
        best_score = score_fp_b
        best_a = fp_b.copy()

    return best_a, best_score


def subgradient_adaptive(a0, time_limit, n_active_frac=0.04):
    """
    Adaptive subgradient descent with Polyak step size on the simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    f_lb = 2.0 * n / (2.0 * n - 1.0)

    t0 = time.monotonic()
    lr = 0.3
    no_improve = 0

    for step in range(300000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        current_score = 2.0 * n * max_b

        threshold = max_b * (1.0 - n_active_frac)
        active = np.where(b >= threshold)[0]

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        gap = current_score - f_lb
        if gap < 1e-12:
            break
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            break
        alpha = gap / (g_norm_sq + 1e-15)
        alpha = min(alpha, lr)

        a_new = a - alpha * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            lr = min(lr * 1.02, 1.0)
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.7
                no_improve = 0
            lr = max(lr, 1e-7)

        if (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.6f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_list=None):
    """L-BFGS refinement with LogSumExp smooth approximation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()

    if T_list is None:
        T_list = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]

    for T in T_list:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func_T(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b_in = fast_convolve(a_in)
            max_Tb = T * b_in
            max_val = max_Tb.max()
            return (2.0 * n / T) * (max_val + np.log(np.exp(max_Tb - max_val).sum()))

        def jac_T(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b_in = fast_convolve(a_in)
            max_Tb = T * b_in
            max_val = max_Tb.max()
            exp_shifted = np.exp(max_Tb - max_val)
            w = exp_shifted / exp_shifted.sum()

            grad_a = np.zeros(n)
            for k in range(len(b_in)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_in[k - idx]

            S = np.dot(grad_a, a_in)
            return a_in * (grad_a - S)

        iters = max(30, min(3000, int(remaining / max(0.5, T * 0.05))))
        try:
            res = minimize(
                func_T, z, jac_T, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def peak_local_search(a0, time_limit, max_steps=80000):
    """Targeted local search attacking specific peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_val = b.max()
        threshold = 0.93 * peak_val
        peaks = np.where(b >= threshold)[0]

        improved = False
        for k in peaks[:3]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]

            for _ in range(15):
                dst = rng.integers(n)
                if dst == src or a[src] < 1e-14:
                    continue
                for frac in [0.02, 0.05, 0.08]:
                    amount = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amount
                    a_trial[dst] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                noise = rng.standard_normal(n) * 0.01
                a = np.maximum(a + noise, 0)
                a = a / a.sum()
                no_improve = 0
            elif step > 5000 and no_improve > 3000:
                break

        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakLocal {elapsed:.0f}s: score={best_score:.6f}")

    return best_a, best_score


def interpolate_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
            best_seq = parent.copy()
            best_score = parent_score
    except Exception:
        pass

    if parent is None:
        n_init = 1000
        x = np.linspace(0, 1, n_init)
        best_seq = 0.5 + 0.5 * np.cos(2.0 * np.pi * x)
        best_seq = best_seq ** 1.5
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    n_best = len(best_seq)

    # === Phase 1: Fast alternating with FP (MW best response) ===
    print(f"\n=== Phase 1: HB-Alt FP / MW (best={best_score:.6f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.28, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a1, _ = heavy_ball_alternate_fp(best_seq, t_alloc, use_mw=True,
                                         mw_iters=60, eta_mw=0.35)
        try_update(a1, "hb_fp_mw")

    # === Phase 2: Subgradient descent ===
    print(f"\n=== Phase 2: Subgradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        a2, _ = subgradient_adaptive(best_seq, t_alloc, n_active_frac=0.03)
        try_update(a2, "subgrad")

    # === Phase 3: L-BFGS polish ===
    print(f"\n=== Phase 3: L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 280.0)
    if t_alloc > 20:
        a3 = lbgfs_refine(best_seq, t_alloc,
                          T_list=[30.0, 15.0, 7.0, 3.0, 1.5, 0.7, 0.3, 0.15, 0.08])
        try_update(a3, "lbgfs")

    # === Phase 4: Peak local search ===
    print(f"\n=== Phase 4: Peak Local (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 15:
        a4, _ = peak_local_search(best_seq, t_alloc)
        try_update(a4, "peak_local")

    # === Phase 5: Second alternating round with PG ===
    print(f"\n=== Phase 5: HB-Alt FP / PG (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        a5, _ = heavy_ball_alternate_fp(best_seq, t_alloc, use_mw=False,
                                         pg_iters=150, pg_lr=0.02)
        try_update(a5, "hb_fp_pg")

    # === Phase 6: Multi-length ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.3, 1.6, 0.8]:
        alt_n = int(n_best * mult)
        if 400 <= alt_n <= 3500 and alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:2]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_total = min(rem * 0.25, 200.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_to_length(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        if np.isfinite(sc_alt) and sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()
            print(f"    Interp n={alt_n}: {sc_alt:.6f}")

        t1 = t_total * 0.55
        a_alt, _ = heavy_ball_alternate_fp(init_alt, t1, use_mw=True,
                                            mw_iters=40, eta_mw=0.3)
        try_update(a_alt, f"alt{alt_n}_hb")

        rem = deadline - time.monotonic()
        if rem > 15:
            t2 = min(rem * 0.3, 60.0)
            if t2 > 10:
                a_alt2 = lbgfs_refine(best_seq, t2, T_list=[10, 5, 2, 1, 0.5, 0.2])
                try_update(a_alt2, f"alt{alt_n}_lbgfs")

    # === Phase 7: Final combined polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.35, 100.0)
        a_fin = lbgfs_refine(best_seq, t_fin,
                             T_list=[5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05])
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 25:
        t_pr = min(rem * 0.3, 60.0)
        a_pr, _ = peak_local_search(best_seq, t_pr)
        try_update(a_pr, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_alt = min(rem * 0.4, 50.0)
        a_fa, _ = heavy_ball_alternate_fp(best_seq, t_alt, use_mw=True,
                                           mw_iters=30, eta_mw=0.25)
        try_update(a_fa, "final_hb")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
