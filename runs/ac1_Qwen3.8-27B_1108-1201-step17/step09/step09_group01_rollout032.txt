The FW method works as follows:

2. At each iteration, solve the linear mini-batch: find s in simplex that minimizes grad(f(a))^T s

This is a very structured step: it moves mass to the "gap" in the shifted sequence.

The step size t can be chosen as t = 2/(iter+2) (standard FW) or via line search.

1. Peak-Deflation Gradient Flow (novel): multiplicative updates targeting top peaks
2. Frank-Wolfe method for global moves
3. Exact coordinate descent with score-based steps
4. LP fictitious play for escape from local optima

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def peak_deflation(a0, time_limit, eta_init=0.1, top_k=5):
    """
    Novel peak-smoothing via multiplicative updates.
    At each step:
    1. Find top-k peaks of the convolution
    2. Compute pressure field: P(i) = sum of a_{k-i} over top peaks
    3. Update: a_i *= exp(-eta * P(i) / P_max)
    4. Renormalize to simplex
    5. Adapt eta based on improvement
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()
    eta = eta_init
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Find top-k peaks
        if top_k < len(conv):
            # Use argpartition for efficiency
            top_indices = np.argpartition(conv, -top_k)[-top_k:]
        else:
            top_indices = np.arange(len(conv))
        
        # Compute pressure field
        pressure = np.zeros(n)
        for k in top_indices:
            w = conv[k] / M  # weight proportional to peak value
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = i_start + (k - i_end)
            # a[k-i] for i in [i_start, i_end]
            for i in range(i_start, i_end + 1):
                pressure[i] += w * a[k - i]
        
        # Normalize pressure
        p_max = pressure.max()
        if p_max < 1e-30:
            break
        pressure = pressure / p_max
        
        # Multiplicative update
        a_new = a * np.exp(-eta * pressure)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        M_new = float(conv_new.max())
        
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            M = M_new
            no_improve = 0
            # Slightly increase eta
            eta = min(eta * 1.02, 2.0)
        else:
            no_improve += 1
            eta *= 0.9
            if no_improve > 50:
                # Try smaller eta
                eta = max(eta * 0.5, 0.001)
                no_improve = 0
                if eta < 0.001:
                    break
        
    return best_a, best_M * n * 2  # score = 2*n*M (with sum=1)

def frank_wolfe(a0, time_limit, max_iters=5000):
    """
    Frank-Wolfe method for min max_k (a*a)[k] s.t. a in simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()
    it = 0
    
    while it < max_iters and time.monotonic() - t0 < time_limit:
        it += 1
        
        # Find active peak
        k_star = int(conv.argmax())
        
        # Gradient: grad_i = 2 * a[k_star - i] for valid i
        grad = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        for i in range(i_start, i_end + 1):
            grad[i] = 2.0 * a[k_star - i]
        
        # FW linear mini-batch: find s in simplex minimizing grad^T s
        # This is the coordinate with minimum grad
        i_min = int(np.argmin(grad))
        s_vec = np.zeros(n)
        s_vec[i_min] = 1.0
        
        # Step size (standard FW: t = 2/(it+2))
        t = 2.0 / (it + 2.0)
        
        a_new = (1 - t) * a + t * s_vec
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        M_new = float(conv_new.max())
        
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()
        
        a = a_new
        conv = conv_new
        M = M_new
    
    return best_a, best_M * 2 * n

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_score(a):
    """Compute score = 2*n*max(conv)/sum^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    M = float(fast_convolve(a).max())
    return 2.0 * len(a) * M / (s * s)


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    if '_lp_solver_cached' in globals() and _lp_solver_cached is not None:
        return _lp_solver_cached
    _lp_solver_cached = None
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.GLOP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.SCS
    return cp.SCS


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.PDLP, cp.GLOP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def peak_deflation(a0, time_limit, top_k=8, eta_init=0.08):
    """
    Novel peak-smoothing via multiplicative updates targeting top peaks.
    The pressure field is computed from the top-k peaks of the convolution.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()
    eta = eta_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Find top-k peaks
        if top_k < n:
            top_indices = np.argpartition(conv, -top_k)[-top_k:]
        else:
            top_indices = np.arange(len(conv))

        # Compute pressure field
        pressure = np.zeros(n)
        for k in top_indices:
            w = max(conv[k] - M * 0.8, 0) / (M * 0.2 + 1e-30)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                pressure[i] += w * a[k - i]

        p_max = pressure.max()
        if p_max < 1e-30:
            break
        pressure = pressure / p_max

        # Multiplicative update with clipping
        a_new = a * np.exp(-eta * pressure)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        M_new = float(conv_new.max())

        if M_new < best_M - 1e-13:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            M = M_new
            no_improve = 0
            eta = min(eta * 1.01, 1.0)
        else:
            no_improve += 1
            eta *= 0.92
            if no_improve > 80:
                if eta < 0.005:
                    break
                no_improve = 0

    return best_a, 2.0 * n * best_M


def coord_descent_fast(a0, time_limit, verbose=False):
    """
    Fast coordinate descent with incremental convolution updates.
    Tries multiple step sizes per coordinate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-18:
                continue

            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            unaff = 0.0
            if k_lo > 0:
                unaff = float(conv[:k_lo].max())
            if k_hi < 2 * n - 2:
                unaff = max(unaff, float(conv[k_hi + 1:].max()))

            # Affected positions
            ks = np.arange(k_lo, k_hi + 1)
            a_idx = ks - i
            a_vals = a[a_idx]
            conv_vals = conv[k_lo:k_hi + 1]

            best_delta = 0.0
            best_new_M = M

            # Try multiple step sizes
            for f in [0.5, 0.7, 0.85, 0.95, 0.99, -0.02, -0.05, -0.1, -0.2, -0.35, -0.5, 0.1, 0.3]:
                delta = f * ai
                new_val = ai + delta
                if new_val < -1e-15:
                    continue

                new_conv = conv_vals + 2.0 * delta * a_vals
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    idx_2i = 2 * i - k_lo
                    new_conv = new_conv.copy()
                    new_conv[idx_2i] += delta * delta

                new_M = max(unaff, float(new_conv.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0:
                a[i] = ai + best_delta
                s += best_delta
                # Update convolution incrementally
                conv[k_lo:k_hi + 1] = conv_vals + 2.0 * best_delta * a_vals
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    conv[2 * i] += best_delta * best_delta
                M = max(unaff, float(conv[k_lo:k_hi + 1].max()))

                cur_score = 2.0 * n * M / (s * s)
                if cur_score < best_score - 1e-13:
                    best_score = cur_score
                    best_a = (a / s).copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 5:
                break
        else:
            no_improve_sweeps = 0

        if verbose and (sweeps := int(time.monotonic() - t0)) % 30 == 0:
            print(f"    CD: t={time.monotonic()-t0:.0f}s M={M:.8f} score={2.0*n*M/(s*s):.6f}")

    # Final exact computation
    s_final = a.sum()
    if s_final > 1e-15:
        a_final = a / s_final
    else:
        a_final = np.ones(n) / n
    M_final = float(fast_convolve(a_final).max())
    score_final = 2.0 * n * M_final
    if score_final < best_score:
        best_score = score_final
        best_a = a_final.copy()

    return best_a, best_score

def frank_wolfe(a0, time_limit, max_iters=10000):
    """Frank-Wolfe method for min max_k(a*a)[k] on simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    it = 0

    while it < max_iters and time.monotonic() - t0 < time_limit:
        it += 1

        k_star = int(conv.argmax())

        # Gradient at active peak
        grad = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        for i in range(i_start, i_end + 1):
            grad[i] = 2.0 * a[k_star - i]

        # FW step: move toward vertex with minimum gradient
        i_min = int(np.argmin(grad))
        t = 2.0 / (it + 2.0)
        a_new = (1.0 - t) * a + t * np.eye(n)[i_min]

        conv_new = fast_convolve(a_new)
        M_new = float(conv_new.max())
        score_new = 2.0 * n * M_new

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new
        conv = conv_new
        M = M_new

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def random_perturb(a0, rng, strength=0.05):
    """Random perturbation to escape local optima."""
    a = a0.copy()
    noise = rng.standard_normal(len(a)) * strength
    a = a * (1.0 + noise)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def structured_perturb(a0, rng, frac=0.2):
    """Zero out a fraction of coordinates and redistribute."""
    a = a0.copy()
    n = len(a)
    mask = rng.random(n) > frac
    n_keep = max(mask.sum(), n // 5)
    if mask.sum() < n_keep:
        idx = np.where(mask)[0]
        extra = rng.choice(np.where(~mask)[0], size=n_keep - len(idx), replace=False)
        mask = np.zeros(n, dtype=bool)
        mask[idx] = True
        mask[extra] = True
    a[~mask] = 0
    a[mask] = a[mask] * (1.0 + rng.standard_normal(mask.sum()) * 0.3)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s

            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"cd_fw_r{rnd}")

            # Perturb for next round
            if rnd < 3:
                if rnd % 2 == 0:
                    current = random_perturb(best_seq, rng, strength=0.03 + 0.02 * rnd)
                else:
                    current = structured_perturb(best_seq, rng, frac=0.1 + 0.05 * rnd)

    # === Phase 6: Try different lengths (200s) ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.6f}) ===")
    t6 = time.monotonic()
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for alt in [n_best + 128, n_best + 256, n_best - 128, 1536, 2048, 512]:
            if 256 <= alt <= 3000 and alt != n_best:
                if alt not in alt_lengths:
                    alt_lengths.append(alt)
        alt_lengths = alt_lengths[:3]

        for alt_n in alt_lengths:
            rem = deadline - time.monotonic() - 100
            if rem < 40:
                break
            rem = deadline - time.monotonic() - 50
            if rem < 30:
                break

            init_alt = interpolate_seq(best_seq, alt_n)
            t_cd_alt = min(rem * 0.6, 100.0)
            a_alt, s_alt = coord_descent_fast(init_alt, t_cd_alt)
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

            rem = deadline - time.monotonic() - 30
            if rem > 20:
                a_fine, s_fine = fine_coord_descent(best_seq, min(rem * 0.5, 50.0))
                best_seq, best_score = check_and_update(a_fine, best_seq, best_score, f"fine_n{len(best_seq)}")

    # === Phase 7: Another LP round if time remains (100s) ===
    print(f"\n=== Phase 7: Final LP + Polish (best={best_score:.6f}) ===")
    t7 = time.monotonic()
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 60:
            t_lp2 = min(rem * 0.4, 60.0)
            a_lp2, s_lp2, _ = lp_fictitious_play(best_seq, t_lp2, solver=solver)
            best_seq, best_score = check_and_update(a_lp2, best_seq, best_score, "lp_final")

        rem = deadline - time.monotonic() - 20
        if rem > 20:
            a_cd_final, _ = coord_descent_fast(best_seq, min(rem * 0.6, 60.0))
            best_seq, best_score = check_and_update(a_cd_final, best_seq, best_score, "cd_final")

    # === Phase 8: Fine polishing (remaining time) ===
    print(f"\n=== Phase 8: Fine Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_fine, s_fine = fine_coord_descent(best_seq, rem)
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
