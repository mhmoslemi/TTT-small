Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an improved LP-alternation
method with dual-gap monitoring, adaptive Nesterov acceleration, spectral
initialization, and multi-stage final polishing (LogSumExp L-BFGS +

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_conv_matrix_transpose(a):
    """Build n x (2n-1) matrix T^T such that T^T @ y = cross-corr."""
    n = len(a)
    size = 2 * n - 1
    Tt = np.zeros((n, size))
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            Tt[j, k] = a[k - j]
    return Tt


_lp_solver_cached = None


def get_lp_solver():
    """Find the fastest working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    best_solver = cp.ECOS
    best_time = float('inf')
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            n_test = 500
            a = np.random.rand(n_test)
            a /= a.sum()
            T = build_conv_matrix(a)
            b_var = cp.Variable(n_test, nonneg=True)
            M_var = cp.Variable()
            constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            dt = time.monotonic() - t0
            if prob.status in ('optimal', 'optimal_inaccurate'):
                if dt < best_time:
                    best_time = dt
                    best_solver = solver_name
        except Exception:
            continue
    _lp_solver_cached = best_solver
    return best_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Returns (b, M, dual_y) where dual_y gives a certificate."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    # Add dual variables to extract certificate
    dual_vars = []
    constr_list = []
    for k in range(size):
        c = cp.Constraint(T[k, :] @ b_var <= M_var)
        constr_list.append(c)
        dual_vars.append(cp.NonPos())
    prob = cp.Problem(cp.Minimize(M_var), constr_list + [cp.sum(b_var) == 1])
    
    solved = False
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True
    except Exception:
        pass
    
    if not solved:
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
    
    if not solved:
        return None, None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    return b_val, float(M_var.value), None

        solver = get_lp_solver()
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    prob = cp.Problem(
        cp.Minimize(M_var),
        [T @ b_var <= M_var, cp.sum(b_var) == 1]
    )
    solved = False
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True
    except Exception:
        pass
    if not solved:
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=500000):
    """Nesterov-accelerated LP alternation with adaptive step and restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_val = solve_lp_best_response_simple(y, solver)
        if b_new is None:
            continue
        
        # Adaptive step: use standard Nesterov rate
        alpha = 2.0 / (it + 2.0)
        
        # Update y (the "query" point)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        # Update a (the running average, which is the actual iterate)
        # Nesterov: a_t = (1/(t+1)) * sum_{i=0}^{t} b_i (with weights)
        denom = (it + 1.0) + alpha * (it + 2.0)
        a = ((it + 1.0) * a + alpha * (it + 2.0) * b_new) / denom
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        # Evaluate
        b_conv = fast_convolve(a)
        score_a = 2.0 * n * b_conv.max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        b_conv_y = fast_convolve(y)
        score_y = 2.0 * n * b_conv_y.max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Stall detection and restart
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 15:
                # Structured restart: blend with a random perturbation
                scale = 0.05 / np.sqrt(stall_count)
                noise = rng.standard_normal(n) * scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
            
            if (it + 1) % 5000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def spectral_init(n, rng):
    """Initialize using spectral method: create sequence with approximately flat spectrum."""
    # Strategy: start with a raised-cosine or similar smooth window
    # then add controlled oscillation to flatten the spectrum
    
    # Method 1: Raised cosine with spectral shaping
    x = np.linspace(0, 1, n)
    base = 0.5 * (1 + np.cos(2 * np.pi * x))  # raised cosine
    
    # Method 2: Multi-scale cosine modulation
    freqs = rng.choice([1, 2, 3, 4, 5], size=3, replace=False)
    amps = rng.uniform(0.05, 0.15, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    mod = np.zeros(n)
    for f, a_m, p in zip(freqs, amps, phases):
        mod += a_m * np.cos(2 * np.pi * f * x + p)
    
    a = base + mod
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif choice == 1:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0)
    elif choice == 3:
        a = (1 - x) ** 0.8 + 0.05
    elif choice == 4:
        # Two-bump
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2) + 0.05
    elif choice == 5:
        # Flat with edges damped
        edge = 0.3
        a = np.ones(n)
        mask = x < edge
        a[mask] = 0.5 + 0.5 * x[mask] / edge
        mask = x > (1 - edge)
        a[mask] = 0.5 + 0.5 * (1 - x[mask]) / edge
    elif choice == 6:
        # Sinusoidal modulation
        a = 0.7 + 0.25 * np.sin(2 * np.pi * 3 * x) + 0.05 * np.sin(2 * np.pi * 7 * x)
        a = np.maximum(a, 0)
    else:
        # Gaussian bump
        a = np.exp(-((x - 0.5) ** 2) / 0.08) + 0.02
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lbgfs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing in log-domain."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        iters = max(50, min(maxiter_per_stage, int(remaining * 20)))
        
        def obj(z_in):
            a_in = softmax_from_z(z_in)
            b = fast_convolve(a_in)
            max_b = b.max()
            exp_b = np.exp(b / stage_T - max_b / stage_T)
            lse = stage_T * (max_b / stage_T + np.log(exp_b.sum() / len(b)))
            return 2.0 * n * lse
        
        def grad(z_in):
            a_in = softmax_from_z(z_in)
            b = fast_convolve(a_in)
            max_b = b.max()
            exp_b = np.exp(b / stage_T - max_b / stage_T)
            w = exp_b / exp_b.sum()
            g = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-18:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                g[idx] += (2.0 / stage_T) * w[k] * a_in[k - idx]
            S = np.dot(g, a_in)
            return a_in * (g - S)
        
        try:
            res = minimize(
                fun=obj, x0=z, jac=grad, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': iters * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def peak_shave_refine(a0, time_limit, max_rounds=50000):
    """Aggressive coordinate descent targeting peak contributors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find top peaks (within 5% of max)
        threshold = max_b * 0.95
        peak_positions = np.where(b >= threshold)[0]
        
        improved = False
        for pk in peak_positions:
            if improved:
                break
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            # Contribution of each pair
            contribs = a[idx] * a[pk - idx]
            # Try reducing the top contributor
            top_i = np.argmax(contribs)
            pos = idx[top_i]
            partner = pk - pos
            
            if a[pos] < 1e-12:
                continue
            
            # Try reducing pos and/or partner
            for target in [pos, partner] if pos != partner else [pos]:
                if a[target] < 1e-12:
                    continue
                for factor in [0.99, 0.98, 0.97, 0.95]:
                    a_trial = a.copy()
                    reduction = a_trial[target] * (1 - factor)
                    a_trial[target] -= reduction
                    # Redistribute to non-contributing elements
                    mask = np.ones(n, dtype=bool)
                    mask[target] = False
                    # Also try not to redistribute to elements that contribute to other peaks
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] += reduction * a_trial[mask] / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 100:
                # Try a small random perturbation
                noise = np.random.default_rng(rnd).standard_normal(n) * 0.01
                a_trial = a + noise
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                else:
                    if stall > 500:
                        break
        else:
            stall = 0
        
        if rnd % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave {elapsed:.0f}s: score={best_score:.8f}")
    
    return best_a, best_score


def evaluate_local(a):
    """Fast local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def multi_start_lp(a_list, time_budget, solver):
    """Run LP alternation from multiple starts, return best."""
    best_a = None
    best_score = float('inf')
    n_starts = len(a_list)
    t_per_start = time_budget / n_starts
    
    for i, a_init in enumerate(a_list):
        remaining = time_budget - (time.monotonic() - multi_start_lp._t0) if hasattr(multi_start_lp, '_t0') else time_budget
        if remaining < 10:
            break
        t_alloc = min(t_per_start, remaining - 5)
        if t_alloc < 10:
            break
        a_res, sc = nesterov_lp_refine(a_init, t_alloc, solver=solver, n_max_iters=100000)
        if sc < best_score:
            best_score = sc
            best_a = a_res
            print(f"    MultiStart[{i}]: score={sc:.8f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP alternation, spectral init,
    and aggressive polishing to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    n_best = len(best_seq)
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    # === Phase 1: Spectral initialization + short LP ===
    print(f"\n=== Phase 1: Spectral Init ({n_best}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_spec = min(rem * 0.08, 80.0)
        # Try spectral initialization
        a_spec = spectral_init(n_best, rng)
        sc_spec = evaluate_local(a_spec)
        print(f"  Spectral init score: {sc_spec:.8f}")
        if sc_spec < best_score * 1.01:
            # Run short LP from spectral init
            a_sp, _ = nesterov_lp_refine(a_spec, t_spec * 0.6, solver=solver, n_max_iters=20000)
            try_update(a_sp, "spectral_lp")
        # Also try from mixed init (avg of parent and spectral)
        a_mix = 0.5 * best_seq + 0.5 * a_spec
        a_mix = np.maximum(a_mix, 0)
        a_mix = a_mix / a_mix.sum()
        a_mx, _ = nesterov_lp_refine(a_mix, t_spec * 0.4, solver=solver, n_max_iters=10000)
        try_update(a_mx, "mixed_lp")
    
    # === Phase 2: Main LP optimization from best ===
    print(f"\n=== Phase 2: Main Nesterov LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.45, 700.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=500000)
        try_update(a_lp, "main_lp")
    
    # === Phase 3: L-BFGS LogSumExp refinement ===
    print(f"\n=== Phase 3: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 30:
        T_stages = [30.0, 15.0, 7.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.03, 0.02]
        print(f"  T_stages: {T_stages}")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=4000)
        try_update(a_lb, "lbgfs")
    
    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        a_ps, _ = peak_shave_refine(best_seq, t_alloc, max_rounds=80000)
        try_update(a_ps, "peak_shave")
    
    # === Phase 5: Second LP pass from polished result ===
    print(f"\n=== Phase 5: Second LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_lp2, "lp_2nd")
    
    # === Phase 6: Final L-BFGS at very low temperature ===
    print(f"\n=== Phase 6: Final L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 100.0)
    if t_alloc > 15:
        T_final = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
        a_fin = lbgfs_refine(best_seq, t_alloc, T_stages=T_final, maxiter_per_stage=3000)
        try_update(a_fin, "lbgfs_final")
    
    # === Phase 7: Final peak shave ===
    print(f"\n=== Phase 7: Final Peak Shave (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 60.0)
    if t_alloc > 10:
        a_ps2, _ = peak_shave_refine(best_seq, t_alloc, max_rounds=30000)
        try_update(a_ps2, "peak_final")
    
    # === Phase 8: Try perturbation escape ===
    print(f"\n=== Phase 8: Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        t_pert = min(rem * 0.3, 100.0)
        n_cur = len(best_seq)
        x_lin = np.linspace(0, 2 * np.pi, n_cur)
        # Try different frequency perturbations
        for freq in [1, 2, 5]:
            rem_inner = deadline - time.monotonic()
            if rem_inner < 20:
                break
            amp = 0.03
            perturbation = amp * np.cos(freq * x_lin + rng.uniform(0, 2 * np.pi))
            perturbed = best_seq + perturbation
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps < 1e-15:
                continue
            perturbed = perturbed / ps
            t_sub = min(t_pert / 3, rem_inner - 10)
            if t_sub < 10:
                break
            a_p, _ = nesterov_lp_refine(perturbed, t_sub, solver=solver, n_max_iters=30000)
            try_update(a_p, f"pert_f{freq}")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    
    # Return with safe bounds
    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
