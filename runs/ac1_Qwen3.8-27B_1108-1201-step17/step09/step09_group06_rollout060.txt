- The active constraints are those where (T@b)_k ≈ M (the current peak).
- Solve the LP with only these constraints.
- Check if the new solution violates any of the excluded constraints.
- If so, add them and re-solve.

This is a "constraint generation" approach. In practice, only a small number of constraints (maybe 10-50 out of 1919) will be active, so the LP will be much faster.

For the second-order approach:
- The objective is `f(a) = max_k a^T Q_k a` where Q_k are the "flip" matrices.
- At the optimum, the active constraints are those where a^T Q_k a = M.
- We can use a KKT-based method: find a, M, and multipliers lambda_k such that:
  - sum_k lambda_k = 1
  - sum_k lambda_k Q_k a = mu * 1 (for the sum constraint)
  - a_j > 0 implies the constraint is satisfied
  - a^T Q_k a <= M for all k

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Multi-stage L-BFGS on log-sum-exp with adaptive temperature,
(2) Reduced-constraint LP best-response with Nesterov acceleration,
(3) Dual-variable tracking for warm starts, (4) Coordinate descent polish,
(5) Multi-length exploration with interpolation.
Key innovations: constraint-reduced LPs for speed, Nesterov with adaptive
restarts, and exploiting the Toeplitz structure of the convolution matrix.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fft_conv(a, b=None):
    """Fast convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    size = n + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def normalize_nonneg(a):
    """Clamp to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def log_sum_exp(x):
    """Numerically stable log-sum-exp."""
    x_max = x.max()
    if x_max > 700:
        return x_max + np.log(np.exp(x - x_max).sum())
    return np.log(np.exp(x).sum())


def compute_grad_lse(a, T, n):
    """Compute gradient of (2n/T)*log(sum exp(T*conv(a,a))) w.r.t. a."""
    conv = fft_conv(a)
    Tconv = T * conv
    Tconv_shift = Tconv - Tconv.max()
    exp_shifted = np.exp(Tconv_shift)
    p = exp_shifted / exp_shifted.sum()

    # grad_a_j = 4n * sum_k p_k * a[k-j]
    # = 4n * cross_correlation(a, p)_j  where p has length 2n-1
    # Using FFT: cross_corr(a, p)_j = sum_m a_m * p[j+m]
    p_ext = p  # length 2n-1
    corr_len = n + (2 * n - 1) - 1
    fft_corr_size = 1 << (corr_len - 1).bit_length()
    A_c = rfft(a, fft_corr_size)
    P_c = rfft(p_ext, fft_corr_size)
    cross = irfft(A_c * np.conj(P_c), fft_corr_size)[:n]

    grad_a = 4.0 * n * cross
    return grad_a, p


def lse_obj_and_grad_z(z, T, n):
    """Objective and gradient w.r.t. z (softmax parameterization)."""
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    a = exp_z / exp_z.sum()

    conv = fft_conv(a)
    Tconv = T * conv
    val = (2.0 * n / T) * (Tconv.max() + np.log(np.exp(Tconv - Tconv.max()).sum()))

    grad_a, _ = compute_grad_lse(a, T, n)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return val, grad_z


def z_from_a(a):
    """Convert a (on simplex) to z (log-domain)."""
    return np.log(np.maximum(a, 1e-30))


def softmax_from_z(z):
    """Convert z to a on simplex."""
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def build_conv_matrix_rows(a, active_indices):
    """Build only the rows of the conv matrix for active indices."""
    n = len(a)
    T_active = np.zeros((len(active_indices), n))
    for i, k in enumerate(active_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        T_active[i, js:je+1] = a[k - np.arange(js, je+1)]
    return T_active


def solve_reduced_lp(a, b_init, active_frac=0.15, solver=None):
    """
    Solve the best-response LP with only active/near-active constraints.
    Returns (b_opt, M_opt, active_indices).
    """
    n = len(a)
    conv_with_b = fft_conv(a, b_init)
    M_curr = conv_with_b.max()
    threshold = M_curr * (1.0 - active_frac)
    active = np.where(conv_with_b >= threshold)[0]
    if len(active) < 3:
        active = np.argsort(conv_with_b)[::-1][:10]

    # Build reduced constraint matrix
    T_act = build_conv_matrix_rows(a, active)
    n_cons = len(active)

    # LP: min M s.t. T_act @ b <= M, sum(b)=1, b>=0
    # Variables: [b_0,...,b_{n-1}, M] (total n+1)
    # Constraints: T_act @ b - M <= 0 (n_cons), sum(b) = 1
    # Using linprog: min c^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds

    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M

    A_ub = np.hstack([T_act, -np.ones((n_cons, 1))])
    b_ub = np.zeros(n_cons)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if res.success:
            b_opt = res.x[:n]
            M_opt = res.x[n]
            b_opt = np.maximum(b_opt, 0)
            s = b_opt.sum()
            if s < 1e-15:
                return None, None, active
            b_opt /= s
            return b_opt, float(M_opt), active
    except Exception:
        pass

    # Fallback: full LP via cvxpy
    return solve_full_lp(a, solver)


def solve_full_lp(a, solver=None):
    """Solve the full best-response LP via cvxpy."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build full constraint matrix
    T = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        T[k, js:je+1] = a[k - np.arange(js, je+1)]

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val /= s
    return b_val, float(M_var.value), None


def nesterov_lp_refine(a0, time_limit, solver=None, max_iters=200000,
                        use_reduced_lp=True, active_frac=0.12):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = normalize_nonneg(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fft_conv(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    rng = np.random.default_rng(42)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if use_reduced_lp:
            b_new, M_b, active = solve_reduced_lp(a, y, active_frac=active_frac)
        else:
            b_new, M_b, active = solve_full_lp(y, solver)

        if b_new is None:
            b_new, M_b, active = solve_full_lp(y, solver)
            if b_new is None:
                continue

        # Nesterov update
        alpha = 2.0 / (it + 2)
        y_new = (1 - alpha) * y + alpha * b_new
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new

        a_new = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a_new / sa

        # Track best
        score_a = 2.0 * n * fft_conv(a).max()
        score_y = 2.0 * n * fft_conv(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection and restart
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 10:
                # Perturb to escape local optimum
                noise = rng.random(n) * 0.01
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = a + noise * 0.5
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M={M_b:.8f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=500.0, T_end=0.05, n_stages=10):
    """Multi-stage L-BFGS with log-sum-exp smoothing."""
    n = len(a0)
    z = z_from_a(a0)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    best_z = z.copy()
    best_score = 2.0 * n * fft_conv(a0).max()

    for stage, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        maxiter = max(50, min(2000, int(remaining * 200)))

        def make_fun(T_val):
            def func(z_in):
                if time.monotonic() - t0 > time_limit:
                    return 1e15
                return lse_obj_and_grad_z(z_in, T_val, n)[0]
            return func

        def make_jac(T_val):
            def jac(z_in):
                if time.monotonic() - t0 > time_limit:
                    return np.zeros_like(z_in)
                return lse_obj_and_grad_z(z_in, T_val, n)[1]
            return jac

        try:
            res = minimize(
                fun=make_fun(T),
                x0=z,
                jac=make_jac(T),
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': maxiter * 200,
                         'disp': False}
            )
            z = res.x
            a_curr = softmax_from_z(z)
            sc = 2.0 * n * fft_conv(a_curr).max()
            if sc < best_score:
                best_score = sc
                best_z = z.copy()
        except Exception as e:
            pass

    return softmax_from_z(best_z)


def mirror_descent(a0, time_limit, n_active_frac=0.05):
    """Mirror descent with subgradient and adaptive step size."""
    n = len(a0)
    z = z_from_a(a0).copy()
    a = softmax_from_z(z)
    best_a = a.copy()
    best_score = 2.0 * n * fft_conv(a).max()
    t0 = time.monotonic()
    lr = 0.3
    no_improve = 0

    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fft_conv(a)
        max_b = conv.max()
        threshold = max_b - n_active_frac * max_b
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])

        # Subgradient: average over active
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Mirror descent update in z-space
        g_scaled = a * (g - np.dot(g, a))  # project to tangent space
        g_norm = np.linalg.norm(g_scaled)
        if g_norm < 1e-15:
            break

        z = z - lr * g_scaled / g_norm
        a = softmax_from_z(z)
        score = 2.0 * n * fft_conv(a).max()

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 50:
                lr *= 0.8
                no_improve = 0
        lr = min(lr * 1.02, 3.0)

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_rounds=5000):
    """Coordinate descent: try adjusting pairs of coordinates to reduce peak."""
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fft_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fft_conv(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]

        # Find coordinates that contribute most to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]

        # Try moving mass from high-contribution to low-contribution positions
        if len(contrib) > 2:
            src = idx[np.argmax(contrib)]
            dst = idx[np.argmin(contrib)]

            if a[src] > 1e-10 and src != dst:
                for frac in [0.01, 0.03, 0.05, 0.08]:
                    amount = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amount
                    a_trial[dst] += amount
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    sc = 2.0 * n * fft_conv(a_trial).max()
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        break

        # Random perturbation every 500 rounds
        if rnd % 500 == 499:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                amount = a[i] * 0.05
                a_trial = a.copy()
                a_trial[i] -= amount
                a_trial[j] += amount
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s > 1e-15:
                    a_trial = a_trial / s
                    sc = 2.0 * n * fft_conv(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()

    return best_a, best_score


def structured_init(n, rng, seed_type=0):
    """Generate diverse structured initializations."""
    if seed_type == 0:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    elif seed_type == 1:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 0.5
    elif seed_type == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 5.0 * (x - 0.5) ** 2)
    elif seed_type == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 1.0 + 0.2 * np.cos(3 * x) + 0.1 * np.cos(7 * x) + 0.05 * np.cos(11 * x)
    elif seed_type == 4:
        # Bimodal
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.2) ** 2) + np.exp(-20 * (x - 0.8) ** 2) + 0.1
    else:
        a = np.ones(n)
        a = a + rng.random(n) * 0.1
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    except Exception:

        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=1000.0, T_end=0.1, n_stages=12)
        try_update(a_lb, "lbgfs_smooth")

    # === Phase 2: Nesterov LP (main workhorse, reduced constraints) ===
    print(f"\n=== Phase 2: Nesterov LP Reduced (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 600.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_np, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver,
                                      max_iters=300000, use_reduced_lp=True,
                                      active_frac=0.10)
        try_update(a_np, "nesterov_lp_1")

    # === Phase 3: L-BFGS sharp (low temperature) ===
    print(f"\n=== Phase 3: L-BFGS Sharp (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 240.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb2 = lbgfs_refine(best_seq, t_alloc, T_start=50.0, T_end=0.01, n_stages=8)
        try_update(a_lb2, "lbgfs_sharp")

    # === Phase 4: Nesterov LP round 2 (tighter) ===
    print(f"\n=== Phase 4: Nesterov LP R2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 360.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_np2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver,
                                       max_iters=200000, use_reduced_lp=True,
                                       active_frac=0.05)
        try_update(a_np2, "nesterov_lp_2")

    # === Phase 5: Mirror descent ===
    print(f"\n=== Phase 5: Mirror Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        a_md, _ = mirror_descent(best_seq, t_alloc, n_active_frac=0.03)
        try_update(a_md, "mirror_desc")

    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: Coord Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_descent(best_seq, t_alloc, n_rounds=10000)
        try_update(a_cd, "coord_desc")

    # === Phase 7: Alternative lengths ===
    print(f"\n=== Phase 7: Alt Lengths (best={best_score:.8f}) ===")
    alt_lengths = sorted(set([
        int(n_best * 0.85), int(n_best * 1.1),
        int(n_best * 1.25), int(n_best * 1.5),
        n_best + 60, n_best - 60
    ]))
    alt_lengths = [n for n in alt_lengths if 384 <= n <= 2500 and n != n_best]
    alt_lengths = alt_lengths[:3]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 90:
            break
        t_total = min(rem * 0.20, 180.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()
            print(f"  Interpolated to n={alt_n}: {sc_alt:.8f}")

        # Quick Nesterov LP on new length
        t_lp = t_total * 0.6
        a_alt, _ = nesterov_lp_refine(init_alt, t_lp, solver=solver,
                                       max_iters=50000, use_reduced_lp=True,
                                       active_frac=0.10)
        try_update(a_alt, f"lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None and len(best_seq) == alt_n:
            t_lb = min(rem * 0.3, 60.0)
            if t_lb > 15:
                a_alt2 = lbgfs_refine(best_seq, t_lb, T_start=100.0, T_end=0.05, n_stages=6)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 8: Final combined polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t1 = min(rem * 0.3, 60.0)
        a_f1 = lbgfs_refine(best_seq, t1, T_start=20.0, T_end=0.01, n_stages=6)
        try_update(a_f1, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 30:
        t2 = min(rem * 0.3, 50.0)
        a_f2, _ = nesterov_lp_refine(best_seq, t2, solver=solver,
                                      max_iters=50000, use_reduced_lp=True,
                                      active_frac=0.03)
        try_update(a_f2, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 20:
        t3 = min(rem * 0.3, 40.0)
        a_f3, _ = coordinate_descent(best_seq, t3, n_rounds=5000)
        try_update(a_f3, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_start
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={total_time:.0f}s")

    return [float(x) for x in best_seq]
