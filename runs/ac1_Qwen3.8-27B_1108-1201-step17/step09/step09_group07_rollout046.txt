def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv(a,a))."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix_sparse(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b) using banded structure."""
    n = len(a)
    size = 2 * n - 1
    # T is banded: T[k, j] = a[j] if 0 <= k-j < n, else 0
    # i.e., T[k, j] = a[j] * 1[0 <= k-j <= n-1] = a[j] * 1[j-n+1 <= k <= j]
    # So column j has a[j] in rows j through j+n-1
    # We'll use a sparse representation via the banded structure
    # For LP, we'll construct it densely if n is small enough, else use scipy.sparse
    from scipy.sparse import diags, eye as sp_eye
    # Actually for the LP, let's just build it as a 2D array if feasible
    # For n <= 2000, (2n-1) x n = ~4M entries, which is fine
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b) s.t. b>=0, sum(b)=1."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_sparse(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue

                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

def lp_nesterov_refine(a0, time_limit, solver=None, n_max_iters=100000):
    """

    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    last_print = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_best_response(y, solver)

            stall_count += 1

                y = project_simplex(y)
                stall_count = 0
            continue
        
        # Nesterov dual averaging update
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)

        # Evaluate both y and a
        score_a = score_normalized(a)
        score_y = score_normalized(y)
        
        if score_a < best_score:

            best_a = a.copy()
            stall_count = 0

def make_fewer_peaks_init(n, n_peaks=3, rng=None):

def make_cosine_init(n, k=1, rng=None):
    """Cosine-based initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 + 0.3 * np.cos(2 * np.pi * k * x + rng.uniform(0, 2*np.pi) if rng else 0)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def make_staggered_init(n, period, rng=None):
    """Staggered pattern initialization."""
    if rng is None:
        rng = np.random.default_rng(0)
    a = np.ones(n)
    for i in range(n):
        if (i % period) < period // 2:
            a[i] = 1.0 + 0.5 * rng.uniform(0, 1)
        else:
            a[i] = 1.0 - 0.3 * rng.uniform(0, 1)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np

from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv(a,a))."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                # Small random perturbation to escape
                rng_local = np.random.default_rng(step + 31337)
                a_pert = a + 0.003 * rng_local.standard_normal(n)
                a_pert = project_simplex(a_pert)
                score_pert = score_normalized(a_pert)
                if score_pert < best_score:
                    a = a_pert
                    best_score = score_pert
                    best_a = a.copy()
                    no_improve = 0
                elif no_improve > 150:
                    break

    return best_a, best_score


def make_cosine_init(n, k, phase=0.0):
    """Cosine-based initialization with given frequency and phase."""
    x = np.linspace(0, 1, n)
    a = 1.0 + 0.4 * np.cos(2 * np.pi * k * x + phase)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def make_staggered_init(n, period, rng):
    """Staggered pattern with random amplitudes per block."""
    a = np.ones(n)
    for block_start in range(0, n, period):
        block_end = min(block_start + period, n)
        amp = rng.uniform(0.7, 1.3)
        a[block_start:block_end] *= amp
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def make_multi_gauss_init(n, n_peaks, rng):
    """Multi-Gaussian bump initialization."""
    x = np.linspace(0, 1, n)
    centers = np.sort(rng.uniform(0.05, 0.95, n_peaks))
    widths = rng.uniform(0.02, 0.08, n_peaks)
    amps = rng.uniform(0.5, 1.5, n_peaks)
    a = np.zeros(n)
    for c, w, amp in zip(centers, widths, amps):
        a += amp * np.exp(-0.5 * ((x - c) / w) ** 2)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def make_ramp_init(n, rng):
    """Ramp-like initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 + rng.uniform(-0.5, 0.5) * (2 * x - 1) + 0.2 * np.sin(4 * np.pi * x)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def update_best(best_state, a, label):
    """Try to update the best state dict with a new candidate."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: LP Nesterov (primary), coordinate descent (polish),
    diverse inits, length exploration, alternating final polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_state = {'score': float('inf'), 'seq': None}
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except NameError:
        pass
    except Exception:
        pass

    if best_state['seq'] is None:
        n_init = 960
        best_state['seq'] = make_cosine_init(n_init, 1)
        best_state['score'] = evaluate_safe(best_state['seq'])
        print(f"Default init: n={n_init}, score={best_state['score']:.6f}")

    n_best = len(best_state['seq'])

    # === Phase 1: Primary LP refinement (longest allocation) ===
    print(f"\n=== Phase 1: LP Refinement n={n_best} (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 660.0)
    if t_alloc > 30:
        print(f"  Time budget: {t_alloc:.0f}s")
        a_lp, _ = lp_nesterov_refine(best_state['seq'], t_alloc, solver=solver, n_max_iters=300000)
        update_best(best_state, a_lp, "lp_p1")

    # === Phase 2: Coordinate descent polish ===
    print(f"\n=== Phase 2: Coord Descent (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 15:
        a_cd, _ = direct_coordinate_descent(best_state['seq'], t_alloc, n_max_steps=100000)
        update_best(best_state, a_cd, "cd_p2")

    # === Phase 3: Second LP round ===
    print(f"\n=== Phase 3: LP Round 2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 300.0)
    if t_alloc > 20:
        a_lp2, _ = lp_nesterov_refine(best_state['seq'], t_alloc, solver=solver, n_max_iters=200000)
        update_best(best_state, a_lp2, "lp_p3")

    # === Phase 4: Diverse initializations with brief LP ===
    print(f"\n=== Phase 4: Diverse Inits (best={best_state['score']:.6f}) ===")
    init_list = [
        ('cos1_p1', make_cosine_init(n_best, 1, np.pi)),
        ('cos2', make_cosine_init(n_best, 2, 0.5)),
        ('cos3', make_cosine_init(n_best, 3, 1.0)),
        ('stag3', make_staggered_init(n_best, 3, rng)),
        ('stag7', make_staggered_init(n_best, 7, rng)),
        ('gauss4', make_multi_gauss_init(n_best, 4, rng)),
        ('gauss6', make_multi_gauss_init(n_best, 6, rng)),
        ('ramp', make_ramp_init(n_best, rng)),
    ]

    for name, a_init in init_list:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_sub = min(rem * 0.12, 90.0)
        if t_sub < 25:
            break
        print(f"  Init: {name}, budget={t_sub:.0f}s, init_score={evaluate_safe(a_init):.4f}")
        a_sub, _ = lp_nesterov_refine(a_init, t_sub * 0.75, solver=solver, n_max_iters=30000)
        update_best(best_state, a_sub, f"lp_{name}")
        rem = deadline - time.monotonic()
        if rem > 15:
            a_sub2, _ = direct_coordinate_descent(a_sub, min(rem * 0.1, 15.0), n_max_steps=5000)
            update_best(best_state, a_sub2, f"cd_{name}")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length Exploration (best={best_state['score']:.6f}) ===")
    alt_lengths = []
    for delta in [-40, -20, -10, 10, 20, 40, 64]:
        alt_n = n_best + delta
        if 300 <= alt_n <= 2500:
            alt_lengths.append(alt_n)
    for fixed in [512, 640, 768, 1024, 1280, 1536]:
        if fixed != n_best:
            alt_lengths.append(fixed)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 45:
            break
        t_total = min(rem * 0.14, 80.0)
        if t_total < 20:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_state['seq'], alt_n)
        a_alt, _ = lp_nesterov_refine(init_alt, t_total, solver=solver, n_max_iters=20000)
        update_best(best_state, a_alt, f"lp_n{alt_n}")

    # === Phase 6: Final alternating polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        print(f"  Time remaining: {rem:.0f}s")
        while time.monotonic() < deadline - 5:
            t_rem = deadline - time.monotonic()
            if t_rem < 15:
                break
            t_lp = min(t_rem * 0.55, 50.0)
            a_p, _ = lp_nesterov_refine(best_state['seq'], t_lp, solver=solver, n_max_iters=40000)
            update_best(best_state, a_p, "polish_lp")
            t_rem = deadline - time.monotonic()
            if t_rem < 10:
                break
            t_cd = min(t_rem * 0.4, 25.0)
            if t_cd > 5:
                a_cd2, _ = direct_coordinate_descent(a_p, t_cd, n_max_steps=15000)
                update_best(best_state, a_cd2, "polish_cd")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])})")
    return [float(x) for x in best_state['seq']]
