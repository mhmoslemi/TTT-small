```python
def coord_descent_step(a, b, j, ratios):
    """Try changing a[j] by each ratio, return best a[j] value."""
    n = len(a)
    S = a.sum()
    v = a[j]
    if v < 1e-15:
        return v  # can't do much with a zero element
    
    # Sensitivity vector: s[k] = a[k-j] if 0 <= k-j < n
    s = np.zeros(2*n-1)
    k_min = j  # k-j >= 0 => k >= j
    k_max = j + n - 1  # k-j < n => k < j+n
    s[k_min:k_max+1] = a[0:n]
    
    # For each ratio, compute d and the new b
    best_score = np.inf
    best_d = 0.0
    for r in ratios:
        d = (r - 1.0) * v
        if d == 0.0:
            b_new = b
        else:
            b_new = b + 2.0 * d * s
            # Add d^2 at position 2j (if valid)
            pos = 2 * j
            if pos < 2 * n - 1:
                b_new = b_new.copy()
                b_new[pos] += d * d
        
        new_S = S + d
        if new_S < 1e-15:
            continue
        score = n * b_new.max() / (new_S * new_S)
        if score < best_score:
            best_score = score
            best_d = d
    
    return v + best_d
```

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def softmax_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def score_from_b(b, n, S):
    if S < 1e-15:
        return np.inf
    return 2.0 * n * b.max() / (S * S)


def coordinate_descent(a, time_limit, n_passes_target=20000, verbose=True):
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    S = 1.0

    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    ratios = np.array([0.3, 0.6, 0.8, 1.0, 1.25, 1.6, 2.5])

    pass_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1

        if pass_count % 50 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        improved_this_pass = False

        for j in order:
            v = a[j]
            if v < 1e-15:
                try_val = S / (10 * n)
                d = try_val
                s_vec = np.zeros(2 * n - 1)
                s_vec[j:j + n] = a
                b_new = b + 2.0 * d * s_vec
                pos = 2 * j
                if pos < 2 * n - 1:
                    b_new = b_new.copy()
                    b_new[pos] += d * d
                new_S = S + d
                sc = 2.0 * n * b_new.max() / (new_S * new_S)
                if sc < best_score - 1e-12:
                    a[j] = try_val
                    S = new_S
                    b = fast_convolve(a)
                    best_score = sc
                    best_a = a.copy()
                    improved_this_pass = True
                continue

            s_vec = np.zeros(2 * n - 1)
            s_vec[j:j + n] = a

            d_arr = (ratios - 1.0) * v
            b_new = b[None, :] + 2.0 * d_arr[:, None] * s_vec[None, :]
            pos = 2 * j
            if pos < 2 * n - 1:
                b_new = b_new.copy()
                b_new[:, pos] += d_arr ** 2

            S_new = S + d_arr
            valid = S_new > 1e-10
            if not valid.any():
                continue

            b_valid = b_new[valid]
            S_valid = S_new[valid]
            scores = 2.0 * n * b_valid.max(axis=1) / (S_valid ** 2)

            best_idx = scores.argmin()
            best_sc = scores[best_idx]

            if best_sc < best_score - 1e-12:
                d_best = d_arr[best_idx]
                a[j] = v + d_best
                S = S + d_best
                b = b_new[best_idx].copy()
                best_score = best_sc
                best_a = a.copy()
                improved_this_pass = True

        if not improved_this_pass:
            no_improve += 1
            if no_improve >= 20:
                # Reheat: add small noise
                noise = rng.normal(0, 0.005, n)
                a = np.maximum(a + noise, 0)
                S = a.sum()
                if S < 1e-15:
                    break
                a = a / S
                S = 1.0
                b = fast_convolve(a)
                best_score = 2.0 * n * b.max()
                no_improve = 0
            elif no_improve >= 10:
                # Try larger ratio range
                ratios = np.array([0.1, 0.4, 0.7, 1.0, 1.5, 2.5, 5.0])
        else:
            no_improve = 0
            if pass_count % 500 == 0 and verbose:
                elapsed = time.monotonic() - t0
                print(f"    CoordDesc pass={pass_count} t={elapsed:.0f}s score={best_score:.8f}")

    return best_a, best_score


def mirror_descent_fast(a0, time_limit, n_active_frac=0.05):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    a = a / S
    z = np.log(a).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.3
    no_improve = 0

    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b - n_active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active[:50]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        g_norm = np.abs(g).max()
        if g_norm < 1e-15:
            break

        gap = best_score - 2.0 * n / (2.0 * n - 1.0)
        if gap < 1e-10:
            break

        polyak_lr = gap / (2.0 * g_norm * best_score + 1e-15)
        alpha = min(lr, polyak_lr)
        alpha = max(alpha, 0.005)

        z = z - alpha * g
        a_new = softmax_z(z)
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            lr = min(lr * 1.03, 1.5)
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.6
                no_improve = 0
            lr = max(lr, 0.003)

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc t={elapsed:.0f}s score={best_score:.8f} lr={lr:.4f}")

    return best_a, best_score


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def nesterov_lp(a0, time_limit, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    a = a / S
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        improved = False
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            improved = True
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            improved = True

        if not improved:
            stall += 1
        else:
            stall = 0

        if stall >= 10:
            rng = np.random.default_rng(it * 13 + 7)
            noise = rng.random(n) * 0.01
            y = y + noise
            y = np.maximum(y, 0)
            y = y / y.sum()
            stall = 0

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP it={it+1} t={elapsed:.0f}s score={best_score:.8f}")

    return best_a, best_score


def simulated_annealing(a0, time_limit, T_start=0.01, T_end=0.0001):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    a = a / S
    z = np.log(a).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    elapsed_est = time_limit
    total_steps = int(elapsed_est * 5000)
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        frac = min(1.0, step / max(1, total_steps))
        T = T_start * (T_end / T_start) ** frac

        n_pert = max(1, int(n * 0.03))
        idx = rng.integers(0, n, n_pert)
        noise_scale = 0.05 + 0.1 * (1 - frac)
        z_trial = z.copy()
        z_trial[idx] += rng.normal(0, noise_scale, n_pert)

        a_trial = softmax_z(z_trial)
        b_trial = fast_convolve(a_trial)
        score_trial = 2.0 * n * b_trial.max()

        a_curr = softmax_z(z)
        score_curr = 2.0 * n * fast_convolve(a_curr).max()

        delta = score_trial - score_curr
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            z = z_trial
            a = a_trial
            if score_trial < best_score:
                best_score = score_trial
                best_a = a.copy()

        if step % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Anneal step={step} t={elapsed:.0f}s T={T:.5f} best={best_score:.8f}")

    return best_a, best_score


def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.5):
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-15)).copy()
    t0 = time.monotonic()

    def lse_obj(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp

    def lse_grad(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        a = softmax_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-14:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    n_stages = 6
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, 300 // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def obj_wrapper(z_in):
            return lse_obj(z_in, stage_T)

        def jac_wrapper(z_in):
            return lse_grad(z_in, stage_T)

        try:
            res = minimize(
                fun=obj_wrapper,
                x0=z,
                jac=jac_wrapper,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 100,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_z(z)


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 1.5
    elif choice == 2:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.15 * np.cos(5 * x) + 0.05 * np.cos(7 * x)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * (x - 0.5) ** 2)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = (1 - (2 * x - 1) ** 4) ** 0.5
    elif choice == 6:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.4 + 0.3 * np.cos(2 * x) + 0.2 * np.cos(4 * x) + 0.1 * np.cos(6 * x)
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * x * (1 - x)) * (1 + 0.3 * np.cos(6 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
