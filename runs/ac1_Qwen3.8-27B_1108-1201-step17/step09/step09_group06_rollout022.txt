import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def score_from_conv(b, n):
    return 2.0 * n * b.max()


def proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - s))[0][-1]
    theta = (cv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_subgradient_conv(a, b, n_active_frac=0.1):
    """Compute subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(2 * n - 1)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g_a = np.zeros(n)
    for j in range(n):
        g_a[j] = g[j] + (g[2*n-2-j] if 2*n-2-j < 2*n-1 else 0)
    # Actually for conv(a,a)[k] = sum_j a[j]*a[k-j], d/d(a[j]) = a[k-j] + a[j-k] (with bounds)
    # Let's compute directly:
    g_a = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            g_a[j] += a[k - j]
    g_a /= len(active)
    return g_a, max_b


def nesterov_subgrad(a0, time_limit, n_active_frac=0.08, max_iters=200000):
    """Nesterov-accelerated subgradient descent on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = score_from_conv(b, n)
    t0 = time.monotonic()
    stall = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute subgradient at y
        b_y = fast_convolve(y)
        g, _ = compute_subgradient_conv(y, b_y, n_active_frac)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        alpha = 2.0 / (it + 2)

        # Gradient step in y
        step = alpha / (g_norm + 1e-15)
        y_new = y - step * g
        y_new = proj_simplex(y_new)

        # Nesterov update
        a_new = (1 - alpha) * a + alpha * y_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y_new
        else:
            a_new = a_new / sa

        a = a_new
        y = y_new

        # Evaluate
        b_a = fast_convolve(a)
        score_a = score_from_conv(b_a, n)
        b_y = fast_convolve(y)
        score_y = score_from_conv(b_y, n)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        if (it + 1) % 2000 == 0:
            if stall >= 15:
                rng = np.random.default_rng(it * 31 + 7)
                noise = rng.random(n) * 0.01
                y = y + noise
                y = proj_simplex(y)
                a = a + noise * 0.5
                a = proj_simplex(a)
                stall = 0
            stall += 1
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Solve: min M s.t. conv(a_fixed, b)_k <= M for all k, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def nesterov_lp_refine(a0, time_limit, solver=None, max_iters=100000):
    """Nesterov-accelerated LP best response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = score_from_conv(fast_convolve(a), n)
    t0 = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = score_from_conv(fast_convolve(a), n)
        score_y = score_from_conv(fast_convolve(y), n)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        if (it + 1) % 500 == 0:
            stall += 1
            if stall >= 10:
                rng = np.random.default_rng(it * 13 + 3)
                noise = rng.random(n) * 0.015
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0
            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.1, maxiter=12000):
    """L-BFGS on smooth log-sum-exp approximation."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for stage_idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_in, T_val=stage_T):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_tmp = softmax_from_z(z_in)
            b = fast_convolve(a_tmp)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp

        def jac(z_in, T_val=stage_T):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a_tmp = softmax_from_z(z_in)
            b = fast_convolve(a_tmp)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_tmp[k - idx]
            S = np.dot(grad_a, a_tmp)
            return a_tmp * (grad_a - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        if (stage_idx + 1) % 5 == 0:
            a_curr = softmax_from_z(z)
            sc = score_from_conv(fast_convolve(a_curr), n)
            print(f"    L-BFGS stage {stage_idx+1}/{n_stages} (T={stage_T:.2f}): score={sc:.6f}")

    return softmax_from_z(z)


def deflation_search(a0, time_limit, n_steps=50000):
    """Targeted local search: identify convolution peaks and adjust contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = score_from_conv(b, n)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        threshold = max_b * 0.95
        peaks = np.where(b >= threshold)[0]

        # For each peak, find top contributors and try to reduce them
        improved = False
        for pk in peaks[:5]:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            # Contribution of element j to b[pk] is a[j] * a[pk-j]
            contribs = a[js:je+1] * a[pk - (je:js-1):-1]
            top_idx = np.argsort(contribs)[::-1][:3]
            top_idx_abs = js + top_idx

            # Find low-contributing targets to move mass to
            all_contrib = np.zeros(n)
            for j in range(n):
                # How much does element j contribute to total max?
                all_contrib[j] = 0
            # Just pick random low-mass positions
            low_positions = np.argsort(a)[:5]

            for si in top_idx_abs:
                if a[si] < 1e-12:
                    continue
                for di in low_positions:
                    if di == si:
                        continue
                    for frac in [0.03, 0.07, 0.12]:
                        amount = a[si] * frac
                        a_trial = a.copy()
                        a_trial[si] -= amount
                        a_trial[di] += amount
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        sc = score_from_conv(fast_convolve(a_trial), n)
                        if sc < best_score - 1e-12:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
                if improved:
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            # Random perturbation
            if no_improve > 20:
                n_perturb = rng.integers(2, min(6, n // 10) + 1)
                idx = rng.choice(n, size=n_perturb, replace=False)
                amounts = rng.random(n_perturb) * 0.05
                a_trial = a.copy()
                for i in idx:
                    a_trial[i] += amounts[list(idx).index(i)] if i in idx else 0
                # Simpler: add random noise
                noise = rng.random(n) * 0.02 - 0.01
                a_trial = a + noise
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    sc = score_from_conv(fast_convolve(a_trial), n)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        no_improve = 0

            if no_improve > 200:
                break

    return best_a, best_score


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.sin(np.pi * x)
    elif choice == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        a = np.exp(-2 * (x - 0.5) ** 2 / (0.15 * (1 - 0.15)))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(8 * np.pi * x)
    elif choice == 4:
        # Triangular wave
        a = 1 - 4 * np.abs(x - 0.5) + 0.3 * np.cos(6 * np.pi * x)
    elif choice == 5:
        # Polynomial
        a = 1.0 - 0.5 * (2 * x - 1) ** 4
    elif choice == 6:
        # Bump + oscillation
        a = np.exp(-4 * (x - 0.5) ** 2) * (1 + 0.3 * np.cos(10 * np.pi * x))
    else:
        a = np.ones(n) * (0.8 + 0.2 * rng.random())
    a = np.maximum(a, 0.01)
    s = a.sum()
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_best = len(best_seq) if best_seq is not None else 960

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        n_best = n_init
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    rem = deadline - time.monotonic()
    print(f"\nTotal time budget: {rem:.0f}s, starting n={n_best}, score={best_score:.6f}")

    # Phase 1: Quick Nesterov subgradient (fast, no LP)
    print(f"\n=== Phase 1: Nesterov Subgrad ({best_score:.6f}) ===")
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = nesterov_subgrad(best_seq, t_alloc, n_active_frac=0.06)
        try_update(a_md, "subgrad_1")

    # Phase 2: LP Nesterov (more accurate)
    print(f"\n=== Phase 2: LP Nesterov ({best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, max_iters=150000)
        try_update(a_lp, "lp_1")

    # Phase 3: L-BFGS smoothing
    print(f"\n=== Phase 3: L-BFGS ({best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=80.0, T_end=0.05, maxiter=15000)
        try_update(a_lb, "lbgfs_1")

    # Phase 4: Deflation local search
    print(f"\n=== Phase 4: Deflation Search ({best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_def, _ = deflation_search(best_seq, t_alloc, n_steps=80000)
        try_update(a_def, "deflation_1")

    # Phase 5: Second round of LP (refined)
    print(f"\n=== Phase 5: LP Nesterov r2 ({best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 15:
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, max_iters=100000)
        try_update(a_lp2, "lp_2")

    # Phase 6: Alternative lengths
    print(f"\n=== Phase 6: Alt Lengths ({best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = []
    for mult in [1.2, 1.5, 0.8, 2.0, 0.65, 1.3]:
        alt_n = int(n_best * mult)
        if 384 <= alt_n <= 4000 and alt_n != n_best:
            # Prefer lengths that are powers of 2 or near them for FFT efficiency
            alt_lengths.append(alt_n)
    # Deduplicate and sort by closeness to n_best
    alt_lengths = sorted(set(alt_lengths), key=lambda x: abs(x - n_best))[:3]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_total = min(rem * 0.3, 180.0)
        if t_total < 30:
            break
        print(f"  Trying n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        cur = init_alt
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()
            print(f"  Interpolation improved: {sc_alt:.6f}")

        # Quick LP
        t_lp = t_total * 0.6
        a_alt, _ = nesterov_lp_refine(cur, t_lp, solver=solver, max_iters=50000)
        try_update(a_alt, f"lp_n{alt_n}")

        # Quick deflation
        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None and len(best_seq) == alt_n:
            t_def = min(rem * 0.2, 40.0)
            a_def2, _ = deflation_search(best_seq, t_def, n_steps=30000)
            try_update(a_def2, f"def_n{alt_n}")

    # Phase 7: Final polish
    print(f"\n=== Phase 7: Final Polish ({best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.35, 80.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=15.0, T_end=0.05, maxiter=10000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 30:
        t_ls = min(rem * 0.3, 50.0)
        a_ls2, _ = deflation_search(best_seq, t_ls, n_steps=30000)
        try_update(a_ls2, "final_deflation")

    rem = deadline - time.monotonic()
    if rem > 25:
        t_lp_fin = min(rem * 0.4, 40.0)
        a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, solver=solver, max_iters=50000)
        try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
