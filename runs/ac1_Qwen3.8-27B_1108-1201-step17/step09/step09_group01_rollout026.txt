"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact 1D coordinate descent via golden section search on convex subproblems
2. Vectorized convolution updates using FFT
3. Multi-phase optimization: explore lengths, intensive CD, perturbation-restart
4. Structured perturbations (peak-targeted, random restructure) for global exploration
5. Adaptive time allocation with early stopping on convergence
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def score_normalized(a):
    """Compute 2*n*M/s^2 for sequence a."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    M = fast_convolve(a / s).max()
    return float(2.0 * n * M)


def _f_i(delta, a, conv, k_start, quad_idx, n):
    """Compute max of convolution after hypothetical change of a[i] by delta.
    Precomputed: k_start = i, quad_idx = i, slice length = n."""
    vals = conv[k_start:k_start + n] + 2.0 * delta * a
    vals[quad_idx] += delta * delta
    return vals.max()


def _apply_update(conv, a, i, delta, n):
    """Apply coordinate update: a[i] += delta, update conv in-place. Return new M."""
    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)
    conv[k_start:k_end + 1] += 2.0 * delta * a[0:(k_end - k_start + 1)]
    quad_idx = 2 * i - k_start
    if 0 <= quad_idx < (k_end - k_start + 1):
        conv[quad_idx] += delta * delta
    a[i] += delta
    return float(conv.max())


def exact_cd(a0, time_limit, verbose=False, label="", max_sweeps=1000,
             gs_iters=35, restart_after=50, rng=None):
    """
    Exact coordinate descent with golden section search.
    For each coordinate i, minimizes the (convex) 1D function:
        f(delta) = max_k (conv[k] + 2*delta*a[k-i] + [k==2i]*delta^2)
    subject to a[i] + delta >= 0.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0
    GR = (np.sqrt(5.0) - 1.0) / 2.0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and sweep < max_sweeps:
        sweep += 1
        n_improvements = 0

        for i in range(n):
            ai = a[i]
            k_start = i
            quad_idx = i  # 2i - k_start = 2i - i = i

            if ai < 1e-16:
                # Coordinate is zero; try a small positive value
                delta_try = max(1e-6, M * 0.005)
                f_try = _f_i(delta_try, a, conv, k_start, quad_idx, n)
                if f_try < M - 1e-14:
                    M = _apply_update(conv, a, i, delta_try, n)
                    n_improvements += 1
                continue

            # Bracket the minimum
            lo = -ai
            hi = ai * 1.5

            # Additionally check if a larger step might help
            f_hi = _f_i(hi, a, conv, k_start, quad_idx, n)
            if f_hi < M - 1e-12:
                hi = ai * 3.0

            # Golden section search
            a_gs = lo + (1.0 - GR) * (hi - lo)
            b_gs = lo + GR * (hi - lo)
            fa = _f_i(a_gs, a, conv, k_start, quad_idx, n)
            fb = _f_i(b_gs, a, conv, k_start, quad_idx, n)

            for _ in range(gs_iters):
                if fa < fb:
                    hi = b_gs
                    b_gs = a_gs
                    fb = fa
                    a_gs = lo + (1.0 - GR) * (hi - lo)
                    fa = _f_i(a_gs, a, conv, k_start, quad_idx, n)
                else:
                    lo = a_gs
                    a_gs = b_gs
                    fa = fb
                    b_gs = lo + GR * (hi - lo)
                    fb = _f_i(b_gs, a, conv, k_start, quad_idx, n)

            best_delta = (lo + hi) * 0.5
            best_M_new = _f_i(best_delta, a, conv, k_start, quad_idx, n)

            if best_M_new < M - 1e-14:
                M = _apply_update(conv, a, i, best_delta, n)
                n_improvements += 1

        current_score = 2.0 * n * M
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD[{label}] sweep={sweep} ({elapsed:.0f}s) "
                  f"score={best_score:.8f} M={M:.12f} imp={n_improvements}")

        if n_improvements == 0:
            break

        # Restart mechanism: perturb if stuck
        if no_improve >= restart_after and rng is not None:
            # Small random perturbation
            noise_scale = M * 0.001
            noise = rng.standard_normal(n) * noise_scale
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp > 1e-10:
                a = a_pert / sp
                conv = fast_convolve(a)
                M = float(conv.max())
                no_improve = 0

    return best_a, best_score, sweep


def peak_targeted_perturb(a0, rng, strength=0.05):
    """Perturb by reducing mass at peak-contributing positions."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    if M < 1e-15:
        return a

    peak_threshold = M * 0.90
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            product = a[i] * a[j]
            pressure[i] += product
            if j != i:
                pressure[j] += product * 0.5

    p_max = pressure.max()
    if p_max > 1e-30:
        pressure = pressure / p_max

    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.2 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def random_restructure(a0, rng, frac=0.2):
    """Randomly zero out fraction and redistribute."""
    n = len(a0)
    a = a0.copy()
    keep = rng.random(n) > frac
    n_keep = keep.sum()
    if n_keep < max(10, n // 10):
        keep = np.ones(n, dtype=bool)
        n_keep = n
    a[~keep] = 0
    a[keep] *= (1.0 + rng.standard_normal(n_keep) * 0.3)
    a = np.maximum(a, 0)
    zero_mask = (a < 1e-15)
    if zero_mask.any():
        n_add = max(1, int(zero_mask.sum() * 0.15))
        add_idx = np.where(zero_mask)[0]
        add_idx = rng.choice(add_idx, size=min(n_add, len(add_idx)), replace=False)
        a[add_idx] = rng.random(n_add) * a.max() * 0.3
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def smooth_gradient_step(a0, time_limit, lr=0.01, beta=50.0, verbose=False, label=""):
    """
    Subgradient method with smooth max (logsumexp) approximation.
    Minimizes logsumexp(beta * conv) as surrogate for max(conv).
    Uses mirror descent (multiplicative updates) on the simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    momentum = 0.0
    v = np.zeros(n)  # momentum buffer

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Logsumexp-based gradient
        scaled = beta * (conv - M)
        scaled = np.clip(scaled, -50, 0)
        w = np.exp(scaled)
        Z = w.sum()
        if Z < 1e-30:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
            Z = 1.0
        w = w / Z

        # Gradient of logsumexp w.r.t. a
        grad = np.zeros(n)
        active_k = np.where(w > 1e-8)[0]
        for k in active_k:
            wk = w[k]
            if wk < 1e-10:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * wk * a[k - j]

        # Momentum
        momentum = 0.9 * momentum + 0.1
        v = momentum * v + grad

        # Mirror descent: multiplicative update
        log_a = np.log(a + 1e-30)
        log_a -= lr * v
        a_new = np.exp(log_a - log_a.max())
        a = simplex_project(a_new)

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        # Adaptive learning rate
        if it % 100 == 0:
            lr *= 0.98
            beta *= 1.05
        if lr < 1e-10:
            break

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGrad[{label}] it={it} ({elapsed:.0f}s) score={best_score:.8f}")

    return best_a, best_score


def init_symmetric(n, rng):
    """Symmetric initialization with cosine superposition."""
    half = (n + 1) // 2
    t = np.linspace(0, 1, half)
    n_comp = int(rng.integers(3, 8))
    profile = np.ones(half) * 0.5
    for j in range(n_comp):
        freq = rng.uniform(1.0, 5.0)
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0.01)
    if n % 2 == 0:
        a = np.concatenate([profile[:n//2], profile[:n//2][::-1]])
    else:
        a = np.concatenate([profile, profile[:-1][::-1]])
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_bimodal(n, rng):
    """Bimodal Gaussian initialization."""
    c1 = n * rng.uniform(0.25, 0.4)
    c2 = n * rng.uniform(0.6, 0.75)
    w1 = n * rng.uniform(0.05, 0.12)
    w2 = n * rng.uniform(0.05, 0.12)
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - c2) / w2) ** 2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_fourier(n, rng):
    """Initialize using Fourier features to create flat-ish convolution."""
    # Try to create a sequence whose power spectrum is flat
    # (this would give a delta-like convolution, but we need non-negative values)
    # Instead, use a random superposition of complex exponentials, take real part + shift
    n_freq = int(rng.integers(3, 10))
    a = np.zeros(n)
    for _ in range(n_freq):
        freq = rng.uniform(1, n / 3)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 1.0)
        t = np.arange(n)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a = a - a.min() + 0.1
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
