"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated mirror descent
with multi-peak subgradient and peak-aware coordinate descent.
Key innovations: larger sequence length, adaptive step sizes, multi-peak
softmax-weighted subgradient, and targeted coordinate updates.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def get_subgradient(conv, a, top_k=50, temperature=0.01):
    """Compute weighted subgradient using top-k peaks."""
    n = len(a)
    k_act = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks]
    pmax = pv.max()
    w = np.exp((pv - pmax) / (temperature * max(pmax, 1e-10)))
    w = w / w.sum()
    
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # For k != 2i, contribution is a[k-i]
        # For k == 2i, contribution is 2*a[i] (but we approximate as a[k-i] = a[i])
        sg[i_start:i_end+1] += w[idx] * a[k - i_end:k - i_start + 1][::-1]
    
    return sg, peaks

def mirror_descent(a0, time_limit, top_k=100, lr_init=0.1, verbose=True):
    """Accelerated mirror descent with multi-peak subgradient."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    
    # Nesterov momentum
    y = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            lr *= 0.7
            no_improve = 0
            if lr < 1e-10:
                break
        
        # Compute subgradient at y (momentum point)
        y = 0.5 * (a + y)  # Simple momentum
        conv_y = fast_convolve(y)
        sg, peaks = get_subgradient(conv_y, y, top_k=top_k, temperature=0.01)
        
        # Mirror descent step
        a_new = a * np.exp(-lr * sg / max(M, 1e-10))
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn
        
        a = a_new
        conv = fast_convolve(a)
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto probability simplex {x >= 0, sum(x)=1} in O(n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def compute_subgradient(conv, a, top_k=100, temperature=0.01):
    """Compute weighted subgradient using top-k peaks with softmax weighting."""
    n = len(a)
    k_act = min(top_k, len(conv))
    if k_act == len(conv):
        peaks = np.arange(len(conv))
        pv = conv.copy()
    else:
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
    pmax = float(pv.max())
    if pmax < 1e-30:
        return np.zeros(n), peaks
    t = temperature * pmax
    w = np.exp((pv - pmax) / t)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(peaks)) / len(peaks)
    else:
        w = w / ws
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-15:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        seg = a[k - i_end:k - i_start + 1][::-1]
        sg[i_start:i_end + 1] += w[idx] * seg
    return sg, peaks


def mirror_descent_main(a0, time_limit, top_k=150, lr_init=0.08, verbose=True):
    """
    Mirror descent (multiplicative weights) with multi-peak subgradient.
    Uses adaptive step size and periodic restarts.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    avg_a = a.copy().astype(np.float64)

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = float(conv.max())
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 300:
            lr *= 0.75
            no_improve = 0
            if lr < 1e-10:
                break

        # Compute subgradient
        sg, _ = compute_subgradient(conv, a, top_k=top_k, temperature=0.008)

        # Mirror descent step: a_new[i] = a[i] * exp(-lr * sg[i] / M)
        scaled_lr = lr / max(M, 1e-15)
        a_new = a * np.exp(-scaled_lr * sg)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn

        # Running average for convergence
        avg_a = (it * avg_a + a_new) / (it + 1)

        a = a_new
        conv = fast_convolve(a)

        # Periodically check running average
        if it % 50 == 0:
            avg_score = 2.0 * n * float(fast_convolve(avg_a).max())
            if avg_score < best_score:
                best_score = avg_score
                best_a = avg_a.copy()

        # Periodic restart: blend with best
        if it % 500 == 0:
            a = 0.7 * a + 0.3 * best_a
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.0f}s): best={best_score:.7f} lr={lr:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True, label=""):
    """
    Peak-aware coordinate descent.
    For each coordinate, finds the optimal delta that minimizes the max
    over the affected range of the convolution.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    # Precompute for speed: we'll iterate over coordinates cyclically
    # Use a shuffled order each sweep for better convergence
    order = np.arange(n)

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        np.random.shuffle(order)

        for i in order:
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = i + n - 1
            k_end = min(k_end, 2 * n - 2)

            # Unaffected max (outside [k_start, k_end])
            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            # Affected values: conv[k] for k in [k_start, k_end]
            # conv[k] changes by 2*delta*a[k-i] (plus delta^2 at k=2i)
            a_shifted = a[k_start - i:k_end - i + 1]  # a[k-i] for k in range

            # Try several candidate deltas
            base = a[i]
            candidates = [
                -0.9 * base, -0.7 * base, -0.5 * base, -0.3 * base,
                -0.15 * base, -0.05 * base, 0.05 * base, 0.15 * base,
                0.3 * base, 0.5 * base, 0.7 * base, 0.9 * base
            ]

            best_delta = 0.0
            best_new_M = M

            for delta in candidates:
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                # New affected values
                new_affected = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted

                # Fix quadratic term at k = 2i
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                # Apply update
                old_val = a[i]
                a[i] = max(a[i] + best_delta, 0.0)

                # Update convolution incrementally
                delta_actual = a[i] - old_val
                changes = 2.0 * delta_actual * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * delta_actual * old_val + delta_actual * delta_actual
                conv[k_start:k_end + 1] += changes

                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 30:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score

def skewed_gaussian_init(n, rng):

    return a

def resize_sequence(a, new_n):
    """Resize sequence via linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_t = np.linspace(0, 1, old_n)
    new_t = np.linspace(0, 1, new_n)
    a_resized = np.interp(new_t, old_t, a)
    a_resized = np.maximum(a_resized, 0)
    s = a_resized.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_resized / s

        t_cd2 = min(remaining * 0.5, 150.0)
        a_cd2, s_cd2 = coordinate_descent(best_seq, t_cd2, verbose=True, label="final_cd")
        best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "final_cd2")

    # === Final fine polish ===
    remaining = deadline - time.monotonic() - 10
    if remaining > 20:
        t_fp2 = min(remaining * 0.7, 100.0)
        a_fp2, s_fp2 = fine_polish(best_seq, t_fp2, verbose=True)
        best_seq, best_score = check_and_update(a_fp2, best_seq, best_score, "final_fine")

    # === Final verification ===
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in np.maximum(best_seq, 0.0)]
