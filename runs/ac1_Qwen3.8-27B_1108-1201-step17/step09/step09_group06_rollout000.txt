import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score_normalized(a):
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a = a / s
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_objective_and_grad(z, T):
    a = softmax_fn(z)
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    s = e.sum()
    if s < 1e-300:
        return 0.0, np.zeros(n)
    w = e / s
    logse = m + np.log(s)
    obj = (2.0 * n / T) * logse

    grad_a = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def adam_anneal(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=10,
                lr_init=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = np.log(a)
    z -= z.mean()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 5000
    lr = lr_init

    no_improve_count = 0

    for si, T_val in enumerate(temps):
        for it in range(iters_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            t += 1
            try:
                obj, grad = lse_objective_and_grad(z, T_val)
            except Exception:
                continue

            if not np.all(np.isfinite(grad)):
                grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if it % 500 == 0:
                a = softmax_fn(z)
                score = compute_score_normalized(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1
                if no_improve_count >= 4:
                    lr *= 0.8
                    no_improve_count = 0

        a = softmax_fn(z)
        score = compute_score_normalized(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if (si + 1) % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AdamAnneal stage={si+1}/{n_stages} T={T_val:.1f} "
                  f"score={best_score:.6f} t={elapsed:.0f}s")

    return best_a, best_score


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver_name = None


def get_lp_solver():
    global _lp_solver_name
    if _lp_solver_name is not None:
        return _lp_solver_name
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_name = solver_name
                return _lp_solver_name
        except Exception:
            continue
    _lp_solver_name = cp.ECOS
    return _lp_solver_name


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def ema_lp_alternation(a0, time_limit, solver=None, n_max_iters=50000,
                       ema_alpha_init=0.3, ema_alpha_min=0.05):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b_ema = a.copy()
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (n_max_iters * 0.7))
        alpha = ema_alpha_init - (ema_alpha_init - ema_alpha_min) * progress

        b_resp, M_val = solve_lp_best_response(a, solver)
        if b_resp is None:
            continue

        b_ema = (1 - alpha) * b_ema + alpha * b_resp
        b_ema = np.maximum(b_ema, 0)
        sb = b_ema.sum()
        if sb < 1e-15:
            break
        b_ema = b_ema / sb

        a = b_ema.copy()
        score = compute_score_normalized(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            if it % 200 == 0 and best_score >= prev_best:
                stall += 1
            prev_best = best_score

        if stall >= 10:
            rng = np.random.default_rng(it * 31 + 7)
            noise = rng.random(n) * 0.05
            b_ema = b_ema + noise
            b_ema = np.maximum(b_ema, 0)
            b_ema = b_ema / b_ema.sum()
            a = b_ema.copy()
            stall = 0
            prev_best = best_score

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    EMALP iter={it+1} ({elapsed:.0f}s): "
                  f"score={best_score:.6f} alpha={alpha:.3f}")

    return best_a, best_score


def lbfgs_polish(a0, time_limit, T_vals=None, maxiter_per_stage=2000):
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z -= z.mean()
    t0 = time.monotonic()

    if T_vals is None:
        T_vals = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5]

    for T_val in T_vals:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_fn(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            m = Tb.max()
            e = np.exp(Tb - m)
            return (2.0 * n / T_val) * (m + np.log(e.sum()))

        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_fn(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            m = Tb.max()
            e = np.exp(Tb - m)
            s = e.sum()
            if s < 1e-300:
                return np.zeros_like(z_in)
            w = e / s
            grad_a = np.zeros(n)
            active = np.where(w > 1e-12)[0]
            for k in active:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': maxiter_per_stage * 100,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_fn(z)


def kkt_peak_shave(a0, time_limit, n_iters=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b - 0.02 * max_b
        active = np.where(b >= threshold)[0]

        if len(active) == 0:
            active = np.array([np.argmax(b)])

        residual = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            residual[idx] += a[k - idx]

        support = np.where(a > 1e-10)[0]
        if len(support) < 2:
            break

        r_support = residual[support]
        r_min = r_support.min()
        r_max = r_support.max()

        if r_max - r_min < 1e-10:
            if it > 1000:
                break
            move_to = rng.integers(len(support))
            move_from = rng.integers(len(support))
            if move_to == move_from:
                continue
        else:
            move_from = support[np.argmax(r_support)]
            CANDIDATES = support[np.where(r_support < (r_min + r_max) / 2)[0]]
            if len(CANDIDATES) == 0:
                CANDIDATES = support[np.argmin(r_support)]
                if np.ndim(CANDIDATES) == 0:
                    CANDIDATES = np.array([CANDIDATES])
            move_to = CANDIDATES[rng.integers(len(CANDIDATES))]

        if a[move_from] < 1e-12:
            continue

        fracs = [0.01, 0.02, 0.05, 0.1, 0.2]
        improved = False
        for frac in fracs:
            amount = a[move_from] * frac
            a_trial = a.copy()
            a_trial[move_from] -= amount
            a_trial[move_to] += amount
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break

        if not improved and it > 500:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j or a[i] < 1e-12:
                continue
            amount = a[i] * 0.03
            a_trial = a.copy()
            a_trial[i] -= amount
            a_trial[j] += amount
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKTPeak it={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def basin_hopping(a0, time_limit, n_hops=10, hop_strength=0.1,
                  local_time_frac=0.7, solver=None):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    local_budget = min(30.0, (time_limit - 10) / max(n_hops, 1))

    for hop in range(n_hops):
        if time.monotonic() - t0 > time_limit:
            break

        noise_scale = hop_strength * (1.0 + 0.5 * rng.random())
        noise = rng.standard_normal(n) * noise_scale
        a_perturbed = a + noise
        a_perturbed = np.maximum(a_perturbed, 0)
        sp = a_perturbed.sum()
        if sp < 1e-15:
            continue
        a_perturbed = a_perturbed / sp

        score_pert = compute_score_normalized(a_perturbed)

        a_local, _ = kkt_peak_shave(a_perturbed, local_budget, n_iters=5000)
        score_local = compute_score_normalized(a_local)

        if score_local < best_score:
            best_score = score_local
            best_a = a_local.copy()
            a = a_local.copy()
            print(f"    BasinHop {hop}: improved to {best_score:.6f}")
        else:
            a = a_local.copy()

    return best_a, best_score


def structured_init(n, rng, style=0):
    if style == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 4
    elif style == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif style == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif style == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif style == 4:
        k = np.linspace(0, 1, n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * k) + 0.1 * np.cos(2 * np.pi * 7 * k)
    elif style == 5:
        a = np.ones(n) / n
    elif style == 6:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2 + 0.1
    elif style == 7:
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * x) * (1 + 0.3 * np.sin(20 * np.pi * x))
    else:
        a = rng.random(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if not a_list:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 960

    if parent is None:
        n_base = 960
        best_seq = structured_init(n_base, rng, style=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    print(f"Starting: n={n_base}, best={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # Phase 1: Adam annealing (main workhorse)
    print(f"\n=== Phase 1: Adam Anneal (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_adam, _ = adam_anneal(best_seq, t_alloc, T_start=80.0, T_end=0.3,
                                n_stages=8, lr_init=0.08)
        try_update(a_adam, "adam_1")

    # Phase 2: EMA LP alternation
    print(f"\n=== Phase 2: EMA LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ema, _ = ema_lp_alternation(best_seq, t_alloc, solver=solver,
                                      n_max_iters=50000,
                                      ema_alpha_init=0.25, ema_alpha_min=0.03)
        try_update(a_ema, "ema_lp_1")

    # Phase 3: KKT Peak Shaving
    print(f"\n=== Phase 3: KKT Peak Shave (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_kkt, _ = kkt_peak_shave(best_seq, t_alloc, n_iters=80000)
        try_update(a_kkt, "kkt_shave_1")

    # Phase 4: L-BFGS Polish
    print(f"\n=== Phase 4: L-BFGS Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = lbfgs_polish(best_seq, t_alloc,
                            T_vals=[30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5],
                            maxiter_per_stage=1500)
        try_update(a_lb, "lbfgs_1")

    # Phase 5: Basin Hopping
    print(f"\n=== Phase 5: Basin Hopping (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_hop, _ = basin_hopping(best_seq, t_alloc, n_hops=6,
                                 hop_strength=0.08, solver=solver)
        try_update(a_hop, "basin_hop_1")

    # Phase 6: Second round of refinement on best
    print(f"\n=== Phase 6: Refinement Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t1 = min(rem * 0.3, 150.0)
        if t1 > 20:
            a_adam2, _ = adam_anneal(best_seq, t1, T_start=30.0, T_end=0.2,
                                     n_stages=6, lr_init=0.03)
            try_update(a_adam2, "adam_2")

        rem = deadline - time.monotonic()
        if rem > 30:
            t2 = min(rem * 0.4, 120.0)
            if t2 > 20:
                a_ema2, _ = ema_lp_alternation(best_seq, t2, solver=solver,
                                               n_max_iters=30000,
                                               ema_alpha_init=0.15, ema_alpha_min=0.02)
                try_update(a_ema2, "ema_lp_2")

        rem = deadline - time.monotonic()
        if rem > 20:
            t3 = min(rem * 0.3, 60.0)
            if t3 > 10:
                a_kkt2, _ = kkt_peak_shave(best_seq, t3, n_iters=50000)
                try_update(a_kkt2, "kkt_shave_2")

    # Phase 7: Alternative lengths
    print(f"\n=== Phase 7: Alt Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = []
    for mult in [1.25, 1.5, 0.75, 2.0, 1.1]:
        alt_n = int(n_base * mult)
        if 400 <= alt_n <= 2500 and alt_n != n_base:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:2]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_total = min(rem * 0.4, 180.0)
        if t_total < 30:
            break
        print(f"  Trying n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)

        t_sub1 = t_total * 0.5
        a_alt, _ = ema_lp_alternation(init_alt, t_sub1, solver=solver,
                                      n_max_iters=20000,
                                      ema_alpha_init=0.3, ema_alpha_min=0.05)
        try_update(a_alt, f"alt_lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None:
            t_sub2 = min(rem * 0.3, 60.0)
            if t_sub2 > 10:
                a_alt2, _ = kkt_peak_shave(best_seq, t_sub2, n_iters=20000)
                try_update(a_alt2, f"alt_kkt_n{alt_n}")

    # Phase 8: Final polish
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 60.0)
        a_fin = lbfgs_polish(best_seq, t_fin,
                             T_vals=[10.0, 5.0, 2.0, 1.0, 0.5],
                             maxiter_per_stage=1000)
        try_update(a_fin, "final_lbfgs")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_ls = min(rem * 0.4, 30.0)
        a_ls, _ = kkt_peak_shave(best_seq, t_ls, n_iters=20000)
        try_update(a_ls, "final_kkt")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
