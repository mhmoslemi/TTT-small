Actually, for non-convex problems, a better approach is **mirror descent** or **exponential weighting**:
- Maintain weights w_j (in log domain: z_j = log(w_j))
- At each step, find the peak k*, compute the "cost" c_j = 2*a_{k*-j} (the subgradient)
- Update: z_j -= eta * c_j (in log domain)
- Renormalize: a_j = exp(z_j) / sum(exp(z))

This is the **multiplicative weights** or **exponential weighting** algorithm. It's equivalent to mirror descent with entropy regularization.

The key advantage is that it naturally maintains non-negativity and we only need to handle the normalization.

```python

    a = a0.copy()
    z = np.log(a + 1e-300)
    best_a = a.copy()
    best_score = 2*len(a)*fast_convolve(a).max()
    
    for t in range(n_steps):
        # Find peak
        b = fast_convolve(a)
        k_star = np.argmax(b)
        
        # Compute subgradient
        g = np.zeros(len(a))
        js = max(0, k_star - len(a) + 1)
        je = min(len(a) - 1, k_star)
        for j in range(js, je+1):
            if j == k_star - j:
                g[j] = a[j]
            else:
                g[j] = 2 * a[k_star - j]
        
        # Mirror descent step in log domain
        eta = eta0 / (1 + 0.01 * t)
        z = z - eta * g
        
        # Renormalize
        z = z - z.max()
        a_new = np.exp(z)
        a_new = a_new / a_new.sum()
        
        # Check improvement
        score = 2*len(a_new)*fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score
```

```python
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
```

```python
    return a / s


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_smooth_gradient(a, w, fft_size=None):
    """
    Compute gradient of (1/T)*log(sum_k exp(T*(a*a)_k)) w.r.t. a.
    g_j = 2*(a*cross_corr*w)_j - w[2j]*a_j (correction)
    Uses FFT for the cross-correlation part.
    """
    n = len(a)
    if fft_size is None:
        total_len = n + (2 * n - 1) - 1
        fft_size = 1 << (total_len - 1).bit_length()
    flip_a = a[::-1]
    FA = rfft(flip_a, fft_size)
    FW = rfft(w, fft_size)
    c = irfft(FA * FW, fft_size)
    g = 2.0 * c[n - 1:2 * n - 1]
    # Correction for diagonal terms (k=2j)
    max_2j = min(2 * n - 1, 2 * n - 2)
    for j in range(n):
        idx = 2 * j
        if idx < len(w):
            g[j] -= w[idx] * a[j]
    return g


def mirror_descent_refine(a0, time_limit, T_start=200.0, T_end=0.005, n_stages=30, eta_scale=1.0):
    """
    Mirror descent with log-sum-exp smoothing and geometric temperature schedule.
    Uses FFT-based gradient for O(n log n) per iteration.
    """
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    z = z - z.max()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    total_target_iters = max(50000, int(time_limit * 30000))
    iters_per_stage = total_target_iters // n_stages

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        # Step size: scale inversely with T and with iteration count
        eta_base = eta_scale * 2.0 * n / (T_val * n)  # = 2.0/T_val
        eta_base = min(eta_base, 10.0 / T_val)

        for it in range(iters_per_stage):
            if time.monotonic() - t0 > time_limit:
                break

            a = np.exp(z - z.max())
            a = a / a.sum()

            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            w = np.exp(Tb - max_Tb)
            w_sum = w.sum()
            if w_sum < 1e-300:
                break
            w = w / w_sum

            g = compute_smooth_gradient(a, w)

            # Adaptive step size
            eta = eta_base / (1.0 + 0.0005 * it)
            z = z - eta * g
            z = z - z.max()

            # Track best (every 10 iterations to save time)
            if it % 10 == 0:
                a = np.exp(z - z.max())
                a = a / a.sum()
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

    # Final evaluation
    a = np.exp(z - z.max())
    a = a / a.sum()
    score = 2.0 * n * fast_convolve(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.005, n_stages=20, maxiter_per_stage=2000):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    z = z - z.max()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            a = np.exp(z_in - z_in.max())
            a = a / a.sum()
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            a = np.exp(z_in - z_in.max())
            a = a / a.sum()
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            w = np.exp(Tb - max_Tb)
            w = w / w.sum()
            g = compute_smooth_gradient(a, w)
            S = np.dot(g, a)
            return a * (g - S)
        return jac

    temps = np.geomspace(T_start, T_end, n_stages)
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': maxiter_per_stage * 500, 'disp': False}
            )
            z = res.x
            z = z - z.max()
        except Exception:
            pass

    a = np.exp(z - z.max())
    a = a / a.sum()
    return a


def _get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=30000):
    """Nesterov dual averaging LP refinement."""
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_score = best_score
    stall_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        if it % 200 == 0:
            if best_score < prev_score - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            if stall_count >= 8:
                rng_local = np.random.default_rng(777 + it)
                noise = rng_local.random(n) * 0.05
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

    return best_a, best_score


def peak_shave_refine(a0, time_limit, n_steps=50000):
    """Adaptive peak shaving with smart redistribution to low-impact coordinates."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find the peak (excluding boundaries which are naturally small)
        b_work = b.copy()
        if len(b_work) > 4:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b[peak_k]

        # Compute contribution to peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total_contrib = contribs.sum()
        if total_contrib < 1e-15:
            break

        # Find coordinates that contribute to peak
        top_contrib = np.argsort(contribs)[::-1]

        # Find coordinates with low overall impact (low autocorrelation contribution everywhere)
        # Approximate: coordinates where a[i] is small
        low_mask = a < np.median(a)
        low_coords = np.where(low_mask)[0]

        improved = False
        # Try reducing top contributors and redistributing
        for ci in top_contrib[:10]:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.90, 0.85, 0.78, 0.70]:
                diff = a[i] * (1.0 - factor)
                if diff < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] -= diff
                # Redistribute to low-impact coords
                if len(low_coords) > 0:
                    redist = a_trial[low_coords] + 1e-12
                    a_trial[low_coords] = redist + diff * redist / redist.sum()
                else:
                    others = np.ones(n)
                    others[i] = 0
                    a_trial = a_trial + diff * others / others.sum()
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    no_improve = 0
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair reduction
            for ci in top_contrib[:5]:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n or j < 0:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.92, 0.85, 0.78]:
                    for f2 in [0.92, 0.85, 0.78]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = norm_sequence(a_trial)
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve > 300:
                break

    return best_a, best_score


def structured_init(n, mode=0):
    """Create structured initial sequence."""
    if mode == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif mode == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-x * (1 - x) * 8) * (1 + 0.3 * np.cos(4 * np.pi * x))
    elif mode == 6:
        # Tukey-ish window
        x = np.linspace(0, 1, n)
        r = 0.25
        a = np.ones(n)
        a[:int(n * r)] = 0.5 * (1 + np.cos(np.pi * np.linspace(1, 0, int(n * r))))
        a[-int(n * r):] = 0.5 * (1 + np.cos(np.pi * np.linspace(1, 0, int(n * r))))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def subgradient_lp_hybrid(a0, time_limit, solver, n_steps=20000):
    """
    Hybrid method: use LP to get dual weights, then do mirror descent with those weights.
    Combines the global information from LP with the speed of mirror descent.
    """
    n = len(a0)
    a = norm_sequence(a0)
    z = np.log(np.maximum(a, 1e-300)).copy()
    z = z - z.max()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # LP-informed dual weights (updated periodically)
    w_dual = None
    lp_interval = 50

    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        a = np.exp(z - z.max())
        a = a / a.sum()

        if it % lp_interval == 0 or w_dual is None:
            b, M_b = solve_lp_best_response(a, solver)
            if b is not None:
                # Use b as the "target" direction
                w_dual = b

        if w_dual is None:
            continue

        # Compute gradient using LP-informed weighting
        conv_b = fast_convolve(a)
        # Weight by proximity to LP target
        T_val = 5.0
        Tb = T_val * conv_b
        max_Tb = Tb.max()
        w_smooth = np.exp(Tb - max_Tb)
        w_smooth = w_smooth / w_smooth.sum()

        g = compute_smooth_gradient(a, w_smooth)

        # Mirror descent step
        eta = 1.0 / np.sqrt(it + 1) * 0.5
        z = z - eta * g
        z = z - z.max()

        # Track best
        if it % 5 == 0:
            a = np.exp(z - z.max())
            a = a / a.sum()
            score = 2.0 * n * fast_convolve(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def multi_peak_mirror_descent(a0, time_limit, T_start=50.0, T_end=0.01, n_stages=20):
    """
    Mirror descent that tracks multiple peaks simultaneously.
    Instead of pure log-sum-exp, uses a weighted combination of top peaks.
    """
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    z = z - z.max()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 10000

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break

        for it in range(iters_per_stage):
            if time.monotonic() - t0 > time_limit:
                break

            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)

            # Use top-K peaks with geometric weighting
            K = min(5, len(b))
            top_idx = np.argsort(b)[::-1][:K]
            top_vals = b[top_idx]
            weights = np.exp(T_val * (top_vals - top_vals[0]))
            weights = weights / weights.sum()

            # Build effective weight vector
            w_eff = np.zeros_like(b)
            for ki, k in enumerate(top_idx):
                w_eff[k] = weights[ki]

            g = compute_smooth_gradient(a, w_eff)

            eta = 2.0 / (T_val * (1.0 + 0.001 * it))
            z = z - eta * g
            z = z - z.max()

            if it % 10 == 0:
                a = np.exp(z - z.max())
                a = a / a.sum()
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Mirror descent (fast, millions of iterations)
    2. L-BFGS polish (very low temperature)
    3. LP alternating (global structure)
    4. Peak shaving (targeted local moves)
    5. Multi-length exploration
    6. Perturbation + re-optimize
    7. Final polish
    """
    global height_sequence_1
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    solver = _get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1024
        best_seq = structured_init(n_init, mode=6)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_sequence(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Mirror Descent (fast global optimization) ===
    print(f"\n=== Phase 1: Mirror Descent n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, T: 200->0.005")
        a_md, _ = mirror_descent_refine(best_seq, t_alloc, T_start=200.0, T_end=0.005, n_stages=30)
        try_update(a_md, "mirror_descent_1")

    # === Phase 2: L-BFGS with very low final temperature ===
    print(f"\n=== Phase 2: L-BFGS Extended (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, T: 80->0.005, 25 stages")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=80.0, T_end=0.005, n_stages=25, maxiter_per_stage=1500)
        try_update(a_lb, "lbgfs_main")

    # === Phase 3: LP Alternating with Nesterov ===
    print(f"\n=== Phase 3: Nesterov LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=50000)
        try_update(a_lp, "nesterov_lp_1")

    # === Phase 4: Multi-peak Mirror Descent ===
    print(f"\n=== Phase 4: Multi-Peak MD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        a_mp, _ = multi_peak_mirror_descent(best_seq, t_alloc, T_start=50.0, T_end=0.01, n_stages=20)
        try_update(a_mp, "multi_peak_md")

    # === Phase 5: Peak Shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_ps, _ = peak_shave_refine(best_seq, t_alloc, n_steps=80000)
        try_update(a_ps, "peak_shave")

    # === Phase 6: Second round of mirror descent + L-BFGS ===
    print(f"\n=== Phase 6: MD + L-BFGS Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_md2 = min(rem * 0.12, 150.0)
    if t_md2 > 15:
        a_md2, _ = mirror_descent_refine(best_seq, t_md2, T_start=50.0, T_end=0.003, n_stages=25)
        try_update(a_md2, "mirror_descent_2")

    rem = deadline - time.monotonic()
    t_lb2 = min(rem * 0.08, 80.0)
    if t_lb2 > 10:
        a_lb2 = lbgfs_refine(best_seq, t_lb2, T_start=20.0, T_end=0.003, n_stages=15, maxiter_per_stage=1000)
        try_update(a_lb2, "lbgfs_polish")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 0.8, 2.0, 1.1]:
        alt_n = int(n_best * mult)
        if 256 <= alt_n <= 4096 and alt_n != n_best:
            alt_lengths.append(alt_n)
    for fixed_n in [512, 768, 1024, 1280, 1536, 2048]:
        if 256 <= fixed_n <= 4096 and fixed_n != n_best:
            alt_lengths.append(fixed_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 45:
            break
        t_total = min(rem * 0.20, 120.0)
        if t_total < 20:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)

        t_md_alt = t_total * 0.6
        a_alt, _ = mirror_descent_refine(init_alt, t_md_alt, T_start=100.0, T_end=0.01, n_stages=20)
        try_update(a_alt, f"md_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 15 and best_seq is not None and len(best_seq) == alt_n:
            t_lb_alt = min(rem * 0.3, 40.0)
            if t_lb_alt > 10:
                a_alt2 = lbgfs_refine(best_seq, t_lb_alt, T_start=20.0, T_end=0.01, n_stages=10, maxiter_per_stage=800)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 8: Perturbation + re-optimize ===
    print(f"\n=== Phase 8: Perturbation Escape (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        for scale in [0.05, 0.1, 0.2]:
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            perturbed = best_seq + scale * rng.standard_normal(n_cur)
            perturbed = norm_sequence(perturbed)

            t_sub = min(60.0, rem * 0.6)
            a_pert, _ = mirror_descent_refine(perturbed, t_sub, T_start=50.0, T_end=0.01, n_stages=15)
            try_update(a_pert, f"pert_md_{scale}")

            rem = deadline - time.monotonic()
            if rem > 15:
                t_lb_pert = min(30.0, rem * 0.5)
                if t_lb_pert > 8:
                    a_pert2 = lbgfs_refine(best_seq, t_lb_pert, T_start=10.0, T_end=0.01, n_stages=8, maxiter_per_stage=500)
                    try_update(a_pert2, f"pert_lb_{scale}")

    # === Phase 9: Final comprehensive polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin1 = min(rem * 0.25, 60.0)
        a_fin = lbgfs_refine(best_seq, t_fin1, T_start=5.0, T_end=0.002, n_stages=12, maxiter_per_stage=1000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin2 = min(rem * 0.25, 40.0)
        a_fin2, _ = mirror_descent_refine(best_seq, t_fin2, T_start=10.0, T_end=0.002, n_stages=15)
        try_update(a_fin2, "final_md")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin3 = min(rem * 0.3, 30.0)
        a_fin3, _ = peak_shave_refine(best_seq, t_fin3, n_steps=30000)
        try_update(a_fin3, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin4 = min(rem * 0.4, 30.0)
        a_fin4, _ = nesterov_lp_refine(best_seq, t_fin4, solver=solver, n_max_iters=3000)
        try_update(a_fin4, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
