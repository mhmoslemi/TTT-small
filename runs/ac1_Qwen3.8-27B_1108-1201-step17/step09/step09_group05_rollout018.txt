import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def evaluate_safe(a):
    """Evaluate sequence safely using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

Key improvements over baseline: lower final temperatures, FFT-based gradients,
more iterations at low T, and better time allocation.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


# Global state for L-BFGS callbacks (to avoid closures)
_LBFGS_STATE = {'T': 1.0, 'n': 0, 't0': 0.0, 'limit': 0.0}


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a) is already 1."""
    b = fast_convolve(a)
    return float(2.0 * len(a) * b.max())


def evaluate_safe(a):
    """Evaluate sequence safely using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_objective(z_in):
    """Log-sum-exp smoothed objective for L-BFGS. Uses _LBFGS_STATE."""
    if time.monotonic() - _LBFGS_STATE['t0'] > _LBFGS_STATE['limit']:
        return 1e15
    n = _LBFGS_STATE['n']
    T_val = _LBFGS_STATE['T']
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T_val) * log_sum_exp


def lse_gradient(z_in):
    """Gradient of log-sum-exp smoothed objective. Uses _LBFGS_STATE."""
    if time.monotonic() - _LBFGS_STATE['t0'] > _LBFGS_STATE['limit']:
        return np.zeros_like(z_in)
    n = _LBFGS_STATE['n']
    T_val = _LBFGS_STATE['T']
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    # FFT-based gradient: grad_a[i] = 2 * sum_m w[i+m] * a[m]
    # = 2 * (convolve(a[::-1], w))[n-1 : 2n-1]
    a_flip = a[::-1]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    Af = rfft(a_flip, fft_size)
    Wf = rfft(w, fft_size)
    conv_full = irfft(Af * Wf, fft_size)[:3 * n - 2]
    grad_a = 2.0 * conv_full[n - 1:2 * n - 1]

    # Softmax Jacobian: d/dz (f(a)) = a * (df/da - a·df/da)
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_single_stage(a0, T_val, time_limit, n_iter):

    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    _LBFGS_STATE['T'] = T_val
    _LBFGS_STATE['n'] = n
    _LBFGS_STATE['t0'] = time.monotonic()
    _LBFGS_STATE['limit'] = time_limit

    try:
        res = minimize(
            fun=lse_objective,
            x0=z,
            jac=lse_gradient,
            method='L-BFGS-B',
            options={'maxiter': n_iter, 'ftol': 1e-17, 'gtol': 1e-13,
                     'maxfun': n_iter * 500, 'disp': False}

        z = res.x
    except Exception:
        pass

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement with adaptive perturbation."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = compute_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if it % 300 == 299:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 8:
                rng_local = np.random.default_rng(77 + it)
                noise = rng_local.random(n) * 0.04
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0

    return best_a, best_score


def peak_shave_gradient(a0, time_limit, n_steps=20000):
    """Gradient-informed peak shaving: reduce mass at high-gradient positions,
    redirect to low-gradient positions to flatten the convolution."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find top peaks (excluding boundaries which are naturally small)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        peak_idx = np.argpartition(-b_work, min(5, len(b_work)-1))[:5]
        peak_k = int(peak_idx[0])

        # Gradient at the peak: grad_a[i] = 2*a[peak_k - i] (with bounds)
        grad_a = np.zeros(n)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        for i in range(js, je + 1):
            grad_a[i] = 2.0 * a[peak_k - i]

        # We want to reduce mass where grad is high, increase where grad is low
        # Use top-2 sources and top-2 targets
        top_sources = np.argsort(grad_a)[::-1][:3]
        low_grad = np.argsort(grad_a)[:n // 5]
        # Filter to positions with non-zero mass for sources
        top_sources = [i for i in top_sources if a[i] > 1e-12]
        if not top_sources:
            break

        improved = False
        for src in top_sources:
            if a[src] < 1e-14:
                continue
            # Try different fractions to move
            for frac in [0.3, 0.2, 0.1, 0.05]:
                delta = a[src] * frac
                # Distribute to low-gradient positions proportionally
                targets = low_grad[low_grad != src]
                if len(targets) == 0:
                    continue
                # Weight by inverse gradient (prefer lowest grad targets)
                weights = 1.0 / (grad_a[targets] + 1e-10)
                weights = weights / weights.sum()
                a_trial = a.copy()
                a_trial[src] -= delta
                a_trial[targets] += delta * weights
                a_trial = norm_sequence(a_trial)
                sc = compute_score(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 100:
                # Try a different peak
                for pk in peak_idx[1:]:
                    pk = int(pk)
                    if pk == 0 or pk == len(b) - 1:
                        continue
                    grad_a2 = np.zeros(n)
                    js2 = max(0, pk - n + 1)
                    je2 = min(n - 1, pk)
                    for i in range(js2, je2 + 1):
                        grad_a2[i] = 2.0 * a[pk - i]
                    top_src2 = np.argsort(grad_a2)[::-1][:3]
                    top_src2 = [i for i in top_src2 if a[i] > 1e-12]
                    low2 = np.argsort(grad_a2)[:n // 5]
                    for src2 in top_src2:
                        for frac2 in [0.2, 0.1, 0.05]:
                            delta2 = a[src2] * frac2
                            targets2 = low2[low2 != src2]
                            if len(targets2) == 0:
                                continue
                            w2 = 1.0 / (grad_a2[targets2] + 1e-10)
                            w2 = w2 / w2.sum()
                            a_trial = a.copy()
                            a_trial[src2] -= delta2
                            a_trial[targets2] += delta2 * w2
                            a_trial = norm_sequence(a_trial)
                            sc2 = compute_score(a_trial)
                            if sc2 < best_score - 1e-15:
                                a = a_trial
                                best_score = sc2
                                best_a = a.copy()
                                improved = True
                                stall = 0
                                break
                        if improved:
                            break
                    if improved:
                        break
            if stall > 300:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def frank_wolfe_away(a0, time_limit, n_steps=30000):
    """Frank-Wolfe with away steps for local optimization."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_star = int(np.argmax(b))

        # Subgradient
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        g[idx] = a[k_star - idx]

        # Toward step
        i_toward = int(np.argmin(g))
        # Away step
        mask = a > 1e-10
        if mask.sum() > 2:
            g_masked = g.copy()
            g_masked[~mask] = -np.inf
            i_away = int(np.argmax(g_masked))
        else:
            i_away = -1

        best_trial = None
        best_trial_score = best_score

        # Try toward steps
        for t_val in [0.4, 0.25, 0.15, 0.08, 0.04]:
            a_trial = a.copy()
            a_trial[i_toward] += t_val * (1.0 - a[i_toward])
            a_trial = norm_sequence(a_trial)
            sc = compute_score(a_trial)
            if sc < best_trial_score:
                best_trial_score = sc
                best_trial = a_trial.copy()

        # Try away steps
        if i_away >= 0:
            for gamma in [0.4, 0.25, 0.15, 0.08]:
                a_trial = a.copy()
                a_trial[i_away] *= (1.0 - gamma)
                a_trial = norm_sequence(a_trial)
                sc = compute_score(a_trial)
                if sc < best_trial_score:
                    best_trial_score = sc
                    best_trial = a_trial.copy()

        # Try mix: toward + away simultaneously
        if i_away >= 0 and i_toward != i_away:
            for t_val, g_val in [(0.3, 0.3), (0.2, 0.2), (0.1, 0.15)]:
                a_trial = a.copy()
                a_trial[i_toward] += t_val * (1.0 - a[i_toward])
                a_trial[i_away] *= (1.0 - g_val)
                a_trial = norm_sequence(a_trial)
                sc = compute_score(a_trial)
                if sc < best_trial_score:
                    best_trial_score = sc
                    best_trial = a_trial.copy()

        if best_trial is not None:
            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 150:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Nesterov LP (global)
    2. Multi-stage L-BFGS (global-to-local refinement)
    3. Peak shaving (local)
    4. Nesterov LP round 2
    5. Low-T L-BFGS polish
    6. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        x = np.linspace(0, np.pi, n_init)
        best_seq = np.cos(x / 2) ** 2
        best_seq = norm_sequence(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_sequence(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Nesterov LP (main global optimization) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 380.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_lp, "nesterov_lp_1")

    # === Phase 2: Multi-stage L-BFGS (aggressive annealing) ===
    print(f"\n=== Phase 2: L-BFGS Multi-Stage (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 520.0)
    if t_alloc > 60:
        T_stages = [80.0, 40.0, 20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        # More iterations at low temperatures
        iters = 2000
        print(f"  T stages: {len(T_stages)}, iters/stage: {iters}, budget={t_alloc:.0f}s")
        a_lb, _ = lbgfs_multistage(best_seq, t_alloc, T_stages, iters)
        try_update(a_lb, "lbgfs_main")

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.06, 80.0)
    if t_alloc > 10:
        a_ps, _ = peak_shave_gradient(best_seq, t_alloc, n_steps=30000)
        try_update(a_ps, "peak_shave")

    # === Phase 4: Frank-Wolfe ===
    print(f"\n=== Phase 4: Frank-Wolfe (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 10:
        a_fw, _ = frank_wolfe_away(best_seq, t_alloc, n_steps=50000)
        try_update(a_fw, "fw")

    # === Phase 5: Nesterov LP round 2 ===
    print(f"\n=== Phase 5: Nesterov LP r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=100000)
        try_update(a_lp2, "nesterov_lp_2")

    # === Phase 6: Low-T L-BFGS polish ===
    print(f"\n=== Phase 6: L-BFGS Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 30:
        T_polish = [2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01, 0.005]
        iters_polish = 3000
        print(f"  T: 2->0.005, iters={iters_polish}, budget={t_alloc:.0f}s")
        a_lb2, _ = lbgfs_multistage(best_seq, t_alloc, T_polish, iters_polish)
        try_update(a_lb2, "lbgfs_polish")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 2.0, 0.85, 1.1]:
        alt_n = int(n_best * mult)
        if 512 <= alt_n <= 3000 and alt_n != n_best:
            alt_lengths.append(alt_n)
    for fixed_n in [768, 896, 1024, 1152, 1280]:
        if 512 <= fixed_n <= 3000 and fixed_n != n_best:
            alt_lengths.append(fixed_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_total = min(rem * 0.2, 150.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)

        # Quick LP
        t_lp = t_total * 0.5
        a_alt, _ = nesterov_lp_refine(init_alt, t_lp, solver=solver, n_max_iters=30000)
        try_update(a_alt, f"lp_n{alt_n}")

        # L-BFGS polish
        rem = deadline - time.monotonic()
        if rem > 30:
            t_lb = min(rem * 0.3, 60.0)
            if t_lb > 15:
                T_short = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.01]
                a_alt2, _ = lbgfs_multistage(best_seq if len(best_seq) == alt_n else init_alt,
                                             t_lb, T_short, 1000)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 8: Final polish on best sequence ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.4, 80.0)
        T_final = [1.0, 0.5, 0.3, 0.1, 0.05, 0.02, 0.01]
        a_fin, _ = lbgfs_multistage(best_seq, t_fin, T_final, 3000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.3, 40.0)
        a_ps2, _ = peak_shave_gradient(best_seq, t_ps, n_steps=20000)
        try_update(a_ps2, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_fw = min(rem * 0.3, 30.0)
        a_fw2, _ = frank_wolfe_away(best_seq, t_fw, n_steps=30000)
        try_update(a_fw2, "final_fw")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
