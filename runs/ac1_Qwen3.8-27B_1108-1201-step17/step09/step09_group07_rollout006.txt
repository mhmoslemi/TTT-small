import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_gradient_smooth(a, T):
    """Gradient of (1/T)*log(sum_k exp(T*conv_k)) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    # grad[i] = 2 * sum_j a[j] * w[j+i] (cross-correlation)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    # correlate(a, w)[i] = sum_j conj(a[j]) * w[j+i] for real = sum_j a[j]*w[j+i]
    corr = irfft(np.conj(np.conj(A)) * W, fft_size)[:n]
    # Actually for real: correlate(a,w)[i] = sum_j a[j] * w[j+i]
    # In FFT: this is the same as the "fractional" convolution
    # Let's use: ifft(conj(A) * W) gives correlation
    corr = irfft(np.conj(A) * W, fft_size)[:n]
    return 2.0 * corr, w


def fgm_optimize(a0, time_limit, T_start=3.0, T_end=400.0, n_stages=15, steps_per_stage=2000):
    """FGM (conditional gradient) with away steps and temperature annealing."""
    n = len(a0)
    x = project_simplex(a0.copy())
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    total_steps = 0
    n_improve = 0

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            t_iter = total_steps + 1
            gamma = 2.0 / (t_iter + 2.0)

            # Compute gradient
            grad, w = compute_gradient_smooth(x, T)

            # Forward vertex
            vf_idx = int(np.argmin(grad))

            # Forward step candidate
            x_forward = x.copy()
            x_forward[vf_idx] += 1.0
            x_forward /= x_forward.sum()

            # Try forward step
            x_f = (1.0 - gamma) * x + gamma * x_forward

            # Away step: find component to move away from
            support = np.where(x > 1e-8)[0]
            x_a = None
            if len(support) > 2:
                # Away index: the one with largest gradient among support
                i_a = support[np.argmax(grad[support])]
                # Build away direction
                denom = 1.0 - x[i_a]
                if denom > 1e-12:
                    x_away_dir = x.copy()
                    x_away_dir[i_a] = 0.0
                    x_away_dir /= denom
                    # Away step: increase all other components, decrease i_a
                    x_a = (1.0 + gamma) * x - gamma * x_away_dir
                    x_a = np.maximum(x_a, 0)
                    s = x_a.sum()
                    if s > 1e-12:
                        x_a /= s

            # Choose better of forward and away
            score_f = None  # We'll just use the score after both
            x_trial = x_f
            if x_a is not None:
                # Quick comparison using gradient projection
                proj_f = np.dot(grad, x_f - x)
                proj_a = np.dot(grad, x_a - x)
                if proj_a < proj_f:
                    x_trial = x_a

            x = x_trial
            total_steps += 1

            # Periodic exact evaluation
            if total_steps % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_a = x.copy()
                    n_improve += 1
                    if n_improve % 5 == 0:
                        elapsed = time.monotonic() - t0
                        print(f"    FGM step={total_steps} ({elapsed:.0f}s): score={best_score:.6f} T={T:.1f}")

    return best_a, best_score


def nesterov_pg_optimize(a0, time_limit, T_start=2.0, T_end=300.0,
                         lr_init=0.08, n_stages=12, steps_per_stage=2500):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            lr = lr_init / (1.0 + 0.003 * step_count)

            # Nesterov extrapolation
            t_curr = (1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count)) / 2.0
            if step_count > 0:
                t_prev = (1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) ** 2)) / 2.0
                beta = (t_prev - 1.0) / t_curr
            else:
                beta = 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)

            grad, _ = compute_gradient_smooth(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1

            if step_count % 80 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
                    if step_count % 800 == 0:
                        elapsed = time.monotonic() - t0
                        print(f"    PG step={step_count} ({elapsed:.0f}s): score={best_score:.6f} T={T:.1f}")

    return best_a, best_score


def build_conv_matrix(a_fixed, n_b):
    """Build (2n-1) x n_b matrix T such that T @ b = convolve(a_fixed, b)."""
    n_a = len(a_fixed)
    size = n_a + n_b - 1
    T = np.zeros((size, n_b))
    for j in range(n_b):
        T[j:j + n_a, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b) subject to b>=0, sum(b)=1."""
    if solver is None:

    try:

    except Exception:

            if alt == solver:

        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)

    s = b_val.sum()

def coordinate_descent_polish(a0, time_limit, n_steps=20000):
    """Coordinate descent that tries reducing top peak contributors."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        if len(b) > 2:
            b_work = b.copy()
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        else:
            b_work = b
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]

        # Find top contributors
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = np.zeros(len(idx))
        for m, i in enumerate(idx):
            j = peak_k - i
            if 0 <= j < n and j != i:
                contribs[m] = 2 * a[i] * a[j]
            elif i == j:
                contribs[m] = a[i] ** 2
        top_m = np.argsort(contribs)[::-1][:8]

        improved = False
        for m in top_m:
            i = idx[m]
            if a[i] < 1e-14:
                continue
            for factor in [0.96, 0.93, 0.89, 0.85]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Pair reduction
            for m in top_m[:4]:
                i = idx[m]
                j = peak_k - i
                if i == j or j < 0 or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.90]:
                    for f2 in [0.95, 0.90]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = project_simplex(a_trial)
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    # === Phase 2: Nesterov PG (different params) ===
    print(f"\n=== Phase 2: Nesterov PG (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 330.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a2, _ = nesterov_pg_optimize(best_seq, t_alloc, T_start=5.0, T_end=600.0,
                                     lr_init=0.04, n_stages=10, steps_per_stage=3000)
        try_update(a2, "pg_1")

    # === Phase 3: LP refinement ===
    print(f"\n=== Phase 3: LP Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 420.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a3, _ = lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=100000)
        try_update(a3, "lp_1")

    # === Phase 4: Redistribution peak shaving ===
    print(f"\n=== Phase 4: Redistribution PS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a4, _ = redistribution_peak_shave(best_seq, t_alloc, n_trials=20000)
        try_update(a4, "redist_ps")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: CD Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 5:
        a5, _ = coordinate_descent_polish(best_seq, t_alloc, n_steps=30000)
        try_update(a5, "cd_polish")

    # === Phase 6: Multi-start with diverse inits ===
    print(f"\n=== Phase 6: Multi-Start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        for variant in [1, 3, 5, 7]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_sub = min(rem * 0.25, 120.0)
            if t_sub < 30:
                break
            a_init = make_diverse_init(n_best, variant, rng)
            # Short FGM
            a_sub, _ = fgm_optimize(a_init, t_sub * 0.5, T_start=2.0, T_end=300.0,
                                    n_stages=8, steps_per_stage=2000)
            try_update(a_sub, f"diverse_{variant}")
            # Short LP
            rem = deadline - time.monotonic()
            if rem > 20:
                a_sub2, _ = lp_refine(a_sub, min(rem * 0.3, 40.0), solver=solver,
                                      n_max_iters=3000)
                try_update(a_sub2, f"diverse_{variant}_lp")

    # === Phase 7: Alternative lengths ===
    print(f"\n=== Phase 7: Alt Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = []
    for delta in [-80, -40, -20, 20, 40, 80, 128, -128]:
        alt_n = n_best + delta
        if 256 <= alt_n <= 3000:
            alt_lengths.append(alt_n)
    # Also try some "round" numbers
    for cand in [512, 768, 896, 1024, 1280, 1536]:
        if 256 <= cand <= 3000 and cand != n_best:
            alt_lengths.append(cand)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_total = min(rem * 0.20, 120.0)
        if t_total < 25:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = fgm_optimize(init_alt, t_total * 0.4, T_start=3.0, T_end=300.0,
                                n_stages=6, steps_per_stage=1500)
        try_update(a_alt, f"alt_n{alt_n}_fgm")
        rem = deadline - time.monotonic()
        if rem > 20:
            a_alt2, _ = lp_refine(a_alt, min(rem * 0.3, 35.0), solver=solver,
                                  n_max_iters=2000)
            try_update(a_alt2, f"alt_n{alt_n}_lp")

    # === Phase 8: Second round of LP on best ===
    print(f"\n=== Phase 8: LP Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.30, 120.0)
        a_fin, _ = lp_refine(best_seq, t_fin, solver=solver, n_max_iters=50000)
        try_update(a_fin, "lp_round2")

    # === Phase 9: Final peak shave + CD ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.25, 40.0)
        a_ps, _ = redistribution_peak_shave(best_seq, t_ps, n_trials=10000)
        try_update(a_ps, "final_ps")
        rem = deadline - time.monotonic()
        if rem > 5:
            a_cd, _ = coordinate_descent_polish(best_seq, min(rem * 0.5, 15.0), n_steps=5000)
            try_update(a_cd, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
