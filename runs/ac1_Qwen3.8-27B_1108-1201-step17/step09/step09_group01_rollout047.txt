from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def solve_best_response(a, method='higs'):
    """Solve LP: min M s.t. T_a @ b <= M, sum(b)=1, b>=0"""
    n = len(a)
    # Build constraint matrix
    # Variables: [b_0, ..., b_{n-1}, M]
    m = 2 * n - 1  # number of convolution positions
    
    # A_ub: for each k, sum_i a_{max(0,k-n+1):min(n-1,k)+1} * b_{k-i} - M <= 0
    # This is expensive to build as a dense matrix for large n
    # But it's a Toeplitz-like structure
    
    # Actually, let's build it differently.
    # The convolution (a*b)[k] = sum_i a[k-i] * b[i] for valid i
    # So (T_a @ b)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[k-i] * b[i]
    
    # Build A_ub as (2n-1) x (n+1) matrix
    # Column 0 to n-1: the T_a matrix
    # Column n: -ones(2n-1)
    
    # For efficiency, build T_a using the structure
    T = np.zeros((m, n))
    for i in range(n):
        # T[k, i] = a[k-i] if 0 <= k-i < n
        # k-i >= 0 => k >= i
        # k-i <= n-1 => k <= i+n-1
        k_start = i
        k_end = min(i + n, m)
        T[k_start:k_end, i] = a[:k_end - i]
    
    A_ub = np.hstack([T, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # b_i >= 0, M free
    # Actually M should be >= 0 for numerical stability
    bounds[n] = (0, None)
    
    res = linprog(
        c=np.concatenate([np.zeros(n), [1.0]]),
        A_ub=A_ub, b_ub=b_ub,
        A_eq=A_eq, b_eq=b_eq,
        bounds=bounds,
        method=method
    )
    
    if res.status != 0:
        return None, None
    
    b = res.x[:n]
    M = res.x[n]
    b = np.maximum(b, 0)
    s = b.sum()
    if s < 1e-15:
        return None, None
    b = b / s
    return b, M

def smooth_subgradient_method(a0, time_limit, beta=50, eta=0.01):
    n = len(a0)
    a = a0.copy()
    # Track running average for Nesterov
    a_avg = a.copy()
    best_a = a.copy()
    best_score = 2*n*fast_convolve(a).max()
    
    t = time.monotonic()
    it = 0
    while time.monotonic() - t < time_limit:
        it += 1
        
        # Compute convolution
        conv = fast_convolve(a)
        M = conv.max()
        
        # Softmax weights
        scaled = beta * (conv - M)
        scaled = np.clip(scaled, -50, 0)
        y = np.exp(scaled)
        y /= y.sum()
        
        # Subgradient: g_i = sum_k y_k * a_{k-i} for valid k-i
        g = np.zeros(n)
        for i in range(n):
            k_min = max(0, i)  # k - i >= 0 => k >= i
            k_max = min(2*n-2, i+n-1)  # k - i <= n-1 => k <= i+n-1
            if k_min <= k_max:
                g[i] = np.dot(y[k_min:k_max+1], a[k_min-i:k_max+1-i])
        
        # Normalize gradient
        g_norm = g.sum()
        
        # Multiplicative update (mirror descent on simplex)
        # log a_new = log a - eta * g
        # a_new ∝ a * exp(-eta * g)
        log_a = np.log(a + 1e-30) - eta * g
        a_new = np.exp(log_a - log_a.max())
        a_new /= a_new.sum()
        
        # Nesterov averaging
        alpha = 2.0 / (it + 2.0)
        a_avg = (1 - alpha) * a_avg + alpha * a_new
        
        # Evaluate
        for cand in [a_new, a_avg]:
            score = 2 * n * fast_convolve(cand).max()
            if score < best_score:
                best_score = score
                best_a = cand.copy()
        
        a = a_new
        
        # Adaptive eta
        if it % 100 == 0:
            eta = 0.01 / np.sqrt(it)  # standard rate
        
        # Increase beta gradually
        if it % 1000 == 0:
            beta *= 1.5
    
    return best_a, best_score

def mirror_descent_step(a, n_steps, beta=50.0, eta_init=0.05):
    """Run several mirror descent subgradient steps on a (normalized, non-negative)."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    for step in range(n_steps):
        conv = fast_convolve(a)
        M = conv.max()
        if M < 1e-15:
            break
        # Softmax weights for smooth max
        scaled = beta * (conv - M)
        scaled = np.clip(scaled, -80.0, 0.0)
        y = np.exp(scaled)
        y_sum = y.sum()
        if y_sum < 1e-30:
            y = np.zeros_like(conv)
            y[conv.argmax()] = 1.0
        else:
            y /= y_sum
        # Subgradient: g_i = sum_j a[j] * y[i+j] = correlate(y, a, 'valid')
        g = np.correlate(y, a, mode='valid')
        # Normalize gradient for stable step
        g_sum = g.sum()
        if g_sum > 1e-15:

        eta = eta_init / (1.0 + 0.1 * step)
        # Multiplicative (mirror) update

        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new /= a_new.sum()

            best_a = a_new.copy()
        a = a_new
        # Increase beta periodically for sharper max approximation
        if step % 200 == 199:
            beta = min(beta * 1.3, 200.0)
    
    return best_a, best_score

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))
            
            affected = conv[k_start:k_end + 1].copy()
            
            best_delta = 0.0
            best_new_max = M
            # Try shrinking and growing
            for factor in [0.98, 0.95, 0.9, 0.8, 0.5, 1.1, 1.2, 1.5, 2.0]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-16:
                    continue
                new_affected = affected + 2.0 * delta * a
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a.copy()
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_random_symmetric(n, rng):
    """Random symmetric initialization."""
    half = (n + 1) // 2
    t = np.linspace(0, 1, half)
    n_comp = int(rng.integers(2, 7))
    profile = np.ones(half) * 0.5
    for j in range(n_comp):
        freq = rng.uniform(0.5, 5.0)
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0.01)
    if n % 2 == 0:
        a = np.concatenate([profile[:half], profile[:half][::-1]])
    else:
        a = np.concatenate([profile, profile[-2::-1]])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_bimodal(n, rng):
    """Bimodal initialization."""
    c1 = n * rng.uniform(0.15, 0.35)
    c2 = n * rng.uniform(0.65, 0.85)
    w1 = n * rng.uniform(0.05, 0.15)
    w2 = n * rng.uniform(0.05, 0.15)
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - c2) / w2) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
