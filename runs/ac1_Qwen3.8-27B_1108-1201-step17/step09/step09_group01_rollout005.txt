import numpy as np

import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def score_normalized(a):

    return float(2.0 * n * conv.max())


def compute_score(a):

    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return float(2.0 * len(a) * fast_convolve(a / s).max())


def coord_descent_level(a, conv, n, i, eps=1e-12):
    """
    Find the optimal delta for coordinate i using the leveling approach.
    Returns the best delta (change to a[i]).
    
    Key insight: all affected conv entries have non-negative slope in delta,
    so the optimal delta is the one that minimizes max(affected).
    Since the max is non-decreasing in delta, the best is to decrease a[i]
    as much as possible, BUT we need to balance with the global objective.
    
    Actually, for coordinate descent on max(conv), the optimal delta makes
    the max affected entry as small as possible, which is at delta = -a[i]
    (set a[i] to 0). But this is too aggressive. Instead, we use a damped
    version: bring the max affected down to the unaffected max U.
    """
    ai = a[i]
    if ai < eps:
        return 0.0
    
    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)
    
    # Compute unaffected max
    if k_start > 0:
        unaff_left = float(conv[:k_start].max())
    else:
        unaff_left = -1e30
    if k_end < 2 * n - 2:
        unaff_right = float(conv[k_end + 1:].max())
    else:
        unaff_right = -1e30
    U = max(unaff_left, unaff_right)
    
    # Current max in affected region
    affected = conv[k_start:k_end + 1]
    cur_max = float(affected.max())
    
    if cur_max <= U + 1e-15:
        # No need to decrease; try small increase
        # (might help other coordinates)
        return 0.0
    
    # Leveling: find delta such that max_affected + 2*a[k-i]*delta = U
    # For the argmax position k*, delta_level = (U - cur_max) / (2 * a[k* - i])
    argmax_k = k_start + int(affected.argmax())
    slope = 2.0 * a[argmax_k - i]
    
    if slope < 1e-15:
        # The argmax entry doesn't depend on a[i] (shouldn't happen for k* in affected)
        return 0.0
    
    delta_level = (U - cur_max) / slope
    
    # Clamp: can't go below -a[i] (i.e., can't make a[i] negative)
    if delta_level < -ai:
        delta_level = -ai
    
    # Damping factor to avoid oscillation
    # Try several fractions of the leveling step
    candidates = [0.0, delta_level, 0.5 * delta_level, 0.75 * delta_level,
                  0.9 * delta_level, 0.25 * delta_level]
    
    best_delta = 0.0
    best_max = cur_max
    
    for d in candidates:
        new_ai = ai + d
        if new_ai < -1e-15:
            continue
        new_ai = max(new_ai, 0.0)
        
        # Compute new affected max
        # conv_new[k] = conv[k] + 2*a[k-i]*d + (1 if k==2i else 0)*d^2
        new_affected = affected + 2.0 * d * np.array([a[k - i] for k in range(k_start, k_end + 1)])
        if 2 * i >= k_start and 2 * i <= k_end:
            idx = 2 * i - k_start
            new_affected = new_affected.copy()
            new_affected[idx] = conv[2 * i] + 2.0 * d * a[i] + d * d
        
        new_max = max(U, float(new_affected.max()))
        if new_max < best_max - 1e-15:
            best_max = new_max
            best_delta = d
    
    return best_delta


def full_sweep(a, time_limit, start_time, damping=1.0, eps=1e-12):
    """One full sweep of coordinate descent. Returns updated a, conv, M."""
    n = len(a)
    conv = fast_convolve(a)
    M = float(conv.max())
    improved = False
    
    for i in range(n):
        if time.monotonic() - start_time > time_limit:
            break
        
        ai = a[i]
        if ai < eps:
            continue
        
        delta = coord_descent_level(a, conv, n, i, eps)
        delta *= damping
        
        if abs(delta) < 1e-18:
            continue
        
        new_ai = ai + delta
        if new_ai < -1e-15:
            continue
        new_ai = max(new_ai, 0.0)
        
        # Update conv in-place
        k_start = i
        k_end = min(i + n - 1, 2 * n - 2)
        
        # Compute changes: conv[k] changes by 2*delta*a[k-i] + (1 if k==2i)*delta^2
        k_range = np.arange(k_start, k_end + 1)
        a_shifted = a[k_range - i]  # a[k-i] for k in range
        changes = 2.0 * delta * a_shifted
        if 2 * i >= k_start and 2 * i <= k_end:
            idx = 2 * i - k_start
            changes = changes.copy()
            changes[idx] = 2.0 * delta * a[i] + delta * delta
        
        conv[k_start:k_end + 1] += changes
        a[i] = new_ai
        
        new_M = max(float(np.max(conv[:k_start])) if k_start > 0 else -1e30,
                    float(conv[k_start:k_end+1].max()),
                    float(np.max(conv[k_end+1:])) if k_end < 2*n-2 else -1e30)
        
        if new_M < M - 1e-16:
            M = new_M
            improved = True
        else:
            M = max(M, new_M)
    
    return a, conv, M, improved


def subgradient_step(a, conv, step_size, n_peaks=50):
    """
    Subgradient step: push down the coordinates that contribute most to peaks.
    Returns updated a (projected to simplex).
    """
    n = len(a)
    M = float(conv.max())
    
    # Find top peak positions
    top_idx = np.argsort(conv)[::-1][:n_peaks]
    
    # Compute subgradient: grad[i] = sum_{k in top} 2*a[k-i] (weighted)
    # Use softmax weights for smoothness
    peaks = conv[top_idx]
    weights = np.exp(-(peaks - peaks.min()) * 5.0)  # weight closer to uniform
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        grad[j] += 2.0 * w * a[k - j]
    
    # Update: a_new = a - step_size * grad, then project
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s


def peak_targeted_perturb(a, rng, strength=0.03):
    """Perturb coordinates that contribute to peaks."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    
    peak_threshold = M * 0.92
    peak_positions = np.where(conv >= peak_threshold)[0]
    
    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a[i] * a[j]
    
    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max
    
    noise = rng.standard_normal(n) * 0.1
    a_new = a * (1.0 - strength * pressure + noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def random_restart(a, rng, frac=0.2):
    """Randomly zero out some coordinates and redistribute."""
    n = len(a)
    a_new = a.copy()
    mask = rng.random(n) < frac
    a_new[mask] = 0
    a_new[~mask] *= (1.0 + rng.standard_normal((~mask).sum()) * 0.15)
    a_new = np.maximum(a_new, 0)
    # Add small mass to some zeroed positions
    zero_mask = (a_new < 1e-15)
    if zero_mask.any():
        n_add = max(1, int(zero_mask.sum() * 0.2))
        add_idx = np.where(zero_mask)[0]
        add_idx = rng.choice(add_idx, size=min(n_add, len(add_idx)), replace=False)
        a_new[add_idx] = rng.random(n_add) * a.max() * 0.3
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def reactivate_zeros(a, rng, n_try=20):
    """Try to reactivate zero coordinates."""
    n = len(a)
    a_new = a.copy()
    zero_mask = (a_new < 1e-12)
    if not zero_mask.any():
        return a_new
    
    zero_indices = np.where(zero_mask)[0]
    n_active = min(n_try, len(zero_indices))
    chosen = rng.choice(zero_indices, size=n_active, replace=False)
    
    for idx in chosen:
        # Try small values
        for val in [a.max() * 0.01, a.max() * 0.05, a.max() * 0.1]:
            a_new[idx] = val
            break  # Just set one value
    
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def init_symmetric(n, rng):
    """Symmetric initialization with cosine profile."""
    half = n // 2 + 1
    t = np.linspace(0, 1, half)
    # Superposition of cosines
    profile = np.ones(half) * 0.5
    n_comp = int(rng.integers(2, 6))
    for j in range(n_comp):
        freq = rng.uniform(1.0, 3.0)
        amp = rng.uniform(0.2, 0.5)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0)
    
    if n % 2 == 0:
        a = np.concatenate([profile[:half], profile[:half-1][::-1]])
    else:
        a = np.concatenate([profile, profile[:half-1][::-1]])
    
    # Zero out boundaries for better edge behavior
    a[0] = 0
    a[-1] = 0
    if n > 2:
        a[1] *= 0.3
        a[-2] *= 0.3
    
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_banded(n, rng):
    """Initialize with mass concentrated in a band (good for flat interior conv)."""
    # Place mass in a contiguous band in the middle
    band_width = int(n * rng.uniform(0.3, 0.7))
    start = (n - band_width) // 2
    a = np.zeros(n)
    t = np.linspace(0, 1, band_width)
    # Cosine bell within the band
    a[start:start + band_width] = np.cos(np.pi * t) ** 2
    # Add small overshoot at edges
    if start > 0:
        a[start - 1] = a[start] * 0.1
    if start + band_width < n:
        a[start + band_width] = a[start + band_width - 1] * 0.1
    
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_riple(n, rng):
    """Flat base with periodic ripples."""
    period = max(3, int(n / rng.integers(4, 12)))
    t = np.arange(n)
    amp = rng.uniform(0.2, 0.5)
    a = np.ones(n) * (1.0 - amp)
    a += amp * np.cos(2 * np.pi * t / period + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0)
    # Zero boundaries
    a[0] = 0
    a[-1] = 0
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    damping = 1.0
    rng = np.random.default_rng(hash(label) % (2**31))
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        elapsed = time.monotonic() - t0
        
        # Periodic perturbation to escape local optima
        if sweeps % 80 == 0 and no_improve > 30:
            if no_improve % 3 == 0:
                a = peak_targeted_perturb(a, rng, strength=0.02 + 0.01 * (no_improve // 100))
            elif no_improve % 3 == 1:
                a = random_restart(a, rng, frac=0.1 + 0.05 * min(no_improve // 200, 3))
            else:
                a = reactivate_zeros(a, rng, n_try=30)
            conv = fast_convolve(a)
            M = float(conv.max())
            no_improve = 0
        
        # Full sweep
        a, conv, new_M, improved = full_sweep(a, time_limit - elapsed, t0, damping=damping)
        M = new_M
        
        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive damping: reduce when stuck
        if no_improve > 20:
            damping = max(0.3, 1.0 - 0.1 * (no_improve - 20) // 10)
        else:
            damping = 1.0
        
        # Subgradient step every 30 sweeps if stuck
        if no_improve > 50 and sweeps % 30 == 0:
            step_size = 0.01 / (1 + no_improve / 100)
            a = subgradient_step(a, conv, step_size, n_peaks=min(50, n // 10))
            conv = fast_convolve(a)
            M = float(conv.max())
        
        if verbose and sweeps % 50 == 0:
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f} M={M:.10f} no_impr={no_improve}")
    
    return best_a, best_score


def optimize_hybrid(a0, time_limit, verbose=False, label=""):
    """
    Hybrid optimizer alternating between:
    - Coordinate descent (fast local convergence)
    - Iterative watermarking / subgradient (global moves)
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 2:
            break
        
        # Alternate: CD sweeps then subgradient
        if it % 4 == 0:
            # Subgradient phase (3 iterations)
            for _ in range(3):
                if time.monotonic() - t0 > time_limit - 5:
                    break
                conv = fast_convolve(a)
                M = float(conv.max())
                step_size = 0.005 * (1.0 / (1.0 + it * 0.01))
                a = subgradient_step(a, conv, step_size, n_peaks=min(80, n // 8))
        else:
            # CD phase
            time_left = min(remaining * 0.4, 30.0)
            a, _, sc_cd, _ = optimize_cd(a, time_left, verbose=False, label=f"{label}_it{it}")
            if sc_cd < best_score:
                best_score = sc_cd
                best_a = a.copy()
        
        # Check current
        conv = fast_convolve(a)
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
        
        # Perturbation every 20 iterations
        if it % 20 == 0:
            strength = 0.02 + 0.01 * min(it // 50, 5)
            a_pert = peak_targeted_perturb(best_a, rng, strength)
            conv_p = fast_convolve(a_pert)
            if 2.0 * n * float(conv_p.max()) < best_score:
                a = a_pert
                best_a = a_pert.copy()
                best_score = 2.0 * n * float(conv_p.max())
        
        if verbose and it % 10 == 0:
            print(f"    Hybrid {label} it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Combines exact coordinate descent, subgradient methods, and structured
    perturbations with adaptive time allocation.
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(42)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass
    
    # === Phase 1: Diverse initialization + quick CD ===
    print("\n=== Phase 1: Diverse Init + Quick CD ===")
    t1_start = time.monotonic()
    t1_budget = 250.0
    
    n_values = [768, 896, 1024, 1152, 1280, 1536]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)
    
    candidates = []
    
    for n in n_values:
        elapsed_p1 = time.monotonic() - t1_start
        if elapsed_p1 > t1_budget - 10:
            break
        
        init_cands = []
        if parent is not None:
            init_cands.append(("parent", interpolate_seq(parent, n)))
        init_cands.append(("symmetric", init_symmetric(n, rng)))
        init_cands.append(("banded", init_banded(n, rng)))
        init_cands.append(("ripple", init_riple(n, rng)))
        init_cands.append(("uniform", np.ones(n) / n))
        
        for name, cand in init_cands:
            t_left = t1_budget - (time.monotonic() - t1_start)
            if t_left < 5:
                break
            t_alloc = max(3.0, min(8.0, t_left / 4.0))
            a_opt, s_opt = optimize_cd(cand, t_alloc, verbose=False, label=f"p1_{name}_n{n}")
            candidates.append((s_opt, a_opt, n, name))
    
    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, top-5: " +
          ", ".join([f"{c[0]:.6f}(n={c[2]},{c[3]})" for c in candidates[:5]]))
    
    for s_c, a_c, n_c, it_c in candidates[:6]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{it_c}_n{n_c}")
    
    # === Phase 2: Intensive optimization on best ===
    print(f"\n=== Phase 2: Intensive CD (best={best_score:.8f}) ===")
    t2_start = time.monotonic()
    t2_budget = 600.0
    
    if best_seq is not None:
        n_best = len(best_seq)
        current = best_seq.copy()
        
        for rnd in range(8):
            rem = deadline - time.monotonic() - 400
            if rem < 20:
                break
            t_alloc = min(rem * 0.25, 120.0)
            print(f"  CD round {rnd+1}/8, n={n_best}, budget={t_alloc:.0f}s, best={best_score:.8f}")
            
            a_opt, s_opt = optimize_cd(current, t_alloc, verbose=True, label=f"p2r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"p2_cd_r{rnd}")
            
            # Perturbation for next round
            if rnd < 6:
                if rnd % 3 == 0:
                    current = peak_targeted_perturb(best_seq, rng, strength=0.02 + 0.01 * rnd)
                elif rnd % 3 == 1:
                    current = random_restart(best_seq, rng, frac=0.1 + 0.03 * rnd)
                else:
                    current = reactivate_zeros(best_seq, rng, n_try=30)
            else:
                current = best_seq.copy()
    
    # === Phase 3: Hybrid optimization ===
    print(f"\n=== Phase 3: Hybrid (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_hybrid = min(rem * 0.5, 200.0)
            print(f"  Hybrid n={len(best_seq)}, budget={t_hybrid:.0f}s")
            a_h, s_h = optimize_hybrid(best_seq, t_hybrid, verbose=True, label="p3")
            best_seq, best_score = check_and_update(a_h, best_seq, best_score, "p3_hybrid")
    
    # === Phase 4: Try different lengths ===
    print(f"\n=== Phase 4: Alt Lengths (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for delta_n in [128, 256, -128, -256, 512]:
            alt = n_best + delta_n
            if 512 <= alt <= 2048 and alt != n_best:
                alt_lengths.append(alt)
        
        for alt_n in alt_lengths[:4]:
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            t_alt = min(rem * 0.3, 60.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, s_alt = optimize_cd(init_alt, t_alt, verbose=False, label=f"alt{alt_n}")
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")
    
    # === Phase 5: Final intensive polishing ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Multiple rounds of CD with decreasing perturbation
        for polish in range(4):
            rem = deadline - time.monotonic() - 30
            if rem < 15:
                break
            t_polish = min(rem * 0.4, 80.0)
            a_p, s_p = optimize_cd(best_seq, t_polish, verbose=False, label=f"polish{polish}")
            best_seq, best_score = check_and_update(a_p, best_seq, best_score, f"polish{polish}")
            
            # Tiny perturbation
            if polish < 3:
                a_tiny = best_seq + rng.standard_normal(len(best_seq)) * 0.001
                a_tiny = np.maximum(a_tiny, 0)
                a_tiny = a_tiny / a_tiny.sum()
                best_seq, best_score = check_and_update(a_tiny, best_seq, best_score, f"tiny{polish}")
    
    # === Phase 6: Subgradient refinement on best ===
    print(f"\n=== Phase 6: Subgradient Refine (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            a = best_seq.copy()
            sub_iter = 0
            while time.monotonic() < deadline - 15:
                sub_iter += 1
                conv = fast_convolve(a)
                M = float(conv.max())
                step_size = 0.002 / (1 + sub_iter * 0.05)
                a = subgradient_step(a, conv, step_size, n_peaks=min(100, len(a) // 5))
                cur_score = 2.0 * len(a) * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
