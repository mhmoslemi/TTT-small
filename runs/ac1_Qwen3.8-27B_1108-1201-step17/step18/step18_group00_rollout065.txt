from scipy.optimize import linprog


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_to(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0.0)
    ss = result.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return result / ss


def lp_fw(a0, time_limit, n, K_initial=400, K_max=2500, verbose=True):

    best_m = fast_autocorr(a).max()

    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:K_initial])

        elapsed = time.monotonic() - t0

            break

            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr

        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        K = len(constraint_list)
        if K == 0:
            break

        A_ub = np.zeros((K, n))
        for ii, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                A_ub[ii, idx] = y[k_int - idx]

        A_full = np.column_stack([A_ub, -np.ones(K)])
        b_full = np.zeros(K)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        bounds = [(0, None)] * n + [(0, None)]

        try:
            res = linprog(c_obj, A_ub=A_full, b_ub=b_full, A_eq=A_eq, b_eq=b_eq,

        except Exception:
            continue

        if res.status != 0 or res.x is None:
            if len(active_set) > K_initial:
                active_list = list(active_set)
                active_set = set(int(active_list[i]) for i in range(min(K_initial, len(active_list))))
            continue

        g_val = np.maximum(res.x[:n], 0)
        g_sum = g_val.sum()
        if g_sum < 1e-12:

        default_alpha = 2.0 / (t + 2.0)

    J_final = 2.0 * n * best_m
    if verbose:
        print("    LP-FW done: J=" + repr(round(J_final, 8)) + " iters=" + str(iter_count) + " time=" + str(round(time.monotonic() - t0, 1)) + "s")

def lse_mirror_descent(a0, time_limit, n, verbose=True):
    t0 = time.monotonic()
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    x = a.copy()
    x_prev = a.copy()
    rng = np.random.default_rng(123)
    stall = 0
    tau_start = 0.08
    tau_end = 0.008

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 2000:
            noise = 0.015 / (1.0 + 0.1 * (stall // 2000))
            x = best_a + rng.normal(0, noise, n) * np.sqrt(best_a + 1e-10)
            x = norm_sum1(x)
            x_prev = x.copy()
            stall = 0
            continue

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-14)
        s_y = y.sum()
        if s_y < 1e-14:
            y = x.copy()
        else:
            y = y / s_y

        ac = fast_autocorr(y)
        tau = tau_end + (tau_start - tau_end) * np.exp(-t / 1500.0)

        ac_shifted = ac - ac.max()
        w = np.exp(ac_shifted / tau)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(ac)) / len(ac)
        else:
            w = w / w_sum

        a_rev = y[::-1]
        conv_size = len(w) + n - 1
        fft_sz = 1 << (conv_size - 1).bit_length()
        Wf = np.fft.rfft(w, fft_sz)
        Af = np.fft.rfft(a_rev, fft_sz)
        conv_result = np.fft.irfft(Wf * Af, fft_sz)[:conv_size]
        grad = (2.0 / tau) * conv_result[n - 1:2 * n - 1]

        grad_centered = grad - grad.mean()
        grad_norm = np.sqrt(np.sum(grad_centered ** 2))
        if grad_norm < 1e-15:
            grad_centered = rng.normal(0, 0.01, n)

        best_eta = 0.0
        best_obj_val = fast_autocorr(y).max()
        eta_candidates = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

        for eta in eta_candidates:
            scaled_eta = eta * 10.0 / (grad_norm + 1e-15)
            log_a = np.log(y) - scaled_eta * grad_centered
            log_a = log_a - log_a.max()
            a_trial = np.exp(log_a)
            a_trial = np.maximum(a_trial, 1e-25)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_obj_val:
                best_obj_val = M_trial
                best_eta = eta
                best_trial = a_trial.copy()

        if best_eta > 0:
            x_prev = x
            x = best_trial
        else:
            x = y + 0.005 * rng.normal(0, 1, n) * np.sqrt(y + 1e-10)
            x = norm_sum1(x)
            x_prev = x.copy()

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if verbose and t % 50 == 0:
            el = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LSE-MD t=" + str(t) + " (" + str(round(el, 1)) + "s): J=" + repr(round(J, 8)) + " tau=" + str(round(tau, 5)))

    J_final = 2.0 * n * best_m
    if verbose:
        print("    LSE-MD done: J=" + repr(round(J_final, 8)) + " time=" + str(round(time.monotonic() - t0, 1)) + "s")
    return best_a


def coordinate_descent(a0, time_limit, n, verbose=True):

                continue
            idx = np.arange(js, je + 1)

            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:

            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)

                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    j_vals = k2i - idx2
                    valid = (j_vals >= 0) & (j_vals < n)
                    if np.any(valid):
                        scores[idx2[valid]] += a[j_vals[valid]]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:6]:
                k2i = int(k2)

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)
            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])

            if dst == src:

def parametric_init(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = np.ones(n) / n
    best_val = fast_autocorr(best_a).max()

    candidates = []
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    for alpha in [0.5, 1.0, 2.0]:
        for beta in [0.5, 1.0, 2.0]:
            candidates.append(np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta)
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3, 0.5]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    for freq in [1, 2, 3, 5]:
        candidates.append(np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x)) / 2.0, 0.01))
    for rate in [1, 2, 5, 10]:
        candidates.append(np.exp(-rate * x) + np.exp(rate * (x - 1.0)))
    for _ in range(30):
        candidates.append(rng.random(n) ** rng.uniform(0.5, 4.0))

    for cand in candidates:
        a = norm_sum1(cand)
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    return best_a
