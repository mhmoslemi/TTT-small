def _spectral_waterfill(a, n_iters=10, strength=0.3):
    """
    Apply spectral water-filling to flatten the power spectrum.
    This helps flatten the autocorrelation by equalizing the frequency content.
    """
    n = len(a)
    a = _norm1(a.copy())
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for _ in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        
        # Water-filling: reduce peaks in power spectrum
        P_mean = P.mean()
        # Soft water-filling: cap the peaks
        P_flat = np.minimum(P, P_mean * (1.0 + strength))
        P_flat = np.maximum(P_flat, P_mean * (1.0 - strength))
        
        # Reconstruct with new magnitude, same phase
        phase = A / np.maximum(np.abs(A), 1e-30)
        A_new = np.sqrt(P_flat) * phase
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
    
    return a

def _targeted_coord_descent(a0, time_limit, verbose=True):

                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()

                    break

            stall += 1

    J = 2.0 * n * best_m
    if verbose:
        print(f"    TCD done ({elapsed:.0f}s): J={J:.6f}")
