def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def my_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgrad(a, conv, top_k=40, temperature=0.0005):
    n = len(a)
    conv_max = conv.max()
    log_w = (conv - conv_max) / temperature
    log_w -= log_w.max()
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    if top_k < len(conv):
        top_idx = np.argsort(w)[::-1][:top_k]
        w_new = np.zeros_like(conv)
        w_new[top_idx] = w[top_idx]
        ws = w_new.sum()
        if ws > 0:
            w_new /= ws
        w = w_new

    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            contrib = 2.0 * w[ki] * a[ki - idx]
            diag = (2 * idx == ki)
            if np.any(diag):
                contrib[diag] = w[ki] * a[idx[diag]]
            grad[idx] += contrib
    return grad

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def make_hann(n):
    x = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (x + 0.5) / n) ** 2
    return norm_sum1(a)


def make_flat_top(n):
    """5-term flat-top window."""

    a1 = 0.21557895

         - a4 * np.cos(6 * np.pi * x) + a5 * np.cos(8 * np.pi * x))
    w = w ** 2
    w = np.maximum(w, 0.0)
    return norm_sum1(w)


def make_blackman(n):
    x = np.arange(n, dtype=np.float64) / n
    w = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    return norm_sum1(np.maximum(w, 0.0))


def make_blackman_harris(n):
    x = np.arange(n, dtype=np.float64) / n
    w = (0.36358115 - 0.4891775 * np.cos(2 * np.pi * x)
         + 0.1365995 * np.cos(4 * np.pi * x) - 0.01064115 * np.cos(6 * np.pi * x))
    w = w ** 2
    return norm_sum1(np.maximum(w, 0.0))


def make_gaussian(n, sigma_frac=0.4):
    x = np.arange(n, dtype=np.float64) / n - 0.5
    sigma = sigma_frac * 0.5
    a = np.exp(-0.5 * (x / sigma) ** 2)
    return norm_sum1(a)


def make_rp_power(n, p=2.0):
    x = np.abs(np.arange(n, dtype=np.float64) / n - 0.5) * 2.0
    a = np.maximum(1.0 - x, 0.0) ** p
    return norm_sum1(a)


def make_asymmetric_gauss(n, center=0.4, sigma_frac=0.35, taper=2.0):
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-0.5 * ((x - center) / sigma_frac) ** 2)
    edge = 1.0 - taper * x * (1.0 - x)
    a = a * np.maximum(edge, 0.1)
    return norm_sum1(a)


def make_b_spline_like(n, order=4):
    """Approximate B-spline by repeated convolution of box car."""
    box = np.ones(max(2, order)) / order
    a = box.copy()
    for _ in range(order - 1):
        a = np.convolve(a, box, mode='full')
        # Center and truncate/pad to length n
        center = len(a) // 2
        half = n // 2
        start = max(0, center - half)
        end = start + n
        if end > len(a):
            a = np.pad(a, (start, end - len(a)))
        else:
            a = a[start:end]
    return norm_sum1(a)


def make_decreasing_power(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_sum1((n - x + 1) ** power)


def make_centered_window(n, center=0.5, width=0.3, power=2.0):

        t6a = min(remaining * 0.40, 180.0)
        if t6a > 10:
            a_in = norm_sum1(_best_tracker['seq'])
            a_out, _ = mirror_descent(a_in, t6a, top_k_init=60, temp_init=0.0002,
                                      c0_init=2.5, restart_every=8000, seed=2024, verbose=False)
            try_update(a_out, "md_final1")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6b = min(remaining * 0.35, 130.0)
            if t6b > 8:
                a_in = norm_sum1(_best_tracker['seq'])
                a_out, _ = coord_descent_dense(a_in, t6b, seed=27182, verbose=False)
                try_update(a_out, "cd_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t6c = min(remaining * 0.40, 100.0)
            if t6c > 8:
                a_in = norm_sum1(_best_tracker['seq'])
                a_out, _ = mirror_descent(a_in, t6c, top_k_init=80, temp_init=0.0001,
                                          c0_init=2.0, restart_every=10000, seed=31415, verbose=False)
                try_update(a_out, "md_final2")

    # Final verification
    if _best_tracker['seq'] is not None:
        fs = my_score(_best_tracker['seq'])
        if np.isfinite(fs):
            _best_tracker['score'] = fs
        print(f"\nFINAL: score={_best_tracker['score']:.6f}, n={len(_best_tracker['seq'])}, "
              f"t={time.monotonic()-t_global:.0f}s")
    else:
        n_fb = 4000
        _best_tracker['seq'] = make_hann(n_fb)
        _best_tracker['score'] = my_score(_best_tracker['seq'])
        print(f"Fallback: score={_best_tracker['score']:.6f}")

    return [float(v) for v in _best_tracker['seq']]
