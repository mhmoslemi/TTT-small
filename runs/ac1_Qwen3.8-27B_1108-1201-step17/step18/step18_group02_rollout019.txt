            stall = 0
            pert_scale = 0.05 + 0.03 * np.random.default_rng(t).uniform()
            pert = rng.normal(0, pert_scale, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            s = x.sum()
            if s > 1e-15:
                a = x / s
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            if verbose:
                elapsed = time.monotonic() - t_start
                print(f"    Smooth restart t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    Smooth t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}")
    
    elapsed = time.monotonic() - t_start
    if verbose:
        print(f"    Smooth done ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    return best_a, best_m


def _lp_frank_wolfe(a0, time_limit, solver, initial_K=800, max_K=8000,
                    add_per_iter=100, verbose=False, seed=7, use_away=False):
    """
    Enhanced LP Frank-Wolfe method for minimizing max(conv(a,a)).
    
    At each iteration:
    1. Compute current convolution and peak M
    2. Solve LP: max sum(b) s.t. b>=0, (y*b)[k]<=M for k in active set
    3. Take step towards LP solution with backtracking
    4. Update active constraint set
    """
    n = len(a0)
    y = _normalize(a0)
    ac = _fast_conv(y)
    M_curr = ac.max()
    
    best_a = y.copy()
    best_m = M_curr
    t_start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Initial constraint set: top-K positions of convolution
    init_idx = np.argsort(ac)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)
    
    iter_count = 0
    lp_fail_count = 0
    no_improve_count = 0
    last_best_m = best_m
    
    while time.monotonic() - t_start < time_limit:
        ac = _fast_conv(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Manage constraint set
        constraint_list = sorted(constraint_set)
        
        # Remove constraints that are far from active
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        # Also periodically remove very inactive constraints (even if under max_K)
        if len(constraint_list) > initial_K * 2:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            # Keep at least initial_K most active, plus some margin
            threshold = np.percentile(margins, 50)
            keep = np.where(margins <= threshold)[0]
            if len(keep) < initial_K:
                keep = np.argsort(margins)[:initial_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K < 10:
            break
        
        # Build constraint matrix
        # Row k: (y * b)[k] = sum_i y[k-i] * b[i] for valid i
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [coeffs @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail_count += 1
            if lp_fail_count > 10:
                # Reset constraint set
                constraint_set = set(int(k) for k in init_idx[:initial_K])
                lp_fail_count = 0
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail_count += 1
            if lp_fail_count > 15:
                constraint_set = set(int(k) for k in init_idx[:max(200, initial_K // 2)])
                lp_fail_count = 0
            continue
        
        lp_fail_count = 0
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-12:
            continue
        b_dir = b_val / b_sum
        
        # Backtracking line search
        best_alpha = 0.5
        best_obj = M_curr
        for alpha in [0.9, 0.7, 0.5, 0.3, 0.2, 0.1]:
            y_trial = (1.0 - alpha) * y + alpha * b_dir
            y_trial = np.maximum(y_trial, 0.0)
            s = y_trial.sum()
            if s < 1e-15:
                continue
            y_trial = y_trial / s
            M_trial = _fast_conv(y_trial).max()
            if M_trial < best_obj:
                best_obj = M_trial
                best_alpha = alpha
        
        # Apply best step
        alpha = best_alpha
        y_new = (1.0 - alpha) * y + alpha * b_dir
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s
        y = y_new
        
        # Add violated constraints
        ac_new = _fast_conv(y)
        violations = np.where(ac_new > M_curr + 1e-10)[0]
        if len(violations) > 0:
            # Add the most violated ones
            viol_amt = ac_new[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))
        
        iter_count += 1
        
        # Check for convergence
        if no_improve_count > 200 and iter_count > 100:
            # Try a restart with different initial constraints
            if verbose:
                print(f"    LP-FW restart at iter {iter_count}")
            ac = _fast_conv(y)
            new_init = np.argsort(ac)[::-1][:initial_K]
            # Mix with random
            random_idx = rng.integers(0, 2*n-1, size=initial_K//2)
            combined = np.concatenate([new_init, random_idx])
            constraint_set = set(int(k) for k in combined)
            no_improve_count = 0
        
        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t_start
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}, alpha={best_alpha:.2f}")
    
    elapsed = time.monotonic() - t_start
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={2*n*best_m:.6f}, iters={iter_count}")
    return best_a, best_m


def _irls_qp(a0, time_limit, solver, verbose=False):
    """
    Iteratively Reweighted optimization.
    At each step, solve a QP: minimize sum_k w[k] * (a*a)[k] s.t. a>=0, sum(a)=1
    where w[k] is concentrated on the peak positions.
    Then update w to focus more on the current peak.
    """
    n = len(a0)
    a = _normalize(a0)
    ac = _fast_conv(a)
    best_a = a.copy()
    best_m = ac.max()
    t_start = time.monotonic()
    
    # Initial weights: emphasize the peak
    p = ac / ac.sum()
    # Sharpen weights
    w = p ** 2
    w = w / w.sum()
    
    for iteration in range(1, 1000):
        if time.monotonic() - t_start > time_limit:
            break
        
        # Sharpen weights based on current peak
        ac = _fast_conv(a)
        M = ac.max()
        # Exponential weighting based on proximity to peak
        tau_w = max(M * 0.1, 1e-10)
        w = np.exp((ac - M) / tau_w)
        w = w / w.sum()
        
        # Solve QP: minimize sum_k w[k] * (a*a)[k] s.t. a >= 0, sum(a) = 1
        # (a*a)[k] = sum_i a[i] * a[k-i]
        # Objective = sum_k w[k] * sum_i a[i]*a[k-i] = a^T * Q * a
        # where Q[i,j] = w[i+j] (for valid i+j)
        
        # Build Q (Hankel-like matrix)
        # Q[i,j] = w[i+j] if i+j < len(w), else 0
        indices = np.arange(n)
        Q = w[np.ix_(indices + indices[:, None])]  # Q[i,j] = w[i+j]
        # Make symmetric (it should already be symmetric since w[i+j] = w[j+i])
        Q = (Q + Q.T) / 2.0
        
        # Solve QP using cvxpy
        a_var = cp.Variable(n, nonneg=True)
        obj = cp.quad_form(a_var, cp.psd_wrap(Q))
        constraints = [cp.sum(a_var) == 1]
        prob = cp.Problem(cp.Minimize(obj), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if verbose:
                print(f"    IRLS QP failed at iter {iteration}")
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        ac_new = _fast_conv(a_new)
        M_new = ac_new.max()
        
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and iteration % 5 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    IRLS iter={iteration} ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    
    elapsed = time.monotonic() - t_start
    if verbose:
        print(f"    IRLS done ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    return best_a, best_m
