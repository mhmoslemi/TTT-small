    return 2.0 * n * c.max() / (S * S)

def smooth_gradient(a, conv, tau):
    """Gradient of tau*log(sum(exp(conv_k/tau))) w.r.t. a."""
    n = len(a)
    c_shifted = conv - conv.max()
    p = np.exp(c_shifted / tau)
    p /= p.sum()
    # g[i] = 2 * sum_j a[j] * p[i+j]
    # = 2 * convolve(a_flip, p)[n-1 : 2n-1]
    a_flip = a[::-1].copy()
    total_len = n + len(p) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    Af = np.fft.rfft(a_flip, fft_size)
    Pf = np.fft.rfft(p, fft_size)
    full = np.fft.irfft(Af * Pf, fft_size)
    g = 2.0 * full[n - 1:2 * n - 1]
    return g

def norm1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    x = np.maximum(a, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return x / s

def sfw_smoothed(a0, time_limit, tau_decay=0.9997, tau_min_frac=0.003, verbose=True):
    """Smoothed Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm1(a0)
    y = a.copy()
    x = a.copy()
    best_a = a.copy()
    conv_init = fast_conv(a)
    best_m = conv_init.max()
    tau0 = conv_init.max() * 0.25
    tau_min = conv_init.max() * tau_min_frac
    t0 = time.monotonic()
    t = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        # Evaluate at extrapolated point x
        x_proj = project_simplex(x)
        conv = fast_conv(x_proj)
        M = conv.max()

        if M < best_m - 1e-14:
            best_m = M
            best_a = x_proj.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive tau
        tau = max(tau0 * (tau_decay ** t), tau_min)

        # Compute gradient at x_proj
        g = smooth_gradient(x_proj, conv, tau)

        # FW step: s = e_argmin(g)
        j = int(np.argmin(g))
        s = np.zeros(n)
        s[j] = 1.0

        # SFW update
        theta = 2.0 / (t + 2.0)
        y_new = (1.0 - theta) * y + theta * s
        y_new = project_simplex(y_new)

        if t > 0:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
        else:
            x_new = y_new.copy()
        x_new = project_simplex(x_new)

        y = y_new
        x = x_new
        t += 1

        # Restart if stalled
        if stall > 3000:
            rng = np.random.default_rng(t)
            perturb = rng.normal(0, 0.02, n) * best_a
            x_new = best_a + perturb
            x_new = project_simplex(x_new)
            y = x_new.copy()
            x = x_new.copy()
            stall = 0
            if verbose:
                print(f"    SFW restart t={t}: m={best_m:.8f}")

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SFW t={t} ({elapsed:.0f}s): obj={2*n*best_m:.6f}, tau={tau:.2e}")

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    SFW done ({elapsed:.0f}s): obj={2*n*best_m:.6f}")
    return best_a, best_m

def stagewise_fw(a0, time_limit, K=10, verbose=True):
    """Frank-Wolfe using stagewise subgradient (average of top-K lags)."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    conv = fast_conv(a)
    best_m = conv.max()
    t0 = time.monotonic()
    t = 0
    y = a.copy()
    x = a.copy()

    while time.monotonic() - t0 < time_limit:
        x_proj = project_simplex(x)
        conv = fast_conv(x_proj)
        M = conv.max()

        if M < best_m - 1e-14:
            best_m = M
            best_a = x_proj.copy()

        # Stagewise: use top-K lags with softmax weights
        top_k = np.argsort(conv)[::-1][:K]
        weights = np.zeros(len(conv))
        max_cv = conv[top_k]
        w = np.exp((max_cv - max_cv.max()) / (max_cv.max() * 0.1 + 1e-10))
        w /= w.sum()
        for i, k in enumerate(top_k):
            weights[k] = w[i]

        # Gradient = sum_k weights[k] * 2*a[k-i] (masked)
        # = 2 * sum_k weights[k] * a[(k-i)] where valid
        g = np.zeros(n)
        for k in top_k:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                g[i_start:i_end] += weights[k_int] * 2.0 * x_proj[k_int - np.arange(i_start, i_end)]

        # FW direction
        j = int(np.argmin(g))
        s = np.zeros(n)
        s[j] = 1.0

        theta = 2.0 / (t + 2.0)
        y_new = (1.0 - theta) * y + theta * s
        y_new = project_simplex(y_new)

        if t > 0:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
        else:
            x_new = y_new.copy()
        x_new = project_simplex(x_new)

        y = y_new
        x = x_new
        t += 1

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    StagFW t={t} ({elapsed:.0f}s): obj={2*n*best_m:.6f}")

    return best_a, best_m

def lp_fw(a0, time_limit, solver=None, initial_K=300, max_K=2500, verbose=True):
    """LP-based Frank-Wolfe with dynamic constraint management."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    conv_init = fast_conv(a)
    best_m = conv_init.max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    # Initial constraint set: top K lags
    init_idx = np.argsort(conv_init)[::-1][:min(initial_K, 2 * n - 1)]
    constraint_set = list(int(k) for k in init_idx)
    constraint_dict = {int(k): i for i, k in enumerate(constraint_set)}

    if solver is None:
        solver = cp.CLARABEL

    t = 0
    lp_count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        x_proj = project_simplex(x)
        conv_y = fast_conv(y)
        conv_x = fast_conv(x_proj)

        if conv_y.max() < best_m - 1e-14:
            best_m = conv_y.max()
            best_a = y.copy()
        if conv_x.max() < best_m - 1e-14:
            best_m = conv_x.max()
            best_a = x_proj.copy()

        # Check for violations and add constraints
        violations = np.where(conv_y > best_m * 0.95)[0]
        new_constr = 0
        for k in violations[:30]:
            k_int = int(k)
            if k_int not in constraint_dict and len(constraint_set) < max_K:
                constraint_set.append(k_int)
                constraint_dict[k_int] = len(constraint_set) - 1
                new_constr += 1

        # Trim constraint set if too large
        if len(constraint_set) > max_K:
            # Keep only constraints with tightest slack
            slacks = []
            for k in constraint_set:
                slacks.append(conv_y[k] - conv_y.min())
            sorted_idx = np.argsort(slacks)[::-1][:max_K]
            constraint_set = [constraint_set[i] for i in sorted_idx]
            constraint_dict = {k: i for i, k in enumerate(constraint_set)}

        K = len(constraint_set)
        if K == 0:
            break

        # Build LP: min M s.t. sum(g)=1, g>=0, (y*g)[k] <= M for k in constraint_set
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1]

        # Build constraint matrix rows
        y_copy = y.copy()
        for k in constraint_set:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                row = np.zeros(n)
                row[idx] = y_copy[k_int - idx]
                constraints.append(row @ g_var <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if lp_count % 50 == 0:
                # Reduce constraints
                constraint_set = constraint_set[:max(100, K // 2)]
                constraint_dict = {k: i for i, k in enumerate(constraint_set)}
            lp_count += 1
            continue

        lp_count += 1
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = g_val.sum()
        if gs < 1e-15:
            g_dir = np.ones(n) / n
        else:
            g_dir = g_val / gs

        # FW update
        theta = 2.0 / (t + 2.0)
        y_new = (1.0 - theta) * y + theta * g_dir
        y_new = project_simplex(y_new)

        if t > 0:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
        else:
            x_new = y_new.copy()
        x_new = project_simplex(x_new)

        # Check gap for stall detection
        lp_obj = M_var.value
        gap = abs(lp_obj - conv_y[g_dir.argmax()] * 0 + 0)  # simplified
        conv_g = fast_conv(g_dir)
        gap = abs(np.dot(y, conv_g) - lp_obj)  # crude gap

        if lp_obj < best_m * 0.999 or gap < 1e-7:
            stall = 0
        else:
            stall += 1

        y = y_new
        x = x_new
        t += 1

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW t={t} ({elapsed:.0f}s): obj={2*n*best_m:.6f}, K={len(constraint_set)}")

    return best_a, best_m

def peak_shaving(a0, time_limit, verbose=True):
    """Direct peak-shaving: transfer mass from peak contributors to non-peak areas."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        conv = fast_conv(a)
        M = conv.max()

        # Find top peaks
        n_peaks = min(10, 2 * n - 1)
        top_lags = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for k in top_lags[:5]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue

            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            # Find best destination: minimize impact on all top peaks
            scores = np.zeros(n)
            for k2 in top_lags[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                idx2 = np.arange(s2, e2)
                if len(idx2) > 0:
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            # Prefer positions with low score and some mass
            min_score = scores.min()
            max_score = scores.max()
            if max_score - min_score < 1e-20:
                candidates = np.arange(n)
            else:
                threshold = min_score + 0.3 * (max_score - min_score)
                candidates = np.where(scores <= threshold)[0]

            if len(candidates) == 0:
                candidates = np.arange(n)

            dst = candidates[np.argmin(scores[candidates])]
            if dst == src or a[src] < 1e-15:
                continue

            # Try transfer
            for frac in [0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = project_simplex(a_trial)
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = project_simplex(a)
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        if verbose and stall > 0 and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave ({elapsed:.0f}s): obj={2*n*best_m:.6f}, stall={stall}")

    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Resample sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def parametric_init(n, seed=42):
    """Generate diverse parametric initial sequences."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []

    # Sine-squared family
    for freq in [1, 2, 3]:
        for phase in [0.0, 0.2, 0.5, 0.8]:
            a = np.maximum(np.sin(np.pi * freq * x + phase) ** 2, 0.001)
            candidates.append(a / a.sum())

    # Raised cosine
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    candidates.append(a / a.sum())

    # Power law
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = (i_arr + 1) ** p
        candidates.append(a / a.sum())
        a = (n - i_arr) ** p
        candidates.append(a / a.sum())
        a = (np.abs(i_arr - n / 2) + 1) ** p
        candidates.append(a / a.sum())

    # Two bumps
    for c1 in [0.2, 0.3, 0.4]:
        for c2 in [c1 + 0.2, c1 + 0.3, c1 + 0.5]:
            if c2 < 1.0:
                for sig in [0.05, 0.1, 0.15]:
                    a = np.exp(-((x - c1) ** 2) / (2 * sig ** 2)) + np.exp(-((x - c2) ** 2) / (2 * sig ** 2))
                    candidates.append(a / a.sum())

    # Gaussian
    for c in [0.3, 0.5, 0.7]:
        for sig in [0.08, 0.15, 0.25]:
            a = np.exp(-((x - c) ** 2) / (2 * sig ** 2))
            candidates.append(a / a.sum())

    # Modulated
    for freq in [5, 10, 20, 50]:
        a = np.maximum(np.sin(np.pi * x) + 0.3 * np.sin(2 * np.pi * freq * x), 0.001)
        candidates.append(a / a.sum())

    # Random smooth
    for _ in range(5):
        pts = rng.random(8)
        a = np.interp(x, np.linspace(0, 1, 8), pts)
        a = np.maximum(a, 0.001)
        candidates.append(a / a.sum())

    # Uniform
    candidates.append(np.ones(n) / n)

    return candidates

def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(v) for v in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def get_solver():
    """Find a working LP solver."""
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.SCS

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent = None
    parent_n = 0
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 100 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            parent_n = len(parent)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm1(parent)
                best_score = p_score
                print(f"Parent: n={parent_n}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        best_seq = norm1(np.sin(np.pi * x) ** 2)
        best_score = full_obj(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = full_obj(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Diverse initialization ===
    n0 = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.05, 60.0)
    print(f"\n=== Phase 1: Init search (n={n0}) ===")
    if t1 > 5:
        candidates = parametric_init(n0, seed=99)
        for i, cand in enumerate(candidates):
            if time.monotonic() - t_global > time_budget - t1:
                break
            sc = full_obj(cand)
            if sc < best_score:
                try_update(cand, f"param{i}")

    n_best = len(best_seq)

    # === Phase 2: SFW ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: SFW (n={n_best}) ===")
    if t2 > 30:
        a_in = norm1(best_seq)
        a_out, _ = sfw_smoothed(a_in, t2, tau_decay=0.9998, tau_min_frac=0.005, verbose=True)
        try_update(a_out, f"sfw_n{n_best}")

    # === Phase 3: LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.25, 420.0)
    if t3 > 30 and n_best <= 6000:
        a_in = norm1(best_seq)
        a_out, _ = lp_fw(a_in, t3, solver=solver, initial_K=200, max_K=2000, verbose=True)
        try_update(a_out, f"lpfw_n{n_best}")

    # === Phase 4: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Peak shaving (n={n_best}) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        a_in = norm1(best_seq)
        a_out, _ = peak_shaving(a_in, t4, verbose=True)
        try_update(a_out, f"shave_n{n_best}")

    # === Phase 5: Stagewise FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Stagewise FW (n={n_best}) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 20:
        a_in = norm1(best_seq)
        a_out, _ = stagewise_fw(a_in, t5, K=8, verbose=True)
        try_update(a_out, f"stagew_n{n_best}")

    # === Phase 6: SFW again (refinement) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: SFW refine (n={n_best}) ===")
    t6 = min(remaining * 0.15, 250.0)
    if t6 > 20:
        a_in = norm1(best_seq)
        a_out, _ = sfw_smoothed(a_in, t6, tau_decay=0.9999, tau_min_frac=0.001, verbose=True)
        try_update(a_out, f"sfw2_n{n_best}")

    # === Phase 7: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    nearby = []
    for delta in [400, -400, 800, -800, 1200]:
        new_n = n_best + delta
        if 2000 <= new_n <= 8000:
            nearby.append(new_n)

    seen = set()
    unique_near = []
    for ln in nearby:
        if ln not in seen:
            seen.add(ln)
            unique_near.append(ln)

    for new_n in unique_near[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print(f"\n=== Phase 7: n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.3, 120.0)
        if t7a > 15:
            a_out, _ = sfw_smoothed(a_long, t7a, tau_decay=0.9999, tau_min_frac=0.002, verbose=False)
            try_update(a_out, f"sfw_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 5500:
            t7b = min(remaining * 0.3, 100.0)
            if t7b > 15:
                a_out, _ = lp_fw(norm1(best_seq), t7b, solver=solver, initial_K=200, max_K=1500, verbose=False)
                try_update(a_out, f"lpfw_n{new_n}")

    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish (n={n_best}) ===")
    if remaining > 20:
        t8 = min(remaining * 0.4, 80.0)
        if t8 > 10:
            a_in = norm1(best_seq)
            a_out, _ = peak_shaving(a_in, t8, verbose=False)
            try_update(a_out, f"shave_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm1(np.sin(np.pi * x) ** 2)
        best_score = full_obj(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(v) for v in best_seq]
