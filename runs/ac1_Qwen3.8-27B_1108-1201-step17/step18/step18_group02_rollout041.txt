import numpy as np
import time
import math

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a[~np.isfinite(a)] = 0.0
    a[a < 0.0] = 0.0
    s = float(a.sum())
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s

def fast_convolution(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        v = float(a[0]) * float(a[0])
        return np.array([v], dtype=np.float64)
    size = 2*n - 1
    nfft = next_pow2(size)
    A = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(A * A, n=nfft)[:size]
    c[c < 0.0] = 0.0
    s = float(c.sum())
    if s > 0.0:
        c = c / s
    return c

def internal_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    q = fast_convolution(a)
    if q.size == 0:
        return float('inf')
    return float(2.0 * float(a.size) * float(q.max()))

def evaluate_safe(a):
    try:
        arr = np.asarray(a, dtype=np.float64)
        vals = [float(x) for x in arr]
        sc = evaluate_sequence(vals)
        return float(sc)
    except Exception:
        return float('inf')

def cross_state(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    lw = int(w.size)
    if lw == 0:
        return np.zeros(n, dtype=np.float64)
    size = n + lw - 1
    nfft = next_pow2(size)
    Af = np.fft.rfft(a[::-1], n=nfft)
    Wf = np.fft.rfft(w, n=nfft)
    c = np.fft.irfft(Af * Wf, n=nfft)
    return c[n-1:n-1+n]

def interpolate_to(a, m):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.ones(m, dtype=np.float64) / max(1, m)
    if n == m:
        return normalize(a)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    y = np.interp(x_new, x_old, a)
    return normalize(y)

"""
Searches for nonnegative step-function heights minimizing
    2 * n * max(conv(a,a)) / (sum(a))^2.

The main optimizer is a fast log-domain multiplicative update that actively
balances high and low convolution coefficients:
  - high lags are down-weighted,
  - low lags are up-weighted,
  - the update is damped and accepted with a decaying temperature.

This is combined with:
  - multiple parametric/random initializations,
  - parent-sequence continuation when available,
  - length interpolation and local length exploration,
  - an active-set two-point transfer polish,
  - final validation against the provided evaluator.
"""

import numpy as np
import time
import math


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a[~np.isfinite(a)] = 0.0
    a[a < 0.0] = 0.0
    s = float(a.sum())
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def fast_convolution(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        v = float(a[0]) * float(a[0])
        return np.array([v], dtype=np.float64)
    if n <= 128:
        c = np.convolve(a, a)
        c = np.asarray(c, dtype=np.float64)
        c[c < 0.0] = 0.0
        s = float(c.sum())
        if s > 0.0:
            c = c / s
        return c

    size = 2 * n - 1
    nfft = next_pow2(size)
    A = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(A * A, n=nfft)[:size]
    c[c < 0.0] = 0.0
    s = float(c.sum())
    if s > 0.0:
        c = c / s
    return c


def internal_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    q = fast_convolution(a)
    if q.size == 0:
        return float("inf")
    return float(2.0 * float(a.size) * float(q.max()))


def evaluate_safe(a):
    try:
        arr = np.asarray(a, dtype=np.float64)
        vals = [float(x) for x in arr]
        sc = evaluate_sequence(vals)
        return float(sc)
    except Exception:

def cross_state(a, w):
    """
    Returns S_i = sum_j a_j w_{i+j} for i=0..n-1, where w has length m.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    lw = int(w.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    if lw == 0:
        return np.zeros(n, dtype=np.float64)

    size = n + lw - 1
    nfft = next_pow2(size)
    Af = np.fft.rfft(a[::-1], n=nfft)
    Wf = np.fft.rfft(w, n=nfft)
    c = np.fft.irfft(Af * Wf, n=nfft)
    return c[n - 1:n - 1 + n]


def interpolate_to(a, m):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if m <= 0:
        return np.ones(1, dtype=np.float64)
    if n == 0:
        return np.ones(m, dtype=np.float64) / float(m)
    if n == m:
        return normalize(a)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    y = np.interp(x_new, x_old, a)
    return normalize(y)


def make_starts(n, parent_arr=None, seed=0):
    if n <= 0:
        return [np.ones(1, dtype=np.float64)]

    out = []

    if parent_arr is not None:
        p = np.asarray(parent_arr, dtype=np.float64)
        if p.size > 0:
            p = np.maximum(p, 0.0)
            ps = float(p.sum())
            if ps > 1e-15:
                p = p / ps
                if p.size != n:
                    p = interpolate_to(p, n)
                out.append(normalize(p))

    x = np.linspace(0.0, 1.0, n)
    idx = np.arange(n, dtype=np.float64)

    out.append(normalize(np.ones(n, dtype=np.float64)))

    s = np.sin(math.pi * x)
    out.append(normalize(s * s))

    tri = 1.0 - np.abs(2.0 * x - 1.0)
    out.append(normalize(tri))

    for e in (0.25, 0.5, 1.0):
        u = np.power(idx + 1.0, e) * np.power(n - idx, e)
        out.append(normalize(u))

    out.append(normalize(np.power(idx + 1.0, 0.5)))
    out.append(normalize(np.power(n - idx, 0.5)))

    for f in (1, 2, 3):
        for ph in (0.0, 0.25, 0.5, 0.75):
            a = 0.1 + 0.9 * np.abs(np.sin(2.0 * math.pi * f * x + ph * math.pi)) ** 2.0
            out.append(normalize(a))

    rng = np.random.default_rng(seed)

    r = np.abs(rng.normal(size=n))
    win = max(1, n // 100)
    kern = np.ones(win, dtype=np.float64) / float(win)
    rs = np.convolve(r, kern, mode="same")
    out.append(normalize(rs))

    r2 = rng.beta(0.5, 0.5, size=n)
    out.append(normalize(r2))

    return out


def balance_optimize(a0, budget_s, seed=0, alt_list=None,
                     start_eta=0.25, min_eta=0.008, max_eta=0.60,
                     rho_start=1.0, rho_end=2.5, verbose=False):
    t0 = time.monotonic()

    a = normalize(a0)
    n = int(a.size)
    if n <= 0:
        return a, float("inf")
    if budget_s <= 1.0:
        q = fast_convolution(a)
        return a, float(2.0 * n * q.max())

    q = fast_convolution(a)
    cur_score = float(2.0 * n * q.max())
    best_a = a.copy()
    best_score = cur_score

    eta = start_eta
    rho = rho_start
    stall = 0
    rng = np.random.default_rng(seed)
    it = 0
    last_print = t0

    while True:
        rem = budget_s - (time.monotonic() - t0)
        if rem <= 0.5:
            break

        M = float(q.max())
        if M <= 1e-300 or not np.isfinite(M):
            break

        a = np.maximum(a, 1e-18)
        sa = float(a.sum())
        if sa <= 1e-15:
            a = best_a.copy()
        else:
            a = a / sa

        z = q / M
        z = np.clip(z, 0.0, 1.0)

        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(1e-9, budget_s))

        target_rho = rho_start + (rho_end - rho_start) * frac
        rho = 0.9 * rho + 0.1 * target_rho

        ph = np.power(z, rho)
        ph_s = float(ph.sum())
        if ph_s <= 1e-300:
            ph = np.ones_like(z, dtype=np.float64) / float(z.size)
        else:
            ph = ph / ph_s

        v = 1.0 - z
        v = np.maximum(v, 1e-12)
        pl = np.power(v, rho)
        pl_s = float(pl.sum())
        if pl_s <= 1e-300:
            pl = np.ones_like(z, dtype=np.float64) / float(z.size)
        else:
            pl = pl / pl_s

        uni = 1.0 / float(z.size)

        ph = 0.03 * uni + 0.97 * ph
        ph_s = float(ph.sum())
        if ph_s > 0.0:
            ph = ph / ph_s
        else:
            ph = uni

        pl = 0.03 * uni + 0.97 * pl
        pl_s = float(pl.sum())
        if pl_s > 0.0:
            pl = pl / pl_s
        else:
            pl = uni

        W = pl - ph
        S = cross_state(a, W)
        S = S * float(n)
        S = S - float(np.mean(S))
        smax = float(np.max(np.abs(S)))

        if smax < 1e-9 or not np.isfinite(smax):
            stall = 200
        else:
            update = eta * (S / smax)
            maxupd = 2.0 * eta
            np.clip(update, -maxupd, maxupd, out=update)

            loga = np.log(np.maximum(a, 1e-300)) + update
            loga -= float(np.max(loga))
            a2 = np.exp(loga)
            a2 = np.where(np.isfinite(a2), a2, 0.0)
            a2 = np.maximum(a2, 1e-18)
            s2 = float(a2.sum())
            if s2 < 1e-15:
                a2 = a.copy()
            else:
                a2 = a2 / s2

            q2 = fast_convolution(a2)
            score2 = float(2.0 * n * q2.max())
            if not np.isfinite(score2):
                score2 = float("inf")

            temp = 1e-7
            if elapsed < budget_s:
                temp = max(1e-7,
                           0.0012 * (1.0 - frac) +
                           0.02 * max(0.0, cur_score - best_score))

            accepted = False
            if np.isfinite(score2) and score2 <= cur_score + temp:
                if not (elapsed > 0.6 * budget_s and score2 > best_score + 0.0004):
                    accepted = True

            if accepted:
                a = a2
                q = q2
                cur_score = score2
                eta = min(max_eta, eta * 1.02)
                if score2 < best_score - 1e-12:
                    best_score = score2
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1

                if verbose and (it % 2000 == 0 or (time.monotonic() - last_print) > 15.0):
                    print("    balance it=%d t=%.0f cur=%.6f best=%.6f eta=%.3f rho=%.2f"
                          % (it, elapsed, cur_score, best_score, eta, rho))
                    last_print = time.monotonic()
            else:
                eta = max(min_eta, eta * 0.7)
                stall += 1

        if stall > 120:
            if alt_list is not None and len(alt_list) > 0:
                alt = alt_list[(it // 100) % len(alt_list)]
                alt = normalize(alt)
                w = 0.80 + 0.20 * float(rng.random())
                mixed = w * best_a + (1.0 - w) * alt
            else:
                mixed = best_a.copy()

            logb = np.log(np.maximum(mixed, 1e-18))
            sigma = 0.08 + 0.08 * (1.0 - frac)
            logb = logb + rng.normal(0.0, sigma, size=n)
            logb -= float(np.max(logb))

            a = np.exp(logb)
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.maximum(a, 1e-18)
            s = float(a.sum())
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s

            q = fast_convolution(a)
            cur_score = float(2.0 * n * q.max())

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()

            eta = max(start_eta * 0.5, min_eta)
            rho = max(rho_start, rho * 0.8)
            stall = 0

        it += 1

    return best_a, float(best_score)


def transfer_polish(a0, budget_s, seed=555, verbose=False):

    return best_a, float(best_score)
