def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_J(a):
    """Compute 2n*max(autocorr) for normalized a."""
    ac = fast_autocorr(a)
    return 2.0 * len(a) * ac.max()


def soft_max_weights(ac, beta):
    """Compute softmax weights over the autocorrelation values."""
    ac_max = ac.max()
    shifted = ac - ac_max
    exp_vals = np.exp(beta * shifted)
    w = exp_vals / exp_vals.sum()
    return w


def compute_soft_gradient(y, ac, beta, n):
    """

    grad[j] = sum_k w_k * (a[k-j] + a[j-(k-j)] * indicator)
    For (a*a)[k] = sum_i a[i]*a[k-i]:

    # Only use top values where w is significant

    active = np.where(w > threshold)[0]
    
    grad = np.zeros(n)

        wk = w[k]
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        partners = k_int - idx
        grad[idx] += 2.0 * wk * y[partners]
    # Fix: for positions where 2*j == k, we double-counted
    # Actually the formula d/dx_j [sum_i a[i]*a[k-i]] = a[k-j] + a[k-j] = 2*a[k-j]
    # unless k-j == j, in which case it's just a[j]
    # But in our sum, when i=j and k-i=j, these are the same term, counted once.
    # Let me redo: (a*a)[k] = sum_i a[i]*a[k-i]
    # d/dx_j: terms where either index equals j
    # i=j: a[j]*a[k-j], derivative contribution: a[k-j]
    # k-i=j, i=k-j: a[k-j]*a[j], derivative contribution: a[j]
    # If j != k-j: total = 2*a[k-j] (= 2*a[j] when they're the same... no)
    # If j == k-j (i.e., 2j=k): only one term, contribution = a[j]
    
    # My formula above: for each k, for each i in valid range, idx[i] gets 2*w_k*y[k-idx[i]]
    # This double-counts when idx[i] == k-idx[i], i.e., 2*idx[i] == k
    # Fix: subtract w_k*y[j] for positions where 2j==k
    
    for k in active:
        wk = w[k]
        k_int = int(k)
        if k_int % 2 == 0:
            j = k_int // 2
            if 0 <= j < n:
                grad[j] -= wk * y[j]
    
    return grad


def nesterov_soft_descent(a0, time_limit, beta_start=5.0, beta_end=50.0,
                           c0=2.0, max_iters=500000, verbose=True):
    """Nesterov-accelerated descent on soft approximation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    y = a.copy()
    t = 1
    gamma = c0
    
    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Time-varying beta (annealing)
        frac = min(1.0, (it - 1) / (max_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Nesterov momentum
        t_new = t + 1
        mu = (t - 0.5 + np.sqrt(2.25 + 2.0 * t * t)) / (t + 0.5 + np.sqrt(2.25 + 2.0 * (t_new - t + t) * (t_new - t + t)))
        # Simplified Nesterov: mu = (1 + sqrt(1 + 4*t^2)) / (2*(1 + t)) ≈ 1 - 1/(t+1)
        mu = (t - 1.0) / t if t > 1 else 0.0
        y = x + mu * (x - y)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        # Compute convolution and gradient
        ac = fast_autocorr(y)
        grad = compute_soft_gradient(y, ac, beta, n)
        
        # Learning rate
        eta = gamma / (np.sqrt(it) + 1.0)
        
        # Update
        x_new = y - eta * grad
        x_new = project_simplex(x_new, 1.0)
        
        x = x_new
        t += 1
        
        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = x.copy()
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Nesterov it=" + str(it) + " (" + str(elapsed) + "s): J=" + repr(J) + ", beta=" + repr(beta))
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    Nesterov done (" + str(elapsed) + "s): J=" + repr(J))
    return best_a, best_m
