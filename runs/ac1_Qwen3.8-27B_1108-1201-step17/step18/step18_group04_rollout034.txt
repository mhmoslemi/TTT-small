def fast_conv(a):
    """Fast full convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize a to be non-negative with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score_from_conv(a, conv):
    """Compute the score from a and its precomputed convolution."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * conv.max() / (s * s)


def compute_grad_peak(a, k):
    """Compute gradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return g
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


def compute_subgrad(a, conv, n_peaks=10, temperature=None):
    """Compute subgradient of max(conv) using softmax-weighted peaks."""
    conv_max = conv.max()
    if temperature is None:
        temperature = 0.001  # very sharp
    diff = (conv - conv_max) / temperature
    diff = np.minimum(diff, 0)  # only peaks near max
    w = np.exp(diff)
    w_sum = w.sum()
    if w_sum < 1e-30:
        # Just use the argmax
        k = np.argmax(conv)
        return compute_grad_peak(a, int(k))
    w = w / w_sum

    top_w = top_w / top_w.sum()
    
    for wi, k in zip(top_w, top_idx):
        g += wi * compute_grad_peak(a, int(k))
    return g

def interpolate_seq(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive mirror descent with per-step step-size selection
2. Sharper subgradient focusing on top peaks with adaptive temperature
3. Decaying momentum for better late convergence
4. Targeted coordinate descent with peak-avoidance logic
5. Multi-phase time allocation with diverse initializations
6. Fine-grained polishing with very small moves on active peaks

import numpy as np
import time

_best_tracker = {'seq': None, 'score': float('inf')}


def fast_conv(a):
    """Fast full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def grad_at_peak(a, k):
    """Gradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return g
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


def compute_subgrad_v2(a, conv, n_peaks=6, temperature=0.001):
    """Subgradient with sharp softmax over top peaks."""
    conv_max = conv.max()
    # Use relative threshold
    threshold = conv_max - max(temperature * 10.0, conv_max * 0.001)
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    if len(active) > n_peaks:
        order = np.argsort(conv[active])[::-1][:n_peaks]
        active = active[order]
    # Weight by how close to max
    diffs = conv_max - conv[active]
    max_diff = diffs.max()
    if max_diff < 1e-15:
        w = np.ones(len(active))
    else:
        w = np.exp(-diffs / max(temperature, max_diff * 0.1))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active))
    else:
        w = w / w_sum
    g = np.zeros(len(a))
    for wi, k in zip(w, active):
        g += wi * grad_at_peak(a, int(k))
    return g


def amd_adaptive(a0, time_limit, seed=42, verbose=False):
    """
    Adaptive Mirror Descent:
    - Tries multiple step sizes per iteration
    - Uses sharp subgradient
    - Decaying momentum
    - Adaptive restarts
    """
    n = len(a0)
    x = norm_sum1(a0)
    best_x = x.copy()
    best_conv = fast_conv(x)
    best_m = best_conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Parameters
    base_eta = 2.0
    eta_schedules = [0.2, 0.5, 1.0, 2.0, 5.0]
    n_peaks = 6
    temperature = 0.001
    momentum = 0.8
    stall_count = 0
    restart_every = 2500
    restart_count = 0
    step = 0
    y = x.copy()  # momentum point

    while time.monotonic() - t0 < time_limit:
        # Compute subgradient at momentum point
        conv_y = fast_conv(y)
        g = compute_subgrad_v2(y, conv_y, n_peaks=n_peaks, temperature=temperature)

        # Try multiple step sizes
        best_m_trial = best_m
        best_x_trial = x.copy()
        current_m = fast_conv(x).max()

        for em in eta_schedules:
            eta = base_eta * em
            log_update = -eta * g
            max_lu = np.max(log_update)
            a_trial = x * np.exp(log_update - max_lu)
            a_trial = np.maximum(a_trial, 1e-25)
            s_t = a_trial.sum()
            if s_t < 1e-12:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m_trial:
                best_m_trial = m_trial
                best_x_trial = a_trial.copy()

        x = best_x_trial
        step += 1

        # Update momentum (decaying)
        mom = momentum * (0.999 ** step)
        y = mom * y + (1.0 - mom) * x

        # Track best
        if best_m_trial < best_m - 1e-16:
            best_m = best_m_trial
            best_x = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Decay base eta slowly
        base_eta *= 0.9998

        # Restart logic
        if stall_count >= restart_every:
            stall_count = 0
            restart_count += 1
            # Perturb best solution
            strength = min(0.08, 0.02 + 0.01 * min(restart_count, 8))
            pert = rng.normal(0, strength, n) * best_x
            x_r = best_x + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            y = x_r.copy()
            # Reset parameters
            base_eta = max(0.5, 2.0 / (1.0 + 0.2 * restart_count))
            temperature = max(0.0003, 0.001 * (0.85 ** restart_count))
            n_peaks = min(12, 6 + restart_count)
            restart_every = max(1500, 2500 - 100 * restart_count)

    return best_x, best_m


def amd_fine(a0, time_limit, seed=42, verbose=False):
    """
    Fine-grained AMD with very small steps, no momentum.
    Focuses on the single worst peak.
    """
    n = len(a0)
    x = norm_sum1(a0)
    best_x = x.copy()
    best_m = fast_conv(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = 0.3
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv(x)
        k_peak = int(np.argmax(conv))

        # Gradient at the single worst peak
        g = grad_at_peak(x, k_peak)

        # Also include a small contribution from the 2nd and 3rd peaks
        top_k = np.argsort(conv)[::-1]
        for j in range(1, min(3, len(top_k))):
            k2 = int(top_k[j])
            if conv[k2] > conv[k_peak] * 0.95:
                g += 0.3 * grad_at_peak(x, k2)

        # Mirror descent step
        for em in [0.3, 0.6, 1.0, 2.0, 4.0]:
            eta_try = eta * em
            log_upd = -eta_try * g
            max_lu = np.max(log_upd)
            a_trial = x * np.exp(log_upd - max_lu)
            a_trial = np.maximum(a_trial, 1e-25)
            s_t = a_trial.sum()
            if s_t < 1e-12:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-16:
                best_m = m_trial
                best_x = a_trial.copy()
                x = a_trial
                stall = 0
                break

        if stall > 0:
            stall += 1
            if stall % 500 == 0:
                # Tiny random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and x[i] > 1e-10:
                    amt = x[i] * 0.002
                    x[i] -= amt
                    x[j] += amt
                    x = np.maximum(x, 0.0)
                    x = x / x.sum()
                    m_new = fast_conv(x).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_x = x.copy()
                        stall = 0

    return best_x, best_m


def coord_descent_targeted(a0, time_limit, seed=42, verbose=False):
    """
    Coordinate descent that targets the worst peaks.
    For each peak, finds the largest contributing pair and tries to
    move mass to a position that minimizes impact on all active peaks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        conv = fast_conv(a)
        n_top = min(8, 2 * n - 1)
        top_peaks = np.argsort(conv)[::-1][:n_top]

        improved = False
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            lo = max(0, k_int - n + 1)
            hi = min(n - 1, k_int)
            idx = np.arange(lo, hi + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_int - idx]

            # Top contributors to this peak
            n_src = max(3, min(10, len(idx) // 10))
            top_src = idx[np.argsort(contrib)[::-1][:n_src]]

            for src in top_src:
                src_i = int(src)
                if a[src_i] < 1e-12:
                    continue

                # Compute cost vector: how much each position contributes to top peaks
                costs = np.zeros(n)
                for k2 in top_peaks:
                    k2i = int(k2)
                    lo2 = max(0, k2i - n + 1)
                    hi2 = min(n - 1, k2i)
                    idx2 = np.arange(lo2, hi2 + 1)
                    if len(idx2) > 0:
                        costs[idx2] += a[idx2] * a[k2i - idx2]

                # Best destinations (lowest cost)
                cands = np.argsort(costs + 1e-12 * np.log1p(a))
                cands = [int(c) for c in cands if int(c) != src_i][:12]

                for dst in cands:
                    found = False
                    for frac in [0.02, 0.05, 0.1, 0.15, 0.25, 0.4, 0.6]:
                        amt = a[src_i] * frac
                        a_trial = a.copy()
                        a_trial[src_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-12:
                            continue
                        a_trial = a_trial / s_t
                        m_trial = fast_conv(a_trial).max()
                        if m_trial < best_m - 1e-15:
                            a = a_trial
                            best_m = m_trial
                            best_a = a.copy()
                            improved = True
                            found = True
                            stall = 0
                            break
                    if found:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random small move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.04)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_conv(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def pair_swap_optimize(a0, time_limit, seed=42):
    """
    Optimize by considering pairs of coordinates (i,j) and finding optimal
    mass transfer between them. Uses golden section search on the fraction.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Precompute which positions are important (high mass)
    # and which peaks are active

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv(a)
        top_peaks = np.argsort(conv)[::-1][:5]

        # Get top mass positions
        top_mass = np.argsort(a)[::-1][:30]

        improved = False
        for i in top_mass:
            if time.monotonic() - t0 > time_limit:
                break
            i_int = int(i)
            if a[i_int] < 1e-10:
                continue

            # Compute cost for each destination
            costs = np.zeros(n)
            for k2 in top_peaks:
                k2i = int(k2)
                lo2 = max(0, k2i - n + 1)
                hi2 = min(n - 1, k2i)
                idx2 = np.arange(lo2, hi2 + 1)
                if len(idx2) > 0:
                    costs[idx2] += a[idx2] * a[k2i - idx2]

            # Try best low-cost destinations
            cands = np.argsort(costs)[:50]
            cands = [int(c) for c in cands if int(c) != i_int][:8]

            for j in cands:
                j_int = int(j)
                # Golden section search on fraction
                best_frac = None
                best_val = best_m
                lo_frac = 0.01
                hi_frac = min(0.8, a[i_int] / max(a[j_int], 1e-15) * 0.5)
                if hi_frac <= lo_frac:
                    continue
                for _ in range(8):
                    m1 = hi_frac - (hi_frac - lo_frac) * 0.382
                    m2 = lo_frac + (hi_frac - lo_frac) * 0.382
                    for frac in [m1, m2]:
                        amt = a[i_int] * frac
                        a_t = a.copy()
                        a_t[i_int] -= amt
                        a_t[j_int] += amt
                        a_t = np.maximum(a_t, 0.0)
                        s_t = a_t.sum()
                        if s_t < 1e-12:
                            continue
                        a_t = a_t / s_t
                        m_t = fast_conv(a_t).max()
                        if m_t < best_val:
                            best_val = m_t
                            best_frac = frac
                    # Narrow interval
                    a_t1 = a.copy()
                    amt1 = a[i_int] * m1
                    a_t1[i_int] -= amt1
                    a_t1[j_int] += amt1
                    a_t1 = np.maximum(a_t1, 0.0)
                    a_t1 = a_t1 / a_t1.sum()
                    m1_val = fast_conv(a_t1).max()

                    a_t2 = a.copy()
                    amt2 = a[i_int] * m2
                    a_t2[i_int] -= amt2
                    a_t2[j_int] += amt2
                    a_t2 = np.maximum(a_t2, 0.0)
                    a_t2 = a_t2 / a_t2.sum()
                    m2_val = fast_conv(a_t2).max()

                    if m1_val < m2_val:
                        hi_frac = m2
                    else:
                        lo_frac = m1

                if best_frac is not None:
                    amt = a[i_int] * best_frac
                    a[i_int] -= amt
                    a[j_int] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    m_new = fast_conv(a).max()
                    if m_new < best_m - 1e-16:
                        best_m = m_new
                        best_a = a.copy()
                        improved = True
            if improved:
                break

        if not improved:
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.001, 0.02)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_conv(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
            else:
                break  # Can't improve

    return best_a, best_m


def interpolate_seq(a, new_n):
    """Interpolate to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def make_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    a = np.sinc((x - 0.5) * bw) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_decreasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_sum1((n - x + 1) ** power)


def make_mixed(n, pos1=0.2, pos2=0.65, width=0.1, ratio=1.0):
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((x - pos1) / width) ** 2) + ratio * np.exp(-((x - pos2) / (width * 1.5)) ** 2)
    return norm_sum1(a)


def make_asymmetric(n, skew=0.3):
    """Asymmetric bell curve."""
    x = np.arange(n, dtype=np.float64) / n
    center = 0.5 + skew
    width = 0.15
    a = np.exp(-((x - center) / width) ** 2)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

            try_update(a_out, f"fin_cycle_{cycle}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 15:
                break
            a_out, _ = coord_descent_targeted(a_in, t_c * 0.4, seed=2000 + cycle)
            try_update(a_out, f"cd_cycle_{cycle}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 15:
                break
            a_out, _ = pair_swap_optimize(a_in, t_c * 0.2, seed=3000 + cycle)
            try_update(a_out, f"ps_cycle_{cycle}")
        if _best_tracker['score'] <= best_score_before:
            print(f"  No improvement in final polish")

    # === Phase 7: Try nearby n values ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_best_tracker['seq'])
    print(f"\n=== Phase 7: Multi-n (base={n_best}) ===")
    for delta in [300, -300, 600, -600]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        new_n = n_best + delta
        if new_n < 2000 or new_n > 8000:
            continue
        print(f"  n={new_n}, remaining={remaining:.0f}s")
        a_interp = interpolate_seq(_best_tracker['seq'], new_n)
        try_update(a_interp, f"intp{new_n}")
        t_m = min(remaining * 0.3, 120.0)
        if t_m > 10:
            a_out, _ = amd_adaptive(a_interp, t_m, seed=5000 + new_n)
            try_update(a_out, f"amd{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_m2 = min(remaining * 0.2, 60.0)
            if t_m2 > 8:
                a_out, _ = coord_descent_targeted(a_interp, t_m2, seed=6000 + new_n)
                try_update(a_out, f"cd{new_n}")

    # Final evaluation
    if _best_tracker['seq'] is not None:
        fs = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        if np.isfinite(fs):
            _best_tracker['score'] = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={_best_tracker['score']:.6f}, n={len(_best_tracker['seq'])}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        _best_tracker['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Fallback: score={_best_tracker['score']:.6f}")

    return [float(v) for v in _best_tracker['seq']]
