            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SpectMul iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, "
                  f"beta={beta:.1f}, eta={eta:.4f}, stall={stall}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K_init=200, top_K_max=None):
    """LP-based Frank-Wolfe for polishing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Backtracking line search
        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0), 0.3, 0.2, 0.1]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1

        if iter_count % 5 == 0:
            K = min(K + 20, top_K_max)

        if verbose and iter_count % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def pair_swap_polish(a0, time_limit, verbose=True, rng_seed=100):
    """Pair-swap coordinate descent for final polishing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200
    n_top = 30

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        # Build sensitivity map
        sens = np.zeros(n)
        for k in top_k[:15]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[ki - idx]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-sensitivity destination
            low_sens = np.argsort(sens)[:20]
            candidates = [c for c in low_sens if c != src and sens[c] < sens[src] * 0.5]
            if not candidates:
                candidates = [c for c in low_sens[:5] if c != src]

            found = False
            for dst in candidates[:5]:
                if found:
                    break
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

            if not found and stall > 50:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PairSwap ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PairSwap final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def spectral_multiplicative_two_phase(a0, time_limit, verbose=True, rng_seed=42):
    """Two-phase: coarse (low beta) then fine (high beta) multiplicative optimization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    # Phase 1: Coarse (lower beta, larger steps)
    t1 = time_limit * 0.4
    a, m1 = spectral_multiplicative_optimize(
        a, t1, verbose=False, beta_init=15.0, beta_final=60.0,
        eta_init=2.0, rng_seed=rng_seed
    )
    if m1 < best_m:
        best_m = m1
        best_a = a.copy()

    # Phase 2: Fine (higher beta, smaller steps)
    remaining = time_limit - (time.monotonic() - t0)
    a2, m2 = spectral_multiplicative_optimize(
        best_a, remaining, verbose=verbose, beta_init=60.0, beta_final=300.0,
        eta_init=0.5, rng_seed=rng_seed + 1
    )
    if m2 < best_m:
        best_m = m2
        best_a = a2.copy()

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-init spectral multiplicative ===
    print(f"\n=== Phase 1: Multi-init Spectral Multiplicative (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.25, 400.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_u_shape(n0, sharpness=2.0))
        labels.append("u2.0")
        inits.append(make_u_shape(n0, sharpness=4.0))
        labels.append("u4.0")
        inits.append(make_bimodal(n0, width=0.18))
        labels.append("bimodal")
        inits.append(make_cosine(n0))
        labels.append("cosine")
        inits.append(spectral_init(n0, n_modes=40, seed=77))
        labels.append("spectral")
        inits.append(make_random_flat(n0, seed=123))
        labels.append("randflat")

        if parent is not None:
            blend = 0.5 * norm_sum1(parent) + 0.5 * make_u_shape(n0, 3.0)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Remove near-duplicates
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 40)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(20.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 15:
                break
            print(f"  SpectMul init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = spectral_multiplicative_optimize(
                init, t_avail, verbose=False,
                beta_init=20.0, beta_final=100.0,
                eta_init=1.5, rng_seed=42 + idx
            )
            try_update(a_out, f"spec_{lbl}")

    # === Phase 2: Extended two-phase on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Two-phase Spectral (n={n_best}) ===")
    t2 = min(remaining * 0.35, 600.0)
    if t2 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_multiplicative_two_phase(
            a_in, t2, verbose=True, rng_seed=200
        )
        try_update(a_out, "spec_2phase")

    # === Phase 3: LP-FW polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.18, 300.0)
    if t3 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 1500)
        top_K_init = min(300, 2 * n_best - 1)
        top_K_max = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp_main")

    # === Phase 4: More spectral multiplicative (different params) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Spectral r2 (n={n_best}) ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        # Different parameter regime
        a_out, _ = spectral_multiplicative_optimize(
            a_in, t4, verbose=False,
            beta_init=80.0, beta_final=400.0,
            eta_init=0.3, rng_seed=500
        )
        try_update(a_out, "spec_r2")

    # === Phase 5: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW r2 (n={n_best}) ===")
    t5 = min(remaining * 0.10, 180.0)
    if t5 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(500, 2 * n_best - 1)
        top_K_max = min(5000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=False,
                                    top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp_r2")

    # === Phase 6: Pair swap polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Pair Swap (n={n_best}) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = pair_swap_polish(a_in, t6, verbose=True, rng_seed=789)
        try_update(a_out, "pairswap")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")

    trial_lengths = []
    for delta in [500, 1000, -500, 1500, -1000, 2000]:
        new_n = n_best + delta
        if 800 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7 = min(remaining * 0.5, 200.0)
        if t7 > 20:
            a_out, _ = spectral_multiplicative_optimize(
                a_long, t7, verbose=False,
                beta_init=20.0, beta_final=150.0,
                eta_init=1.0, rng_seed=new_n % 500
            )
            try_update(a_out, f"spec_n{new_n}")

    # === Phase 8: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish (n={n_best}) ===")
    if remaining > 20:
        # Quick spectral
        t8a = min(remaining * 0.4, 100.0)
        if t8a > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = spectral_multiplicative_optimize(
                a_in, t8a, verbose=False,
                beta_init=100.0, beta_final=500.0,
                eta_init=0.2, rng_seed=999
            )
            try_update(a_out, "spec_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            # Quick pair swap
            t8b = min(remaining * 0.5, 80.0)
            if t8b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = pair_swap_polish(a_in, t8b, verbose=False, rng_seed=1000)
                try_update(a_out, "pair_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
