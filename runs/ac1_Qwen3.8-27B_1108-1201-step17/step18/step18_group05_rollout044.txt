def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_u_shape(n, sharpness=2.0):

def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0, restart_every=6000,
                 adaptive_beta=True, verbose=False, rng_seed=42):

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if adaptive_beta and stall_count > restart_every // 2:
            cur_beta = max(cur_beta * 0.9, 2.0)
            cur_top_k = min(cur_top_k + 5, len(ac))

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 150)
            cur_beta = max(cur_beta - 1.0, 2.0)

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=True, top_K_init=200, 
                       top_K_max=None, random_mix=0.1, look_ahead=5):
    """
    Improved LP-Frank-Wolfe with:
    - Random lag mixing to avoid over-fitting
    - Better line search
    - Look-ahead: consider top-k lags for objective tracking
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0
    stale_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        
        # Mix top lags with random lags to avoid over-fitting
        if random_mix > 0:
            n_random = max(1, int(K_sel * random_mix))
            n_top = K_sel - n_random
            top_idx = np.argsort(ac)[::-1][:n_top]
            all_idx = np.arange(2 * n - 1)
            remaining = np.setdiff1d(all_idx, top_idx)
            if len(remaining) > 0:
                rand_idx = rng.choice(remaining, size=min(n_random, len(remaining)), replace=False)
                binding_idx = np.concatenate([top_idx, rand_idx])
            else:
                binding_idx = top_idx
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build constraint matrix
        K_actual = len(binding_idx)
        T_sel = np.zeros((K_actual, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):

            if stale_count > 20:
                K = min(K + 100, top_K_max)
                stale_count = 0
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:

        g_dir = b_val / b_sum

        # Better line search: try more alphas with Armijo-like condition
        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)

                best_trial_m = m_trial

        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a.copy()
            stale_count = 0
        else:
            stale_count += 1

            else:
                K = min(K + 80, top_K_max)
        
        # If very stale, try a perturbation

def smooth_gradient_optimize_v2(a0, time_limit, verbose=True, tau_schedule=None,
                                 eta_init=0.5, rng_seed=100):
    """
    Smoothed projected gradient with geometric temperature scheduling.
    tau decreases over time (annealing), allowing finer resolution of peaks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    if tau_schedule is None:
        tau_schedule = np.geomspace(200.0, 2.0, num=20)

    eta = eta_init
    stall = 0
    max_stall = 300
    total_iters = 0

    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        phase_start = time.monotonic()
        phase_iters = 0
        stall = 0

        while (time.monotonic() - phase_start) < (time_limit * 0.3 / len(tau_schedule)) + 10:
            if time.monotonic() - t0 > time_limit or stall > max_stall:
                break

            ac = fast_autocorr(a)
            ac_max = ac.max()
            ac_shifted = tau * (ac - ac_max)
            exp_ac = np.exp(ac_shifted)
            p = exp_ac / exp_ac.sum()

            p_cum = np.concatenate([[0.0], np.cumsum(p)])
            k_lo = np.arange(n)
            k_hi = np.minimum(k_lo + n, 2 * n - 1)
            S = p_cum[k_hi] - p_cum[k_lo]
            cross = sp_correlate(p, a, mode='valid')
            g = a * S + cross
            g_tangent = g - g.mean()

            a_new = project_simplex(a - eta * g_tangent)
            m_new = fast_autocorr(a_new).max()

            if m_new < best_m - 1e-15:
                best_m = m_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                eta = min(eta * 1.01, 3.0)
            else:
                a = a_new
                stall += 1
                if stall > 30:
                    eta *= 0.85
                if stall > 100:
                    pert = rng.normal(0, 0.003, n) * a
                    a_p = np.maximum(a + pert, 0.0)
                    a = a_p / a_p.sum()
                    stall = 50

            phase_iters += 1
            total_iters += 1

        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad2 tau={tau:.1f} ({elapsed:.0f}s): J={J:.6f}, iters={phase_iters}")

    return best_a, best_m


def stochastic_local_search(a0, time_limit, verbose=True, rng_seed=777):
    """
    Stochastic local search with susceptibility-based move selection.
    Key idea: identify "hot" positions (high susceptibility to autocorrelation peaks)
    and "cold" positions (low susceptibility), then transfer mass from hot to cold.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    # Compute susceptibility: for each position i, how much does increasing a[i]
    # increase the peak autocorrelation?
    def compute_susceptibility(a):
        ac = fast_autocorr(a)
        M = ac.max()
        # Top lags
        top_k = np.argsort(ac)[::-1][:30]
        sens = np.zeros(n)
        for k in top_k:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            # d/d a[i] of a[i]*a[ki-i] = a[ki-i] + a[i]*d/d a[i] a[ki-i]
            # For the product a[i]*a[ki-i], the sensitivity is a[ki-i] for position i
            # and a[i] for position ki-i (which is the same as the other term)
            # So the total sensitivity of R(k) w.r.t. a[i] is a[ki-i] (when i is in range)
            sens[idx] += a[ki - idx] * (ac[int(k)] / M)  # normalized by peak
        return sens

    stall = 0
    max_stall = 200
    iter_count = 0
    temperature = 0.01  # initial step size
    cool_rate = 0.9995

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 5:
            break

        # Periodically recompute susceptibility
        if iter_count % 100 == 0:
            sens = compute_susceptibility(a)
            hot = np.argsort(sens)[::-1][:n // 4]  # top 25% hot positions
            cold = np.argsort(sens)[:n // 4]  # bottom 25% cold positions

        # Choose a move
        if rng.random() < 0.7:
            # Smart move: hot to cold
            if len(hot) > 0 and len(cold) > 0:
                src = hot[rng.integers(len(hot))]
                dst = cold[rng.integers(len(cold))]
            else:
                src = rng.integers(n)
                dst = rng.integers(n)
        else:
            # Random move
            src = rng.integers(n)
            dst = rng.integers(n)

        if src == dst:
            continue

        # Step size
        frac = rng.uniform(0.001, temperature)
        amt = a[src] * frac
        if amt < 1e-15:
            continue

        a_trial = a.copy()
        a_trial[src] -= amt
        a_trial[dst] += amt
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t

        M_trial = fast_autocorr(a_trial).max()

        # Accept if better, or with some probability if worse (simulated annealing)
        if M_trial < best_m:
            a = a_trial
            best_m = M_trial
            best_a = a.copy()
            stall = 0
        elif M_trial < best_m * (1.0 + 0.001 * np.exp(-iter_count / 5000)):
            # Accept small worsening with decreasing probability
            if rng.random() < np.exp(-iter_count / 10000):
                a = a_trial
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        # Cool down
        temperature *= cool_rate
        if temperature < 0.001:
            temperature = 0.01
            # Reset with perturbation
            pert = rng.normal(0, 0.005, n) * a
            a_p = np.maximum(a + pert, 0.0)
            a = a_p / a_p.sum()
            temperature = 0.01

        iter_count += 1

        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SLS iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, temp={temperature:.4f}, stall={stall}")

    return best_a, best_m


def multi_scale_optimize(a0, time_limit, solver, verbose=True):
    """
    Multi-scale: downsample, optimize, upscale, refine.
    """
    n = len(a0)
    best_a = norm_sum1(a0)
    best_m = fast_autocorr(best_a).max()
    t0 = time.monotonic()

    # Downsample to n/2, optimize, then upscale
    n_small = max(500, n // 2)
    a_small = interpolate_sequence(best_a, n_small)
    
    # Optimize small
    t1 = min(time_limit * 0.3, (time.monotonic() - t0))
    if t1 > 20:
        a_small_out, _ = amd_optimize(a_small, t1, top_k=40, beta=5.0, c0=4.0,
                                       restart_every=3000, verbose=False)
        # Upscale and blend
        a_up = interpolate_sequence(a_small_out, n)
        # Blend with original
        for blend_w in [0.5, 0.7, 0.3]:
            a_blend = (1 - blend_w) * best_a + blend_w * a_up
            a_blend = norm_sum1(a_blend)
            m_blend = fast_autocorr(a_blend).max()
            if m_blend < best_m:
                best_m = m_blend
                best_a = a_blend.copy()

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator. Uses:
    1. Refined LP-Frank-Wolfe with random lag mixing
    2. Smoothed gradient with geometric temperature schedule
    3. Stochastic local search with susceptibility-based moves
    4. Multi-length exploration
    5. Final polish
    """

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):

        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    PHASES = []

    # === Phase 1: LP-FW v2 (main optimizer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 50 and n0 <= 6000:
        PHASES.append(("LP-FW2", t1))

    # === Phase 2: Smooth gradient v2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    # Adjust based on what's left
    t2 = min(remaining * 0.2, 300.0)
    if t2 > 30:
        PHASES.append(("SmoothGrad2", t2))

    # === Phase 3: LP-FW2 round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.2, 300.0)
    if t3 > 50 and n0 <= 6000:
        PHASES.append(("LP-FW2-r2", t3))

    # === Phase 4: Stochastic Local Search ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        PHASES.append(("SLS", t4))

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 50:
        PHASES.append(("LengthExploration", t5))

    # === Phase 6: Final LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.08, 150.0)
    if t6 > 30 and n0 <= 6000:
        PHASES.append(("LP-FW-final", t6))

    # === Phase 7: Final SLS polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.10, 100.0)
    if t7 > 20:
        PHASES.append(("SLS-final", t7))

    # Execute phases
    for phase_name, phase_time in PHASES:

            break

        actual_time = min(phase_time, remaining - 10)
        if actual_time < 15:
            break

        n_best = len(best_seq)

        if phase_name == "LP-FW2":

            use_all = (n_best <= 1500)
            top_K_init = min(300, 2 * n_best - 1)
            top_K_max = min(4000, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe_v2(a_in, actual_time, solver, verbose=True,
                                          top_K_init=top_K_init, top_K_max=top_K_max,
                                          random_mix=0.15, look_ahead=5)
            try_update(a_out, phase_name)

        elif phase_name == "LP-FW2-r2":
            a_in = norm_sum1(best_seq)
            use_all = (n_best <= 1500)
            top_K_init = min(500, 2 * n_best - 1)
            top_K_max = min(6000, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe_v2(a_in, actual_time, solver, verbose=True,
                                          top_K_init=top_K_init, top_K_max=top_K_max,

            try_update(a_out, phase_name)

        elif phase_name == "LengthExploration":

            trial_lengths = []
            for delta in [500, -500, 1000, -1000]:

            trial_lengths = [l for l in trial_lengths if l != n_best]

            t_per = actual_time / max(1, len(trial_lengths[:2]))
            for new_n in trial_lengths[:2]:
                if time.monotonic() - t_global > time_budget - 30:
                    break
                print(f"  Trying n={new_n}...")
                a_long = interpolate_sequence(best_seq, new_n)
                
                # AMD on new length
                a_out, _ = amd_optimize(a_long, t_per * 0.6, top_k=60, beta=6.0,
                                         c0=4.0, restart_every=4000, verbose=False,
                                         rng_seed=new_n % 1000)
                try_update(a_out, f"amd_n{new_n}")
                
                # LP on new length if time allows
                remaining_inner = time_budget - (time.monotonic() - t_global)

                                          top_K_init=top_K_init, top_K_max=top_K_max,

            try_update(a_out, phase_name)

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200

        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
