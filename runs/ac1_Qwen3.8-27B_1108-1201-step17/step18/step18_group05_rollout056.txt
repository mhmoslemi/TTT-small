def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def cross_corr_weights(w, a):
    """Compute xc[i] = sum_j w[j+i] * a[j] for i=0..n-1.
    w has length 2n-1, a has length n.
    """
    n = len(a)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    xc = np.fft.irfft(np.conj(A) * W, fft_size)
    return xc[:n]


def variance_gradient(c, a):
    """Compute gradient of sum_k c[k]^2 with respect to a.
    g[i] = 4 * xc[i] - 2 * c[2i] * a[i] (if 2i < len(c))
    """
    n = len(a)
    xc = cross_corr_weights(c, a)
    g = 4.0 * xc
    # Diagonal correction
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < len(c)
    g[valid] -= 2.0 * c[diag_idx[valid]] * a[valid]
    return g


def softmax_gradient(c, a, beta):
    """Compute gradient of (1/beta)*log(sum_k exp(beta*c[k])) with respect to a.
    g[i] = (2/beta)*xc_p[i] - (1/beta)*p[2i]*a[i]
    """
    n = len(a)
    shifted = beta * (c - c.max())
    exp_c = np.exp(shifted)
    p = exp_c / exp_c.sum()
    xc_p = cross_corr_weights(p, a)
    g = (2.0 / beta) * xc_p
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < len(c)
    g[valid] -= (1.0 / beta) * p[diag_idx[valid]] * a[valid]
    return g


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()

    return a / s


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K_init=300, top_K_max=None):
    """LP-based Frank-Wolfe with adaptive K."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = objective_value(a)
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = min(3000, 2 * n - 1)
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        M_curr = c.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(c)[::-1][:K_sel]

        # Build constraint matrix T: T[ii, j] = a[ki - j] where ki = binding_idx[ii]
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start < j_end:
                T_sel[ii, j_start:j_end] = a[ki - np.arange(j_start, j_end)]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Backtracking line search
        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0),
                      min(1.0, best_alpha * 3.0), 0.5, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_convolve(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_convolve(a).max()
        obj_new = 2.0 * n * m_new
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()

        iter_count += 1
        if iter_count % 10 == 0:
            K = min(K + 30, top_K_max)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={best_obj:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

    return best_a, best_obj


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent for final polishing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = objective_value(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_convolve(a)
        M_curr = c.max()
        top_k = np.argsort(c)[::-1][:15]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            idx = np.arange(s, e)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-contribution destinations
            in_peak = np.zeros(n, dtype=bool)
            for kk in top_k[:8]:
                kk_i = int(kk)
                ss = max(0, kk_i - n + 1)
                ee = min(n, kk_i + 1)
                in_peak[ss:ee] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            # Pick a few candidates
            top_cands = rng.choice(candidates, size=min(5, len(candidates)), replace=False)

            found = False
            for dst in top_cands:
                if dst == src or found:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_convolve(a_trial).max()
                    if M_trial < M_curr - 1e-14:
                        a = a_trial
                        M_curr = M_trial
                        obj = 2.0 * n * M_curr
                        if obj < best_obj:
                            best_obj = obj
                            best_a = a.copy()
                        improved = True
                        found = True
                        break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_convolve(a).max()
                if M_new < M_curr:
                    stall = 0
                    obj = 2.0 * n * M_new
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()

        if verbose and stall % 50 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord ({elapsed:.0f}s): J={best_obj:.6f}, stall={stall}")

    return best_a, best_obj


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-start gradient optimization ===
    print(f"\n=== Phase 1: Multi-start gradient (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.20, 300.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_u_shape(n0, sharpness=2.0))
        labels.append("u2")
        inits.append(make_u_shape(n0, sharpness=4.0))
        labels.append("u4")
        inits.append(make_bimodal(n0, width=0.2))
        labels.append("bimodal")
        inits.append(make_cosine(n0))
        labels.append("cosine")
        inits.append(spectral_init_flat(n0, seed=42))
        labels.append("flat_spec")

        if parent is not None:
            blend = 0.7 * norm_sum1(parent) + 0.3 * make_u_shape(n0, 3.0)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Remove near-duplicates
        seen_sigs = []
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 30)], 4))
            is_dup = False
            for ss in seen_sigs:
                if np.allclose(sig, ss, atol=0.05):
                    is_dup = True
                    break
            if not is_dup:
                seen_sigs.append(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(15.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  [{lbl}] combined_grad (t_avail={t_avail:.0f}s)...")
            a_out, _ = combined_gradient_descent(init, t_avail, verbose=False, rng_seed=42 + idx)
            try_update(a_out, f"comb_{lbl}")

    # === Phase 2: Softmax momentum on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Softmax momentum (n={n_best}) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = softmax_momentum_descent(a_in, t2, verbose=True,
                                             beta_init=15.0, beta_max=400.0,
                                             eta_init=0.2, momentum=0.9, rng_seed=100)
        try_update(a_out, "softmom")

    # === Phase 3: Variance gradient on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Variance gradient (n={n_best}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = variance_grad_descent(a_in, t3, verbose=True, eta_init=0.4, rng_seed=300)
        try_update(a_out, "vargrad")

    # === Phase 4: LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 40 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(400, 2 * n_best - 1)
        top_K_max = min(4000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True,
                                   top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp1")

    # === Phase 5: Combined gradient round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Combined gradient r2 (n={n_best}) ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = combined_gradient_descent(a_in, t5, verbose=True, rng_seed=500)
        try_update(a_out, "comb_r2")

    # === Phase 6: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LP-FW r2 (n={n_best}) ===")
    t6 = min(remaining * 0.12, 200.0)
    if t6 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(600, 2 * n_best - 1)
        top_K_max = min(5000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=True,
                                   top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp2")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration (base n={n_best}) ===")

    trial_lengths = []
    for delta in [200, 500, -200, 800, -500, 1000]:
        new_n = n_best + delta
        if 3000 <= new_n <= 8000 and new_n != n_best:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.35, 150.0)
        if t7a > 20:
            a_out, _ = combined_gradient_descent(a_long, t7a, verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"comb_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 6000:
            t7b = min(remaining * 0.3, 100.0)
            if t7b > 20:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                top_K = min(3000, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe(a_in, t7b, solver, verbose=False,
                                           top_K_init=min(300, 2 * new_n - 1),
                                           top_K_max=top_K)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 8: Final intensive optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final optimization (n={n_best}) ===")
    if remaining > 40:
        t8 = min(remaining * 0.3, 150.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = softmax_momentum_descent(a_in, t8, verbose=False,
                                                 beta_init=50.0, beta_max=500.0,
                                                 eta_init=0.1, momentum=0.95, rng_seed=700)
            try_update(a_out, "softmom_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and n_best <= 6000:
            t8b = min(remaining * 0.35, 120.0)
            if t8b > 20:
                a_in = norm_sum1(best_seq)
                top_K_init = min(800, 2 * n_best - 1)
                top_K_max = min(6000, 2 * n_best - 1)
                a_out, _ = lp_frank_wolfe(a_in, t8b, solver, verbose=False,
                                           top_K_init=top_K_init, top_K_max=top_K_max)
                try_update(a_out, "lp_final")

    # === Phase 9: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 9: Coordinate polish (n={n_best}) ===")
        t9 = min(remaining * 0.5, 80.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t9, verbose=False, rng_seed=999)
            try_update(a_out, "coord_final")

        # One more round of combined gradient after coordinate
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_in = norm_sum1(best_seq)
            t9b = min(remaining * 0.5, 40.0)
            if t9b > 10:
                a_out, _ = combined_gradient_descent(a_in, t9b, verbose=False, rng_seed=1111)
                try_update(a_out, "comb_very_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
