def fast_autocorr(a):

def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs

def coordinate_polish_exact(a0, time_limit, verbose=True):
    """Coordinate descent with exact 1D optimization using bisection on M."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    ac = fast_autocorr(a)
    # Focus on coordinates that contribute to the top peaks
    top_k = np.argsort(ac)[::-1][:min(50, len(ac))]

    no_improve_rounds = 0
    max_no_improve_rounds = 150

    for round_num in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve_rounds > max_no_improve_rounds:
            break

        ac = fast_autocorr(a)
        current_m = ac.max()
        round_improved = False

        # Determine which coordinates to update: those that appear in peak convolutions
        peak_indices = set()
        for k in top_k[:20]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            for i in range(i_start, i_end + 1):
                peak_indices.add(i)
                peak_indices.add(k_int - i)
            if len(peak_indices) > 200:
                break

        coord_list = sorted(peak_indices)
        np.random.seed(round_num)
        np.random.shuffle(coord_list)

        for j in coord_list[:150]:
            if time.monotonic() - t0 > time_limit:
                break

            j_int = int(j)
            # Compute rest_k, lin_k, quad_k for all relevant k
            a_j = a[j_int]

            # We need to check constraints at the top convolution positions
            relevant_k = list(top_k[:30])

            rest = np.zeros(len(relevant_k))
            lin = np.zeros(len(relevant_k))
            quad = np.zeros(len(relevant_k))

            for kk, k in enumerate(relevant_k):
                k_int = int(k)
                # conv[k] = sum_i a[i]*a[k-i]
                # Terms involving a[j]:
                # i = j: a[j]*a[k-j] if 0 <= k-j < n
                # i = k-j: a[k-j]*a[j] if 0 <= k-j < n
                if 0 <= k_int - j_int < n:
                    partner = k_int - j_int
                    if partner != j_int:
                        lin[kk] = 2.0 * a[partner]
                        rest[kk] = ac[k_int] - lin[kk] * a_j
                    else:
                        quad[kk] = 1.0
                        rest[kk] = ac[k_int] - a_j * a_j
                else:
                    rest[kk] = ac[k_int]

            # Bisection on x = a[j] to minimize max_k (rest[k] + lin[k]*x + quad[k]*x^2)
            # Feasible range: x >= 0, and for each k with lin[k] > 0: x <= (M - rest[k])/lin[k]
            # We want the minimum M such that there exists x >= 0 satisfying all constraints.

            x_lo = 0.0
            x_hi = a_j * 3.0 + 1.0 / n  # upper bound

            # Find minimum M by bisection
            M_lo = 0.0
            M_hi = current_m * 1.5

            for _ in range(40):
                M_mid = (M_lo + M_hi) / 2.0
                # Check feasibility: exists x in [x_lo, x_hi] s.t. all constraints satisfied
                feas_lo = x_lo
                feas_hi = x_hi

                for kk in range(len(relevant_k)):
                    if quad[kk] > 0.5:
                        # x^2 + lin[kk]*x + (rest[kk] - M_mid) <= 0
                        # discriminant = lin^2 - 4*(rest-M)
                        disc = lin[kk] * lin[kk] - 4.0 * (rest[kk] - M_mid)
                        if disc < 0:
                            if kk > 0:
                                pass
                            # If rest > M_mid, infeasible
                            if rest[kk] > M_mid:
                                break
                        else:
                            sq = np.sqrt(disc)
                            r1 = (-lin[kk] - sq) / 2.0
                            r2 = (-lin[kk] + sq) / 2.0
                            feas_lo = max(feas_lo, r1)
                            feas_hi = min(feas_hi, r2)
                    else:
                        if lin[kk] > 1e-15:
                            if rest[kk] > M_mid:
                                break
                            feas_lo = max(feas_lo, 0.0)
                            feas_hi = min(feas_hi, (M_mid - rest[kk]) / lin[kk])
                        elif rest[kk] > M_mid:
                            break

                if feas_lo <= feas_hi + 1e-12:
                    # Feasible: find optimal x in [feas_lo, feas_hi]
                    # Optimal x minimizes max_k (rest + lin*x + quad*x^2)
                    # Try a few candidates
                    candidates = [feas_lo, feas_hi, (feas_lo + feas_hi) / 2.0]
                    if a_j > 0 and feas_lo <= a_j <= feas_hi:
                        candidates.append(a_j)
                    # Also try where the max of active constraints is minimized
                    best_x = a_j
                    best_val = float('inf')
                    for xc in candidates:
                        xc = max(0.0, xc)
                        vals = rest + lin * xc + quad * xc * xc
                        vm = vals.max()
                        if vm < best_val:
                            best_val = vm
                            best_x = xc
                    if best_val < M_lo:
                        # Shouldn't happen but safety
                        M_lo = best_val
                    # Feasible at M_mid, try smaller
                    M_hi = M_mid
                else:
                    M_lo = M_mid

            # M_lo is approximately the minimum achievable M
            if M_lo < current_m - 1e-10:
                # Apply the update
                a[j_int] = max(0.0, best_x)
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m - 1e-15:
                        best_m = M_new
                        best_a = a.copy()
                        round_improved = True
                        # Update top_k
                        ac = fast_autocorr(a)
                        top_k = np.argsort(ac)[::-1][:min(50, len(ac))]

        if not round_improved:
            no_improve_rounds += 1
        else:
            no_improve_rounds = 0

        if verbose and round_num % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    CoordExact round=" + str(round_num) + " (" + str(elapsed) + "s): J=" + repr(J))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    CoordExact done (" + str(elapsed) + "s): J=" + repr(J))
    return best_a, best_m


def mwd_nesterov(a0, time_limit, top_k=80, beta=5.0, c0=4.0,
                 restart_every=3000, verbose=False):
    """Multiplicative weight descent with Nesterov momentum and AdaGrad-like step."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    restart_count = 0

    # AdaGrad accumulators
    grad_sq_sum = np.ones(n) * 1e-8

    for t in range(1, 30000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        ac = fast_autocorr(y)
        # Compute soft gradient: weighted sum of gradients of top_k terms
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        # AdaGrad step size
        eta = gamma / (np.sqrt(t) + 1.0)
        grad_sq_sum += grad * grad
        eta_adaptive = eta / np.sqrt(grad_sq_sum)

        log_x = np.log(y) - eta_adaptive * grad
        max_log = log_x.max()
        a_new = np.exp(log_x - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            restart_count += 1
            noise_level = 0.04 / (1.0 + 0.05 * restart_count)
            pert = rng.normal(0, noise_level, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r < 1e-15:
                x_r = best_a.copy()
            else:
                x_r = x_r / s_r
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 15, 150)
            cur_beta = max(cur_beta - 0.3, 2.0)
            gamma = max(gamma * 0.9, 1.0)
            grad_sq_sum = np.ones(n) * 1e-8
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    MWD restart t=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J))

    return best_a, best_m


def symmetric_lp_fw(a0, time_limit, solver, initial_K=200, max_K=1500,
                    add_per_iter=15, verbose=True):
    n = len(a0)
    half = (n + 1) // 2
    a = norm_sum1(a0)

    a_sym = np.zeros(n)
    for i in range(n):
        a_sym[i] = (a[i] + a[n - 1 - i]) / 2.0
    a_sym = norm_sum1(a_sym)

    best_a = a_sym.copy()
    best_m = fast_autocorr(a_sym).max()
    t0 = time.monotonic()

    y = a_sym.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 800:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs_half = np.zeros((K, half))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            vals = y[k_int - idx]
            for ii in range(len(idx)):
                i_full = idx[ii]
                i_half = min(i_full, n - 1 - i_full)
                if i_half < half:
                    coeffs_half[jj, i_half] += vals[ii]

        g_half = cp.Variable(half, nonneg=True)
        M_var = cp.Variable()
        if n % 2 == 0:
            sum_constraint = 2.0 * cp.sum(g_half)
        else:
            sum_constraint = 2.0 * cp.sum(g_half) - g_half[half - 1]

        constraints = [sum_constraint == 1, coeffs_half @ g_half <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_half_val = np.maximum(np.asarray(g_half.value, dtype=np.float64), 0.0)

        g_full = np.zeros(n)
        for i in range(half):
            g_full[i] = g_half_val[i]
            g_full[n - 1 - i] = g_half_val[i]
        if n % 2 == 1:
            g_full[half - 1] = g_half_val[half - 1]

        g_sum = g_full.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_full / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Backtracking
        alpha = min(1.0, 2.0 / (t + 2.0))
        best_trial_M = M_curr
        best_alpha = 0.0
        for attempt in range(8):
            cur_alpha = alpha * (0.7 ** attempt)
            if cur_alpha < 0.01:
                break
            trial = (1.0 - cur_alpha) * y + cur_alpha * g_dir
            trial = np.maximum(trial, 0.0)
            for i in range(n // 2):
                avg = (trial[i] + trial[n - 1 - i]) / 2.0
                trial[i] = avg
                trial[n - 1 - i] = avg
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < M_curr - 1e-12:
                if M_trial < best_trial_M:
                    best_trial_M = M_trial
                    best_alpha = cur_alpha
                break

        if best_alpha > 0:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
        else:
            y_new = y.copy()

        for i in range(n // 2):
            avg = (y_new[i] + y_new[n - 1 - i]) / 2.0
            y_new[i] = avg
            y_new[n - 1 - i] = avg
        s_y = y_new.sum()
        if s_y > 1e-15:
            y = y_new / s_y

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Sym-LP iter=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J))

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Total time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        best_seq = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1((n_base - np.arange(n_base, dtype=np.float64)) ** 1.0)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: Parametric search
    print("\n=== Phase 1: Parametric search (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 10:
        a_param, _ = parametric_search(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "parametric", t_global)

    # Phase 2: Main LP-FW with backtracking
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: LP-FW backtrack (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.40, 550.0)
    if t2 > 30 and n_best <= 8000:
        a_out, _ = lp_fw_with_backtrack(norm_sum1(best_seq), t2, solver,
                                        initial_K=500, max_K=3000,
                                        add_per_iter=30, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_bt_n" + str(n_best), t_global)

    # Phase 3: Coordinate descent exact
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: CoordExact (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_out, _ = coordinate_polish_exact(norm_sum1(best_seq), t3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coordexact_n" + str(n_best), t_global)

    # Phase 4: MWD with Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: MWD Nesterov (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out, _ = mwd_nesterov(norm_sum1(best_seq), t4, top_k=80, beta=5.0,
                                c0=4.0, restart_every=3000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "mwd_n" + str(n_best), t_global)

    # Phase 5: Second LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: LP-FW round 2 (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 30 and n_best <= 8000:
        a_out, _ = lp_fw_with_backtrack(norm_sum1(best_seq), t5, solver,
                                        initial_K=600, max_K=3500,
                                        add_per_iter=40, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp2_n" + str(n_best), t_global)

    # Phase 6: Final coord exact
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Final CoordExact (n=" + str(n_best) + ") ===")
    if remaining > 30:
        t6 = min(remaining * 0.25, 150.0)
        if t6 > 15:
            a_out, _ = coordinate_polish_exact(norm_sum1(best_seq), t6, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coordexact2_n" + str(n_best), t_global)

    # Phase 7: Try one longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        new_n = min(len(best_seq) + 1000, 7500)
        print("\n=== Phase 7: Scale to n=" + str(new_n) + " ===")
        a_long = interpolate_sequence(best_seq, new_n)
        t7a = min(remaining * 0.4, 150.0)
        if t7a > 40:
            a_out, _ = lp_fw_with_backtrack(norm_sum1(a_long), t7a, solver,
                                            initial_K=400, max_K=2500,
                                            add_per_iter=25, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t7b = min(remaining * 0.5, 100.0)
            if t7b > 20:
                a_out, _ = coordinate_polish_exact(norm_sum1(best_seq), t7b, verbose=True)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n" + str(new_n), t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(elapsed) + "s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = norm_sum1((n_fb - np.arange(n_fb, dtype=np.float64)) ** 1.0)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
