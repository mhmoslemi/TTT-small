"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

def fast_autocorr(a):
    """Compute full autocorrelation (conv(a,a)) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the global evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interp_sequence(a, new_n):
    """Interpolate sequence to new length, maintaining non-negativity and sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    return norm_sum1(a_interp)


def compute_jscore(a):
    """Compute J = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def compute_lse_objective(a, t):
    """Compute t * log(sum(exp(conv(a,a)/t))) with numerical stability."""
    n = len(a)
    ac = fast_autocorr(a)
    ac_max = ac.max()
    if ac_max < 1e-30:
        return 0.0
    shifted = (ac - ac_max) / t
    # Clip to avoid underflow issues (exp of very negative is ~0, that's fine)
    log_sum = ac_max / t + np.log(np.sum(np.exp(shifted)))
    return t * log_sum


def compute_smooth_grad(a, t):
    """
    Compute gradient of t*log(sum(exp(conv(a,a)/t))) w.r.t. a.
    Uses FFT-based cross-correlation and convolution.
    Returns gradient of length n.
    """
    n = len(a)
    ac = fast_autocorr(a)
    ac_max = ac.max()
    if ac_max < 1e-30:
        return np.zeros(n)
    
    # Softmax weights p of length 2n-1
    shifted = (ac - ac_max) / t
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        p = p / p_sum
    
    # grad[j] = sum_k p[k] * (a[k-j] + a[j-k]) with boundary handling
    # = cross_corr(p, a)[j] + conv(p, a)[j] for j in [0, n)
    # cross_corr(p, a)[j] = sum_m p[m+j] * a[m] = ifft(FFT(p) * conj(FFT(a)))[:n]
    # conv(p, a)[j] = sum_k p[k] * a[j-k] = ifft(FFT(p) * FFT(a))[:n]
    
    fft_size = 1 << (3 * n - 2).bit_length()
    
    a_p = np.zeros(fft_size)
    a_p[:n] = a
    p_p = np.zeros(fft_size)
    p_p[:len(p)] = p
    
    A_f = np.fft.rfft(a_p)
    P_f = np.fft.rfft(p_p)
    
    first_term = np.fft.irfft(P_f * np.conj(A_f), fft_size)[:n]
    second_term = np.fft.irfft(P_f * A_f, fft_size)[:n]
    
    return first_term + second_term

def parametric_init(n, parent_seq=None):
    """Generate diverse parametric initializations and return the best one."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []
    
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    
    # Bilinear
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        for beta in [0.5, 1.0, 1.5, 2.0, 3.0]:
            candidates.append(np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta)
    
    # Gaussian bumps
    for c in [0.25, 0.35, 0.5, 0.65, 0.75]:
        for sigma in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    
    # Double bumps
    for c1 in [0.2, 0.3, 0.4]:
        for c2 in [0.6, 0.7, 0.8]:
            for sigma in [0.05, 0.1, 0.15, 0.2]:
                candidates.append(np.exp(-((x - c1) ** 2) / (2.0 * sigma * sigma)) +
                                   np.exp(-((x - c2) ** 2) / (2.0 * sigma * sigma)))
    
    # Cosine families
    for freq in [1, 2, 3, 4, 6]:
        for ph in [0.0, 0.3, 0.5]:
            candidates.append(np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x + ph)) / 2.0, 0.001))
    
    # Sigmoid
    for c in [0.3, 0.5, 0.7]:
        for s in [3, 5, 10, 20, 40]:
            candidates.append(1.0 / (1.0 + np.exp(-s * (x - c))))
    
    # Symmetric exponential
    for rate in [0.5, 1, 2, 3, 5, 10]:
        candidates.append(np.exp(-rate * x) + np.exp(rate * (x - 1.0)))
    
    # U-shaped
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(x ** p + (1.0 - x) ** p)
    
    # Flat-top
    for w in [0.2, 0.3, 0.4, 0.5]:
        mask = (x > w / 2) & (x < 1.0 - w / 2)
        candidates.append(mask.astype(np.float64) + 0.001)
    
    # Triangular
    candidates.append(1.0 - np.abs(2.0 * x - 1.0))
    
    # If parent provided, add it and blends
    if parent_seq is not None and len(parent_seq) == n:
        p_norm = norm_sum1(parent_seq)
        candidates.append(p_norm)
        for w in [0.2, 0.4, 0.6, 0.8]:
            for cand in candidates[:5]:
                candidates.append(w * p_norm + (1.0 - w) * norm_sum1(cand))
    
    # Random
    rng = np.random.default_rng(123)
    for trial in range(50):
        candidates.append(rng.random(n) ** rng.uniform(0.5, 4.0))
    
    # Evaluate all
    best_a = None
    best_j = float('inf')
    for c in candidates:
        a = norm_sum1(c)
        j = compute_jscore(a)
        if j < best_j:
            best_j = j
            best_a = a
    
    return best_a, best_j


def dual_ascent_polish(a0, time_limit, verbose=True):
    """
    Dual ascent: binary search on M (max convolution value) with feasibility check
    via iterative reduction of over-constrained lags.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_j = compute_jscore(a)
    t0 = time.monotonic()
    
    M_lo = fast_autocorr(a).max() * 0.8
    M_hi = fast_autocorr(a).max() * 1.05
    
    for outer in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_mid = (M_lo + M_hi) / 2.0
        # Try to find a feasible a with max conv <= M_mid
        # Iterative reduction: reduce values at positions contributing to violations
        a_try = a.copy()
        feasible = False
        
        for inner in range(500):
            if time.monotonic() - t0 > time_limit:
                break
            ac = fast_autocorr(a_try)
            if ac.max() <= M_mid:
                feasible = True
                break
            
            # Find violating lags
            violations = np.where(ac > M_mid)[0]
            if len(violations) == 0:
                feasible = True
                break
            
            # Compute pressure: for each position, how much it contributes to violations
            pressure = np.zeros(n)
            for k in violations[:20]:
                k_int = int(k)
                js = max(0, k_int - n + 1)
                je = min(n - 1, k_int)
                if js > je:
                    continue
                idx = np.arange(js, je + 1)
                contrib = a_try[idx] * a_try[k_int - idx]
                # Pressure on source positions
                pressure[idx] += contrib / max(a_try[idx], 1e-15)
                pressure[k_int - idx] += contrib / max(a_try[k_int - idx], 1e-15)
            
            # Reduce high-pressure positions
            top_pressure = np.argsort(pressure)[::-1][:n // 5]
            for i in top_pressure:
                if a_try[i] > 1e-10:
                    a_try[i] *= 0.95  # reduce by 5%
            
            a_try = np.maximum(a_try, 0.0)
            s_t = a_try.sum()
            if s_t < 1e-15:
                break
            a_try = a_try / s_t
        
        if feasible:
            M_hi = M_mid
            j_try = compute_jscore(a_try)
            if j_try < best_j:
                best_j = j_try
                best_a = a_try.copy()
                a = a_try.copy()
        else:
            M_lo = M_mid
        
        elapsed = time.monotonic() - t0
        if verbose and outer % 5 == 0:
            print("  DA iter=" + str(outer) + " M=[" + repr(M_lo) + "," + repr(M_hi) + "] (" + str(elapsed) + "s): J=" + repr(best_j))
    
    return best_a, best_j
