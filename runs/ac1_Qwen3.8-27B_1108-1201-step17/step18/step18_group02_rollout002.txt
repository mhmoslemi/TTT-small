def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def lse_smooth(a, ac, tau):
    """Compute smoothed objective and gradient."""
    n = len(a)
    ac_scaled = ac / tau
    ac_scaled -= ac_scaled.max()
    exp_vals = np.exp(ac_scaled)
    p = exp_vals / exp_vals.sum()
    obj = tau * (np.log(exp_vals.sum()) + ac.max() / tau)
    
    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] (with zero padding)
    a_flip = a[::-1].copy()
    size = len(ac)
    fft_size = 1 << (size - 1).bit_length()
    Af = np.fft.rfft(a_flip, fft_size)
    Pf = np.fft.rfft(p, fft_size)
    conv_full = np.fft.irfft(Af * Pf, fft_size)
    grad = 2.0 * conv_full[n - 1:n - 1 + n]
    return obj, p, grad

def nesterov_projected(a0, time_limit, tau_init=None, tau_decay=0.9995, 
                        tau_min=None, lr=0.05, verbose=True):
    """Smoothed projected gradient with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_score = 2.0 * n * ac.max()
    t0 = time.monotonic()
    
    if tau_init is None:
        tau_init = max(ac.max() * 0.3, 1e-8)
    if tau_min is None:
        tau_min = max(ac.max() * 0.001, 1e-12)
    
    tau = tau_init
    a_prev = a.copy()
    momentum = np.zeros(n)
    stall = 0
    lr_current = lr
    rng = np.random.default_rng(777)
    
    step_count = 0
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Nesterov: evaluate at lookahead point
        a_look = project_simplex(a + momentum)
        ac_look = fast_autocorr(a_look)
        
        tau = max(tau * tau_decay, tau_min)
        _, p, grad = lse_smooth(a_look, ac_look, tau)
        
        # Gradient step with projection
        a_new = project_simplex(a_look - lr_current * grad)
        momentum = (a - a_look)  # Nesterov momentum
        
        a = a_new
        step_count += 1
        
        # Adaptive learning rate
        if stall > 500:
            lr_current *= 0.5
            if lr_current < 1e-5:
                # Restart with perturbation
                pert = rng.normal(0, 0.02, n) * best_a
                a = project_simplex(best_a + pert)
                a_prev = a.copy()
                momentum = np.zeros(n)
                lr_current = max(lr, 0.02)
                stall = 0
                if verbose:
                    elapsed = time.monotonic() - t0
                    print(f"    Nesterov restart t={t} ({elapsed:.0f}s): score={best_score:.6f}")
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov t={t} ({elapsed:.0f}s): score={best_score:.6f}, tau={tau:.2e}, lr={lr_current:.4f}")
    
    elapsed = time.monotonic() - t0
    print(f"    Nesterov done ({elapsed:.0f}s): score={best_score:.6f}, steps={step_count}")
    return best_a, best_score

def lp_frank_wolfe(a0, time_limit, solver, initial_K=300, max_K=6000,
                   add_per_iter=100, verbose=True):
    """LP Frank-Wolfe with boundary-aware constraints."""
    n = len(a0)
    a = norm_sum1(a0)
    ac_init = fast_autocorr(a)
    best_a = a.copy()
    best_M = ac_init.max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    # Initialize constraint set: top peaks + boundaries
    init_idx = np.argsort(ac_init)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)
    # Add boundary constraints
    for bnd in list(range(min(50, n))) + list(range(2*n-2-min(50,n), 2*n-1)):
        constraint_set.add(bnd)
    
    iter_count = 0
    lp_fail = 0
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_M - 1e-14:
            best_M = M_curr
            best_a = y.copy()
        
        ac_x = fast_autocorr(x)
        if ac_x.max() < best_M - 1e-14:
            best_M = ac_x.max()
            best_a = x.copy()
        
        # Trim constraints
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            # Always keep boundaries
            boundary_mask = [k < 50 or k >= 2*n-2-50 for k in constraint_list]
            interior = np.array([i for i, bm in enumerate(boundary_mask) if not bm])
            n_remove = len(constraint_list) - max_K
            if n_remove > 0:
                interior_margins = margins[interior]
                remove_idx = interior[np.argsort(interior_margins)[:n_remove]]
                for i in remove_idx:
                    constraint_set.discard(constraint_list[i])
                constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix: (y*a)[k] = sum_i y[k-i]*a[i] <= M
        # For each k, the coefficient of a[i] is y[k-i] (if valid)
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail += 1
            if lp_fail > 10:
                constraint_set = set(int(k) for k in constraint_list[:max(200, initial_K)])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail += 1
            if lp_fail > 15:
                constraint_set = set(int(k) for k in constraint_list[:max(100, initial_K//2)])
            continue
        
        lp_fail = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Check for violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # FW step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new /= sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx > 1e-15:
                x_new /= sx
            else:
                x_new = y_new.copy()
        
        y = y_new
        x = x_new
        iter_count += 1
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_M
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_M
