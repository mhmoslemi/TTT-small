"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Approach (deliberately different from the previous program, which only used
plain MGD + block + coordinate moves and did NOT use the LP direction from the
state-of-the-art):
  * Smoothed multiplicative gradient descent (softmax-weighted top peaks) for a
    much better, lower-variance subgradient.
  * True state-of-the-art LP direction finding: at the current point a with
    M=max(a*a), solve the primal LP  max sum(b) s.t. (a*b)_k<=M, b>=0  (the
    dual is an interval covering LP). Blend a <- (1-t)a + t*norm(b).
  * Symmetric MGD variant (soft symmetry) as a multi-start that often reaches a
    different, lower local minimum.
  * Simulated annealing with random block moves to escape local minima.
  * Aggressive coordinate descent that moves mass out of the peak-contributing
    pairs of the top lags.
  * Multi-length exploration (interpolate to a longer n and re-optimize, since
    larger n gives more degrees of freedom to flatten the autocorrelation).

All helpers are top-level, no closures, no lambdas.
"""

import numpy as np

def fast_autocorr(a):
    """Linear autocorrelation a*a computed through a real FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Clip to >=0 and rescale so the sum is 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Call evaluate_sequence defensively."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float("inf")


def score(a):
    """Fast internal score (scale-invariant), used during search."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return float("inf")
    return 2.0 * n * fast_autocorr(a).max() / (S * S)


def interpolate_sequence(f, new_n):
    """Linearly resample a normalized non-negative sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def soft_top_peaks(ac, top_k):
    """Return (index array, normalized softmax weights) of the top |ac| peaks."""
    kk = int(min(top_k, len(ac)))
    order = np.argsort(ac)[::-1][:kk]
    vals = ac[order]
    shifted = vals - vals.max()
    w = np.exp(50.0 * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(kk) / kk
    else:
        w = w / ws
    return order, w


def compute_grad_weighted(a, lags, weights):
    """Subgradient of sum_k w_k*(a*a)_k w.r.t. a (each peak contributes a band)."""
    n = len(a)
    grad = np.zeros(n)
    for wi, k in zip(weights, lags):
        ki = int(k)
        s = max(0, ki - n + 1)
        e = min(n, ki + 1)
        if s < e:
            idx = np.arange(s, e)
            grad[idx] += 2.0 * wi * a[ki - idx]
    grad = grad - grad.mean()   # project out the simplex-conserved direction
    return grad

def mgd_smooth(a0, time_limit, top_k=12, eta0=0.03, sym=False, verbose=False):
    """Multiplicative gradient descent on the simplex with a softened max."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    eta = eta0
    patience = 0
    max_patience = 600
    it = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience > max_patience:
            break

        lags, w = soft_top_peaks(ac, top_k)
        grad = compute_grad_weighted(a, lags, w)
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        if sym:
            a_new = 0.5 * (a_new + a_new[::-1])
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        ac_new = fast_autocorr(a_new)
        M_new = ac_new.max()

        if M_new <= M + 1e-10:
            a = a_new
            eta *= 0.999
        else:
            eta *= 0.7
            if eta < 1e-4:
                eta = 0.012
                patience = max(0, patience - 250)

        it += 1
        if verbose and it % 200 == 0:
            print("    mgd_smooth it=%d n=%d J=%.6f eta=%.5f pat=%d" %
                  (it, n, 2.0 * n * fast_autocorr(best_a).max(), eta, patience))

    return best_a

def build_conv_matrix_csc(a):
    """Matrix C with C[k,i]=a[i] for 0<=k-i<n (the linear-convolution matrix).

    Built directly in compressed-sparse-column form. Each column i has the value
    a[i] in exactly n consecutive rows i..i+n-1.
    """
    n = len(a)
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    indptr = np.arange(n + 1) * n
    data = np.tile(a, n)
    return csc_matrix((data, rows, indptr), shape=(2 * n - 1, n))


def lp_direction(a, frac_of_max, max_time=15.0):
    """Solve max sum(b) s.t. (a*b)_k <= M, b>=0 and return the normalized b.

    Returns None on failure / timeout.
    """
    n = len(a)
    ac = fast_autocorr(a)
    M = float(ac.max())
    if M < 1e-18:
        return None
    C = build_conv_matrix_csc(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method="highs",
                      options={"time_limit": max_time, "presolve": True,
                               "mip_rel_gap": 0.0})
        if res.status == 0:
            b = np.asarray(res.x, dtype=np.float64)
            b = np.maximum(b, 0.0)
            bs = b.sum()
            if bs < 1e-15:
                return None
            return b / bs
    except Exception:
        return None
    return None


def lp_iterate(a0, time_limit, n_iters, tblend, verbose=True):
    """State-of-the-art loop: solve the LP, blend, repeat."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    lp_time = min(15.0, max(3.0, (time_limit / max(1, n_iters)) * 0.6))

    for r in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        g = lp_direction(a, 1.0, max_time=lp_time)
        if g is None:
            break
        a = (1.0 - tblend) * a + tblend * g
        a = norm_sum1(a)
        m = fast_autocorr(a).max()
        if m < best_m - 1e-15:
            best_m = m
            best_a = a.copy()
            if verbose:
                print("    lp_iter %d J=%.6f" % (r, 2.0 * n * m))

    return best_a

def anneal_search(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)

    T_start = 0.04
    T_end = 0.0004
    it = 0
    block_sizes = (4, 8, 16, 32, 64)

    while time.monotonic() - t0 < time_limit:
        # batch a couple thousand moves per temperature sweep for a smooth curve
        prog = min(1.0, (time.monotonic() - t0) / max(1e-9, time_limit))
        T = T_start * (T_end / T_start) ** prog

        for _ in range(2000):
            if time.monotonic() - t0 > time_limit:
                break
            bs = block_sizes[int(rng.integers(len(block_sizes)))]
            start = int(rng.integers(0, max(1, n - bs + 1)))
            end = min(n, start + bs)
            block = a[start:end].copy()
            base = block.sum() + 1e-15
            # perturbation that preserves the block's mass on average
            pert = rng.normal(0.0, 0.12 * base / max(1, bs), end - start)
            perturb = rng.normal(0.0, 0.03 * base, end - start)
            pert = pert - pert.mean() + perturb
            new_block = np.maximum(block + pert, 0.0)

            a_new = a.copy()
            a_new[start:end] = new_block
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s

            m_new = fast_autocorr(a_new).max()
            delta = m_new - m
            if delta < 0.0 or (T > 0.0 and
                               rng.random() < math.exp(-max(0.0, delta) / T)):
                a = a_new
                m = m_new

            if m_new < best_m:
                best_m = m_new
                best_a = a_new.copy()

        it += 1
        if verbose and it % 5 == 0:
            print("    anneal it=%d J=%.6f T=%.4f" % (it, 2.0 * n * best_m, T))

    return best_a

# Aggressive coordinate descent on peak-contributing pairs
# --------------------------------------------------------------------------- #
def coord_polish(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(7)
    stall = 0
    max_stall = 40

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        top_lags = np.argsort(ac)[::-1][:8]

        improved = False
        # Build a "pressure" map: sum over active lags of the partner value.
        pressure = np.zeros(n)
        for k in top_lags:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            if s < e:
                idx = np.arange(s, e)
                pressure[idx] += a[ki - idx]

        for k in top_lags[:4]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            if s >= e:
                continue
            idx = np.arange(s, e)
            pairs = a[idx] * a[ki - idx]
            src_pos = idx[np.argmax(pairs)]
            amt0 = a[src_pos]
            if amt0 < 1e-10:
                continue
            # destination = low pressure, not near this peak
            cand = np.where(pressure <= pressure.max() * 0.5)[0]
            if len(cand) == 0:
                cand = np.arange(n)
            dst = int(cand[np.argmin(pressure[cand])])
            if dst == src_pos:
                continue
            for frac in (0.05, 0.10, 0.18, 0.30):
                amt = amt0 * frac
                a_t = a.copy()
                a_t[src_pos] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t / st
                m_t = fast_autocorr(a_t).max()
                if m_t < best_m - 1e-14:
                    a = a_t
                    best_m = m_t
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            # random escape move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * float(rng.uniform(0.02, 0.10))
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                if fast_autocorr(a).max() < best_m:
                    best_m = fast_autocorr(a).max()
                    best_a = a.copy()
                    stall = 0

    return best_a


# --------------------------------------------------------------------------- #
# Parametric starting families
# --------------------------------------------------------------------------- #
def parametric_init(n, ref, ref_score):
    """Try a handful of smooth families; return the best normalized one."""
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    i = np.arange(n, dtype=np.float64)
    fams = []
    for p in (0.3, 0.6, 1.0, 1.5, 2.5):
        fams.append((i + 1.0) ** p)
        fams.append((n - i).clip(0.5) ** p)
    for c in (0.5, 0.6, 0.7):
        for sg in (0.10, 0.20, 0.35):
            fams.append(np.exp(-((x - c) ** 2) / (2.0 * sg * sg)))
    for fr in (1, 2):
        fams.append(np.maximum(np.sin(np.pi * fr * x) ** 2, 0.02))
    fams.append(np.ones(n))
    fams.append(np.sin(np.pi * x) ** 2 + 0.05)

    best = ref.copy()
    best_s = ref_score
    for f in fams:
        fn = norm_sum1(f)
        s = score(fn)
        if s < best_s:
            best_s = s
            best = fn.copy()
    return best, best_s

# Orchestrator
# --------------------------------------------------------------------------- #
def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    TIME = min(float(budget_s) - 60.0, 1740.0)
    MAX_N = 7000

    best_seq = None
    best_score = float("inf")

    def remaining():
        return TIME - (time.monotonic() - t_global)

    # ---- Load parent ------------------------------------------------------
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            pn = norm_sum1(p)
            sc = evaluate_safe(pn)
            if np.isfinite(sc):
                best_seq = pn
                best_score = sc
                print("Parent n=%d score=%.6f" % (len(p), sc))
    except Exception:
        pass

    if best_seq is None:
        n0 = 4000
        x = np.linspace(0.0, 1.0, n0, endpoint=False)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2 + 0.05)
        best_score = evaluate_safe(best_seq)
        print("Default init n=%d score=%.6f" % (n0, best_score))

    n = min(len(best_seq), MAX_N)

    # ---- Phase 1 : parametric multi-start (cheap) -------------------------
    rem = remaining()
    if rem > 30:
        t1 = min(rem * 0.03, 40.0)
        ap, sp = parametric_init(n, best_seq, best_score)
        scp = evaluate_safe(ap)
        if np.isfinite(scp) and scp < best_score:
            best_score = scp
            best_seq = ap.copy()
            print("Parametric improved to %.6f" % scp)

    # ---- Phase 2 : primary smooth MGD ------------------------------------
    rem = remaining()
    if rem > 60:
        t2 = min(rem * 0.26, 420.0)
        out = mgd_smooth(norm_sum1(best_seq), t2, top_k=12, eta0=0.03,
                         sym=False, verbose=True)
        sc = evaluate_safe(out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = out.copy()
            print("*** NEW BEST mgd %.6f (n=%d)" % (sc, len(best_seq)))

    # ---- Phase 3 : state-of-the-art LP direction -------------------------
    rem = remaining()
    if rem > 60 and len(best_seq) <= 5600:
        n_lp = len(best_seq)
        iters = max(6, int(min(14, (rem * 0.16) / 20.0)))
        t3 = min(rem * 0.16, 260.0)
        print("\n=== LP SOTA (n=%d, ~%d iters) ===" % (n_lp, iters))
        out = lp_iterate(norm_sum1(best_seq), t3, iters, tblend=0.35, verbose=True)
        sc = evaluate_safe(out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = out.copy()
            print("*** NEW BEST lp %.6f" % sc)
        # short MGD polish right after LP to re-converge
        rem = remaining()
        if rem > 40:
            out = mgd_smooth(norm_sum1(best_seq), min(rem * 0.25, 120.0),
                             top_k=10, eta0=0.02, verbose=False)
            sc = evaluate_safe(out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = out.copy()
                print("*** NEW BEST lp+mgd %.6f" % sc)

    # ---- Phase 4 : symmetric MGD multi-start -----------------------------
    rem = remaining()
    if rem > 50:
        a_sym = norm_sum1(0.5 * (best_seq + best_seq[::-1]))
        t4 = min(rem * 0.10, 150.0)
        out = mgd_smooth(a_sym, t4, top_k=10, eta0=0.025, sym=True, verbose=False)
        sc = evaluate_safe(out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = out.copy()
            print("*** NEW BEST sym %.6f" % sc)

    # ---- Phase 5 : annealing to escape -----------------------------------
    rem = remaining()
    if rem > 60 and len(best_seq) <= 5600:
        t5 = min(rem * 0.16, 250.0)
        out = anneal_search(norm_sum1(best_seq), t5, verbose=True)
        sc = evaluate_safe(out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = out.copy()
            print("*** NEW BEST anneal %.6f" % sc)
        rem = remaining()
        if rem > 40:
            out = mgd_smooth(norm_sum1(best_seq), min(rem * 0.3, 120.0),
                             top_k=10, eta0=0.02, verbose=False)
            sc = evaluate_safe(out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = out.copy()
                print("*** NEW BEST anneal+mgd %.6f" % sc)

    # ---- Phase 6 : multi-length (larger n -> more freedom) ----------------
    rem = remaining()
    n_cur = len(best_seq)
    for delta in (400, 900, 1600):
        new_n = min(n_cur + delta, MAX_N)
        if new_n <= n_cur:
            continue
        rem = remaining()
        if rem < 80:
            break
        print("\n=== longer n=%d ===" % new_n)
        a_long = interpolate_sequence(best_seq, new_n)
        t6 = min(rem * 0.30, 130.0)
        out = mgd_smooth(a_long, t6, top_k=12, eta0=0.025, verbose=False)
        sc = evaluate_safe(out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = out.copy()
            print("*** NEW BEST n=%d %.6f" % (new_n, sc))
        rem = remaining()
        if rem > 40 and new_n <= 5600:
            out = coord_polish(norm_sum1(best_seq), min(rem * 0.4, 80.0),
                               verbose=False)
            sc = evaluate_safe(out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = out.copy()
                print("*** NEW BEST cc n=%d %.6f" % (len(best_seq), sc))

    # ---- Phase 7 : coordinate polish + final MGD -------------------------
    rem = remaining()
    if rem > 30 and len(best_seq) <= 6000:
        t7 = min(rem * 0.35, 200.0)
        print("\n=== coordinate polish ===")
        out = coord_polish(norm_sum1(best_seq), t7, verbose=True)
        sc = evaluate_safe(out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = out.copy()
            print("*** NEW BEST cc %.6f" % sc)

    rem = remaining()
    if rem > 25:
        t8 = min(rem * 0.6, 150.0)
        out = mgd_smooth(norm_sum1(best_seq), t8, top_k=10, eta0=0.01,
                         verbose=False)
        sc = evaluate_safe(out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = out.copy()
            print("*** NEW BEST final mgd %.6f" % sc)

    # ---- Final validation ------------------------------------------------
    if best_seq is None:
        n_fb = 4000
        x = np.linspace(0.0, 1.0, n_fb, endpoint=False)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2 + 0.05)
        best_score = evaluate_safe(best_seq)
        print("Fallback used score=%.6f" % best_score)

    fs = evaluate_safe(best_seq)
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    final = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    print("\nFINAL score=%.6f n=%d time=%.0fs" %
          (best_score, len(final), time.monotonic() - t_global))
    return [float(v) for v in final]
