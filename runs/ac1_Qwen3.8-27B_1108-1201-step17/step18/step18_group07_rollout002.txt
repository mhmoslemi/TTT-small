import numpy as np
import time
import math

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0]*a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, n=fft_size)[:size]

def fast_conv_max(a):
    """Fast: compute only max of conv(a,a) for sum-1 normalized a."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    c = np.fft.irfft(A * A, n=fft_size)[:size]
    return c.max(), c

def simplex_project(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    # Simple projection via bisection on Lagrange multiplier
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    result = np.maximum(v - lam, 0.0)
    s = result.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return result / s

def smooth_max(c, beta, top_k=None):
    """Compute smoothed max of c using log-sum-exp on top_k elements."""
    if top_k is not None and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        vals = c[idx]
    else:
        vals = c
    m = vals.max()
    if m < 1e-30:
        return 0.0, np.ones(len(vals), dtype=np.float64) / len(vals)
    exps = np.exp(beta * (vals - m))
    exps = np.maximum(exps, 1e-300)
    w = exps / exps.sum()
    return m + np.log(exps.sum()) / beta, w

def spectral_equalize(a, alpha=0.2):
    """Flatten the magnitude spectrum of a."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    mag = np.abs(A)
    target_mag = np.full_like(mag, mag.mean())
    new_mag = (1 - alpha) * mag + alpha * target_mag
    phase = np.angle(A)
    A_new = new_mag * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n=fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def obj_score(a):
    """Compute 2*n*max(conv(a,a)) for sum-1 normalized a."""
    n = len(a)
    c_max, _ = fast_conv_max(a)
    return 2.0 * n * c_max

def ftrl_optimize(a0, time_limit, verbose=False):
    """FTRL-style mirror descent with smoothed max and spectral equalization."""
    n = len(a0)
    a = simplex_project(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    x = a.copy()  # Nesterov position
    
    # Annealing parameters
    beta_init = 10.0
    beta_final = 200.0
    top_k_init = 20
    top_k_final = 3
    eta_init = 0.5
    eta_decay = 0.1
    
    steps = 0
    while time.monotonic() - t0 < time_limit:
        steps += 1
        progress = min(1.0, steps / 30000.0)
        
        # Anneal parameters
        beta = beta_init * (beta_final / beta_init) ** progress
        top_k = max(top_k_final, int(top_k_init - (top_k_init - top_k_final) * progress))
        eta = eta_init / (1.0 + eta_decay * steps ** 0.5)
        
        # Nesterov extrapolation
        momentum = 0.9 * progress  # Ramp up momentum
        y = x + momentum * (x - a)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # Compute convolution and subgradient
        c = fast_conv(y)
        c_max, w_peaks = smooth_max(c, beta=beta, top_k=top_k)
        
        # Build subgradient: g[j] = 2 * sum_k w_k * a[k-j]
        g = np.zeros(n)
        # Get the actual peak indices
        if top_k < len(c):
            peak_idx = np.argpartition(c, -top_k)[-top_k:]
        else:
            peak_idx = np.arange(len(c))
        
        for wi, k in zip(w_peaks, peak_idx):
            if wi < 1e-10:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)

            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            g[idx] += 2.0 * wi * y[j_idx]
        
        # Mirror descent step (exponential family)
        log_a = np.log(np.maximum(y, 1e-15)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x = a
        a = a_new
        
        # Track best
        c_new = fast_conv(a)
        M_new = c_new.max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
        
        # Periodic spectral equalization

            alpha_spec = 0.3 * (1.0 - 0.5 * progress)
            a_eq = spectral_equalize(a, alpha_spec)
            c_eq = fast_conv(a_eq)
            if c_eq.max() < best_m:
                best_m = c_eq.max()
                best_a = a_eq.copy()
                a = a_eq
                x = a_eq.copy()
        
        # Occasional random perturbation (simulated annealing escape)
        if steps % 5000 == 0 and progress < 0.8:
            rng_pert = np.random.default_rng(steps)
            pert = rng_pert.normal(0, 0.01 * (1 - progress), n)
            a_trial = a + pert
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                c_trial = fast_conv(a_trial)
                if c_trial.max() < best_m * (1.0 + 0.05 * np.exp(-steps / 5000.0)):
                    a = a_trial
                    x = a_trial.copy()

    return best_a, best_m

def spectral_optimize(a0, time_limit, verbose=False):

    n = len(a0)

    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    alpha = 0.2
    for it in range(50000):

            break
        
        a_new = spectral_equalize(a, alpha)
        c_new = fast_conv(a_new)
        M_new = c_new.max()
        
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            alpha = min(0.4, alpha * 1.02)
        else:

            # Revert half-way
            a = 0.5 * a + 0.5 * best_a
            a = simplex_project(a)

def pair_rejection(a0, time_limit, verbose=False):

        c = fast_conv(a)
        M_curr = c.max()
        
        # Get top peaks
        n_top = min(5, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]

            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            
            # Find the pair with highest contribution
            top_pairs = np.argsort(contrib)[::-1][:3]
            
            for pi in top_pairs:
                src = idx[pi]
                partner = k_int - src
                src_val = a[src]
                if src_val < 1e-10:
                    continue

                    if dst == src or dst == partner:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt

                    s_t = a_trial.sum()
                    if s_t < 1e-15:

                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial

                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation

            j = int(rng.integers(n))

                a[i] -= amt

                a = np.maximum(a, 0)

                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_m:

def differential_evolution(a0, time_limit, pop_size=15, verbose=False):

def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.1
    s = a.sum()
    return a / s

def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.05
    s = a.sum()
    return a / s

def make_init_ramp(n, power=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    s = a.sum()
    return a / s

def make_init_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_init_random_smooth(n, seed=42):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 15):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    s = a.sum()
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
