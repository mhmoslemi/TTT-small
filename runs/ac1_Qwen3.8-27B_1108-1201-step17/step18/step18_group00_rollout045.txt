def smoothed_gradient_descent(s, time_limit, tau_schedule='linear', n_restarts=5):
    n = len(s)
    s = np.maximum(s, 0)
    s = s / s.sum()
    
    best_s = s.copy()
    best_val = 2.0 * n * fast_autocorr(s).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.01
    
    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Reset optimizer state on restart
        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        
        # Add perturbation on restart (except first)
        if restart > 0:
            noise_level = 0.05 / (1 + restart * 0.5)
            s_pert = best_s + rng.normal(0, noise_level, n)
            s_pert = np.maximum(s_pert, 0)
            s = s_pert / s_pert.sum()
        
        # Temperature schedule
        n_iters = 5000
        for t in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Temperature: anneal from tau_min to tau_max
            progress = t / n_iters
            tau = 10.0 + 100.0 * progress  # from 10 to 110
            
            # Compute autocorrelation
            ac = fast_autocorr(s)
            max_ac = ac.max()
            current_val = 2.0 * n * max_ac
            
            if current_val < best_val - 1e-10:
                best_val = current_val
                best_s = s.copy()
            
            # Compute gradient
            scaled = tau * ac
            max_scaled = scaled.max()
            exp_scaled = np.exp(scaled - max_scaled)
            p = exp_scaled / exp_scaled.sum()
            
            g = compute_gradient_fft(s, p, n)
            
            # Adam update
            t_adam += 1
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * g * g
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            s_new = s - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Project: clip and normalize
            s_new = np.maximum(s_new, 0)
            s_sum = s_new.sum()
            if s_sum < 1e-15:
                s_new = np.ones(n) / n
            else:
                s_new = s_new / s_sum
            
            s = s_new
        
        # Reduce learning rate
        lr *= 0.5
    
    return best_s, best_val

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_grad_fft(s, p, n):
    """Compute g[j] = 4n * sum_m s[m] * p[m+j] for j=0,...,n-1 using FFT."""
    p_flip = p[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    S = np.fft.rfft(s, fft_size)
    P = np.fft.rfft(p_flip, fft_size)
    conv = np.fft.irfft(S * P, fft_size)
    q = conv[n - 1:2 * n - 1][::-1]
    return 4.0 * n * q


def simplex_project(x):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(x)
    if n == 0:
        return x
    if x.sum() < 1e-15:
        return np.ones(n) / n
    # Sort in descending order
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not np.any(cond):
        return np.ones(n) / n
    k = np.max(ind[cond])
    theta = cssv[k - 1] / k
    proj = np.maximum(x - theta, 0.0)
    s = proj.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return proj / s


def scaled_value(s):
    """Compute 2*n*max(ac) for sum-normalized s."""
    n = len(s)
    ac = fast_autocorr(s)
    return 2.0 * n * ac.max()


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
