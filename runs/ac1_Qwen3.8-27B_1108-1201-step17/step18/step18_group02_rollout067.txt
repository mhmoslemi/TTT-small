import numpy as np

_cached_solver = None


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def get_solver():
    global _cached_solver
    if _cached_solver is not None:
        return _cached_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _cached_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _cached_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _cached_solver

def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k - idx]
    return coeffs


def lp_frank_wolfe(a0, time_limit, solver, initial_K=1000, max_K=8000,
                   away_interval=30, max_iters=200000, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()

    y = a.copy()
    rng = np.random.default_rng(777)

    init_idx = np.argsort(ac)[::-1][:initial_K]
    constraint_set = list(set(int(k) for k in init_idx))

    stall = 0
    iter_count = 0

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        if stall > 800:
            pert = rng.normal(0, 0.03, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-15)
            x = x / x.sum()
            y = x
            ac_new = fast_autocorr(y)
            constraint_set = list(set(int(k) for k in np.argsort(ac_new)[::-1][:initial_K]))
            stall = 0
            continue

        if len(constraint_set) > max_K:
            margins = np.array([M_curr - ac_y[k] for k in constraint_set])
            keep = np.argsort(margins)[:max_K]
            constraint_set = [constraint_set[i] for i in keep]

        violations = np.where(ac_y > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = ac_y[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:80]]
            for k in top_v:
                ki = int(k)
                if ki not in constraint_set:
                    constraint_set.append(ki)

        if t % 200 == 0:
            rks = rng.integers(0, 2 * n - 1, size=20)
            for rk in rks:
                ri = int(rk)
                if ri not in constraint_set:
                    constraint_set.append(ri)

        if len(constraint_set) == 0:
            break

        coeffs = build_constraint_matrix(y, constraint_set, n)
        K = len(constraint_set)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            constraint_set = constraint_set[:initial_K]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if len(constraint_set) > initial_K:
                constraint_set = constraint_set[:initial_K]
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        if t % away_interval == 0:
            away_idx = int(np.argmax(y))
            g_away = np.zeros(n)
            g_away[away_idx] = 1.0
            g_comb = 0.7 * g_dir + 0.3 * (1.0 - g_away)
            g_comb = np.maximum(g_comb, 0.0)
            g_comb_sum = g_comb.sum()
            if g_comb_sum > 1e-10:
                g_dir = g_comb / g_comb_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        y = y_new
        iter_count += 1

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |C|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def coord_descent(a0, time_limit, rng_seed=42, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            top_dsts = candidates[np.argsort(cand_scores)[:20]]

            for dst in top_dsts[:10]:
                if dst == src:
                    continue

                lo = 0.001
                hi = 0.4
                best_frac = None
                best_trial_m = best_m

                for _ in range(12):
                    mid = (lo + hi) / 2.0
                    amt = src_val * mid
                    if amt < 1e-15:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        break
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_trial_m:
                        best_trial_m = M_trial
                        best_frac = mid
                        lo = mid
                    else:
                        hi = mid

                if best_frac is not None:
                    amt = src_val * best_frac
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    best_m = best_trial_m
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if improved:
                break

        if not improved:
            stall += 1
            if stall % 50 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
            if verbose and stall % 100 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def multiplicative_descent(a0, time_limit, lr=0.05, smooth_tau=1.0, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(55)

    tau = max(fast_autocorr(a).max() * 0.5, 1e-6)

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M = ac.max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()

        tau = max(tau * 0.9998, 1e-8)
        ac_scaled = ac / tau
        ac_scaled -= ac_scaled.max()
        exp_vals = np.exp(ac_scaled)
        p = exp_vals / exp_vals.sum()

        a_flip = a[::-1].astype(np.float64)
        size = len(ac)
        fft_size = 1 << (size - 1).bit_length()
        Af = np.fft.rfft(a_flip, fft_size)
        Pf = np.fft.rfft(p.astype(np.float64), fft_size)
        conv_full = np.fft.irfft(Af * Pf, fft_size)
        grad = 2.0 * conv_full[n - 1:n - 1 + n]
        grad = np.maximum(grad, 0.0)

        log_a = np.log(np.maximum(a, 1e-30)) - lr * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        a = a_new

        if t % 5000 < 100:
            pert = rng.normal(0, 0.01, n) * a
            a = np.maximum(a + pert, 1e-15)
            a = a / a.sum()

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MultDescent t={t} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.2e}")

    # Phase 6: Final LP-FW aggression on best
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final LP-FW (n={n_best}) ===")
    if remaining > 50 and n_best <= 6000:
        t6 = min(remaining * 0.40, 250.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t6, solver, initial_K=1500, max_K=12000,
                                  away_interval=15, verbose=True)
        check_and_update(a_out, f"lp_final_n{n_best}")

    # Phase 7: Final coordinate descent
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final coord (n={n_best}) ===")
    if remaining > 15:
        t7 = min(remaining * 0.50, 100.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_v2(a_in, t7, rng_seed=314, verbose=False)
        check_and_update(a_out, f"coord2_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
