    idx = np.argsort(ac)[::-1][:top_k]

    m = vals.max()
    exp_vals = np.exp(beta * (vals - m))

def block_optimizer(a0, time_limit, block_size=20, verbose=False):
    """
    Optimizes blocks of the sequence by trying local perturbations.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 200
    n_blocks = max(1, (n - block_size) // block_size)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        ac = fast_autocorr(a)
        M = ac.max()
        
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Random block position
            start = int(rng.integers(0, max(1, n - block_size + 1)))
            end = min(n, start + block_size)
            bs = end - start
            if bs < 3:
                continue
            
            block = a[start:end].copy()
            mean_b = block.mean() + 1e-10
            
            # Try several perturbation strategies
            for strategy in range(3):
                if strategy == 0:
                    # Gaussian perturbation
                    pert = rng.normal(0, 0.05 * mean_b, bs)
                elif strategy == 1:
                    # Move mass within block
                    i1 = int(rng.integers(bs))
                    i2 = int(rng.integers(bs))
                    if i1 == i2:
                        continue
                    amt = block[i1] * 0.1
                    pert = np.zeros(bs)
                    pert[i1] = -amt
                    pert[i2] = amt
                else:
                    # Scale a subset
                    mask = rng.random(bs) < 0.5
                    pert = np.zeros(bs)
                    pert[mask] = 0.03 * mean_b
                    pert[~mask] = -0.03 * mean_b
                    
                new_block = np.maximum(block + pert, 0.0)
                if abs(new_block.sum() - block.sum()) < 1e-15:
                    continue
                
                a_trial = a.copy()
                a_trial[start:end] = new_block
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                
                if M_trial < M - 1e-14:
                    a = a_trial
                    M = M_trial
                    if M < best_m:
                        best_m = M
                        best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
        
        if verbose and stall % 50 == 0:
            J = 2.0 * n * best_m
            elapsed = time.monotonic() - t0
            print(f"    Block iter stall={stall} ({elapsed:.0f}s): J={J:.6f}")
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Block done: J={J:.6f}")
    return best_a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Softmax-smoothed multiplicative gradient descent with annealed temperature
2. Multi-start with diverse parametric initializations (exploiting asymmetry)
3. Coordinate descent directly on the true max objective
4. Block-level local search with adaptive strategies
5. Multi-length exploration via interpolation
Key insight: optimal sequences are asymmetric; the max autocorrelation peak
is NOT at the center lag, enabling scores below 2.

import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear autocorrelation (convolution with itself) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_top_weights(ac, beta, top_k=40):
    """Get softmax weights and indices for top-k lags of autocorrelation."""
    k = min(top_k, len(ac))
    idx = np.argsort(ac)[::-1][:k]
    vals = ac[idx]
    m = vals.max()
    exp_vals = np.exp(beta * (vals - m))
    w_sum = exp_vals.sum()
    if w_sum < 1e-30:
        w = np.ones(k) / k
    else:
        w = exp_vals / w_sum
    return idx, w


def compute_grad_weighted(a, idx, w):
    """
    Compute gradient of weighted sum of autocorrelation values.
    d(conv(a,a)[k])/da[i] = 2*a[k-i] if 0<=k-i<n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    for wi, k in zip(w, idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n - 1, ki)
        if i_start <= i_end:
            ii = np.arange(i_start, i_end + 1)
            grad[ii] += 2.0 * wi * a[ki - ii]
    return grad


def mgd_annealed(a0, time_limit, beta_start, beta_end, top_k, eta0,
                 patience_max=300, verbose=False):
    """
    Multiplicative Gradient Descent with annealed soft-max objective.
    Beta increases over time to transition from smooth to sharp objective.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_true_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    eta = eta0
    patience = 0
    iter_count = 0
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1.0))
        # Anneal beta with a smooth schedule
        beta = beta_start + (beta_end - beta_start) * (progress ** 0.7)

        ac = fast_autocorr(a)
        true_m = ac.max()

        if true_m < best_true_m - 1e-15:
            best_true_m = true_m
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > patience_max:
            break

        k_use = min(cur_top_k, len(ac))
        idx, w = softmax_top_weights(ac, beta, k_use)
        grad = compute_grad_weighted(a, idx, w)

        # Center gradient for simplex
        grad = grad - grad.mean()

        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax

        # Multiplicative update (mirror descent on simplex)
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        ac_new = fast_autocorr(a_new)
        m_new = ac_new.max()

        if m_new <= true_m + 1e-12:
            a = a_new
            eta *= 0.9992
        else:
            eta *= 0.5
            if eta < 0.0005:
                cur_top_k = max(5, cur_top_k // 2)
                eta = max(0.005, eta0 * 0.5)
                patience = max(0, patience - 50)

        iter_count += 1
        if verbose and iter_count % 150 == 0:
            J = 2.0 * n * best_true_m
            print(f"      iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, "
                  f"beta={beta:.1f}, eta={eta:.5f}, tk={cur_top_k}")

    if verbose:
        J = 2.0 * n * best_true_m
        print(f"      MGD-anneal done: J={J:.6f}, iters={iter_count}")
    return best_a


def coord_descent_direct(a0, time_limit, verbose=False):
    """
    Coordinate descent directly minimizing the true max autocorrelation.
    Moves mass from peak-contributing positions to low-impact positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        n_top = min(10, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start > i_end:
                continue

            ii = np.arange(i_start, i_end + 1)
            contrib = a[ii] * a[ki - ii]
            if contrib.max() < 1e-20:
                continue
            src = ii[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score each position by total peak contribution
            peak_impact = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    peak_impact[idx2] += a[k2i - idx2]
                    peak_impact[k2i - idx2] += a[idx2]

            peak_impact[src] = 1e30
            candidates = np.where(peak_impact < 1e29)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            dst = candidates[np.argmin(peak_impact[candidates])]

            for frac in [0.008, 0.015, 0.025, 0.04, 0.06]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < M - 1e-15:
                    a = a_trial
                    M = M_trial
                    if M < best_m:
                        best_m = M
                        best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random escape perturbation
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.04)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    break

    if verbose:
        J = 2.0 * n * best_m
        print(f"      CoordDescent done: J={J:.6f}")
    return best_a


def block_local_search(a0, time_limit, block_size=20, verbose=False):
    """
    Block-level local search: perturb random blocks, keep improvements.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)

    stall = 0
    max_stall = 250

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        improved = False

        # Focus on peak-contributing regions
        top_lags = np.argsort(ac)[::-1][:8]

        for attempt in range(40):
            if time.monotonic() - t0 > time_limit:
                break

            if attempt < 12 and len(top_lags) > 0:
                k_pick = int(top_lags[attempt % len(top_lags)])
                js = max(0, k_pick - n + 1)
                je = min(n - 1, k_pick)
                center = (js + je) // 2
                start = max(0, min(n - block_size,
                                   center - block_size // 2 + int(rng.integers(-8, 8))))
            else:
                start = int(rng.integers(0, max(1, n - block_size + 1)))

            end = min(n, start + block_size)
            bs = end - start
            if bs < 3:
                continue

            block = a[start:end].copy()
            mean_b = block.mean() + 1e-10
            scale = rng.uniform(0.02, 0.06)
            pert = rng.normal(0, scale * mean_b, bs)
            new_block = np.maximum(block + pert, 0.0)

            if abs(new_block.sum() - block.sum()) < 1e-15:
                continue

            a_trial = a.copy()
            a_trial[start:end] = new_block
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()

            if M_trial < M - 1e-14:
                a = a_trial
                M = M_trial
                if M < best_m:
                    best_m = M
                    best_a = a.copy()
                improved = True
                if attempt >= 25:
                    break

        if not improved:
            stall += 1
            # Escape: random redistribution
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        J = 2.0 * n * best_m
        print(f"      BlockSearch done: J={J:.6f}")
    return best_a


def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def diverse_initializations(n, best_a, best_score):
    """
    Generate diverse parametric initializations and return the best one.
    Key: try asymmetric shapes since optimal sequences are asymmetric.
    """
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    results = []

    # Asymmetric shapes (key insight: optimal is asymmetric)
    for alpha in [1.0, 2.0, 3.0, 5.0, 8.0]:
        f = np.exp(alpha * (x - 1.0))  # concentrated at right
        results.append(norm_sum1(f))
        f = np.exp(-alpha * x)  # concentrated at left
        results.append(norm_sum1(f))

    # Half-sine variations (asymmetric)
    for shift in [0.0, 0.1, 0.2, -0.1, -0.2]:
        f = np.sin(np.pi * np.clip(x + shift, 0, 1) / 2.0) ** 2
        results.append(norm_sum1(f))

    # Power-law shapes
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        results.append(norm_sum1((x + 0.01) ** p))
        results.append(norm_sum1((1.0 - x + 0.01) ** p))

    # Gaussian bumps at various positions
    for center in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.3]:
            f = np.exp(-((x - center) ** 2) / (2.0 * sigma ** 2))
            results.append(norm_sum1(f))

    # Sine squared (symmetric, good baseline)
    results.append(norm_sum1(np.sin(np.pi * x) ** 2))

    # Triangle (asymmetric)
    for peak in [0.3, 0.5, 0.7]:
        f = np.maximum(0, 1 - np.abs(x - peak) / (min(peak, 1 - peak) + 0.01))
        results.append(norm_sum1(f))

    # Try the best existing sequence as well
    results.append(norm_sum1(best_a))

    best_local = best_a.copy()
    best_local_score = best_score

    for fam in results:
        ac = fast_autocorr(fam)
        val = 2.0 * n * ac.max()
        if val < best_local_score:
            best_local_score = val
            best_local = fam.copy()

    return best_local, best_local_score

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish (n={n_best}) ===")

    if remaining > 50:
        t8a = min(remaining * 0.4, 150.0)
        if t8a > 25:
            a_out = coord_descent_direct(best_seq, t8a, verbose=True)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** Final coord: {sc:.6f}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t8b = min(remaining * 0.5, 100.0)
            if t8b > 20:
                a_out = mgd_annealed(best_seq, t8b, beta_start=50.0, beta_end=200.0,
                                     top_k=15, eta0=0.008, verbose=False)
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  *** Final MGD: {sc:.6f}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8c = min(remaining * 0.6, 60.0)
            if t8c > 15:
                a_out = block_local_search(best_seq, t8c, block_size=15, verbose=False)
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  *** Final block: {sc:.6f}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
