"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP Frank-Wolfe with Away Step and sparse constraint management  
2. Log-sum-exp smoothed multiplicative Adam optimizer
3. Coordinate descent with peak-targeted transfers

import numpy as np
import time

_MODULE_SOLVER = None

def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _MODULE_SOLVER


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / max(len(a), 1)
    return a / s


def build_csr(y, constraint_list, n):
    K = len(constraint_list)
    if K == 0:
        return csr_matrix((0, n), dtype=np.float64)
    
    constraint_list = np.asarray(constraint_list, dtype=np.int64)
    i_starts = np.maximum(0, constraint_list - (n - 1))
    i_ends = np.minimum(n, constraint_list + 1)
    lengths = i_ends - i_starts
    lengths = np.maximum(lengths, 0)
    
    total_nnz = int(lengths.sum())
    if total_nnz == 0:
        return csr_matrix((K, n), dtype=np.float64)
    
    rows_all = np.repeat(np.arange(K, dtype=np.int32), lengths)
    
    # Build cols
    cols_parts = []
    for jj in range(K):
        s = int(i_starts[jj])
        e = int(i_ends[jj])
        cols_parts.append(np.arange(s, e, dtype=np.int32))
    if cols_parts:
        cols_all = np.concatenate(cols_parts)
    else:
        cols_all = np.zeros(0, dtype=np.int32)
    
    # Build vals
    vals_all = np.zeros(total_nnz, dtype=np.float64)
    pos = 0
    for jj in range(K):
        s = int(i_starts[jj])
        e = int(i_ends[jj])
        k_val = int(constraint_list[jj])
        L = e - s
        if L > 0:
            vals_all[pos:pos+L] = y[k_val - np.arange(s, e)]
        pos += L
    
    return csr_matrix((vals_all, (rows_all, cols_all)), shape=(K, n), dtype=np.float64)


def lp_frank_wolfe_afw(a0, time_limit, solver, initial_K=800, max_K=5000,
                       add_per_iter=100, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_conv(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    away_set = [a.copy()]  # Store vertices for AFW
    
    # Initial constraint set: top K convolution elements
    init_idx = np.argsort(ac)[::-1][:initial_K]
    constraint_list = sorted([int(k) for k in init_idx])
    
    iter_count = 0
    lp_fail = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac_y = fast_conv(y)
        M_curr = ac_y.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
        
        # Also check x
        if t % 5 == 0:
            ac_x = fast_conv(x)
            M_x = ac_x.max()
            if M_x < best_m - 1e-14:
                best_m = M_x
                best_a = x.copy()
        
        # Constraint management
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_list = sorted([constraint_list[i] for i in keep])
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build sparse matrix
        A_mat = build_csr(y, constraint_list, n)
        
        # Solve LP: min M s.t. sum(g)=1, g>=0, A@g <= M
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A_mat @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_value), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail += 1
            if lp_fail > 10:
                constraint_list = constraint_list[:max(300, initial_K // 2)]
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail += 1
            if lp_fail > 20:
                constraint_list = constraint_list[:max(200, initial_K // 3)]
            continue
        
        lp_fail = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = g_val.sum()
        if gs < 1e-15:
            continue
        g_dir = g_val / gs
        
        # Add violated constraints
        # Compute y * g_dir convolution
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        ygt = np.fft.irfft(Yf * Gf, fft_size)[:2*n-1]
        
        violations = np.where(ygt > M_curr + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = ygt[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_constraints = [int(k) for k in top_v if int(k) not in set(constraint_list)]
            constraint_list.extend(new_constraints)
            constraint_list = sorted(constraint_list)
        
        # AFW step
        alpha = 2.0 / (t + 2.0)
        
        # Away step: find best s in away_set
        lambda_ = 0.0
        s_away = None
        if len(away_set) >= 2 and t > 50:
            # For each s in away_set, find optimal lambda
            # Optimize: min ||(1-alpha-lambda)*x + alpha*g + lambda*s - target||^2
            # Simplified: (g - x) . (s - x) > 0 indicates away step helps
            best_obj = float('inf')
            for s in away_set[-20:]:  # Only check recent vertices
                gs2 = (g_dir - x).dot(s - x)
                if gs2 > 0:
                    # Optimal lambda = gs2 / ||x - s||^2, clipped
                    xs2 = np.sum((x - s)**2)
                    if xs2 > 1e-15:
                        lam_candidate = min(gs2 / xs2, 1.0 - alpha)
                        if lam_candidate > 0:
                            new_obj = np.sum(((1 - alpha - lam_candidate)*x + alpha*g_dir + lam_candidate*s - x)**2)
                            if new_obj < best_obj:
                                best_obj = new_obj
                                lambda_ = lam_candidate
                                s_away = s
        
        if s_away is not None and lambda_ > 0:
            y_new = (1.0 - alpha - lambda_) * y + alpha * g_dir + lambda_ * s_away
        else:
            y_new = (1.0 - alpha) * y + alpha * g_dir
        
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        # Update x (AVG or step)
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        # Update away set
        away_set.append(g_dir.copy())
        if len(away_set) > 50:
            away_set = away_set[-50:]
        
        y = y_new
        x = x_new
        iter_count += 1
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AFW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_list)}, M={M_curr:.6f}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    AFW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def smooth_adam(a0, time_limit, tau_decay=0.9997, lr=0.7, beta1=0.9, beta2=0.999,
                restart_every=2000, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_conv(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    
    tau = max(ac.max() * 0.3, 1e-6)
    tau_min = max(ac.max() * 0.003, 1e-10)
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    stall = 0
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(a)
        M = ac.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        tau = max(tau * tau_decay, tau_min)
        
        # LSE gradient
        ac_scaled = ac / tau
        ac_scaled -= ac_scaled.max()
        exp_vals = np.exp(ac_scaled)
        p = exp_vals / exp_vals.sum()
        
        # grad[i] = 2 * sum_j p[i+j] * a[j] = 2 * (flip(a) * p)[i+n-1]
        a_flip = a[::-1]
        size = len(ac)
        fft_size = 1 << (size - 1).bit_length()
        Af = np.fft.rfft(a_flip, fft_size)
        Pf = np.fft.rfft(p, fft_size)
        conv_full = np.fft.irfft(Af * Pf, fft_size)
        grad = 2.0 * conv_full[n - 1:n - 1 + n]
        grad = np.maximum(grad, 0.0)
        
        m_hat = beta1 * m_hat + (1.0 - beta1) * grad
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad * grad)
        
        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2
        
        step = lr / (np.sqrt(v_c) + 1e-8)
        
        log_a = np.log(np.maximum(a, 1e-30)) - step * m_c
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        a = a_new / s
        
        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.05, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            a = x.copy()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}")
    
    elapsed = time.monotonic() - t0
    print(f"    Adam done ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    return best_a, best_m


def coord_descent(a0, time_limit, rng_seed=42, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        
        top_k = np.argsort(ac)[::-1][:min(10, len(ac))]
        improved = False
        
        for k in top_k[:6]:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find destination with minimal impact
            scores = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]
            
            # Prefer positions not in peak regions
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            
            dst = candidates[np.argmin(scores[candidates])]
            if dst == src:
                continue
            
            for frac in [0.01, 0.02, 0.05, 0.1]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t / st
                Mt = fast_conv(a_t).max()
                if Mt < best_m - 1e-16:
                    a = a_t
                    best_m = Mt
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    Mn = fast_conv(a).max()
                    if Mn < best_m:
                        best_m = Mn
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 40 == 0 and stall > 0:
            print(f"    Coord ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}, stall={stall}")
    
    return best_a, best_m


def interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_n)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
