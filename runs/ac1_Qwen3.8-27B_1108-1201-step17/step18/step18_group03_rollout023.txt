import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def splyn_optimize(a0, time_limit, target_flat=True, verbose=True):
    """Splyn-like algorithm for PAPR reduction with non-negativity constraint."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 2).bit_length()
    
    # Target power spectrum
    if target_flat:
        # Flat target
        target = np.ones(fft_size // 2 + 1)
    else:
        # Raised cosine target
        f = np.linspace(0, 1, fft_size // 2 + 1)
        target = 0.5 * (1 + np.cos(2 * np.pi * f))
    
    stall = 0
    max_stall = 500
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute FFT
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        
        # Compute new amplitude
        P_safe = np.maximum(P, 1e-30)
        new_amp = np.sqrt(P_safe * target)  # Geometric mean
        
        # New complex spectrum
        phase = np.angle(A)
        A_new = new_amp * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project onto non-negative
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
            if verbose and (t < 100 or t % 1000 == 0):
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    Splyn iter={t} ({elapsed:.0f}s): J={J:.6f}")
        else:
            stall += 1
        
        a = a_new
        
        if stall > max_stall:
            break
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    Splyn done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def compute_smoothed_grad(a, ac, beta, top_k=None):
    """Compute gradient of (1/beta)*log(sum exp(beta*ac[k])) w.r.t. a."""
    n = len(a)
    if top_k is None:
        top_k = min(200, len(ac))
    
    # Get top lags by value
    if len(ac) > top_k:
        top_idx = np.argsort(ac)[::-1][:top_k]
    else:
        top_idx = np.arange(len(ac))
    
    top_vals = ac[top_idx]
    
    # Compute softmax weights
    max_v = top_vals.max()
    exp_vals = np.exp(beta * (top_vals - max_v))
    w = exp_vals / exp_vals.sum()
    
    # Compute gradient: grad[j] = sum_k w[k] * d(conv[k])/d(a[j])
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        # d(conv[k])/d(a[j]) = a[k-j] if 0<=k-j<n (and factor of 2 except at j=k/2)
        # Simplification: use 2*a[k-j] for j where k-j is valid
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            # For each j in idx, the contribution is a[k_int - j] (times 2 approx)
            grad[idx] += wi * a[k_int - idx] * 2.0
    
    return grad

                grad[idx] += a[k_int - idx]
        src_candidates = np.where(grad > grad.mean())[0]
        if len(src_candidates) == 0:
            src = int(rng.integers(n))
        else:
            src = int(src_candidates[rng.integers(len(src_candidates))])
        if a[src] < 1e-10:
            continue
        dst_candidates = np.where(grad < grad.mean())[0]
        if len(dst_candidates) == 0:
            dst = int(rng.integers(n))
        else:
            dst = int(dst_candidates[rng.integers(len(dst_candidates))])
        if src == dst:
            continue
        frac = float(rng.uniform(0.01, 0.1))
        amt = a[src] * frac
        a_trial = a.copy()
        a_trial[src] -= amt
        a_trial[dst] += amt
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t
        M_trial = fast_autocorr(a_trial).max()
        delta = M_trial - M_curr
        if delta < 0:
            a = a_trial
            accept_count += 1
        else:
            if T > T_min:
                log_accept = -delta / T
                if log_accept > -50 and np.log(float(rng.random())) < log_accept:
                    a = a_trial
                    accept_count += 1
        T = max(T * cool_rate, T_min)
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.5)
            T = best_T
            pert = rng.normal(0, 0.03, n) * best_a
            a = norm_sum1(best_a + pert)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    GLS restart t={t} ({elapsed:.1f}s): J={J:.6f}")
        if max_stall > 20000:
            break
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    GLS done ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def spectral_mirror(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    rng = np.random.default_rng(42)
    lr = 0.3
    stall = 0
    max_stall = 1000
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        ac = fast_autocorr(a)
        P_mean = P.mean()
        P_dev = np.abs(P - P_mean) / max(P_mean, 1e-30)
        beta_spec = 3.0
        w_spec = np.exp(beta_spec * P_dev)
        w_spec = w_spec / w_spec.sum()
        grad_freq = A * w_spec * 2.0
        grad_time = np.fft.irfft(grad_freq, fft_size)[:n]
        eta = lr / (1.0 + 0.02 * np.sqrt(t))
        a_new = a * np.exp(-eta * grad_time)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and (t < 20 or t % 500 == 0):
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    SM t={t} ({elapsed:.1f}s): J={J:.6f}")
        else:
            stall += 1
            a = a_new
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.4)
            lr *= 0.7
            pert = rng.normal(0, 0.02, n) * best_a
            a = norm_sum1(best_a + pert)
            if max_stall > 10000:
                break
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    Spectral Mirror done ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n):
    i_arr = np.arange(n, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    candidates = []
    for p in [0.5, 0.7, 1.0, 1.3, 1.5]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
    for p in [0.5, 1.0, 1.5]:
        candidates.append(norm_sum1((n - i_arr) ** p))
    for alpha in [0.5, 1.0, 1.5]:
        for beta in [0.5, 1.0, 1.5]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            candidates.append(norm_sum1(a))
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.2, 0.3, 0.4]:
            candidates.append(norm_sum1(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))))
    for c1 in [0.25, 0.3]:
        for c2 in [0.7, 0.75]:
            for sigma in [0.1, 0.15, 0.2]:
                a = np.exp(-((x - c1) ** 2) / (2.0 * sigma * sigma)) + np.exp(-((x - c2) ** 2) / (2.0 * sigma * sigma))
                candidates.append(norm_sum1(a))
    candidates.append(norm_sum1(np.ones(n)))
    best_c = None
    best_v = float('inf')
    for c in candidates:
        ac = fast_autocorr(c)
        v = 2.0 * n * ac.max()
        if v < best_v:
            best_v = v
            best_c = c.copy()
    return best_c, best_v


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    best_seq = None
    best_score = float('inf')
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    if best_seq is None:
        n_base = 5000
        best_seq, _ = parametric_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n0 = len(best_seq)

    print(f"\n=== Phase 1: Splyn pre-processing (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 5:
        a_out, _ = splyn_optimize(best_seq, t1, verbose=False)
        check_and_update(a_out, "splyn_init")

    print(f"\n=== Phase 2: Mirror Descent main (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent(a_in, t2, beta0=20.0, beta_max=200.0,
                                  top_k=150, c0=2.0, verbose=True)
        check_and_update(a_out, f"md_main_n{n_cur}")

    print(f"\n=== Phase 3: Spectral Mirror (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_mirror(a_in, t3, verbose=False)
        check_and_update(a_out, f"specmirror_n{n_cur}")

    print(f"\n=== Phase 4: LP Cutting Plane (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 20 and n_cur <= 6500:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, initial_K=400, max_K=2500,
                                   add_per_iter=50, verbose=True)
        check_and_update(a_out, f"lp_n{n_cur}")

    print(f"\n=== Phase 5: Gradient Local Search (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = gradient_local_search(a_in, t5, verbose=True)
        check_and_update(a_out, f"gls_n{n_cur}")

    print(f"\n=== Phase 6: MD Refinement (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent(a_in, t6, beta0=50.0, beta_max=500.0,
                                  top_k=200, c0=1.0, verbose=True)
        check_and_update(a_out, f"md_refine_n{n_cur}")

    print(f"\n=== Phase 7: Length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    length_candidates = []
    for delta in [300, -300, 600, -600, 1000, -1000]:
        new_n = n_cur + delta
        if 2000 <= new_n <= 7000:
            length_candidates.append(new_n)
    length_candidates = sorted(set(length_candidates))
    for new_n in length_candidates[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)
        t7a = min(remaining * 0.4, 120.0)
        if t7a > 15:
            a_out, _ = mirror_descent(a_long, t7a, beta0=30.0, beta_max=200.0,
                                       top_k=100, c0=2.0, verbose=False)
            check_and_update(a_out, f"md_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and new_n <= 6000:
            t7b = min(remaining * 0.5, 100.0)
            if t7b > 15:

                a_out, _ = lp_frank_wolfe(a_in, t7b, solver, initial_K=300,
                                           max_K=2000, add_per_iter=40, verbose=False)
                check_and_update(a_out, f"lp_n{new_n}")

    print(f"\n=== Phase 8: Final LP polish (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 40 and n_cur <= 6500:
        t8 = min(remaining * 0.5, 200.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t8, solver, initial_K=500,
                                       max_K=3000, add_per_iter=60, verbose=True)
            check_and_update(a_out, f"lp_final_n{n_cur}")

    print(f"\n=== Phase 9: Final GLS polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t9 = min(remaining * 0.6, 100.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = gradient_local_search(a_in, t9, verbose=False)
            check_and_update(a_out, "gls_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t10 = min(remaining * 0.7, 60.0)
        if t10 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = mirror_descent(a_in, t10, beta0=100.0, beta_max=500.0,
                                       top_k=300, c0=0.5, verbose=False)
            check_and_update(a_out, "md_ultra_final")

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq, _ = parametric_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    return [float(x) for x in best_seq]
