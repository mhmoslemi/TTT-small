y_t = x_t + (t-1)/(t+1) * (x_t - x_{t-1})
x_{t+1} = Proj_simplex(y_t - eta_t * grad f(y_t))
eta_t = 1/sqrt(L*t) or similar

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def spectral_init(n, n_modes=30, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def compute_smooth_gradient(a, ac, tau, top_k=200):
    n = len(a)
    ac_max = ac.max()
    ac_shifted = tau * (ac - ac_max)
    exp_ac = np.exp(ac_shifted)
    p = exp_ac / exp_ac.sum()

    p_cum = np.concatenate([[0.0], np.cumsum(p)])
    k_lo = np.arange(n)
    k_hi = np.minimum(k_lo + n, 2 * n - 1)
    S = p_cum[k_hi] - p_cum[k_lo]
    cross = sp_correlate(p, a, mode='valid')
    g = a * S + cross
    return g, p


def nesterov_continuation(a0, time_limit, tau_schedule=None, verbose=True, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    if tau_schedule is None:
        tau_schedule = [5.0, 10.0, 20.0, 40.0, 80.0, 150.0, 300.0, 600.0]

    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        phase_start = time.monotonic()
        phase_limit = min(time_limit - (phase_start - t0), time_limit / len(tau_schedule) * 1.5)

        x = a.copy()
        x_prev = a.copy()
        eta = 1.0
        stall = 0
        iter_count = 0

        while time.monotonic() - phase_start < phase_limit:
            iter_count += 1
            t_iter = float(iter_count)
            momentum = (t_iter - 1.0) / (t_iter + 1.0)
            y = x + momentum * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

            ac = fast_autocorr(y)
            g, p = compute_smooth_gradient(y, ac, tau, top_k=min(300, 2 * n - 1))
            g_tangent = g - g.mean()

            step = 1.0 / (tau * np.sqrt(t_iter) + 1.0)
            a_trial = y - step * g_tangent
            a_new = project_simplex(a_trial)

            m_new = fast_autocorr(a_new).max()
            if m_new < best_m - 1e-15:
                best_m = m_new
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1

            if stall > 500:
                a_new = np.maximum(a_new + rng.normal(0, 0.002, n) * a_new, 0.0)
                a_new = a_new / a_new.sum()
                m_new = fast_autocorr(a_new).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a_new.copy()
                stall = 0

            x_prev = x
            x = a_new
            a = a_new

            if iter_count % 2000 == 0:
                if time.monotonic() - phase_start > phase_limit:
                    break

        a = best_a.copy()
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NestCont tau={tau:.0f} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K_init=200, top_K_max=None, use_all=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0
    last_improve = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if use_all or K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            iter_count += 1
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 50, top_K_max)
            iter_count += 1
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            iter_count += 1
            continue
        g_dir = b_val / b_sum

        # Standard FW step size with improvement
        alpha_fw = 2.0 / (iter_count + 2.0)
        best_alpha = alpha_fw
        best_trial_m = M_curr

        # Try FW step and a few alternatives
        for alpha_try in [alpha_fw, min(1.0, alpha_fw * 1.5), min(1.0, alpha_fw * 2.0), 0.5 / np.sqrt(iter_count + 1)]:
            a_trial = (1.0 - alpha_try) * a + alpha_try * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha_try

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()
            last_improve = iter_count
        else:
            if iter_count - last_improve > 100:
                # Perturb to escape
                pert = rng_perturb(a, 0.01, rng_seed=iter_count)
                m_pert = fast_autocorr(pert).max()
                if m_pert < best_m:
                    a = pert
                    best_m = m_pert
                    best_a = a.copy()
                    last_improve = iter_count

        iter_count += 1
        if iter_count % 10 == 0:
            K = min(K + 20, top_K_max)

        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def rng_perturb(a, scale, rng_seed=0):
    rng = np.random.default_rng(rng_seed)
    pert = rng.normal(0, scale, len(a)) * a
    result = np.maximum(a + pert, 0.0)
    s = result.sum()
    if s < 1e-15:
        return a.copy()
    return result / s


def fine_coord_descent(a0, time_limit, verbose=False, rng_seed=777):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:30]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            if len(idx) < 2:
                continue
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-impact destinations
            in_peak = np.zeros(n, dtype=bool)
            for kk in top_k[:15]:
                s2 = max(0, int(kk) - n + 1)
                e2 = min(n - 1, int(kk))
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            # Pick a few candidates with low current value
            sorted_cands = candidates[np.argsort(a[candidates])[:5]]

            found = False
            for dst in sorted_cands:
                if dst == src or found:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

        if not improved:
            stall += 1
            # Random small perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        else:
            stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordDesk ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def block_multiplicative_update(a0, time_limit, block_size=40, overlap=20, verbose=False, rng_seed=55):
    """Block-based multiplicative update (MISO) for accelerated convergence."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stride = block_size - overlap
    n_blocks = (n - overlap) // stride + 1
    top_k_lags = 100

    for sweep in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        active_lags = np.argsort(ac)[::-1][:top_k_lags]

        # Shuffle block order
        block_starts = list(range(0, min(n_blocks, (n - overlap) // stride + 1) * stride, stride))
        rng.shuffle(block_starts)

        for b_start in block_starts:
            if time.monotonic() - t0 > time_limit:
                break
            b_end = min(b_start + block_size, n)
            m = b_end - b_start
            if m < 5:
                continue

            # Extract block
            block = a[b_start:b_end].copy()
            if block.sum() < 1e-15:
                continue

            # Compute gradient contributions for this block
            # For each active lag, compute the subgradient w.r.t. block variables
            grad_block = np.zeros(m)
            for k in active_lags[:50]:
                ki = int(k)
                # conv[ki] = sum_j a[j]*a[ki-j]
                # For block variable a[b_start + i], it appears as:
                #   a[b_start+i] * a[ki - (b_start+i)] (as first factor)
                #   a[ki - (b_start+i)] * a[b_start+i] (as second factor, same term for real)
                # Actually conv[ki] = sum_j a[j]*a[ki-j], so a[b_start+i] appears
                # when j = b_start+i: term is a[b_start+i]*a[ki-b_start-i]
                # when ki-j = b_start+i, i.e., j = ki-b_start-i: term is a[ki-b_start-i]*a[b_start+i]
                # Both give the same contribution: 2*a[b_start+i]*a[ki-b_start-i] (if both valid)
                # But they're the same term counted once in the sum.
                # Let me just compute the gradient directly.
                j1 = b_start + np.arange(m)
                # Term 1: a[j1] * a[ki - j1]
                j1_valid = (j1 >= 0) & (j1 < n) & (ki - j1 >= 0) & (ki - j1 < n)
                # grad w.r.t. a[b_start+i] from a[j1]*a[ki-j1]:
                # = a[ki - j1] for valid j1
                contrib1 = np.zeros(m)
                valid1 = j1_valid
                contrib1[valid1] = a[ki - j1[valid1]]
                # Term 2: a[j2]*a[ki-j2] where ki-j2 = b_start+i, j2 = ki-b_start-i
                # This is the same as term 1 (same product), so the gradient is just contrib1
                # Actually for a real sequence, conv[k] = sum_j a[j]*a[k-j]
                # d(conv[k])/da[i] = a[k-i] + a[i] * 0 ... no
                # d(sum_j a[j]*a[k-j])/da[i] = a[k-i] (from j=i term) + a[i] (from k-j=i term, i.e., j=k-i)
                # = a[k-i] + (a[i] if k-i != i else a[i])
                # = a[k-i] + a[k-i] = 2*a[k-i] if k-i != i
                # = a[i] if k-i == i (i.e., k=2i)
                # So grad_i conv[k] = a[k-i] + (a[k-i] if k-i != i else a[i])
                # Since a[k-i] is the value at position k-i, and if k-i==i then it's a[i] itself.
                # So: grad_i conv[k] = a[k-i] + a[i] * [k-2i==0] ... no this is wrong.

                # Let me be very careful:
                # conv[k] = sum_{j=0}^{n-1} a[j] * a[k-j]  (where k-j is in [0,n-1])
                # d(conv[k])/da[i] = a[k-i] (from the term j=i)
                #                    + a[i] (from the term k-j=i, i.e., j=k-i, if k-i is valid and k-i != i)
                # If k-i == i (i.e., k=2i), both terms are the same: 2*a[i]... no, it's just a[i]*a[i] and d/d a[i] = 2*a[i]
                # Actually: conv[k] = sum_j a[j]*a[k-j]. The term with j=i is a[i]*a[k-i].
                # The term with k-j=i (j=k-i) is a[k-i]*a[i].
                # If k-i != i, these are two different terms in the sum, each contributing a[i] to the gradient.
                # If k-i == i, it's one term a[i]*a[i], contributing 2*a[i] to the gradient.
                # In either case: d(conv[k])/da[i] = a[k-i] + a[i]*[k-i valid] ...

                # Simpler: just use the symmetry. For i in [0,n-1]:
                # grad_i conv[k] = a[k-i] (if 0 <= k-i < n) + a[i] * 0
                # Wait no. Let me just compute it numerically for correctness.

                # Actually the cleanest way:
                # conv[k] = sum_j a[j] * a[k-j]
                # d/d a[i]: the only terms involving a[i] are:
                #   j=i: a[i]*a[k-i] -> derivative is a[k-i]
                #   k-j=i (j=k-i): a[k-i]*a[i] -> derivative is a[k-i]
                # If i == k-i (i.e., k=2i): only one term, derivative is 2*a[i]
                # If i != k-i: two terms, derivative is 2*a[k-i]
                # In all cases: derivative = a[k-i] + a[k-i] = 2*a[k-i] (when k-i is valid)
                # But if k-i is not in [0,n-1], the term doesn't exist.
                # And if k=2i and k-i=i, derivative = 2*a[i] (which equals 2*a[k-i]).
                # So: grad_i conv[k] = 2*a[k-i] if 0<=k-i<n, else 0.

                # Wait that can't be right either. If k-i is valid:
                # When j=i: term is a[i]*a[k-i], d/d a[i] = a[k-i]
                # When j=k-i: term is a[k-i]*a[i], d/d a[i] = a[k-i]
                # Total: 2*a[k-i]
                # But if i = k-i (k=2i): there's only ONE term (j=i=k-i), which is a[i]*a[i], d/d a[i] = 2*a[i] = 2*a[k-i]. Same formula!
                # So indeed: grad_i conv[k] = 2*a[k-i] if 0<=k-i<n, else 0.

                grad_i = np.zeros(m)
                ki_minus_i = ki - j1  # = ki - (b_start + i) for i=0..m-1
                valid = (ki_minus_i >= 0) & (ki_minus_i < n)
                grad_i[valid] = 2.0 * a[ki_minus_i[valid]]
                # But we need to subtract the self-interaction: when ki_minus_i == j1 (i.e., ki - b_start - i == b_start + i, ki - 2*b_start - 2*i == 0)
                # In that case, the two terms are the same and we've double-counted.
                # Actually no - I showed above that in that case the derivative is still 2*a[i] = 2*a[ki_minus_i], so the formula is correct.

                grad_block += grad_i

            # Multiplicative update: a_i <- a_i * (1 - eta * grad_i / (a_i + eps))
            # Or the standard MISO: a_i <- a_i * (target / a_i)^eta
            # For minimizing sum_k p_k * conv[k], the update is:
            # a_i <- a_i * (1 - eta * grad_i)^{+}
            # i.e., multiplicative: a_i <- a_i / (1 + eta * grad_i) if grad_i > 0

            # Actually for the max (or smoothed max), the MISO is:
            # a_i <- a_i * exp(-eta * grad_i)  (for the exponential smoothing)
            # Or a_i <- a_i / (1 + eta * grad_i) (for the log-based MISO)

            # Let's use: a_i <- a_i * (1 + eta * grad_i)^{-1} for positive grad,
            # and a_i * (1 - eta * |grad_i|) for negative grad (rare here)
            eta = 0.05
            block_new = block / (1.0 + eta * np.maximum(grad_block, 0.0))
            block_new = np.maximum(block_new, 0.0)

            # Also try to reduce elements that contribute most to the peak
            # Move mass from high-gradient elements to low-gradient ones within the block
            if grad_block.max() > 1e-10:
                move_frac = 0.05
                src_idx = np.argmax(grad_block)
                # Find element with minimum gradient in block
                dst_idx = np.argmin(grad_block)
                if src_idx != dst_idx and block[src_idx] > 1e-10:
                    amt = block[src_idx] * move_frac
                    block_new[src_idx] = block[src_idx] - amt
                    block_new[dst_idx] = block[dst_idx] + amt

            # Update
            a[b_start:b_end] = block_new

        # Renormalize
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a / s

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()

        if verbose and sweep % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    BlockMISO sweep={sweep} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        best_seq = make_semicircle(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Nesterov continuation on parent ===
    print(f"\n=== Phase 1: Nesterov continuation (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_continuation(a_in, t1, verbose=True, rng_seed=42)
        try_update(a_out, "nest_cont_1")

    # === Phase 2: Nesterov continuation from diverse init ===
    print(f"\n=== Phase 2: Diverse Nesterov (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 40:
        inits = [
            (make_semicircle(n0), "semi"),
            (make_raised_cosine(n0), "cos"),
            (make_gaussian(n0, 0.2), "gauss"),
            (make_bimodal(n0, 0.18), "bimod"),
            (make_flat_top(n0, 0.35), "flat"),
        ]
        # Blend with current best
        for i, (init_a, lbl) in enumerate(inits[:3]):
            blend = 0.4 * norm_sum1(best_seq) + 0.6 * init_a
            blend = norm_sum1(blend)
            t_per = t2 / 3.0
            if time.monotonic() - t_global > time_budget - 60:
                break
            a_out, _ = nesterov_continuation(blend, t_per, verbose=False, rng_seed=100 + i)
            try_update(a_out, f"nest_blend_{lbl}")

    # === Phase 3: LP-FW polishing ===
    print(f"\n=== Phase 3: LP-FW (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 50 and n0 <= 6000:
        a_in = norm_sum1(best_seq)
        use_all = (n0 <= 2000)
        top_K_init = min(400, 2 * n0 - 1)
        top_K_max = min(3000, 2 * n0 - 1)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max,
                                    use_all=use_all)
        try_update(a_out, "lp_fw_1")

    # === Phase 4: Block MISO ===
    print(f"\n=== Phase 4: Block MISO (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = block_multiplicative_update(a_in, t4, block_size=50, overlap=20,
                                                verbose=False, rng_seed=55)
        try_update(a_out, "block_miso")

    # === Phase 5: LP-FW round 2 with larger active set ===
    print(f"\n=== Phase 5: LP-FW r2 (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 40 and n0 <= 6000:
        a_in = norm_sum1(best_seq)
        use_all = (n0 <= 1500)
        top_K_init = min(800, 2 * n0 - 1)
        top_K_max = min(5000, 2 * n0 - 1)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max,
                                    use_all=use_all)
        try_update(a_out, "lp_fw_2")

    # === Phase 6: Nesterov continuation (fine tau) ===
    print(f"\n=== Phase 6: Fine Nesterov (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 30:
        a_in = norm_sum1(best_seq)
        tau_fine = [100.0, 200.0, 400.0, 800.0, 1500.0]
        a_out, _ = nesterov_continuation(a_in, t6, tau_schedule=tau_fine,
                                          verbose=True, rng_seed=200)
        try_update(a_out, "nest_fine")

    # === Phase 7: Fine coordinate descent ===
    print(f"\n=== Phase 7: Coord descent (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.08, 120.0)
    if t7 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = fine_coord_descent(a_in, t7, verbose=False, rng_seed=777)
        try_update(a_out, "coord_fine")

    # === Phase 8: Length exploration ===
    print(f"\n=== Phase 8: Length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    trial_lengths = [n0 - 400, n0 + 400, n0 - 800, n0 + 800]
    trial_lengths = [l for l in trial_lengths if 2000 <= l <= 8000 and l != n0]

    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t8a = min(remaining * 0.3, 150.0)
        if t8a > 20:
            a_out, _ = nesterov_continuation(a_long, t8a, verbose=False, rng_seed=new_n % 500)
            try_update(a_out, f"nest_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 5000:
            t8b = min(remaining * 0.3, 100.0)
            if t8b > 20:
                a_in = norm_sum1(a_long)
                top_K = min(2000, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe(a_in, t8b, solver, verbose=False,
                                           top_K_init=min(300, 2 * new_n - 1),
                                           top_K_max=top_K, use_all=False)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 9: Final LP-FW + coord polish ===
    print(f"\n=== Phase 9: Final polish (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        t9a = min(remaining * 0.5, 100.0)
        if t9a > 20 and n_best <= 6000:
            a_in = norm_sum1(best_seq)
            top_K_init = min(1000, 2 * n_best - 1)
            top_K_max = min(6000, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe(a_in, t9a, solver, verbose=False,
                                       top_K_init=top_K_init, top_K_max=top_K_max,
                                       use_all=False)
            try_update(a_out, "lp_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t9b = min(remaining * 0.5, 60.0)
            if t9b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = fine_coord_descent(a_in, t9b, verbose=False, rng_seed=999)
                try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        best_seq = make_semicircle(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
