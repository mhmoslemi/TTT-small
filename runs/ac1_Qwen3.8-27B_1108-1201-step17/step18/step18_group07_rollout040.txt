import numpy as np

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project onto probability simplex: v >= 0, sum(v) = 1."""
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones_like(v) / len(v)
    return v / s


def obj_max_conv(a):
    """Compute max(conv(a,a)) assuming sum(a)=1."""
    return fast_conv(a).max()


def score_from_maxconv(a):
    """Compute the full score from a sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return float(2.0 * n * c.max() / (s * s))

def spectral_flatten(a, strength=0.3, iterations=3):
    """Iterate spectral flattening steps."""
    a = simplex_project(a)
    for _ in range(iterations):
        n = len(a)
        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        mag_mean = mag.mean()
        if mag_mean < 1e-30:
            break
        
        ratio = mag / (mag_mean + 1e-30)
        mag_new = mag_mean * np.power(ratio, 1.0 - strength)

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        a = simplex_project(a_new)
    return a

def subgradient_weighted(a, c, top_k=20, beta=50.0):
    """Compute weighted subgradient from top peaks."""
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = c[top_idx].copy()
    
    max_v = top_vals[0]
    if max_v < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        exponents = beta * (top_vals - max_v) / (max_v + 1e-30)
        exponents = np.maximum(exponents, -50.0)
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-12:

        k_int = int(k)
        i_start = max(0, k_int - n + 1)

            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_smooth_random(n, seed=42):
    """Generate a smooth random non-negative sequence."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    for f in range(1, 15):
        coeff = rng.normal(0, 0.3 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.01)
    return simplex_project(a)


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return simplex_project(a)


def make_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return simplex_project(a)

import numpy as np
import time


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project onto probability simplex: v >= 0, sum(v) = 1."""
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones_like(v) / len(v)
    return v / s


def obj_max_conv(a):
    """Compute max(conv(a,a)) assuming sum(a)=1."""
    return fast_conv(a).max()


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_top_peaks(c, top_k=10):
    """Get indices and values of top peaks in autocorrelation."""
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    return top_idx, c[top_idx]


def subgradient_weighted(a, c, top_k=20, beta=50.0):
    """Compute weighted subgradient from top peaks."""
    n = len(a)
    top_idx, top_vals = compute_top_peaks(c, top_k)
    
    max_v = top_vals[0]
    if max_v < 1e-30:
        w = np.ones(len(top_idx)) / len(top_idx)
    else:
        exponents = beta * (top_vals - max_v) / (max_v + 1e-30)
        exponents = np.maximum(exponents, -50.0)
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-12:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def mirror_descent_step(a, g, eta):
    """One mirror descent step on simplex: a_new[i] prop a[i]*exp(-eta*g[i])."""
    log_shift = -eta * g
    max_shift = log_shift.max()
    scaled = a * np.exp(log_shift - max_shift)
    scaled = np.maximum(scaled, 1e-15)
    s = scaled.sum()
    if s < 1e-15:
        return a.copy()
    return scaled / s


def optimize_mirror_descent(a0, time_limit, verbose=False):
    """Mirror descent with backtracking line search on simplex."""
    n = len(a0)
    a = simplex_project(a0)
    best_a = a.copy()
    best_m = obj_max_conv(a)
    t0 = time.monotonic()
    
    eta = 0.3
    stall_count = 0
    max_stall = 5000
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall_count > max_stall:
            break
        
        progress = min(1.0, t / 100000.0)
        cur_top_k = max(5, int(40 * (1 - 0.7 * progress) + 5))
        cur_beta = 20.0 + 60.0 * progress
        
        c = fast_conv(a)
        g = subgradient_weighted(a, c, top_k=cur_top_k, beta=cur_beta)
        
        # Try step with backtracking
        improved = False
        for shrink in range(6):
            cur_eta = eta * (0.5 ** shrink)
            a_new = mirror_descent_step(a, g, cur_eta)
            m_new = obj_max_conv(a_new)
            
            if m_new < best_m - 1e-15:
                best_m = m_new
                best_a = a_new.copy()
                a = a_new.copy()
                eta = min(1.0, eta * 1.02)
                improved = True
                stall_count = 0
                break
        if not improved:
            eta = max(0.001, eta * 0.5)
            stall_count += 1
            # Occasional perturbation
            if stall_count % 200 == 0:
                rng_p = np.random.default_rng(t)
                pert = rng_p.normal(0, 0.005, n)
                a_pert = simplex_project(a + pert)
                m_pert = obj_max_conv(a_pert)
                if m_pert < best_m:
                    best_m = m_pert
                    best_a = a_pert.copy()
                    a = a_pert.copy()
                    stall_count = 0
                    eta = 0.3
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD t={t} ({elapsed:.0f}s): J={2*n*best_m:.8f}, eta={eta:.4f}")
    
    return best_a, best_m


def spectral_peak_shave(a, strength=0.3, iterations=3):
    """Frequency-domain peak shaving: flatten |A| to reduce autocorrelation peaks."""
    a = simplex_project(a)
    for _ in range(iterations):
        n = len(a)
        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        mag_mean = mag.mean()
        if mag_mean < 1e-30:
            break
        
        ratio = mag / (mag_mean + 1e-30)
        mag_new = mag_mean * np.power(ratio, 1.0 - strength)
        mag_new = np.maximum(mag_new, 1e-30)
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        a = simplex_project(a_new)
    return a


def spectral_flatten_adaptive(a, strength=0.4, max_iter=10):

    a = simplex_project(a)

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        else:

def freq_domain_gradient_step(a, c, eta=0.1, top_k=5):
    """Frequency-domain gradient step: reduce |A| at bins contributing to peaks."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    top_idx, top_vals = compute_top_peaks(c, top_k)
    
    # For each peak at lag k, the frequency bins m contribute with weight
    # Re(exp(i*2*pi*m*k/fft_size))
    # Reduce mag at bins where this is large for the top peaks
    sensitivity = np.zeros_like(mag)
    for k in top_idx:
        k_int = int(k)
        m_idx = np.arange(len(mag))
        weights = np.cos(2.0 * np.pi * m_idx * k_int / fft_size)
        sensitivity += np.maximum(weights, 0) ** 2
    
    # Scale: reduce more where sensitivity is high
    sens_norm = sensitivity / (sensitivity.max() + 1e-30)
    mag_reduction = 1.0 - eta * sens_norm
    mag_new = mag * np.maximum(mag_reduction, 0.5)
    mag_new = np.maximum(mag_new, 1e-30)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    return simplex_project(a_new)


def pair_move_search(a0, time_limit, num_peaks=5, verbose=False):
    """Local search using pair moves targeting peak reduction."""
    n = len(a0)
    a = simplex_project(a0)
    best_a = a.copy()
    best_m = obj_max_conv(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        top_idx, _ = compute_top_peaks(c, num_peaks)
        
        improved = False
        
        for k in top_idx:
            if time.monotonic() - t0 > time_limit:
                break
            
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            best_trial_m = best_m
            best_trial = None
            
            # Try various destinations
            dests = rng.integers(0, n, size=min(20, n))
            for dst in dests:
                if int(dst) == int(src):
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[int(src)] -= amt
                    a_trial[int(dst)] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = obj_max_conv(a_trial)
                    if m_trial < best_trial_m:
                        best_trial_m = m_trial
                        best_trial = a_trial.copy()
            
            if best_trial is not None:
                a = best_trial
                best_m = best_trial_m
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.05, 0.15)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = obj_max_conv(a_trial)
                if m_trial < best_m:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    stall = 0
                    break
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return simplex_project(a)


def make_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return simplex_project(a)


def make_asymmetric(n, decay=2.0):
    """Asymmetric sequence: exponential decay from left."""
    x = np.linspace(0, 1, n)
    a = np.exp(-decay * x) + 0.1 * np.exp(-decay * (1 - x))
    return simplex_project(a)


def make_smooth_random(n, seed=42):
    """Generate a smooth random non-negative sequence."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 15):
        coeff = rng.normal(0, 0.4 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.01)
    return simplex_project(a)


def make_dual_peak(n, pos1=0.25, pos2=0.7, width=0.12):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.05
    return simplex_project(a)


def try_update_global(state, a, label, t_global):
    """Try to update the global best. state is a dict with 'best_seq' and 'best_score'."""
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < state['best_score']:
        state['best_score'] = sc
        state['best_seq'] = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

        t4 = remaining * 0.6
        a_out, _ = optimize_combined(a_in, t4, verbose=True)
        try_update_global(state, a_out, "final_polish", t_global)
        
        # Extra spectral pass
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_spec = spectral_flatten_adaptive(a_out, strength=0.5, max_iter=15)
            try_update_global(state, a_spec, "final_spectral", t_global)
    
    # === Phase 5: Final pair move polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        n_best = len(state['best_seq'])
        a_in = simplex_project(state['best_seq'])
        t5 = remaining * 0.4
        a_out, _ = pair_move_search(a_in, t5, num_peaks=10)
        try_update_global(state, a_out, "final_pair", t_global)
    
    # Final validation
    if state['best_seq'] is not None:
        fs = evaluate_safe(state['best_seq'])
        if np.isfinite(fs):
            state['best_score'] = min(state['best_score'], fs)
        print(f"\nFINAL: score={state['best_score']:.8f}, n={len(state['best_seq'])}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        state['best_seq'] = np.ones(4000) / 4000
        state['best_score'] = evaluate_safe(state['best_seq'])
        print(f"Fallback: score={state['best_score']:.8f}")
    
    return [float(x) for x in state['best_seq']]
