def norm_nonneg(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a)/len(a)
    return a / s

def make_flat_spectrum(n, rng):
    fft_size = 1 << (2*n-1).bit_length()
    half = fft_size//2 + 1
    X = np.zeros(half, dtype=np.complex128)
    X[0] = rng.uniform(0.9, 1.1)
    for k in range(1, half):
        X[k] = np.exp(1j * rng.uniform(0, 2*np.pi))
    a = np.fft.irfft(X, fft_size)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return None
    return a / s

def make_smooth_bump(n, alpha=2.0, center=0.5):

    a = np.maximum(a, 0)

def compute_subgrad(a, conv, n_top=50, temperature=1.0):
    n = len(a)
    conv_len = len(conv)
    # Soft-max weights
    conv_clipped = np.minimum(conv, conv.max())
    log_w = np.log(np.maximum(conv_clipped, 1e-30)) / temperature
    log_w -= log_w.max()
    w = np.exp(log_w)
    w /= w.sum()
    
    # Focus on top peaks
    top_idx = np.argsort(conv)[::-1][:n_top]
    grad = np.zeros(n)
    for k in top_idx:
        ki = int(k)
        wi = w[ki]
        if wi < 1e-12:
            continue
        # d conv[ki]/d a[j] for valid j
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        pair = ki - idx
        # Each pair (j, ki-j) contributes 2*a[ki-j] to grad[j]
        # but if j == ki-j (i.e., ki = 2j), it contributes only a[ki-j]
        contrib = 2.0 * a[pair]
        # Fix the diagonal: where idx == pair, i.e., ki-idx == idx, ki = 2*idx
        diag = (idx == pair)
        if np.any(diag):
            contrib[diag] = a[pair[diag]]
        grad[idx] += wi * contrib
    return grad

def spectral_flatten_step(a, alpha=0.5):
    """One step of spectral flattening."""
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.abs(A)**2
    P_max = P.max()
    if P_max < 1e-30:
        return a.copy()
    # Compress the dynamic range
    P_new = P / (P + 0.5*P_max) ** (1-alpha)
    # Actually, let's use: P_new = P^alpha / (P^alpha).mean() * P.mean()
    # Simpler: P_new = P / (P + lambda) * scale
    lam = 0.1 * P_max
    P_new = P / (P + lam)
    # Normalize so that the total "energy" is preserved
    P_new *= P.sum() / P_new.sum()
    # Take sqrt to get new magnitude
    mag_new = np.sqrt(P_new)
    phase = A / np.maximum(np.abs(A), 1e-30)
    A_new = mag_new * phase
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s
